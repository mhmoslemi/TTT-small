    n = len(a)

    best_a = a.copy()
    best_obj = compute_obj(a)

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that row k of (A @ z) gives (a*z)_k."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        hi = min(m, j + n)
        A[j:hi, j] = a[:hi - j]
    return A


def solve_lp_reduced(a, A, constraint_mask):
    """Solve LP with subset of constraints. Returns (z, M, dual_y_for_masked_rows)."""
    n = len(a)
    A_sub = A[constraint_mask, :]
    n_cons = int(np.sum(constraint_mask))

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A_sub @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y_sub = None
                    try:
                        dual_y_sub = np.maximum(np.array(c1.dual_value), 0)
                    except:
                        pass
                    return z_val, float(M.value), dual_y_sub
        except:
            continue
    return None, None, None


def solve_lp_full(a, A):
    """Solve the full LP with all constraints."""
    n = len(a)
    mask = np.ones(2 * n - 1, dtype=bool)
    return solve_lp_reduced(a, A, mask)


def cutting_plane_lp(a, A, max_iter=20, verbose=False):
    """
    Cutting plane approach to solve the LP.
    Start with top-K constraints, add violated ones iteratively.
    Returns (z_star, M_star, active_mask, dual_y_full).
    """
    n = len(a)
    m = 2 * n - 1

    # Initial constraint set: top K based on current a*a
    conv_aa = A @ a  # = (a*a) using the matrix
    K = max(20, min(m, 5 * n // 2))
    top_idx = np.argsort(conv_aa)[-K:]
    active_mask = np.zeros(m, dtype=bool)
    active_mask[top_idx] = True

    z_star = None
    M_star = np.inf

    for it in range(max_iter):
        z, M, dual_sub = solve_lp_reduced(a, A, active_mask)
        if z is None:
            # Fallback to full LP
            z, M, dual_sub = solve_lp_full(a, A)
            if z is None:
                break
            active_mask = np.ones(m, dtype=bool)
            break

        z_star = z
        M_star = M

        # Check for violated constraints
        conv_az = A @ z
        violations = conv_az > M * 1.001
        n_viol = int(np.sum(violations & ~active_mask))

        if n_viol == 0:
            break

        # Add most violated constraints
        all_violated = np.where(violations & ~active_mask)[0]
        if len(all_violated) > 0:
            # Add up to 10 most violated
            cross_vals = conv_az[all_violated]
            worst = all_violated[np.argsort(cross_vals)[-min(10, len(all_violated)):]]
            active_mask[worst] = True

        # If too many constraints, switch to full LP
        if np.sum(active_mask) > m * 0.6:
            result = solve_lp_full(a, A)
            if result[0] is not None:
                z_star, M_star = result[0], result[1]
            active_mask = np.ones(m, dtype=bool)
            break

    # Map dual back to full size
    dual_y_full = np.zeros(m)
    if dual_sub is not None:
        active_indices = np.where(active_mask)[0]
        dual_y_full[active_indices] = dual_sub

    return z_star, M_star, active_mask, dual_y_full


def golden_section_search(a, z, a_best, f_best, n_iter=12):
    """
    Golden section search for optimal mixing t in [0,1]: a_t = (1-t)*a + t*z.
    Returns (a_opt, f_opt).
    """
    n = len(a)

    def eval_at(t):
        a_t = (1.0 - t) * a + t * z
        a_t = np.maximum(a_t, 0)
        s = float(np.sum(a_t))
        if s < 1e-10:
            return np.inf
        a_t = a_t / s
        return compute_obj(a_t)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0

    # Endpoints
    f_lo = f_best  # eval_at(0)
    f_hi = eval_at(1.0)

    best_f = min(f_lo, f_hi)
    best_t = 0.0 if f_lo <= f_hi else 1.0

    for _ in range(n_iter):
        m1 = lo + (1.0 - gr) * (hi - lo)
        m2 = lo + gr * (hi - lo)
        f_m1 = eval_at(m1)
        f_m2 = eval_at(m2)

        if f_m1 < f_m2:
            hi = m2
            f_hi = f_m2
        else:
            lo = m1
            f_lo = f_m1

        f_mid = eval_at((lo + hi) / 2.0)
        if f_mid < best_f:
            best_f = f_mid
            best_t = (lo + hi) / 2.0

    # Also check the current best is still competitive
    if f_best < best_f:
        best_f = f_best
        best_t = 0.0

    a_opt = (1.0 - best_t) * a + best_t * z
    a_opt = np.maximum(a_opt, 0)
    a_opt = a_opt / float(np.sum(a_opt))
    return a_opt, best_f

def resample_sequence(a, new_n):
    """Resample to new length via interpolation with slight smoothing."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    # Slight smoothing to avoid interpolation artifacts
    if new_n > 4:
        kernel = np.array([0.25, 0.5, 0.25])
        a_new = np.convolve(a_new, kernel, mode='same')
    return np.maximum(a_new, 0.0)

        # Structured perturbation: add small sinusoidal component
        t = np.linspace(0, 1, n_best)
        perturbation_options = [
            best_seq * (1.0 + 0.05 * np.sin(2 * np.pi * 3 * t)),
            best_seq * (1.0 + 0.03 * np.cos(2 * np.pi * 7 * t)),
            best_seq * np.exp(rng.normal(0, 0.03, n_best)),
            best_seq * (1.0 + 0.04 * np.sin(2 * np.pi * 1 * t + rng.uniform(0, 2 * np.pi))),
        ]
        for pert_opt in perturbation_options:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_r = np.maximum(pert_opt, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr
            alloc = min(80.0, remaining * 0.25)
            a_opt = lp_alternating_improved(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  ** Restart: obj={best_score:.8f} **")
            else:
                print(f"  Restart: obj={obj_r:.8f}")

    # Phase 5: Final LP polish
    print(f"\n=== Phase 5: Final LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(100.0, remaining * 0.4)
        a_opt = lp_alternating_improved(best_seq, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Phase 6: Coordinate polish
    print(f"\n=== Phase 6: Coordinate polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(60.0, remaining * 0.4)
        a_opt = quick_coordinate_polish(best_seq, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Polish: obj={best_score:.8f} **")
        else:
            print(f"  Polish: obj={obj:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
