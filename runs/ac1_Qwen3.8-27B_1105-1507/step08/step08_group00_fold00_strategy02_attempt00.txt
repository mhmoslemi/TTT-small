def compute_conv(sequence):
    """Computes the full linear convolution of the sequence using FFT.
    Returns the array of length 2*n-1 containing the convolution values.
    """
    n = len(sequence)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = fft(sequence, size)
    return ifft(fa * np.conj(fa)).real[:2*n-1]
