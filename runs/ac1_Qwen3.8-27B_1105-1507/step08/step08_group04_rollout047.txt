def coordinate_descent(a, time_budget=30.0, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    max_sweeps = 100
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        # Random order for diversity
        order = rng.permutation(n)
        improved = False
        
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            
            # Try a range of values for a[j]
            old_val = a[j]
            a_without_j = a.copy()
            a_without_j[j] = 0
            s_without_j = float(np.sum(a_without_j))
            
            if s_without_j < 1e-10:
                continue
            
            # We want to find the best value of a[j]
            # The constraint is sum(a) = 1, so a[j] = 1 - sum(others)
            # But we're not fixing the others, we're just changing a[j]
            # and re-normalizing.
            
            # Try several values
            candidates = [0.0, old_val * 0.5, old_val * 0.75, old_val, old_val * 1.25, old_val * 1.5, 
                         old_val * 2.0, s_without_j * 0.1, s_without_j * 0.5]
            candidates = [max(0, c) for c in candidates]
            
            obj_threshold = best_obj
            for c in candidates:
                a_try = a.copy()
                a_try[j] = c
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
            # Reset if we didn't update
            if not improved or a is best_a:
                pass
        
        if not improved and sweep > 5:
            break

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(x_old, a, bc_type='not-a-knot')
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                    n_bumps = rng.integers(1, 4)
                    for _ in range(n_bumps):
                        center = rng.integers(0, n)
                        width = rng.integers(max(2, n // 25), max(3, n // 8))
                        amp = rng.uniform(0.05, 0.3)
                        x = np.arange(n)
                        a += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
                elif pert_type == 2:
                    # Block reweighting
                    block_size = rng.integers(max(2, n // 60), max(3, n // 12))
                    n_blocks = n // block_size
                    a = best_a.copy()
                    for b in range(n_blocks):
                        mul = rng.uniform(0.6, 1.4)
                        a[b * block_size:(b + 1) * block_size] *= mul
                elif pert_type == 3:
                    # Circular shift + noise
                    shift = rng.integers(1, max(2, n // 8))
                    a = np.roll(best_a, rng.integers(-shift, shift + 1))
                    a = a * np.exp(rng.normal(0, 0.03, n))
                else:
                    # Compress/stretch
                    factor = rng.uniform(0.7, 1.3)
                    x = np.arange(n)
                    x_new = np.minimum(np.round(x * factor), n - 1).astype(int)
                    a = best_a[x_new]
                    a = a * np.exp(rng.normal(0, 0.02, n))

                a = np.maximum(a, 0)
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new
                stall = 0
                if verbose:
                    elapsed = time.monotonic() - start
                    print(f"    Perturbed (type={pert_type}), {elapsed:.0f}s")

    return best_a


def coordinate_descent(a, time_budget=30.0):
    """Coordinate descent polish: optimize individual coordinates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    for sweep in range(200):
        if time.monotonic() - start > time_budget:
            break
        order = rng.permutation(n)
        improved = False
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            old_val = a[j]
            if old_val < 1e-8:
                continue
            factors = [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.1]
            for f in factors:
                a_try = a.copy()
                a_try[j] = old_val * f
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-11:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
        if not improved and sweep > 10:
            break

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    try:
        cs = CubicSpline(x_old, a, bc_type='not-a-knot')
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 7)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-30 * (x - 0.5) ** 2)
    elif choice == 1:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif choice == 2:
        a = np.exp(-50 * (x - 0.25) ** 2) + 0.7 * np.exp(-50 * (x - 0.75) ** 2)
    elif choice == 3:
        # Fourier-based
        n freqs = n // 2 + 1
        freqs = np.arange(n // 2 + 1)
        coeffs = rng.normal(0, 1, size=len(freqs))
        a = np.zeros(n)
        for k in range(len(freqs)):
            a += coeffs[k] * np.cos(2 * np.pi * freqs[k] * x)
        a = np.abs(a) + 0.05
    elif choice == 4:
        a = (x + 0.1) ** rng.uniform(0.5, 2.5)
    elif choice == 5:
        # Smoothed random
        a = np.abs(rng.normal(0, 1, n))
        ks = max(3, n // 40)
        kernel = np.ones(ks) / ks
        a = np.convolve(a, kernel, mode='same')
    else:
        # Triangle wave
        a = 1.0 - 2.0 * np.abs(x - 0.5)
        a = a ** rng.uniform(1.0, 3.0)
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0.001)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating with adaptive blending, structured perturbation,
    coordinate descent polish, multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        print(f"Parent: n={len(a)}, obj={compute_obj(a):.8f}")
    except Exception:
        a = np.ones(800) / 800
        print(f"No parent, uniform n=800, obj={compute_obj(a):.8f}")

    best_a = a.copy()
    best_obj = compute_obj(a)
    n_base = len(best_a)

    # Phase 1: Primary LP alternating
    print(f"\n=== Phase 1: LP Alt (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 350.0)
    a_p1 = lp_alternate(best_a, time_budget=p1_budget, verbose=True, rng=rng)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  P1: obj={best_obj:.8f} **")
    else:
        print(f"  P1: obj={obj_p1:.8f}")

    # Phase 2: Coordinate descent
    print(f"\n=== Phase 2: CoordDescent ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.08, 45.0)
        a_cd = coordinate_descent(best_a, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd
            print(f"  CD: obj={best_obj:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # Phase 3: LP alternating round 2
    print(f"\n=== Phase 3: LP Alt Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 250.0)
        a_p3 = lp_alternate(best_a, time_budget=alloc, verbose=True, rng=rng)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            print(f"  P3: obj={best_obj:.8f} **")
        else:
            print(f"  P3: obj={obj_p3:.8f}")

    # Phase 4: Multi-n exploration
    print(f"\n=== Phase 4: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_base = len(best_a)
    n_options = sorted(set([
        n_base - 400, n_base - 200, n_base - 100,
        n_base,
        n_base + 100, n_base + 200, n_base + 400, n_base + 600,
    ]), key=lambda x: abs(x - n_base))
    n_options = [x for x in n_options if 400 <= x <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
            else:
                continue
        alloc = min(35.0, remaining * 0.18)
        a_opt = lp_alternate(a_try, time_budget=alloc, verbose=False, rng=rng)
        obj_try = compute_obj(a_opt)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        elif remaining > 50:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 5: Diverse initializations
    print(f"\n=== Phase 5: Diverse Inits ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    for d in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        a_div = diverse_initialization(n_best, rng)
        alloc = min(30.0, remaining * 0.12)
        a_opt = lp_alternate(a_div, time_budget=alloc, verbose=False, rng=rng)
        obj_div = compute_obj(a_opt)
        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.8f} **")
        elif remaining > 35:
            print(f"  Diverse {d}: obj={obj_div:.8f}")

    # Phase 6: Final polish
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 12:
        alloc = min(remaining * 0.30, 50.0)
        a_f = lp_alternate(best_a, time_budget=alloc, verbose=True, rng=rng)
        obj_f = compute_obj(a_f)
        if obj_f < best_obj:
            best_obj = obj_f
            best_a = a_f
            print(f"  Final LP: obj={best_obj:.8f} **")
        else:
            print(f"  Final LP: obj={obj_f:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.20, 25.0)
        a_cd = coordinate_descent(best_a, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd
            print(f"  Final CD: obj={best_obj:.8f} **")
        else:
            print(f"  Final CD: obj={obj_cd:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 8:
        alloc = min(remaining * 0.35, 20.0)
        a_last = lp_alternate(best_a, time_budget=alloc, verbose=False, rng=rng)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.8f} **")

    # Validation
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")
    return result
