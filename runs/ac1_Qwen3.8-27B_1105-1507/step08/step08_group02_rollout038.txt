The objective is to minimize: `2 * n * max(conv(a, a)) / (sum(a))^2`

where `conv(a, a)` is the full linear convolution of the sequence with itself.

Key observations:
1. The objective is scale-invariant (normalizing a doesn't change it)
2. For a uniform sequence of length n, conv(a,a) has a triangular shape with max at the center = n * (1/n)^2 = 1/n, so the objective = 2*n*(1/n)/1 = 2
3. We need to get below 1.5027, which means we need sequences where the max autocorrelation is as small as possible relative to the sum squared.

The theoretical minimum relates to the "autocorrelation problem" or "aperiodic autocorrelation" problem. For binary sequences, the best known values relate to Legendre sequences and similar constructions. But here we have non-negative real-valued sequences.

Let me think about what makes a good sequence:
- We want the autocorrelation to be as flat as possible (low peak-to-average ratio)
- The sum of all autocorrelation values = (sum a)^2
- There are 2n-1 autocorrelation values
- So the average autocorrelation value = (sum a)^2 / (2n-1)
- The minimum possible max is at least (sum a)^2 / (2n-1)
- This gives a lower bound of 2n * (sum a)^2 / (2n-1) / (sum a)^2 = 2n/(2n-1) ≈ 1 for large n

So theoretically we can get close to 1, but in practice with non-negative sequences it's harder.

For non-negative sequences, the key insight is that a[0] * a[k] + a[k] * a[0] + ... contributes to conv[k]. Since all terms are non-negative, we can't use cancellation.

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj(a, beta):
    """Smooth approximation: 2n * logsumexp(conv) / beta (normalized)."""
    conv = compute_conv(a)
    m = conv.max()
    shifted = beta * (conv - m)
    lse = m + np.log(np.sum(np.exp(shifted))) / beta
    return 2.0 * len(a) * lse


def smooth_grad(a, beta):
    """Gradient of smooth objective w.r.t. a (on the simplex)."""
    n = len(a)
    conv = compute_conv(a)
    m = conv.max()
    shifted = beta * (conv - m)
    w = np.exp(shifted)
    w /= w.sum()
    
    # Gradient of max approx: g[j] = 2 * sum_k w[k] * a[k-j]
    # where the sum is over valid k-j indices
    g = np.zeros(n)
    for j in range(n):
        k_start = max(0, j)
        k_end = min(2 * n - 2, j + n - 1)
        # a[k-j] for k in [k_start, k_end]
        idx = np.arange(k_start, k_end + 1) - j
        g[j] = 2.0 * np.dot(w[k_start:k_end + 1], a[idx])
    return 2.0 * n * g

def build_conv_matrix(a_norm):
    """Build matrix A such that A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp(a, time_limit=30.0):
    """
    Solve: min M s.t. conv(a, g)[k] <= M, sum(g)=1, g>=0.
    Returns g or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    # Try cvxpy with CLARABEL first (faster for this size)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > time_limit:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    
    # Fallback: scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': time_limit})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    
    return None

def subgradient_descent(a, time_budget=100.0, topK=5, verbose=False):

    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:topK]
        
        # Weighted subgradient from top peaks
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break
        weights = np.exp(-2.0 * (max_val - peak_vals) / max_val)
        weights /= weights.sum()
        
        g = np.zeros(n)

        y_new = y_new / y_new.sum()

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
    elif mode == 2:
        # Cosine hump
        a = np.abs(np.cos(np.pi * x)) ** 2
    elif mode == 3:
        # Broad top-hat
        a = np.exp(-((x - 0.5) ** 2) / 0.05)
    elif mode == 4:
        # Two humps
        a = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02)
    elif mode == 5:
        # Cosine power
        a = np.abs(np.cos(np.pi * x)) ** 4
    elif mode == 6:
        # Sinc-like
        t = (x - 0.5) * 10
        a = np.sinc(t) ** 2 + 0.1
    elif mode == 7:
        # Polynomial bump
        a = (4 * x * (1 - x)) ** 0.5
    elif mode == 8:
        # Asymmetric
        a = x ** 0.3 + (1 - x) ** 1.5 + 0.1
    elif mode == 9:
        # High-frequencycosine mix
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0.05)
    elif mode == 10:
        # Beta-like distribution
        a = (x * (1 - x) + 0.01) ** 0.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def coordinate_descent(a, time_budget=30.0):
    """Fast coordinate descent: adjust pairs of coordinates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    
    rounds = 0
    while time.monotonic() - start < time_budget and rounds < 10:
        rounds += 1
        improved = False
        
        # Find top contributors to peak
        pairs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                pairs.append((a[i] * a[j], i, j))
        pairs.sort(reverse=True)
        
        # Find non-contributors (small a values)
        sorted_idx = np.argsort(a)
        low_idx = sorted_idx[:n // 4]
        
        for contrib, i, j in pairs[:20]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break
            
            src = i if a[i] >= a[j] else j
            if src == i and i == j:
                src = i
            
            for target in low_idx[:5]:
                if target == src or target == i or target == j:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[src]
                    if delta <= 0 or delta >= a[src]:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
        
        if not improved:
            break
    
    return best_a

                a_r = lp_alternate(a_r, time_budget=min(remaining * 0.1, 40.0))
        
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Init {r} (mode={mode}): obj={best_score:.8f} ** (init was {init_obj:.8f})")
        else:
            print(f"  Init {r} (mode={mode}): obj={obj_r:.8f} (init was {init_obj:.8f})")
    
    # === Phase 6: Final intensive LP polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    # Extended LP
    if remaining > 50:
        lp_alloc = min(remaining * 0.35, 150.0)
        a_lp = lp_alternate(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
    
    # Nesterov polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        md_alloc = min(remaining * 0.2, 60.0)
        a_md = nesterov_smooth(best_seq, time_budget=md_alloc, beta_start=100, beta_end=8000)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final Nesterov: obj={best_score:.8f} **")
    
    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        cd_alloc = min(remaining * 0.2, 30.0)
        a_cd = coordinate_descent(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
    
    # One more short LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternate(best_seq, time_budget=min(remaining * 0.4, 40.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")
    
    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
