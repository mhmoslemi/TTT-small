            a_try[i] -= delta
            # Redistribute to random other coordinates
            others = np.delete(np.arange(n), i)
            a_try[others] += delta / len(others)
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()

            if obj_try < best_obj:

                improved = True

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses targeted peak-equalization LP, adaptive constraint tightening, line search,
coordinate-level refinement, multi-scale exploration, and structured restarts.
Key improvements: LP that specifically targets the current peak (not symmetric min-M),
iterative constraint tightening, golden-section line search along LP directions,
and peak-equalization heuristic that balances top-K convolution peaks.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def standard_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def targeted_lp(a, peak_k, tighten=0.0, verbose=False):
    """
    Solve LP that specifically targets reducing the peak at lag peak_k.
    min: A[peak_k] @ g
    s.t.: A[k] @ g <= (1+tighten) * c[k] for all k
          sum(g) = 1, g >= 0
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    c = A @ a_norm
    g = cp.Variable(n, nonneg=True)
    rhs = c * (1.0 + tighten)
    rhs[peak_k] = c[peak_k]
    constraints = [A @ g <= rhs, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(A[peak_k] @ g), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def peak_equalize_lp(a, top_k=3, tighten=0.02):
    """
    LP that equalizes the top-k peaks: minimize the sum of top-k convolution values.
    s.t.: A @ g <= (1+tighten) * c, sum(g)=1, g>=0
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    c = A @ a_norm
    g = cp.Variable(n, nonneg=True)
    top_indices = np.argsort(c)[::-1][:top_k]
    rhs = c * (1.0 + tighten)
    constraints = [A @ g <= rhs, cp.sum(g) == 1]
    obj = cp.sum([A[k] @ g for k in top_indices])
    prob = cp.Problem(cp.Minimize(obj), constraints)
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    return None


def golden_line_search(a, direction, max_t=1.0):
    """
    Golden section line search: find best t in [0, max_t]
    such that a + t*(direction - a) minimizes objective.
    """
    best_obj = compute_obj(a)
    best_a = a.copy()
    best_t = 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max_t
    for _ in range(25):
        if hi - lo < 1e-10:
            break
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        a1 = a + m1 * (direction - a)
        a1 = np.maximum(a1, 0)
        s1 = a1.sum()
        if s1 < 1e-15:
            obj1 = np.inf
        else:
            obj1 = compute_obj(a1 / s1)
        a2 = a + m2 * (direction - a)
        a2 = np.maximum(a2, 0)
        s2 = a2.sum()
        if s2 < 1e-15:
            obj2 = np.inf
        else:
            obj2 = compute_obj(a2 / s2)
        if obj1 < obj2:
            hi = m2
        else:
            lo = m1
        t_mid = (lo + hi) / 2.0
        a_mid = a + t_mid * (direction - a)
        a_mid = np.maximum(a_mid, 0)
        s_mid = a_mid.sum()
        if s_mid < 1e-15:
            continue
        a_mid = a_mid / s_mid
        obj_mid = compute_obj(a_mid)
        if obj_mid < best_obj:
            best_obj = obj_mid
            best_a = a_mid.copy()
            best_t = t_mid
    return best_a, best_t, best_obj


def lp_polish_targeted(a, time_budget=100.0, verbose=False):
    """
    Iterated LP polishing with targeted peaks, constraint tightening, and line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    tighten = 0.0
    stall = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_k = int(np.argmax(conv))
        peak_val = float(conv[peak_k])
        top_k = min(5, len(conv))
        top_frac = conv[:top_k].sum() / (conv.sum() + 1e-30)
        if it % 3 == 0 or top_frac > 0.3:
            g_lp = targeted_lp(a, peak_k, tighten=tighten)
        else:
            g_lp = peak_equalize_lp(a, top_k=3, tighten=tighten)
        if g_lp is None:
            g_lp, _ = standard_lp(a)
        if g_lp is None:
            break
        a_ls, t_used, obj_ls = golden_line_search(a, g_lp, max_t=1.0)
        if obj_ls < best_obj - 1e-14:
            best_obj = obj_ls
            best_a = a_ls.copy()
            a = a_ls
            stall = 0
            tighten = max(0.0, tighten - 0.01)
        else:
            stall += 1
            tighten = min(0.5, tighten + 0.02)
            if stall > 15:
                g_std, _ = standard_lp(a)
                if g_std is not None:
                    a_ls, t_used, obj_ls = golden_line_search(a, g_std, max_t=0.5)
                    if obj_ls < best_obj - 1e-14:
                        best_obj = obj_ls
                        best_a = a_ls.copy()
                        a = a_ls
                        stall = 0
                        tighten = max(0.0, tighten - 0.02)
                    else:
                        a = a * np.exp(np.random.default_rng(it).normal(0, 0.01, n))
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        stall = 0
                        tighten = 0.0
                else:
                    break
        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    LPT it{it}: obj={best_obj:.8f}, tighten={tighten:.3f}, t={elapsed:.0f}s")
    return best_a


def coordinate_refine(a, time_budget=60.0, verbose=False):
    """
    Coordinate-level refinement: adjust individual coordinates to reduce the peak.
    Uses the structure of convolution to efficiently identify best moves.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_k = int(np.argmax(conv))
        i_start = max(0, peak_k - (n - 1))
        i_end = min(n - 1, peak_k)
        contributions = np.zeros(n)
        for i in range(i_start, i_end + 1):
            j = peak_k - i
            if 0 <= j < n:
                contributions[i] += a[j]
                if i != j:
                    contributions[j] += a[i]
        top_count = min(30, n)
        top_idx = np.argsort(contributions)[::-1][:top_count]
        bottom_count = min(15, n)
        bottom_idx = np.argsort(contributions)[:bottom_count]
        improved = False
        for i in top_idx:
            delta = a[i] * rng.uniform(0.005, 0.03)
            if delta < 1e-15:
                continue
            a_try = a.copy()
            a_try[i] -= delta
            others = np.delete(np.arange(n), i)
            weights = a[others] + 1e-10
            weights = weights / weights.sum()
            a_try[others] += delta * weights
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                a = a_try.copy()
                improved = True
        if not improved:
            for i in bottom_idx:
                max_i = int(np.argmax(a))
                delta = a[max_i] * rng.uniform(0.003, 0.015)
                if delta < 1e-15:
                    continue
                a_try = a.copy()
                a_try[max_i] -= delta
                a_try[i] += delta
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    a = a_try.copy()
                    improved = True
        if not improved:
            top_peak2 = np.argsort(conv)[::-1][1]
            for i in range(max(0, top_peak2 - (n-1)), min(n, top_peak2 + 1)):
                if i >= n:
                    break
                delta = a[i] * 0.01
                if delta < 1e-15:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                others = np.delete(np.arange(n), i)
                a_try[others] += delta / len(others)
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    a = a_try.copy()
                    improved = True
        if not improved:
            break
        it += 1
        if verbose and it % 100 == 0:
            print(f"    Coord it{it}: obj={best_obj:.8f}")
    return a


def smoothed_gradient_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0):
    """
    Nesterov accelerated gradient descent on log-sum-exp smoothed max of convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        size = 1
        while size < 3 * n:
            size <<= 1
        P = np.fft.fft(p, size)
        A = np.fft.fft(y, size)
        g_full = np.fft.ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9997)
        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 / (np.sqrt(1 - 4 * x * (1 - x) + 1e-8))
        a = a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = x ** rng.uniform(0.5, 3.0)
        a = a / a.sum()
    elif mode == 5:
        c1 = n // 3
        c2 = 2 * n // 3
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) * 0.7
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2)) * 1.3
        a = a / a.sum()
    elif mode == 6:
        a = np.ones(n)
        for _ in range(3):
            center = rng.integers(0, n)
            width = rng.integers(n // 8, n // 3)
            strength = rng.uniform(0.3, 1.0)
            a -= strength * np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = rng.uniform(0.5, 1.5, n)
        a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: targeted LP + peak equalization + coordinate refinement + multi-n + restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    print(f"\n=== Phase 1: Targeted LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 30:
        a_lp = lp_polish_targeted(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase1 LPT: obj={best_score:.8f} **")
        else:
            print(f"  Phase1 LPT: obj={obj_lp:.8f}")

    print(f"\n=== Phase 2: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 150.0)
    if p2_budget > 20:
        a_sgd = smoothed_gradient_descent(best_seq, time_budget=p2_budget,
                                          tau_init=5.0, tau_end=400.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Phase2 GD: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 GD: obj={obj_sgd:.8f}")

    print(f"\n=== Phase 3: Coordinate Refinement ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 20:
        a_coord = coordinate_refine(best_seq, time_budget=p3_budget, verbose=True)
        obj_coord = compute_obj(a_coord)
        if obj_coord < best_score:
            best_score = obj_coord
            best_seq = a_coord.copy()
            print(f"  Phase3 Coord: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 Coord: obj={obj_coord:.8f}")

    print(f"\n=== Phase 4: Intensive LP with Tightening ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.2, 250.0)
    if p4_budget > 30:
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p4_budget / (3 - rnd), remaining * 0.5)
            if alloc < 20:
                break
            a_lp = lp_polish_targeted(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Phase4 LP rnd{rnd}: obj={best_score:.8f} **")
            else:
                print(f"  Phase4 LP rnd{rnd}: obj={obj_lp:.8f}")
                if rnd < 2:
                    scale = 0.015 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    n_options = sorted(set([
        max(400, n_current - 150), max(400, n_current - 50),
        n_current + 50, n_current + 100, n_current + 200,
        600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_current) > 30]

    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        a_try = resample_sequence(best_seq, n_try)
        alloc1 = min(remaining * 0.2, 60.0)
        if alloc1 > 15:
            a_try = smoothed_gradient_descent(a_try, time_budget=alloc1,
                                              tau_init=10.0, tau_end=300.0)
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 50.0)
        if alloc2 > 15:
            a_try = lp_polish_targeted(a_try, time_budget=alloc2)
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 20:
            half = alloc * 0.5
            a_r = smoothed_gradient_descent(a_r, time_budget=half,
                                            tau_init=10.0, tau_end=200.0)
            remaining = deadline - time.monotonic()
            alloc_lp = min(remaining * 0.1, 40.0)
            if alloc_lp > 10:
                a_r = lp_polish_targeted(a_r, time_budget=alloc_lp)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    print(f"\n=== Phase 7: Final Intensive Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(remaining * 0.3, 100.0)
        a_final = lp_polish_targeted(best_seq, time_budget=alloc, verbose=True)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 80.0)
        a_coord = coordinate_refine(best_seq, time_budget=alloc)
        obj_c = compute_obj(a_coord)
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_coord.copy()
            print(f"  Final Coord: obj={best_score:.8f} **")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 60.0)
        a_sgd = smoothed_gradient_descent(best_seq, time_budget=alloc,
                                          tau_init=50.0, tau_end=800.0)
        obj_s = compute_obj(a_sgd)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_sgd.copy()
            print(f"  Final GD: obj={best_score:.8f} **")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 50.0)
        a_final = lp_polish_targeted(best_seq, time_budget=alloc)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
