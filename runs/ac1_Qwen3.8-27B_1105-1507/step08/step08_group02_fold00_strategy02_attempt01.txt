def compute_conv(a):
    """Compute the full convolution using FFT."""
    n_len = len(a)
    if n_len == 0:
        return np.array([])

    size = 1

        size *= 2
    fa = rfft(a.astype(np.float64), size)
    conv_result = irfft(fa * fa, size)[:2 * n_len - 1]
    return conv_result
