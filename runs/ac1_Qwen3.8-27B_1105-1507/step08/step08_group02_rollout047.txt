"""
Optimized step function search using FFT-based subgradient methods with
Nesterov acceleration, LP alternating with dual-guided perturbation,
and multi-length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_of_max_conv(a, K=None):

    Compute subgradient of max_k (a*a)_k w.r.t. a.

    # This is the cross-correlation of ind and a at lag i

def smooth_subgradient(a, beta):
    """
    Compute smooth subgradient using log-sum-exp.
    grad_i = 2 * sum_k w_k * a_{k-i} where w_k = softmax(beta * conv[k])
    """
    n = len(a)
    conv = compute_conv(a)
    m = conv.max()
    shifted = beta * (conv - m)
    w = np.exp(shifted)
    w /= w.sum()
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    ind = np.zeros(size)
    ind[:2*n-1] = w
    
    fa = rfft(a, size)
    fi = rfft(ind, size)
    result = irfft(fi * np.conj(fa), size)[:n]
    return 2.0 * result


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimized step function search using FFT-based subgradient methods, LP alternating
with dual-guided perturbation, mirror descent with top-K peaks, and multi-length
exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective for sequence: 2*n*max(conv)/(sum)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def obj_normalized(a):
    """Objective assuming a is already normalized (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_topk(a, K=1, temperature=1.0):
    """
    Compute weighted subgradient of max(conv(a,a)) using top-K positions.
    Uses softmax weighting over top-K convolution values.
    Returns gradient vector of length n (for descent, use negative).
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Get top-K positions
    if K >= 2 * n - 1:
        # Use full softmax
        m = conv.max()
        w = np.exp(temperature * (conv - m))
        w /= w.sum()
        sizes = 1
        while sizes < 2 * n - 1:
            sizes <<= 1
        ind = np.zeros(sizes)
        ind[:2*n-1] = w
        fa = rfft(a, sizes)
        fi = rfft(ind, sizes)
        result = irfft(fi * np.conj(fa), sizes)[:n]
        return 2.0 * result
    
    # Top-K with softmax
    top_idx = np.argsort(conv)[::-1][:K]
    top_vals = conv[top_idx]
    m = top_vals[0]
    weights = np.exp(temperature * (top_vals - m))
    weights /= weights.sum()
    
    # grad_i = 2 * sum_k w_k * a_{k-i}
    sizes = 1
    while sizes < 2 * n - 1:
        sizes <<= 1
    
    ind = np.zeros(sizes)
    for k, w in zip(top_idx, weights):
        ind[int(k)] = w
    
    fa = rfft(a, sizes)
    fi = rfft(ind, sizes)
    result = irfft(fi * np.conj(fa), sizes)[:n]
    return 2.0 * result


def solve_lp_direction(a):
    """
    Solve LP: min M s.t. conv(a,g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value).
    """
    n = len(a)
    a_s = float(np.sum(a))
    if a_s < 1e-10:
        return None, None
    a_norm = a / a_s

    # Build constraint matrix efficiently
    # A_ub: (2n-1) x (n+1), last column is -1 for M
    # A_ub[k, j] = a_norm[k-j] if 0 <= k-j < n else 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, n] = -1.0

    for j in range(n):
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        length = k_hi - k_lo + 1
        A_ub[k_lo:k_hi + 1, j] = a_norm[:length]

    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    c = np.zeros(n + 1)
    c[n] = 1.0

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass

    return None, None

def coordinate_refine(a, time_budget=100.0):
    """
    Coordinate descent: redistribute mass from peak-contributing positions
    to non-contributing positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()

    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    max_rounds = 10
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        if k_star >= len(conv):
            break

        improved = False
        
        # Find top contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        # Find non-contributing positions
        non_contrib = set(range(n))
        for _, i, j in contributors[:15]:
            non_contrib.discard(i)
            if i != j:
                non_contrib.discard(j)
        non_contrib_list = sorted(non_contrib, key=lambda x: -a[x])[:12]

        if not non_contrib_list:
            break

        top_count = min(20, len(contributors))
        for idx in range(top_count):
            if time.monotonic() - start > time_budget:
                break
            contrib, i, j = contributors[idx]
            if contrib < 1e-10:
                break

            for target in non_contrib_list[:6]:
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    if i == j:
                        delta = frac * a[i]
                    else:
                        delta = frac * min(a[i], a[j])
                    if delta < 1e-12 or delta >= min(a[i], a[j]) * 0.9:
                        continue

                    src = i if a[i] >= a[j] else j if i != j else i
                    orig_val = a[src].copy()
                    a[src] -= delta
                    a[target] += delta

                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))

                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        a[src] = orig_val
                        a[target] -= delta
                        break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if new_n == 1:
        return np.array([float(np.sum(a))])
    
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):

        # Hanning window
        x = np.arange(n) / (n - 1)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 3:
        # Raised cosine squared
        x = np.arange(n) / (n - 1)
        a = (0.5 * (1 + np.cos(2 * np.pi * x)))**2
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 4:
        # Gaussian bump
        c = n // 2
        w = n // 6
        a = np.exp(-((np.arange(n) - c)**2) / (2 * w**2))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 5:
        # Two bumps
        a = np.zeros(n)
        for c in [n//4, 3*n//4]:
            w = n // 10
            a += np.exp(-((np.arange(n) - c)**2) / (2 * w**2))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 6:
        # Inverse U (concentrated at edges)
        x = np.arange(n) / (n - 1)
        a = 0.5 + 0.5 * np.cos(np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Power law
        x = np.arange(1, n + 1) / n
        a = x**(-0.5) + (1-x)**(-0.5) + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: hybrid LP alternating, mirror descent, Fourier perturbation,
    coordinate descent, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = obj_normalized(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = obj_normalized(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}")

    # === Phase 1: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 500.0)
    if p1_budget > 30:
        a_lp = lp_alternating_optimize(best_seq, time_budget=p1_budget, verbose=True,
                                       use_dual_perturb=True)
        obj_lp = obj_normalized(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  P1 LP: obj={best_score:.8f} **")
        else:
            print(f"  P1 LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Mirror Descent with Top-K ===
    print(f"\n=== Phase 2: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 200.0)
    if p2_budget > 20:
        a_md = mirror_descent_topk(best_seq, time_budget=p2_budget, verbose=True)
        obj_md = obj_normalized(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  P2 MD: obj={best_score:.8f} **")
        else:
            print(f"  P2 MD: obj={obj_md:.8f} (no improvement)")

    # === Phase 3: Fourier Domain Search ===
    print(f"\n=== Phase 3: Fourier Perturbation ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 150.0)
    if p3_budget > 20:
        a_ft = fourier_perturb_search(best_seq, time_budget=p3_budget, verbose=True)
        obj_ft = obj_normalized(a_ft)
        if obj_ft < best_score:
            best_score = obj_ft
            best_seq = a_ft.copy()
            print(f"  P3 FT: obj={best_score:.8f} **")
        else:
            print(f"  P3 FT: obj={obj_ft:.8f} (no improvement)")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(400, n_best - 100), max(400, n_best - 50),
        n_best + 50, n_best + 100, n_best + 200,
        600, 700, 900, 1000, 1100, 1200, 1300
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and n != n_best]
    n_options = n_options[:8]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 80.0)

        # Quick LP optimization
        if alloc > 20:
            a_opt = lp_alternating_optimize(a_try, time_budget=alloc * 0.6, verbose=False)
            obj_opt = obj_normalized(a_opt)
            # Mirror descent refinement
            remaining2 = deadline - time.monotonic()
            if remaining2 > 30 and obj_opt < best_score:
                a_opt = mirror_descent_topk(a_opt, time_budget=min(remaining2 * 0.1, 30.0), verbose=False)
                obj_opt = obj_normalized(a_opt)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")
        else:
            obj_try = obj_normalized(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** n={n_try} (raw): obj={best_score:.8f} **")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r + 2) % 8
        a_r = diverse_init(n_use, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.15, 60.0)
        if restart_budget > 30:
            a_r = lp_alternating_optimize(a_r, time_budget=restart_budget, verbose=False)

        obj_r = obj_normalized(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 50:
        lp_alloc = min(remaining * 0.35, 150.0)
        a_lp = lp_alternating_optimize(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = obj_normalized(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Mirror descent
    remaining = deadline - time.monotonic()
    if remaining > 30:
        md_alloc = min(remaining * 0.25, 60.0)
        a_md = mirror_descent_topk(best_seq, time_budget=md_alloc, verbose=False)
        obj_md = obj_normalized(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd_alloc = min(remaining * 0.3, 50.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_alloc)
        obj_cd = obj_normalized(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more LP round if time permits
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_last = lp_alternating_optimize(best_seq, time_budget=min(remaining * 0.5, 50.0), verbose=False)
        obj_last = obj_normalized(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = obj_normalized(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
