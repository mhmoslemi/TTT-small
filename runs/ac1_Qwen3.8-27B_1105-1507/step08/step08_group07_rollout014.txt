**Coordinate descent with exact line search:**
For a fixed coordinate `a_i`, the objective `max_k conv(a,a)(k)` as a function of `a_i` is the max of quadratic functions (each `conv(a,a)(k)` is quadratic in `a_i` when other coordinates are fixed). So we can find the optimal `a_i` by checking the minimax point.

Actually, `conv(a,a)(k)` as a function of `a_i` (with other a's fixed) is:
`conv(a,a)(k) = 2 * a_i * (sum of relevant a's) + (constant terms not involving a_i)`

Wait, let me be more precise. `conv(a,a)(k) = sum_j a_j * a_{k-j}`. The terms involving `a_i` are:
- `a_i * a_{k-i}` (when j = i)
- `a_{k-i} * a_i` (when k-j = i, i.e., j = k-i)

These are the same term if `k-i = i` (k = 2i). So:
- If `k ≠ 2i`: contribution is `2 * a_i * a_{k-i}`
- If `k = 2i`: contribution is `a_i^2`

So `conv(a,a)(k)` is linear in `a_i` for `k ≠ 2i` and quadratic (specifically `a_i^2 + linear`) for `k = 2i`.

The max of a set of linear functions and one quadratic function is a convex function of `a_i` (since each is convex and max of convex is convex). Wait, `a_i^2 + c` is convex, and `2*a_i*d + c` is both convex and concave (linear). The max of convex functions is convex. So yes, `max_k conv(a,a)(k)` is convex in `a_i` when other coordinates are fixed!

This means coordinate descent with exact line search will work! For each coordinate, we minimize a convex piecewise-linear (plus one quadratic) function. The minimum can be found by:
1. Identifying the active constraints (which k values achieve the max)
2. The minimum is at a point where either the derivative is zero (if the quadratic is active) or two constraints are tight (for the piecewise-linear part)

Actually, this is a 1D minimization of a convex function, which can be done by bisection or ternary search.

**But**: The overall objective is not convex (as we discussed), so coordinate descent may cycle or get stuck. However, with a good initialization and careful implementation, it can still be effective as a polish step.

```python
"""
Combined optimizer: LP alternating + smoothed gradient + coordinate descent
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. A@a_cross(z) <= M, sum(z)=1, z>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    prob = cp.Problem(cp.Minimize(M), [A @ z <= M, cp.sum(z) == 1])
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(prob.constraints[0].dual_value), 0)
                    except:
                        try:
                            dual_y = np.maximum(np.array(prob.constraints[1].dual_value), 0)
                        except:
                            pass
                    return z_val, float(M.value), dual_y
        except:
            continue
    
    # Fallback
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except:
        pass
    
    return None, None, None


def lp_alternating(a, time_budget=200.0, verbose=False):
    """LP alternating with 2-cycle breaking and momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    prev_a = None
    prev_prev_a = None
    g_momentum = a.copy()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp, dual_y = result
        
        # 2-cycle detection
        if prev_prev_a is not None:
            if np.max(np.abs(a - prev_prev_a)) < 1e-7:
                a_break = 0.5 * (a + prev_a)
                a_break = np.maximum(a_break, 0)
                a_break = a_break / np.sum(a_break)
                obj_break = compute_obj(a_break)
                if obj_break < best_obj:
                    best_obj = obj_break
                    best_a = a_break.copy()
                a = a_break
                prev_a = None
                prev_prev_a = None
                g_momentum = a.copy()
                continue
        
        # Update momentum
        g_momentum = 0.5 * g_momentum + 0.5 * g_lp
        g_momentum = np.maximum(g_momentum, 0)
        gs = float(np.sum(g_momentum))
        if gs > 1e-10:
            g_momentum = g_momentum / gs
        
        # Try multiple mixing ratios
        best_cand_obj = best_obj
        best_cand = a.copy()
        
        for target in [g_lp, g_momentum]:
            for mix in [0.3, 0.5, 0.7, 1.0]:
                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj:
                    best_cand_obj = obj_try
                    best_cand = a_try.copy()
        
        if best_cand_obj < best_obj - 1e-12:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            prev_a = a.copy()
            prev_prev_a = prev_a
            a = best_a
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 20:
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev_prev_a = None
                g_momentum = a.copy()
                stall = 0
    
    return best_a


def smooth_gradient(a, t):
    """Compute gradient of t*log(sum(exp(conv(a,a)/t))) with respect to a.
    Returns grad of length n."""
    n = len(a)
    c = compute_conv(a)  # length 2n-1
    c_norm = c - np.max(c)
    p = np.exp(c_norm / t)
    p_sum = np.sum(p)
    if p_sum < 1e-300:
        p = np.ones(2*n-1) / (2*n-1)
    else:
        p = p / p_sum
    
    # grad[m] = 2 * sum_j a[j] * p[m+j] (with p zero outside [0, 2n-2])
    # = convolve(a[::-1], p_padded)[n-1+m]
    q = np.zeros(3 * n)
    q[:2*n-1] = p
```

Combined optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
1. LP alternating with adaptive line search
2. Smoothed gradient descent polish
3. Multi-n exploration with resampling
4. Random restarts for escape from local optima

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_cross_conv(a, b):
    """Compute linear convolution a*b using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def proj_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. cross_conv(a, z) <= M, sum(z)=1, z>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ z <= M, cp.sum(z) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(constraints[0].dual_value), 0)
                    except:
                        pass
                    return z_val, float(M.value), dual_y
        except:
            continue
    
    return None, None, None


def line_search_lp_direction(a, g, max_iter=50):
    """
    Find t* in [0,1] minimizing f((1-t)*a + t*g) using golden section search.
    f(t) = 2n * max_k conv((1-t)a+tg, (1-t)a+tg)(k) (with normalization sum=1)
    """
    n = len(a)
    # Precompute
    aa_conv = compute_conv(a)  # conv(a,a)
    ag_conv = compute_cross_conv(a, g)  # conv(a,g)
    gg_conv = compute_conv(g)  # conv(g,g)
    
    def f_of_t(t):
        # conv((1-t)a+tg, (1-t)a+tg)(k) = (1-t)^2 * aa(k) + 2t(1-t) * ag(k) + t^2 * gg(k)
        val = (1 - t)**2 * aa_conv + 2 * t * (1 - t) * ag_conv + t**2 * gg_conv
        return 2.0 * n * float(np.max(val))
    
    # Golden section search on [0, 1]
    gr = (np.sqrt(5) - 1) / 2  #golden ratio ~0.618
    a_low, a_high = 0.0, 1.0
    c = a_high - gr * (a_high - a_low)
    d = a_low + gr * (a_high - a_low)
    fc = f_of_t(c)
    fd = f_of_t(d)
    
    for _ in range(max_iter):
        if fc < fd:
            a_high = d
            d = c
            fd = fc
            c = a_high - gr * (a_high - a_low)
            fc = f_of_t(c)
        else:
            a_low = c
            c = d
            fc = fd
            d = a_low + gr * (a_high - a_low)
            fd = f_of_t(d)
    
    t_star = (a_low + a_high) / 2
    return t_star
