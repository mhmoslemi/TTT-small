def compute_conv(a):
    """Compute linear autoconvolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:2 * n - 1]
    return conv


def compute_obj_normalized(a):
    """Compute 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))
