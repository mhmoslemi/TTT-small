        if np.isnan(x) or np.isinf(x):

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def active_set_loop(a, time_budget=100.0):

    while time.monotonic() - start < time_budget:

        if result[0] is None:
            break
        g_full, M_full, dual_y = result

        if dual_y is not None and np.max(dual_y) > 0:

            n_active = int(np.sum(active_mask))

            if 0 < n_active < 2 * n - 1:

                if z_red is not None:

                    max_all = float(np.max(conv_all))
                    if max_all <= M_full * 1.005:
                        obj_red = compute_obj(z_red)
                        if obj_red < best_obj - 1e-12:
                            best_obj = obj_red
                            best_a = z_red.copy()
                            a = z_red
                            continue

        # Standard update
        a_new = 0.4 * a + 0.6 * g_full
        a_new = np.maximum(a_new, 0)
        a_new = a_new / np.sum(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            improved = False
            for step in [0.1, 0.2, 0.3]:
                a_try = (1 - step) * a + step * g_full
                a_try = np.maximum(a_try, 0)

                    a = a_try

                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def pairwise_exchange(a, time_budget=40.0):
    """
    Pairwise exchange polish: try swapping mass between pairs of coordinates.
    More effective than single-coordinate moves.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for _ in range(20):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            for frac in [0.05, 0.1, 0.2, 0.02]:
                delta = frac * min(a[i], a[j])
                if delta > 0.5 * min(a[i], a[j]):
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved and it > 30:
            break

    return best_a
