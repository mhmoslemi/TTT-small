def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_descent_v2(a, time_budget=60.0, verbose=False):
    """Mirror descent with adaptive multi-temperature weighting."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)

    min_lr = 1e-5

    rng = np.random.default_rng(99)

        conv = compute_conv(a)

        if max_val < 1e-15:
            break
        
        # Multi-temperature softmax weighting over ALL peaks
        temps = [1.0, 3.0, 10.0, 30.0]
        all_w = np.zeros(len(conv))
        for temp in temps:
            w = np.exp(-(conv - max_val) / (temp * max_val + 1e-15))
            w /= w.sum()
            all_w += w
        all_w /= len(temps)
        
        # Use only significant peaks for efficiency
        threshold = all_w.max() * 0.01
        active = all_w > threshold
        if not active.any():
            active[int(np.argmax(conv))] = True
        active_w = all_w[active]
        active_w /= active_w.sum()
        active_idx = np.where(active)[0]
        
        g = np.zeros(n)
        for w, k in zip(active_w, active_idx):
            if w > 1e-8:
                g += w * get_subgradient(a, int(k))
        
        v = momentum * v + g
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = max(min_lr, lr * 0.998)
            momentum = min(0.7, momentum + 0.001)
        else:
            stall += 1

def fourier_perturb(a, rng, n_modes=5, scale=0.1):
    """Perturb a using low-frequency Fourier modes to explore structure."""
    n = len(a)
    a_pert = a.copy()
    x = np.arange(n)
    for m in range(1, n_modes + 1):
        freq = m / n
        amp = rng.normal(0, scale / m)
        phase = rng.uniform(0, 2 * np.pi)
        a_pert = a_pert * np.exp(amp * np.cos(2 * np.pi * freq * x + phase))
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s > 1e-10:
        a_pert = a_pert / s
    return a_pert

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def get_subgradient(a, k):
    """Gradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            if i == j:
                g[i] += 2.0 * a[i]
            else:
                g[i] += a[j]
    return g


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_min_max(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """

    a = a / a.sum()
    A = build_conv_matrix(a)

    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = g.sum()
        if s > 1e-10:
            g = g / s
        M = float(res.x[n])
        return g, M
    return None, None


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_lse_optimize(a, time_budget=40.0, T=20.0):
    """Smooth the objective via log-sum-exp and use L-BFGS-B."""
    n = len(a)
    a = np.maximum(a, 1e-12).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)

    def obj_grad(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = compute_conv(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        e = np.exp(z)
        sum_e = e.sum()
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)
        w = e / sum_e
        g = np.zeros(n)
        active = w > 1e-10
        for k in np.where(active)[0]:
            g += w[k] * get_subgradient(x, int(k))
        g *= 2.0 * n
        return obj_val, g

    try:
        res = sp_minimize(obj_grad, a, method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': 100, 'maxfun': 200, 'disp': False})
        a_opt = np.maximum(res.x, 0)
        s_opt = a_opt.sum()
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
    except Exception:
        pass

    return best_a


def fourier_perturb(a, rng, n_modes=5, scale=0.12):
    """Perturb using low-frequency Fourier modes."""
    n = len(a)
    a_pert = a.copy()
    x = np.arange(n)
    for m in range(1, n_modes + 1):
        freq = m / max(n, 1)
        amp = rng.normal(0, scale / m)
        phase = rng.uniform(0, 2 * np.pi)
        a_pert = a_pert * np.exp(amp * np.cos(2 * np.pi * freq * x + phase))
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s > 1e-10:
        a_pert = a_pert / s
    return a_pert


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        a_opt = subgradient_nesterov(best_seq, time_budget=alloc, K=10)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.5, 25.0)
        a_opt = lp_alternating_aggressive(best_seq, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
