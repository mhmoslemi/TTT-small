"""
Smooth gradient-based optimizer for flat autocorrelation sequences.
Uses power-sum approximation of max with gradient descent,
combined with LP polishing and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def softmax(x):
    """Stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / np.sum(e)

def obj_grad_smooth(x, p):
    """
    Smooth objective and gradient using power-sum approximation.
    a = softmax(x), objective = 2n * (sum (a*a)^p)^(1/p)
    """
    a = softmax(x)
    n = len(a)
    conv = compute_conv(a)
    conv_pos = np.maximum(conv, 1e-30)
    
    Sp = np.sum(conv_pos ** p)
    f = 2.0 * n * Sp ** (1.0 / p)
    
    if not np.isfinite(f):
        return f, np.zeros_like(x)
    
    alpha = 2.0 * n * Sp ** (1.0/p - 1.0)
    weights = conv_pos ** (p - 1.0)
    
    # Compute corr[i] = sum_j a[j] * weights[j+i] for valid indices
    # Using FFT for efficiency
    size = 1
    while size < 3 * n:
        size <<= 1
    
    a_fft = rfft(a, size)
    w_fft = rfft(weights, size)
    
    # corr[i] = sum_j a[j] * weights[j+i]
    # This is the cross-correlation of a and weights
    # = ifft(conj(a_fft) * w_fft)... let me verify
    # 
    # In FFT: if A = FFT(a), W = FFT(w), then:
    # ifft(conj(A) * W)[k] = sum_j a[j] * w[(k-j) mod size]
    # = sum_j a[j] * w[j-k] (with appropriate wrapping)
    # 
    # Hmm, that gives w[j-k], not w[j+k].
    # 
    # Let me use: ifft(A * conj(W))[k] = sum_j a[j] * w[k+j]... no
    # 
    # Standard identity: ifft(FFT(a) * FFT(b))[k] = sum_j a[j] * b[k-j] (circular conv)
    # ifft(FFT(a) * conj(FFT(b)))[k] = sum_j a[j] * b[j-(k mod N)]... 
    # 
    # This is getting confusing. Let me just use the direct method.
    
    # Direct computation: O(n^2)
    corr = np.zeros(n)
    for i in range(n):
        # corr[i] = sum_{j=0}^{min(n-1, 2n-2-i)} a[j] * weights[j+i]
        j_max = min(n, 2*n-1-i)
        if j_max > 0:
            corr[i] = np.dot(a[:j_max], weights[i:i+j_max])
    
    # Diagonal correction
    diag_correction = np.zeros(n)
    valid_diag = 2 * np.arange(n) < 2*n - 1
    diag_correction[valid_diag] = weights[2 * np.arange(n)[valid_diag]] * a[valid_diag]
    
    grad_f = 2.0 * corr - diag_correction
    grad_f *= alpha
    
    # Transform to x-space
    mean_gf = np.dot(grad_f, a)
    grad_x = a * (grad_f - mean_gf)
    
    return f, grad_x

def smooth_optimize(a_init, p_values=None, max_iter=500, time_budget=50.0):
    """
    Optimize using smooth power-sum objective with increasing p.
    a_init: initial sequence (will be normalized)
    p_values: list of p values to anneal through
    """
    if p_values is None:
        p_values = [2, 4, 8, 16, 32, 64, 128, 256]
    
    a = np.maximum(a_init, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a_init
    a = a / s
    
    # Convert to x-space
    x = np.log(np.maximum(a, 1e-15))
    
    start_time = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for p in p_values:
        if time.monotonic() - start_time > time_budget:
            break
        
        t_p_start = time.monotonic()
        p_budget = min(30.0, time_budget * 0.8 / len(p_values))
        
        # Use L-BFGS-B on x
        def fun_and_grad(x_eval):
            f, g = obj_grad_smooth(x_eval, p)
            return f, g
        
        try:
            result = minimize(fun_and_grad, x, method='L-BFGS-B',
                           jac=True, options={'maxiter': 200, 'gtol': 1e-8,
                                             'maxls': 20})
            if result.success or result.fun < best_obj:
                a_new = softmax(result.x)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = result.x.copy()
                    a = a_new.copy()
        except Exception as e:
            pass
        
        # Also try a few Adam steps
        lr = 0.01 / np.sqrt(p)
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        
        for step in range(200):
            if time.monotonic() - t_p_start > p_budget:
                break
            f, g = obj_grad_smooth(x, p)
            t_adam = step + 1
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * g**2
            m_hat = m / (1 - beta1**t_adam)
            v_hat = v / (1 - beta2**t_adam)
            x -= lr * m_hat / (np.sqrt(v_hat) + eps)
            
            a_step = softmax(x)
            obj_step = compute_obj(a_step)
            if obj_step < best_obj:
                best_obj = obj_step
                best_a = a_step.copy()
    
    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_simple(a):
    """Solve LP: minimize M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        prob = cp.Problem(cp.Minimize(M), [A @ g <= M, cp.sum(g) == 1])
        for solver in [cp.CLARABEL, cp.PDLP, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        return g_val / gs
            except:
                continue
    except:
        pass
    
    # Fallback: use scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except:
        pass
    
    return None

def lp_alternating(a, time_budget=30.0, n_iter_max=50):
    """LP alternating method."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(n_iter_max):
        if time.monotonic() - start > time_budget:
            break
        
        g = solve_lp_simple(a)
        if g is None:
            break
        
        # Try different mixing ratios
        improved = False
        for mix in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - mix) * a + mix * g
            a_try = np.maximum(a_try, 0)
            st = np.sum(a_try)
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            # Small perturbation to escape
            rng = np.random.default_rng(hash((it, n, time.monotonic())) % (2**31))
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
            best_obj = compute_obj(a)
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def coordinate_descent(a, time_budget=20.0):
    """Simple coordinate descent polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        # Try moving mass from one coordinate to another
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        for frac in [0.01, 0.05, 0.1, 0.2]:
            delta = frac * min(a[i], a[j])
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            if a_try[i] < 0 or a_try[j] < 0:

"""
Smooth gradient-based optimizer (log-sum-exp smoothed max) combined with
LP alternating, simulated annealing, and multi-n exploration for minimizing
the peak linear autocorrelation of non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_fn(x):
    """Stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    s = np.sum(e)
    return e / s


def obj_grad_lse(x, T):
    """
    Log-sum-exp smoothed objective and gradient.
    Objective: 2n * (1/T) * log(sum_k exp(T * conv[k]))
    where conv = a*a and a = softmax(x).
    """
    a = softmax_fn(x)
    n = len(a)
    conv = compute_conv(a)
    conv_pos = np.maximum(conv, 1e-30)

    Tconv = T * conv_pos
    max_Tconv = np.max(Tconv)
    log_sum_exp = max_Tconv + np.log(np.sum(np.exp(Tconv - max_Tconv)))
    f = 2.0 * n * log_sum_exp / T

    if not np.isfinite(f):
        return f, np.zeros_like(x)

    weights = np.exp(Tconv - log_sum_exp)

    # corr[i] = sum_j a[j] * weights[j+i]
    corr = np.correlate(weights, a, 'full')[n-1 : 2*n-2]

    # Diagonal correction: d(conv[2i])/da[i] = a[i] not 2*a[i]
    diag_idx = 2 * np.arange(n)
    diag = weights[diag_idx] * a

    grad_f = (2.0 * n / T) * (2.0 * corr - diag)

    # Transform to x-space: grad_x[j] = a[j] * (grad_f[j] - <grad_f, a>)
    mean_gf = np.dot(grad_f, a)
    grad_x = a * (grad_f - mean_gf)

    return f, grad_x


def smooth_optimize_lse(a_init, T_values=None, time_budget=100.0, max_lbfgs_iter=300):
    """
    Optimize using log-sum-exp smoothing with increasing T (annealing).
    """
    if T_values is None:
        T_values = [5, 20, 50, 100, 200, 500, 1000, 2000, 4000, 8000]

    a = np.maximum(a_init, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a_init.copy()
    a = a / s

    x = np.log(np.maximum(a, 1e-15))
    start_time = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    for T in T_values:
        elapsed = time.monotonic() - start_time
        if elapsed > time_budget:
            break

        remaining_budget = time_budget - elapsed
        T_budget = min(25.0, remaining_budget * 0.9)
        t_start = time.monotonic()

        # L-BFGS optimization
        lbfgs_iter = max(50, max_lbfgs_iter // max(1, len(T_values)))

        def fun_and_grad(x_eval):
            return obj_grad_lse(x_eval, T)

        try:
            result = minimize(fun_and_grad, x, method='L-BFGS-B',
                              jac=True, options={'maxiter': lbfgs_iter,
                                                 'gtol': 1e-7, 'maxls': 30})
            if result.fun < best_obj:
                a_new = softmax_fn(result.x)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = result.x.copy()
                    a = a_new.copy()
        except Exception:
            pass

        # Adam fine-tuning
        lr = 0.5 / T * np.sqrt(np.mean(a**2)) / max(np.std(a), 1e-10)
        lr = min(max(lr, 1e-5), 0.1)
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1, beta2, eps = 0.8, 0.95, 1e-8

        n_adam = 200
        for step in range(n_adam):
            if time.monotonic() - t_start > T_budget:
                break
            f_s, g_s = obj_grad_lse(x, T)
            t_adam = step + 1
            m = beta1 * m + (1 - beta1) * g_s
            v = beta2 * v + (1 - beta2) * g_s**2
            m_hat = m / (1 - beta1**t_adam)
            v_hat = v / (1 - beta2**t_adam)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            a_step = softmax_fn(x)
            obj_step = compute_obj(a_step)
            if obj_step < best_obj:
                best_obj = obj_step
                best_a = a_step.copy()
                a = a_step.copy()

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)

    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        prob = cp.Problem(cp.Minimize(M), [A @ g <= M, cp.sum(g) == 1])
        for solver in [cp.CLARABEL, cp.PDLP, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        return g_val / gs
            except Exception:
                continue
    except Exception:
        pass

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass

    return None


def lp_alternating(a, time_budget=100.0, n_max_iter=80):
    """LP alternating with momentum and cycle detection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = a.copy()
    prev_a = None
    prev_prev_a = None
    stall = 0

    for it in range(n_max_iter):
        if time.monotonic() - start > time_budget:
            break

        g = solve_lp(a)
        if g is None:
            break

        # 2-cycle detection
        if prev_prev_a is not None:
            if np.max(np.abs(a - prev_prev_a)) < 1e-8:
                a = 0.5 * (a + prev_a)
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev_prev_a = None
                momentum = a.copy()
                continue

        # Update momentum
        momentum = 0.5 * momentum + 0.5 * g
        momentum = np.maximum(momentum, 0)
        momentum = momentum / np.sum(momentum)

        # Try multiple targets and mixing ratios
        improved = False
        for target in [g, momentum]:
            for mix in [0.3, 0.5, 0.7, 0.9, 1.0]:
                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = np.sum(a_try)
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    prev_a = a.copy()
                    prev_prev_a = prev_a
                    a = a_try
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 10:
                rng = np.random.default_rng((it * 7919 + n) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev_prev_a = None
                momentum = a.copy()
                stall = 0

    return best_a


def simulated_annealing(a_init, time_budget=60.0, T_start=0.01, T_end=0.0001):
    """Simulated annealing with various perturbation strategies."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj

    start = time.monotonic()
    n_steps = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        n_steps += 1
        progress = n_steps / max(1, (time.monotonic() - start) / (time_budget / 200.0 + 1))
        T = T_start * (T_end / T_start) ** min(1.0, progress / 50.0)

        # Choose perturbation type
        ptype = rng.integers(0, 4)
        a_try = a.copy()

        if ptype == 0:
            # Gaussian
            scale = 0.02 * (1.0 + T * 50)
            a_try = a * np.exp(rng.normal(0, scale, n))
        elif ptype == 1:
            # Block
            n_blocks = max(3, n // 200)
            block_size = max(1, n // n_blocks)
            b = rng.integers(0, n // block_size)
            si = b * block_size
            ei = min(si + block_size, n)
            a_try[si:ei] *= rng.uniform(0.3, 2.0)
        elif ptype == 2:
            # Pair swap
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n)
            if i1 != i2:
                frac = rng.uniform(0.05, 0.3)
                delta = frac * min(a[i1], a[i2])
                a_try[i1] -= delta
                a_try[i2] += delta
        else:
            # Sparse
            n_sparse = max(3, n // 50)
            idx = rng.choice(n, n_sparse, replace=False)
            a_try[idx] *= rng.uniform(0.3, 2.0, n_sparse)

        a_try = np.maximum(a_try, 0)
        st = np.sum(a_try)
        if st < 1e-10:
            continue
        a_try = a_try / st

        obj_try = compute_obj(a_try)
        delta = obj_try - current_obj

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_try
            current_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent(a, time_budget=30.0):
    """Random coordinate descent polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        improved = False
        for frac in [0.01, 0.03, 0.1, 0.2]:
            delta = frac * min(a[i], a[j])
            if delta <= 0:
                continue
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            a_try = a_try / np.sum(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

    return best_a
