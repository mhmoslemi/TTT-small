import numpy as np
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def gradient_inner(fixed, var, beta):
    """
    Gradient of log-sum-exp(beta * conv(fixed, var)) w.r.t. var.
    fixed and var both have length n.
    """
    n = len(fixed)
    conv = compute_conv_cross(fixed, var)
    m = conv.max()
    shifted = beta * (conv - m)
    weights = np.exp(shifted)
    weights /= weights.sum()
    # grad[i] = sum_k fixed[k-i] * weights[k] for valid k
    # = cross-correlation of weights and fixed
    # grad[i] = sum_j fixed[j] * weights[j + i] for j = 0..n-1, j+i = 0..2n-2
    # Use FFT: corr = ifft(fft(weights, N) * conj(fft(fixed, N)))
    N = 1
    while N < 3 * n:
        N *= 2
    W = np.fft.rfft(weights, N)
    F = np.fft.rfft(fixed, N)
    corr = np.fft.irfft(W * np.conj(F), N)
    return corr[:n]

def accelerated_mirror_descent(fixed, x0, time_budget, beta_start=20, beta_end=1000):
    """
    Accelerated mirror descent to minimize max_k conv(fixed, x) over the simplex.
    """
    n = len(x0)
    x = x0.copy()
    x_prev = x0.copy()
    x = np.maximum(x, 1e-15)
    x = x / x.sum()
    x_prev = x.copy()
    
    best_x = x.copy()
    best_obj = float(np.max(compute_conv_cross(fixed, x)))
    
    start = time.monotonic()
    t = 1
    eta = 1.0
    
    while time.monotonic() - start < time_budget:
        # Adaptive beta
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Nesterov extrapolation
        gamma = (t - 1) / (t + 2)
        y = x + gamma * (x - x_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Gradient at y
        grad = gradient_inner(fixed, y, beta)
        
        # Mirror descent step
        log_x = np.log(x)
        log_x_new = log_x - eta * grad
        log_x_new -= log_x_new.max()
        x_new = np.exp(log_x_new)
        x_new /= x_new.sum()
        
        # Evaluate
        conv_new = compute_conv_cross(fixed, x_new)
        obj_new = float(np.max(conv_new))
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_x = x_new.copy()
        
        x_prev = x
        x = x_new
        t += 1
        eta = 1.0 / np.sqrt(t)
    
    return best_x

"""
Optimized step function search using fast accelerated mirror descent
for the inner convex subproblem, combined with LP polishing,
coordinate descent refinement, and multi-dimension exploration.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n <= 1:
        return np.array([a[0] * a[0]] if n == 1 else [])
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def fast_gradient_inner(fixed, var, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta * conv(fixed,var))) w.r.t. var.
    Both fixed and var have length n.
    """
    n = len(fixed)
    conv = compute_conv_cross(fixed, var)
    m = conv.max()
    shifted = beta * (conv - m)
    weights = np.exp(shifted)
    weights /= weights.sum()
    # grad[i] = sum_j fixed[j] * weights[j+i] for j=0..n-1, j+i in [0, 2n-2]
    # This is cross-correlation
    N = 1
    while N < 3 * n:
        N *= 2
    W = rfft(weights, N)
    F = rfft(fixed, N)
    corr = irfft(W * np.conjugate(F), N)
    return corr[:n]


def inner_solve(fixed, x_init, time_budget, beta_start=10.0, beta_end=500.0, momentum_init=0.5):
    """
    Minimize max_k conv(fixed, x) over x in simplex using accelerated mirror descent.
    Returns the best x found.
    """
    n = len(x_init)
    x = np.maximum(x_init, 1e-15).astype(np.float64)
    x = x / x.sum()
    
    best_x = x.copy()
    best_val = float(np.max(compute_conv_cross(fixed, x)))
    
    start = time.monotonic()
    v = np.zeros(n)  # velocity in log-space
    momentum = momentum_init
    t = 0
    
    while time.monotonic() - start < time_budget:
        # Adaptive beta
        frac = min(1.0, (time.monotonic() - start) / max(time_budget, 1e-10))
        beta = beta_start * (beta_end / beta_start) ** (frac ** 0.7)
        
        # Compute gradient
        grad = fast_gradient_inner(fixed, x, beta)
        
        # Update velocity (heavy ball in log-space)
        v = momentum * v - grad
        momentum = min(0.9, momentum + 0.001)
        
        # Update x
        log_x = np.log(x)
        log_x += 0.5 * v  # step size
        log_x -= log_x.max()
        x_new = np.exp(log_x)
        s = x_new.sum()
        if s > 1e-15:
            x_new /= s
        else:
            x_new = best_x.copy()
        
        # Evaluate
        val = float(np.max(compute_conv_cross(fixed, x_new)))
        if val < best_val - 1e-12:
            best_val = val
            best_x = x_new.copy()
        
        x = x_new
        t += 1
    
    return best_x


def solve_lp_inner(fixed):
    """
    Solve LP: min M s.t. conv(fixed, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g or None.
    """
    n = len(fixed)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = fixed[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2 * n - 1),
                      A_eq=A_eq, b_eq=[1.0],
                      bounds=[(0, None)] * n + [(None, None)], method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    
    return None

def coordinate_refine(a, time_budget):
    """
    Coordinate descent: move mass from peak contributors to non-contributors.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()

    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    
    for rnd in range(20):
        if time.monotonic() - start > time_budget:
            break
        
        # Find top contributors to the peak
        pairs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                pairs.append((a[i] * a[j], i, j))
        pairs.sort(reverse=True)
        
        # Non-contributor positions
        contrib_set = set()
        for _, i, j in pairs[:20]:
            contrib_set.add(i)
            contrib_set.add(j)
        non_contrib = [i for i in range(n) if i not in contrib_set]
        
        if not non_contrib:
            break
        
        improved = False
        for contrib_val, i, j in pairs[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib_val < 1e-10:
                break
            
            src = i if a[i] >= a[j] else (j if i != j else i)
            for target in non_contrib[:5]:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = frac * a[src]
                    if delta >= a[src] * 0.9:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
        
        if not improved:
            break
    
    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        x = np.arange(n)
        a = np.exp(-((x - n*0.25)**2) / (2*(n*0.1)**2)) + np.exp(-((x - n*0.75)**2) / (2*(n*0.1)**2))
        return a / a.sum()
    elif mode == 3:
        # Triangle
        x = np.linspace(-1, 1, n)
        a = np.maximum(1 - np.abs(x), 0.01)
        return a / a.sum()
    elif mode == 4:
        # Arcsine-like (high at edges)
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x**2)
        return a / a.sum()
    elif mode == 5:
        # Sine-squared
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        return a / a.sum()
    elif mode == 6:
        # Frequency-modulated
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        # Inverse triangle (high in middle, low at edges)
        x = np.linspace(0, 1, n)
        a = 4 * x * (1 - x) + 0.05
        return a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n))
        return a


def resample(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
