def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=10.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def fista_smooth(a, time_budget=120.0, T_start=50.0, T_end=5.0, verbose=False):
    """
    FISTA on the smooth log-sum-exp approximation of the max.
    Temperature anneals from T_start to T_end.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # FISTA parameters
    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()

    # Temperature schedule
    n_iters_est = 500  # estimate for scheduling
    for it in range(n_iters_est):
        if time.monotonic() - start > time_budget:
            break

        # Anneal temperature
        frac = it / max(1, n_iters_est - 1)
        # Logarithmic annealing
        T = T_start * (T_end / T_start) ** frac

        conv = compute_conv(y)
        max_c = float(np.max(conv))
        z = (conv - max_c) / T
        e = np.exp(z)
        sum_e = float(np.sum(e))
        w = e / sum_e

        # Gradient of the smooth approximation
        g = np.zeros(n)
        for k in range(len(conv)):
            if w[k] > 1e-7:
                g += w[k] * grad_conv_at_peak(y, k)
        g *= 2.0 * n

        # FISTA step
        lr = 1.0 / (2.0 * n * max_c + 1e-10)  # rough Lipschitz estimate
        # Actually, let's use a simpler step size
        lr = 0.5 / (n * max_c + 1e-10)

        a_new = y - lr * g
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = y.copy()
        else:
            a_new = a_new / s

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)
        y_new = np.maximum(y_new, 0)
        s_y = float(np.sum(y_new))
        if s_y < 1e-10:
            y_new = a_new.copy()
        else:
            y_new = y_new / s_y

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a_prev = a_new.copy()
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        y = y_new.copy()
        t_k = t_new

        if verbose and it % 200 == 0:
            print(f"    FISTA it{it}: obj={best_obj:.8f}, T={T:.2f}, elapsed={time.monotonic()-start:.1f}s")

def softmax_subgradient(a, time_budget=60.0, K=10, verbose=False):
    """Softmax-weighted subgradient with adaptive temperature and momentum."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Momentum
    v = np.zeros(n)
    beta = 0.9
    lr = 1.0 / np.sqrt(n)
    stall = 0
    patience = 50

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        if max_val < 1e-15:
            break

        # Adaptive temperature
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        spread = (peak_vals[0] - peak_vals[-1]) / max_val
        T = 2.0 / (1.0 + spread * 5.0)

        # Softmax weights over top-K
        w = np.exp((peak_vals - max_val) / T)
        w /= w.sum()

        g = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            g += wi * grad_conv_at_peak(a, int(ki))

        # Momentum update
        v = beta * v + (1 - beta) * g
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            lr = max(1e-6, lr * 0.999)
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                v = np.zeros(n)
                lr = max(1e-6, lr * 0.5)
                stall = 0
            else:
                lr = max(1e-6, lr * 0.98)

        it += 1
        if verbose and it % 300 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, lr={lr:.6f}, el={time.monotonic()-start:.1f}s")

    return best_a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0, 1, n)
    elif mode == 2:
        # Two bumps
        centers = [0.25, 0.75]
        widths = [0.1, 0.1]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a += 0.05
    elif mode == 3:
        # Triangle
        a = 1.0 - 2.0 * np.abs(x - 0.5)
        a = np.maximum(a, 0.01)
    elif mode == 4:
        # Arcsine-like
        a = 1.0 / (np.sqrt(1 - (2*x - 1)**2 + 0.01) + 0.5)
    elif mode == 5:
        # Sinusoidal
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
    elif mode == 6:
        # U-shape
        a = (x - 0.5)**2 + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

    c = np.zeros(n + 1)

    A_ub = np.zeros((2 * n - 1, n + 1))

    LP alternating with momentum and adaptive step size.
    Key difference: maintains a velocity vector to smooth out oscillations.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    beta = 0.3
    best_t = 0.4
    stall = 0
    patience = 12
    rng = np.random.default_rng(54321)
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Direction with momentum
        direction = (1.0 - beta) * (g_lp - a) + beta * v
        v = direction.copy()

        # Adaptive line search
        t_candidates = [best_t, max(0.02, best_t * 0.6), min(1.0, best_t * 1.4),
                        max(0.02, best_t - 0.15), min(1.0, best_t + 0.15), 0.2, 0.5, 0.8]
        t_candidates = sorted(set([round(t, 4) for t in t_candidates if 0.02 <= t <= 1.0]))

        best_t_obj = np.inf
        best_t_val = best_t
        best_local = best_a.copy()

        for t in t_candidates:
            a_try = a + t * direction
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_local.copy()
            a = best_a.copy()
            best_t = best_t_val
            stall = 0
            if verbose and it % 3 == 0:
                print(f"    LP-Mom it{it}: obj={best_obj:.8f}, t={best_t_val:.4f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > patience:
                strategy = stall % 4
                if strategy == 0:
                    a = best_a * np.exp(rng.normal(0, 0.08 + 0.01 * (stall // patience), n))
                elif strategy == 1:
                    shift = rng.integers(1, max(2, n // 15))
                    a = np.concatenate([best_a[-shift:], best_a[:-shift]])
                elif strategy == 2:
                    # Deflation: reduce the peak
                    conv_a = compute_conv(best_a)
                    k_peak = int(np.argmax(conv_a))
                    a = best_a.copy()
                    # Reduce elements contributing to the peak
                    for i in range(n):
                        j = k_peak - i
                        if 0 <= j < n and i <= j:
                            a[i] *= 0.95
                            a[j] *= 0.95
                    s_a = float(np.sum(a))
                    if s_a > 1e-10:
                        a = a / s_a
                else:
                    a = best_a * rng.uniform(0.7, 1.3, n)

                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                v = np.zeros(n)
                best_obj = compute_obj(a)
                stall = 0
                best_t = 0.4
                patience = int(patience * 1.2)

    return best_a


def peak_deflation(a, time_budget=60.0, verbose=False):
    """
    Targeted peak deflation: identify the top convolution peaks and
    redistribute mass away from positions that contribute to them.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_k = np.argsort(conv)[::-1][:5]
        peak_vals = conv[top_k]
        max_val = peak_vals[0]

        # Compute "pressure" at each position: sum of contributions to top peaks
        pressure = np.zeros(n)
        for k in top_k:
            g = grad_conv_at_peak(a, int(k))
            weight = (peak_vals[list(top_k).index(k)] - max_val * 0.5) / (max_val * 0.5 + 1e-10)
            weight = max(0.1, weight)
            pressure += weight * g

        pressure /= (pressure.max() + 1e-10)

        # Reduce high-pressure positions, increase low-pressure
        lr = 0.1
        a_new = a * (1.0 - lr * pressure)
        # Add mass back to low-pressure positions
        low_mask = pressure < np.percentile(pressure, 30)
        deficit = a.sum() - a_new.sum()
        if deficit > 0 and low_mask.any():
            a_new[low_mask] += deficit / low_mask.sum()

        a_new = np.maximum(a_new, 1e-10)
        s_new = float(np.sum(a_new))
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            it += 1
        else:
            # Try smaller step
            for lr_try in [0.05, 0.02, 0.01]:
                a_try = a * (1.0 - lr_try * pressure)
                deficit = a.sum() - a_try.sum()
                if deficit > 0 and low_mask.any():
                    a_try[low_mask] += deficit / low_mask.sum()
                a_try = np.maximum(a_try, 1e-10)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break
            else:
                # No improvement, perturb
                if time.monotonic() - start > 5:
                    rng = np.random.default_rng(it * 31 + n)
                    a = best_a * np.exp(rng.normal(0, 0.1, n))
                    a = np.maximum(a, 1e-10)
                    a = a / a.sum()
                    best_obj = compute_obj(a)
                else:
                    break
            it += 1

        if verbose and it % 50 == 0:
            print(f"    Deflate it{it}: obj={best_obj:.8f}, el={time.monotonic()-start:.1f}s")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
    elif mode == 2:
        centers = rng.uniform(0.2, 0.8, size=3)
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((x - c) ** 2) / (2 * 0.08 ** 2))
        a += 0.05
    elif mode == 3:
        a = 1.0 - np.abs(2.0 * x - 1.0)
        a = np.maximum(a, 0.05)
    elif mode == 4:
        a = 1.0 / (0.3 + 0.7 * np.sqrt(1 - (2*x - 1)**2 + 0.05))
    elif mode == 5:
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n) * 0.8
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
    elif mode == 6:
        a = (x - 0.5)**2 + 0.15
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def optimize_at_n(a, time_budget=200.0, verbose=False):
    """Full optimization pipeline at a fixed n."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    best = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Step 1: FISTA (40% of budget)
    alloc = min(time_budget * 0.4, time_budget - 10)
    if alloc > 15:
        a1 = fista_anneal(a, time_budget=alloc, T_start=60.0, T_end=3.0, verbose=verbose)
        obj1 = compute_obj(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best = a1.copy()

    # Step 2: LP momentum (35% of budget)
    remaining = time_budget - (time.monotonic() - start)
    alloc = min(remaining * 0.7, remaining - 5)
    if alloc > 20:
        a2 = lp_momentum(best, time_budget=alloc, verbose=verbose)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best = a2.copy()

    # Step 3: Peak deflation (20% of budget)
    remaining = time_budget - (time.monotonic() - start)
    alloc = min(remaining * 0.6, remaining - 3)
    if alloc > 10:
        a3 = peak_deflation(best, time_budget=alloc, verbose=verbose)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best = a3.copy()

    # Step 4: Quick FISTA polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a4 = fista_anneal(best, time_budget=remaining * 0.5, T_start=20.0, T_end=2.0)
        obj4 = compute_obj(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best = a4.copy()

    return best


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-n exploration with FISTA + LP momentum + peak deflation.
    Key strategy: try multiple n values, optimize each thoroughly, pick best.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_parent = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default (n=800): obj={best_score:.8f}")

    # === Phase 1: Optimize at parent n (30% budget) ===
    print(f"\n=== Phase 1: Optimize n={len(best_seq)} ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.30, 350.0)
    if p1 > 50:
        a_opt = optimize_at_n(best_seq, time_budget=p1, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 result: obj={obj_opt:.8f}")

    # === Phase 2: Multi-n exploration (40% budget) ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    p2_total = min(remaining * 0.40, 450.0)
    n_cur = len(best_seq)

    # Generate n candidates: focus on diverse range
    n_candidates = []
    for n_try in [150, 200, 250, 300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1400]:
        if 100 <= n_try <= 2000 and n_try != n_cur:
            n_candidates.append(n_try)
    # Sort by closeness to current (try nearby first)
    n_candidates.sort(key=lambda x: abs(x - n_cur))

    n_tried = 0
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40 or n_tried >= 8:
            break

        alloc = p2_total / max(1, len(n_candidates) - n_tried)
        alloc = min(alloc, remaining * 0.3, 120.0)
        if alloc < 30:
            continue

        # Initialize by resampling best sequence
        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        a_opt = optimize_at_n(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 3: Intensive polish at best n (20% budget) ===
    print(f"\n=== Phase 3: Intensive Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p3 = min(remaining * 0.25, 200.0)
    if p3 > 30:
        a_opt = optimize_at_n(best_seq, time_budget=p3, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_opt:.8f}")

    # === Phase 4: Diverse restarts (10% budget) ===
    print(f"\n=== Phase 4: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with best
        blend = 0.4 + 0.1 * r
        a_r = (1.0 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 1e-10)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.15, 80.0)
        a_opt = optimize_at_n(a_r, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Restart {r}: obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_opt:.8f}")

    # === Phase 5: Final LP polish ===
    print(f"\n=== Phase 5: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_lp = lp_momentum(best_seq, time_budget=min(remaining * 0.5, 80.0), verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # === Final FISTA polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_f = fista_anneal(best_seq, time_budget=min(remaining * 0.6, 50.0),
                           T_start=15.0, T_end=1.5, verbose=False)
        obj_f = compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final FISTA: obj={best_score:.8f} **")

    # Validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
