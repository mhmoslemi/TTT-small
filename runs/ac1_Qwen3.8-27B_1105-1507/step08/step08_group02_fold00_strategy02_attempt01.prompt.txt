def evaluate_sequence(sequence: list[float]) -> float:
    """

        if np.isnan(x) or np.isinf(x):
            return np.inf

    sequence = [float(x) for x in sequence]

    sequence = [max(0, x) for x in sequence]

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    s = float(np.sum(a))

def smooth_max_conv(a, beta):
    """Smooth approximation of max(conv(a,a)) using log-sum-exp."""
    conv = compute_conv(a)
    m = conv.max()
    shifted = beta * (conv - m)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def smooth_grad_conv(a, beta):
    """
    Gradient of smooth max(conv) approximation w.r.t. a.
    Uses the identity: grad_i = 2 * sum_m a_m * w_{i+m}
    where w_k are softmax weights over convolution values.
    """

    conv = compute_conv(a)
    m = conv.max()
    shifted = beta * (conv - m)
    weights = np.exp(shifted)
    weights /= weights.sum()

    g = np.zeros(n)

        # sum over m where i+m is in [0, 2n-2]

        g[i] = 2.0 * np.dot(a[m_start:m_end + 1], weights[i + m_start:i + m_end + 1])
    return g


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse_info(a):
    """
    For LP: we need A such that A @ g = conv(a, g).
    A is (2n-1) x n, with A[k, j] = a[k-j] if 0<=k-j<n else 0.
    We build it as a dense matrix for simplicity (n up to ~1500 is fine).
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).

    a_norm = a / np.sum(a)
    A = build_conv_matrix_sparse_info(a_norm)

    # Try cvxpy solvers
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))

                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback to scipy HiGHS
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

            s = float(np.sum(g_val))

    return None, None

        # Mirror descent step (exponential family)

        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try smaller step

            log_a_new2 -= np.max(log_a_new2)
            a_new2 = np.exp(log_a_new2)

                best_a = a_new2.copy()
                a = a_new2.copy()

            print(f"      Smooth it{total_iters}: obj={best_obj:.8f}, beta={beta:.1f}, t={time.monotonic()-start:.1f}s")

def coordinate_descent_refine(a, time_budget=50.0):
    """
    Coordinate descent: try adjusting each coordinate to reduce the max convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    max_rounds = 5
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        # Try moving mass from peak-contributing positions to others
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib = a[i] * a[j]
                contributors.append((contrib, i, j))
        contributors.sort(reverse=True)

        non_contrib = set(range(n))
        for _, i, j in contributors:
            non_contrib.discard(i)
            if i != j:
                non_contrib.discard(j)
        non_contrib_list = list(non_contrib)

        if not non_contrib_list:
            break

        for contrib, i, j in contributors[:30]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-8:
                break
            for target in non_contrib_list[:8]:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15]:
                    delta = frac * min(a[i], a[j] if i != j else a[i])
                    if delta < 0 or delta >= a[i]:
                        continue
                    # Move mass from the larger contributor
                    if i != j:
                        src = i if a[i] >= a[j] else j
                    else:
                        src = i
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        if not improved:
            break

    return best_a


def mirror_descent_topk(a, time_budget=100.0, topK=3, verbose=False):
    """
    Mirror descent with top-K peak subgradients.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    momentum = 0.0
    v = np.zeros(n)
    stall = 0
    patience = 150

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:topK]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        weights = np.exp((peak_vals - max_val) * 2.0 / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        log_a = np.log(a)
        log_a_new = log_a - lr * v
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            momentum = min(0.85, momentum + 0.002)
            lr = max(1e-5, lr * 0.998)
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, min(0.3, 0.05 + 0.02 * stall // patience), n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = 0.0
                lr = max(1e-5, lr * 0.5)
                stall = 0
            else:
                lr = max(1e-5, lr * 0.97)

        it += 1
        if verbose and it % 100 == 0:
            print(f"      MD it{it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x**3 + (1-x)**3 + 0.05
        a = a / a.sum()
    elif mode == 7:
        # Chirp-like
        x = np.linspace(0, 1, n)
        freq = 3 + 7 * x
        a = np.abs(np.sin(2 * np.pi * freq * x**2)) + 0.01
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
