"""
Optimized step function height sequence minimizer.
Uses LP alternating with QP polishing, Nesterov subgradient, 
and multi-n exploration to minimize 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp_min_max(a):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Use CVXPY with Clarabel (fast for LPs)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    best_g = None
    best_M = np.inf

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            elapsed = time.monotonic() - t0
            if elapsed > 30:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10 and float(M.value) < best_M:
                    best_g = g_val / gs
                    best_M = float(M.value)
        except Exception:
            continue

    if best_g is not None:
        return best_g, best_M

    # Fallback to scipy
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def solve_qp_for_active(a, k_active, max_iters=50):
    """Solve QP: min (g*g)[k] for k in k_active, s.t. sum(g)=1, g>=0.
    Uses projected gradient for the QP."""
    n = len(a)
    g = np.ones(n) / n  # Start from uniform

    # For each active k, compute the quadratic form g^T M_k g
    # M_k has (M_k)[i,j] = 1 if i+j=k
    # (g*g)[k] = sum_i g[i]*g[k-i] (valid indices)
    
    best_g = g.copy()
    best_val = np.inf
    lr = 0.1 / n
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > 5:
            break

        # Compute gradient of weighted average of active QPs
        grad = np.zeros(n)
        val = 0.0
        for k in k_active:
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    grad[j] += 2.0 * g[kj]
                    val += g[j] * g[kj]
            # Note: when j = k-j, the term is g[j]^2, counted once in val
            # but gradient gets 2*g[j] which is correct
        grad /= len(k_active)
        val /= len(k_active)

        # Projected gradient step on simplex
        g_new = g - lr * grad
        g_new = np.maximum(g_new, 0)
        s = float(np.sum(g_new))
        if s < 1e-10:
            g_new = np.ones(n) / n
        else:
            g_new = g_new / s

        # Line search
        best_local = val
        best_local_g = g_new.copy()
        for alpha in [0.5, 0.25, 0.1, 0.05]:
            g_try = (1 - alpha) * g + alpha * g_new
            g_try = np.maximum(g_try, 0)
            s_try = float(np.sum(g_try))
            if s_try < 1e-10:
                continue
            g_try = g_try / s_try
            v_try = 0.0
            for k in k_active:
                for j in range(n):
                    kj = k - j
                    if 0 <= kj < n:
                        v_try += g_try[j] * g_try[kj]
            v_try /= len(k_active)
            if v_try < best_local:
                best_local = v_try
                best_local_g = g_try.copy()

        if best_local < best_val - 1e-15:
            best_val = best_local
            best_g = best_local_g.copy()
        g = best_local_g

        if abs(val - best_val) < 1e-12:
            break

    return best_g, best_val
