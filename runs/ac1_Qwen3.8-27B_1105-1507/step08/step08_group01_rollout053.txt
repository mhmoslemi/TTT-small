def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    n = len(a)
    m = len(b)

    while size < n + m - 1:
        size <<= 1

    fb = rfft(b, size)

    n = len(v)
    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous approach:
- Correct spectral gradient: grad = 2 * IFFT(conj(FA)*FW)[:n]
- Adaptive active-set LP with dual-guided constraint selection

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_rows(a, indices):
    """Build rows of convolution matrix for specific indices.
    Row k of the matrix A satisfies (A @ g)[k] = (a * g)[k].
    A[k, j] = a[k - j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    rows = len(indices)
    A = np.zeros((rows, n))
    for idx in range(rows):
        k = indices[idx]
        # A[k, j] = a[k-j] if 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[idx, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_with_active(a, active_indices):
    """Solve LP: min M s.t. conv(a,g)_k <= M for k in active, sum(g)=1, g>=0."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    
    A = build_conv_matrix_rows(a, active_indices)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_cons = len(active_indices)
    A_ub = np.zeros((n_cons, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_cons)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_norm = g_val / s
            # Verify against ALL constraints
            all_indices = np.arange(2 * n - 1)
            A_full = build_conv_matrix_rows(a, all_indices)
            true_M = float(np.max(A_full @ g_norm))
            return g_norm, true_M
    return None, None


def solve_lp_full(a):
    """Solve LP with all constraints."""
    n = len(a)
    all_indices = np.arange(2 * n - 1)
    return solve_lp_with_active(a, all_indices)


def convex_line_search(a, g, n_evals=12):
    """Find optimal t for blend (1-t)*a + t*g using convexity of the objective."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)
    
    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)
    
    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))
    
    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    return t_opt, a_new / s


def get_active_set(a, top_fraction=0.15, min_count=20, max_count=200):
    """Get active set indices: top peaks of convolution."""
    conv = compute_conv(a)
    n = len(a)
    # Use symmetry: only top n peaks matter (rest are mirror)
    half_len = n
    top_k = min(max_count, max(min_count, int(half_len * top_fraction)))
    top_indices_full = np.argsort(conv)[::-1][:top_k]
    return top_indices_full

def compute_gradient_smooth(a, weights):
    """Compute gradient of weighted convolution sum using FFT.
    grad[j] = 2 * sum_k weights[k] * a[k-j] (where 0<=k-j<n)
    = 2 * cross_correlation(a, weights)[j]
    """
    n = len(a)
    size_fn = 1
    while size_fn < 2 * n:
        size_fn <<= 1
    fa = rfft(a, size_fn)
    fw = rfft(weights, size_fn)
    # cross[j] = sum_i a[i] * w[i+j] = IFFT(conj(FA) * FW)[j]
    cross = irfft(np.conj(fa) * fw, size_fn)
    return 2.0 * cross[:n].copy()

def block_coordinate_descent(a, time_budget=80.0, block_size=4, verbose=False):

    pass_count = 0
    step_scale = 0.01

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    if n >= 4:
        from scipy.interpolate import CubicSpline
        cs = CubicSpline(x_old, a, bc_type='natural')
        a_new = cs(x_new)
    else:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)

    except Exception:
        pass
    
    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init

    elapsed = time.monotonic() - start_time
    remaining = deadline - time.monotonic()
    phase1_budget = min(300.0, remaining * 0.3)
    if phase1_budget > 30:
        a_opt = lp_alternating_v2(a, time_budget=phase1_budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  Phase 1: obj={best_score:.8f}")
        else:
            print(f"  Phase 1: obj={obj_opt:.8f} (no improvement)")
    
    # === Phase 2: Smooth Max Gradient Descent ===
    print(f"\n=== Phase 2: Smooth Max GD (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase2_budget = min(150.0, remaining * 0.2)
        a_sm = smooth_max_gd_v2(a, time_budget=phase2_budget, T_init=0.002, T_min=0.000005, verbose=True)

            a = a_sm.copy()

    # === Phase 3: LP Alternating Round 2 ===
    print(f"\n=== Phase 3: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase3_budget = min(200.0, remaining * 0.25)
        a_opt = lp_alternating_v2(a, time_budget=phase3_budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:

            best_seq = a_opt.copy()

    # === Phase 4: Peak Equalization ===

    remaining = deadline - time.monotonic()

    # === Phase 5: Smooth Max GD Round 2 (lower T) ===
    print(f"\n=== Phase 5: Smooth Max GD Fine ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(500, n_best - 100), max(500, n_best - 50), max(500, n_best - 25),
        n_best + 25, n_best + 50, n_best + 100, n_best + 150, n_best + 200,
        600, 650, 700, 750, 800, 850, 900, 950, 1000, 1050, 1100, 1200

    n_options = [n_o for n_o in n_options if 500 <= n_o <= 1500]
    
    for n_try in n_options:

            if obj_try2 < obj_try:
                a_opt = a_opt2
                obj_try = obj_try2
        
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()

    if best_seq is None:

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Verify with evaluate_sequence
    try:
        val = evaluate_sequence(result)

        pass
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
