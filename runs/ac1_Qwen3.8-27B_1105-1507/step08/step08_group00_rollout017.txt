"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. LP-driven steepest descent: each step uses the LP solution as exact first-order direction
   with optimal line search along that direction.
2. Peak-shaving heuristic: directly reduces contributions to top autocorrelation peaks.
3. Multi-n exploration with adaptive resampling.
4. Multiplicative weight updates for better exploitation of the non-negativity constraint.
5. Better time management with early stopping and reallocation.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None) on failure.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of Q_k(a) using FFT.
    Returns (gradient, convolution_array).
    """
    n = len(a)
    conv = compute_conv(a)

    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    return g, conv


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.9, 1e-6))

        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 8.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-9, lr * 0.9997)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def lp_driven_descent(a, time_budget=200.0, verbose=False):
    """
    LP-driven steepest descent: each step solves LP for optimal first-order direction,
    then performs line search along that direction.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    blend_base = 0.3

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Line search: find best t in [(1-t)*a + t*g_lp]
        best_t = 0.0
        best_t_obj = best_obj
        best_t_a = best_a.copy()

        # Try a range of t values, concentrating around blend_base
        t_candidates = np.unique(np.concatenate([
            np.linspace(0.05, 1.0, 20),
            np.linspace(max(0.01, blend_base * 0.5), min(0.99, blend_base * 1.5), 10)
        ]))

        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_t_a = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a
            blend_base = best_t
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP-DD it{it}: obj={best_obj:.8f}, t={best_t:.3f}, M={M_lp:.6f}")
        else:
            stall += 1
            if stall > 10:
                # Perturb and reduce blend
                rng = np.random.default_rng(it * 7919)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                blend_base = max(0.1, blend_base * 0.5)
                stall = 0
                if blend_base < 0.05:
                    break
            else:
                blend_base = max(0.05, blend_base * 0.8)

        # Every 15 iterations, run a few steps of smoothed GD to escape
        if it % 15 == 0:

                obj_gd = compute_obj(a_gd)
                if obj_gd < best_obj:
                    best_obj = obj_gd
                    best_a = a_gd.copy()
                    a = a_gd

    return best_a


def peak_shaving(a, time_budget=50.0, verbose=False):
    """
    Direct peak-shaving: identify top autocorrelation peaks and reduce
    the contributions to those peaks by adjusting specific a_i values.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    rng = np.random.default_rng(42)
    lr = 0.05

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        top_k = 3
        peak_indices = np.argsort(conv)[::-1][:top_k]

        # For each peak, find which a_i contribute most and reduce them
        gradient_pressure = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            # conv[k] = sum_i a[i]*a[k-i]
            # d(conv[k])/d(a[j]) = a[k-j] (if valid) + a[j] (if k-2j+... )
            # Actually: d/d(a[j]) sum_i a[i]*a[k-i] = a[k-j] + a[j] (with bounds)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            for j in range(i_min, i_max + 1):
                other = k - j
                if 0 <= other < n:
                    # Contribution of a[j] to conv[k] is a[other]
                    # And a[other]'s contribution includes a[j]
                    gradient_pressure[j] += max_val * max(0, a[other])
                    if other != j:
                        gradient_pressure[other] += max_val * max(0, a[j])

        # Apply multiplicative update: reduce high-pressure elements
        a_new = a * np.exp(-lr * gradient_pressure / (max_val + 1e-10))
        # Add small random perturbation
        a_new *= np.exp(rng.normal(0, 0.01, n))
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.3)
        else:
            lr *= 0.9
            if lr < 0.005:
                break

    return best_a


def multiplicative_weight_descent(a, time_budget=100.0, verbose=False):
    """
    Multiplicative weight (exponential gradient) method on the smoothed objective.
    Naturally handles non-negativity. Often better than additive GD for this problem.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    tau = 20.0
    it = 0
    best_it = 0
    momentum = np.zeros(n)
    mom_coeff = 0.7

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.9, 1e-6))
        tau = 10.0 * (400.0 / 10.0) ** (progress ** 0.8)

        g, conv = smoothed_gradient_fft(a, tau)

        # Momentum on gradient
        momentum = mom_coeff * momentum + (1 - mom_coeff) * g
        g_avg = momentum

        # Multiplicative update: a[i] *= exp(-lr * g_avg[i])
        # Center the gradient to preserve sum
        g_centered = g_avg - float(np.mean(g_avg))
        a_new = a * np.exp(-lr * g_centered)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            a = a_new

        it += 1
        lr = max(0.1 / np.sqrt(n), lr * 0.999)

        if it - best_it > 20000:
            a = best_a.copy()
            momentum = np.zeros(n)
            lr = 1.0 / np.sqrt(n)
            best_it = it
            if verbose:
                print(f"    MWD restart at it{it}, obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussians
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Arcsine (U-shaped)
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        # Multi-frequency
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f in [2, 3, 5, 7]:
            a += 0.2 * np.sin(2 * np.pi * f * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1, 1)
        a = a / a.sum()
    elif mode == 6:
        # Cosine bell
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        # Three Gaussians
        centers = [n // 6, n // 2, 5 * n // 6]
        w = max(1, n // 15)
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Smooth step (S-curve)
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
        a = a / a.sum()
    elif mode == 9:
        # Random smooth (low-pass filtered noise)
        a = rng.uniform(0, 1, n)
        # Low-pass filter
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

    print(f"\n=== Phase 3: Multiplicative Weight Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 200.0)
    if p3_budget > 20:
        a_mwd = multiplicative_weight_descent(best_seq, time_budget=p3_budget, verbose=True)
        obj_mwd = compute_obj(a_mwd)
        if obj_mwd < best_score:
            best_score = obj_mwd
            best_seq = a_mwd.copy()
            print(f"  Phase3 MWD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 MWD: obj={obj_mwd:.8f}")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 100.0)
    if p4_budget > 15:
        a_ps = peak_shaving(best_seq, time_budget=p4_budget, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Phase4 PS: obj={best_score:.8f} **")
        else:
            print(f"  Phase4 PS: obj={obj_ps:.8f}")

    # === Phase 5: Intensive LP polishing ===
    print(f"\n=== Phase 5: Intensive LP Polishing ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.3, 400.0)
    if p5_budget > 30:
        for rnd in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p5_budget / (6 - rnd), remaining * 0.6)
            if alloc < 20:
                break
            a_pol = polish_with_lp_iterations(best_seq, time_budget=alloc, verbose=True)
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_pol:.8f}")
                if rnd < 5:
                    scale = 0.02 * (rnd + 1)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()

    # === Phase 6: Diverse restarts with strong polishing ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = (r * 3) % 10
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # MWD then LP polish
        alloc1 = min(remaining * 0.15, 60.0)
        if alloc1 > 10:
            a_r = multiplicative_weight_descent(a_r, time_budget=alloc1)

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.12, 50.0)
        if alloc2 > 10:
            a_r = polish_with_lp_iterations(a_r, time_budget=alloc2)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau smoothed GD
    if remaining > 30:
        alloc = min(remaining * 0.25, 100.0)
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                          tau_init=50.0, tau_end=1000.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final SGD: obj={best_score:.8f} **")

    # LP polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 120.0)
        a_lp = polish_with_lp_iterations(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # MWD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 60.0)
        a_mwd = multiplicative_weight_descent(best_seq, time_budget=alloc)
        obj_mwd = compute_obj(a_mwd)
        if obj_mwd < best_score:
            best_score = obj_mwd
            best_seq = a_mwd.copy()
            print(f"  Final MWD: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 80.0)
        a_lp = polish_with_lp_iterations(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
