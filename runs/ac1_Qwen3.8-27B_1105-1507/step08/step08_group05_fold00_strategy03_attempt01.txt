        a = np.ones(n) / n

    # Compute full self-convolution
    conv = compute_conv(a)
    max_conv = np.max(conv)

    # The objective is 2*n*max(conv_val) / (sum_val * sum_val)
    return float(2 * n * max_conv) / (sum_val * sum_val)

def crossover(parent1, parent2, point):
    # Perform crossover at the given point
    child = np.concatenate([parent1[:point+1], parent2[point+1:]])
    return child

def mutate(child, mutation_rate):
    # Apply mutation to the child sequence
    # For each index, with probability mutation_rate, change the element
    # to a random value in [0, 1000]
    # We don't mutate if the parent is the best ever and has been stable for many generations
    L = len(child)
    mask = np.random.rand(L) < mutation_rate * 0.01  # very small mutation per element to be efficient
    # But we want to apply mutation to the whole chromosome, so we'll use a lower mutation rate for the whole generation,
    # but then we have to do point mutations per element.
    # Alternatively, we can do:
    #   For each element, if random() < mutation_rate, then set to random.uniform(0, 1000)
    mutated_child = child.copy()
    # Chance to mutate a value: mutation_rate_per_element is very small
    for i, x in enumerate(mutated_child):
        if np.random.random() < mutation_rate * (0.01 if True else None):
            # We require the mutation to be with positive probability, so set:
            mutated_child[i] = np.random.uniform(0.0001, 0.01, 1)[0]  # Why this? Because we want small changes sometimes.
    # Actually, we should mutate to any value in [0,1000] so:
    #   new_value = random.uniform(0, 1000)
    # So let's do:
    for i in range(L):
        if np.random.random() < mutation_rate * 0.1:
            mutated_child[i] = np.random.uniform(0, 1000)
    return mutated_child

def tournament_selection(candidates, scores):
    # Pick three random indices (without replacement) and return the best candidate
    if len(candidates) == 0:
        return None, None
    indices = np.random.choice(len(candidates), size=3, replace=False)
    return candidates[indices], scores[indices]

            candidate = candidate / s * 1000  # This isn't necessary because the evaluator will do it, but we want to keep magnitudes.

            else:
                # Add the candidate without duplicates

                    population.append(candidate)
        # Then, we have to generate more until we have 5? Or keep each unique candidate.
    else:

        best_global = current_best_sequence

                best_global = child1

            new_pop.append(child1)

        combined_population = population + new_pop
        # And keep only the best from the combined
        combined_scores = population_scores + new_pop_scores
        best_index = np.argmin(combined_scores, 0)

        if combined_scores[best_index] < best_score:
            best_score = combined_scores[best_index]
            best_global = combined_population[best_index]

        # Update population
        population = combined_population
        population_scores = combined_scores

        if combined_population[best_index] == population[best_index]:
            stagnation_count += 1
        else:
            stagnation_count = 0

        # Check for stagnation: number of best individuals found so far is stagnation_count, but note we always keep the best global, which might be from a previous winner.
        # Actually, we are incrementing the evaluation count by the number of individuals, not generations.

        # Add to evaluation count
        evaluation_count += len(new_pop)

        # Safety: break if too many evaluations
        if evaluation_count > 3000000:  # 3e6 evaluations is 1500 seconds at 0.001s per convolve
            break

    # Return the best candidate (best_global)

    return best_global
