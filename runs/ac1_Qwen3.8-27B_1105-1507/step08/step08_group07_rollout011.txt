def compute_conv(a):

    n = len(a)
    size = 1

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""

        return np.inf
    a_norm = a / s

    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    active = conv >= threshold_frac * max_val
    active_k = np.where(active)[0]

    if len(active_k) == 0:
        active_k = np.array([int(np.argmax(conv))])

    grad = np.zeros(n)
    for k in active_k:
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        idx = k - np.arange(m_lo, m_hi + 1)
        grad[m_lo:m_hi + 1] += 2.0 * a[idx]
    grad /= len(active_k)
    return grad, max_val


def mirror_descent_step(a, grad, eta):
    """One mirror descent (multiplicative weights) step."""
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def frank_wolfe_step(a, grad, t):
    """One Frank-Wolfe (conditional gradient) step."""
    j = int(np.argmin(grad))
    v = np.zeros(len(a))
    v[j] = 1.0
    a_new = (1 - t) * a + t * v
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def combined_optimize(a, time_budget=200.0, verbose=False):
    """
    Main optimization loop: LP alternating + mirror descent + Frank-Wolfe.
    Uses adaptive strategy to balance large LP steps with small gradient steps.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    no_improve_count = 0
    eta = 0.08
    prev_a = None
    prev_prev_a = None
    rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
    lp_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 2.0:
            break

        # Solve LP (the expensive step)
        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp, dual_y = result
        lp_count += 1

        # 2-cycle detection
        if prev_prev_a is not None:
            if np.max(np.abs(a - prev_prev_a)) < 3e-7:
                # Escape via mirror descent burst
                a_esc = best_a.copy()
                for _ in range(12):
                    grad, _ = batch_subgradient(a_esc)
                    a_esc = mirror_descent_step(a_esc, grad, 0.04)
                obj_esc = compute_obj(a_esc)
                if obj_esc < best_obj:
                    best_obj = obj_esc
                    best_a = a_esc.copy()
                a = 0.5 * best_a + 0.5 * a_esc
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall_count = 0
                no_improve_count = 0
                prev_a = None
                prev_prev_a = None
                eta = 0.08
                continue

        # Try LP mixing with multiple ratios
        improved = False
        best_try_obj = best_obj
        best_try_a = None

        mix_ratios = [0.5, 0.7, 0.9, 1.0, 0.3, 0.6]
        for mix in mix_ratios:
            a_try = (1.0 - mix) * a + mix * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_try_obj:
                best_try_obj = obj_try
                best_try_a = a_try.copy()

        if best_try_a is not None and best_try_obj < best_obj - 1e-14:
            best_obj = best_try_obj
            best_a = best_try_a.copy()
            a = best_try_a
            if prev_a is not None:
                prev_prev_a = prev_a
            prev_a = a.copy()
            stall_count = 0
            no_improve_count = 0
            # Slightly increase eta (more confident)
            eta = min(eta * 1.005, 0.25)
            if verbose and lp_count % 5 == 0:
                print(f"    [LP] obj={best_obj:.8f} t={elapsed:.0f}s lp#{lp_count}")
        else:
            no_improve_count += 1
            stall_count += 1

            if no_improve_count >= 3:
                # Use gradient-based steps
                grad, _ = batch_subgradient(a)

                # Try mirror descent
                a_try = mirror_descent_step(a, grad, eta)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    stall_count = 0
                    no_improve_count = 0
                else:
                    # Try Frank-Wolfe with adaptive step
                    t_fw = max(0.05, 2.0 / (no_improve_count + 2))
                    a_try = frank_wolfe_step(a, grad, t_fw)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stall_count = 0
                        no_improve_count = 0
                    else:
                        # Decay eta
                        eta = max(eta * 0.97, 0.01)

                        # If stuck for a while, do a larger escape
                        if stall_count > 15:
                            # Structured perturbation using dual info
                            if dual_y is not None and np.max(dual_y) > 0:
                                A_mat = build_conv_matrix(a)
                                active_idx = np.where(dual_y > 0.05 * np.max(dual_y))[0]
                                if len(active_idx) > 0:
                                    col_imp = np.sum(A_mat[active_idx, :], axis=0)
                                    pert = -np.sign(col_imp - np.mean(col_imp)) * 0.008
                                    pert += rng.normal(0, 0.008, n)
                                else:
                                    pert = rng.normal(0, 0.02, n)
                            else:
                                pert = rng.normal(0, 0.025, n)

                            a = best_a * np.exp(pert)
                            a = np.maximum(a, 0)
                            a = a / np.sum(a)
                            stall_count = 0
                            no_improve_count = 0
                            prev_a = None
                            prev_prev_a = None
                            eta = 0.08
                            if verbose:
                                print(f"    [Escaped] obj={compute_obj(a):.8f} t={elapsed:.0f}s")
                        elif stall_count > 8:
                            # Moderate perturbation
                            pert = rng.normal(0, 0.015, n)
                            a = best_a * np.exp(pert)
                            a = np.maximum(a, 0)
                            a = a / np.sum(a)
                            no_improve_count = 0
                            prev_a = None
                            prev_prev_a = None

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fine_polish(a, time_budget=30.0):
    """Fine-grained pairwise coordinate exchange polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        for frac in [0.003, 0.007, 0.015, 0.002]:
            delta = frac * min(a[i], a[j])
            if delta <= 0:
                continue
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            if a_try[i] < 0 or a_try[j] < 0:
                continue
            a_try = a_try / np.sum(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-15:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break

    return best_a


def diverse_initializations(rng, n):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n)

    # Raised cosine
    w1 = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(np.maximum(w1, 0.01))

    # Sine squared
    w2 = np.sin(np.pi * t) ** 2
    inits.append(np.maximum(w2, 0.01))

    # Flat with slight modulation
    w3 = 1.0 - 0.2 * np.cos(4 * np.pi * t)
    inits.append(np.maximum(w3, 0.01))

    # Triangular
    w4 = 1.0 - np.abs(2 * t - 1)
    inits.append(np.maximum(w4, 0.01))

    # Gaussian-like
    w5 = np.exp(-4 * (t - 0.5) ** 2)
    inits.append(np.maximum(w5, 0.01))

    # Random smooth
    w6 = 1.0 + 0.3 * rng.normal(0, 1, n)
    inits.append(np.maximum(w6, 0.01))

    for w in inits:
        s = float(np.sum(w))
        if s > 0.01:
            inits[-1] = w / s  # fix last one, should fix all
    # Re-normalize all properly
    result = []
    for w in inits:
        s = float(np.sum(w))
        if s > 0.01:
            result.append(w / s)
    return result


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating, mirror descent, Frank-Wolfe,
    multi-n exploration, diverse restarts, and fine polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        # Try structured initializations
        for n_init in [600, 800, 1000, 1200]:
            t = np.linspace(0, 1, n_init)
            w = 0.5 + 0.5 * np.cos(2 * np.pi * t)
            a_init = np.maximum(w, 0.01)
            a_init = a_init / np.sum(a_init)
            obj_init = compute_obj(a_init)
            if obj_init < best_score:
                best_score = obj_init
                best_seq = a_init.copy()

        if best_seq is None:
            best_seq = np.ones(800) / 800.0
            best_score = compute_obj(best_seq)
        print(f"Initial (n={len(best_seq)}): obj={best_score:.8f}")

    n_parent = len(best_seq)
    a = best_seq.copy()

    # === Phase 1: Primary LP+MD+FW optimization (45% of time) ===
    print(f"\n=== Phase 1: Primary Optimize (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    alloc1 = remaining * 0.45
    print(f"  Time allocated: {alloc1:.0f}s")
    a_new = combined_optimize(a, time_budget=alloc1, verbose=True)
    obj_new = compute_obj(a_new)
    if obj_new < best_score:
        best_score = obj_new
        best_seq = a_new.copy()
        a = a_new.copy()
        print(f"  Phase 1 done: obj={best_score:.8f} **")
    else:
        print(f"  Phase 1 done: obj={obj_new:.8f}")

    # === Phase 2: Multi-n exploration (25% of time) ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    alloc2 = remaining * 0.25

    # Generate candidate n values around the current best
    n_candidates = sorted(set([
        n_parent - 100, n_parent - 50, n_parent - 20,
        n_parent + 20, n_parent + 50, n_parent + 100,
        n_parent + 200, n_parent + 400,
        500, 600, 700, 900, 1000, 1100, 1200, 1400, 1500,
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2500]
    # Sort by proximity to n_parent
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    time_per_n = alloc2 / len(n_candidates)
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(time_per_n, remaining * 0.3)
        if alloc < 10:
            alloc = 10
        a_opt = combined_optimize(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Diverse restarts (15% of time) ===
    print(f"\n=== Phase 3: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    alloc3 = remaining * 0.15
    n_best = len(best_seq)
    max_restarts = max(3, int(alloc3 // 30))

    for r in range(max_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if r % 4 == 0:
            # Gaussian perturbation
            sigma = 0.04 + 0.015 * r
            a_r = best_seq * np.exp(rng.normal(0, sigma, n_best))
        elif r % 4 == 1:
            # Block perturbation
            a_r = best_seq.copy()
            n_blocks = max(5, n_best // 80)
            block_size = max(1, n_best // n_blocks)
            n_perturb = max(2, n_blocks // 4)
            blocks = rng.choice(n_blocks, min(n_perturb, n_blocks), replace=False)
            for b in blocks:
                si = int(b) * block_size
                ei = min(si + block_size, n_best)
                a_r[si:ei] *= rng.uniform(0.5, 1.6)
        elif r % 4 == 2:
            # Sparse large perturbation
            a_r = best_seq.copy()
            n_sparse = max(10, n_best // 20)
            idx = rng.choice(n_best, n_sparse, replace=False)
            a_r[idx] *= rng.uniform(0.2, 2.5, n_sparse)
        else:
            # Mix with diverse initialization
            inits = diverse_initializations(rng, n_best)
            a_r = 0.5 * best_seq + 0.5 * inits[r % len(inits)]

        a_r = np.maximum(a_r, 0)
        sr = float(np.sum(a_r))
        if sr > 1e-10:
            a_r = a_r / sr

        alloc = min(remaining * 0.35, 35.0)
        a_opt = combined_optimize(a_r, time_budget=alloc)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (n={n_best}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 4: Final focused optimization on best (remaining time) ===
    print(f"\n=== Phase 4: Final Focused Optimize ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc4 = min(remaining * 0.6, 120.0)
        print(f"  Time allocated: {alloc4:.0f}s, n={len(best_seq)}")
        a_final = combined_optimize(best_seq, time_budget=alloc4, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final optimize: obj={best_score:.8f} **")
        else:
            print(f"  Final optimize: obj={obj_final:.8f}")

        # Fine polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc_polish = min(remaining * 0.5, 40.0)
            a_polished = fine_polish(best_seq, time_budget=alloc_polish)
            obj_polished = compute_obj(a_polished)
            if obj_polished < best_score:
                best_score = obj_polished
                best_seq = a_polished.copy()
                print(f"  Polish: obj={best_score:.8f} **")
            else:
                print(f"  Polish: obj={obj_polished:.8f}")

    # Final short LP run
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = combined_optimize(best_seq, time_budget=min(remaining * 0.7, 20.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
