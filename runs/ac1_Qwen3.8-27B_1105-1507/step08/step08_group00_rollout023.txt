"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive LP polishing with dual-informed gradient steps, adaptive line search,

Key innovations: LP dual feedback loop, parabolic line search, and spectral peak reduction.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M, dual_vars) where dual_vars are the dual variables for the peak constraints.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    peak_constraints = [A @ g <= M]
    constraints = peak_constraints + [cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    g_val = None
    M_val = None
    dual_vals = None

    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    # Try to get dual variables
                    try:
                        dual_vals = np.array(peak_constraints[0].dual_value, dtype=np.float64)
                        if dual_vals.shape == ():
                            dual_vals = np.array([dual_vals.item()])
                    except:
                        dual_vals = None
                if g_val is not None:
                    break
        except Exception:
            continue

    if g_val is None:
        return None, None, None
    return g_val, M_val, dual_vals


def parabolic_line_search(a, g, a0_obj, max_iter=20):
    """
    Parabolic/golden section line search along direction (a -> g).
    Finds t in [0, 1] minimizing obj((1-t)*a + t*g).
    """
    best_t = 0.0
    best_obj = a0_obj

    # First, evaluate at a few points to bracket
    t_points = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
    t_objs = [a0_obj]
    for t in t_points:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        t_objs.append(obj_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t

    # Find the best t from the grid
    if best_t == 0.0:
        return 0.0, a0_obj

    # Refine around best_t using golden section
    t_lo = max(0.0, best_t - 0.1)
    t_hi = min(1.0, best_t + 0.1)
    
    # Golden section
    gr = (np.sqrt(5) - 1) / 2
    a1 = t_hi - gr * (t_hi - t_lo)
    b1 = t_lo + gr * (t_hi - t_lo)
    
    def eval_at(t):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        return compute_obj(a_try)
    
    f_a1 = eval_at(a1)
    f_b1 = eval_at(b1)
    
    for _ in range(max_iter):
        if f_a1 < f_b1:
            t_hi = b1
            b1 = a1
            f_b1 = f_a1
            a1 = t_hi - gr * (t_hi - t_lo)
            f_a1 = eval_at(a1)
        else:
            t_lo = a1
            a1 = b1
            f_a1 = f_b1
            b1 = t_lo + gr * (t_hi - t_lo)
            f_b1 = eval_at(b1)
        
        if abs(t_hi - t_lo) < 0.005:
            break
        
        if f_a1 < best_obj:
            best_obj = f_a1
            best_t = a1
        if f_b1 < best_obj:
            best_obj = f_b1
            best_t = b1

    return best_t, best_obj

def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of Q_k(a) using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=10.0, tau_end=500.0, verbose=False):

def peak_shave_coordinate_descent(a, time_budget=100.0, K=5, verbose=False):
    """
    Coordinate descent that focuses on reducing the top-K autocorrelation peaks.
    For each coordinate, compute the exact change in max peak and update.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute: for each coordinate i and peak k, the coefficient is a[k-i] (if valid)
    # When we change a[i] by delta (and compensate elsewhere), the change in peak k is:
    # d(peak_k)/da_i = 2 * a[k-i] (for the cross term)
    
    # Strategy: identify top K peaks, then for coordinates that contribute strongly
    # to those peaks, reduce them and redistribute to coordinates that contribute
    # less to the peaks.
    
    rng = np.random.default_rng(42)
    lr = 0.05
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        max_val = conv[top_idx[0]]
        
        if max_val < 1e-15:
            break
        
        # Compute "peak sensitivity" for each coordinate
        # sensitivity[i] = how much a[i] contributes to the top peaks
        sensitivity = np.zeros(n)
        for k in top_idx:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    sensitivity[i] += a[j] / (max_val + 1e-15)
        
        # We want to reduce high-sensitivity coordinates and increase low-sensitivity ones
        # Use a "flow" approach: move mass from high-sensitivity to low-sensitivity coords
        sens_rank = np.argsort(sensitivity)
        n_low = n // 4
        n_high = n // 4
        
        low_coords = sens_rank[:n_low]
        high_coords = sens_rank[-n_high:]
        
        # Move a small amount from high to low
        amount = lr * np.mean(a[high_coords]) * n_high / max(1, n_low)
        
        for i in high_coords:
            transfer = min(a[i] * 0.1, amount)
            a[i] -= transfer
            # Distribute to low coordinates proportionally
            for j in low_coords:
                a[j] += transfer * a[j] / max(float(np.sum(a[low_coords])), 1e-15)
        
        a = np.maximum(a, 0)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
        
        obj_new = compute_obj(a)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a.copy()
        else:
            a = best_a.copy()
            lr *= 0.95
            if lr < 0.001:
                break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        c1 = rng.integers(0, n // 3)
        c2 = rng.integers(2 * n // 3, n)
        w = max(1, n // 12)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Sine-squared profile
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
        a = a / a.sum()
    elif mode == 9:
        # Truncated Gaussian
        x = np.linspace(-3, 3, n)
        a = np.exp(-x**2)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

def lp_active_set(a, max_constraints=50, max_iter=20):
    """
    Active set LP: iteratively add constraints until the solution is feasible.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Initial active set: top peaks of (a*a)
    conv = compute_conv(a_norm)
    active_set = list(np.argsort(conv)[::-1][:10])
    
    g = np.ones(n) / n  # Initial guess
    M = np.max(conv)
    
    for iteration in range(max_iter):
        # Build LP with only active constraints
        A_sub = np.zeros((len(active_set), n))
        for idx, k in enumerate(active_set):
            for j in range(n):
                r = k - j
                if 0 <= r < n:
                    A_sub[idx, j] = a_norm[r]
        
        # Solve: min M s.t. A_sub @ g <= M, sum(g) = 1, g >= 0
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A_sub @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=cp.MOSEK)
        except:
            try:
                prob.solve(solver=cp.CLARABEL)
            except:
                prob.solve(solver=cp.ECOS)
        
        if g_var.value is None:
            break
        
        g_new = np.maximum(g_var.value, 0)
        s = float(np.sum(g_new))
        if s < 1e-10:
            break
        g_new = g_new / s
        M_new = float(M_var.value)
        
        # Check feasibility against all constraints
        full_conv = A_full @ g_new  # (a * g_new)[k] for all k
        violated = np.where(full_conv > M_new + 1e-8)[0]
        
        if len(violated) == 0:
            # Feasible! Add a few more for robustness
            g = g_new
            M = M_new
            break
        
        # Add most violated constraints
        most_violated = violated[np.argsort(full_conv[violated])[::-1][:5]]
        for k in most_violated:
            if k not in active_set:
                active_set.append(k)
        
        # Keep active set bounded
        if len(active_set) > max_constraints:
            # Remove least violated
            active_set.sort(key=lambda k: full_conv[int(k)])
            active_set = active_set[-max_constraints:]
        
        g = g_new
        M = M_new
    
    return g, M

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses active-set LP polishing with dual-informed feedback, Nesterov smoothed GD,
peak-shaving, and multi-length exploration.
Key innovations: active-set LP for fast iterations, parabolic line search,
dual-weighted gradient steps between LP solves, and adaptive active set management.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))
