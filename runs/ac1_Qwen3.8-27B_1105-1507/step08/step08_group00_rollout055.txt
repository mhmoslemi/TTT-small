"""
Optimizes non-negative step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated gradient methods, linearized LP directions, peak-targeting refinement,
and multi-length exploration with diverse restarts.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of convolution using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        if it % 5000 == 0 and verbose and it > 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def linearized_lp_direction(a, max_iter=50):
    """
    Solve LP: minimize M subject to conv(a,a)[k] + 2*conv(a, delta)[k] <= M,
    sum(delta) = 0, a + delta >= 0.
    Returns the direction delta and the predicted M.
    """
    n = len(a)
    a = a / np.sum(a)
    conv_aa = compute_conv(a)

    # Build the linearized constraint matrix
    # conv(a, delta)[k] = sum_j a[j] * delta[k-j] (for valid j, k-j)
    # We need: conv_aa[k] + 2*conv(a, delta)[k] <= M for all k

    # Build matrix B such that B @ delta = conv(a, delta)
    # B is (2n-1) x n
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        # delta[j] contributes to conv(a, delta)[k] for k = j, j+1, ..., j+n-1
        # with coefficients a[0], a[1], ..., a[n-1]
        for k_offset in range(n):
            B[j + k_offset, j] = a[k_offset]

    delta = cp.Variable(n)
    M = cp.Variable()
    constraints = [
        conv_aa + 2 * (B @ delta) <= M,
        cp.sum(delta) == 0,
        a + delta >= 0
    ]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                d = delta.value.copy()
                d = np.maximum(d, -a + 1e-15)  # ensure a + d >= 0
                return d, float(M.value)
        except Exception:
            continue
    return None, None


def lp_direction_search(a, time_budget=100.0, verbose=False):
    """
    Iteratively find LP directions and take line search steps.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        delta, M_pred = linearized_lp_direction(a)
        if delta is None:
            break

        # Line search on a + alpha * delta
        # Find best alpha in (0, 1]
        best_alpha = 0.0
        best_alpha_obj = best_obj

        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = a + alpha * delta
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_alpha_obj:
                best_alpha_obj = obj_try
                best_alpha = alpha

        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-15:
            a = best_a + best_alpha * delta  # recompute from best
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj_new = compute_obj(a)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
                if verbose and it % 5 == 0:
                    print(f"    LP-LS it{it}: obj={best_obj:.8f}, alpha={best_alpha:.2f}, M_pred={M_pred:.6f}")
        else:
            # No improvement, try smaller step or break
            improved = False
            for alpha in [0.01, 0.05, 0.1]:
                a_try = a + alpha * delta
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = best_a.copy()
                    improved = True
                    break
            if not improved:
                # Stagnation - add small perturbation
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                if it > 50:
                    break

        it += 1

    return best_a


def peak_target_refine(a, time_budget=100.0, K=3, verbose=False):
    """Target specific peaks using constrained LP."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Find top K peaks (exclude the center which is always the max for non-neg)
        # Actually include all peaks
        top_idx = np.argsort(conv)[::-1][:K]
        target_M = float(conv[top_idx[0]]) * 0.95  # try to reduce peak by 5%

        # LP: minimize sum of violations subject to constraints
        B = np.zeros((2 * n - 1, n))
        for j in range(n):
            for k_offset in range(n):
                B[j + k_offset, j] = a[k_offset]

        delta = cp.Variable(n)
        obj = cp.Minimize(cp.sum(cp.abs(2 * (B @ delta))))
        constraints = [
            cp.sum(delta) == 0,
            a + delta >= 0
        ]
        # Add peak reduction constraints
        for k in top_idx:
            k = int(k)
            constraints.append(conv[k] + 2 * (B[k] @ delta) <= target_M)

        prob = cp.Problem(obj, constraints)
        solved = False
        for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            # Relax constraints
            target_M = float(conv[top_idx[0]]) * 1.01
            constraints = [cp.sum(delta) == 0, a + delta >= 0]
            for k in top_idx:
                k = int(k)
                constraints.append(conv[k] + 2 * (B[k] @ delta) <= target_M)
            prob = cp.Problem(obj, constraints)
            for solver in [cp.CLARABEL, cp.SCS]:
                try:
                    prob.solve(solver=solver)
                    if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        solved = True
                        break
                except Exception:
                    continue

        if not solved:
            break

        delta = delta.value
        a_new = a + delta
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = best_a
            it += 1
            if it > 30:
                break

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 3:
        # Gaussian
        x = np.arange(n)
        a = np.exp(-((x - n/2)**2) / (2 * (n/4)**2))
        a = a / a.sum()
    elif mode == 4:
        # Multi-frequency
        freqs = rng.integers(1, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-3, 3)
        a = a / a.sum()
    elif mode == 6:
        # Two bumps
        c1, c2 = n // 3, 2 * n // 3
        w = n // 10
        a = np.exp(-((np.arange(n) - c1)**2) / (2*w**2)) + np.exp(-((np.arange(n) - c2)**2) / (2*w**2))
        a = a / a.sum()
    elif mode == 7:
        # Periodic
        period = max(2, n // rng.integers(3, 10))
        a = 1.0 + 0.5 * np.sin(2 * np.pi * np.arange(n) / period)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

But actually, for the linearized LP, I'm solving:
minimize M

This has 2n-1 + n + 1 = 3n variables (n for delta, 1 for M, and the constraints). For n=1000, that's about 3000 variables and about 3000 constraints. This should be solvable by solvers like CLARABEL or MOSEK in reasonable time.

Actually, let me reconsider. The number of constraints is (2n-1) + 1 + n = 3n. The number of variables is n + 1. So it's a fairly standard LP size.

Let me also think about whether there's a more efficient way to express the constraint `B @ delta`. Since B has a special structure (it's a convolution matrix), we could use FFT to compute B @ delta efficiently. But for the LP solver, we need to provide it in matrix form.

For n=1000, B is 1999x1000 with about 1000*500 = 500K nonzeros (since each column has at most n nonzeros, and the total is n*(n+1)/2 ≈ 500K for n=1000). This is sparse, which is good.

Actually, let me count more carefully. Column j of B has nonzeros at rows j, j+1, ..., j+n-1 (clamped to [0, 2n-2]). So column j has min(n, 2n-1-j) nonzeros. Total nonzeros = sum_{j=0}^{n-1} min(n, 2n-1-j).

For j = 0,...,n-1: 2n-1-j >= n (since j <= n-1, so 2n-1-j >= n). So each column has exactly n nonzeros. Total = n^2.

For n=1000, that's 1 million nonzeros. Manageable for sparse solvers.

Let me also think about whether I should use a different approach entirely. What about:

**Approach: Direct non-convex optimization using trust-region methods**

The objective `f(a) = max_k conv(a,a)[k]` (with sum(a)=1, a>=0) is non-convex. But we can use trust-region methods:
1. At current point a, compute the top K peaks.
2. Build a quadratic model: f(a + δ) ≈ f(a) + ∇f(a)^T δ + δ^T H δ / 2
   where H is the Hessian of the active peak (or a blend of peaks).
3. Solve the trust-region subproblem.
4. Take a step.

The Hessian of `conv(a,a)[k]` with respect to a is:
`H[i,j] = δ_{i,k-j} + δ_{j,k-i}` (with bounds checking)

This is a sparse matrix! For each peak k, the Hessian has at most 2n nonzeros.

This could be much more efficient than the LP approach for the final polishing steps.

Let me implement a trust-region method:

"""
Optimizes non-negative step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated gradient methods, linearized LP directions, peak-targeting refinement,
and multi-length exploration with iterative polishing loops.
Key innovations: principled linearized LP for direction finding, adaptive peak targeting,
and iterative multi-method polishing for final convergence.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """Compute gradient of log-sum-exp smoothed max of convolution using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def nesterov_smoothed_descent(a, time_budget=200.0, tau_init=10.0, tau_end=500.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.85, 1.0))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if it % 8000 == 0 and verbose and it > 0:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        y = a + beta * v

        a_new = y + v

def build_conv_matrix(a):
    """Build matrix B (2n-1 x n) such that B @ delta = conv(a, delta)."""
    n = len(a)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = a
    return B


def linearized_lp_direction(a):
    """
    Solve LP: minimize M s.t. conv(a,a)[k] + 2*conv(a, delta)[k] <= M,
    sum(delta) = 0, a + delta >= 0.
    Returns direction delta or None.
    """
    n = len(a)
    a = a / float(np.sum(a))
    conv_aa = compute_conv(a)

    B = build_conv_matrix(a)

    delta = cp.Variable(n)
    M = cp.Variable()
    constraints = [
        conv_aa + 2.0 * (B @ delta) <= M,
        cp.sum(delta) == 0,
        a + delta >= 0
    ]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if delta.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                d = delta.value.copy()
                return d, float(M.value)
        except Exception:
            continue
    return None, None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
