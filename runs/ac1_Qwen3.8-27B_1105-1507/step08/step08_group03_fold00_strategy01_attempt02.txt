def prepare_bases(size):
    """Transform a size specification intor a paired basis system."""
    ends = [0, size-1]
    while len(ends) < 2:
        ends.append((size-1)//2)
        # Alternate between extreme points and their complements
        if len(ends) % 2 == 0:
            ends.append(size-1-len(ends))
    # Filter for non-trivial dimensions, ensure distinct pairs
    base_dim = min(max(20, size//2), 100)
    if base_dim > len(ends):
        bases = [1 for _ in range(base_dim)]
    else:
        bases = ends[:base_dim]
    return bases

def compute_conv_bases(bases):
    """Compute the self-convolution in the desired basis form."""
    # Apply transformations onto the bases space
    conv_vals = [sum(1 for x in bases)]
