from scipy.optimize import linprog, minimize as sp_minimize


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:

    a_norm = a / s
    conv = compute_conv(a_norm)

def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def build_and_solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).

    a = a / float(np.sum(a))

    # A[k,j] = a[k-j] if 0 <= k-j < n

    A_ub[:, n] = -1.0

def golden_section_search(a, g, n, max_evals=25):
    """
    Golden section search for optimal t in [0,1] minimizing obj((1-t)*a + t*g).
    The objective is convex in t since it's a max of quadratics.
    """
    inv_phi = (np.sqrt(5) - 1) / 2.0
    lo, hi = 0.0, 1.0

    # Evaluate at golden points
    t1 = lo + (1.0 - inv_phi) * (hi - lo)
    t2 = lo + inv_phi * (hi - lo)

    a1 = (1 - t1) * a + t1 * g
    a2 = (1 - t2) * a + t2 * g
    f1 = compute_obj(a1)
    f2 = compute_obj(a2)

    evals = 2
    for _ in range(max_evals - 2):
        if f1 < f2:
            hi = t2
            t2 = t1
            f2 = f1
            t1 = lo + (1.0 - inv_phi) * (hi - lo)
            at1 = (1 - t1) * a + t1 * g
            f1 = compute_obj(at1)
        else:
            lo = t1
            t1 = t2
            f1 = f2
            t2 = lo + inv_phi * (hi - lo)
            at2 = (1 - t2) * a + t2 * g
            f2 = compute_obj(at2)
        evals += 1

    t_best = (lo + hi) / 2.0
    a_best = (1 - t_best) * a + t_best * g
    f_best = compute_obj(a_best)

    s = x.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    x = x / s
    conv = compute_conv(x)
    max_c = float(conv.max())
    z = T * (conv - max_c)
    e = np.exp(z)
    sum_e = float(e.sum())
    obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)
    w = e / sum_e
    g = np.zeros(n)
    active = w > 1e-6
    for k in np.where(active)[0]:
        g += w[k] * grad_conv_at_peak(x, int(k))
    g *= 2.0 * n
    # Subtract gradient component along the sum direction (for normalization)
    g = g - np.sum(g) / n * np.ones(n)
    return obj_val, g


def smooth_refine(a, time_budget=60.0, T=50.0):
    """Refine using L-BFGS-B on the smooth approximation."""
    n = len(a)
    a = np.maximum(a, 1e-12).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_wrapper(x):
        return smooth_obj_grad(x, n, T)

    try:
        res = sp_minimize(obj_wrapper, a, jac=True, method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': 300, 'maxfun': 600})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
    except Exception:
        pass

    return best_a


def nesterov_subgrad(a, time_budget=60.0, K=5):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 60
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        spread = (peak_vals[0] - peak_vals[-1]) / max_val
        temp = 5.0 / (1.0 + spread * 5.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_peak(y, int(k))

        lr = lr0 / t_k
        a_new = y - lr * g
        # Project onto simplex
        v = np.sort(a_new)[::-1]
        cssv = np.cumsum(v)
        idx = np.arange(1, n + 1)
        cond = v * idx > (cssv - 1.0)
        rho = np.where(cond)[0]
        if len(rho) == 0:
            a_new = np.ones(n) / n
        else:
            rho = rho[-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_new - theta, 0.0)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.5)

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations with emphasis on asymmetry."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        # Geometric decay (right-leaning)
        r = rng.uniform(0.95, 0.995)
        a = r ** np.arange(n)
        a = a / a.sum()
    elif mode == 2:
        # Geometric decay (left-leaning)
        r = rng.uniform(0.95, 0.995)
        a = r ** (n - 1 - np.arange(n))
        a = a / a.sum()
    elif mode == 3:
        # Linear ramp
        a = np.arange(1, n + 1, dtype=float)
        a = a / a.sum()
    elif mode == 4:
        # Inverse linear ramp
        a = np.arange(n, 0, -1, dtype=float)
        a = a / a.sum()
    elif mode == 5:
        # Sine bump (centered)
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
        a = a / a.sum()
    elif mode == 6:
        # Sine bump (off-center)
        x = np.linspace(0, 1, n)
        center = rng.uniform(0.2, 0.8)
        a = np.sin(np.pi * (x - center + 0.5)) ** 2
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 7:
        # Exponential bump
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a = a / a.sum()
    elif mode == 8:
        # Cosine window (asymmetric)
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x * rng.uniform(1, 3))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 9:
        # Two bumps
        c1, c2 = rng.uniform(0.1, 0.4), rng.uniform(0.6, 0.9)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - c1) ** 2) / 0.02) + np.exp(-((x - c2) ** 2) / 0.02)
        a = a / a.sum()
    elif mode == 10:
        # Polynomial profile
        p = rng.uniform(1, 5)
        x = np.linspace(0, 1, n)
        a = x ** p
        a = a / a.sum()
    elif mode == 11:
        # Inverse polynomial
        p = rng.uniform(1, 5)
        x = np.linspace(0, 1, n)
        a = (1 - x) ** p
        a = a / a.sum()
    else:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def lp_alternating(a, time_budget, rng, verbose=False):
    """
    LP alternating with golden-section line search and adaptive restarts.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        # Solve LP
        g_lp, M_lp = build_and_solve_lp(a)
        if g_lp is None:
            # Fallback: use current a as direction
            g_lp = a.copy()
            M_lp = float(np.max(compute_conv(a)))

        # Golden section line search
        t_best, f_best, a_new = golden_section_search(a, g_lp, n, max_evals=20)

        it += 1
        if f_best < best_obj - 1e-14:
            best_obj = f_best
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 5 == 0:
                print(f"      LP it{it}: obj={best_obj:.8f}, t={t_best:.4f}, M={M_lp:.8f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 10:
                # Adaptive restart: symmetry-breaking perturbation
                noise_scale = min(0.5, 0.05 * stall)
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
            elif stall > 5:
                # Try a larger step
                a_try = 0.7 * a + 0.3 * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stall = 0

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid LP alternating + smooth optimization + multi-n exploration.
    Focuses on asymmetric sequences to suppress the center convolution peak.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    best_n = 800
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                best_n = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    # === Phase 1: Diverse initialization screening (150s) ===
    print(f"\n=== Phase 1: Diverse Init Screening ===")
    p1_end = start_time + min(150.0, (deadline - start_time) * 0.1)

    # Try several asymmetric initializations
    for mode in range(12):
        if time.monotonic() > p1_end:
            break
        a_init = diverse_init(best_n, rng, mode=mode)
        # Quick LP polish (30s max per init)
        alloc = min(30.0, p1_end - time.monotonic())
        if alloc < 5:
            break
        a_opt = lp_alternating(a_init, time_budget=alloc, rng=rng)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  mode={mode}: obj={best_score:.8f} **")
        else:
            print(f"  mode={mode}: obj={obj:.8f}")

    # === Phase 2: Intensive LP alternating on best (600s) ===
    print(f"\n=== Phase 2: Intensive LP (n={len(best_seq)}) ===")
    p2_end = min(deadline, start_time + 750.0)

    # Run multiple rounds with increasing perturbation
    for rnd in range(4):
        remaining = p2_end - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.35, 200.0)
        if alloc < 20:
            break

        if rnd == 0:
            a_start = best_seq.copy()
        else:
            scale = 0.02 * rnd
            a_start = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        a_lp = lp_alternating(a_start, time_budget=alloc, rng=rng, verbose=(rnd == 0))
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  Round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Smooth optimization refinement (200s) ===
    print(f"\n=== Phase 3: Smooth Refinement ===")
    p3_end = min(deadline, start_time + 950.0)

    for T_val in [30.0, 80.0, 200.0]:
        remaining = p3_end - time.monotonic()
        if remaining < 15:
            break
        alloc = min(remaining * 0.3, 60.0)
        if alloc < 10:
            continue
        a_sm = smooth_refine(best_seq, time_budget=alloc, T=T_val)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Smooth T={T_val}: obj={best_score:.8f} **")
        else:
            print(f"  Smooth T={T_val}: obj={obj_sm:.8f}")

    # === Phase 4: Multi-n exploration (250s) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    p4_end = min(deadline, start_time + 1200.0)

    n_current = len(best_seq)
    n_options = sorted(set([
        max(200, n_current - 100),
        max(200, n_current - 50),
        n_current + 50,
        n_current + 100,
        n_current + 200,
        n_current + 300,
        600, 700, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000]
    n_options.sort(key=lambda x: abs(x - n_current))
    # Remove current n
    n_options = [n for n in n_options if n != n_current]

    n_tried = 0
    for n_try in n_options[:8]:
        remaining = p4_end - time.monotonic()
        if remaining < 40 or n_tried >= 6:
            break
        n_tried += 1

        a_try = resample_sequence(best_seq, n_try)

        # Quick LP optimization
        alloc_lp = min(remaining * 0.12, 50.0)
        if alloc_lp >= 10:
            a_try = lp_alternating(a_try, time_budget=alloc_lp, rng=rng)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            best_n = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts with LP (150s) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    p5_end = min(deadline, start_time + 1350.0)
    n_best = len(best_seq)

    for r in range(6):
        remaining = p5_end - time.monotonic()
        if remaining < 30:
            break

        mode = (r * 3 + 1) % 12
        a_r = diverse_init(n_best, rng, mode=mode)
        # Blend with best to stay in good region
        blend = 0.3 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = float(np.sum(a_r))
        if s_r > 1e-10:
            a_r = a_r / s_r

        alloc = min(remaining * 0.15, 50.0)
        if alloc < 10:
            break
        a_opt = lp_alternating(a_r, time_budget=alloc, rng=rng)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 30:
        alloc = min(remaining * 0.4, 100.0)
        a_lp = lp_alternating(best_seq, time_budget=alloc, rng=rng, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Nesterov polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 50.0)
        a_ns = nesterov_subgrad(best_seq, time_budget=alloc, K=7)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Final NS: obj={best_score:.8f} **")

    # Smooth final
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 40.0)
        a_sm = smooth_refine(best_seq, time_budget=alloc, T=50.0)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Final Smooth: obj={best_score:.8f} **")

    # One more LP if time
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 40.0), rng=rng)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with the actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
