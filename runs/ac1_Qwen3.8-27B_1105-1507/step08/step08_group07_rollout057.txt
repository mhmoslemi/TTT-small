def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:

    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)

        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternating_v2(a, time_budget=200.0, verbose=False):
    """LP alternating with over-relaxation and adaptive step sizes."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    gamma = 0.6

    prev_a = None
    prev_prev_a = None

    while time.monotonic() - start < time_budget:
        result = solve_lp_fast(a)
        if result[0] is None:
            break
        g, M_lp = result

        # 2-cycle detection
        if prev_prev_a is not None:
            diff = np.max(np.abs(a - prev_prev_a))
            if diff < 1e-8:
                a_break = 0.5 * (a + prev_a)
                a_break = np.maximum(a_break, 0)
                sb = float(np.sum(a_break))
                if sb > 1e-10:
                    a_break = a_break / sb
                    obj_break = compute_obj(a_break)
                    if obj_break < best_obj:
                        best_obj = obj_break
                        best_a = a_break.copy()
                    a = a_break
                    prev_a = None
                    prev_prev_a = None
                    continue

        # Try multiple step sizes including over-relaxation
        best_step_obj = np.inf
        best_step_a = a.copy()
        best_step = 0.5

        steps = [0.2, 0.3, 0.5, 0.7, 0.9, 1.0, 1.2, 1.4, 1.6]
        for step in steps:
            a_try = a + step * (g - a)
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step_a = a_try.copy()
                best_step = step

        if best_step_obj < best_obj - 1e-12:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            prev_prev_a = prev_a
            prev_a = a.copy()
            a = best_step_a
            stall = 0
            if best_step > 0.8:
                gamma = min(gamma * 1.03, 1.5)
            else:
                gamma = max(gamma * 0.97, 0.3)
        else:
            stall += 1
            if stall > 12:
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                pert_scale = 0.02 + 0.01 * min(stall - 12, 5)
                a = best_a * np.exp(rng.normal(0, pert_scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev_prev_a = None
                stall = 0
                gamma = 0.5

    return best_a


def coordinate_descent(a, time_budget=80.0, verbose=False):
    """
    Exact coordinate descent: for each j, find optimal a_j given others fixed.
    (a*a)_k is linear in a_j, so the 1D subproblem is convex (max of linear).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        improved = False
        pass_improvement = 0.0

        indices = np.arange(n)
        rng.shuffle(indices)

        for j in indices:
            if time.monotonic() - start > time_budget:
                break

            aj = a[j]
            d = np.zeros(2 * n - 1)

            # d_k = a_{k-j} (if 0<=k-j<n) + a_{j+k} (if 0<=j+k<n)
            # First term: k from j to j+n-1, d[k] += a[k-j]
            lo1 = j
            hi1 = min(2 * n - 1, j + n)
            if hi1 > lo1:
                d[lo1:hi1] += a[:hi1 - lo1]

            # Second term: k from 0 to n-j-1, d[k] += a[j+k]
            n2 = n - j
            if n2 > 0:
                d[:n2] += a[j:j + n2]

            # Double counting at k=2j: a_j appears in both terms
            k_dc = 2 * j
            if 0 <= k_dc < 2 * n - 1:
                d[k_dc] -= a[j]

            c = conv - aj * d

            # Minimize max_k(c_k + t*d_k) over t >= 0 using bisection
            max_c = np.max(c)
            active_at_0 = c >= max_c - 1e-14
            max_d_active = np.max(d[active_at_0])

            if max_d_active <= 1e-14:
                opt_t = 0.0
            else:
                lo_t = 0.0
                hi_t = max(3.0 * aj, 0.05)
                for _ in range(50):
                    mid_t = (lo_t + hi_t) * 0.5
                    vals = c + mid_t * d
                    max_val = np.max(vals)
                    active = vals >= max_val - 1e-14
                    subgrad = np.mean(d[active])
                    if subgrad > 1e-13:
                        hi_t = mid_t
                    elif subgrad < -1e-13:
                        lo_t = mid_t
                    else:
                        break
                opt_t = (lo_t + hi_t) * 0.5

            # Verify t=0 vs opt_t
            obj_opt = np.max(c + opt_t * d)
            if obj_opt >= max_c - 1e-14:
                opt_t = 0.0

            new_max = np.max(c + opt_t * d)
            if new_max < np.max(conv) * (1.0 - 1e-10):
                a[j] = opt_t
                improved = True
                pass_improvement += (np.max(conv) - new_max)

        # Renormalize
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s

        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        if not improved:
            break
        if pass_improvement < 1e-15 and it > 5:
            break

    return best_a


def fista_smoothed(a, time_budget=120.0, verbose=False):
    """
    FISTA on smoothed objective f_tau(a) = (1/tau)*log(sum_k exp(tau*(a*a)_k)).
    Uses adaptive tau (annealing) and backtracking step size.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = 10.0 / n
    x = a.copy()
    y = a.copy()
    t_k = 1.0
    step = 1.0 / (4.0 * n)

    no_improve = 0

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        # Compute gradient of smoothed objective at y
        conv_y = compute_conv(y)
        max_cv = np.max(conv_y)
        exp_vals = np.exp((conv_y - max_cv) / tau)
        p = exp_vals / np.sum(exp_vals)

        # Gradient: grad_j = sum_k p_k * (a_{k-j} + a_{j+k}) with bounds
        grad = np.zeros(n)
        for j in range(n):
            lim1 = min(n, 2 * n - 1 - j)
            if lim1 > 0:
                grad[j] += np.dot(p[j:j + lim1], y[:lim1])
            lim2 = n - j
            if lim2 > 0:
                grad[j] += np.dot(p[:lim2], y[j:j + lim2])

        grad *= (1.0 / tau)

        # Backtracking line search
        obj_y = compute_obj(y)
        for k in range(30):
            x_try = y - step / (2.0 * n) * grad
            x_try = np.maximum(x_try, 0)
            st = float(np.sum(x_try))
            if st > 1e-10:
                x_try = x_try / st
                obj_try = compute_obj(x_try)
                # Descent condition for smoothed: f(x_try) <= f(y) + grad*(x_try-y) + (1/(2L))||x_try-y||^2
                # Simplified: just check if it's an improvement or near-flat
                if obj_try <= obj_y + 1e-10:
                    break
            step *= 0.5

        x_new = x_try
        obj_x = compute_obj(x_new)
        if obj_x < best_obj - 1e-12:
            best_obj = obj_x
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy > 1e-10:
            y = y / sy
        x = x_new
        t_k = t_new

        # Anneal tau
        if it % 200 == 0 and it > 0:
            tau *= 0.92
            step = 1.0 / (4.0 * n * tau)

        if no_improve > 5000:
            break

        if verbose and it % 1000 == 0:
            print(f"    FISTA it={it}, obj={best_obj:.8f}, tau={tau:.5f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def pairwise_exchange(a, time_budget=50.0):
    """Pairwise exchange polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(4242)

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for _ in range(40):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.25]:
                delta = frac * min(a[i], a[j])
                if delta > 0.5 * min(a[i], a[j]):
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < -1e-10 or a_try[j] < -1e-10:
                    continue
                a_try = np.maximum(a_try, 0)
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved and it > 60:
            break

    return best_a


def triple_swap(a, time_budget=40.0):
    """Try moving mass between triples of coordinates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(8888)

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for _ in range(20):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            k = rng.integers(0, n)
            if len(set([i, j, k])) < 3:
                continue

            src = max(i, j, k)
            d1, d2 = min(i, j), max(i, j)
            d1 = d1 if d1 != src else d2

            for frac in [0.02, 0.05, 0.1]:
                delta = frac * a[src]
                d1_amt = delta * rng.uniform(0.3, 0.7)
                d2_amt = delta - d1_amt

                if a[src] - delta < -1e-10:
                    continue
                a_try = a.copy()
                a_try[src] -= delta
                a_try[d1] += d1_amt
                a_try[d2] += d2_amt
                if np.any(a_try < -1e-10):
                    continue
                a_try = np.maximum(a_try, 0)
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved and it > 30:
            break

    return best_a


def structured_init(n, method='cosine'):
    """Create structured initial sequences."""
    t = np.linspace(0, 1, n)
    if method == 'cosine':
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    elif method == 'sine_sq':
        a = np.sin(np.pi * t) ** 2
    elif method == 'gaussian':
        a = np.exp(-4 * (t - 0.5) ** 2)
    elif method == 'chirp_mag':
        # Magnitude of a Zadoff-Chu-like sequence
        k = np.arange(n)
        phase = -np.pi * k * (k + 1) / n
        a = np.abs(np.exp(1j * phase) + np.exp(1j * 2 * phase))
    elif method == 'triangle':
        a = 1.0 - 2.0 * np.abs(t - 0.5)
    elif method == 'flat_mod':
        a = 1.0 + 0.2 * np.cos(4 * np.pi * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternation + coordinate descent + FISTA + n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        for n_init in [500, 700, 800, 1000, 1200]:
            for method in ['cosine', 'sine_sq', 'gaussian', 'chirp_mag', 'flat_mod']:
                a_init = structured_init(n_init, method)
                obj_init = compute_obj(a_init)
                if obj_init < best_score:
                    best_score = obj_init
                    best_seq = a_init.copy()
        if best_seq is None:
            best_seq = np.ones(800) / 800
            best_score = compute_obj(best_seq)
        print(f"Initial (n={len(best_seq)}): obj={best_score:.8f}")

    a = best_seq.copy()
    n_parent = len(a)

    # === Phase 1: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alt (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(300.0, remaining * 0.30)
    a_new = lp_alternating_v2(a, time_budget=alloc, verbose=True)
    obj = compute_obj(a_new)
    if obj < best_score:
        best_score = obj
        best_seq = a_new.copy()
        a = a_new.copy()
        print(f"  LP Alt: obj={best_score:.8f} **")
    else:
        print(f"  LP Alt: obj={obj:.8f}")

    # === Phase 2: Coordinate Descent ===
    print(f"\n=== Phase 2: Coord Descent (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(100.0, remaining * 0.10)
        a_new = coordinate_descent(a, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Coord Descent: obj={best_score:.8f} **")
        else:
            print(f"  Coord Descent: obj={obj:.8f}")

    # === Phase 3: FISTA ===
    print(f"\n=== Phase 3: FISTA (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(120.0, remaining * 0.12)
        a_new = fista_smoothed(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  FISTA: obj={best_score:.8f} **")
        else:
            print(f"  FISTA: obj={obj:.8f}")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_candidates = sorted(set([
        max(300, n_best - 200), max(300, n_best - 100),
        n_best - 50 if n_best > 350 else 350,
        n_best,
        n_best + 50, n_best + 100, n_best + 200, n_best + 400,
        400, 500, 600, 700, 900, 1000, 1100, 1200, 1500, 2000,
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2500]
    n_candidates.sort(key=lambda x: abs(x - n_best))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == n_best:
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(40.0, remaining * 0.08)
        a_opt = lp_alternating_v2(a_try, time_budget=alloc)
        # Quick coord descent polish
        obj_try = compute_obj(a_opt)
        if obj_try < best_score + 0.005:
            a_opt = coordinate_descent(a_opt, time_budget=min(15.0, remaining * 0.03))
            obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Second LP round on best ===
    print(f"\n=== Phase 5: LP Round 2 (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(200.0, remaining * 0.25)
        a_new = lp_alternating_v2(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  LP Round 2: obj={best_score:.8f} **")
        else:
            print(f"  LP Round 2: obj={obj:.8f}")

        # Coord descent
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(60.0, remaining * 0.10)
            a_new = coordinate_descent(best_seq, time_budget=alloc)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Coord Descent 2: obj={best_score:.8f} **")

    # === Phase 6: Pairwise + Triple Polish ===
    print(f"\n=== Phase 6: Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(50.0, remaining * 0.15)
        a_new = pairwise_exchange(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Pairwise: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(40.0, remaining * 0.15)
            a_new = triple_swap(best_seq, time_budget=alloc)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Triple: obj={best_score:.8f} **")

    # === Phase 7: Random Restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(8, int(remaining // 35))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if r % 4 == 0:
                scale = 0.03 + 0.015 * r
                a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            elif r % 4 == 1:
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 80)
                block_size = n_best // n_blocks
                n_blocks_actual = n_best // block_size
                blocks = rng.choice(n_blocks_actual, min(4, n_blocks_actual), replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = si + block_size
                    a_r[si:ei] *= rng.uniform(0.4, 1.6)
            elif r % 4 == 2:
                a_r = best_seq.copy()
                n_sparse = max(10, n_best // 15)
                idx = rng.choice(n_best, n_sparse, replace=False)
                a_r[idx] *= rng.uniform(0.2, 3.0, n_sparse)
            else:
                # Fresh structured init at same or nearby n
                n_fresh = n_best + rng.integers(-200, 200)
                n_fresh = max(400, min(2000, n_fresh))
                method = rng.choice(['cosine', 'sine_sq', 'gaussian', 'chirp_mag', 'flat_mod', 'triangle'])
                a_r = structured_init(n_fresh, method)

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            alloc = min(remaining * 0.3, 30.0)
            a_opt = lp_alternating_v2(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 20.0)
        a_final = coordinate_descent(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final Coord: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_final = pairwise_exchange(best_seq, time_budget=min(remaining * 0.5, 10.0))
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final Pairwise: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
