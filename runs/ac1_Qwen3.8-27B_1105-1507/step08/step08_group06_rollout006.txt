"""
Hybrid peak-gradient + LP-alternating optimizer for minimizing
2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key ideas:
- Exact peak-tracking gradient method with Nesterov acceleration
- Sparse Hessian exploitation for Newton steps when peak is unique
- LP-based direction for diversification and escaping local minima
- Multi-peak subgradient handling via active-set QP
- Diverse initializations and length sweep
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv / sum^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_obj_normalized(a):
    """Compute objective assuming a is normalized to sum=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_response(a, max_time=10.0):
    """
    Solve LP: min M s.t. conv(a, g) <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    start = time.monotonic()
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        if time.monotonic() - start > max_time:
            break
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:

            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                return g_val, M_val
    except Exception:
        pass

    return None, None


def negate_and_shift(a, k_star, n):
    """
    Compute the gradient direction for reducing peak at k_star.
    grad_j = 2*a[k_star - j] if 0 <= k_star - j < n, else 0
    Projected to sum-0: grad_j - mean(grad)
    Returns the NEGATIVE projected gradient (descent direction).
    """
    grad = np.zeros(n)
    for j in range(n):
        k_j = k_star - j
        if 0 <= k_j < n:
            grad[j] = 2.0 * a[k_j]
        else:
            # Check if j - k_star is valid (for the symmetric term)
            k_j2 = j - k_star
            if 0 <= k_j2 < n:
                grad[j] = 2.0 * a[k_j2]
    # Project to tangent space of simplex (sum = 0)
    grad_centered = grad - np.mean(grad)
    return -grad_centered  # descent direction


def peak_gradient_step(a, direction, max_iter=50):
    """
    Line search along a - t * direction (projected to simplex at each step).
    Finds t that minimizes the objective.
    """
    n = len(a)
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()
    best_t = 0.0

    # Find max feasible t (where a - t*dir would hit zero)
    pos_mask = direction > 1e-12
    if np.any(pos_mask):
        t_max = np.min(a[pos_mask] / direction[pos_mask]) * 0.99
        t_max = max(t_max, 0.0)
    else:
        t_max = 1.0

    if t_max < 1e-10:
        return a, best_obj

    # Try a few t values via golden section or simple bisection
    t_candidates = np.linspace(0.001, min(t_max, 0.5), 20)
    # Also add some larger values
    t_candidates = np.unique(np.concatenate([t_candidates, [0.5, 1.0, 2.0, 5.0, 10.0]]))
    t_candidates = t_candidates[t_candidates <= t_max]

    for t in t_candidates:
        a_try = a - t * direction
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj_normalized(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            best_t = t

    # Refine around best_t if found
    if best_t > 0:
        t_lo = max(0, best_t * 0.5)
        t_hi = best_t * 1.5 if best_t * 1.5 <= t_max else t_max
        for _ in range(10):
            t_mid = (t_lo + t_hi) / 2
            a_mid = np.maximum(a - t_mid * direction, 0)
            s_mid = float(np.sum(a_mid))
            if s_mid < 1e-10:
                break
            a_mid = a_mid / s_mid
            obj_mid = compute_obj_normalized(a_mid)
            if obj_mid < best_obj:
                best_obj = obj_mid
                best_a = a_mid.copy()
                best_t = t_mid
                t_lo = t_mid
            else:
                t_hi = t_mid
            if t_hi - t_lo < 1e-6:
                break

    return best_a, best_obj


def active_subgradient_direction(a, active_peaks, n):
    """
    Compute the worst-case subgradient direction (most negative)
    among the convex hull of gradients at active peaks.
    For simplicity, use the average gradient direction.
    """
    total_grad = np.zeros(n)
    for k_star in active_peaks:
        d = negate_and_shift(a, k_star, n)
        total_grad += d
    total_grad /= len(active_peaks)
    return total_grad


def peak_gradient_optimizer(a, time_budget=100.0, verbose=False):
    """
    Main optimizer: Nesterov-accelerated projected gradient with line search,
    with periodic LP-based diversification.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    # Nesterov momentum
    y = a.copy()  # lookahead
    momentum = np.zeros(n)
    beta = 0.7
    lr = 0.02
    stall = 0
    lp_counter = 0
    best_lp_dir = None

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        # Compute convolution at current point
        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Find active peaks (within 1% of max)
        threshold = max_val * 0.99
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = [int(np.argmax(conv))]
        # Limit to top 10 peaks
        if len(active) > 10:
            active = np.argsort(conv)[::-1][:10]

        # Compute direction
        if len(active) == 1:
            direction = negate_and_shift(a, int(active[0]), n)
        else:
            direction = active_subgradient_direction(a, active, n)

        # Nesterov update
        t_k = 2.0 / (np.sqrt(max(stall, 1)) + 2.0)
        momentum = beta * momentum - lr * direction
        y = a + t_k * momentum
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy > 1e-10:
            y = y / sy

        # Take step from y
        a_new, obj_new = peak_gradient_step(y, np.zeros(n) if not np.any(momentum != 0) else -momentum, max_iter=30)

        # Actually, let's just do a direct step
        a_try = y - lr * 0.5 * direction
        a_try = np.maximum(a_try, 0)
        sa = float(np.sum(a_try))
        if sa > 1e-10:
            a_try = a_try / sa
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                stall = 0
            else:
                # Try smaller step
                for frac in [0.5, 0.25, 0.1]:
                    a_try2 = a - lr * frac * direction
                    a_try2 = np.maximum(a_try2, 0)
                    sa2 = float(np.sum(a_try2))
                    if sa2 < 1e-10:
                        continue
                    a_try2 = a_try2 / sa2
                    obj_try2 = compute_obj_normalized(a_try2)
                    if obj_try2 < best_obj:
                        best_obj = obj_try2
                        best_a = a_try2.copy()
                        a = a_try2
                        stall = 0
                        break
                else:
                    stall += 1

        # Update a to best found
        a = best_a if obj_new < best_obj else a

        # Periodic LP diversification
        lp_counter += 1
        if stall > 20 and (lp_counter % 30 == 0 or best_lp_dir is None):
            if time.monotonic() - start < time_budget - 20:
                g_lp, M_lp = solve_lp_for_response(a, max_time=min(15.0, time_budget - (time.monotonic() - start) - 5))
                if g_lp is not None:
                    best_lp_dir = g_lp
                    # Mix with LP direction
                    mix = 0.3 + 0.1 * min(stall // 10, 5)
                    a_mix = (1 - mix) * best_a + mix * g_lp
                    a_mix = np.maximum(a_mix, 0)
                    sm = float(np.sum(a_mix))
                    if sm > 1e-10:
                        a_mix = a_mix / sm
                        obj_mix = compute_obj_normalized(a_mix)
                        if obj_mix < best_obj:
                            best_obj = obj_mix
                            best_a = a_mix.copy()
                            a = a_mix
                            stall = 0
                            if verbose:
                                print(f"    LP mix improved: {best_obj:.8f} at t={elapsed:.0f}s")
                    momentum = np.zeros(n)
                else:
                    # Random perturbation
                    rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                    scale = 0.02 + 0.01 * min(stall // 20, 5)
                    a_pert = best_a * np.exp(rng.normal(0, scale, n))
                    a_pert = np.maximum(a_pert, 0)
                    sp = float(np.sum(a_pert))
                    if sp > 1e-10:
                        a_pert = a_pert / sp
                        obj_pert = compute_obj_normalized(a_pert)
                        # Accept even if worse (simulated annealing style)
                        if obj_pert < best_obj * 1.05:
                            a = a_pert
                            stall = 0
                    momentum = np.zeros(n)

        # Adaptive learning rate
        if stall > 10:
            lr = max(lr * 0.95, 0.001)
        elif stall == 0:
            lr = min(lr * 1.02, 0.05)

        # Check convergence
        if stall > 200:
            break

    return best_a


def newton_peak_optimizer(a, time_budget=100.0, verbose=False):
    """
    Newton-type optimizer exploiting the sparse anti-diagonal Hessian structure.
    Works on the simplex by parameterizing with n-1 free variables.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        k_star = int(np.argmax(conv))

        # Check if peak is unique
        threshold = max_val * 0.999
        n_active = np.sum(conv >= threshold)

        if n_active > 1:
            # Multi-peak: fall back to gradient step
            direction = active_subgradient_direction(a, np.where(conv >= threshold)[0], n)
            a_new, obj_new = peak_gradient_step(a, direction)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 50:
                    break
            continue

        # Unique peak: use Newton-like step
        # Gradient of P = (a*a)_{k*}: grad_j = 2*a[k*-j] (if valid)
        grad = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                grad[j] = 2.0 * a[k_j]

        # Project gradient to tangent space
        grad_proj = grad - np.mean(grad)

        # Hessian of P is anti-diagonal: H[i,j] = 2 if i+j = k_star, else 0
        # For Newton step on simplex: solve (H - lambda*1*1') d = -grad_proj
        # Approximate: just use gradient descent with Barzillow-Borwein step
        # (Full Newton is expensive in high dimensions)

        # BB step size
        if hasattr(newton_peak_optimizer, '_prev_grad') and newton_peak_optimizer._prev_grad is not None:
            gy = grad_proj - newton_peak_optimizer._prev_grad
            ys = a - newton_peak_optimizer._prev_x
            denom = np.dot(gy, ys)
            if abs(denom) > 1e-12:
                lr_bb = np.dot(gy, gy) / denom
                lr_bb = np.clip(lr_bb, 1e-6, 1.0)
            else:
                lr_bb = 0.01
        else:
            lr_bb = 0.01

        newton_peak_optimizer._prev_grad = grad_proj.copy()
        newton_peak_optimizer._prev_x = a.copy()

        a_try = a - lr_bb * grad_proj
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            stall += 1
            continue
        a_try = a_try / st
        obj_try = compute_obj_normalized(a_try)

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
        else:
            # Backtracking
            t = lr_bb
            improved = False
            for _ in range(10):
                t *= 0.5
                a_bt = np.maximum(a - t * grad_proj, 0)
                sbt = float(np.sum(a_bt))
                if sbt < 1e-10:
                    break
                a_bt = a_bt / sbt
                obj_bt = compute_obj_normalized(a_bt)
                if obj_bt < best_obj - 1e-14:
                    best_obj = obj_bt
                    best_a = a_bt.copy()
                    a = a_bt
                    stall = 0
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall > 30:
                    # Perturb
                    rng = np.random.default_rng(int(time.monotonic() * 1e5) % (2**31))
                    a = best_a * np.exp(rng.normal(0, 0.02, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    stall = 0
                if stall > 100:
                    break

    newton_peak_optimizer._prev_grad = None
    newton_peak_optimizer._prev_x = None
    return best_a


def lp_alternating_simple(a, time_budget=100.0, verbose=False):
    """Simplified LP alternating with adaptive mixing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    stall = 0
    mix = 0.3

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_for_response(a, max_time=min(20.0, time_budget - (time.monotonic() - start) - 5))
        if g_lp is None:
            break

        # Try multiple mixing ratios
        best_local_obj = best_obj
        best_local_a = best_a.copy()

        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_local_a
            stall = 0
            mix = min(mix * 1.1, 0.8)
            if verbose:
                print(f"    LP-alt: obj={best_obj:.8f}, t={mix:.2f}, time={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            mix = max(mix * 0.8, 0.05)
            if stall % 20 == 0:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.02 + 0.005 * min(stall // 20, 8), n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0

        if stall > 150:
            break

    return best_a


def coordinate_descent_refine(a, time_budget=60.0):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given all others fixed, minimizing the peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    for outer in range(50):
        if time.monotonic() - start > time_budget:
            break
        improved = False

        # Process coordinates in random order
        rng = np.random.default_rng(42 + outer)
        order = rng.permutation(n)

        for idx in order:
            if time.monotonic() - start > time_budget:
                break
            j = int(idx)
            current_val = a[j]
            if current_val < 1e-10:
                # Try to increase
                for frac in [0.001, 0.005, 0.01, 0.02]:
                    a[j] = frac
                    s = float(np.sum(a))
                    a = a / s
                    obj = compute_obj_normalized(a)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = 0
                        s = float(np.sum(a))
                        a = a / s
                        break
            else:
                # Try decreasing
                for frac in [0.5, 0.8, 0.9, 0.95]:
                    new_val = current_val * frac
                    a[j] = new_val
                    s = float(np.sum(a))
                    if s < 1e-10:
                        a[j] = current_val
                        break
                    a = a / s
                    obj = compute_obj_normalized(a)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a.copy()
                        current_val = new_val
                        improved = True
                    else:
                        a[j] = current_val
                        s = float(np.sum(a))
                        a = a / s
                        break

        if not improved and outer > 5:
            break

    return best_a


def generate_initialization(kind, n, rng):
    """Generate diverse initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'bilinear':
        a = np.linspace(0.5, 1.5, n)
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 8)
        a[:edge] = 0.3
        a[-edge:] = 0.3
        return a / np.sum(a)
    elif kind == 'two_peaks':
        a = np.ones(n) * 0.3
        a[n//4] = 1.0
        a[3*n//4] = 1.0
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'chirp_real':
        # Real part of a chirp (Zadoff-Chu like)
        x = np.arange(n)
        a = np.abs(np.exp(1j * np.pi * x * x / n)) + 0.1
        return a / np.sum(a)
    elif kind == 'alternating':
        a = np.ones(n)
        a[::2] = 0.3
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def hybrid_optimize(a, time_budget=120.0, verbose=False):
    """
    Hybrid optimizer: alternates between peak-gradient, Newton, and LP methods.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    # Phase A: Peak gradient (40% of time)
    t_a = time_budget * 0.4
    a_res = peak_gradient_optimizer(a, time_budget=t_a, verbose=verbose)
    obj_res = compute_obj_normalized(a_res)
    if obj_res < best_obj:
        best_obj = obj_res
        best_a = a_res.copy()
        a = a_res
        if verbose:
            print(f"    PeakGrad: {best_obj:.8f}")

    # Phase B: LP alternating (30% of time)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t_b = remaining * 0.4
        a_res = lp_alternating_simple(a, time_budget=t_b, verbose=verbose)
        obj_res = compute_obj_normalized(a_res)
        if obj_res < best_obj:
            best_obj = obj_res
            best_a = a_res.copy()
            a = a_res
            if verbose:
                print(f"    LP-alt: {best_obj:.8f}")

    # Phase C: Newton (20% of time)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t_c = remaining * 0.4
        a_res = newton_peak_optimizer(a, time_budget=t_c, verbose=verbose)
        obj_res = compute_obj_normalized(a_res)
        if obj_res < best_obj:
            best_obj = obj_res
            best_a = a_res.copy()
            a = a_res
            if verbose:
                print(f"    Newton: {best_obj:.8f}")

    # Phase D: Coordinate descent polish (10% of time)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_res = coordinate_descent_refine(a, time_budget=remaining * 0.5)
        obj_res = compute_obj_normalized(a_res)
        if obj_res < best_obj:
            best_obj = obj_res
            best_a = a_res.copy()
            if verbose:
                print(f"    CoordDesc: {best_obj:.8f}")

    # Phase E: Final LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_res = lp_alternating_simple(best_a, time_budget=remaining * 0.6, verbose=verbose)
        obj_res = compute_obj_normalized(a_res)
        if obj_res < best_obj:
            best_obj = obj_res
            best_a = a_res.copy()
            if verbose:
                print(f"    FinalLP: {best_obj:.8f}")

    return best_a

        if remaining < 20:
            break
        a_init = generate_initialization(kind, n_parent, rng)
        obj_init = compute_obj(a_init)
        all_candidates.append((obj_init, a_init.copy(), kind))
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init '{kind}': obj={obj_init:.8f} **")
        else:
            print(f"  Init '{kind}': obj={obj_init:.8f}")

    # A few random inits
    for i in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        a_init = generate_initialization('random', n_parent, rng)
        obj_init = compute_obj(a_init)
        all_candidates.append((obj_init, a_init.copy(), f'rand_{i}'))
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Random init {i}: obj={obj_init:.8f} **")

    # Sort candidates by initial objective
    all_candidates.sort(key=lambda x: x[0])
    print(f"  Best after Phase 1: obj={best_score:.8f}")

    # === Phase 2: Primary Optimization on Best Candidate ===
    print(f"\n=== Phase 2: Primary Optimization (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    primary_budget = min(500.0, remaining * 0.4)

    a_current = best_seq.copy()
    a_opt = hybrid_optimize(a_current, time_budget=primary_budget, verbose=True)
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  ** Primary: obj={best_score:.8f} **")
    else:
        print(f"  Primary: obj={obj_opt:.8f}")

    # === Phase 3: Second round from best with different strategy ===
    print(f"\n=== Phase 3: Second Round (different emphasis) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        round2_budget = min(300.0, remaining * 0.25)

        # Try LP-heavy approach
        a_lp = lp_alternating_simple(best_seq, time_budget=round2_budget * 0.5, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LP-heavy: obj={best_score:.8f} **")
        else:
            print(f"  LP-heavy: obj={obj_lp:.8f}")

        # Try Newton-heavy approach
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_newton = newton_peak_optimizer(best_seq, time_budget=min(100.0, remaining * 0.3), verbose=True)
            obj_newton = compute_obj(a_newton)
            if obj_newton < best_score:
                best_score = obj_newton
                best_seq = a_newton.copy()
                print(f"  ** Newton: obj={best_score:.8f} **")
            else:
                print(f"  Newton: obj={obj_newton:.8f}")

    # === Phase 4: Multi-start from diverse candidates ===
    print(f"\n=== Phase 4: Multi-start from diverse candidates ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Take top 3 diverse candidates (not the same as best)
        diverse_starts = [c for c in all_candidates if not np.allclose(c[1], best_seq, atol=0.01)][:3]

        for i, (obj_init, a_start, kind) in enumerate(diverse_starts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(80.0, remaining * 0.2)
            print(f"  Multi-start {i} ({kind}, init={obj_init:.6f}):")
            a_ms = hybrid_optimize(a_start, time_budget=alloc, verbose=False)
            obj_ms = compute_obj(a_ms)
            if obj_ms < best_score:
                best_score = obj_ms
                best_seq = a_ms.copy()
                print(f"  ** Multi-start {i}: obj={best_score:.8f} **")
            else:
                print(f"  Multi-start {i}: obj={obj_ms:.8f}")

    # === Phase 5: Length Sweep ===
    print(f"\n=== Phase 5: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(100, n_best - 200),
        max(100, n_best - 100),
        max(100, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 400,
    ]))
    n_options = [n for n in n_options if 100 <= n <= 8000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization
        alloc = min(40.0, remaining * 0.3)
        a_opt = hybrid_optimize(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final Intensive Polish ===
    print(f"\n=== Phase 6: Final Intensive Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    # Multiple short hybrid rounds with perturbations
    polish_count = 0
    while remaining > 20 and polish_count < 5:
        alloc = min(remaining * 0.35, 80.0)
        # Slight perturbation for diversity
        scale = 0.01 * (polish_count + 1)
        a_pert = best_seq * np.exp(rng.normal(0, scale, n_final))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / np.sum(a_pert)

        a_polish = hybrid_optimize(a_pert, time_budget=alloc, verbose=True)
        obj_polish = compute_obj(a_polish)

        if obj_polish < best_score:
            best_score = obj_polish
            best_seq = a_polish.copy()
            print(f"  ** Polish {polish_count}: obj={best_score:.8f} **")
        else:
            print(f"  Polish {polish_count}: obj={obj_polish:.8f}")

        polish_count += 1
        remaining = deadline - time.monotonic()

        # Also try from the unperturbed best
        if remaining > 20:
            a_polish2 = hybrid_optimize(best_seq, time_budget=min(30.0, remaining * 0.3), verbose=False)
            obj_polish2 = compute_obj(a_polish2)
            if obj_polish2 < best_score:
                best_score = obj_polish2
                best_seq = a_polish2.copy()
                print(f"  ** Polish {polish_count} (clean): obj={best_score:.8f} **")
            remaining = deadline - time.monotonic()

    # === Phase 7: Final coordinate descent ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd = coordinate_descent_refine(best_seq, time_budget=min(remaining * 0.4, 30.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
