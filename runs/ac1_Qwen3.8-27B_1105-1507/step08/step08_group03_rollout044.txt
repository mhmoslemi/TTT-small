def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def grad_conv_at_peak(a, k):

        j_end = k - i_start

                cached_A = build_conv_matrix(a)
            if verbose and it % 10 == 0:
                print(f"    LP2 it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 12:
                # Escape: multiplicative perturbation
                noise_scale = min(0.4, 0.05 * (1 + stall // 12))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                cached_A = build_conv_matrix(a)
                stall = 0
                best_t = 0.3

    return best_a


def mirror_descent_v2(a, time_budget=60.0, verbose=False):
    """
    Improved mirror descent with:
    - Multi-temperature softmax over all peaks
    - Momentum with adaptive decay
    - Periodic re-normalization in log space
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    min_lr = 1e-6
    momentum = 0.0
    v = np.zeros(n)
    stall = 0
    rng = np.random.default_rng(999 + n)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(conv.max())
        if max_val < 1e-15:
            break

        # Multi-temperature weighting
        temps = [1.0, 3.0, 10.0, 50.0]
        all_w = np.zeros(len(conv))
        for temp in temps:
            z = (conv - max_val) / (temp * max_val + 1e-15)
            w = np.exp(z)
            w /= w.sum()
            all_w += w
        all_w /= len(temps)

        # Only use active weights (top 10%)
        if len(conv) > 100:
            threshold = np.percentile(all_w, 90)
            active = all_w > threshold
            active_w = all_w.copy()
            active_w[~active] = 0
        else:
            active_w = all_w
        active_w_sum = active_w.sum()
        if active_w_sum > 0:
            active_w /= active_w_sum

        # Compute gradient
        g = np.zeros(n)
        active_idx = np.where(active_w > 1e-12)[0]
        for k in active_idx:
            if active_w[k] > 1e-12:
                g += active_w[k] * grad_conv_at_peak(a, int(k))

        # Momentum update
        momentum = min(0.7, momentum + 0.005)
        v = momentum * v + g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm

        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = max(min_lr, lr * 0.998)
        else:
            stall += 1
            if stall > 80:
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                lr = max(min_lr, lr * 0.5)
                stall = 0
            else:
                lr = max(min_lr, lr * 0.98)

        it += 1
        if verbose and it % 200 == 0:
            print(f"    MD2 it{it}: obj={best_obj:.8f}, lr={lr:.6f}, el={time.monotonic()-start:.1f}s")

    return best_a


def penalty_refine(a, time_budget=60.0, verbose=False):
    """
    Penalty-based refinement: minimize sum_k soft_max((a*a)[k]) with penalty increasing.
    Uses projected gradient in the simplex.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T_schedule = [50.0, 30.0, 15.0, 7.0, 3.0, 1.5, 0.7]
    lr = 0.5 / np.sqrt(n)

    for T in T_schedule:
        if time.monotonic() - start > time_budget:
            break
        seg_budget = (time_budget - (time.monotonic() - start)) / len(T_schedule)
        seg_start = time.monotonic()

        for it in range(200):
            if time.monotonic() - seg_start > seg_budget:
                break

            a = np.maximum(a, 1e-15)
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s

            conv = compute_conv(a)
            max_c = conv.max()
            z = T * (conv - max_c)
            e = np.exp(z)
            sum_e = e.sum()
            w = e / sum_e

            # Gradient of smooth approximation
            g = np.zeros(n)
            for k in range(len(conv)):
                if w[k] > 1e-6:
                    g += w[k] * grad_conv_at_peak(a, int(k))
            g *= 2.0 * n

            # Projected gradient step
            a_new = a - lr * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()

            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                a = best_a.copy()
                lr *= 0.8

        if verbose:
            print(f"    Penalty T={T}: obj={best_obj:.8f}")

    return best_a


def spectral_init(n, rng):
    """
    Generate initializations using spectral methods.
    Try to make |FFT(a)|^2 flat (which minimizes peak autocorrelation for complex sequences).
    For real non-negative: use squared magnitudes of good sequences.
    """
    candidates = []

    # 1. Uniform
    candidates.append(np.ones(n) / n)

    # 2. Raised cosine
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 1e-6)
    candidates.append(a / a.sum())

    # 3. Gaussian centered
    centers = [n // 3, 2 * n // 3]
    width = n // 5
    a = np.zeros(n)
    for c in centers:
        a += np.exp(-((np.arange(n) - c) ** 2) / (2 * width ** 2))
    candidates.append(a / a.sum())

    # 4. Sinc-like
    x = np.linspace(-1, 1, n)
    a = np.sinc(x * 2) ** 2 + 0.01
    candidates.append(a / a.sum())

    # 5. Random positive
    a = rng.uniform(0.1, 1.0, n)
    candidates.append(a / a.sum())

    # 6. Two-hump with phase
    x = np.linspace(0, 1, n)
    a = 1 + 0.5 * np.cos(4 * np.pi * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.01)
    candidates.append(a / a.sum())

    # 7. Non-uniform periodic
    period = max(2, n // 7)
    pattern = rng.uniform(0.2, 1.0, period)
    a = np.tile(pattern, n // period + 1)[:n]
    candidates.append(a / a.sum())

    # 8. Triangular
    x = np.linspace(0, 1, n)
    a = 1 - 2 * np.abs(x - 0.5) + 0.05
    candidates.append(a / a.sum())

    # Evaluate all
    best = None
    best_obj = np.inf
    for a in candidates:
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best = a.copy()

    return best, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def symmetric_refine(a, time_budget=30.0):
    """
    Refine assuming the sequence has approximate symmetry.
    Optimize the first half and mirror it.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    half_n = (n + 1) // 2
    half = a[:half_n].copy()

    best_obj = compute_obj(a)
    best_half = half.copy()
    start = time.monotonic()
    lr = 0.1

    def build_full(h):
        full = np.zeros(n)
        full[:half_n] = h
        full[n - 1 - np.arange(half_n)] = h
        s = full.sum()
        return full / s

    while time.monotonic() - start < time_budget:
        full = build_full(half)
        conv = compute_conv(full)
        max_c = conv.max()

        # Gradient with respect to half
        g_half = np.zeros(half_n)
        k_star = int(np.argmax(conv))
        g_full = grad_conv_at_peak(full, k_star)
        for i in range(half_n):
            g_half[i] = g_full[i]
            if n - 1 - i != i:
                g_half[i] += g_full[n - 1 - i]

        half_new = half - lr * g_half
        half_new = np.maximum(half_new, 1e-10)

        full_new = build_full(half_new)
        obj_new = compute_obj(full_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_half = half_new.copy()
            half = half_new
            lr *= 0.99
        else:
            half = best_half.copy()
            lr *= 0.8

    return build_full(best_half)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP alternating, mirror descent,
    penalty refinement, spectral initialization, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 800

    # === Phase 0: Spectral initialization and selection ===
    print(f"\n=== Phase 0: Initialization (n={n_base}) ===")
    a_spec, obj_spec = spectral_init(n_base, rng)
    print(f"  Spectral init: obj={obj_spec:.8f}")

    if obj_spec < best_score:
        best_score = obj_spec
        best_seq = a_spec.copy()
        print(f"  -> New best from spectral: {best_score:.8f}")

    # Also try different n for initialization
    for n_try in [n_base - 50, n_base + 50, n_base + 100]:
        if n_try < 100:
            continue
        a_t, obj_t = spectral_init(n_try, rng)
        if obj_t < best_score:
            best_score = obj_t
            best_seq = a_t.copy()
            n_base = n_try
            print(f"  -> New best n={n_try}: {best_score:.8f}")

    # === Phase 1: LP Alternating (main workhorse) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 1: LP Alternating (n={len(best_seq)}, budget={min(remaining*0.4, 450.0):.0f}s) ===")
    p1_budget = min(remaining * 0.4, 450.0)
    if p1_budget > 30:
        a_lp = lp_alternating_v2(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Mirror Descent ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 2: Mirror Descent (budget={min(remaining*0.15, 150.0):.0f}s) ===")
    p2_budget = min(remaining * 0.15, 150.0)
    if p2_budget > 20:
        a_md = mirror_descent_v2(best_seq, time_budget=p2_budget, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD: obj={best_score:.8f} **")
        else:
            print(f"  MD: obj={obj_md:.8f} (no improvement)")

    # === Phase 3: LP again after MD (often improves) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 3: LP Round 2 (budget={min(remaining*0.25, 300.0):.0f}s) ===")
    p3_budget = min(remaining * 0.25, 300.0)
    if p3_budget > 30:
        a_lp2 = lp_alternating_v2(best_seq, time_budget=p3_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP2: obj={best_score:.8f} **")
        else:
            print(f"  LP2: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 4: Penalty refinement ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 4: Penalty Refinement (budget={min(remaining*0.1, 80.0):.0f}s) ===")
    p4_budget = min(remaining * 0.1, 80.0)
    if p4_budget > 15:
        a_pen = penalty_refine(best_seq, time_budget=p4_budget, verbose=True)
        obj_pen = compute_obj(a_pen)
        if obj_pen < best_score:
            best_score = obj_pen
            best_seq = a_pen.copy()
            print(f"  Penalty: obj={best_score:.8f} **")

    # === Phase 5: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    print(f"\n=== Phase 5: Multi-n Exploration (current n={n_current}) ===")

    n_options = sorted(set([
        n_current - 20, n_current - 10, n_current + 10, n_current + 20,
        n_current + 40, n_current + 80,
        max(200, n_base - 100), max(200, n_base + 100),
        500, 600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500 and n != n_current]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 30 or n_tried >= 6:
            break
        n_tried += 1

        a_try = resample_sequence(best_seq, n_try)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Quick LP polish
        alloc = min(remaining * 0.1, 40.0)
        if alloc > 10:
            a_try = lp_alternating_v2(a_try, time_budget=alloc)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Diverse Restarts ===")

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = r % 8
        if mode == 0:
            a_r = np.ones(n_best) / n_best
        elif mode == 1:
            a_r = rng.uniform(0.1, 1.0, n_best)
            a_r = a_r / a_r.sum()
        elif mode == 2:
            x = np.linspace(0, 1, n_best)
            a_r = 1 + 0.4 * np.cos(2 * np.pi * (r + 2) * x + rng.uniform(0, 6.28))
            a_r = np.maximum(a_r, 0.01)
            a_r = a_r / a_r.sum()
        elif mode == 3:
            a_r = best_seq * (1 + 0.3 * rng.normal(0, 1, n_best))
            a_r = np.maximum(a_r, 1e-6)
            a_r = a_r / a_r.sum()
        elif mode == 4:
            # Periodic
            period = n_best // 5
            pattern = rng.uniform(0.3, 1.0, period)
            a_r = np.tile(pattern, n_best // period + 1)[:n_best]
            a_r = a_r / a_r.sum()
        else:
            # Blend of parent and random
            a_r = 0.5 * best_seq + 0.5 * rng.uniform(0, 1, n_best)
            a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.12, 50.0)
        if restart_budget < 10:
            continue
        a_opt = lp_alternating_v2(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 7: Final Polish (remaining={remaining:.0f}s) ===")

    # LP with long budget
    if remaining > 40:
        lp_alloc = min(remaining * 0.35, 120.0)
        a_final = lp_alternating_v2(best_seq, time_budget=lp_alloc, verbose=True)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Penalty refinement
    remaining = deadline - time.monotonic()
    if remaining > 20:
        pen_alloc = min(remaining * 0.25, 50.0)
        a_pen = penalty_refine(best_seq, time_budget=pen_alloc)
        obj_p = compute_obj(a_pen)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_pen.copy()
            print(f"  Final Penalty: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternating_v2(best_seq, time_budget=min(remaining * 0.4, 40.0))
        obj_l = compute_obj(a_last)
        if obj_l < best_score:
            best_score = obj_l
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure non-negative and valid
    best_seq = np.maximum(best_seq, 0.0)
    result = [float(min(1000.0, x)) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
