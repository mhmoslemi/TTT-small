- Fictitious play
- The exPost algorithm

The value of this game is exactly M* (the optimal LP objective).

Uses: (1) LP alternating with exact HiGHS solver as primary driver,
(2) Fast dual multiplicative-weights inner solve as fallback,
(3) Adaptive line search with golden-section refinement,
(4) L-BFGS polish on log-sum-exp smoothed objective,
(5) Structured escape via dual-informed perturbations,
(6) Multi-n exploration with resampling and re-optimization.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as sp_minimize, linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2, with a normalized to sum=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    c = compute_conv(a_n)
    return 2.0 * n * float(np.max(c))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_inner_lp(a, timeout=3.0):

    a_n = a / max(float(np.sum(a)), 1e-15)
    A = build_conv_matrix(a_n)

    for t in range(T):
        # col_vals[j] = sum_m a_n[m] * y[m+j] = ifft(fft(y) * conj(fa)).real[:n]
        fy = rfft(y, N)
        col_vals = irfft(fy * np.conj(fa), N)[:n]

        j_star = int(np.argmax(col_vals))
        col_counts[j_star] += 1

        # MW update on y
        lo = j_star
        hi = lo + n
        if hi <= L:
            y[lo:hi] *= np.exp(-eta * a_n)
        y_sum = float(np.sum(y))
        if y_sum < 1e-300:
            y = np.ones(L) / L
        else:
            y /= y_sum

    g = col_counts / max(col_counts.sum(), 1e-15)
    g = np.maximum(g, 0)
    s = g.sum()
    if s > 0:
        g /= s
    M_est = float(np.max(np.convolve(a_n, g, 'full')))
    return g, M_est


def grad_conv_at_peak(a, k):
    """Gradient of conv(a,g)[k] with respect to g, evaluated at g=a.
    This is a[k-j] for j where 0<=k-j<n, i.e., the k-th row of A."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        g[lo:hi + 1] = a[k - hi:k - lo + 1]
    return g


def line_search(a, g, t_init=0.35, n_trials=7):
    """Adaptive line search for a_new = (1-t)*a + t*g. Returns best (a_try, t, obj)."""
    s_a = float(np.sum(a))
    s_g = float(np.sum(g))
    if s_a < 1e-10 or s_g < 1e-10:
        return a, t_init, compute_obj(a)

    ts = [t_init, t_init * 0.5, t_init * 1.5, 0.15, 0.25, 0.5, 0.7, 0.9]
    ts = sorted(set(t for t in ts if 0.02 <= t <= 1.0))

    best_obj = np.inf
    best_a = a.copy()
    best_t = t_init

    for t in ts:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            best_t = t

    return best_a, best_t, best_obj


def lp_alternating(a, time_budget=300.0, verbose=False):
    """
    Main LP alternating optimizer with adaptive line search and escape mechanisms.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    best_t = 0.35
    rng = np.random.default_rng(12345 + n)
    it = 0
    use_lp = True

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        if use_lp:
            t0 = time.monotonic()
            g_inner, M_lp = solve_inner_lp(a, timeout=min(3.0, remaining * 0.3))
            lp_time = time.monotonic() - t0
            if g_inner is None:
                g_inner, M_lp = solve_inner_mw(a, T=15000)
                if g_inner is None:
                    break
        else:
            g_inner, M_lp = solve_inner_mw(a, T=15000)
            if g_inner is None:
                break

        best_a_trial, best_t_val, best_obj_trial = line_search(a, g_inner, t_init=best_t)

        it += 1
        if best_obj_trial < best_obj - 1e-14:
            best_obj = best_obj_trial
            best_a = best_a_trial.copy()
            a = best_a
            best_t = best_t_val
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 12:
                # Structured escape: dual-informed perturbation
                # Use the current LP solution to guide the perturbation
                noise_scale = min(0.25, 0.04 + 0.03 * (stall // 12))
                # Blend with a fresh random direction weighted toward the LP solution
                a_pert = best_a * np.exp(rng.normal(0, noise_scale, n))
                # Also try blending with the LP direction
                a_pert2 = 0.7 * best_a + 0.3 * g_inner
                a_pert2 = np.maximum(a_pert2, 0)
                s2 = a_pert2.sum()
                if s2 > 1e-10:
                    a_pert2 /= s2

                obj_pert = compute_obj(a_pert)
                obj_pert2 = compute_obj(a_pert2)

                if obj_pert2 < obj_pert:
                    a = a_pert2
                else:
                    a = a_pert / max(a_pert.sum(), 1e-10)

                obj_now = compute_obj(a)
                if obj_now < best_obj:
                    best_obj = obj_now
                    best_a = a.copy()
                stall = 0
                best_t = 0.3 + 0.1 * rng.random()

    return best_a


def smooth_optimize(a, time_budget=60.0, T_smooth=40.0, verbose=False):
    """L-BFGS on log-sum-exp smoothed objective."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_grad(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = compute_conv(x)
        max_c = conv.max()
        z = T_smooth * (conv - max_c)
        z -= z.max()  # numerical stability
        e = np.exp(z)
        sum_e = e.sum()
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T_smooth)
        w = e / sum_e
        # Gradient
        g = np.zeros(n)
        # Only compute for significant weights
        thresh = 1e-7
        for k in range(len(conv)):
            if w[k] > thresh:
                g += w[k] * grad_conv_at_peak(x, k)
        g *= 2.0 * n
        return obj_val, g

    try:
        max_iter = min(300, int(time_budget * 10))
        res = sp_minimize(obj_grad, a, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 3, 'disp': False})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                if verbose:
                    print(f"    Smooth: obj={best_obj:.8f}")
    except Exception:
        pass

    return best_a


def nesterov_subgradient(a, time_budget=60.0, K=5, verbose=False):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    a_curr = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 100

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        spread = (peak_vals[0] - peak_vals[-1]) / max_val
        temp = 3.0 / (1.0 + spread * 10.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_peak(y, int(k))

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.4)

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

        if verbose and it % 300 == 0:
            print(f"    NS it{it}: obj={best_obj:.8f}, t_k={t_k:.1f}, el={time.monotonic()-start:.1f}s")

    return best_a


def mirror_descent(a, time_budget=60.0, verbose=False):
    """Mirror descent with multi-temperature softmax subgradient."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.5 / np.sqrt(n)
    min_lr = 1e-5
    stall = 0
    momentum = 0.3
    v = np.zeros(n)
    rng = np.random.default_rng(999 + n)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        if max_val < 1e-15:
            break

        # Multi-temperature softmax weights
        temps = [2.0, 5.0, 12.0]
        all_w = np.zeros(len(conv))
        for temp in temps:
            z = (conv - max_val) / (temp * max_val + 1e-15)
            w = np.exp(z)
            w /= w.sum()
            all_w += w
        all_w /= len(temps)

        # Use top 30% of weights
        threshold = np.percentile(all_w, 70)
        active = all_w > threshold
        if not active.any():
            active[np.argmax(conv)] = True
        active_w = all_w[active]
        active_w /= active_w.sum()
        active_idx = np.where(active)[0]

        g = np.zeros(n)
        for w, k in zip(active_w, active_idx):
            if w > 1e-10:
                g += w * grad_conv_at_peak(a, int(k))

        v = momentum * v + g
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = max(min_lr, lr * 0.998)
            momentum = min(0.7, momentum + 0.001)
        else:
            stall += 1
            if stall > 80:
                a = best_a * np.exp(rng.normal(0, 0.12, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                v = np.zeros(n)
                lr = max(min_lr, lr * 0.4)
                momentum = 0.3
                stall = 0
                best_obj = compute_obj(a)
                best_a = a.copy()
            else:
                lr = max(min_lr, lr * 0.98)

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / max(a.sum(), 1e-15)
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 5, n // 5]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / max(a.sum(), 1e-15)
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (1.0 + 0.5 * x**2)
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 7, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.25 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 + 0.1
        a = a / a.sum()
    elif mode == 7:
        # Two-peaked
        a = np.exp(-((np.arange(n) - n*0.2)**2) / (n*0.05) + 3) 
        a += np.exp(-((np.arange(n) - n*0.8)**2) / (n*0.05) + 3)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid optimization combining LP alternating (primary),
    Nesterov subgradient, mirror descent, smooth L-BFGS, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Quick Nesterov (warm the ball) ~60s ===
    print(f"\n=== Phase 1: Nesterov Warmup (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.06, 60.0)
    if p1_budget > 15:
        a_ns = nesterov_subgradient(best_seq, time_budget=p1_budget, K=5, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_ns:.8f} (no improvement)")

    # === Phase 2: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP Alternating (main) ===")
    remaining = deadline - time.monotonic()
    p2_total = min(remaining * 0.5, 600.0)

    # Multiple rounds with perturbation
    n_rounds = 3
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p2_total / max(1, (n_rounds - rnd)), remaining * 0.55)
        if alloc < 20:
            break

        if rnd > 0:
            # Perturb to escape potential local minima
            scale = 0.02 * rnd
            a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            s_pert = a_pert.sum()
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert
            else:
                a_pert = best_seq.copy()
        else:
            a_pert = best_seq.copy()

        a_lp = lp_alternating(a_pert, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement, keeping best)")

    # === Phase 3: Mirror Descent ~80s ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 80.0)
    if p3_budget > 15:
        a_md = mirror_descent(best_seq, time_budget=p3_budget, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Mirror: obj={best_score:.8f} **")

    # === Phase 4: Smooth L-BFGS polish ===
    print(f"\n=== Phase 4: Smooth L-BFGS Polish ===")
    remaining = deadline - time.monotonic()
    p4_total = min(remaining * 0.12, 120.0)

    for T_val in [30.0, 60.0, 120.0]:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        t_budget = min(p4_total / 3.0, remaining * 0.4)
        if t_budget < 10:
            continue
        a_sm = smooth_optimize(best_seq, time_budget=t_budget, T_smooth=T_val, verbose=True)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Smooth T={T_val}: obj={best_score:.8f} **")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 100),
        max(200, n_current - 50),
        n_current + 50,
        n_current + 100,
        n_current + 200,
        400, 500, 600, 700, 800, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= 5:
            break
        if n_try == n_current:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        # Quick optimize: Nesterov + LP
        alloc_ns = min(remaining * 0.08, 30.0)
        if alloc_ns > 10:
            a_try = nesterov_subgradient(a_try, time_budget=alloc_ns, K=3)

        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.08, 40.0)
        if alloc_lp > 10:
            a_try = lp_alternating(a_try, time_budget=alloc_lp, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        s_r = a_r.sum()
        if s_r > 1e-10:
            a_r = a_r / s_r

        restart_budget = min(remaining * 0.12, 50.0)
        a_opt = lp_alternating(a_r, time_budget=restart_budget, verbose=False)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        lp_alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_alternating(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        ns_alloc = min(remaining * 0.2, 40.0)
        a_ns = nesterov_subgradient(best_seq, time_budget=ns_alloc, K=7)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Final NS: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        sm_alloc = min(remaining * 0.2, 30.0)
        a_sm = smooth_optimize(best_seq, time_budget=sm_alloc, T_smooth=80.0)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Final Smooth: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 25.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
