def compute_conv(a):

    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

    s = a.sum()
    if s < 0.01:

    a = a / s
    c = compute_conv(a)
    return 2.0 * n * c.max()

    w = w / w_sum

- There exist dual variables p_k >= 0 and multiplier λ such that:
  2 * sum_k p_k * a[k-i] = λ + μ_i for each i
  where μ_i >= 0 and μ_i * a_i = 0 (complementary slackness)
- sum_k p_k * (c[k] - M) = 0 (complementary slackness)
- sum_k p_k = 1 (normalization)

If we can find p and a that satisfy these, we have a certificate of optimality.

The condition 2 * sum_k p_k * a[k-i] = λ + μ_i means that the "weighted cross-correlation" of a with p is constant (plus a non-negative correction at the boundary). In the interior (where a_i > 0), this means:
sum_k p_k * a[k-i] = λ/2 for all i with a_i > 0.

This is a system of equations! If we know which a_i are positive (the support), we can solve for a and λ.

This gives us a way to verify (and potentially find) the optimum:
1. Guess the support S = {i : a_i > 0}
2. Solve the system: sum_k p_k * a[k-i] = λ/2 for i ∈ S, sum_{i∈S} a_i = 1
3. Check if p_k >= 0 and c[k] <= M for all k
4. If not, adjust the guess

This is similar to the "active set" method. But finding the right support is the hard part.

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def cross_corr(a, w):
    """
    Compute r[i] = sum_j a[j] * w[i+j] for i=0..n-1.
    a has length n, w has length 2n-1.
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    w_rev = w[::-1]
    size = 1
    while size < n + len(w) - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w_rev, size)
    cross = irfft(fa * fw, size)[:n + len(w) - 1]
    return cross[n - 1:2 * n - 1][::-1]


def smooth_obj_grad(a, T):
    """
    Compute smooth objective and gradient for f_T(a) = (2n/T)*log(sum exp(T*c_k))
    where c = a*a and a is on the simplex. Returns (obj, grad) where grad is
    the gradient w.r.t. a (not projected).
    """
    n = len(a)
    a = np.maximum(a, 1e-15)
    s = a.sum()
    a = a / s

    c = compute_conv(a)
    c_max = c.max()

    z = T * (c - c_max)
    w = np.exp(z)
    w_sum = w.sum()
    w = w / w_sum

    log_sum_exp = T * c_max + np.log(w_sum)
    obj = 2.0 * n * log_sum_exp / T

    grad_a = 2.0 * n * cross_corr(a, w)
    return obj, grad_a


def softmax_obj_grad(z, T):
    """
    Objective and gradient in softmax parameterization.
    z is length n, a = softmax(z). Returns (obj, grad_z).
    """
    n = len(z)
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()

    obj, grad_a = smooth_obj_grad(a, T)

    # Chain rule: grad_z[j] = a[j] * (grad_a[j] - dot(grad_a, a))
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a)
    return obj, grad_z


def lBFGS_smooth(a_init, T_values, max_iter_per_T=500, time_budget=100.0):

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_b(a, M_target=None):
    """
    Solve LP: min M s.t. (A(a)@b)[k] <= M, sum(b)=1, b>=0.
    Or if M_target given: max sum(b) s.t. (A(a)@b)[k] <= M_target, b>=0.
    Returns normalized b or None.
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if M_target is None:
        # min M s.t. A@b <= M*1, sum(b)=1, b>=0
        if HAVE_CVXPY:
            b_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [A @ b_var <= M_var, cp.sum(b_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
                try:
                    prob.solve(solver=solver, verbose=False)
                    if b_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        b_val = np.maximum(b_var.value, 0)
                        s = float(np.sum(b_val))
                        if s > 1e-10:
                            return b_val / s
                except Exception:
                    continue

        # scipy fallback
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            b_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(b_val))
            if s > 1e-10:
                return b_val / s
    else:
        # max sum(b) s.t. A@b <= M_target, b >= 0
        if HAVE_CVXPY:
            b_var = cp.Variable(n, nonneg=True)
            constraints = [A @ b_var <= M_target]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
                try:
                    prob.solve(solver=solver, verbose=False)
                    if b_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        b_val = np.maximum(b_var.value, 0)
                        s = float(np.sum(b_val))
                        if s > 1e-10:
                            return b_val / s
                except Exception:
                    continue

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_target)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            b_val = np.maximum(res.x, 0)
            s = float(np.sum(b_val))
            if s > 1e-10:
                return b_val / s
    return None

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_smooth(a, time_budget=60.0, K=7):

def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # U-shaped
        x = np.linspace(0, 1, n)
        a = 1.0 / (x * (1 - x) + 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Triangular
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like
        x = np.linspace(0.001, 0.999, n)
        a = 1.0 / np.sqrt(x * (1 - x))
        a = a / a.sum()
    elif mode == 5:
        # Cosine bump
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Random Fourier
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Two bumps
        c1, c2 = n // 3, 2 * n // 3
        w = n // 8
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def kenkarri_polish(a, time_budget=30.0):
    """
    Fine-grained coordinate descent: move small mass from peak-contributing
    positions to non-contributing positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Find contributing pairs
        contrib = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib.append((a[i] * a[j], i, j))
        contrib.sort(reverse=True)

        # Find non-contributing positions
        non_contrib = []
        for i in range(n):
            j = k_star - i
            if j < 0 or j >= n:
                non_contrib.append(i)

        if not non_contrib:
            break

        improved = False
        for (val, i, j) in contrib[:20]:
            if val < 1e-14:
                break
            for target in non_contrib[:10]:
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    src = i if a[i] >= a[j] else j
                    delta = frac * a[src]
                    if delta < 1e-15:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
            if improved:
                break

        if not improved:
            break

    return best_a

            print(f"  L-BFGS high-T: obj={best_score:.8f} **")
        else:
            print(f"  L-BFGS high-T: obj={obj:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current - 100, n_current - 50, n_current,
        n_current + 50, n_current + 100, n_current + 200,
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1500
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= 5:
            break
        if n_try == n_current:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        # Quick L-BFGS
        alloc = min(remaining * 0.12, 50.0)
        if alloc > 15:
            T_vals = [50.0, 100.0, 300.0]
            a_try = lBFGS_smooth(a_try, T_vals, max_iter_per_T=100, time_budget=alloc)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r + 1) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # Quick optimization
        alloc = min(remaining * 0.12, 50.0)
        if alloc > 15:
            T_vals = [20.0, 100.0, 300.0]
            a_opt = lBFGS_smooth(a_r, T_vals, max_iter_per_T=100, time_budget=alloc * 0.6)
            remaining = deadline - time.monotonic()
            alloc_lp = min(remaining * 0.08, 30.0)
            if alloc_lp > 10:
                a_opt = lp_alternating_v2(a_opt, time_budget=alloc_lp)

        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 40:
        lp_alloc = min(remaining * 0.25, 100.0)
        a_lp = lp_alternating_v2(best_seq, time_budget=lp_alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Coordinate refinement
    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd_alloc = min(remaining * 0.2, 30.0)
        a_cd = kenkarri_polish(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more L-BFGS
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = lBFGS_smooth(best_seq, [200.0, 500.0, 1000.0],
                               max_iter_per_T=100, time_budget=min(remaining * 0.3, 40.0))
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final L-BFGS: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternating_v2(best_seq, time_budget=min(remaining * 0.4, 30.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
