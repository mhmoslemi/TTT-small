import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(v, 0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_set = np.where(cond)[0]
    if len(rho_set) == 0:
        return np.ones(n) / n
    rho = rho_set[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of log-sum-exp smoothed max of convolution values.
    F(a) = (1/tau) * log(sum_k exp(tau * conv_k(a)))
    grad F(i) = 2 * sum_k p_k * a[k-i] (with appropriate range)
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    P = rfft(p, N)
    A = rfft(a, N)
    # Cross-correlation: g[i] = sum_j p[j] * a[j-i]... 
    # Actually we need: g[i] = 2 * sum_{j: i+j < 2n-1} p[i+j] * a[j]... 
    # Let me use the full FFT approach
    Pfull = np.fft.fft(p, N)
    Afull = np.fft.fft(a, N)
    g_full = np.fft.ifft(Pfull * np.conj(Afull)).real
    g = 2.0 * g_full[:n]
    
    return g, conv


def nesterov_descent(a, time_budget=100.0, tau_init=20.0, tau_end=500.0):
    """
    Nesterov accelerated gradient descent on smoothed objective over simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.9, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        # Gradient at y
        g, conv = smoothed_gradient_fft(y, tau)
        
        # BB step size every other iteration
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.7 * lr + 0.3 * min(max(bb_lr, 1e-8), 5.0 / np.sqrt(n))
        
        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9999)
        
        # Reset if stuck
        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns the optimal g and M.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

def subgradient_refine(a, time_budget=60.0, K=8):
    """
    Subgradient method with top-K peaks, momentum, and adaptive step.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.5 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        
        # Weight by peak value (exponential weighting)
        weights = np.exp((peak_vals - max_pv) * 10.0 / max_pv)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += 2.0 * w * a[j_start:j_end + 1][::-1]
        
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                lr *= 0.5
                v *= 0.2
                a = best_a.copy()
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.998
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.arange(n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Asymmetric: concentrated on left
        a = np.exp(-x / (n * 0.3))
        a = a / a.sum()
    elif mode == 3:
        # Asymmetric: concentrated on right
        a = np.exp(-(n - 1 - x) / (n * 0.3))
        a = a / a.sum()
    elif mode == 4:
        # Triangular (peak at center)
        center = n / 2
        a = 1.0 - np.abs(x - center) / (n / 2 + 1)
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 5:
        # Sine-based
        a = 0.5 + 0.5 * np.sin(np.pi * x / n)
        a = a / a.sum()
    elif mode == 6:
        # Two blobs (asymmetric positions)
        c1, c2 = int(n * 0.2), int(n * 0.7)
        a = np.exp(-((x - c1) ** 2) / (2 * (n * 0.1) ** 2)) + \
            np.exp(-((x - c2) ** 2) / (2 * (n * 0.15) ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Power law (asymmetric)
        a = (x + 1) ** (-0.5)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def peak_localization_refine(a, time_budget=60.0):

        pressure = np.zeros(n)

            for i in range(i_start, i_end + 1):

                pressure[i] += c
                if i != j:
                    pressure[j] += c

        pressure = pressure / (pressure.sum() + 1e-15)
        
        # Transfer mass from high-pressure to low-pressure
