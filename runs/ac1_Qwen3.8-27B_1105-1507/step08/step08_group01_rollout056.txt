Actually, let me reconsider the multiplicative update approach more carefully.

We want to minimize `f(a) = max_k (a*a)_k` subject to `sum(a) = 1, a >= 0`.

The "pressure" on coordinate j is `p_j = 2 * sum_k w_k * a_{k-j}` where w_k are weights on the peaks.

The idea is: if p_j is high, coordinate j is "pressing" on the peaks and should be decreased. If p_j is low, it should be increased.

A natural update is: `a_j <- a_j * (c / p_j)^alpha` for some constant c and exponent alpha.

The constant c should be chosen so that the new sum is 1:
`sum_j a_j * (c / p_j)^alpha = 1`
`c^alpha * sum_j a_j / p_j^alpha = 1`
`c = (sum_j a_j / p_j^alpha)^(-1/alpha)`

This is similar to the "mirror descent" or "exponential weights" update.

In the limit of small alpha, this becomes:
`a_j <- a_j * (1 - alpha * (log(p_j) - log(c)))`
`a_j <- a_j * (1 - alpha * (log(p_j) - mean_weighted(log(p))))`

Which is gradient descent in log-space: `log(a_j) <- log(a_j) - alpha * (log(p_j) - mean)`.

This is equivalent to: `log(a_j) <- log(a_j) - alpha * grad_log_j` where `grad_log_j` is the gradient of the objective in the log-parameterization.

The log-parameterization is nice because it automatically handles the non-negativity constraint (as long as we stay in the domain).

```python
def multiplicative_update(a, conv, T=0.001, alpha=0.1, top_k=None):
    """
    Multiplicative (log-space) update to reduce max convolution.
    a: current sequence (sum=1, non-negative)
    conv: convolution of a with itself
    T: temperature for softmax
    alpha: step size
    top_k: if set, use only top-k peaks
    """
    n = len(a)
    max_c = np.max(conv)
    
    if top_k is not None:
        # Use top-k peaks with equal weight
        idx = np.argpartition(-conv, top_k)[:top_k]
        w = np.zeros(len(conv))
        w[idx] = 1.0 / top_k
    else:
        # Softmax weighting
        shifted = (conv - max_c) / T
        w = np.exp(np.clip(shifted, -50, 0))
        w /= np.sum(w)
    
    # Pressure: p_j = 2 * sum_k w[k] * a[k-j]
    p = np.zeros(n)
    for j in range(n):
        k_lo = max(0, j)
        k_hi = min(len(w)-1, j+n-1)
        if k_lo <= k_hi:
            p[j] = 2.0 * np.dot(w[k_lo:k_hi+1], a[k_lo-j:k_hi-j+1])
    
    # Update: a_j <- a_j * (c / p_j)^alpha
    # c = (sum a_j / p_j^alpha)^(-1/alpha)
    # Handle p_j = 0: keep a_j unchanged (or set to small value)
    mask = p > 1e-15
    if not np.all(mask):
        # For zero-pressure coordinates, set pressure to a small value
        p = np.where(mask, p, np.min(p[mask]) if np.any(mask) else 1.0)
    
    log_a = np.log(a + 1e-30)  # avoid log(0)
    log_p = np.log(p)
    
    # log(a_new_j) = log(a_j) + alpha * (log(c) - log(p_j))
    # = log(a_j) - alpha * log(p_j) + alpha * log(c)
    # We want sum(exp(log(a_new))) = 1
    # sum(exp(log(a_j) - alpha * log(p_j))) * exp(alpha * log(c)) = 1
    # exp(alpha * log(c)) = 1 / sum(exp(log(a_j) - alpha * log(p_j)))
    # So: log(a_new_j) = log(a_j) - alpha * log(p_j) - log(sum(exp(log(a) - alpha*log(p))))
    
    log_weights = log_a - alpha * log_p
    log_weights -= np.max(log_weights)  # numerical stability
    log_sum = np.log(np.sum(np.exp(log_weights)))
    log_a_new = log_weights - log_sum
    
    a_new = np.exp(log_a_new)
    # Ensure non-negative
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-15:
        a_new /= s
    return a_new
```

"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: smooth max gradient descent with Nesterov, LP alternating with active sets,
multiplicative pressure updates, coordinate descent polish, and multi-start/multi-n exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp(a, active_indices=None):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    try:
        import cvxpy as cp
    except ImportError:
        return _solve_lp_scipy(a, active_indices)

    n = len(a)
    a = a / np.sum(a)
    A_full = build_conv_matrix(a)

    if active_indices is not None and len(active_indices) < 2 * n - 1:
        A = A_full[active_indices, :]
    else:
        A = A_full

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = ['CLARABEL', 'MOSEK', 'ECOS', 'SCS', 'HIGHS']
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Verify against all constraints
                    conv_check = A_full @ g_val
                    true_M = float(np.max(conv_check))
                    return g_val, true_M
        except Exception:
            continue
    return _solve_lp_scipy(a, active_indices)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_val = g_val / s
            return g_val, float(res.x[n])
    return None, None


def convex_line_search(a, g, n_evals=12):
    """Find optimal t in [0,1] for a_new = (1-t)*a + t*g using convex structure."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    a_new = a_new / s
    return t_opt, a_new


def compute_pressure(w, a):
    """
    Compute pressure vector p_j = 2 * sum_k w[k] * a[k-j].
    w: weights over convolution lags (length 2n-1)
    a: sequence (length n)
    Returns pressure of length n.
    """
    n = len(a)
    nw = len(w)
    p = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(nw - 1, j + n - 1)
        if k_lo <= k_hi:
            p[j] = 2.0 * np.dot(w[k_lo:k_hi + 1], a[0:k_hi - k_lo + 1])
    return p


def multiplicative_pressure_update(a, conv, T=0.0005, alpha=0.05, top_k=None):
    """
    Multiplicative (log-space) update to reduce max convolution peak.
    """
    n = len(a)
    max_c = np.max(conv)

    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(-conv, top_k)[:top_k]
        w = np.zeros(len(conv))
        w[idx] = 1.0 / top_k
    else:
        shifted = (conv - max_c) / T
        w = np.exp(np.clip(shifted, -50, 0))
        sw = np.sum(w)
        if sw < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= sw

    p = compute_pressure(w, a)

    # Handle zero pressure
    p_min = np.min(p[p > 0]) if np.any(p > 0) else 1.0
    p = np.where(p > 1e-15, p, p_min * 1e-3)

    log_a = np.log(np.maximum(a, 1e-30))
    log_p = np.log(p)

    log_weights = log_a - alpha * log_p
    log_weights -= np.max(log_weights)
    log_sum = np.log(np.sum(np.exp(log_weights)))
    log_a_new = log_weights - log_sum

    a_new = np.exp(log_a_new)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-15:
        a_new /= s
    return a_new


def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating with convex line search and adaptive active set."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    active_set = None

    max_iters = 300
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Determine active set
        conv_cur = compute_conv(a)
        max_conv = np.max(conv_cur)
        if active_set is None or it % 5 == 0:
            threshold = max_conv * 0.95
            active_set = np.where(conv_cur >= threshold)[0]
            if len(active_set) > 4 * n:
                active_set = np.argsort(-conv_cur)[:4 * n]

        # Solve LP
        g_lp, M_lp = solve_lp(a, active_set)
        if g_lp is None:
            active_set = None
            g_lp, M_lp = solve_lp(a, None)
            if g_lp is None:
                break

        # Convex line search
        t_opt, a_new = convex_line_search(a, g_lp)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 10 == 0:
                print(f"          LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, time={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 15:
                # Perturb to escape
                rng = np.random.default_rng((it * 7919 + n) % (2**31))
                scale = 0.01 + 0.01 * (stall // 10)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a = a_pert / s_p
                active_set = None
                stall = 0
                if stall > 50:
                    break

    return best_a


def smooth_max_descent(a, time_budget=60.0, T_init=0.0003, T_min=0.00002):
    """Smooth max gradient descent with Nesterov momentum and adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.005
    momentum = 0.8
    v = np.zeros(n)
    stall = 0

    max_iters = 3000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 0)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        if total < 1e-30:
            break
        weights = exp_vals / total

        # Gradient: grad[i] = 2 * sum_j weights[j] * a[j-i]
        grad = np.zeros(n)
        for j in range(n):
            k_lo = max(0, j - n + 1)
            k_hi = min(2 * n - 2, j)
            if k_lo <= k_hi:
                # weights[k] * a[k - (j-k)]... let me redo
                pass

        # Direct computation of subgradient
        for i in range(n):
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            if k_lo <= k_hi:
                grad[i] = 2.0 * np.dot(weights[k_lo:k_hi + 1], a[0:k_hi - k_lo + 1])

        # Center gradient (remove mean component for simplex projection)
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 50 == 0 and lr < 0.02:
                lr *= 1.05
        else:
            stall += 1
            if stall > 30:
                lr *= 0.7
                if lr < 1e-6:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0

        if it > 50:
            T = max(T_min, T_init * (0.997 ** it))

    return best_a


def multiplicative_descent(a, time_budget=60.0, verbose=False):
    """
    Multiplicative pressure-based descent.
    Alternates between stiff (top_k=1) and soft (top_k larger) updates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    alpha = 0.03
    stall = 0

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)

        # Adaptive top_k: start with 1, grow as we converge
        if it < 100:
            top_k = 1
            T = 0.0001
        elif it < 300:
            top_k = 3
            T = 0.0003
        elif it < 600:
            top_k = 10
            T = 0.001
        else:
            top_k = 30
            T = 0.003

        a_new = multiplicative_pressure_update(a, conv, T=T, alpha=alpha, top_k=top_k)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 200 == 0:
                print(f"          MP it{it}: obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 50:
                alpha *= 0.8
                if alpha < 0.001:
                    break
                a = best_a.copy()
                stall = 0

    return best_a


def coordinate_descent_polish(a, time_budget=60.0, delta=1e-4, verbose=False):
    """Coordinate descent: try tiny perturbations of individual coordinates."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    pass_count = 0
    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count * 31 + n % 10000)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            for delta_try in [delta, delta * 0.5, delta * 2]:
                a_try = a.copy()
                a_try[i] += delta_try
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 0.01:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                continue

            for delta_try in [delta, delta * 0.5]:
                if a[i] < delta_try * 0.3:
                    continue
                a_try = a.copy()
                a_try[i] -= delta_try
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 0.01:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if verbose and pass_count % 3 == 0:
            print(f"          CD pass {pass_count}: obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")

        if not improved and pass_count > 3:
            delta *= 0.6
            if delta < 1e-7:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def optimize_single(a, time_budget=100.0, verbose=False):
    """
    Full optimization pipeline for a single starting point.
    Combines multiplicative pressure, smooth max, LP alternating, and CD polish.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Multiplicative pressure (structure-preserving)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t1 = min(0.25 * time_budget, remaining * 0.3)
        a_p1 = multiplicative_pressure_update  # just check it exists
        a_p1 = multiplicative_descent(a, time_budget=t1)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_obj:
            best_obj = obj_p1
            best_a = a_p1
            if verbose:
                print(f"          MP: obj={best_obj:.8f}")

    # Phase 2: Smooth max gradient descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        t2 = min(0.25 * time_budget, remaining * 0.3)
        a_p2 = smooth_max_descent(best_a, time_budget=t2)
        obj_p2 = compute_obj(a_p2)
        if obj_p2 < best_obj:
            best_obj = obj_p2
            best_a = a_p2
            if verbose:
                print(f"          SM: obj={best_obj:.8f}")

    # Phase 3: LP alternating
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 30:
        t3 = min(0.3 * time_budget, remaining * 0.4)
        a_p3 = lp_alternating(best_a, time_budget=t3, verbose=verbose)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            if verbose:
                print(f"          LP: obj={best_obj:.8f}")

    # Phase 4: Another round of multiplicative (finer)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        t4 = min(0.15 * time_budget, remaining * 0.3)
        a_p4 = multiplicative_descent(best_a, time_budget=t4)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_obj:
            best_obj = obj_p4
            best_a = a_p4

    # Phase 5: CD polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t5 = min(0.15 * time_budget, remaining * 0.5)
        a_p5 = coordinate_descent_polish(best_a, time_budget=t5, delta=1e-4)
        obj_p5 = compute_obj(a_p5)
        if obj_p5 < best_obj:
            best_obj = obj_p5
            best_a = a_p5

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start with various initializations, then intensive optimization,
    multi-n exploration, and final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0

    rng = np.random.default_rng(seed)

    best_a = None
    best_obj = np.inf

    # --- Get starting points ---
    starts = []

    # 1. Parent
    n_default = 800
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        if parent.size > 0 and np.isfinite(parent).all():
            sp = float(np.sum(parent))
            if sp > 0.01:
                parent = parent / sp
                starts.append(('parent', parent))
                n_default = len(parent)
    except Exception:
        pass

    # 2. Uniform
    starts.append(('uniform', np.ones(n_default) / n_default))

    # 3. Cosine bell
    x = np.linspace(0, 2 * np.pi, n_default)
    cos_bell = (1 + np.cos(x)) / 2
    starts.append(('cosine', cos_bell / np.sum(cos_bell)))

    # 4. Exponential decay pair
    exp_a = np.exp(-0.5 * np.linspace(0, 1, n_default) ** 2)
    starts.append(('gaussian', exp_a / np.sum(exp_a)))

    # 5. Random (exponential distribution)
    rand = rng.exponential(1.0, n_default)
    starts.append(('random', rand / np.sum(rand)))

    # 6. Parent with perturbation (if parent exists)
    if len(starts) > 1 and starts[0][0] == 'parent':
        pert = starts[0][1] * np.exp(rng.normal(0, 0.02, n_default))
        pert = np.maximum(pert, 0)
        starts.append(('parent_pert', pert / np.sum(pert)))

    # --- Phase 1: Multi-start exploration ---
    print(f"=== Phase 1: Multi-start exploration ({len(starts)} starts) ===")
    per_start_budget = min(80.0, (deadline - time.monotonic()) / max(1, len(starts)) * 0.7)

    for name, a_init in starts:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        t_bud = min(per_start_budget, remaining * 0.15)
        a_opt = optimize_single(a_init.copy(), time_budget=t_bud, verbose=False)
        obj = compute_obj(a_opt)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            print(f"  {name} (n={len(a_init)}): obj={obj:.8f} [BEST]")
        else:
            print(f"  {name} (n={len(a_init)}): obj={obj:.8f}")

    if best_a is None:
        best_a = np.ones(n_default) / n_default
        best_obj = compute_obj(best_a)
        print(f"  Default init: obj={best_obj:.8f}")

    print(f"\n  After Phase 1: best={best_obj:.8f} (n={len(best_a)})")

    # --- Phase 2: Intensive optimization on best ---
    print(f"\n=== Phase 2: Intensive optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t2 = min(350.0, remaining * 0.45)
        a_int = optimize_single(best_a.copy(), time_budget=t2, verbose=True)
        obj_int = compute_obj(a_int)
        if obj_int < best_obj:
            best_obj = obj_int
            best_a = a_int
            print(f"  Phase 2: obj={best_obj:.8f}")
        else:
            print(f"  Phase 2: obj={obj_int:.8f} (no improvement)")

    # --- Phase 3: Targeted LP refinement ---
    print(f"\n=== Phase 3: Targeted LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        t3 = min(150.0, remaining * 0.3)
        a_lp = lp_alternating(best_a.copy(), time_budget=t3, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp
            print(f"  Phase 3 LP: obj={best_obj:.8f}")
        else:
            print(f"  Phase 3 LP: obj={obj_lp:.8f} (no improvement)")

    # --- Phase 4: Multi-n exploration ---
    print(f"\n=== Phase 4: Multi-n exploration ===")
    n_best = len(best_a)
    n_options = sorted(set([
        max(400, n_best - 75), max(400, n_best - 50), max(400, n_best - 25),

        650, 700, 750, 800, 850, 900, 950, 1000, 1100
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1400]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        if n_try == n_best:
            a_try = best_a.copy()
        else:
            a_try = resample_sequence(best_a, n_try)

        t_bud = min(60.0, remaining * 0.12)
        a_opt = optimize_single(a_try, time_budget=t_bud)
        obj_try = compute_obj(a_opt)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # --- Phase 5: Final polish ---
    print(f"\n=== Phase 5: Final polish ===")
    remaining = deadline - time.monotonic()

            if obj_f1 < best_obj:
                best_obj = obj_f1
                best_a = a_f1
                print(f"  Final LP: obj={best_obj:.8f}")

        # MP polish
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_mp = min(40.0, remaining * 0.25)
            a_f2 = multiplicative_descent(best_a.copy(), time_budget=t_mp)
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < best_obj:
                best_obj = obj_f2
                best_a = a_f2
                print(f"  Final MP: obj={best_obj:.8f}")

        # CD polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_cd = min(50.0, remaining * 0.4)
            a_f3 = coordinate_descent_polish(best_a.copy(), time_budget=t_cd, delta=5e-5)
            obj_f3 = compute_obj(a_f3)
            if obj_f3 < best_obj:
                best_obj = obj_f3
                best_a = a_f3
                print(f"  Final CD: obj={best_obj:.8f}")

    return result
