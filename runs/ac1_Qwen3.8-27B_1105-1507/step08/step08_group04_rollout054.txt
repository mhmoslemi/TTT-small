```python
"""
Fast alternating optimizer using top-K LP approximations.
Key idea: Instead of solving the full LP (2n-1 constraints), solve a reduced LP
with only the top-K peak constraints. This allows 10-100x more iterations,
leading to better convergence within the time budget.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_topk_lp(a, K, solver='highs'):
    """
    Solve: min M s.t. conv(a,g)[k] <= M for k in top_K, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a = a / np.sum(a)
    conv_aa = compute_conv(a)
    
    # Get top K peak positions
    if K >= len(conv_aa):
        top_k_idx = np.arange(len(conv_aa))
    else:
        top_k_idx = np.argsort(conv_aa)[::-1][:K]
    
    # Build A_K matrix: A_K[k_idx, j] = a[k_idx - j] if 0 <= k_idx - j < n
    A_K = np.zeros((K, n))
    for i, k in enumerate(top_k_idx):
        k = int(k)
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                A_K[i, j] = a[m]
    
    # LP: min M
    # s.t. A_K @ g - M <= 0  (K constraints)
    #      sum(g) = 1
    #      g >= 0, M >= 0
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    # Inequality constraints: A_K @ g - M <= 0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = A_K
    A_ub[:, n] = -1.0
    b_ub = np.zeros(K)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception as e:
        pass
    
    return None, None
```

```python
def build_topk_matrix(a, top_k_idx):
    """Build A_K matrix where A_K[i,j] = a[top_k_idx[i] - j] if valid."""
    n = len(a)
    K = len(top_k_idx)
    A_K = np.zeros((K, n))
    for i in range(K):

        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:

        elapsed = time.monotonic() - start
        progress = elapsed / time_budget
        K = int(K_start + (K_end - K_start) * min(1, progress * 1.5))
```

```python
"""
Fast alternating optimizer using top-K peak LP approximations.
Key insight: Instead of solving the full LP (2n-1 constraints) each iteration,
solve a reduced LP with only the top-K convolution peak constraints. This allows
10-100x more iterations, leading to significantly better convergence within
the 1500s time budget. Combined with periodic full-LP refinement, multi-n
exploration, and diverse restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_topk_matrix(a, top_k_idx):
    """Build A_K where A_K[i,j] = a[top_k_idx[i] - j] if 0<=top_k_idx[i]-j<n."""
    n = len(a)
    K = len(top_k_idx)
    A_K = np.zeros((K, n))
    for i in range(K):
        k = int(top_k_idx[i])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A_K[i, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A_K


def solve_topk_lp(a, top_k_idx):
    """Solve min M s.t. conv(a,g)[k]<=M for k in top_k_idx, sum(g)=1, g>=0."""
    n = len(a)
    a = a / float(np.sum(a))
    K = len(top_k_idx)
    A_K = build_topk_matrix(a, top_k_idx)

    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = A_K
    A_ub[:, n] = -1.0
    b_ub = np.zeros(K)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_full_lp(a):
    """Solve full LP: min M s.t. conv(a,g)[k]<=M for ALL k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / float(np.sum(a))
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(2 * n - 2, j + n - 1)
        A[j:k_end + 1, j] = a[:k_end - j + 1]

    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    return None, None


def fast_alternate(a, time_budget, K_start=20, K_end=80, t_base=0.3, verbose=True):
    """Fast alternating with top-K LP, adaptive K and t."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    iteration = 0
    t_current = t_base

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1))
        K = int(K_start + (K_end - K_start) * min(1.0, progress * 2.0))
        K = max(10, min(K, 2 * n - 1))

        conv = compute_conv(a)
        if K < len(conv):
            top_k_idx = np.argsort(conv)[::-1][:K]
        else:
            top_k_idx = np.arange(len(conv))

        g, M = solve_topk_lp(a, top_k_idx)
        if g is None:
            break

        best_local_obj = best_obj
        best_local_a = best_a
        best_t = 0.0

        t_candidates = [t_current * 0.5, t_current, t_current * 1.5, t_current * 2.0]
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_t = t

        if best_t > 0:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a.copy()
            stall = 0
            t_current = best_t
            if verbose and iteration % 1000 == 0:
                print(f"    FastAlt: it={iteration}, K={K}, obj={best_obj:.8f}, t={best_t:.3f}, time={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 50:
                t_current = min(t_current * 0.5, 0.05)
            if stall > 200:
                rng = np.random.default_rng(hash((n, stall, iteration)) % (2**31))
                scale = 0.05 + 0.05 * min(stall // 200, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
                t_current = t_base

        iteration += 1

    if verbose:
        print(f"    FastAlt done: it={iteration}, obj={best_obj:.8f}, time={time.monotonic()-start:.0f}s")
    return best_a

def subgradient_polish(a, time_budget, top_k=20):
    """Nesterov-accelerated subgradient with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    lr = 2.0 / n
    x = a.copy()

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(x)
        if top_k >= len(conv):
            k_indices = np.arange(len(conv))
        else:
            k_indices = np.argsort(conv)[::-1][:top_k]

        n_fft = 1
        while n_fft < 4 * n:
            n_fft <<= 1
        w = np.zeros(2 * n - 1)
        for k in k_indices:
            w[int(k)] = 1.0 / len(k_indices)

        A_fft = rfft(x, n_fft)
        W_fft = rfft(w, n_fft)
        R = irfft(np.conj(A_fft) * W_fft, n_fft)
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = R[(n_fft - j) % n_fft]
        grad *= 4.0 * n
        grad -= np.mean(grad)

        momentum = 0.9 * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        t_k = 2.0 / (it + 2.0)
        x = a_new + (1 - t_k) * (a_new - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        if it % 500 == 499:
            lr *= 0.97

    return best_a


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng):
    """Generate diverse structured initialization."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-30 * (x - 0.5) ** 2)
    elif choice == 1:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 2 * x)
    elif choice == 2:
        a = np.exp(-60 * (x - 0.3) ** 2) + np.exp(-60 * (x - 0.7) ** 2)
    elif choice == 3:
        a = np.ones(n)
        for _ in range(4):
            center = rng.integers(0, n)
            width = max(1, rng.integers(n // 10, n // 3))
            for i in range(n):
                d = min(abs(i - center), n - abs(i - center))
                a[i] *= 1.0 + rng.standard_normal() * np.exp(-d / (width * 0.5))
        a = np.maximum(a, 0.01)
    elif choice == 4:
        a = x ** rng.uniform(0.3, 2.0)
    else:
        a = np.exp(-15 * (x - 0.5) ** 2) * (1 + 0.3 * np.cos(2 * np.pi * 5 * x))
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

        a_last = fast_alternate(best_a, time_budget=alloc, K_start=40, K_end=80, t_base=0.15, verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Final fast LP: obj={best_obj:.8f} **")

    # Validation
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")
    return result
```
