import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv / sum^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_response(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                return g_val, M_val
    except Exception:
        pass

    return None, None

def subgradient_polish(a, time_budget=100.0, verbose=False):
    """
    Nesterov-accelerated subgradient method targeting active peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()
    y = a.copy()
    beta = 0.7
    lr = 0.05
    momentum = np.zeros(n)

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(x)
        max_val = float(np.max(conv))
        threshold = max_val * 0.85
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = [int(np.argmax(conv))]
        if len(active) > 40:
            active = np.argsort(conv)[::-1][:40]

        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += a[k - indices]
        grad /= len(active)
        grad_centered = grad - np.mean(grad)

        momentum = beta * momentum - lr * grad_centered
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new

        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        y = y_new
        obj = compute_obj(y)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()

        if it % 300 == 299:
            lr = max(lr * 0.92, 0.001)

    return best_a


def multiplicative_update(a, time_budget=100.0):
    """
    Multiplicative weight / mirror descent update.
    Reduces elements contributing to the peak, increases others.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10).copy()
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.01
    lr_decay = 0.995

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        k_star = int(np.argmax(conv))

        # Compute contribution of each element to the peak
        contrib = np.zeros(n)
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        for i in range(i_min, i_max + 1):
            contrib[i] = a[k_star - i]

        # Multiplicative update: reduce hot elements, boost cold ones
        for i in range(i_min, i_max + 1):
            a[i] *= (1.0 - eta * contrib[i] / max_val)
        for i in range(n):
            if i < i_min or i > i_max:
                a[i] *= (1.0 + eta * 0.5)

        a = np.maximum(a, 1e-10)
        s = float(np.sum(a))
        a = a / s

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        else:
            eta *= lr_decay

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            if s > 0.01:
                parent = parent_arr / s
                print(f"Parent loaded (n={parent.size})")
    except Exception:
        pass

    n_base = 800

    # === Phase 1: Generate and evaluate diverse initial shapes ===
    print(f"\n=== Phase 1: Diverse Initializations (n={n_base}) ===")
    shapes = ['linear', 'sqrt', 'log', 'cubic', 'power03', 'power15',
              'cosine_ramp', 'exp_ramp', 'bimodal', 'uniform']

    shape_results = []
    for kind in shapes:
        a_init = generate_shape(kind, n_base, rng)
        obj_init = compute_obj(a_init)
        shape_results.append((obj_init, kind, a_init))
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Shape '{kind}': obj={obj_init:.8f} **")
        else:
            print(f"  Shape '{kind}': obj={obj_init:.8f}")

    # Also evaluate parent if it's better
    if parent is not None:
        obj_parent = compute_obj(parent)
        print(f"  Parent: obj={obj_parent:.8f}")
        if obj_parent < best_score:
            best_score = obj_parent
            best_seq = parent.copy()

    # Sort by objective
    shape_results.sort(key=lambda x: x[0])
    print(f"  Best initial: obj={best_score:.8f}")

    a = best_seq.copy()
    n_current = len(a)

    # === Phase 2: LP Alternation from best shape ===
    print(f"\n=== Phase 2: LP Alternation (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(500.0, remaining * 0.4)

    # Multiple rounds with different momentum
    momentum_vals = [0.3, 0.5, 0.7, 0.9]
    for round_idx in range(len(momentum_vals)):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        beta = momentum_vals[round_idx]
        alloc = min(lp_budget / (len(momentum_vals) - round_idx), remaining * 0.5)
        print(f"  LP Round {round_idx} (beta={beta}), budget={alloc:.0f}s")
        a_new = lp_alternating(a, time_budget=alloc, momentum_beta=beta, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  ** LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj:.8f}")
            if round_idx > 0:
                # Perturb from best
                scale = 0.02
                a = best_seq * np.exp(rng.normal(0, scale, n_current))
                a = np.maximum(a, 0)
                a = a / np.sum(a)

    # === Phase 3: Subgradient Polish ===
    print(f"\n=== Phase 3: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        sg_budget = min(120.0, remaining * 0.12)
        a_sg = subgradient_polish(a, time_budget=sg_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a = a_sg.copy()
            print(f"  ** Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 4: Multiplicative Update ===
    print(f"\n=== Phase 4: Multiplicative Update ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        mu_budget = min(80.0, remaining * 0.08)
        a_mu = multiplicative_update(best_seq, time_budget=mu_budget)
        obj_mu = compute_obj(a_mu)
        if obj_mu < best_score:
            best_score = obj_mu
            best_seq = a_mu.copy()
            a = a_mu.copy()
            print(f"  ** Multi: obj={best_score:.8f} **")
        else:
            print(f"  Multi: obj={obj_mu:.8f}")

    # === Phase 5: Try other shapes with LP ===
    print(f"\n=== Phase 5: LP on Other Shapes ===")
    remaining = deadline - time.monotonic()
    # Try the top 3 other shapes (excluding the one we already used)
    used_kind = shapes[shape_results[0][1] if isinstance(shape_results[0][1], int) else shape_results[0][1]]
    for obj_s, kind_s, a_s in shape_results[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if kind_s == 'linear' and len(best_seq) == n_base:
            # Already tried the best shape, skip
            if obj_s >= best_score * 0.99:
                continue
        # Resample to current best length
        a_try = resample_sequence(a_s, n_current) if len(a_s) != n_current else a_s.copy()
        alloc = min(60.0, remaining * 0.3)
        a_new = lp_alternating(a_try, time_budget=alloc, momentum_beta=0.5, verbose=False)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  ** Shape '{kind_s}'+LP: obj={best_score:.8f} **")
        else:
            print(f"  Shape '{kind_s}'+LP: obj={obj_new:.8f}")

    # === Phase 6: Length Sweep ===
    print(f"\n=== Phase 6: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(100, n_best - 200),
        max(100, n_best - 100),
        max(100, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 400,
    ]))
    n_options = [n for n in n_options if 100 <= n <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick LP optimization
        alloc_lp = min(30.0, remaining * 0.35)
        alloc_sg = min(15.0, remaining * 0.15)

        a_opt = lp_alternating(a_try, time_budget=alloc_lp, momentum_beta=0.5, verbose=False)
        a_opt = subgradient_polish(a_opt, time_budget=alloc_sg, verbose=False)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final LP rounds from best (remaining time) ===
    print(f"\n=== Phase 7: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    a_final = best_seq.copy()

    # Multiple short LP rounds with perturbation
    while remaining > 25:
        alloc = min(remaining * 0.35, 50.0)
        if alloc < 10:
            break
        a_new = lp_alternating(a_final, time_budget=alloc, momentum_beta=0.5, verbose=False)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_final = a_new.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            # Try subgradient as alternative
            alloc_sg = min(remaining * 0.2, 30.0)
            if alloc_sg > 10:
                a_sg = subgradient_polish(a_final, time_budget=alloc_sg, verbose=False)
                obj_sg = compute_obj(a_sg)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    a_final = a_sg.copy()
                    print(f"  Final SG: obj={best_score:.8f} **")
            # Perturb
            scale = 0.01 + 0.01 * (1 - remaining / max(1, deadline - start_time))
            a_final = best_seq * np.exp(rng.normal(0, scale, n_final))
            a_final = np.maximum(a_final, 0)
            a_final = a_final / np.sum(a_final)
            print(f"  Final: obj={obj:.8f} (perturbing, scale={scale:.4f})")

        remaining = deadline - time.monotonic()

    # === Final validation ===
    if best_seq is None:
        best_seq = np.arange(1, 801, dtype=np.float64)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [float(i + 1) for i in range(800)]
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
