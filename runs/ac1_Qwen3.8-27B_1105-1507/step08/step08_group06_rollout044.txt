"""
Enhanced step function search: LP-alternating with geometric interpolation, 
active-set perturbations, peak shaving, parameterized family exploration,
and multi-scale optimization.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv / sum^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, g_ub=None):
    """
    Solve LP: min M s.t. conv(a, g) <= M for all k, sum(g)=1, g>=0.
    Optionally upper-bound g elements.
    Returns (g_normalized, M_opt, active_indices) or (None, None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None, None
    a = a / s
    A = build_conv_matrix(a)
    m = 2 * n - 1

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    if g_ub is not None:
        constraints.append(g <= g_ub)

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    conv_val = A @ g_val
                    tol = max(1e-6, M_val * 1e-4)
                    active = np.where(conv_val >= M_val - tol)[0]
                    return g_val, M_val, active
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                conv_val = A @ g_val
                tol = max(1e-6, M_val * 1e-4)
                active = np.where(conv_val >= M_val - tol)[0]
                return g_val, M_val, active
    except Exception:
        pass

    return None, None, None


def solve_lp_top_k(a, K=20):
    """
    Solve LP with only top-K peak constraints (faster for large n).
    First compute conv(a,a) to identify top peaks, then solve reduced LP.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None, None
    a = a / s

    # Find top K peaks of conv(a,a)
    conv_aa = compute_conv(a)
    top_k = np.argsort(conv_aa)[::-1][:K]

    A_full = build_conv_matrix(a)
    A = A_full[top_k, :]  # Only top K rows
    m = len(top_k)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    # Check all constraints (not just top K)
                    conv_check = A_full @ g_val
                    M_val = float(np.max(conv_check))
                    active = np.where(conv_check >= M_val - max(1e-6, M_val * 1e-4))[0]
                    return g_val, M_val, active
        except Exception:
            continue

    return None, None, None


def lp_alternating_v2(a, time_budget=300.0, momentum_beta=0.5, use_geometric=False,
                      g_ub_mult=3.0, top_k_only=False, verbose=False):
    """
    Enhanced LP alternating with:
    - Geometric interpolation option
    - Adaptive step size
    - Active-set perturbations
    - Upper bound on g
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    momentum = np.zeros(n)
    last_t = 0.3
    iteration = 0

    while time.monotonic() - start < time_budget:
        iteration += 1
        if top_k_only and n > 200:
            g_lp, M_lp, active_idx = solve_lp_top_k(a, K=min(30, 2 * n - 1))
        else:
            g_ub = g_ub_mult / n  # Upper bound to prevent concentration
            g_lp, M_lp, active_idx = solve_lp(a, g_ub=g_ub)
        
        if g_lp is None:
            break

        # Try different interpolation methods and step sizes
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = a.copy()

        step_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
        if 0 < last_t < 5.0:
            step_candidates = sorted(set(step_candidates + [max(0.05, last_t * 0.7), min(3.0, last_t * 1.3)]))

        for t in step_candidates:
            if use_geometric:
                # Geometric interpolation: a * (g*a_neg)^t
                # Where we handle zeros: use a + epsilon
                eps = 1e-10
                a_safe = np.maximum(a, eps)
                g_safe = np.maximum(g_lp, eps)
                # Geometric mean interpolation
                a_try = a_safe * (g_safe / a_safe) ** min(t, 2.0)
                # This equals a_safe * (g_safe/a_safe)^t = a^(1-t) * g^t (geometric interp)
            else:
                # Linear with momentum
                direction = g_lp - a
                momentum = momentum_beta * momentum + (1 - momentum_beta) * direction
                a_try = a + t * momentum

            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and iteration % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP v2 iter {iteration}: obj={best_obj:.8f}, t={best_t_val:.3f}, time={elapsed:.0f}s")
        else:
            stall += 1
            momentum *= 0.7
            if stall >= 5:
                # Active-set-based perturbation
                if len(active_idx) > 0:
                    # Perturb coordinates that contribute to active peaks
                    n_perturb = max(10, n // 20)
                    rng = np.random.default_rng(iteration * 7919)
                    # Get contributing coordinates for active peaks
                    contrib_set = set()
                    for k in active_idx[:10]:
                        k = int(k)
                        i_min = max(0, k - n + 1)
                        i_max = min(n - 1, k)
                        for i in range(i_min, i_max + 1):
                            contrib_set.add(i)
                    if len(contrib_set) > n_perturb:
                        contrib_list = list(contrib_set)
                        perturb_idx = rng.choice(contrib_list, size=n_perturb, replace=False)
                    else:
                        perturb_idx = np.array(list(contrib_set))
                    
                    scale = 0.05 + 0.02 * min(stall // 5, 10)
                    a[perturb_idx] *= np.exp(rng.normal(0, scale, len(perturb_idx)))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                else:
                    # Random perturbation
                    rng = np.random.default_rng(iteration * 104729)
                    scale = 0.05 + 0.02 * min(stall // 5, 10)
                    a *= np.exp(rng.normal(0, scale, n))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                
                stall = 0
                last_t = 0.5
                momentum = np.zeros(n)

    return best_a


def peak_shaving(a, time_budget=120.0, verbose=False):
    """
    Targeted peak shaving: for the dominant peak, find the best
    single-coordinate or pair-coordinate move that reduces the objective.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for iteration in range(2000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Find contributing positions to this peak
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        
        if i_max < i_min:
            break

        # Contributions: a[i] * a[k_star - i]
        contribs = []
        for i in range(i_min, i_max + 1):
            j = k_star - i
            contribs.append((i, j, a[i] * a[j]))
        
        # Sort by contribution (highest first)
        contribs.sort(key=lambda x: -x[2])

        improved = False
        for (i, j, c) in contribs[:30]:
            if time.monotonic() - start > time_budget:
                break
            
            # Try moving mass from position i (or j) to a "good" target
            # Good target: a position t where a[t]*a[k_star-t] is small
            # and the move reduces the peak
            
            # Option 1: Move from i to some target t
            for frac in [0.01, 0.03, 0.05]:
                delta = frac * a[i]
                if delta <= 0 or delta >= a[i]:
                    continue
                
                # Try all possible targets (sampled)
                # Best targets are where a[t] is small AND a[k_star-t] is small
                # (so adding mass there doesn't create a new peak)
                
                # Sample targets
                targets = []
                for t in range(n):
                    if t == i:
                        continue
                    tj = k_star - t
                    if 0 <= tj < n:
                        new_contrib = (a[t] + delta) * a[tj]
                        old_contrib = a[i] * a[k_star - i]
                        # Net change in peak: new_contrib - (a[i]-delta)*a[k_star-i] - (a[i]*a[k_star-i])
                        # Actually: removing delta from i reduces contribution by delta*a[k_star-i]
                        # Adding delta to t increases contribution by delta*a[k_star-t] (if k_star-t valid)
                        # Net change: delta * (a[k_star-t] - a[k_star-i])
                        # We want this negative: a[k_star-t] < a[k_star-i]
                        if a[tj] < a[k_star - i] * 0.5:
                            targets.append((t, a[t] + delta) * a[tj] - a[t] * a[tj])
                
                if len(targets) > 50:
                    targets.sort(key=lambda x: x[1])
                    targets = targets[:20]
                
                for (t, _new_contrib) in targets[:5]:
                    a[i] -= delta
                    a[t] += delta
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s < 1e-10:
                        a[i] += delta
                        a[t] -= delta
                        break
                    a = a / s
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-10:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        # Revert
                        a = best_a.copy()  # revert to best
                        # Actually we should revert properly
                        a[i] += delta  # This is wrong if a was modified
                        # Let's just re-copy
                        a = best_a.copy()
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random pair moves
            rng = np.random.default_rng(iteration * 31)
            for _ in range(10):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.02 * a[i]
                if delta <= 0 or delta >= a[i]:
                    continue
                a_save = a.copy()
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s < 1e-10:
                    continue
                a = a / s
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-10:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = a_save

            if not improved:
                # Try moving mass from peak contributors to the "quiet" region
                # The quiet region is where conv is smallest
                conv = compute_conv(a)
                quiet = np.argsort(conv)[:10]
                rng2 = np.random.default_rng(iteration * 47)
                for kq in quiet:
                    kq = int(kq)
                    # Move mass to positions that contribute to quiet peaks
                    i_min_q = max(0, kq - n + 1)
                    i_max_q = min(n - 1, kq)
                    if i_max_q < i_min_q:
                        continue
                    target_q = rng2.integers(i_min_q, i_max_q + 1)
                    source = i_min + rng2.integers(0, i_max - i_min + 1)
                    if source == target_q:
                        continue
                    delta = 0.01 * a[source]
                    if delta <= 0:
                        continue
                    a_save = a.copy()
                    a[source] -= delta
                    a[target_q] += delta
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s < 1e-10:
                        continue
                    a = a / s
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-10:
                        best_obj = new_obj
                        best_a = a.copy()
                        a = best_a.copy()
                        improved = True
                        break
                    else:
                        a = a_save

            if not improved and iteration > 200:
                break

    if verbose:
        print(f"  Peak shaving: obj={best_obj:.8f}")
    return best_a


def subgradient_polish_v2(a, time_budget=100.0, verbose=False):
    """
    Improved subgradient method with:
    - Multi-peak averaging
    - Adaptive step size (Barzilai-Borwein)
    - Projection onto simplex
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()
    y = a.copy()
    lr = 0.03
    prev_y = None
    prev_grad = None
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        
        conv = compute_conv(x)
        max_val = float(np.max(conv))
        
        # Get active peaks (within 20% of max)
        threshold = max_val * 0.8
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = [int(np.argmax(conv))]
        if len(active) > 40:
            active = np.argsort(conv)[::-1][:40]

        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                grad[i] += a[k - i]
        grad /= len(active)
        # Multiply by 2 for the derivative of a*a
        grad *= 2.0

        # Center gradient
        grad_centered = grad - np.mean(grad)

        # Barzilai-Borwein step size
        if prev_y is not None and prev_grad is not None:
            sy = y - prev_y
            gy = grad_centered - prev_grad
            sy_norm2 = float(np.dot(sy, sy))
            gy_norm2 = float(np.dot(gy, gy))
            if sy_norm2 > 1e-15 and gy_norm2 > 1e-15:
                sg = float(np.dot(sy, gy))
                if abs(sg) > 1e-15:
                    # BB step
                    lr_bb = sy_norm2 / abs(sg)
                    lr = 0.5 * lr + 0.5 * min(max(lr_bb, 0.001), 0.5)

        y_new = y - lr * grad_centered
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new

        # Nesterov
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        prev_y = y.copy()
        prev_grad = grad_centered.copy()
        y = y_new

        obj = compute_obj(y)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = y.copy()

    if verbose:
        print(f"  Subgrad v2: obj={best_obj:.8f}")
    return best_a
