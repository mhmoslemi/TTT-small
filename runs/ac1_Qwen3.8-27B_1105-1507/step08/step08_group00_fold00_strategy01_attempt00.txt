                    n_val = n_initial + 500
                    best_score = float('inf')
                
                # Check active learning regret bounds
                if best_score < missed_penalty and (time.monotonic() - start_time < time_budget * 0.9):
                    # More aggressive update for early regulation
                    n_val = n_initial + 100
                    best_score = float('inf')
                elif best_score >= missed_penalty and (time.monotonic() - start_time < time_budget * 0.9):
                    # Shrink for later phases
                    pass
            
            iterations += 1
            learning_rate *= learning_rate
            
            # Safety backup with strong zero-padding
            if learning_rate < 0.0000001 and iterations > 1000:
                # Final improvement vector and score
                return [float(x) for x in best_sequence], best_score
        
        # Second phase: Polyak averaging of the gradients
        duration2 = min(250, d_remaining * 0.3)
        
        # Cuadratic cost conservative gradient bounds
        learning_cooldown = 0.0005
        
        #Polish the solution with adaptive smoothing
        # Use arithmetic mean convolution with probability weight
        smooth_factor = 0.8
        
        # We'll blend the best_sequence with noise-based improvement
        # Only update if it significantly improves the score
        while (time.monotonic() - start_time < time_budget):
            raw_seq = [float(x) * smooth_factor for x in best_sequence]
            obj_smooth = evaluate_sequence(raw_seq)
            
            # If improvement, incorporate update
            if obj_smooth < best_score:
                best_sequence = [float(x * (1 - learning_cooldown)) + float(learning_cooldown * raw_seq) for x in raw_seq]
                best_score = obj_smooth
            else:
                # Stuck improvement
                break
    
    except Exception as e:
        print("Error in candidate optimization: ", e)
        # Return fallback if exceeded or interrupted
        fallback_seq = list([float(1) if 10==1 else float(0.1)])
        fallback_score = getattr(fallback_seq, 'inf', evaluate_sequence(fallback_seq))
        
        if fallback_score is not None and float('inf') > fallback_score:
            return fallback_seq, fallback_score
        
    finally:
        # Extract the final sequence and score
        final_seq = [float(x) for x in best_sequence]
        final_score = float(best_score) if best_score is not None else np.inf

# Additional utilities for decay management
def decay_safe(seq, safety_margin=0.005):
    try:
        seq_clean = [x if isinstance(x, (int, float)) else float(0.0001) for x in seq]
        non_zero_count = sum(1 for x in seq if x != 0)
        total_sum = sum(seq)
        
        if non_zero_count < len(seq):
            # Zero-pad to ensure full convolution length
            max_len = len(seq) + (len(seq) - non_zero_count)
            padded_seq = np.zeros(int(max_len))
            padded_seq += np.array(seq)
            return [float(x) * (1/safety_margin) for x in padded_seq], safety_margin / total_sum]
        
        return seq_clean, total_sum
    except:
        margin = 0.5 * len(seq)
        return list(seq), float(margin) if any(seq) and total_sum > 0 else np.inf

# Convolution wrapper with safety margins
def safe_convolution(a, b):
    try:
        a = np.array(a)
        b = np.array(b)
        
        # Avoid generating long sequences unnecessarily
        # aliasing may occur, but truncated for large outputs
        
        # Use only up to the maximum index needed
        window_size = min(len(a), len(b), 1000)
        
        a_trimmed = a[:window_size]
        b_trimmed = b[:window_size]
        
        # Compute probabilities for the convolution step
        convolve_result = np.convolve(a_trimmed, b_trimmed, mode='full')
        
        # Keep max and then scale accordingly
        max_step = np.max(convolve_result)
        max_index = np.argmax(convolve_result)
        
        # Truncate to active length for repetition
        return list(convolve_result[:window_size]), convolve_result[max_index] * (window_size**2)
    except:
        fallback_size = 500
        fallback_arr = [float(1)/fallback_size] * fallback_size
        return fallback_arr, evaluate_sequence(fallback_arr)
