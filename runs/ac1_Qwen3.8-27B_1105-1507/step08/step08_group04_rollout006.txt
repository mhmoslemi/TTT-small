import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A

    prob = cp.Problem(cp.Minimize(M), constraints)

        try:

                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    
    return None, None

def pnorm_gradient(a, p):
    """Compute gradient of (sum_k conv_k^p)^(1/p) at a (sum(a)=1)."""
    n = len(a)
    conv = compute_conv(a)
    # Avoid issues with negative values (shouldn't happen for non-negative a)
    conv = np.maximum(conv, 1e-30)
    
    # Weights: w[k] = conv[k]^(p-1)
    w = conv ** (p - 1.0)
    
    # Normalization: f_p = (sum w[k] * conv[k])^(1/p) = (sum conv[k]^p)^(1/p)
    # grad[j] = (sum conv^p)^(1/p - 1) * sum_k conv[k]^(p-1) * 2*a[k-j] * 1[valid]
    
    f_val = (w @ conv) ** (1.0 / p)
    scale = f_val ** (1.0 - p)  # = (sum conv^p)^(1/p - 1)
    
    # Cross-correlation: R[j] = sum_k w[k] * a[k-j] * 1[0<=k-j<n]
    N = 1
    while N < 3 * n:
        N <<= 1
    FA = rfft(a, N)
    FW = rfft(w, N)
    R = irfft(np.conj(FA) * FW, N)[:n]
    
    grad = 2.0 * scale * R
    return grad, f_val, conv

def adam_optimize_pnorm(a, time_budget, p_start=2.0, p_end=64.0, lr=0.01):
    """Adam optimizer with p-norm smooth objective, adaptive p schedule."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        # Adaptive p: start small, increase over time
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        # Use exponential schedule
        p = p_start * (p_end / p_start) ** progress
        
        grad, f_val, conv = pnorm_gradient(a, p)
        
        # Center gradient for simplex
        grad = grad - np.mean(grad)
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive learning rate
        lr_t = lr / (1 + t * 0.001)
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Decay learning rate
        if t % 500 == 0:
            lr *= 0.95
    
    return best_a

def lp_alternate_fast(a, time_budget=100.0, verbose=False):

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Hybrid optimizer: MWU (multiplicative weight updates) for fast exploration,
LP alternating for precise polishing, log-sum-exp smooth gradient for escaping
local minima. Targets 2*n*max(conv(a,a))/(sum(a))^2 < 1.5027.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp(a):
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > 30:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                # Recompute M
                conv_ag = np.convolve(a, g_val, mode='full')
                return g_val, float(np.max(conv_ag))
    except Exception:
        pass
    
    return None, None


def mwu_step(a, eta, top_k=None):
    """One step of multiplicative weight updates.
    Returns updated a (normalized to sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    if top_k is not None and top_k < len(conv):
        # Use top-k peaks for the subgradient
        k_indices = np.argsort(conv)[::-1][:top_k]
        w = np.zeros(len(conv))
        for k in k_indices:
            w[k] = 1.0 / top_k
    else:
        w = np.ones(len(conv)) / len(conv)
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] * 1[0<=k-j<n]
    N = 1
    while N < 3 * n:
        N <<= 1
    FA = rfft(a, N)
    FW = rfft(w, N)
    R = irfft(np.conj(FA) * FW, N)[:n]
    grad = 2.0 * R
    
    # Center for simplex
    grad = grad - np.mean(grad)
    
    # Multiplicative update
    a_new = a * np.exp(-eta * grad / max_conv)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s


def mwu_optimize(a, time_budget, eta_init=0.1, top_k=None, verbose=False):

        a = mwu_step(a, eta, top_k=top_k)
        it += 1
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        
        # Decay eta
        if it % 100 == 0:
            eta *= 0.98
        
        # Adapt top_k: start with few, increase over time
        if top_k is None:
            pass  # top_k is fixed
    
    if verbose:
        elapsed = time.monotonic() - start
        print(f"    MWU: {it} iters, obj={best_obj:.8f}, time={elapsed:.1f}s")
    
    return best_a


def logsumexp_gradient(a, beta):
    """Compute gradient of (1/beta)*log(sum_k exp(beta*conv_k)) at normalized a."""
    n = len(a)
    conv = compute_conv(a)
    
    # Softmax weights (numerically stable)
    bc = beta * conv
    bc_max = np.max(bc)
    exp_bc = np.exp(bc - bc_max)
    p = exp_bc / np.sum(exp_bc)
    
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] * 1[0<=k-j<n]
    N = 1
    while N < 3 * n:
        N <<= 1
    FA = rfft(a, N)
    FP = rfft(p, N)
    R = irfft(np.conj(FA) * FP, N)[:n]
    grad = 2.0 * R
    
    return grad


def logsumexp_optimize(a, time_budget, beta_start=5.0, beta_end=500.0, lr=0.05, verbose=False):
    """Optimize using log-sum-exp smooth objective with Adam."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        grad = logsumexp_gradient(a, beta)
        grad = grad - np.mean(grad)
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr_t = lr / (1.0 + 0.01 * t)
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if t % 1000 == 0 and verbose:
            print(f"    LSE: t={t}, beta={beta:.1f}, obj={best_obj:.8f}")
    
    return best_a
