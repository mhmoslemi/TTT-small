    e = np.exp(theta - m)
    s = e.sum()

def subsample_conv_grad(conv, n, topK):
    """Use only top-K convolution values for a sparser gradient (faster)."""
    if topK >= len(conv):
        return np.ones(len(conv))
    idx = np.argpartition(conv, -topK)[-topK:]
    m = conv[idx].max()
    w = np.zeros_like(conv)
    w[idx] = np.exp(conv[idx] - m)
    w /= w.sum()
    return w


def lbfgs_pass(a, beta, max_time=60.0, maxiter=300, verbose=False, tag=""):
    """
    Single L-BFGS-B pass on the smooth objective at fixed beta.
    Returns optimized a on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    theta = np.log(a)

    start = time.monotonic()
    state = {'elapsed': 0.0, 'n_calls': 0}

    def time_callback(xk):
        state['n_calls'] += 1
        elapsed = time.monotonic() - start
        state['elapsed'] = elapsed
        if elapsed > max_time:
            return True
        return None

    try:
        res = minimize(
            smooth_obj_grad,
            theta.copy(),
            args=(beta,),
            method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': maxiter,
                'ftol': 1e-12,
                'gtol': 1e-7,
                'maxls': 30,
                'callback': time_callback,
            }
        )
        a_new = softmax(res.x)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
    except Exception:
        a_new = a.copy()

    obj = 2.0 * n * float(np.max(compute_conv(a_new)))
    if verbose:
        print(f"    {tag} L-BFGS beta={beta:.0f}: obj={obj:.8f}, "
              f"iters={state['n_calls']}, t={time.monotonic()-start:.1f}s, "
              f"status={res.status if 'res' in dir() else 'err'}")

    return a_new


def multi_pass_lbfgs(a, time_budget, beta_schedule=None, verbose=False):
    """
    Multiple L-BFGS-B passes with increasing beta.
    """
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    if beta_schedule is None:
        beta_schedule = [30, 100, 300, 1000, 3000, 10000]

    start = time.monotonic()
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    for i, beta in enumerate(beta_schedule):
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 5:
            break

        # Allocate time: later passes get more time
        frac_remaining = (len(beta_schedule) - i) / len(beta_schedule)
        pass_time = min(remaining * 0.6, remaining * frac_remaining)
        pass_time = max(pass_time, 5.0)

        a = lbfgs_pass(a, beta, max_time=pass_time,
                       maxiter=min(500, int(pass_time * 40)),
                       verbose=verbose, tag=f"pass{i}")

        obj = 2.0 * n * float(np.max(compute_conv(a)))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if verbose:
        print(f"    Multi-pass L-BFGS: final obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def subgradient_polish(a, time_budget=60.0, verbose=False):
    """
    Subgradient descent on max(conv(a,a)) with projection onto simplex.
    Uses momentum and adaptive step size.
    """
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    momentum = 0.3
    v = np.zeros(n)
    lr = 0.5 / n
    stall = 0
    patience = 80
    topK = 5

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Get top-K indices for subgradient
        if topK >= len(conv):
            idx = np.arange(len(conv))
        else:
            idx = np.argpartition(conv, -topK)[-topK:]

        m = conv.max()
        if m < 1e-15:
            break

        # Soft weights over top-K
        w_top = np.exp(3.0 * (conv[idx] - conv[idx].max()))
        w_top /= w_top.sum()

        # Subgradient: weighted sum of gradients of top-K convolution values
        g = np.zeros(n)
        for w_k, k in zip(w_top, idx):
            k = int(k)
            # d(conv_k)/d(a_p) = 2*a_{k-p} for 0<=k-p<n
            p_min = 0
            p_max = n - 1
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            # g[p] += 2 * w_k * a[k-p] for p in [k-(n-1), k] intersected with [0, n-1]
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * w_k * a[k - hi:k - lo + 1][::-1]

        # Momentum
        v = momentum * v + g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        # Adaptive step size
        a_new = a - lr * v / v_norm * np.sqrt(n)
        # Project onto simplex
        a_new = project_simplex(a_new)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            lr = min(0.5 / n, lr * 1.01)
            momentum = min(0.9, momentum + 0.002)
        else:
            stall += 1
            if stall > patience:
                # Restart with perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                noise = 0.02 * rng.standard_normal(n)
                a = best_a * (1.0 + noise)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                v = np.zeros(n)
                lr = 0.5 / n
                momentum = 0.3
                stall = 0
                topK = min(20, topK + 5)
            else:
                lr = max(1e-6, lr * 0.98)
                if stall % 20 == 0:
                    topK = min(2 * n - 1, topK + 2)

        if verbose and stall == 0:
            print(f"    SubGrad: obj={best_obj:.8f}, lr={lr:.2e}, "
                  f"topK={topK}, t={time.monotonic()-start:.1f}s")

    return best_a


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)
        a = a / a.sum()
    elif mode == 3:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-8))
        a = a / a.sum()
    elif mode == 4:
        # Exponential decay
        x = np.arange(n, dtype=float)
        a = np.exp(-0.5 * x / n)
        a = a / a.sum()
    elif mode == 5:
        # Bump at center
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * (x - 0.5)**2) + 0.1
        a = a / a.sum()
    elif mode == 6:
        # Two bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-80 * (x - 0.3)**2) + np.exp(-80 * (x - 0.7)**2) + 0.1
        a = a / a.sum()
    elif mode == 7:
        # Linear
        x = np.arange(n, dtype=float)
        a = (x + 1)
        a = a / a.sum()
    elif mode == 8:
        # Cosine
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return a


def coordinate_descent_refine(a, time_budget=30.0):
    """
    Coordinate descent: try moving mass from peak-contributing positions.
    """
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    max_rounds = 10
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        # Find non-contributing positions
        contrib_set = set()
        for _, i, j in contributors:
            contrib_set.add(i)
            contrib_set.add(j)

        non_contrib = [i for i in range(n) if i not in contrib_set]
        if not non_contrib:
            non_contrib = list(range(n))

        improved = False
        for contrib_val, ci, cj in contributors[:20]:
            if time.monotonic() - start > time_budget:
                break
            if contrib_val < 1e-10:
                break

            for target in non_contrib[:10]:
                for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                    src = ci if a[ci] >= a[cj] else cj
                    if src == target:
                        continue
                    delta = frac * a[src]
                    if delta < 1e-12 or delta > a[src] * 0.5:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_obj = 2.0 * n * float(np.max(compute_conv(a)))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        if not improved:
            break

    return best_a


def anneal_lbfgs(a, time_budget=200.0, n_restarts=4, verbose=False):
    """
    L-BFGS-B with simulated annealing: perturb and restart to escape local minima.
    """
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    temperature = 0.08
    rng = np.random.default_rng(42 + n)

    for restart in range(n_restarts + 1):
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 10:
            break

        # Beta schedule for this restart
        betas = [50, 200, 800, 3000]
        if restart == 0:
            betas = [30, 100, 500, 2000, 8000]

        for beta in betas:
            remaining = time_budget - (time.monotonic() - start)
            if remaining < 5:
                break
            pass_time = min(remaining * 0.35, 40.0)
            a = lbfgs_pass(a, beta, max_time=pass_time,
                           maxiter=min(300, int(pass_time * 30)),
                           verbose=False)
            obj = 2.0 * n * float(np.max(compute_conv(a)))
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        if verbose and (restart + 1) % 2 == 0:
            print(f"    Anneal-LBFGS restart {restart}: obj={best_obj:.8f}, "
                  f"t={time.monotonic()-start:.1f}s")

        # Perturb for next restart
        if restart < n_restarts:
            noise = temperature * rng.standard_normal(n)
            a = best_a * (1.0 + noise)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            temperature *= 0.8

    return best_a
