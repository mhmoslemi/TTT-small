def toep_matvec(p, a, fft_size=None, precomputed fv=None):

    n = len(a)
    if n == 1:
        return p[0] * a
    if fft_size is None:
        fft_size = 1
        while fft_size < 2 * n - 1:
            fft_size <<= 1

def build_toep_fft(p, fft_size):
    """Precompute FFT of circulant embedding for Toeplitz matvec."""
    n = len(p)
    v = np.zeros(fft_size)
    v[0] = p[0]
    if n > 1:
        v[1:n] = p[1:n]
        v[n:2*n-1] = p[1:n][::-1]
    return rfft(v, fft_size)


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

            print(f"    MinMMX it{it}: obj={best_obj:.8f}, temp={temperature:.3f}, "
                  f"lr={lr:.4f}, pat={patience}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def direct_subgrad_mirror_descent(a_init, n, time_budget, verbose=False):
    """Direct mirror descent using subgradient of max_k conv(a,a)_k."""
    start = time.monotonic()
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_obj = np.inf
    best_a = a.copy()
    lr = 3.0 / np.sqrt(n)
    momentum = np.zeros(n)
    mu = 0.5  # momentum coefficient
    it = 0
    patience = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        conv_n = conv[:n]
        max_val = conv_n.max()
        
        # Softmax weights over lags
        beta = 5.0
        w = np.exp(beta * (conv_n - max_val))
        w = w / w.sum()
        
        # Compute subgradient: g_i = sum_k w_k * (a_{i+k} + a_{i-k})
        g = np.zeros(n)
        for k in range(n):
            if w[k] < 1e-8:
                continue
            # conv(a,a)_k = sum_i a_i * a_{i+k}
            # d/d a_i of conv_k = a_{i+k} + a_{i-k}
            wk = w[k]
            if k == 0:
                g += wk * 2.0 * a
            else:
                # a_{i+k} for i=0..n-1-k
                g[:n-k] += wk * a[k:]
                # a_{i-k} for i=k..n-1
                g[k:] += wk * a[:n-k]
        
        # Momentum
        momentum = mu * momentum + g
        log_a = np.log(a)
        log_a_new = log_a - lr * momentum
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)[:n]))
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            patience = 0
            lr = max(0.5 / np.sqrt(n), lr * 0.9995)
            mu = min(0.9, mu + 0.001)
        else:
            patience += 1
            lr = max(0.1 / np.sqrt(n), lr * 0.98)
            if patience > 100:
                rng = np.random.default_rng(it * 17 + n)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                momentum = np.zeros(n)
                lr = 3.0 / np.sqrt(n)
                patience = 0
        
        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    SubGrad it{it}: obj={best_obj:.8f}, lr={lr:.4f}, mu={mu:.3f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def lp_polish(a, time_budget, verbose=False):
    """LP alternating polish: solve LP for g given a, interpolate, repeat."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Build constraint matrix A (2n-1) x n: A[k,j] = a_norm[k-j] if valid
    A = np.zeros((2 * n - 1, n))
    a_norm = a / a.sum()
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j+r, j] = a_norm[:r]
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Solve LP: min M s.t. A @ g <= M, sum(g)=1, g>=0
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                break
            g_val = np.maximum(res.x[:n], 0)
            s_g = g_val.sum()
            if s_g < 1e-10:
                break
            g_val = g_val / s_g
        except Exception:
            break
        
        # Line search for best interpolation
        best_t = 0
        best_t_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g_val
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            a = (1 - best_t) * a + best_t * g_val
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_a = a.copy()
            # Rebuild A with new a
            a_norm = a / a.sum()
            A = np.zeros((2 * n - 1, n))
            for j in range(n):
                r = min(n, 2 * n - 1 - j)
                A[j:j+r, j] = a_norm[:r]
        else:
            # No improvement, break
            break
        
        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}, t_elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def coordinate_descent_fast(a, time_budget, verbose=False):
    """Fast coordinate descent: move mass from peak contributors to non-peak positions."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    k_star = int(np.argmax(conv[:n]))
    
    it = 0
    while time.monotonic() - start < time_budget:
        improved = False
        conv = compute_conv(a)
        k_star = int(np.argmax(conv[:n]))
        peak_val = conv[k_star]
        
        # Find top contributors to conv[k_star]
        if k_star == 0:
            contribs = [(a[i]*a[i], i, i) for i in range(n)]
        else:
            contribs = [(a[i]*a[i+k_star], i, i+k_star) for i in range(n-k_star)]
        contribs.sort(reverse=True)
        
        # Non-contributor positions
        contrib_set = set()
        for _, i, j in contribs[:20]:
            contrib_set.add(i)
            contrib_set.add(j)
        non_contrib = [i for i in range(n) if i not in contrib_set]
        
        if not non_contrib:
            break
        
        for contrib, i, j in contribs[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break
            for target in non_contrib[:5]:
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * min(a[i], a[j])
                    if delta < 1e-12:
                        continue
                    src = i if a[i] >= a[j] else j
                    old_val = a[src]
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_peak = float(np.max(new_conv[:n]))
                    new_obj = 2.0 * n * new_peak
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        k_star = int(np.argmax(new_conv[:n]))
                        break
                    else:
                        a[src] = old_val
                        a[target] -= delta
        
        if not improved:
            break
        it += 1
    
    return best_a, best_obj


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4*np.abs(x-0.5)**2) + 0.01
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = (1 - x**2) ** 0.5 + 0.01
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2*np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2*np.pi*f*x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        a = rng.dirichlet(np.ones(n) * 0.5)
    elif mode == 6:
        # Beta-like distribution
        a = rng.beta(2, 2, n) + 0.01
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)

            s = float(np.sum(parent_arr))

                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj

                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

        n0 = 800
        best_seq = np.ones(n0) / n0

        print(f"Default init (n={n0}): obj={best_score:.8f}")

    print(f"\n=== Phase 1: Minimax Mirror Descent (n={n_current}) ===")
    remaining = deadline - time.monotonic()

    if p1_budget > 30:

    # --- Phase 3: LP Polishing ---

    remaining = deadline - time.monotonic()

    if p3_budget > 30:
        a_lp, obj_lp = lp_polish(best_seq, p3_budget, verbose=True)

            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f} (no improvement)")
    
    # --- Phase 4: Multi-n exploration ---
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    n_options = sorted(set([

    return result
