Hybrid optimization for minimizing peak autocorrelation of non-negative step functions.
Combines LP alternating with spectral shaping, targeted mass transfer, n-sweep,
and adaptive restarts. Target: score < 1.5027.

import numpy as np
import time

from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.PDLP, cp.SCS]:
        try:
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None

def spectral_shaping(a, time_budget, verbose=False):
    """
    Try to flatten the power spectrum to reduce autocorrelation peaks.
    Iteratively: compute spectrum, flatten it, invert, project to simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        if time.monotonic() - start > time_budget:
            break
        for it in range(20):
            if time.monotonic() - start > time_budget:
                break
            # Compute power spectrum
            size = 1
            while size < 2 * n:
                size <<= 1
            A = rfft(a, size)
            P = np.abs(A) ** 2

            # Flatten: P_new = (1-alpha)*P + alpha*mean(P)
            P_flat = (1 - alpha) * P + alpha * np.mean(P)
            P_flat = np.maximum(P_flat, 0)

            # Invert to get candidate sequence
            A_new = np.sqrt(P_flat) * (A / (np.abs(A) + 1e-30))
            a_new = irfft(A_new, size)[:n]
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                if verbose:
                    print(f"          Spectral alpha={alpha}, it{it}: obj={best_obj:.8f}")
            else:
                break

    return best_a

        round_num += 1
        conv = compute_conv(a)
        # Find top 3 peak lags
        peak_lags = np.argsort(conv)[::-1][:3]

        # Find "quiet" positions: positions i where a[i] is small
        # and that don't contribute much to peak lags
        quiet_score = np.zeros(n)
        for i in range(n):
            contrib = 0.0
            for k in peak_lags:
                k = int(k)
                j = k - i
                if 0 <= j < n:
                    contrib += abs(conv[k]) * a[j]
            quiet_score[i] = -contrib  # more negative = more "active"

        quiet_positions = np.argsort(quiet_score)[-5:]  # least active positions

        # For each peak lag, try to reduce it
        improved = False
        for k_star in peak_lags:
            if time.monotonic() - start > time_budget:
                break
            k_star = int(k_star)
            # Find contributors to this lag
            contributors = []
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contributors.append((a[i] * a[j], i, j))
            contributors.sort(reverse=True)

            for contrib, i, j in contributors[:10]:
                if time.monotonic() - start > time_budget:
                    break
                if contrib < 1e-10:
                    continue
                src = i if a[i] >= a[j] else j
                for target in quiet_positions:
                    if target == src:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                        delta = frac * a[src]
                        if delta <= 0 or delta >= a[src]:
                            continue
                        a[src] -= delta
                        a[target] += delta
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))
                        if new_obj < best_obj - 1e-12:
                            best_obj = new_obj
                            best_a = a.copy()
                            conv = new_conv
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[target] -= delta
                            break
                    if improved:
                        break

        if not improved and round_num > 10:
            break

        if verbose and round_num % 10 == 0:
            print(f"          Targeted round {round_num}: obj={best_obj:.8f}")

    return best_a


def smooth_optimize(a, time_budget, beta_start=50.0, beta_end=3000.0, verbose=False):
    """
    Smooth optimization using log-sum-exp surrogate with mirror descent.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    it = 0
    total_iters = max(1, int(time_budget * 3))

    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / total_iters)
        beta = beta_start * (beta_end / beta_start) ** frac

        conv = compute_conv(a)
        m = conv.max()
        shifted = beta * (conv - m)
        weights = np.exp(shifted)
        weights /= weights.sum()

        # Gradient: g[i] = 2 * sum_k weights[k] * a[k-i] (cross-correlation)
        g = np.zeros(n)
        for i in range(n):
            k_start = 0
            k_end = min(2 * n - 1 - 1, i + n - 1)
            a_start = max(0, k_start - i)
            a_end = min(n - 1, k_end - i)
            if a_start <= a_end:
                ks = i + a_start
                ke = i + a_end
                g[i] = 2.0 * np.dot(a[a_start:a_end + 1], weights[ks:ke + 1])

        # Mirror descent
        log_a = np.log(a)
        log_a_new = log_a - lr * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = max(1e-6, lr * 0.995)
        else:
            lr = max(1e-6, lr * 0.9)

        it += 1
        if verbose and it % 100 == 0:
            print(f"          Smooth it{it}: obj={best_obj:.8f}, beta={beta:.0f}")

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode % 8 == 0:
        a = np.ones(n) / n
    elif mode % 8 == 1:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode % 8 == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 8 == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode % 8 == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.3) ** 2) + np.exp(-2 * (x - 0.7) ** 2)
        a = a / a.sum()
    elif mode % 8 == 5:
        x = np.linspace(0, 1, n)
        a = x**2 + (1 - x)**2 + 0.1
        a = a / a.sum()
    elif mode % 8 == 6:
        freqs = rng.integers(2, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent(a, time_budget):
    """Refined coordinate descent using LP-guided moves."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for rnd in range(20):
        if time.monotonic() - start > time_budget:
            break

        # Get LP solution to guide moves
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Find the largest discrepancies
        diff = g_lp - a
        # Move towards LP solution in the largest difference coordinates
        idx = np.argsort(np.abs(diff))[::-1]

        improved = False
        for i in idx[:15]:
            if time.monotonic() - start > time_budget:
                break
            for target in idx[:15]:
                if target == i:
                    continue
                for frac in [0.01, 0.02, 0.05]:
                    delta = frac * abs(diff[i])
                    if delta <= 0:
                        continue
                    a[i] += delta
                    a[target] -= delta
                    if a[target] < 0:
                        a[i] -= delta
                        a[target] += delta
                        continue
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s < 1e-10:
                        a[i] -= delta
                        a[target] += delta
                        continue
                    a = a / s
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] -= delta
                        a[target] += delta
                        a = np.maximum(a, 0)
                        break

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating, spectral shaping,
    targeted mass transfer, n-sweep, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, obj={best_score:.8f}, target=1.5027")

    # === Phase 1: Quick LP polish (10%) ===
    print(f"\n=== Phase 1: Quick LP Polish ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.10, 100.0)
    if p1_budget > 20:
        a_p1 = lp_alternate(best_seq, time_budget=p1_budget, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f}")

    # === Phase 2: n-sweep (20%) ===
    print(f"\n=== Phase 2: n-Sweep ===")
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.20
    n_options = sorted(set([
        max(400, n_current - 100),
        max(400, n_current - 50),
        max(400, n_current - 30),
        max(400, n_current - 10),
        n_current,
        n_current + 10,
        n_current + 30,
        n_current + 50,
        n_current + 100,
        n_current + 150,
        750, 768, 780, 820, 840, 850, 860, 880, 900, 960, 1000, 1024,
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1200]

    per_n_budget = p2_budget / max(1, len(n_options))
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_n_budget, remaining * 0.15)
        if alloc < 15:
            alloc = 15

        a_try = resample_sequence(best_seq, n_try)
        # Quick LP alternating
        a_opt = lp_alternate(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)
        marker = " **" if obj_opt < best_score else ""
        print(f"  n={n_try}: obj={obj_opt:.8f}{marker}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = n_try

    # === Phase 3: Intensive LP alternating on best n (35%) ===
    print(f"\n=== Phase 3: Intensive LP (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.35, 500.0)

    # Run in multiple rounds with perturbations
    n_rounds = 4
    per_round = p3_budget / n_rounds
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        alloc = min(per_round, remaining * 0.25)
        if alloc < 30:
            break
        if rnd == 0:
            a_start = best_seq.copy()
        else:
            rng_r = np.random.default_rng(rnd * 7919 + n_current)
            scale = 0.03 + 0.02 * rnd
            a_start = best_seq * np.exp(rng_r.normal(0, scale, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        a_lp = lp_alternate(a_start, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        marker = " **" if obj_lp < best_score else ""
        print(f"  LP round {rnd}: obj={obj_lp:.8f}{marker}")
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()

    # === Phase 4: Spectral shaping + targeted moves (15%) ===
    print(f"\n=== Phase 4: Spectral + Targeted ===")
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.15

    # Spectral shaping
    alloc_s = min(p4_budget * 0.4, 60.0)
    if alloc_s > 10:
        a_spec = spectral_shaping(best_seq, time_budget=alloc_s, verbose=True)
        obj_spec = compute_obj(a_spec)
        marker = " **" if obj_spec < best_score else ""
        print(f"  Spectral: obj={obj_spec:.8f}{marker}")
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()

    # Targeted mass transfer
    remaining = deadline - time.monotonic()
    alloc_t = min(remaining * 0.08, 60.0)
    if alloc_t > 10:
        a_targ = targeted_mass_transfer(best_seq, time_budget=alloc_t, verbose=True)
        obj_targ = compute_obj(a_targ)
        marker = " **" if obj_targ < best_score else ""
        print(f"  Targeted: obj={obj_targ:.8f}{marker}")
        if obj_targ < best_score:
            best_score = obj_targ
            best_seq = a_targ.copy()

    # === Phase 5: Diverse restarts (10%) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.10
    n_use = len(best_seq)
    n_restarts = 4
    per_restart = p5_budget / max(1, n_restarts)

    for r in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(per_restart, remaining * 0.2)
        if alloc < 20:
            break

        mode = (r * 3 + 1) % 8
        a_r = diverse_init(n_use, rng, mode=mode)
        # Blend with best
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        # Quick smooth + LP
        half = alloc / 2
        if half > 10:
            a_r = smooth_optimize(a_r, time_budget=half, beta_start=30, beta_end=1000)
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.1, 40.0)
        if lp_alloc > 15:
            a_r = lp_alternate(a_r, time_budget=lp_alloc, verbose=False)

        obj_r = compute_obj(a_r)
        marker = " **" if obj_r < best_score else ""
        print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}{marker}")
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 6: Final polish (10%) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP alternating polish
    alloc = min(remaining * 0.4, 80.0)
    if alloc > 20:
        a_final = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.3, 40.0)
    if alloc > 10:
        a_cd = coordinate_descent(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more LP round
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternate(best_seq, time_budget=min(remaining * 0.5, 30.0), verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    print(f"=== Target: 1.5027, Achieved: {best_score:.8f}, Gap: {best_score - 1.5027:.8f} ===")

    return result
