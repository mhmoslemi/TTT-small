def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smoothed_obj_and_grad(a, tau):
    """
    Log-sum-exp smoothed objective and gradient.
    F(a) = (n/(sum(a)^2)) * (1/tau) * log(sum_k exp(tau * conv_k(a)/sum(a)^2))
    Actually, let's work with normalized a (sum=1) for simplicity.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, np.zeros(n)
    a_norm = a / s
    conv = compute_conv(a_norm)
    max_conv = float(np.max(conv))
    
    # Softmax weights
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    F = (1.0 / tau) * float(np.log(np.sum(exp_vals))) + max_conv
    
    # Gradient: dF/da[i] = 2 * sum_k p[k] * a_norm[k-i mod ...]
    # More precisely: d(conv[k])/d(a_norm[i]) = a_norm[k-i] + a_norm[i-k] (for valid indices)
    # But since conv is symmetric, d(conv[k])/da[i] = 2*a_norm[k-i] for k != n-1, and 2*a_norm[n-1-i] for k=n-1
    # Actually for linear conv: d(conv[k])/da[i] = a_norm[k-i] if 0<=k-i<n, else 0
    # Plus the a_norm[i] term... wait no.
    
    # conv[k] = sum_j a[j]*a[k-j] (a defined as 0 outside [0,n-1])
    # d(conv[k])/da[i] = a[k-i] (if 0 <= k-i <= n-1) + a[i-k] (if 0 <= i-k <= n-1)
    # But since conv is a convolution (not correlation), and a is the same on both sides:
    # d(conv[k])/da[i] = a[k-i] + a[k-i] = 2*a[k-i] (if 0 <= k-i <= n-1)
    # Wait no. conv[k] = sum_j a[j]*a[k-j]. d/d(a[i]) = a[k-i] (from the j=i term in first factor)
    # + a[k-i] (from the j=k-i term in second factor, i.e., a[k-j] with j = k-i)
    # But only if both i and k-i are in [0, n-1].
    # Actually: d/d(a[i]) sum_j a[j]*a[k-j] = a[k-i]*1_{0<=k-i<=n-1} + a[k-i]*1_{0<=k-i<=n-1}
    # Wait, that's the same term counted twice only if i = k-i, i.e., k = 2i.
    # Let me be more careful:
    # conv[k] = sum_j a[j] * a[k-j]
    # d/d(a[i]) conv[k] = sum_j [delta_{j,i}*a[k-j] + a[j]*delta_{k-j,i}]
    # = a[k-i] (if 0 <= k-i <= n-1) + a[k-i] (if 0 <= k-i <= n-1)
    # = 2*a[k-i] if 0 <= k-i <= n-1, else 0
    
    # Hmm wait, that's not right either. Let me redo:
    # conv[k] = sum_j a[j] * a[k-j]
    # The derivative w.r.t. a[i]:
    # = sum_j [delta_{j,i} * a[k-j] + a[j] * delta_{k-j,i}]
    # = a[k-i] * 1_{0<=k-i<=n-1} + a[k-i] * 1_{0<=k-i<=n-1}
    # Wait, delta_{k-j,i} means k-j=i, so j=k-i. The term is a[k-i] * 1_{0<=k-i<=n-1}.
    # And delta_{j,i} means j=i, giving a[k-i] * 1_{0<=k-i<=n-1}.
    # So total: 2*a[k-i] * 1_{0<=k-i<=n-1}
    # EXCEPT when i = k-i (i.e., k=2i), in which case it's a[i]^2 and the derivative is 2*a[i] (not 2*2*a[i]).
    # Wait no: if k=2i, then both terms give a[i], so the derivative is a[i] + a[i] = 2*a[i]. That's the same formula.
    # So d(conv[k])/da[i] = 2*a[k-i] if 0 <= k-i <= n-1, else 0.
    
    # Grad F w.r.t. a[i] (a_norm[i]):
    # = 2 * sum_k p[k] * a_norm[k-i] * 1_{0<=k-i<=n-1} / sum(a_norm) ... 
    # Actually since we're working with a_norm = a/sum(a), the gradient w.r.t. a_norm is:
    # dF/d(a_norm[i]) = 2 * sum_k p[k] * a_norm[k-i] (for 0<=k-i<=n-1)
    
    # Use FFT: g[i] = 2 * sum_k p[k] * a_norm[k-i]
    # This is the cross-correlation of p and a_norm!
    # g = 2 * ifft(conj(fft(p, N)) * fft(a_norm, N)).real[:n]
    # Wait, cross-correlation: (p ⋆ a)[i] = sum_k p[k] * a[k-i]
    # In FFT: ifft(fft(p, N) * conj(fft(a, N))).real[:n] gives sum_k p[k] * conj(a)[k-i]
    # For real a: = sum_k p[k] * a[k-i]
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    P = fft(p, N)
    A = fft(a_norm, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    
    return 2.0 * n * F, g

def lbfgs_optimize(a, time_budget=200.0, tau_init=10.0, tau_end=800.0):
    """
    L-BFGS optimization with annealed log-sum-exp smoothing.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 1.0  # mixing parameter for annealing
    tau = tau_init
    stall_count = 0
    
    def fun_and_grad(x):
        nonlocal tau, best_obj, best_a, stall_count
        x = np.maximum(x, 1e-15)
        sx = float(np.sum(x))
        if sx < 1e-10:
            return 1e10, np.zeros(n)
        x = x / sx
        f, g = smoothed_obj_and_grad(x, tau)
        return f, g
    
    it = 0
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)
        
        # L-BFGS with the smoothed objective
        res = minimize(fun_and_grad, a, method='L-BFGS-B', 
                      bounds=[(0, None)] * n,
                      options={'maxiter': 50, 'maxfun': 200, 'ftol': 1e-12, 'gtol': 1e-8})
        
        if res.x is not None:
            a_new = np.maximum(res.x, 0)
            sx = float(np.sum(a_new))
            if sx > 1e-10:
                a_new = a_new / sx
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    stall_count = 0
                else:
                    stall_count += 1
                    if stall_count > 5:
                        a = best_a * np.exp(np.random.default_rng(it*7).normal(0, 0.05, n))
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        stall_count = 0
        
        it += 1
        if time.monotonic() - start > time_budget:
            break
    
    return best_a

def mirror_descent(a, time_budget=100.0, K=10):
    """
    Mirror descent (multiplicative weight updates) with peak-aware weighting.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / np.sqrt(n)
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Weight peaks by importance (top K)
        sorted_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[sorted_idx]
        
        # Build gradient: for each peak k, push a in the direction that reduces conv[k]
        # d(conv[k])/da[i] = 2*a[k-i] (for valid k-i)
        # So to reduce conv[k], we should decrease a[i] where a[k-i] is large
        # Equivalently, the subgradient at peak k is: g[i] += 2*a[k-i] for valid i
        
        g = np.zeros(n)
        for k in sorted_idx:
            k = int(k)
            # d(conv[k])/da[i] = 2*a[k-i] if 0 <= k-i <= n-1
            i_min = max(0, k - (n-1))
            i_max = min(n-1, k)
            if i_min <= i_max:
                # a[k-i] for i in [i_min, i_max] means a[k-i_min] ... a[k-i_max]
                # which is a[k : k-i_min-1 : -1] ... 
                # a[k-i] for i from i_min to i_max: indices k-i_min down to k-i_max
                g[i_min:i_max+1] += 2.0 * a[k-i_max:k-i_min+1][::-1]
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        # Multiplicative update: a_new = a * exp(-lr * g), then normalize
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 5.0 / np.sqrt(n))
        else:
            lr = max(lr * 0.95, 0.01 / np.sqrt(n))
        
        it += 1
    
    return best_a

def build_hankel_matrix(a, k):
    """
    Build the matrix Q_k such that a^T Q_k a = conv(a,a)[k].
    conv[k] = sum_j a[j]*a[k-j]
    This is a^T H_k a where H_k is a Hankel matrix with H_k[j, k-j] = 1/2 for valid indices.
    Actually: a^T H a = sum_{i,j} a[i] H[i,j] a[j]
    We want: sum_j a[j] a[k-j] = sum_i a[i] a[k-i]
    So H_k[i, k-i] = 1 (for valid k-i in [0,n-1]), and by symmetry H_k[k-i, i] = 1.
    But then a^T H_k a = 2*sum_i a[i]*a[k-i] (counting each pair twice except when i = k-i).
    So we need H_k[i, k-i] = 1/2 (and H_k[k-i, i] = 1/2) to get the correct value.
    Actually, let's just use: conv[k] = sum_i a[i] * a[k-i] * 1_{0<=k-i<n}
    This can be written as a^T B_k a where B_k[i,j] = 1 if i+j = k, else 0.
    Then a^T B_k a = sum_{i,j} a[i] B_k[i,j] a[j] = sum_i a[i] a[k-i] (for valid k-i).
    That's exactly what we want!
    """
    n = len(a)
    B = np.zeros((n, n))
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            B[i, j] = 1.0
    return B

def solve_lp_improved(a, top_k=20):
    """
    Improved LP: minimize sum of top-k peaks + lambda * max peak.
    This is harder than just min max, but might give better results.
    Actually, let's just do min M with an additional slack variable approach.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build A matrix: A[k, j] = a_norm[k-j] if 0 <= k-j < n, else 0
    # Then A @ g = conv(a_norm, g)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # A[k, j] = a_norm[k-j] for valid k-j
        k_min = j
        k_max = j + n - 1
        A[k_min:k_max+1, j] = a_norm[:n]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

def lp_polish(a, time_budget=100.0):
    """LP polishing with adaptive blending."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    iter_count = 0
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_improved(a)
        if g_lp is None:
            break
        
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        iter_count += 1
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
        else:
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if iter_count > 20:
                break
    
    return best_a

def supersample(a, factor=2):
    """Insert zeros between elements and slightly widen."""
    n = len(a)
    a_new = np.zeros(n * factor)
    a_new[::factor] = a
    # Slightly widen to avoid zeros
    a_new = np.maximum(a_new, 1e-10)
    return a_new

def downsample(a, factor=2):
    """Average blocks to reduce length."""
    n = len(a)
    new_n = n // factor
    a_new = np.zeros(new_n)
    for i in range(new_n):
        a_new[i] = a[i*factor:(i+1)*factor].mean()
    return a_new

def coarse_to_fine(a_short, target_n, budget_per_scale=30.0):
    """
    Multiscale optimization: start from short, gradually increase length.
    """
    n = len(a_short)
    while n < target_n:
        n = min(n * 2, target_n)
        # Upsample by inserting zeros
        a_up = np.zeros(n)
        a_up[0:len(a_short)] = a_short  # Not quite right, let's do it properly
        # Actually, let's just resample
        x_old = np.linspace(0, 1, len(a_short))
        x_new = np.linspace(0, 1, n)
        a_up = np.interp(x_new, x_old, a_short)
        a_up = np.maximum(a_up, 0)
        a_up = a_up / a_up.sum()
        
        # Optimize at this scale
        a_up = lbfgs_optimize(a_up, time_budget=budget_per_scale, tau_init=10, tau_end=200)
        a_short = a_up
    
    # Final scale
    if len(a_short) < target_n:
        x_old = np.linspace(0, 1, len(a_short))
        x_new = np.linspace(0, 1, target_n)
        a_short = np.interp(x_new, x_old, a_short)
        a_short = np.maximum(a_short, 0)
        a_short = a_short / a_short.sum()
    
    return a_short

def freq_perturb(a, rng, strength=0.1):
    """Perturb in frequency domain to create structured diversification."""
    n = len(a)
    A = fft(a, n)
    
    # Add random phase perturbation to non-DC components
    phase = rng.normal(0, strength, n)
    phase[0] = 0  # Keep DC
    A_pert = A * np.exp(1j * phase)
    
    a_pert = np.real(ifft(A_pert, n))
    a_pert = np.maximum(a_pert, 0)
    s = float(np.sum(a_pert))
    if s > 1e-10:
        a_pert = a_pert / s
    else:
        a_pert = a.copy()
    
    return a_pert

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_obj_and_grad(a, tau):
    """
    Log-sum-exp smoothed objective and gradient for normalized a (sum=1).
    Returns (2*n*F, gradient) where F = (1/tau)*log(sum_k exp(tau*conv_k)).
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, np.zeros(n)
    a_n = a / s
    conv = compute_conv(a_n)
    max_conv = float(np.max(conv))

    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    F = (1.0 / tau) * float(np.log(np.sum(exp_vals))) + max_conv

    N = 1
    while N < 3 * n:
        N <<= 1

    P = fft(p, N)
    A = fft(a_n, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]

    return 2.0 * n * F, g


def lbfgs_single_tau(a, tau, maxiter=80):
    """Run L-BFGS at a fixed tau value. Returns optimized sequence."""
    n = len(a)

    def fun_grad(x):
        x = np.maximum(x, 1e-15)
        sx = float(np.sum(x))
        if sx < 1e-10:
            return 1e10, np.zeros(n)
        x = x / sx
        f, g = smoothed_obj_and_grad(x, tau)
        return f, g

    res = minimize(fun_grad, a, method='L-BFGS-B',
                   bounds=[(0, None)] * n,
                   options={'maxiter': maxiter, 'maxfun': 500, 'ftol': 1e-14, 'gtol': 1e-10})
    if res.x is not None:
        x = np.maximum(res.x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            return x / sx
    return a.copy()


def lbfgs_anneal(a, time_budget=200.0, taus=None):
    """
    L-BFGS with annealed tau: multiple calls at increasing tau values.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)

    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        a_new = lbfgs_single_tau(a, tau, maxiter=60)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        # Keep a as the L-BFGS result even if obj didn't improve (for smoothing)

    return best_a


def mirror_descent_steps(a, time_budget=80.0, K=15):
    """
    Multiplicative weight (mirror) descent with peak-aware subgradient.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.5 / np.sqrt(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        # Top-K peaks
        top_k = min(K, len(conv))
        top_idx = np.argsort(conv)[::-1][:top_k]

        # Build subgradient
        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            if i_min <= i_max:
                # d(conv[k])/da[i] = 2*a[k-i] for i in [i_min, i_max]
                # a[k-i] for i=i_min gives a[k-i_min], for i=i_max gives a[k-i_max]
                # So indices in a are from k-i_max to k-i_min (descending)
                idx = np.arange(k - i_max, k - i_min + 1)
                g[i_min:i_max + 1] += 2.0 * a[idx][::-1]

        # Normalize and apply multiplicative update
        g_max = float(np.max(np.abs(g)))
        if g_max > 1e-15:
            g = g / g_max
            a_new = a * np.exp(-lr * g)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()

            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                if stall > 50:
                    lr *= 0.7
                    stall = 0
                if lr < 0.001 / np.sqrt(n):
                    break
        else:
            break

    return best_a


def coordinate_descent(a, time_budget=60.0):
    """
    Coordinate descent: optimize one element at a time.
    For each element a[i], the objective as a function of a[i] is a piecewise
    quadratic (since conv[k] is linear in a[i] for each k).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Precompute: for each i, the "other" contributions to each conv[k]
    # conv[k] = sum_j a[j]*a[k-j]
    # When we vary a[i], the terms involving a[i] are:
    #   a[i]*a[k-i] for valid k-i (from the j=i term)
    #   a[i]*a[k-i] for valid k-i (from the k-j=i term, i.e., j=k-i)
    # So conv[k] = base[k] + 2*a[i]*a[k-i] (for k != 2i)
    #              = base[k] + a[i]^2 (for k = 2i)
    # where base[k] = conv[k] - (terms involving a[i])

    it = 0
    improved_total = True
    while improved_total and time.monotonic() - start < time_budget:
        improved_total = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            # Save current value
            old_val = a[i]
            a[i] = 0.0
            base_conv = compute_conv(a)
            # Now conv[k] = base_conv[k] + 2*val*a[k-i] (for valid k-i)
            # plus val^2 if k = 2i

            # The objective is 2n * max_k (base_conv[k] + 2*val*a[k-i]*1_{0<=k-i<n} + val^2*1_{k=2i})
            # We want to minimize this over val >= 0.
            # This is equivalent to minimizing max_k (base_conv[k] + c_k*val + d_k*val^2)
            # where c_k = 2*a[k-i] (if valid) and d_k = 1 if k=2i else 0.

            # Binary search for optimal val
            # The function f(val) = max_k g_k(val) is convex in val >= 0.
            # Use ternary search or bisection on derivative.

            # First find upper bound: val unlikely to exceed 2*old_val or 1/n
            lo = 0.0
            hi = max(2.0 * old_val, 2.0 / n)

            # Quick bisection on the argmax to find min
            # f(val) is convex, so minimum is where the "active" constraints cross.
            # Simple approach: grid search + refine
            candidates = [0.0, old_val * 0.5, old_val, old_val * 1.5, hi]
            best_val = old_val
            best_f = np.inf

            for cv in candidates:
                a[i] = cv
                s_chk = float(np.sum(a))
                if s_chk < 1e-10:
                    a[i] = old_val
                    break
                # Compute max conv for this val
                # conv[k] = base_conv[k] + 2*cv*a[k-i] (valid) + cv^2 (if k=2i)
                conv_est = base_conv.copy()
                k_min = max(0, i)
                k_max = min(2 * n - 2, 2 * i)
                # 2*cv*a[k-i] for k where 0 <= k-i < n
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                if k_lo <= k_hi:
                    conv_est[k_lo:k_hi + 1] += 2.0 * cv * a[max(0, k_lo - i):k_hi - i + 1]
                # cv^2 at k=2i
                if 2 * i < 2 * n - 1:
                    conv_est[2 * i] += cv * cv

                f_val = float(np.max(conv_est))
                if f_val < best_f:
                    best_f = f_val
                    best_val = cv

            # Refine with a few more points around best_val
            step = max(1e-6, 0.1 * best_val)
            for frac in [-0.5, -0.2, 0.0, 0.2, 0.5]:
                cv = max(0, best_val + frac * step)
                conv_est = base_conv.copy()
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                if k_lo <= k_hi:
                    conv_est[k_lo:k_hi + 1] += 2.0 * cv * a[max(0, k_lo - i):k_hi - i + 1]
                if 2 * i < 2 * n - 1:
                    conv_est[2 * i] += cv * cv
                f_val = float(np.max(conv_est))
                if f_val < best_f:
                    best_f = f_val
                    best_val = cv

            a[i] = best_val
            s_new = float(np.sum(a))
            if s_new < 1e-10:
                a[i] = old_val
            else:
                a = a / s_new
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved_total = True

        it += 1

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[:n]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_topk(a, K=10):
    """
    LP that minimizes the sum of the top-K peaks plus the max peak.
    This gives a flatter peak distribution than just minimizing max.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    # Auxiliary variables for each constraint (slack)
    slacks = cp.Variable(2 * n - 1, nonneg=True)

    constraints = [A @ g + slacks >= 0, cp.sum(g) == 1]
    # conv(a,g) - slacks = 0 means slacks = conv(a,g) (elementwise, nonneg)
    # We want to minimize sum of top K of conv + max(conv)
    # Simplification: minimize t + mu * sum(slacks) where t >= conv[k] for all k
    # Actually let's just do: minimize M + 0.1 * sum(slacks)
    # where M >= conv[k] and slacks[k] = conv[k] (enforced by conv - slacks = 0, slacks >= 0)

    # Simpler: just minimize M + lambda * (sum of conv)
    # sum(conv) = (sum a)(sum g) = sum(a) since sum(g)=1... actually sum(conv(a,g)) = sum(a)*sum(g) = sum(a)
    # So that's constant. Let's just do regular LP.

    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
