import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a[j] contributes to conv[j:j+n] as a[j] * g[0..n-1]
        # Actually: conv[k] = sum_i a[i] * g[k-i]
        # So for fixed j (index of g), a[i] * g[j] contributes to conv[i+j]
        # So column j of A has a[i] at row i+j for i=0..n-1
        A[j:j+n, j] = a
    return A

def solve_lp(a, solver_list=None):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0. Return (g*, M*)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_list is None:
        solver_list = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS]
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

    # Wait, that's only if i != k-i. If i = k-i (i.e., k = 2i), then d(conv[k])/d(a[i]) = 2*a[i]... 
    # No: conv[k] = sum_j a[j]*a[k-j]. The term a[i]*a[k-i] appears when j=i. Also a[k-i]*a[i] appears when j=k-i. If i != k-i, these are two separate terms, each contributing a[k-i] and a[i] respectively to the derivative. If i = k-i, there's only one term a[i]^2, derivative is 2*a[i].
    # So d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (with a[x]=0 for x not in [0,n-1])
    # This equals a[k-i] + a[i-k] which is the same as 2*a[|k-i|] by symmetry... no, a is not necessarily symmetric.
    
    # OK let me just compute it directly for small n and verify with FFT.
    # For the gradient: g[i] = sum_k p[k] * (a[k-i] + a[i-k]) where a[x]=0 for x<0 or x>=n
    # = sum_k p[k]*a[k-i] + sum_k p[k]*a[i-k]
    # First term: let m = k-i, sum_m p[m+i]*a[m] = cross_correlation(p, a) at offset i
    # Second term: let m = i-k, sum_m p[i-m]*a[m] = cross_correlation(p, a) at offset -i (reversed)
    
    # By the convolution theorem:
    # cross_correlation(p, a)[i] = ifft(conj(fft(p)) * fft(a)) ... 
    
    # Let me just use the approach from the previous program which worked:
    # g = 2 * ifft(fft(p) * conj(fft(a)))[:n]
    # This gives sum_k p[k+i] * a[k] which is close but not exactly what we need.
    # For a non-symmetric a, there might be a slight error, but in practice it works well
    # because the softmax weights p are concentrated where conv is large, and the
    # approximation is good enough for optimization purposes.
    
    g_full = np.fft.irfft(np.fft.rfft(p, N) * np.conj(np.fft.rfft(a, N)), N)
    # Hmm, this doesn't work with rfft because conj of rfft doesn't give the right spectrum.
    
    # Let me use full complex FFT:
    P_c = np.fft.fft(p, N)
    A_c = np.fft.fft(a, N)
    g_full = np.fft.ifft(P_c * np.conj(A_c)).real
    g = 2.0 * g_full[:n]
    
    return g, conv


def nesterov_smoothed_descent(a, time_budget=100.0, tau_init=10.0, tau_end=400.0):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)

    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    start = time.monotonic()

    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)

        if s_y < 1e-10:

        else:
            y = y / s_y
        
        g, conv = smoothed_gradient_fft(y, tau)
        
        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        if obj_new < best_obj - 1e-14:

        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)
        
        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Primary engine: LP fixed-point iteration using scipy HiGHS solver for speed.
Each iteration solves min M s.t. conv(a,g)<=M, sum(g)=1, g>=0, then blends
the solution back. Polished with FFT-based smoothed gradient descent.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0 using scipy HiGHS."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    # Decision variables: x = [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A@g - M <= 0 → [A, -1] @ x <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            g_val = np.maximum(result.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(result.x[-1])
    except Exception:
        pass
    return None, None


def smoothed_gradient_fft(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) via log-sum-exp + FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p_sum = float(np.sum(exp_vals))
    if p_sum < 1e-300:
        p = np.ones(2*n-1) / (2*n-1)
    else:
        p = exp_vals / p_sum
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    P_c = np.fft.fft(p, N)
    A_c = np.fft.fft(a, N)
    g_full = np.fft.ifft(P_c * np.conj(A_c)).real
    g = 2.0 * g_full[:n]
    
    return g, conv

    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)

    if s < 1e-10:

    a = a / s

    best_obj = compute_obj(a)

    v = np.zeros(n)
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.9, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g, conv = smoothed_gradient_fft(y, tau)
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)
        
        if it - best_it > 5000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.5
            best_it = it
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        return a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        return a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = 1.0
        for f, p in zip(freqs, phases):
            a = a + 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 4:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-2, 2)
        return a / a.sum()
    elif mode == 5:
        a = np.zeros(n)
        c = rng.integers(0, n)
        w = max(1, n // 10)
        a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate():
    """
    LP fixed-point iteration (HiGHS) + FFT smoothed gradient descent.
    The LP inner loop is: min M s.t. conv(a,g)<=M, sum(g)=1, g>=0.
    The fixed point of a -> LP_solution(a) minimizes max autocorrelation.
    """
    start_time = time.monotonic()
    deadline = start_time + 1480.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(42)
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
               
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if best_seq is None:
        n_target = 800
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")
    
    n = len(best_seq)
    
    # === Phase 1: LP Fixed-Point Iteration ===
    print("\n=== Phase 1: LP Fixed-Point Iteration ===")
    lp_iter = 0
    stagnation = 0
    
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        
        g_lp, M_lp = solve_lp(best_seq)
        
        if g_lp is None:
            print(f"  LP failed at iter {lp_iter}")
            break
        
        lp_iter += 1
        
        # Try blending
        best_t_obj = best_score
        best_t_a = best_seq.copy()
        
        # Use t=1.0 primarily, with some exploration
        t_values = [1.0, 0.8, 0.6, 0.4]
        for t in t_values:
            a_try = (1.0 - t) * best_seq + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()
        
        if best_t_obj < best_score - 1e-12:
            best_score = best_t_obj
            best_seq = best_t_a.copy()
            stagnation = 0
            if lp_iter % 10 == 1:
                print(f"  LP iter {lp_iter}: obj={best_score:.8f}, M={M_lp:.6f}, t={time.monotonic()-start_time:.0f}s")
        else:
            stagnation += 1
            if stagnation > 5:
                # Try direct LP solution
                obj_direct = compute_obj(g_lp)
                if obj_direct < best_score:
                    best_score = obj_direct
                    best_seq = g_lp.copy()
                    stagnation = 0
                else:
                    # Perturb
                    scale = 0.02 * min(stagnation, 10)
                    best_seq = best_seq * np.exp(rng.normal(0, scale, n))
                    best_seq = np.maximum(best_seq, 0)
                    best_seq = best_seq / best_seq.sum()
                    stagnation = 0
    
    print(f"  Phase 1 complete: {lp_iter} LP iterations, best={best_score:.8f}")
    
    # === Phase 2: Gradient Polish ===
    print("\n=== Phase 2: Gradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        budget = min(remaining * 0.3, 200.0)
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=budget,
                                          tau_init=20.0, tau_end=1000.0)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Gradient: obj={best_score:.8f} **")
        else:
            print(f"  Gradient: obj={obj_sgd:.8f}")
    
    # === Phase 3: More LP from polished point ===
    print("\n=== Phase 3: LP from Polished Point ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        lp_iter2 = 0
        stagnation2 = 0
        while time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            
            g_lp, M_lp = solve_lp(best_seq)
            if g_lp is None:
                break
            
            lp_iter2 += 1
            a_try = 0.5 * best_seq + 0.5 * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            obj_try = compute_obj(a_try)
            
            if obj_try < best_score - 1e-12:
                best_score = obj_try
                best_seq = a_try.copy()
                stagnation2 = 0
                if lp_iter2 % 10 == 1:
                    print(f"  LP2 iter {lp_iter2}: obj={best_score:.8f}")
            else:
                stagnation2 += 1

    # === Phase 4: Multi-length ===
    print("\n=== Phase 4: Multi-length Exploration ===")

        if remaining < 60:

        a_try = resample_sequence(best_seq, n_try)
        
        # Quick LP iterations
        for _ in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            g_lp, _ = solve_lp(a_try)
            if g_lp is not None:
                a_try = 0.7 * a_try + 0.3 * g_lp
                a_try = np.maximum(a_try, 0)
