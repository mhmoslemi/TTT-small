`∂f/∂a[i] = 2n * (F_β'[i] - 2*F_β)`

But wait, this ignores the constraint that S should remain 1. The proper way is to project onto the simplex `{a >= 0, sum a = 1}`.

Actually, since the objective is scale-invariant, we can just:
1. Compute gradient of `F_β(a)` (not the full objective)
2. Subtract mean to enforce `sum = 0` (tangent to the level set of S)
3. Take a step
4. Project onto `a >= 0`
5. Re-normalize to `sum = 1`

This is like doing gradient descent on the sphere (well, simplex).

Let me code this up. The key computation per iteration:
1. Compute `c = conv(a, a)` (length 2n-1) via FFT: O(n log n)
2. Compute `w = exp(β * c)` (vector exp): O(n)
3. Compute `F_β = (1/β) log(sum(w))`: O(n)
4. Compute gradient via two FFT convolutions: O(n log n)
5. Adam update + projection: O(n)

Total per iteration: O(n log n). For n = 800, this is very fast. We can do millions of iterations in 300 seconds.

Let me also think about the coordinate descent approach. For a fixed coordinate i, the objective is:
`f(a) = 2n * max_k (c_k + a_i * (a[k-i] + a[i-k])) / S^2`

where `c_k = conv(a,a)[k] - a_i * (a[k-i] + a[i-k])` (the convolution without the terms involving a_i).

Wait, this is getting complicated. Let me just use the pairwise exchange approach for fine-tuning.

Let me now think about what's different from the previous approach that could give a better result:

1. **Soft-max gradient with Adam**: This is a completely different optimization paradigm. The LP approach finds the best "direction" at each step, but it's limited by the LP structure. The gradient approach can navigate the smooth landscape more efficiently.

2. **Adaptive β schedule**: Start with small β (smooth approximation) to navigate the landscape, then increase β to sharpen towards the true max.

3. **Multiple scales**: Optimize at different levels of granularity (coarse to fine).

4. **Better restarts**: Instead of random perturbations, use structured perturbations based on the gradient landscape.

"""
Hybrid optimizer combining:
1. LP alternating (proven baseline)
2. Soft-max gradient descent with Adam (new)
3. Coordinate/pairwise polish
4. Multi-n exploration
5. Structured restarts
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    n = len(a)
    A = build_conv_matrix(a)
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    prob = cp.Problem(cp.Minimize(M), [A @ z <= M, cp.sum(z) == 1])
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    return z_val, float(M.value)
        except:
            continue
    return None, None


def soft_max_grad(a, beta):
    """
    Compute soft-max of conv(a,a) and its gradient.
    Returns (F_beta, grad) where grad = dF_beta/d(a) (before normalization).
    """
    n = len(a)
    conv = compute_conv(a)  # length 2n-1
    w = np.exp(beta * conv - beta * np.max(conv))  # shifted for numerical stability
    sw = float(np.sum(w))
    F_beta = (1.0 / beta) * (np.log(sw) + beta * np.max(conv))
    
    # grad[i] = sum_k (w[k]/sw) * (a[k-i] + a[i-k])
    # cross2[i] = sum_k w[k] * a[i-k] = conv(w, a)[i] for i=0..n-1
    # cross1[i] = sum_k w[k] * a[k-i] = conv(a, w_flip)[2n-2-i] for i=0..n-1
    
    # cross2: first n elements of conv(w, a)
    size = 1
    while size < (2*n-1) + n - 1:
        size <<= 1
    fw = rfft(w, size)
    fa = rfft(a, size)
    cross2_full = irfft(fw * fa, size)
    cross2 = cross2_full[:n]  # positions 0 to n-1
    
    # cross1: conv(a, w_flip)[2n-2-i] for i=0..n-1
    w_flip = w[::-1]  # length 2n-1, reversed
    size2 = 1
    while size2 < n + (2*n-1) - 1:
        size2 <<= 1
    fa2 = rfft(a, size2)
    fwf = rfft(w_flip, size2)
    cross1_full = irfft(fa2 * fwf, size2)
    # Need positions 2n-2-i for i=0..n-1, i.e., positions n-1 to 2n-2
    cross1 = cross1_full[n-1:2*n-1][::-1]  # position 2n-2 down to n-1
    
    grad = (cross1 + cross2) / sw
    return F_beta, grad


def soft_max_gradient_descent(a, time_budget=300.0, verbose=True):
    """
    Adam-based gradient descent on soft-max approximation of the objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Adaptive beta schedule
    beta = 5.0  # start smooth
    
    stall = 0
    last_improve = start
    
    while time.monotonic() - start < time_budget:
        # Increase beta over time (sharpen the approximation)
        elapsed = time.monotonic() - start
        beta = 5.0 + 20.0 * min(1.0, elapsed / time_budget)
        
        F_beta, grad = soft_max_grad(a, beta)
        
        # Subtract mean to stay on the "sum=1" manifold (approximately)
        grad_centered = grad - np.mean(grad)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_centered
        v = beta2 * v + (1 - beta2) * grad_centered ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            last_improve = time.monotonic()
            # Decay learning rate slowly
            lr = max(0.001, lr * 0.9999)
        else:
            a = a_new  # still move (helps exploration)
            stall += 1
            if stall > 100:
                # Reschedule
                lr = 0.01
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                # Small perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
        
        if verbose and int((time.monotonic() - start) * 2) % 20 == 0:
            print(f"    SM-GD: obj={best_obj:.8f}, beta={beta:.1f}, t={time.monotonic()-start:.0f}s")
    
    return best_a


def lp_alternating(a, time_budget=200.0, verbose=False):
    """LP alternating with cycle breaking."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    prev_a = None
    
    while time.monotonic() - start < time_budget:
        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, _ = result
        
        # Cycle detection
        if prev_a is not None and np.max(np.abs(a - prev_a)) < 1e-8:
            a_break = 0.5 * (a + prev_a)
            a_break = np.maximum(a_break, 0)
            sb = float(np.sum(a_break))
            if sb > 1e-10:
                a_break = a_break / sb
            obj_break = compute_obj(a_break)
            if obj_break < best_obj:
                best_obj = obj_break
                best_a = a_break.copy()
            a = a_break
            prev_a = None
            continue
        
        prev_a = a.copy()
        
        for mix in [0.5, 0.7, 1.0]:
            a_try = (1 - mix) * a + mix * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st > 1e-10:
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    if verbose:
                        print(f"    LP-alt: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
                    break
    
    return best_a


def pairwise_polish(a, time_budget=60.0):
    """Pairwise exchange polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    no_improve = 0
    while time.monotonic() - start < time_budget:
        improved = False
        for _ in range(50):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for frac in [0.05, 0.1, 0.15, 0.25]:
                delta = frac * min(a[i], a[j])
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0:
                    continue
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    no_improve = 0
                    break
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
    
    return best_a

    Coordinate descent: for each coordinate, find the optimal value given all others.
    For coordinate i, the objective is:
    2n * max_k (c_k + a_i*(a[k-i] + a[i-k])) / (S_old + a_i - a_i_old)^2

    """

    a = np.maximum(a, 0).astype(np.float64).copy()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    while time.monotonic() - start < time_budget:
        # Random coordinate order
        order = rng.permutation(n)
        improved = False
        for idx in order:
            i = int(idx)
            a_i_old = a[i]
            if a_i_old < 1e-15:
                continue

            # conv[a_i * x] where x contributes to positions k where a[k-i] or a[i-k] != 0
            # conv[a,a][k] = sum_j a[j]*a[k-j]
            # Terms involving a[i]: a[i]*a[k-i] (when j=i) + a[k-i]*a[i] (when k-j=i, j=k-i)
            # So terms = a[i] * (a[k-i] + a[k-i]) = 2*a[i]*a[k-i] when k-i is in [0,n-1]
            # Wait no: a[i]*a[k-i] when j=i, and a[j]*a[i] when k-j=i i.e. j=k-i
            # These are the same term! a[i]*a[k-i] = a[k-i]*a[i]
            # So the total contribution is: for each k where both i and k-i are in [0,n-1],
            # the term a[i]*a[k-i] appears in conv[k]. It appears once when j=i (giving a[i]*a[k-i])
            # and once when j=k-i (giving a[k-i]*a[i]). But these are the same product!

            # The term with j=k-i is a[k-i]*a[i]. These are different terms in the sum
            # (they're at different j values) but they have the same value.
            # So the total contribution of a[i] to conv[k] is:
            #   a[i]*a[k-i] (from j=i) + a[k-i]*a[i] (from j=k-i) = 2*a[i]*a[k-i]
            # UNLESS i = k-i, i.e., k = 2i, in which case it's just a[i]^2 (one term).
            
            # Actually wait, conv[k] = sum_j a[j]*a[k-j]
            # If j = i: term is a[i]*a[k-i] (valid if 0 <= k-i < n)
            # If j = k-i: term is a[k-i]*a[i] (valid if 0 <= k-i < n, same condition)
            # These are two separate terms in the sum (at j=i and j=k-i), unless i = k-i.
            # If i = k-i (k=2i), then j=i=k-i, so it's one term: a[i]*a[i] = a[i]^2.
            
            # So: contribution of a[i] to conv[k]:
            #   If k = 2i: a[i]^2
            #   If k != 2i and 0 <= k-i < n: 2*a[i]*a[k-i]
            #   Otherwise: 0
            
            # For the objective, we need max_k conv[k]. This is complex.
            # Let's just use a ternary search or golden section on a[i].
            
            # Simpler: try a few candidate values for a[i]
            S_rest = float(np.sum(a)) - a_i_old
            c = compute_conv(a)  # current full convolution
            # Remove a[i]'s contribution
            # This is expensive to do exactly, so let's just try candidate values
            
            candidates = [0.0, a_i_old * 0.5, a_i_old * 0.8, a_i_old * 1.2, a_i_old * 1.5,
                         a_i_old * 0.3, a_i_old * 0.7, a_i_old * 1.3]
            best_val = a_i_old
            best_obj_i = best_obj
            
            for val in candidates:
                a_try = a.copy()
                a_try[i] = val
                st = float(np.sum(a_try))
                if st < 1e-10:

                    best_a = a.copy()

Wait, this is O(n^2) per sweep (n coordinates × n candidates × O(n) to compute max). That's O(n^3) per sweep = 512M for n=800. Too slow.

"""
Hybrid optimizer combining LP alternating, soft-max gradient descent (Adam),
vectorized coordinate descent, multi-n exploration, and structured restarts
to minimize 2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. A@a_conv(z) <= M, sum(z)=1, z>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    prob = cp.Problem(cp.Minimize(M), [A @ z <= M, cp.sum(z) == 1])
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(prob.constraints[0].dual_value), 0)
                    except:
                        pass
                    return z_val, float(M.value), dual_y
        except:
            continue
    return None, None, None


def soft_max_grad(a, beta):
    """Compute soft-max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    c_max = float(np.max(conv))
    w = np.exp(beta * (conv - c_max))
    sw = float(np.sum(w))
    F_beta = c_max + (1.0 / beta) * np.log(sw)

    # grad[i] = sum_k (w[k]/sw) * (a[k-i] + a[i-k]) for valid indices
    # cross2[i] = sum_k w[k]*a[i-k] = conv(w,a)[i] for i=0..n-1
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    fw = rfft(w, size)
    fa = rfft(a, size)
    cross2_full = irfft(fw * fa, size)
    cross2 = cross2_full[:n]

    # cross1[i] = sum_k w[k]*a[k-i] = conv(a, w_flip)[2n-2-i]
    w_flip = w[::-1]
    size2 = 1
    while size2 < n + (2 * n - 1) - 1:
        size2 <<= 1
    fa2 = rfft(a, size2)
    fwf = rfft(w_flip, size2)
    cross1_full = irfft(fa2 * fwf, size2)
    # Position 2n-2-i for i=0..n-1 → positions 2n-2 down to n-1
    cross1 = cross1_full[n - 1:2 * n - 1][::-1]

    grad = (cross1 + cross2) / sw
    return F_beta, grad


def soft_max_gd(a, time_budget=250.0, verbose=True):
    """Adam-based gradient descent on soft-max approximation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.005
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta = 8.0
    stall = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        beta = 8.0 + 30.0 * min(1.0, elapsed / time_budget)

        F_beta, grad = soft_max_grad(a, beta)
        grad_centered = grad - np.mean(grad)

        t += 1
        m = beta1 * m + (1 - beta1) * grad_centered
        v = beta2 * v + (1 - beta2) * grad_centered ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a_new = a_new / s

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = max(0.0005, lr * 0.9998)
        else:
            a = a_new
            stall += 1
            if stall > 150:
                lr = 0.005
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0

        if verbose and int(elapsed) % 30 == 0 and int(elapsed) > int(elapsed - 0.5):
            print(f"    SM-GD: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.5f}, t={elapsed:.0f}s")

    return best_a


def lp_alternating(a, time_budget=200.0, verbose=False):
    """LP alternating with 2-cycle detection and adaptive mixing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    prev_a = None
    prev_prev_a = None
    g_momentum = a.copy()

    while time.monotonic() - start < time_budget:
        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp, dual_y = result

        if prev_prev_a is not None:
            if np.max(np.abs(a - prev_prev_a)) < 1e-7:
                a_break = 0.5 * (a + prev_a)
                a_break = np.maximum(a_break, 0)
                sb = float(np.sum(a_break))
                if sb > 1e-10:
                    a_break = a_break / sb
                obj_break = compute_obj(a_break)
                if obj_break < best_obj:
                    best_obj = obj_break
                    best_a = a_break.copy()
                a = a_break
                prev_a = None
                prev_prev_a = None
                g_momentum = a.copy()
                continue

        g_momentum = 0.5 * g_momentum + 0.5 * g_lp
        g_momentum = np.maximum(g_momentum, 0)
        gm_s = float(np.sum(g_momentum))
        if gm_s > 1e-10:
            g_momentum = g_momentum / gm_s

        targets = [g_lp, g_momentum]
        best_cand_obj = np.inf
        best_cand = a.copy()

        for target in targets:
            for mix in [0.3, 0.5, 0.7, 1.0]:
                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj:
                    best_cand_obj = obj_try
                    best_cand = a_try.copy()

        if best_cand_obj < best_obj - 1e-12:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            prev_prev_a = prev_a
            prev_a = a.copy()
            a = best_a
            if verbose:
                print(f"    LP-alt: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
        else:
            if prev_a is not None:
                pass
            prev_prev_a = prev_a
            prev_a = a.copy()
            # Small perturbation to break stagnation
            rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
            if dual_y is not None and np.max(dual_y) > 0:
                A = build_conv_matrix(a)
                active_idx = np.where(dual_y > 0.05 * np.max(dual_y))[0]
                if len(active_idx) > 0:
                    col_imp = np.sum(A[active_idx, :], axis=0)
                    pert = -np.sign(col_imp - np.mean(col_imp)) * 0.01
                    pert += rng.normal(0, 0.01, n)
                else:
                    pert = rng.normal(0, 0.02, n)
            else:
                pert = rng.normal(0, 0.02, n)
            a = best_a * np.exp(pert)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            prev_a = None
            prev_prev_a = None
            g_momentum = a.copy()

def fast_coordinate_descent(a, time_budget=150.0, verbose=False):
    """Vectorized coordinate descent using O(n) convolution update per coordinate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(a).copy()
    S = 1.0
    start = time.monotonic()
    rng = np.random.default_rng(42)

    factors = np.array([0.3, 0.6, 0.8, 1.25, 1.5, 2.0])
    no_improve_sweeps = 0

    while time.monotonic() - start < time_budget:
        order = rng.permutation(n)
        improved = False

        for idx in order:
            i = int(idx)
            old = a[i]
            if old < 1e-14:
                continue

            best_val = old
            best_obj_i = 2.0 * n * np.max(conv) / (S * S)

            for f in factors:
                val = old * f
                delta = val - old
                if abs(delta) < 1e-12:
                    continue

                # Update: conv[i:i+n] += 2*delta*a, conv[2i] += delta^2
                new_block = conv[i:i + n] + 2.0 * delta * a
                new_block[i] += delta * delta

                # Max outside affected range
                if i > 0:
                    max_left = np.max(conv[:i])
                else:
                    max_left = -np.inf
                if i + n < 2 * n - 1:
                    max_right = np.max(conv[i + n:])
                else:
                    max_right = -np.inf
                max_outside = max(max_left, max_right)
                new_max = max(max_outside, np.max(new_block))

                S_new = S + delta
                if S_new <= 1e-10:
                    continue
                obj_try = 2.0 * n * new_max / (S_new * S_new)
                if obj_try < best_obj_i:
                    best_obj_i = obj_try
                    best_val = val

            if best_val != old and best_obj_i < best_obj - 1e-14:
                delta = best_val - old
                conv[i:i + n] += 2.0 * delta * a
                conv[2 * i] += delta * delta
                a[i] = best_val
                S += delta
                best_obj = best_obj_i
                best_a = a.copy()
                improved = True

        if verbose:
            elapsed = time.monotonic() - start
            print(f"    CoordDesc: obj={best_obj:.8f}, t={elapsed:.0f}s")

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 10:
                break
        else:
            no_improve_sweeps = 0

    best_a = best_a / np.sum(best_a)
    best_obj = compute_obj(best_a)
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.001)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP alternating, soft-max gradient descent,
    vectorized coordinate descent, multi-n exploration, and structured restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        # Structured initialization
        best_init = None
        best_init_obj = np.inf
        for n_init in [400, 600, 800, 1000, 1200, 1400]:
            t = np.linspace(0, 1, n_init)
            shapes = [
                np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.1)), 0.01),
                np.sin(np.pi * t) ** 2 + 0.01,
                np.maximum(1.0 - 0.5 * np.cos(3 * np.pi * t), 0.01),
                np.maximum(np.exp(-2 * (t - 0.3) ** 2) + 0.3 * np.exp(-2 * (t - 0.7) ** 2), 0.01),
            ]
            for sh in shapes:
                a_init = sh / np.sum(sh)
                obj_init = compute_obj(a_init)
                if obj_init < best_init_obj:
                    best_init_obj = obj_init
                    best_init = a_init.copy()
        best_seq = best_init
        best_score = best_init_obj
        print(f"Initial (n={len(best_seq)}): obj={best_score:.8f}")

    a = best_seq.copy()
    n_parent = len(a)

    # === Phase 1: LP Alternating (250s) ===
    print(f"\n=== Phase 1: LP Alternating (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(250.0, remaining * 0.25)
    if alloc > 30:
        a_new = lp_alternating(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 1 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 done: obj={obj:.8f}")

    # === Phase 2: Soft-max Gradient Descent (250s) ===
    print(f"\n=== Phase 2: Soft-Max GD (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(250.0, remaining * 0.25)
    if alloc > 30:
        a_new = soft_max_gd(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 done: obj={obj:.8f}")

    # === Phase 3: Coordinate Descent (150s) ===
    print(f"\n=== Phase 3: Coord Descent (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(150.0, remaining * 0.2)
    if alloc > 20:
        a_new = fast_coordinate_descent(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 3 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 done: obj={obj:.8f}")

    # === Phase 4: LP + SM-GD + Coord cycle (200s) ===
    print(f"\n=== Phase 4: Combined Refinement ===")
    remaining = deadline - time.monotonic()
    alloc = min(200.0, remaining * 0.2)
    if alloc > 60:
        # LP
        sub_alloc = alloc * 0.4
        a_new = lp_alternating(a, time_budget=sub_alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  P4-LP: obj={best_score:.8f} **")

        # SM-GD
        remaining = deadline - time.monotonic()
        sub_alloc = min(alloc * 0.3, remaining * 0.3)
        if sub_alloc > 30:
            a_new = soft_max_gd(a, time_budget=sub_alloc, verbose=True)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  P4-SM: obj={best_score:.8f} **")

        # Coord
        remaining = deadline - time.monotonic()
        sub_alloc = min(alloc * 0.3, remaining * 0.3)
        if sub_alloc > 20:
            a_new = fast_coordinate_descent(a, time_budget=sub_alloc, verbose=True)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  P4-CD: obj={best_score:.8f} **")

    # === Phase 5: Multi-n Exploration (200s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_candidates = sorted(set([
        n_parent - 100, n_parent - 50, n_parent + 50, n_parent + 100,
        n_parent + 200, n_parent + 400, n_parent + 600,
        500, 700, 900, 1100, 1300, 1500,
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 3000 and n != n_parent]
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        a_try = resample_sequence(best_seq, n_try)
        # Quick optimization: short LP + coord
        sub_alloc = min(40.0, remaining * 0.15)
        a_opt = lp_alternating(a_try, time_budget=sub_alloc * 0.6)
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_opt = fast_coordinate_descent(a_opt, time_budget=min(20, remaining * 0.3))

        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Structured Restarts (200s) ===
    print(f"\n=== Phase 6: Structured Restarts ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_seq)
        max_restarts = min(8, int(remaining // 35))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if r % 4 == 0:
                # Smooth Gaussian
                a_r = best_seq * np.exp(rng.normal(0, 0.05, n_best))
            elif r % 4 == 1:
                # Block modulation
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 80)
                block_size = max(1, n_best // n_blocks)
                n_mod = min(4, n_blocks)
                blocks = rng.choice(n_blocks, n_mod, replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = min(si + block_size, n_best)
                    a_r[si:ei] *= rng.uniform(0.4, 1.6)
            elif r % 4 == 2:
                # Asymmetry push
                ramp = np.linspace(0.5, 1.5, n_best)
                a_r = best_seq * ramp
            else:
                # Sparse perturbation
                a_r = best_seq.copy()
                n_sparse = max(5, n_best // 20)
                idx = rng.choice(n_best, n_sparse, replace=False)
                a_r[idx] *= rng.uniform(0.2, 3.0, n_sparse)

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            # Optimize: LP + SM-GD
            alloc_lp = min(remaining * 0.3, 25.0)
            a_opt = lp_alternating(a_r, time_budget=alloc_lp)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc_sm = min(remaining * 0.3, 20.0)
                a_opt = soft_max_gd(a_opt, time_budget=alloc_sm)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_opt = fast_coordinate_descent(a_opt, time_budget=min(15, remaining * 0.3))

            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                a = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final Polish (remaining) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # LP
        alloc = min(remaining * 0.4, 30.0)
        a_new = lp_alternating(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

        # Coord descent
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.4, 25.0)
            a_new = fast_coordinate_descent(best_seq, time_budget=alloc)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

        # One more SM-GD pass
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.4, 20.0)
            a_new = soft_max_gd(best_seq, time_budget=alloc)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Final SM: obj={best_score:.8f} **")

            # Coord again
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_new = fast_coordinate_descent(best_seq, time_budget=min(10, remaining * 0.5))
                obj = compute_obj(a_new)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_new.copy()
                    print(f"  Final CD2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
