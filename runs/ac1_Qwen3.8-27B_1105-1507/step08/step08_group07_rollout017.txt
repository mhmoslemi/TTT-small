```python
"""
Multi-phase optimizer for minimum peak autocorrelation step functions.
Uses diverse spectral initializations, momentum LP-CD, subgradient polishing,
systematic n-sweep, and structured restarts.
"""
import numpy as np
from scipy.fft import rfft, irfft, next_fast_len
import time
import cvxpy as cp

# ===================== Core Computation =====================

def compute_conv(a):
    n = len(a)
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def obj_val(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    an = a / s
    c = compute_conv(an)
    return 2.0 * n * np.max(c)

def subgradient(a, k_star):
    """Subgradient of g_{k_star}(a) = sum_i a[i]*a[k_star-i]."""
    n = len(a)
    sg = np.zeros(n)
    # sg[j] = 2*a[k_star - j] if 0 <= k_star - j < n
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    for j in range(lo, hi + 1):
        sg[j] = 2.0 * a[k_star - j]
    return sg

def soft_subgradient(a, tau=100.0):
    """Smoothed subgradient using soft-max over all peaks."""
    n = len(a)
    c = compute_conv(a)
    # Soft-max weights
    c_max = np.max(c)
    w = np.exp(tau * (c - c_max))
    w = w / np.sum(w)
    # Weighted sum of subgradients
    sg = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        sg += w[k] * subgradient(a, k)
    return sg, c_max

# ===================== LP Solver =====================

def solve_lp(a, time_limit=5.0):
    """Solve LP: min M s.t. cross_conv(a,z) <= M, sum(z)=1, z>=0."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    prob = cp.Problem(cp.Minimize(M), [A @ z <= M, cp.sum(z) == 1])
    
    start = time.monotonic()
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        if time.monotonic() - start > time_limit:
            break
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                zv = np.maximum(z.value, 0)
                s = np.sum(zv)
                if s > 1e-12:
                    zv /= s
                    return zv, float(M.value), prob
        except:
            continue
    return None, None, None

# ===================== Optimization Methods =====================

def lp_momentum_cd(a, time_budget=300.0, verbose=False):
    """LP-CD with momentum and adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = obj_val(a)
    v = np.zeros(n)  # momentum
    beta = 0.7  # momentum coefficient
    stall = 0
    start = time.monotonic()
    
    prev_a = a.copy()
    
    while time.monotonic() - start < time_budget:
        z_lp, M_lp, _ = solve_lp(a)
        if z_lp is None:
            break
        
        # Momentum update
        direction = z_lp - a
        v = beta * v + (1 - beta) * direction
        v = np.maximum(v, -a * 0.5)  # don't go too negative
        
        # Line search
        best_step_obj = np.inf
        best_step_a = a.copy()
        for step in [0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
            a_try = a + step * v
            a_try = np.maximum(a_try, 0)
            s = np.sum(a_try)
            if s < 1e-12:
                continue
            a_try /= s
            o = obj_val(a_try)
            if o < best_step_obj:
                best_step_obj = o
                best_step_a = a_try.copy()
        
        if best_step_obj < best_obj - 1e-12:
            stall = 0
            prev_a = a.copy()
            a = best_step_a
            best_a = a.copy()
            best_obj = best_step_obj
            # Reduce momentum periodically
            beta = max(0.3, beta * 0.995)
            if verbose and int((time.monotonic() - start) * 2) % 10 == 0:
                print(f"    LP-mom: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
        else:
            stall += 1

            if o_raw < best_obj - 1e-12:
                a = z_lp.copy()
                best_a = a.copy()
                best_obj = o_raw

                a = best_a * np.exp(rng.normal(0, 0.03, n))

                best_obj = obj_val(a)

    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    tau = 50.0  # soft-max parameter (higher = sharper)
    
    # Polyak-like step size
    f_lower = 2 * n / (2 * n - 1)  # trivial lower bound
    step_size = 1.0 / (tau * np.max(compute_conv(a)))
    
    while time.monotonic() - start < time_budget:
        sg, c_max = soft_subgradient(a, tau)
        # Scale subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm
        
        # Try several step sizes
        improved = False
        for eta in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_try = a - eta * sg * tau  # scale by tau since sg is per-unit
            a_try = np.maximum(a_try, 0)
            s = np.sum(a_try)
            if s < 1e-12:
                continue
            a_try /= s
            o = obj_val(a_try)
            if o < best_obj - 1e-10:
                a = a_try
                best_a = a.copy()
                best_obj = o
                improved = True
                break
        
        if not improved:
            # Adapt tau
            tau *= 0.9
            if tau < 5:
                break
        else:
            if verbose and int((time.monotonic() - start) * 5) % 10 == 0:
                print(f"    SG: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
    
    return best_a

def freq_domain_optimize(a, time_budget=60.0, verbose=False):
    """
    Frequency-domain iterative optimization.
    Alternates between time-domain non-negativity constraint and 
    frequency-domain spectrum flattening.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / np.sum(a)
    N = next_fast_len(2 * n - 1)
    
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        # Zero-pad and FFT
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = rfft(a_pad)
        S = np.abs(A)**2
        
        # Compute autocorrelation
        R = np.zeros(N)
        AR = np.zeros(N)
        AR[:len(A)] = A
        # Actually, let's just use the direct computation
        R_full = irfft(A * np.conj(A), N)  # This is the cyclic autocorrelation
        # Wait, that's not right. The cyclic autocorrelation is IFFT(FFT(a) * conj(FFT(a)))
        # But A is already rfft, so... let me just use the direct convolution
        c = compute_conv(a)
        
        # Find peak
        k_star = np.argmax(c)
        
        # In frequency domain, the peak at lag k_star is:
        # c[k_star] = sum_m S[m] * cos(2*pi*m*k_star/N) / N (approximately)
        # To reduce c[k_star], reduce S[m] where cos aligns
        
        # Simpler approach: reduce the top spectral components
        S_sorted_idx = np.argsort(S)[::-1]
        # Reduce top few% of spectral components
        n_reduce = max(1, N // 20)
        for idx in S_sorted_idx[:n_reduce]:
            if idx == 0:
                continue  # Don't touch DC (it's fixed by sum constraint)
            S[idx] *= 0.95
        
        # Reconstruct with modified spectrum (same phase)
        A_mag = np.sqrt(S)
        A_phase = A / (np.abs(A) + 1e-15)
        A_new = A_mag * A_phase
        a_new = irfft(A_new, N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-12:
            break
        a_new /= s
        
        o = obj_val(a_new)
        if o < best_obj:
            best_a = a_new.copy()
            best_obj = o
            a = a_new
            if verbose:
                print(f"    Freq: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
        else:
            # Blend
            for blend in [0.3, 0.5, 0.7]:
                a_mix = (1-blend) * best_a + blend * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix = a_mix / np.sum(a_mix)
                o_mix = obj_val(a_mix)
                if o_mix < best_obj:
                    best_a = a_mix.copy()
                    best_obj = o_mix
                    a = a_mix
                    break
            else:
                # No improvement, reduce aggressiveness
                break
    
    return best_a

# ===================== Initialization =====================

def generate_initializations(n, rng):
    """Generate diverse initializations of length n."""
    t = np.linspace(0, 1, n)
    inits = {}
    
    inits['uniform'] = np.ones(n)
    inits['cosine'] = 0.5 + 0.5*np.cos(2*np.pi*t)
    inits['sine2'] = np.sin(np.pi*t)**2
    inits['hann'] = 0.5 + 0.5*np.cos(2*np.pi*t)
    inits['raised_cos4'] = 0.5 + 0.5*np.cos(4*np.pi*t)
    
    # Chirp-like (real part of complex chirp, rectified)
    chirp = np.cos(2*np.pi*(0.5*t**2))*np.exp(-2*t*(1-t))
    inits['chirp'] = np.maximum(chirp, 0.01)
    
    # Banded spectrum: sum of a few sinusoids
    for freq in [3, 5, 8, 13]:
        sig = 1.0 + 0.3*np.sin(2*np.pi*freq*t)
        inits[f'sine_{freq}'] = np.maximum(sig, 0.05)
    
    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        bump = np.exp(-10*(t-center)**2)
        inits[f'gauss_{center}'] = 0.1 + 0.9*bump
    
    # Random smooth
    noise = rng.standard_normal(n)
    # Smooth with a kernel
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(noise, kernel, mode='same')
    inits['smooth_rand'] = np.maximum(smooth, 0) + 0.1
    
    # Triangular
    tri = 1 - 2*np.abs(t - 0.5)
    inits['triangular'] = np.maximum(tri, 0.05)
    
    return inits

def generate_best_init(n, rng, time_budget=20.0):
    """Generate and evaluate diverse inits, return best."""
    start = time.monotonic()
    inits = generate_initializations(n, rng)
    
    best_a = None
    best_obj = np.inf
    
    for name, a in inits.items():
        if time.monotonic() - start > time_budget:
            break
        a = np.maximum(a, 0).astype(np.float64)
        s = np.sum(a)
        if s < 1e-12:
            continue
        a = a / s
        o = obj_val(a)
        if o < best_obj:
            best_obj = o
            best_a = a.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = obj_val(best_a)
    
    return best_a, best_obj

# ===================== Multi-n =====================

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

# ===================== Coordinate Descent =====================

def coordinate_polish(a, time_budget=30.0):
    """Single-coordinate and pairwise exchange polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    while time.monotonic() - start < time_budget:
        improved = False
        # Try random coordinate pairs
        for _ in range(30):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for frac in [0.02, 0.05, 0.1, 0.2]:
                delta = frac * min(a[i], a[j])
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0:
                    continue
                a_try = a_try / np.sum(a_try)
                o = obj_val(a_try)
                if o < best_obj - 1e-12:
                    best_obj = o
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        if not improved:
            # Try single coordinate
            for _ in range(20):
                if time.monotonic() - start > time_budget:
                    break
                i = rng.integers(0, n)
                for mult in [0.5, 0.7, 1.3, 1.5, 2.0]:
                    a_try = a.copy()
                    a_try[i] *= mult
                    a_try = np.maximum(a_try, 0)
                    s = np.sum(a_try)
                    if s < 1e-12:
                        continue
                    a_try = a_try / s
                    o = obj_val(a_try)
                    if o < best_obj - 1e-12:
                        best_obj = o
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
            if not improved:
                break
    
    return best_a

# ===================== Main =====================

def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimizer:
    1. Diverse spectral initializations
    2. Momentum LP-CD
    3. Frequency-domain optimization
    4. Subgradient polishing
    5. Systematic multi-n sweep
    6. Final polish + restarts
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        if parent.size > 0 and np.isfinite(parent).all():
            s = float(np.sum(parent))
            if s > 0.01:
                parent = parent / s
                obj = obj_val(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except:
        pass
    
    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = obj_val(best_seq)
        print(f"Default init: obj={best_score:.8f}")
    
    n_best = len(best_seq)
    
    # === Phase 1: Diverse initializations with quick opt ===
    print(f"\n=== Phase 1: Diverse inits + quick LP-CD ===")
    remaining = deadline - time.monotonic()
    n_init = n_best
    
    # Try to find a better init
    a_init, obj_init = generate_best_init(n_init, rng, time_budget=min(15, remaining*0.02))
    if obj_init < best_score:
        best_score = obj_init
        best_seq = a_init.copy()
        print(f"  Better init found: obj={best_score:.8f}")
    else:
        a_init = best_seq.copy()
    
    # Quick LP-CD on best init
    if remaining > 100:
        alloc = min(100.0, remaining * 0.1)
        a_opt = lp_momentum_cd(a_init, time_budget=alloc, verbose=True)
        o = obj_val(a_opt)
        if o < best_score:
            best_score = o
            best_seq = a_opt.copy()
            print(f"  After quick LP-CD: obj={best_score:.8f}")
    
    # === Phase 2: Frequency-domain optimization ===
    print(f"\n=== Phase 2: Freq-domain opt ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        alloc = min(80.0, remaining * 0.08)
        a_freq = freq_domain_optimize(best_seq, time_budget=alloc, verbose=True)
        o = obj_val(a_freq)
        if o < best_score:
            best_score = o
            best_seq = a_freq.copy()
            print(f"  After freq-domain: obj={best_score:.8f}")
        else:
            print(f"  Freq-domain: obj={o:.8f} (no improvement)")
    
    # === Phase 3: Main LP-CD (long) ===
    print(f"\n=== Phase 3: Main LP-CD ===")
    remaining = deadline - time.monotonic()
    if remaining > 200:
        alloc = min(400.0, remaining * 0.35)
        a_opt = lp_momentum_cd(best_seq, time_budget=alloc, verbose=True)
        o = obj_val(a_opt)
        if o < best_score:
            best_score = o
            best_seq = a_opt.copy()
        print(f"  After main LP-CD: obj={best_score:.8f}")
    
    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(60.0, remaining * 0.07)
        a_sg = subgradient_polish(best_seq, time_budget=alloc, verbose=True)
        o = obj_val(a_sg)
        if o < best_score:
            best_score = o
