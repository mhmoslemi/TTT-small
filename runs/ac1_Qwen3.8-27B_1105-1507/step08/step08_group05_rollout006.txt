def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def active_peaks(conv, tol=1e-6):
    """Find peaks that are within tol of the maximum."""
    max_val = np.max(conv)
    threshold = max_val - tol * max_val
    return np.where(conv >= threshold)[0]

def subgradient_at_active(a, active_k, max_val):
    """Compute subgradient of max_k conv(a,a)[k] at a, using active peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_k:
        k = int(k)
        # grad[m] = 2 * a[k-m] if 0 <= k-m < n
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            m_indices = k - np.arange(i_start, i_end + 1)
            grad[m_indices] += 2.0 * a[np.arange(i_start, i_end + 1)]
    # Average over active peaks (subgradient of max)
    grad /= len(active_k)
    return grad

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i-1] / i
    return np.maximum(v - theta, 0)

    grad_obj_a = 2.0 * n * grad_a
    
    mean_grad = np.dot(a, grad_obj_a)
    grad_z = a * (grad_obj_a - mean_grad)
    
    return obj, grad_z

def lbfgs_smooth(a0, tau_init=5.0, tau_final=1000.0, time_budget=100.0):
    """Optimize smoothed objective with increasing tau."""
    n = len(a0)
    start = time.monotonic()
    best_a = a0.copy()
    best_obj = compute_obj(a0)
    
    # Start from current a, convert to z
    a_norm = a0 / np.sum(a0)
    z = np.log(a_norm + 1e-30)
    
    taus = [tau_init]
    # Geometric progression
    while taus[-1] < tau_final:
        taus.append(min(taus[-1] * 2.0, tau_final))
    
    for tau in taus:
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        try:
            res = minimize(
                lambda z: smooth_obj_grad(z, tau)[0],
                z,
                jac=lambda z: smooth_obj_grad(z, tau)[1],
                method='L-BFGS-B',
                options={'maxiter': 200, 'gtol': 1e-6, 'maxfun': 500}
            )
            z = res.x
            a_new = np.exp(z - np.max(z))
            a_new = a_new / np.sum(a_new)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a
