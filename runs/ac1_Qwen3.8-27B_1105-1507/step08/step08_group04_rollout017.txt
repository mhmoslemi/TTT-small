from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import cvxpy as cp
import time

def compute_conv_fft(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def smooth_max_grad(u, beta):
    """Compute smooth max of conv(u,u) and its gradient.
    Assumes sum(u) = 1, u >= 0.
    Returns (smooth_max_value, gradient)."""
    n = len(u)
    conv = compute_conv_fft(u)
    # Softmax weights
    max_conv = np.max(conv)
    exp_conv = np.exp(beta * (conv - max_conv))
    p = exp_conv / np.sum(exp_conv)
    
    # Smooth max value
    smooth_val = max_conv + np.log(np.sum(exp_conv)) / beta
    
    # Gradient: grad[j] = 2*n/beta * sum_m p[m] * 2*u[m-j]
    # = 4*n/beta * (p cross-correlate u)[j]
    # Using FFT: r = ifft(conj(fft(p)) * fft(u))
    N = 1
    while N < 2*n:
        N <<= 1
    P = rfft(p, N)  # p has length 2n-1, zero-padded to N
    U = rfft(u, N)  # u has length n, zero-padded to N
    r = irfft(np.conj(P) * U, N)  # cross-correlation
    grad = (4.0 * n / beta) * r[:n]
    
    return smooth_val, grad

```python
"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth max approximation (log-sum-exp) with exact gradients
2. Projected gradient descent with backtracking line search and Nesterov acceleration
3. LP alternating for polishing
4. Multi-n exploration and diverse restarts
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time

# Global deadline
_global_deadline = None

def compute_conv_fft(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv_fft(a)
    return 2.0 * n * float(np.max(conv))

def smooth_max_and_grad(u, beta):
    """Compute smooth max of conv(u,u) and its gradient w.r.t. u.
    Assumes sum(u) = 1, u >= 0.
    Returns (smooth_value, gradient)."""
    n = len(u)
    conv = compute_conv_fft(u)  # length 2n-1
    
    # Numerically stable log-sum-exp
    max_c = np.max(conv)
    shifted = beta * (conv - max_c)
    exp_c = np.exp(shifted)
    log_sum = max_c * beta + np.log(np.sum(exp_c))
    smooth_val = log_sum / beta
    
    # Softmax weights
    p = exp_c / np.sum(exp_c)
    
    # Gradient: grad[j] = (4n/beta) * sum_m p[m] * u[m-j]
    # C[j] = sum_m p[m] * u[m-j] = convolve(p, u[::-1])[j + n - 1]
    N = 1
    while N < (2 * n - 1) + n - 1:
        N <<= 1
    P = rfft(p, N)
    U_rev = rfft(u[::-1], N)
    C_full = irfft(P * U_rev, N)
    r = C_full[n - 1:2 * n - 1]  # length n
    
    grad = (4.0 * n / beta) * r
    return smooth_val, grad

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_gradient_optimize(a, time_budget, beta_start=10.0, beta_end=500.0, 
                              max_iters=5000, verbose=False):
    """Optimize using projected gradient with smooth max approximation.
    Uses adaptive beta (continuation method)."""
    global _global_deadline
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov momentum
    momentum = np.zeros(n)
    beta_mom = 0.9
    
    # Beta schedule
    num_phases = 5
    phase_budget = time_budget / num_phases
    
    for phase in range(num_phases):
        phase_start = time.monotonic()
        # Beta for this phase
        frac = phase / max(num_phases - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = max(beta, 5.0)
        
        x = a.copy()  # Nesterov lookahead
        a_curr = a.copy()
        
        it = 0
        while time.monotonic() - phase_start < phase_budget \
              and time.monotonic() < _global_deadline:
            if it % 10 == 0 and (time.monotonic() - phase_start) > 2:
                t_curr = time.monotonic()
                if t_curr >= _global_deadline or t_curr - phase_start > phase_budget:
                    break
            
            # Compute gradient at x (Nesterov point)
            smooth_val, grad = smooth_max_and_grad(x, beta)
            
            # Nesterov update
            momentum = beta_mom * momentum - grad
            a_new = a_curr + momentum
            a_new = project_simplex(a_new)
            
            # Update Nesterov lookahead
            t_k = 2.0 / (it + 2.0)
            x = a_new + (1 - t_k) * (a_new - a_curr)
            x = project_simplex(x)
            
            a_curr = a_new
            it += 1
            
            # Periodically check objective
            if it % 50 == 0:
                obj = compute_obj(a_curr)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_curr.copy()
                    if verbose:
                        print(f"      Phase {phase}: it={it}, beta={beta:.1f}, obj={best_obj:.8f}")
        
        # Update best
        obj_final = compute_obj(a_curr)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_curr.copy()
    
    return best_a
```

def smooth_max_and_grad(u, beta):
    """Returns (smooth_max_value, gradient_vector)."""
    n = len(u)
    conv = compute_conv_fft(u)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    exp_sum = np.exp(shifted).sum()
    smooth_val = max_c + np.log(exp_sum) / beta
    p = np.exp(shifted) / exp_sum
    
    # Gradient
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    P = rfft(p, N)
    Urev = rfft(u[::-1], N)
    C = irfft(P * Urev, N)
    r = C[n-1:2*n-1]
    grad = (4.0 * n / beta) * r
    return smooth_val, grad

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n <= 1:
        return np.maximum(v, 0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(2 * n - 2, j + n - 1)
        A[j:k_end+1, j] = a[:k_end - j + 1]
    return A
