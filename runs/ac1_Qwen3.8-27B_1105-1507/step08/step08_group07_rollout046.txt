"""
Optimized minimizer for 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Entropy-smoothed projected gradient descent with Nesterov momentum (fast basin finding)
2. Pairwise Frank-Wolfe with away steps and backtracking line search
3. Careful coordinate descent polish
4. Multi-n exploration with focused runs
5. Diverse restarts guided by dual information
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_smooth_gradient(a, tau):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient w.r.t. a.
    Assumes sum(a)=1, a>=0.
    Returns (smooth_max_value, gradient_on_simplex_tangent)
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    # Handle overflow: subtract max again (already done)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    smooth_max = max_conv + log_sum / tau
    weights = exp_vals / np.sum(exp_vals)

    # corr[i] = sum_j a[j] * weights[i+j] for valid indices
    # = convolve(flip(a), weights)[n-1 : 2n-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = rfft(a[::-1], size)
    fw = rfft(weights, size)
    corr_full = irfft(fa * fw, size)
    corr = corr_full[n - 1:2 * n - 1]

    grad = 2.0 * corr
    # Project to tangent space of simplex
    grad = grad - np.mean(grad)
    return smooth_max, grad


def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = np.max(np.where(cond)[0])
    theta = cssv[k] / (k + 1)
    return np.maximum(v - theta, 0)


def smooth_optimize(a0, time_budget=100.0, tau_start=500.0, tau_end=50.0, verbose=True):
    """
    Projected gradient descent with Nesterov momentum on the smoothed objective.
    Anneals tau from tau_start to tau_end.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    v = np.zeros(n)  # momentum velocity

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    total_iters = 0

    # Adaptive learning rate
    lr = 0.01
    momentum = 0.9

    while time.monotonic() - start < time_budget:
        # Current tau (exponential annealing)
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        tau = tau_start * (tau_end / tau_start) ** frac

        smooth_val, grad = compute_smooth_gradient(a, tau)
        total_iters += 1

        # Nesterov lookahead
        a_lookahead = a + momentum * v
        a_lookahead = np.maximum(a_lookahead, 0)
        a_lookahead = a_lookahead / np.sum(a_lookahead)

        smooth_val_la, grad_la = compute_smooth_gradient(a_lookahead, tau)

        # Gradient step
        v = momentum * v - lr * grad_la
        a_new = a + v
        a_new = proj_simplex(a_new)

        # Backtracking if smooth objective increases
        obj_new = compute_obj(a_new)
        if obj_new > best_obj * 1.01:
            # Step back
            a_new = a.copy()
            v *= 0.5
            lr = max(lr * 0.8, 0.001)
        else:
            a = a_new
            # Adaptive lr
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
                lr = min(lr * 1.05, 0.1)
            else:
                lr = max(lr * 0.95, 0.001)

        # Reduce momentum over time
        momentum = 0.9 * (1.0 - frac * 0.3)

        if verbose and total_iters % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"      Smooth: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.4f}, t={elapsed:.0f}s")

        # Check convergence
        if frac > 0.5 and total_iters % 100 == 0:
            improvement = 0
            for _ in range(50):
                if time.monotonic() - start > time_budget:
                    break
                _, g = compute_smooth_gradient(a, tau)
                a_try = proj_simplex(a - 0.01 * g)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-9:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improvement += 1
            if improvement == 0 and frac > 0.8:
                break

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ z gives cross-conv a*z."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_fast(a):
    """
    Solve LP: min M s.t. A@z <= M, sum(z)=1, z>=0.
    Uses scipy linprog with HiGHS.
    Returns (z_normalized, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        z_val = np.maximum(res.x[:n], 0)
        zs = float(np.sum(z_val))
        if zs > 1e-10:
            z_val = z_val / zs
            return z_val, float(res.x[n])
    return None, None

def coordinate_polish(a, time_budget=60.0, verbose=True):

    """

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(a)
    active_k = int(np.argmax(conv))

    # Single coordinate moves
    improved_any = True
    passes = 0
    while improved_any and time.monotonic() - start < time_budget:
        improved_any = False
        passes += 1

        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            for frac in [0.02, 0.05, 0.1, 0.2, 0.01]:
                if a[i] < frac * 1e-6:
                    continue

                # Determine best j using gradient info
                # d(conv[k])/d(a[i]) = 2*a[k-i], d(conv[k])/d(a[j]) = 2*a[k-j]
                # We want to decrease the active convolution
                # Moving mass from i to j changes conv[k] by:
                # 2*frac*a_i*(a[j-(k-i)] ... ) complex, just try random j's
                for _ in range(3):
                    j = np.random.randint(0, n)
                    if j == i:
                        continue
                    delta = frac * best_a[i]
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    if a_try[i] < 0:
                        continue
                    a_try = a_try / np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved_any = True
                        break
                if improved_any:
                    break

        if verbose and passes % 5 == 0:
            print(f"      Coord polish: pass={passes}, obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    from scipy.interpolate import interp1d
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = f(x_new)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: smooth init -> FW with away steps -> coordinate polish -> multi-n -> restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        # Generate diverse initializations
        candidates = []
        for n_init in [400, 600, 800, 1000, 1200]:
            t = np.linspace(0, 1, n_init)
            windows = [
                0.5 + 0.5 * np.cos(2 * np.pi * t),
                np.sin(np.pi * t) ** 2,
                1.0 - 0.2 * np.cos(3 * np.pi * t),
                np.exp(-2.0 * (t - 0.5) ** 2 / 0.1),
            ]
            for w in windows:
                a_init = np.maximum(w, 0.001)
                a_init = a_init / np.sum(a_init)
                obj_init = compute_obj(a_init)
                candidates.append((obj_init, a_init))
        candidates.sort(key=lambda x: x[0])
        best_score = candidates[0][0]
        best_seq = candidates[0][1].copy()
        print(f"Initial (n={len(best_seq)}): obj={best_score:.8f}")

    a = best_seq.copy()
    n_parent = len(a)

    # === Phase 1: Smooth Optimization (200s) ===
    print(f"\n=== Phase 1: Smooth Opt (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(200.0, remaining * 0.25)
        a_smooth = smooth_optimize(a, time_budget=alloc, tau_start=200.0, tau_end=30.0, verbose=True)
        obj_smooth = compute_obj(a_smooth)
        if obj_smooth < best_score:
            best_score = obj_smooth
            best_seq = a_smooth.copy()
            a = a_smooth.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_smooth:.8f} (no improvement)")

    # === Phase 2: FW with Away Steps (400s) ===
    print(f"\n=== Phase 2: FW Away (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc = min(400.0, remaining * 0.4)
        a_fw = fw_with_away_steps(a, time_budget=alloc, verbose=True)
        obj_fw = compute_obj(a_fw)
        if obj_fw < best_score:
            best_score = obj_fw
            best_seq = a_fw.copy()
            a = a_fw.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_fw:.8f}")

    # === Phase 3: Coordinate Polish (60s) ===
    print(f"\n=== Phase 3: Coord Polish (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(60.0, remaining * 0.1)
        a_pol = coordinate_polish(best_seq, time_budget=alloc, verbose=True)
        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_pol:.8f}")

    # === Phase 4: Multi-n Exploration (300s) ===
    print(f"\n=== Phase 4: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_candidates = sorted(set([
        max(300, n_parent - 200), max(300, n_parent - 100),
        n_parent,
        n_parent + 100, n_parent + 200, n_parent + 400,
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 2500]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if abs(n_try - len(best_seq)) < 50:
            continue

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / np.sum(a_try)

        # Quick smooth opt
        alloc_smooth = min(40.0, remaining * 0.2)
        a_s = smooth_optimize(a_try, time_budget=alloc_smooth, tau_start=100.0, tau_end=20.0, verbose=False)
        obj_s = compute_obj(a_s)

        # Quick FW
        remaining2 = deadline - time.monotonic()
        if remaining2 > 40:
            alloc_fw = min(60.0, remaining2 * 0.4)
            a_f = fw_with_away_steps(a_s, time_budget=alloc_fw, verbose=False)
            obj_f = compute_obj(a_f)
            if obj_f < obj_s:
                a_s = a_f
                obj_s = obj_f

        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_s.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
            # Re-polish at new length
            remaining3 = deadline - time.monotonic()
            if remaining3 > 30:
                a_p = coordinate_polish(best_seq, time_budget=min(20.0, remaining3 * 0.2), verbose=False)
                obj_p = compute_obj(a_p)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
        else:
            print(f"  n={n_try}: obj={obj_s:.8f}")

    # === Phase 5: Final FW Run (remaining time) ===
    print(f"\n=== Phase 5: Final FW (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(remaining * 0.5, 200.0)
        a_final = fw_with_away_steps(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Phase 5: obj={best_score:.8f} **")
        else:
            print(f"  Phase 5: obj={obj_final:.8f}")

        # Final coordinate polish
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_pol = coordinate_polish(best_seq, time_budget=min(30.0, remaining * 0.4), verbose=True)
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  Final polish: obj={best_score:.8f} **")

    # === Phase 6: Diverse Restarts (if time) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 6: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(4, int(remaining // 50))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            if r % 2 == 0:
                # Smooth perturbation
                a_r = best_seq * np.exp(rng.normal(0, 0.03, n_best))
            else:
                # Structural perturbation: modify a segment
                a_r = best_seq.copy()
                seg_len = max(20, n_best // 10)
                seg_start = rng.integers(0, n_best - seg_len)
                t_seg = np.linspace(0, 1, seg_len)
                modification = 1.0 + 0.3 * np.cos(2 * np.pi * rng.uniform(0.5, 3.0) * t_seg)
                a_r[seg_start:seg_start + seg_len] *= modification

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            alloc = min(remaining * 0.4, 50.0)
            a_opt = fw_with_away_steps(a_r, time_budget=alloc, verbose=False)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
