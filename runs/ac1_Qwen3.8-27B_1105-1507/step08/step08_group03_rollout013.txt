import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fft_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = fft_conv(a_norm)
    return 2.0 * n * conv.max()

def build_conv_matrix(a):
    n = len(a)
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = a  # a[0], a[1], ..., a[n-1] at positions i, i+1, ..., i+n-1
    return A

def lp_minpeak(a):
    """Solve min M s.t. conv(a,g) <= M, sum(g)=1, g>=0"""
    n = len(a)
    a = a / a.sum()
    A = build_conv_matrix(a)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # min M (last variable)
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # Constraints: A @ g - M <= 0 for each row
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = g.sum()
        if s > 1e-10:
            g = g / s
            return g, res.x[-1]
    return None, None

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using aggressive LP alternating,
frequency-domain insights, adaptive n exploration, and structured escapes.
Key innovations: flat-target LP variant, dual-informed perturbations,
multi-n resampling, and deterministic chirp-based initializations.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = fft_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def lp_minpeak(a):
    """Solve min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g, M_opt)."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s, float(res.x[-1])
    return None, None


def lp_flat_target(a):
    """
    Solve min t s.t. conv(a,g)[k] <= c+t for all k, sum(g)=1, g>=0.
    where c = 1/(2n-1) is the theoretical average.

    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)
    c_target = 1.0 / (2 * n - 1)

    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.full(2 * n - 1, c_target)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s, float(res.x[-1])
    return None, None


def lp_maxsum(a, M_target):
    """Solve max sum(g) s.t. conv(a,g)[k] <= M_target, g>=0. Returns g_normalized."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x, 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s
    return None


def chirp_init(n, rng):
    """Generate a chirp-like non-negative initialization."""
    x = np.arange(n) / n
    # Multiple chirp frequencies for diversity
    freqs = [1, 2, 3, 5]
    phases = rng.uniform(0, 2 * np.pi, len(freqs))
    a = np.ones(n)
    for f, p in zip(freqs, phases):
        a += 0.5 * np.sin(2 * np.pi * f * x + p)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def tanh_profile(n, rng):
    """Generate a smooth non-symmetric profile."""
    x = np.arange(n) / n
    center = rng.uniform(0.2, 0.8)
    width = rng.uniform(0.1, 0.4)
    a = 1.0 / (1.0 + np.exp(-(x - center) / width))
    a += 1.0 / (1.0 + np.exp((x - center) / width))
    a += 0.1
    a = a / a.sum()
    return a


def interleaved_init(n, rng):
    """Interleave high and low values to reduce local autocorrelation."""
    a = np.zeros(n)
    step = rng.integers(2, 5)
    for i in range(0, n, step):
        a[i] = 1.0
    a += rng.uniform(0, 0.3, n)
    a = a / a.sum()
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def gradient_polish(a, time_budget, K=10, verbose=False):
    """Smooth gradient-based polish using log-sum-exp approximation."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    T = 50.0  # temperature for log-sum-exp
    momentum = 0.0
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = fft_conv(a)
        max_c = conv.max()
        if max_c < 1e-15:
            break

        # Log-sum-exp gradient (smooth approximation of max)
        z = T * (conv - max_c)
        z = np.clip(z, -50, 50)
        e = np.exp(z)
        sum_e = e.sum()
        w = e / sum_e

        # Gradient of smooth objective
        g = np.zeros(n)
        # Only process significant weights for speed
        thresh = np.percentile(w, 80)
        active = w > thresh
        for k in np.where(active)[0]:
            if w[k] > 1e-8:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w[k] * a[j_start:j_end + 1][::-1]

        g *= 2.0 * n  # Scale by 2n (from objective)

        v = momentum * v + g
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        it += 1

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = max(1e-6, lr * 0.999)
            momentum = min(0.7, momentum + 0.001)
            # Gradually decrease temperature
            T = max(10.0, T * 0.999)
            if verbose and it % 100 == 0:
                print(f"    Grad it{it}: obj={best_obj:.8f}, T={T:.1f}, "
                      f"el={time.monotonic()-start:.1f}s")
        else:
            lr = max(1e-6, lr * 0.97)
            if it % 500 == 0:
                a = best_a.copy()
                v = np.zeros(n)

    return best_a

            print(f"  After warmup: obj={best_obj:.8f} **")
        else:
            print(f"  After warmup: obj={obj1:.8f} (no improvement)")

    # === Phase 2: Intensive LP alternating (35% time) ===
    print("\n=== Phase 2: Intensive LP ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 450.0)

    # Run multiple rounds with increasing aggressiveness
    n_rounds = 4
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p2_budget / (n_rounds - rnd), remaining * 0.5)
        if alloc < 20:
            break

        if rnd > 0:
            # Perturb for diversity
            noise_scale = 0.02 * (rnd + 1)
            a_pert = best_a * np.exp(rng.normal(0, noise_scale, len(best_a)))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        else:
            a_pert = best_a.copy()

        a_lp = lp_alternating(a_pert, alloc, rng, use_flat_target=(rnd % 2 == 1),
                              verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            n_current = len(best_a)
            print(f"  Round {rnd}: obj={best_obj:.8f} **")
        else:
            print(f"  Round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Gradient polish (15% time) ===
    print(f"\n=== Phase 3: Gradient Polish (n={len(best_a)}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 15:
        a_grad = gradient_polish(best_a, p3_budget, K=10, verbose=True)
        obj_grad = compute_obj(a_grad)
        if obj_grad < best_obj:
            best_obj = obj_grad
            best_a = a_grad.copy()
            print(f"  Gradient: obj={best_obj:.8f} **")
        else:
            print(f"  Gradient: obj={obj_grad:.8f} (no improvement)")

    # === Phase 4: Multi-n exploration (25% time) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()

    n_options = sorted(set([
        max(100, n_current - 200),
        max(100, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
    ]))
    # Add some fixed options
    for fn in [300, 500, 600, 700, 800, 1000, 1200, 1500, 2000]:
        n_options.append(fn)
    n_options = sorted(set(n for n in n_options if 100 <= n <= 3000))
    # Sort by closeness to current
    n_options.sort(key=lambda x: abs(x - n_current))

    n_explored = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40 or n_explored >= 8:
            break
        if n_try == n_current:
            continue
        if n_try > 2500:
            continue  # Too slow for LP

        a_try = resample_sequence(best_a, n_try)
        n_explored += 1

        # Quick optimization: LP + gradient
        alloc_lp = min(remaining * 0.1, 40.0)
        if alloc_lp > 10:
            a_try = lp_alternating(a_try, alloc_lp, rng)

        remaining = deadline - time.monotonic()
        alloc_grad = min(remaining * 0.05, 20.0)
        if alloc_grad > 10:
            a_try = gradient_polish(a_try, alloc_grad)

        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts (10% time) ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # Different initialization strategies
        if r == 0:
            a_r = chirp_init(n_best, rng)
        elif r == 1:
            a_r = tanh_profile(n_best, rng)
        elif r == 2:
            a_r = interleaved_init(n_best, rng)
        elif r == 3:
            a_r = rng.uniform(0, 1, n_best)
            a_r = a_r / a_r.sum()
        else:
            # Blend best with uniform
            a_r = 0.3 * best_a + 0.7 * np.ones(n_best) / n_best
            a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.12, 60.0)
        a_opt = lp_alternating(a_r, restart_budget, rng)
        obj_r = compute_obj(a_opt)

        if obj_r < best_obj:
            best_obj = obj_r
            best_a = a_opt.copy()
            print(f"  Restart {r}: obj={best_obj:.8f} **")
        else:
            print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        lp_alloc = min(remaining * 0.4, 80.0)
        a_final = lp_alternating(best_a, lp_alloc, rng, use_flat_target=True,
                                 verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final LP: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        grad_alloc = min(remaining * 0.3, 40.0)
        a_final = gradient_polish(best_a, grad_alloc, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final Grad: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_final = lp_alternating(best_a, min(remaining * 0.5, 20.0), rng)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final LP2: obj={best_obj:.8f} **")

    # Final validation
    if best_a is None:
        best_a = np.ones(200)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_obj:
            best_obj = val
    else:
        result = [1.0] * 100
        best_obj = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
