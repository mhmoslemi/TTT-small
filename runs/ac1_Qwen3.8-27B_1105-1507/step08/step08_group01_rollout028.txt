def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)

    if s < 0.01:

    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_full(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""

def _solve_lp_scipy(a):
    """Fallback LP solver using scipy linprog (HiGHS)."""

    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        s = float(np.sum(g_val))
        if s > 1e-10:

    return None, None

    #     # a[i+k] is valid when 0 <= i+k < n => -i <= k < n-i
    #     k1_start = i
    #     k1_end = min(2*n-1, i+n)

    #     k2_end = min(2*n-1, n-i)

    # When k≠2i: first term gives a[k-i] (if valid), second gives a[k-i] (if valid), total = 2*a[k-i]
    # But we need to be careful: the two terms are always the same value a[k-i] (when valid),
    # so d(conv[k])/d(a[i]) = 2*a[k-i] if 0<=k-i<n, else 0.
    # (The case k=2i is handled correctly: gives 2*a[i].)
    
    # So: grad[i] = 2 * sum_k w[k] * a[k-i] * indicator(0<=k-i<n)
    # = 2 * sum_{j=0}^{n-1} w[j+i] * a[j] (where j = k-i, so k = j+i, need 0<=j+i<2n-1)

    if 2 * n - 1 > 1:
        w_rev[1:2*n-1] = weights[2*n-2:0:-1]

    # Let me try: ifft(FA * FW)[i] = sum_m a[m] * w[(i-m) mod N]

    grad = 2.0 * corr[:n]
    
    return smooth_val, grad


def smooth_max_gd_nesterov(a, time_budget, T_init=None, T_min=None, verbose=False):
    """Nesterov accelerated gradient descent with smooth max approximation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    if T_init is None:
        T_init = 1.0 / (10 * n)
    if T_min is None:
        T_min = T_init / 1000.0

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov parameters
    t_val = 1.0
    v = np.zeros(n)
    alpha = 0.05  # base learning rate
    max_alpha = 0.5
    min_alpha = 1e-6
    stall_count = 0
    T = T_init
    T_decay = 0.9995  # per-iteration decay
    patience = 50

    x = a.copy()
    y = a.copy()

    max_iters = 20000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute gradient at y
        smooth_val, grad = compute_smooth_max_grad(y, T)
        grad = grad - np.mean(grad)  # Remove mean to respect simplex constraint

        # Nesterov update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        alpha_t = alpha * t_val / t_next

        # Gradient step
        step = x - y
        y_new = x + (t_val - 1.0) / t_next * step - alpha_t * grad
        x_new = simplex_project(y_new)

        # Evaluate
        obj_new = compute_obj(x_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = x_new.copy()
            x = x_new
            y = y_new
            t_val = t_next
            stall_count = 0
            # Increase alpha slightly
            alpha = min(max_alpha, alpha * 1.01)
        else:
            stall_count += 1
            if stall_count == patience:
                # Reset momentum, reduce alpha
                alpha *= 0.5
                if alpha < min_alpha:
                    break
                t_val = 1.0
                x = best_a.copy()
                y = best_a.copy()
                stall_count = 0
            elif stall_count > patience:
                # Nudge back
                x = best_a.copy()
                y = best_a + 0.01 * np.random.default_rng(it).normal(0, 1, n)
                y = simplex_project(y)
                t_val = 1.0
                stall_count = 0
                alpha *= 0.8

        # Decrease temperature
        T = max(T_min, T * T_decay)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"            GD it{it}: obj={best_obj:.8f}, T={T:.6f}, alpha={alpha:.4f}, t={t_val:.1f}, time={elapsed:.1f}s")

    return best_a


def coordinate_descent_1d(a, time_budget, verbose=False):
    """Coordinate descent with 1D line search per coordinate."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv_cur = compute_conv(a)
    max_conv = float(np.max(conv_cur))

    pass_count = 0
    delta_base = 0.1 / n  # Initial step size
    
    while time.monotonic() - start < time_budget:
        pass_count += 1
        any_improved = False
        
        rng = np.random.default_rng(pass_count * 7919 + n)
        order = rng.permutation(n)

        for i in order[:n]:  # Process all coordinates
            if time.monotonic() - start > time_budget:
                break

            v = a[i]
            if v < 1e-10:
                # Try small positive values
                for delta in [delta_base * 0.1, delta_base * 0.5, delta_base]:
                    a_try = a.copy()
                    a_try[i] = delta
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        any_improved = True
                        break
                continue

            # Try several delta values (both positive and negative)
            candidates = [
                v + delta_base, v - delta_base,
                v + delta_base * 0.3, v - delta_base * 0.3,
                v + delta_base * 0.1, v - delta_base * 0.1,
                v * 1.1, v * 0.9,
                v * 1.01, v * 0.99,
            ]
            candidates = [max(0.0, c) for c in candidates]

            best_delta_obj = best_obj
            best_delta_a = None

            for a_try_val in candidates:
                if abs(a_try_val - v) < 1e-15:
                    continue
                a_try = a.copy()
                a_try[i] = a_try_val
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_delta_obj:
                    best_delta_obj = obj_try
                    best_delta_a = a_try

            if best_delta_a is not None:
                best_obj = best_delta_obj
                best_a = best_delta_a.copy()
                a = best_delta_a
                any_improved = True

        if verbose and pass_count % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"            CD pass {pass_count}: obj={best_obj:.8f}, time={elapsed:.1f}s")

        if not any_improved:
            delta_base *= 0.6
            if delta_base < 1e-9:
                break

    return best_a


def lp_alternating(a, time_budget, max_iters=100, verbose=False):
    """LP alternating method for global improvements."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp_full(a)
        if g_lp is None:
            break

        # Line search
        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t > 0 and best_t_obj < best_obj - 1e-14:
            a = (1 - best_t) * a + best_t * g_lp
            a = np.maximum(a, 0)
            s2 = float(np.sum(a))
            a = a / s2
            best_obj = best_t_obj
            best_a = a.copy()
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"            LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}, M={M_lp:.6f}, time={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 10:
                break

    return best_a


def peak_shaving(a, time_budget, verbose=False):
    """Targeted optimization focusing on the top peaks of the autocorrelation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = np.max(conv)
        
        # Find top peaks
        sort_idx = np.argsort(conv)[::-1]
        top_k = min(10, len(sort_idx))
        top_indices = sort_idx[:top_k]

        # Try to reduce the top peaks by adjusting coordinates
        # For each top peak at index k, the contributing coordinates are those
        # where i + j = k, i.e., pairs (i, k-i)
        
        improved = False
        for k in top_indices[:3]:
            # Find coordinates that contribute to peak k
            # conv[k] = sum_{i+j=k} a[i]*a[j]
            # To reduce this, we can decrease some a[i] where i participates
            # and increase others that don't contribute to this peak
            
            contributing = []
            for i in range(min(n, k + 1)):
                j = k - i
                if 0 <= j < n:
                    contributing.append(i)
            
            if not contributing:
                continue
            
            # Try reducing a random contributing coordinate
            rng = np.random.default_rng(it * 31 + k)
            if len(contributing) > 0:
                i_idx = contributing[rng.integers(len(contributing))]
                for frac in [0.9, 0.8, 0.7]:
                    a_try = a.copy()
                    amount = a_try[i_idx] * (1 - frac)
                    a_try[i_idx] *= frac
                    # Redistribute to a coordinate that doesn't contribute to peak k
                    non_contrib = [j for j in range(n) if j not in contributing]
                    if non_contrib:
                        j_idx = non_contrib[rng.integers(len(non_contrib))]
                        a_try[j_idx] += amount
