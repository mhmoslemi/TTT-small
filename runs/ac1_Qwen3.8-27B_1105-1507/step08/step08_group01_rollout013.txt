"""

import numpy as np
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_rows(a, indices):
    """Build only the specified rows of the conv matrix A where A@g = conv(a,g)."""
    n = len(a)
    A = np.zeros((len(indices), n))
    for idx, k in enumerate(indices):
        # b[k] = sum_i a[i]*g[k-i] where 0<=i<n, 0<=k-i<n
        # So g[j] appears with coefficient a[k-j] where j = k-i, i.e., i = k-j
        # Valid: 0 <= k-j < n => k-n < j < k => max(0,k-n+1) <= j <= min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A[idx, j] = a[k - j]
    return A


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_active(a, active_indices):
    """Solve LP: min M s.t. (a*g)[k] <= M for k in active_indices, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A_active = build_conv_matrix_rows(a, active_indices)
    K = len(active_indices)

    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = A_active
    A_ub[:, n] = -1.0
    b_ub = np.zeros(K)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def solve_lp_full(a, max_constraints=200):
    """Solve LP with top max_constraints binding constraints as active set."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    # Get indices where conv is close to max
    threshold = max_conv - 0.01 * max_conv
    active = np.where(conv >= threshold)[0]
    if len(active) > max_constraints:
        # Take the top ones
        active = np.argsort(conv)[::-1][:max_constraints]
    return solve_lp_active(a, active)


def projected_gradient_step(a, lr=0.01, n_steps=50):
    """Projected gradient descent on the exact max of convolution."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        # Gradient of b[k_star] w.r.t. a[i]: a[k_star - i] for valid i
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        grad = np.zeros(n)
        # b[k] = sum_j g[j] * a[k-j], so d(b[k])/d(g[j]) = a[k-j]
        # Here we're treating a as the variable, so grad[j] = a[k_star - j] for valid j
        for j in range(j_lo, j_hi + 1):
            grad[j] = a[k_star - j]
        # Center to maintain sum
        grad = grad - np.mean(grad)
        a_new = simplex_project(a - lr * grad)
        # Check if improvement
        obj_old = compute_obj(a)
        obj_new = compute_obj(a_new)
        if obj_new < obj_old:
            a = a_new
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return a


def structural_perturbation(a, rng, strength=0.3):
    """Apply a structural perturbation to escape local minima."""
    n = len(a)
    a = a.copy()
    op = rng.integers(0, 5)

    if op == 0:
        # Segment swap
        seg_len = rng.integers(n // 10, n // 3)
        i = rng.integers(0, n - seg_len)
        j = rng.integers(0, n - seg_len)
        if abs(i - j) < seg_len:
            j = (j + seg_len + 1) % max(1, n - seg_len)
        a[i:i+seg_len], a[j:j+seg_len] = a[j:j+seg_len].copy(), a[i:i+seg_len].copy()
    elif op == 1:
        # Segment mirror
        seg_len = rng.integers(n // 10, n // 4)
        i = rng.integers(0, n - seg_len)
        a[i:i+seg_len] = a[i:i+seg_len][::-1].copy()
    elif op == 2:
        # Circular shift of segment
        seg_len = rng.integers(n // 8, n // 3)
        i = rng.integers(0, n - seg_len)
        shift = rng.integers(1, seg_len)
        a[i:i+seg_len] = np.roll(a[i:i+seg_len], shift)
    elif op == 3:
        # Gaussian bump addition/subtraction
        center = rng.integers(0, n)
        width = rng.integers(n // 20, n // 5)
        x = np.arange(n)
        bump = strength * np.exp(-0.5 * ((x - center) / width) ** 2)
        sign = 1 if rng.random() < 0.5 else -1
        a = a + sign * bump
    else:
        # Random noise
        a = a + strength * 0.1 * rng.normal(0, 1, n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a

        if best_t > 0 and best_obj_t < best_obj - 1e-13:
            a_new = (1 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                best_obj = best_obj_t
                best_a = a.copy()
                stall = 0
                # Shrink active set slightly for efficiency
                active_K = max(20, active_K - 2)
        else:
            stall += 1
            if stall % 10 == 0:
                active_K = min(active_K + 5, 100)
            if stall > 50:
                break

        if verbose and it % 50 == 0:
            print(f"      LP it{it}: obj={best_obj:.8f}, K={active_K}, t={best_t:.2f}, time={time.monotonic()-start:.1f}s")

    return best_a


def gradient_refine(a, time_budget=30.0):
    """Refine using projected gradient on exact max."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.05

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        grad = np.zeros(n)
        for j in range(j_lo, j_hi + 1):
            grad[j] = a[k_star - j]
        grad = grad - np.mean(grad)

        a_new = simplex_project(a - lr * grad)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.8
            if lr < 1e-5:
                break
    return best_a


def coordinate_descent(a, time_budget=60.0):
    """Coordinate descent with adaptive step sizes."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    delta = 1e-4

    while time.monotonic() - start < time_budget:
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            # Try small changes
            for d in [delta, -delta, delta * 3, -delta * 0.3]:
                if a[i] + d < 0:
                    continue
                a_try = a.copy()
                a_try[i] += d
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        if not improved:
            delta *= 0.5
            if delta < 1e-8:
                break
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n <= 1:
        return np.full(new_n, float(a[0]) if n == 1 else 0.0)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_structured_init(n, kind, rng):
    """Create structured initial sequences for multi-start."""
    if kind == 0:
        return np.ones(n) / n  # uniform
    elif kind == 1:
        # Ramp
        a = np.linspace(0, 1, n)
        return a / np.sum(a)
    elif kind == 2:
        # Anti-ramp (reverse ramp)
        a = np.linspace(1, 0, n)
        return a / np.sum(a)
    elif kind == 3:
        # Sine
        a = np.maximum(np.sin(np.pi * np.arange(n) / n) + 0.1, 0)
        return a / np.sum(a)
    elif kind == 4:
        # Cosine
        a = np.maximum(np.cos(np.pi * np.arange(n) / n) + 0.3, 0)
        return a / np.sum(a)
    elif kind == 5:
        # Random
        a = rng.random(n)
        return a / np.sum(a)
    elif kind == 6:
        # Two-tone
        a = np.where(np.arange(n) < n // 2, 1.0, 0.3)
        return a / np.sum(a)
    elif kind == 7:
        # Exponential decay
        a = np.exp(-np.arange(n) / (n / 3))
        return a / np.sum(a)
    elif kind == 8:
        # Triangular
        a = 1 - 2 * np.abs(np.arange(n) / (n - 1) - 0.5)
        return a / np.sum(a)
    else:
        # Weighted random with structure
        a = rng.random(n) * np.sin(np.pi * np.arange(n) / n)
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        return a
