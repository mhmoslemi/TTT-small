    i_star = int(np.argmin(grad))
    v = np.zeros(n)
    v[i_star] = 1.0
    return v

def fw_step_g(a, current_g):
    """FW step for 'fix a, optimize g' subproblem."""
    n = len(a)
    conv = compute_conv(a, current_g)

    v[j_star] = 1.0
    return v

Hmm wait, I realize the FW step as I've described it always picks a single vertex (one-hot vector). This means the sequence becomes increasingly sparse over time, which is probably not what we want.

The issue is that FW for this problem is not very effective because the "descent direction" is always towards the edges (where the gradient is zero). This is a known issue with FW for problems where the optimum is in the interior.

**Better approach: Use "away steps" (FW with away steps, aka SAVE or FRW)**

In FRW (Frank-Wolfe with away steps):
- Standard step: move towards v (the FW vertex)
- Away step: move away from the "least useful" vertex in the current support
- Pairwise step: move from one support vertex to the FW vertex

This keeps the solution less sparse.

But actually, for our problem, the optimum is NOT sparse (we want a relatively flat distribution). So standard FW is not ideal.

**Alternative: Use "conditional gradient with lazy cat" or "slided"**

Or better yet, let me just use a **modified multiplicative weights** approach:

```
Update: g_new[j] = g[j] * exp(-eta * grad[j])
Normalize
```

This is the "exponential weights" or "Hedge" algorithm, which is a natural fit for the simplex. It's equivalent to mirror descent with the negative entropy regularizer.

The key advantage over FW is that it spreads mass more evenly (not just to a single vertex).

"""
Optimized step function search using hybrid Frank-Wolfe-inspired mirror descent,
fast LP alternating with adaptive line search, systematic multi-n exploration,
and structured initialization.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a, b=None):
    """Full linear convolution via FFT. O(n log n)."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build the constraint matrix A (2n-1 x n)
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n else 0
    # Using sparse construction
    from scipy.sparse import diags, eye, vstack, hstack
    
    # For each lag k, the constraint is: sum_j A[k,j] * g[j] <= M
    # A[k, j] = a_norm[k-j] for valid j
    
    # Build using the structure: A is a "hankel-like" matrix
    # Row k has a_norm shifted by k
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            rows.append(k)
            cols.append(j)
            vals.append(a_norm[k - j])
    
    # Add -1 for M variable (column n)
    for k in range(2 * n - 1):
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    from scipy.sparse import coo_matrix
    A_ub = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.status == 0 and res.x is not None:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    
    return None, None


def mirror_descent_alternating(a, time_budget=200.0, verbose=False):
    """
    Alternating mirror descent: optimize a (with g=a fixed), then optimize g (with a fixed).
    Uses softmax-smoothed subgradients.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / np.sqrt(n)
    beta = 20.0
    stall = 0
    patience = 100
    iteration = 0
    
    while time.monotonic() - start < time_budget:
        # Step 1: Optimize a (treating current a as both a and g)
        conv = compute_conv(a, a)
        m = conv.max()
        if m < 1e-15:
            break
        shifted = beta * (conv - m)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Gradient for a: grad[i] = sum_k weights[k] * a[k-i]
        grad_a = np.zeros(n)
        active = np.where(weights > 1e-8)[0]
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            j_min = k - i_max
            j_max = k - i_min
            grad_a[i_min:i_max+1] += weights[k] * a[j_min:j_max+1]
        
        # Mirror descent update for a
        log_a = np.log(a) - lr * grad_a
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Step 2: Optimize g (treat a_new as 'g', current a as 'a')
        # Actually, let's alternate: first update a, then update g
        # For the g update, the gradient is: grad_g[j] = sum_k weights2[k] * a_new[k-j]
        conv2 = compute_conv(a_new, a)
        m2 = conv2.max()
        shifted2 = beta * (conv2 - m2)
        weights2 = np.exp(shifted2)
        weights2 /= weights2.sum()
        
        grad_g = np.zeros(n)
        active2 = np.where(weights2 > 1e-8)[0]
        for k in active2:
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            i_min = k - j_max
            i_max = k - j_min
            grad_g[j_min:j_max+1] += weights2[k] * a_new[i_min:i_max+1]
        
        log_g = np.log(a) - lr * grad_g
        log_g -= log_g.max()
        g_new = np.exp(log_g)
        g_new = np.maximum(g_new, 1e-15)
        g_new = g_new / g_new.sum()
        
        # Average the two updates
        a_try = 0.5 * a_new + 0.5 * g_new
        a_try = np.maximum(a_try, 1e-15)
        a_try = a_try / a_try.sum()
        
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try.copy()
            stall = 0
            lr = max(1e-6, lr * 0.999)
        else:
            # Try just a_new
            obj_an = compute_obj(a_new)
            if obj_an < best_obj - 1e-12:
                best_obj = obj_an
                best_a = a_new.copy()
                a = a_new.copy()
                stall = 0
            else:
                stall += 1
                if stall > patience:
                    # Restart with perturbation
                    rng = np.random.default_rng(iteration * 31 + n)
                    a = best_a * np.exp(rng.normal(0, 0.1, n))
                    a = np.maximum(a, 1e-15)
                    a = a / a.sum()
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    lr = max(1e-6, lr * 0.5)
                    beta = max(5.0, beta * 0.5)
                    stall = 0
                else:
                    lr = max(1e-6, lr * 0.98)
        
        # Anneal beta
        beta = min(100.0, beta * 1.001)
        iteration += 1
        
        if verbose and iteration % 50 == 0:
            print(f"      MD-Alt it{iteration}: obj={best_obj:.8f}, lr={lr:.6f}, beta={beta:.1f}, t={time.monotonic()-start:.1f}s")
    
    return best_a

def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, np.maximum(a, 0))
    return a_new


def structured_init(n, rng, mode=0):
    """Generate structured initializations."""

def compute_conv(a, b=None):
    """Full linear convolution via FFT. O(n log n)."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_gradient(a, beta):
    """
    Compute smoothed subgradient of max_k conv(a,a)_k w.r.t. a.
    Uses softmax weights over convolution values.
    Returns gradient vector of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    m = conv.max()
    if m < 1e-15:
        return np.zeros(n)
    shifted = beta * (conv - m)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights /= wsum

    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for k in active:
        k = int(k)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        j_min = k - i_max
        j_max = k - i_min
        grad[i_min:i_max + 1] += weights[k] * a[j_min:j_max + 1]
