"""
Optimized step function search: smoothed-max gradient with KKT plateau enforcement,
LP alternating, adaptive length sweep. Targets sub-1.5027 score.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_response(a, solvers=None):
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    if solvers is None:
        solvers = [cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue
    try:
        from scipy.optimize import linprog
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                return g_val, M_val
    except Exception:
        pass
    return None, None


def smooth_gradient_step(a, beta, lr, n_active=10):
    """One step of smoothed-max gradient descent. Returns (new_a, obj)."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))

    # Compute softmax weights (temperature-controlled)
    shifted = beta * (conv - max_val)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)

    # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j]
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(w[i:i + n], a)

    # Center to stay on sum=1 hyperplane
    grad_centered = grad - np.mean(grad)

    # Step
    a_new = a - lr * grad_centered
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a, np.inf
    a_new = a_new / s
    return a_new, compute_obj(a_new)


def kkt_plateau_enforce(a, alpha=0.3, kkt_tol=0.01):
    """
    Enforce KKT plateau structure: for active peak k*, a should be constant
    on T = {k*-i : i in supp(a)}. Blend with current a using alpha.
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    threshold = max_val * (1.0 - kkt_tol)
    active_peaks = np.where(conv >= threshold)[0]
    if len(active_peaks) == 0:
        return a

    S_mask = a > 1e-8
    S_indices = np.where(S_mask)[0]
    if len(S_indices) == 0:
        return a

    new_a = a.copy()
    for k in active_peaks:
        k = int(k)
        # Compute T = {k - i : i in S, 0 <= k-i < n}
        T = set()
        for i in S_indices:
            j = k - int(i)
            if 0 <= j < n:
                T.add(j)
        if len(T) < 2:
            continue
        T_list = list(T)
        plateau_mean = float(np.mean(new_a[T_list]))
        for j in T_list:
            new_a[j] = (1.0 - alpha) * new_a[j] + alpha * plateau_mean

    # Cap shadow values
    for k in active_peaks:
        k = int(k)
        for i in range(n):
            if not S_mask[i]:
                j = k - i
                if 0 <= j < n:
                    T_check = set()
                    for ii in S_indices:
                        jj = k - int(ii)
                        if 0 <= jj < n:
                            T_check.add(jj)
                    if j not in T_check:
                        cap_val = float(max(new_a[T_list]) if len(T_list) > 0 else 0)
                        if new_a[j] > cap_val * 1.05:
                            new_a[j] = (1.0 - alpha * 0.5) * new_a[j] + alpha * 0.5 * cap_val
        break  # Only use first active peak for capping to avoid conflicts

    s = float(np.sum(new_a))
    if s > 1e-10:
        new_a = new_a / s
    return new_a


def smoothed_gradient_optimize(a, time_budget=200.0, beta_start=5.0, beta_end=50.0,
                                lr_start=0.02, verbose=True):
    """Main smoothed gradient optimizer with adaptive parameters and KKT enforcement."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_start
    stall = 0
    beta = beta_start

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget
        # Animate beta from start to end
        beta = beta_start + (beta_end - beta_start) * min(1.0, progress * 2.0)

        # Try a few step sizes
        improved = False
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 2.0]:
            a_try, obj_try = smooth_gradient_step(a, beta, lr_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                lr = lr_try
                stall = 0
                improved = True
                break

        if not improved:
            stall += 1
            lr *= 0.7
            if stall > 20:
                # KKT enforcement
                a = kkt_plateau_enforce(a, alpha=0.3)
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
                lr = lr_start

        # Periodic KKT enforcement
        if stall % 10 == 5:
            a_kkt = kkt_plateau_enforce(a, alpha=0.2)
            obj_kkt = compute_obj(a_kkt)
            if obj_kkt < best_obj:
                best_obj = obj_kkt
                best_a = a_kkt.copy()
                a = a_kkt

        if verbose and int(elapsed) > 0 and int(elapsed) % 10 == 0 and int(elapsed) != int(elapsed - 1):
            print(f"    Smooth-grad: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.5f}, stall={stall}")

    return best_a

        best_t_val = 0.3
        best_a_local = best_a.copy()

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Try adjusting pairs that contribute to the peak
        # (a*a)[k_star] = sum_j a[j]*a[k_star-j]
        # To reduce it, decrease products: either reduce a[j] or a[k_star-j]
        contributions = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:

        for i in sorted_idx[:15]:
            if time.monotonic() - start > time_budget:

                if not target_candidates:

                target = target_candidates[np.random.randint(0, len(target_candidates))] \
                    if len(target_candidates) > 0 else (i + 1) % n

                a[i] -= delta
                a[target] += delta
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] = old_i
                    a[target] = old_t

        if not improved:
            # Random perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = 0.01 * min(a[i], 1.0 / n)

    return best_a


def generate_initializer(kind, n, rng):
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.1
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 8)
        a[:edge] = 0.4
        a[-edge:] = 0.4
        return a / np.sum(a)
    elif kind == 'two_block':
        a = np.zeros(n)
        q = n // 4
        a[:q] = 1.0
        a[3 * q:4 * q] = 1.0
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"Starting optimization with n={n_parent}, score={best_score:.8f}")

    # === Phase 1: Quick diverse initializations (60s) ===
    print("\n=== Phase 1: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    init_budget = min(60.0, remaining * 0.05)

    init_kinds = ['cosine', 'sine_taper', 'gaussian', 'random', 'plateau', 'two_block']
    for kind in init_kinds:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        a_init = generate_initializer(kind, n_parent, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init '{kind}': obj={obj_init:.8f} **")
        else:
            print(f"  Init '{kind}': obj={obj_init:.8f}")

    # Quick LP on best init
    a = best_seq.copy()
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_lp = lp_alternating_improved(a, time_budget=min(30.0, remaining * 0.3))
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp
            print(f"  Quick LP: obj={best_score:.8f} **")

    # === Phase 2: Smoothed gradient with KKT (500s) ===
    print(f"\n=== Phase 2: Smoothed Gradient + KKT (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    sg_budget = min(400.0, remaining * 0.35)

    # Multiple passes with different beta ranges
    for pass_idx in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(sg_budget / (3 - pass_idx), remaining * 0.6)
        beta_s = [5.0, 15.0, 30.0][pass_idx % 3]
        beta_e = [30.0, 80.0, 200.0][pass_idx % 3]
        lr_s = [0.02, 0.01, 0.005][pass_idx % 3]

        a_new = smoothed_gradient_optimize(a, time_budget=alloc,
                                            beta_start=beta_s, beta_end=beta_e,
                                            lr_start=lr_s, verbose=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  ** Pass {pass_idx}: obj={best_score:.8f} **")
        else:
            print(f"  Pass {pass_idx}: obj={obj_new:.8f}")
            # Perturb for next pass
            a = best_seq * np.exp(rng.normal(0, 0.02, n_parent))
            a = np.maximum(a, 0)
            a = a / np.sum(a)

    # === Phase 3: LP Alternating (300s) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(300.0, remaining * 0.25)

    for round_idx in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        beta_mom = [0.3, 0.6, 0.8][round_idx % 3]
        alloc = min(lp_budget / (3 - round_idx), remaining * 0.5)
        a_new = lp_alternating_improved(a, time_budget=alloc,
                                         momentum_beta=beta_mom, verbose=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  ** LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj_new:.8f}")
            a = best_seq * np.exp(rng.normal(0, 0.015, n_parent))
            a = np.maximum(a, 0)
            a = a / np.sum(a)

    # === Phase 4: Coordinate Descent (100s) ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        cd_budget = min(80.0, remaining * 0.1)
        a_cd = coordinate_descent_obj(best_seq, time_budget=cd_budget)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  ** CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 5: Length Sweep (200s) ===
    print(f"\n=== Phase 5: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 200),
        max(200, n_best - 100),
        max(200, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 400,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 4000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
            a_try = a_try / np.sum(a_try)

        # Quick optimization at this length
        alloc_sg = min(25.0, remaining * 0.3)
        alloc_lp = min(20.0, remaining * 0.25)

        a_opt = smoothed_gradient_optimize(a_try, time_budget=alloc_sg,
                                            beta_start=10.0, beta_end=50.0,
                                            lr_start=0.01, verbose=False)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_opt = lp_alternating_improved(a_opt, time_budget=alloc_lp,
                                             momentum_beta=0.5, verbose=False)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final Polish (remaining) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    a_final = best_seq.copy()

    while remaining > 15:
        # Alternate between smooth grad and LP
        phase = int((start_time - start_time) + (time.monotonic() - start_time)) % 2
        alloc = min(remaining * 0.3, 40.0)

        if remaining > 40:
            a_new = smoothed_gradient_optimize(a_final, time_budget=alloc,
                                                beta_start=20.0, beta_end=100.0,
                                                lr_start=0.003, verbose=False)
        else:
            a_new = lp_alternating_improved(a_final, time_budget=alloc,
                                             momentum_beta=0.5, verbose=False)

        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_final = a_new
            print(f"  Final: obj={best_score:.8f} **")
        else:
            a_final = best_seq * np.exp(rng.normal(0, 0.01, n_final))
            a_final = np.maximum(a_final, 0)
            a_final = a_final / np.sum(a_final)
            print(f"  Final: obj={obj_new:.8f} (perturbing)")

        remaining = deadline - time.monotonic()

    # Final KKT + CD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_kkt = kkt_plateau_enforce(best_seq, alpha=0.3)
        obj_kkt = compute_obj(a_kkt)
        if obj_kkt < best_score:
            best_score = obj_kkt
            best_seq = a_kkt
            print(f"  Final KKT: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_cd = coordinate_descent_obj(best_seq, time_budget=min(remaining * 0.6, 20.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
