import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ z gives convolution a*z."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_fast(a):
    """
    Solve LP: min M s.t. (a*z)_k <= M for all k, sum(z)=1, z>=0.
    Returns (z_normalized, M_opt, active_mask) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    
    # Decision variables: [z_0, ..., z_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    # Inequality constraints: A @ z - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality constraint: sum(z) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.status == 0:
        z_val = np.maximum(res.x[:n], 0)
        zs = float(np.sum(z_val))
        if zs > 1e-10:
            z_val = z_val / zs
            M_val = float(res.x[n])
            # Get active constraints from dual
            try:
                dual = res.ineqlin  # dual values for inequality constraints
                if dual is not None:
                    active_mask = np.array(dual) > 1e-6
                else:
                    active_mask = np.zeros(2*n-1, dtype=bool)
            except:
                active_mask = np.zeros(2*n-1, dtype=bool)
            return z_val, M_val, active_mask
    
    return None, None, None

def coordinate_descent(a, time_budget, verbose=False):
    """
    Coordinate descent: try adjusting each coordinate to reduce the objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)
    
    while time.monotonic() - start < time_budget:
        improved = False
        # Random order of coordinates
        order = rng.permutation(n)
        
        for idx in order:
            if time.monotonic() - start > time_budget:
                    break
            
            # Try different values for a[idx]

def pairwise_refine(a, time_budget, verbose=False):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ z gives convolution a*z."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_fast(a):
    """
    Solve LP: min M s.t. (a*z)_k <= M for all k, sum(z)=1, z>=0.
    Returns (z_normalized, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})

    if res.status == 0:
        z_val = np.maximum(res.x[:n], 0)
        zs = float(np.sum(z_val))
        if zs > 1e-10:
            z_val = z_val / zs
            M_val = float(res.x[n])
            return z_val, M_val

    return None, None


def subgradient_step(a, lr, peak_idx=None):
    """
    Take a subgradient step to reduce the peak of the convolution.
    The subgradient of max_k (a*a)_k at a[j] is 2*a[k*-j] (if valid).
    """
    n = len(a)
    if peak_idx is None:
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))

    v = np.zeros(n)
    for j in range(n):
        k_minus_j = peak_idx - j
        if 0 <= k_minus_j < n:
            v[j] = 2.0 * a[k_minus_j]

    a_new = a - lr * v
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def make_asymmetric_init(n, rng):
    """Create an asymmetric initial sequence."""
    t = np.linspace(0, 1, n)
    choice = rng.integers(0, 4)

        # Tilted uniform
        a = 1.0 + 0.5 * t
    elif choice == 1:
        # Exponential decay
        a = np.exp(-2.0 * t)
    elif choice == 2:
        # Two-bump asymmetric
        a = np.exp(-20.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-20.0 * (t - 0.7) ** 2)
    else:
        # Cosine window tilted
        a = (0.5 + 0.5 * np.cos(2 * np.pi * t)) * (1.0 + 0.3 * t)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)

        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
