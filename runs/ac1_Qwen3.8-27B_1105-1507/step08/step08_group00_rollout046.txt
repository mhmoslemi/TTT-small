def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    smooth_obj = (1.0 / tau) * (np.log(float(np.sum(exp_vals))) + tau * max_conv)

def backtracking_smooth_descent(a, time_budget=300.0, tau_init=5.0, tau_end=1000.0, verbose=False):
    """
    Smoothed gradient descent with backtracking line search on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    beta = 0.9  # momentum
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        
        # Anneal tau
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        if it % 500 == 0 and verbose:
            print(f"    BSGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient at y
        g, smooth_obj, conv, p = smoothed_grad_and_obj(y, tau)
        
        # Project gradient to tangent space of simplex (remove mean)
        g_tangent = g - np.mean(g)
        
        # Backtracking line search
        armijo_c = 1e-4
        rho = 0.5
        cur_obj = compute_obj(y)
        step = lr
        
        for _ in range(10):
            v = beta * v - step * g_tangent
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            
            new_obj = compute_obj(a_new)
            if new_obj <= cur_obj + armijo_c * step * float(np.dot(g_tangent, a_new - y)):
                break
            step *= rho
        else:
            # Line search failed, use small step
            v = beta * v - step * g_tangent
            a_new = y + v
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = best_a.copy()
            else:
                a_new = a_new / s_new
            new_obj = compute_obj(a_new)

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            best_it = it

        a = a_new
        it += 1
        
        # Decay learning rate
        lr = max(1e-10, lr * 0.9999)
        
        # Reset if stuck
        if it - best_it > 10000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def spectral_shaping(a, time_budget=100.0, verbose=False):
    """
    Gerchberg-SAXON style spectral shaping.
    Alternates between:
    - Time domain: enforce non-negativity and support [0, n-1]
    - Frequency domain: flatten |A(f)|^2 (the power spectrum)
    
    The goal is to make the convolution as flat as possible.
    Since conv = IDFT(A^2), flat conv means A^2 is concentrated at DC,
    i.e., |A| is roughly constant and phase is roughly 0 or π.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # DFT size
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    it = 0
    stall = 0
    damping = 0.5
    
    while time.monotonic() - start < time_budget:
        # Forward DFT
        A = fft(a, N)
        
        # In frequency domain: we want |A|^2 to be flat
        # But we also need A^2 to have small IDFT peak
        # Strategy: modify A to reduce peaks in IDFT(A^2)
        
        # Compute current convolution
        B = A * A
        b = ifft(B).real[:2*n-1]
        max_b = float(np.max(b))
        
        # Find lags with high values
        # Try to reduce them by modifying A
        
        # Approach: for each frequency bin j, the contribution to b[k] is A[j]^2 * exp(2πijk/N) / N
        # We want to reduce the peak. A simple approach: reduce |A[j]| for bins that
        # contribute most to the peak.
        
        # Alternative: use a "flatness" target
        # Target: |A_target[j]| = sqrt(mean(|A|^2)) for all j
        # Blend: A_new = (1-d)*A + d * A * target/|A|
        
        mag_A = np.abs(A)
        mean_mag2 = np.mean(mag_A ** 2)
        target_mag = np.sqrt(mean_mag2)
        
        # Avoid division by zero
        safe_mag = np.maximum(mag_A, 1e-10)
        
        # Create target spectrum with flat magnitude
        # Preserve phase, adjust magnitude
        phase_A = np.angle(A)
        A_flat = target_mag * np.exp(1j * phase_A)
        
        # Blend
        A_new = (1.0 - damping) * A + damping * A_flat
        
        # Also try to make A^2 more "DC-like"
        # A^2 should be close to constant (real, positive)
        # This means A should have roughly constant phase (0 or π) and constant magnitude
        
        # Time domain: enforce constraints
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                damping *= 0.9
                if damping < 0.01:
                    break
                stall = 0
        
        # Adaptively increase damping
        if it % 20 == 0:
            damping = min(0.9, damping * 1.05)
        
        it += 1
        
        if it % 100 == 0 and verbose:
            elapsed = time.monotonic() - start
            print(f"    Spectral it{it}: obj={best_obj:.8f}, damping={damping:.3f}, t={elapsed:.0f}s")

    return best_a

def solve_qp_equalize(a, M_bound, time_budget=30.0):
    """
    Solve QP: min ||conv(a,g) - c||^2 s.t. sum(g)=1, g>=0, conv(a,g) <= M_bound
    where c = mean(conv(a,g)) = (sum a)(sum g)/(2n-1) = sum(a)/(2n-1) (since sum(g)=1)
    
    This is a convex QP that tries to equalize the convolution.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    num_lags = 2 * n - 1
    
    # Build convolution matrix
    A_matrix = np.zeros((num_lags, n))
    for j in range(n):
        for k in range(max(0, j), min(num_lags, j + n)):
            i = k - j
            if 0 <= i < n:
                A_matrix[k, j] = a_norm[i]
    
    # Target: c = sum(a_norm) / (2n-1) = 1/(2n-1)
    c_val = 1.0 / (2 * n - 1)
    target = np.full(num_lags, c_val)
    
    # QP: min (A@x - target)'(A@x - target) s.t. 1'x = 1, x >= 0, A@x <= M_bound
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g) == 1, A_matrix @ g <= M_bound]
    objective = cp.Minimize(cp.sum_squares(A_matrix @ g - target))
    prob = cp.Problem(objective, constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    
    return None

3. **New: "Peak reduction" heuristic**:

   - The peak is at conv[k*] = sum_i a[i]*a[k*-i]

   - But reducing a[i] might increase other conv values

4. **New: "Iterative reweighting"**:
   - Start with uniform weights
   - At each iteration, identify which a[i] contribute most to the peak
   - Decrease their weight, increase others
   - This is like a multiplicative weight update

5. **Exploit symmetry**: If a has some symmetry, the convolution has structure. We can try to impose or discover symmetry.

Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Active-set LP (only enforce near-max constraints) for much faster LP solves
2. Multiplicative weight updates for peak reduction

4. Spectral domain shaping (GS-like)

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad_fft(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*conv[k])) w.r.t. a.
    Returns (gradient, smoothed_value, conv, softmax_weights).
    """
    n = len(a)
    conv = compute_conv(a)
    
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    smooth_val = (1.0 / tau) * (np.log(float(np.sum(exp_vals))) + tau * max_conv)
    
    # Gradient: g[m] = 2 * sum_k p[k] * a[k-m] (where 0<=k-m<n)
    N = 1
    while N < 2 * n:
        N <<= 1
    A = fft(a, N)
    P = fft(p, N)
    g = 2.0 * ifft(A * np.conj(P)).real[:n]
    
    return g, smooth_val, conv, p


def lse_descent(a, time_budget=200.0, tau_init=5.0, tau_end=2000.0, verbose=False):
    """Log-sum-exp smoothed descent with adaptive step and momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    mu = 0.95  # momentum (higher than before for smoother trajectory)
    it = 0
    best_it = 0
    patience = 5000

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.5)

        if it % 1000 == 0 and verbose:
            print(f"  LSE it{it}: obj={best_obj:.8f}, tau={tau:.0f}, lr={lr:.6f}, t={elapsed:.0f}s")

        # Nesterov
        y = a + mu * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, _, conv, p = smooth_grad_fft(y, tau)
        
        # Tangent space projection (for simplex)
        g_tan = g - g.mean()
        
        # Step
        v = mu * v - lr * g_tan
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            # Increase lr slightly on improvement
            lr = min(lr * 1.001, 5.0 / np.sqrt(n))
        else:
            lr = max(lr * 0.999, 1e-10)

        a = a_new
        it += 1

        if it - best_it > patience:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 3.0 / np.sqrt(n) * 0.3
            best_it = it
            patience = max(1000, patience // 2)

    return best_a
