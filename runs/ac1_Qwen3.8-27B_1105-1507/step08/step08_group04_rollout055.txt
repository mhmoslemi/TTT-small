   def _fft_size(n):
       size = 1
       while size < n:
           size <<= 1
       return size

   def full_conv(a):
       n = len(a)
       size = _fft_size(2*n - 1)
       fa = rfft(a, size)
       return irfft(fa * fa, size)[:2*n-1]

   def compute_obj_normalized(a):
       """Assumes sum(a) = 1. Returns 2n * max(conv(a,a))."""
       n = len(a)
       conv = full_conv(a)

       return a / s

   def compute_subgradient(a, K):
       """Compute normalized subgradient for active set K.
       Returns g of length n where g[m] = (2/|K|) * sum_{k in K} a[k-m] * 1[0<=k-m<n]"""

       scaled -= np.max(scaled)  # numerical stability
