def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
    return g


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            a_try = (1.0 - t) * a + t * direction
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                objs.append(np.inf)
            else:
                a_try = a_try / s
                objs.append(compute_obj(a_try))
        if min(objs) < np.inf:
            # Parabolic interpolation
            denom = (objs[0] - 2 * objs[1] + objs[2])
            if abs(denom) > 1e-15:
                t_vertex = t1 - 0.5 * (objs[0] - objs[2]) / (objs[0] - 2 * objs[1] + objs[2]) * 0.2
                t_vertex = np.clip(t_vertex, 0.05, 1.5)
                a_try = (1.0 - t_vertex) * a + t * direction
                a_try = np.maximum(a_try, 0)
                s = float(np.sum(a_try))
                if s > 1e-10:
                    a_try = a_try / s
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_t = t_vertex

    a_best = (1.0 - best_t) * a + best_t * direction
    a_best = np.maximum(a_best, 0)
    s = float(np.sum(a_best))
    if s > 1e-10:
        a_best = a_best / s
    return a_best, best_t, best_obj


def lp_alternating(a, budget_s, m, rows, indptr, verbose=False):
    """
    Aggressive LP alternating with thorough line search and escape mechanisms.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    it = 0
    use_maxsum = False

    while time.monotonic() - start < budget_s:
        remaining = budget_s - (time.monotonic() - start)
        if remaining < 5.0:
            break

        it += 1

        if use_maxsum:
            conv_a = compute_conv(a)
            M_cur = float(np.max(conv_a))
            g_lp = solve_lp_maxsum(a, M_cur, m, rows, indptr)
            if g_lp is None:
                g_lp, M_lp = solve_lp_minpeak(a, m, rows, indptr)
                if g_lp is None:
                    break
        else:
            g_lp, M_lp = solve_lp_minpeak(a, m, rows, indptr)
            if g_lp is None:
                conv_a = compute_conv(a)
                M_cur = float(np.max(conv_a))
                g_lp = solve_lp_maxsum(a, M_cur, m, rows, indptr)
                if g_lp is None:
                    break
                M_lp = M_cur

        # Line search
        a_new, t_best, obj_new = line_search(a, g_lp, best_obj)

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = best_a
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_best:.3f}, M={M_lp:.6f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 20:
                # Spectral perturbation: add small sinusoid at dominant frequency
                conv_a = compute_conv(a)
                k_peak = int(np.argmax(conv_a))
                # Perturb with a shifted bump
                center = rng.integers(n // 4, 3 * n // 4)
                width = max(5, n // 20)
                pert = np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
                # Also try an asymmetric perturbation
                pert2 = np.exp(-((np.arange(n) - n * 0.3) ** 2) / (2 * (n // 15) ** 2))
                choice = rng.integers(0, 3)
                if choice == 0:
                    a = a * (1.0 + 0.15 * pert / (pert.max() + 1e-10))
                elif choice == 1:
                    a = a + 0.05 * pert2 / (pert2.sum() + 1e-10)
                else:
                    a = a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                use_maxsum = not use_maxsum
                if verbose:
                    print(f"    LP it{it}: ESCAPE obj={best_obj:.8f}, el={time.monotonic()-start:.1f}s")

    return best_a


def subgradient_method(a, budget_s, K=8, verbose=False):
    """
    Fast subgradient method with adaptive temperature.
    Much faster per iteration than LP (no LP solve).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)
    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 3.0 / np.sqrt(n)
    stall = 0
    patience = 50
    it = 0

    while time.monotonic() - start < budget_s:
        conv = compute_conv(y)
        max_val = conv.max()
        if max_val < 1e-15:
            break

        # Top-K peaks with temperature weighting
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        spread = (peak_vals[0] - peak_vals[-1]) / (max_val + 1e-15)
        temp = 5.0 / (1.0 + spread * 20.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / (max_val + 1e-15))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-10:
                g += w * grad_conv_at_peak(y, int(k))

        # Normalize subgradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        lr = lr0 / (1.0 + t_k * 0.1)
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)
        y_new = project_simplex(y_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Restart with perturbation
                a_new = best_a * np.exp(rng.normal(0, 0.1, n))
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                y = a_new.copy()
                a_prev = a_new.copy()
                y_new = a_new.copy()
                t_k = 1.0
                stall = 0
                patience = int(patience * 1.5)

        a_prev = a_new.copy()
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        y = y_new
        t_k = t_new
        it += 1

        if verbose and it % 200 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, t_k={t_k:.0f}, el={time.monotonic()-start:.1f}s")

    return best_a


def smooth_laplace(a, budget_s, T=50.0, verbose=False):
    """
    Smooth minimization using L-BFGS-B on log-sum-exp approximation.
    """
    from scipy.optimize import minimize as sp_minimize
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_grad(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = compute_conv(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        e = np.exp(z)
        sum_e = e.sum()
        lse = max_c + np.log(sum_e) / T
        obj_val = 2.0 * n * lse
        w = e / sum_e
        g = np.zeros(n)
        # Only compute gradient for significant weights
        active = w > 1e-6
        for k in np.where(active)[0]:
            g += w[k] * grad_conv_at_peak(x, int(k))
        g *= 2.0 * n
        # Normalize gradient (since we're on the simplex via constraint)
        g = g - np.sum(g * x) * x  # project to tangent space
        return obj_val, g

    try:
        res = sp_minimize(obj_grad, a, method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': 300, 'maxfun': 600, 'disp': False})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                if verbose:
                    print(f"    Smooth: obj={best_obj:.8f}")
    except Exception:
        pass

    return best_a


def direct_peak_reduction(a, k_peak, budget_s):
    """
    Directly try to reduce the peak at lag k_peak by moving mass.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - start < budget_s:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Find which pairs (i, j) contribute to c_k_star: i + j = k_star (in convolution sense)
        # Actually (a*a)_k = sum_i a_i * a_{k-i}
        # So contributing pairs are (i, k-i) for valid i
        contributing = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributing.append((a[i] * a[j], i, j))
        contributing.sort(reverse=True)

        # Find non-contributing positions
        non_contributing = set(range(n))
        for _, i, j in contributing:
            non_contributing.discard(i)
            non_contributing.discard(j)
        non_contributing = list(non_contributing)

        if not non_contributing:
            break

        improved = False
        for ci, (val, i, j) in enumerate(contributing[:20]):
            if val < 1e-14 or ci >= 20:
                break
            for target in non_contributing[:10]:
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                    src = i if a[i] >= a[j] else j
                    delta = frac * a[src]
                    if delta < 1e-15:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        # Asymmetric Gaussian
        center = n * 0.4
        width = n * 0.25
        a = np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 - 0.5 * x
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 4:
        # Asymmetric: more mass on left
        x = np.linspace(0, 1, n)
        a = (1.0 - x) ** 2 + 0.1
        a = a / a.sum()
    elif mode == 5:
        # Frequency-based
        freqs = rng.integers(1, 4, size=2)
        phases = rng.uniform(0, 2 * np.pi, size=2)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 6:
        # Two bumps
        centers = [n * 0.25, n * 0.7]
        widths = [n * 0.1, n * 0.15]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Power law decay from left
        x = np.arange(n, 0, -1)
        a = x ** 0.5
        a = a / a.sum()
    elif mode == 8:
        # Sinusoidal modulation of uniform
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x)
        a = np.maximum(a, 0.1)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid optimization combining:
    - Fast subgradient for quick initial improvement
    - LP alternating with thorough line search (main workhorse)
    - Smooth L-BFGS optimization
    - Direct peak reduction
    - Multi-n exploration
    - Diverse restarts with structured perturbations
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_best = len(best_seq)
    m_best = 2 * n_best - 1
    m_best, rows_best, indptr_best = build_lp_matrices(n_best)

    # === Phase 1: Quick subgradient (~120s) ===
    print(f"\n=== Phase 1: Fast Subgradient (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 120.0)
    if p1_budget > 10:
        a_sg = subgradient_method(best_seq, time_budget=p1_budget, K=8, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgradient: obj={best_score:.8f} **")
        else:
            print(f"  Subgradient: obj={obj_sg:.8f} (no improvement)")
    else:
        print("  Skipped (insufficient time)")

    # === Phase 2: LP alternating (main workhorse, ~700s) ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p2_total = min(remaining * 0.45, 700.0)

    # Split into rounds
    n_rounds = 3
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if rnd == 0:
            alloc = p2_total * 0.5
        elif rnd == 1:
            alloc = p2_total * 0.3
        else:
            alloc = p2_total * 0.2

        alloc = min(alloc, remaining * 0.7)
        if alloc < 20:
            break

        # For later rounds, perturb to escape
        if rnd > 0 and best_seq is not None:
            n_cur = len(best_seq)
            noise_scale = 0.03 * (rnd + 1)
            a_pert = best_seq * np.exp(rng.normal(0, noise_scale, n_cur))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        else:
            a_pert = best_seq.copy()

        m_cur = 2 * len(a_pert) - 1
        m_cur, rows_cur, indptr_cur = build_lp_matrices(len(a_pert))

        a_lp = lp_alternating(a_pert, time_budget=alloc, m=m_cur, rows=rows_cur,
                              indptr=indptr_cur, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Smooth optimization (~150s) ===
    print(f"\n=== Phase 3: Smooth L-BFGS ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 150.0)
    if p3_budget > 15:
        for T_val in [30.0, 80.0]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            t_budget = min(p3_budget / 2.0, remaining * 0.5)
            if t_budget < 10:
                continue
            a_sm = smooth_laplace(best_seq, time_budget=t_budget, T=T_val, verbose=True)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Smooth T={T_val}: obj={best_score:.8f} **")

    # === Phase 4: Direct peak reduction (~80s) ===
    print(f"\n=== Phase 4: Direct Peak Reduction ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.06, 80.0)
    if p4_budget > 10:
        conv_best = compute_conv(best_seq)
        k_peak = int(np.argmax(conv_best))
        a_pr = direct_peak_reduction(best_seq, k_peak, time_budget=p4_budget)
        obj_pr = compute_obj(a_pr)
        if obj_pr < best_score:
            best_score = obj_pr
            best_seq = a_pr.copy()
            print(f"  Peak reduction: obj={best_score:.8f} **")

    # === Phase 5: Multi-n exploration (~250s) ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(100, n_current - 50),
        max(100, n_current - 20),
        n_current + 20,
        n_current + 50,
        n_current + 100,
        400, 600, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 100 <= n <= 2000 and n != n_current]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= 5:
            break
        n_tried += 1

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()
        m_try = 2 * n_try - 1
        m_try, rows_try, indptr_try = build_lp_matrices(n_try)

        # Quick subgradient
        alloc_sg = min(remaining * 0.05, 30.0)
        if alloc_sg > 10:
            a_try = subgradient_method(a_try, time_budget=alloc_sg, K=5)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.08, 50.0)
        if alloc_lp > 15:
            a_try = lp_alternating(a_try, time_budget=alloc_lp, m=m_try,
                                   rows=rows_try, indptr=indptr_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts (~150s) ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best_curr = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r * 3 + 1) % 9
        a_r = diverse_init(n_best_curr, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.12, 60.0)
        m_r = 2 * n_best_curr - 1
        m_r, rows_r, indptr_r = build_lp_matrices(n_best_curr)
        a_opt = lp_alternating(a_r, time_budget=restart_budget, m=m_r,
                               rows=rows_r, indptr=indptr_r)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    m_final = 2 * n_final - 1
    m_final, rows_final, indptr_final = build_lp_matrices(n_final)

    # LP polish
    if remaining > 40:
        lp_alloc = min(remaining * 0.35, 100.0)
        a_lp = lp_alternating(best_seq, time_budget=lp_alloc, m=m_final,
                              rows=rows_final, indptr=indptr_final, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        ns_alloc = min(remaining * 0.2, 50.0)
        a_ns = subgradient_method(best_seq, time_budget=ns_alloc, K=10)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    # Peak reduction
    remaining = deadline - time.monotonic()
    if remaining > 20:
        conv_best = compute_conv(best_seq)
        k_peak = int(np.argmax(conv_best))
        pr_alloc = min(remaining * 0.15, 30.0)
        a_pr = direct_peak_reduction(best_seq, k_peak, time_budget=pr_alloc)
        obj_pr = compute_obj(a_pr)
        if obj_pr < best_score:
            best_score = obj_pr
            best_seq = a_pr.copy()
            print(f"  Final PR: obj={best_score:.8f} **")

    # One more LP if time
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.4, 30.0),
                                m=m_final, rows=rows_final, indptr=indptr_final)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
