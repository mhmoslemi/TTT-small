def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.

Key innovations over previous approach:
1. LP-based fixed-point iteration as PRIMARY optimizer (not just polishing)
2. Semismooth Newton method on log-sum-exp smoothed objective
3. Log-space (multiplicative) parameterization for natural non-negativity
4. Multi-scale tau exploration (coarse-to-fine annealing)
5. Multi-length search with spectral initialization

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv_k(a))) and the conv."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_c))
    p = exp_vals / float(np.sum(exp_vals))
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    return g, conv


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g) (cross-conv)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[:n]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None

def semismooth_newton(a, time_budget=200.0, tau=200.0, verbose=False):
    """
    Semismooth Newton method on log-sum-exp smoothed objective.
    Uses the Hessian (sum p^2 * outer(a,a) terms) for superlinear convergence.
    Works in log-space for natural non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    log_a = np.log(a)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(np.exp(log_a))
        cur_a = np.exp(log_a)
        cur_a = cur_a / cur_a.sum()

        max_c = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_c))
        p = exp_vals / float(np.sum(exp_vals))

        # Gradient of smoothed objective
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(cur_a, N)
        g_full = ifft(P * np.conj(A)).real
        grad = 2.0 * g_full[:n]

        # Compute Hessian-vector product direction (Newton step)
        # H = 2 * sum_k p[k] * (outer(a, a) at lag k)
        # In log-space: J = diag(a), H_w = J^T H J
        # Newton step: H_w @ d = -grad
        # Simplified: use inverse of diagonal of H as preconditioner + line search

        # Diagonal of H in original space: H[i,i] = 2 * sum_k p[k] * a[i]^2 (for valid k)
        # In log-space: H_log[i,i] = H[i,i] * a[i]^2 = 2 * sum_k p[k] * a[i]^4
        # Simplified preconditioner: use a[i] as weight
        # d_log[i] = -grad[i] / (a[i] * H_diag[i]) * a[i]^2 = -grad[i] * a[i] / H_diag[i]

        # Actually, let's use a simpler approach: d = -grad / (grad^2 + eps)
        # This is like preconditioned gradient descent with per-coordinate step

        h_diag = a ** 2 + 1e-10
        d = -grad / (h_diag + np.linalg.norm(grad) * 1e-6)

        # Line search in log-space
        best_step = 0.01
        best_f = np.inf
        cur_f = float(np.max(compute_conv(cur_a)))
        for step in [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005]:
            log_a_try = log_a + step * d
            a_try = np.exp(log_a_try)
            a_try = a_try / a_try.sum()
            f_try = float(np.max(compute_conv(a_try)))
            if f_try < best_f:
                best_f = f_try
                best_step = step

        log_a = log_a + best_step * d
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            it = 0
        else:
            it += 1
            # Reduce tau slightly to smooth more
            if it % 50 == 0:
                tau = max(5.0, tau * 0.5)

        if verbose and it % 20 == 0:
            print(f"    SN it{it}: obj={best_obj:.8f}, tau={tau:.1f}, elapsed={time.monotonic()-start:.0f}s")

        if it > 500:
            reset_t = time.monotonic() - start
            if reset_t > time_budget * 0.5:
                break
            log_a = np.log(best_a)
            tau = max(5.0, tau * 0.5)
            it = 0

    return best_a
