def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        lo = j
        hi = min(m - 1, j + n - 1)
        length = hi - lo + 1
        A[lo:hi + 1, j] = a[:length]
    return A

def compute_subgradient(a, frac=0.3):
    """Compute subgradient of max(conv(a,a)) with weighted peaks."""
    n = len(a)
    conv = compute_conv(a)
    peak = np.max(conv)
    thresh = (1.0 - frac) * peak
    mask = conv >= thresh
    k_indices = np.where(mask)[0]
    
    # Weight by height above threshold
    w = conv[k_indices] - thresh
    w_sum = np.sum(w)
    if w_sum < 1e-10:
        w = np.ones(len(k_indices)) / len(k_indices)
    else:
        w = w / w_sum
    
    # grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices
    # Use direct computation (OK for n <= 5000)
    w_full = np.zeros(2 * n - 1)
    w_full[k_indices] = w
    
    grad = np.zeros(n)
    for j in range(n):
        # Term 1: sum_{k=j}^{j+n-1} w[k] * a[k-j]
        lo1 = j
        hi1 = min(2 * n - 2, j + n - 1)
        if lo1 <= hi1:
            grad[j] += np.dot(w_full[lo1:hi1+1], a[:hi1-lo1+1])
        # Term 2: sum_{k=max(0,j-n+1)}^{j} w[k] * a[j-k]
        lo2 = max(0, j - n + 1)
        hi2 = j
        if lo2 <= hi2:
            grad[j] += np.dot(w_full[lo2:hi2+1], a[1:hi2-lo2+2][::-1])
    
    # The full subgradient of 2n*max(conv)/sum^2 (with sum=1) is 2n * (subgrad of max(conv))
    # But we need to account for the constraint sum(a)=1, so we center the gradient
    grad *= 2.0 * n
    # Project onto the tangent space of the simplex: subtract the mean
    grad -= np.mean(grad)
    return grad

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
