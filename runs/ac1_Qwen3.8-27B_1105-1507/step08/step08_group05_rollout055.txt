```python
def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A

        return None, None

    threshold = max_conv * (1.0 - active_fraction * 0.5)
```

**Approach: Direct optimization using the structure of the problem**

The key insight is that `conv(a,a)[k]` for the peak position `k*` is:
`conv[k*] = sum_{i} a[i] * a[k*-i]`

This is a sum of products. To minimize it, we want to minimize the "overlap" between `a` and its reverse (shifted by `k*`).

For a fixed `k*`, this is like a "minimum overlap" problem. The optimal solution would spread the mass of `a` such that it minimizes the overlap with its shifted reverse.

4. For each hot spot, try to reduce one of the pair by moving mass to a "cold spot" (a position that doesn't contribute to the peak).
5. Use a small step size and check if the objective improved.

The objective is `f(a) = 2n * max_k conv(a,a)[k] / (sum a)^2`.

Let `k*` be the argmax. Then:
`df/da[m] = 2n * (d/d a[m] conv(a,a)[k*] * (sum a)^2 - max_k conv * 2 * (sum a) * 1) / (sum a)^4`
`= 2n * (2*a[k*-m] * (sum a)^2 - max_k conv * 2 * (sum a)) / (sum a)^4`
`= 4n * (a[k*-m] * (sum a) - max_k conv) / (sum a)^3`

When `sum a = 1`:
`df/da[m] = 4n * (a[k*-m] - max_k conv)`

This is very simple! The gradient is proportional to `a[k*-m] - max_k conv`. When `a[k*-m] > max_k conv`, the gradient is positive (increase in `a[m]` increases the objective), and vice versa.

This gives us a very clean update rule:
- Decrease `a[m]` where `a[k*-m]` is large (i.e., where the "reflected" value is large).
- Increase `a[m]` where `a[k*-m]` is small.

```python
def peak_gradient_step(a, step_size=0.1):
    """One step of the peak gradient method."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    max_conv = float(conv[k_star])
    
    # Gradient: 4n * (a[k*-m] - max_conv) for each m
    # d/da[m] = 4n * (a[k*-m] - max_conv) (when sum=1)
    # We want to move in the negative gradient direction
    grad = np.zeros(n)
    for m in range(n):
        km = k_star - m
        if 0 <= km < n:
            grad[m] = a[km]
        else:
            grad[m] = 0
    
    # Negative gradient: -(a[k*-m] - max_conv) = max_conv - a[k*-m]
    # So we increase a[m] where a[k*-m] < max_conv
    # and decrease a[m] where a[k*>-m] > max_conv
    
    # Update: a_new = a - step_size * grad (projected to simplex)
    a_new = a - step_size * (grad - np.mean(grad))  # centering helps
    a_new = project_simplex(a_new)
    return a_new
```

Actually, since it's the max of quadratics, it's piecewise quadratic and convex. We can use a three-interval search (golden section) to find the minimum.

```python
def line_search_max_quad(a, g, C_aa, C_ag, C_gg):
    """Minimize max_k [(1-t)^2 C_aa[k] + 2t(1-t) C_ag[k] + t^2 C_gg[k]] over t in [0,1]."""
    # This is a convex function of t (max of convex quadratics)
    # Use golden section search
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    def f(t):
        c = (1-t)**2 * C_aa + 2*t*(1-t) * C_ag + t**2 * C_gg
        return np.max(c)
    
    a_g = 0.0
    b_g = 1.0
    c_ = b_g - phi * (b_g - a_g)
    d_ = a_g + phi * (b_g - a_g)
    fc = f(c_)
    fd = f(d_)
    
    for _ in range(50):
        if fc < fd:
            b_g = d_
            d_ = c_
            fd = fc
            c_ = b_g - phi * (b_g - a_g)
            fc = f(c_)
        else:
            a_g = c_
            c_ = d_
            fc = fd
            d_ = a_g + phi * (b_g - a_g)
            fd = f(d_)
    
    t_opt = (a_g + b_g) / 2
    return t_opt
```

This is more efficient than `minimize_scalar` because we're evaluating the max directly without recomputing the convolution.

- Line search: O(1) evaluations (if we precompute the convolutions)

**Key insight for LP speed**: We can reduce the number of constraints by only including the "active" ones. The active set at the optimal solution consists of the constraints that are tight (i.e., `conv(a,g)[k] = M`). Typically, only a small number of constraints are active (maybe 5-20 out of 2n-1).

The problem is that we don't know which constraints are active in advance. But we can use the previous solution as a warm start: if constraint `k` was active in the previous LP, it's likely to be active again.

**Active set LP**: 
1. Start with all constraints (or a large subset).
2. Solve the LP.
3. Identify the active constraints (those with `|conv(a,g)[k] - M| < tol`).
4. Solve the LP again with only the active constraints.
5. Iterate until the active set stabilizes.

This can dramatically reduce the LP size.

```python
"""
Improved optimizer: LP alternating with golden-section line search using
precomputed cross-convolutions, multi-peak Nesterov subgradient, adaptive
perturbation, and n-sweep. Focuses on squeezing the last 0.001-0.003 from
the objective through tighter LP iteration and better 1D optimization.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def build_conv_matrix_rows(a, indices, n_total):
    """Build rows of convolution matrix for given peak indices."""
    n = len(a)
    A = np.zeros((len(indices), n))
    for ii, k in enumerate(indices):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A[ii, i] = a[k - i]
    return A


def solve_lp_minpeak(a, n_constraints=None):
    """
    LP: min M s.t. conv(a,g)[k] <= M for selected k, sum(g)=1, g>=0.
    Uses top n_constraints peaks to keep LP tractable.
    """
    n = len(a)
    conv_a = compute_conv(a)
    max_conv = float(np.max(conv_a))
    if max_conv < 1e-15:
        return None, None

    if n_constraints is not None and n_constraints < len(conv_a):
        top_idx = np.argsort(conv_a)[::-1][:n_constraints]
    else:
        top_idx = np.arange(len(conv_a))

    a_n = a / float(np.sum(a))
    A = build_conv_matrix_rows(a_n, top_idx, len(conv_a))

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CBC, cp.GLPK, cp.SCIP, cp.XPRESS, cp.MOSEK, cp.GUROBI,
                   cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def golden_section_line_search(a, g):
    """
    Golden section search for optimal t in [0,1] minimizing
    max_k conv((1-t)*a + t*g, (1-t)*a + t*g)[k].
    Uses precomputed convolutions for speed.
    """
    C_aa = compute_conv(a)
    C_ag = compute_conv(a)  # placeholder, will compute properly
    # Actually we need conv(a,g) not conv(a,a)
    # Let me compute cross-convolutions properly
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    C_ag = irfft(fa * fg, size)[:2 * n - 1]
    C_gg = irfft(fg * fg, size)[:2 * n - 1]
    C_aa = irfft(fa * fa, size)[:2 * n - 1]

    def f_of_t(t):
        c = (1.0 - t) ** 2 * C_aa + 2.0 * t * (1.0 - t) * C_ag + t ** 2 * C_gg
        return float(np.max(c))

    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = hi - phi * (hi - lo)
    d_pt = lo + phi * (hi - lo)
    fc = f_of_t(c_pt)
    fd = f_of_t(d_pt)

    for _ in range(40):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - phi * (hi - lo)
            fc = f_of_t(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + phi * (hi - lo)
            fd = f_of_t(d_pt)

    return (lo + hi) / 2.0


def lp_alternate_fast(a0, time_budget=60.0, verbose=False, n_constraints=None):
    """LP alternating with golden-section line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a, n_constraints=n_constraints)
        if g_lp is None:
            break

        t_opt = golden_section_line_search(a, g_lp)
        a_try = (1.0 - t_opt) * a + t_opt * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall = 0
            if verbose:
                print(f"    LP fast: obj={best_obj:.8f}, t={t_opt:.4f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 12:
                break

    return best_a


def nesterov_subgrad(a0, time_budget=60.0, top_k=3, verbose=False):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.06
    patience = 0

    for it in range(500000):
        if time.monotonic() - start > time_budget:
            break
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                m_start = k - i_end
                m_end = k - i_start
                grad[m_start:m_end + 1] += 2.0 * y[i_start:i_end + 1][::-1]
        grad /= len(top_idx)
        grad -= np.mean(grad)

        y_new = y - step_size * grad
        a_new = project_simplex(y_new)
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = project_simplex(y)
        a = a_new

        if it % 4 == 0 or it < 10:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 400:
                step_size = max(step_size * 0.5, 0.001)
                patience = 0

        if it % 500 == 499:
            step_size = max(step_size * 0.98, 0.002)

    return best_a


def peak_deflation(a0, time_budget=30.0, verbose=False):
    """
    Targeted peak deflation: identify the peak position, find contributing
    pairs, and move mass from hot to cold spots iteratively.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for outer in range(200):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_val = float(conv[k_star])

        # Find contributing pairs
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(i_start, i_end + 1):
            m = k_star - i
            contrib[i] += a[m]
            if m != i:
                contrib[m] += a[i]

        # Top contributors and bottom contributors
        top_contrib = np.argsort(contrib)[::-1][:5]
        bottom_contrib = np.argsort(contrib)[:5]

        improved = False
        for src in top_contrib:
            if a[int(src)] < 1e-8:
                continue
            for dst in bottom_contrib:
                if int(src) == int(dst):
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[int(src)]
                    a[int(src)] -= delta
                    a[int(dst)] += delta
                    a = np.maximum(a, 0)
                    st = float(np.sum(a))
                    if st < 1e-10:
                        a[int(src)] += delta
                        a[int(dst)] -= delta
                        continue
                    a = a / st
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[int(src)] += delta
                        a[int(dst)] -= delta
                        a = a / st
                        break
        if not improved and outer > 30:
            break

    return best_a


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
```
