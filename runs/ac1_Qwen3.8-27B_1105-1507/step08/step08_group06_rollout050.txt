"""
Optimized step function search: Fast LP-alternating with direct scipy linprog,
Nesterov subgradient polish, geometric updates, and aggressive length sweep.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv / sum^2 for given a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_best_response_lp(a):
    """
    Solve LP: min M s.t. conv(a, g) <= M, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None).
    Uses scipy linprog with HiGHS for speed.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)
    m = 2 * n - 1
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize M -> c = [0, ..., 0, 1]
    # Constraints: A @ g - M <= 0 for each row -> [A, -1] @ [g, M] <= 0
    #              sum(g) = 1
    #              g >= 0, M free
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(max(res.x[n], np.max(A @ g_val)))
                return g_val, M_val
    except Exception:
        pass
    
    return None, None

def subgradient_descent(a, time_budget=100.0):
    """
    Nesterov-accelerated subgradient on the max-convolution objective.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:

    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    x = a.copy()
    momentum = np.zeros(n)
    beta = 0.8
    lr = 0.03
    count = 0

    while time.monotonic() - start < time_budget:
        # Compute convolution at y
        conv = compute_conv(y)
        max_val = float(np.max(conv))
        
        # Active set: peaks within 5% of max
        threshold = max_val * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 50:
            active = np.argsort(conv)[::-1][:50]

        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += 2.0 * y[k - indices]
        grad /= len(active)
        
        # Center
        grad = grad - np.mean(grad)

        # Nesterov update
        t_k = 2.0 / (np.sqrt(count) + 2.0)
        momentum = beta * momentum - lr * grad
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
        else:
            y_new = y_new / s_new

        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        y = y_new
        count += 1

        if count % 5 == 0:
            obj = compute_obj(y)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = y.copy()

        # Adaptive LR
        if count % 500 == 0:
            lr *= 0.9

    return best_a


def geometric_update(a, g, t):
    """Geometric (multiplicative) interpolation."""
    # a_new = a^(1-t) * g^t, handling zeros
    eps = 1e-30
    a_safe = np.maximum(a, eps)
    g_safe = np.maximum(g, eps)
    a_new = np.exp((1-t) * np.log(a_safe) + t * np.log(g_safe))
    a_new = a_new / np.sum(a_new)
    return a_new

def mirror_descent_update(a, conv, eta):
    """Mirror descent: a_new = a * exp(-eta * (conv_a_k - M)) for active k."""
    n = len(a)
    max_val = float(np.max(conv))
    # For each active k, the "pressure" on a_i is 2*a_{k-i}
    # Approximate: reduce a_i proportionally to how much it contributes to peaks
    pressure = np.zeros(n)
    threshold = max_val * 0.9
    active = np.where(conv >= threshold)[0]
    for k in active[:20]:
        k = int(k)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_max >= i_min:
            indices = np.arange(i_min, i_max + 1)
            pressure[indices] += a[k - indices]
    
    a_new = a * np.exp(-eta * pressure)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new

"""
Step function optimization via LP-alternating with Anderson acceleration,
geometric updates, Nesterov subgradient polish, and length sweep.
Uses scipy HiGHS for fast LP solves.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_best_response_lp(a):
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)
    m = 2 * n - 1
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(np.max(A @ g_val))
                return g_val, M_val
    except Exception:
        pass
    return None, None


def anderson_accelerate(x_history, F_history, m=5):
    """
    Anderson acceleration step.
    x_history: list of recent x values (oldest first)
    F_history: list of recent F(x) = BR(x) - x values (residuals)
    Returns the accelerated next step direction.
    """
    n_hist = min(m, len(x_history))
    if n_hist < 2:
        # Simple step
        return F_history[-1]
    
    # Build the matrix
    # We want to find alpha_0...alpha_{k-1} minimizing ||r_k - sum alpha_i (r_k - r_i)||^2
    # where r_k = F(x_k) is the residual
    
    r_k = F_history[-1]
    
    # gamma_k = r_k - sum_{i=k-m+1}^{k} something... 
    # Simplified: use the "residual difference" formulation
    
    # Gather residuals
    residuals = F_history[-n_hist:]
    rs = r_k - residuals[-1]  # This is 0 by definition... 
    
    # Let me use the standard form:
    # Given x_0, ..., x_k and f(x_i) = x_i - F(x_i) (residual)
    # Find x_{k+1} = x_k + sum_i alpha_i (x_i - x_{i-1})
    # minimizing ||f(x_k) + sum_i alpha_i (f(x_k) - f(x_{k-i}))||^2
    
    # This is getting complex. Let me use a simpler "G-MRES" style approach.
    # x_{k+1} = x_k + V @ alpha where V is the matrix of recent steps
    # minimizing ||r_k + V_r @ alpha||^2 where V_r is residual differences
    
    steps = []
    res_diffs = []
    for i in range(1, n_hist):
        step = x_history[-i] - x_history[-i-1]
        steps.append(step)
        rd = residuals[-1] - residuals[-(i+1)] if i+1 <= len(residuals) else r_k
        res_diffs.append(rd)
    
    if not steps:
        return r_k
    
    V = np.column_stack(steps)  # n x k
    Vr = np.column_stack(res_diffs)  # n x k
    
    # Solve min ||r_k + Vr @ alpha||^2
    # Normal equations: Vr^T Vr @ alpha = -Vr^T r_k
    try:
        A_mat = Vr.T @ Vr + 1e-12 * np.eye(len(Vr))
        b_vec = -Vr.T @ r_k
        alpha = np.linalg.solve(A_mat, b_vec)
        direction = V @ alpha
    except np.linalg.LinAlgError:
        direction = r_k
    
    return direction

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    y = a.copy()
    momentum = np.zeros(n)
    beta = 0.8
    lr = 0.02
    count = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        max_val = float(np.max(conv))
        
        threshold = max_val * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 50:
            active = np.argsort(conv)[::-1][:50]

        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += 2.0 * y[k - indices]
        grad /= len(active)
        grad = grad - np.mean(grad)

        momentum = beta * momentum - lr * grad
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
        else:
            y_new = y_new / s_new

        t_k = 2.0 / (np.sqrt(count) + 2.0)
        x_new = y_new + (1 - t_k) * (y_new - y)
        x_new = np.maximum(x_new, 0)
        sx = float(np.sum(x_new))
        if sx > 1e-10:
            x_new = x_new / sx

        y = y_new
        count += 1

        if count % 10 == 0:
            obj = compute_obj(y)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = y.copy()

        if count % 1000 == 0:
            lr *= 0.95

    return best_a


def geometric_lp(a, time_budget=150.0):
    """LP alternating with geometric (multiplicative) updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_best_response_lp(a)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_a_local = best_a.copy()
        
        for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
            eps = 1e-30
            a_safe = np.maximum(a, eps)
            g_safe = np.maximum(g_lp, eps)
            a_try = np.exp((1-t) * np.log(a_safe) + t * np.log(g_safe))
            a_try = a_try / np.sum(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall >= 5:
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0

    return best_a


def resample_sequence(a, new_n):
    n_old = len(a)
    if n_old == new_n:
        return a.copy()
    if n_old == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def direct_subgradient(a, time_budget=100.0):
    """
    Direct subgradient with adaptive step size (Nesterov's method).
    Optimizes 2n * max_k (a*a)[k] subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    y = a.copy()
    v = np.zeros(n)
    beta = 0.85
    lr = 0.01
    count = 0
    best_lr = lr

    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        max_val = float(np.max(conv))
        
        # Identify active peaks
        threshold = max_val * 0.97
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        # Cap active set
        if len(active) > 30:
            active = np.argsort(conv)[::-1][:30]
        
        # Subgradient of max_k (a*a)[k]
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += 2.0 * y[k - indices]
        grad /= len(active)
        
        # Project to simplex tangent (sum = 0)
        grad = grad - np.mean(grad)
        
        # Clip gradient for stability
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-6:
            grad = grad / grad_norm  # Normalize
        
        # Update with momentum
        v = beta * v - lr * grad
        y_new = y + v
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
        else:
            y_new = y_new / s_new
        
        # Nesterov lookahead
        t_k = 2.0 / (np.sqrt(count) + 2.0)
        y = y_new
        count += 1
        
        if count % 5 == 0:
            obj = compute_obj(y)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = y.copy()
                best_lr = lr
        
        # Adaptive learning rate
        if count % 200 == 0:
            lr = max(lr * 0.92, 0.0005)

    return best_a
