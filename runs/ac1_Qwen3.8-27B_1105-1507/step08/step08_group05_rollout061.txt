def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond]) - 1
    theta = cssv[i] / (i + 1)
    return np.maximum(v - theta, 0)


def smoothed_gradient(conv, a, lam):
    """
    Compute gradient of (1/lam)*log(sum exp(lam*conv_k)) w.r.t. a.
    Uses FFT cross-correlation for efficiency.
    Returns grad_a of length n.
    """
    n = len(a)
    # Softmax weights over conv entries
    shifted = lam * (conv - np.max(conv))
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        return np.zeros(n)
    w /= w_sum
    # grad_a[m] = 2 * sum_k w[k] * a[k-m] for valid indices
    # = 2 * cross_correlation(w, a)[m]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    W = rfft(w, size)
    A = rfft(a, size)
    R = irfft(W * np.conj(A), size)
    return 2.0 * R[:n]


def subgradient_max(conv, a, top_k=5):
    """
    Subgradient of max(conv) using top-k peaks.
    Returns grad_a of length n.
    """
    n = len(a)

    top_idx = sorted(set(int(k) for k in top_idx))
    grad = np.zeros(n)
    for k in top_idx:

        i_end = min(n - 1, k)

            m_end = k - i_start
            grad[m_start:m_end + 1] += 2.0 * a[i_start:i_end + 1][::-1]
    grad /= len(top_idx)
