"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with backtracking line search, FFT-based smooth gradients,
adaptive temperature log-sum-exp, and structured escape mechanisms.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 after normalizing sum to 1."""
    n = len(a)

    s = float(np.sum(a))
    if s < 0.01:

def build_conv_matrix(a):

    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):

        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a, max_time=8.0):

def smooth_gradient_direct(a, T):

def project_simplex(v, target_sum=1.0):

def nesterov_smooth(a, time_budget=100.0, T_init=50.0, verbose=False):

    T_min = 5.0
    T_decay = 0.995
    lr = 3.0 / np.sqrt(n)
    lr_decay = 0.999

    stall = 0
    patience = 60
    it = 0

    while time.monotonic() - start < time_budget:

        # Nesterov update
        v = momentum * v + grad

        a_new = project_simplex(a_new, 1.0)

        it += 1
        if obj_new < best_obj - 1e-14:

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, 1.0)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new

def diverse_init(n, rng, mode=0):

    s = a.sum()

        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default uniform init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Initial LP alternating (fast convergence) ===
    print(f"\n=== Phase 1: LP Alternating ({min(300, (deadline - time.monotonic())*0.25):.0f}s) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 10:
        a_lp = lp_alternating_improved(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Phase 1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Smooth gradient (Nesterov) ===
    print(f"\n=== Phase 2: Nesterov Smooth Gradient ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 120.0)
    if p2_budget > 10:
        best_for_phase2 = best_seq.copy()
        for T_val in [30.0, 60.0]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            t_bud = min(p2_budget / 2.0, remaining * 0.4)
            if t_bud < 10:
                continue
            a_ns = nesterov_smooth(best_for_phase2, time_budget=t_bud, T_init=T_val, verbose=True)
            obj_ns = compute_obj(a_ns)
            if obj_ns < best_score:
                best_score = obj_ns
                best_seq = a_ns.copy()
                best_for_phase2 = a_ns.copy()
                print(f"  Nesterov T={T_val}: obj={best_score:.8f} **")
            else:
                print(f"  Nesterov T={T_val}: obj={obj_ns:.8f}")

    # === Phase 3: Extended LP alternating (main workhorse) ===
    print(f"\n=== Phase 3: Extended LP Alternating ===")
    remaining = deadline - time.monotonic()
    p3_total = min(remaining * 0.40, 500.0)

    # Run in two sub-phases with perturbation between
    for sub in range(2):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(p3_total / (2 - sub), remaining * 0.55)
        if alloc < 15:
            break

        if sub == 1:
            # Perturb before second sub-phase
            a_pert = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        else:
            a_pert = best_seq.copy()

        a_lp = lp_alternating_improved(a_pert, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP sub{sub}: obj={best_score:.8f} **")
        else:
            print(f"  LP sub{sub}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 4: More smooth gradient with different T ===
    print(f"\n=== Phase 4: Additional Smooth Optimization ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.10, 80.0)
    if p4_budget > 10:
        a_ns = nesterov_smooth(best_seq, time_budget=p4_budget, T_init=100.0, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Smooth: obj={best_score:.8f} **")
        else:
            print(f"  Smooth: obj={obj_ns:.8f}")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coordinate Descent Polish ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.08, 60.0)
    if p5_budget > 5:
        a_cd = coordinate_descent(best_seq, time_budget=p5_budget)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        n_current,
        max(100, n_current - 50),
        n_current + 50,
        n_current + 100,
    ]))
    n_options = [n for n in n_options if 50 <= n <= 2000]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30 or n_tried >= 4:
            break
        if n_try == n_current:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        # Quick LP optimization
        alloc = min(remaining * 0.15, 45.0)
        if alloc > 10:
            a_try = lp_alternating_improved(a_try, time_budget=alloc)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        mode = (seed + r) % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.15 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.15, 50.0)
        a_opt = lp_alternating_improved(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive LP polish ===
    print(f"\n=== Phase 8: Final LP Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        lp_alloc = min(remaining * 0.4, 100.0)
        a_final = lp_alternating_improved(best_seq, time_budget=lp_alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # One more coordinate descent if time allows
    remaining = deadline - time.monotonic()
    if remaining > 15:
        cd_alloc = min(remaining * 0.3, 30.0)
        a_cd = coordinate_descent(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation and return
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
