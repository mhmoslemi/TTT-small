"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses a novel spectral (Gerchberg-Saxton style) iterative method combined with
LP alternating, monotone-shape exploitation, adaptive n exploration, and
smooth log-sum-exp optimization. Key innovations: fast O(n log n) spectral
iterations that exploit FFT structure, phase-preserving peak shaving in
frequency domain, and monotone/asymmetric initializations that reduce the
central convolution peak via the rearrangement inequality.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize as sp_minimize

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except Exception:
    HAVE_CVXPY = False


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=5.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def spectral_refine(a, time_budget=60.0, clip_factor=0.95, verbose=False):
    """
    Fast Gerchberg-Saxton style spectral iteration.
    Alternates between:
    - Time domain: project to simplex (a >= 0, sum(a) = 1)
    - Frequency domain: reduce peak of autocorrelation via amplitude modification
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    # Momentum for the amplitude update
    amp_ema = np.zeros(size // 2 + 1)
    momentum = 0.7

    it = 0
    stall = 0
    patience = 200

    while time.monotonic() - start < time_budget:
        # Compute current FFT
        A = rfft(a, size)
        amp = np.abs(A)
        phase = np.angle(A)
        power = amp * amp

        # Get linear convolution
        conv = irfft(power, size)[:2 * n - 1]
        peak_idx = np.argmax(conv)
        peak_val = conv[peak_idx]

        curr_obj = 2.0 * n * peak_val
        if curr_obj < best_obj - 1e-12:
            best_obj = curr_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Escape: perturb and reset
                a = best_a * np.exp(np.random.default_rng(it).normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                patience = int(patience * 1.2)
                amp_ema = np.zeros(size // 2 + 1)
                continue

        # Target peak: slightly below current
        target_peak = max(peak_val * clip_factor, best_obj / (2.0 * n))

        # Shave the convolution
        conv_shaved = np.minimum(conv, target_peak)

        # Build full circular convolution
        conv_full = np.zeros(size)
        conv_full[:2 * n - 1] = conv_shaved
        # Mirror for circular symmetry (needed for real signal)
        # Actually, for valid circular autocorrelation, we need:
        # r[k] = r[size-k] for the IFFT to give a real power spectrum
        for k in range(1, (size + 1) // 2):
            conv_full[size - k] = conv_full[k]

        # FFT of shaved convolution gives new power spectrum
        new_power = np.real(rfft(conv_full))
        new_power = np.maximum(new_power, 0)
        new_amp = np.sqrt(new_power)

        # Smooth the amplitude update (prevent oscillations)
        amp_ema = momentum * amp_ema + (1 - momentum) * new_amp
        # Use the smoothed version if it has better objective
        A_candidate = amp_ema * np.exp(1j * phase)
        a_candidate = irfft(A_candidate, size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s_cand = a_candidate.sum()

        if s_cand > 1e-10:
            a_candidate = a_candidate / s_cand
            obj_cand = 2.0 * n * float(np.max(compute_conv(a_candidate)))
            # Also try without smoothing
            A_direct = new_amp * np.exp(1j * phase)
            a_direct = irfft(A_direct, size)[:n]
            a_direct = np.maximum(a_direct, 0)
            s_dir = a_direct.sum()
            if s_dir > 1e-10:
                a_direct = a_direct / s_dir
                obj_dir = 2.0 * n * float(np.max(compute_conv(a_direct)))

            if obj_cand <= obj_dir:
                a = a_candidate
            else:
                a = a_direct
                amp_ema = new_amp.copy()
        else:
            a = best_a.copy()
            amp_ema = np.zeros(size // 2 + 1)

        it += 1
        if verbose and it % 2000 == 0:
            print(f"    Spectral it{it}: obj={best_obj:.8f}, peak={peak_val:.8f}, "
                  f"stall={stall}, elapsed={time.monotonic()-start:.1f}s")

    return best_a

def lp_alternating(a, time_budget=200.0, verbose=False):
    """
    LP alternating with adaptive line search, momentum, and escape mechanisms.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    best_t = 0.4
    rng = np.random.default_rng(12345)
    it = 0
    lp_direction = None  # Cache LP direction

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        # Solve LP (or use cached direction if stalling)
        if stall < 10 or it % 5 == 0:
            g_lp, M_lp = solve_lp_minpeak(a)
            if g_lp is not None:
                lp_direction = g_lp
            else:
                lp_direction = None

        if lp_direction is None:
            # Fallback: perturb
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0
            it += 1
            continue

        # Golden-section-like line search
        ts = [best_t, max(0.05, best_t * 0.6), min(1.0, best_t * 1.5),
              0.2, 0.5, 0.8, 1.0]
        ts = sorted(set([t for t in ts if 0.02 < t <= 1.0]))

        best_t_obj = np.inf
        best_t_val = best_t
        best_a_local = best_a.copy()

        for t in ts:
            a_try = (1.0 - t) * a + t * lp_direction
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_t = np.clip(best_t_val, 0.1, 1.0)
            stall = 0
            if verbose and it % 10 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, "
                      f"elapsed={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 20:
                # Structured escape
                noise_scale = min(0.4, 0.05 + 0.05 * (stall // 20))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                best_t = 0.4
                lp_direction = None

    return best_a


def smooth_optimize(a, time_budget=60.0, T=30.0, verbose=False):
    """Smooth optimization using log-sum-exp approximation with L-BFGS-B."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_grad(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = compute_conv(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        e = np.exp(z)
        sum_e = e.sum()
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)
        w = e / sum_e
        g = np.zeros(n)
        for k in range(len(conv)):
            if w[k] > 1e-8:
                g += w[k] * grad_conv_at_peak(x, int(k))
        g *= 2.0 * n
        return obj_val, g

    max_iter = 300
    try:
        res = sp_minimize(obj_grad, a, method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                   'disp': False, 'gtol': 1e-8})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                if verbose:
                    print(f"    Smooth: obj={best_obj:.8f}")
    except Exception:
        pass

    return best_a


def nesterov_subgradient(a, time_budget=60.0, K=5, verbose=False):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 100

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        spread = (peak_vals[0] - peak_vals[-1]) / max_val
        temp = 3.0 / (1.0 + spread * 10.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_peak(y, int(k))

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.3)

        a_prev = a_new.copy()
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        y = y_new.copy()
        t_k = t_new
        it += 1

        if verbose and it % 1000 == 0:
            print(f"    NS it{it}: obj={best_obj:.8f}, elapsed={time.monotonic()-start:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def monotone_init(n, mode=0, rng=None):
    """Generate monotone or near-monotone initializations that exploit asymmetry."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.linspace(0, 1, n)

    if mode == 0:
        # Gentle linear ramp
        a = 1.0 + x
    elif mode == 1:
        # Quadratic ramp
        a = 1.0 + x * x
    elif mode == 2:
        # S-curve (logistic)
        a = 1.0 + 1.0 / (1.0 + np.exp(-5.0 * (x - 0.5)))
    elif mode == 3:
        # Power law
        p = rng.uniform(0.3, 1.5)
        a = 1.0 + x ** p
    elif mode == 4:
        # Symmetric (baseline)
        a = np.ones(n)
    elif mode == 5:
        # Exponential ramp
        a = np.exp(rng.uniform(0.5, 2.0) * x)
    elif mode == 6:
        # Triangle (peak in middle)
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    elif mode == 7:
        # Inverted triangle (valley in middle)
        a = 0.5 + 2.0 * np.abs(x - 0.5)
    else:
        a = 1.0 + 0.5 * x

    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        a = 1.0 - np.abs(2 * x - 1)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        a = 1.0 / (np.sqrt(1 - 4 * (x - 0.5) ** 2 + 0.05))
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        a = x ** 2 + (1 - x) ** 2 + 0.05
        a = a / a.sum()
    elif mode == 7:
        # Flat-top window
        a = np.ones(n)
        edge = n // 8
        for i in range(edge):
            t = i / edge
            a[i] = 0.5 * (1 + np.cos(np.pi * t))
            a[n - 1 - i] = a[i]
        a = a / a.sum()
    elif mode == 8:
        # Blackman window
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = monotone_init(n, mode % 5, rng)
    return a


def proposed_candidate_inner(seed=42, budget_s=1500):
    """Main optimization routine."""
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_base = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                n_base = parent.size
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = monotone_init(n_base, mode=0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_base}): obj={best_score:.8f}")

    # === Phase 0: Quick multi-n scan (30s) ===
    print(f"\n=== Phase 0: Multi-n Quick Scan ===")
    n_options = sorted(set([
        max(200, n_base - 50), max(300, n_base - 100),
        n_base, n_base + 50, n_base + 100,
        600, 700, 800, 900, 1000, 1100
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1200]
    n_options.sort(key=lambda x: abs(x - n_base))

    n_best = n_base
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        if n_try == len(best_seq):
            continue
        a_try = resample_sequence(best_seq, n_try)
        # Very quick spectral
        a_try = spectral_refine(a_try, time_budget=min(3.0, remaining * 0.3))
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 1: Intensive spectral refinement (300s) ===
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: Spectral Refinement (n={n_cur}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 300.0)
    if p1_budget > 20:
        a_spec = spectral_refine(best_seq, time_budget=p1_budget * 0.5, clip_factor=0.97, verbose=True)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral v1: obj={best_score:.8f} **")
        a_spec2 = spectral_refine_v2(best_seq, time_budget=p1_budget * 0.5, verbose=True)
        obj_spec2 = compute_obj(a_spec2)
        if obj_spec2 < best_score:
            best_score = obj_spec2
            best_seq = a_spec2.copy()
            print(f"  Spectral v2: obj={best_score:.8f} **")

    # === Phase 2: LP Alternating (300s) ===
    print(f"\n=== Phase 2: LP Alternating (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 20:
        # Multiple rounds with perturbation
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(p2_budget / (3 - rnd), remaining * 0.5)
            if alloc < 15:
                break
            if rnd > 0:
                a_pert = best_seq * np.exp(rng.normal(0, 0.02 * rnd, len(best_seq)))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
            else:
                a_pert = best_seq.copy()
            a_lp = lp_alternating(a_pert, time_budget=alloc, verbose=(rnd == 0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")

    # === Phase 3: Smooth optimization (100s) ===
    print(f"\n=== Phase 3: Smooth Optimization ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 15:
        for T_val in [15.0, 40.0, 80.0]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            t_budget = min(p3_budget / 3.0, remaining * 0.4)
            if t_budget < 10:
                continue
            a_sm = smooth_optimize(best_seq, time_budget=t_budget, T=T_val)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Smooth T={T_val}: obj={best_score:.8f} **")

    # === Phase 4: More spectral + LP (200s) ===
    print(f"\n=== Phase 4: Combined Spectral+LP ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.2, 200.0)
    if p4_budget > 20:
        # Alternate: 50s spectral, 50s LP, repeat
        cycles = 2
        for cyc in range(cycles):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            t_spec = min(p4_budget / (2 * cycles), remaining * 0.3)
            t_lp = min(p4_budget / (2 * cycles), remaining * 0.3)
            if t_spec > 10:
                a_s = spectral_refine_v2(best_seq, time_budget=t_spec)
                obj_s = compute_obj(a_s)
                if obj_s < best_score:
                    best_score = obj_s
                    best_seq = a_s.copy()
            remaining = deadline - time.monotonic()
            if t_lp > 10 and remaining > 15:
                a_l = lp_alternating(best_seq, time_budget=min(t_lp, remaining * 0.5))
                obj_l = compute_obj(a_l)
                if obj_l < best_score:
                    best_score = obj_l
                    best_seq = a_l.copy()

    # === Phase 5: Diverse restarts with monotone shapes (150s) ===
    print(f"\n=== Phase 5: Monotone Restarts ===")
    remaining = deadline - time.monotonic()
    n_best_cur = len(best_seq)

    restart_configs = [
        (0, 0.0), (1, 0.0), (2, 0.0), (5, 0.0),  # Pure monotone inits
        (0, 0.3), (2, 0.3),  # Blended with best
    ]
    for mode, blend in restart_configs:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        a_r = monotone_init(n_best_cur, mode=mode, rng=rng)
        if blend > 0:
            a_r = (1 - blend) * a_r + blend * best_seq
            a_r = np.maximum(a_r, 0)
            a_r = a_r / a_r.sum()

        # Quick optimization: spectral + LP
        t_alloc = min(remaining * 0.15, 40.0)
        if t_alloc > 15:
            a_r = spectral_refine(a_r, time_budget=t_alloc * 0.5)
            remaining = deadline - time.monotonic()
            t_lp = min(remaining * 0.1, 25.0)
            if t_lp > 10:
                a_r = lp_alternating(a_r, time_budget=t_lp)

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Monotone restart (mode={mode}, blend={blend}): obj={best_score:.8f} **")
        else:
            print(f"  Monotone restart (mode={mode}, blend={blend}): obj={obj_r:.8f}")

    # === Phase 6: Diverse non-monotone restarts (100s) ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        mode = [1, 4, 7, 8][r % 4]
        a_r = diverse_init(n_best_cur, rng, mode=mode)
        blend = 0.2
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.15, 40.0)
        a_opt = lp_alternating(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish (80s) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        t = min(remaining * 0.3, 30.0)
        a_final = lp_alternating(best_seq, time_budget=t)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        t = min(remaining * 0.3, 25.0)
        a_final = spectral_refine_v2(best_seq, time_budget=t)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final Spectral: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        t = min(remaining * 0.3, 20.0)
        a_final = nesterov_subgradient(best_seq, time_budget=t, K=7)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final NS: obj={best_score:.8f} **")

    # One last LP if time permits
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 20.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Very Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining spectral iterations, LP alternating,
    monotone-shape exploitation, smooth optimization, and multi-n exploration.
    
    Key innovations over previous approach:
    1. Fast O(n log n) Gerchberg-Saxton spectral iterations for peak reduction
    2. Monotone/asymmetric initializations exploiting the rearrangement inequality
    3. Adaptive spectral clipping with momentum-stabilized amplitude updates
    4. Interleaved spectral + LP optimization for faster convergence
    """
    return proposed_candidate_inner(seed=seed, budget_s=budget_s)
