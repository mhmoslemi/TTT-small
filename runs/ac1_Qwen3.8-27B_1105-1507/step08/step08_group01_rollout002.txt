import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def grad_smooth_max(a, T):
    """Compute gradient of smooth max of conv(a,a) using temperature T.
    Returns (gradient, smooth_max_value)
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    smooth_val = max_c + T * np.log(total)
    
    # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i]
    # = 2 * cross_correlation(a, weights)[i]
    size = 1
    while size < n + (2*n-1) - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(weights, size)
    corr = irfft(np.conj(fa) * fw, size)
    grad = 2.0 * corr[:n]
    
    return grad, smooth_val

def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_max_gd(a, time_budget=200.0, T_init=0.01, T_min=0.00001, verbose=False):

    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()

    lr = 0.01
    stall = 0

        if time.monotonic() - start > time_budget:
            break
        
        grad, smooth_val = grad_smooth_max(a, T)
        # Center the gradient (since we're on the simplex, the direction matters)
        grad = grad - np.mean(grad)
        
        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(20):
            a_try = simplex_project(a - alpha * grad)

                best_a = a_try.copy()
                a = a_try

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
