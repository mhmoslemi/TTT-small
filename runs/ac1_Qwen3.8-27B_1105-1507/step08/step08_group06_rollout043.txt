```python
"""
Search for a non-negative step function minimizing
    2*n*max(conv(a,a)) / (sum(a))**2.

Main ideas in this implementation:
1. Work with scale-normalized positive sequences.
2. Use an annealed log-sum-exp approximation of the peak objective.
3. Optimize in log-ratio coordinates with L-BFGS-B.
4. Finish with an active-peak subgradient / line-search polish.
5. Sweep nearby and larger lengths, since longer sequences often improve
   the asymptotic bound.
"""

import numpy as np
import time
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global state for the smooth L-BFGS callback.
# No closures are used; this global is written by soft_optimize before
# calling scipy.optimize.minimize.
# ----------------------------------------------------------------------
_SOFT = {"n": 0, "beta": 1.0, "deadline": 1.0e300}
_LOG_BOUND = 40.0
_FFT_CACHE = {}


def _next_pow2(m: int) -> int:
    size = 1
    while size < int(m):
        size <<= 1
    return size


def _fft_sizes(n: int):
    n = int(n)
    if n in _FFT_CACHE:
        return _FFT_CACHE[n]
    cs = _next_pow2(2 * n - 1)
    ss = _next_pow2(3 * n - 2) if n > 1 else 1
    _FFT_CACHE[n] = (cs, ss)
    return _FFT_CACHE[n]


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a, clipped to nonnegative."""
    n = int(a.size)
    if n == 0:
        return np.zeros(0)
    size = _fft_sizes(n)[0]
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return np.maximum(c, 0.0)


def weighted_shift(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Returns H_i = sum_k w_k * a_{k-i} over valid indices.
    This is the half-gradient of sum_k w_k*(a*a)_k with respect to a.
    """
    n = int(a.size)
    if n == 0:
        return np.zeros(0)
    need = 2 * n - 1
    if w.size < need:
        tmp = np.zeros(need)
        m = min(w.size, need)
        tmp[:m] = w[:m]
        w = tmp
    elif w.size > need:
        w = w[:need]

    size = _fft_sizes(n)[1]
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a[::-1], size)
    res = np.fft.irfft(fw * fa, size)[n - 1: 2 * n - 1]
    return np.maximum(res, 0.0)


def objective_from_c(n: int, c: np.ndarray) -> float:
    if c.size == 0:
        return np.inf
    return float(2.0 * n * np.max(c))


def objective(a: np.ndarray) -> float:
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    c = conv_self(a / s)
    return objective_from_c(n, c)


def normalize_positive(a: np.ndarray, floor: float = 1e-12) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.zeros(0)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, floor)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def logit_a(a: np.ndarray) -> np.ndarray:
    """Log-ratio coordinates with the last coordinate fixed to 0."""
    n = int(a.size)
    if n <= 1:
        return np.zeros(0, dtype=np.float64)
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-300)
    la = np.log(a)
    x = la - la[-1]
    return np.clip(x[:-1], -_LOG_BOUND, _LOG_BOUND).astype(np.float64)


def a_from_logit(x: np.ndarray, n: int) -> np.ndarray:
    if n <= 1:
        return np.ones(max(1, int(n)), dtype=np.float64)
    x_full = np.empty(n, dtype=np.float64)
    x = np.asarray(x, dtype=np.float64)
    m = min(x.size, n - 1)
    if m > 0:
        x_full[:m] = x[:m]
    x_full[m:n] = 0.0
    x_full = np.clip(x_full, -_LOG_BOUND, _LOG_BOUND)
    xm = float(np.max(x_full))
    ex = np.exp(x_full - xm)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return ex / s


def _soft_callback(x: np.ndarray):
    """
    L-BFGS objective/gradient for the smoothed problem:
        F_beta(a) = (2n/beta) log(sum_k exp(beta * (a*a)_k)),
    with variables x in log-ratio coordinates, last coordinate fixed to 0.
    """
    n = int(_SOFT["n"])
    beta = float(_SOFT["beta"])
    deadline = float(_SOFT["deadline"])

    if n <= 1:
        return 0.0, np.zeros(0, dtype=np.float64)

    if time.monotonic() > deadline:
        return 0.0, np.zeros(n - 1, dtype=np.float64)

    x_full = np.empty(n, dtype=np.float64)
    x = np.asarray(x, dtype=np.float64)
    m = min(x.size, n - 1)
    if m > 0:
        x_full[:m] = x[:m]
    x_full[m:n] = 0.0
    x_full = np.clip(x_full, -_LOG_BOUND, _LOG_BOUND)

    xm = float(np.max(x_full))
    ex = np.exp(x_full - xm)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 1e-300:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = ex / s

    c = conv_self(a)
    smax = float(np.max(c))
    z = beta * (c - smax)
    ez = np.exp(z)
    s2 = float(np.sum(ez))

    if not np.isfinite(s2) or s2 <= 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)
        log_s = 0.0
    else:
        w = ez / s2
        log_s = float(np.log(s2))

    H = weighted_shift(w, a)
    g_a = 2.0 * H
    shift = float(np.dot(a, g_a))
    grad_full = a * (g_a - shift)

    fun = float((2.0 * n / beta) * (smax + log_s))
    if not np.isfinite(fun):
        fun = 1.0e100
        grad_full = np.zeros(n, dtype=np.float64)

    return fun, grad_full[: n - 1].astype(np.float64)


def soft_optimize(a: np.ndarray, time_budget: float, quick: bool = False, verbose: bool = True) -> np.ndarray:
    """
    Annealed log-sum-exp L-BFGS optimization.
    """
    a = normalize_positive(a)
    n = int(a.size)
    start = time.monotonic()
    local_deadline = start + max(0.0, float(time_budget))

    best = a.copy()
    best_obj = objective(best)
    if n <= 1 or local_deadline - start < 1.0:
        return best

    x = logit_a(best)

    if quick:
        targets = [8.0, 40.0, 160.0]
    else:
        targets = [3.0, 10.0, 30.0, 100.0, 300.0, 800.0]

    last_beta = -1.0
    n_targets = len(targets)

    for idx, target in enumerate(targets):
        now = time.monotonic()
        if now >= local_deadline - 2.0:
            break

        remaining = local_deadline - now
        if idx == n_targets - 1:
            alloc = remaining
        else:
            alloc = min(remaining * 0.55, 80.0 if not quick else 40.0)
        if alloc < 4.0:
            break

        p = best_obj / max(1.0, 2.0 * n)
        if not np.isfinite(p) or p <= 1e-15:
            p = 1.0 / (2.0 * n)

        beta = float(target / p)
        if beta < 10.0:
            beta = 10.0
        if beta > 800000.0:
            beta = 800000.0
        if last_beta > 0.0 and abs(beta - last_beta) <= 0.02 * max(1.0, beta):
            beta = last_beta * 1.25
        last_beta = beta

        # Profile one callback to choose a conservative maxfun.
        _SOFT["n"] = int(n)
        _SOFT["beta"] = beta
        _SOFT["deadline"] = time.monotonic() + 10.0
        t0 = time.monotonic()
        f0, _ = _soft_callback(x)
        tper = max(0.0005, time.monotonic() - t0)

        if not np.isfinite(f0):
            continue

        cap = 500 if n <= 1200 else 260
        maxcalls = int(min(cap, max(30, (alloc * 0.45) / tper)))
        if quick:
            maxcalls = min(maxcalls, 120)
        maxiter = int(min(60, max(5, maxcalls // 10)))

        _SOFT["deadline"] = time.monotonic() + alloc

        try:
            res = minimize(
                _soft_callback,
                x,
                method="L-BFGSB",
                jac=True,
                options={
                    "maxiter": maxiter,
                    "maxfun": maxcalls,
                    "ftol": 1.0e-12,
                    "gtol": 1.0e-6,
                    "maxls": 20,
                    "disp": False,
                },
            )
        except Exception:
            res = None

        if res is not None and hasattr(res, "x") and res.x is not None:
            x_new = np.asarray(res.x, dtype=np.float64)
            if x_new.size == n - 1 and np.isfinite(x_new).all():
                a_new = a_from_logit(x_new, n)
                obj_new = objective(a_new)
                if np.isfinite(obj_new) and obj_new < best_obj - 1e-14:
                    best = a_new
                    best_obj = obj_new
                    x = logit_a(best)
                    if verbose:
                        print(f"      soft beta={beta:.0f} obj={best_obj:.9f}")

        if verbose:
            now = time.monotonic()
            print(f"    soft[{idx + 1}/{n_targets}] t={now - start:.0f}s best={best_obj:.9f}")

    return best


def active_polish(a: np.ndarray, time_budget: float, verbose: bool = False) -> np.ndarray:
    """
    Active-peak subgradient polish with backtracking on the true objective.
    """
    a = normalize_positive(a)
    n = int(a.size)
    start = time.monotonic()
    deadline = start + max(0.0, float(time_budget))

    best = a.copy()
    best_obj = objective(best)
    if n <= 1 or deadline - start < 1.0:
        return best

    x = logit_a(best)
    top_count = max(10, min(100, n // 8))
    max_iters = 250 if n <= 1200 else 120
    no_accept = 0

    primary_alphas = [0.10, 0.05, 0.20, 0.02, 0.40, 0.01]
    tiny_alphas = [0.005, 0.002, 0.001]

    rng = np.random.default_rng(12345 + n)

    for _ in range(max_iters):
        if time.monotonic() >= deadline - 1.0:
            break

        x = np.clip(x, -_LOG_BOUND, _LOG_BOUND)
        a_cur = a_from_logit(x, n)
        c_cur = conv_self(a_cur)
        cur_obj = objective_from_c(n, c_cur)

        if np.isfinite(cur_obj) and cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best = a_cur.copy()
            no_accept = 0

        c_size = 2 * n - 1
        if top_count < c_size:
            idx = np.argpartition(c_cur, -top_count)[-top_count:]
        else:
            idx = np.arange(c_size)

        w = np.zeros(c_size, dtype=np.float64)
        w[idx] = 1.0 / float(len(idx))

        H = weighted_shift(w, a_cur)
        shift = float(np.dot(a_cur, H))
        grad_full = 2.0 * a_cur * (H - shift)
        grad = grad_full[: n - 1].astype(np.float64)

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm <= 1.0e-12:
            break

        direction = -grad / gnorm
        accepted = False

        for alpha in primary_alphas:
            x_try = np.clip(x + alpha * direction, -_LOG_BOUND, _LOG_BOUND)
            a_try = a_from_logit(x_try, n)
            obj_try = objective(a_try)
            if np.isfinite(obj_try) and obj_try < cur_obj - 1e-14:
                x = x_try
                best = a_try.copy()
                best_obj = obj_try
                accepted = True
                no_accept = 0
                break

        if not accepted:
            for alpha in tiny_alphas:
                x_try = np.clip(x + alpha * direction, -_LOG_BOUND, _LOG_BOUND)
                a_try = a_from_logit(x_try, n)
                obj_try = objective(a_try)
                if np.isfinite(obj_try) and obj_try < cur_obj - 1e-14:
                    x = x_try
                    best = a_copy = None
                    best_obj = obj_try
                    best = a_try.copy()
                    accepted = True
                    no_accept = 0
                    break

        if not accepted:
            no_accept += 1
            if no_accept >= 60:
                break
            if no_accept % 30 == 0:
                # Small controlled escape kick; best is preserved.
                x = np.clip(x + rng.normal(0.0, 0.03, n - 1), -_LOG_BOUND, _LOG_BOUND)
                a_try = a_from_logit(x, n)
                obj_try = objective(a_try)
                if np.isfinite(obj_try) and obj_try < best_obj:
                    best_obj = obj_try
                    best = a_try.copy()
                else:
                    x = logit_a(best)

    return best


def resample_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    new_n = int(new_n)
    if n <= 0 or new_n <= 0:
        return np.ones(1, dtype=np.float64)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize_positive(a_new)


def _anti_sequence(n: int, r1: float, r2: float) -> np.ndarray:
    n = int(n)
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n) - 0.5
    a = 1.0 + r1 * np.sin(2.0 * np.pi * t) + r2 * np.sin(6.0 * np.pi * t)
    return normalize_positive(np.maximum(a, 0.05))


def generate_initialization(kind: str, n: int, rng: np.random.Generator) -> np.ndarray:
    n = int(n)
    if kind == "uniform_perturb":
        a = 1.0 + 0.03 * rng.normal(0.0, 1.0, n)
        return normalize_positive(np.maximum(a, 0.05))
    if kind == "anti1":
        return _anti_sequence(n, 0.30, 0.12)
    if kind == "anti2":
        return _anti_sequence(n, 0.55, 0.18)
    if kind == "anti3":
        return _anti_sequence(n, 0.75, 0.20)
    if kind == "random":
        a = 1.0 + 0.45 * rng.normal(0.0, 1.0, n)
        return normalize_positive(np.maximum(a, 0.05))
    if kind == "two_pulse":
        a = np.ones(n, dtype=np.float64) * 0.5
        a[max(0, n // 4)] = 1.6
        a[max(0, 3 * n // 4)] = 1.6
        kernel = np.array([0.2, 0.3, 0.4, 0.3, 0.2], dtype=np.float64)
        a = np.convolve(a, kernel, mode="same")
        return normalize_positive(np.maximum(a, 0.05))
    return np.ones(n, dtype=np.float64) / n


def propose_candidate(seed: int = 42, budget_s: float = 1500.0):
    """
    Search routine. Returns a Python list of nonnegative floats.
    """
    start_time = time.monotonic()

    # Leave a safety margin before the hard deadline.
    hard_deadline = start_time + max(10.0, float(budget_s)) - 30.0
    rng = np.random.default_rng(seed)

    best = None
    best_obj = np.inf

    # ------------------------------------------------------------------
    # Try to use the previous best construction if it is available.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.where(np.isfinite(parent), parent, 0.0)
        parent = np.maximum(parent, 0.0)
        if parent.size > 0 and float(np.sum(parent)) > 1e-9:
            parent = normalize_positive(parent)
            if parent.size > 4000:
                parent = resample_sequence(parent, 3000)
            best = parent.copy()
            best_obj = objective(best)
            print(f"Parent sequence: n={best.size}, local objective={best_obj:.9f}")
    except Exception:
        best = None

    # ------------------------------------------------------------------
    # If no parent is available, create a diverse starting point.
    # ------------------------------------------------------------------
    if best is None:
        n0 = 800
        for kind in ["anti1", "anti2", "anti3", "random", "uniform_perturb", "two_pulse"]:
            a_try = generate_initialization(kind, n0, rng)
            obj_try = objective(a_try)
            if np.isfinite(obj_try) and obj_try < best_obj:
                best_obj = obj_try
                best = a_try.copy()
        if best is None:
            best = np.ones(800, dtype=np.float64) / 800.0
            best_obj = objective(best)
        print(f"Generated initial sequence: n={best.size}, objective={best_obj:.9f}")

    # ------------------------------------------------------------------
    # First strong optimization on the parent length.
    # ------------------------------------------------------------------
    now = time.monotonic()
    remaining = hard_deadline - now
    if remaining > 120.0:
        alloc = min(380.0, remaining * 0.30)
        improved = soft_optimize(best, alloc, quick=False, verbose=True)
        obj_improved = objective(improved)
        if np.isfinite(obj_improved) and obj_improved < best_obj:
            best = improved.copy()
            best_obj = obj_improved
        print(f"After first full polish: n={best.size}, objective={best_obj:.9f}")

    # ------------------------------------------------------------------
    # A few controlled perturbations to avoid staying in one basin.
    # ------------------------------------------------------------------
    now = time.monotonic()
    remaining = hard_deadline - now
    if remaining > 180.0:
        n_best = best.size
        for scale in [0.04, 0.10, 0.18]:
            now = time.monotonic()
            remaining = hard_deadline - now
            if remaining < 70.0:
                break
            a_pert = best * np.exp(rng.normal(0.0, scale, n_best))
            a_pert = normalize_positive(a_pert)
            alloc = min(80.0, remaining * 0.18)
            a_opt = soft_optimize(a_pert, alloc, quick=True, verbose=False)
            obj_opt = objective(a_opt)
            if np.isfinite(obj_opt) and obj_opt < best_obj:
                best = a_opt.copy()
                best_obj = obj_opt
                print(f"Perturbation scale={scale}: objective={best_obj:.9f}")
        print(f"After perturbations: n={best.size}, objective={best_obj:.9f}")

    # ------------------------------------------------------------------
    # Length sweep. Longer lengths often improve the asymptotic score,
    # but we keep the cap moderate so the evaluator and search stay fast.
    # ------------------------------------------------------------------
    now = time.monotonic()
    remaining = hard_deadline - now
    current_n = int(best.size)

    raw_lengths = [current_n]
    for mult in [0.90, 1.10, 1.25, 1.50, 1.80, 2.20, 2.70, 3.30]:
        n_candidate = int(round(current_n * mult))
        raw_lengths.append(n_candidate)

    for n_abs in [512, 640, 800, 1000, 1200, 1500, 1800, 2200, 2600, 3000]:
        raw_lengths.append(n_abs)

    unique_lengths = []
    for n_cand in raw_lengths:
        n_cand = int(n_cand)
        if 256 <= n_cand <= 3000:
            if n_cand not in unique_lengths:
                unique_lengths.append(n_cand)

    pairs = []
    for n_cand in unique_lengths:
        pairs.append((abs(n_cand - current_n), n_cand))
    pairs.sort()
    candidate_lengths = [n_cand for _, n_cand in pairs]

    for n_cand in candidate_lengths:
        now = time.monotonic()
        remaining = hard_deadline - now
        if remaining < 80.0:
            break
        if n_cand == len(best):
            continue

        budget_len = min(110.0, remaining * 0.14)
        if budget_len < 40.0:
            continue

        a_try = resample_sequence(best, n_cand)
        obj_try = objective(a_try)
        if np.isfinite(obj_try) and obj_try < best_obj:
            best = a_try.copy()
            best_obj = obj_try
            print(f"Resampled to n={n_cand}: objective={best_obj:.9f}")

        a_opt = soft_optimize(a_try, budget_len, quick=True, verbose=False)
        obj_opt = objective(a_opt)
        if np.isfinite(obj_opt) and obj_opt < best_obj:
            best = a_opt.copy()
            best_obj = obj_opt
            print(f"Optimized length n={n_cand}: objective={best_obj:.9f}")

    # ------------------------------------------------------------------
    # Final polish on the best sequence found so far.
    # ------------------------------------------------------------------
    now = time.monotonic()
    remaining = hard_deadline - now
    if remaining > 90.0:
        alloc = min(260.0, remaining * 0.55)
        improved = soft_optimize(best, alloc, quick=False, verbose=True)
        obj_improved = objective(improved)
        if np.isfinite(obj_improved) and obj_improved < best_obj:
            best = improved.copy()
            best_obj = obj_improved
        print(f"After final full polish: n={best.size}, objective={best_obj:.9f}")

    now = time.monotonic()
    remaining = hard_deadline - now
    if remaining > 45.0:
        improved = active_polish(best, min(120.0, remaining * 0.45), verbose=False)
        obj_improved = objective(improved)
        if np.isfinite(obj_improved) and obj_improved < best_obj:
            best = improved.copy()
            best_obj = obj_improved
        print(f"After active polish: n={best.size}, objective={best_obj:.9f}")

    # ------------------------------------------------------------------
    # Final validation. Return positive, bounded floats.
    # ------------------------------------------------------------------
    best = np.asarray(best, dtype=np.float64)
    best = np.where(np.isfinite(best), best, 0.0)
    best = np.maximum(best, 0.0)

    s = float(np.sum(best))
    if s > 0.0 and np.isfinite(s):
        best = best / s
    else:
        best = np.ones(800, dtype=np.float64)
        s = float(np.sum(best))
        best = best / s

    result = [float(max(0.0, min(1000.0, x))) for x in best]

    try:
        val = evaluate_sequence(result)
    except Exception:
        val = np.inf

    if np.isfinite(val):
        best_obj = float(val)
    else:
        result = [1.0] * 800

    elapsed = time.monotonic() - start_time
    print(f"FINAL: n={len(result)}, objective={best_obj:.9f}, elapsed={elapsed:.1f}s")
    return result
```
