"""
Optimized step function search using enhanced LP alternating with golden-section
line search, symmetric optimization track, coordinate descent polishing,
multi-scale n exploration, and diverse structured initializations.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """FFT-based full linear convolution of a with b."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2 with scale invariance."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_raw(a):
    """Compute the raw objective without normalization (matches evaluate_sequence)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return float(2.0 * n * max_b / (s * s))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = 0
        k_end = min(2 * n - 2, j + n - 1)
        length = k_end - k_start + 1
        A[k_start:k_end + 1, j] = a[j - k_end:(j - k_start) + 1]
    return A


def solve_lp(a, time_limit=30.0):

    Returns (g_normalized, M_value) or (None, None)."""

    A = build_conv_matrix(a)

            g_val = np.maximum(g.value, 0)

def line_search_golden(a, g, max_evals=30):
    """Golden section search for optimal t in [0,1] minimizing obj((1-t)*a + t*g)."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
    lo = 0.0
    hi = 1.0

    # Initial evaluations
    b = hi - phi * (hi - lo)
    c = lo + phi * (hi - lo)
    f_b = _obj_blend(a, g, b)
    f_c = _obj_blend(a, g, c)

    best_t = 0.0
    best_f = compute_obj(a)  # t=0

    # Also check t=1
    f_1 = _obj_blend(a, g, 1.0)
    if f_1 < best_f:
        best_f = f_1
        best_t = 1.0

    for _ in range(max_evals - 2):
        if f_b < f_c:
            hi = c
            c = b
            f_c = f_b
            b = hi - phi * (hi - lo)
            f_b = _obj_blend(a, g, b)
        else:
            lo = b
            b = c
            f_b = f_c
            c = lo + phi * (hi - lo)
            f_c = _obj_blend(a, g, c)

        if f_b < best_f:
            best_f = f_b
            best_t = b
        if f_c < best_f:
            best_f = f_c
            best_t = c

    return best_t, best_f


def _obj_blend(a, g, t):
    """Compute objective of (1-t)*a + t*g."""
    h = (1.0 - t) * a + t * g
    h = np.maximum(h, 0)
    s = float(np.sum(h))
    if s < 1e-10:
        return np.inf
    h = h / s
    return compute_obj(h)

            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
                    stall = 0
                    if verbose:
                        elapsed = time.monotonic() - start
                        print(f"    LP iter: obj={best_obj:.10f}, t={best_t:.4f}, time={elapsed:.0f}s")
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        if stall > 5:
            # Structured perturbation using subgradient
            conv = compute_conv(a)
            max_val = np.max(conv)
            # Find peaks
            peak_mask = conv > max_val * 0.99
            n_peaks = np.sum(peak_mask)
            if n_peaks > 0:
                # Compute subgradient direction
                w = np.zeros(len(conv))
                w[peak_mask] = 1.0 / n_peaks
                # grad[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)
                n_fft = 1
                while n_fft < 4 * n:
                    n_fft <<= 1
                A_fft = rfft(a, n_fft)
                W_fft = rfft(w, n_fft)
                R = irfft(np.conj(A_fft) * W_fft, n_fft)
                grad = np.zeros(n)
                for j in range(n):
                    grad[j] = R[(n_fft - j) % n_fft]
                grad *= 2.0
                grad -= np.mean(grad)

                # Apply perturbation
                scale = 0.1 + 0.05 * min(stall // 5, 3)
                a = a - scale * grad + rng.normal(0, 0.01, n)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s

            stall = 0

    return best_a


def coordinate_descent(a, time_budget=60.0, verbose=False):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)

    order = np.arange(n)
    sweep = 0

    while time.monotonic() - start < time_budget:
        rng.shuffle(order)
        for j in order:
            if time.monotonic() - start > time_budget:
                    break

            # Save current value
            x_old = a[j]
            a[j] = 0.0

            # Compute b[k] without contribution from a[j]
            # b[k] = sum_i a[i]*a[k-i], excluding terms with index j
            # For efficiency, compute full conv and subtract
            conv_base = compute_conv(a)  # conv with a[j]=0

            # Now add back terms with x = a[j]:
            # b[k] = conv_base[k] + D[k]*x + E[k]*x^2 for k in [j, j+n-1]
            # D[k] = 2*a[k-j] if k != 2j and 0<=k-j<n
            # E[k] = 1 if k == 2j

            # Find valid k range: k from j to j+n-1 (where 0 <= k-j < n)
            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)

            # Get D values
            idx = np.arange(k_lo, k_hi + 1)
            d_vals = 2.0 * a[idx - j]  # 2*a[k-j] for all k in range
            # Fix k=2j: D should be 0 there (it's E term)
            if 2 * j >= k_lo and 2 * j <= k_hi:
                d_vals[2 * j - k_lo] = 0.0

            # E: only at k=2j
            # b[k] = conv_base[k] + d_vals[k-k_lo]*x + (x^2 if k==2j else 0)

            # Minimize max_k (conv_base[k] + d[k]*x + e[k]*x^2) for x >= 0
            # Bisection on M
            c_vals = conv_base.copy()
            # For k outside [k_lo, k_hi], b[k] = conv_base[k] (no x dependence)
            # So the constraint is conv_base[k] <= M for all such k

            # Pre-compute the max of conv_base outside the active range
            if k_lo > 0:
                max_outside_left = np.max(c_vals[:k_lo])
            else:
                max_outside_left = 0.0
            if k_hi < 2 * n - 2:
                max_outside_right = np.max(c_vals[k_hi + 1:])
            else:
                max_outside_right = 0.0
            max_outside = max(max_outside_left, max_outside_right)

            # For active range: b[k] = c_vals[k] + d_vals[k-k_lo]*x + (x^2 if k==2j)
            c_active = c_vals[k_lo:k_hi + 1]
            d_active = d_vals

            has_quadratic = (2 * j >= k_lo and 2 * j <= k_hi)

            # Bisection on M
            # Lower bound: max(max_outside, min_x max(c_active + d_active*x + e*x^2))
            # Upper bound: current objective
            M_lo = max_outside
            M_hi = max(max_outside, np.max(c_active) + np.max(d_active) * max(x_old, 1.0) + max(x_old, 1.0)**2)

            # Actually, let's just find the best x by bisection
            # f(x) = max(max_outside, max_k (c_active[k] + d_active[k]*x + (x^2 if k==2j else 0)))
            # f is convex in x >= 0

            # Binary search for minimum of f(x)
            x_lo = 0.0
            x_hi = max(x_old * 3, 1.0)

            # First check if f is decreasing at x=0
            f_0 = _cd_eval(0.0, c_active, d_active, has_quadratic, 2*j - k_lo if has_quadratic else -1, max_outside)
            f_small = _cd_eval(0.01, c_active, d_active, has_quadratic, 2*j - k_lo if has_quadratic else -1, max_outside)

            if f_small >= f_0:
                x_opt = 0.0
            else:
                # Find where f stops decreasing (within [0.01, x_hi])
                # Binary search on derivative sign
                for _ in range(50):
                    x_mid = (x_lo + x_hi) / 2.0
                    f_mid = _cd_eval(x_mid, c_active, d_active, has_quadratic, 2*j - k_lo if has_quadratic else -1, max_outside)
                    f_mid_plus = _cd_eval(x_mid + 0.001, c_active, d_active, has_quadratic, 2*j - k_lo if has_quadratic else -1, max_outside)
                    if f_mid_plus >= f_mid:
                        x_hi = x_mid
                    else:
                        x_lo = x_mid
                x_opt = (x_lo + x_hi) / 2.0

            # Clamp
            x_opt = max(0.0, x_opt)

            a[j] = x_opt
            s_new = float(np.sum(a))
            if s_new < 1e-10:
                a[j] = x_old
            else:
                a = a / s_new
                obj_new = compute_obj(a)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a.copy()

        sweep += 1
        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, time={elapsed:.0f}s")

    return best_a


def _cd_eval(x, c_active, d_active, has_quad, quad_idx, max_outside):
    """Evaluate max(max_outside, max_k (c[k] + d[k]*x + (x^2 if quad))) for given x."""
    vals = c_active + d_active * x
    if has_quad and 0 <= quad_idx < len(vals):
        vals[quad_idx] += x * x
    return max(max_outside, float(np.max(vals))) if len(vals) > 0 else max_outside


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Hann window
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 1:
        # Blackman window
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif choice == 2:
        # Gaussian bump
        a = np.exp(-50 * (x - 0.5) ** 2)
    elif choice == 3:
        # Two bumps
        a = np.exp(-80 * (x - 0.25) ** 2) + np.exp(-80 * (x - 0.75) ** 2)
    elif choice == 4:
        # Tapered: high in middle, low at edges
        a = np.sin(np.pi * x) ** rng.uniform(1.0, 4.0)
    elif choice == 5:
        # Oscillatory with envelope
        a = np.sin(np.pi * x) * (0.5 + 0.5 * np.cos(2 * np.pi * rng.integers(2, 8) * x))
        a = np.abs(a)
    elif choice == 6:
        # Power law with taper
        a = np.minimum(x, 1 - x) ** rng.uniform(0.5, 2.0)
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(3):
            center = rng.integers(0, n)
            width = rng.integers(max(2, n // 20), n // 3)
            d = np.abs(np.arange(n) - center)
            a *= 1.0 + rng.standard_normal() * np.exp(-d / (width * 0.5))
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def symmetric_lp_alternate(a, time_budget=120.0, verbose=False):
    """LP alternating on symmetric sequences (a[i] = a[n-1-i])."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = 0.5 * (a + a[::-1])  # Enforce symmetry
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        # Solve LP (full, but we'll symmetrize the result)
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        g_lp = 0.5 * (g_lp + g_lp[::-1])  # Symmetrize
        gs = float(np.sum(g_lp))
        if gs > 1e-10:
            g_lp = g_lp / gs

        # Line search
        best_t, best_f = line_search_golden(a, g_lp, max_evals=25)

        if best_f < best_obj - 1e-12:
            a_new = (1.0 - best_t) * a + best_t * g_lp
            a_new = 0.5 * (a_new + a_new[::-1])
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
                    stall = 0
                    if verbose:
                        print(f"    SymLP: obj={best_obj:.10f}, t={best_t:.4f}")
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        if stall > 6:
            # Perturb
            a = a * np.exp(rng.normal(0, 0.1, n))
            a = 0.5 * (a + a[::-1])
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            stall = 0

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses enhanced LP alternating with golden-section
    line search, symmetric optimization, coordinate descent polishing,
    multi-n exploration, and diverse initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0  # 30s safety margin

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        print(f"Parent loaded: n={len(a)}, obj={compute_obj(a):.10f}")
    except Exception:
        a = np.ones(800) / 800
        print(f"No parent, using uniform n=800, obj={compute_obj(a):.10f}")

    best_a = a.copy()
    best_obj = compute_obj(a)
    n_base = len(best_a)
    print(f"Initial best: n={n_base}, obj={best_obj:.10f}")

    # === Phase 1: Primary LP alternating from parent ===
    print(f"\n=== Phase 1: LP Alternating (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 350.0)
    a_p1 = lp_alternate_improved(best_a, time_budget=p1_budget, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  Phase 1 result: obj={best_obj:.10f} **")
    else:
        print(f"  Phase 1: no improvement (obj={obj_p1:.10f})")

    # === Phase 2: Coordinate descent polish ===
    print(f"\n=== Phase 2: Coordinate Descent Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = min(remaining * 0.12, 90.0)
        a_cd = coordinate_descent(best_a, time_budget=p2_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd
            print(f"  CD result: obj={best_obj:.10f} **")
        else:
            print(f"  CD: no improvement (obj={obj_cd:.10f})")

    # === Phase 3: LP alternating round 2 (from improved point) ===
    print(f"\n=== Phase 3: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(remaining * 0.2, 180.0)
        a_p3 = lp_alternate_improved(best_a, time_budget=p3_budget, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            print(f"  Phase 3 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 3: no improvement (obj={obj_p3:.10f})")

    # === Phase 4: Symmetric optimization ===
    print(f"\n=== Phase 4: Symmetric LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p4_budget = min(remaining * 0.12, 100.0)
        # Start from symmetrized best
        a_sym_init = 0.5 * (best_a + best_a[::-1])
        a_sym_init = a_sym_init / np.sum(a_sym_init)
        a_sym = symmetric_lp_alternate(a_sym_init, time_budget=p4_budget, verbose=True)
        obj_sym = compute_obj(a_sym)
        if obj_sym < best_obj:
            best_obj = obj_sym
            best_a = a_sym
            print(f"  Symmetric result: obj={best_obj:.10f} **")
        else:
            print(f"  Symmetric: no improvement (obj={obj_sym:.10f})")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    # Try a range of n values around the best
    n_candidates = sorted(set([
        max(200, n_best - 300),
        max(200, n_best - 200),
        max(200, n_best - 100),
        n_best,
        n_best + 100,
        n_best + 200,
        n_best + 400,
        n_best + 600,
    ]))
    n_candidates = [x for x in n_candidates if 300 <= x <= 3000]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

        # Quick LP optimization
        alloc = min(45.0, remaining * 0.2)
        a_opt = lp_alternate_improved(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        # Also try coordinate descent
        if remaining > 30:
            alloc_cd = min(15.0, remaining * 0.08)
            a_opt_cd = coordinate_descent(a_opt, time_budget=alloc_cd)
            obj_try_cd = compute_obj(a_opt_cd)
            if obj_try_cd < obj_try:
                obj_try = obj_try_cd
                a_opt = a_opt_cd

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 6: Diverse initializations ===
    print(f"\n=== Phase 6: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    for d in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_div = diverse_initialization(n_best, rng)
        alloc = min(35.0, remaining * 0.15)
        a_opt = lp_alternate_improved(a_div, time_budget=alloc)
        obj_div = compute_obj(a_opt)

        # Quick CD polish
        if remaining > 25:
            alloc_cd = min(10.0, remaining * 0.05)
            a_opt_cd = coordinate_descent(a_opt, time_budget=alloc_cd)
            obj_div_cd = compute_obj(a_opt_cd)
            if obj_div_cd < obj_div:
                obj_div = obj_div_cd
                a_opt = a_opt_cd

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse init {d}: obj={best_obj:.10f} **")
        else:
            print(f"  Diverse init {d}: obj={obj_div:.10f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Final LP
        alloc_lp = min(remaining * 0.3, 60.0)
        a_final = lp_alternate_improved(best_a, time_budget=alloc_lp, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.10f} **")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_cd = min(remaining * 0.25, 40.0)
        a_cd = coordinate_descent(best_a, time_budget=alloc_cd)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd
            print(f"  Final CD: obj={best_obj:.10f} **")

    # One more LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_last = min(remaining * 0.4, 25.0)
        a_last = lp_alternate_improved(best_a, time_budget=alloc_last)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.10f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - t_start
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
