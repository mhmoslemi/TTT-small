import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    from scipy.optimize import linprog

    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

def convex_line_search(a, g, n_evals=20):
    """Find optimal t in [0,1] for a_new = (1-t)*a + t*g."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    a_new = a_new / s
    return t_opt, a_new

def coordinate_descent(a, time_budget, delta=1e-4):
    """Coordinate descent polish."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    pass_count = 0
    while time.monotonic() - start < time_budget and delta > 1e-8:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count * 7919 + hash(n) % 100000)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                    break

            # Try small perturbations
            for d in [delta, -delta, delta*0.3, -delta*0.3]:
                if a[i] + d < -1e-15:
                    continue
                a_try = a.copy()
                a_try[i] = max(0.0, a[i] + d)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            delta *= 0.5

    return best_a


def multi_pair_swap(a, time_budget):
    """Try pairwise swaps of mass between coordinates."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        for eps in [0.001, 0.0005, 0.002]:
            if a[i] < eps:
                continue
            a_try = a.copy()
            a_try[i] -= eps
            a_try[j] += eps
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: LP alternating with active-set, FFT-based subgradient un-stick,
adaptive smooth-max gradient descent, pair-swap local search,
multi-n exploration, and perturbation restarts.
Key improvements over baseline: smarter active-set management,
gradient un-stick using FFT, better restart strategy with increasing
perturbation scales, and more aggressive final polishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def solve_lp_reduced(a, active_indices):
    """Solve LP with only a subset of constraints (active set reduction)."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A_full = build_conv_matrix(a)
    A = A_full[active_indices, :]
    m_active = len(active_indices)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((m_active, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m_active)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_norm = g_val / s
            conv_check = A_full @ g_norm
            true_M = float(np.max(conv_check))
            return g_norm, true_M
    return None, None


def convex_line_search(a, g, n_evals=20):
    """Find optimal t in [0,1] for a_new = (1-t)*a + t*g using convexity."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    a_new = a_new / s
    return t_opt, a_new


def lp_alternating(a, time_budget, use_reduced=True, verbose=False):
    """LP alternating with convex line search and adaptive active set."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    active_set = None
    total_iters = 0

    max_iters = 2000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        if use_reduced and active_set is not None and 0 < len(active_set) < 2 * n - 1:
            g_lp, M_lp = solve_lp_reduced(a, active_set)
            if g_lp is None:
                g_lp, M_lp = solve_lp(a)
        else:
            g_lp, M_lp = solve_lp(a)

        if g_lp is None:
            break

        total_iters += 1

        conv_cur = compute_conv(a)
        max_conv = np.max(conv_cur)
        # Adaptive threshold: start tight, loosen if stalling
        if stall_count < 5:
            threshold = max_conv * 0.995
        elif stall_count < 15:
            threshold = max_conv * 0.97
        else:
            threshold = max_conv * 0.90
        active_set = np.where(conv_cur >= threshold)[0]

        t_opt, a_new = convex_line_search(a, g_lp, n_evals=15)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose and total_iters % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{total_iters}: obj={best_obj:.8f}, t={t_opt:.4f}, time={elapsed:.1f}s")
        else:
            stall_count += 1
            if stall_count > 20:
                # Try progressively smaller steps
                improved = False
                for step_size in [0.1, 0.05, 0.02, 0.01]:
                    a_try = (1 - step_size) * a + step_size * g_lp
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            stall_count = 5
                            improved = True
                            break
                if not improved:
                    active_set = None
                    stall_count = 0
                    if stall_count > 50:
                        break

    return best_a


def compute_weighted_gradient(weights, a):
    """
    Compute grad[i] = 2 * sum_k weights[k] * a[k-i] for valid indices.
    Uses FFT for O(n log n) efficiency.
    """
    n = len(a)
    b = a[::-1]
    conv_wb = compute_conv_cross(weights, b)
    # grad[i] = 2 * conv_wb[n-1+i] for i=0,...,n-1
    grad = 2.0 * conv_wb[n - 1:2 * n - 1]
    return grad


def subgradient_step(a, n_steps=100, time_budget=60.0, verbose=False):
    """
    Subgradient method to un-stick from local minima.
    Uses active-set weighted gradient with FFT acceleration.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)

        # Soft weights: exponential over active constraints
        T_sub = max_c * 0.01
        shifted = (conv - max_c) / T_sub
        shifted = np.clip(shifted, -300, 300)
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)

        # Compute gradient via FFT
        grad = compute_weighted_gradient(weights, a)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)

        if grad_norm < 1e-15:
            break

        # Adaptive step size
        eta = 0.02 * max_c / (grad_norm * grad_norm + 1e-30)
        eta = min(eta, 0.05)

        a_new = simplex_project(a - eta * grad_centered)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Try smaller step
                improved = False
                for scale in [0.5, 0.1, 0.05]:
                    a_try = simplex_project(a - eta * scale * grad_centered)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stall = 5
                        improved = True
                        break
                if not improved:
                    break

    return best_a


def smooth_max_descent(a, time_budget, T_init=0.0008, T_min=0.00002, lr_init=0.004):
    """Smooth max gradient descent with Nesterov momentum and adaptive scheduling."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = lr_init
    momentum = 0.9
    v = np.zeros(n)
    stall = 0

    max_iters = 20000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -400, 400)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total

        # Gradient via FFT
        grad = compute_weighted_gradient(weights, a)
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 40:
                lr *= 0.75
                momentum = 0.85
                v = np.zeros(n)
                a = best_a.copy()
                stall = 0
                if lr < 1e-7:
                    break

        # Temperature schedule: exponential decay
        if it > 10:
            T = max(T_min, T_init * np.exp(-0.003 * it))
        # Learning rate warmup then slow decay
        if it < 50:
            lr = lr_init * (1 + it / 50.0)
        elif it % 1000 == 0:
            lr = max(lr * 0.98, 1e-6)

    return best_a


def coordinate_descent(a, time_budget, delta=1e-4):
    """Coordinate descent: try tiny perturbations of individual coordinates."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    pass_count = 0
    while time.monotonic() - start < time_budget and delta > 1e-8:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count * 104729 + n * 31)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            for d in [delta, -delta, 0.3 * delta, -0.3 * delta]:
                if a[i] + d < -1e-15:
                    continue
                a_try = a.copy()
                a_try[i] = max(0.0, a[i] + d)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            delta *= 0.6

    return best_a


def pair_swap_search(a, time_budget):
    """Try pairwise mass transfers between coordinates."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99991)

    eps_schedule = [0.002, 0.001, 0.0005, 0.0002]
    eps_idx = 0

    while time.monotonic() - start < time_budget:
        if eps_idx >= len(eps_schedule):
            break
        eps = eps_schedule[eps_idx]
        
        improved_this_round = False
        for _ in range(50):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            if a[i] < eps * 0.5:
                continue
            a_try = a.copy()
            a_try[i] -= eps
            a_try[j] += eps
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved_this_round = True

        if not improved_this_round:
            eps_idx += 1

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):

            s = float(np.sum(parent_arr))

                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()

        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init

    print(f"\n{'='*60}")

            best_seq = a2.copy()
            a = a2.copy()

        if remaining > 60:

            a2r = lp_alternating(a, time_budget=p2r_budget, use_reduced=True)
            obj2r = compute_obj(a2r)
            print(f"  Re-LP: obj={obj2r:.8f}")
            if obj2r < best_score:

                best_seq = a2r.copy()
                a = a2r.copy()

    # === Phase 3: Smooth Max Gradient Descent ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Smooth Max Gradient")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(100.0, remaining * 0.15)
        a3 = smooth_max_descent(a, time_budget=p3_budget, T_init=0.0006, T_min=0.00002, lr_init=0.003)
        obj3 = compute_obj(a3)

        if obj3 < best_score:

            best_seq = a3.copy()

        if remaining > 40:

                best_seq = a3l.copy()

    print(f"Phase 4: Perturbation Restarts")
    remaining = deadline - time.monotonic()

    for r in range(n_restarts):

        elif strategy == 1:

        a_n = lp_alternating(a_try, time_budget=b_n, use_reduced=True)
        obj_n = compute_obj(a_n)

        if remaining > 25 and obj_n < best_score + 0.004:
            a_n = coordinate_descent(a_n, time_budget=min(15.0, remaining * 0.3), delta=1e-4)
            obj_n = compute_obj(a_n)
