**Mirror Descent (Multiplicative Weights) for this problem:**

We want to minimize f(a) = max_k (A@g)_k where the constraint is a in simplex.

Wait, actually we're minimizing f(a) = max_k (a*a)_k subject to a in simplex (sum=1, a>=0).

∂f/∂a_i = sum_j a_j * δ_{i+j, k*} = a_{k*-i} (if valid)

For non-unique maxima, we take a convex combination.

Then normalize: a = a / sum(a)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. U-shaped initialization based on spectral analysis of the sum distribution
2. Geometric (multiplicative) path following for LP blending
3. Dual-guided mirror descent with adaptive step size
4. Multi-scale optimization (coarse shape + fine tuning)

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def geometric_interpolate(a, b, t, eps=1e-15):
    """Geometric (multiplicative) interpolation on the simplex.
    a_t = normalize(a^(1-t) * b^t)
    This is the geodesic in the information geometry of the simplex.
    """
    n = len(a)
    # Handle zeros by adding epsilon
    a_safe = np.maximum(a, eps)
    b_safe = np.maximum(b, eps)
    # Compute in log space for numerical stability
    log_result = (1 - t) * np.log(a_safe) + t * np.log(b_safe)
    # Shift for numerical stability
    log_result -= np.max(log_result)
    result = np.exp(log_result)
    s = float(np.sum(result))
    if s < 1e-30:
        return b.copy()
    return result / s


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal, dual_variables) or (None, None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    # We need dual variables for the convolution constraints
    conv_constraints = A @ g <= M
    constraints = [conv_constraints, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    M_val = float(M.value)
                    # Try to get dual variables
                    dual_y = None
                    try:
                        dual_y = conv_constraints.dual_value
                        if dual_y is not None:
                            dual_y = np.maximum(np.asarray(dual_y), 0)
                            ds = float(np.sum(dual_y))
                            if ds > 1e-15:
                                dual_y = dual_y / ds
                    except Exception:
                        pass
                    return g_val, M_val, dual_y
        except Exception:
            continue

    return None, None, None


def geometric_lp_iteration(a, time_budget=100.0, verbose=False):
    """
    Iteratively solve LP and move along geometric path.
    At each step:
    1. Solve LP to find g*
    2. Find optimal t on geometric path a -> g*
    3. Update a
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        g_lp, M_lp, dual_y = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Find optimal t on geometric path
        # Use golden section search
        a_lo, a_hi = 0.05, 1.0
        obj_lo = compute_obj(geometric_interpolate(a, g_lp, a_lo))
        obj_hi = compute_obj(geometric_interpolate(a, g_lp, a_hi))
        
        # Golden section
        gr = (np.sqrt(5) - 1) / 2
        c = a_hi - gr * (a_hi - a_lo)
        d = a_lo + gr * (a_hi - a_lo)
        obj_c = compute_obj(geometric_interpolate(a, g_lp, c))
        obj_d = compute_obj(geometric_interpolate(a, g_lp, d))
        
        for _ in range(15):  # Fixed number of golden section iterations
            if obj_c < obj_d:
                a_hi, obj_hi = d, obj_d
                d, obj_d = c, obj_c
                c = a_hi - gr * (a_hi - a_lo)
                obj_c = compute_obj(geometric_interpolate(a, g_lp, c))
            else:
                a_lo, obj_lo = c, obj_c
                c, obj_c = d, obj_d
                d = a_lo + gr * (a_hi - a_lo)
                obj_d = compute_obj(geometric_interpolate(a, g_lp, d))
        
        t_opt = (a_lo + a_hi) / 2
        a_new = geometric_interpolate(a, g_lp, t_opt)
        obj_new = compute_obj(a_new)
        
        it += 1
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 3 == 0:
                print(f"    GeomLP it{it}: obj={best_obj:.8f}, t={t_opt:.3f}, M={M_lp:.6f}")
        else:
            # Try perturbation
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if it > 20:
                break

    return best_a


def mirror_descent_peaks(a, time_budget=100.0, verbose=False):
    """
    Mirror descent (multiplicative weights) focusing on the top peaks.
    Uses softmax-weighted subgradient for smoothness.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 5.0 / np.sqrt(n)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        
        # Softmax weights over top peaks
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break
        tau = 50.0
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))
        
        # Compute weighted subgradient via FFT
        # g[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation)
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        it += 1
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slightly increase eta if making good progress
            if it % 10 == 0:
                eta *= 1.02
            if verbose and it % 50 == 0:
                print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.4f}")
        else:
            stall += 1
            eta *= 0.7
            if stall > 50:
                # Reset with perturbation
                a = best_a * np.exp(np.random.default_rng(it).normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                eta = 5.0 / np.sqrt(n)
                stall = 0
                if it > 500:
                    break

    return best_a


def u_shaped_init(n, alpha):
    """Generate U-shaped initialization: high at ends, low in middle."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + alpha * x**2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def cosine_init(n, alpha, freq=1.0):
    """Cosine-based initialization."""
    x = np.linspace(0, np.pi, n)
    a = 1.0 + alpha * np.cos(freq * x)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def smoothed_gradient_fft(a, tau):
    """Compute smoothed gradient using FFT."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv


def nesterov_descent(a, time_budget=200.0, tau_init=10.0, tau_end=300.0, verbose=False):
    """Nesterov accelerated smoothed gradient descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        # Barzilai-Borwein
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.7 * lr + 0.3 * min(max(bb_lr, 1e-8), 5.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9997)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations:
1. U-shaped initialization based on sum-distribution flattening analysis
2. Geometric (multiplicative) path following for LP blending
3. Dual-guided mirror descent with adaptive step size
4. Multi-scale exploration with diverse restarts
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def geometric_interpolate(a, b, t, eps=1e-15):
    """Geometric (multiplicative) interpolation on the simplex.
    a_t = normalize(a^(1-t) * b^t)
    """
    n = len(a)
    a_safe = np.maximum(a, eps)
    b_safe = np.maximum(b, eps)
    log_result = (1.0 - t) * np.log(a_safe) + t * np.log(b_safe)
    log_result -= np.max(log_result)
    result = np.exp(log_result)
    s = float(np.sum(result))
    if s < 1e-30:
        return b.copy()
    return result / s


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_optimal, M_optimal) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    return g_val, float(M.value)
        except Exception:
            continue
    return None, None


def geometric_lp_iteration(a, time_budget=100.0, verbose=False):
    """
    Iteratively solve LP and move along geometric path with golden section search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    rng = np.random.default_rng(777 + n)
    no_improve = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Golden section search on geometric path
        a_lo, a_hi = 0.03, 1.0
        gr = (np.sqrt(5) - 1) / 2
        c = a_hi - gr * (a_hi - a_lo)
        d = a_lo + gr * (a_hi - a_lo)

        obj_c = compute_obj(geometric_interpolate(a, g_lp, c))
        obj_d = compute_obj(geometric_interpolate(a, g_lp, d))

        for _ in range(12):
            if obj_c < obj_d:
                a_hi, obj_d, d = a_hi, obj_c, c
                c = a_hi - gr * (a_hi - a_lo)

            else:

                obj_d = compute_obj(geometric_interpolate(a, g_lp, d))

def mirror_descent_peaks(a, time_budget=100.0, verbose=False):
    """
    Mirror descent (multiplicative weights) with softmax-weighted subgradient.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 3.0 / np.sqrt(n)
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        # Softmax weights
        tau = 40.0
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        # Weighted subgradient via FFT cross-correlation
        N = 1
        while N < 3 * n:
            N <<= 1
        P = fft(p, N)
        A = fft(a, N)
        g_full = ifft(P * np.conj(A)).real
        g = 2.0 * g_full[:n]

        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        it += 1

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 20 == 0:
                eta = min(eta * 1.03, 20.0 / np.sqrt(n))

                print(f"    MD it{it}: obj={best_obj:.8f}, eta={eta:.4f}")
        else:
            stall += 1
            eta *= 0.75
            if stall > 60:

                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                eta = 3.0 / np.sqrt(n)
                stall = 0
                if it > 800:
                    break

    return best_a


def u_shaped_init(n, alpha):
    """U-shaped: high at ends, low in middle."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + alpha * x**2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def cosine_init(n, alpha, freq=1.0):
    """Cosine-based initialization."""
    x = np.linspace(0, np.pi, n)
    a = 1.0 + alpha * np.cos(freq * x)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def bimodal_init(n, center_frac, width_frac):
    """Two Gaussian bumps at specified positions."""
    a = np.zeros(n)
    c1 = int(center_frac * n)
    c2 = n - 1 - c1
    w = max(1, int(width_frac * n))
    x = np.arange(n)
    a += np.exp(-((x - c1)**2) / (2.0 * w**2))
    a += np.exp(-((x - c2)**2) / (2.0 * w**2))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def smoothed_gradient_fft(a, tau):
    """Compute smoothed gradient using FFT cross-correlation."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    N = 1
    while N < 3 * n:
        N <<= 1
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv


def nesterov_descent(a, time_budget=200.0, tau_init=10.0, tau_end=300.0, verbose=False):
    """Nesterov accelerated smoothed gradient descent on simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget * 0.9, 1e-6))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        g, conv = smoothed_gradient_fft(y, tau)

        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.7 * lr + 0.3 * min(max(bb_lr, 1e-8), 5.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9997)

        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    p2_budget = min(remaining * 0.35, 400.0)
    if p2_budget > 30:
        a_glp = geometric_lp_iteration(best_seq, time_budget=p2_budget, verbose=True)
        obj_glp = compute_obj(a_glp)
        if obj_glp < best_score:
            best_score = obj_glp
            best_seq = a_glp.copy()
            print(f"  Phase2 GeomLP: obj={best_score:.8f} **")
        else:
            print(f"  Phase2 GeomLP: obj={obj_glp:.8f}")

        # Second round with perturbation
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc = min(remaining * 0.25, 150.0)
            if alloc > 25:
                a_pert = best_seq * np.exp(rng.normal(0, 0.03, len(best_seq)))
                a_pert = np.maximum(a_pert, 1e-10)
                a_pert = a_pert / a_pert.sum()
                a_glp2 = geometric_lp_iteration(a_pert, time_budget=alloc, verbose=True)
                obj_glp2 = compute_obj(a_glp2)
                if obj_glp2 < best_score:
                    best_score = obj_glp2
                    best_seq = a_glp2.copy()
                    print(f"  Phase2 GeomLP2: obj={best_score:.8f} **")

    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 20:
        a_md = mirror_descent_peaks(best_seq, time_budget=p3_budget, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Phase3 MD: obj={best_score:.8f} **")
        else:
            print(f"  Phase3 MD: obj={obj_md:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 200),
        n_current + 100, n_current + 300, n_current + 500,
        600, 900, 1100, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and abs(n - n_current) > 50]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_res = resample_sequence(best_seq, n_try)
        a_u = u_shaped_init(n_try, 2.0)

        if compute_obj(a_u) < compute_obj(a_res):
            a_try = a_u
        else:
            a_try = a_res

        alloc1 = min(remaining * 0.2, 50.0)
        if alloc1 > 15:
            a_sgd = nesterov_descent(a_try, time_budget=alloc1,
                                     tau_init=10.0, tau_end=200.0)
            if compute_obj(a_sgd) < compute_obj(a_try):
                a_try = a_sgd

        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 35.0)
        if alloc2 > 15:
            a_try = geometric_lp_iteration(a_try, time_budget=alloc2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    restart_inits = [
        u_shaped_init(n_best, 3.0),
        cosine_init(n_best, 1.2, freq=2.0),
        bimodal_init(n_best, 0.2, 0.12),
    ]

    for r, a_base in enumerate(restart_inits):
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break

        blend = 0.2 + 0.15 * r
        a_r = (1.0 - blend) * a_base + blend * best_seq
        a_r = np.maximum(a_r, 1e-10)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.22, 100.0)
        if alloc > 40:
            gd_alloc = alloc * 0.5
            a_r = nesterov_descent(a_r, time_budget=gd_alloc,
                                   tau_init=10.0, tau_end=300.0)
            remaining = deadline - time.monotonic()
            lp_alloc = min(remaining * 0.25, 50.0)
            if lp_alloc > 20:
                a_r = geometric_lp_iteration(a_r, time_budget=lp_alloc)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        alloc = min(remaining * 0.4, 180.0)
        a_glp = geometric_lp_iteration(best_seq, time_budget=alloc, verbose=True)
        obj_glp = compute_obj(a_glp)
        if obj_glp < best_score:
            best_score = obj_glp
            best_seq = a_glp.copy()
            print(f"  Final GeomLP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 50.0)
        a_md = mirror_descent_peaks(best_seq, time_budget=alloc)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 100.0)
        a_glp = geometric_lp_iteration(best_seq, time_budget=alloc)
        obj_glp = compute_obj(a_glp)
        if obj_glp < best_score:
            best_score = obj_glp
            best_seq = a_glp.copy()
            print(f"  Final GeomLP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
