def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # conv(a,g)[k] += a[j] * g[k-j]
        # So for fixed j, A[k, j] = a[j] when k-j is in [0, n-1], i.e., k in [j, j+n-1]
        k_start = j
        k_end = min(2*n-1, j+n)
        A[k_start:k_end, j] = a[:k_end-k_start]
    return A

    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

                s = float(np.sum(g_val))
                if s > 1e-10:

    return A


def solve_lp(a, tol=1e-8):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / (float(np.sum(a)) + 1e-30)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            if solver == cp.SCS:
                prob.solve(solver=solver,.eps=tol)
            elif solver == cp.MOSEK:
                prob.solve(solver=solver, msdk_params={'MSDK_DPAR_INTPNT_CO_TOL_REL_GAP': tol})
            else:
                prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_iteration(a, time_budget=300.0, use_anderson=True, verbose=True, seed=42):
    """
    LP fixed-point iteration with optional Anderson acceleration.
    Iterates: a -> (1-t)*a + t*LP(a) where LP(a) = argmin_g max_k conv(a,g).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Anderson acceleration state
    m = 4  # memory
    history_a = []
    history_g = []
    T = 0.5  # blend parameter
    stall_count = 0
    last_improve = best_obj

    it = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Solve LP
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            break

        if use_anderson and len(history_a) >= 2:
            # Anderson acceleration
            d = min(m, len(history_a))
            S_a = np.array(history_a[-d:])  # d x n
            S_g = np.array(history_g[-d:])  # d x n
            residuals = S_a - S_g  # d x n

            # Build Gram matrix
            Q = (residuals * residuals).sum(axis=1)  # d x d (Frobenius inner products)
            gamma = 1.0 / (1.0 + 0.5 * it)

            try:
                alpha = np.linalg.solve(Q + gamma * np.eye(d), np.ones(d))
                alpha = np.maximum(alpha, 0)
                a_sum = alpha.sum()
                if a_sum > 1e-10:
                    alpha /= a_sum
                else:
                    alpha = np.ones(d) / d
            except Exception:
                alpha = np.ones(d) / d

            # Anderson step
            a_new = (alpha * S_g).sum(axis=0)
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new = (1 - T) * a + T * g_lp
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            else:
                a_new = a.copy()
        else:
            # Standard step
            a_new = (1.0 - T) * a + T * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            else:
                a_new = a.copy()

        obj_new = compute_obj(a_new)

        # Update history for Anderson
        history_a.append(a.copy())
        history_g.append(a_new.copy())
        if len(history_a) > m:
            history_a = history_a[-m:]
            history_g = history_g[-m:]

        # Track best
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_count = 0
            last_improve = best_obj
        else:
            stall_count += 1

        a = a_new
        it += 1

        # Adaptive T: decrease over time for finer steps
        T = max(0.2, 0.5 * (0.97 ** (it // 20)))

        # Perturb if stuck
        if stall_count > 30:
            scale = 0.01 + 0.005 * min(stall_count // 10, 5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            history_a = [a.copy()]
            history_g = []
            stall_count = 0
            T = 0.5

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, M={M_lp:.6f}, T={T:.3f}, t={elapsed:.0f}s")

        # Early stop if converged
        if it > 50 and abs(obj_new - last_improve) < 1e-12 and best_obj - last_improve < 1e-12:
            break

    return best_a


def smoothed_gradient_descent(a, time_budget=200.0, tau_init=20.0, tau_end=500.0, verbose=True):
    """
    Nesterov accelerated gradient descent on log-sum-exp smoothed objective.
    Work on simplex (sum=1, non-negative).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute smoothed gradient via FFT
        conv = compute_conv(y)
        max_conv = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_conv))
        p = exp_vals / float(np.sum(exp_vals))

        # Cross-correlation for gradient
        N_fft = 1
        while N_fft < 3 * n:
            N_fft <<= 1
        P = fft(p, N_fft)
        A_f = fft(y, N_fft)
        g_full = ifft(P * np.conj(A_f)).real
        g = 2.0 * g_full[:n]

        # BB step size (alternate)
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 5.0 / np.sqrt(n))

        # Update
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-10, lr * 0.9997)

        # Reset if stuck
        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

        if verbose and it % 5000 == 0:
            print(f"    SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = x ** rng.uniform(0.3, 3) + (1-x) ** rng.uniform(0.3, 3)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 7:
        a = np.zeros(n)
        centers = rng.integers(0, n, size=2)
        widths = rng.integers(n//10, n//5, size=2)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        if a.sum() < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def high_precision_polish(a, time_budget=100.0, verbose=True):
    """
    Very fine LP polishing with tiny blend factors and multiple perturbations.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    it = 0
    T = 0.1
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # Try very small blend factors
        for t_try in [T, T * 0.5, T * 1.5, T * 0.3]:
            a_try = (1.0 - t_try) * a + t_try * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break

        it += 1
        T = max(0.02, T * 0.98)

        # Periodic perturbation
        if it % 20 == 0:
            scale = 0.005
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and it % 5 == 0:
            print(f"    HP it{it}: obj={best_obj:.8f}, T={T:.4f}, M={M_lp:.6f}")

    return best_a

                print(f"  n={n_try}: obj={obj_out:.8f}")

        # Quick gradient polish
        remaining = deadline - time.monotonic()

            if alloc > 20:
                a_sg = smoothed_gradient_descent(best_seq, time_budget=alloc, verbose=False)

            else:

    # === Phase 7: Final micro-polish ===
    print(f"\n=== Phase 7: Final micro-polish ===")

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
