def compute_conv(a):
    n = len(a)
    if n==0:
        return np.array([0.0])
    #Convolvefull-lengthsequences
    cur = np.array(a, dtype=np.float64)
    #FFT-basedconvolution
    size_needed = 2*n-1
    size = 1
    while size < size_needed:
        size *= 2
    return irfft(rfft(cur, size) * rfft(cur, size))[:2*n-1]
