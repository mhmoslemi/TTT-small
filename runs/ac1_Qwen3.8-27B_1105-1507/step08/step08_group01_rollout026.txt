"""
LP-alternating with simulated annealing, parabolic line search,
multi-n exploration, and adaptive coordinate descent polish.
Optimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return None, None
    a = a / s

    solved = False
    g_val = None
    M_val = None

    try:
        import cvxpy as cp
        A = build_conv_matrix(a)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        for solver in ['CLARABEL', 'ECOS', 'SCS', 'MOSEK']:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    ss = float(np.sum(g_val))
                    if ss > 1e-10:
                        M_val = float(M.value)
                        g_val = g_val / ss
                        solved = True
                        break
            except Exception:
                continue
    except ImportError:
        pass

    if not solved:
        from scipy.optimize import linprog
        A = build_conv_matrix(a)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            ss = float(np.sum(g_val))
            if ss > 1e-10:
                g_val = g_val / ss
                M_val = float(res.x[n])

    return g_val, M_val


def parabolic_line_search(a, g, n):
    """
    Parabolic line search for a_new = (1-t)*a + t*g.
    Uses 3-point quadratic fit then refines.
    """
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    # Initial 3-point fit
    t1, t2, t3 = 0.1, 0.4, 0.8
    f1, f2, f3 = obj_at_t(t1), obj_at_t(t2), obj_at_t(t3)

    best_t = t1
    best_f = f1
    if f2 < best_f:
        best_t, best_f = t2, f2
    if f3 < best_f:
        best_t, best_f = t3, f3

    # Parabolic interpolation
    denom = (t1 - t2) * (t1 - t3) * (t2 - t3)
    if abs(denom) > 1e-15:
        tp = t1 * (f2 * (t1 - t3)) / (f2 - f1) / (f2 - f3) + \
             t2 * (f3 * (t1 - t2)) / (f3 - f1) / (f3 - f2) + \
             t3 * (f1 * (t2 - t3)) / (f1 - f2) / (f1 - f3)
        if 0 < tp < 1:
            fp = obj_at_t(tp)
            if fp < best_f:
                best_t, best_f = tp, fp

    # Refine with golden section around best_t
    lo = max(0.0, best_t - 0.2)
    hi = min(1.0, best_t + 0.2)
    gr = (np.sqrt(5) - 1) / 2
    for _ in range(15):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        f1g = obj_at_t(m1)
        f2g = obj_at_t(m2)
        if f1g < f2g:
            hi = m2
        else:
            lo = m1
    t_opt = (lo + hi) / 2.0
    f_opt = obj_at_t(t_opt)
    if f_opt > best_f:
        t_opt = best_t

    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a_norm
    return a_new / s

def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

"""

multi-n exploration, and adaptive coordinate descent polish.
Optimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return None, None
    a = a / s

    solved = False
    g_val = None
    M_val = None

    try:
        import cvxpy as cp
        A = build_conv_matrix(a)
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)

        for solver in ['CLARABEL', 'ECOS', 'SCS']:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    ss = float(np.sum(g_val))
                    if ss > 1e-10:
                        M_val = float(M.value)
                        g_val = g_val / ss
                        solved = True
                        break
            except Exception:
                continue
    except ImportError:
        pass

    if not solved:
        from scipy.optimize import linprog
        A = build_conv_matrix(a)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            ss = float(np.sum(g_val))
            if ss > 1e-10:
                g_val = g_val / ss
                M_val = float(res.x[n])

    return g_val, M_val


def parabolic_line_search(a, g, n):
    """
    Parabolic line search for a_new = (1-t)*a + t*g.
    Uses 3-point quadratic fit then golden section refine.
    """
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    # Initial 3-point evaluation
    t_pts = [0.05, 0.3, 0.6, 0.9]
    f_pts = [obj_at_t(t) for t in t_pts]

    best_idx = int(np.argmin(f_pts))
    best_t = t_pts[best_idx]
    best_f = f_pts[best_idx]

    # Parabolic interpolation using 3 closest points
    if best_idx > 0 and best_idx < len(t_pts) - 1:
        i1, i2, i3 = best_idx - 1, best_idx, best_idx + 1
    elif best_idx == 0:
        i1, i2, i3 = 0, 1, 2
    else:
        i1, i2, i3 = len(t_pts) - 3, len(t_pts) - 2, len(t_pts) - 1

    x1, x2, x3 = t_pts[i1], t_pts[i2], t_pts[i3]
    y1, y2, y3 = f_pts[i1], f_pts[i2], f_pts[i3]

    denom = (x1 - x2) * (x1 - x3) * (x2 - x3)
    if abs(denom) > 1e-15:
        tp = (x1 * x1 * (y2 * (x1 - x3) - y3 * (x1 - x2)) / (
                    (y1 - y2) * (y1 - y3))) + \
             (x2 * x2 * (y3 * (x1 - x2) - y1 * (x2 - x3)) / (
                    (y2 - y1) * (y2 - y3))) + \
             (x3 * x3 * (y1 * (x2 - x3) - y2 * (x1 - x3)) / (
                    (y3 - y1) * (y3 - y2)))
        # Simpler formula for vertex of parabola through 3 points
        d1 = (y1 - y2) / (x1 - x2) if x1 != x2 else 0
        d2 = (y2 - y3) / (x2 - x3) if x2 != x3 else 0
        if abs(d2 - d1) > 1e-15:
            tp = x2 - (y2 - y1) / (2 * (d1 - d2)) if abs(d1 - d2) > 1e-15 else x2
        else:
            tp = (x1 + x3) / 2

        if 0.01 < tp < 0.99:
            fp = obj_at_t(tp)
            if fp < best_f:
                best_t, best_f = tp, fp

    # Golden section refinement around best
    half = 0.15
    lo = max(0.0, best_t - half)
    hi = min(1.0, best_t + half)
    if hi - lo < 0.01:
        lo, hi = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(20):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        f1g = obj_at_t(m1)
        f2g = obj_at_t(m2)
        if f1g < f2g:
            hi = m2
        else:
            lo = m1
    t_opt = (lo + hi) / 2.0
    f_opt = obj_at_t(t_opt)
    if f_opt > best_f:
        t_opt = best_t

    a_new = (1.0 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a_norm
    return a_new / s_new


def lp_alternating_sa(a, time_budget=200.0, verbose=False):
    """LP alternating with simulated annealing for escaping local minima."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    temp = 0.008
    min_temp = 0.00005
    rng = np.random.default_rng(12345 + n)

    max_iters = 3000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        a_cand = parabolic_line_search(a, g_lp, n)
        obj_cand = compute_obj(a_cand)

        if obj_cand < best_obj - 1e-14:
            best_obj = obj_cand
            best_a = a_cand.copy()
            a = a_cand
            stall = 0
            if verbose and it % 25 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, el={time.monotonic() - start:.1f}s")
        else:
            # Try smaller step
            a_small = 0.95 * a + 0.05 * g_lp
            a_small = np.maximum(a_small, 0)
            ss = float(np.sum(a_small))
            if ss > 1e-10:
                a_small = a_small / ss
                obj_small = compute_obj(a_small)
                if obj_small < best_obj - 1e-14:
                    best_obj = obj_small
                    best_a = a_small.copy()
                    a = a_small
                    stall = 0
                    continue

            stall += 1
            if stall > 8:
                # SA jump
                noise = rng.normal(0, 1, n)
                a_pert = a + temp * noise
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    delta_obj = obj_pert - best_obj
                    if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(temp, 1e-8)):
                        a = a_pert
                        if obj_pert < best_obj:
                            best_obj = obj_pert
                            best_a = a_pert.copy()
                            stall = 0
                    else:
                        a = a_pert
                temp = max(min_temp, temp * 0.93)
                stall = 4

        if stall > 60:
            a = best_a.copy()
            temp = max(min_temp, temp * 0.5)
            stall = 0

    return best_a


def coordinate_descent(a, time_budget=60.0, init_delta=0.005, verbose=False):
    """Adaptive coordinate descent."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999 + n)
    delta = init_delta

    passes = 0
    while time.monotonic() - start < time_budget:
        passes += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            a_try = a.copy()
            a_try[i] += delta
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st > 1e-10:
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    continue

            if a[i] > delta * 0.3:
                a_try = a.copy()
                a_try[i] -= delta
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st > 1e-10:
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True

        if verbose and passes % 5 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.8f}, d={delta:.6f}, t={time.monotonic() - start:.1f}s")

        if not improved:
            delta *= 0.55
            if delta < 1e-7:
                break

    return best_a


def nesterov_smooth_max(a, time_budget=40.0, verbose=False):
    """Nesterov-accelerated gradient on smooth max."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777 + n)

    T = 0.0004
    lr = 0.0015
    momentum = 0.9
    v = np.zeros(n)
    stall = 0

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = np.clip((conv - max_c) / T, -350, 350)
        exp_v = np.exp(shifted)
        total = np.sum(exp_v)
        if total < 1e-30:
            weights = np.zeros_like(conv)
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights = exp_v / total

        # Compute gradient using FFT-based correlation
        # grad[j] = sum_k w_k * (a[(k-j) mod valid] + a[|j-k|] valid)
        # More precisely: grad[j] = sum_k w_k * cross_corr(a,a,k)[j]
        # where cross_corr at lag k evaluated at j is a[k-j] + a[j-k] (if in range)
        # This equals: ifft(weights) convolved with a, plus a convolved with ifft(weights)
        # Actually let's just do it directly for the top-weighted terms
        grad = np.zeros(n)
        top_k = min(50, 2 * n - 1)
        idx = np.argsort(weights)[::-1][:top_k]
        for k in idx:
            if weights[k] < 1e-8:
                break
            wk = weights[k]
            # Contribution: for each j, a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            # a[k-j] valid when j in [max(0,k-n+1), min(n-1,k)]
            j_lo1 = max(0, k - n + 1)
            j_hi1 = min(n - 1, k)
            if j_lo1 <= j_hi1:
                grad[j_lo1:j_hi1 + 1] += wk * a[k - j_hi1:k - j_lo1 + 1][::-1]
            # a[j-k] valid when j in [k, min(n-1, k+n-1)] = [k, min(n-1, 2k+n-1)]
            j_lo2 = max(0, k)
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                grad[j_lo2:j_hi2 + 1] += wk * a[0:j_hi2 - j_lo2 + 1]

        grad_centered = grad - np.mean(grad)
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 200 == 0:
                print(f"    NM it{it}: obj={best_obj:.8f}, T={T:.6f}, t={time.monotonic() - start:.1f}s")
        else:
            stall += 1
            if stall > 40:
                lr *= 0.65
                if lr < 1e-7:
                    break
                a = best_a.copy()
                a = a + rng.normal(0, 0.0005, n)
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                v = np.zeros(n)
                stall = 0

        if it > 50:
            T = max(0.00003, T * 0.996)

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def random_init(n, rng):
    """Generate structured random initialization."""
    choice = rng.random()
    if choice < 0.25:
        a = np.linspace(0.1, 1.0, n)
    elif choice < 0.45:
        x = np.linspace(0, 1, n)
        center = 0.25 + 0.5 * rng.random()
        a = np.exp(-8 * (x - center) ** 2)
    elif choice < 0.65:
        x = np.linspace(0, 1, n)
        freq = 1.0 + 2.0 * rng.random()
        phase = rng.random() * 2 * np.pi
        a = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + phase)
    elif choice < 0.8:
        # Two-bump asymmetric
        x = np.linspace(0, 1, n)
        c1 = 0.2 + 0.1 * rng.random()
        c2 = 0.6 + 0.2 * rng.random()
        a = np.exp(-20 * (x - c1) ** 2) + np.exp(-20 * (x - c2) ** 2)
    else:
        a = rng.random(n)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1500):

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={len(parent)}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)

    # === Phase 1: LP + SA on parent ===
    n_parent = len(best_seq)
    print(f"\n=== Phase 1: LP+SA (n={n_parent}) ===")

    a_opt = lp_alternating_sa(best_seq, time_budget=p1_budget, verbose=True)
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_score:
        best_score = obj_opt

        print(f"  Phase 1: obj={best_score:.8f}")
    else:
        print(f"  Phase 1: obj={obj_opt:.8f}")

    # === Phase 2: Multi-n ===
    print(f"\n=== Phase 2: Multi-n ===")
    n_base = len(best_seq)
    n_cands = sorted(set([

        n_base,
        n_base + 40, n_base + 80, n_base + 120,
        700, 750, 850, 900, 950, 1000, 1050, 1100, 1200
    ]))
    n_cands = [nn for nn in n_cands if 500 <= nn <= 1300]

    for n_try in n_cands:

        if remaining < 50:
            break

        if n_try == n_base:
            a_try = best_seq.copy()
        else:

            a_try = a_try / s_try

        alloc = min(remaining * 0.12, 35.0)
        a_opt_n = lp_alternating_sa(a_try, time_budget=alloc, verbose=False)
        obj_n = compute_obj(a_opt_n)

        if obj_n < best_score:
            best_score = obj_n
            best_seq = a_opt_n.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_n:.8f}")

    # === Phase 3: Deep LP on best ===
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Deep LP (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc = min(200.0, remaining * 0.3)
        a_opt = lp_alternating_sa(best_seq, time_budget=alloc, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()

        else:

        remaining = deadline - time.monotonic()

    # === Phase 7: Final micro-polish ===
    remaining = deadline - time.monotonic()
    if remaining > 12:
        alloc = min(remaining * 0.35, 30.0)
        a_fp = coordinate_descent(best_seq, time_budget=alloc, init_delta=0.0002)

        if obj_fp < best_score:
            best_score = obj_fp
            best_seq = a_fp.copy()
            print(f"  Final: obj={best_score:.8f}")

    # Validate
    if best_seq is None:

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:

        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
