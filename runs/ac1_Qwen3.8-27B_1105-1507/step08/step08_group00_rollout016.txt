from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Subgradient descent with Polyak step, momentum, and restarts."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lb = 2.0 * n / (2 * n - 1)
    lr = 2.0 / np.sqrt(n)
    momentum = 0.5
    v = np.zeros(n)
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.85))
        k_cur = max(3, int(top_k * (0.3 + 0.7 * progress)))
        
        g, conv, peaks = get_subgradient(a, k_cur)
        
        # Polyak step
        gap = best_obj - lb
        if gap < 1e-10:
            break
        gg = float(np.dot(g, g))
        if gg < 1e-20:
            break
        lr_p = gap / (gg + 1e-20)
        lr = np.clip(lr_p, lr * 0.5, min(10.0 / np.sqrt(n), lr * 2.0))
        
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
            if stall > 10000:
                a = best_a.copy()
                v = np.zeros(n)
                lr = 2.0 / np.sqrt(n)
                momentum = 0.3
                stall = 0
        
        it += 1
        if verbose and it % 50000 == 0:
