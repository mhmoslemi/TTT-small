    Returns g.

    best_val = np.max(compute_conv_a_b(a, g))

        cur_max = c.max()

        g_new = g * (1 - t_plus)
        g_new[j_plus] += t_plus

        # Away step FW (Cortes et al.)
        g_away = g.copy()
        # Move away from j_minus

            best_g = g.copy()

"""
Optimized step function search using fast Frank-Wolfe alternating optimization,
mirror descent, and adaptive restarts. Key innovation: replace slow LP solves
with O(n log n) Frank-Wolfe iterations for 10-100x speedup, enabling far more
iterations within the time budget.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv_full(a, b=None):
    """Linear convolution a*b via FFT. If b is None, computes a*a."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size) if b is not a else fa
    return irfft(fa * fb, size)[:n + m - 1]


def compute_conv_self(a):
    """Linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_self(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def frank_wolfe_inner(a, max_iter=2000, tol=1e-10):
    """
    Given fixed a (sum=1), find g on simplex minimizing max_k conv(a,g)_k.
    Uses Frank-Wolfe with away steps. O(n log n) per iteration.
    """
    n = len(a)
    g = np.ones(n) / n
    
    best_g = g.copy()
    best_val = np.inf
    
    for it in range(max_iter):
        c = compute_conv_full(a, g)
        cur_max = c.max()
        k_star = int(c.argmax())
        
        if cur_max < best_val:
            best_val = cur_max
            best_g = g.copy()
        
        # Subgradient: s[j] = a[k_star - j] if 0 <= k_star - j < n
        s = np.zeros(n)
        j_lo = max(0, k_star - (n - 1))
        j_hi = min(n - 1, k_star)
        s[j_lo:j_hi+1] = a[k_star - j_hi : k_star - j_lo + 1][::-1]
        
        # FW: move towards argmin of s
        j_plus = int(np.argmin(s))
        
        # Standard FW step
        t = 2.0 / (it + 2)
        g_fw = g * (1 - t)
        g_fw[j_plus] += t
        
        # Away-step: consider moving away from high-s support point
        support = np.where(g > 1e-12)[0]
        g_away = None
        if len(support) > 1:
            j_minus = support[np.argmax(s[support])]
            if j_minus != j_plus:
                # Away step
                g_away = g.copy()
                g_away[j_minus] = 0.0
                # Redistribute: scale up the rest, add to j_plus
                rest_sum = 1.0 - g[j_plus]
                if rest_sum > 1e-15:
                    g_away = g * (t / rest_sum)  # Hmm this isn't quite right
                # Actually, proper away-step FW:
                # g_new = (1-t)*g + t*e_{j+} - t*g[j-]*e_{j-}/(1-g[j-])
                # Let me just use the simple version
                g_away = g.copy()
                alpha = min(t, g[j_minus] - 1e-15)
                g_away[j_minus] -= alpha
                g_away[j_plus] += alpha
                g_away = np.maximum(g_away, 0)
                g_away /= g_away.sum()
        
        # Pick best update
        val_fw = compute_conv_full(a, g_fw).max()
        if g_away is not None:
            val_away = compute_conv_full(a, g_away).max()
            if val_away < val_fw:
                g = g_away
                cur_max = val_away
            else:
                g = g_fw
                cur_max = val_fw
        else:
            g = g_fw
            cur_max = val_fw
        
        # Convergence check
        if it > 10 and cur_max > best_val - tol:
            # No improvement from best
            pass  # Continue anyway for a bit
    
    return best_g, best_val


def fw_alternate(a, time_budget=100.0, verbose=False):
    """
    Alternating Frank-Wolfe: fix a, optimize g; update a; repeat.
    """
    n = len(a)
    a = a / a.sum()
    a = np.maximum(a, 0)

def mirror_descent_direct(a, time_budget=100.0, verbose=False):
    """
    Direct mirror descent on the non-convex objective max_k conv(a,a)_k.
    Uses the fact that the subgradient is sparse.
    """
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 1.0 / np.sqrt(n)
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    patience = 300
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv_self(a)
        k_star = int(conv.argmax())
        max_val = conv[k_star]
        
        if max_val < 1e-15:
            break
        
        # Subgradient: s[i] = 2*a[k_star-i] if 0<=k_star-i<n
        s = np.zeros(n)
        j_lo = max(0, k_star - (n - 1))
        j_hi = min(n - 1, k_star)
        s[j_lo:j_hi+1] = 2.0 * a[k_star - j_hi : k_star - j_lo + 1][::-1]
        
        # Also handle the boundary case: when k_star-i = i (i.e., k_star = 2i)
        # The gradient is still 2*a[i] in that case (both terms contribute a[i])
        # So the formula above is correct.
        
        # Momentum
        v = momentum * v + s
        
        # Mirror descent update (exponential family)
        log_a = np.log(a)
        log_a -= lr * v
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        
        # Line search
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            # Increase momentum slightly
            momentum = min(0.9, momentum + 0.001)
        else:
            # Try smaller step
            log_a2 = np.log(a)
            log_a2 -= lr * 0.3 * v
            log_a2 -= log_a2.max()
            a2 = np.exp(log_a2)
            a2 = a2 / a2.sum()
            obj2 = compute_obj(a2)
            if obj2 < best_obj:
                best_obj = obj2
                best_a = a2.copy()
                a = a2.copy()
                stall = 0
            else:
                stall += 1
                lr *= 0.95
                if stall > patience:
                    # Restart
                    rng = np.random.default_rng(it * 7 + n)
                    a = best_a * np.exp(rng.normal(0, 0.2, n))
                    a = np.maximum(a, 1e-15)
                    a = a / a.sum()
                    v = np.zeros(n)
                    momentum = 0.3
                    lr = 1.0 / np.sqrt(n)
                    stall = 0
        
        it += 1
        if verbose and it % 200 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, lr={lr:.4f}, t={time.monotonic()-start:.1f}s")
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimized step function search using fast Frank-Wolfe alternating with incremental
convolution updates (O(n) per inner iteration instead of O(n log n)), combined with
direct mirror descent on the non-convex objective, adaptive multi-dimension exploration,
and diverse restarts. Key innovation: incremental FW convolution update eliminates the
FFT bottleneck in the inner loop, enabling 100x more effective iterations than LP-based
alternating within the same time budget.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv_full(a, b=None):
    """Linear convolution a*b via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size) if b is not a else fa
    return irfft(fa * fb, size)[:n + m - 1]


def compute_conv_self(a):
    """Linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_self(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    g = np.ones(n) / n

    # Initial convolution via FFT
    conv = compute_conv_full(fixed_a, g).copy()
    cur_max = conv.max()

    best_g = g.copy()
    best_max = cur_max

    prev_max = cur_max

    for it in range(max_iter):
        k_star = int(np.argmax(conv))
        cur_max = conv[k_star]

        if cur_max < best_max:
            best_max = cur_max
            best_g = g.copy()

        j_hi = min(n - 1, k_star)
        s[j_lo:j_hi + 1] = fixed_a[k_star - j_hi:k_star - j_lo + 1][::-1]

        if support_mask.sum() > 1:
            s_on_support = s.copy()
            s_on_support[~support_mask] = -np.inf
            j_minus = int(np.argmax(s_on_support))
            if j_minus != j_plus and s[j_minus] > s[j_plus]:
                g_away_accept = True

        # Step size
        t = 2.0 / (it + 2)

        # Precompute delta_conv for e_{j_plus}: conv(fixed_a, e_j)[k] = a[k-j]
        delta_plus = np.zeros(2 * n - 1)
        delta_plus[j_plus:j_plus + n] = fixed_a

        if g_away_accept:
            delta_minus = np.zeros(2 * n - 1)
            delta_minus[j_minus:j_minus + n] = fixed_a

        # Evaluate FW step
        conv_fw = (1.0 - t) * conv + t * delta_plus
        val_fw = conv_fw.max()

        if g_away_accept:
            # Away step: reduce g[j_minus], increase g[j_plus]
            alpha = min(t, g[j_minus] - 1e-15)
            if alpha > 1e-15:
                conv_away = conv + alpha * (delta_plus - delta_minus)
                val_away = conv_away.max()
                if val_away < val_fw:
                    conv = conv_away
                    g = g.copy()
                    g[j_minus] -= alpha
                    g[j_plus] += alpha
                    g = np.maximum(g, 0)
                    g /= g.sum()
                    cur_max = val_away
                else:
                    conv = conv_fw
                    g = g * (1.0 - t)
                    g[j_plus] += t
                    cur_max = val_fw
            else:
                conv = conv_fw
                g = g * (1.0 - t)
                g[j_plus] += t
                cur_max = val_fw
        else:
            conv = conv_fw
            g = g * (1.0 - t)
            g[j_plus] += t
            cur_max = val_fw

        # Periodic FFT refresh to avoid numerical drift
        if (it + 1) % 2000 == 0:
            conv = compute_conv_full(fixed_a, g)

        # Convergence: if no improvement in last 500 iters and values are stable
        if it > 500 and abs(cur_max - best_max) < tol and abs(prev_max - cur_max) < tol:
            if it > 1000:
                break
        prev_max = cur_max

    return best_g, best_max


def fw_alternate_optimize(a, time_budget=200.0, verbose=False):
    """
    Alternating Frank-Wolfe: fix a, optimize g via FW; update a; repeat.
    Tracks actual objective max_k conv(a,a)_k.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    g = a.copy()
    stall = 0
    patience = 100
    best_t = 0.3
    temperature = 0.1

    it = 0
    while time.monotonic() - start < time_budget:
        # Inner FW: fix a, find best g
        g, g_val = fw_inner_fast(a, max_iter=2000)

        # Line search: try several blend factors
        best_local_obj = best_obj
        best_local_t = 0
        best_local_a = best_a.copy()

        ts = [best_t * 0.5, best_t * 0.7, best_t, best_t * 1.3, best_t * 1.5,
              0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        ts = [t for t in ts if 0 < t <= 1.0]

        for t in ts:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_t = t
                best_local_a = a_try.copy()

        it += 1

        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_local_a.copy()
            best_t = best_local_t
            stall = 0
            temperature = max(0.02, temperature * 0.99)
        else:
            stall += 1
            if stall > patience:
                # Restart with perturbation
                rng = np.random.default_rng(it * 2654435761 % (2**31) + n)
                scale = temperature
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                g = a.copy()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                best_t = 0.3
                temperature = min(0.3, temperature * 1.5 + 0.05)
            elif stall > patience // 2:
                # Try a different step
                rng = np.random.default_rng(it * 7919)
                best_t = float(np.clip(best_t + rng.normal(0, 0.1), 0.05, 1.0))

        if verbose and it % 30 == 0:
            print(f"    FW-Alt it{it}: obj={best_obj:.8f}, stall={stall}, t={time.monotonic()-start:.1f}s")

    return best_a, best_obj


def bi_directional_fw(a, time_budget=200.0, verbose=False):
    """
    Bi-directional FW alternating: alternately fix a->optimize g and fix g->optimize a.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    g = a.copy()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall = 0
    patience = 80
    it = 0

    while time.monotonic() - start < time_budget:
        # Step 1: Fix a, optimize g
        g_new, _ = fw_inner_fast(a, max_iter=1500)
        g = g_new

        # Step 2: Fix g, optimize a
        a_new, _ = fw_inner_fast(g, max_iter=1500)
        a = a_new

        # Check actual objective
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng.normal(0, 0.15, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                g = a.copy()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

        it += 1
        if verbose and it % 20 == 0:
            print(f"    BiDir it{it}: obj={best_obj:.8f}, stall={stall}, t={time.monotonic()-start:.1f}s")

    return best_a, best_obj


def mirror_descent_direct(a, time_budget=150.0, verbose=False):
    """
    Direct mirror descent on the non-convex objective max_k conv(a,a)_k on the simplex.
    Uses the sparse subgradient structure.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 5.0 / np.sqrt(n)
    decay = 0.9995
    momentum = 0.3
    v = np.zeros(n)
    stall = 0
    patience = 200

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv_self(a)
        k_star = int(conv.argmax())
        max_val = conv[k_star]

        if max_val < 1e-15:
            break

        # Subgradient: s[i] = 2*a[k_star-i] if 0<=k_star-i<n
        s = np.zeros(n)
        j_lo = max(0, k_star - (n - 1))
        j_hi = min(n - 1, k_star)
        s[j_lo:j_hi + 1] = 2.0 * a[k_star - j_hi:k_star - j_lo + 1][::-1]

        # Momentum
        v = momentum * v + s

        # Mirror descent update
        log_a = np.log(a)
        log_a -= lr * v
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            momentum = min(0.85, momentum + 0.002)
        else:
            # Halve step
            log_a2 = np.log(a)
            log_a2 -= lr * 0.25 * v
            log_a2 -= log_a2.max()
            a2 = np.exp(log_a2)
            a2 = a2 / a2.sum()
            obj2 = compute_obj(a2)
            if obj2 < best_obj - 1e-14:
                best_obj = obj2
                best_a = a2.copy()
                a = a2.copy()
                stall = 0
            else:
                stall += 1
                lr *= decay
                if stall > patience:
                    rng = np.random.default_rng(it * 7 + n)
                    noise = 0.1 + 0.1 * min(stall / patience, 2.0)
                    a = best_a * np.exp(rng.normal(0, noise, n))
                    a = np.maximum(a, 1e-15)
                    a = a / a.sum()
                    v = np.zeros(n)
                    momentum = 0.2
                    lr = 5.0 / np.sqrt(n)
                    stall = 0
                    best_a = a.copy()
                    best_obj = compute_obj(a)

        it += 1
        if verbose and it % 500 == 0:
            print(f"    MD it{it}: obj={best_obj:.8f}, lr={lr:.5f}, t={time.monotonic()-start:.1f}s")

    return best_a, best_obj


def coordinate_descent_polish(a, time_budget=30.0):
    """
    Coordinate descent: try moving mass from peak contributors to reduce max conv.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv_self(a)
    k_star = int(np.argmax(conv))

    for rnd in range(100):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv_self(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to conv[k_star]
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                c = a[i] * a[j]
                contribs.append((c, i, j))
        contribs.sort(reverse=True)

        non_contrib = set(range(n))
        for _, i, j in contribs[:20]:
            non_contrib.discard(i)
            if i != j:
                non_contrib.discard(j)

        if not non_contrib:
            break

        nc_list = list(non_contrib)[:]
        rng = np.random.default_rng(rnd)
        rng.shuffle(nc_list)

        improved = False
        for contrib, i, j in contribs[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break

            src = i if a[i] >= a[j] else (j if i != j else i)
            for tgt in nc_list[:10]:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = frac * a[src]
                    if delta < 1e-12 or delta >= a[src]:
                        continue
                    a[src] -= delta
                    a[tgt] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[tgt] -= delta
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
        a = a / a.sum()
    elif mode == 2:
        a = np.exp(-((x - 0.25)**2) / 0.02) + np.exp(-((x - 0.75)**2) / 0.02) + 0.01
        a = a / a.sum()
    elif mode == 3:
        a = (x - 0.5)**2 * 4 + 0.1
        a = a / a.sum()
    elif mode == 4:
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.05
        a = a / a.sum()
    elif mode == 5:
        a = 1 + 0.3 * np.sin(4 * np.pi * x) + 0.2 * np.sin(7 * np.pi * x + 1) + 0.05
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        a = 1 - np.abs(2 * x - 1)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 7:
        a = rng.uniform(0.5, 1.5, n)
        for _ in range(3):
            a = np.convolve(a, np.ones(5) / 5, mode='same')
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 8:
        # Based on Ramanujan sum structure - periodic modulation
        period = max(2, n // 7)
        a = np.ones(n)
        for i in range(n):
            a[i] = 1.0 + 0.4 * np.cos(2 * np.pi * i / period)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 9:
        # Inverse of triangular (peaked at edges)
        a = 1.0 / (np.abs(2 * x - 1) + 0.3)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines fast Frank-Wolfe alternating (with O(n) incremental
    convolution updates), bi-directional FW, direct mirror descent, coordinate
    descent polishing, multi-dimension exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    print(f"\nInitial: n={len(best_seq)}, obj={best_score:.8f}")

    # === Phase 1: Fast FW alternating from parent ===
    print(f"\n=== Phase 1: FW Alternating ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.20, 250.0)
    if p1_budget > 30:
        a1, obj1 = fw_alternate_optimize(best_seq, time_budget=p1_budget, verbose=True)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  FW-Alt: obj={best_score:.8f} **")
        else:
            print(f"  FW-Alt: obj={obj1:.8f} (no improvement)")

    # === Phase 2: Direct mirror descent ===
    print(f"\n=== Phase 2: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.15, 180.0)
    if p2_budget > 20:
        a2, obj2 = mirror_descent_direct(best_seq, time_budget=p2_budget, verbose=True)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  MD: obj={best_score:.8f} **")
        else:
            print(f"  MD: obj={obj2:.8f} (no improvement)")

    # === Phase 3: Bi-directional FW ===
    print(f"\n=== Phase 3: Bi-directional FW ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 250.0)
    if p3_budget > 30:
        a3, obj3 = bi_directional_fw(best_seq, time_budget=p3_budget, verbose=True)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  BiDir: obj={best_score:.8f} **")
        else:
            print(f"  BiDir: obj={obj3:.8f} (no improvement)")

    # === Phase 4: Multi-dimension exploration ===
    print(f"\n=== Phase 4: Multi-Dimension ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best + 50, n_best + 100,
        n_best + 200, n_best + 300,
        500, 600, 700, 900, 1000, 1200, 1400
    ]))
    n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_best]

    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Quick FW optimization
        alloc = min(remaining * 0.12, 80.0)
        if alloc > 20:
            a_opt, obj_opt = fw_alternate_optimize(a_try, time_budget=alloc)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r + 1) % 10
        a_r = diverse_init(n_use, rng, mode=mode)
        # Blend with best
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.12, 70.0)
        if restart_budget > 30:
            a_r, obj_r = fw_alternate_optimize(a_r, time_budget=restart_budget * 0.6)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_r, obj_r = mirror_descent_direct(a_r, time_budget=min(remaining * 0.08, 25.0))

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # FW polish
    if remaining > 40:
        alloc = min(remaining * 0.3, 100.0)
        a_lp, obj_lp = fw_alternate_optimize(best_seq, time_budget=alloc, verbose=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final FW: obj={best_score:.8f} **")

    # MD polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 60.0)
        a_md, obj_md = mirror_descent_direct(best_seq, time_budget=alloc)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Final MD: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 25.0)
        a_cd, obj_cd = coordinate_descent_polish(best_seq, time_budget=alloc)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more short FW
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last, obj_last = fw_alternate_optimize(best_seq, time_budget=min(remaining * 0.5, 30.0))
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last FW: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
