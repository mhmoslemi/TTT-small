```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp

# Helper functions (all top-level)

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj_normalized(a):
    """Objective assuming sum(a) = 1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv)

def compute_obj(a):
    """Full objective."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    return compute_obj_normalized(a_norm)

def smooth_obj_grad(b, tau):
    """
    Compute smoothed objective and gradient w.r.t. b where a = exp(b).
    a is NOT normalized (objective is scale-invariant).
    """
    n = len(b)
    a = np.exp(b)
    s = np.sum(a)
    conv = compute_conv(a)
    
    # Log-sum-exp of tau * conv
    max_c = np.max(conv)
    shifted = tau * (conv - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    lse = max_c + np.log(sum_exp) / tau  # This is (1/tau)*log(sum exp(tau*conv))
    
    # Softmax weights
    p = exp_shifted / sum_exp
    
    # Gradient of LSE w.r.t. a:
    # grad_a[l] = sum_k p[k] * a[k-l] (with a[k-l]=0 if out of range)
    # = cross-correlation
    N = 1
    while N < 2*n - 1:
        N <<= 1
    Fa = rfft(a, N)
    Fp = rfft(p, N)
    grad_a = irfft(np.conj(Fa) * Fp, N)[:n]
    
    # Objective: 2n * lse / s^2
    obj = 2.0 * n * lse / (s * s)
    
    # Gradient w.r.t. a:
    # d(2n*lse/s^2)/da_l = 2n * (grad_a[l]/s^2 - 2*lse*a[l]/s^3)
    grad_obj_a = 2.0 * n * (grad_a / (s*s) - 2.0 * lse * a / (s*s*s))
    
    # Chain rule: a = exp(b), da_l/db_l = a_l
    grad_obj_b = grad_obj_a * a
    
    return obj, grad_obj_b

def lbfsgs_smooth(a_init, tau_list=None, max_iter=100, time_budget=100.0):
    """Run L-BFGS with adaptive tau."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    b_init = np.log(a_init)
    
    if tau_list is None:
        tau_list = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0]
    
    best_a = a_init / np.sum(a_init)
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    
    b = b_init.copy()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget - 10:
            break
        
        def fun(g):
            return smooth_obj_grad(g, tau)
        
        try:
            result = minimize(fun, b, method='L-BFGS-B', jac=True,
                           options={'maxiter': max_iter, 'maxls': 50,
                                   'ftol': 1e-12, 'gtol': 1e-8})
            if result.success or result.fun < best_obj:
                a_new = np.exp(result.x)
                a_new = a_new / np.sum(a_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    b = result.x.copy()
        except Exception as e:
            pass
    
    return best_a, best_obj

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        r = min(n, 2*n - 1 - j)
        A[j:j+r, j] = a[:r]
    return A

def solve_lp(a):

    A = build_conv_matrix(a)

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x, a)
    return np.maximum(a_new, 0)
```

```python
"""
Hybrid optimizer combining smooth L-BFGS (log-sum-exp), LP alternating,
multi-n exploration, and targeted restarts to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_obj_grad(b, tau):
    """
    Smoothed objective and gradient w.r.t. b where a = exp(b).
    Uses log-sum-exp approximation of max for differentiability.
    """
    n = len(b)
    a = np.exp(b)
    s = float(np.sum(a))
    conv = compute_conv(a)

    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    lse = max_c + np.log(sum_exp) / tau

    p = exp_shifted / sum_exp

    # Gradient of LSE w.r.t. a via cross-correlation
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    Fa = rfft(a, N)
    Fp = rfft(p, N)
    grad_a = irfft(np.conj(Fa) * Fp, N)[:n]

    obj = 2.0 * n * lse / (s * s)
    grad_obj_a = 2.0 * n * (grad_a / (s * s) - 2.0 * lse * a / (s * s * s))
    grad_obj_b = grad_obj_a * a

    return obj, grad_obj_b


def lbfsgs_smooth(a_init, tau_list=None, max_iter=120, time_budget=100.0):
    """Run L-BFGS on smoothed objective with exp parametrization."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    b_init = np.log(a_init)

    if tau_list is None:
        tau_list = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0]

    best_a = a_init / float(np.sum(a_init))
    best_obj = compute_obj(best_a)
    start = time.monotonic()

    b = b_init.copy()

    for tau in tau_list:
        if time.monotonic() - start > time_budget - 5:
            break

        def fun(g):
            return smooth_obj_grad(g, tau)

        try:
            result = minimize(fun, b, method='L-BFGS-B', jac=True,
                              options={'maxiter': max_iter, 'maxls': 50,
                                       'ftol': 1e-14, 'gtol': 1e-10,
                                       'max_fun': 5000})
            a_new = np.exp(result.x)
            a_new = a_new / float(np.sum(a_new))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                b = result.x.copy()
        except Exception:
            pass

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A such that A @ g = cross-convolution a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. (a*g)(k) <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except Exception:
                        pass
                    return z_val, float(M.value), dual_y
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except Exception:
        pass

    return None, None, None


def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating with 2-cycle detection, momentum, and adaptive mixing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    prev_a = a.copy()
    prev_prev_a = None
    momentum = a.copy()
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        result = solve_lp(a)
        if result[0] is None:
            break
        g, M_opt, dual = result

        # 2-cycle detection
        if prev_prev_a is not None:
            diff = np.max(np.abs(a - prev_prev_a))
            if diff < 1e-8:
                a = 0.5 * (a + prev_a)
                a = np.maximum(a, 0)
                sa = float(np.sum(a))
                if sa > 1e-10:
                    a = a / sa
                prev_a = a.copy()
                prev_prev_a = None
                result = solve_lp(a)
                if result[0] is None:
                    break
                g, M_opt, dual = result

        # Update momentum
        momentum = 0.5 * momentum + 0.5 * g
        momentum = np.maximum(momentum, 0)
        ms = float(np.sum(momentum))
        if ms > 1e-10:
            momentum = momentum / ms

        # Try multiple mixing strategies
        improved = False
        targets = [g, momentum]
        for target in targets:
            for mix in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3]:
                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    prev_prev_a = prev_a.copy()
                    prev_a = a.copy()
                    a = a_try
                    improved = True
                    stall = 0
                    if verbose and it % 20 == 0:
                        print(f"    LP iter {it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 12:
                # Dual-guided perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                if dual is not None and np.max(dual) > 0:
                    A = build_conv_matrix(a)
                    active = np.where(dual > 0.05 * np.max(dual))[0]
                    if len(active) > 0:
                        col_imp = np.sum(A[active, :], axis=0)
                        pert = -np.sign(col_imp - np.mean(col_imp)) * 0.015
                        pert += rng.normal(0, 0.02, n)
                    else:
                        pert = rng.normal(0, 0.03, n)
                else:
                    pert = rng.normal(0, 0.03, n)

                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                sa = float(np.sum(a))
                if sa > 1e-10:
                    a = a / sa
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = a.copy()
                prev_prev_a = None
                momentum = a.copy()
                stall = 0
                if verbose:
                    print(f"    [Perturbed] obj={best_obj:.8f}")

    return best_a


def coordinate_descent(a, time_budget=30.0):
    """Coordinate descent polish on unsmoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(77)
    stall = 0

    while time.monotonic() - start < time_budget:
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            for frac in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]:
                a_try = a.copy()
                a_try[i] = old_val * frac
                a_try = a_try / float(np.sum(a_try))
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall >= 4:
                break
        else:
            stall = 0

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def pairwise_exchange(a, time_budget=20.0):
    """Targeted pairwise mass exchange."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(314)

    for _ in range(100):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for _ in range(30):
            if time.monotonic() - start > time_budget:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            for frac in [0.03, 0.08, 0.15]:
                delta = frac * min(a[i], a[j])
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0:
                    continue
                a_try = a_try / float(np.sum(a_try))
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines smooth L-BFGS, LP alternating, multi-n exploration,
    coordinate descent, and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_a = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_a = parent_arr / s
                best_score = compute_obj(best_a)
                print(f"Parent (n={parent_arr.size}): obj={best_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_a is None:
        n_init = 800
        t = np.linspace(0, 1, n_init)
        best_a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        best_a = np.maximum(best_a, 0.01)
        best_a = best_a / float(np.sum(best_a))
        best_score = compute_obj(best_a)
        print(f"Initial (n={n_init}): obj={best_score:.8f}")

    a = best_a.copy()
    n_cur = len(a)

    # === Phase 1: L-BFGS smooth (up to 250s) ===
    print(f"\n=== Phase 1: L-BFGS Smooth (n={n_cur}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(250.0, remaining * 0.3)
        a_new, obj_new = lbfsgs_smooth(a, tau_list=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0],
                                       max_iter=150, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            n_cur = len(a)
            print(f"  L-BFGS: obj={best_score:.8f} **")
        else:
            print(f"  L-BFGS: obj={obj_new:.8f}")

    # === Phase 2: LP Alternating (up to 300s) ===
    print(f"\n=== Phase 2: LP Alternating (n={n_cur}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(300.0, remaining * 0.3)
        a_new = lp_alternating(a, time_budget=alloc, verbose=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            n_cur = len(a)
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_new:.8f}")

    # === Phase 3: Multi-n exploration (up to 250s) ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_list = sorted(set([
        max(400, n_cur - 200), max(400, n_cur - 100), n_cur,
        n_cur + 100, n_cur + 200, n_cur + 400, n_cur + 600,
        600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600,
    ]))
    n_list = [n for n in n_list if 400 <= n <= 2000]
    n_list_sorted = sorted(set(n_list), key=lambda x: abs(x - n_cur))

    for n_try in n_list_sorted:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == n_cur:
            continue

        a_try = resample(best_a, n_try)
        ts = float(np.sum(a_try))
        if ts < 1e-10:
            continue
        a_try = a_try / ts

        # LP optimization
        alloc_lp = min(40.0, remaining * 0.15)
        a_opt = lp_alternating(a_try, time_budget=alloc_lp)
        obj_opt = compute_obj(a_opt)

        # L-BFGS refinement
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20 and obj_opt < best_score + 0.01:
            a_lb, _ = lbfsgs_smooth(a_opt, tau_list=[2.0, 5.0, 10.0, 20.0],
                                    max_iter=60, time_budget=min(30.0, remaining2 * 0.4))
            obj_lb = compute_obj(a_lb)
            if obj_lb < obj_opt:
                a_opt = a_lb
                obj_opt = obj_lb

        if obj_opt < best_score:
            best_score = obj_opt
            best_a = a_opt.copy()
            a = a_opt.copy()
            n_cur = len(a)
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 4: L-BFGS from absolute best (up to 150s) ===
    print(f"\n=== Phase 4: L-BFGS Refinement (n={len(best_a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(150.0, remaining * 0.2)
        a_new, _ = lbfsgs_smooth(best_a, tau_list=[1.0, 2.0, 5.0, 10.0, 20.0, 50.0],
                                 max_iter=200, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            print(f"  L-BFGS: obj={best_score:.8f} **")
        else:
            print(f"  L-BFGS: obj={obj_new:.8f}")

    # === Phase 5: LP Alternating on best (up to 150s) ===
    print(f"\n=== Phase 5: LP on Best (n={len(best_a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(150.0, remaining * 0.2)
        a_new = lp_alternating(best_a, time_budget=alloc, verbose=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_new:.8f}")

    # === Phase 6: Coordinate + Pairwise polish (up to 60s) ===
    print(f"\n=== Phase 6: Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(30.0, remaining * 0.15)
        a_new = coordinate_descent(best_a, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_a = a_new.copy()
            print(f"  Coord: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(20.0, remaining * 0.15)
            a_new = pairwise_exchange(best_a, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_a = a_new.copy()
                print(f"  Pairwise: obj={best_score:.8f} **")

    # === Phase 7: Random restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Restarts ===")
        n_best = len(best_a)
        max_restarts = min(6, int(remaining // 35))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            if r % 3 == 0:
                a_r = best_a * np.exp(rng.normal(0, 0.04 + 0.01 * r, n_best))
            elif r % 3 == 1:
                a_r = best_a.copy().astype(np.float64)
                n_blocks = max(5, n_best // 80)
                block_size = n_best // n_blocks
                for _ in range(3):
                    b = int(rng.integers(0, n_blocks))
                    si = b * block_size
                    ei = min(si + block_size, n_best)
                    a_r[si:ei] *= rng.uniform(0.4, 1.6)
            else:
                # Fourier perturbation
                a_r = np.array(best_a, dtype=np.float64)
                A = rfft(a_r, len(a_r))
                for k in range(1, len(A) // 4):
                    A[k] *= (1.0 + rng.normal(0, 0.04)).real
                a_r = irfft(A, len(a_r))
                a_r = np.maximum(np.real(a_r), 0.001)

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            alloc = min(30.0, remaining * 0.4)
            a_opt = lp_alternating(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)

            # Quick L-BFGS on restart result
            remaining2 = deadline - time.monotonic()
            if remaining2 > 15:
                a_lb, _ = lbfsgs_smooth(a_opt, tau_list=[5.0, 10.0, 20.0],
                                        max_iter=50, time_budget=min(15.0, remaining2 * 0.5))
                obj_lb = compute_obj(a_lb)
                if obj_lb < obj_r:
                    a_opt = a_lb
                    obj_r = obj_lb

            if obj_r < best_score:
                best_score = obj_r
                best_a = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 8: Final Polish ===")
        alloc = min(remaining * 0.4, 30.0)
        a_new = lp_alternating(best_a, time_budget=alloc)
        obj_new = compute_obj(a_new)
        if obj_new < best_score:
            best_score = obj_new
            best_a = a_new.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(remaining * 0.4, 15.0)
            a_new, _ = lbfsgs_smooth(best_a, tau_list=[10.0, 20.0, 50.0],
                                     max_iter=80, time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_a = a_new.copy()
                print(f"  Final L-BFGS: obj={best_score:.8f} **")

    # Final validation
    if best_a is None:
        best_a = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
