"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent, LP alternating, smooth gradient descent, and QP polish.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp_full(a):
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

def convex_line_search(a, g, n_evals=12):
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)
    
    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)
    
    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))
    
    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    a_new = a_new / s
    return t_opt, a_new

def lp_alternating(a, time_budget=100.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, _ = solve_lp_full(a)
        if g_lp is None:
            break
        
        t_opt, a_new = convex_line_search(a, g_lp, n_evals=12)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose and it % 10 == 0:
                print(f"          LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}")
        else:
            stall_count += 1
            if stall_count > 30:
                break
    
    return best_a

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines exponential tilting (mirror descent), LP alternating, SOR peak reduction,
structured initializations, and multi-n exploration to escape local minima.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp(a):
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def convex_line_search(a, g, n_evals=15):
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)
    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1.0 - t) ** 2 * conv_aa + 2.0 * t * (1.0 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t_opt = (lo + hi) / 2.0
    a_new = (1.0 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    a_new = a_new / s
    return t_opt, a_new

        k_star = int(np.argmax(conv))
        # Gradient of (a*a)_{k*} w.r.t. a[i] = 2*a[k*-i] (if valid)
        grad = np.zeros(n)

        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total
        # Gradient: grad[i] = sum_k weights[k] * 2 * a[k-i]
        grad = np.zeros(n)
        for i in range(n):
            lo_k = i
            hi_k = i + n
            if lo_k < len(weights) and hi_k > 0:
                segment = weights[lo_k:min(hi_k, len(weights))]
                a_segment = a[:len(segment)]
                grad[i] = 2.0 * np.dot(segment, a_segment)
        # Center gradient
        grad_mean = np.mean(grad)
        grad_centered = grad - grad_mean
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if it % 20 == 1:
                lr = min(lr * 1.02, 0.05)
            if verbose and it % 200 == 0:
                print(f"          SMG it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.5f}, time={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                if lr < 1e-5:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
        if it > 50:
            T = max(T_min, T_init * (0.997 ** it))
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_init(n, kind='hamming'):
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'hamming':
        w = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / max(1, n - 1))
        w = np.maximum(w, 0)
        return w / np.sum(w)
    elif kind == 'blackman':
        w = 0.42 - 0.5 * np.cos(2 * np.pi * np.arange(n) / max(1, n - 1)) + 0.08 * np.cos(4 * np.pi * np.arange(n) / max(1, n - 1))
        w = np.maximum(w, 0)
        return w / (np.sum(w) + 1e-30)
    elif kind == 'raised_cosine':
        w = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / max(1, n - 1)))
        return w / np.sum(w)
    elif kind == 'triangle':
        w = 1 - np.abs(2 * np.arange(n) / max(1, n - 1) - 1)
        w = np.maximum(w, 0)
        return w / np.sum(w)
    elif kind == 'chirp':
        w = 1 + 0.3 * np.cos(2 * np.pi * np.arange(n) ** 2 / (2 * n))
        w = np.maximum(w, 0)
        return w / np.sum(w)
    elif kind == 'random':
        rng = np.random.default_rng(42)
        w = rng.random(n)
        return w / np.sum(w)
    elif kind == 'bipolar_mapped':
        # Create a pattern similar to Barker but non-negative
        rng = np.random.default_rng(123)
        w = rng.random(n) * 0.5 + 0.5
        # Add some structure
        w += 0.2 * np.cos(2 * np.pi * np.arange(n) / n * 7)
        w = np.maximum(w, 0)
        return w / np.sum(w)
    else:
        return np.ones(n) / n

                    a_try[i] -= d
