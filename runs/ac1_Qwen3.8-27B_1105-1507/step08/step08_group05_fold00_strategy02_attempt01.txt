            top_k_indices = np.argsort(conv_vals)[::-1][:10]  # 15 points total, but we'll focus on top 10

            grad_vector_new = np.zeros(n, dtype=np.float64)

            for k in top_k_indices:
                # For each active peak in the convolution, compute its gradient contribution
                grad_vector_new += conv_grad_at_k(best_seq, k)

            grad_vector_new = grad_vector_new / len(top_k_indices)  # average over distinct peaks

            # Update the sequence using momentum (moving towards the gradient direction, but normalized)
            # We scale by alpha_step_size, but also add a damping factor
            grad_magnitude = np.linalg.norm(grad_vector_new)
            if grad_magnitude < 0.01:
                # No significant change
                continue
                
            grad_vector_new = grad_vector_new / grad_magnitude  # normalize direction
            grad_vector_new = alpha_step_size * grad_vector_new

            # We'll update the sequence by moving in the negative gradient
            # But to ensure it stays in the simplex, we project
            try:
                # Compute step as: current_seq - grad_step
                candidate_seq = best_seq - grad_vector_new
                
                # Clip negative coefficients and project to simplex
                s_val = np.sum(candidate_seq)
                if s_val < 0.01:
                    # Need to scale up candidate_seq
                    candidate_seq = candidate_seq * (0.01 / s_val)

                # If the new sequence does not sum to at least 0.01, skip
                s_check = np.sum(candidate_seq)
                if s_check >= 0.01:
                    # Update the candidate only if positive
                    candidate_seq = np.maximum(candidate_seq, 0)
                    s_check_val = np.sum(candidate_seq)
                    # Re-normalize again to avoid small sum issues
                    if s_check_val >= 0.01:
                        # Only consider candidates that are in bounds and positive
                        score_new = evaluate_sequence(candidate_seq)

                        # We check if an improvement was made: whether the new score is lower than the current best or above a threshold?
                        if score_new < best_score_val:
                            best_score_val = score_new
                            best_seq = candidate_seq.copy()
                            grad_vector = grad_vector_new.copy()
                            # Reset the step size occasionally
                            # But don't print for every improvement, print only every 100.
                            # We can continue without resetting step size.

                    # Else: continue with original candidate

            except Exception as e:
                # Optimization not converging, also consider other methods?
                continue

            # Also, we might want to update the gradient vector to be the new average gradient (which could be used to update further)
            # But we are already updating in the direction of the gradient.

        except Exception as e:
            # Sometimes the compute_conv can fail, reexamine and fallback
            if np.any(candidate_seq < 0) or (i % 100 == 0 and np.linalg.norm(grad_vector_new) < 0.1):
                # Fallback to one-dimension update
                try:
                    # Use the previous gradient descent to update without repeating compute_conv
                    grad_vector_new = alpha_step_size * grad_vector
                    candidate_seq = best_seq - grad_vector_new
                    s_val = np.sum(candidate_seq)
                    if s_val >= 0.01:
                        candidate_seq = np.maximum(candidate_seq, 0)
                        candidate_seq = candidate_seq / s_val   # normalize so sum is 1
                        score_new = evaluate_sequence(candidate_seq)
                        if score_new < best_score_val:
                            best_score_val = score_new
                            best_seq = candidate_seq.copy()
                            print(f"Phase2: improvement after fallback at iteration {i}")
                        grad_vector = grad_vector_new.copy()
                    else:
                        print(f"Phase2: candidate_seq sum too low at {i}")
                except:
                    continue
            else:
                continue

        # Also, include the step size decay mechanism from the previous program
        # But we'll rely on time_budget to trigger transitions between phases

    # Phase 3: Exploring different lengths to reduce the objective
    if time.time() - time_start_phase2 >= time_budget:
        phase_budget_reached = True
    else:
        phase_budget_reached = False

    # Define the scenario: 
    # But note, the consistency of simplex projection is important because the objective is invariant to scalar multiples.

    # Submit the best candidate found so far
    return best_seq, best_score_val
