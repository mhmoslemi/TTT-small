def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                g_lp, M_lp = solve_lp(a, weights=weights)
            else:
                g_lp, M_lp = solve_lp(a)
        elif stall > 10 and (it % 2 == 0):
            # Mildly weighted LP
            conv_a = compute_conv(a)
            max_c = np.max(conv_a)
            if max_c > 1e-15:
                weights = 0.7 + 0.3 * (conv_a / max_c)
                g_lp, M_lp = solve_lp(a, weights=weights)
            else:
                g_lp, M_lp = solve_lp(a)
        else:
            g_lp, M_lp = solve_lp(a)

        if g_lp is None:
            break

        # Line search
        a_new, obj_new = line_search(a, g_lp, best_obj)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: obj={best_obj:.8f}, M={M_lp:.6f}, el={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 35:
                noise_scale = min(0.4, 0.03 + 0.04 * (stall // 35))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s2 = float(np.sum(a))
                if s2 > 1e-10:
                    a = a / s2
                best_obj = compute_obj(a)
                best_a = a.copy()
                stall = 0

    return best_a


def subgradient_polish(a, time_budget=60.0, verbose=False):
    """
    Mirror descent (multiplicative updates) with adaptive learning rate.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.2 / np.sqrt(n)
    min_lr = 1e-6
    momentum = 0.4
    v = np.zeros(n)
    stall = 0

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = np.max(conv)
        if max_c < 1e-15:
            break

        # Multi-temperature softmax weighting
        T_values = [3.0, 8.0, 20.0]
        all_w = np.zeros(len(conv))
        for T in T_values:
            w = np.exp((conv - max_c) / (T * max_c + 1e-15))
            w /= w.sum()
            all_w += w
        all_w /= len(T_values)

        # Use top 10% for efficiency
        threshold = np.percentile(all_w, 90)
        active = all_w > threshold
        if not active.any():
            active[np.argmax(conv)] = True
        active_w = all_w[active]
        active_w /= active_w.sum()
        active_idx = np.where(active)[0]

        g = np.zeros(n)
        for w, k in zip(active_w, active_idx):
            if w > 1e-10:
                k = int(k)
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]

        v = momentum * v + g
        log_a = np.log(a)
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = max(min_lr, lr * 0.9997)
            momentum = min(0.7, momentum + 0.001)
        else:
            stall += 1
            if stall > 80:
                rng_esc = np.random.default_rng(it * 31 + n)
                a = best_a * np.exp(rng_esc.normal(0, 0.12, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                lr = max(min_lr, 1.0 / np.sqrt(n))
                momentum = 0.4
                stall = 0
            else:
                lr = max(min_lr, lr * 0.985)

        it += 1
        if verbose and it % 200 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, lr={lr:.6f}, el={time.monotonic()-start:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    a = None
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.arange(n)
        center = n / 2.0
        sigma = n / 4.0
        a = np.exp(-(x - center) ** 2 / (2 * sigma ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.arange(n)
        a = np.exp(-((x - 0.25 * n) ** 2) / (2 * (0.08 * n) ** 2)) + \
            np.exp(-((x - 0.75 * n) ** 2) / (2 * (0.08 * n) ** 2))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1 - x ** 2, 0.01))
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, 1, n)
        freq = rng.integers(2, 7)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.25 * np.sin(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-5.0 * (x - 0.5)))
        a = a / a.sum()
    elif mode == 7:
        x = np.arange(1, n + 1)
        a = 1.0 / (x ** 0.3 + 1.0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid optimization combining enhanced LP alternating,
    subgradient polishing, multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Enhanced LP Alternating (main workhorse, ~40% time) ===
    print(f"\n=== Phase 1: LP Alternating (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.40, 500.0)
    if p1_budget > 30:
        a_lp = lp_alternating(best_seq, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: obj={best_score:.8f} **")
        else:
            print(f"  LP: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Subgradient polish (~12% time) ===
    print(f"\n=== Phase 2: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.12, 100.0)
    if p2_budget > 15:
        a_sg = subgradient_polish(best_seq, time_budget=p2_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SG: obj={best_score:.8f} **")
        else:
            print(f"  SG: obj={obj_sg:.8f} (no improvement)")

    # === Phase 3: LP with perturbation (~15% time) ===
    print(f"\n=== Phase 3: LP with Perturbation ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 20:
        a_pert = best_seq * np.exp(rng.normal(0, 0.025, len(best_seq)))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_lp2 = lp_alternating(a_pert, time_budget=p3_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  LP pert: obj={best_score:.8f} **")
        else:
            print(f"  LP pert: obj={obj_lp2:.8f} (no improvement)")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_curr = len(best_seq)

    n_options = sorted(set([
        max(100, n_curr - 200),
        max(100, n_curr - 100),
        max(100, n_curr - 50),
        n_curr + 50,
        n_curr + 100,
        n_curr + 200,
        n_curr + 400,
        n_curr + 800,
        400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000, 2500
    ]))
    n_options = [n for n in n_options if 100 <= n <= 3000]
    n_options.sort(key=lambda x: abs(x - n_curr))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25 or n_tried >= 10:
            break
        if n_try == n_curr:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        alloc = min(remaining * 0.06, 35.0)
        if alloc > 10:
            a_try = lp_alternating(a_try, time_budget=alloc)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_curr = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.25 + 0.08 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.1, 45.0)
        a_opt = lp_alternating(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 25:
        lp_alloc = min(remaining * 0.4, 80.0)
        a_lp = lp_alternating(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        sg_alloc = min(remaining * 0.3, 30.0)
        a_sg = subgradient_polish(best_seq, time_budget=sg_alloc)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 25.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
