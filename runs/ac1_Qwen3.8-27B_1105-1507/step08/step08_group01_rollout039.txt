```python
def smooth_max_grad(a, T):
    n = len(a)
    conv = compute_conv(a)  # length 2n-1
    # Softmax weights
    shifted = conv / T
    shifted -= np.max(shifted)  # for numerical stability
    exp_v = np.exp(shifted)
    p = exp_v / np.sum(exp_v)
    # Gradient: grad[j] = sum_k p[k] * d(conv[k])/d(a[j])
    # d(conv[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n, j-k != k-j)
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum over k where 0<=k-j<n, i.e., j <= k <= j+n-1
        k_lo = max(0, j)
        k_hi = min(2*n-2, j+n-1)
        # a[k-j] for k in [k_lo, k_hi]: a[0], a[1], ..., a[min(n-1, k_hi-j)]
        idx = np.arange(k_lo - j, k_hi - j + 1)  # indices into a
        grad[j] += np.dot(p[k_lo:k_hi+1], a[idx])
        # Second term: sum over k where 0<=j-k<n AND j-k != k-j (i.e., k != 2j)
        # j-k in [0, n-1] means k in [j-n+1, j], clipped to [0, 2n-2]
        k_lo2 = max(0, j-n+1)
        k_hi2 = min(2*n-2, j)
        # a[j-k] for k in [k_lo2, k_hi2]: a[n-1], a[n-2], ..., a[n - (k_hi2 - j + 1)]
        # j-k ranges from j-k_lo2 to j-k_hi2, i.e., from min(n-1, j) to max(0, j-n+1)
        # Actually: when k = k_lo2, j-k = j - k_lo2; when k = k_hi2, j-k = j - k_hi2
        idx2 = j - np.arange(k_lo2, k_hi2+1)  # these are the indices into a
        # But we need to exclude k = 2j (where j-k = k-j)
        mask = np.arange(k_lo2, k_hi2+1) != 2*j
        if np.any(mask):
            grad[j] += np.dot(p[k_lo2:k_hi2+1][mask], a[idx2[mask]])
    return grad
```

```python
"""
Optimized minimization of 2n*max(conv(a,a))/(sum(a))^2 using:
- Efficient O(n log n) smooth-max gradient via FFT-based correlate
- Adam optimizer with simplex projection
- Adaptive temperature annealing
- LP-based direction finding (active set)
- Multi-restart with diverse initializations
- Coordinate descent polish
- Multi-n exploration

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    na = len(a)
    nb = len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:na + nb - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1) / rho
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_grad(a, T):
    """
    Efficient gradient of smooth max of conv(a,a).
    Uses the identity: d(conv[k])/d(a[j]) = 2*a[k-j] (if 0<=k-j<n)
    So grad[j] = 2 * sum_d p[j+d] * a[d] = 2 * correlate(p, a, 'valid')[j]
    Returns (grad, p, conv) where p is softmax weight vector.
    """
    n = len(a)
    conv = compute_conv(a)
    shifted = conv / T
    shifted -= np.max(shifted)
    exp_v = np.exp(shifted)
    p = exp_v / np.sum(exp_v)
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad, p, conv


def smooth_max_value(a, T):
    """Compute smooth max approximation of max(conv(a,a))."""
    conv = compute_conv(a)
    shifted = conv / T
    shifted -= np.max(shifted)
    return T * (float(np.max(shifted)) + float(np.log(np.sum(np.exp(shifted)))))


def optimize_smooth_max(a, time_budget=300.0, T_init=0.0005, T_min=0.00001,
                        lr_init=0.05, use_adam=True, verbose=True, max_iters=200000):
    """
    Adam-based gradient descent on simplex with smooth max surrogate.
    Efficient O(n log n) per iteration.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = lr_init
    stall = 0
    best_stall_lr = lr

    if use_adam:
        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        t_adam = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad, p, conv = smooth_max_grad(a, T)
        grad_centered = grad - np.mean(grad)

        if use_adam:
            t_adam += 1
            m = beta1 * m + (1.0 - beta1) * grad_centered
            v = beta2 * v + (1.0 - beta2) * grad_centered ** 2
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
        else:
            step = lr * grad_centered

        a_new = a - step
        a_new = simplex_project(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Occasionally increase lr if making steady progress
            if stall == 0 and it % 200 == 0 and lr < lr_init * 2:
                lr = min(lr * 1.02, lr_init * 2)
        else:
            stall += 1
            if stall == 30:
                lr *= 0.8
                a = best_a.copy()
                if use_adam:
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t_adam = 0
                stall = 0
            elif stall > 300:
                # Perturb and restart
                rng = np.random.default_rng(it + n)
                a_pert = best_a * np.exp(rng.normal(0, 0.002, n))
                a_pert = simplex_project(a_pert)
                a = a_pert
                lr = lr_init * 0.5
                if use_adam:
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t_adam = 0
                T = max(T_min, T * 1.5)
                stall = 0

        # Anneal temperature
        if it > 100:
            T = max(T_min, T_init * (0.9997 ** it))

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"      SM-GD it{it}: obj={best_obj:.9f}, T={T:.7f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[::-1][:n]
    # Fix: A[k, j] = a[j] if k >= j and k-j < n, and we want conv(a,g)[k] = sum_j a[k-j]*g[j]
    # Actually let me rebuild properly
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(max(0, j - n + 1), min(2 * n - 1, j + n)):
            # conv(a,g)[k] includes a[k-j]*g[j] if 0 <= k-j < n
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    return A


def build_conv_matrix_eff(a):
    """Efficient build of convolution matrix using Toeplitz structure."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # Column j: A[k, j] = a[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        lo = j
        hi = min(m, j + n)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp_active(a, active_indices, time_budget=10.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for k in active, sum(g)=1, g>=0."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix_eff(a)
    A_sub = A[active_indices, :]

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((len(active_indices), n + 1))
    A_ub[:, :n] = A_sub
    A_ub[:, n] = -1.0
    b_ub = np.zeros(len(active_indices))

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_norm = g_val / s
            # Verify against all constraints
            conv_check = A @ g_norm
            true_M = float(np.max(conv_check))
            return g_norm, true_M
    return None, None


def solve_lp_full(a, time_budget=20.0):
    """Solve LP with all constraints."""
    n = len(a)
    active = np.arange(2 * n - 1)
    return solve_lp_active(a, active, time_budget)


def lp_blend_step(a, time_budget=15.0):
    """
    One LP-based step: solve LP, find optimal blend via line search.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv = compute_conv(a_norm)
    max_c = np.max(conv)

    # Get active set (within 5% of max)
    threshold = max_c * 0.95
    active = np.where(conv >= threshold)[0]

    # Solve LP
    g_lp, M_lp = solve_lp_active(a_norm, active, time_budget=time_budget)
    if g_lp is None:
        g_lp, M_lp = solve_lp_full(a_norm, time_budget=time_budget)
    if g_lp is None:
        return None, None

    # Convex line search for optimal t
    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_lp)
    conv_gg = compute_conv(g_lp)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(15):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj_at_t(m1) < obj_at_t(m2):
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_lp
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return None, None
    a_new = a_new / s
    return a_new, M_lp

    """LP alternating with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

def coordinate_descent(a, time_budget=100.0, delta_init=1e-4, verbose=True):
    """Coordinate descent with adaptive step sizes."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    delta = delta_init
    pass_num = 0

    while time.monotonic() - start < time_budget:
        pass_num += 1
        improved = False
        rng = np.random.default_rng(pass_num * 1337 + n)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            # Try positive perturbation
            for d in [delta, delta * 0.1, delta * 5]:
                a_try = a.copy()
                a_try[i] += d
                a_try = simplex_project(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                # Also try negative
                for d in [delta, delta * 0.1]:
                    if a[i] < d * 0.3:
                        continue
                    a_try = a.copy()
                    a_try[i] -= d
                    a_try = simplex_project(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                continue

            # Try negative perturbation
            for d in [delta, delta * 0.5]:
                if a[i] < d * 0.3:
                    continue
                a_try = a.copy()
                a_try[i] -= d
                a_try = simplex_project(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if verbose and pass_num % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"      CD pass {pass_num}: obj={best_obj:.9f}, delta={delta:.2e}, elapsed={elapsed:.1f}s")

        if not improved:
            delta *= 0.5
            if delta < 1e-8:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def multi_temperature_optimize(a, time_budget=200.0):
    """Run smooth max GD with multiple temperature settings and keep best."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Multiple temperature settings from hot to cold
    T_settings = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    remaining = time_budget - (time.monotonic() - start)
    per_setting = remaining / len(T_settings)

    for T_init in T_settings:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        alloc = min(per_setting, remaining)

        a_cur = best_a.copy()

                                    T_min=T_init * 0.01, lr_init=0.03,

        obj_opt = compute_obj(a_opt)
        if obj_opt < best_obj - 1e-13:
            best_obj = obj_opt
            best_a = a_opt.copy()
            print(f"    T={T_init:.5f}: obj={best_obj:.9f}")

    return best_a


def diverse_initialization(n, rng):
    """Generate diverse random initializations."""
    type_id = rng.integers(0, 5)
    if type_id == 0:
        # Pure random
        a = rng.random(n)
    elif type_id == 1:
        # Exponential decay from center
        center = n / 2.0
        a = np.exp(-((np.arange(n) - center) ** 2) / (2 * (n * 0.2) ** 2))
    elif type_id == 2:
        # Alternating pattern
        a = np.ones(n)
        a[::2] = 1.5
        a[1::2] = 0.7
        a += rng.normal(0, 0.1, n)
    elif type_id == 3:
        # Two peaks
        peak1 = rng.integers(0, n // 3)
        peak2 = rng.integers(2 * n // 3, n)
        a = np.zeros(n)
        a[peak1] = 1.0
        a[peak2] = 1.0
        a = np.convolve(a, np.ones(max(2, n // 10)) / max(2, n // 10), mode='same')
        a += rng.normal(0, 0.05, n)
    else:
        # Low-frequency cos
        freq = rng.uniform(1, 3)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * np.arange(n) / n)
        a += rng.normal(0, 0.1, n)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    return a / np.sum(a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining efficient smooth-max GD, LP, CD, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_init = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={len(best_seq)}): obj={best_score:.9f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.9f}")

    a = best_seq.copy()
    n = len(a)

    # === Phase 1: Multi-temperature smooth max GD (main workhorse) ===
    print(f"\n=== Phase 1: Multi-T Smooth Max GD (n={n}) ===")
    remaining = deadline - time.monotonic()
    phase1_budget = min(400.0, remaining * 0.35)
    print(f"  Budget: {phase1_budget:.0f}s")

    a_opt = multi_temperature_optimize(a, time_budget=phase1_budget)
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_score - 1e-13:
        best_score = obj_opt
        best_seq = a_opt.copy()
        a = a_opt.copy()
        print(f"  Phase 1 best: obj={best_score:.9f}")
    else:
        print(f"  Phase 1: no improvement (obj={obj_opt:.9f})")

    # === Phase 2: LP-based alternating ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase2_budget = min(200.0, remaining * 0.2)
        a_opt = lp_alternating(a, time_budget=phase2_budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score - 1e-13:
            best_score = obj_opt
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  Phase 2 best: obj={best_score:.9f}")
        else:
            print(f"  Phase 2: no improvement (obj={obj_opt:.9f})")

    # === Phase 3: More smooth max GD (cold) ===
    print(f"\n=== Phase 3: Cold Smooth Max GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        phase3_budget = min(200.0, remaining * 0.25)
        a_opt = optimize_smooth_max(a, time_budget=phase3_budget, T_init=0.00005,
                                    T_min=0.000005, lr_init=0.01, use_adam=True,
                                    verbose=True, max_iters=500000)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score - 1e-13:
            best_score = obj_opt
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  Phase 3 best: obj={best_score:.9f}")
        else:
            print(f"  Phase 3: no improvement (obj={obj_opt:.9f})")

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        phase4_budget = min(150.0, remaining * 0.25)
        a_opt = coordinate_descent(a, time_budget=phase4_budget, delta_init=5e-5, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score - 1e-13:
            best_score = obj_opt
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  Phase 4 best: obj={best_score:.9f}")
        else:
            print(f"  Phase 4: no improvement (obj={obj_opt:.9f})")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        600, 700, 750, 800, 850, 900, 950, 1000, 1050, 1100, 1200, 1300,
        n_best - 50, n_best - 25, n_best + 25, n_best + 50
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1400]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization
        alloc = min(remaining * 0.1, 40.0)
        a_opt = optimize_smooth_max(a_try, time_budget=alloc, T_init=0.0003,
                                    T_min=0.00002, lr_init=0.02, use_adam=True,
                                    verbose=False, max_iters=100000)
        obj_opt = compute_obj(a_opt)

        # If promising, do more
        remaining = deadline - time.monotonic()
        if remaining > 30 and obj_opt < best_score + 0.003:
            alloc2 = min(remaining * 0.1, 30.0)
            a_opt2 = optimize_smooth_max(a_opt, time_budget=alloc2, T_init=0.00005,
                                         T_min=0.00001, lr_init=0.005, use_adam=True,
                                         verbose=False, max_iters=100000)
            obj_opt2 = compute_obj(a_opt2)
            if obj_opt2 < obj_opt:
                a_opt = a_opt2
                obj_opt = obj_opt2

        if obj_opt < best_score - 1e-13:
            best_score = obj_opt
            best_seq = a_opt.copy()
            a = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.9f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.9f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_diverse = min(4, int(remaining / 50))
        for d in range(n_diverse):
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            alloc = min(remaining * 0.15, 60.0)

            # Create diverse perturbation of best
            scale = 0.01 + 0.01 * d
            a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
            a_pert = simplex_project(a_pert)

            a_opt = optimize_smooth_max(a_pert, time_budget=alloc, T_init=0.0005,
                                        T_min=0.00002, lr_init=0.03, use_adam=True,
                                        verbose=False, max_iters=100000)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score - 1e-13:
                best_score = obj_opt
                best_seq = a_opt.copy()
                a = a_opt.copy()
                print(f"  Restart {d} (scale={scale:.3f}): obj={best_score:.9f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Very cold smooth max
        alloc = min(60.0, remaining * 0.4)
        a_opt = optimize_smooth_max(best_seq, time_budget=alloc, T_init=0.00002,
                                    T_min=0.000002, lr_init=0.003, use_adam=True,
                                    verbose=True, max_iters=500000)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score - 1e-13:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final SM: obj={best_score:.9f}")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(50.0, remaining * 0.4)
            a_opt = coordinate_descent(best_seq, time_budget=alloc, delta_init=1e-5, verbose=True)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score - 1e-13:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Final CD: obj={best_score:.9f}")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(40.0, remaining * 0.5)
            a_opt = lp_alternating(best_seq, time_budget=alloc, verbose=False)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score - 1e-13:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Final LP: obj={best_score:.9f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.9f}, time={elapsed:.1f}s ===")

    return result
```
