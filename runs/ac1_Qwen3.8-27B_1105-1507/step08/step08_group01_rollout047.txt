"""
Optimized step function coefficient search using fast smooth-max gradient descent,
multiplicative updates, palindromic symmetry exploitation, and multi-start exploration.
Replaces slow LP solves with O(n log n) FFT-based gradient computations for maximum
iteration count within the time budget.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_cross_conv(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_weights(conv, T):
    """Compute softmax weights on convolution coefficients."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_vals / total


def compute_gradient(weights, a):
    """Compute gradient of max_k B_k w.r.t. a using FFT cross-correlation.
    grad_p = 2 * sum_j weights[p+j] * a_j
    This is a valid cross-correlation of weights (len 2n-1) and a (len n).
    """
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    fw = rfft(weights, size)
    fa = rfft(a, size)
    # cross-correlation: (weights * reversed_a) in frequency domain
    # Actually: grad_p = sum_j w_{p+j} * a_j for p = 0..n-1
    # This is the "valid" convolution of w and a with specific alignment
    # ufunc: grad = irfft(fw * fa_conj) won't work directly
    # Let's use: grad_p = sum_j w[p+j]*a[j], which is cross-correlation
    # In FFT terms: correlate(w, a) = irfft(conj(rfft(w)) * rfft(a)) 
    # But we want a specific slice. Let me just do it directly with a trick.
    
    # Alternative: pad a to length 2n-1 and do element-wise
    # Actually the simplest: for each p, grad_p = sum_j w[p+j] * a[j]
    # This is (w * a_reversed) convolved, taking the right part
    # Let me use the direct FFT approach:
    # Define b of length n: b[j] = a[j]
    # grad_p = sum_{j=0}^{n-1} w[p+j] * b[j]
    # = (w conv b)_p where conv is cross-correlation
    # = irfft(conj(rfft(w, size)) * rfft(b, size)) ... no
    
    # Simplest correct approach: 
    # grad = np.zeros(n)
    # for p in range(n):
    #     grad[p] = np.dot(weights[p:p+n], a)
    # But this is O(n^2). Use FFT:
    # Let c = irfft(fw * conj(fa)) which gives correlation
    # Actually: (w ⋆ a)[k] = sum_j w[j] * a[j+k] (cross-correlation)
    # We want: grad[p] = sum_j w[p+j] * a[j] = sum_j a[j] * w[p+j]
    # = (a ⋆ w)[p] where (a ⋆ w)[p] = sum_j a[j] * w[j+p]
    # Standard cross-correlation: (x ⋆ y)[n] = sum_k x[k] * conj(y[k+n])
    # For real: (a ⋆ w)[p] = sum_k a[k] * w[k+p]
    # So grad = (a ⋆ w)[:n] -- the first n elements of cross-correlation
    
    # Using FFT: (a ⋆ w) = irfft(conj(rfft(a, size)) * rfft(w, size))
    fa_full = rfft(a, size)
    fw_full = rfft(weights, size)
    corr = irfft(np.conj(fa_full) * fw_full, size)
    return 2.0 * corr[:n]


def smooth_max_gradient_step(a, T, lr, momentum_vec, momentum_alpha):
    """One step of smooth-max mirror descent. Returns (new_a, new_momentum, obj)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    weights = smooth_max_weights(conv, T)
    grad = compute_gradient(weights, a)
    
    # Center the gradient (remove mean for constraint handling)
    grad_mean = np.mean(grad)
    grad_centered = grad - grad_mean
    
    # Momentum update
    momentum_vec = momentum_alpha * momentum_vec - lr * grad_centered
    
    # Mirror descent (multiplicative) update
    log_a = np.log(np.maximum(a, 1e-300)) + momentum_vec
    log_a -= np.max(log_a)  # numerical stability
    a_new = np.exp(log_a)
    s = np.sum(a_new)
    a_new /= s
    
    obj = 2.0 * n * max_c  # using old conv for speed, will verify after
    return a_new, momentum_vec

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Cosine window
    x = np.linspace(0, np.pi, n)
    c = np.cos(x) ** 2
    inits.append(c / np.sum(c))
    # Sine window (bump in middle)
    s = np.sin(x) ** 2
    inits.append(s / np.sum(s))
    # Linear ramp up then down (triangle)
    t = np.linspace(0, 1, n)
    tri = 1.0 - np.abs(2.0 * t - 1.0)
    inits.append(tri / np.sum(tri))
    # U-shaped (large at edges)
    u = np.abs(2.0 * t - 1.0)
    inits.append(u / np.sum(u))
    # Random
    r = rng.random(n)
    inits.append(r / np.sum(r))
    # Two bumps
    bump1 = np.exp(-((t - 0.25) ** 2) / 0.02)
    bump2 = np.exp(-((t - 0.75) ** 2) / 0.02)
    two_bumps = bump1 + bump2
    inits.append(two_bumps / np.sum(two_bumps))
    # Exponential decay from edges
    e1 = np.exp(-t * 3.0)
    e2 = np.exp(-(1 - t) * 3.0)
    e_shape = e1 + e2
    inits.append(e_shape / np.sum(e_shape))
    # Gaussian centered
    g = np.exp(-((t - 0.5) ** 2) / 0.1)
    inits.append(g / np.sum(g))
    return inits

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={len(parent)}): obj={best_score:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")
    
    n_best = len(best_seq)
    
    # Phase 1: Intensive gradient descent on parent
    print(f"\n=== Phase 1: Gradient Descent (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(200.0, remaining * 0.25)
    a_opt = optimize_gradient_descent(best_seq, time_budget=alloc, T_init=0.0004, T_min=0.00001,
                                       lr_init=0.004, momentum_init=0.82, verbose=True)
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
    print(f"  GD1 done: obj={best_score:.8f}")
    
    # Phase 2: Multiplicative weight update
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 2: Multiplicative Weights (n={len(best_seq)}) ===")
        alloc = min(120.0, remaining * 0.15)
        a_opt = optimize_multiplicative(best_seq, time_budget=alloc, T_init=0.0008, T_min=0.00002,
                                          lr_init=0.006, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
        print(f"  MW done: obj={best_score:.8f}")
    
    # Phase 3: Palindromic optimization
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: Palindromic GD (n={len(best_seq)}) ===")
        alloc = min(120.0, remaining * 0.15)
        a_pali = optimize_palindromic(best_seq, time_budget=alloc, T_init=0.0004, T_min=0.00001,
                                       lr_init=0.004, verbose=True)
        obj_pali = compute_obj(a_pali)
        if obj_pali < best_score:
            best_score = obj_pali
            best_seq = a_pali.copy()
            print(f"  ** Palindromic improved to {best_score:.8f} **")
        else:
            print(f"  Palindromic: obj={obj_pali:.8f} (no improvement)")
    
    # Phase 4: Multi-start with diverse initializations
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 4: Multi-Start Exploration ===")
        n_options = sorted(set([
            n_best, n_best - 50, n_best + 50, n_best - 100, n_best + 100,
            700, 750, 850, 900, 950, 1000, 1100, 1200
        ]))
        n_options = [n for n in n_options if 500 <= n <= 1400]
        
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            
            inits = generate_diverse_inits(n_try, rng)
            # If we have a good best_seq, also include resampled version
            if n_try != n_best:
                inits.append(resample_sequence(best_seq, n_try) / np.sum(resample_sequence(best_seq, n_try)))
            
            best_for_n = None
            best_obj_for_n = np.inf
            
            per_init_budget = min(15.0, remaining / (len(inits) + 1) * 0.7)
            if per_init_budget < 5.0:
                # Too little time, just try the best init
                inits = inits[:2] if len(inits) > 2 else inits
            
            for init_idx, init in enumerate(inits):
                remaining = deadline - time.monotonic()
                if remaining < 10:
                    break
                if per_init_budget < 3.0:
                    per_init_budget = 3.0
                
                # Quick gradient descent
                a_try = optimize_gradient_descent(init, time_budget=per_init_budget,
                                                   T_init=0.0005, T_min=0.00002,
                                                   lr_init=0.005, momentum_init=0.8)
                obj_try = compute_obj(a_try)
                
                if obj_try < best_obj_for_n:
                    best_obj_for_n = obj_try
                    best_for_n = a_try.copy()
                
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    n_best = n_try
                    print(f"    n={n_try}, init{init_idx}: obj={best_score:.8f} **NEW BEST**")
            
            if best_for_n is not None and best_obj_for_n > best_score:
                print(f"    n={n_try}: best={best_obj_for_n:.8f}")
    
    # Phase 5: Refine the best with multiple gradient methods
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 5: Intensive Refinement (n={len(best_seq)}) ===")
        
        # GD with different parameters
        alloc = min(100.0, remaining * 0.25)
        a_opt = optimize_gradient_descent(best_seq, time_budget=alloc, T_init=0.0002, T_min=0.000005,
                                           lr_init=0.003, momentum_init=0.88, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
        print(f"  GD2 done: obj={best_score:.8f}")
        
        # Multiplicative with different T
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc = min(80.0, remaining * 0.2)
            a_opt = optimize_multiplicative(best_seq, time_budget=alloc, T_init=0.0003, T_min=0.00001,
                                             lr_init=0.004, verbose=True)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
            print(f"  MW2 done: obj={best_score:.8f}")
        
        # Palindromic again
        remaining = deadline - time.monotonic()
        if remaining > 40:
            alloc = min(80.0, remaining * 0.2)
            a_pali = optimize_palindromic(best_seq, time_budget=alloc, T_init=0.0002, T_min=0.000005,
                                           lr_init=0.003, verbose=True)
            obj_pali = compute_obj(a_pali)
            if obj_pali < best_score:
                best_score = obj_pali
                best_seq = a_pali.copy()
                print(f"  ** Palindromic2 improved to {best_score:.8f} **")
            else:
                print(f"  Palindromic2: obj={obj_pali:.8f}")
    
    # Phase 6: Coordinate descent polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 6: Coordinate Descent Polish ===")
        alloc = min(80.0, remaining * 0.3)
        a_cd = coordinate_descent_polish(best_seq, time_budget=alloc, delta=5e-5, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
        print(f"  CD done: obj={best_score:.8f}")
    
    # Phase 7: Pairwise swap polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 7: Pairwise Swap Polish ===")
        alloc = min(50.0, remaining * 0.3)
        a_ps = pairwise_swap_polish(best_seq, time_budget=alloc, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
        print(f"  PS done: obj={best_score:.8f}")
    
    # Phase 8: Final micro-polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 8: Final Micro-Polish ===")
        # Very small T gradient descent
        alloc = min(40.0, remaining * 0.4)
        a_final = optimize_gradient_descent(best_seq, time_budget=alloc, T_init=0.0001, T_min=0.000005,
                                             lr_init=0.002, momentum_init=0.9, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
        
        # CD with tiny delta
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(30.0, remaining * 0.5)
            a_cd = coordinate_descent_polish(best_seq, time_budget=alloc, delta=1e-5)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
        
        print(f"  Final: obj={best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
