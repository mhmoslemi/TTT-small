from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:

    size = 1

    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def gradient_fft(a, beta, topK=5):
    """
    Compute gradient of smooth max(conv(a,a)) using FFT.
    Returns gradient vector of same length as a.
    """
    n = len(a)
    conv = compute_conv(a)
    m = conv.max()
    if m < 1e-15:
        return np.zeros(n)
    
    # Use top-K peaks for focus
    if topK is not None and topK < len(conv):
        # Softmax only over top-K peaks, zero elsewhere
        idx = np.argsort(conv)[::-1][:topK]
        weights = np.zeros(len(conv))
        vals = conv[idx] - m
        exp_vals = np.exp(beta * vals)
        weights[idx] = exp_vals
        weights /= weights.sum()
    else:
        weights = np.exp(beta * (conv - m))
        weights /= weights.sum()
    
    # Gradient: g[i] = 2 * sum_{k} weights[k] * a[k-i] (for valid k-i)
    # This is a cross-correlation: g = 2 * cross_corr(weights, a)
    # Using FFT for efficiency
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    # cross_corr(weights, a)[i] = sum_k weights[k] * a[k-i] for valid k-i in [0,n-1]
    # = sum_j weights[i+j] * a[j] for valid i+j
    # This is the same as (weights * a_flipped) convolved appropriately
    # Actually: g[i] = sum_{j=0}^{n-1} a[j] * weights[i+j]
    # Let's compute this as a correlation
    
    # Method: use the fact that g[i] = sum_j a[j] * weights[i+j]
    # This can be computed as: for each i, dot product of a with weights[i:i+n]
    # Using FFT: this is a "sliding dot product" which is a cross-correlation
    
    # Actually, let's just use the direct computation with FFT:
    # Define w_extended as weights placed in a larger array
    # Then g[i] is the i-th element of the cross-correlation of a and weights
    
    # Simpler: g = 2 * np.correlate(weights, a, mode='valid') won't work directly
    # Let me think...
    
    # g[i] = sum_{j=0}^{n-1} a[j] * weights[i+j]
    # Let's flip a: af = a[::-1]
    # Then g[i] = sum_j af[n-1-j] * weights[i+j]
    # Hmm, this is getting complicated. Let me use a different approach.
    
    # Alternative: g[i] = sum_k weights[k] * a[k-i] for 0 <= k-i < n
    # = sum_k weights[k] * a[k-i] * 1(0 <= k-i < n)
    
    # This is the cross-correlation of weights and a, but truncated.
    # For our specific case, we can compute it as:
    # g = 2 * ifft(fft(weights) * conj(fft(a))) -- but this gives circular correlation
    
    # Let me just do it directly with a loop over the support of weights:
    # If weights is sparse (only top-K is non-zero), this is fast:
    non_zero = np.nonzero(weights)[0]
    g = np.zeros(n)
    for k in non_zero:
        # g[i] += 2 * weights[k] * a[k-i] for i where 0 <= k-i < n
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end+1] += 2.0 * weights[k] * a[j_start:j_end+1]
    
    return g

def gradient_fft_fast(a, beta, topK=10):
    """
    Faster gradient computation using the structure of the problem.
    For the peak at position k, grad = 2*a reversed and shifted.
    For multiple peaks (softmax weighted), it's a weighted sum of such shifts.
    """
    n = len(a)
    conv = compute_conv(a)
    m = conv.max()
    if m < 1e-15:
        return np.zeros(n)
    
    # Get top-K peaks
    if topK is not None and topK < len(conv):
        idx = np.argsort(conv)[::-1][:topK]
        vals = conv[idx] - m
        w = np.exp(beta * vals)
        w /= w.sum()
    else:
        w = np.exp(beta * (conv - m))
        w /= w.sum()
    
    # Compute gradient: g[i] = 2 * sum_k w_k * a[k-i] (for valid k-i in [0,n))
    # For each peak position k, the contribution to g is 2*w_k * a reversed and shifted by k
    g = np.zeros(n)
    for wk, k in zip(w, idx if topK is not None else range(len(conv))):
        if wk < 1e-10:
            continue
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end+1] += 2.0 * wk * a[j_start:j_end+1]
    
    return g

def mirror_descent_fast(a, beta, eta, topK=10):
    """
    One step of mirror descent with FFT-accelerated gradient.
    """
    n = len(a)
    g = gradient_fft_fast(a, beta, topK)
    # Mirror descent: a_new = a * exp(-eta * g)
    log_a = np.log(np.maximum(a, 1e-30))
    log_a_new = log_a - eta * g
    log_a_new -= np.max(log_a_new)
    a_new = np.exp(log_a_new)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    return a_new

def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g or None.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n else 0
    # conv(a_norm, g)[k] = sum_j a_norm[k-j] * g[j]
    # We need this <= M for all k
    
    # Build A as a matrix (might be large but for n~800 it's fine)
    # A is (2n-1) x n
    rows, cols, vals = [], [], []
    for j in range(n):
        for k in range(j, j + n):
            # A[k, j] = a_norm[k-j]
            vals.append(a_norm[k - j])
            rows.append(k)
            cols.append(j)
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue
    
    return None

def spectral_shape(a, alpha=0.1, n_iter=100):
    """
    Gerchberg-Saxton style spectral shaping.
    Iteratively flatten the power spectrum while maintaining non-negativity.
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    
    N = 1
    while N < 2 * n:
        N <<= 1
    
    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        P = np.abs(A)**2
        phase = np.angle(A)
        
        # Interpolate P toward its mean (flatten)
        P_mean = P.mean()
        P_new = P * (1 - alpha) + P_mean * alpha
        
        # Or geometric interpolation:
        # P_new = P ** (1-alpha) * P_mean ** alpha
        
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    
    return a

The LP is:
min M
s.t. for each k in {0, ..., 2n-2}:
       sum_{j: 0<=k-j<n} a[k-j] * g[j] <= M
       sum(g) = 1
       g >= 0

The constraint matrix A has a special banded structure: each row k has nonzeros only in columns j where 0 <= k-j < n, i.e., j in {max(0,k-n+1), ..., min(k, n-1)}.

Row k has at most n nonzeros (at k=n-1, all n columns are active). So the matrix is (2n-1) x n with at most n*(n + (n-1) + ... ) ≈ n^2 nonzeros total. For n=800, that's about 512K nonzeros. This is very sparse.

With a sparse LP solver, this should be fast.

"""
Hybrid optimization for minimum peak autocorrelation step function.
Combines FFT-accelerated mirror descent, LP refinement, and coordinate descent.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    s2 = s * s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / s2

def compute_obj_normalized(a):
    """Assume a is already normalized (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def gradient_topk(a, beta, topK=5):
    """Compute weighted gradient of peak autocorrelation."""
    n = len(a)
    conv = compute_conv(a)
    m = float(np.max(conv))
    if m < 1e-15:
        return np.zeros(n)
    
    # Get top-K indices
    if topK < len(conv):
        idx = np.argpartition(conv, -topK)[-topK:]
        idx = idx[np.argsort(conv[idx])[::-1]]
        vals = conv[idx] - m
        w = np.exp(beta * vals)
        total = w.sum()
        if total < 1e-30:
            w = np.zeros(len(idx))
            w[0] = 1.0
        else:
            w = w / total
    else:
        idx = np.arange(len(conv))
        w = np.exp(beta * (conv - m))
        w = w / w.sum()
    
    # Compute gradient: g[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    g = np.zeros(n)
    for wi, k in zip(w, idx):
        if wi < 1e-12:
            continue
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end+1] += 2.0 * wi * a[j_start:j_end+1]
    
    return g

def mirror_step(a, beta, eta, topK=5):
    """One mirror descent step."""
    n = len(a)
    g = gradient_topk(a, beta, topK)
    log_a = np.log(np.maximum(a, 1e-30))
    log_new = log_a - eta * g
    log_new -= np.max(log_new)
    a_new = np.exp(log_new)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / np.sum(a)
    
    # Build sparse A matrix: A[k,j] = a_n[k-j] if 0<=k-j<n
    rows = []
    cols = []
    vals = []
    for j in range(n):
        ks = range(j, j + n)
        for k in ks:
            vals.append(a_n[k - j])
            rows.append(k)
            cols.append(j)
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                s = float(np.sum(gv))
                if s > 1e-10:
                    return gv / s, float(M.value)
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[n] = 1.0
        A_ub = A.toarray()
        A_ub_full = np.hstack([A_ub, -np.ones((2*n-1, 1))])
        b_ub = np.zeros(2*n-1)
        A_eq = np.hstack([np.ones((1, n)), np.zeros((1,1))])
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            gv = np.maximum(res.x[:n], 0)
            s = float(np.sum(gv))
            if s > 1e-10:
                return gv / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
