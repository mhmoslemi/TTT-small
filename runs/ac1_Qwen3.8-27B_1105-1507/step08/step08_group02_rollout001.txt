import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)

    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

    """Build (2n-1) x n convolution matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)

    for j in range(n):

    return A

          0.6, 0.7, 0.8, 0.9, 1.0]
    best_t = 0.0
    best_f = best_obj

    for t in ts:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        f_try = compute_obj(a_try)
        if f_try < best_f:
            best_f = f_try
            best_t = t

    if best_t == 0.0:
        return best_obj, None, 0.0

    # Refine around best_t with golden section
    if 0.05 < best_t < 0.95:
        lo = max(0.0, best_t - 0.1)
        hi = min(1.0, best_t + 0.1)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)

        for _ in range(15):
            f_c1 = compute_obj(np.maximum((1 - c1) * a + c1 * g, 0) /
                               np.sum(np.maximum((1 - c1) * a + c1 * g, 0)))
            f_c2 = compute_obj(np.maximum((1 - c2) * a + c2 * g, 0) /
                               np.sum(np.maximum((1 - c2) * a + c2 * g, 0)))
            if f_c1 < f_c2:
                hi = c2
                c2 = c1
                c1 = hi - gr * (hi - lo)
            else:
                lo = c1
                c1 = c2
                c2 = lo + gr * (hi - lo)
            if hi - lo < 1e-6:
                break

        t_fine = (lo + hi) / 2.0
        a_fine = np.maximum((1 - t_fine) * a + t_fine * g, 0)
        a_fine = a_fine / a_fine.sum()
        f_fine = compute_obj(a_fine)
        if f_fine < best_f:
            best_f = f_fine
            best_t = t_fine

    a_best = np.maximum((1.0 - best_t) * a + best_t * g, 0)
    a_best = a_best / a_best.sum()
    return best_f, a_best, best_t


def subgradient_step(a, lr, n):
    """
    One subgradient step on the simple max peak.
    Returns new a (on simplex).
    """
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    # Subgradient: g_i = 2 * a[k_star - i] if 0 <= k_star - i < n
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] = 2.0 * a[j]

    # Mirror descent step
    log_a = np.log(np.maximum(a, 1e-30))
    log_a_new = log_a - lr * g
    log_a_new -= np.max(log_a_new)
    a_new = np.exp(log_a_new)
    a_new = a_new / a_new.sum()
    return a_new


def smooth_optimize(a, time_budget, beta_start=50.0, beta_end=3000.0):
    """Smooth optimization using log-sum-exp with mirror descent."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    lr = 3.0 / np.sqrt(n)

    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / max(1, int(time_budget * 3)))
        beta = beta_start * (beta_end / beta_start) ** frac

        conv = compute_conv(a)
        m = conv.max()
        shifted = beta * (conv - m)
        weights = np.exp(shifted)
        weights /= weights.sum()

        # Gradient: g_i = 2 * sum_j a[j] * w_{i+j}
        g = np.zeros(n)
        for i in range(n):
            j_start = 0
            j_end = min(n, 2 * n - 1 - i)
            if j_start <= j_end:
                g[i] = 2.0 * np.dot(a[j_start:j_end], weights[i + j_start:i + j_end])

        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = max(1e-5, lr * 0.997)
        else:
            lr = max(1e-5, lr * 0.95)

        it += 1

    return best_a


def lp_anneal_main(a, time_budget, rng, verbose=True):
    """
    Main LP-alternating optimization with simulated annealing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall = 0
    patience = 30
    temperature = 0.1
    iteration = 0
    last_good_t = 0.5

    while time.monotonic() - start < time_budget:
        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            # Fallback: subgradient step
            g_lp = subgradient_step(a, 1.0 / np.sqrt(n), n)

        # Check similarity
        diff_norm = np.linalg.norm(a - g_lp)
        if diff_norm < 1e-8:
            stall += 3
        else:
            stall += 1

        # Line search
        obj_new, a_new, t_best = line_search(a, g_lp, best_obj, n)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            last_good_t = t_best
            stall = 0
            temperature = max(0.005, temperature * 0.98)
        else:
            stall += 1
            if stall > patience:
                # Annealing restart
                noise = temperature * rng.normal(0, 1, n)
                a = best_a * (1.0 + noise)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()

                # Try blending with random direction
                if rng.random() < 0.4:
                    random_dir = rng.dirichlet(np.ones(n) * 0.5)
                    blend = 0.1 + 0.3 * rng.random()
                    a = (1 - blend) * best_a + blend * random_dir
                    a = np.maximum(a, 0)
                    a = a / a.sum()

                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                temperature = min(0.3, temperature * 2.5)
                last_good_t = 0.5
            else:
                temperature = max(0.01, temperature * 0.99)

        iteration += 1
        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP-Anneal it{iteration}: obj={best_obj:.8f}, "
                  f"t={last_good_t:.3f}, stall={stall}, temp={temperature:.3f}, "
                  f"elapsed={elapsed:.1f}s")

    return best_a


def coordinate_descent(a, time_budget):
    """
    Coordinate descent: find optimal scaling of individual coordinates.
    For each coordinate, try to reduce it (moving mass to others).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    max_rounds = 20
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        improved = False

        # Identify top contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib = a[i] * a[j]
                contributors.append((contrib, i, j))
        contributors.sort(reverse=True)

        # Non-contributor targets
        contributor_set = set()
        for _, i, j in contributors[:20]:
            contributor_set.add(i)
            if i != j:
                contributor_set.add(j)

        non_contrib = [x for x in range(n) if x not in contributor_set]
        if not non_contrib:
            non_contrib = list(range(n))

        for contrib, i, j in contributors[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break

            for target in non_contrib[:5]:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    if delta <= 0 or delta >= a[i]:
                        continue
                    src = i if a[i] >= a[j] else j
                    old_val = a[src]
                    a[src] -= delta
                    a[target] += delta
                    a = a / a.sum()
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        a[src] = old_val
                        a[target] -= delta
                        a = a / a.sum()
                        break

        if not improved:
            # Try pairing swaps
            for c1, c2 in zip(contributors[:10], contributors[10:20]):
                if time.monotonic() - start > time_budget:
                    break
                i1, j1 = c1[1], c1[2]
                i2, j2 = c2[1], c2[2]
                if i1 == i2 or i1 == j2:
                    continue
                delta = 0.05 * a[i2]
                if delta <= 0:
                    continue
                old_i2 = a[i2]
                old_i1 = a[i1]
                a[i2] -= delta
                a[i1] += delta
                a = a / a.sum()
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = compute_conv(a)
                    k_star = int(np.argmax(conv))
                    improved = True
                    break
                else:
                    a[i2] = old_i2
                    a[i1] = old_i1
                    a = a / a.sum()
            if not improved:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] * n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-alternating with annealing, multi-dimension exploration,
    smooth optimization, and coordinate descent polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # --- Load parent ---
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, 1000.0)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nStarting: n={n_current}, obj={best_score:.8f}")

    # --- Phase 1: Smooth optimization warmup ---
    print(f"\n=== Phase 1: Smooth Warmup ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 100.0)
    if p1_budget > 20:
        a_smooth = smooth_optimize(best_seq, time_budget=p1_budget,
                                   beta_start=50.0, beta_end=2000.0)
        obj_s = compute_obj(a_smooth)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_smooth.copy()
            print(f"  Smooth: obj={best_score:.8f} **")
        else:
            print(f"  Smooth: obj={obj_s:.8f} (no improvement)")

    # --- Phase 2: Main LP-alternating optimization ---
    print(f"\n=== Phase 2: LP Alternating (Main) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.55, 700.0)

    # Multiple rounds with different perturbations
    n_rounds = 3
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        alloc = min(p2_budget / (n_rounds - rnd), remaining * 0.55)
        if alloc < 30:
            break

        if rnd == 0:
            a_start = best_seq.copy()
        else:
            # Perturb for diversity
            scale = 0.03 + 0.02 * rnd
            rng_r = np.random.default_rng(seed + 1000 * (rnd + 1))
            a_start = best_seq * (1.0 + scale * rng_r.normal(0, 1, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
            # Also try a blend with random
            if rnd == 1:
                rand_dir = rng_r.dirichlet(np.ones(n_current) * 0.3)
                a_start = 0.8 * best_seq + 0.2 * rand_dir
                a_start = a_start / a_start.sum()

        print(f"  Round {rnd}: budget={alloc:.0f}s")
        a_lp = lp_anneal_main(a_start, time_budget=alloc, rng=rng)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            n_current = len(best_seq)
            print(f"  Round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  Round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # --- Phase 3: Multi-dimension exploration ---
    print(f"\n=== Phase 3: Multi-Dimension ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = []
    for delta in [-200, -100, -50, 50, 100, 200, 300]:
        n_try = n_best + delta
        if 300 <= n_try <= 2000:
            n_options.append(n_try)
    # Also try some fixed sizes
    for n_fix in [400, 600, 1000, 1200, 1500]:
        if n_fix != n_best:
            n_options.append(n_fix)

    n_options = sorted(set(n_options))

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 80.0)

        # Quick LP optimization
        a_opt = lp_anneal_main(a_try, time_budget=alloc, rng=rng, verbose=False)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = len(best_seq)
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # --- Phase 4: More LP rounds on best so far ---
    print(f"\n=== Phase 4: Extended LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        lp_alloc = min(remaining * 0.3, 200.0)
        rng2 = np.random.default_rng(seed + 9999)
        a_lp2 = lp_anneal_main(best_seq, time_budget=lp_alloc, rng=rng2)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Extended LP: obj={best_score:.8f} **")
        else:
            print(f"  Extended LP: obj={obj_lp2:.8f}")

    # --- Phase 5: Coordinate descent polish ---
    print(f"\n=== Phase 5: Coordinate Descent Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd_alloc = min(remaining * 0.25, 100.0)
        a_cd = coordinate_descent(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # --- Phase 6: Final LP + CD alternating ---
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        half = remaining * 0.5
        if half > 20:
            a_f1 = lp_anneal_main(best_seq, time_budget=half, rng=rng)
            obj_f1 = compute_obj(a_f1)
            if obj_f1 < best_score:
                best_score = obj_f1
                best_seq = a_f1.copy()
                print(f"  Final LP: obj={best_score:.8f} **")
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_f2 = coordinate_descent(best_seq, time_budget=min(remaining, 30.0))
                obj_f2 = compute_obj(a_f2)
                if obj_f2 < best_score:
                    best_score = obj_f2
                    best_seq = a_f2.copy()
                    print(f"  Final CD: obj={best_score:.8f} **")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 200
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
