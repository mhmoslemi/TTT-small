"""

4. Gradient-based coordinate polish

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_fast(a):
    """Solve LP: min M s.t. A@z <= M, sum(z)=1, z>=0. Returns (z, M, dual)."""
    n = len(a)
    A = build_conv_matrix(a)
    
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ z <= M, cp.sum(z) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual = None
                    try:
                        dual = np.maximum(np.array(constraints[0].dual_value), 0)
                    except:
                        pass
                    return z_val, float(M.value), dual
        except:
            continue
    
    # Fallback
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
        A_eq = np.hstack([np.ones((1, n)), np.zeros((1,1))])
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2*n-1), A_eq=A_eq, b_eq=[1.0],
                      bounds=[(0,None)]*n + [(None,None)], method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except:
        pass
    
    return None, None, None

            a = np.maximum(a, 0) / np.sum(a)

            best_a = a.copy() if best_obj == compute_obj(a) else best_a
            prev_a = None
            prev_prev_a = None
            stall_count = 0
            continue
        
        # Backtracking line search
        best_mix = 0
        best_try_obj = best_obj

def coordinate_polish(a, time_budget=30.0):

            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)

                a_try[i] *= scale

                    continue
                a_try = a_try / st

                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break
        if not improved:
            # Try pairwise
            for _ in range(20):
                if time.monotonic() - start > time_budget:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                for frac in [0.02, 0.05, 0.1, -0.02, -0.05, -0.1]:
                    delta = frac * (best_a[i] + best_a[j]) / 2
                    a_try = best_a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    a_try = a_try / np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
            if not improved:
                break
    
    return best_a

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimized step function search using:
1. Fast LP alternating with adaptive backtracking step size
2. Dual-informed perturbation for escaping local optima
3. Aggressive multi-n exploration with resampling
4. Gradient-based flattening perturbation
5. Fine coordinate/pairwise polish for final refinement

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ z gives cross-conv a*z."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. A@a conv z <= M, sum(z)=1, z>=0.
    Returns (z_normalized, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except:
                        pass
                    return z_val, float(M.value), dual_y
        except:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2 * n - 1),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except:
        pass

    return None, None, None

                a_avg = np.maximum(a_avg, 0)
                sa = float(np.sum(a_avg))
                if sa > 1e-10:

                obj_avg = compute_obj(a_avg)
                if obj_avg < best_obj:

                    best_a = a_avg.copy()
                a = a_avg
                prev_a = None
                prev_prev_a = None

        found_step = False

                a = a_try
                best_obj = obj_try
                best_a = a.copy()
                stall = 0
                found_step = True

            sp = float(np.sum(a_pert))
            if sp > 1e-10:

                prev_a = None

                print(f"    [perturb it={it_count}] obj={compute_obj(a):.8f}, best={best_obj:.8f}, t={elapsed:.0f}s")

        if verbose and it_count % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP-alt: it={it_count}, obj={best_obj:.8f}, t={elapsed:.0f}s, stall={stall}")

        # Time check for verbose
        if verbose and it_count % 50 == 0:
            if time.monotonic() - start > time_budget * 0.9:
                break

    return best_a


def flatten_perturb(a, time_budget=40.0):
    """
    Try to flatten the autocorrelation profile by directing mass
    toward lags that are below the maximum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(best_a)
        max_val = float(np.max(conv))
        if max_val < 1e-10:
            break

        # Find lags significantly below the max
        threshold = 0.7 * max_val
        below = np.where(conv < threshold)[0]
        if len(below) == 0:
            break

        # Pick a random below-threshold lag
        k = int(rng.choice(below))

        # Direction: increase a[j] proportional to a[k-j] to boost conv[k]
        direction = np.zeros(n)
        valid_js = np.arange(max(0, k - n + 1), min(n, k + 1))
        for j in valid_js:
            kj = k - j
            if 0 <= kj < n:
                direction[j] = best_a[kj]

        # Also subtract a bit from coords that contribute to active (max) lags
        active_lags = np.where(conv >= 0.95 * max_val)[0]
        for kl in active_lags[:5]:
            for j in valid_js:
                kj = kl - j
                if 0 <= kj < n:
                    direction[j] -= 0.5 * best_a[kj]

        dnrm = np.linalg.norm(direction)
        if dnrm < 1e-10:
            continue
        direction /= dnrm

        # Try small steps
        for step in [0.02, 0.01, 0.005, 0.002]:
            a_try = best_a + step * direction
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                break

    return best_a


def coordinate_polish(a, time_budget=30.0):
    """
    Fine coordinate and pairwise polish.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)

    while time.monotonic() - start < time_budget:
        improved = False

        # Single coordinate
        for _ in range(30):
            if time.monotonic() - start > time_budget:
                break
            i = int(rng.integers(0, n))
            for scale in [0.92, 0.96, 0.88, 1.04, 1.08, 1.12]:
                a_try = best_a.copy()
                a_try[i] *= scale
                if a_try[i] < 0:
                    continue
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break

        # Pairwise
        for _ in range(15):
            if time.monotonic() - start > time_budget:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            for frac in [0.03, 0.06, 0.015]:
                delta = frac * (best_a[i] + best_a[j]) * 0.5
                for sign in [1, -1]:
                    a_try = best_a.copy()
                    a_try[i] -= sign * delta
                    a_try[j] += sign * delta
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    a_try = a_try / np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try a few more rounds
            any_improve = False
            for _ in range(10):
                i = int(rng.integers(0, n))
                for scale in [0.97, 1.03]:
                    a_try = best_a.copy()
                    a_try[i] *= scale
                    if a_try[i] < 0:
                        continue
                    a_try = a_try / np.sum(a_try)
                    if compute_obj(a_try) < best_obj - 1e-14:
                        best_obj = compute_obj(a_try)
                        best_a = a_try.copy()
                        any_improve = True
                        break
            if not any_improve:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining fast LP alternating, flattening perturbation,
    multi-n exploration, restarts, and coordinate polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        for n_init in [600, 800, 1000, 1200]:
            t = np.linspace(0, 1, n_init)
            windows = [
                0.5 + 0.5 * np.cos(2 * np.pi * t),
                np.sin(np.pi * t) ** 2,
                1.0 - 0.2 * np.cos(3 * np.pi * t),
            ]
            for w in windows:
                a_init = np.maximum(w, 0.01)
                a_init = a_init / np.sum(a_init)
                obj_init = compute_obj(a_init)
                if obj_init < best_score:
                    best_score = obj_init
                    best_seq = a_init.copy()
        if best_seq is None:
            best_seq = np.ones(800) / 800.0
            best_score = compute_obj(best_seq)
        print(f"Initial: obj={best_score:.8f}")

    # === Phase 1: Main LP optimization (450s) ===
    print(f"\n=== Phase 1: LP Alt (n={len(best_seq)}, budget~450s) ===")
    remaining = deadline - time.monotonic()
    alloc = min(450.0, remaining * 0.4)
    a_opt = lp_alternate(best_seq, time_budget=alloc, verbose=True)
    obj = compute_obj(a_opt)
    if obj < best_score:
        best_score = obj
        best_seq = a_opt.copy()
        print(f"  Phase 1: obj={best_score:.8f} **")
    else:
        print(f"  Phase 1: obj={obj:.8f}")

    # Flattening perturbation after main LP
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(40.0, remaining * 0.06)
        a_opt = flatten_perturb(best_seq, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Flatten: obj={best_score:.8f} **")
        else:
            print(f"  Flatten: obj={obj:.8f}")
        # Follow up with LP on the flattened result
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(30.0, remaining * 0.05)
            a_opt = lp_alternate(best_seq, time_budget=alloc)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Post-flatten LP: obj={best_score:.8f} **")

    # === Phase 2: Multi-n exploration (250s total) ===
    print(f"\n=== Phase 2: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_parent = len(best_seq)
    n_candidates = sorted(set([
        max(400, n_parent - 200), max(400, n_parent - 100),
        n_parent + 100, n_parent + 200, n_parent + 400,
        500, 600, 700, 900, 1000, 1100, 1200, 1400, 1600,
    ]))
    n_candidates = [n for n in n_candidates if 400 <= n <= 1800 and n != n_parent]
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    per_n_time = max(20.0, min(35.0, remaining * 0.18 / max(len(n_candidates), 1)))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(per_n_time, remaining * 0.15)
        a_opt = lp_alternate(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_parent = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Targeted LP on best (200s) ===
    print(f"\n=== Phase 3: Targeted LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(200.0, remaining * 0.35)
        a_opt = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj:.8f}")

    # === Phase 4: Restarts (120s) ===
    remaining = deadline - time.monotonic()
    if remaining > 35:
        print(f"\n=== Phase 4: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(5, int(remaining // 30))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            if r == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.04, n_best))
            elif r == 1:
                a_r = best_seq * np.exp(rng.normal(0, 0.08, n_best))
            elif r == 2:
                a_r = best_seq.copy()
                n_blocks = max(10, n_best // 80)
                bs = n_best // n_blocks
                blocks = rng.choice(n_blocks, min(4, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * bs
                    ei = si + bs
                    a_r[si:ei] *= rng.uniform(0.3, 1.8)
            elif r == 3:
                # Mirror/asymmetry perturbation
                a_r = best_seq.copy()
                half = n_best // 2
                a_r[half:] = a_r[half:][::-1] * rng.uniform(0.7, 1.3)
            else:
                a_r = best_seq * np.exp(rng.normal(0, 0.06, n_best))

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr
            alloc = min(remaining * 0.25, 25.0)
            a_opt = lp_alternate(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 5: Flatten + LP on best (40s) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 5: Flatten + LP ===")
        alloc = min(20.0, remaining * 0.2)
        a_opt = flatten_perturb(best_seq, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Flatten: obj={best_score:.8f} **")
        else:
            print(f"  Flatten: obj={obj:.8f}")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(20.0, remaining * 0.4)
            a_opt = lp_alternate(best_seq, time_budget=alloc)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Post-flatten LP: obj={best_score:.8f} **")

    # === Phase 6: Coordinate polish (30s) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 6: Coordinate Polish ===")
        alloc = min(30.0, remaining * 0.4)
        a_opt = coordinate_polish(best_seq, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Polish: obj={best_score:.8f} **")
        else:
            print(f"  Polish: obj={obj:.8f}")

    # === Phase 7: Final LP squeeze (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.5, 20.0)
        a_opt = lp_alternate(best_seq, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Final squeeze: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
