    size = 2 * n - 1
    nfft = 1
    while nfft < size:
        nfft <<= 1
    fa = rfft(a, nfft)
    return irfft(fa * fa, nfft)[:size]

def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        length = k_end - k_start + 1
        A[k_start:k_start + length, j] = a[:length]
    return A

def solve_lp(a):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.ones((1, n + 1))
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            return g_val / gs, float(res.x[n])
    return None, None

def lse_gradient(a, beta):
    """Compute gradient of LSE(beta * conv(a,a)) w.r.t. a (unnormalized)."""
    n = len(a)
    conv = compute_conv(a)
    
    # Softmax weights
    conv_max = np.max(conv)
    exp_vals = np.exp(beta * (conv - conv_max))
    w = exp_vals / np.sum(exp_vals)
    
    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
    # = 2 * (a cross-correlate w)[j]
    nfft = 1
    while nfft < 4 * n:
        nfft <<= 1
    
    A_fft = rfft(a, nfft)
    W_fft = rfft(w, nfft)
    R = irfft(np.conj(A_fft) * W_fft, nfft)
    grad = 2.0 * R[:n]
    
    return grad, w

def projected_gradient_step(a, grad, lr):
    """Take a projected gradient step on the simplex."""
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s

def backtracking_line_search(a, direction, obj_current, max_iter=50):
    """Find step size alpha such that obj(a - alpha*direction) < obj(a).
    direction is the projected gradient (pointing downhill)."""
    alpha = 1.0
    for _ in range(max_iter):
        a_try = a - alpha * direction
        a_try = np.maximum(a_try, 0)
        s = np.sum(a_try)
        if s < 1e-10:
            alpha *= 0.5
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < obj_current - 0.01 * alpha * np.sum(np.abs(direction) * a):
            return a_try, alpha
        alpha *= 0.5
    return a, 0.0

def smoothed_gradient_optimize(a, time_budget=100.0, beta_start=100.0, beta_end=5000.0):
    """Optimize using smoothed gradient with adaptive beta."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 1.0 / n
    total_iters = 5000
    
    for it in range(total_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive beta
        progress = it / total_iters
        beta = beta_start * (beta_end / beta_start) ** progress
        
        grad, _ = lse_gradient(a, beta)
        # Project onto simplex tangent space
        grad = grad - np.mean(grad)
        
        # Line search
        a_new, alpha = backtracking_line_search(a, grad, compute_obj(a))
        
        if alpha > 0:
            a = a_new
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        
        # Decay learning rate
        if it % 500 == 499:
            lr *= 0.9
        
        if it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SG: it={it}, beta={beta:.0f}, obj={best_obj:.8f}, time={elapsed:.0f}s")
    
    return best_a

def lp_alternate_linesearch(a, time_budget=200.0, verbose=False):

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_initialization(n, rng):
    """Generate a diverse structured initialization."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * (x - 0.5) ** 2)
    elif choice == 1:
        x = np.linspace(0, 1, n)
        freq = rng.uniform(2, 8)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        c1, c2 = sorted(rng.uniform(0, 1, 2))
        a = np.exp(-80 * (x - c1) ** 2) + np.exp(-80 * (x - c2) ** 2)
    elif choice == 3:
        a = np.ones(n)
        for _ in range(4):
            center = rng.integers(0, n)
            width = rng.integers(n // 10, n // 3)
            for i in range(n):
                d = min(abs(i - center), n - abs(i - center))
                a[i] *= 1.0 + rng.standard_normal() * 0.5 * np.exp(-d / (width * 0.5))
        a = np.maximum(a, 0.01)
    elif choice == 4:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(0.3, 4.0)
    else:
        # Random with smoothing
        a = rng.random(n)
        kernel_size = max(3, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: LP alternating with 1D line search, smoothed LSE gradient polishing,
multi-n exploration, and diverse structured restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize_scalar
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    nfft = 1
    while nfft < size:
        nfft <<= 1
    fa = rfft(a, nfft)
    return irfft(fa * fa, nfft)[:size]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(m - 1, j + n - 1)
        length = k_end - k_start + 1
        A[k_start:k_start + length, j] = a[:length]
    return A


def solve_lp(a, reduced=False, threshold=0.9):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s
    A_full = build_conv_matrix(a)
    m_full = 2 * n - 1

    if reduced:
        conv_a = compute_conv(a)
        max_conv = np.max(conv_a)
        active = np.where(conv_a >= threshold * max_conv)[0]
        A = A_full[active, :]
        m = len(active)
    else:
        A = A_full
        m = m_full

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.ones((1, n + 1))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    return None, None


def lse_gradient_direction(a, beta):
    """Compute scaled gradient direction of LSE(beta * conv(a,a)).
    Returns (direction, lse_value) where direction points uphill."""
    n = len(a)
    conv = compute_conv(a)
    conv_max = np.max(conv)
    scaled = beta * (conv - conv_max)
    # Numerically stable LSE
    lse_val = conv_max + np.log(np.sum(np.exp(scaled))) / beta
    w = np.exp(scaled) / np.sum(np.exp(scaled))

    # grad[j] = 2 * sum_k w[k] * a[k-j]
    nfft = 1
    while nfft < 4 * n:
        nfft <<= 1
    A_fft = rfft(a, nfft)
    W_fft = rfft(w, nfft)
    R = irfft(np.conj(A_fft) * W_fft, nfft)
    grad = 2.0 * R[:n]
    return grad, lse_val

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        p3_budget = min(remaining * 0.12, 90.0)
        a_p3 = smoothed_grad_optimize(best_a, time_budget=p3_budget,
                                      beta_start=2000.0, beta_end=20000.0)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            print(f"  Phase 3: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_p3:.8f}")

    # === Phase 4: LP alternating round 2 (from polished point) ===
    print(f"\n=== Phase 4: LP + Line Search Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = min(remaining * 0.30, 250.0)
        a_p4 = lp_alternate_ls(best_a, time_budget=p4_budget, verbose=True)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_obj:
            best_obj = obj_p4
            best_a = a_p4
            print(f"  Phase 4: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 4: obj={obj_p4:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        max(100, n_best - 400), max(100, n_best - 200),
        n_best + 100, n_best + 200, n_best + 400,
    ]))
    n_options = [x for x in n_options if 100 <= x <= 2500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_seq(best_a, n_try)
        a_try = a_try / float(np.sum(a_try))
        obj_start = compute_obj(a_try)

        alloc = min(50.0, remaining * 0.20)
        a_opt = lp_alternate_ls(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f} (start={obj_start:.8f})")

    # === Phase 6: Diverse initializations at best n ===
    print(f"\n=== Phase 6: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_a)

    for d in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_div = diverse_initialization(n_use, rng)
        alloc = min(35.0, remaining * 0.12)
        a_opt = lp_alternate_ls(a_div, time_budget=alloc)
        obj_div = compute_obj(a_opt)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse {d}: obj={obj_div:.8f}")

    # === Phase 7: Final polish with remaining time ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.45, 60.0)
        a_final = lp_alternate_ls(best_a, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.35, 30.0)
        a_sg = smoothed_grad_optimize(best_a, time_budget=alloc,
                                      beta_start=5000.0, beta_end=50000.0)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  Final SG: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
