"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses sparse-constraint LP alternating, exponentiated gradient updates,
and multi-scale n exploration with coordinate polishing.
Key improvements: sparse LP (top-K constraints only), fast HiGHS solves,
adaptive constraint management, and structured initializations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv_fft(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fft(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_sparse_lp(a, active_indices, K=30):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for k in active_indices,
    sum(g) = 1, g >= 0.
    
    Returns g (normalized) or None.
    """
    n = len(a)
    a = a / float(np.sum(a))
    
    # Build sparse constraint matrix for active indices only
    # For each active index k, the constraint is:
    #   sum_j a[k-j] * g[j] <= M  (for valid j)
    
    m_constraints = len(active_indices)
    
    # Use dense matrix (should be fine for n up to ~2000 and K up to ~50)
    A_ub = np.zeros((m_constraints + 1, n + 1))
    b_ub = np.zeros(m_constraints + 1)
    
    for i, k in enumerate(active_indices):
        # conv(a,g)[k] = sum_j a[k-j]*g[j] for valid j
        # j ranges from max(0, k-n+1) to min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[i, j] = a[k - j]
        A_ub[i, n] = -1.0  # -M
        b_ub[i] = 0.0
    
    # Sum(g) = 1 constraint (as equality)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # Minimize M
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s
    return None


def lp_alternating_sparse(a, time_budget=300.0, K=30, verbose=False):
    """
    LP alternating with sparse constraints (only top-K lags constrained).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    best_t = 0.5
    rng = np.random.default_rng(12345)
    it = 0
    
    conv = compute_conv_fft(a)
    active = np.argsort(conv)[::-1][:K]
    
    while time.monotonic() - start < time_budget:
        if time_budget - (time.monotonic() - start) < 1.0:
            break
        
        # Solve LP with current active constraints
        g_star = solve_sparse_lp(a, active, K)
        if g_star is None:
            # Fallback: use uniform
            g_star = np.ones(n) / n
        
        # Line search
        ts = [best_t, best_t * 0.5, best_t * 1.5, 0.1, 0.3, 0.5, 0.7, 1.0]
        ts = sorted(set(t for t in ts if 0.01 <= t <= 1.0))
        
        improved = False
        for t in ts:
            a_trial = (1.0 - t) * a + t * g_star
            a_trial = np.maximum(a_trial, 0)
            s_trial = float(np.sum(a_trial))
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial
            obj_trial = compute_obj(a_trial)
            if obj_trial < best_obj - 1e-12:
                best_obj = obj_trial
                best_a = a_trial.copy()
                a = a_trial
                best_t = t
                improved = True
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Escape: perturb and reset
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                best_t = 0.5
        
        # Update active set
        conv = compute_conv_fft(a)
        active = np.argsort(conv)[::-1][:K]
        
        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t:.2f}, el={elapsed:.1f}s")
    
    return best_a


def exp_gradient_method(a, time_budget=120.0, verbose=False):
    """
    Exponentiated gradient (mirror descent) with adaptive temperature.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 2.0 / np.sqrt(n)
    it = 0
    stall = 0
    rng = np.random.default_rng(999)
    
    while time.monotonic() - start < time_budget:
        if time_budget - (time.monotonic() - start) < 0.5:
            break
        
        conv = compute_conv_fft(a)
        max_val = conv.max()
        if max_val < 1e-15:
            break
        
        # Softmax weights on the convolution
        T = 5.0 * max_val  # Temperature
        w = np.exp((conv - max_val) / T)
        w = w / w.sum()
        
        # Compute gradient: grad[j] = sum_k w[k] * a[k-j]
        # Use the cross-convolution structure
        # grad = cross_correlation(w, a) in a specific sense
        # Actually: d(conv(a,a)[k])/da[j] involves a[k-j]
        # The weighted gradient is: sum_k w[k] * a[k-j] (approximately)
        
        # Compute via FFT: grad[j] = sum_k w[k] * a[k-j]
        # This is the cross-correlation of w and a
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        # We want: grad[j] = sum_k w[k] * a[k-j] for valid k,j
        # This is like a correlation. Let's compute it directly for efficiency
        # when n is moderate, or use FFT.
        
        # Using the fact that: sum_k w[k] * a[k-j] = (w * a_flip)[j + offset]
        a_flip = a[::-1]
        # (w * a_flip)[m] = sum_k w[k] * a_flip[m-k] = sum_k w[k] * a[n-1-(m-k)]
        # = sum_k w[k] * a[n-1-m+k]
        # We want sum_k w[k] * a[k-j], so set n-1-m+k = k-j => m = n-1+j
        # So grad = conv(w, a_flip)[n-1 : n-1+n]
        
        conv_wa = compute_conv_cross(w, a_flip)
        grad = conv_wa[n-1:n-1+n]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = max(0.01, eta * 0.999)
        else:
            stall += 1
            if stall > 50:
                a = best_a * np.exp(rng.normal(0, 0.15, n))
                a
