def smooth_optimize(sequence, time_budget=200.0):
    """
    Perform smooth optimization using gradient descent with top-K gradient truncation.
    """
    sequence = np.array(sequence)
    sequence_sum = sequence.sum()
    sequence = sequence / sequence_sum if sequence_sum > 0.01 else sequence * 0.01 / 0.01
    n = len(sequence)

    # Compute smooth gradient approximation
    conv_result = np.convolve(sequence, sequence, mode='full')
    max_val = conv_result.max()
    smooth_term = 0.1 * (conv_result - max_val)
    gradient_shifted = sequence * np.exp(smooth_term / 0.1) if max_val > 0.0 else np.ones(n) * 0.01

    learning_rate = 0.5 / (n * max_val) if max_val > 0.0 else 0.01
    new_sequence = sequence - learning_rate * gradient_shifted

    target_sum = 0.01 * learning_rate
    new_sequence_sum = np.sum(new_sequence)
    new_sequence = new_sequence / new_sequence_sum * (target_sum / learning_rate) if new_sequence_sum > 0.01 else new_sequence * 0.01 / 0.01

    # Ensure nonnegative values and normalize
    new_sequence = np.maximum(new_sequence, 0)
    new_sum = np.sum(new_sequence)
    new_sequence = new_sequence / new_sum if new_sum > 0.01 else np.ones(n) * 0.01 / n

    # Recalculate values after updates
    conv_new = np.convolve(new_sequence, new_sequence, mode='full')
    new_objective = 2.0 * n * conv_new.max() / (np.sum(new_sequence))**2
    objective_prev = 2.0 * n * conv_result.max() / (np.sum(sequence))**2

    return new_sequence if objective_prev >= new_objective else sequence

def mirror_descent_topk(sequence, time_budget=60.0):

    """

    # Compute element-wise gradient
