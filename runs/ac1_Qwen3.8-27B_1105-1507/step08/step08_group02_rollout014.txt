active-set LP, and multi-scale exploration.
Key improvements over previous: O(n log n) gradient computation via FFT,
faster LP with active constraints only, and better overall scheduling.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_gradient_fft(a, weights):
    """
    Compute gradient of smooth max approximation using FFT.
    weights: softmax weights over convolution values (length 2n-1)
    Returns gradient (length n).
    g[i] = 2 * sum_m a[m] * w[i+m] for valid m.
    Uses FFT-based convolution for O(n log n) computation.

    a_rev = a[::-1]

    while size < 3 * n - 2:

    fw = rfft(weights, size)
    conv_full = irfft(fa * fw, size)
    # C[k] = conv_full[n-1+k] for k = 0, ..., n-1
    C = conv_full[n-1:n-1+n]
    return 2.0 * C

"""
Optimized step function search using FFT-accelerated gradient methods,
active-set LP refinement, and adaptive multi-phase optimization.
Key innovations: O(n log n) gradient via FFT, QP-based direction finding,
adaptive top-K smoothing, and aggressive multi-scale exploration.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time
import cvxpy as cp


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def conv_cross(a, g):
    """Compute conv(a, g) where both have length n. Result length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:2 * n - 1]


def aptv(a, v):
    """Compute A^T * v where (A*g)_k = conv(a,g)_k.
    a has length n, v has length 2n-1, result has length n.
    (A^T v)_i = sum_j a[j] * v[j+i] for valid j.
    """
    n = len(a)
    a_rev = a[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fa = rfft(a_rev, size)
    fv = rfft(v, size)
    conv_full = irfft(fa * fv, size)
    return conv_full[n - 1:n - 1 + n]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_at_peak(a, k_star):
    """Exact subgradient of max(conv(a,a)) at the peak position k_star.
    g[i] = 2 * sum of a[j]*a[k_star - j] contributions to position i.
    For the subgradient: if conv[k_star] is the max, then
    d/d a[i] conv[k_star] = 2 * a[k_star - i] (for valid index).
    """
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] = 2.0 * a[j]
    return g


def smooth_grad_topk(a, K, beta):
    """Compute smooth gradient using top-K weighting. O(n log n) via FFT."""
    conv = compute_conv(a)
    max_val = conv.max()
    if max_val < 1e-15:
        return np.zeros(len(a)), conv
    
    top_idx = np.argsort(conv)[::-1][:K]
    top_vals = conv[top_idx]
    
    weights = np.zeros(2 * len(a) - 1)
    shifted = beta * (top_vals - top_vals[0])
    w_top = np.exp(shifted)
    w_top /= w_top.sum()
    for i, idx in enumerate(top_idx):
        weights[idx] = w_top[i]
    
    # FFT-based gradient computation
    g = aptv(a, 2.0 * weights)
    return g, conv


def smooth_optimize_fft(a, time_budget=100.0, K=5, beta_start=5.0, beta_end=200.0,
                        verbose=False):
    """
    Fast smooth optimization using FFT-based gradient with adaptive parameters.
    Uses projected gradient with backtracking line search on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / n
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / max(1, 5000))
        beta = beta_start * (beta_end / beta_start) ** frac
        # Adaptive K: use larger K early, smaller K late
        K_cur = max(1, int(K * (1.0 - 0.5 * frac)))

        g, conv = smooth_grad_topk(a, K_cur, beta)
        
        # Projected gradient step with line search
        improved = False
        for ls in range(20):
            a_new = project_simplex(a - lr * g)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                lr = min(lr * 1.05, 5.0 / n)
                stall = 0
                improved = True
                break
            lr *= 0.5
            lr = max(lr, 1e-10)

                log_a_new -= np.max(log_a_new)
                a_new = np.exp(log_a_new)

                a_new = a_new / a_new.sum()

            print(f"    SmoothFFT it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={time.monotonic()-start:.1f}s")

def qp_direction(a, max_iters=200):
    """
    Solve QP: min ||A g||^2 s.t. sum(g)=1, g>=0
    using projected gradient with FFT-based matrix-vector products.
    Returns optimal g (the best response direction).
    """
    n = len(a)
    a_norm = a / a.sum()
    
    # Initialize with uniform
    g = np.ones(n) / n
    
    best_g = g.copy()
    v = conv_cross(a_norm, g)
    best_obj = float(np.sum(v ** 2))
    
    # Estimate Lipschitz constant for step size
    # L = 2 * ||A||^2 ≈ 2 * (sum(a))^2 * 2 (rough bound)
    lr = 1.0 / (2.0 * float(np.sum(a_norm ** 2)) * 2 * n + 1e-10)
    lr = min(lr, 1.0 / (4.0 * n))
    
    for it in range(max_iters):
        v = conv_cross(a_norm, g)
        grad = 2.0 * aptv(a_norm, v)
        
        # Line search
        improved = False
        for ls in range(15):
            g_new = project_simplex(g - lr * grad)
            v_new = conv_cross(a_norm, g_new)
            obj_new = float(np.sum(v_new ** 2))
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_g = g_new.copy()
                g = g_new.copy()
                lr = min(lr * 1.1, 1.0 / (4.0 * n))
                improved = True
                break
            lr *= 0.5
            lr = max(lr, 1e-15)
        
        if not improved:
            break
    
    return best_g


def solve_lp_active(a, top_m=30, time_budget=5.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for top M peaks, sum(g)=1, g>=0.
    Uses only the top M most constraining positions for speed.
    """
    n = len(a)
    a_norm = a / a.sum()
    
    # Find the top positions in conv(a, a) as guide for active constraints
    conv_aa = compute_conv(a_norm)
    # Use positions near the maximum as active constraints
    top_idx = np.argsort(conv_aa)[::-1][:top_m]
    
    # Try with more constraints too
    top_idx_ext = np.argsort(conv_aa)[::-1][:min(2n-1, top_m + 20)]
    
    # Build constraint matrix for selected rows
    # A[k, j] = a_norm[k-j] if 0 <= k-j < n else 0
    rows = sorted(set(top_idx_ext.tolist()))
    
    A_sub = np.zeros((len(rows), n))
    for i, k in enumerate(rows):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_sub[i, j] = a_norm[idx]
    
    # Solve LP with cvxpy
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_sub @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    start = time.monotonic()
    solvers = [cp.CLARABEL, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((len(rows), n + 1))
        A_ub[:, :n] = A_sub
        A_ub[:, n] = -1.0
        b_ub = np.zeros(len(rows))
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', time_limit=time_budget)
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    
    return None, None

def subgradient_polish(a, time_budget=30.0):
    """
    Final polishing with exact subgradient method.
    Uses the true subgradient at the argmax of conv.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    it = 0
    avg_a = a.copy()
    avg_obj = best_obj

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        g = subgradient_at_peak(a, k_star)
        
        # Subgradient step with projection
        a_new = project_simplex(a - lr * g)
        obj_new = compute_obj(a_new)
        
        # Polyak averaging
        avg_a = (it * avg_a + a_new) / (it + 1)
        avg_obj = compute_obj(avg_a)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        elif avg_obj < best_obj - 1e-14:
            best_obj = avg_obj
            best_a = avg_a.copy()
            a = avg_a.copy()
        
        lr = 2.0 / np.sqrt(it + 1) / np.sqrt(n)
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (x - 0.3) ** 2) + np.exp(-5.0 * (x - 0.7) ** 2) + 0.05
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 1.0 / (0.1 + x * (1 - x) * 4)
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Concentrated in middle
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2) + 0.02
        a = a / a.sum()
    elif mode == 6:
        # U-shaped
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 + 0.05
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def coordinate_refine(a, time_budget=30.0):
    """
    Targeted coordinate refinement: identify peak-contributing pairs
    and redistribute mass to reduce the peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for rnd in range(20):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_val = conv[k_star]
        
        # Find top contributing pairs
        contributions = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions.append((a[i] * a[j], i, j))
        contributions.sort(reverse=True)
        
        # Find non-contributing positions
        contrib_set = set()
        for _, i, j in contributions:
            contrib_set.add(i)
            contrib_set.add(j)
        non_contrib = [i for i in range(n) if i not in contrib_set]
        
        if not non_contrib:
            break
        
        improved = False
        for contrib, i, j in contributions[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break
            
            src = i if a[i] >= (a[j] if i != j else 0) else j
            for target in non_contrib[:5]:
                for frac in [0.005, 0.01, 0.02, 0.04, 0.06]:
                    delta = frac * a[src]
                    if delta < 0 or delta >= a[src] * 0.5:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
            if improved:
                break
        
        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining FFT-accelerated smooth
    methods, QP direction finding, active LP, subgradient polishing, and
    multi-scale exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial best: n={n_current}, obj={best_score:.8f}")

    # === Phase 1: Fast FFT smooth optimization ===
    print(f"\n=== Phase 1: FFT Smooth Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 180.0)
    if p1_budget > 30:
        a_s = smooth_optimize_fft(best_seq, time_budget=p1_budget, K=5,
                                  beta_start=5.0, beta_end=300.0, verbose=True)
        obj_s = compute_obj(a_s)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_s.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_s:.8f}")

    # === Phase 2: QP-based LP-ALM ===
    print(f"\n=== Phase 2: QP/LP-ALM ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.35, 450.0)
    if p2_budget > 60:
        a_qp = lp_anneal_fast(best_seq, time_budget=p2_budget, verbose=True)
        obj_qp = compute_obj(a_qp)
        if obj_qp < best_score:
            best_score = obj_qp
            best_seq = a_qp.copy()
            print(f"  Phase 2: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj_qp:.8f}")

    # === Phase 3: Another round of FFT smooth (with different params) ===
    print(f"\n=== Phase 3: FFT Smooth (refined) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.15, 150.0)
    if p3_budget > 30:
        a_s2 = smooth_optimize_fft(best_seq, time_budget=p3_budget, K=3,
                                   beta_start=20.0, beta_end=1000.0, verbose=True)
        obj_s2 = compute_obj(a_s2)
        if obj_s2 < best_score:
            best_score = obj_s2
            best_seq = a_s2.copy()
            print(f"  Phase 3: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj_s2:.8f}")

    # === Phase 4: Multi-dimension exploration ===
    print(f"\n=== Phase 4: Multi-Dimension Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([n_best - 100, n_best - 50, n_best + 50, n_best + 100,
                            n_best + 200, 600, 700, 850, 950, 1000, 1100, 1200]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and n != n_best]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 80.0)
        
        if alloc > 20:
            a_opt = smooth_optimize_fft(a_try, time_budget=alloc * 0.5, K=5,
                                        beta_start=5, beta_end=200)
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_opt = lp_anneal_fast(a_opt, time_budget=min(remaining * 0.1, 40.0))
            obj_opt = compute_obj(a_opt)
        else:
            a_opt = a_try
            obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r + 1) % 7
        a_r = diverse_init(n_use, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.12, 60.0)
        if restart_budget > 25:
            a_r = smooth_optimize_fft(a_r, time_budget=restart_budget * 0.5, K=5,
                                      beta_start=5, beta_end=100)
            remaining = deadline - time.monotonic()
            if remaining > 25:
                a_r = lp_anneal_fast(a_r, time_budget=min(remaining * 0.1, 30.0))

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Subgradient polish ===
    print(f"\n=== Phase 6: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_pg = subgradient_polish(best_seq, time_budget=min(remaining * 0.15, 40.0))
        obj_pg = compute_obj(a_pg)
        if obj_pg < best_score:
            best_score = obj_pg
            best_seq = a_pg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_pg:.8f}")

    # === Phase 7: Coordinate refinement ===
    print(f"\n=== Phase 7: Coordinate Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd = coordinate_refine(best_seq, time_budget=min(remaining * 0.15, 30.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CoordDesc: obj={best_score:.8f} **")
        else:
            print(f"  CoordDesc: obj={obj_cd:.8f}")

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_final = lp_anneal_fast(best_seq, time_budget=min(remaining * 0.5, 80.0),
                                  verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # === Phase 9: Final subgradient polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_sg = subgradient_polish(best_seq, time_budget=min(remaining * 0.5, 20.0))
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final Subgrad: obj={best_score:.8f} **")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
