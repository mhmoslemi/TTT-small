Uses LP alternating with proper line search, spectral initialization,

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(2 * n - 2, j + n - 1)
        A[j:k_end + 1, j] = a[:k_end - j + 1]
    return A


def solve_lp(a):

        return np.ones(1), 1.0

    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.SCS, cp.CBC]:

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def subgradient_step(a, top_k=20):
    """Compute subgradient of max(conv(a,a)) at a (sum(a)=1)."""
    n = len(a)
    conv = compute_conv(a)
    if top_k >= len(conv):
        k_indices = np.arange(len(conv))
    else:
        k_indices = np.argsort(conv)[::-1][:top_k]

    w = np.zeros(2 * n - 1)
    w[k_indices] = 1.0 / len(k_indices)

    # grad[j] = sum_k w[k] * a[k-j] for valid indices (k-j in [0,n-1])
    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1
    A_fft = rfft(a, n_fft)
    W_fft = rfft(w, n_fft)
    R = irfft(np.conj(A_fft) * W_fft, n_fft)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = R[(n_fft - j) % n_fft]
    grad *= 2.0 * n
    grad -= np.mean(grad)
    return grad

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with golden-section line search, diverse spectral
initializations, stochastic perturbation for local-minima escape,
multi-n exploration, and subgradient polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(2 * n - 2, j + n - 1)
        A[j:k_end + 1, j] = a[:k_end - j + 1]
    return A


def solve_lp(a):
    """Solve: min M s.t. conv(a,g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None)."""
    n = len(a)
    if n < 2:
        return np.ones(1), 1.0
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Try cvxpy solvers in order of preference
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.CBC]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > 90:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    # Fallback to scipy HiGHS
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def line_search_blend(a, g):
    """Find optimal t in [0,1] for (1-t)*a + t*g via golden section search.
    Returns (best_a, best_obj, best_t)."""

    def objective(t):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        return compute_obj(a_try)

    # Evaluate endpoints and a few interior points
    f0 = objective(0.0)
    f1 = objective(1.0)
    f_mid = objective(0.5)

    # Golden section search in [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = objective(c)
    fd = objective(d)

    for _ in range(25):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = objective(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = objective(d)

    candidates = [(0.0, f0), (1.0, f1), (c, fc), (d, fd), (0.5, f_mid)]
    best_t, best_f = min(candidates, key=lambda x: x[1])

    a_best = (1.0 - best_t) * a + best_t * g
    a_best = np.maximum(a_best, 0)
    s = float(np.sum(a_best))
    if s > 1e-10:
        a_best = a_best / s
    return a_best, best_f, best_t


def lp_alternate(a, time_budget=200.0, verbose=False):

    n = len(a)

    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 10:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        a_new, obj_new, t_used = line_search_blend(a, g_lp)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP: obj={best_obj:.8f}, t={t_used:.3f}, M={M_lp:.6f}, t+={elapsed:.0f}s")
        else:
            obj_g = compute_obj(g_lp)
            if obj_g < best_obj - 1e-13:
                best_obj = obj_g
                best_a = g_lp.copy()
                a = g_lp.copy()
                stall = 0
                if verbose:
                    print(f"    LP-direct: obj={best_obj:.8f}")
            else:
                stall += 1
                if stall > 4:

                    a = best_a * np.exp(scale * noise)
                    a = np.maximum(a, 0)
                    s2 = float(np.sum(a))
                    if s2 > 1e-10:

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def subgradient_step(a, top_k=20):
    """Compute subgradient of max(conv(a,a)) at a (sum(a)=1)."""
    n = len(a)
    conv = compute_conv(a)
    if top_k >= len(conv):
        k_indices = np.arange(len(conv))
    else:
        k_indices = np.argsort(conv)[::-1][:top_k]

    w = np.zeros(2 * n - 1)
    w[k_indices] = 1.0 / len(k_indices)

    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1
    A_fft = rfft(a, n_fft)
    W_fft = rfft(w, n_fft)
    R = irfft(np.conj(A_fft) * W_fft, n_fft)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = R[(n_fft - j) % n_fft]
    grad *= 2.0 * n
    grad -= np.mean(grad)
    return grad


def subgradient_optimize(a, time_budget=30.0, top_k=15):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()

    x = a.copy()

    for it in range(30000):
        if time.monotonic() - start > time_budget:

        grad = subgradient_step(x, top_k=top_k)
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        t_k = 2.0 / (it + 2.0)
        x = a_new + (1.0 - t_k) * (a_new - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))

        a = a_new
        if it % 100 == 0:

            if obj < best_obj:

        if it % 800 == 799:
            lr *= 0.93

                continue
            a_try = a_try / s_try

        alloc = min(50.0, remaining * 0.18)
        a_opt = lp_alternate(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_opt)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            n_best_cur = n_try
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Diverse spectral initializations at best n ===
    print(f"\n=== Phase 4: Spectral Initialization (n={len(best_a)}) ===")
    remaining = deadline - time.monotonic()
    n_curr = len(best_a)

    for d in range(15):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_div = spectral_init(n_curr, rng, mode=d % 11)
        alloc = min(35.0, remaining * 0.15)
        a_opt = lp_alternate(a_div, time_budget=alloc, verbose=False)
        obj_div = compute_obj(a_opt)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Spectral {d} (mode={d % 11}): obj={best_obj:.8f} **")
        else:
            print(f"  Spectral {d} (mode={d % 11}): obj={obj_div:.8f}")

    # === Phase 5: Subgradient polish ===
    print(f"\n=== Phase 5: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        for top_k in [8, 20, 40]:
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break
            alloc = min(25.0, remaining * 0.25)
            a_sg = subgradient_optimize(best_a, time_budget=alloc, top_k=top_k)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                print(f"  SubGrad top_k={top_k}: obj={best_obj:.8f} **")
            else:
                print(f"  SubGrad top_k={top_k}: obj={obj_sg:.8f}")

    # === Phase 6: Final LP polish ===
    print(f"\n=== Phase 6: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 120.0)
        a_final = lp_alternate(best_a, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()
            print(f"  Final LP: obj={best_obj:.8f} **")

    # === Phase 7: One more round of spectral + LP if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Extra Exploration ===")
        n_curr = len(best_a)
        for d in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break
            # Try at a slightly different n too
            n_try = n_curr + rng.integers(-100, 101)
            n_try = max(200, min(3000, n_try))
            a_div = spectral_init(n_try, rng, mode=rng.integers(0, 11))
            alloc = min(30.0, remaining * 0.2)
            a_opt = lp_alternate(a_div, time_budget=alloc, verbose=False)
            obj_div = compute_obj(a_opt)
            if obj_div < best_obj:
                best_obj = obj_div
                best_a = a_opt.copy()
                print(f"  ** Extra {d} (n={n_try}): obj={best_obj:.8f} **")
            else:
                print(f"  Extra {d} (n={n_try}): obj={obj_div:.8f}")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
