Lagrangian: M + sum_k y_k (A g)_k - M sum_k y_k - lambda (1^T g - 1) - sum_j mu_j g_j

= M(1 - sum y_k) + g^T (A^T y - lambda * 1 - mu) + lambda

For the dual to be bounded, we need:
- 1 - sum(y) = 0, i.e., sum(y) = 1
- A^T y - lambda * 1 - mu = 0, i.e., (A^T y)_j >= lambda for all j (with mu_j >= 0)

Dual: maximize lambda subject to sum(y) = 1, y >= 0, (A^T y)_j >= lambda for all j.

So the dual is:
maximize lambda
subject to: sum_k y_k = 1
            y_k >= 0
            sum_k y_k * A[k,j] >= lambda for all j

The optimal lambda* equals the optimal M* (strong duality holds since the problem is feasible and bounded).

The dual variables y* give us the "optimal weights" on the peaks. This is useful because:
- It tells us which peaks are "binding" (y*_k > 0)
- The complementary slackness condition says: if (Ag)_k < M*, then y*_k = 0

Advanced optimizer combining IRL (Iteratively Reweaved L-infinity) optimization
with LP alternating and Nesterov subgradient methods.
Key innovation: IRL approach that reframes the L-infinity minimization as a
sequence of QP subproblems with adaptive reweighting, combined with FFT-based

import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate as sp_correlate
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_grad_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        m_start = k - i_end
        m_end = k - i_start
        grad[m_start:m_end + 1] = 2.0 * a[i_start:i_end + 1][::-1]
        # Fix the diagonal element (where m == k-m, i.e., 2m == k)
        if k % 2 == 0:
            m_diag = k // 2
            if 0 <= m_diag < n:
                grad[m_diag] = a[m_diag]  # Should be a[k-m_diag] = a[m_diag], but counted once
    return grad

"""

Key innovations:
1. IRL approach using FFT-based Hankel matrix-vector products for efficient QP subproblems
2. Log-sum-exp smooth optimization with adaptive temperature for global progress

4. Interleaved multi-method strategy to escape local minima

import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate as sp_correlate
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_grad_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a[j] = 2*a[k-j] for valid j."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        m_start = k - i_end
        m_end = k - i_start
        grad[m_start:m_end + 1] = 2.0 * a[i_start:i_end + 1][::-1]
    return grad


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def hankel_matvec(w, a):
    """Compute Q(w) @ a where Q[i,j] = w[i+j]. Uses FFT-based correlation."""
    return sp_correlate(w, a, mode='valid')


def irl_optimize(a0, time_budget=100.0, verbose=False):
    """
    Iteratively Reweighed L-infinity optimization.
    Alternates between solving QP subproblems and updating peak weights.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adaptive parameters
    base_step = 0.4
    no_improve_count = 0

    while time.monotonic() - start < time_budget:
        # Compute current conv and update weights
        conv = compute_conv(a)
        m = float(np.max(conv))
        if m < 1e-15:
            break

        # Soft-max weights with adaptive temperature
        elapsed = time.monotonic() - start
        frac = elapsed / time_budget
        tau = 2.0 + 50.0 * frac  # increasing sharpness over time

        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum

        # Solve QP: min a^T Q(w) a s.t. a >= 0, sum(a) = 1
        # Using projected gradient with backtracking
        a_qp = a.copy()
        objective_val = float(np.sum(a_qp * hankel_matvec(w, a_qp)))

def logsumexp_optimize(a0, time_budget=100.0, verbose=False):

    Optimize using log-sum-exp smooth approximation to the max.
    f(a) = (1/tau) * log(sum_k exp(tau * c[k]))
    """

    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.3
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        frac = elapsed / time_budget
        tau = 5.0 + 80.0 * frac  # adaptive temperature

        # Compute smooth gradient
        conv = compute_conv(a)
        m = float(np.max(conv))
        exponents = tau * (conv - m)
        # Clip for numerical stability
        exponents = np.clip(exponents, -500, 500)
        weights = np.exp(exponents)
        w_sum = np.sum(weights)
        if w_sum < 1e-30:
            weights = np.zeros(len(conv))
            weights[np.argmax(conv)] = 1.0
        else:
            weights = weights / w_sum

        # Gradient: 2 * correlate(weights, a, 'valid')
        grad = 2.0 * hankel_matvec(weights, a)

        # Nesterov momentum
        if not hasattr(logsumexp_optimize, '_y'):
            logsumexp_optimize._y = a.copy()
            logsumexp_optimize._prev_a = a.copy()

        y = logsumexp_optimize._y
        # Recompute gradient at y for Nesterov
        conv_y = compute_conv(y)
        m_y = float(np.max(conv_y))
        exponents_y = tau * (conv_y - m_y)
        exponents_y = np.clip(exponents_y, -500, 500)
        weights_y = np.exp(exponents_y)
        wsum_y = np.sum(weights_y)
        if wsum_y > 1e-30:
            weights_y = weights_y / wsum_y
        else:
            weights_y = np.zeros(len(conv_y))
            weights_y[np.argmax(conv_y)] = 1.0
        grad_y = 2.0 * hankel_matvec(weights_y, y)

        # Nesterov step
        t_k = 2.0 / (no_improve + 2.0)
        a_new = project_simplex(y - step_size * grad_y)
        y = a_new + (1.0 - t_k) * (a_new - logsumexp_optimize._prev_a)
        y = project_simplex(y)
        logsumexp_optimize._prev_a = a
        a = a_new

        obj_a = compute_obj(a)
        if obj_a < best_obj - 1e-12:
            best_obj = obj_a
            best_a = a.copy()
            no_improve = 0
            if verbose and no_improve == 0:

        else:
            no_improve += 1
            if no_improve > 100:
                step_size = max(step_size * 0.8, 0.01)
                logsumexp_optimize._y = best_a.copy()
                logsumexp_optimize._prev_a = best_a.copy()
                no_improve = 0

    # Clean up
    if hasattr(logsumexp_optimize, '_y'):
        del logsumexp_optimize._y
        del logsumexp_optimize._prev_a

    return best_a


def nesterov_subgrad(a0, time_budget=60.0, top_k=5, verbose=False):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    prev_a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05
    patience = 0
    it = 0

    while it < 500000 and time.monotonic() - start < time_budget:
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            grad += conv_grad_at_k(y, int(k))
        grad /= len(top_idx)
        grad = grad - np.mean(grad)

        t_k = 2.0 / (it + 2.0)
        a_new = project_simplex(y - step_size * grad)
        y = a_new + (1.0 - t_k) * (a_new - prev_a)
        y = project_simplex(y)
        prev_a = a
        a = a_new
        it += 1

        if it % 20 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
                if verbose and it % 200 == 0:
                    print(f"      Nest it={it}: best={best_obj:.8f}, step={step_size:.4f}")
            else:
                patience += 1
                if patience > 150:
                    step_size = max(step_size * 0.7, 0.002)
                    patience = 0
                    y = best_a.copy()
                    prev_a = best_a.copy()
                    a = best_a.copy()

        if it % 2000 == 0:
            step_size = max(step_size * 0.97, 0.003)

    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, warm_start=None):
    """
    LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_norm, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    best_solution = None
    best_M = np.inf
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS, cp.CVXOPT]:
        try:
            if warm_start is not None and solver in [cp.SCS, cp.CLARABEL]:
                prob.solve(solver=solver, warm_start=True)
            else:
                prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    gv = gv / gs
                    M_val = float(M.value)
                    if M_val < best_M:
                        best_M = M_val
                        best_solution = gv
        except Exception:
            continue

    if best_solution is not None:
        return best_solution, best_M
    return None, None


def lp_alternate(a0, time_budget=100.0, verbose=False):
    """LP alternating with 1D line search and warm starts."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_g = None

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a, warm_start=last_g)
        if g_lp is None:
            break
        last_g = g_lp

        # 1D line search
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)

        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 40, 'xatol': 0.005})
            best_t = res.x
        except Exception:
            best_t = 0.5

        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall_count = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}, t_elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 20:
                break

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def coordinate_descent(a0, time_budget=60.0, verbose=False):
    """Coordinate descent: optimize one coordinate at a time with exact line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        # Random coordinate ordering
        order = rng.permutation(n)
        improved = False
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            j = int(j)
            if a[j] < 1e-12:
                # Try to increase from zero
                for frac in [0.001, 0.005, 0.01, 0.05]:
                    a_try = a.copy()
                    total = 1.0 - a[j]
                    if total < 1e-10:
                        continue
                    # Reduce others proportionally
                    a_try[:] = a_try * (1.0 - frac)
                    a_try[j] += frac
                    a_try = np.maximum(a_try, 0)
                    st = np.sum(a_try)
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                continue

            # Try to adjust a[j]
            for delta_frac in [0.1, -0.1, 0.05, -0.05, 0.2, -0.2]:
                a_try = a.copy()
                delta = delta_frac * a[j]
                a_try[j] = a[j] + delta
                if a_try[j] < 0:
                    continue
                # Redistribute to others
                others = np.ones(n)
                others[j] = 0
                others_sum = np.sum(others)
                redistribution = -delta / others_sum
                a_try += redistribution * others
                a_try = np.maximum(a_try, 0)
                st = np.sum(a_try)
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining IRL, Log-Sum-Exp, LP alternating,
    Nesterov subgradient, and coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = hann_window(600)
        best_score = compute_obj(best_seq)
        print(f"Hann init (n=600): obj={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: Quick diverse initialization (80s) ===
    print(f"\n=== Phase 1: Diverse Initializations ===")
    inits = [best_seq.copy()]

    # Different n values
    for n_try in [n_base, max(100, n_base - 200), n_base + 200, 400, 1000]:
        if 50 <= n_try <= 3000:
            a_try = resample_seq(best_seq, n_try)
            inits.append(a_try)

    # Special shapes
    inits.append(hann_window(n_base))
    gauss = np.exp(-0.5 * ((np.arange(n_base) - n_base / 2.0) / (n_base / 4.0)) ** 2)
    inits.append(np.maximum(gauss, 0) / np.sum(gauss))
    # Asymmetric: mass at one end
    asym = np.exp(-np.arange(n_base) / (n_base * 0.2))
    inits.append(asym / np.sum(asym))
    # Two-bump
    bump1 = np.exp(-0.5 * ((np.arange(n_base) - n_base * 0.25) / (n_base * 0.1)) ** 2)
    bump2 = np.exp(-0.5 * ((np.arange(n_base) - n_base * 0.75) / (n_base * 0.1)) ** 2)
    inits.append((bump1 + bump2) / np.sum(bump1 + bump2))

    for i, init in enumerate(inits):
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        obj = compute_obj(init)
        if obj < best_score:
            best_score = obj
            best_seq = init.copy()
            print(f"  Init {i} (n={len(init)}): obj={best_score:.8f} **")
        else:
            print(f"  Init {i} (n={len(init)}): obj={obj:.8f}")
        if remaining > 15 and i < 5:
            alloc = min(8.0, remaining * 0.15)
            a_opt = irl_optimize(init, time_budget=alloc)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Init {i}+IRL: obj={best_score:.8f} **")

    # === Phase 2: IRL Optimization (350s) ===
    print(f"\n=== Phase 2: IRL Optimization ===")
    remaining = deadline - time.monotonic()
    alloc = min(350.0, remaining * 0.35)
    a_irl = irl_optimize(best_seq, time_budget=alloc, verbose=True)
    obj_irl = compute_obj(a_irl)
    if obj_irl < best_score - 1e-12:
        best_score = obj_irl
        best_seq = a_irl.copy()
        print(f"  IRL result: obj={best_score:.8f} **")
    else:
        print(f"  IRL result: obj={obj_irl:.8f}")

    # === Phase 3: Log-Sum-Exp Optimization (200s) ===
    print(f"\n=== Phase 3: Log-Sum-Exp Optimization ===")
    remaining = deadline - time.monotonic()
    alloc = min(200.0, remaining * 0.25)
    a_lse = logsumexp_optimize(best_seq, time_budget=alloc, verbose=True)
    obj_lse = compute_obj(a_lse)
    if obj_lse < best_score - 1e-12:
        best_score = obj_lse
        best_seq = a_lse.copy()
        print(f"  LSE result: obj={best_score:.8f} **")
    else:
        print(f"  LSE result: obj={obj_lse:.8f}")

    # === Phase 4: LP Alternating (400s) ===
    print(f"\n=== Phase 4: LP Alternating ===")
    for round_idx in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(70.0, remaining * 0.25)
        a_lp = lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-12:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj_lp:.8f}")
            # Perturb to escape
            n_cur = len(best_seq)
            best_seq = best_seq * np.exp(rng.normal(0, 0.02, n_cur))
            best_seq = np.maximum(best_seq, 0)
            best_seq /= np.sum(best_seq)

    # === Phase 5: IRL + Nesterov Interleaved (200s) ===
    print(f"\n=== Phase 5: IRL + Nesterov Interleaved ===")
    for k_val in [3, 5, 8]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        # Nesterov
        alloc_ns = min(40.0, remaining * 0.25)
        a_ns = nesterov_subgrad(best_seq, time_budget=alloc_ns, top_k=k_val, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score - 1e-12:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nest k={k_val}: obj={best_score:.8f} **")

        # IRL
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc_irl = min(40.0, remaining * 0.25)
        a_irl2 = irl_optimize(best_seq, time_budget=alloc_irl, verbose=True)
        obj_irl2 = compute_obj(a_irl2)
        if obj_irl2 < best_score - 1e-12:
            best_score = obj_irl2
            best_seq = a_irl2.copy()
            print(f"  IRL k={k_val}: obj={best_score:.8f} **")

    # === Phase 6: n-sweep (150s) ===
    print(f"\n=== Phase 6: n-Sweep ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best, n_best - 50, n_best + 50, n_best - 100, n_best + 100,
        n_best - 200, n_best + 200, 300, 500, 700, 900, 1200, 1500, 2000
    ]))
    n_options = [n for n in n_options if 50 <= n <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        if n_try == n_best:
            continue
        a_try = resample_seq(best_seq, n_try)
        alloc = min(12.0, remaining * 0.15)
        a_opt = irl_optimize(a_try, time_budget=alloc * 0.6)
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_opt = lp_alternate(a_opt, time_budget=min(alloc * 0.4, remaining * 0.3))
        obj_try = compute_obj(a_opt)
        if obj_try < best_score - 1e-12:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Coordinate Descent + Final LP (remaining) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(60.0, remaining * 0.3)
        a_cd = coordinate_descent(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score - 1e-12:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(40.0, remaining * 0.4)
        a_f = lp_alternate(best_seq, time_budget=alloc * 0.5, verbose=True)
        obj_f = compute_obj(a_f)
        if obj_f < best_score - 1e-12:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(30.0, remaining * 0.5)
        a_lse2 = logsumexp_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_lse2 = compute_obj(a_lse2)
        if obj_lse2 < best_score - 1e-12:
            best_score = obj_lse2
            best_seq = a_lse2.copy()
            print(f"  Final LSE: obj={best_score:.8f} **")

    # === Phase 8: Random Restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 8: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(15):
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break
            scale = 0.01 + 0.003 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr
            alloc = min(remaining * 0.25, 12.0)
            a_opt = irl_optimize(a_r, time_budget=alloc * 0.5)
            remaining = deadline - time.monotonic()
            if remaining > 5:
                a_opt = lp_alternate(a_opt, time_budget=min(alloc * 0.5, remaining * 0.4))
            obj_r = compute_obj(a_opt)
            if obj_r < best_score - 1e-12:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  R{r}: obj={best_score:.8f} **")
            else:
                print(f"  R{r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
