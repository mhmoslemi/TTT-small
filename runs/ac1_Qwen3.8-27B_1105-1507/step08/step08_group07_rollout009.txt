"""
Hybrid optimizer combining:
1. FFT-accelerated exponentiated gradient (mirror descent) with adaptive softmax temperature
2. Nesterov momentum for accelerated convergence

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_smoothed_grad(a, T):
    """
    Compute gradient of soft-max smoothed autocorrelation peak.
    Returns (grad, conv, p) where p is the softmax weight over lags.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    exp_val = np.exp(T * (conv - max_c))
    p = exp_val / np.sum(exp_val)

    # grad_i = (2/T) * sum_k p_k * a_{k-i} for valid k-i
    # Using FFT: r = convolve(a, p_reversed), then grad_i = 2*r[2n-2-i]/T
    p_rev = p[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    r = irfft(rfft(a, size) * rfft(p_rev, size), size)[:3 * n - 2]
    # r[k] = sum_j a[j] * p_rev[k-j] = sum_j a[j] * p[2n-2-k+j]
    # We want: grad_i = 2/T * sum_{j=0}^{n-1} p[i+j] * a[j] for valid i+j < 2n-1
    # = 2/T * r[2n-2-i]  (verified in analysis)
    grad = 2.0 * r[2 * n - 2 : n - 1 : -1] / T
    return grad, conv, p


def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def exponentiated_gradient_optimize(a, time_budget, T_start=30.0, T_end=300.0,
                                     lr_start=2.0, lr_end=0.1, momentum=0.85,
                                     verbose=False, seed=42):
    """
    Mirror descent (exponentiated gradient) with:
    - Adaptive softmax temperature (geometric schedule)
    - Nesterov momentum
    - Adaptive step size with backtracking
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    v = np.zeros(n)  # momentum velocity
    a_prev = a.copy()
    lr = lr_start
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(elapsed / time_budget, 1.0)
        # Geometric temperature schedule
        T = T_start * (T_end / T_start) ** progress
        # Decay learning rate
        lr_cur = lr_start * (lr_end / lr_start) ** progress

        # Nesterov: compute gradient at extrapolated point
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        ys = float(np.sum(y))
        if ys > 1e-10:
            y = y / ys
            grad, conv, p = compute_smoothed_grad(y, T)
        else:
            grad, conv, p = compute_smoothed_grad(a, T)

        # Update momentum
        v = momentum * v + grad

        # Exponentiated update (mirror step)
        step = lr_cur / (T + 1.0)
        a_new = a * np.exp(-step * v)

        # Ensure non-negativity and normalize
        a_new = np.maximum(a_new, 0)
        sn = float(np.sum(a_new))
        if sn < 1e-10 or not np.isfinite(sn):
            lr_cur *= 0.1
            a_new = a * np.exp(-step * lr_cur * v)
            a_new = np.maximum(a_new, 0)
            sn = float(np.sum(a_new))
            if sn < 1e-10:
                a_new = a.copy()
                sn = 1.0
        a_new = a_new / sn

        # Evaluate
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            # Backtracking: reduce step if no improvement for a while
            if stall % 50 == 0:
                lr_cur *= 0.5
                a_new = a.copy()  # go back
                sn = 1.0
                if stall > 200:
                    # Perturb to escape
                    a_new = a * np.exp(rng.normal(0, 0.03, n))
                    a_new = np.maximum(a_new, 0)
                    a_new = a_new / np.sum(a_new)
                    v *= 0.5
                    stall = 0

        a_prev = a
        a = a_new

        if verbose and int(elapsed * 2) % 30 == 0:
            print(f"    EG: obj={best_obj:.8f}, T={T:.1f}, lr={lr_cur:.4f}, t={elapsed:.0f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ z <= M, cp.sum(z) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    return z_val, float(M.value)
        except:
            continue

    # Fallback
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n])
    except:
        pass

    return None, None


def lp_alternating(a, time_budget, max_iters=50, verbose=False):
    """
    LP alternating (Frank-Wolfe) with adaptive mixing and 2-cycle detection.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    prev_a = None
    prev_prev_a = None
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp = result

        # 2-cycle detection
        if prev_prev_a is not None:
            if np.max(np.abs(a - prev_prev_a)) < 1e-8:
                a_break = 0.5 * (a + prev_a)
                a_break = np.maximum(a_break, 0)
                a_break = a_break / np.sum(a_break)
                obj_break = compute_obj(a_break)
                if obj_break < best_obj:
                    best_obj = obj_break
                    best_a = a_break.copy()
                a = a_break
                prev_a = None
                prev_prev_a = None
                continue

        # Try multiple mixing ratios
        best_cand_obj = best_obj
        best_cand = a.copy()
        for mix in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - mix) * a + mix * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_cand_obj:
                best_cand_obj = obj_try
                best_cand = a_try.copy()

        if best_cand_obj < best_obj - 1e-12:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            prev_a = a.copy()
            a = best_a
            prev_prev_a = prev_a
            stall = 0
            if verbose:
                print(f"    LP alt {it}: obj={best_obj:.8f}")
        else:
            stall += 1
            if stall > 10:
                # Perturb and restart
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev_prev_a = None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def peak_directed_perturb(a, strength=0.05, n_peaks=3):
    """
    Perturb a to specifically reduce the top peaks in autocorrelation.
    Pushes mass away from coordinates that contribute to peak lags.
    """
    n = len(a)
    conv = compute_conv(a)

    # Find top peak lags (excluding lag 0 which is always the sum squared)
    peak_lags = np.argsort(conv)[::-1][:n_peaks]

    # For each peak lag k*, the contribution to coordinate i is proportional to a_{|k*-i|}
    # We want to reduce the peak, so push mass from high-contribution coordinates
    pert = np.zeros(n)
    for k_star in peak_lags:
        if k_star == 0:
            continue
        # Contribution of coordinate i to peak at k_star: a_{|k_star - i|} if valid
        for i in range(n):
            j = abs(k_star - i)
            if j < n:
                pert[i] += a[j]

    # Normalize perturbation
    pert_max = np.max(np.abs(pert))
    if pert_max > 1e-10:
        pert = pert / pert_max

    # Apply: reduce high-perturbation coordinates, increase low ones
    a_new = a * (1.0 - strength * pert + strength * 0.5)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
        return a_new
    return a


def coordinate_polish(a, time_budget, verbose=False):
    """
    Randomized coordinate descent: optimize one coordinate at a time.
    For coordinate i, the objective as a function of a_i is:
    max_k [c_k + d_k * a_i] (piecewise linear, convex)
    We minimize this using ternary search (since it's convex).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    def eval_with_changed_a(a_base, idx, val):
        """Evaluate objective with a[idx] = val, renormalized."""
        a_try = a_base.copy()
        a_try[idx] = val
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        conv = compute_conv(a_try)
        return 2.0 * n * float(np.max(conv))

    for sweep in range(200):
        if time.monotonic() - start > time_budget:
            break

        # Random order of coordinates
        order = rng.permutation(n)
        improved = False

        for idx in order:
            if time.monotonic() - start > time_budget:
                break
            # Ternary search for optimal a[idx]
            lo = 0.0
            hi = a[idx] * 5.0 + 1.0 / n
            obj_lo = eval_with_changed_a(a, idx, lo)
            obj_hi = eval_with_changed_a(a, idx, hi)

            # Golden section search
            gr = (np.sqrt(5) - 1) / 2
            for _ in range(30):
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)
                f1 = eval_with_changed_a(a, idx, m1)
                f2 = eval_with_changed_a(a, idx, m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            best_val = (lo + hi) / 2
            best_f = eval_with_changed_a(a, idx, best_val)

            if best_f < best_obj - 1e-13:
                a[idx] = best_val
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_obj = best_f
                best_a = a.copy()
                improved = True

        if not improved and sweep > 20:
            break

    return best_a

                    best_seq = a_init.copy()

        if best_seq is None:
            best_seq = np.ones(800) / 800
            best_score = compute_obj(best_seq)
        print(f"Initial (n={len(best_seq)}): obj={best_score:.8f}")

    # === Phase 1: Exponentiated Gradient (400s) ===
    print(f"\n=== Phase 1: Exponentiated Gradient (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(400.0, remaining * 0.35)
    a_new = exponentiated_gradient_optimize(best_seq, time_budget=alloc,
                                             T_start=20.0, T_end=500.0,
                                             lr_start=3.0, lr_end=0.5,
                                             momentum=0.8, verbose=True)
    obj = compute_obj(a_new)
    if obj < best_score:
        best_score = obj
        best_seq = a_new.copy()
        print(f"  Phase 1 done: obj={best_score:.8f} **")
    else:
        print(f"  Phase 1 done: obj={obj:.8f}")

    # === Phase 2: LP Alternating Polish (250s) ===
    print(f"\n=== Phase 2: LP Alternating (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        alloc = min(250.0, remaining * 0.25)
        a_new = lp_alternating(best_seq, time_budget=alloc, max_iters=30, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 done: obj={obj:.8f}")

    # === Phase 3: Peak-directed + EG (200s) ===
    print(f"\n=== Phase 3: Peak-directed EG (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(200.0, remaining * 0.2)
        # Apply peak-directed perturbation then EG
        a_pert = peak_directed_perturb(best_seq, strength=0.03, n_peaks=5)
        a_new = exponentiated_gradient_optimize(a_pert, time_budget=alloc,
                                                 T_start=50.0, T_end=800.0,
                                                 lr_start=2.0, lr_end=0.3,
                                                 momentum=0.85, seed=123)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 3 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 done: obj={obj:.8f}")

    # === Phase 4: Multi-n Exploration (200s) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_parent = len(best_seq)
    n_candidates = sorted(set([
        n_parent - 100, n_parent - 50, n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200,
        500, 600, 700, 900, 1000, 1100, 1200, 1400,
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2500]
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(40.0, remaining * 0.12)
        # Quick EG optimization
        a_opt = exponentiated_gradient_optimize(a_try, time_budget=alloc,
                                                 T_start=30.0, T_end=300.0,
                                                 lr_start=2.5, lr_end=0.5,
                                                 momentum=0.8, seed=n_try)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: LP Polish on Best (150s) ===
    print(f"\n=== Phase 5: LP Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(150.0, remaining * 0.2)
        a_new = lp_alternating(best_seq, time_budget=alloc, max_iters=40, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 5 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 5 done: obj={obj:.8f}")

    # === Phase 6: More EG with different params (150s) ===
    print(f"\n=== Phase 6: EG Different Params (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(150.0, remaining * 0.2)
        # Higher momentum, different T schedule
        a_new = exponentiated_gradient_optimize(best_seq, time_budget=alloc,
                                                 T_start=100.0, T_end=2000.0,
                                                 lr_start=1.5, lr_end=0.1,
                                                 momentum=0.9, seed=999)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 6 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 6 done: obj={obj:.8f}")

    # === Phase 7: Coordinate Polish (80s) ===
    print(f"\n=== Phase 7: Coordinate Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(80.0, remaining * 0.15)
        a_new = coordinate_polish(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 7 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 7 done: obj={obj:.8f}")

    # === Phase 8: Random Restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 8: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(5, int(remaining // 40))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if r % 3 == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.03 + 0.01 * r, n_best))
            elif r % 3 == 1:
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 100)
                block_size = n_best // n_blocks
                blocks = rng.choice(n_blocks, min(3, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = si + block_size
                    a_r[si:ei] *= rng.uniform(0.5, 1.5)
            else:
                a_r = peak_directed_perturb(best_seq, strength=0.05, n_peaks=5)
                a_r = a_r * np.exp(rng.normal(0, 0.02, n_best))

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            alloc = min(remaining * 0.35, 30.0)
            a_opt = exponentiated_gradient_optimize(a_r, time_budget=alloc,
                                                     T_start=50.0, T_end=500.0,
                                                     lr_start=2.0, lr_end=0.3,
                                                     momentum=0.85, seed=1000 + r)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 9: Final short LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.6, 40.0)
        a_final = lp_alternating(best_seq, time_budget=alloc, max_iters=20)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid for the evaluator
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
