        # Time tracking: if approach exceeds budget, auto-break
        if time.time() > start_time + time_budget:
            break
    
    # Post-optimization validation and time recovery reporting
    final_eval = _evaluate_sequence(best_vectors.tolist())
    best_obj = final_eval.tolist()
    
    # Ensure consistency with IEEE standards
    valid_sequence = best_vectors.tolist()
    for idx, val in enumerate(valid_sequence):
        valid_sequence[idx] = max(0, min(1000, val))
    final_brs = 2.0 * len(valid_sequence) * np.max(np.convolve(valid_sequence, mode='full')) / (np.sum(valid_sequence)**2)
    
    if final_brs < best_obj * 0.95:  # In case of numerical instability
        best_sequence = valid_sequence
        
    # Final timer and cleanup
    time_spent = time.time() - start_time
    hours_frac = int(np.floor(time_spent/3600)) + (time_spent/100) % 1
    brs_str = f"optimize_brs:score={final_brs:.7f},time={time_spent:.0f}s"
    
    # Terminal return with reinforcements
    return valid_sequence.tolist()  # Not normalized, but evaluated correctly
