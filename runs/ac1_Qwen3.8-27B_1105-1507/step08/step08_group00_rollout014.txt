"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FISTA with FFT-accelerated smoothed gradients, LP-based direction finding,
pairwise coordinate descent, and multi-length exploration.
Improvements over previous: better step size (backtracking), line search for LP blend,
pairwise CD for fine polishing, and more aggressive exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_gradient_fft(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*Q_k(a))) using FFT.
    Returns (gradient, conv) where gradient has same shape as a.
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Softmax weights
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))

    # = 2 * (p ⊛ a)[i] in a cross-correlation sense
    # Using FFT: g = 2 * real(ifft(fft(p, N) * conj(fft(a, N))))[:n]
    N = 1
    while N < 2 * n + n:  # Need enough room to avoid circular wraparound
        N <<= 1
    
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    
    return g, conv

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))

    # Zero out invalid entries

        start = j
        end = j + n
        # conv[k] = sum_j a[j] * g[k-j] where k-j must be in [0, n-1]
        # So for fixed j, conv[k] gets contribution a[j]*g[k-j] for k in [j, j+n-1]
        # i.e., row k, col j has value a[j] if 0 <= k-j <= n-1
    # Let me redo this properly

    for j in range(n):
        for k in range(2 * n - 1):
            # conv[k] includes a[j] * g[k-j] if 0 <= k-j < n

def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    import cvxpy as cp
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None

def lp_polish(a, time_budget=100.0, verbose=False):
    """
    LP-based polishing with line search on blend factor.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Line search on blending factor t
        # Try a range of t values and pick the best
        t_candidates = []
        # Coarse grid
        for t in np.arange(0.05, 1.01, 0.05):
            t_candidates.append(t)
        # Fine grid around 0.2-0.5 (usually good range)
        for t in np.arange(0.1, 0.6, 0.02):
            t_candidates.append(t)
        
        best_t = 0.0
        best_t_obj = best_obj
        best_a_local = best_a.copy()
        
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()
        
        iter_count += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            if verbose and iter_count % 2 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t:.3f}, M_lp={M_lp:.6f}")
        else:
            # Perturb to escape
            scale = 0.01 + 0.005 * min(iter_count, 20)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))

def coordinate_descent(a, time_budget=100.0, verbose=False):
    """
    Pairwise coordinate descent: for each pair (i,j), find optimal transfer.
    Focus on high-gradient coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)
    
    # Precompute which pairs to consider
    # Focus on pairs where one coordinate is "high" and another is "low"
    # Use gradient to identify important coordinates
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Compute gradient (subgradient of max)
        max_idx = np.argmax(conv)
        
        # Subgradient at the max
        # d(max_k conv[k])/da[i] = a[i+k-(n-1)] + a[i-k+(n-1)] (if valid) for k = max_idx
        k = max_idx
        g = np.zeros(n)
        for i in range(n):
            idx1 = i + k - (n - 1)
            idx2 = i - k + (n - 1)
            if 0 <= idx1 < n:
                g[i] += a[idx1]
            if 0 <= idx2 < n and idx2 != idx1:
                g[i] += a[idx2]
        
        # Find best pair: high g[i] (should decrease) and low g[j] (should increase)
        # Or: find i with max g[i] and j with min g[j] subject to a[j] > 0
        i_max = np.argmax(g)
        
        # Try transferring from i_max to several j's
        # The change in conv when transferring delta from i to j:
        # conv[k'] changes by delta * (coeff[i,k'] - coeff[j,k'])
        # Where coeff[i,k'] = a[i+k'-(n-1)] + a[i-k'+(n-1)] (if valid)
        
        # For the peak at k=max_idx:
        # coeff[i, k] - coeff[j, k] determines if transfer helps
        ci = g[i_max]
        
        # Try several j's
        candidates = []
        for j in range(n):
            if j == i_max:
                continue
            cj = g[j]
            if a[j] > 0 or cj < ci:  # Can only increase j if it helps
                candidates.append((ci - cj, j))
        
        candidates.sort(reverse=True)
        
        improved = False
        for _, j in candidates[:5]:
            if a[i_max] < 1e-15:
                break
            # Binary search for optimal delta
            # We want to minimize max_k conv[k] after transfer
            # conv_new[k] = conv[k] + delta * (coeff_coeff[i,k] - coeff_coeff[j,k])
            # where coeff_coeff[i,k] is the coefficient of a[i] in conv[k]
            
            # Compute coefficients
            coeff_i = np.zeros(2*n-1)  # coefficient of a[i_max] in conv
            coeff_j = np.zeros(2*n-1)  # coefficient of a[j] in conv
            for k in range(2*n-1):
                idx_i1 = i_max + k - (n-1)
                idx_i2 = i_max - k + (n-1)
                if 0 <= idx_i1 < n:
                    coeff_i[k] += a[idx_i1]
                if 0 <= idx_i2 < n and idx_i2 != idx_i1:
                    coeff_i[k] += a[idx_i2]
                
                idx_j1 = j + k - (n-1)
                idx_j2 = j - k + (n-1)
                if 0 <= idx_j1 < n:
                    coeff_j[k] += a[idx_j1]
                if 0 <= idx_j2 < n and idx_j2 != idx_j1:
                    coeff_j[k] += a[idx_j2]
            
            diff = coeff_i - coeff_j  # change in conv per unit transfer
            
            # Find optimal delta in [0, a[i_max]]
            # min max_k (conv[k] + delta * diff[k])
            # Binary search
            lo, hi = 0.0, a[i_max]
            for _ in range(20):
                mid = (lo + hi) / 2
                conv_test = conv + mid * diff
                max_test = float(np.max(conv_test))
                conv_test0 = conv  # delta=0
                max_test0 = float(np.max(conv))
                # We want to find where max is minimized
                # Use ternary search since it's convex in delta
                mid1 = lo + (hi - lo) / 3
                mid2 = hi - (hi - lo) / 3
                f1 = float(np.max(conv + mid1 * diff))
                f2 = float(np.max(conv + mid2 * diff))
                if f1 < f2:
                    hi = mid2
                else:
                    lo = mid1
            
            delta = (lo + hi) / 2
            if delta > 1e-15:
                a_new = a.copy()
                a_new[i_max] -= delta
                a_new[j] += delta
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new > 0:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj - 1e-15:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        improved = True
        
        if not improved:
            # Try random pairs
            for _ in range(10):
                i, j = rng.integers(0, n, 2)
                if i == j or a[i] < 1e-15:
                    continue
                # Small random transfer
                delta = min(a[i], rng.uniform(0.001, 0.01))
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-15:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
            if not improved:
                break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.sin(2 * np.pi * x)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = 1 + 0.3 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01) / np.sum(np.maximum(a, 0.01))
    elif mode == 4:
        # Two peaks
        c1, c2 = n // 3, 2 * n // 3
        w = n // 10
        a = np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 1.5
        a = a / a.sum()
    elif mode == 6:
        # Cosine window
        x = np.linspace(-1, 1, n)
        a = 1 - x ** 2
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

d(conv[4])/da[0] = 0
d(conv[4])/da[1] = 0
d(conv[4])/da[2] = 2*a2

So the gradient of the smoothed objective (with weights p[k]):
g[0] = 2*p[0]*a0 + 2*p[1]*a1 + 2*p[2]*a2
g[1] = 2*p[1]*a0 + 2*p[2]*a1 + 2*p[3]*a2
g[2] = 2*p[2]*a0 + 2*p[3]*a1 + 2*p[4]*a2

Now let's check the FFT formula. We compute:
g_full = ifft(fft(p, N) * conj(fft(a, N))).real

Let's say N >= 3n = 9 (for n=3). p has length 5 (2n-1=5), a has length 3.

fft(p, N) = [P_0, P_1, ..., P_{N-1}] where P_k = sum_j p[j] * exp(-2*pi*i*j*k/N) for j=0..4
fft(a, N) = [A_0, A_1, ..., A_{N-1}] where A_k = sum_j a[j] * exp(-2*pi*i*j*k/N) for j=0..2

conj(fft(a, N))_k = sum_j a[j] * exp(+2*pi*i*j*k/N)

ifft(fft(p, N) * conj(fft(a, N)))[i] = (1/N) * sum_k (sum_j p[j]*exp(-2pi i jk/N)) * (sum_l a[l]*exp(2pi i lk/N)) * exp(2pi i ik/N)

= (1/N) * sum_j sum_l p[j] * a[l] * sum_k exp(2pi i k (l + i - j)/N)

= (1/N) * sum_j sum_l p[j] * a[l] * N * delta_{j, (l+i) mod N}

= sum_j sum_l p[j] * a[l] * delta_{j, (l+i) mod N}

= sum_l p[(l+i) mod N] * a[l]

For i=0: sum_l p[l] * a[l] (with l=0,1,2 since a has only 3 elements)
= p[0]*a0 + p[1]*a1 + p[2]*a2

For i=1: sum_l p[l+1] * a[l]
= p[1]*a0 + p[2]*a1 + p[3]*a2

For i=2: sum_l p[l+2] * a[l]
= p[2]*a0 + p[3]*a1 + p[4]*a2

So g_full[i] = sum_l p[l+i] * a[l] for i=0,1,2 (with p having indices 0..4).

Then the formula gives g = 2 * g_full[:n] = 2 * [p[0]*a0+p[1]*a1+p[2]*a2, p[1]*a0+p[2]*a1+p[3]*a2, p[2]*a0+p[3]*a1+p[4]*a2]

But the correct gradient is:
g[0] = 2*(p[0]*a0 + p[1]*a1 + p[2]*a2)
g[1] = 2*(p[1]*a0 + p[2]*a1 + p[3]*a2)
g[2] = 2*(p[2]*a0 + p[3]*a1 + p[4]*a2)

These match! Great, so the FFT gradient formula is correct.
