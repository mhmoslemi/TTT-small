        return None
    a = a / s

    c = compute_conv(a)
    max_c = float(np.max(c))
    # Find active peaks (within 5% of max)
    threshold = max_c * 0.95
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = [int(np.argmax(c))]
    if len(active) > num_active:
        active = np.argsort(c)[::-1][:num_active]

    # Compute pressure: p_j = sum_{k in active} a[k-j] for valid k-j
    # This tells us how much position j contributes to the peaks
    size = 1
    while size < 3 * n - 1:
        size <<= 1

    # Build the active indicator
    active_ind = np.zeros(len(c))
    for k in active:
        active_ind[int(k)] = 1.0

    # pressure_j = sum_k active_ind[k] * a[k-j]
    AI = rfft(active_ind, size)
    A = rfft(a, size)
    pressure = irfft(AI * np.conj(A), size)[:n]

    # g_j proportional to 1/(pressure_j + eps)
    eps = np.max(pressure) * 0.01 + 1e-10
    g = 1.0 / (pressure + eps)
    g = np.maximum(g, 0)
    gs = float(np.sum(g))
    if gs > 1e-10:
        g = g / gs
    else:
        g = np.ones(n) / n

    return g


def lp_alternating_heuristic(a, time_budget=60.0, verbose=False):
    """
    LP alternating using heuristic response (no cvxpy needed).
    Fast version of the original LP alternating method.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    last_t = 0.3

    while time.monotonic() - start < time_budget:
        g_lp = heuristic_response(a)
        if g_lp is None:
            break

        # Try several blend factors
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()

        step_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if 0 < last_t < 1.0:
            step_candidates = sorted(set(step_candidates +
                                          [max(0.05, last_t * 0.7),
                                           min(1.5, last_t * 1.3)]))

        for t in step_candidates:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP-heur: obj={best_obj:.8f}, t={best_t_val:.2f}, time={elapsed:.1f}s")
        else:
            # Perturb to escape
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            scale = 0.02
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            a = a / s
            last_t = 0.3

    return best_a


def peak_chasing(a, time_budget=60.0, verbose=False):
    """
    Directly chase and reduce the peak by optimal mass redistribution.
    Moves mass from high-contributing positions to low-contributing ones.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        c = compute_conv(a)
        k_star = int(np.argmax(c))
        peak_val = float(c[k_star])

        # Compute contribution of each position to the peak
        # (a*a)[k_star] = sum_j a[j] * a[k_star - j]
        contrib = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                contrib[j] = a[j] * a[kj]

        # Total contribution should equal peak_val
        # Find high and low contribution positions
        total_contrib = float(np.sum(contrib))
        if total_contrib < 1e-15:
            break

        # Normalized contribution (fraction of peak from each position)
        norm_contrib = contrib / total_contrib

        # We want to move mass from high-norm-contrib to low-norm-contrib
        # that doesn't hurt other peaks
        # Compute "safety": how much each position contributes to ALL peaks
        size = 1
        while size < 3 * n - 1:
            size <<= 1
        A = rfft(a, size)
        C = rfft(c, size)
        # total_contribution_j = sum_k c_k_weighted * a[k-j]
        # Actually, let's just use the peak and nearby peaks
        # Safety: low is good (safe to reduce)
        # Use the convolution with the top few peaks
        top_k = min(5, 2 * n - 1)
        peak_weights = np.zeros(2 * n - 1)
        top_indices = np.argsort(c)[::-1][:top_k]
        for idx in top_indices:
            peak_weights[int(idx)] = c[int(idx)]
        peak_weights /= np.sum(peak_weights)

        PW = rfft(peak_weights, size)
        xcorr = irfft(PW * np.conj(A), size)[:n]

        # Move mass: from positions with high norm_contrib AND high xcorr
        # to positions with low xcorr
        danger = norm_contrib * (xcorr / (np.max(xcorr) + 1e-10))

        # Source: highest danger positions
        # Target: lowest xcorr positions
        source_idx = np.argsort(danger)[::-1][:5]
        target_idx = np.argsort(xcorr)[:5]

        improved = False
        for src in source_idx:
            for tgt in target_idx:
                if src == tgt:
                    continue
                # Try small transfer
                for frac in [0.005, 0.01, 0.02]:
                    delta = frac * a[int(src)]
                    if delta <= 0 or delta > a[int(src)]:
                        continue
                    a[int(src)] -= delta
                    a[int(tgt)] += delta
                    a = np.maximum(a, 0)
                    s_new = float(np.sum(a))
                    if s_new > 1e-10:
                        a = a / s_new
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[int(src)] += delta
                        a[int(tgt)] -= delta
                        a = a / float(np.sum(a))
            if improved:
                break

        if not improved:
            # Small random perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e4) % (2**31))
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = 0.005 * min(a[int(i)], a[int(j)])
                if delta > 0:
                    a[int(i)] -= delta
                    a[int(j)] += delta
                    a = np.maximum(a, 0)
                    a = a / float(np.sum(a))
                    new_obj = compute_obj(a)
                    if new_obj < best_obj:
                        best_obj = new_obj
                        best_a = a.copy()

    return best_a


def generate_initialization(kind, n, rng):
    """Generate diverse initialization of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.1
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 8)
        a[:edge] = 0.2
        a[-edge:] = 0.2
        return a / np.sum(a)
    elif kind == 'bimodal':
        a = np.ones(n) * 0.3
        a[n // 4] = 1.5
        a[3 * n // 4] = 1.5
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'spectral_flat':
        # Try to create a sequence with flat power spectrum
        a = np.ones(n)
        # Random phase in frequency domain
        size = 1
        while size < n:
            size <<= 1
        freqs = rfft(a, size)
        magnitudes = np.abs(freqs)
        phases = rng.uniform(0, 2 * np.pi, size // 2 + 1)
        new_freqs = magnitudes * np.exp(1j * phases)
        a_new = irfft(new_freqs, size)[:n]
        a_new = np.abs(a_new) + 0.01
        return a_new / np.sum(a_new)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: soft-max GD + heuristic LP + peak chasing + length sweep.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Quick diverse initializations (60s) ===
    print(f"\n=== Phase 1: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    init_kinds = ['cosine', 'gaussian', 'plateau', 'bimodal', 'spectral_flat', 'random']

    for kind in init_kinds:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        a_init = generate_initialization(kind, n_parent, rng)
        # Quick soft-max GD on each init (5s each)
        a_opt = softmax_gradient_descent(a_init, time_budget=min(5.0, remaining * 0.1))
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Init '{kind}': obj={obj:.8f} **")
        else:
            print(f"  Init '{kind}': obj={obj:.8f}")

    a = best_seq.copy()
    print(f"  Best after Phase 1: obj={best_score:.8f}")

    # === Phase 2: Intensive Soft-max GD (500s) ===
    print(f"\n=== Phase 2: Intensive Soft-max GD ===")
    remaining = deadline - time.monotonic()
    sg_budget = min(500.0, remaining * 0.4)

    # Multiple rounds with different parameters
    sg_configs = [
        {'t_init': 30, 't_max': 500, 'lr_init': 0.08, 'beta': 0.8},
        {'t_init': 50, 't_max': 1000, 'lr_init': 0.05, 'beta': 0.85},
        {'t_init': 100, 't_max': 2000, 'lr_init': 0.03, 'beta': 0.9},
        {'t_init': 20, 't_max': 300, 'lr_init': 0.1, 'beta': 0.75},
    ]

    for cfg_idx, cfg in enumerate(sg_configs):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(sg_budget / (len(sg_configs) - cfg_idx), remaining * 0.5)
        print(f"  SoftMax GD round {cfg_idx} (t={cfg['t_init']}, lr={cfg['lr_init']}): ")
        a_new = softmax_gradient_descent(a, time_budget=alloc,
                                          t_init=cfg['t_init'], t_max=cfg['t_max'],
                                          lr_init=cfg['lr_init'], beta=cfg['beta'],
                                          verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  ** SoftMax R{cfg_idx}: obj={best_score:.8f} **")
        else:
            print(f"  SoftMax R{cfg_idx}: obj={obj:.8f}")
            # Perturb for next round
            scale = 0.02
            a = best_seq * np.exp(rng.normal(0, scale, n_parent))
            a = np.maximum(a, 0)
            a = a / np.sum(a)

    # === Phase 3: Heuristic LP Alternating (200s) ===
    print(f"\n=== Phase 3: Heuristic LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        lp_budget = min(200.0, remaining * 0.2)
        a_lp = lp_alternating_heuristic(a, time_budget=lp_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  ** LP-heur: obj={best_score:.8f} **")
        else:
            print(f"  LP-heur: obj={obj_lp:.8f}")

    # === Phase 4: Peak Chasing (100s) ===
    print(f"\n=== Phase 4: Peak Chasing ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        pc_budget = min(100.0, remaining * 0.12)
        a_pc = peak_chasing(a, time_budget=pc_budget, verbose=True)
        obj_pc = compute_obj(a_pc)
        if obj_pc < best_score:
            best_score = obj_pc
            best_seq = a_pc.copy()
            a = a_pc.copy()
            print(f"  ** PeakChase: obj={best_score:.8f} **")
        else:
            print(f"  PeakChase: obj={obj_pc:.8f}")

    # === Phase 5: Length Sweep (250s) ===
    print(f"\n=== Phase 5: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 200),
        max(200, n_best - 100),
        max(200, n_best - 50),
        max(200, n_best - 10),
        n_best,
        n_best + 10,
        n_best + 50,
        n_best + 100,
        n_best + 200,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Optimize at this length
        alloc_sg = min(20.0, remaining * 0.3)
        alloc_lp = min(15.0, remaining * 0.2)

        a_opt = softmax_gradient_descent(a_try, time_budget=alloc_sg,
                                          t_init=50, t_max=500, lr_init=0.05)
        if alloc_lp > 0:
            a_opt = lp_alternating_heuristic(a_opt, time_budget=alloc_lp)

        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final intensive soft-max (remaining time) ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    a_final = best_seq.copy()

    while remaining > 20:
        # Alternate between soft-max GD and peak chasing
        alloc1 = min(remaining * 0.4, 60.0)
        alloc2 = min(remaining * 0.2, 30.0)

        a_new = softmax_gradient_descent(a_final, time_budget=alloc1,
                                          t_init=100, t_max=3000, lr_init=0.02, beta=0.92)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_final = a_new.copy()
            print(f"  Final SoftMax: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_new2 = peak_chasing(a_final, time_budget=alloc2)
            obj2 = compute_obj(a_new2)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_new2.copy()
                a_final = a_new2.copy()
                print(f"  Final PeakChase: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            # Perturb and try again
            scale = 0.01
            a_final = best_seq * np.exp(rng.normal(0, scale, n_final))
            a_final = np.maximum(a_final, 0)
            a_final = a_final / np.sum(a_final)

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
