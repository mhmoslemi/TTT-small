import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_ab(a, b):
    """Compute conv(a, b) using FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp_for_response(a):
    """Solve: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M, dual_y) or (None, None, None)."""
    n = len(a)
    a = a / np.sum(a)
    
    # Build the constraint matrix A (2n-1 x n)
    # A[k, i] = a[k-i] if 0<=k-i<n else 0
    # For efficiency, we can use the fact that this is a convolution matrix
    # But for the LP, we need the explicit matrix
    
    # Actually, for n=800, the matrix is 1599 x 800 = ~1.3M entries
    # That's fine for HiGHS
    
    A = np.zeros((2*n-1, n))
    for i in range(n):
        # Column i: A[k, i] = a[k-i] for k = i, i+1, ..., i+n-1
        k_start = i
        k_end = min(2*n-2, i+n-1)
        length = k_end - k_start + 1
        A[k_start:k_end+1, i] = a[:length]
    
    # LP: min [0,...,0,1] @ [g, M]
    # s.t. A@g - M <= 0 (i.e., [A, -I]@[g,M] <= 0)
    #      sum(g) = 1
    #      g >= 0, M >= 0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.eye(2*n-1)])
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs < 1e-10:

"""

with dual-guided gradient polishing, adaptive blending, and multi-n exploration.
Key differences from previous approach:

- Adaptive perturbation based on dual information
- Better time management with progressive refinement

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    """Compute linear convolution of a and b using FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(seq):
    """Build matrix A (2n-1 x n) such that A @ g = conv(seq, g)."""
    n = len(seq)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = seq
    return A


def solve_response_lp(seq):
    """Solve: min M s.t. conv(seq,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M, dual_y) or (None, None, None)."""
    n = len(seq)
    s = float(np.sum(seq))
    if s < 1e-10:
        return None, None, None
    seq_norm = seq / s

    A = build_conv_matrix(seq_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual': False})
    except Exception:
        return None, None, None

    if res.status != 0:
        return None, None, None

    g = np.maximum(res.x[:n], 0)
    gs = float(np.sum(g))
    if gs < 1e-10:
        return None, None, None
    g = g / gs
    M = float(res.x[n])

    # Extract dual variables
    dual_y = None
    try:
        if hasattr(res.ineqlin, 'marginals'):
            raw_dual = np.array(res.ineqlin.marginals[:2 * n - 1])
            dual_y = np.maximum(raw_dual, 0)
            dy_sum = float(np.sum(dual_y))
            if dy_sum < 1e-12:
                dual_y = None
            else:
                dual_y = dual_y / dy_sum
    except Exception:
        dual_y = None

    if dual_y is None:
        # Fallback: uniform over active constraints
        conv_val = A @ g
        tol = max(1e-10, 1e-8 * M)
        active = np.where(np.abs(conv_val - M) < tol)[0]
        if len(active) > 0:
            dual_y = np.zeros(2 * n - 1)
            dual_y[active] = 1.0 / len(active)
        else:
            dual_y = np.ones(2 * n - 1) / (2 * n - 1)

    return g, M, dual_y


def dual_guided_gradient(seq, dual_y):
    """Compute gradient s[i] = 2 * sum_k dual_y[k] * seq[k-i] for valid indices.
    Returns array of length n."""
    n = len(seq)
    # s[i] = sum_{j=0}^{n-1} seq[j] * dual_y[i+j] (where i+j < 2n-1)
    # = conv(flip(seq), dual_y)[n-1 + i]
    seq_flipped = seq[::-1]
    conv_sy = compute_conv_ab(seq_flipped, dual_y)
    s = conv_sy[n - 1:2 * n - 1]
    return 2.0 * s


def smooth_subgradient(seq, temperature=0.1, top_k=50):
    """Compute a smooth approximation of the subgradient of max(conv(seq,seq))."""
    n = len(seq)
    conv = compute_conv(seq)
    
    if top_k >= len(conv):
        weights = np.ones(len(conv)) / len(conv)
    else:
        # Softmax-style weights over top peaks
        sorted_idx = np.argsort(conv)[::-1][:top_k]
        values = conv[sorted_idx]
        # Temperature-based weights
        exp_vals = np.exp((values - values[0]) / (temperature + 1e-15))
        weights = np.zeros(len(conv))
        weights[sorted_idx] = exp_vals / exp_vals.sum()
    
    # Gradient: s[i] = 2 * sum_k weights[k] * seq[k-i]
    return dual_guided_gradient(seq, weights)


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0)
    s = float(np.sum(v))
    if s < 1e-15:
        return np.full(len(v), target_sum / len(v))
    return v * (target_sum / s)

def subgradient_optimize(a, time_budget=60.0, top_k=30, temperature=0.05, verbose=False):

    n = len(a)

            lr *= 0.95
            a = a_new  # still update to explore

        if it % 500 == 499:
            lr = max(lr, 0.01 / n)
            temperature *= 0.95

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng):
    """Generate diverse structured initialization."""
    choice = rng.integers(0, 7)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-30 * (x - 0.5) ** 2)
    elif choice == 1:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * rng.integers(2, 8) * x)
        a = np.maximum(a, 0.01)
    elif choice == 2:
        c1, c2 = rng.uniform(0.2, 0.4), rng.uniform(0.6, 0.8)
        w = rng.uniform(5, 20)
        a = np.exp(-w * (x - c1) ** 2) + np.exp(-w * (x - c2) ** 2)
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            center = rng.integers(0, n)
            width = rng.integers(max(1, n // 20), max(2, n // 4))
            for i in range(n):
                d = abs(i - center)
                a[i] += rng.standard_normal() * 0.5 * np.exp(-d / width)
        a = np.maximum(a, 0.01)
    elif choice == 4:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    elif choice == 5:
        # Bilinear
        a = np.abs(x - 0.3) + np.abs(x - 0.7)
        a = 1.0 / (a + 0.1)
    else:
        # Smooth random via cosine series
        coeffs = rng.normal(0, 1, min(n, 50))
        a = np.ones(n)
        for k in range(1, min(n, 50)):
            a += coeffs[k] * np.cos(2 * np.pi * k * x) * (1.0 / k)
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: symmetric LP alternating with dual-guided gradient polish,
    multi-n exploration, and diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        n_base = len(a)
        print(f"Parent loaded: n={n_base}, obj={compute_obj(a):.8f}")
    except Exception:
        a =np.ones(800) / 800
        n_base = 800
        print(f"No parent, using uniform: n={n_base}, obj={compute_obj(a):.8f}")

    best_a = a.copy()
    best_obj = compute_obj(a)
    print(f"Initial best: obj={best_obj:.8f}")

    # === Phase 1: Primary symmetric LP alternating ===
    print(f"\n=== Phase 1: Symmetric LP Alternating (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.40, 500.0)
    a_p1 = lp_alternate_symmetric(best_a, time_budget=p1_budget, verbose=True, max_per_lp=3)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  Phase 1 result: obj={best_obj:.8f} **")
    else:
        print(f"  Phase 1 result: obj={obj_p1:.8f}")

    # === Phase 2: Subgradient polish ===
    print(f"\n=== Phase 2: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for params in [(20, 0.05, 80), (50, 0.1, 60), (10, 0.02, 40)]:
            top_k, temp, alloc = params
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(alloc, remaining * 0.3)
            a_sg = subgradient_optimize(best_a, time_budget=alloc, top_k=top_k, temperature=temp)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg
                print(f"  SG (top_k={top_k}, t={temp}): obj={best_obj:.8f} **")
            else:
                print(f"  SG (top_k={top_k}, t={temp}): obj={obj_sg:.8f}")

    # === Phase 3: LP alternating round 2 ===
    print(f"\n=== Phase 3: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.25, 250.0)
        a_p3 = lp_alternate_symmetric(best_a, time_budget=p3_budget, verbose=True, max_per_lp=5)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            print(f"  Phase 3 result: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 3 result: obj={obj_p3:.8f}")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        n_best,
        n_best + 100, n_best + 200, n_best + 400,
        n_best - 100, n_best - 200,
    ]))
    n_options = [x for x in n_options if 200 <= x <= 4000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == n_best:
            continue  # already optimized

        a_try = resample_seq(best_a, n_try)
        a_try = project_simplex(a_try)

        alloc = min(40.0, remaining * 0.2)
        a_opt = lp_alternate_symmetric(a_try, time_budget=alloc, max_per_lp=2)
        obj_try = compute_obj(a_opt)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    for d in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_div = diverse_init(n_best, rng)
        alloc = min(35.0, remaining * 0.15)
        a_opt = lp_alternate_symmetric(a_div, time_budget=alloc, max_per_lp=2)
        obj_div = compute_obj(a_opt)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse {d}: obj={obj_div:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # More aggressive LP alternating with more gradient steps
        alloc_lp = min(remaining * 0.5, 120.0)
        a_final = lp_alternate_symmetric(best_a, time_budget=alloc_lp, verbose=True, max_per_lp=5)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.8f} **")

    # Subgradient micro-polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_sg = min(remaining * 0.3, 50.0)
        a_sg = subgradient_optimize(best_a, time_budget=alloc_sg, top_k=20, temperature=0.03)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  Final SG: obj={best_obj:.8f} **")

    # One more LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_last = min(remaining * 0.5, 40.0)
        a_last = lp_alternate_symmetric(best_a, time_budget=alloc_last, max_per_lp=5)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
