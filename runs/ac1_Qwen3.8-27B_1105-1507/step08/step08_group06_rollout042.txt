import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    a_n = a / s
    conv = fast_conv(a_n)
    return 2.0 * n * float(np.max(conv))

def soft_gradient(a, p, n):
    """Compute g[i] = 2 * sum_j a[j] * p[i+j] using FFT."""
    # a_rev = a reversed
    a_rev = a[::-1]
    # convolve a_rev with p
    total_len = n + len(p) - 1  # = n + 2n - 1 - 1 = 3n - 2
    size = 1
    while size < total_len:
        size <<= 1
    fa = fft(a_rev, size)
    fp = fft(p, size)
    conv_result = ifft(fa * fp, size)[:total_len]
    # g[i] = 2 * conv_result[i + n - 1]
    g = 2.0 * conv_result[n-1 : n-1+n]
    return g

        conv = fast_conv(a)
        max_conv = float(np.max(conv))

    best_obj = compute_obj(a)

    x = a.copy()  # Nesterov lookahead

        p = np.exp(logits)

def lp_direction(a, timeout=8.0):
    """
    Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Returns optimal g or None.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None
    a = a / s
    M_mat = build_conv_matrix(a)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [M_mat @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([M_mat, -np.ones((m, 1))])
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': timeout})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass

    return None


def lp_interate(a, time_budget, verbose=False):
    """LP alternating with adaptive blending."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    it = 0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        it += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        g = lp_direction(a, timeout=min(6.0, remaining - 3))
        if g is None:
            break

        improved = False
        for t in [0.15, 0.25, 0.4, 0.6, 0.8]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                if verbose:
                    print(f"    LP: it={it}, t={t:.2f}, obj={best_obj:.8f}")
                break

        if not improved and it % 5 == 0:
            # Perturb to escape
            scale = 0.01 + 0.005 * min(it // 5, 10)
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def generate_init(kind, n, rng):
    """Generate diverse initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n, dtype=float)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 8)
        for i in range(edge):
            frac = i / max(1, edge)
            a[i] = 0.4 + 0.6 * frac
            a[-1 - i] = 0.4 + 0.6 * frac
        return a / np.sum(a)
    elif kind == 'u_shape':
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.5 * x ** 2
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'two_peak':
        a = np.ones(n) * 0.3
        a[n // 4] = 1.0
        a[3 * n // 4] = 1.0
        k = max(3, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    else:
        return np.ones(n) / n

    rng = np.random.default_rng(seed)

    # Load parent
    best_seq = None
    best_score = np.inf
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Diverse initializations with quick optimization (90s) ===
    print(f"\n=== Phase 1: Diverse Initializations (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 90:
        init_kinds = ['cosine', 'sine', 'gaussian', 'plateau', 'u_shape', 'two_peak', 'random']
        per_init = min(12.0, remaining / (len(init_kinds) + 2) * 0.7)
        for kind in init_kinds:
            remaining = deadline - time.monotonic()

            a_init = generate_init(kind, n_parent, rng)
            obj_init = compute_obj(a_init)

        best_score = obj_main
        best_seq = a_main.copy()
        print(f"  After Nesterov: obj={best_score:.8f} **")
    else:
        print(f"  After Nesterov: obj={obj_main:.8f}")

    # === Phase 3: LP iteration (25% of remaining) ===
    print(f"\n=== Phase 3: LP Alternating ===")

    lp_budget = min(remaining * 0.25, 200.0)

                                          lr_init=0.02)
