import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import cvxpy as cp
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp(a, time_limit=30):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    
    # Use scipy linprog (more reliable than cvxpy for this size)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.ones((1, n + 1))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

        g = cp.Variable(n, nonneg=True)

        prob.solve(solver=cp.CLARABEL)

            gs = float(np.sum(g_val))

        pass

def log_sum_exp_obj(a, beta):

    a = a / np.sum(a)
    conv = compute_conv(a)

    max_c = np.max(conv)

        return 0.0
    scaled = beta * (conv - max_c)
    lse = max_c + np.log(np.sum(np.exp(scaled))) / beta
    return 2.0 * n * lse

def mirror_descent_optimize(a, time_budget=60.0, eta0=0.1, verbose=False):
    """
    Mirror descent (exponentiated gradient) for the self-convolution minimization.
    Uses the entropy regularized Bregman divergence.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta0
    patience = 0
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute current convolution
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        obj = 2.0 * n * max_c
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 200:
            eta *= 0.9
            patience = 0
            if eta < 1e-6:
                break
        
        # Gradient step in mirror descent
        # The Bregman gradient for entropy is: a_new[m] ∝ a[m] * exp(-eta * grad[m])
        # grad[m] = 2n * sum_k 1[k in argmax] * (a[k-m] + a[m]) * 1[0<=k-m<n]
        # Use all peaks with weight proportional to how close they are to max
        
        # Soft max weights
        threshold = max_c * 0.95
        peak_mask = conv >= threshold
        if not np.any(peak_mask):
            peak_mask = np.array([True] + [False] * (len(conv) - 1))
        
        # Compute gradient
        grad = np.zeros(n)
        peak_indices = np.where(peak_mask)[0]
        for k in peak_indices:
            k_int = int(k)
            wk = conv[k_int] / max_c  # weight proportional to peak height
            m_min = max(0, k_int - (n - 1))
            m_max = min(n - 1, k_int)
            for m in range(m_min, m_max + 1):
                grad[m] += wk * (a[k_int - m] + a[m])
        
        grad *= 2.0 * n
        
        # Mirror descent update
        log_a = np.log(a)
        log_a -= eta * grad
        # Normalize to sum = 1
        log_a -= np.max(log_a)  # for numerical stability
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)
        
        # Damping
        a = 0.7 * a + 0.3 * a_new
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD: it={it}, obj={best_obj:.8f}, eta={eta:.6f}, time={elapsed:.0f}s")
    
    return best_a


def lp_alterate_improved(a, time_budget=100.0, verbose=False):
    """Improved LP alternating with better perturbation and line search."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp(a, time_limit=20)
        if g_lp is None:
            break
        
        # Line search over blend parameter
        best_local_obj = np.inf
        best_local_a = best_a.copy()
        
        t_values = np.linspace(0.05, 1.0, 20)
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
        
        if best_local_obj < best_obj - 1e-13:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 5:
                # More aggressive perturbation
                scale = min(0.1 + 0.05 * stall, 0.5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
    
    return best_a


def lbfsgs_optimize(a, time_budget=60.0, beta=50.0, verbose=False):
    """L-BFGS optimization with log-sum-exp smoothing."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64).copy()
    s = float(np.sum(a))
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # We optimize in log-space with the constraint sum(exp(x)) = 1
    # Use the softmax parameterization: a = softmax(x)
    x = np.log(a + 1e-30)
    
    def objective_and_grad(x_flat):
        a_local = np.exp(x_flat - np.max(x_flat))
        a_local = a_local / np.sum(a_local)
        conv = compute_conv(a_local)
        max_c = np.max(conv)
        if max_c <= 0:
            return 2.0 * n * 1e-30, np.zeros(n)
        
        # Log-sum-exp
        scaled = beta * (conv - max_c)
        max_scaled = 0.0
        sum_exp = np.sum(np.exp(scaled - max_scaled))
        lse = max_c + (max_scaled + np.log(sum_exp)) / beta
        obj_val = 2.0 * n * lse
        
        # Gradient of lse w.r.t. conv
        weights = np.exp(scaled - max_scaled) / sum_exp
        
        # Gradient w.r.t. a
        grad_a = np.zeros(n)
        significant = np.where(weights > 1e-10)[0]
        for k in significant:
            k_int = int(k)
            wk = weights[k_int]
            m_min = max(0, k_int - (n - 1))
            m_max = min(n - 1, k_int)
            for m in range(m_min, m_max + 1):
                grad_a[m] += wk * (a_local[k_int - m] + a_local[m])
        grad_a *= 2.0 * n
        
        # Gradient w.r.t. x (softmax chain rule)
        # a_i = exp(x_i) / sum(exp(x))
        # d(a_i)/d(x_j) = a_i * (delta_ij - a_j)
        # grad_x[j] = sum_i grad_a[i] * a_i * (delta_ij - a_j)
        #           = grad_a[j] * a_j - a_j * sum_i grad_a[i] * a_i
        grad_x = grad_a * a_local - a_local * np.dot(grad_a, a_local)
        
        return obj_val, grad_x
    
    # Adaptive beta: start high, decrease
    current_beta = beta
    for round_ in range(5):
        if time.monotonic() - start > time_budget:
            break
        
        def obj_wrapper(x_flat):
            nonlocal current_beta
            return objective_and_grad(x_flat)[0]
        
        def grad_wrapper(x_flat):
            nonlocal current_beta
            return objective_and_grad(x_flat)[1]
        
        # Constraint: sum(exp(x)) = 1 is handled by softmax parameterization
        # But we still need to ensure the result is valid
        try:
            res = minimize(obj_wrapper, x, jac=grad_wrapper, method='L-BFGS-B',
                          options={'maxiter': 200, 'maxfun': 500,
                                   'ftol': 1e-12, 'gtol': 1e-8})
            
            x_opt = res.x
            a_opt = np.exp(x_opt - np.max(x_opt))
            a_opt = a_opt / np.sum(a_opt)
            a_opt = np.maximum(a_opt, 0)
            
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                x = x_opt
            else:
                break
        except Exception:
            break
        
        # Decrease beta for next round
        current_beta = max(current_beta * 0.5, 5.0)
    
    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    from scipy.interpolate import PchipInterpolator
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    try:
        interp = PchipInterpolator(x_old, a)
        a_new = interp(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    
    if choice == 0:
        # Gaussian
        a = np.exp(-50 * (x - 0.5) ** 2)
    elif choice == 1:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 2:
        # Two Gaussians
        a = np.exp(-100 * (x - 0.3) ** 2) + np.exp(-100 * (x - 0.7) ** 2)
    elif choice == 3:
        # Power law
        a = x ** rng.uniform(0.3, 3.0) + (1-x) ** rng.uniform(0.3, 3.0)
    elif choice == 4:
        # Random smooth (sum of Gaussians)
        a = np.ones(n) * 0.1
        for _ in range(rng.integers(2, 6)):
            center = rng.uniform(0.1, 0.9)
            width = rng.uniform(0.05, 0.3)
            amp = rng.uniform(0.5, 2.0)
            a += amp * np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif choice == 5:
        # Fourier-based (ensure non-negative)
        coeffs = rng.standard_normal(10) * 0.1
        a = 1.0
        for k in range(1, 10):
            a += coeffs[k] * np.cos(2 * np.pi * k * x)
        a = np.maximum(a, 0.01)
    elif choice == 6:
        # Triangular
        a = 1 - np.abs(2 * x - 1)
    else:
        # Uniform with noise
        a = 1 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * x)
        a = np.maximum(a, 0.01)
    
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
