import numpy as np

import time

def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_and_peak(a):
    """Return (conv, peak_value, peak_indices) for normalized a."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, np.inf, np.array([])
    a_n = a / s
    conv = compute_conv(a_n)
    peak = float(np.max(conv))
    # Find all indices within 1% of peak
    thresh = peak * 0.99
    peak_idx = np.where(conv >= thresh)[0]
    return conv, peak, peak_idx


def smoothed_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*conv(a,a)))) w.r.t. a.
    Uses FFT for efficiency.
    """
    n = len(a)
    conv = compute_conv(a)
    # Stable softmax
    conv_shifted = conv - np.max(conv)
    exp_vals = np.exp(beta * conv_shifted)
    p = exp_vals / np.sum(exp_vals)
    
    # grad[m] = 2 * sum_{j=0}^{n-1} a[j] * p[j+m] for m = 0..n-1
    size = 1
    while size < 4 * n:
        size <<= 1
    F_p = rfft(p, size)
    F_a = rfft(a, size)
    # corr2[k] = sum_j p[j+k] * a[j]
    corr2 = irfft(F_p * np.conj(F_a), size)
    grad = 2.0 * corr2[:n]
    return grad


def mirror_descent_smoothed(a0, time_budget=100.0, beta_start=2.0, beta_end=80.0,
                            eta_start=0.3, eta_end=0.01, verbose=False):

    Mirror descent with smoothed max objective.
    a_new = a * exp(-eta * grad) / sum(...)
    """

    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    n_iters = 0
    patience = 0
    max_patience = 500
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        # Anneal beta
        beta = beta_start + (beta_end - beta_start) * progress
        # Anneal eta
        eta = eta_start * (eta_end / eta_start) ** progress
        
        try:
            grad = smoothed_gradient(a, beta)
        except Exception:
            break
        
        # Center gradient (for simplex constraint)
        grad_centered = grad - np.mean(grad)
        
        # Mirror descent update
        a_raw = a * np.exp(-eta * grad_centered)
        s_raw = np.sum(a_raw)
        if s_raw < 1e-30:
            a_new = np.ones(n) / n
        else:
            a_new = a_raw / s_raw
        a_new = np.maximum(a_new, 0)
        a_new = a_new / np.sum(a_new)

            obj = compute_obj(a_new)

                best_a = a_new.copy()
                patience = 0
            else:

                a_new = a  # revert

                # Reduce eta and try again

                if eta < 0.001:

        a = a_new
        
        if verbose and n_iters % 500 == 0:

            print(f"    MD: it={n_iters}, obj={best_obj:.8f}, beta={beta:.1f}, eta={eta:.4f}, t={elapsed:.1f}s")
    
    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=100.0, verbose=False):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_good_t = 0.3
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        # 1D line search
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)
        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 25, 'xatol': 0.01})
            best_t = res.x if res.success else last_good_t
        except Exception:
            best_t = last_good_t
        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            last_good_t = best_t
            stall_count = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 20:
                break
    
    return best_a

                continue

            # Compute impact on peak
            # Moving delta from i to j changes conv[k] by:

            # If we change a[i] -> a[i]-delta and a[j] -> a[j]+delta:

            best_i_obj = best_obj
            best_i_delta = 0

def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def raised_cosine(n, p=1.0):
    """Raised cosine family: (1 + cos(2*pi*(i+0.5)/n))^p normalized."""
    x = np.arange(n)
    w = (1.0 + np.cos(2.0 * np.pi * (x + 0.5) / n)) ** p
    w = np.maximum(w, 1e-15)
    return w / np.sum(w)


def gaussian_seq(n, sigma_frac=0.3):
    """Gaussian centered at n/2."""
    x = np.arange(n)
    sigma = n * sigma_frac
    w = np.exp(-0.5 * ((x - n / 2.0) / sigma) ** 2)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def plateau_seq(n, edge_frac=0.1, sharpness=2.0):
    """Plateau in middle, tapered edges."""
    x = np.arange(n) / (n - 1.0)
    # Taper at edges
    left = x / edge_frac
    right = (1.0 - x) / edge_frac
    edge = np.minimum(left, right)
    w = np.where(edge > 1.0, 1.0, edge ** sharpness)
    w = np.maximum(w, 1e-15)
    return w / np.sum(w)


def propose_candidate():
    """
    Main search: Smoothed-max mirror descent + LP alternating + coordinate descent.
    Uses FFT-based softmax gradient for efficient optimization on the simplex.
    """
    start_time = time.monotonic()
    deadline = start_time + 1500.0 - 30.0
    rng = np.random.default_rng(12345)
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = hann_window(800)
        best_score = compute_obj(best_seq)
        print(f"Hann init (n=800): obj={best_score:.8f}")
    
    n_base = len(best_seq)
    
    # === Phase 1: Diverse inits + quick mirror descent (150s) ===
    print(f"\n=== Phase 1: Diverse Initializations (150s) ===")
    phase1_end = start_time + 150.0
    inits = [
        ("parent", best_seq.copy()),
        ("hann", hann_window(n_base)),
        ("gauss_0.25", gaussian_seq(n_base, 0.25)),
        ("gauss_0.35", gaussian_seq(n_base, 0.35)),
        ("gauss_0.5", gaussian_seq(n_base, 0.5)),
        ("plateau_0.1", plateau_seq(n_base, 0.1, 2.0)),
        ("plateau_0.15", plateau_seq(n_base, 0.15, 3.0)),
        ("raised_cos_1", raised_cosine(n_base, 1.0)),
        ("raised_cos_2", raised_cosine(n_base, 2.0)),
        ("raised_cos_0.5", raised_cosine(n_base, 0.5)),
        ("random", None),
    ]
    inits[-1] = ("random", rng.random(n_base))
    inits[-1] = ("random", inits[-1][1] / np.sum(inits[-1][1]))
    
    for name, init in inits:
        remaining = phase1_end - time.monotonic()
        if remaining < 10:
            break
        obj = compute_obj(init)
        if obj < best_score:
            best_score = obj
            best_seq = init.copy()
            print(f"  {name} (n={n_base}): obj={best_score:.8f} **")
        else:
            print(f"  {name} (n={n_base}): obj={obj:.8f}")
        
        if remaining > 20:
            alloc = min(12.0, remaining * 0.3)
            a_md = mirror_descent_smoothed(init, time_budget=alloc,
                                           beta_start=2.0, beta_end=40.0,
                                           eta_start=0.2, eta_end=0.02)
            obj_md = compute_obj(a_md)
            if obj_md < best_score:
                best_score = obj_md
                best_seq = a_md.copy()
                print(f"  {name}+MD: obj={best_score:.8f} **")
    
    # === Phase 2: Mirror descent main (300s) ===
    print(f"\n=== Phase 2: Mirror Descent Main (300s) ===")
    phase2_end = start_time + 450.0
    a = best_seq.copy()
    
    # Multiple passes with different parameters
    md_configs = [
        {"beta_start": 1.0, "beta_end": 30.0, "eta_start": 0.3, "eta_end": 0.03},
        {"beta_start": 5.0, "beta_end": 60.0, "eta_start": 0.15, "eta_end": 0.01},
        {"beta_start": 2.0, "beta_end": 100.0, "eta_start": 0.25, "eta_end": 0.005},
    ]
    
    for cfg in md_configs:
        remaining = phase2_end - time.monotonic()
        if remaining < 30:
            break
        alloc = min(90.0, remaining * 0.4)
        a_new = mirror_descent_smoothed(a, time_budget=alloc, verbose=True, **cfg)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  MD main: obj={best_score:.8f} **")
        else:
            print(f"  MD main: obj={obj_new:.8f}")
            # Keep best
            a = best_seq.copy()
    
    # === Phase 3: LP alternating (250s) ===
    print(f"\n=== Phase 3: LP Alternating (250s) ===")
    phase3_end = start_time + 700.0
    a = best_seq.copy()
    n_lp_rounds = 5
    for r in range(n_lp_rounds):
        remaining = phase3_end - time.monotonic()
        if remaining < 40:
            break
        alloc = min(60.0, remaining * 0.35)
        a_new = lp_alternate(a, time_budget=alloc, verbose=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"  LP R{r}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{r}: obj={obj_new:.8f}")
            # Perturb to escape
            a = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
            a = np.maximum(a, 0)
            a /= np.sum(a)
    
    # === Phase 4: n-sweep (150s) ===
    print(f"\n=== Phase 4: n-Sweep (150s) ===")
    phase4_end = start_time + 850.0
    n_options = [400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000]
    n_options = sorted(set(n_options + [n_base, max(200, n_base - 200), n_base + 200]) &
                       set(range(200, 2501)))
    
    for n_try in n_options:
        remaining = phase4_end - time.monotonic()
        if remaining < 15:
            break
        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
        alloc = min(15.0, remaining * 0.2)
        a_md = mirror_descent_smoothed(a_try, time_budget=alloc,
                                       beta_start=2.0, beta_end=50.0,
                                       eta_start=0.2, eta_end=0.02)
        obj_try = compute_obj(a_md)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_md.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 5: Coordinate descent (150s) ===
    print(f"\n=== Phase 5: Coordinate Descent (150s) ===")
    phase5_end = start_time + 1000.0
    remaining = phase5_end - time.monotonic()
    if remaining > 30:
        alloc = min(120.0, remaining * 0.6)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, n_sweeps=100)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score - 1e-12:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")
    
    # === Phase 6: Combined polishing (250s) ===
    print(f"\n=== Phase 6: Combined Polishing (250s) ===")
    phase6_end = start_time + 1250.0
    a = best_seq.copy()
    
    # Alternate between MD and LP
    for round_idx in range(6):
        remaining = phase6_end - time.monotonic()
        if remaining < 30:
            break
        alloc_md = min(40.0, remaining * 0.25)
        alloc_lp = min(40.0, remaining * 0.25)
        
        # Mirror descent step
        a_md = mirror_descent_smoothed(a, time_budget=alloc_md,
                                       beta_start=10.0, beta_end=100.0,
                                       eta_start=0.1, eta_end=0.005)
        obj_md = compute_obj(a_md)
        if obj_md < best_score - 1e-12:
            best_score = obj_md
            best_seq = a_md.copy()
            a = a_md
            print(f"  P6 MD R{round_idx}: obj={best_score:.8f} **")
        else:
            a = best_seq.copy()
        
        # LP step
        a_lp = lp_alternate(a, time_budget=alloc_lp)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-12:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp
            print(f"  P6 LP R{round_idx}: obj={best_score:.8f} **")
    
    # === Phase 7: Random restarts + final polish (remaining) ===
    print(f"\n=== Phase 7: Random Restarts ===")
    n_best = len(best_seq)
    for r in range(15):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        scale = 0.01 + 0.005 * r
        a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
        a_r = np.maximum(a_r, 0)
        sr = float(np.sum(a_r))
        if sr < 1e-10:
            continue
        a_r = a_r / sr
        alloc = min(remaining * 0.25, 20.0)
        a_md = mirror_descent_smoothed(a_r, time_budget=alloc * 0.5,
                                       beta_start=5.0, beta_end=80.0,
                                       eta_start=0.15, eta_end=0.01)
        a_lp = lp_alternate(a_md, time_budget=alloc * 0.5)
        obj_r = compute_obj(a_lp)
        if obj_r < best_score - 1e-12:
            best_score = obj_r
            best_seq = a_lp.copy()
            print(f"  R{r}: obj={best_score:.8f} **")
        else:
            print(f"  R{r}: obj={obj_r:.8f}")
    
    # === Final: one more LP + MD on best ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.5, 60.0)
        a_final = lp_alternate(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-12:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
    
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.6, 40.0)
        a_final2 = mirror_descent_smoothed(best_seq, time_budget=alloc,
                                           beta_start=20.0, beta_end=150.0,
                                           eta_start=0.05, eta_end=0.001)
        obj_final2 = compute_obj(a_final2)
        if obj_final2 < best_score - 1e-12:
            best_score = obj_final2
            best_seq = a_final2.copy()
            print(f"  Final MD: obj={best_score:.8f} **")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
