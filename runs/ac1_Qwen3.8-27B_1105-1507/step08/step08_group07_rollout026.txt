3. Add a small random perturbation proportional to the dual variables

Another improvement: **use warm-starting for the LP solver**. If we have a previous solution, we can provide it as a warm start to the solver, which can speed things up significantly.

Let me also consider using **interior point methods** directly on the non-convex problem. The problem is:
min M
s.t. (a*a)(k) ≤ M ∀k
sum(a) = 1
a ≥ 0

This is a quadratically constrained problem. We could use a sequential quadratic programming (SQP) approach, but that's complex to implement.

Instead, let me use a **penalty method**:
min M + μ * sum_k max(0, (a*a)(k) - M)^2
s.t. sum(a) = 1, a ≥ 0

For large μ, this approximates the original problem. We can solve this with scipy's SLSQP or trust-constr.

Actually, for the penalty method, the inner problem (fixing M) is:
min μ * sum_k max(0, (a*a)(k) - M)^2
s.t. sum(a) = 1, a ≥ 0

This is a smooth (C^1) optimization problem with a convex constraint set. We can use projected gradient or L-BFGS-B with the sum constraint handled by normalization.

import numpy as np

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ z gives cross-conv a*z."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. A@z <= M, sum(z)=1, z>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except:
                        pass
                    return z_val, float(M.value), dual_y
        except:
            continue
    
    return None, None, None

def smoothed_gradient_descent(a, time_budget=100.0, tau_start=20.0, tau_end=200.0, verbose=False):

        tau = tau_start * (tau_end / tau_start) ** progress

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        scaled = (conv - max_conv) * tau
        scaled = np.clip(scaled, -500, 500)
        exp_vals = np.exp(scaled)
        total_exp = np.sum(exp_vals)

        improved = False

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    t = np.linspace(0, 1, n)

    inits.append(np.sin(np.pi * t) ** 2)

    inits.append(rng.exponential(1.0, n) + 0.1)
    # Random (uniform)

    # Alternating (high-low pattern)
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.7
    inits.append(alt)
    
    return inits

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ z gives cross-conv a*z."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. A@z <= M, sum(z)=1, z>=0. Returns (z, M, dual)."""
    n = len(a)
    A = build_conv_matrix(a)
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except Exception:
                        pass
                    return z_val, float(M.value), dual_y
        except Exception:
            continue
    return None, None, None


def lp_alternating(a, time_budget=200.0, verbose=False):
    """LP alternating with 2-cycle detection, dual-guided escape, and momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    prev_a = None
    prev_prev_a = None
    stall = 0
    cycle_count = 0
    momentum = a.copy()
    iters = 0

    while time.monotonic() - start < time_budget:
        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp, dual_y = result
        iters += 1

        # Update momentum
        momentum = 0.5 * momentum + 0.5 * g_lp
        momentum = np.maximum(momentum, 0)
        ms = float(np.sum(momentum))
        if ms > 1e-10:
            momentum = momentum / ms

        # 2-cycle detection
        cyclic = False
        if prev_prev_a is not None:
            diff = np.max(np.abs(a - prev_prev_a))
            if diff < 1e-6:
                cyclic = True

        if cyclic:
            escaped = False
            # Strategy 1: Average
            a_avg = 0.5 * (a + prev_a)
            a_avg = np.maximum(a_avg, 0)
            a_avg = a_avg / np.sum(a_avg)
            obj_avg = compute_obj(a_avg)
            if obj_avg < best_obj - 1e-12:

                a = a_avg

                        a_new = np.maximum(a_new, 0)
                        a_new = a_new / np.sum(a_new)
                        obj_new = compute_obj(a_new)
                        if obj_new < best_obj - 1e-12:

                            a = a_new

                            break

                        a = a_new

                a = best_a.copy()

            cycle_count += 1
            prev_a = None
            prev_prev_a = None
            momentum = a.copy()
            stall = 0
            if verbose and cycle_count % 3 == 0:
                print(f"    [Cycle {cycle_count}] obj={best_obj:.8f}")
            continue

        # Normal update: try multiple targets and mixing ratios
        best_cand_obj = best_obj
        best_cand = a.copy()
        targets = [g_lp, momentum]

                a_try = np.maximum(a_try, 0)

            stall += 1

                a = best_a * np.exp(rng.normal(0, scale, n))

def smoothed_gradient_descent(a, time_budget=100.0, verbose=False):
    """Gradient descent on smoothed (log-sum-exp) objective with momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    vel = np.zeros(n)
    beta = 0.9
    base_lr = 0.05
    iters = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(elapsed / time_budget, 1.0)
        tau = 15.0 * (300.0 / 15.0) ** progress

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        scaled = (conv - max_conv) * tau
        scaled = np.clip(scaled, -500, 500)
        exp_vals = np.exp(scaled)
        total_exp = np.sum(exp_vals)
        weights = exp_vals / total_exp

        # Gradient via FFT cross-correlation
        size = 1
        while size < 3 * n:
            size <<= 1
        fa = rfft(a, size)
        fw = rfft(weights, size)
        cross_corr = irfft(np.conj(fa) * fw, size)
        grad = 4.0 * n * cross_corr[:n]
        grad = grad - np.mean(grad)

        # Momentum
        vel = beta * vel + (1.0 - beta) * grad
        grad_eff = vel

        # Line search
        lr = base_lr * (1.0 - 0.5 * progress)
        improved = False
        for _ in range(15):
            a_try = a - lr * grad_eff
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                lr *= 0.5
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                no_improve = 0
                break
            lr *= 0.5

        if not improved:
            no_improve += 1
            if no_improve > 100:
                # Restart with perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                vel = np.zeros(n)
                no_improve = 0

        iters += 1
        if verbose and iters % 30 == 0:
            print(f"    Grad: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.5f}, t={elapsed:.0f}s")

    return best_a


def fast_coordinate_descent(a, time_budget=60.0, verbose=False):
    """Coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    conv = compute_conv(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    best_conv = conv.copy()
    start = time.monotonic()
    rng = np.random.default_rng(777)
    sweep = 0

    while time.monotonic() - start < time_budget:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget + 5:
                break

            for frac in [0.1, 0.3, 0.5, -0.1, -0.3]:
                delta = frac * a[i]
                if delta == 0:
                    continue
                # Update conv incrementally: conv[k] changes by 2*delta*a[k-i]
                conv_try = conv.copy()
                lo = i
                hi = min(2 * n - 1, i + n)
                conv_try[lo:hi] += 2.0 * delta * a[i:hi - i]

                a_new_val = a[i] + delta
                if a_new_val < 0:
                    continue
                s_new = float(np.sum(a)) + delta
                if s_new < 1e-10:
                    continue

                max_try = float(np.max(conv_try)) / (s_new * s_new)
                obj_try = 2.0 * n * max_try

                if obj_try < best_obj - 1e-13:
                    a[i] += delta
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    best_conv = conv.copy()
                    improved = True

        sweep += 1
        if not improved and sweep > 8:
            break
        if verbose and sweep % 5 == 0:
            # Renormalize periodically
            a = a / np.sum(a)
            conv = compute_conv(a)
            best_a = a.copy()
            print(f"    CoordDesc: sweep={sweep}, obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    t = np.linspace(0, 1, n)

    # Various window shapes

    inits.append(np.sin(np.pi * t) ** 2)
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(6 * np.pi * t))

    # Flat with tapered edges
    edge = max(5, n // 8)
    flat = np.ones(n)
    ramp = np.arange(edge, dtype=float) / edge
    flat[:edge] = ramp
    flat[n - edge:] = ramp[::-1]
    inits.append(flat)

    # Gaussian-like (narrow and wide)
    inits.append(np.exp(-8 * (t - 0.5) ** 2))
    inits.append(np.exp(-2 * (t - 0.5) ** 2))

    # Two bumps
    inits.append(np.exp(-15 * (t - 0.3) ** 2) + np.exp(-15 * (t - 0.7) ** 2))

    # Power law decay from center
    center = np.abs(t - 0.5)
    inits.append(1.0 / (1.0 + 10 * center))

    # Linear triangle
    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    inits.append(np.maximum(tri, 0.01))

    # Random
    inits.append(10 ** rng.uniform(-1.5, 0.5, n))
    inits.append(rng.exponential(1.0, n) + 0.05)
    inits.append(rng.uniform(0.1, 1.0, n))

    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.3
    alt[1::2] = 0.8
    inits.append(alt)

    # Chirp-like (use absolute value of a sinusoid with varying frequency)
    phase = 2 * np.pi * (5 * t ** 2)
    inits.append(0.5 + 0.5 * np.abs(np.sin(phase)))

    return inits

        400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000
    ]))
    n_values = [n for n in n_values if 300 <= n <= 2500]
    n_values.sort(key=lambda x: abs(x - n_parent))

    time_per_n = max(5.0, min(12.0, remaining * 0.25 / len(n_values)))

    for n_try in n_values:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        inits = generate_initializations(n_try, rng)
        inits.append(resample_sequence(best_seq, n_try))

        best_local = None
        best_local_obj = np.inf

        for init in inits:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            init = np.maximum(init, 0.001)
            init = init / np.sum(init)
            obj_init = compute_obj(init)
            if obj_init < best_local_obj:
                best_local_obj = obj_init
                best_local = init.copy()

        if best_local is not None and best_local_obj < best_score:
            best_score = best_local_obj
            best_seq = best_local.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f}")

        # Short LP from best local init
        remaining = deadline - time.monotonic()
        if remaining > 10 and best_local is not None:
            alloc = min(time_per_n, remaining * 0.08)
            if alloc > 3:
                a_opt = lp_alternating(best_local, time_budget=alloc)
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  ** n={n_try}: LP obj={best_score:.8f}")

    # === Phase 2: Main LP alternating (350s) ===
    print(f"\n=== Phase 2: Main LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alloc = min(350.0, remaining * 0.3)
        a_new = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 2 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 LP: obj={obj:.8f} (no improvement)")

    # === Phase 3: Gradient descent polish (150s) ===
    print(f"\n=== Phase 3: Gradient Descent (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(150.0, remaining * 0.15)
        a_new = smoothed_gradient_descent(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Gradient: obj={best_score:.8f} **")
        else:
            print(f"  Gradient: obj={obj:.8f}")

    # === Phase 4: LP after gradient (150s) ===
    print(f"\n=== Phase 4: LP After Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(150.0, remaining * 0.15)
        a_new = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 4 LP: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4 LP: obj={obj:.8f}")

    # === Phase 5: Gradient again (80s) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: Gradient Round 2 ===")
        alloc = min(80.0, remaining * 0.1)
        a_new = smoothed_gradient_descent(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Grad2: obj={best_score:.8f} **")
        else:
            print(f"  Grad2: obj={obj:.8f}")

    # === Phase 6: Multi-n fine-tuning (120s) ===
    print(f"\n=== Phase 6: Multi-n Fine-tuning ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    nearby_ns = sorted(set([
        n_best - 100, n_best - 50, n_best - 25,
        n_best + 25, n_best + 50, n_best + 100, n_best + 200,
    ]))
    nearby_ns = [n for n in nearby_ns if 300 <= n <= 3000 and n != n_best]

    for n_try in nearby_ns:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(25.0, remaining * 0.12)
        a_opt = lp_alternating(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Coordinate descent polish (60s) ===
    print(f"\n=== Phase 7: Coord Descent Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(60.0, remaining * 0.15)
        a_new = fast_coordinate_descent(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  CoordDesc: obj={best_score:.8f} **")
        else:
            print(f"  CoordDesc: obj={obj:.8f}")

    # === Phase 8: Final combined polish (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Final Polish (remaining={remaining:.0f}s) ===")
        # Alternating short LP + gradient
        sub_budget = min(remaining * 0.4, 50.0)
        if sub_budget > 10:
            a_new = lp_alternating(best_seq, time_budget=sub_budget)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Final LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        sub_budget = min(remaining * 0.4, 40.0)
        if sub_budget > 5:
            a_new = smoothed_gradient_descent(best_seq, time_budget=sub_budget)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Final Grad: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        sub_budget = min(remaining * 0.3, 30.0)
        if sub_budget > 5:
            a_new = lp_alternating(best_seq, time_budget=sub_budget)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
