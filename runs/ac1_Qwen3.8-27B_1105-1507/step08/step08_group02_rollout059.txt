    # Shift for numerical stability

    u = np.sort(v)[::-1]

    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1.0)
    rho_final = np.where(cond)[0][-1]
    theta = (cssv[rho_final] - 1.0) / (rho_final + 1)
    return np.maximum(v - theta, 0.0)

and adaptive dimension exploration.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 with a >= 0.

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    b = compute_conv(a_norm)
    return 2.0 * n * float(np.max(b))


def gradient_smoothed(a, beta):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*b[k]))) w.r.t. a.
    Returns (gradient, max_b, weights).
    """
    n = len(a)
    b = compute_conv(a)
    m = b.max()
    shifted = beta * (b - m)
    exp_b = np.exp(shifted)
    total = exp_b.sum()
    w = exp_b / total  # softmax weights over convolution

    fw = rfft(w_rev, size)
    conv_result = irfft(fa * fw, size)

    # Indices: 2n-2, 2n-3, ..., n-1
    g = 2.0 * conv_result[n - 1:2 * n - 1][::-1]

    return g, m, w

    m = b.max()
    shifted = beta * (b - m)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_optimize(a, time_budget, beta_start=15.0, beta_end=8000.0,
                    eta=0.005, verbose=False):
    """
    Smoothed gradient optimization using log-sum-exp surrogate.
    Works in log-space (mirror descent on simplex).
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 5.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    patience = 60

    it = 0
    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        beta = beta_start * (beta_end / beta_start) ** (frac * 0.6)

        g_conv, m_b, w = gradient_smoothed(a, beta)
        g_entropy = eta * (np.log(np.maximum(a, 1e-30)) + 1.0)
        g = g_conv / beta + g_entropy

        # Momentum
        v = momentum * v + g

        # Mirror descent step in log-space
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.5
            it += 1
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            # Slowly increase learning rate
            lr = min(lr * 1.01, 2.0)
        else:
            stall += 1
            if stall > patience:
                # Annealing restart
                rng = np.random.default_rng((it * 2654435761 + n) % (2**31))
                noise_scale = min(0.15, 0.03 + 0.02 * (stall // patience))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                lr = max(0.5 / np.sqrt(n), lr * 0.4)
                momentum = 0.5
                stall = 0
            else:
                lr = max(0.1 / np.sqrt(n), lr * 0.97)

        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"      Smooth it{it}: obj={best_obj:.8f}, beta={beta:.1f}, "
                  f"lr={lr:.4f}, t={elapsed:.1f}s")

    return best_a


def subgradient_step(a, topK=5, lr=0.1):
    """
    Single subgradient step using top-K peak indices.
    Returns new a (not yet normalized to sum=1, caller should normalize).
    """
    n = len(a)
    b = compute_conv(a)
    S = a.sum()
    M = b.max()

    # Get top K peak indices
    top_idx = np.argsort(b)[::-1][:topK]
    weights = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        weights[k] = 1.0 / topK

    # Subgradient of M w.r.t. a[m] = sum_k weights[k] * 2*a[k-m] (if valid)
    sg = np.zeros(n)
    for k in top_idx:
        k = int(k)
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                sg[m] += (1.0 / topK) * 2.0 * a[j]

    # Gradient of f = 2*n*M/S^2 w.r.t. a[m]:
    # df/da[m] = 2*n * (sg[m]/S - 2*M/S^2)
    grad = 2.0 * n * (sg / S - 2.0 * M / (S * S))

    # Proportional descent (multiplicative)
    a_new = a * np.exp(-lr * grad / (lr * np.abs(grad).max() + 1e-10))
    a_new = np.maximum(a_new, 1e-15)
    return a_new


def subgradient_optimize(a, time_budget, topK=5, verbose=False):
    """Subgradient optimization with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.5 / np.sqrt(n)
    stall = 0

    it = 0
    while time.monotonic() - start < time_budget:
        a_new = subgradient_step(a, topK=topK, lr=lr)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            lr = min(lr * 1.02, 2.0)
        else:
            stall += 1
            if stall > 80:
                rng = np.random.default_rng((it * 12345 + n) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = 0.2 / np.sqrt(n)
                stall = 0
            else:
                lr = max(0.01 / np.sqrt(n), lr * 0.96)

        it += 1
        if verbose and it % 100 == 0:
            print(f"      Subgrad it{it}: obj={best_obj:.8f}, lr={lr:.5f}, "
                  f"t={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_descent(a, time_budget=30.0):
    """
    Coordinate descent: try small adjustments to individual or pairs of coordinates.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    iterations = 0
    while time.monotonic() - start < time_budget:
        # Pick a random pair to swap mass between
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try moving a small amount from i to j or vice versa
        amounts = [0.001, 0.005, 0.01, 0.02, 0.05]
        for frac in amounts:
            delta = frac * min(a[i], a[j])
            if delta < 1e-12:
                continue

            # Move from i to j
            a[i] -= delta
            a[j] += delta
            obj_try = compute_obj(a)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a.copy()
                break
            else:
                a[i] += delta
                a[j] -= delta

            # Move from j to i
            a[j] -= delta
            a[i] += delta
            obj_try = compute_obj(a)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a.copy()
                break
            else:
                a[j] += delta
                a[i] -= delta

        a = best_a.copy()
        iterations += 1

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
    elif mode == 2:
        # Exponential decay
        a = np.exp(-np.linspace(0, 3, n))
    elif mode == 3:
        # Cosine taper
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 4:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x - np.pi))
    elif mode == 5:
        # Front-loaded
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2 + 0.05
    elif mode == 6:
        # Back-loaded
        x = np.linspace(0, 1, n)
        a = x ** 2 + 0.05
    elif mode == 7:
        # Gaussian bump
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) / 0.05)
    elif mode == 8:
        # Sine wave
        x = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + 0.5 * np.sin(x)
    elif mode == 9:
        # Polynomial
        x = np.linspace(0, 1, n)
        a = 1.0 - x + 0.5 * x * (1 - x)
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def multi_scale_optimize(a, time_budget, verbose=False):
    """
    Multi-scale optimization: start with low beta (coarse), increase to high beta (fine).
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Coarse to fine scales
    scales = [
        (10, 100, 0.3),    # beta 10->100, 30% of time
        (50, 500, 0.25),   # beta 50->500, 25% of time
        (200, 2000, 0.25), # beta 200->2000, 25% of time
        (1000, 10000, 0.2),# beta 1000->10000, 20% of time
    ]

    for beta_s, beta_e, frac in scales:
        alloc = time_budget * frac
        if time.monotonic() - start + alloc > time_budget:
            alloc = time_budget - (time.monotonic() - start) - 1.0
        if alloc < 5:
            break

        result = smooth_optimize(best_a, time_budget=alloc,
                                  beta_start=beta_s, beta_end=beta_e, eta=0.003)
        obj = compute_obj(result)
        if obj < best_obj:
            best_obj = obj
            best_a = result.copy()
            if verbose:
                print(f"      MultiScale beta[{beta_s},{beta_e}]: obj={best_obj:.8f}")

    return best_a


def alternating_optimize(a, time_budget, verbose=False):
    """
    Alternate between smoothed gradient and subgradient steps.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall = 0
    mode = 0  # 0 = smooth, 1 = subgradient

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        step_budget = min(remaining, max(5.0, remaining * 0.05))

        if mode == 0:
            result = smooth_optimize(best_a, time_budget=step_budget,
                                      beta_start=50, beta_end=2000, eta=0.002)
        else:
            result = subgradient_optimize(best_a, time_budget=step_budget, topK=5)

        obj = compute_obj(result)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = result.copy()
            stall = 0
        else:
            stall += 1
            if stall > 4:
                # Switch mode
                mode = 1 - mode
                stall = 0
                # Add perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
                best_a = best_a * np.exp(rng.normal(0, 0.05, n))
                best_a = np.maximum(best_a, 1e-15)
                best_a = best_a / best_a.sum()
                best_obj = compute_obj(best_a)

        mode = 1 - mode

        if verbose:
            print(f"      Alt mode={mode}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines multi-scale smoothed optimization,
    subgradient method, coordinate descent, and multi-dimension exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 50 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent loaded: n={parent.size}, obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init: n={n0}, obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial: n={n_current}, score={best_score:.8f}")

    # === Phase 1: Multi-scale optimization on current best ===
    print(f"\n=== Phase 1: Multi-scale Optimization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 350.0)
    if p1_budget > 30:
        result = multi_scale_optimize(best_seq, time_budget=p1_budget, verbose=True)
        obj = compute_obj(result)
        if obj < best_score:
            best_score = obj
            best_seq = result.copy()
            print(f"  Phase 1 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 result: obj={obj:.8f} (no improvement)")

    # === Phase 2: Alternating optimization ===
    print(f"\n=== Phase 2: Alternating Optimization ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.20, 250.0)
    if p2_budget > 20:
        result = alternating_optimize(best_seq, time_budget=p2_budget, verbose=True)
        obj = compute_obj(result)
        if obj < best_score:
            best_score = obj
            best_seq = result.copy()
            print(f"  Phase 2 result: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 result: obj={obj:.8f} (no improvement)")

    # === Phase 3: Diverse restarts from different initializations ===
    print(f"\n=== Phase 3: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_seq)
    modes_to_try = [2, 3, 5, 7, 0, 4]  # exponential, cosine, front-loaded, gaussian, uniform, raised-cos

    for i, mode in enumerate(modes_to_try):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # Generate diverse init and blend with best
        a_init = diverse_init(n_use, rng, mode=mode)
        blend = 0.4 + 0.1 * i
        a_init = (1 - blend) * a_init + blend * best_seq
        a_init = np.maximum(a_init, 1e-10)
        a_init = a_init / a_init.sum()

        budget_i = min(remaining * 0.12, 120.0)
        result = multi_scale_optimize(a_init, time_budget=budget_i)
        obj = compute_obj(result)
        if obj < best_score:
            best_score = obj
            best_seq = result.copy()
            print(f"  Restart mode={mode}: obj={best_score:.8f} **")
        else:
            print(f"  Restart mode={mode}: obj={obj:.8f}")

    # === Phase 4: Dimension exploration ===
    print(f"\n=== Phase 4: Dimension Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = []
    for delta in [-200, -100, -50, 50, 100, 200, 300, 500]:
        n_try = n_best + delta
        if 300 <= n_try <= 2500 and n_try != n_best:
            n_options.append(n_try)
    # Add some fixed sizes
    for n_fixed in [500, 600, 700, 900, 1000, 1200, 1500]:
        if 300 <= n_fixed <= 2500 and n_fixed != n_best and n_fixed not in n_options:
            n_options.append(n_fixed)

    n_options = sorted(set(n_options))[:8]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        budget_i = min(remaining * 0.10, 100.0)

        result = multi_scale_optimize(a_try, time_budget=budget_i)
        obj = compute_obj(result)
        if obj < best_score:
            best_score = obj
            best_seq = result.copy()
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj:.8f}")

    # === Phase 5: Intensive polish on best ===
    print(f"\n=== Phase 5: Intensive Polish ===")
    remaining = deadline - time.monotonic()

    # High-beta smooth optimization
    if remaining > 30:
        alloc = min(remaining * 0.3, 200.0)
        result = smooth_optimize(best_seq, time_budget=alloc,
                                  beta_start=500, beta_end=15000, eta=0.001, verbose=True)
        obj = compute_obj(result)
        if obj < best_score:
            best_score = obj
            best_seq = result.copy()
            print(f"  Polish (high-beta): obj={best_score:.8f} **")

    # Subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.25, 100.0)
        result = subgradient_optimize(best_seq, time_budget=alloc, topK=3)
        obj = compute_obj(result)
        if obj < best_score:
            best_score = obj
            best_seq = result.copy()
            print(f"  Polish (subgrad): obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.2, 50.0)
        result = coordinate_descent(best_seq, time_budget=alloc)
        obj = compute_obj(result)
        if obj < best_score:
            best_score = obj
            best_seq = result.copy()
            print(f"  Polish (coord): obj={best_score:.8f} **")

    # Final smooth pass
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 100.0)
        result = smooth_optimize(best_seq, time_budget=alloc,
                                  beta_start=1000, beta_end=20000, eta=0.001)
        obj = compute_obj(result)
        if obj < best_score:
            best_score = obj
            best_seq = result.copy()
            print(f"  Final smooth: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure valid output: non-negative, finite, reasonable size
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
