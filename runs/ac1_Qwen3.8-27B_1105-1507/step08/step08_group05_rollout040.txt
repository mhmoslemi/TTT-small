import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

def smooth_max_obj(a, T):
    """Compute smooth max of conv(a,a)."""
    conv = compute_conv(a)
    conv_shifted = conv - np.max(conv)
    return float(np.max(conv) + np.log(np.sum(np.exp(T * conv_shifted))) / T)


def smooth_max_gradient_descent(a0, time_budget=60.0, T_schedule=None, verbose=False):
    """
    Projected gradient descent on the smooth max of conv(a,a).
    a is on the simplex (sum=1, a>=0).
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # T schedule: start small, increase over time
    if T_schedule is None:
        T_schedule = [10, 50, 100, 200, 500, 1000, 2000, 5000]
    
    phase = 0
    step_size = 1.0 / n  # initial step size
    backtrack_factor = 0.5
    max_backtracks = 20
    
    total_steps = 0
    
    while time.monotonic() - start < time_budget:
        T = T_schedule[min(phase, len(T_schedule) - 1)]
        
        grad, conv, weights = smooth_max_grad(a, T)
        
        # Center the gradient (remove mean for better behavior on simplex)
        grad_centered = grad - np.mean(grad)
        
        # Projected gradient step with backtracking
        step = step_size
        a_trial = a
        for _ in range(max_backtracks):
            a_trial = project_simplex(a - step * grad_centered)
            obj_trial = compute_obj(a_trial)
            if obj_trial < best_obj or _ == 0:
                break
            step *= backtrack_factor
        
        a = a_trial
        total_steps += 1
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        
        # Adaptive step size
        # Decrease step size as T increases (higher T = steeper gradients)
        if verbose and total_steps % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"      SMGD phase={phase} T={T} step={step:.6f} obj={best_obj:.8f} t={elapsed:.1f}s")
        
        # Check if we should advance to next phase
        phase += 1
        
        # Reset step size for new T
        step_size = min(step_size / (1 + 0.001 * phase), 5.0 / (n * T))
        step_size = max(step_size, 1e-10)
    
    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_norm, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=60.0, verbose=False):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)
        
        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 30, 'xatol': 0.01})
            best_t = res.x if res.success else 0.5
        except Exception:
            best_t = 0.5
        
        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall_count = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 10:
                break
    
    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def hamming_window(n):
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def blackman_window(n):
    w = (0.42 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1))
         + 0.08 * np.cos(4.0 * np.pi * np.arange(n) / max(1, n - 1)))
    w = np.maximum(w, 0)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def kaiser_window(n, beta):
    from scipy.signal import get_window
    w = get_window(('kaiser', beta), n)
    w = np.maximum(w, 0)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def conv_gradient_full(a, conv, weights):
    """
    Compute sum_k w_k * grad_k where grad_k[m] = 2*a[k-m] if 0<=k-m<n.
    Uses direct computation.
    """
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(weights)):
        wk = weights[k]
        if wk < 1e-18:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        m_start = k - i_end
        m_end = k - i_start
        grad[m_start:m_end + 1] += 2.0 * wk * a[i_start:i_end + 1][::-1]
    return grad

def smooth_max_gradient_descent(a0, time_budget=60.0, verbose=False):

    Projected gradient descent on smooth max of conv(a,a) with backtracking.
    """

    a = a0.astype(np.float64).copy()

        return a0.copy()

    best_obj = compute_obj(a)
    start = time.monotonic()

    step_size = 2.0 / n
    T_increase = 1.01
    stall_count = 0

    while time.monotonic() - start < time_budget:

        conv_shifted = conv - np.max(conv)
        exp_vals = np.exp(T * conv_shifted)
        weights = exp_vals / np.sum(exp_vals)

            step_size = alpha * 0.9

        T *= T_increase
        step_size = max(step_size, 1e-8)

        if stall_count > 200:
            break

        if verbose and (int(time.monotonic() - start) * 2) % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"      SMGD T={T:.0f} step={step_size:.6f} best={best_obj:.8f} t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_norm, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=60.0, verbose=False):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)

        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 30, 'xatol': 0.01})
            best_t = res.x if res.success else 0.5
        except Exception:
            best_t = 0.5

        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall_count = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 10:
                break

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def hamming_window(n):
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def blackman_window(n):
    w = (0.42 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1))
         + 0.08 * np.cos(4.0 * np.pi * np.arange(n) / max(1, n - 1)))
    w = np.maximum(w, 0)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def kaiser_window(n, beta):
    try:
        w = get_window(('kaiser', beta), n)
    except Exception:
        w = hann_window(n) * n
    w = np.maximum(w, 0)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def shifted_gaussian(n, center_frac, width_frac):
    """Gaussian bumped at center_frac * n, with width width_frac * n."""
    center = center_frac * (n - 1)
    width = max(1.0, width_frac * n)
    x = np.arange(n, dtype=np.float64)
    w = np.exp(-0.5 * ((x - center) / width) ** 2)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def one_sided_power(n, power):
    """One-sided power law: w[i] = (i+1)^power, normalized."""
    w = (np.arange(n) + 1.0) ** power
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def sine_hump(n, frac=0.5):
    """Sine hump: w[i] = sin(pi * i / (n*frac)) for i < n*frac, 0 otherwise."""
    w = np.zeros(n)
    k = int(n * frac)
    if k < 2:
        k = 2
    w[:k] = np.sin(np.pi * np.arange(k) / k)
    w[:k] = np.maximum(w[:k], 0)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Mirror descent + smooth-max gradient + LP alternating + multi-start.
    Exploits asymmetry for objective < 2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
                # Also try reversed parent
                parent_rev = parent[::-1].copy()
                obj_rev = compute_obj(parent_rev)
                if obj_rev < best_score:
                    best_score = obj_rev
                    best_seq = parent_rev.copy()
                    print(f"Parent reversed: obj={best_score:.8f} **")
    except Exception:
        pass

    if best_seq is None:
        best_seq = hann_window(800)
        best_score = compute_obj(best_seq)
        print(f"Hann init (n=800): obj={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: Diverse asymmetric initializations (200s) ===
    print("\n=== Phase 1: Diverse Asymmetric Initializations ===")
    inits = {}
    inits['parent'] = best_seq.copy()
    inits['hann'] = hann_window(n_base)
    inits['hamming'] = hamming_window(n_base)
    inits['blackman'] = blackman_window(n_base)
    inits['kaiser4'] = kaiser_window(n_base, 4.0)
    inits['kaiser8'] = kaiser_window(n_base, 8.0)

    # Asymmetric inits (key for breaking the 2-barrier)
    inits['gauss_025'] = shifted_gaussian(n_base, 0.25, 0.15)
    inits['gauss_033'] = shifted_gaussian(n_base, 0.33, 0.15)
    inits['gauss_04'] = shifted_gaussian(n_base, 0.4, 0.12)
    inits['gauss_06'] = shifted_gaussian(n_base, 0.6, 0.12)
    inits['gauss_075'] = shifted_gaussian(n_base, 0.75, 0.15)
    inits['power1'] = one_sided_power(n_base, 1.0)
    inits['power2'] = one_sided_power(n_base, 2.0)
    inits['power05'] = one_sided_power(n_base, 0.5)
    inits['sine_05'] = sine_hump(n_base, 0.5)
    inits['sine_07'] = sine_hump(n_base, 0.7)
    inits['sine_04'] = sine_hump(n_base, 0.4)
    # Mixed: Gaussian * linear taper
    t = np.arange(n_base, dtype=np.float64) / max(1, n_base - 1)
    mixed = shifted_gaussian(n_base, 0.4, 0.15) * (1.0 - 0.5 * t)
    mixed = np.maximum(mixed, 0)
    inits['mixed1'] = mixed / np.sum(mixed)
    mixed2 = shifted_gaussian(n_base, 0.6, 0.15) * (0.5 + 0.5 * t)
    mixed2 = np.maximum(mixed2, 0)
    inits['mixed2'] = mixed2 / np.sum(mixed2)

    # Evaluate all inits
    for name, init in inits.items():
        obj = compute_obj(init)
        if obj < best_score:
            best_score = obj
            best_seq = init.copy()
            print(f"  {name}: obj={best_score:.8f} **")
        else:
            print(f"  {name}: obj={obj:.8f}")

    # Quick optimization on top-5 inits
    init_objs = [(compute_obj(init), name, init) for name, init in inits.items()]
    init_objs.sort(key=lambda x: x[0])
    top_inits = init_objs[:6]

    print("\n  Quick MD + SMGD on top inits:")
    for obj_val, name, init in top_inits:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(12.0, remaining * 0.2)
        a_md = mirror_descent_optimize(init, time_budget=alloc * 0.5)
        obj_md = compute_obj(a_md)
        a_sm = smooth_max_gradient_descent(a_md, time_budget=alloc * 0.5)
        obj_sm = compute_obj(a_sm)
        best_local = a_md if obj_md < obj_sm else a_sm
        best_local_obj = min(obj_md, obj_sm)
        if best_local_obj < best_score:
            best_score = best_local_obj
            best_seq = best_local.copy()
            print(f"  {name}: MD={obj_md:.8f} SMGD={obj_sm:.8f} ** best={best_score:.8f}")
        else:
            print(f"  {name}: MD={obj_md:.8f} SMGD={obj_sm:.8f}")

    # === Phase 2: n-sweep (200s) ===
    print("\n=== Phase 2: n-Sweep ===")
    n_options = [300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000]
    n_options = sorted(set(n_options + [n_base]))
    n_options = [n for n in n_options if 200 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        a_try = resample_seq(best_seq, n_try)
        alloc = min(10.0, remaining * 0.2)
        a_opt = mirror_descent_optimize(a_try, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

        # Also try asymmetric init at this n
        a_asym = shifted_gaussian(n_try, 0.35, 0.14)
        a_asym_obj = compute_obj(a_asym)
        if a_asym_obj < best_score:
            a_opt2 = mirror_descent_optimize(a_asym, time_budget=alloc * 0.5)
            obj_opt2 = compute_obj(a_opt2)
            if obj_opt2 < best_score:
                best_score = obj_opt2
                best_seq = a_opt2.copy()
                print(f"  ** n={n_try} asym: obj={best_score:.8f} **")

    # === Phase 3: LP Alternating main (400s) ===
    print("\n=== Phase 3: LP Alternating (main) ===")
    a = best_seq.copy()
    for round_idx in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(50.0, remaining * 0.25)
        a_new = lp_alternate(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new
            print(f"  LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj:.8f}")
            # Perturb to escape
            n_cur = len(best_seq)
            a = best_seq * np.exp(rng.normal(0, 0.015, n_cur))
            a = np.maximum(a, 0)
            a /= np.sum(a)

    # === Phase 4: Mirror Descent + SMGD polish (250s) ===
    print("\n=== Phase 4: MD + SMGD Polish ===")
    for method_name, func in [('MD', mirror_descent_optimize), ('SMGD', smooth_max_gradient_descent)]:
        for _ in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(30.0, remaining * 0.3)
            a_opt = func(best_seq, time_budget=alloc, verbose=False)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score - 1e-12:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  {method_name}: obj={best_score:.8f} **")
            else:
                print(f"  {method_name}: obj={obj_opt:.8f}")
                break

    # === Phase 5: Alternating LP + MD (200s) ===
    print("\n=== Phase 5: Alternating LP + MD ===")
    for alt_r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        # LP step
        alloc_lp = min(35.0, remaining * 0.3)
        a_lp = lp_alternate(best_seq, time_budget=alloc_lp)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score - 1e-12:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Alt LP {alt_r}: obj={best_score:.8f} **")

        # MD step
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc_md = min(25.0, remaining * 0.3)
        a_md = mirror_descent_optimize(best_seq, time_budget=alloc_md)
        obj_md = compute_obj(a_md)
        if obj_md < best_score - 1e-12:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Alt MD {alt_r}: obj={best_score:.8f} **")

    # === Phase 6: Random restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 25:
        print("\n=== Phase 6: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(12):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            scale = 0.01 + 0.008 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:

            a_r = a_r / sr
            alloc = min(remaining * 0.3, 20.0)
            # Try both MD and LP

            if obj_r < best_score - 1e-12:

                best_seq = a_opt.copy()

                obj_opt = compute_obj(a_opt)
                a_opt = lp_alternate(a_opt, time_budget=10.0)
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  Shape: obj={best_score:.8f} **")

    # Final validation and return

    return result
