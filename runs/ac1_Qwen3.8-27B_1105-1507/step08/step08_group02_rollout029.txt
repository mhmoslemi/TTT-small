    s = float(np.sum(a))
    if s < 0.01: return np.inf
    a_n = a / s
    conv = compute_conv(a_n)
    return 2.0 * n * float(np.max(conv))

    Fa = rfft(a, size2)
    corr = irfft(Fw * np.conj(Fa), size2)
    sg = p * corr[:n]
    
    return sg, conv

**Spectral Shaping Algorithm:**

The idea: In the frequency domain, we want to make `|FFT(a)|^2` as flat as possible. We do this by iteratively:

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated p-norm convolution minimization in log-space,
spectral shaping for exploration, LP refinement, and multi-dimension search.

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1 << (2 * n - 2).bit_length()
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def pnorm_conv_gradient(a, p=4):
    """
    Gradient of sum_k conv(a,a)[k]^p with respect to a.
    Uses FFT for O(n log n) computation.
    Returns (gradient, convolution).
    """
    n = len(a)
    size1 = 1 << (2 * n - 2).bit_length()
    fa = rfft(a, size1)
    conv = irfft(fa * fa, size1)[:2 * n - 1]

    conv_clipped = np.maximum(conv, 0)
    w = conv_clipped ** (p - 1)

    size2 = 1 << (3 * n - 3).bit_length()
    Fw = rfft(w, size2)
    Fa = rfft(a, size2)
    corr = irfft(Fw * np.conj(Fa), size2)
    sg = p * corr[:n]

    return sg, conv


def nesterov_pnorm_optimize(a, time_budget, p=4, verbose=False):
    """
    Nesterov-accelerated optimization of p-norm of convolution.
    Works in log-space (mirror descent) for automatic simplex constraint.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = np.log(a)
    x_prev = x.copy()
    beta = 0.0
    lr = 1.0 / np.sqrt(n)

    it = 0
    stall = 0
    patience = 400
    p_switch_counter = 0

    while time.monotonic() - start < time_budget:
        beta_curr = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        if it > 0:
            mu = (beta - 1.0) / beta_curr
            y = x + mu * (x - x_prev)
        else:
            y = x.copy()

        y_shifted = y - np.max(y)
        a_y = np.exp(y_shifted)
        a_y = a_y / a_y.sum()

        sg, conv = pnorm_conv_gradient(a_y, p)
        mean_sg = float(np.dot(a_y, sg))
        dx = a_y * (sg - mean_sg)

        x_new = y - lr * dx
        x_prev = x.copy()
        x = x_new
        beta = beta_curr

        x_shifted = x - np.max(x)
        a_new = np.exp(x_shifted)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            # Very slowly increase lr
            lr = min(lr * 1.0005, 2.0 / np.sqrt(n))
        else:
            stall += 1

                noise_std = min(0.4, 0.05 + 0.03 * (stall // patience))
                a_new = best_a * np.exp(rng.normal(0, noise_std, n))
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / a_new.sum()
                x = np.log(a_new)
                x_prev = x.copy()
                beta = 0.0
                lr = max(0.01, lr * 0.6)
                best_obj = compute_obj(a_new)
                if best_obj > obj_new:
                    best_a = a_new.copy()
                stall = 0
                # Switch p on restart

        p_switch_counter += 1
        if p_switch_counter == 500:

        it += 1

            print(f"      NPoly it{it}: obj={best_obj:.8f}, lr={lr:.5f}, p={p}, t={elapsed:.1f}s")

def spectral_shaping_optimize(a, time_budget, verbose=False):
    """
    Iterative spectral shaping: modify the frequency domain to flatten the spectrum,
    then project back to the non-negative simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1 << (3 * n - 3).bit_length()

    it = 0
    alpha = 0.2
    stall = 0
    patience = 100

    while time.monotonic() - start < time_budget:
        A = np.fft.rfft(a, N)

        mag = np.abs(A)
        mag_mean = mag[1:].mean() if len(mag) > 1 else 1.0
        if mag_mean < 1e-15:
            mag_mean = 1.0

        # Power compression: reduce dynamic range
        ratio = mag / mag_mean
        ratio_clipped = np.maximum(ratio, 1e-10)
        mag_new = mag * ratio_clipped ** (-alpha)

        # Preserve DC
        A_new = A.copy()
        for idx in range(1, len(A)):
            if mag[idx] > 1e-15:
                A_new[idx] = A[idx] * (mag_new[idx] / mag[idx])

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = best_a.copy()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            # Increase alpha slowly
            alpha = min(0.8, alpha * 1.02)
        else:
            a = a_new.copy()
            stall += 1
            if stall > patience:
                alpha = max(0.05, alpha * 0.7)

                    a = best_a * np.exp(rng.normal(0, 0.08, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    alpha = 0.2

def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)

    # Build constraint matrix: rows are convolution positions, columns are g indices
    # conv(a,g)[k] = sum_j a[j] * g[k-j] (with bounds)
    # For each k, the constraint is: sum_j a[k-j] * g[j] <= M
    # where a[k-j] = 0 if k-j not in [0, n-1]

    from scipy.optimize import linprog

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, n] = -1.0
    for j in range(n):
        # g[j] appears in conv[k] for k = j, j+1, ..., j+n-1
        # coefficient is a[k-j]
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A_ub[k_start:k_end + 1, j] = a_norm[0:k_end - k_start + 1]

    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s_g = float(np.sum(g_val))
            if s_g > 1e-10:
                return g_val / s_g, float(res.x[n])
    except Exception:
        pass

    return None, None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
    elif mode == 2:
        a = np.exp(-((x - 0.5) * 4) ** 2) + 0.1
    elif mode == 3:
        a = 1.0 / (1.0 + 5.0 * (x - 0.5) ** 2)
    elif mode == 4:
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.05)
    elif mode == 5:
        # Two-bump
        a = np.exp(-((x - 0.25) * 6) ** 2) + np.exp(-((x - 0.75) * 6) ** 2) + 0.05
    elif mode == 6:
        # Chirp
        a = np.abs(np.sin(2 * np.pi * (2 + 4 * x) * x)) + 0.05
    elif mode == 7:
        # Power law
        a = x ** 0.3 + (1 - x) ** 0.3 + 0.1
    else:
        a = np.ones(n)
    a = a / a.sum()
    return a

        if obj_fn < best_score:
            best_score = obj_fn
            best_seq = a_fn.copy()
            print(f"  Final NPoly: obj={best_score:.8f} **")

    # LP polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.25, 80.0)
        a_lp2, obj_lp2 = lp_alternating_optimize(best_seq, time_budget=alloc, verbose=False)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd, obj_cd = coordinate_descent_refine(best_seq, time_budget=min(remaining * 0.3, 30.0))
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
