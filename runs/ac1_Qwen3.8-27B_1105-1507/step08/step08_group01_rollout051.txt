"""

Key innovations over previous approach:

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def normalize(a):
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    return a / s


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def parametric_init(n, family='gaussian', **kwargs):
    """Generate parametric initial sequences."""
    x = np.linspace(0, 1, n)
    xc = x - 0.5
    
    if family == 'gaussian':
        sigma = kwargs.get('sigma', 0.25)
        a = np.exp(-0.5 * (xc / sigma) ** 2)
    elif family == 'raised_cosine':
        a = np.cos(np.pi * xc) ** 2
        a = np.maximum(a, 0)
    elif family == 'sinc':
        beta = kwargs.get('beta', 1.0)
        a = np.sinc(beta * xc) ** 2  # (sin(beta*pi*x)/(beta*pi*x))^2
        a = np.maximum(a, 0)
    elif family == 'uniform':
        a = np.ones(n)
    elif family == 'parabolic':
        p = kwargs.get('p', 1.0)
        a = (1 - (2 * xc) ** 2) ** p
        a = np.maximum(a, 0)
    elif family == 'sech':
        alpha = kwargs.get('alpha', 5.0)
        a = 1.0 / np.cosh(alpha * xc)
    elif family == 'bessel':
        from scipy.special import j0
        xi = kwargs.get('xi', 2.0)
        a = j0(xi * np.abs(xc) * 2) ** 2
        a = np.maximum(a, 0)
    elif family == 'super_gaussian':
        p = kwargs.get('p', 4)
        width = kwargs.get('width', 0.3)
        a = np.exp(-(np.abs(xc) / width) ** p)
    else:
        a = np.ones(n)
    
    return normalize(a)


def solve_qp_active_set(n, active_indices, a_init=None, solver_list=None):
    """Solve the active-set QP: min M s.t. (a*a)[k] <= M for k in active_set, sum(a)=1, a>=0."""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    
    if solver_list is None:
        solver_list = ['MOSEK', 'CLARABEL', 'ECOS', 'SCS']
    
    a_var = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_indices:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        indices = np.arange(i_start, i_end + 1)
        rev_indices = k - indices
        conv_k = cp.sum(cp.multiply(a_var[indices], a_var[rev_indices]))
        constraints.append(conv_k <= M)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if a_init is not None:
        a_var.value = a_init
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.maximum(a_var.value, 0)
                s = float(np.sum(a_val))
                if s > 1e-10:
                    return a_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None


def solve_qp_full(n, a_init=None, max_constraints=100):
    """Solve QP with top max_constraints peaks as active set."""
    # First estimate which peaks to include
    if a_init is not None:
        conv = compute_conv(a_init)
        max_conv = np.max(conv)
        threshold = max_conv * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) > max_constraints:
            sort_idx = np.argsort(conv)[::-1]
            active = sort_idx[:max_constraints]
    else:
        # Default: include center peaks
        active = np.arange(n - max_constraints // 2, n + max_constraints // 2)
        active = active[(active >= 0) & (active < 2 * n - 1)]
    
    return solve_qp_active_set(n, active, a_init)


def ring_player_qp(a, time_budget=300.0, verbose=True):
    """
    Ring-player using active-set QP.
    Iteratively:
    1. Find current peaks
    2. Solve QP to minimize max of those peaks
    3. Check if new peaks appear
    4. Repeat
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    active_set = None
    stall = 0
    max_iters = 200
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_conv = np.max(conv)
        
        # Determine active set: peaks within 98% of max
        threshold = max_conv * 0.98
        new_active = sorted(set(np.where(conv >= threshold)[0].tolist()))
        
        # Expand active set if needed (always include the actual max)
        max_idx = int(np.argmax(conv))
        if max_idx not in new_active:
            new_active.append(max_idx)
            new_active.sort()
        
        # Limit active set size
        if len(new_active) > 150:
            sort_idx = np.argsort(conv)[::-1][:150]
            new_active = sorted(set(np.where(conv >= conv[sort_idx[-1]])[0].tolist()))
        
        # Solve QP
        if verbose and it % 5 == 0:
            print(f"    QP ring-player it{it}: |active|={len(new_active)}, max_conv={max_conv:.8f}, time={time.monotonic()-start:.1f}s")
        
        a_qp, M_qp = solve_qp_active_set(n, new_active, a_init=a)
        
        if a_qp is None:
            if verbose:
                print(f"    QP failed at it{it}, stopping.")
            break
        
        # Check if QP solution actually improves
        obj_qp = compute_obj(a_qp)
        if obj_qp < best_obj - 1e-14:
            best_obj = obj_qp
            best_a = a_qp.copy()
            a = a_qp.copy()
            active_set = new_active
            stall = 0
        else:
            # Try blending
            for t in [0.5, 0.3, 0.7, 0.1, 0.9]:
                a_blend = (1 - t) * a + t * a_qp
                a_blend = normalize(a_blend)
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_obj - 1e-14:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
                    a = a_blend.copy()
                    active_set = new_active
                    stall = 0
                    break
            else:
                stall += 1
                if stall > 10:
                    # Perturb
                    rng = np.random.default_rng(hash((it, n)) % (2**32))
                    a_pert = a * np.exp(rng.normal(0, 0.01, n))
                    a = normalize(a_pert)
                    active_set = None
                    stall = 0
                    if it > 50:
                        break
    
    return best_a

def coordinate_descent(a, time_budget=60.0, verbose=False):
    """Coordinate descent with pair updates exploiting symmetry."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    delta = 1e-5
    pass_count = 0
    no_improve_passes = 0
    
    conv = compute_conv(a)
    max_conv = np.max(conv)
    
    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count * 7919 + n)
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            # Try increasing
            for d in [delta, delta * 0.1, delta * 10]:
                a_try = a.copy()
                a_try[i] += d
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            
            if improved:
                continue
            
            # Try decreasing
            if a[i] > delta * 0.5:
                for d in [delta, delta * 0.1]:
                    a_try = a.copy()
                    a_try[i] -= d
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
        
        if verbose and pass_count % 3 == 0:
            print(f"      CD pass {pass_count}: obj={best_obj:.8f}, delta={delta:.2e}, time={time.monotonic()-start:.1f}s")
        
        if not improved:
            no_improve_passes += 1
            delta *= 0.5
            if delta < 1e-9 or no_improve_passes > 8:
                break
        else:
            no_improve_passes = 0
    
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize(np.maximum(a_new, 0))

```python
"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations:
1. Direct convex QP active-set solving (ring-player) for peak minimization
2. Parametric initialization sweep over analytic families
3. Smooth-max gradient descent with Nesterov momentum and adaptive temperature
4. Mirror descent (multiplicative weights) as alternative optimizer
5. Aggressive multi-n exploration with resampling
6. Coordinate descent polish
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    return a / s


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def parametric_init(n, family='gaussian', **kwargs):
    """Generate parametric initial sequences from analytic families."""
    x = np.linspace(0, 1, n)
    xc = x - 0.5

    if family == 'gaussian':
        sigma = kwargs.get('sigma', 0.25)
        a = np.exp(-0.5 * (xc / sigma) ** 2)
    elif family == 'raised_cosine':
        a = np.cos(np.pi * xc) ** 2
        a = np.maximum(a, 0)
    elif family == 'sinc':
        beta = kwargs.get('beta', 1.5)
        z = beta * np.pi * xc
        a = np.ones(n)
        mask = np.abs(z) > 1e-10
        a[mask] = (np.sin(z[mask]) / z[mask]) ** 2
        a = np.maximum(a, 0)
    elif family == 'uniform':
        a = np.ones(n)
    elif family == 'parabolic':
        p = kwargs.get('p', 1.0)
        a = (1 - (2 * xc) ** 2) ** p
        a = np.maximum(a, 0)
    elif family == 'sech':
        alpha = kwargs.get('alpha', 5.0)
        a = 1.0 / np.cosh(alpha * xc)
    elif family == 'super_gaussian':
        p = kwargs.get('p', 4)
        width = kwargs.get('width', 0.25)
        a = np.exp(-(np.abs(xc) / width) ** p)
    elif family == 'cos_gaussian':
        sigma = kwargs.get('sigma', 0.3)
        freq = kwargs.get('freq', 2.0)
        a = np.exp(-0.5 * (xc / sigma) ** 2) * (1 + 0.3 * np.cos(2 * np.pi * freq * x))
        a = np.maximum(a, 0)
    elif family == 'bump':
        width = kwargs.get('width', 0.2)
        a = np.zeros(n)
        center_mask = np.abs(xc) < width
        a[center_mask] = 1.0 - (np.abs(xc[center_mask]) / width) ** 2
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)

    return normalize(a)


def solve_qp_active(n, active_indices, a_init=None):
    """
    Solve active-set QP: min M s.t. (a*a)[k] <= M for k in active_indices,
    sum(a)=1, a>=0.
    Returns (a_opt, M_opt) or (None, None) on failure.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None, None

    a_var = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active_indices:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        indices = np.arange(i_start, i_end + 1)
        rev_indices = k - indices
        conv_k_expr = cp.sum(cp.multiply(a_var[indices], a_var[rev_indices]))
        constraints.append(conv_k_expr <= M)

    prob = cp.Problem(cp.Minimize(M), constraints)

    if a_init is not None:
        a_var.value = a_init

    solvers = ['MOSEK', 'CLARABEL', 'ECOS', 'SCS']
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.maximum(a_var.value, 0)
                s = float(np.sum(a_val))
                if s > 1e-10:
                    return a_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def ring_player_qp(a, time_budget=300.0, verbose=True, max_active=100):
    """
    Ring-player using active-set QP.
    Iteratively solves QP to minimize the maximum over active peaks,
    expanding the active set as new peaks emerge.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall = 0
    max_iters = 300
    active_set = None

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = np.max(conv)

        # Determine active set: peaks within 97% of max
        threshold = max_conv * 0.97
        new_active = set(np.where(conv >= threshold)[0].tolist())

        # Always include the actual maximum
        max_idx = int(np.argmax(conv))
        new_active.add(max_idx)

        # If active set is too large, keep top ones
        if len(new_active) > max_active:
            conv_sorted = np.sort(conv)[::-1]
            cutoff = conv_sorted[min(max_active, len(conv_sorted)) - 1]
            new_active = set(np.where(conv >= cutoff)[0].tolist())

        new_active_list = sorted(new_active)

        if verbose and it % 5 == 0:
            print(f"    QP it{it}: |active|={len(new_active_list)}, max_conv={max_conv:.10f}, obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

        a_qp, M_qp = solve_qp_active(n, new_active_list, a_init=a)

        if a_qp is None:
            if verbose:
                print(f"    QP solve failed at it{it}")
            break

        obj_qp = compute_obj(a_qp)
        if obj_qp < best_obj - 1e-14:
            best_obj = obj_qp
            best_a = a_qp.copy()
            a = a_qp.copy()
            active_set = new_active_list
            stall = 0
        else:
            # Try blending
            found = False
            for t in [0.3, 0.5, 0.7, 0.1, 0.9]:
                a_blend = normalize((1 - t) * a + t * a_qp)
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_obj - 1e-14:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
                    a = a_blend.copy()
                    stall = 0
                    found = True
                    break
            if not found:
                stall += 1
                if stall > 8:
                    # Perturb to escape
                    rng = np.random.default_rng((it * 31 + n * 7) % (2**32))
                    scale = 0.005 + 0.003 * stall
                    a_pert = a * np.exp(rng.normal(0, scale, n))
                    a = normalize(a_pert)
                    stall = 0
                    if it > 80:
                        break

    if verbose:
        print(f"    QP ring-player done: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    return best_a


def smooth_max_grad(a, time_budget=100.0, T_init=None, T_min=0.000005, verbose=False):
    """Smooth max gradient descent with Nesterov momentum."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if T_init is None:
        conv = compute_conv(a)
        T_init = float(np.max(conv)) * 0.0005

    T = T_init
    lr = 0.003
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    max_iters = 10000

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = float(np.max(conv))
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        if total < 1e-300:
            break
        weights = exp_vals / total

        # Gradient: grad[i] = 2 * (weights correlate a)[i]
        grad = 2.0 * np.correlate(weights, a, mode='valid')
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 40:
                lr *= 0.75
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
                if lr < 1e-8:
                    break

        if it > 50:
            T = max(T_min, T_init * (0.9995 ** it))

        if verbose and it % 500 == 0:
            print(f"      SMGD it{it}: obj={best_obj:.8f}, T={T:.8f}, lr={lr:.6f}, t={time.monotonic()-start:.1f}s")

    return best_a


def mirror_descent(a, time_budget=100.0, verbose=False):
    """Multiplicative weight (mirror descent) on the simplex."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = float(np.max(compute_conv(a))) * 0.001
    lr = 0.02
    stall = 0
    max_iters = 5000

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = float(np.max(conv))
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)

        grad = 2.0 * np.correlate(weights, a, mode='valid')
        grad_centered = grad - np.mean(grad)

        # Mirror descent: a_new = a * exp(-lr * grad)
        a_new = a * np.exp(-lr * grad_centered)
        a_new = normalize(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                lr *= 0.7
                a = best_a.copy()
                stall = 0
                if lr < 1e-7:
                    break

        if it > 30:
            T = max(0.000002, T_init * (0.9997 ** it)) if 'T_init' in dir() else max(0.000002, T * 0.9997)

        if verbose and it % 500 == 0:
            print(f"      MD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, t={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_descent(a, time_budget=60.0, verbose=False):
    """Coordinate descent with adaptive step sizes."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    delta = 5e-5
    pass_count = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count * 104729 + n * 31)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            for d in [delta, delta * 0.1, delta * 5]:
                a_try = a.copy()
                a_try[i] += d
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                continue

            if a[i] > delta * 0.5:
                for d in [delta, delta * 0.1]:
                    a_try = a.copy()
                    a_try[i] -= d
                    a_try = normalize(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if verbose and pass_count % 3 == 0:
            print(f"      CD pass {pass_count}: obj={best_obj:.8f}, delta={delta:.1e}, t={time.monotonic()-start:.1f}s")

        if not improved:
            no_improve += 1
            delta *= 0.4
            if delta < 1e-9 or no_improve > 6:
                break
        else:
            no_improve = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)
