import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear autocorrelation a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:2 * n - 1]
    return np.maximum(conv, 0)  # Numerical safety


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ z gives conv(a,z)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # A[i, j] = a[i - j] if 0 <= i - j < n
        rows = np.arange(j, min(2 * n - 1, j + n))
        cols = np.zeros(len(rows), dtype=int) + j
        A[rows, cols] = a[:len(rows)]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,z) <= M, sum(z)=1, z>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ z <= M, cp.sum(z) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(constraints[0].dual_value), 0)
                    except:
                        pass
                    return z_val, float(M.value), dual_y
        except:
            continue
    
    # Fallback
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except:
        pass
    
    return None, None, None

def frank_wolfe_polish(a, time_budget=100.0, verbose=False):

def coordinate_descent(a, time_budget=60.0, verbose=False):
    """
    Coordinate descent: for each coordinate, find the best value

        order = rng.permutation(n)
        
        for k in order:
            if time.monotonic() - start > time_budget:
                break
            
            # Try several values for coordinate k
            current = a[k]
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]:
                new_val = current * mult
                if new_val < 0:
                    continue
                a_try = a.copy()

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Enhanced step function optimizer combining LP alternating with lazy constraints,
Frank-Wolfe conditional gradient, gradient-based peak shaving, and adaptive
multi-n exploration to minimize 2*n*max(autocorr)/(sum)^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear autocorrelation a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:2 * n - 1]
    return np.maximum(conv, 0.0)


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ z)_j = conv(a,z)_j."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_full(a, dual=True):
    """Solve LP: min M s.t. conv(a,z) <= M, sum(z)=1, z>=0.
    Returns (z, M, dual_y) or (None, None, None)."""
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    if dual:
                        try:
                            dual_y = np.maximum(np.asarray(c1.dual_value, dtype=np.float64), 0)
                        except:
                            pass
                    return z_val, float(M.value), dual_y
        except:
            continue

    # Fallback to scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except:
        pass

    return None, None, None

    n = len(a)
    A = build_conv_matrix(a)
    total_rows = 2 * n - 1
    rng = np.random.default_rng(seed)

    # Start with center and a few random constraints
    init_set = {n - 1}
    extra = rng.choice(total_rows, min(30, total_rows), replace=False)
    init_set.update(extra.tolist())

    active = set(init_set)

    z_val = None
    M_val = np.inf
    start = time.monotonic()

    for iteration in range(50):
        if time.monotonic() - start > time_budget:
            break

        rows = np.array(sorted(active))
        A_sub = A[rows, :]
        nr = len(rows)

        # Solve reduced LP using scipy (fast for small problems)
        try:
            from scipy.optimize import linprog
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.zeros((nr, n + 1))
            A_ub[:, :n] = A_sub
            A_ub[:, n] = -1.0
            b_ub = np.zeros(nr)
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(None, None)]
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                break
            z_cand = np.maximum(res.x[:n], 0)
            zc = float(np.sum(z_cand))
            if zc < 1e-10:
                break
            z_cand = z_cand / zc
            M_cand = float(res.x[n])
        except:
            break

        violated = np.where(conv_check > M_cand * 1.0001 + 1e-12)[0]

            M_val = M_cand

        else:

def frank_wolfe_polish(a, time_budget=80.0, verbose=False):
    """Frank-Wolfe conditional gradient for peak autocorrelation reduction."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    t_cur = 0.4
    stall = 0
    iter_count = 0

    while time.monotonic() - start < time_budget:
        iter_count += 1
        conv = compute_conv(a)
        peak = float(np.max(conv))

        # Near-peak lags
        threshold = peak * 0.995
        peak_idx = np.where(conv >= threshold)[0]

        # Compute subgradient at peak lags
        grad = np.zeros(n)
        for j in peak_idx:
            for k in range(n):
                v1 = a[j - k] if (0 <= j - k < n) else 0.0
                v2 = a[k - j] if (0 <= k - j < n) else 0.0
                grad[k] += v1 + v2
        grad /= max(len(peak_idx), 1)

        # Candidate vertices: argmin grad + a few random
        k_min = int(np.argmin(grad))
        candidates = [k_min]
        for _ in range(2):
            candidates.append(int(rng.integers(0, n)))

        improved = False
        for k_t in candidates:
            if improved:
                break
            v = np.zeros(n)
            v[k_t] = 1.0
            for step in [t_cur, t_cur * 0.5, t_cur * 0.25, 0.05, 0.02]:
                a_try = (1.0 - step) * a + step * v
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    t_cur = min(t_cur + 0.02, 0.7)
                    break

        if not improved:
            stall += 1
            t_cur = max(t_cur * 0.85, 0.03)
            if stall > 50:
                # Perturb to escape
                a = best_a * np.exp(rng.normal(0, 0.015, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                t_cur = 0.3

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    FW iter {iter_count}: obj={best_obj:.8f}, t={elapsed:.0f}s")

    return best_a


def peak_shave(a, time_budget=50.0, verbose=False):
    """Direct peak shaving via mass redistribution from high to low peak contributors."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak = float(np.max(conv))
        threshold = peak * 0.99
        peak_idx = np.where(conv >= threshold)[0]

        # Compute peak contribution per coordinate
        peak_contrib = np.zeros(n)
        for j in peak_idx:
            for k in range(n):
                v1 = a[j - k] if (0 <= j - k < n) else 0.0
                v2 = a[k - j] if (0 <= k - j < n) else 0.0
                peak_contrib[k] += v1 + v2
        peak_contrib /= max(len(peak_idx), 1)

        # Rank by contribution
        ranked = np.argsort(peak_contrib)
        top_q = n // 5
        high_idx = ranked[n - top_q:]
        low_idx = ranked[:top_q]

        improved = False
        for _ in range(80):
            if time.monotonic() - start > time_budget:
                break

            src_cands = [k for k in high_idx if a[k] > 5e-5]
            tgt_cands = [k for k in low_idx if k not in set(src_cands)]
            if not src_cands or not tgt_cands:
                break

            src = int(rng.choice(src_cands))
            tgt = int(rng.choice(tgt_cands))

            for frac in [0.01, 0.02, 0.04, 0.08]:
                delta = frac * a[src]
                if delta >= 0.9 * a[src]:
                    continue
                a_try = a.copy()
                a_try[src] -= delta
                a_try[tgt] += delta
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall > 8:
                break

    return best_a


def coordinate_polish(a, time_budget=50.0, verbose=False):
    """Coordinate descent: adjust individual coordinates to reduce peak."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(333)

    for sweep in range(20):
        if time.monotonic() - start > time_budget:
            break

        improved_sweep = False
        order = rng.permutation(n)

        for k in order:
            if time.monotonic() - start > time_budget:
                break
            current = a[k]
            if current < 1e-8:
                continue
            for mult in [0.6, 0.8, 1.25, 1.5, 0.4, 2.0, 0.7, 1.4]:
                new_val = current * mult
                a_try = a.copy()
                a_try[k] = new_val
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    current = new_val
                    improved_sweep = True

        if not improved_sweep:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
