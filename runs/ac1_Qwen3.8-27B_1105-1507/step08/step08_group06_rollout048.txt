def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, g):
    """Compute cross-convolution a * g via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:2 * n - 1]

def compute_obj(a):

def build_conv_matrix(a):

    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A

"""
Optimized step function search using accelerated LP-alternating with Nesterov momentum,
adaptive line search, diverse restarts, subgradient polish, and length sweep.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                return g_val, M_val
    except Exception:
        pass

    return None, None


def lp_step(a):
    """One LP alternating step: solve LP and return the optimal g."""
    g, M = solve_lp(a)
    return g

def subgradient_polish(a, time_budget=100.0):
    """
    Nesterov-accelerated subgradient method targeting active peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    lr = 0.04
    beta = 0.8
    momentum = np.zeros(n)

    for it in range(8000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(y)
        max_val = float(np.max(conv))
        threshold = max_val * 0.85
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 40:
            active = np.argsort(conv)[::-1][:40]

        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += y[k - indices]
        grad /= len(active)

        # Center gradient
        grad_centered = grad - np.mean(grad)

        # Momentum step
        momentum = beta * momentum - lr * grad_centered
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = s
        y_new = y_new / s_new

        # Nesterov lookahead
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        y_next = y_new + (1.0 - t_k) * (y_new - y)
        y_next = np.maximum(y_next, 0)
        sy = float(np.sum(y_next))
        if sy > 1e-10:
            y_next = y_next / sy
        y = y_next

        obj = compute_obj(y)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()

        if it % 300 == 299:
            lr = max(lr * 0.9, 0.003)
            beta = max(beta * 0.999, 0.5)

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n <= 1:

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            a_try = a_try / np.sum(a_try)

        alloc_lp = min(80.0, remaining * 0.4)
        alloc_sg = min(30.0, remaining * 0.15)

        a_opt = momentum_alternating(a_try, time_budget=alloc_lp, beta_init=0.6)
        a_opt = subgradient_polish(a_opt, time_budget=alloc_sg)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score - 1e-14:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Final Polish Cycles ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    polish_round = 0
    while remaining > 20 and polish_round < 5:
        alloc1 = min(60.0, remaining * 0.35)
        alloc2 = min(30.0, remaining * 0.2)
        alloc3 = min(20.0, remaining * 0.1)

        a1 = momentum_alternating(best_seq, time_budget=alloc1, beta_init=0.5)
        a2 = subgradient_polish(a1, time_budget=alloc2)
        a3 = coordinate_polish(a2, time_budget=alloc3)
        obj = compute_obj(a3)

        if obj < best_score - 1e-14:
            best_score = obj
            best_seq = a3.copy()
            print(f"  ** Final {polish_round}: obj={best_score:.8f}")
            polish_round += 1
        else:
            # Try with perturbation
            scale = 0.01 + 0.005 * polish_round
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_final))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / np.sum(a_pert)
            a_p1 = momentum_alternating(a_pert, time_budget=alloc1, beta_init=0.6)
            a_p2 = subgradient_polish(a_p1, time_budget=alloc2)
            obj_p = compute_obj(a_p2)
            if obj_p < best_score - 1e-14:
                best_score = obj_p
                best_seq = a_p2.copy()
                print(f"  ** Final {polish_round} (pert): obj={best_score:.8f}")
                polish_round += 1
            else:
                print(f"  Final {polish_round}: obj={obj:.8f} (plateau)")
                break

        remaining = deadline - time.monotonic()

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [100.0] * len(best_seq)
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
