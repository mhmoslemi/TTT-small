def compute_conv(a):

    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp(a):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)

    # (A @ g)[k] = sum_j a[k-j] * g[j] for valid indices
    # = conv(a, g)[k]

            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    
    # LP: min M s.t. A@g - M <= 0, sum(g) = 1, g >= 0, M free
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

**Acceleration idea**: Use over-relaxation. Instead of a_new = (1-t)*a + t*g, use a_new = (1-t)*a + alpha*t*g with alpha > 1 (extrapolation). This can accelerate convergence but risks divergence.

**Another idea**: Use the LP dual to compute a "certified lower bound" on the objective. If the lower bound meets the upper bound, we've found the optimum.

The LP dual for fixed a is:
max sum_k y_k s.t. sum_k y_k * a[k-j] <= 1 for all j, y_k >= 0

The optimal value of this dual equals M (the min max conv value). So at each iteration, we get a certified value M. If we also have an upper bound (from the current a), we can measure the gap.

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with adaptive line-search blending, peak-weighted
subgradient polishing, QP regularization, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
from scipy.optimize import linprog


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_with(a, b):
    """FFT-based full linear convolution of a and b."""
    n = len(a)
    m = len(b)
    if n != m:
        # Pad shorter to match
        if n < m:
            a = np.concatenate([a, np.zeros(m - n)])
        else:
            b = np.concatenate([b, np.zeros(n - m)])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (conv(a,g))[k] = sum_j a[k-j] * g[j]
        # For fixed j, a[k-j] is nonzero when 0 <= k-j < n, i.e., j <= k <= j+n-1
        k_min = j
        k_max = min(2 * n - 2, j + n - 1)
        if k_min <= k_max:
            idx = np.arange(k_min, k_max + 1)
            A[idx, j] = a[idx - j]
    return A


def solve_lp(a, use_qp=False, lam=0.0):
    """Solve: min M (+ lam/2 * ||g||^2) s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    
    if use_qp and lam > 0:
        objective = cp.Minimize(M + 0.5 * lam * cp.sum_squares(g))
    else:
        objective = cp.Minimize(M)
    
    prob = cp.Problem(objective, constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            elapsed = time.monotonic() - t0
            if elapsed > 120:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    return None, None

def subgradient_optimize(a, time_budget=60.0, top_k=20, weighted=False, verbose=False):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.85
    lr = 1.0 / np.sqrt(n)
    x = a.copy()

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        grad = subgradient_step(x, top_k=top_k, weighted=weighted)
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Nesterov update
        t_k = 2.0 / (it + 2.0)
        x = a_new + (1 - t_k) * (a_new - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        if it % 500 == 499:
            lr *= 0.97

        if verbose and it % 10000 == 0:
            print(f"    SubGrad: it={it}, obj={best_obj:.8f}")

    return best_a

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng):
    """Generate a diverse structured initialization."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-50 * (x - 0.5) ** 2)
    elif choice == 1:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 3 * x)
    elif choice == 2:
        a = np.exp(-80 * (x - 0.3) ** 2) + np.exp(-80 * (x - 0.7) ** 2)
    elif choice == 3:
        # Random smooth
        a = np.ones(n)
        for _ in range(4):
            center = rng.integers(0, n)
            width = max(1, rng.integers(n // 10, n // 3))
            d = np.abs(np.arange(n) - center)
            d = np.minimum(d, n - d)
            a *= 1.0 + rng.standard_normal() * np.exp(-d / (width * 0.5))
        a = np.maximum(a, 0.01)
    elif choice == 4:
        # Power law
        a = x ** rng.uniform(0.5, 3.0)
    else:
        # sinc-like
        a = np.sinc(3 * (x - 0.5)) ** 2
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
