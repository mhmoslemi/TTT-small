"""
Optimized step function search using Nesterov-accelerated smooth optimization,
fast HiGHS LP alternating with parabolic line search, peak-targeting refinement,
and multi-n exploration. Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def convolution_matrix(a):
    """Build the convolution matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = convolution_matrix(a_norm)

    # LP: variables are [g_0, ..., g_{n-1}, M]
    # min M => c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Constraints: A @ g - M <= 0  =>  [A, -1] @ [g; M] <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1 => [1,...,1, 0] @ [g; M] = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def smooth_max_and_grad(a, beta):
    """Compute smooth max approximation and its gradient."""
    n = len(a)
    conv = compute_conv(a)
    m = conv.max()
    shifted = beta * (conv - m)
    # Handle numerical issues
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros(2 * n - 1)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= wsum

    smooth_val = m + np.log(wsum) / beta if wsum > 1e-300 else m

    # Gradient: g_i = 2 * sum_{m=0}^{n-1} a_m * w_{i+m}
    g = np.zeros(n)
    for i in range(n):
        end = min(2 * n - 1, i + n)
        g[i] = 2.0 * np.dot(a, weights[i:end])

    return smooth_val, g


def nesterov_smooth_optimize(a, time_budget=250.0, beta_start=20.0, beta_end=2000.0, verbose=False):
    """
    Nesterov-accelerated gradient method on log-sum-exp smooth approximation.
    Works on the simplex {a >= 0, sum(a) = 1} using projection.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov momentum
    y = a.copy()  # momentum point
    it = 0
    total_time_estimate = 1e-4 * n  # rough time per iteration

    while time.monotonic() - start < time_budget:
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        beta = beta_start * (beta_end / beta_start) ** frac

        smooth_val, g = smooth_max_and_grad(y, beta)

        # Nesterov step size: t_k = 2 / (k + 1) for step = 1/(L*t_k)
        # L ~ 2*beta*n (Lipschitz constant of gradient)
        k = it + 1
        step = 2.0 / (beta * n * k * k)
        step = min(step, 0.5 / (beta * np.sqrt(n)))

        # Gradient step from y
        x = y - step * g
        x = np.maximum(x, 0)
        sx = x.sum()
        if sx > 1e-15:
            x = x / sx
        else:
            x = y.copy()

        # Nesterov momentum update
        t_prev = 2.0 / (k - 1 + 2) if k > 1 else 1.0
        t_curr = 2.0 / (k + 1)
        # Standard Nesterov: y_new = x + ((t_curr - 1)/(t_curr + 1)) * (x - y)
        # Simpler: use constant momentum
        momentum = min(0.9, 0.5 + 0.3 * frac)
        y = x + momentum * (x - y)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = x.copy()

        # Evaluate
        obj_y = compute_obj(y)
        if obj_y < best_obj - 1e-12:
            best_obj = obj_y
            best_a = y.copy()

        it += 1
        if verbose and it % 50 == 0:
            print(f"      Nest it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={time.monotonic()-start:.1f}s")

    return best_a

def peak_targeting_refine(a, time_budget=200.0, verbose=False):
    """
    Targeted refinement: identify top peaks and make small moves to reduce them.
    Uses subgradient information to guide perturbations.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    iteration = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Find top K peaks
        K = min(5, len(conv))
        top_idx = np.argsort(conv)[::-1][:K]
        top_vals = conv[top_idx]

        # Build subgradient from top peaks
        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            weight = conv[k] / (conv[top_idx].sum() + 1e-30)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += weight * a[j]

        # Try moving in -g direction (reduces peaks)
        # But project onto simplex
        for scale in [0.001, 0.003, 0.01, 0.03, 0.05, 0.1]:
            a_try = a - scale * g
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                iteration += 1
                break
        else:
            # No improvement from subgradient direction
            # Try targeted pairwise moves
            improved = False
            for k in top_idx[:3]:
                k = int(k)
                # Find top contributing pairs for this peak
                pairs = []
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        pairs.append((a[i] * a[j], i, j))
                pairs.sort(reverse=True)

                for contrib, i, j in pairs[:5]:
                    if contrib < 1e-10:
                        break
                    # Try moving mass from i (or j) to a random position
                    rng = np.random.default_rng(iteration * 1000 + i * 10 + j)
                    for _ in range(3):
                        target = rng.integers(0, n)
                        if target == i or target == j:
                            continue
                        for frac in [0.01, 0.03, 0.05]:
                            delta = frac * min(a[i], a[j] if i != j else a[i])
                            if delta < 1e-10 or delta >= a[i]:
                                continue
                            src = i if a[i] >= a[j] else j
                            a[src] -= delta
                            a[target] += delta
                            obj_try = compute_obj(a)
                            if obj_try < best_obj - 1e-12:
                                best_obj = obj_try
                                best_a = a.copy()
                                iteration += 1
                                improved = True
                                break
                            else:
                                a[src] += delta
                                a[target] -= delta
                                break
                    if improved:
                        break
                if improved:
                    break

            if not improved:
                # Try a random restart perturbation
                rng = np.random.default_rng(iteration * 31 + n)
                scale = 0.02 + 0.03 * (iteration % 5)
                a = best_a * (1.0 + scale * rng.normal(0, 1, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                best_a = a.copy()
                best_obj = compute_obj(a)

        if verbose and iteration % 20 == 0:
            print(f"      Peak it{iteration}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent_refine(a, time_budget=40.0):
    """
    Coordinate descent: try adjusting pairs of coordinates to reduce max convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributing pairs
        pairs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                pairs.append((a[i] * a[j], i, j))
        pairs.sort(reverse=True)

        improved = False
        rng = np.random.default_rng(int(time.monotonic() * 1000))

        for contrib, i, j in pairs[:10]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break

            # Try moving small amount from the larger of i,j to several targets
            src = i if a[i] >= a[j] else j
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                delta = frac * a[src]
                if delta < 1e-12:
                    break
                # Try a few random targets
                targets = rng.choice(n, size=min(5, n), replace=False)
                for target in targets:
                    if target == src or target == (i if src == j else j):
                        continue
                    a[src] -= delta
                    a[target] += delta
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / (a.sum() + 1e-30)
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 10, n // 10]
        x = np.arange(n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / (a.sum() + 1e-30)
    elif mode == 3:
        # Triangle
        x = np.linspace(-1, 1, n)
        a = np.maximum(1.0 - np.abs(x), 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(np.abs(1 - x**2) + 0.01))
        a = a / a.sum()
    elif mode == 5:
        # Sum of sinusoids
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Chirp
        x = np.linspace(0, 1, n)
        freq = 2 + 6 * x
        a = np.abs(np.sin(np.pi * freq * x)) + 0.01
        a = a / a.sum()
    elif mode == 7:
        # Random smooth (low-pass filtered noise)
        a = rng.uniform(0, 1, n)
        # Simple moving average smoothing
        k = max(3, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

            print(f"  Peak: obj={obj_pt:.8f} (no improvement)")

    # --- Phase 4: Multi-n exploration ---
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Generate candidate n values
    n_candidates = set()
    for offset in [-200, -100, -50, 50, 100, 200, 300]:
        n_try = n_best + offset
        if 400 <= n_try <= 2000:
            n_candidates.add(n_try)
    # Also try some absolute values
    for n_abs in [500, 600, 700, 900, 1000, 1200, 1400]:
        if n_abs != n_best and 400 <= n_abs <= 2000:
            n_candidates.add(n_abs)

    n_candidates = sorted([n for n in n_candidates if n != n_best])

    for n_try in n_candidates[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 80.0)

        # Quick Nesterov
        half = alloc / 2
        if half > 15:
            a_s = nesterov_smooth_optimize(a_try, time_budget=half,
                                           beta_start=20, beta_end=800)
            if compute_obj(a_s) < compute_obj(a_try):
                a_try = a_s

        # Quick LP
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.08, 50.0)
        if lp_alloc > 15:
            a_opt = lp_alternate_optimize(a_try, time_budget=lp_alloc)
            obj_opt = compute_obj(a_opt)
        else:
            a_opt = a_try
            obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_current = len(best_seq)
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # --- Phase 5: Diverse restarts ---
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        mode = (r + 3) % 8
        a_r = diverse_init(n_use, rng, mode=mode)
        # Blend with best
        blend = 0.2 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / (a_r.sum() + 1e-30)

        restart_budget = min(remaining * 0.12, 60.0)
        if restart_budget > 30:
            a_r = nesterov_smooth_optimize(a_r, time_budget=restart_budget * 0.5,
                                           beta_start=20, beta_end=500)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_r = lp_alternate_optimize(a_r, time_budget=min(remaining * 0.15, 30.0))

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            n_current = len(best_seq)
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # --- Phase 6: Final intensive polish ---
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 30:
        lp_alloc = min(remaining * 0.3, 100.0)
        a_lp = lp_alternate_optimize(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            n_current = len(best_seq)
            print(f"  Final LP: obj={best_score:.8f} **")

    # Peak targeting
    remaining = deadline - time.monotonic()
    if remaining > 25:
        pt_alloc = min(remaining * 0.2, 60.0)
        a_pt = peak_targeting_refine(best_seq, time_budget=pt_alloc)
        obj_pt = compute_obj(a_pt)
        if obj_pt < best_score:
            best_score = obj_pt
            best_seq = a_pt.copy()
            print(f"  Final Peak: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        cd_alloc = min(remaining * 0.2, 30.0)
        a_cd = coordinate_descent_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more LP round
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_last = lp_alternate_optimize(best_seq, time_budget=min(remaining * 0.4, 30.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # --- Final validation ---
    print(f"\n=== Final Validation ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
