"""
Step function optimization: LP alternating with HiGHS solver, 
flattening heuristics, and length sweep.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_max = min(m - 1, j + n - 1)
        A[j:k_max+1, j] = a[:k_max - j + 1]
    return A


def solve_lp(a, A=None):
    """Solve LP: min M s.t. A@conv_matrix*a * g <= M, sum(g)=1, g>=0.
    Returns g, M, dual_info or (None, None, None)."""
    n = len(a)
    if A is None:
        A = build_conv_matrix(a)
    m = 2 * n - 1
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs > 1e-12:
            g = g / gs
            M = float(res.x[n])
            return g, M, res
    return None, None, None


def lp_alternate(a, max_iters=500, time_budget=300.0, verbose=False):
    """Pure LP alternating (t=1) with restarts."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    A = build_conv_matrix(a)  # This changes when a changes!
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Rebuild A since a has changed
        A = build_conv_matrix(a)
        
        g, M, res = solve_lp(a, A)
        if g is None:
            break
        
        # Update: a = g (pure LP, t=1)
        a = g.copy()
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            
            if verbose and it % 100 == 0:
                elapsed = time.monotonic() - start
                print(f"    it={it}, obj={best_obj:.10f}, M={M:.8f}, time={elapsed:.0f}s")
    
    return best_a


def lp_alternate_adaptive(a, max_iters=500, time_budget=300.0, verbose=False):
    """LP alternating with adaptive step size and periodic restarts."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        A = build_conv_matrix(a)
        g, M, res = solve_lp(a, A)
        if g is None:
            break
        
        # Try different step sizes
        best_t_obj = np.inf
        best_t = 1.0
        best_a_try = a.copy()
        
        for t in [1.0, 0.8, 0.6, 0.4]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-12:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_try = a_try
        
        if best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = best_a_try.copy()
            a = best_a_try
            stall = 0
        else:
            stall += 1
            if stall == 20:
                # Perturb and restart
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_a * (1 + 0.02 * rng.standard_normal(n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    it={it}, obj={best_obj:.10f}, stall={stall}, time={elapsed:.0f}s")
    
    return best_a


def flatten_conv(a, time_budget=60.0, verbose=False):
    """
    Try to flatten the convolution by moving mass from positions
    that contribute most to the peak to positions that don't.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find positions contributing to the peak
        # conv[k_star] = sum_j a[j] * a[k_star - j]
        contrib = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contrib[j] = a[j] * a[k_j]
        
        # Also consider the second and third highest peaks
        peak_indices = np.argsort(conv)[::-1][:5]
        for k_p in peak_indices:
            k_p = int(k_p)
            for j in range(n):
                k_j = k_p - j
                if 0 <= k_j < n:
                    contrib[j] += 0.5 * a[j] * a[k_j]
        
        # Try moving mass from high-contribution to low-contribution positions
        sorted_idx = np.argsort(contrib)[::-1]
        improved = False
        
        for i in sorted_idx[:10]:
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-10:
                continue
            
            # Find a "good" target position (low contribution, far from peak)
            target_candidates = np.argsort(contrib)[:10]
            for j_idx in target_candidates:
                if j_idx == i:
                    continue
                for frac in [0.01, 0.02, 0.05]:
                    delta = frac * a[i]
                    if delta > a[i]:
                        continue
                    a[i] -= delta
                    a[j_idx] += delta
                    s = float(np.sum(a))
                    a = a / s
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j_idx] -= delta
                        a = a / s
                if improved:
                    break
            if improved:
                break
        
        if not improved and it > 50:
            # Random perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e4) % (2**31))
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = 0.01 * min(a[i], a[j])
                if delta > 0:
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a.copy()
                    else:
                        a = best_a.copy()
    
    return best_a


def generate_init(kind, n, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
    elif kind == 'gaussian':
        x = np.arange(n, dtype=float)
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif kind == 'triangle':
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - x**2)
    elif kind == 'raised_cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif kind == 'bell':
        x = np.arange(n, dtype=float)
        center = (n - 1) / 2.0
        sigma = n / 5.0
        a = np.exp(-((x - center) / sigma) ** 2)
    elif kind == 'random':
        a = rng.random(n) + 0.01
    else:
        a = np.ones(n)
    return a / np.sum(a)


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
