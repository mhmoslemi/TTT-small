import numpy as np

from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_rows(a, indices):
    """Build only rows of the conv matrix for given indices."""
    n = len(a)
    A = np.zeros((len(indices), n))
    for row, k in enumerate(indices):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[row, j] = a[i]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def convex_line_search(a, g, n_evals=20):
    """Find optimal t in [0,1] for a_new = (1-t)*a + t*g using golden section search."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    # Golden section search
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    lo, hi = 0.0, 1.0
    d = phi
    c = lo + (1 - d) * (hi - lo)
    bb = lo + d * (hi - lo)
    fc = obj_at_t(c)
    fb = obj_at_t(bb)

    for _ in range(n_evals):
        if fc < fb:
            hi = bb
            bb = c
            fb = fc
            d = d * phi
            c = lo + (1 - d) * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = bb
            fc = fb
            d = d * phi
            bb = lo + d * (hi - lo)
            fb = obj_at_t(bb)

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm.copy()
    return t_opt, a_new / s

        # Compute smooth max gradient
        conv_y = compute_conv(y)
        max_c = np.max(conv_y)
        shifted = (conv_y - max_c) / T
        shifted = np.clip(shifted, -400, 400)
        exp_vals = np.exp(shifted)
        total = float(np.sum(exp_vals))
        p = exp_vals / total  # softmax weights over conv peaks

        # Gradient: grad[i] = sum_k p[k] * a[k-i] = conv(p, a)
        # But we need gradient wrt a, and conv(p,a)[k] = sum_i p[k-i]*a[i]
        # Actually grad of smooth_max wrt a[i] = sum_k p[k] * (d conv(a,a)[k]/da[i])
        # d conv(a,a)[k] / da[i] = a[k-i] (for valid k-i)
        # So grad[i] = sum_k p[k] * a[k-i] = conv(p, a)[i]... 
        # Let me verify: conv(p, a)[m] = sum_j p[j] * a[m-j]
        # We want grad[i] = sum_k p[k] * a[k-i]
        # Let m = k-i, so k = m+i: grad[i] = sum_m p[m+i] * a[m]
        # This is conv(p_shifted, a) where p_shifted[m] = p[m+i]... 
        # Actually it's cross-correlation: grad = correlate(p, a, 'full')[offset]
        # Let's use: grad[i] = sum_k p[k] * a[k-i] for max(0,k-n+1) <= i <= min(n-1,k)
        # = sum_{k=i}^{i+n-1} p[k] * a[k-i] (with p[k]=0 outside [0,2n-2])
        # This is the convolution (p * a)[i] if we define it as sum_k p[k] a[k-i]
        # Which is correlate(p, a)[i] with appropriate offset

        # Use FFT for efficiency
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fp = rfft(p, size)
        fa = rfft(a, size)
        # We want: grad[i] = sum_k p[k] * a[k-i]
        # This is the cross-correlation <p, shift_i(a)>
        # In FFT: IFFT(conj(FP) * FA) gives correlation
        # Actually: cross_corr[i] = sum_k p[k] * a[k+i] = IFFT(conj(FP) * FA)[i]
        # But we want sum_k p[k] * a[k-i] which is cross_corr[-i] = cross_corr[2n-1-i] (cyclic)
        # Simpler: just compute conv(p, a) and take the right part
        # conv(p,a)[m] = sum_j p[j] * a[m-j]
        # We want grad[i] = sum_k p[k] * a[k-i]
        # Let j=k, m-k=i => m = i+k... no
        # conv(p,a)[m] = sum_j p[j] a[m-j]
        # Set m-j = k-i => j = m - k + i... this is getting complicated
        # Let me just use: grad[i] = sum_k p[k] a[k-i]
        # Define q[k] = p[k], then grad = q * a (cross-correlation with shift)
        # numpy correlate: np.correlate(p, a, 'valid') gives sum_k p[k+j]*a[j] for j=0..n-1
        # That's not quite right either.
        
        # Simplest correct approach: 
        # grad[i] = sum_k p[k] * a[k-i] where 0<=k-i<n, 0<=k<2n-1
        # = sum_{k=max(0,i)}^{min(2n-2, i+n-1)} p[k] * a[k-i]
        # = (p * a)[i] in the "cross-correlation" sense
        # In numpy: np.correlate(p, a, mode='full')[2n-2-i]... 
        # Actually let's just use the direct formula via FFT:
        # Define the "anti-convolution": (p ⋆ a)[i] = sum_k p[k] a[k-i]
        # In FFT: DFT(p ⋆ a) = conj(DFT(p)) * DFT(a) (for circular)
        # For linear, we need to be careful with padding.
        
        # Let me just compute it directly for now (n=800 is small enough)
        # Actually for n=800, the direct computation is O(n*(2n-1)) which is too slow.
        # Let's use the FFT approach properly.
        
        # (p ⋆ a)[i] = sum_k p[k] a[k-i]
        # Let b[k] = p[2n-2-k] (reverse p)
        # (p ⋆ a)[i] = sum_k b[2n-2-k] a[k-i]
        # Let j = 2n-2-k, so k = 2n-2-j
        # = sum_j b[j] a[2n-2-j-i]
        # = (a * b)[2n-2-i] (standard convolution)
        # So grad[i] = (a * b)[2n-2-i] where b = reverse(p)
        
        b = p[::-1]  # reverse
        size2 = 1
        while size2 < 2 * n - 1:
            size2 <<= 1
        fa2 = rfft(a, size2)
        fb2 = rfft(b, size2)
        conv_ab = irfft(fa2 * fb2, size2)[:2 * n - 1]
        grad = conv_ab[2 * n - 2 - np.arange(n)]  # grad[i] = conv_ab[2n-2-i]
        
        # Center the gradient (remove mean for simplex constraint)
        grad_centered = grad - np.mean(grad)
        
        # Nesterov update
        v = momentum * v - lr * grad_centered
        y_new = simplex_project(y + v)
        
        # Update y with Nesterov momentum
        t_k = (1 + np.sqrt(1 + 4 * it * it)) / 2
        alpha = (it - 1) / t_k if it > 0 else 0
        y_new = y_new + alpha * (y_new - y)
        y = simplex_project(y_new)
        
        # Evaluate
        obj_y = compute_obj(y)
        if obj_y < best_obj - 1e-12:
            best_obj = obj_y
            best_a = y.copy()
            a = y.copy()
            stall = 0
            # Adaptive LR: increase if improving
            if it % 500 == 0 and it > 100:
                lr = min(lr * 1.02, 0.05)
        else:
            stall += 1
            if stall > 200:
                lr *= 0.8
                if lr < 1e-7:
                    break
                y = best_a.copy()
                v = np.zeros(n)
                stall = 0

        # Decay temperature
        if it > 50:
            T = max(T_min, T_init * (0.9995 ** it))

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"            SubGrad it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}, time={elapsed:.1f}s")

    return best_a


def lp_alternating_fast(a, time_budget=400.0, top_k=15, verbose=False):
    """
    LP alternating with aggressive active set (only top_k constraints).
    Uses golden-section line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    active_set = None
    blend_schedule = []

    max_iters = 2000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Update active set every 5 iterations or if not set
        if active_set is None or it % 5 == 0:
            conv_cur = compute_conv(a)
            max_conv = np.max(conv_cur)
            # Use a tight threshold: only truly active constraints
            sorted_conv = np.sort(conv_cur)[::-1]
            # Adaptive k: use constraints within 1% of max
            threshold = max_conv * 0.99
            candidates = np.where(conv_cur >= threshold)[0]
            active_set = candidates[:min(top_k, len(candidates))]
            if len(active_set) < 3:
                active_set = np.argsort(conv_cur)[::-1][:top_k]

        # Solve LP with reduced active set
        g_lp, M_lp = solve_lp_reduced(a, active_set)
        if g_lp is None:
            # Fallback: use the "peaked" direction
            conv_cur = compute_conv(a)
            peak_idx = np.argmax(conv_cur)
            g_lp = np.zeros(n)
            for j in range(n):
                i = peak_idx - j
                if 0 <= i < n:
                    g_lp[j] = a[i]
            s_g = float(np.sum(g_lp))
            if s_g > 1e-10:
                g_lp = g_lp / s_g
            else:
                break

        # Verify LP solution against all constraints
        conv_check = compute_conv_cross(a, g_lp)
        true_M = float(np.max(conv_check))
        constraints_violated = np.where(conv_check > M_lp * 1.001)[0]

        if len(constraints_violated) > 0:
            # Add violated constraints to active set
            active_set = np.unique(np.concatenate([active_set, constraints_violated[:5]]))[:top_k + 5]

        # Convex line search
        t_opt, a_new = convex_line_search(a, g_lp, n_evals=18)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - start
                print(f"            LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, |A|={len(active_set)}, time={elapsed:.1f}s")
        else:
            stall_count += 1
            if stall_count > 30:
                # Try aggressive blend
                for t_try in [0.3, 0.5, 0.7, 0.9]:
                    a_try = (1 - t_try) * a + t_try * g_lp
                    a_try = np.maximum(a_try, 0)
                    st = float(np.sum(a_try))
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stall_count = 5
                        break
                else:
                    # Perturb to escape
                    rng = np.random.default_rng((it * 7919 + n) % (2**32))
                    scale = 0.003
                    a_pert = best_a * np.exp(rng.normal(0, scale, n))
                    a_pert = np.maximum(a_pert, 0)
                    s_p = float(np.sum(a_pert))
                    if s_p > 1e-10:
                        a = a_pert / s_p
                    active_set = None
                    stall_count = 0
                    if stall_count > 100:
                        break

    return best_a


def hybrid_lp_subgradient(a, time_budget=200.0, top_k=12, subgrad_per_lp=200, verbose=False):
    """
    Hybrid: alternate between 1 LP step and multiple subgradient steps.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    active_set = None

    while time.monotonic() - start < time_budget:
        # LP step
        if time.monotonic() - start > time_budget:
            break
        
        if active_set is None:
            conv_cur = compute_conv(a)
            max_conv = np.max(conv_cur)
            threshold = max_conv * 0.995
            active_set = np.where(conv_cur >= threshold)[0][:top_k]
            if len(active_set) < 3:
                active_set = np.argsort(conv_cur)[::-1][:top_k]

        g_lp, M_lp = solve_lp_reduced(a, active_set)
        if g_lp is not None:
            t_opt, a_new = convex_line_search(a, g_lp, n_evals=15)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            active_set = None  # Force refresh

        # Subgradient steps
        subgrad_budget = min(2.0, (time_budget - (time.monotonic() - start)) * 0.3)
        if subgrad_budget > 0.5:
            a_sub = nesterov_subgradient(a, time_budget=subgrad_budget, T_init=0.0002, T_min=0.00005, lr_init=0.003)
            obj_sub = compute_obj(a_sub)
            if obj_sub < best_obj - 1e-14:
                best_obj = obj_sub
                best_a = a_sub.copy()
                a = a_sub

        if verbose:
            elapsed = time.monotonic() - start
            print(f"            Hybrid: obj={best_obj:.8f}, time={elapsed:.1f}s")

    return best_a


def spectral_coordinate_descent(a, time_budget=150.0, verbose=False):
    """
    Coordinate descent in the frequency domain.
    Modify individual DFT bins while maintaining a >= 0.
    This allows for "global" moves that real-space CD can't see.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    # DFT size (padded for linear convolution)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute DFT of current a (zero-padded to size)
        a_padded = np.zeros(size)
        a_padded[:n] = a
        FA = rfft(a_padded, size)

        improved = False
        # Try modifying a few random frequency bins
        n_try = min(5, size // 2)
        for _ in range(n_try):
            if time.monotonic() - start > time_budget:
                break
            freq_idx = rng.integers(0, size // 2 + 1)
            
            if freq_idx == 0:
                continue  # DC component is fixed by sum constraint

            # Try scaling this frequency bin
            for scale in [0.9, 0.95, 1.05, 1.1]:
                FA_new = FA.copy()
                FA_new[freq_idx] *= scale
                if freq_idx < size // 2:
                    FA_new[-freq_idx] = FA_new[freq_idx]  # maintain real signal
                a_new = irfft(FA_new, size)[:n]
                a_new = np.maximum(a_new, 0)
                s_new = float(np.sum(a_new))
                if s_new < 0.01:
                    continue
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    FA = FA_new
                    improved = True
                    break

        # Also try real-space coordinate descent
        i = rng.integers(0, n)
        for delta in [1e-4, 3e-4, 1e-3, -1e-4, -3e-4, -1e-3]:
            a_try = a.copy()
            a_try[i] = max(0, a_try[i] + delta)
            s_try = float(np.sum(a_try))
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved and it > 100:
            # Reduce perturbation scale
            pass

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"            Spectral CD it{it}: obj={best_obj:.8f}, time={elapsed:.1f}s")

    return best_a


def fine_coordinate_descent(a, time_budget=120.0, delta=5e-5, verbose=False):
    """Fine coordinate descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    pass_count = 0

    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count * 31 + n)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            best_delta = 0
            best_obj_i = best_obj
            for delta_try in [delta, delta * 0.3, delta * 3, -delta, -delta * 0.3]:
                a_try = a.copy()
                a_try[i] = max(0.0, a_try[i] + delta_try)
                s_try = float(np.sum(a_try))
                if s_try < 0.01:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj_i - 1e-14:
                    best_obj_i = obj_try
                    best_delta = delta_try

            if best_delta != 0:
                a[i] = max(0.0, a[i] + best_delta)
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                best_obj = best_obj_i
                best_a = a.copy()
                improved = True

        if not improved and pass_count > 3:
            delta *= 0.6
            if delta < 1e-8:
                break

        if verbose and pass_count % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"            Fine CD pass {pass_count}: obj={best_obj:.8f}, delta={delta:.2e}, time={elapsed:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    # Use scipy for cubic
    from scipy.interpolate import CubicSpline
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    try:
        cs = CubicSpline(x_old, a, bc_type='natural')
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                parent = parent_arr / s

        pass

            obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()

        # One more LP round
        if remaining > 40:
            alloc = min(60.0, remaining * 0.3)
            a_lp = lp_alternating_fast(a_final, time_budget=alloc, top_k=10)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_final = a_lp.copy()
                print(f"  Final LP: obj={best_score:.8f}")

        # Fine CD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(40.0, remaining * 0.4)
            a_cd = fine_coordinate_descent(a_final, time_budget=alloc, delta=3e-5)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: obj={best_score:.8f}")

        # Quick subgradient
        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(20.0, remaining * 0.5)
            a_sm = nesterov_subgradient(best_seq, time_budget=alloc, T_init=0.0001, T_min=0.00002, lr_init=0.002)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Final SubGrad: obj={best_score:.8f}")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
