    # Cross-correlation (a x w)[i] = sum_j a[j]*w[i+j]
    # = conv(a_rev, w)[(n-1)+i] ... let me verify
    # a_rev[j] = a[n-1-j]
    # conv(a_rev, w)[k] = sum_j a_rev[j]*w[k-j] = sum_j a[n-1-j]*w[k-j]
    # Let m = n-1-j: = sum_m a[m]*w[k-(n-1-m)] = sum_m a[m]*w[k-n+1+m]
    # We want sum_m a[m]*w[i+m], so k-n+1 = i, k = i+n-1
    # So term1[i] = conv(a_rev, w)[i+n-1]
    
    a_rev = a[::-1]
    conv_ar_w = compute_conv_cross(a_rev, w)
    term1 = conv_ar_w[n-1:2*n-1]
    
    return (term1 + term2)


def compute_subgradient_optimal(a, active_set):
    """
    Find optimal lambda on active set that makes subgradient as flat as possible.
    Then return the resulting subgradient.
    """
    n = len(a)
    k_active = len(active_set)
    
    if k_active <= 2:
        return compute_subgradient(a, active_set)
    
    # Compute grad_k for each active k
    # grad_k[i] = a[k-i] + a[i-k] (with bounds)
    # We want to find lambda minimizing ||sum_k lambda_k * grad_k - c*1||^2
    # subject to lambda >= 0, sum lambda = 1
    
    # This is equivalent to: min ||G^T lambda - mean(G^T lambda)||^2
    # where G[i,k] = grad_k[i]
    
    # For efficiency, we only need G^T lambda for given lambda
    # G^T lambda[i] = sum_k lambda_k * (a[k-i] + a[i-k])
    # = sum_k lambda_k * a[k-i] + sum_k lambda_k * a[i-k]
    
    # Let's build the matrix G implicitly and solve the QP
    # For small k_active (say < 100), we can solve directly
    
    if k_active > 200:
        # Too many active constraints, use uniform
        return compute_subgradient(a, active_set)
    
    # Build G matrix (n x k_active) where G[i,k] = grad_k[i]
    G = np.zeros((n, k_active))
    for idx, k in enumerate(active_set):
        for i in range(n):
            val = 0.0
            ki = k - i
            if 0 <= ki < n:
                val += a[ki]
            ik = i - k
            if 0 <= ik < n:
                val += a[ik]
            G[i, idx] = val
    
    # Solve: min ||G^T lambda - mean(G^T lambda)*1||^2 s.t. lambda>=0, sum lambda=1
    # = min lambda^T (G G^T) lambda - (1^T G G^T lambda)^2 / n  ... this is getting complex
    # Simpler: just use uniform or a few iterations of projection
    
    # For now, use a simple approach: iterate lambda proportional to 1/grad_k_max
    # to balance the contributions
    w = np.ones(k_active) / k_active
    
    for _ in range(10):
        # Compute current subgradient
        sg = G @ w
        sg_centered = sg - np.mean(sg)
        # Update: increase weight for constraints that have low gradient contribution
        # (this is like multiplicative weights on the dual)
        # Actually, we want to minimize variance of sg
        # Gradient of ||sg - mean(sg)||^2 w.r.t. lambda_k is:
        # 2 * G_k * (G G^T lambda - mean(G G^T lambda)) ... projecting onto simplex
        
        # Simple approach: w_k ∝ 1 / (standard deviation contribution)
        # Just use uniform for now as it works well in practice
        break
    
    return G @ w


def mirror_descent_optimize(a, time_budget=100.0, eta_init=0.5, eta_decay=0.995,
                            active_threshold=0.95, verbose=False):
    """
    Mirror descent (exponential weight) optimization for min max_k conv(a,a)[k]
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    stall = 0

    max_iters = 20000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)

        # Find active set (within active_threshold of max)
        threshold = max_c * active_threshold
        active_set = np.where(conv >= threshold)[0]

        # Compute subgradient
        if len(active_set) > 50:
            # Use top 50 for efficiency
            top_idx = np.argsort(conv)[::-1][:50]
            active_set_top = top_idx
        else:
            active_set_top = active_set

        grad = compute_subgradient(a, active_set_top)
        grad_centered = grad - np.mean(grad)

        # Mirror descent update: a_new[i] ∝ a[i] * exp(-eta * grad_centered[i])
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * grad_centered
        # Normalize
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        s_new = float(np.sum(a_new))
        if s_new < 1e-300:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 50 == 0:
                elapsed = time.monotonic() - start
                print(f"            MD it{it}: obj={best_obj:.8f}, eta={eta:.5f}, "
                      f"|active|={len(active_set_top)}, time={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 100:
                eta *= 0.7
                if eta < 1e-6:
                    break
                a = best_a.copy()
                stall = 0
                active_threshold = min(0.99, active_threshold * 1.02)

        eta = max(eta * eta_decay, 0.0001)

    return best_a


def lp_alternating(a, time_budget=200.0, verbose=False):
    """
    LP alternating method: solve LP to find best direction, then line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    active_set = None

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        # Solve LP: min M s.t. conv(a, b) <= M, sum(b)=1, b>=0
        g_lp, M_lp = _solve_lp(a, active_set)
        if g_lp is None:
            break

        # Update active set
        conv_cur = compute_conv(a)
        max_conv = np.max(conv_cur)
        threshold = max_conv * 0.97
        active_set = np.where(conv_cur >= threshold)[0]
        if len(active_set) > 300:
            active_set = np.argsort(conv_cur)[::-1][:300]

        # Line search
        t_opt, a_new = _convex_line_search(a, g_lp, n_evals=10)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"          LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, time={elapsed:.1f}s")
        else:
            stall_count += 1
            if stall_count > 30:
                # Perturb
                rng = np.random.default_rng((it * 7919 + n) % (2**31))
                scale = 0.01
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a = a_pert / s_p
                active_set = None
                stall_count = 0
                if stall_count > 100:
                    break

    return best_a


def _solve_lp(a, active_set=None):
    """Solve the LP for direction finding."""
    try:
        from scipy.optimize import linprog
    except ImportError:
        return None, None

    n = len(a)
    a_norm = a / np.sum(a)
    A_full = build_conv_matrix(a_norm)

    if active_set is not None and len(active_set) < 2 * n - 1:
        A = A_full[active_set, :]
    else:
        A = A_full

    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((A.shape[0], n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(A.shape[0])

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                g_val = g_val / s
                # Verify against ALL constraints
                conv_check = A_full @ g_val
                true_M = float(np.max(conv_check))
                return g_val, true_M
    except Exception:
        pass
    return None, None


def _convex_line_search(a, g, n_evals=10):
    """Find optimal t in [0,1] for a_new = (1-t)*a + t*g."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    a_new = a_new / s
    return t_opt, a_new


def coordinate_descent_polish(a, time_budget=60.0, delta=1e-4, verbose=False):
    """Coordinate descent with adaptive step sizes."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    pass_count = 0

    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count * 31 + n % 10007)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            # Try several step sizes for increasing
            for d in [delta, delta * 3, delta * 0.3]:
                a_try = a.copy()
                a_try[i] += d
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                continue

            # Try decreasing
            for d in [delta, delta * 0.3]:
                if a[i] < d * 0.5:
                    continue
                a_try = a.copy()
                a_try[i] -= d
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if verbose and pass_count % 3 == 0:
            print(f"          CD pass {pass_count}: obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")

        if not improved:
            delta *= 0.6
            if delta < 1e-8:
                break

    return best_a


def smooth_max_gradient_descent(a, time_budget=60.0, T_init=0.0005, T_min=0.00001):
    """Gradient descent using smooth max approximation with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.005
    momentum = 0.8
    v = np.zeros(n)
    stall = 0

    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -400, 400)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total

        # Gradient via cross-correlation
        # grad[i] = sum_k weights[k] * (a[k-i] + a[i-k])
        grad = compute_subgradient(a, np.arange(len(weights)))
        # But weights are not uniform, need weighted version
        # For efficiency, compute directly:
        w = weights.copy()
        a_rev = a[::-1]
        conv_ar_w = compute_conv_cross(a_rev, w)
        conv_wa = compute_conv_cross(w, a)
        grad = (conv_ar_w[n-1:2*n-1] + conv_wa[:n])

        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.7
                if lr < 1e-7:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0

        if it > 20:
            T = max(T_min, T_init * (0.997 ** it))

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_parametric_init(n, seed=0):
    """Generate a diverse parametric initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 6)

    if choice == 0:
        # Uniform
        a = np.ones(n)
    elif choice == 1:
        # Cosine window
        a = 0.5 * (1 + np.cos(np.pi * np.arange(n) / (n - 1)))
        a = np.maximum(a, 0.01)
    elif choice == 2:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        a = np.maximum(a, 0.01)
    elif choice == 3:
        # Triangular
        mid = (n - 1) / 2.0
        a = 1.0 - np.abs(np.arange(n) - mid) / mid
        a = np.maximum(a, 0.01)
    elif choice == 4:
        # Inverse cosine (peaks at center)
        a = 0.5 * (1 - np.cos(np.pi * np.arange(n) / (n - 1)))
        a = np.maximum(a, 0.01)
    else:
        # Random smooth
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        a = np.maximum(a, 0.01)

    # Add small random noise
    a = a + rng.normal(0, 0.01, n)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    return a / s

    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 50, n_best - 25, n_best + 25, n_best + 50,
        n_best + 100, n_best - 100,
        700, 750, 800, 850, 900, 950, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1400 and n != n_best]

    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_sequence(best_seq, n_try)
        # Quick mirror descent
        alloc_md = min(30.0, remaining * 0.1)
        a_opt = mirror_descent_optimize(a_try, time_budget=alloc_md, eta_init=0.2, eta_decay=0.997)
        obj_try = compute_obj(a_opt)

        # Quick LP if promising
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.01:
            alloc_lp = min(20.0, remaining * 0.1)
            a_opt = lp_alternating(a_opt, time_budget=alloc_lp)
            obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 9: Final Polish ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a = best_seq.copy()

        # Fine mirror descent
        if remaining > 20:
            alloc = min(60.0, remaining * 0.2)
            a_opt = mirror_descent_optimize(a, time_budget=alloc, eta_init=0.05,
                                            eta_decay=0.999, active_threshold=0.995)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                a = a_opt.copy()
                print(f"  Final MD: obj={best_score:.8f}")

        # Very fine CD
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(40.0, remaining * 0.2)
            a_opt = coordinate_descent_polish(a, time_budget=alloc, delta=1e-5)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final CD: obj={best_score:.8f}")

        # One more LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(40.0, remaining * 0.2)
            a_opt = lp_alternating(a, time_budget=alloc)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final LP: obj={best_score:.8f}")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val) and val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
