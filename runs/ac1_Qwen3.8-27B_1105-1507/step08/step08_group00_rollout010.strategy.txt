def project_simplex(v, epsilon=0.0):
    """Projects onto the simplex (sum=1, nonnegative)."""
    n = len(v)
    # Requires normalization if needed
    if np.sum(v) == 0:
        return np.ones(n) / n
    v = v / np.sum(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

def evaluate_candidate(a, force_compute=True):
    """Evaluates `a` against the official evaluate_sequence function. Return inf if any check fails."""
    # If the candidate is just a list of zeros, we skip computation
    if np.product(a) == 0 and len(a) > 1:
        return compute_obj_from_components(a)
    elif np.max(a) > 1000.0:
        a_clip = a.copy()
        a_clip = np.where(a < 1000, a, 1007)
        return evaluate_sequence(a_clip)
    # If we are to compute, do so carefully
    if force_compute:
        # Precisely follow the evaluator's method, but with defensive checks (because the parent was clipped)
        n = len(a)
        if n == 0:
            return np.inf
        sum_a = np.sum(a)
        if sum_a < 0.01:
            # Rescaling to meet the evaluator's minimum? Never change the sum arbitrarily
            # because the objective is invariant under scaling (as long as the max remains)
            # so we stick to the original clipped parent
            return np.inf
        # The official evaluator does this computation
        try:
            seq_max = float(max(a)) if type(a) is not np.ndarray else float(np.max(a))
            a_arr = np.array(a)
            if sum_a >= 0.01 and seq_max < 1000.0 and np.all(a_arr))
                conv = np.convolve(a_arr, a_arr, mode="full")
            else:
                conv = compute_conv_fft_safe(a)
        except Exception:
            # Something went wrong, but we don't know what
            return np.inf
    else:
        conv = compute_conv_fft_safe(a)
    # The objective is 2*n*∞-norm(conv)/sum(a)**2
    # Now, we need to compute the max of the convolution and the sum squared
    max_conv = np.max(conv)
    denom = sum_a ** 2 if sum_a >= 0.01 else 0.01**2
    return float(2 * n * max_conv / denom)

def compute_obj_from_components(a):

    full_a = a if not isinstance(a, list) else list(a)
    n_val = len(full_a)

        return np.inf  # sum must be at least 0.01

def compute_conv_fft_safe(a, normalizer=True):
    """Compute convolution safely, without relying on the official implementation."""
    n = int(len(a))
    size = 2*n - 1
    if size==0:
        return np.array([])
    try:
        fa = np.fft.rfft(a.astype(np.float64), size)// Wait, common error: the length might need adjustment for FFT caching (fourier))

        # Actually, standard FFT convolution: 
        max_size = 2*n - 1
        new_size = 1
        while new_size < max_size:
            new_size <<= 1
        # If new_size is exactly max_size, we can minimize memory allocation
        if new_size == max_size:
            fa = np.fft.rfft(a, size=max_size)
            conv = np.fft.irfft(fa * fa, size=max_size)
        else:
            fa = np.fft.rfft(a, new_size) if new_size > size else np.fft.rfft(a, size)
            conv = np.fft.irfft(fa * fa, new_size)
        return conv[:2*n-1]  # (n*log(n)) operation
    except Exception as e:
        # If something goes wrong, fall back to
        print(f"FFT error: {e}")
        try:
            # Manual convolution in bounded time (acceptable only if n is small)
            conv = np.zeros(2*n-1)
            for i in range(len(a)):
                for j in range(len(a)):
                    if (i+j) < len(conv):
                        conv[i+j] += a[i] * a[j]
            if normalizer:
                norm_val = np.sqrt(2*n)  # Only if we are precisely mimicking the previous program
                conv = conv / norm_val
            return conv
        except:
            return np.full(2*n-1, 10.0)
