"""
Step function optimizer using smooth-max gradient (FFT-accelerated) with Adam,
LP alternating polishing, multi-scale search, and local refinement.
Targets improvement over 1.506330 toward 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_max_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*conv(a,a)[k])) w.r.t. a.
    Returns (gradient, softmax_weights).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    w = np.exp(beta * (conv - max_c))
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(2*n-1)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # cross-correlation: C[m] = sum_k w[k]*a[k-m]
    # C = ifft(FFT(w)*conj(FFT(a))) with zero-padding
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fw = rfft(w, size)
    fa = rfft(a, size)
    C = irfft(fw * np.conj(fa), size)
    grad = 2.0 * C[:n] / beta
    return grad, w


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = int(np.max(ind[cond]))
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

def build_conv_matrix(a_norm):
    """Build (2n-1) x n matrix A where A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp(a):
    """
    LP: min M s.t. conv(a/g)*g <= M, sum(g)=1, g>=0.
    Actually: min M s.t. conv(a_norm, g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None

    stall = 0
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:

def resample(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def local_refine(a0, time_budget, n_peaks=5):
    """Local refinement: move mass between positions to reduce peak convolution."""
    n = len(a0)
    a = a0 / float(np.sum(a0))
    best_a = a.copy()

    start = time.monotonic()

        peak_idx = int(np.argmax(conv))
        # Find contributors to the peak
        i_start = max(0, peak_idx - n + 1)
        i_end = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        for i in range(i_start, i_end + 1):
            m = peak_idx - i
            contrib[m] += a[i]
            if m != i:
                contrib[i] += a[m]
        
        # Try moving mass from high to low contributors
        top_idx = np.argsort(contrib)[::-1][:5]
        bot_idx = np.argsort(contrib)[:5]
        
        improved = False
        for src in top_idx:
            for dst in bot_idx:
                if src == dst or a[src] < 1e-12:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = frac * a[src]
                    a[src] -= delta
                    a[dst] += delta
                    st = float(np.sum(a))
                    a = a / st
                    obj = 2.0 * n * float(np.max(compute_conv(a)))
                    if obj < best_obj - 1e-12:

                        break
                    else:

                        a[dst] -= delta
                        a = a / st
                        break
        if not improved:
            break
    
    return best_a


def hann_window(n):
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    return w / sw if sw > 1e-10 else np.ones(n) / n


def blackman_harris(n):
    """Blackman-Harris window."""
    a0, a1, a2, a3 = 0.35875, 0.48829, 0.14128, 0.01168
    m = np.arange(n)
    w = a0 - a1 * np.cos(2 * np.pi * m / (n - 1)) + a2 * np.cos(4 * np.pi * m / (n - 1)) - a3 * np.cos(6 * np.pi * m / (n - 1))
    sw = np.sum(w)
    return w / sw if sw > 1e-10 else np.ones(n) / n

Step function optimizer using smooth-max gradient (FFT-accelerated) with Adam/Nesterov,
LP alternating polishing, multi-scale search, and local refinement.
Targets improvement over 1.506330 toward 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def _compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def _compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = _compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def _smooth_max_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*conv(a,a)[k])) w.r.t. a.
    Returns (gradient, softmax_weights).
    """
    n = len(a)
    conv = _compute_conv(a)
    max_c = float(np.max(conv))
    w = np.exp(beta * (conv - max_c))
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    # cross-correlation: C[m] = sum_k w[k]*a[k-m]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fw = rfft(w, size)
    fa = rfft(a, size)
    C = irfft(fw * np.conj(fa), size)
    grad = 2.0 * C[:n] / beta
    return grad, w


def _project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = int(np.max(ind[cond]))
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

        a = a / a_sum

        grad_f, _ = _smooth_max_grad(a, beta)
        mean_g = float(np.dot(grad_f, a))
        grad_theta = a * (grad_f - mean_g)

        m = b1 * m + (1 - b1) * grad_theta
        v = b2 * v + (1 - b2) * grad_theta ** 2
        m_hat = m / (1 - b1 ** (t + 1))
        v_hat = v / (1 - b2 ** (t + 1))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 300 == 0:
            obj = 2.0 * n * float(np.max(_compute_conv(a)))
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 40:
                lr *= 0.5
                patience = 0
                if lr < 1e-7:
                    break

    if best_a is None:
        a = np.exp(theta - np.max(theta))
        best_a = a / float(np.sum(a))
    return best_a


def _nesterov_simplex(a0, beta, time_budget, lr=0.05):
    """Nesterov-accelerated projected gradient on the simplex."""
    n = len(a0)
    a = a0 / float(np.sum(a0))
    y = a.copy()
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(_compute_conv(a)))
    start = time.monotonic()
    patience = 0

    for t in range(200000):
        if time.monotonic() - start > time_budget:
            break

        grad_f, _ = _smooth_max_grad(y, beta)
        grad_c = grad_f - float(np.mean(grad_f))

        alpha = 2.0 / (t + 2.0)
        y_new = y - lr * grad_c
        a_new = _project_simplex(y_new)
        y = a_new + (1.0 - alpha) * (a_new - a)
        y = _project_simplex(y)
        a = a_new

        if t % 200 == 0:
            obj = 2.0 * n * float(np.max(_compute_conv(a)))
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 60:
                lr = max(lr * 0.5, 0.001)
                patience = 0
                if lr < 0.002:
                    break

    return best_a


def _build_conv_matrix(a_norm):
    """Build (2n-1) x n matrix A where A @ g = conv(a_norm, g)."""
    n = len(a_norm)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def _solve_lp(a):
    """LP: min M s.t. conv(a_norm, g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = _build_conv_matrix(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def _lp_alternate(a0, time_budget, verbose=False):
    """LP alternating with line search."""
    n = len(a0)
    a = a0 / float(np.sum(a0))
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(_compute_conv(a)))
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = _solve_lp(a)
        if g_lp is None:
            break

        def _line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return 2.0 * n * float(np.max(_compute_conv(a_try)))

        try:
            res = minimize_scalar(_line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 40, 'xatol': 0.005})
            best_t = res.x
        except Exception:
            best_t = 0.5

        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = 2.0 * n * float(np.max(_compute_conv(a_try)))

        if obj_try < best_obj - 1e-10:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            stall += 1
            if stall > 12:
                break

    return best_a


def _resample(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def _local_refine(a0, time_budget, n_peaks=5):
    """Local refinement: move mass between positions to reduce peak convolution."""
    n = len(a0)
    a = a0 / float(np.sum(a0))
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(_compute_conv(a)))
    start = time.monotonic()

    for outer in range(500):
        if time.monotonic() - start > time_budget:
            break
        conv = _compute_conv(a)
        peak_idx = int(np.argmax(conv))
        i_start = max(0, peak_idx - n + 1)
        i_end = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        for i in range(i_start, i_end + 1):
            m = peak_idx - i
            contrib[m] += a[i]
            if m != i:
                contrib[i] += a[m]

        top_idx = np.argsort(contrib)[::-1][:6]
        bot_idx = np.argsort(contrib)[:6]

        improved = False
        for src in top_idx:
            for dst in bot_idx:
                if src == dst or a[src] < 1e-12:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    delta = frac * a[src]
                    a[src] -= delta
                    a[dst] += delta
                    st = float(np.sum(a))
                    a = a / st
                    obj = 2.0 * n * float(np.max(_compute_conv(a)))
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[dst] -= delta
                        a = a / st
                        break
        if not improved:
            break

    return best_a


def _hann_window(n):
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    return w / sw if sw > 1e-10 else np.ones(n) / n


def _blackman_harris(n):
    a0, a1, a2, a3 = 0.35875, 0.48829, 0.14128, 0.01168
    m = np.arange(n)
    w = (a0 - a1 * np.cos(2 * np.pi * m / (n - 1))
         + a2 * np.cos(4 * np.pi * m / (n - 1))
         - a3 * np.cos(6 * np.pi * m / (n - 1)))
    sw = np.sum(w)
    return w / sw if sw > 1e-10 else np.ones(n) / n


def _kaiser_window(n, beta_k=8.0):
    from scipy.special import ive
    m = np.arange(n)
    alpha = (n - 1) / 2.0
    w = np.ones(n)
    for i in range(n):
        x = (i - alpha) / alpha
        w[i] = ive(0, beta_k * np.sqrt(1 - x * x))
    sw = np.sum(w)
    return w / sw if sw > 1e-10 else np.ones(n) / n


def propose_candidate(time_budget=1500):
    """
    Main search combining smooth-max Adam, Nesterov simplex, LP alternating,
    local refinement, multi-scale search, and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(time_budget) - 30.0
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = _compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = _hann_window(800)
        best_score = _compute_obj(best_seq)
        print(f"Hann init (n=800): obj={best_score:.8f}")

    # === Phase 1: Diverse initializations + quick optimization ===
    print("\n=== Phase 1: Diverse Inits + Adam/Nesterov ===")
    n_base = len(best_seq)

    inits = [best_seq.copy()]
    inits.append(_hann_window(n_base))
    inits.append(_blackman_harris(n_base))
    inits.append(_kaiser_window(n_base, 6.0))
    inits.append(_kaiser_window(n_base, 10.0))
    gauss = np.exp(-0.5 * ((np.arange(n_base) - n_base / 2.0) / (n_base / 4.0)) ** 2)
    inits.append(gauss / np.sum(gauss))
    r_init = rng.random(n_base)
    inits.append(r_init / np.sum(r_init))

    for i, init in enumerate(inits):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        obj = _compute_obj(init)
        if obj < best_score:
            best_score = obj
            best_seq = init.copy()
            print(f"  Init {i}: obj={best_score:.8f} **")
        else:
            print(f"  Init {i}: obj={obj:.8f}")

        if remaining > 30:
            alloc = min(20.0, remaining * 0.2)
            theta_init = np.log(init + 1e-12)
            for beta_val in [5.0, 20.0]:
                remaining = deadline - time.monotonic()
                if remaining < 10:
                    break
                a_ad = _adam_softmax_opt(theta_init, beta_val, alloc * 0.4, lr=0.004)
                obj_ad = _compute_obj(a_ad)
                if obj_ad < best_score:
                    best_score = obj_ad
                    best_seq = a_ad.copy()
                    theta_init = np.log(a_ad + 1e-12)
                    print(f"    Adam beta={beta_val}: obj={best_score:.8f} **")
                a_ns = _nesterov_simplex(a_ad, beta_val, alloc * 0.3, lr=0.03)
                obj_ns = _compute_obj(a_ns)
                if obj_ns < best_score:
                    best_score = obj_ns
                    best_seq = a_ns.copy()
                    print(f"    Nest beta={beta_val}: obj={best_score:.8f} **")

    # === Phase 2: n-sweep ===
    print("\n=== Phase 2: n-Sweep ===")
    n_options = [300, 500, 700, 800, 1000, 1200, 1500, 2000, 2500]
    n_options = sorted(set(n_options + [n_base]))
    n_options = [n for n in n_options if 150 <= n <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        if abs(n_try - len(best_seq)) < 20:
            continue
        a_try = _resample(best_seq, n_try)
        alloc = min(12.0, remaining * 0.12)
        theta_t = np.log(a_try + 1e-12)
        a_opt = _adam_softmax_opt(theta_t, 10.0, alloc, lr=0.004)
        obj_opt = _compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 3: LP Alternating (main polishing) ===
    print("\n=== Phase 3: LP Alternating ===")
    lp_seq = best_seq.copy()
    for round_idx in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(55.0, remaining * 0.22)
        a_lp = _lp_alternate(lp_seq, time_budget=alloc, verbose=True)
        obj_lp = _compute_obj(a_lp)
        if obj_lp < best_score - 1e-10:
            best_score = obj_lp
            best_seq = a_lp.copy()
            lp_seq = a_lp.copy()
            print(f"  LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj_lp:.8f}")
            # Perturb to escape
            n_cur = len(lp_seq)
            lp_seq = lp_seq * np.exp(rng.normal(0, 0.012, n_cur))
            lp_seq = np.maximum(lp_seq, 0)
            lp_seq /= float(np.sum(lp_seq))

    # === Phase 4: Adam polish with increasing beta ===
    print("\n=== Phase 4: Adam Polish (increasing beta) ===")
    theta_cur = np.log(best_seq + 1e-12)
    for beta_val in [8.0, 25.0, 80.0, 250.0, 800.0]:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(35.0, remaining * 0.18)
        a_pd = _adam_softmax_opt(theta_cur, beta_val, alloc, lr=0.003)
        obj_pd = _compute_obj(a_pd)
        if obj_pd < best_score:
            best_score = obj_pd
            best_seq = a_pd.copy()
            theta_cur = np.log(a_pd + 1e-12)
            print(f"  Adam beta={beta_val}: obj={best_score:.8f} **")
        else:
            print(f"  Adam beta={beta_val}: obj={obj_pd:.8f}")

    # === Phase 5: Local refinement ===
    print("\n=== Phase 5: Local Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(50.0, remaining * 0.25)
        a_lr = _local_refine(best_seq, time_budget=alloc)
        obj_lr = _compute_obj(a_lr)
        if obj_lr < best_score:
            best_score = obj_lr
            best_seq = a_lr.copy()
            print(f"  Local: obj={best_score:.8f} **")
        else:
            print(f"  Local: obj={obj_lr:.8f}")

    # === Phase 6: Final LP rounds ===
    print("\n=== Phase 6: Final LP ===")
    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        alloc = min(45.0, remaining * 0.3)
        a_f = _lp_alternate(best_seq, time_budget=alloc, verbose=True)
        obj_f = _compute_obj(a_f)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  Final LP {r}: obj={best_score:.8f} **")
        else:
            print(f"  Final LP {r}: obj={obj_f:.8f}")

    # === Phase 7: Random restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 7: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(12):
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break
            scale = 0.02 + 0.008 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr
            alloc = min(remaining * 0.25, 12.0)
            theta_r = np.log(a_r + 1e-12)
            a_opt = _adam_softmax_opt(theta_r, 20.0, alloc * 0.5, lr=0.004)
            a_opt = _lp_alternate(a_opt, time_budget=alloc * 0.5)
            obj_r = _compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  R{r}: obj={best_score:.8f} **")
            else:
                print(f"  R{r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = _compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)

    else:

        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
