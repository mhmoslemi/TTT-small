"""
Optimized step function search using active-set LP with exact line search,
Riemannian mirror descent, and guided perturbations.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project vector onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def active_set_lp(a, K=80, max_time=2.0):
    """
    Solve LP with only top K active constraints.
    min M s.t. (A@g)[k] <= M for k in active_set, sum(g)=1, g>=0
    Returns g or None.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None
    a = a / s
    
    # Compute current convolution to find active set
    conv_current = compute_conv(a)
    # Top K peaks
    top_k = np.argsort(conv_current)[::-1][:K]
    
    # Build reduced constraint matrix
    m = len(top_k)
    A = np.zeros((m, n))
    for idx, k in enumerate(top_k):
        k = int(k)
        i_max = min(n - 1, k)
        i_min = max(0, k - n + 1)
        for i in range(i_min, i_max + 1):
            A[idx, i] = a[k - i]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    start = time.monotonic()
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        if time.monotonic() - start > max_time:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs
        except Exception:
            continue
    
    # Fallback: scipy linprog with reduced constraints
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass
    
    return None


def lp_direction_search(a, time_budget=300.0, K=80, verbose=False):
    """
    LP-based alternating with active sets and line search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iteration = 0
    
    while time.monotonic() - start < time_budget:
        # Get LP direction
        g_lp = active_set_lp(a, K=K)
        if g_lp is None:
            break
        
        # Line search along direction
        direction = g_lp - a
        best_t = 0.5
        best_local_obj = best_obj
        
        # Try multiple step sizes with some smart choices
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = a + t * direction
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_t = t
        
        if best_local_obj < best_obj - 1e-15:
            best_obj = best_local_obj
            a_try = a + best_t * direction
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)
            best_a = a_try.copy()
            a = best_a
            if verbose and iteration % 10 == 0:
                print(f"    LP-AS iter {iteration}: obj={best_obj:.8f}, t={best_t:.2f}")
            iteration += 1
        else:
            # Stall: try perturbing
            iteration += 1
            if iteration % 5 == 0:
                rng = np.random.default_rng(hash((iteration, int(time.monotonic()*1000))) % (2**31))
                scale = 0.02 + 0.005 * min(iteration // 20, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
    
    return best_a


def mirror_descent_polish(a, time_budget=100.0, verbose=False):
    """
    Mirror descent on the simplex (exponential family parameterization).
    Uses the fact that for the non-negative simplex, the mirror map is log.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Work in log-space for mirror descent
    log_a = np.log(a + 1e-15)
    
    lr = 0.01
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Re-normalize a from log_a
        a = np.exp(log_a)
        a = a / np.sum(a)
        
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Find active peaks
        threshold = max_val * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = [int(np.argmax(conv))]
        if len(active) > 50:
            active = np.argsort(conv)[::-1][:50]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += a[k - indices]
        grad /= len(active)
        
        # Mirror descent step: update log_a
        log_a = log_a - lr * grad
        # Project to keep things bounded
        log_a = np.clip(log_a, -30, 30)
        
        # Check objective
        a_new = np.exp(log_a)
        a_new = a_new / np.sum(a_new)
        obj = compute_obj(a_new)
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
        
        # Decay learning rate
        it += 1
        if it % 100 == 0:
            lr *= 0.95
        
        if lr < 0.0001:
            break
    
    return best_a

    """

def generate_initialization(kind, n, rng):
    """Generate a diverse initialization of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'edge_bias':
        # High at edges, low in middle
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (2*x - 1)**2
        return a / np.sum(a)
    elif kind == 'double_peak':
        # Two peaks at 1/4 and 3/4
        x = np.arange(n)
        a = np.exp(-0.5*((x - n//4)/(n//8))**2) + np.exp(-0.5*((x - 3*n//4)/(n//8))**2) + 0.1
        return a / np.sum(a)
    elif kind == 'random_smooth':
        a = rng.random(n)
        # Smooth with a kernel
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use cubic interpolation for smoother results
    from scipy.interpolate import PchipInterpolator
    try:
        interp = PchipInterpolator(x_old, a, extrapolate=False)
        a_new = interp(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def pairwise_exchange(a, time_budget=60.0, n_trials_per_iter=20):

    Pairwise exchange: try moving mass from one position to another.

    a = np.maximum(a.astype(np.float64), 0).copy()

    iteration = 0

        k_star = int(np.argmax(conv))
        peak = conv[k_star]
        
        # Compute contributions to the peak
        i_max = min(n - 1, k_star)
        i_min = max(0, k_star - n + 1)
        contributions = np.zeros(n)
        for i in range(i_min, i_max + 1):
            contributions[i] = a[i] * a[k_star - i]
        
        # Find positions with high contribution
        high_contrib = np.argsort(contributions)[::-1][:30]
        # Find positions with low contribution (potential targets)
        low_contrib = np.argsort(contributions)[:30]
        
        improved = False
        rng = np.random.default_rng(hash((iteration, n)) % (2**31))
        
        for _ in range(n_trials_per_iter):
            i = high_contrib[rng.integers(0, len(high_contrib))]
            # Choose target: prefer positions far from the peak contributors
            targets = [j for j in low_contrib if j != i and abs(j - (k_star - int(i))) > n//4]
            if not targets:
                targets = [j for j in low_contrib if j != i]
            if not targets:
                continue
            j = targets[rng.integers(0, len(targets))]
            
            # Try different fractions
            for frac in [0.01, 0.02, 0.05, 0.1]:
                delta = frac * a[i]
                if delta <= 0 or delta > a[i]:
                    continue
                a[i] -= delta
                a[j] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
        
        if not improved:
            iteration += 1
            if iteration > 100:
                break
    
    return best_a

"""
Optimized step function search using full-constraint LP alternating with
backtracking line search, guided restarts via LP dual, Riemannian mirror

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_full_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_full_lp(a, max_time=5.0):

    Solve the full LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    Returns (g, M_opt, dual_vars) or (None, None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))

        return None, None, None

    A = build_full_conv_matrix(a)
    m = 2 * n - 1

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    start = time.monotonic()
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        if time.monotonic() - start > max_time:
            break
        try:

                    # Extract dual variables for constraint A@g <= M

                    try:

                        if dv is not None:

                        dual = None

        A_ub[:, :n] = A
        A_ub[:, n] = -1.0

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0

def mirror_descent(a, time_budget=100.0, verbose=False):
    """
    Mirror descent on the simplex using log-parameterization.
    Focuses on reducing active peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    log_a = np.log(a + 1e-15)
    lr = 0.02
    it = 0

    while time.monotonic() - start < time_budget:
        a = np.exp(log_a)
        a = a / np.sum(a)

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        threshold = max_val * 0.92
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = [int(np.argmax(conv))]
        if len(active) > 60:
            active = np.argsort(conv)[::-1][:60]

        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += a[k - indices]
        grad /= len(active)

        # Center the gradient (project to hyperplane sum=0 for simplex)
        grad_centered = grad - np.mean(grad)

        log_a = log_a - lr * grad_centered
        log_a = np.clip(log_a, -40, 40)

        it += 1
        if it % 50 == 0:
            lr *= 0.92
            a_new = np.exp(log_a)
            a_new = a_new / np.sum(a_new)
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a_new.copy()
                if verbose and it % 200 == 0:
                    print(f"    MD iter {it}: obj={best_obj:.8f}, lr={lr:.5f}")

        if lr < 0.00005:
            break

    return best_a


def pairwise_exchange_opt(a, time_budget=80.0, verbose=False):
    """
    Focused pairwise exchange: move mass from peak contributors to reduce max conv.
    Uses smart target selection based on convolution structure.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    rng = np.random.default_rng(999 + n)

    no_improve = 0
    while time.monotonic() - start < time_budget:
        k_star = int(np.argmax(conv))

        # Compute gradient of the peak with respect to each a[i]
        # d(conv[k_star])/da[i] = a[k_star - i] (if valid) + a[i - k_star] (if valid)
        # But since we're looking at self-convolution:
        # d(conv[k])/da[i] = a[k-i] if 0<=k-i<n, plus a[i-k] if k!=0 and 0<=i-k<n
        # Actually for self-conv: conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/da[i] = a[k-i] (from j=i term) + a[i-k] (from k-j=i, j=k-i term)
        # If k==i: a[0] + (a[0] if k-i=0 and i!=0... actually let's just compute directly)

        i_max = min(n - 1, k_star)
        i_min = max(0, k_star - n + 1)
        contributions = np.zeros(n)
        for i in range(i_min, i_max + 1):
            contributions[i] = a[i] * a[k_star - i]

        # Sort positions by contribution to peak
        sorted_high = np.argsort(contributions)[::-1]
        top_contributors = sorted_high[:40]

        # For each top contributor, find good exchange targets
        improved = False
        for i in top_contributors:
            if time.monotonic() - start > time_budget:
                break
            i = int(i)
            if a[i] < 1e-10:
                continue

            # Good targets: positions that don't contribute much to ANY of the top peaks
            # Approximate: positions far from k_star - i and k_star + i
            target_score = np.zeros(n)
            # Penalize positions near the main peak complement
            comp = k_star - i
            for j in range(n):
                dist1 = abs(j - comp)
                dist2 = abs(j - i)
                target_score[j] = min(dist1, dist2)
            # Add some randomness
            target_score += rng.random(n) * n * 0.1
            top_targets = np.argsort(target_score)[::-1][:30]

            for j in top_targets:
                j = int(j)
                if j == i:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.04]:
                    delta = frac * a[i]
                    if delta <= 0 or delta >= a[i]:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-15:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                if improved:
                    break

        if not improved:
            no_improve += 1
            # Random perturbation to escape
            if no_improve > 5:
                scale = 0.01 + 0.005 * min(no_improve // 10, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                conv = compute_conv(a)
                no_improve = 0
            if no_improve > 100:
                break

    return best_a


def generate_initialization(kind, n, rng):
    """Generate diverse initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'edge_bias':
        x = np.linspace(0, 1, n)
        a = 0.2 + 0.8 * (2 * x - 1) ** 2
        return a / np.sum(a)
    elif kind == 'double_peak':
        x = np.arange(n)
        a = (np.exp(-0.5 * ((x - n // 4) / (n // 10)) ** 2) +
             np.exp(-0.5 * ((x - 3 * n // 4) / (n // 10)) ** 2) + 0.05)
        return a / np.sum(a)
    elif kind == 'random_smooth':
        a = rng.random(n) + 0.01
        kernel_len = max(3, n // 50)
        kernel = np.ones(kernel_len) / kernel_len
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'four_peak':
        x = np.arange(n)
        a = 0.05
        for frac in [0.15, 0.35, 0.65, 0.85]:
            a = a + np.exp(-0.5 * ((x - frac * n) / (n // 12)) ** 2)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def quick_optimize(a, time_budget=30.0, K=80):
    """Quick optimization: a few LP steps + mirror descent."""
    # LP phase
    lp_time = time_budget * 0.6
    a = lp_alternating_full(a, time_budget=lp_time, use_full_lp=False)
    # MD phase
    md_time = time_budget * 0.4
    a = mirror_descent(a, time_budget=md_time)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: full-constraint LP with backtracking, guided restarts,
    mirror descent, pairwise exchange, and systematic length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                best_seq = parent_arr / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={len(best_seq)}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Full LP alternating with backtracking (500s) ===
    print(f"\n=== Phase 1: Full LP + Backtracking (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(500.0, remaining * 0.4)

    # Three rounds with different settings
    for round_idx in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(lp_budget / 3.5, remaining * 0.5)
        use_full = (round_idx < 2)  # First two rounds use full LP, third uses active set
        a_new = lp_alternating_full(best_seq, time_budget=alloc,
                                     verbose=True, use_full_lp=use_full)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  ** LP R{round_idx} (full={use_full}): obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx} (full={use_full}): obj={obj:.8f}")

    # === Phase 2: Mirror descent polish (200s) ===
    print(f"\n=== Phase 2: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        md_budget = min(200.0, remaining * 0.2)
        a_md = mirror_descent(best_seq, time_budget=md_budget, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  ** MD: obj={best_score:.8f} **")
        else:
            print(f"  MD: obj={obj_md:.8f}")

    # === Phase 3: Pairwise exchange (150s) ===
    print(f"\n=== Phase 3: Pairwise Exchange ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        pe_budget = min(150.0, remaining * 0.15)
        a_pe = pairwise_exchange_opt(best_seq, time_budget=pe_budget, verbose=True)
        obj_pe = compute_obj(a_pe)
        if obj_pe < best_score:
            best_score = obj_pe
            best_seq = a_pe.copy()
            print(f"  ** PE: obj={best_score:.8f} **")
        else:
            print(f"  PE: obj={obj_pe:.8f}")

    # === Phase 4: Another LP round from improved point (200s) ===
    print(f"\n=== Phase 4: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        lp2_budget = min(200.0, remaining * 0.2)
        a_new = lp_alternating_full(best_seq, time_budget=lp2_budget,
                                     verbose=True, use_full_lp=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  ** LP R2: obj={best_score:.8f} **")
        else:
            print(f"  LP R2: obj={obj:.8f}")

    # === Phase 5: Length sweep (250s) ===
    print(f"\n=== Phase 5: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 200),
        max(200, n_best - 100),
        max(200, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 400,
        n_best + 800,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 6000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(remaining * 0.2, 50.0)
        a_opt = quick_optimize(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse initializations at best length (150s) ===
    print(f"\n=== Phase 6: Diverse Inits ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    init_kinds = ['edge_bias', 'double_peak', 'four_peak', 'random_smooth',
                  'cosine', 'gaussian']
    for kind in init_kinds:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        a_init = generate_initialization(kind, n_final, rng)
        alloc = min(remaining * 0.12, 25.0)
        a_opt = quick_optimize(a_init, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** Init '{kind}': obj={best_score:.8f} **")
        else:
            print(f"  Init '{kind}': obj={obj:.8f}")

    # === Phase 7: Final intensive optimization (remaining time) ===
    print(f"\n=== Phase 7: Final Optimization ===")
    remaining = deadline - time.monotonic()

    while remaining > 25:
        # Alternate: LP -> MD -> Pairwise
        phase_alloc = min(remaining * 0.25, 60.0)

        # LP
        a_new = lp_alternating_full(best_seq, time_budget=phase_alloc * 0.5,
                                     use_full_lp=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  ** Final LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # MD
        a_md = mirror_descent(best_seq, time_budget=phase_alloc * 0.3)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  ** Final MD: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Pairwise
        a_pe = pairwise_exchange_opt(best_seq, time_budget=phase_alloc * 0.2)
        obj_pe = compute_obj(a_pe)
        if obj_pe < best_score:
            best_score = obj_pe
            best_seq = a_pe.copy()
            print(f"  ** Final PE: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
