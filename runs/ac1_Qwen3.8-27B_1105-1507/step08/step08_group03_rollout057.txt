"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP alternating with adaptive step sizes, peak-shaving QP,
smooth log-sum-exp optimization, and multi-scale n exploration.
Key improvements over previous: better LP warm-starting, dual-guided updates,
peak-specific penalty, and more efficient convergence.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize as sp_minimize, minimize_scalar

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except Exception:
    HAVE_CVXPY = False


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    # Fallback to scipy
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    if warm_start is not None:
        ws = list(warm_start) + [0]
    else:
        ws = None
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def solve_lp_maxsum(a, M_target):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M_target, g >= 0."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M_target]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x, 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None


def lp_alternating_improved(a, time_budget=200.0, verbose=False):
    """
    Improved LP alternating with:
    - Adaptive step size (not fixed convex combination)
    - Dual-informed weight updates
    - Periodic re-linearization
    - Better stagnation handling
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    t_best = 0.5
    rng = np.random.default_rng(12345)
    it = 0
    
    # Track history for adaptive step size
    obj_history = [best_obj]

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break

        # Solve LP: minimize peak of conv(a, g)
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            # Fallback: use max-sum LP
            conv_a = compute_conv(a)
            M_cur = float(np.max(conv_a))
            g_lp = solve_lp_maxsum(a, M_cur)
            if g_lp is None:
                break

        # Adaptive line search: try multiple step sizes
        # Focus around best_t with some exploration
        base_ts = [t_best, t_best * 0.7, t_best * 1.3, 0.3, 0.5, 0.7, 1.0]
        # Add some random exploration
        base_ts.append(rng.uniform(0.1, 1.0))
        ts = sorted(set([max(0.01, min(1.0, t)) for t in base_ts]))

        best_t_obj = np.inf
        best_t_val = t_best
        best_a_local = best_a.copy()

        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            t_best = best_t_val
            stall = 0
            obj_history.append(best_obj)
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 10:
                # Escape: perturb and restart
                noise_scale = min(0.4, 0.05 + 0.03 * (stall // 10))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                # Re-evaluate
                obj_new = compute_obj(a)
                if obj_new < best_obj:
                    best_a = a.copy()
                    best_obj = obj_new
                stall = 0
                t_best = rng.uniform(0.3, 0.8)
    
    return best_a


def peak_shave(a, time_budget=60.0, K=5):
    """
    Peak shaving: identify top K peaks and solve a QP to reduce them.
    Uses the smooth approximation with explicit peak penalties.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Iteratively identify peaks and add penalties
    T = 10.0  # Temperature for log-sum-exp
    penalty_weights = np.zeros(2 * n - 1)

    for outer in range(10):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = conv.max()
        
        # Update penalty weights: focus on peaks near the max
        threshold = 0.95 * max_val
        active = conv > threshold
        # Distribute weight among active peaks
        active_weights = np.exp((conv[active] - max_val) / (T * max_val))
        penalty_weights[active] = active_weights / active_weights.sum()
        penalty_weights[~active] = 0

        # Define smooth objective with peak penalty
        def obj_pen(x):
            x = np.maximum(x, 1e-15)
            x = x / x.sum()
            conv_x = compute_conv(x)
            max_c = conv_x.max()
            # Log-sum-exp approximation of max
            z = 20.0 * (conv_x - max_c)
            lse = max_c + np.log(np.exp(z).sum()) / 20.0
            # Peak penalty
            peak_pen = sum(penalty_weights[k] * conv_x[k] for k in range(len(conv_x)) if penalty_weights[k] > 1e-6)
            return 2.0 * n * (lse + 0.1 * peak_pen)

        def obj_pen_grad(x):
            x = np.maximum(x, 1e-15)
            x = x / x.sum()
            conv_x = compute_conv(x)
            max_c = conv_x.max()
            z = 20.0 * (conv_x - max_c)
            e = np.exp(z)
            sum_e = e.sum()
            w_lse = e / sum_e
            
            g = np.zeros(n)
            for k in range(len(conv_x)):
                w_total = w_lse[k]
                if penalty_weights[k] > 1e-6:
                    w_total += 0.1 * penalty_weights[k]
                if w_total > 1e-10:
                    g += w_total * grad_conv_at_peak(x, k)
            g *= 2.0 * n
            
            # Subtract mean gradient for simplex constraint
            g -= g.mean()
            return obj_pen(x), g

        # Use L-BFGS-B
        try:
            res = sp_minimize(obj_pen_grad, a, method='L-BFGS-B',
                             bounds=[(1e-12, None)] * n,
                             options={'maxiter': 100, 'disp': False})
            a_opt = np.maximum(res.x, 0)
            s_opt = float(np.sum(a_opt))
            if s_opt > 1e-10:
                a_opt = a_opt / s_opt
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_obj:
                    best_obj = obj_opt
                    best_a = a_opt.copy()
                    a = a_opt
        except Exception:
            pass

    return best_a


def smooth_lse(a, time_budget=60.0, T=30.0):
    """
    Smooth optimization using log-sum-exp with adaptive temperature.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def make_obj_grad(T_val):
        def _obj_grad(x):
            x = np.maximum(x, 1e-15)
            s = x.sum()
            if s < 1e-10:
                return 1e10, np.zeros(len(x))
            x = x / s
            conv = compute_conv(x)
            max_c = conv.max()
            z = T_val * (conv - max_c)
            e = np.exp(z)
            sum_e = e.sum()
            obj_val = 2.0 * n * (max_c + np.log(sum_e) / T_val)
            w = e / sum_e
            g = np.zeros(n)
            for k in range(len(conv)):
                if w[k] > 1e-8:
                    g += w[k] * grad_conv_at_peak(x, k)
            g *= 2.0 * n
            g -= g.mean()  # Project to tangent space of simplex
            return obj_val, g
        return _obj_grad

    # Try different temperatures
    for T_val in [T, T * 0.5, T * 2.0]:
        if time.monotonic() - start > time_budget:
            break
        try:
            og = make_obj_grad(T_val)
            res = sp_minimize(og, a, method='L-BFGS-B',
                             bounds=[(1e-12, None)] * n,
                             options={'maxiter': 150, 'disp': False})
            a_opt = np.maximum(res.x, 0)
            s_opt = float(np.sum(a_opt))
            if s_opt > 1e-10:
                a_opt = a_opt / s_opt
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_obj:
                    best_obj = obj_opt
                    best_a = a_opt.copy()
                    a = a_opt
        except Exception:
            pass

    return best_a


def nesterov_subgradient(a, time_budget=60.0, K=5):
    """
    Nesterov-accelerated subgradient method on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 100

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        # Adaptive temperature
        spread = (peak_vals[0] - peak_vals[-1]) / max_val if max_val > 0 else 1
        temp = 3.0 / (1.0 + spread * 10.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_peak(y, int(k))

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.3)

        a_prev = a_new.copy()
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        y = y_new.copy()
        t_k = t_new
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    n_conv = 2 * n - 1
    n_new_conv = 2 * new_n - 1
    # Resample the convolution, then "deconvolve" approximately
    # Simpler: just resample a directly
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Exponential decay
        a = np.exp(-0.5 * np.arange(n) / n)
        a = a / a.sum()
    elif mode == 3:
        # Triangular (asymmetric)
        x = np.linspace(0, 1, n)
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.3), 0)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Two bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.25)**2) / 0.02) + np.exp(-((x - 0.75)**2) / 0.02)
        a = a / a.sum()
    elif mode == 5:
        # Sine-based
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(0.5, 3.0) + x ** rng.uniform(0.5, 3.0)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def coordinate_refine(a, time_budget=30.0):
    """
    Coordinate descent: small transfers to reduce peaks.
    More efficient than previous version.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Find top contributing pairs to this peak
        contrib = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib.append((a[i] * a[j], i, j))
        contrib.sort(reverse=True)

        # Find positions NOT contributing to this peak
        non_contrib = []
        for i in range(n):
            j = k_star - i
            if j < 0 or j >= n:
                non_contrib.append(i)

        if not non_contrib:
            break

        improved = False
        for ci, (val, i, j) in enumerate(contrib[:20]):
            if val < 1e-12 or ci >= 20:
                break
            for target in non_contrib[:10]:
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    delta = frac * a[min(i, j)]
                    if delta < 1e-15:
                        continue
                    src = min(i, j) if i == j else (i if a[i] > a[j] else j)
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        if not improved:
            # Try reverse: move mass INTO contributing positions
            # to increase "spread" and potentially reduce other peaks
            break
        it += 1

    return best_a


def multiplicative_weights(a, time_budget=60.0, K=5):
    """
    Multiplicative weights (exponentiated gradient) with adaptive learning rate.
    Different from the previous mirror descent: uses a different update rule
    and momentum.
    """
    n = len(a)
    a = np.maximum(a, 1e-12).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    momentum = 0.3
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        if max_val < 1e-15:
            break

        # Soft max weights over all lags (not just top K)
        # This gives a smoother signal
        T_adapt = max_val * 0.1  # Temperature relative to peak
        w = np.exp((conv - max_val) / T_adapt)
        w /= w.sum()

        # Compute gradient
        g = np.zeros(n)
        # Only use lags with significant weight for efficiency
        active = w > np.percentile(w, 90)
        active_w = w[active] / w[active].sum()
        active_idx = np.where(active)[0]

        for wi, k in zip(active_w, active_idx):
            if wi > 1e-10:
                g += wi * grad_conv_at_peak(a, int(k))

        # Update with momentum
        v = momentum * v + (1 - momentum) * g
        a = a * np.exp(-lr * v)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stall = 0
            lr = max(1e-5, lr * 0.998)
        else:
            stall += 1
            if stall > 50:
                # Reset with perturbation
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                v = np.zeros(n)
                lr = max(1e-5, lr * 0.5)
                stall = 0
            else:
                lr = max(1e-5, lr * 0.98)
        it += 1

    return best_a
