```python
    """
    Solve: min M s.t. conv(a, g)[k] <= M for all k, sum(g) = 1, g >= 0.
    Returns (g_normalized, M_value) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    
    if use_cvxpy and HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in ['ECOS', 'SCS', 'CLARABEL', 'MOSEK']:
            try:
                prob.solve(solver=eval('cp.' + solver), verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    
    # scipy fallback
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None
```

```python
    res = minimize_scalar(f, bounds=(0, 1), method='bounded', options={'xatol': 1e-6})
    return res.x, res.fun
```

```python
"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: quadratic line search via precomputed convolution cross-terms,
dual-informed LP alternating with stagnation detection, log-space smooth optimization,
and focused adaptive multi-n exploration with resampling.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize_scalar
import math

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except Exception:
    HAVE_CVXPY = False


def compute_conv(a, b=None):
    """Full linear convolution a*b (or a*a if b is None)."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(2 * n - 1, j + n)
        length = end - start
        A[start:end, j] = a[:length]
    return A


def solve_lp_minpeak(a):
    """
    Solve: min M s.t. conv(a, g)[k] <= M for all k, sum(g) = 1, g >= 0.
    Returns (g_normalized, M_value) or (None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def solve_lp_maxsum(a, M_target):
    """Solve: max sum(g) s.t. conv(a, g)[k] <= M_target, g >= 0."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x, 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None


def quadratic_line_search(C_aa, C_ag, C_gg, n):
    """
    Find t in [0,1] minimizing 2n * max([(1-t)^2 C_aa + 2t(1-t) C_ag + t^2 C_gg]).
    Returns (best_t, best_obj).
    """
    def f(t):
        combined = (1.0 - t) ** 2 * C_aa + 2.0 * t * (1.0 - t) * C_ag + t ** 2 * C_gg
        return 2.0 * n * float(np.max(combined))

    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-8, 'maxiter': 200})
    return res.x, res.fun


def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_subgradient(a, time_budget=60.0, K=5):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 60

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        spread = (peak_vals[0] - peak_vals[-1]) / (max_val + 1e-15)
        temp = 3.0 / (1.0 + spread * 10.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / (max_val + 1e-15))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_peak(y, int(k))

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.5)

        a_prev = a_new.copy()
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        y = y_new.copy()
        t_k = t_new
        it += 1

    return best_a


def lp_alternating_quad(a, time_budget=300.0, verbose=False):
    """
    LP alternating with quadratic line search.
    Main workhorse optimization.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    stall_threshold = 20
    rng = np.random.default_rng(12345)
    it = 0
    lp_time_total = 0.0
    best_M_lp = None

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        t_lp_start = time.monotonic()
        g_lp, M_lp = solve_lp_minpeak(a)
        lp_time = time.monotonic() - t_lp_start
        lp_time_total += lp_time

        if g_lp is None:
            conv_a = compute_conv(a)
            M_cur = float(np.max(conv_a))
            g_lp = solve_lp_maxsum(a, M_cur)
            if g_lp is None:
                break
            M_lp = M_cur

        # Quadratic line search
        C_aa = compute_conv(a, a)
        C_ag = compute_conv(a, g_lp)
        C_gg = compute_conv(g_lp, g_lp)

        best_t, best_obj_try = quadratic_line_search(C_aa, C_ag, C_gg, n)

        it += 1
        if best_obj_try < best_obj - 1e-14:
            best_obj = best_obj_try
            a_new = (1.0 - best_t) * best_a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new > 1e-10:
                a_new = a_new / s_new
            best_a = a_new
            stall = 0
            if best_M_lp is not None and M_lp > best_M_lp * 1.01:
                pass  # LP got worse but objective improved, fine
            if verbose and it % 10 == 0:
                el = time.monotonic() - start
                print(f"      LP-Quad it{it}: obj={best_obj:.8f}, t={best_t:.4f}, M_lp={M_lp:.6f}, el={el:.1f}s")
        else:
            stall += 1
            if stall >= stall_threshold:
                # Escape: perturbation + different strategy
                noise_scale = min(0.2, 0.03 + 0.02 * (stall // stall_threshold))
                a_pert = best_a * np.exp(rng.normal(0, noise_scale, n))
                a_pert = np.maximum(a_pert, 0)
                s_pert = float(np.sum(a_pert))
                if s_pert > 1e-10:
                    a_pert = a_pert / s_pert
                a = a_pert
                stall = 0
                stall_threshold = int(stall_threshold * 1.3)
                it += 1
            else:
                # Try a smaller step or different g
                a = best_a.copy()

    return best_a


def mirror_descent(a, time_budget=60.0):
    """Mirror descent (multiplicative updates) with adaptive learning rate."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    min_lr = 1e-5
    stall = 0
    momentum = 0.0
    v = np.zeros(n)
    rng = np.random.default_rng(999)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        if max_val < 1e-15:
            break

        # Use softmax weighting over convolution values
        T = max_val / 5.0
        w = np.exp((conv - max_val) / T)
        w /= w.sum()

        # Only use top positions for efficiency
        num_active = max(5, len(conv) // 20)
        active_idx = np.argsort(w)[::-1][:num_active]
        active_w = w[active_idx]
        active_w /= active_w.sum()

        g = np.zeros(n)
        for wi, k in zip(active_w, active_idx):
            if wi > 1e-10:
                g += wi * grad_conv_at_peak(a, int(k))

        v = momentum * v + g
        log_a_new = np.log(a) - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new
            a = a_new
            stall = 0
            momentum = min(0.7, momentum + 0.003)
        else:
            stall += 1
            if stall > 80:
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                v = np.zeros(n)
                momentum = 0.0
                lr = max(min_lr, lr * 0.5)
                stall = 0
            else:
                lr = max(min_lr, lr * 0.98)

        it += 1

    return best_a


def smooth_logsumexp_optimize(a, time_budget=100.0, T=50.0):
    """
    Smooth optimization using log-sum-exp approximation of max.
    Uses log parameterization to handle non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    # Work in log-space: x = log(a)
    x0 = np.log(a)
    x0 -= x0.max()  # center for numerical stability

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_from_x(x):
        """Compute objective from log-parameterization."""
        x_centered = x - x.max()
        a_x = np.exp(x_centered)
        a_x = a_x / a_x.sum()
        conv = compute_conv(a_x)
        return 2.0 * n * float(np.max(conv)), a_x

    def smooth_obj(x):
        """Smooth objective using log-sum-exp."""
        x_centered = x - x.max()
        a_x = np.exp(x_centered)
        s = a_x.sum()
        if s < 1e-10:
            return 1e10
        a_x = a_x / s
        conv = compute_conv(a_x)
        max_c = float(np.max(conv))
        z = T * (conv - max_c)
        sum_e = float(np.exp(z).sum())
        return 2.0 * n * (max_c + np.log(sum_e) / T)

    # Use Nesterov in log-space with the smooth objective
    x = x0.copy()
    x_prev = x0.copy()
    t_k = 1.0
    lr = 0.5 / np.sqrt(n)
    stall = 0
    patience = 50
    y = x0.copy()

    it = 0
    while time.monotonic() - start < time_budget:
        # Evaluate at y (Nesterov lookahead point)
        y_centered = y - y.max()
        a_y = np.exp(y_centered)
        a_y = a_y / a_y.sum()
        conv_y = compute_conv(a_y)
        max_val = float(np.max(conv_y))

        # Subgradient via top-K peaks
        K = 7
        top_idx = np.argsort(conv_y)[::-1][:K]
        peak_vals = conv_y[top_idx]
        spread = (peak_vals[0] - peak_vals[-1]) / (max_val + 1e-15)
        temp_T = 5.0 / (1.0 + spread * 5.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp_T / (max_val + 1e-15))
        weights /= weights.sum()

        # Gradient in a-space
        g_a = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g_a += w * grad_conv_at_peak(a_y, int(k))

        # Convert to x-space gradient: dx = da / a
        g_x = g_a / np.maximum(a_y, 1e-15)
        # Subtract mean to handle the normalization (sum constraint)
        g_x -= g_x.mean()

        lr_t = lr / t_k
        x_new = y - lr_t * g_x

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = x_new + (t_k / t_new) * (x_new - x_prev)
        y_new -= y_new.max()  # re-center

        # Check objective
        x_c = x_new - x_new.max()
        a_new = np.exp(x_c)
        a_new = a_new / a_new.sum()
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new
            stall = 0
        else:
            stall += 1
            if stall > patience:
                x = x0.copy()
                x_prev = x0.copy()
                y = x0.copy()
                t_k = 1.0
                lr = max(0.1, lr * 0.5)
                stall = 0
                patience = int(patience * 1.3)

        x_prev = x_new.copy()
        x_new -= x_new.max()
        x = x_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangle
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Arcsine distribution
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(np.maximum(1 - x**2, 0.01))
        a = a / a.sum()
    elif mode == 5:
        # Multi-frequency
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # U-shaped
        x = np.linspace(0, 1, n)
        a = 2.0 * x**2 + 2.0 * (1 - x)**2 + 0.1
        a = a / a.sum()
    elif mode == 7:
        # Sparse peak
        a = np.ones(n) * 0.1
        peak = n // 2
        width = n // 10
        a[peak - width:peak + width] += np.exp(-((np.arange(2*width+1) - width)**2) / (width/2)**2)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def enforce_symmetry(a):
    """Make sequence symmetric: a[i] = a[n-1-i]."""
    n = len(a)
    half = (n + 1) // 2
    sym = np.zeros(n)
    for i in range(half):
        sym[i] = (a[i] + a[n - 1 - i]) / 2.0
        sym[n - 1 - i] = sym[i]
    s = sym.sum()
    if s > 1e-10:
        sym /= s
    return sym


def lp_alternating_symmetric(a, time_budget=200.0, verbose=False):
    """
    LP alternating exploiting symmetry: a[i] = a[n-1-i].
    Reduces LP variables to ceil(n/2).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    a = enforce_symmetry(a)

    half_n = (n + 1) // 2
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(555)
    it = 0

    # Build half-size convolution matrix
    # g is determined by its first half_n elements (symmetric)
    # A_half[k, j] = sum of contributions from g[j] and g[n-1-j]
    A_full = build_conv_matrix(a)
    # For symmetric g: g[j] = g[n-1-j]
    # conv(a, g)[k] = sum_j A_full[k,j] * g[j]
    # With symmetry: = sum_{j < n/2} A_full[k,j] * g[j] + sum_{j >= n/2} A_full[k,j] * g[n-1-j]
    # = sum_{j < n/2} (A_full[k,j] + A_full[k, n-1-j]) * g[j] + (middle term if n odd)

    A_half = np.zeros((2 * n - 1, half_n))
    for j in range(half_n):
        A_half[:, j] = A_full[:, j]
        if j != n - 1 - j:
            A_half[:, j] += A_full[:, n - 1 - j]

    def expand_g(g_half):
        """Expand half g to full g using symmetry."""
        g_full = np.zeros(n)
        g_full[:half_n] = g_half
        g_full[half_n:] = g_half[:n - half_n][::-1]
        return g_full

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        # Solve half-size LP
        c = np.zeros(half_n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, half_n + 1))
        A_ub[:, :half_n] = A_half
        A_ub[:, half_n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, half_n + 1))
        A_eq[0, :half_n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * half_n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if res.status != 0:
            break

        g_half = np.maximum(res.x[:half_n], 0)
        s_g = float(np.sum(g_half))
        if s_g < 1e-10:
            break
        g_half = g_half / s_g
        g_full = expand_g(g_half)
        s_full = float(np.sum(g_full))
        if s_full > 1e-10:
            g_full = g_full / s_full

        # Quadratic line search
        C_aa = compute_conv(a, a)
        C_ag = compute_conv(a, g_full)
        C_gg = compute_conv(g_full, g_full)
        best_t, best_obj_try = quadratic_line_search(C_aa, C_ag, C_gg, n)

        it += 1
        if best_obj_try < best_obj - 1e-14:
            best_obj = best_obj_try
            a_new = (1.0 - best_t) * best_a + best_t * g_full
            a_new = np.maximum(a_new, 0)
            a_new = enforce_symmetry(a_new)
            best_a = a_new
            stall = 0
            if verbose and it % 5 == 0:
                print(f"      LP-Sym it{it}: obj={best_obj:.8f}, t={best_t:.4f}")
        else:
            stall += 1
            if stall > 25:
                a = best_a * np.exp(rng.normal(0, 0.05, half_n))
                a_half = np.maximum(a[:half_n], 0)
                a_full = expand_g(a_half)
                s_a = float(np.sum(a_full))
                if s_a > 1e-10:
                    a_full = a_full / s_a
                a = a_full
                stall = 0

    return best_a


def coordinate_refine(a, time_budget=30.0):
    """
    Coordinate descent: move small amounts of mass to reduce the dominant peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Find positions contributing to this peak
        contrib = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib.append((a[i] * a[j], i, j))
        contrib.sort(reverse=True)

        # Find positions NOT contributing to this peak
        non_contrib = []
        for i in range(n):
            j = k_star - i
            if j < 0 or j >= n:
                non_contrib.append(i)

        if not non_contrib:
            break

        improved = False
        for ci, (val, i, j) in enumerate(contrib[:10]):
            if val < 1e-12 or ci >= 10:
                break
            for target in non_contrib[:6]:
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[min(i, j)]
                    if delta < 1e-15:
                        continue
                    src = min(i, j) if i == j else (i if a[i] > a[j] else j)
                    a[src] -= delta
                    a[target] += delta
                    a[src] = max(a[src], 0)
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid optimization combining LP alternating with
    quadratic line search, Nesterov subgradient, mirror descent, log-space
    smooth optimization, symmetric LP, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_target = 800
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Quick Nesterov (~80s) ===
    print(f"\n=== Phase 1: Nesterov Subgradient (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 15:
        a_ns = nesterov_subgradient(best_seq, time_budget=p1_budget, K=5)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_ns:.8f} (no improvement)")

    # === Phase 2: Main LP Alternating with Quadratic Line Search (~650s) ===
    print(f"\n=== Phase 2: LP Alternating + Quad Line Search (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p2_total = min(remaining * 0.45, 650.0)

    # Multiple rounds with increasing perturbations
    n_rounds = 3
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(p2_total / (n_rounds - rnd), remaining * 0.55)
        if alloc < 30:
            break

        if rnd == 0:
            a_start = best_seq.copy()
        elif rnd == 1:
            a_start = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        else:
            a_start = best_seq * np.exp(rng.normal(0, 0.04, len(best_seq)))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        a_lp = lp_alternating_quad(a_start, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Mirror Descent (~80s) ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.07, 80.0)
    if p3_budget > 15:
        a_md = mirror_descent(best_seq, time_budget=p3_budget)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Mirror: obj={best_score:.8f} **")
        else:
            print(f"  Mirror: obj={obj_md:.8f} (no improvement)")

    # === Phase 4: Log-space smooth optimization (~120s) ===
    print(f"\n=== Phase 4: Log-Space Smooth Optimization ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.09, 120.0)
    if p4_budget > 20:
        for T_val in [30.0, 80.0]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            t_alloc = min(p4_budget / 2.0, remaining * 0.45)
            if t_alloc < 15:
                continue
            a_sm = smooth_logsumexp_optimize(best_seq, time_budget=t_alloc, T=T_val)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  LogSmooth T={T_val}: obj={best_score:.8f} **")

    # === Phase 5: Symmetric LP (~100s) ===
    print(f"\n=== Phase 5: Symmetric LP Alternating ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.08, 100.0)
    if p5_budget > 20:
        a_sym = enforce_symmetry(best_seq)
        a_sym_res = lp_alternating_symmetric(a_sym, time_budget=p5_budget, verbose=True)
        obj_sym = compute_obj(a_sym_res)
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym_res.copy()
            print(f"  Sym LP: obj={best_score:.8f} **")
        else:
            print(f"  Sym LP: obj={obj_sym:.8f} (no improvement)")

    # === Phase 6: Multi-n Exploration (~250s) ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Focus on n values near the current best, plus a few strategic ones
    n_options = sorted(set([
        max(200, n_best - 100),
        max(200, n_best - 50),
        n_best + 50,
        n_best + 100,
        n_best + 150,
        n_best + 200,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500]
    n_options.sort(key=lambda x: abs(x - n_best))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= 5:
            break
        if n_try == n_best:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        # Quick Nesterov
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_try = nesterov_subgradient(a_try, time_budget=min(remaining * 0.1, 30.0), K=3)

        # Quick LP
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_try = lp_alternating_quad(a_try, time_budget=min(remaining * 0.15, 45.0))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = len(best_seq)
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse restarts (~100s) ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r + 2) % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.2, 50.0)
        a_opt = lp_alternating_quad(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final Intensive Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 40:
        lp_alloc = min(remaining * 0.3, 80.0)
        a_lp = lp_alternating_quad(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Nesterov polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        ns_alloc = min(remaining * 0.2, 40.0)
        a_ns = nesterov_subgradient(best_seq, time_budget=ns_alloc, K=7)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Final NS: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd_alloc = min(remaining * 0.2, 25.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 25:
        a_last = lp_alternating_quad(best_seq, time_budget=min(remaining * 0.4, 30.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
