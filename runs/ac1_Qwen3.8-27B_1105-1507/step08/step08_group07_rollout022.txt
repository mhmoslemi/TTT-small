    return irfft(fa * fa, size)[:2 * n - 1]

def build_conv_matrix(a):
    n = len(a)

        A[j:j + r, j] = a[:r]
    return A

def symmetric_optimize(a, time_budget=60.0):
    """Optimize in the symmetric subspace: a_j = a_{n-1-j}."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    # Extract half
    half_n = (n + 1) // 2
    a_sym = (a + a[::-1]) / 2.0
    a_sym = a_sym / a_sym.sum()
    
    def make_symmetric(half):
        if n % 2 == 0:
            full = np.concatenate([half, half[::-1]])
        else:
            full = np.concatenate([half, half[-2::-1]])
        full = np.maximum(full, 0)
        s = full.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return full / s
    
    half_init = make_symmetric(a_sym)[:half_n].copy()
    # Actually, extract the first half
    half_init = a_sym[:half_n].copy() / (a_sym[:half_n].sum() + 1e-10)
    
    def obj_half(h):
        full = make_symmetric(h)
        conv = compute_conv(full)
        return float(np.max(conv))
    
    from scipy.optimize import minimize
    bounds = [(0, None)] * half_n
    constraints = [{'type': 'eq', 'fun': lambda h: h.sum() - 1.0}]
    
    try:
        res = minimize(obj_half, half_init, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 500, 'ftol': 1e-12})
        if res.fun < obj_half(half_init):
            full = make_symmetric(res.x)
            return full
    except:
        pass
    
    return a_sym / a_sym.sum()

def coordinate_descent_polish(a, time_budget=30.0):
    """Coordinate descent: try adjusting each coordinate."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    improved = True
    while improved and time.monotonic() - start < time_budget:
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - start > time_budget:
                break
            for delta_frac in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]:
                delta = delta_frac * a[i]
                if a[i] + delta < 0:
                    continue
                # We need to maintain sum(a) = 1, so distribute delta to others
                # Simple: just scale
                a_try = a.copy()
                a_try[i] += delta
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    a = a_try.copy()
                    improved = True
                    break
    
    return a, best_obj

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def solve_lp_cutting_plane(a, max_iter=50):
    """Solve LP using cutting plane (constraint generation)."""
    n = len(a)
    A = build_conv_matrix(a)
    
    # Start with just the sum constraint and a few peak constraints
    active = list(range(2*n-1))  # Start with all, then reduce
    
    # Actually, let's start with just a few
    conv0 = compute_conv(a)  # a is the "other" side, this gives a*a
    # The peaks of a*a tell us which constraints are likely active
    top_k = np.argsort(conv0)[-min(20, 2*n-1)][:]  # top 20 peaks
    active = list(top_k)
    
    for _ in range(max_iter):
        A_active = A[active, :]
        # Solve: min M s.t. A_active @ z <= M, 1^T z = 1, z >= 0
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([A_active, -np.ones((len(active), 1))])
        b_ub = np.zeros(len(active))
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            break
        
        z = np.maximum(res.x[:n], 0)
        z = z / z.sum()
        M = res.x[-1]
        
        # Check all constraints
        Ax = A @ z
        violations = np.where(Ax > M + 1e-10)[0]
        
        if len(violations) == 0:
            return z, M
        
        # Add violated constraints (top 10)
        new_active = sorted(set(active) | set(violations[:10].tolist()))
        active = new_active
    
    return None, None

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ z gives cross-conv a*z."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_full(a):
    """Solve full LP: min M s.t. A@z <= M, sum(z)=1, z>=0 using HiGHS."""
    n = len(a)
    A = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        z_val = np.maximum(res.x[:n], 0)
        zs = float(np.sum(z_val))
        if zs > 1e-10:
            z_val = z_val / zs
            return z_val, float(res.x[n])
    return None, None


def solve_lp_cutting_plane(a, max_iters=30, init_peaks=15):
    """Solve LP using cutting plane method for speed."""
    n = len(a)
    A = build_conv_matrix(a)
    num_constraints = 2 * n - 1

    # Initialize with top peak constraints
    conv_a = compute_conv(a)
    top_idx = np.argsort(conv_a)[-min(init_peaks, num_constraints):]
    active = set(top_idx.tolist())

    z_prev = None
    M_prev = np.inf

    for _ in range(max_iters):
        active_list = sorted(active)
        A_active = A[active_list, :]

        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([A_active, -np.ones((len(active_list), 1))])
        b_ub = np.zeros(len(active_list))
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            break

        z_val = np.maximum(res.x[:n], 0)
        zs = float(np.sum(z_val))
        if zs < 1e-10:
            break
        z_val = z_val / zs
        M_val = float(res.x[n])

        # Check all constraints
        Ax = A @ z_val
        violations = np.where(Ax > M_val + 1e-12)[0]

        if len(violations) == 0:
            return z_val, M_val

        # Add violated constraints (up to 10 new ones)
        for v in violations[:10]:
            active.add(int(v))

        # If no progress, fall back to full LP
        if len(active) > num_constraints:
            return solve_lp_full(a)

    # Fallback to full LP
    return solve_lp_full(a)


def lp_alternating(a, time_budget=200.0, verbose=False):
    """LP alternating with adaptive mixing and 2-cycle breaking."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    prev_a = a.copy()
    prev_prev_a = None
    cycle_count = 0

    while time.monotonic() - start < time_budget:
        # Use cutting plane for speed, fall back to full LP
        z_lp, M_lp = solve_lp_cutting_plane(a)
        if z_lp is None:
            z_lp, M_lp = solve_lp_full(a)
            if z_lp is None:
                break

        # 2-cycle detection
        if prev_prev_a is not None:
            diff = np.max(np.abs(a - prev_prev_a))
            if diff < 5e-7:
                # 2-cycle detected: try averaging
                a_avg = 0.5 * (a + prev_a)
                a_avg = np.maximum(a_avg, 0)
                sa = float(np.sum(a_avg))
                if sa > 1e-10:
                    a_avg = a_avg / sa
                    obj_avg = compute_obj(a_avg)
                    if obj_avg < best_obj - 1e-12:
                        best_obj = obj_avg
                        best_a = a_avg.copy()
                    a = a_avg.copy()
                prev_a = None
                prev_prev_a = None
                cycle_count += 1
                if cycle_count > 5:
                    # Force escape
                    rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    s2 = float(np.sum(a))
                    if s2 > 1e-10:
                        a = a / s2
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    cycle_count = 0
                continue

        # Try multiple mixing ratios
        best_trial_obj = np.inf
        best_trial = None
        for mix in [0.4, 0.6, 0.8, 1.0]:
            a_try = (1.0 - mix) * a + mix * z_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_trial_obj:
                best_trial_obj = obj_try
                best_trial = a_try.copy()

        if best_trial is not None and best_trial_obj < best_obj - 1e-13:
            best_obj = best_trial_obj
            best_a = best_trial
            prev_prev_a = prev_a
            prev_a = a.copy()
            a = best_a.copy()
            stall = 0
            if verbose and int((time.monotonic() - start) * 2) % 30 == 0:
                print(f"    LP: obj={best_obj:.10f}, t={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 8:
                # Escape with dual-guided perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                # Perturb proportional to the convolution peaks
                conv = compute_conv(a)
                # Coordinates that contribute to peaks get downward pressure
                A = build_conv_matrix(a)
                top_peaks = np.argsort(conv)[-5:]
                col_sum = np.sum(A[top_peaks, :], axis=0)
                col_sum = col_sum / (np.max(col_sum) + 1e-10)
                pert = -0.03 * col_sum + rng.normal(0, 0.02, n)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s2 = float(np.sum(a))
                if s2 > 1e-10:
                    a = a / s2
                else:
                    a = best_a.copy()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                prev_a = None
                prev_prev_a = None
                if verbose:
                    print(f"    [Escaped] obj={best_obj:.10f}")

    return best_a


def scipy_direct_optimize(a, time_budget=80.0, verbose=False):
    """Direct optimization using scipy SLSQP on the full problem."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj_val = compute_obj(a)
    start = time.monotonic()

    def objective(x):
        x_nonneg = np.maximum(x, 0)
        s = x_nonneg.sum()
        if s < 1e-10:
            return 1e6
        x_norm = x_nonneg / s
        conv = compute_conv(x_norm)
        return float(np.max(conv))

    def constraint_sum(x):
        return x.sum() - 1.0

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, None)] * n

    # Try multiple starting points
    starts = [a.copy()]
    rng = np.random.default_rng(777)
    for _ in range(3):
        a_pert = a * np.exp(rng.normal(0, 0.05, n))
        a_pert = np.maximum(a_pert, 0)
        sp = float(np.sum(a_pert))
        if sp > 1e-10:
            starts.append(a_pert / sp)

    for start_point in starts:
        if time.monotonic() - start > time_budget:
            break
        try:
            res = minimize(objective, start_point, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10})
            if res.fun < best_obj_val:
                a_new = np.maximum(res.x, 0)
                s = a_new.sum()
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj_val:
                        best_obj_val = obj_new
                        best_a = a_new.copy()
                        if verbose:
                            print(f"    scipy L-BFGS: obj={best_obj_val:.10f}")
        except Exception as e:
            pass

    # Follow up with SLSQP on the best found
    if time.monotonic() - start < time_budget - 10:
        try:
            res = minimize(objective, best_a, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 300, 'ftol': 1e-14})
            if res.fun < best_obj_val:
                a_new = np.maximum(res.x, 0)
                s = a_new.sum()
                if s > 1e-10:
                    a_new = a_new / s
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj_val:
                        best_obj_val = obj_new
                        best_a = a_new.copy()
                        if verbose:
                            print(f"    scipy SLSQP: obj={best_obj_val:.10f}")
        except:
            pass

    return best_a


def coordinate_polish(a, time_budget=30.0):
    """Coordinate descent polishing with very small steps."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)

    for outer in range(100):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-10:
                continue
            for delta in [0.005 * a[i], -0.005 * a[i], 0.002 * a[i], -0.002 * a[i]]:
                a_try = a.copy()
                a_try[i] += delta
                if a_try[i] < 0:
                    continue
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    a = a_try.copy()
                    improved = True
                    break
        if not improved and outer > 20:
            break

    return a, best_obj


def pairwise_polish(a, time_budget=30.0):
    """Pairwise exchange polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(555)

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for _ in range(10):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for frac in [0.02, 0.05, 0.01]:
                delta = frac * min(a[i], a[j])
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < -1e-15 or a_try[j] < -1e-15:
                    continue
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    a = a_try.copy()
                    improved = True
                    break
        if not improved and it > 50:
            break

    return a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining LP alternating (cutting plane), scipy direct optimization,
    aggressive n-sweep, and multi-stage polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.10f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        # Try structured initializations
        for n_init in [600, 800, 1000, 1200]:
            t = np.linspace(0, 1, n_init)
            windows = [
                0.5 + 0.5 * np.cos(2 * np.pi * t),
                np.sin(np.pi * t) ** 2,
                np.exp(-2 * ((t - 0.5) * 3) ** 2),
            ]
            for w in windows:
                a_init = np.maximum(w, 0.001)
                a_init = a_init / np.sum(a_init)
                obj_init = compute_obj(a_init)
                if obj_init < best_score:
                    best_score = obj_init
                    best_seq = a_init.copy()

        if best_seq is None:
            best_seq = np.ones(800) / 800
            best_score = compute_obj(best_seq)
        print(f"Initial: obj={best_score:.10f}")

    a = best_seq.copy()
    n_parent = len(a)

    # === Phase 1: LP Alternating from parent (250s) ===
    print(f"\n=== Phase 1: LP Alt (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(250.0, remaining * 0.18)
    a1 = lp_alternating(a, time_budget=alloc, verbose=True)
    obj1 = compute_obj(a1)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
        a = a1.copy()
        print(f"  Phase 1 done: obj={best_score:.10f} **")
    else:
        print(f"  Phase 1 done: obj={obj1:.10f}")

    # === Phase 2: scipy direct optimization (100s) ===
    print(f"\n=== Phase 2: scipy direct ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(100.0, remaining * 0.08)
        a2 = scipy_direct_optimize(best_seq, time_budget=alloc, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            a = a2.copy()
            print(f"  Phase 2 done: obj={best_score:.10f} **")
        else:
            print(f"  Phase 2 done: obj={obj2:.10f}")

    # === Phase 3: Aggressive n-sweep (400s) ===
    print(f"\n=== Phase 3: n-sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Generate n candidates: focus near n_best but also explore widely
    n_near = list(range(max(300, n_best - 100), min(2500, n_best + 101), 10))
    n_far = list(range(400, 2001, 100))
    n_candidates = sorted(set(n_near + n_far))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2500]
    # Sort by proximity to current best
    n_candidates.sort(key=lambda x: abs(x - n_best))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        if n_try == n_best and n_try != n_parent:
            continue

        a_try = resample_sequence(best_seq, n_try)
        # Give more time to n values close to current best
        dist = abs(n_try - n_best)
        alloc = min(35.0, remaining * 0.03)
        if dist < 30:
            alloc = min(50.0, remaining * 0.04)

        a_opt = lp_alternating(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.10f} **")
        else:
            if dist % 200 == 0 or dist < 20:
                print(f"  n={n_try}: obj={obj_try:.10f}")

    # === Phase 4: LP on current best (150s) ===
    print(f"\n=== Phase 4: LP on best (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(150.0, remaining * 0.12)
        a4 = lp_alternating(best_seq, time_budget=alloc, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            a = a4.copy()
            print(f"  Phase 4 done: obj={best_score:.10f} **")

    # === Phase 5: scipy on best (60s) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 5: scipy on best ===")
        a5 = scipy_direct_optimize(best_seq, time_budget=min(60.0, remaining * 0.05), verbose=True)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
            print(f"  Phase 5: obj={best_score:.10f} **")

    # === Phase 6: Coordinate + pairwise polish (40s) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 6: Polish ===")
        alloc1 = min(20.0, remaining * 0.02)
        a6, obj6 = coordinate_polish(best_seq, time_budget=alloc1)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Coord polish: obj={best_score:.10f} **")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc2 = min(20.0, remaining * 0.015)
            a7, obj7 = pairwise_polish(best_seq, time_budget=alloc2)
            if obj7 < best_score:
                best_score = obj7
                best_seq = a7.copy()
                print(f"  Pairwise: obj={best_score:.10f} **")

    # === Phase 7: Random restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 7: Restarts ({remaining:.0f}s) ===")
        n_restarts = min(10, int(remaining // 35))
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            n_best_cur = len(best_seq)
            # Choose perturbation type
            if r % 5 == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.03 + 0.005 * r, n_best_cur))
            elif r % 5 == 1:
                # Block perturbation
                a_r = best_seq.copy()
                n_blocks = max(5, n_best_cur // 60)
                bs = max(1, n_best_cur // n_blocks)
                blocks = rng.choice(n_blocks, min(4, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * bs
                    ei = min(si + bs, n_best_cur)
                    a_r[si:ei] *= rng.uniform(0.5, 1.5)
            elif r % 5 == 2:
                # Frequency perturbation
                fa = rfft(a_r) if a_r is not None else rfft(best_seq)
                a_r = best_seq.copy()
                fa = rfft(a_r)
                if len(fa) > 2:
                    fa[1:-1] *= (1.0 + rng.normal(0, 0.03, len(fa[1:-1])))
                a_r = irfft(fa, n_best_cur)
            elif r % 5 == 3:
                # Sparse large perturbation
                a_r = best_seq.copy()
                n_sp = max(5, n_best_cur // 15)
                idx = rng.choice(n_best_cur, n_sp, replace=False)
                a_r[idx] *= rng.uniform(0.2, 3.0, n_sp)
            else:
                # Try different n
                n_alt = n_best_cur + rng.integers(-50, 51)
                if 300 <= n_alt <= 2500:
                    a_r = resample_sequence(best_seq, n_alt)
                else:
                    a_r = best_seq * np.exp(rng.normal(0, 0.05, n_best_cur))

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            alloc = min(remaining * 0.25, 30.0)
            a_opt = lp_alternating(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.10f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.10f}")

    # === Final: Quick LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 15.0)
        a_final = lp_alternating(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.10f} **")

    # Final coordinate polish
    remaining = deadline - time.monotonic()
    if remaining > 5:
        a_fp, obj_fp = coordinate_polish(best_seq, time_budget=min(remaining * 0.3, 8.0))
        if obj_fp < best_score:
            best_score = obj_fp
            best_seq = a_fp.copy()

    # Ensure valid output
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
