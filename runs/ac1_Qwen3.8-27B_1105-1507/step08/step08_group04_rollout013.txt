    c = compute_conv(a)  # length 2n-1

    tau_c = tau * c
    tau_c -= np.max(tau_c)  # for numerical stability

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategy:
1. LP alternating (warm-started) as primary optimizer with adaptive blending
2. Log-sum-exp smoothed projected gradient with Nesterov acceleration as secondary polisher
3. Multi-n exploration with diverse structured initializations
4. Gradient-informed perturbation to escape local optima
5. Adaptive temperature scheduling for LSE smoothing
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time


def compute_conv(a, b=None):
    """FFT-based full linear convolution. If b is None, convolves a with itself."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2 (scale invariant, use sum=1)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp(a, max_time=30.0):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    # Try cvxpy solvers
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > max_time:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    # Fallback to scipy linprog
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': max_time})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def project_simplex(z):
    """Project z onto the standard simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * (np.arange(1, n + 1) - 1) >= cssv - 1)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(z - theta, 0)


def lse_gradient(a, tau):
    """Compute gradient of LSE(tau * conv(a,a)) w.r.t. a, using FFT.
    Returns (gradient, softmax_weights, conv_values)."""
    n = len(a)
    c = compute_conv(a)
    # Compute softmax weights (numerically stable)
    tc = tau * c
    tc -= np.max(tc)
    w = np.exp(tc)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # Compute gradient via FFT correlations
    # grad[j] = tau * (r1[j] + r2[j])
    # r1[j] = sum_m w[j+m]*a[m] = (a * w_rev)[2n-2-j]
    # r2[j] = sum_m a[m]*w[j-m] = (a * w)[j]

    size = 1
    while size < 3 * n:
        size <<= 1

    fa = rfft(a, size)
    fw = rfft(w, size)
    conv_aw = irfft(fa * fw, size)[:3 * n - 2]

    w_rev = w[::-1]
    fwr = rfft(w_rev, size)
    conv_awr = irfft(fa * fwr, size)[:3 * n - 2]

    # r2[j] = conv_aw[j] for j in [0, n-1]
    r2 = conv_aw[:n]
    # r1[j] = conv_awr[2n-2-j] for j in [0, n-1]
    r1 = conv_awr[n - 1:2 * n - 1][::-1]

    grad = tau * (r1 + r2)
    return grad, w, c

    t = 1.0
    prev_a = a.copy()
    x = a.copy()  # Nesterov virtual iterate
    lr = 1.0 / np.sqrt(n)
    momentum = 0.7
    stall_count = 0
    best_stall_obj = best_obj

    for it in range(100000):

        alpha = (t - 1.0) / t_next

        y = project_simplex(y)

        grad, w, c = lse_gradient(y, tau)

        x = a

            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_t = t

        if best_local_obj < best_obj - 1e-13:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.2f}, time={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 5:
                # Gradient-informed perturbation
                try:
                    grad, w, c = lse_gradient(best_a, 10.0 * n)
                    # Perturb in the direction that reduces the gradient variance
                    # (i.e., flatten the peaks)
                    grad_centered = grad - np.mean(grad)
                    # Move opposite to gradient on the simplex
                    scale = 0.05 + 0.03 * min(stall // 5, 5)
                    a = best_a - scale * grad_centered / (np.max(np.abs(grad_centered)) + 1e-10)
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                        # Also add random component
                        a = a * (1.0 + 0.05 * rng.standard_normal(n))
                        a = np.maximum(a, 0)
                        s = float(np.sum(a))
                        if s > 1e-10:
                            a = a / s
                            obj_p = compute_obj(a)
                            if obj_p < best_obj:
                                best_obj = obj_p
                                best_a = a.copy()
                except Exception:
                    a = best_a * np.exp(0.1 * rng.standard_normal(n))
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s

                stall = 0

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 7)
    x = np.linspace(0, 1, n)

    if choice == 0:
        # Smooth bump
        a = np.exp(-30 * (x - 0.5) ** 2)
    elif choice == 1:
        # Oscillatory
        freq = rng.uniform(2, 8)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
    elif choice == 2:
        # Multiple bumps
        centers = rng.uniform(0, 1, size=3)
        a = np.ones(n) * 0.1
        for c in centers:
            width = rng.uniform(0.05, 0.2)
            a += np.exp(-((x - c) / width) ** 2)
    elif choice == 3:
        # Power law
        p = rng.uniform(0.3, 3.0)
        a = np.maximum(x, 0.01) ** p
    elif choice == 4:
        # Triangular wave
        freq = rng.integers(2, 6)
        a = 1.0 - np.abs(2 * (x * freq - np.floor(x * freq + 0.5)))
    elif choice == 5:
        # Gaussian mixture
        a = np.ones(n) * 0.05
        for _ in range(rng.integers(2, 5)):
            mu = rng.uniform(0, 1)
            sigma = rng.uniform(0.05, 0.3)
            a += np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    else:
        # Random smooth (low-freq noise)
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        a = np.ones(n) * 0.5
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating, LSE smoothed gradient, multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        print(f"Parent loaded: n={len(a)}, obj={compute_obj(a):.8f}")
    except Exception:
        a = np.ones(800) / 800
        print(f"No parent, using uniform: n=800, obj={compute_obj(a):.8f}")

    best_a = a.copy()
    best_obj = compute_obj(a)
    n_base = len(best_a)

    # === Phase 1: Primary LP alternating ===
    print(f"\n=== Phase 1: LP Alternating (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 300.0)
    a_p1 = lp_alternate(best_a, time_budget=p1_budget, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  Phase 1 result: obj={best_obj:.8f} **")
    else:
        print(f"  Phase 1: no improvement (obj={obj_p1:.8f})")

    # === Phase 2: LSE Smoothed Gradient ===
    print(f"\n=== Phase 2: LSE Smoothed Gradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = min(remaining * 0.25, 200.0)
        # Try multiple tau values
        tau_options = [3.0 * n_base, 10.0 * n_base, 30.0 * n_base]
        for tau in tau_options:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p2_budget / len(tau_options), remaining * 0.4)
            a_lse = lse_optimize(best_a, time_budget=alloc, init_tau=tau, verbose=False)
            obj_lse = compute_obj(a_lse)
            if obj_lse < best_obj:
                best_obj = obj_lse
                best_a = a_lse
                print(f"  LSE tau={tau:.0f}: obj={best_obj:.8f} **")
            else:
                print(f"  LSE tau={tau:.0f}: obj={obj_lse:.8f}")

    # === Phase 3: LP Alternating Round 2 ===
    print(f"\n=== Phase 3: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = min(remaining * 0.25, 250.0)
        a_p3 = lp_alternate(best_a, time_budget=p3_budget, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            print(f"  Phase 3 result: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 3: no improvement")

    # === Phase 4: LSE again from improved point ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 4: LSE Polish Round 2 ===")
        p4_budget = min(remaining * 0.15, 100.0)
        a_lse2 = lse_optimize(best_a, time_budget=p4_budget, init_tau=20.0 * len(best_a), verbose=False)
        obj_lse2 = compute_obj(a_lse2)
        if obj_lse2 < best_obj:
            best_obj = obj_lse2
            best_a = a_lse2
            print(f"  Phase 4: obj={best_obj:.8f} **")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        max(200, n_best - 400), max(200, n_best - 200),
        n_best,
        n_best + 200, n_best + 400, n_best + 800,
    ]))
    n_options = [x for x in n_options if 200 <= x <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            a_try = a_try / np.sum(a_try)

        # LP optimize
        alloc = min(40.0, remaining * 0.2)
        a_opt = lp_alternate(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        # Quick LSE polish
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30 and obj_try < best_obj + 0.01:
            alloc2 = min(20.0, remaining2 * 0.2)
            a_lse = lse_optimize(a_opt, time_budget=alloc2, init_tau=5.0 * n_try)
            obj_lse = compute_obj(a_lse)
            if obj_lse < obj_try:
                obj_try = obj_lse
                a_opt = a_lse

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse initializations ===
    print(f"\n=== Phase 6: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    n_div = len(best_a)

    for d in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        a_div = diverse_initialization(n_div, rng)
        alloc = min(35.0, remaining * 0.15)
        a_opt = lp_alternate(a_div, time_budget=alloc)
        obj_div = compute_obj(a_opt)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse {d}: obj={obj_div:.8f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # LP polish
        alloc = min(remaining * 0.35, 60.0)
        a_final = lp_alternate(best_a, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # LSE micro-polish
        alloc = min(remaining * 0.3, 40.0)
        a_sg = lse_optimize(best_a, time_budget=alloc, init_tau=10.0 * len(best_a))
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  Final LSE: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # One more LP
        alloc = min(remaining * 0.4, 30.0)
        a_last = lp_alternate(best_a, time_budget=alloc)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
