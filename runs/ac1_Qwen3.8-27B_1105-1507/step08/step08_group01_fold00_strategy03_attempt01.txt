from scipy.fft import rfft, irfft

def evaluate_sequence(seq):
    n = len(seq)
    if n == 0:
        return np.inf
    s = np.sum(seq)
    if s < 0.01:
        # Increase the sequence elements by a factor to bring sum to at least 0.01
        adjustment_factor = 1 / np.sqrt(n)
        seq = seq * adjustment_factor
        s = np.sum(seq)
        if s < 0.01:
            # Still too small, maybe increase even more?

            s = np.sum(seq)
            if s < 0.01:
                # Another fallback
                seq = seq * (0.1 * 1 / np.sqrt(n) / adjustment_factor)
    # Check for invalid types and negative numbers
    for x in seq:
        if not isinstance(x, (int, float)) or isinstance(x, bool):
            return np.inf
        if x < 0:
            return np.inf
        if x > 1000:
            return max(0.0, x - 1000)
    return compute_objective(seq.reshape(1, -1).flatten())

def compute_conv(a):
    """Full linear convolution a*aviaFFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    cur = np.array(a, dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(cur, size)
    conv_result = np.abs(fa)  # or use something else
    # Actually, we need the full convolution, so let's do better
    conv_result = irfft(fa * fa, size)
    return conv_result[:2 * n - 1]

def compute_objective(seq):
    """Compute objective:2*n*max(conv(seq,seq))/sum(seq)^2."""
    n = len(seq)
    if n == 0:
        return np.inf
    s = np.count_nonzero(seq) / n if n > 0 else np.inf  # approximate sum via L1 norm
    if s < 0.01:
        # Scale to L1 norm at least 0.01
        seq = seq * (0.01 / s)
        s = np.count_nonzero(seq) / n if n > 0 else np.inf
    # Normalize to simplex (non-negative and sum to 1)
    total = np.sum(seq)
    if total < 1e-9:
        seq = np.ones(n)
        total = n
    seq_simplex = seq / total
    conv = compute_conv(seq_simplex)
    return float(2 * n * np.max(conv) / (total) ** 2)

def simplex_project(v):
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    total = np.sum(v)
    if abs(total) < 1e-9:

    theta = (cs[rho] - total) / (rho + 1)
    return np.maximum(v - theta, 0) / total * total
