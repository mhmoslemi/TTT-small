```python
# Precompute FFT size
def next_power_of_2(n):
    return 1 << (n - 1).bit_length()

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = next_power_of_2(2 * n - 1)
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > cssv - 1.0
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(a):
    """Build the (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A
```

```python
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_val = g_val / s
                    # Get dual values
                    dual = np.zeros(m)
                    try:
                        dv = con1.dual_value
                        if dv is not None:
                            dual = np.abs(np.asarray(dv)).flatten()
                    except:
                        pass
                    return g_val, float(M.value), dual
        except Exception:
            continue
    
    return None, None, None
```

Now, the dual values give us the weights z_k for each lag constraint. We can use these to compute a weighted subgradient.

The subgradient of f(a) = max_k (a*a)[k] at a point where the active set is S = {k : (a*a)[k] ≈ max} is:
s = (1/|S|) * sum_{k in S} grad_k(a)

Where grad_k(a)[i] = a[k-i] (for valid k-i).

With the LP dual weights, we use:
s = sum_k z_k * grad_k(a) / sum_k z_k

This is a "soft" subgradient that uses all lags (weighted by dual values) instead of just the active set.
