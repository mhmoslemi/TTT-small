"""
LP-alternating optimizer with diverse restarts and smoothed gradient descent.
Focuses on high-throughput LP iterations, geometric-mean updates, and 
systematic local-minimum escaping via structured perturbations.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_cross(a, b):
    """Compute conv(a, b) using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    # A[k, j] = a[k-j] if 0 <= k-j < n
    for j in range(n):
        # For row k, col j: A[k,j] = a[k-j] if 0 <= k-j < n
        # k ranges from j to j+n-1 (clamped to [0, 2n-2])
        k_start = j
        k_end = min(2 * n - 1, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a, solver=None):
    """
    LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS, cp.SCS]
    if solver is not None:
        solvers = [solver] + [s for s in solvers if s != solver]
    
    for sol in solvers:
        try:
            prob.solve(solver=sol)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate_fast(a0, time_budget=60.0, t_fixed=0.3, verbose=False):
    """
    Fast LP alternating with fixed step size (no line search for speed).
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # Fixed step update
        a_try = (1.0 - t_fixed) * a + t_fixed * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall = 0
            if verbose and stall == 0:
                elapsed = time.monotonic() - start
                print(f"      LP: obj={best_obj:.8f}, t={t_fixed}, elapsed={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 50:
                # Try geometric mean update
                with np.errstate(divide='ignore', invalid='ignore'):
                    geo = np.sqrt(np.maximum(a, 0) * np.maximum(g_lp, 0))
                gs = float(np.sum(geo))
                if gs > 1e-10:
                    geo = geo / gs
                    obj_geo = compute_obj(geo)
                    if obj_geo < best_obj - 1e-13:
                        best_obj = obj_geo
                        best_a = geo.copy()
                        a = best_a
                        stall = 0
                        if verbose:
                            print(f"      LP(geo): obj={best_obj:.8f}, elapsed={time.monotonic()-start:.1f}s")
                if stall > 80:
                    break
        
        # Adaptive step size
        if (time.monotonic() - t0) < 0.01 and stall > 0 and stall % 100 == 0:
            t_fixed = max(0.05, t_fixed * 0.8)
    
    return best_a


def lp_alternate_linsearch(a0, time_budget=60.0, verbose=False):
    """LP alternating with 1D line search for better convergence."""
    from scipy.optimize import minimize_scalar
    
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    last_t = 0.5
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)
        
        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 20, 'xatol': 0.02})
            best_t = res.x
        except Exception:
            best_t = last_t
        
        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            last_t = best_t
            stall = 0
            if verbose:
                print(f"      LP(ls): obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 30:
                break
    
    return best_a


def smoothed_gradient_descent(a0, time_budget=60.0, tau=0.01, verbose=False):
    """
    Gradient descent on the smoothed objective: tau * log(sum exp(b[k]/tau)).
    This gives a smooth surrogate for max(b[k]).
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.05
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        # Smoothed gradient: sum_k softmax(conv[k]/tau) * grad(conv[k])
        # softmax weights
        log_w = (conv - np.max(conv)) / tau  # log of softmax (unnormalized)
        w = np.exp(log_w)
        w = w / np.sum(w)
        
        # Gradient of sum_k w[k] * conv[k] w.r.t. a
        # = sum_k w[k] * grad conv[k]
        # grad conv[k][m] = 2 * a[k-m] (if valid)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            # grad conv[k][m] = a[k-m] + a[m-(k-n+1)] ... 
            # Actually conv[k] = sum_j a[j]*a[k-j]
            # d/d a[m] = a[k-m] (if 0<=k-m<n) + a[m] * (if k-m is... )
            # Wait: conv[k] = sum_j a[j] a[k-j]
            # d/d a[m]: term j=m gives a[k-m], term k-j=m gives j=k-m, so a[k-m]
            # So d/d a[m] = a[k-m] + a[k-m] = 2*a[k-m] if 0 <= k-m < n
            # But if m == k-m, i.e., k = 2m, then it's a[m]^2, derivative is 2*a[m] (same formula)
            # So: d/d a[m] conv[k] = 2*a[k-m] if 0 <= k-m < n, else 0
            i = k - m  # wait, this is per-m
            # For fixed k, the gradient w.r.t. a[m] is 2*a[k-m] if 0<=k-m<n
            # So grad[m] += w[k] * 2*a[k-m] for each k where 0<=k-m<n
            pass
        
        # More efficient: use the structure
        # grad[m] = 2 * sum_k w[k] * a[k-m] (where 0 <= k-m < n)
        # = 2 * (w * a)[m] (cross-correlation of w and a)
        # where (w * a)[m] = sum_k w[k] * a[k-m]
        # This is a cross-correlation!
        # Let's compute it via FFT
        # Actually, let me just do it directly for efficiency
        for m in range(n):
            k_lo = max(0, m)
            k_hi = min(2 * n - 2, m + n - 1)
            # sum over k from k_lo to k_hi of w[k] * a[k-m]
            # k-m ranges from k_lo-m to k_hi-m, which is from max(0,m)-m to min(2n-2,m+n-1)-m
            # = max(0,m)-m to min(2n-2-m, n-1)
            # If m >= 0: from 0 to min(2n-2-m, n-1)
            lo = 0
            hi = min(2 * n - 2 - m, n - 1)
            if lo <= hi:
                grad[m] = 2.0 * np.dot(w[lo + m:hi + m + 1], a[lo:hi + 1])
        
        # Project gradient onto tangent space of simplex
        grad = grad - np.mean(grad)
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        a_new = a_new / np.sum(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = best_a.copy()
            lr *= 0.8
            if lr < 0.001:
                break
        
        if verbose and it % 100 == 0:
            print(f"      SGD: obj={best_obj:.8f}, tau={tau}, lr={lr:.4f}, it={it}")
    
    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def perturb_seq(a, rng, method='gauss', scale=0.05):
    """Generate a perturbation of a."""
    n = len(a)
    if method == 'gauss':
        a_pert = a * np.exp(rng.normal(0, scale, n))
    elif method == 'smooth':
        # Multiply by a random smooth function
        freqs = rng.choice([1, 2, 3, 4, 5], size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        amps = rng.uniform(0.1, scale, size=3)
        modulation = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            modulation += amp * np.cos(2 * np.pi * f * np.arange(n) / n + p)
        a_pert = a * modulation
    elif method == 'swap':
        a_pert = a.copy()
        for _ in range(n // 20):
            i, j = rng.integers(0, n, size=2)
            a_pert[i], a_pert[j] = a_pert[j], a_pert[i]
    elif method == 'block':
        # Perturb a random block
        a_pert = a.copy()
        block_size = max(5, n // 10)
        start = rng.integers(0, n - block_size)
        a_pert[start:start+block_size] *= np.exp(rng.normal(0, scale * 2, block_size))
    else:
        a_pert = a * np.exp(rng.normal(0, scale, n))
    
    a_pert = np.maximum(a_pert, 0)
    s = float(np.sum(a_pert))
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw

"""
LP-alternating optimizer with diverse restarts, smoothed gradient, and n-sweeping.
Uses high-throughput LP iterations (fixed step for speed), geometric-mean updates,
structured perturbations for local-minimum escaping, and FFT-based smoothed gradients.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(m, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    solvers = [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    for sol in solvers:
        try:
            prob.solve(solver=sol)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_obj(a, g_lp, t):
    """Objective for line search: obj((1-t)*a + t*g_lp)."""
    a_try = (1.0 - t) * a + t * g_lp
    a_try = np.maximum(a_try, 0)
    st = float(np.sum(a_try))
    if st < 1e-10:
        return np.inf
    a_try = a_try / st
    return compute_obj(a_try)


def lp_alternate_fast(a0, time_budget=60.0, t_fixed=0.4, verbose=False):
    """Fast LP alternating with fixed step size."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    t = t_fixed

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        a_try = (1.0 - t) * a + t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall = 0

        else:

            # Reduce step size

            if stall > 150:

    return best_a


def lp_alternate_ls(a0, time_budget=60.0, verbose=False):
    """LP alternating with line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:

    best_a = a.copy()

    start = time.monotonic()
    stall = 0

        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Line search using a wrapper
        class LSFunc:
            def __init__(self, a_arr, g_arr):
                self.a_arr = a_arr
                self.g_arr = g_arr

                return line_search_obj(self.a_arr, self.g_arr, t)

            best_t = 0.5

        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))

            break

            best_a = a_try.copy()
            a = best_a
            stall = 0
            if verbose:
                print(f"      LP ls: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 25:
                break

    return best_a


def compute_smoothed_grad(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a using FFT cross-correlation."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w, size)
    # cross-correlation: (w ⊛ a)[m] = sum_k w[k] * a[k-m]
    # = IDFT(FW * conj(FA))[m]
    cross = irfft(fw * np.conj(fa), size)
    grad = 2.0 * cross[:n]
    return grad


def smoothed_gradient_descent(a0, time_budget=60.0, tau=0.01, verbose=False):
    """Gradient descent on smoothed objective tau*log(sum exp(b[k]/tau))."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.03
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Softmax weights
        log_w = (conv - np.max(conv)) / tau
        w = np.exp(log_w)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            break
        w = w / w_sum

        grad = compute_smoothed_grad(a, w)
        # Project onto tangent space of simplex
        grad = grad - np.mean(grad)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            it += 1
        else:
            a = best_a.copy()
            lr *= 0.85
            it += 1
            if lr < 0.0005:
                break

        if verbose and it % 50 == 0:
            print(f"      SGD: obj={best_obj:.8f}, tau={tau}, lr={lr:.5f}, it={it}, t={time.monotonic()-start:.1f}s")

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def perturb_seq(a, rng, method='gauss', scale=0.03):
    """Generate a perturbation of a."""
    n = len(a)
    if method == 'gauss':
        a_pert = a * np.exp(rng.normal(0, scale, n))
    elif method == 'smooth':
        freqs = rng.integers(1, 6, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        amps = rng.uniform(0.1, scale, size=3)
        modulation = np.ones(n)
        for i in range(3):
            modulation = modulation + amps[i] * np.cos(2 * np.pi * freqs[i] * np.arange(n) / n + phases[i])
        a_pert = a * modulation
    elif method == 'swap':
        a_pert = a.copy()
        n_swaps = max(3, n // 30)
        for _ in range(n_swaps):
            idx = rng.integers(0, n, size=2)
            a_pert[idx[0]], a_pert[idx[1]] = a_pert[idx[1]], a_pert[idx[0]]
    elif method == 'block':
        a_pert = a.copy()
        bs = max(5, n // 8)
        st = rng.integers(0, max(1, n - bs))
        a_pert[st:st + bs] = a_pert[st:st + bs] * np.exp(rng.normal(0, scale * 2, bs))
    elif method == 'inverse':
        # Multiply by inverse-like modulation
        a_pert = a * (1.0 + scale * np.sin(2 * np.pi * np.arange(n) / n * rng.uniform(1, 4)))
    else:
        a_pert = a * np.exp(rng.normal(0, scale, n))

    a_pert = np.maximum(a_pert, 0)
    s = float(np.sum(a_pert))
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def gaussian_window(n, sigma_frac=0.2):
    """Gaussian window normalized to sum=1."""
    center = (n - 1) / 2.0
    w = np.exp(-0.5 * ((np.arange(n) - center) / (n * sigma_frac)) ** 2)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw

            alloc = min(15.0, remaining * 0.25)
            a_opt = lp_alternate_fast(a_pert, time_budget=alloc, t_fixed=0.5)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score - 1e-13:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Fine R{r} (s={scale:.4f}): obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
        print(f"  evaluate_sequence: {val:.8f}")
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
