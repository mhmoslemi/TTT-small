    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

```python
def project_simplex(v, s=1.0):
    """Project v onto {x: x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - s))[0][-1]
    theta = (cv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

"""
Optimized step function search combining smooth-max (log-sum-exp) gradient descent
on the probability simplex with Nesterov acceleration, LP alternating refinement,
spectral shaping, and adaptive length sweep.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import linprog


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v, s=1.0):
    """Project vector v onto the simplex {x: x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cumulative = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cumulative - s))[0]
    if len(rho_candidates) == 0:
        return np.full(n, s / n)
    rho = rho_candidates[-1]
    theta = (cumulative[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_descent(a, time_budget=300.0, tau_init=5.0, tau_max=200.0,
                       lr_init=0.05, beta=0.9, verbose=False):
    """
    Smooth-max (log-sum-exp) gradient descent on the probability simplex
    with Nesterov-style momentum and adaptive step size.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    momentum = np.zeros(n)
    lr = lr_init
    tau = tau_init
    stall_count = 0
    prev_obj = best_obj

    while time.monotonic() - start < time_budget:
        R = compute_conv(a)
        max_R = float(np.max(R))
        obj = 2.0 * n * max_R

        # Softmax weights with numerical stability
        shifted = tau * (R - max_R)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)

        # Gradient: 2 * correlate(w, a, 'valid')
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # Project to tangent space of simplex
        grad_proj = grad - np.mean(grad)

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad_proj

        # Try step with backtracking
        success = False
        for shrunk_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            a_new = a - shrunk_lr * momentum
            a_new = project_simplex(a_new, 1.0)
            R_new = compute_conv(a_new)
            obj_new = 2.0 * n * float(np.max(R_new))
            if obj_new < obj + 1e-12:
                success = True
                a = a_new
                lr = shrunk_lr
                break

        if not success:
            # Reset and try smaller step
            momentum = np.zeros(n)
            a_new = a - lr * 0.01 * grad_proj
            a_new = project_simplex(a_new, 1.0)
            a = a_new

        # Update best
        R_curr = compute_conv(a)
        obj_curr = 2.0 * n * float(np.max(R_curr))
        if obj_curr < best_obj - 1e-10:
            best_obj = obj_curr
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 200:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            scale = 0.05 + 0.02 * min(stall_count // 200, 5)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s_a = float(np.sum(a))
            if s_a > 1e-10:
                a = a / s_a
            momentum = np.zeros(n)
            lr = lr_init
            tau = max(tau * 0.5, tau_init)
            stall_count = 0
            prev_obj = best_obj

        # Gradually increase tau
        tau = min(tau * 1.005, tau_max)

        # Adaptive learning rate decay
        if int(time.monotonic() - start) % 30 == 0:
            lr = max(lr * 0.95, lr_init * 0.1)

        if verbose and not int((time.monotonic() - start) % 20):
            elapsed = time.monotonic() - start
            print(f"    SGD: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.5f}, time={elapsed:.0f}s")

    return best_a


def lp_alternate(a, time_budget=120.0, momentum_beta=0.5, verbose=False):
    """
    LP alternating: solve LP for best response g to current a,
    then mix a and g with adaptive step size and momentum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    momentum = np.zeros(n)
    stall = 0

    # Build convolution matrix
    def build_A(a_curr):
        m = 2 * n - 1
        A = np.zeros((m, n))
        for j in range(n):
            r = min(n, m - j)
            A[j:j + r, j] = a_curr[:r]
        return A

    while time.monotonic() - start < time_budget:
        A = build_A(a)

        # Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0
        g_val = None
        try:
            m = 2 * n - 1
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.zeros((m, n + 1))
            A_ub[:, :n] = A
            A_ub[:, n] = -1.0
            b_ub = np.zeros(m)
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(None, None)]
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
        except Exception:
            pass

        if g_val is None:
            break

        # Try multiple mixing parameters
        direction = g_val - a
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * direction

        best_t_obj = best_obj
        best_a_local = best_a.copy()

        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = a + t * momentum
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a.copy()
            stall = 0
            if verbose and not int((time.monotonic() - start) % 15):
                print(f"    LP: obj={best_obj:.8f}, time={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            momentum *= 0.5
            if stall % 15 == 0:
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                momentum = np.zeros(n)
                stall = 0

    return best_a


def spectral_shaping(a, time_budget=60.0, verbose=False):
    """
    Spectral shaping: flatten the power spectrum iteratively to reduce autocorrelation peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    alpha = 0.3
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        A = np.fft.fft(a, N)
        power = np.abs(A) ** 2
        phase = np.angle(A)

        # Flatten: reduce dynamic range
        mean_p = np.mean(power)
        min_p = np.min(power[power > 1e-20])
        max_p = np.max(power)
        if max_p > 0 and mean_p > 0:
            # Geometric mean shaping
            new_power = power * (mean_p / np.maximum(power, 1e-30)) ** alpha
            # Also try a floor
            floor = min_p * 0.5
            new_power = np.maximum(new_power, floor)

        A_new = np.sqrt(new_power) * np.exp(1j * phase)
        # Ensure conjugate symmetry
        a_new = np.real(np.fft.ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            break
        a_new = a_new / s_new

        obj = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()

        # Gradually increase flattening
        alpha = min(alpha * 1.02, 0.9)

    return best_a


def coordinate_polish(a, time_budget=60.0):
    """
    Pairwise coordinate swaps to reduce the peak autocorrelation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for iteration in range(2000):
        if time.monotonic() - start > time_budget:
            break

        R = compute_conv(a)
        k_star = int(np.argmax(R))

        # Compute contributions to the peak
        contrib = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contrib[j] = a[j] * a[k_j]

        # Try swapping mass from high-contribution to low-contribution positions
        sorted_idx = np.argsort(contrib)[::-1]
        improved = False

        for frac in [0.01, 0.02, 0.05]:
            if improved:
                break
            for i in sorted_idx[:15]:
                if time.monotonic() - start > time_budget:
                    break
                # Target: position that contributes least and is "far" from peak
                candidates = np.argsort(contrib)[:15]
                for j in candidates:
                    if j == i:
                        continue
                    delta = frac * a[i]
                    if delta <= 1e-10 or delta > a[i] * 0.9:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    R_new = compute_conv(a)
                    obj_new = 2.0 * n * float(np.max(R_new))
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation
            if rng.random() < 0.3:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j and a[i] > 1e-8:
                    delta = 0.02 * a[i]
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0)
                    s_a = float(np.sum(a))
                    if s_a > 1e-10:
                        a = a / s_a
                    obj = 2.0 * n * float(np.max(compute_conv(a)))
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a.copy()

    return best_a


def generate_init(kind, n, rng):
    """Generate diverse initialization of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 8)
        a[:edge] = 0.2
        a[-edge:] = 0.2
        return a / np.sum(a)
    elif kind == 'bimodal':
        a = np.ones(n) * 0.3
        a[n//4] = 1.5
        a[3*n//4] = 1.5
        kernel = np.ones(max(3, n//20)) / max(3, n//20)
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'triangular':
        x = np.arange(n)
        a = 1.0 - 2.0 * np.abs(x / n - 0.5)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif kind == 'other_periodic':
        x = np.arange(n)
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x / n)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            print(f"  Coord: obj={obj:.8f}")

    # === Phase 6: Length Sweep (200s) ===
    print(f"\n=== Phase 6: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 100),
        max(200, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        1200,
        1500,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            continue  # already optimized

        a_try = resample_sequence(best_seq, n_try)
        s_try = float(np.sum(a_try))
        if s_try > 1e-10:
            a_try = a_try / s_try

        # Quick optimize: SGD + LP
        alloc_sgd = min(25.0, remaining * 0.3)
        alloc_lp = min(20.0, remaining * 0.25)

        a_opt = smooth_max_descent(a_try, time_budget=alloc_sgd, tau_init=10.0,
                                    tau_max=200.0, lr_init=0.03, beta=0.9)
        a_opt = lp_alternate(a_opt, time_budget=alloc_lp, momentum_beta=0.5)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

        n_best = len(best_seq)

    # === Phase 7: Final combined polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)

    # Alternate SGD and LP for remaining time
    cycles = 0
    while remaining > 25 and cycles < 10:
        alloc_sgd = min(remaining * 0.4, 60.0)
        alloc_lp = min(remaining * 0.3, 40.0)

        if remaining > 30:
            a_new = smooth_max_descent(best_seq, time_budget=alloc_sgd,
                                        tau_init=30.0, tau_max=500.0,
                                        lr_init=0.01, beta=0.95)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Final SGD {cycles}: obj={best_score:.8f} **")
            else:
                print(f"  Final SGD {cycles}: obj={obj:.8f}")

        if remaining > 25:
            a_new = lp_alternate(best_seq, time_budget=alloc_lp, momentum_beta=0.5)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Final LP {cycles}: obj={best_score:.8f} **")
            else:
                print(f"  Final LP {cycles}: obj={obj:.8f}")

        remaining = deadline - time.monotonic()
        cycles += 1

    # Final coordinate polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd = coordinate_polish(best_seq, time_budget=min(remaining * 0.5, 20.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
