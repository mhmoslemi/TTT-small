def compute_subgradient(a, k_indices):
    """Compute subgradient at a for the max of conv(a,a) at positions k_indices."""
    n = len(a)
    grad = np.zeros(n)
    for k in k_indices:
        k = int(k)
        # grad[j] += a[k-j] for valid j (0 <= j < n, 0 <= k-j < n)
        # j ranges from max(0, k-(n-1)) to min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad[j_start:j_end+1] += a[k - j_end:k - j_start + 1][::-1]
    grad /= len(k_indices)
    return grad

"""
Optimizer using FISTA with adaptive smoothing for the max-convolution objective,
combined with LP alternating for final polish and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2 for a non-negative sequence."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)

def fista_optimize(a, time_budget=600.0, lam_0=50.0, lam_final=5.0, verbose=True):
    """
    FISTA with exponentially weighted smoothed max-convolution objective.
    a: initial sequence (will be normalized)
    time_budget: seconds to spend
    lam_0: initial inverse temperature (large = sharp peak selection)
    lam_final: final inverse temperature
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = a.copy()  # FISTA iterate
    y = a.copy()  # FISTA lookup
    t_k = 1.0
    
    # Lipschitz constant estimate
    L = 4.0 * n
    
    n_iters = 0
    for it in range(2000000):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive lambda: exponential decay
        frac = min(1.0, it / 10000.0)
        lam = lam_0 * np.exp(np.log(lam_final / lam_0) * frac)
        
        # Compute gradient at y
        conv_y = compute_conv(y)
        # Softmax weights (numerically stable)
        conv_max = np.max(conv_y)
        exp_vals = np.exp(lam * (conv_y - conv_max))
        w = exp_vals / np.sum(exp_vals)
        
        # Gradient: grad[j] = sum_k w[k] * a[k-j] = correlate(w, y)[j + n - 1]
        # Using np.correlate
        corr = np.correlate(w, y, mode='full')
        grad = corr[n-1 : 2*n-1]  # lags 0 to n-1, length n
        
        # Backtracking line search
        step = 1.0 / L
        for bs in range(20):
            x_trial = project_simplex(y - step * grad)
            # Check if objective decreased (using exact objective for safety)
            obj_trial = compute_obj(x_trial)
            obj_y = compute_obj(y)
            if obj_trial <= obj_y + 1e-12 + 0.5 * step * np.sum(grad * (x_trial - y)):
                break
            step *= 0.5
        
        # FISTA update
        x_new = x_trial
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + (t_k - 1.0) / t_new * (x_new - x)
        y = project_simplex(y)
        x = x_new
        t_k = t_new
        n_iters += 1
        
        # Track best
        obj_x = compute_obj(x)
        if obj_x < best_obj - 1e-13:
            best_obj = obj_x
            best_a = x.copy()
            if verbose and (n_iters % 1000 == 0 or obj_x < best_obj + 1e-10):
                elapsed = time.monotonic() - start
                print(f"    FISTA: it={n_iters}, obj={best_obj:.8f}, lam={lam:.2f}, time={elapsed:.0f}s")
        
        if verbose and n_iters % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    FISTA: it={n_iters}, cur_obj={compute_obj(x):.8f}, best={best_obj:.8f}, time={elapsed:.0f}s")
    
    if verbose:
        print(f"    FISTA done: {n_iters} iters, best_obj={best_obj:.8f}")
    
    return best_a

def fast_lp_alternate(a, time_budget=100.0, verbose=False):
    """Fast LP alternating using scipy linprog directly."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    # Build the constraint matrix A where A[k,j] = a[k-j] if 0 <= k-j < n
    # conv(a, g)[k] = sum_j A[k,j] * g[j]
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    while time.monotonic() - start < time_budget:
        # Solve LP: min M s.t. A@g <= M*1, sum(g)=1, g>=0
        # Variables: [g_0, ..., g_{n-1}, M]
        # c = [0, ..., 0, 1]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        # Inequality: A@g - M <= 0 => A@g + (-1)*M <= 0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        
        # Equality: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(-np.inf, np.inf)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status != 0 or res.x is None:
            break
        
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs < 1e-10:
            break
        g = g / gs
        
        # Try blending
        best_local = np.inf
        best_local_a = best_a.copy()
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - alpha) * a + alpha * g
            a_try = np.maximum(a_try, 0)

            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local:
                best_local = obj_try
                best_local_a = a_try.copy()
        
        if best_local < best_obj - 1e-12:

            best_a = best_local_a.copy()
            a = best_a.copy()
            stall_count = 0
            if verbose:
                elapsed = time.monotonic() - start

            stall_count += 1

                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:

                stall_count = 0

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

Optimizer combining FISTA with adaptive smoothing for fast convergence,
LP alternating for precise optimization, and multi-n exploration.
Targets the minimum of 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.ndimage import gaussian_filter1d
import time


def compute_conv(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0)


def fista_optimize(a, time_budget=300.0, lam_0=80.0, lam_final=8.0, verbose=True):
    """FISTA with exponential smoothing of the max-convolution objective."""
    n = len(a)

    s = float(np.sum(a))

    best_obj = compute_obj(a)

    y = a.copy()

        lam = lam_0 * (lam_final / lam_0) ** frac

        conv_max = np.max(conv_y)
        exp_vals = np.exp(lam * (conv_y - conv_max))
        w = exp_vals / np.sum(exp_vals)

        corr = np.correlate(w, y, mode='full')
        grad = corr[n - 1:2 * n - 1].copy()

        step = 1.0 / L
        x_new = project_simplex(y - step * grad)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_for_a(a):
