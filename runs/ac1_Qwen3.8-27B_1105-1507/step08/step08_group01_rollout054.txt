"""
Optimized step function coefficient search using L-BFGS with smooth-max
approximation, symmetric constraint exploitation, and multi-resolution strategy.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_max_and_weights(conv, T):
    """Compute smooth max value and softmax weights."""
    conv_max = np.max(conv)
    shifted = (conv - conv_max) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    smooth_val = conv_max + T * np.log(total)
    return smooth_val, weights

def obj_grad_smooth(x, T):
    """
    Objective and gradient for L-BFGS.
    x is the log-parameter; a = softmax(x).
    Returns (f, grad_x).
    """

    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    a = a / np.sum(a)

    smooth_val, weights = smooth_max_and_weights(conv, T)
    f = 2.0 * n * smooth_val
    f -= 2.0 * n * T * 0  # no correction needed
    
    # Cross-correlation: g_cross[i] = sum_j a[j] * weights[i+j]
    g_cross = np.correlate(weights, a, mode='valid')
    
    # grad_a[i] = 4n * g_cross[i]  (but we'll just use the direction)
    grad_a = n * g_cross  # factor of 2n * 2 = 4n, but we can drop the constant
    dot_ag = float(np.dot(a, grad_a))
    grad_x = a * (grad_a - dot_ag)
    
    return f, grad_x

def obj_grad_smooth_sym(x, T, n):
    """
    Objective and gradient for symmetric sequences.
    x has length n//2 (for even n). a is constructed as symmetric.
    """
    half = n // 2
    # Construct symmetric a from x
    xs = np.exp(x - np.max(x))
    xs = xs / np.sum(xs)
    a = np.zeros(n)
    a[:half] = xs
    a[half:] = xs[::-1] if n % 2 == 0 else np.concatenate([xs[::-1], [xs[-1]]])
    # Wait, for even n: a[0..half-1] = xs, a[half..n-1] = xs[::-1]
    # For odd n: a[0..half-1] = xs[:-1], a[half] = xs[-1], a[half+1..] = xs[:-1][::-1]
    # Let me just handle even n for simplicity
    
    if n % 2 == 0:
        a[:half] = xs
        a[half:] = xs[::-1]
    else:
        a[:half] = xs[:half]
        a[half] = xs[half]
        a[half+1:] = xs[:half][::-1]
    
    a = a / np.sum(a)
    
    conv = compute_conv(a)
    smooth_val, weights = smooth_max_and_weights(conv, T)
    f = 2.0 * n * smooth_val
    
    g_cross = np.correlate(weights, a, mode='valid')
    grad_a = n * g_cross
    
    # Now we need d(f)/d(x) through the symmetric construction
    # a[i] for i < half: a[i] = xs[i] = exp(x[i]) / sum(exp(x))
    # a[i] for i >= half: a[i] = xs[n-1-i] = exp(x[n-1-i]) / sum(exp(x))
    
    # This is getting complex. Let me just use the full parameterization for now.
    # Map grad_a back to grad_x
    grad_xs = np.zeros(half)
    for i in range(half):
        grad_xs[i] = grad_a[i]
        ji = n - 1 - i
        if ji >= half:
            grad_xs[i] += grad_a[ji]
    
    # Now grad_xs is the gradient of f w.r.t. xs (the unnormalized softmax inputs)
    # f = g(xs_normalized) where xs_normalized = xs / sum(xs)
    # dfs/dx[i] = xs[i] * (grad_xs[i] / sum(xs) - xs[i] * sum(grad_xs * xs) / (sum(xs))^2)
    # Wait, let me just use the chain rule properly.
    
    # xs[i] = exp(x[i]) / S where S = sum_j exp(x[j])
    # dxs[i]/dx[i] = xs[i] * (1 - xs[i])
    # dxs[i]/dx[j] = -xs[i]*xs[j] for i != j
    # df/dx[i] = sum_k grad_xs[k] * dxs[k]/dx[i]
    # = xs[i] * (grad_xs[i] - xs[i] * sum_k grad_xs[k]*xs[k])
    
    dot_xs_gs = float(np.dot(xs, grad_xs))
    grad_x = xs * (grad_xs - dot_xs_gs)
    
    return f, grad_x

def lbfgs_optimize(a_init, T, maxiter=500, tol=1e-8, verbose=False):
    """Run L-BFGS-B optimization on softmax parameterization."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / np.sum(a_init)
    x0 = np.log(a_init)
    
    def fun_grad(x):
        f, g = obj_grad_smooth(x, T)
        return f, g
    
    res = minimize(fun_grad, x0, jac=True, method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-7,
                            'maxfun': maxiter * 3})
    x_opt = res.x
    a_opt = np.exp(x_opt - np.max(x_opt))
    a_opt = a_opt / np.sum(a_opt)
    return a_opt

def coordinate_descent_polish(a, time_budget=50.0, delta=1e-4):
    """Coordinate descent polish."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count + hash(n) % 10000)
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            for delta_try in [delta, delta * 0.3, delta * 3]:
                a_try = a.copy()
                a_try[i] += delta_try
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if improved:
                continue
            for delta_try in [delta, delta * 0.3]:
                if a[i] < delta_try * 0.5:
                    continue
                a_try = a.copy()
                a_try[i] -= delta_try
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        if not improved and pass_count > 3:
            delta *= 0.5
            if delta < 1e-7:
                break
    
    return best_a

def adaptive_temperature_lbfgs(a_init, time_budget=200.0, T_start=0.01, T_end=0.0001, verbose=False):

def peak_shaving(a, time_budget=60.0):
    """Iteratively reduce the top convolution peaks."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_val = np.max(conv)
        
        # Find top peaks
        peak_indices = np.argsort(conv)[::-1][:5]
        
        for k in peak_indices:
            # Determine which a[i] contribute to conv[k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            contributors = np.arange(lo, hi + 1)
            
            # Reduce the contributor with highest value
            if len(contributors) == 0:
                continue
            vals = a[contributors]
            idx = contributors[np.argmax(vals)]
            
            # Try reducing
            for frac in [0.01, 0.005, 0.001]:
                a_try = a.copy()
                a_try[idx] *= (1 - frac)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimized step function coefficient search using L-BFGS with smooth-max approximation,
QP-alternating refinement, FISTA, coordinate descent, and multi-resolution strategy.
Key innovations over previous approach:
- L-BFGS on softmax parameterization (much faster per iteration than LP)
- Adaptive temperature annealing for smooth->sharp transition
- QP-alternating (direct quadratic optimization instead of LP)
- FISTA for robustness and escaping dead zones
- Diverse structured starting points
- Multi-n exploration with resampling

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_and_weights(conv, T):
    """Compute smooth max value and softmax weights over conv."""
    conv_max = np.max(conv)
    shifted = (conv - conv_max) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    smooth_val = conv_max + T * np.log(total)
    return smooth_val, weights


def obj_grad_softmax(x, T):
    """
    Objective and gradient for L-BFGS with softmax parameterization.
    x is log-parameter; a = softmax(x).
    Minimizes 2n * smooth_max(conv(a,a), T) over a in simplex.
    Returns (f_val, grad_x).
    """
    n = len(x)
    x_shifted = x - np.max(x)
    a = np.exp(x_shifted)
    a_sum = np.sum(a)
    a = a / a_sum

    conv = compute_conv(a)
    smooth_val, weights = smooth_max_and_weights(conv, T)
    f_val = 2.0 * n * smooth_val

    # Cross-correlation: g_cross[i] = sum_j weights[i+j] * a[j]
    g_cross = np.correlate(weights, a, mode='valid')

    # grad_a[i] = 4n * g_cross[i]
    grad_a = 4.0 * n * g_cross
    dot_ag = float(np.dot(a, grad_a))
    # Softmax chain rule: grad_x[i] = a[i] * (grad_a[i] - dot(a, grad_a))
    grad_x = a * (grad_a - dot_ag)

    # Clip gradient for numerical stability
    max_g = np.max(np.abs(grad_x))
    if max_g > 1e6:
        grad_x = grad_x * (1e6 / max_g)

    return f_val, grad_x


def lbfgs_optimize(a_init, T, maxiter=300, maxfun=1000):
    """Run L-BFGS-B optimization on softmax parameterization."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-12)
    a_init = a_init / np.sum(a_init)
    x0 = np.log(a_init)

    def fun_grad(x):
        f, g = obj_grad_softmax(x, T)
        return f, g

    try:
        res = minimize(fun_grad, x0, jac=True, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'maxfun': maxfun,
                                'ftol': 1e-12, 'gtol': 1e-7, 'maxls': 30,
                                'iprint': -1})
        x_opt = res.x
        a_opt = np.exp(x_opt - np.max(x_opt))
        a_opt = a_opt / np.sum(a_opt)
        return a_opt
    except Exception:
        return a_init.copy()

def coordinate_descent_polish(a, time_budget=60.0, delta=1e-4):
    """Coordinate descent with adaptive step sizes."""

        return a

    pass_count = 0

    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count + hash(n) % 100000)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

                a_try = a.copy()
                a_try[i] += delta_try
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                continue

            # Try decreasing
            for delta_try in [delta, delta * 0.3]:
                if a[i] < delta_try * 0.5:
                    continue
                a_try = a.copy()
                a_try[i] -= delta_try
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved and pass_count > 2:
            delta *= 0.5
            if delta < 1e-8:
                break

    return best_a


def peak_shaving(a, time_budget=60.0):
    """Iteratively reduce top convolution peaks by adjusting contributing coefficients."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(300):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        peak_indices = np.argsort(conv)[::-1][:8]

        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            contributors = np.arange(lo, hi + 1)
            if len(contributors) == 0:
                continue

            vals = a[contributors]
            max_idx_local = np.argmax(vals)
            idx = contributors[max_idx_local]

            for frac in [0.02, 0.01, 0.005, 0.001]:
                a_try = a.copy()
                a_try[idx] *= (1.0 - frac)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_symmetric(a):
    """Make a sequence symmetric."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    a_sym[:half] = a[:half]
    if n % 2 == 0:
        a_sym[half:] = a[:half][::-1]
    else:
        a_sym[half:] = a[:half-1][::-1]
    return a_sym
