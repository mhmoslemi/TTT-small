Let me also think about the coordinate descent approach. For coordinate descent:
- Pick coordinate i
- Try changing a[i] by delta (positive or negative)
- The effect on the objective: only conv[k] for k in [i, i+n-1] change, and they change by `2*delta*a[k-i]` (approximately).
- The new max is `max(old_max, max_changed)` where `max_changed` is the max over the affected lags.

This is O(n) per coordinate evaluation (to check all affected lags). With n coordinates, one full pass is O(n^2). For n=1000, that's 10^6 per pass. Very fast.

Actually, I can be smarter. The objective is `max_k conv[k]`. If I increase a[i] by delta, the lags k in [i, i+n-1] increase by `2*delta*a[k-i]` (approximately, ignoring the second-order term `delta^2`). The lags outside this range decrease (because we renormalize).

For a first-order approximation: the new max is approximately `max(max_unaffected, max_affected + 2*delta*a[k-i])`.

This suggests a simple rule: only increase a[i] if it doesn't push any affected lag above the current max. And decrease a[i] if it's contributing to the peak.

Let me implement a "water-filling" type algorithm:
1. Compute conv = a*a
2. Find the peak lag k*
3. For each i, compute the "contribution" to the peak: `a[i]*a[k*-i]`
4. Reduce the coordinates that contribute most to the peak
5. Renormalize
6. Repeat

This is similar to what the LP does, but in a more direct way.

Actually, I think the most effective approach is a combination:
1. **LP alternating** for global optimization (it's essentially solving a related LP at each step)
2. **Smoothed gradient** for local refinement (the smooth max gives a good gradient signal)
3. **Direct peak reduction**: identify the peak lag and directly reduce the coordinates contributing to it

Let me code this up now. I'll focus on making the smoothed gradient approach robust and efficient.

One more thought: the "dual" of the LP gives us the weights p[k] on each lag. These are exactly the softmax weights for the smoothed max (in the limit of the LP, they're put on the active lags). So the LP's dual solution directly gives us the gradient for the smoothed objective!

This means: solve LP -> get dual y[k] -> normalize to get p[k] -> compute gradient -> update a.

This is a nice connection. The LP gives the "exact" subgradient (concentrated on active constraints), while the smoothed version gives a "fuzzy" subgradient (spread over all lags with softmax weights).

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ z gives cross-conv a*z."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. A@z <= M, sum(z)=1, z>=0. Returns (z, M, dual_y)."""
    n = len(a)
    A = build_conv_matrix(a)
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except:
                        pass
                    return z_val, float(M.value), dual_y
        except:
            continue
    return None, None, None

def smooth_objective(a, tau):
    """Smoothed version of max(conv(a,a))."""
    conv = compute_conv(a)
    m = np.max(conv)
    return m + np.log(np.sum(np.exp(tau * (conv - m)))) / tau

def smooth_gradient(a, tau):
    """Gradient of smooth_objective w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    m = np.max(conv)
    exp_vals = np.exp(tau * (conv - m))
    p = exp_vals / np.sum(exp_vals)
    # grad[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i)
    # = 2 * (a_rev * p)[n-1 : 2n-1]
    grad = 2.0 * np.convolve(a[::-1], p, mode='full')[n-1:2*n-1]
    return grad

def mirror_descent_smooth(a, tau, time_budget=100.0, verbose=False):
    """Mirror descent (multiplicative weights) with smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 1.0 / (2.0 * np.max(compute_conv(a)) * n)  # initial step size
    # Actually, let's use a simpler initialization
    eta = 0.5 / (n * compute_obj(a))
    
    no_improve = 0
    while time.monotonic() - start < time_budget:
        grad = smooth_gradient(a, tau)
        
        # Mirror descent step
        a_raw = a * np.exp(-eta * grad)
        a_new = np.maximum(a_raw, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Reduce step size slightly for stability
            eta *= 0.999
        else:
            # Try smaller step
            improved = False
            for frac in [0.5, 0.25, 0.1]:
                a_try = a * np.exp(-eta * frac * grad)
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if not improved:
                no_improve += 1
                if no_improve > 50:
                    break
        
        if verbose:
            elapsed = time.monotonic() - start
            if int(elapsed * 2) % 20 == 0:
                print(f"    MirrorDescent(tau={tau:.1f}): obj={best_obj:.8f}, t={elapsed:.0f}s")
    
    return best_a

        # Try random coordinates
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            # Try reducing a[i] (if it contributes to peak)
            # and increasing a[i] (if it's small)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    """Build sparse matrix A (2n-1 x n) such that A @ z gives cross-conv a*z."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A.tocsr()


def build_conv_matrix_dense(a):
    """Build dense matrix A (2n-1 x n) such that A @ z gives cross-conv a*z."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. A@z <= M, sum(z)=1, z>=0. Returns (z, M, dual_y)."""
    n = len(a)
    use_sparse = n > 500

    if use_sparse:
        A = build_conv_matrix_sparse(a)
    else:
        A = build_conv_matrix_dense(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except Exception:
                        pass
                    return z_val, float(M.value), dual_y
        except Exception:
            continue

    # Fallback: scipy linprog (only for small n)
    if n <= 3000:
        try:
            from scipy.optimize import linprog
            A_full = build_conv_matrix_dense(a)
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = A_full
            A_ub[:, n] = -1.0
            b_ub = np.zeros(2 * n - 1)
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(None, None)]
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                z_val = np.maximum(res.x[:n], 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    return z_val, float(res.x[n]), None
        except Exception:
            pass

    return None, None, None


def smooth_objective(a, tau):
    """Smoothed max of conv(a,a) using log-sum-exp."""
    conv = compute_conv(a)
    m = np.max(conv)
    return m + np.log(np.sum(np.exp(tau * (conv - m)))) / tau


def smooth_gradient(a, tau):
    """Gradient of smooth_objective w.r.t. a, computed via FFT/conv."""
    n = len(a)
    conv = compute_conv(a)
    m = np.max(conv)
    exp_vals = np.exp(tau * (conv - m))
    p = exp_vals / np.sum(exp_vals)
    # grad[i] = 2 * sum_k p[k] * a[k-i] (valid indices only)
    # = 2 * (a_rev * p)[n-1 : 2n-1] where a_rev = a[::-1]
    grad = 2.0 * np.convolve(a[::-1], p, mode='full')[n - 1:2 * n - 1]
    return grad

    if gmax < 1e-15:
        return best_a
    eta = 0.8 / gmax
    eta = min(eta, 2.0 / (n * compute_obj(a) + 1e-10))

    no_improve = 0
    last_print = 0

            if s_new < 1e-10:

def simulated_annealing(a, time_budget=60.0, verbose=False):
    """Simulated annealing with pairwise mass transfers."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(777)

    T = 0.005  # initial temperature (in objective units)
    T_min = 1e-5
    cooling = 0.9997

    n_iter = 0
    while time.monotonic() - start < time_budget and T > T_min:
        # Pick two random coordinates
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Transfer mass
        direction = rng.choice([1, -1])
        delta = direction * rng.uniform(0.01, 0.05) * (a[i] + a[j] + 1e-10)

        a_try = a.copy()
        a_try[i] -= delta
        a_try[j] += delta
        if a_try[i] < 0 or a_try[j] < 0:
            continue
        a_try = a_try / np.sum(a_try)

        obj_try = compute_obj(a_try)
        n_iter += 1

        # Metropolis acceptance
        if obj_try < current_obj:
            a = a_try
            current_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
        elif rng.random() < np.exp(-(obj_try - current_obj) / max(T, 1e-10)):
            a = a_try
            current_obj = obj_try

        T *= cooling

        if verbose and n_iter % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SA: obj={best_obj:.8f}, T={T:.6f}, t={elapsed:.0f}s")

    return best_a
