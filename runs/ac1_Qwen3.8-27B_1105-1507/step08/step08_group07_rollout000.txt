        # For j < n: 2n-2-j >= n-1 (since 2n-2-j >= 2n-2-(n-1) = n-1)
        # So (active ⋆ a)[j] = sum_{m=0}^{n-1} active[j+m] * a[m] = grad[j] ✓
        
        # Great, so cross[:n] gives us grad.
        
        # Ascent direction: we want to DECREASE the peaks, so move opposite to grad
        a_new = a - eta * grad[:n]
        
        # Project onto {a >= 0, sum(a) = 1}
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
    
    return a_new


def nesterov_subgradient(a, time_budget=50.0, verbose=False):
    """
    Nesterov accelerated subgradient method.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov momentum
    y = a.copy()
    momentum = 0.0
    
    # Adaptive step size
    eta = 0.02
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        t_k = 1.0 / (1.0 + momentum)
        y = a + momentum * (a - y) if it > 0 else a.copy()
        # Standard Nesterov: y_k = x_k + (t_{k-1} - 1)/t_k * (x_k - x_{k-1})
        # Simplified: y = x + beta * (x - x_prev)
        
        it += 1
        beta = (1.0 - momentum) / (1.0 + momentum)
        momentum = min(0.9, momentum + 0.01)
        
        # Compute gradient at y
        conv_y = compute_conv(y)
        max_val = float(np.max(conv_y))
        threshold = max_val * 0.9
        active = conv_y >= threshold
        n_active = max(1, int(np.sum(active)))
        
        # Compute subgradient
        size = 1
        while size < 2*n - 1:
            size <<= 1
        fy = rfft(y, size)
        factive = rfft(active.astype(np.float64), size)
        cross = irfft(np.conj(factive) * fy, size)[:n]
        grad = cross / n_active
        grad = grad - np.mean(grad)
        
        # Step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive step size
        if obj_new > best_obj * 1.01:
            eta *= 0.9
        elif it % 100 == 0 and obj_new < best_obj:
            eta = min(eta * 1.05, 0.1)
        
        if verbose and it % 500 == 0:
            print(f"    NSub: it={it}, obj={best_obj:.8f}, eta={eta:.4f}")
    
    return best_a


def solve_lp_fast(a, time_per_lp=2.0):
    """
    Solve LP: min M s.t. (a*g)_k <= M for all k, sum(g)=1, g>=0
    Using scipy's HiGHS with the convolution structure.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    
    # Build sparse convolution matrix
    # (a*g)_k = sum_j a_{k-j} * g_j for valid indices
    # Row k, col j: a[k-j] if 0 <= k-j < n
    
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        # For column j, the entries are a[k-j] for k where 0 <= k-j < n
        # k ranges from j to j+n-1
        k_start = j
        k_end = min(j + n, 2*n - 1)
        for k in range(k_start, k_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    # Add M column: -1 for each row
    total_rows = 2 * n - 1
    rows_m = list(range(total_rows))
    cols_m = [n] * total_rows
    vals_m = [-1.0] * total_rows
    
    all_rows = rows + rows_m
    all_cols = cols + cols_m
    all_vals = vals + vals_m
    
    A_ub = csr_matrix((all_vals, (all_rows, all_cols)), shape=(total_rows, n + 1))
    b_ub = np.zeros(total_rows)
    
    # A_eq: sum(g) = 1
    A_eq = csr_matrix(np.array([[1.0] * n + [0.0]]), shape=(1, n + 1))
    b_eq = np.array([1.0])
    
    # Objective: minimize M (last variable)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=[(0, None)] * n + [(None, None)],
                     method='highs', options={'time_limit': time_per_lp})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                return g_val, float(res.x[n])
    except:
        pass
    
    return None, None


def lp_alternating_fast(a, time_budget=100.0, verbose=False):
    """
    LP alternating with fast solver and 2-cycle detection.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    prev_a = None
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        g, M = solve_lp_fast(a, time_per_lp=2.0)
        if g is None:
            break
        
        it += 1
        
        # 2-cycle detection
        if prev_a is not None:
            diff = np.max(np.abs(a - prev_a))
            if diff < 1e-6:
                # Try averaging
                a_avg = 0.5 * (a + prev_a)
                a_avg = np.maximum(a_avg, 0)
                sa = float(np.sum(a_avg))
                if sa > 1e-10:
                    a_avg = a_avg / sa
                    obj_avg = compute_obj(a_avg)
                    if obj_avg < best_obj:
                        best_obj = obj_avg
                        best_a = a_avg.copy()
                prev_a = None
                stall = 0
                continue
        
        prev_a = a.copy()
        
        # Try multiple mixing ratios
        best_cand = a.copy()
        best_cand_obj = best_obj
        for mix in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - mix) * a + mix * g
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_cand_obj:
                best_cand_obj = obj_try
                best_cand = a_try.copy()
        
        if best_cand_obj < best_obj - 1e-13:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                stall = 0
        
        if verbose and it % 20 == 0:
            print(f"    LP-fast: it={it}, obj={best_obj:.8f}")
    
    return best_a


def coordinate_descent(a, time_budget=30.0):
    """
    Coordinate descent: try adjusting individual coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        improved = False
        
        # Try random coordinates
        for _ in range(50):
            if time.monotonic() - start > time_budget:
                break
            j = rng.integers(0, n)
            for delta_frac in [0.01, 0.05, 0.1, -0.01, -0.05, -0.1]:
                delta = delta_frac * a[j]
                if delta >= 0 and a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[j] += delta
                if a_try[j] < 0:
                    continue
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        if not improved and it > 50:
            break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def spectral_perturb(a, rng, strength=0.05):
    """
    Perturb in the frequency domain to create diverse candidates.
    """
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    
    # Perturb the magnitude spectrum
    phase = np.angle(fa)
    mag = np.abs(fa)
    
    # Add random perturbation to magnitude
    mag_pert = mag * np.exp(rng.normal(0, strength, len(mag)))
    
    # Reconstruct
    fa_new = mag_pert * np.exp(1j * phase)
    a_new = irfft(fa_new, size)[:n]
    
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a
    
    return a_new
