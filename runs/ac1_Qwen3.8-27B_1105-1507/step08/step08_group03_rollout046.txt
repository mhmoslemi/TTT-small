def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_ab(a, b):
    """Compute conv(a, b) via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)

    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g

    return g

        conv_ag = compute_conv_ab(a, g_new)
        violations = conv_ag - M_new
        max_viol = float(np.max(violations))

        if max_viol < 1e-8 * max(1.0, M_new):
            g = g_new
            M = M_new
            if verbose:
                print(f"      CG LP converged in {cg_iter+1} iters, M={M:.8f}")
            break
        else:
            max_viol_idx = int(np.argmax(violations))
            if max_viol_idx not in active:
                active = np.append(active, max_viol_idx)
            g = g_new
            M = M_new

    g = np.maximum(g, 0)
    s = float(np.sum(g))
    if s < 1e-10:
        return None, None
    return g / s, M


def solve_lp_minpeak_full(a, verbose=False):
    """Solve the full LP with all constraints. Used when constraint generation fails."""
    n = len(a)
    a = a / float(np.sum(a))

    A = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    A[:, n] = -1.0

    c = np.zeros(n + 1)
    c[n] = 1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, None)]

    res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, max(0.0, float(res.x[n]))
    return None, None


def lp_alternating_cg(a, time_budget=200.0, verbose=False, max_constraints=30):
    """
    LP alternating with constraint generation and adaptive line search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    best_t = 0.5
    rng = np.random.default_rng(12345 + n)
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        # Solve LP with constraint generation
        g_lp, M_lp = solve_lp_minpeak_cg(a, max_constraints=max_constraints)
        if g_lp is None:
            g_lp, M_lp = solve_lp_minpeak_full(a)
            if g_lp is None:
                break

        # Adaptive line search
        ts = [best_t, best_t * 0.6, best_t * 1.4, 0.3, 0.5, 0.7, 0.9, 1.0]
        ts = sorted(set([t for t in ts if 0.05 < t <= 1.0]))

        best_t_obj = np.inf
        best_t_val = best_t
        best_a_local = best_a.copy()

        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_t = best_t_val
            stall = 0
            if verbose and (it % 5 == 0 or it <= 5):
                print(f"    LP-{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 20:
                noise_scale = min(0.4, 0.05 + 0.05 * (stall // 20))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                best_t = 0.5
                if verbose:
                    print(f"    LP escape (stall={stall}): obj={best_obj:.8f}")

    return best_a


def nesterov_subgradient(a, time_budget=60.0, K_start=3, K_max=20, verbose=False):
    """
    Nesterov-accelerated subgradient method on the simplex with adaptive K.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.5 / np.sqrt(n)
    stall = 0
    patience = 100
    K = K_start

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        current_K = min(K, len(conv))
        top_idx = np.argsort(conv)[::-1][:current_K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        # Adaptive temperature
        spread = (peak_vals[0] - peak_vals[-1]) / max_val
        temp = 4.0 / (1.0 + spread * 8.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        g = weighted_subgradient_fast(y, weights, top_idx)

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            # Gradually increase K as we improve
            if best_obj < 1.55 and K < K_max:
                K += 1
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.2)
                K = max(K_start, K - 2)

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

        if verbose and it % 1000 == 0:
            print(f"    NS it{it}: obj={best_obj:.8f}, K={K}, t_k={t_k:.1f}, el={time.monotonic()-start:.1f}s")

    return best_a


def mirror_descent(a, time_budget=60.0, verbose=False):
    """Mirror descent with multiplicative updates and momentum."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 2.0 / np.sqrt(n)
    min_lr = 1e-5
    stall = 0
    momentum = 0.0
    v = np.zeros(n)
    rng = np.random.default_rng(999 + n)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        if max_val < 1e-15:
            break

        # Multi-temperature Boltzmann weights
        temps = [1.5, 3.0, 7.0]
        all_w = np.zeros(len(conv))
        for temp in temps:
            w = np.exp((conv - max_val) / (temp * max_val + 1e-15))
            w /= w.sum()
            all_w += w
        all_w /= len(temps)

        # Use top 30% of weights
        threshold = np.percentile(all_w, 70)
        active = all_w > threshold
        if not active.any():
            top_idx = np.argsort(conv)[::-1][:5]
            active[top_idx] = True
        active_w = all_w[active]
        active_w /= active_w.sum()
        active_idx = np.where(active)[0]

        g = np.zeros(n)
        for w, k in zip(active_w, active_idx):
            if w > 1e-10:
                g += w * grad_conv_at_peak(a, int(k))

        v = momentum * v + g
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = max(min_lr, lr * 0.9995)
            momentum = min(0.85, momentum + 0.002)
        else:
            stall += 1
            if stall > 150:
                a = best_a * np.exp(rng.normal(0, 0.1 + 0.02 * (stall // 150), n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = 0.0
                lr = max(min_lr, lr * 0.4)
                stall = 0
            else:
                lr = max(min_lr, lr * 0.98)

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 7, n // 7]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangle
        x = np.linspace(-1, 1, n)
        a = np.maximum(0.1, 1.0 - np.abs(x))
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like
        x = np.linspace(-0.99, 0.99, n)
        a = 1.0 / np.sqrt(1 - x**2 + 0.005)
        a = a / a.sum()
    elif mode == 5:
        # Multi-frequency
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Parabolic
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x) + 0.05
        a = a / a.sum()
    elif mode == 7:
        # Power law
        x = np.linspace(0.01, 1, n)
        alpha = rng.uniform(0.3, 1.5)
        a = x**alpha + (1-x)**alpha + 0.05
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def coordinate_descent_peaks(a, time_budget=30.0):
    """
    Coordinate descent: shift mass from peak-contributing positions
    to non-contributing positions to reduce the dominant peak.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Find contributing pairs
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        contrib_indices = list(range(i_start, i_end + 1))
        contrib_vals = []
        for i in contrib_indices:
            j = k_star - i
            contrib_vals.append((a[i] * a[j], i, j))
        contrib_vals.sort(reverse=True)

        # Find non-contributing positions
        non_contrib = [i for i in range(n) if i < i_start or i > i_end]
        if not non_contrib:
            break

        # Also check second peak
        conv_masked = conv.copy()
        conv_masked[k_star] = -np.inf
        k_star2 = int(np.argmax(conv_masked))

        improved = False
        for ci in range(min(10, len(contrib_vals))):
            val, i, j = contrib_vals[ci]
            if val < 1e-13:
                break
            src = i if a[i] >= a[j] else j
            tgt_candidates = non_contrib[:6]
            for target in tgt_candidates:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = frac * a[src]
                    if delta < 1e-15:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    s_cur = float(np.sum(a))
                    a_norm = a / s_cur
                    new_conv = compute_conv(a_norm)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
            if improved:
                break

        if not improved:
            break

    return best_a


def smooth_logsumexp_step(a, T, n_steps=50):
    """
    One step of smooth optimization: minimize log-sum-exp approximation
    of max(conv(a,a)) using projected gradient.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    lr = 1.0 / np.sqrt(n)
    for _ in range(n_steps):
        conv = compute_conv(a)
        max_c = conv.max()
        z = T * (conv - max_c)
        z -= z.max()
        e = np.exp(z)
        w = e / e.sum()

        # Gradient of log-sum-exp wrt a
        g = np.zeros(n)
        active = w > 1e-8
        for k in np.where(active)[0]:
            g += w[k] * grad_conv_at_peak(a, int(k))

        a = a - lr * g / T
        a = project_simplex(a)
        lr *= 0.99

    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining constraint-generation LP alternating,
    adaptive Nesterov, mirror descent, and systematic multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.issfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Quick Nesterov warmup (~80s) ===
    print(f"\n=== Phase 1: Nesterov Warmup ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 20:
        a_ns = nesterov_subgradient(best_seq, time_budget=p1_budget, K_start=3, K_max=8, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")

    # === Phase 2: LP Alternating with constraint generation (~300s) ===
    print(f"\n=== Phase 2: LP Alternating (CG) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 300.0)
    if p2_budget > 30:
        a_lp = lp_alternating_cg(best_seq, time_budget=p2_budget, verbose=True, max_constraints=25)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP-CG: obj={best_score:.8f} **")
        n_current = len(best_seq)

    # === Phase 3: Systematic multi-n exploration (~400s) ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()

    # Generate n options: around current and some spread
    n_base = n_current
    n_options = []
    for delta in [-200, -100, -50, 50, 100, 200, 300, 500]:
        n_try = n_base + delta
        if 300 <= n_try <= 4000:
            n_options.append(n_try)
    # Add some absolute values
    for n_abs in [400, 600, 800, 1000, 1200, 1500, 2000, 2500]:
        if 300 <= n_abs <= 4000 and n_abs not in n_options:
            n_options.append(n_abs)

    n_options = sorted(set(n_options))
    # Sort by priority: closer to current first, then larger
    n_options.sort(key=lambda x: (abs(x - n_base), -x))

    n_tried = 0
    max_n_tries = 8
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= max_n_tries:
            break
        if n_try == n_current:
            continue

        n_tried += 1
        a_try = resample_sequence(best_seq, n_try)

        # Quick Nesterov
        ns_alloc = min(remaining * 0.06, 40.0)
        if ns_alloc > 10:
            a_try = nesterov_subgradient(a_try, time_budget=ns_alloc, K_start=3, K_max=8)

        # Quick LP
        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.07, 50.0)
        if lp_alloc > 15:
            a_try = lp_alternating_cg(a_try, time_budget=lp_alloc, max_constraints=20)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Intensive LP on best n found (~250s) ===
    print(f"\n=== Phase 4: Intensive LP Polish (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.25, 250.0)
    if p4_budget > 30:
        a_lp2 = lp_alternating_cg(best_seq, time_budget=p4_budget, verbose=True, max_constraints=40)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Intensive LP: obj={best_score:.8f} **")

    # === Phase 5: Mirror descent refinement (~80s) ===
    print(f"\n=== Phase 5: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.08, 80.0)
    if p5_budget > 15:
        a_md = mirror_descent(best_seq, time_budget=p5_budget, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  Mirror: obj={best_score:.8f} **")

    # === Phase 6: Coordinate descent refinement (~40s) ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_cd = coordinate_descent_peaks(best_seq, time_budget=min(remaining * 0.04, 40.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CoordDesc: obj={best_score:.8f} **")

    # === Phase 7: Diversified restarts (~100s) ===
    print(f"\n=== Phase 7: Diversified Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.15 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.12, 50.0)
        # Quick LP polish
        a_opt = lp_alternating_cg(a_r, time_budget=restart_budget, max_constraints=20)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final combined polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        a_final_lp = lp_alternating_cg(best_seq, time_budget=min(remaining * 0.35, 60.0),
                                       verbose=True, max_constraints=30)
        obj_fl = compute_obj(a_final_lp)
        if obj_fl < best_score:
            best_score = obj_fl
            best_seq = a_final_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        a_final_ns = nesterov_subgradient(best_seq, time_budget=min(remaining * 0.3, 30.0),
                                          K_start=5, K_max=15)
        obj_fn = compute_obj(a_final_ns)
        if obj_fn < best_score:
            best_score = obj_fn
            best_seq = a_final_ns.copy()
            print(f"  Final NS: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_final_cd = coordinate_descent_peaks(best_seq, time_budget=min(remaining * 0.4, 20.0))
        obj_fc = compute_obj(a_final_cd)
        if obj_fc < best_score:
            best_score = obj_fc
            best_seq = a_final_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One last LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternating_cg(best_seq, time_budget=min(remaining * 0.6, 25.0), max_constraints=30)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
