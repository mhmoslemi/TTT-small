import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp_direction(a, solver=None):
    """Solve LP to find best direction g: min M s.t. conv(a,g) <= M, g>=0, sum(g)=1"""
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Build constraint matrix A where A[k,j] = a_norm[k-j] if valid
    # A is (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver is None:
        solver_list = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    else:
        solver_list = [solver]
    
    for s in solver_list:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                sv = float(np.sum(g_val))
                if sv > 1e-10:
                    return g_val / sv, float(M.value)
        except Exception:
            continue
    
    return None, None

    n = len(a)

    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall_count = 0
    iteration = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        g_dir, M_lp = solve_lp_direction(a)
        if g_dir is None:

        # Check if we're at a fixed point
        diff = np.linalg.norm(a - g_dir)
        if diff < 1e-8:

        else:

        candidates = [t, t * 0.5, t * 1.5, 0.1, 0.3, 0.5, 0.7, 1.0]

        best_a_new = a.copy()
        best_t_val = t

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_A(a):
    """Build the (2n-1) x n matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a, solver_pref=None):
    """Solve the LP: min M s.t. A@g <= M, g >= 0, sum(g) = 1."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_A(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    if solver_pref is None:
        solver_pref = [cp.CLARABEL, cp.MOSEK, cp.ECOS]
    
    for s in solver_pref:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                sv = float(np.sum(gv))
                if sv > 1e-10:
                    return gv / sv, float(M.value)
        except Exception:
            continue
    return None, None

def de_evolve(pop, n, generations, seed):
    """Differential evolution on the simplex."""
    rng = np.random.default_rng(seed)
    pop_size = len(pop)
    F = 0.8  # Scale factor
    CR = 0.9  # Crossover rate
    
    best_idx = np.argmin([compute_obj(p) for p in pop])
    best_ever = pop[best_idx].copy()
    best_ever_obj = compute_obj(best_ever)
    
    for gen in range(generations):
        for i in range(pop_size):
            # Select two distinct random individuals
            r1, r2 = rng.choice(pop_size, size=2, replace=False)
            
            # Mutation
            v = pop[best_idx] + F * (pop[r1] - pop[r2])
            
            # Crossover

            u = np.maximum(u, 0)

            if su < 1e-10:
                continue

                    best_ever = u.copy()
                    best_ever_obj = compute_obj(u)
                    best_idx = i
        # Update best_idx
        objs = [compute_obj(p) for p in pop]
        best_idx = int(np.argmin(objs))
        if objs[best_idx] < best_ever_obj:
            best_ever = pop[best_idx].copy()
            best_ever_obj = objs[best_idx]
    
    return best_ever

"""
Step function optimization: LP-based Frank-Wolfe with multi-n exploration,
adaptive perturb-restart, and subgradient refinement.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Fast linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_A(a):
    """Build (2n-1) x n matrix where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: minimize M subject to A@g <= M, g>=0, sum(g)=1.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_A(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solver_list = [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0.0)
                sv = float(np.sum(gv))
                if sv > 1e-10:
                    return gv / sv, float(M.value)
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            gv = np.maximum(res.x[:n], 0.0)
            sv = float(np.sum(gv))
            if sv > 1e-10:
                return gv / sv, float(res.x[n])
    except Exception:
        pass

    return None, None


def fw_optimize(a, time_budget=200.0, verbose=False):
    """
    LP-based Frank-Wolfe with adaptive step size and perturb-restart.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t = 0.5
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - start < time_budget:
        g_dir, M_lp = solve_lp(a)
        if g_dir is None:
            break

        # Check fixed point
        rel_diff = np.linalg.norm(a - g_dir) / (np.linalg.norm(a) + 1e-15)
        if rel_diff < 1e-7:
            no_improve += 3
        else:
            no_improve += 0

        # Line search with multiple candidates
        candidates = [t, 2.0 * t / 3.0, t * 0.5, min(1.0, t * 1.5),
                      0.05, 0.15, 0.25, 0.35, 0.5, 0.65, 0.8, 0.95, 1.0]
        candidates = sorted(set(max(0.01, min(1.0, c)) for c in candidates))

        best_local_obj = np.inf
        best_local_a = a.copy()
        best_local_t = t

        for tc in candidates:
            a_try = (1.0 - tc) * a + tc * g_dir
            a_try = np.maximum(a_try, 0.0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            oj = compute_obj(a_try)
            if oj < best_local_obj:
                best_local_obj = oj
                best_local_a = a_try.copy()
                best_local_t = tc

        it += 1
        if best_local_obj < best_obj - 1e-13:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_local_a.copy()
            t = best_local_t
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 25:
                # Perturb and restart
                scale = 0.03 + 0.02 * min(no_improve / 25.0, 2.0)
                a = best_a * (1.0 + scale * rng.standard_normal(n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                t = 0.5
                no_improve = 0

        if verbose and it % 15 == 0:
            elapsed = time.monotonic() - start
            print(f"    FW(n={n}) it{it}: obj={best_obj:.8f}, t={t:.3f}, "
                  f"diff={rel_diff:.1e}, elapsed={elapsed:.0f}s")

    return best_a


def subgradient_refine(a, time_budget=30.0):
    """
    Projected subgradient descent on the true (non-smooth) objective.
    Uses the subgradient at the argmax convolution lag.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 3.0 / np.sqrt(n)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Use top-3 lags for a better subgradient
        top_idx = np.argsort(conv)[::-1][:3]
        top_vals = conv[top_idx]
        max_val = top_vals[0]
        if max_val < 1e-15:
            break
        # Softmax weights
        w = np.exp((top_vals - max_val) * 3.0 / max_val)
        w /= w.sum()

        grad = np.zeros(n)
        for wk, k in zip(w, top_idx):
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += wk * 2.0 * a[j]

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        sn = float(np.sum(a_new))
        if sn < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / sn

        oj = compute_obj(a_new)
        if oj < best_obj - 1e-14:
            best_obj = oj
            best_a = a_new.copy()
            a = a_new.copy()
            it += 1
        else:
            lr *= 0.85
            if lr < 1e-14:
                break

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def diverse_init(n, rng, mode):
    """Generate a diverse initialization of length n."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0.1, 1.0, n)
    elif mode == 2:
        # Two bumps
        c1, c2 = n // 3, 2 * n // 3
        a = np.exp(-((np.arange(n) - c1) ** 2) / (n * n / 20.0)) + \
            np.exp(-((np.arange(n) - c2) ** 2) / (n * n / 20.0))
    elif mode == 3:
        # U-shaped
        x = np.linspace(-1, 1, n)
        a = x ** 2 + 0.1
    elif mode == 4:
        # Bell
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3.0)
    elif mode == 5:
        # Wavy
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.1 * np.sin(2 * np.pi * 7 * x)
        a = np.maximum(a, 0.01)
    elif mode == 6:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1.0 - 2.0 * np.abs(x - 0.5)
    else:
        a = rng.uniform(0.05, 1.0, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
