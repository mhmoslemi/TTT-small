    fa = rfft(a, size)

def compute_obj(a):

    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def subgradient_topk(a, K=10):
    """
    Compute subgradient of max(conv(a,a)) using top-K peaks.
    Returns gradient vector of length n.
    """
    n = len(a)
    c = compute_conv(a)
    
    # Find top-K positions
    top_k = np.argsort(c)[-K:][::-1]
    
    # For each peak position k*, subgradient w.r.t. a[i] is a[k*-i] + a[i-k*]
    # (with a[j] = 0 for j outside [0, n-1])
    grad = np.zeros(n)
    for k_star in top_k:
        # a[k_star - i] for i in [0, n-1]: valid when 0 <= k_star - i <= n-1
        # i.e., k_star - (n-1) <= i <= k_star
        i_lo = max(0, k_star - (n - 1))
        i_hi = min(n - 1, k_star)
        if i_hi >= i_lo:
            # a[k_star - i] for i in [i_lo, i_hi]
            idx = k_star - np.arange(i_lo, i_hi + 1)
            grad[i_lo:i_hi+1] += a[idx]
        
        # a[i - k_star] for i in [0, n-1]: valid when 0 <= i - k_star <= n-1
        # i.e., k_star <= i <= k_star + n - 1
        i_lo2 = max(0, k_star)
        i_hi2 = min(n - 1, k_star + n - 1)
        if i_hi2 >= i_lo2:
            idx2 = np.arange(i_lo2, i_hi2 + 1) - k_star
            grad[i_lo2:i_hi2+1] += a[idx2]
    
    return grad / K  # average over peaks

def adam_subgradient(a, time_budget=100.0, K=10, verbose=False):
    """
    Adam optimizer with subgradient method for minimizing max(conv(a,a)).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Adam state
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.01
    t = 0
    
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        t += 1
        grad = subgradient_topk(a, K)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Projected gradient step
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Reduce learning rate
                lr *= 0.9
                stall = 0
            if lr < 1e-6:
                break
        
        if verbose and t % 100 == 0:
            print(f"    Adam: t={t}, obj={best_obj:.8f}, lr={lr:.6f}")
    
    return best_a

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    result = irfft(fa * fa, size)[:2 * n - 1]
    return result

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_soft_gradient(a, T=10.0):
    """
    Compute smooth approximation to gradient of max(conv(a,a)) using soft-max.

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

    """
    Solve LP: min M s.t. conv(a, z) <= M, sum(z)=1, z>=0.
    Returns z_normalized or None.
    """
    n = len(a)
    if n > 3000:  # Too large for LP
        return None
    
    A = build_conv_matrix(a)
    
    try:
        import cvxpy as cp
        z = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        prob = cp.Problem(cp.Minimize(M), [A @ z <= M, cp.sum(z) == 1])
        
        for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    z_val = np.maximum(z.value, 0)
                    zs = float(np.sum(z_val))
                    if zs > 1e-10:
                        return z_val / zs
            except:
                continue
    except:
        pass
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                return z_val / zs
    except:
        pass
    
    return None

def lp_polish(a, time_budget=100.0, verbose=False):
    """LP alternating polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        g_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Mix and check
        for mix in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - mix) * a + mix * g_lp
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break
        else:
            # No improvement, try small perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            a = best_a * np.exp(rng.normal(0, 0.01, n))

            a = a / np.sum(a)
            best_a = a.copy()
            best_obj = compute_obj(a)
        
        if verbose:
            print(f"    LP polish: obj={best_obj:.8f}")
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0).astype(np.float64)

def make_initialization(n, init_type='uniform', rng=None):
    """Create initial sequence of length n."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    if init_type == 'uniform':
        return np.ones(n) / n
    elif init_type == 'sine':
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    elif init_type == 'cosine':
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    elif init_type == 'gaussian':
        t = np.linspace(0, 1, n)
        center = 0.5
        sigma = rng.uniform(0.15, 0.3)
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    elif init_type == 'smoothed_random':
        a = rng.uniform(0.5, 1.5, n)
        # Smooth with a Gaussian kernel
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=max(1, n // 50))
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    elif init_type == 'sinusoidal_mod':
        t = np.linspace(0, 1, n)
        freq = rng.uniform(2, 6)
        amp = rng.uniform(0.1, 0.3)
        a = 1.0 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    elif init_type == 'raised_cosine':
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-6)
        return a / np.sum(a)
    else:
        return np.ones(n) / n

"""
Soft-Gradient Adam optimizer with LP polish for minimizing peak autocorrelation.
Key improvements over previous approach:
1. FFT-accelerated smooth soft-max gradient (much faster than LP for large n)
2. Adam optimizer with adaptive learning rate and momentum
3. Much larger n values (up to 5000) for better theoretical bound
4. Temperature annealing for soft-max (smooth to sharp)
5. Multi-start with diverse structured initializations
6. Periodic LP polish for exact subproblem solving
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    result = irfft(fa * fa, size)[:2 * n - 1]
    return result


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_soft_gradient(a, T=10.0):
    """
    Compute smooth approximation to gradient of max(conv(a,a)) using soft-max.
    Uses FFT for O(n log n) gradient computation.
    Returns (soft_obj, gradient_vector).
    """
    n = len(a)
    c = compute_conv(a)

    c_max = float(np.max(c))
    exp_c = np.exp(T * (c - c_max))
    sum_exp = float(np.sum(exp_c))
    s = exp_c / sum_exp

    soft_max_val = c_max + float(np.log(sum_exp)) / T
    obj = 2.0 * n * soft_max_val

    # Gradient: g_a[i] = conv(a, s_rev)[2n-2-i] + conv(a, s)[i]
    s_rev = s[::-1].copy()

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fs = rfft(s, size)
    fs_rev = rfft(s_rev, size)

    conv_as = irfft(fa * fs, size)
    conv_as_rev = irfft(fa * fs_rev, size)

    grad = conv_as_rev[n - 1:2 * n - 1][::-1] + conv_as[:n]

    return obj, grad

    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    start = time.monotonic()
    stall_count = 0
    lr = lr_start
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        t += 1
        elapsed_frac = min(1.0, (time.monotonic() - start) / time_budget)

        # Adaptive temperature: increase over time (smooth -> sharp)
        T = T_start + (T_end - T_start) * (elapsed_frac ** 0.7)
        # Adaptive LR: decrease over time
        lr = lr_start * (1.0 - 0.6 * elapsed_frac)

        soft_obj, grad = compute_soft_gradient(a, T)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 150:
                # Perturb to escape local minimum
                sigma = 0.015 + 0.01 * (stall_count // 100)
                a = best_a * np.exp(rng.normal(0, min(sigma, 0.05), n))
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr = lr_start * 0.5
                stall_count = 0
                best_obj = compute_obj(a)
                best_a = a.copy()

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    Adam-soft: t={t}, obj={best_obj:.8f}, T={T:.1f}, lr={lr:.5f}, elapsed={elapsed:.0f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a, z) <= M, sum(z)=1, z>=0.
    Returns z_normalized or None.
    """
    n = len(a)
    if n > 3000:
        return None

    A = build_conv_matrix(a)

    try:
        import cvxpy as cp
        z = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        prob = cp.Problem(cp.Minimize(M), [A @ z <= M, cp.sum(z) == 1])

        for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    z_val = np.maximum(z.value, 0)
                    zs = float(np.sum(z_val))
                    if zs > 1e-10:
                        return z_val / zs
            except Exception:
                continue
    except Exception:
        pass

    try:
        from scipy.optimize import linprog
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                return z_val / zs
    except Exception:
        pass

    return None


def lp_polish(a, time_budget=100.0, verbose=False):
    """LP alternating polish with cycle detection."""
    n = len(a)
    if n > 3000:
        return a

    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    prev_a = None
    stall = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        g_lp = solve_lp(a)
        if g_lp is None:
            break

        # 2-cycle detection
        if prev_a is not None and np.max(np.abs(a - prev_a)) < 1e-8:
            a_break = 0.5 * (a + prev_a)
            a_break = np.maximum(a_break, 0)
            s_b = float(np.sum(a_break))
            if s_b > 1e-10:
                a_break = a_break / s_b
                obj_b = compute_obj(a_break)
                if obj_b < best_obj:
                    best_obj = obj_b
                    best_a = a_break.copy()
                    a = a_break
            prev_a = None
            stall += 5
            continue

        improved = False
        for mix in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - mix) * a + mix * g_lp
            a_try = np.maximum(a_try, 0)
            s_t = float(np.sum(a_try))
            if s_t < 1e-10:
                continue
            a_try = a_try / s_t
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                prev_a = a.copy()
                a = a_try
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall > 10:
                a = best_a * np.exp(rng.normal(0, 0.015, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                stall = 0

        if verbose:
            print(f"    LP polish: obj={best_obj:.8f}, elapsed={time.monotonic()-start:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0], dtype=np.float64)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0).astype(np.float64)


def make_initialization(n, init_type, rng):
    """Create initial sequence of length n with given structure."""
    if init_type == 'uniform':
        return np.ones(n, dtype=np.float64) / n
    elif init_type == 'sine':
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-8)
        return a / np.sum(a)
    elif init_type == 'cosine':
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-8)
        return a / np.sum(a)
    elif init_type == 'gaussian':
        t = np.linspace(0, 1, n)
        center = 0.5
        sigma = rng.uniform(0.15, 0.3)
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        a = np.maximum(a, 1e-8)
        return a / np.sum(a)
    elif init_type == 'smoothed_random':
        a = rng.uniform(0.5, 1.5, n)
        # Simple moving average smoothing
        k = max(1, n // 50)
        kernel = np.ones(k) / k
        from scipy.signal import convolve as sig_conv
        a = sig_conv(a, kernel, mode='same')
        a = np.maximum(a, 1e-8)
        return a / np.sum(a)
    elif init_type == 'sinusoidal_mod':
        t = np.linspace(0, 1, n)
        freq = rng.uniform(2, 6)
        amp = rng.uniform(0.1, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * freq * t + phase)
        a = np.maximum(a, 1e-8)
        return a / np.sum(a)
    elif init_type == 'raised_cosine':
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-8)
        return a / np.sum(a)
    elif init_type == 'double_lobe':
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-8)
        return a / np.sum(a)
    else:
        return np.ones(n, dtype=np.float64) / n


def coordinate_polish(a, time_budget=30.0):
    """
    Coordinate-wise polish: for each coordinate i, find optimal value
    given all other coordinates fixed.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    coord_order = np.arange(n)
    improved = True

    while improved and time.monotonic() - start < time_budget:
        improved = False
        np.random.seed(int(time.monotonic() * 1e6) % (2**31))
        np.random.shuffle(coord_order)

        for i in coord_order:
            if time.monotonic() - start > time_budget:
                break

            # Try multiple values for a[i]
            old_val = a[i]
            rest_sum = 1.0 - old_val  # since sum(a) = 1

            # The objective as function of a[i] = x (with rest scaled to sum to 1-x):
            # We need to be careful: if we change a[i], we need to renormalize.
            # Simpler approach: try x in a range around old_val
           candidates = [old_val * 0.5, old_val * 0.8, old_val * 1.2, old_val * 1.5,
                        old_val + 0.001, old_val - 0.001]
            candidates = [max(0.0, c) for c in candidates]

            best_cand = old_val
            best_cand_obj = best_obj

            for x_val in candidates:
                if x_val < 0 or x_val == old_val:
                    continue
                a_try = a.copy()
                a_try[i] = x_val
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj - 1e-14:
                    best_cand_obj = obj_try
                    best_cand = x_val

            if best_cand != old_val:
                a[i] = best_cand
                s_a = float(np.sum(a))
                a = a / s_a
                if best_cand_obj < best_obj:
                    best_obj = best_cand_obj
                    best_a = a.copy()
                    improved = True

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining soft-gradient Adam optimization with LP polish.
    Explores multiple sequence lengths and initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Try to load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # === Phase 1: Quick multi-n, multi-init exploration (180s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    candidates = []

    if best_seq is not None:
        candidates.append(('parent', best_seq.copy(), best_score))

    n_values = [800, 1000, 1200, 1500, 2000, 3000, 4000]
    init_types = ['sine', 'cosine', 'smoothed_random', 'sinusoidal_mod', 'raised_cosine', 'double_lobe']

    for n in n_values:
        for init_type in init_types:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_init = make_initialization(n, init_type, rng)
            alloc = min(12.0, remaining * 0.08)
            if alloc < 5:
                continue
            a_opt = adam_soft_gradient(a_init, time_budget=alloc, T_start=5.0, T_end=40.0, lr_start=0.04)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** n={n}, {init_type}: obj={best_score:.8f}")

            if obj < 1.55:
                candidates.append((f"n{n}_{init_type}", a_opt.copy(), obj))

    # Sort candidates
    candidates.sort(key=lambda x: x[2])
    print(f"  Phase 1 done: best={best_score:.8f}, {len(candidates)} candidates")

    # === Phase 2: Extended Adam on top candidates (400s) ===
    print("\n=== Phase 2: Extended Adam ===")
    remaining = deadline - time.monotonic()

    # Ensure best is in candidates
    has_best = False
    for c in candidates:
        if len(c[1]) == len(best_seq) and np.allclose(c[1], best_seq, atol=1e-10):
            has_best = True
            break
    if not has_best:
        candidates.insert(0, ('best', best_seq.copy(), best_score))

    top_k = min(4, len(candidates))
    for idx in range(top_k):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        name, a_cand, obj_cand = candidates[idx]
        alloc = min(150.0, remaining * 0.35)
        if alloc < 20:
            alloc = 20

        a_opt = adam_soft_gradient(a_cand, time_budget=alloc, T_start=5.0, T_end=100.0,
                                   lr_start=0.025, verbose=False, seed=seed + idx)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** {name} (n={len(a_opt)}): obj={best_score:.8f}")
        else:
            print(f"  {name} (n={len(a_cand)}): obj={obj:.8f}")

    # === Phase 3: LP Polish (200s) ===
    print("\n=== Phase 3: LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 60 and len(best_seq) <= 2500:
        alloc = min(180.0, remaining * 0.3)
        a_opt = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** LP: obj={best_score:.8f}")
        else:
            print(f"  LP: obj={obj:.8f}")
    else:
        print("  Skipping LP (n too large or insufficient time)")

    # === Phase 4: Multi-n resampling (250s) ===
    print("\n=== Phase 4: Multi-n Resampling ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_targets = sorted(set([
        max(500, n_best - 200), max(500, n_best - 500),
        n_best + 200, n_best + 500, n_best + 1000, n_best + 2000, n_best + 3000,
        1000, 1500, 2000, 2500, 3000, 4000, 5000, 6000,
    ]))
    n_targets = [n for n in n_targets if 500 <= n <= 7000 and n != n_best]
    n_targets.sort(key=lambda x: abs(x - n_best))

    for n_try in n_targets[:12]:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(20.0, remaining * 0.15)

        a_opt = adam_soft_gradient(a_try, time_budget=alloc, T_start=5.0, T_end=60.0, lr_start=0.03)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj:.8f}")

    # === Phase 5: Final Adam refinement (200s) ===
    print("\n=== Phase 5: Final Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(180.0, remaining * 0.45)
        a_opt = adam_soft_gradient(best_seq, time_budget=alloc, T_start=10.0, T_end=150.0,
                                   lr_start=0.008, verbose=False, seed=seed + 100)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** Final Adam: obj={best_score:.8f}")
        else:
            print(f"  Final Adam: obj={obj:.8f}")

        # LP round if feasible
        remaining = deadline - time.monotonic()
        if remaining > 40 and len(best_seq) <= 2500:
            alloc = min(100.0, remaining * 0.4)
            a_opt = lp_polish(best_seq, time_budget=alloc)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** Final LP: obj={best_score:.8f}")

    # === Phase 6: Coordinate polish (60s) ===
    remaining = deadline - time.monotonic()
    if remaining > 20 and len(best_seq) <= 2000:
        print("\n=== Phase 6: Coordinate Polish ===")
        alloc = min(50.0, remaining * 0.3)
        a_opt = coordinate_polish(best_seq, time_budget=alloc)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** Coord: obj={best_score:.8f}")

    # === Phase 7: Random restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 7: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(5, int(remaining // 40))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            if r % 3 == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.03 + 0.01 * r, n_best))
            elif r % 3 == 1:
                # Block perturbation
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 100)
                block_size = n_best // n_blocks
                blocks = rng.choice(n_blocks, min(3, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = si + block_size
                    a_r[si:ei] *= rng.uniform(0.6, 1.4)
            else:
                # Sparse perturbation
                a_r = best_seq.copy()
                n_sparse = max(10, n_best // 20)
                idx = rng.choice(n_best, n_sparse, replace=False)
                a_r[idx] *= rng.uniform(0.4, 2.0, n_sparse)

            a_r = np.maximum(a_r, 0)
            s_r = float(np.sum(a_r))
            if s_r > 1e-10:
                a_r = a_r / s_r

            alloc = min(remaining * 0.3, 35.0)
            a_opt = adam_soft_gradient(a_r, time_budget=alloc, T_start=5.0, T_end=50.0,
                                       lr_start=0.02, seed=seed + 200 + r)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** Restart {r}: obj={best_score:.8f}")
            else:
                print(f"  Restart {r}: obj={obj:.8f}")

    # Final short Adam pass
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 20.0)
        a_opt = adam_soft_gradient(best_seq, time_budget=alloc, T_start=20.0, T_end=200.0,
                                   lr_start=0.003, seed=seed + 500)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Final pass: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(800, dtype=np.float64)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
