from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):

    n = len(a)

    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def fast_subgrad(a):
    """
    Fast subgradient of max(conv(a,a)) w.r.t. a.
    O(n * |active_set|) using the structure of autocorrelation.
    Returns (gradient, active_indices, max_val).
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(conv.max())
    
    # Find active set (indices at or near the max)
    tol = 1e-8 * max(1.0, max_val)
    active = np.where(conv >= max_val - tol)[0]
    
    # Compute subgradient
    g = np.zeros(n)
    for k in active:
        k = int(k)
        # g[i] += 2 * a[k-i] for valid i
        # i ranges from max(0, k-n+1) to min(k, n-1)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            # a[k-i] for i in [i_start, i_end]
            # k-i ranges from k-i_start to k-i_end (decreasing)
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end+1] += 2.0 * a[j_start:j_end+1][::-1]
    
    # Normalize by number of active constraints (for smooth subgradient)
    g /= len(active)
    
    return g, active, max_val

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build the convolution matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.

    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None

def smooth_logsumexp_optimize(a, time_budget=100.0, beta_start=50.0, beta_end=5000.0, verbose=False):
    """
    Smooth optimization using log-sum-exp surrogate with mirror descent.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    lr = 1.0 / np.sqrt(n)
    
    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / max(1, int(time_budget)))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        conv = compute_conv(a)
        m = conv.max()
        shifted = beta * (conv - m)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Gradient: g[i] = 2 * sum_m a[m] * w[i+m]
        # This is a cross-correlation
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-15:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end+1] += 2.0 * weights[k] * a[j_start:j_end+1][::-1]
        
        # Mirror descent
        log_a = np.log(a)
        log_a_new = log_a - lr * g
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            lr = max(1e-6, lr * 0.999)
        else:
            lr = max(1e-6, lr * 0.95)
        
        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      Smooth it{it}: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.1f}s")
    
    return best_a

def coordinate_descent_refine(a, time_budget=30.0):
    """
    Coordinate descent: locally optimize pairs of coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    max_rounds = 10
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        # Focus on the peak
        contributors = []
        for i in range(n):

            if 0 <= j < n:

        contributors.sort(reverse=True)
        
        non_peak_set = set()
        for _, i, j in contributors[:20]:
            non_peak_set.add(i)
            if i != j:
                non_peak_set.add(j)

        for contrib, i, j in contributors[:15]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break
            src = i if a[i] >= a[j] else j
            if src == i and src == j:
                src = i
            
            for target in candidates[:5]:
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[src]
                    if delta >= a[src] * 0.9:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
        
        if not improved:
            # Try random moves
            rng = np.random.default_rng(rnd * 1000 + n)
            for _ in range(20):
                if time.monotonic() - start > time_budget:
                    break
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Chirp
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(2 * np.pi * (2 + 3*x) * x)) + 0.01
        a = a / a.sum()
    elif mode == 3:
        # Gaussian mixture
        centers = rng.uniform(0.2, 0.8, size=3)
        widths = rng.uniform(0.05, 0.15, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.01
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Periodic
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * x) + 0.2 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 0.5 + (1-x) ** 0.5
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

import numpy as np

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def fast_subgrad_fft(a, active_weights=None):
    """
    FFT-accelerated subgradient of max(conv(a,a)) w.r.t. a.
    If active_weights is None, uses uniform weights on the exact maximizers.
    Returns (gradient, max_val, num_active).
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(conv.max())

    if active_weights is None:
        # Smooth subgradient: use all positions with soft weighting
        beta = 10.0 / max(1e-10, max_val)
        shifted = beta * (conv - max_val)
        active_weights = np.exp(shifted)
        active_weights /= active_weights.sum()
    else:
        active_weights = np.asarray(active_weights)

    # FFT-based cross-correlation: g[i] = 2 * sum_k w[k] * a[k-i]
    # Use: g[i] = 2 * conv(w, a_flip)[n-1+i]
    a_flip = a[::-1]
    size = 1
    while size < len(active_weights) + n - 1:
        size <<= 1
    fa = rfft(a_flip, size)
    fw = rfft(active_weights, size)
    c = irfft(fa * fw, size)
    g = 2.0 * c[n - 1: n - 1 + n]

    return g, max_val, int(np.sum(active_weights > 1e-10))


def smooth_subgrad(a, beta):
    """Smooth subgradient using log-sum-exp weights."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(conv.max())
    shifted = beta * (conv - max_val)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()

    # FFT cross-correlation
    a_flip = a[::-1]
    size = 1
    while size < len(w) + n - 1:
        size <<= 1
    fa = rfft(a_flip, size)
    fw = rfft(w, size)
    c = irfft(fa * fw, size)
    g = 2.0 * c[n - 1: n - 1 + n]

    return g


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_subgrad_optimize(a, time_budget=120.0, verbose=False):
    """
    Nesterov-accelerated smooth optimization using log-sum-exp surrogate.
    Adaptive beta schedule and Polyak-like step sizes.
    """

    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Lower bound: 2n/(2n-1) (from average of conv)
    lb = 2.0 * n / (2.0 * n - 1.0)

    # Nesterov state
    x = a.copy()
    y = a.copy()
    v = np.zeros(n)
    beta_mom = 0.95

    it = 0
    no_improve = 0
    max_no_improve = 300

    # Adaptive parameters
    beta_lse = 50.0
    lr = 1.0 / np.sqrt(n)

    while time.monotonic() - start < time_budget:
        # Adaptive beta: increase over iterations
        frac = min(1.0, it / 2000.0)
        beta_lse = 50.0 * (5000.0 / 50.0) ** frac

        # Compute smooth gradient at y (Nesterov point)
        g = smooth_subgrad(y, beta_lse)

        # Polyak step size
        current_val = 2.0 * n * float(np.max(compute_conv(y)))
        gap = current_val - lb
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq > 1e-20 and gap > 1e-12:
            polyak_lr = gap / g_norm_sq
        else:
            polyak_lr = lr
        step = min(polyak_lr, 2.0 / (np.sqrt(it + 1)))
        step = max(step, 1e-10)

        # Nesterov update
        x_new = project_simplex(y - step * g)
        v = beta_mom * v + (x_new - x)
        y_new = project_simplex(x_new + beta_mom * v)

        # Evaluate
        obj_new = compute_obj(y_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = y_new.copy()
            no_improve = 0
            beta_mom = min(0.99, beta_mom + 0.0005)
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                # Subgradient-guided restart
                g_curr, _, _ = fast_subgrad_fft(best_a)
                rng = np.random.default_rng(it * 104729 + n)
                # Mix subgradient direction with random perturbation
                g_norm = np.linalg.norm(g_curr)

                else:
                    g_dir = np.zeros(n)
                pert = 0.03 * rng.normal(0, 1, n) + 0.02 * g_dir
                y_new = np.maximum(best_a * (1.0 + pert), 1e-15)
                y_new = y_new / y_new.sum()
                best_a = y_new.copy()
                best_obj = compute_obj(best_a)
                x = best_a.copy()
                y = best_a.copy()
                v = np.zeros(n)
                beta_mom = 0.9
                no_improve = 0
            else:
                # Keep y at current x (no progress)
                y_new = x.copy()

        x = x_new if obj_new < best_obj + 0.01 else x
        y = y_new

        # Decay learning rate
        lr = max(1e-7, lr * 0.9995)

        it += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      Nesterov it{it}: obj={best_obj:.8f}, beta={beta_lse:.1f}, "
                  f"lr={lr:.6f}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build convolution matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses PDLP as primary solver.
    Returns (g_normalized, M_opt, solve_time).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    t0 = time.monotonic()
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    t1 = time.monotonic()
                    return g_val / s, float(M.value), t1 - t0
        except Exception:
            continue

    return None, None, time.monotonic() - t0

def coordinate_descent_refine(a, time_budget=30.0):
    """
    Coordinate descent: move mass from peak-contributing positions.
    """
    n = len(a)

    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    max_rounds = 8
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        improved = False

        # Identify top contributors to the peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        # Non-peak positions (targets for mass transfer)
        peak_set = set()
        for _, i, j in contributors[:20]:
            peak_set.add(i)
            if i != j:
                peak_set.add(j)
        candidates = [i for i in range(n) if i not in peak_set]

        if not candidates:
            break

        for contrib, i, j in contributors[:12]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-12:
                break

            if i != j:
                src = i if a[i] >= a[j] else j
            else:
                src = i

            for target in candidates[:6]:
                for frac in [0.01, 0.03, 0.05, 0.08]:
                    delta = frac * a[src]
                    if delta < 1e-15 or delta >= a[src] * 0.8:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = compute_conv(a)
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        # Random moves if stuck
        if not improved:
            rng = np.random.default_rng(rnd * 7919 + n)
            for _ in range(30):
                if time.monotonic() - start > time_budget:

                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:

                delta = 0.005 * a[i]
                if delta < 1e-15:

                a[i] -= delta
                a[j] += delta
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = compute_conv(a)
                    k_star = int(np.argmax(conv))
                    improved = True
                else:
                    a[i] += delta
                    a[j] -= delta
            if not improved:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

    elif mode == 3:
        x = np.linspace(0, 1, n)
        centers = [0.25, 0.5, 0.75]
        a = np.ones(n) * 0.05
        for c in centers:
            a += np.exp(-((x - c) ** 2) / (2 * 0.08 ** 2))
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 7 * x) + 0.2 * np.cos(2 * np.pi * 3 * x + 1)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0.01, 1, n)
        a = x ** 0.3 + (1 - x) ** 0.3 + 0.05 * np.sin(20 * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Nesterov subgradient warm-up, LP alternating with over-relaxation,
    multi-dimension exploration, diverse restarts, and coordinate descent polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"\nInitial best: n={n_current}, obj={best_score:.8f}")

    # === Phase 1: Nesterov subgradient warm-up ===
    print(f"\n=== Phase 1: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 100.0)
    if p1_budget > 20:
        a_ns = nesterov_subgrad_optimize(best_seq, time_budget=p1_budget, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            n_current = len(best_seq)
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_ns:.8f} (no improvement)")

    # === Phase 2: LP Over-Relaxation (main workhorse) ===
    print(f"\n=== Phase 2: LP Over-Relaxation ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.5, 700.0)

    for rnd in range(3):
        remaining = deadline - time.monotonic()
        alloc = min(p2_budget / (3 - rnd), remaining * 0.5)
        if alloc < 30:
            break

        if rnd > 0:
            rng_r = np.random.default_rng(rnd * 48271 + n_current)
            a_start = best_seq * (1.0 + 0.02 * rng_r.normal(0, 1, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        else:
            a_start = best_seq.copy()

        a_lp = lp_overrelax_optimize(a_start, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            n_current = len(best_seq)
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Multi-dimension exploration ===
    print(f"\n=== Phase 3: Multi-Dimension ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 50, n_best + 50, n_best + 100, n_best + 200,
        600, 700, 800, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and n != n_best]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 80.0)

        if alloc > 30:
            a_try = nesterov_subgrad_optimize(a_try, time_budget=alloc * 0.3)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_try = lp_overrelax_optimize(a_try, time_budget=min(alloc * 0.5, remaining * 0.3))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = len(best_seq)
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Diverse restarts ===
    print(f"\n=== Phase 4: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r + 2) % 6
        a_r = diverse_init(n_use, rng, mode=mode)
        blend = 0.5 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.12, 60.0)
        if restart_budget > 25:
            a_r = nesterov_subgrad_optimize(a_r, time_budget=restart_budget * 0.4)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_r = lp_overrelax_optimize(a_r, time_budget=min(restart_budget * 0.5, remaining * 0.3))

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            n_use = len(best_seq)
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Nesterov polish
    if remaining > 25:
        ns_alloc = min(remaining * 0.15, 50.0)
        a_ns = nesterov_subgrad_optimize(best_seq, time_budget=ns_alloc)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov polish: obj={best_score:.8f} **")

    # LP polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        lp_alloc = min(remaining * 0.35, 120.0)
        a_lp = lp_overrelax_optimize(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP polish: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd_alloc = min(remaining * 0.25, 40.0)
        a_cd = coordinate_descent_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD polish: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_overrelax_optimize(best_seq, time_budget=min(remaining * 0.5, 50.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # Final Nesterov
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_fin = nesterov_subgrad_optimize(best_seq, time_budget=min(remaining * 0.5, 30.0))
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_score:
            best_score = obj_fin
            best_seq = a_fin.copy()
            print(f"  Final Nesterov: obj={best_score:.8f} **")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_cdf = coordinate_descent_refine(best_seq, time_budget=min(remaining * 0.5, 20.0))
        obj_cdf = compute_obj(a_cdf)
        if obj_cdf < best_score:
            best_score = obj_cdf
            best_seq = a_cdf.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
