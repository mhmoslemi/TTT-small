def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""

    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A

def subgradient_polish(a, time_budget=100.0, verbose=False):
    """
    Nesterov-accelerated subgradient method targeting active peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()  # Nesterov lookahead
    y = a.copy()  # actual point
    beta = 0.7
    lr = 0.05
    momentum = np.zeros(n)

    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break

        # Compute convolution at lookahead point
        conv = compute_conv(x)
        max_val = float(np.max(conv))
        # Get active peaks (within 10% of max)
        threshold = max_val * 0.9
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = [int(np.argmax(conv))]
        if len(active) > 30:
            active = np.argsort(conv)[::-1][:30]

        # Compute subgradient: average of gradients of active peaks
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            # d(a*a)[k]/da[i] = a[k-i] if 0 <= k-i < n
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            if i_max >= i_min:
                # grad[i] += a[k-i] for i in [i_min, i_max]
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += a[k - indices]
        grad /= len(active)

        # Center gradient (project to hyperplane sum=0)
        grad_centered = grad - np.mean(grad)

        # Momentum step
        momentum = beta * momentum - lr * grad_centered
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new

        # Nesterov update
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        y = y_new
        obj = compute_obj(y)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()

        # Adaptive learning rate
        if it % 200 == 199:
            lr = max(lr * 0.9, 0.001)

    return best_a


def coordinate_adjust(a, time_budget=60.0):
    """
    Pairwise coordinate adjustment: move mass between pairs to reduce peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    for iteration in range(500):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        improved = False

        # Identify positions contributing most to the peak
        # (a*a)[k_star] = sum_j a[j]*a[k_star-j]
        contributions = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contributions[j] = a[j] * a[k_j]

        # Sort by contribution (highest first)
        sorted_idx = np.argsort(contributions)[::-1]

        # Try moving mass from high-contributors to low-contributors
        for frac in [0.005, 0.01, 0.02, 0.05]:
            done = False
            for i in sorted_idx[:20]:
                if done or time.monotonic() - start > time_budget:
                    break
                # Find a position j that contributes least and is "far" from k_star-i
                target = (k_star - int(i) + n // 2) % n
                if target == i:
                    target = (i + n // 2) % n
                delta = frac * a[i]
                if delta <= 0 or delta > a[i]:
                    continue
                a[i] -= delta
                a[target] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    done = True
                else:
                    a[i] += delta
                    a[target] -= delta

            if done:
                improved = True

        if not improved:
            # Random pair swap
            rng = np.random.default_rng(int(time.monotonic() * 1e4) % (2**31))
            for _ in range(5):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.01 * min(a[i], a[j])
                if delta <= 0:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                a = a / s
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and iteration > 100:
            break

    return best_a


def generate_initialization(kind, n, rng):
    """Generate a diverse initialization of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.1
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'bilinear':
        # Linear ramp (asymmetric)
        a = np.linspace(1, 2, n)
        return a / np.sum(a)
    elif kind == 'plateau':
        # High in middle 60%, low at edges
        a = np.ones(n)
        edge = max(1, n // 10)
        a[:edge] = 0.3
        a[-edge:] = 0.3
        return a / np.sum(a)
    elif kind == 'periodic':
        # Two peaks
        a = np.ones(n) * 0.5
        a[n//4] = 2.0
        a[3*n//4] = 2.0
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
