"""
Optimized step function search using Adam optimizer on log-sum-exp smoothed
convolution objective with softmax parameterization, LP alternating refinement,
and multi-dimension exploration.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def cross_correlation_fft(a, w):
    """
    Compute c[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i=0..n-1.
    w has length 2n-1 (the convolution values).
    Uses the identity: c[i] = conv(flip(a), w)[n-1+i]
    """
    n = len(a)
    L = len(w)
    size = 1
    while size < n + L - 1:
        size <<= 1
    af = a[::-1]
    result = irfft(rfft(af, size) * rfft(w, size), size)
    return result[n - 1:n - 1 + n]


def adam_optimize(a, time_budget, beta_start=10.0, beta_end=300.0, lr=0.01, seed=0):
    """
    Adam optimization on log-sum-exp smoothed objective with softmax parameterization.
    a: initial sequence (will be normalized internally)
    Returns best sequence found (normalized to sum=1) and its objective.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    x = np.log(a) - np.max(np.log(a))  # log-parameterization for stability

    m_x = np.zeros(n)
    v_x = np.zeros(n)
    t = 0
    best_a_local = a.copy()
    best_obj_local = compute_obj_normalized(a)
    start = time.monotonic()
    no_improve_count = 0
    patience = 500

    while time.monotonic() - start < time_budget:
        t += 1
        frac = min(1.0, t / max(1, int(time_budget * 2)))
        beta = beta_start * (beta_end / beta_start) ** frac

        # Compute a from softmax of x
        x_shifted = x - np.max(x)
        a = np.exp(x_shifted)
        a = a / a.sum()

        # Compute convolution
        conv = compute_conv(a)  # length 2n-1

        # Log-sum-exp and softmax weights
        m = np.max(conv)
        shifted = beta * (conv - m)
        shifted -= np.max(shifted)  # extra stability
        exp_shifted = np.exp(shifted)
        weights = exp_shifted / exp_shifted.sum()

        # Gradient w.r.t. a: grad_a[i] = (4*n/beta) * sum_j a[j] * weights[i+j]
        cross = cross_correlation_fft(a, weights)
        grad_a = (4.0 * n / beta) * cross

        # Backprop through softmax: grad_x = a * (grad_a - dot(a, grad_a))
        grad_x = a * (grad_a - np.dot(a, grad_a))

        # Adam update
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        m_x = beta1 * m_x + (1 - beta1) * grad_x
        v_x = beta2 * v_x + (1 - beta2) * grad_x ** 2
        m_hat = m_x / (1 - beta1 ** t)
        v_hat = v_x / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate
        a_new = np.exp(x - np.max(x))
        a_new = a_new / a_new.sum()
        obj_new = compute_obj_normalized(a_new)

        if obj_new < best_obj_local - 1e-12:
            best_obj_local = obj_new
            best_a_local = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Adaptive learning rate
        if no_improve_count > patience:
            # Reset with perturbation

            x += 0.05 * rng.normal(0, 1, n)
            m_x = np.zeros(n)
            v_x = np.zeros(n)
            t = 0
            lr = max(1e-5, lr * 0.5)
            no_improve_count = 0
        elif t % 200 == 0:
            lr = max(1e-5, min(0.1, lr * 0.98))

    return best_a_local, best_obj_local


def lp_refine(a, time_budget):
    """
    LP alternating method: solve min M s.t. conv(a,g)[k]<=M, sum(g)=1, g>=0.
    Then blend with current a via line search.
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a_local = a.copy()
    best_obj_local = compute_obj_normalized(a)
    start = time.monotonic()

    # Pre-build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]

    stall = 0
    last_good_t = 0.3

    while time.monotonic() - start < time_budget:
        # Solve LP
        try:
            from scipy.optimize import linprog
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.zeros((2 * n - 1, n + 1))
            A_ub[:, :n] = A
            A_ub[:, n] = -1.0
            b_ub = np.zeros(2 * n - 1)
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(None, None)]
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
            if res.status != 0:
                break
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs < 1e-10:
                break
            g_val = g_val / gs
        except Exception:
            break

        # Line search
        candidate_ts = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]
        if 0 < last_good_t < 1:
            candidate_ts.append(max(0.02, last_good_t * 0.8))
            candidate_ts.append(min(1.0, last_good_t * 1.2))
        candidate_ts = sorted(set(candidate_ts))

        best_t_obj = np.inf
        best_t = 0
        for t_val in candidate_ts:
            a_try = (1 - t_val) * a + t_val * g_val
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t_val

        if best_t_obj < best_obj_local - 1e-12:
            best_obj_local = best_t_obj
            a = (1 - best_t) * a + best_t * g_val
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            else:
                a = best_a_local.copy()
                break
            best_a_local = a.copy()
            last_good_t = best_t
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break

        # Update A matrix for next iteration
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            r = min(n, 2 * n - 1 - j)
            A[j:j + r, j] = a[:r]

    return best_a_local, best_obj_local


def peak_reduction_polish(a, time_budget):
    """
    Direct peak reduction: iteratively move mass from peak-contributing positions

    """

    if s < 1e-10:

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_val = conv[k_star]

        # Find top contributors to peak
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(key=lambda x: -x[0])

        # Find good targets (low contribution to peak)
        all_pos = list(range(n))

        improved = False
        for contrib, i, j in contribs[:20]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break
            src = i if a[i] >= a[j] else j
            if src == src:  # always true
                # Try moving to various targets
                for target in all_pos:
                    if target == src:
                        continue
                    # Check if target worsens other peaks
                    for frac in [0.02, 0.05, 0.1]:
                        delta = frac * a[src]
                        if delta <= 0:
                            continue
                        a[src] -= delta
                        a[target] += delta
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))

                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[target] -= delta
                            break
                if improved:
                    break

        if not improved:
            break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        a_div = a_div / a_div.sum()
        # Blend with current best
        a_div = 0.5 * a_div + 0.5 * best_seq
        a_div = a_div / a_div.sum()

        alloc = min(remaining * 0.3, 80.0)
        a_div, obj_div = adam_optimize(a_div, time_budget=alloc,
                                        beta_start=20, beta_end=300, lr=0.01, seed=seed + 777)
        if obj_div < best_score:
            best_score = obj_div
            best_seq = a_div.copy()
            print(f"  Diverse restart: obj={best_score:.8f} **")
        else:
            print(f"  Diverse restart: obj={obj_div:.8f}")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = 2.0

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val != np.inf and val < np.inf:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
