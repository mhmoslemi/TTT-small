"""
Smooth max (log-sum-exp) optimization with Adam + multiplicative updates,
combined with LP polishing and multi-scale length sweep.

Key innovation over previous approach: uses a smooth approximation of the max
function (log-sum-exp) to enable gradient-based optimization with adaptive
per-coordinate learning rates (Adam). The epsilon parameter provides a
"zoom level" that transitions from considering all peaks (small eps) to
focusing on the maximum (large eps), enabling escape from local minima that
trap the pure LP-alternating method.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv / (sum)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_raw(a):
    """Compute objective without normalization (for internal use)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_alternate_step(a, max_iters=3):
    """
    Run a few LP alternating iterations.
    Each iteration: solve LP for best response g, then interpolate.
    Returns improved sequence.
    """
    from scipy.optimize import linprog
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(max_iters):
        A = build_conv_matrix(a)
        # LP: min M s.t. A@g <= M*ones, sum(g)=1, g>=0
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 30})
            if res.status != 0:
                break
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs < 1e-10:
                break
            g_val = g_val / gs
        except Exception:
            break

        # Try different step sizes
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_try = ((1 - t) * best_a + t * g_val)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-15:
                best_obj = obj_try
                best_a = a_try.copy()
                improved = True
                break

        if not improved:
            break

    return best_a


def smooth_max_adam(a, time_budget=300.0, eps_start=80.0, eps_end=8.0,
                    lr_init=0.3, verbose=False, anneal_interval=5000):
    """
    Optimize using smooth max (log-sum-exp) + Adam with multiplicative updates.
    
    The key idea: approximate max_k conv_k by (1/eps)*log(sum exp(eps*conv_k)),
    compute its gradient via cross-correlation of softmax weights with a,
    and use Adam with multiplicative (log-space) updates for stability.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Work in log-space: b = log(a + tiny)
    tiny = 1e-30
    b = np.log(a + tiny)

    # Adam state
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t = 0
    lr = lr_init

    # Epsilon schedule: exponential decay from eps_start to eps_end
    eps_decay = (eps_end / eps_start) ** (1.0 / max(anneal_interval, 1))

    while time.monotonic() - start < time_budget:
        t += 1
        eps = eps_start * (eps_decay ** t)
        eps = max(eps, eps_end)

        # Compute convolution
        conv = compute_conv(a)

        # Compute softmax weights (numerically stable)
        conv_max = float(np.max(conv))
        exp_conv = np.exp(eps * (conv - conv_max))
        w = exp_conv / exp_conv.sum()

        # Compute gradient: grad[i] = 4n * sum_{m=0}^{n-1} w[i+m] * a[m]
        # This is 4n * correlate(w, a, mode='valid')
        grad_raw = 4.0 * n * np.correlate(w, a, mode='valid')

        # Project to sum=0 hyperplane (for the constraint sum(a)=1)
        grad = grad_raw - np.mean(grad_raw)

        # Adam update
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad * grad
        m_hat_corr = m_hat / (1.0 - beta1 ** t)
        v_hat_corr = v_hat / (1.0 - beta2 ** t)

        # Multiplicative (log-space) update
        b = b - lr * m_hat_corr / (np.sqrt(v_hat_corr) + eps_adam)

        # Convert back and normalize
        a_new = np.exp(b)
        # Clamp to avoid overflow
        a_new = np.minimum(a_new, 1e6)
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            # Reset to uniform if we blew up
            a_new = np.ones(n) / n
            b = np.log(a_new + tiny)
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            continue
        a_new = a_new / s_new

        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        b = np.log(a + tiny)

        # Adaptive learning rate: decay slowly
        if t % 2000 == 0:
            lr = max(lr * 0.95, 0.01)

        # Periodic re-evaluation with exact max (not soft)
        if t % 1000 == 0:
            exact_obj = compute_obj(a)
            if exact_obj < best_obj - 1e-15:
                best_obj = exact_obj
                best_a = a.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SMA: t={t}, eps={eps:.1f}, lr={lr:.4f}, "
                  f"obj={best_obj:.8f}, time={elapsed:.0f}s")

    return best_a


def smooth_max_subgrad(a, time_budget=200.0, verbose=False):
    """
    Subgradient method targeting the active peak(s), with Nesterov acceleration.
    Different from smooth_max_adam: uses exact max, not log-sum-exp.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()  # Nesterov lookahead
    x = a.copy()  # current point
    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.08

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # Compute convolution at lookahead
        conv = compute_conv(y)
        max_val = float(np.max(conv))

        # Active set: indices within 15% of max
        threshold = max_val * 0.85
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 50:
            active = np.argsort(conv)[::-1][:50]

        # Subgradient: average of gradients of active peaks
        # grad[i] = (1/|A|) * sum_{k in A} 2*a[k-i] for valid k-i
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            # For each k, grad[i] += 2*a[k-i] for i where 0 <= k-i < n
            # i ranges from max(0, k-n+1) to min(n-1, k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += 2.0 * a[k - indices]
        grad /= len(active)

        # Center
        grad = grad - np.mean(grad)

        # Nesterov + momentum
        momentum = beta * momentum - lr * grad
        x_new = x + momentum
        x_new = np.maximum(x_new, 0)
        s_new = float(np.sum(x_new))
        if s_new < 1e-10:
            x_new = x.copy()
            s_new = 1.0
        x_new = x_new / s_new

        # Nesterov update
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        y = x_new + (1 - t_k) * (x_new - x)
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy > 1e-10:
            y = y / sy

        x = x_new
        obj = compute_obj(x)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = x.copy()

        # Adaptive lr
        if it % 3000 == 2999:
            lr = max(lr * 0.92, 0.005)

        if verbose and it % 10000 == 9999:
            elapsed = time.monotonic() - start
            print(f"    SubGrad: it={it}, obj={best_obj:.8f}, lr={lr:.4f}, time={elapsed:.0f}s")

    return best_a


def generate_initialization(kind, n, rng):
    """Generate diverse initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.05
        return a / np.sum(a)
    elif kind == 'bilinear':
        a = np.linspace(0.5, 1.5, n)
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 8)
        a[:edge] = 0.3
        a[-edge:] = 0.3
        return a / np.sum(a)
    elif kind == 'two_peaks':
        a = np.ones(n) * 0.5
        a[n//4] = 3.0
        a[3*n//4] = 3.0
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'lopsided':
        # Asymmetric: more mass on one side
        a = np.exp(-np.linspace(0, 3, n))
        return a / np.sum(a)
    elif kind == 'oscillating':
        # Try an oscillating pattern
        x = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x / n)
        a = np.maximum(a, 0.1)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] * n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  '{kind}': obj={obj_init:.8f} **")
        else:
            print(f"  '{kind}': obj={obj_init:.8f}")

    # A couple random inits
    for i in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        a_init = generate_initialization('random', n_parent, rng)
        obj_init = compute_obj(a_init)
        all_candidates.append((obj_init, a_init.copy(), f"rand_{i}"))
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Random {i}: obj={obj_init:.8f} **")

    # Sort candidates by objective
    all_candidates.sort(key=lambda x: x[0])
    print(f"  Best after Phase 1: obj={best_score:.8f}")

    # === Phase 2: Main smooth max optimization from best (500s) ===
    print(f"\n=== Phase 2: Smooth Max + Adam (main) ===")
    remaining = deadline - time.monotonic()
    main_budget = min(500.0, remaining * 0.35)

    a_current = best_seq.copy()

    # Multiple rounds with different epsilon schedules
    eps_configs = [
        (100.0, 10.0, 0.4),   # Start sharp, end moderate
        (60.0, 5.0, 0.5),     # Start moderate, end flat
        (200.0, 20.0, 0.3),   # Very sharp start
        (40.0, 3.0, 0.6),     # Moderate start, very flat end
    ]

    for round_idx, (eps_s, eps_e, lr_c) in enumerate(eps_configs):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(main_budget / (len(eps_configs) - round_idx), remaining * 0.7)
        print(f"  Round {round_idx}: eps=[{eps_s}->{eps_e}], lr={lr_c}")

        a_new = smooth_max_adam(a_current, time_budget=alloc,
                                eps_start=eps_s, eps_end=eps_e,
                                lr_init=lr_c, verbose=True)
        obj_new = compute_obj(a_new)
        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            a_current = a_new.copy()
            print(f"  ** Round {round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  Round {round_idx}: obj={obj_new:.8f}")
            # Try from a slightly perturbed version
            pert = best_seq * np.exp(rng.normal(0, 0.02, n_parent))
            pert = np.maximum(pert, 0)
            pert = pert / np.sum(pert)
            a_current = pert

    # === Phase 3: Subgradient polish (150s) ===
    print(f"\n=== Phase 3: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        sg_budget = min(150.0, remaining * 0.12)
        a_sg = smooth_max_subgrad(best_seq, time_budget=sg_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score - 1e-12:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  ** SubGrad: obj={best_score:.8f} **")
        else:
            print(f"  SubGrad: obj={obj_sg:.8f}")

    # === Phase 4: LP polishing (150s) ===
    print(f"\n=== Phase 4: LP Polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        lp_budget = min(150.0, remaining * 0.12)
        n_iter = max(3, int(lp_budget / 15))
        a_lp = a_current.copy()
        for i in range(n_iter):
            if time.monotonic() > deadline - 100:
                break
            a_lp = lp_alternate_step(a_lp, max_iters=5)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score - 1e-12:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  ** LP iter {i}: obj={best_score:.8f} **")
            else:
                if i > 2:
                    break
        # Also try LP from best_seq directly
        a_lp2 = best_seq.copy()
        a_lp2 = lp_alternate_step(a_lp2, max_iters=3)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score - 1e-12:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  ** LP direct: obj={best_score:.8f} **")

    # === Phase 5: Length sweep (250s) ===
    print(f"\n=== Phase 5: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Try a range of lengths around the current best
    n_options = sorted(set([
        max(100, n_best - 200),
        max(100, n_best - 100),
        max(100, n_best - 50),
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 400,
        n_best + 800,
    ]))
    # Also try some very different lengths
    n_options = sorted(set(n_options + [200, 400, 1200, 2000, 3000]))
    n_options = [n for n in n_options if 100 <= n <= 4000]
    # Remove duplicates and sort
    n_options = sorted(set(n_options))

    # Try closest to current best first
    n_options.sort(key=lambda x: abs(x - n_best))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        if n_try == n_best:
            continue  # Already optimized at this length

        a_try = resample_sequence(best_seq, n_try)
        print(f"  n={n_try}: initiating optimization...")

        # Quick optimize: smooth max + LP
        alloc_sma = min(20.0, remaining * 0.3)
        alloc_lp = min(10.0, remaining * 0.1)

        a_opt = smooth_max_adam(a_try, time_budget=alloc_sma,
                                eps_start=50.0, eps_end=20.0, lr_init=0.3)
        obj_sma = compute_obj(a_opt)

        a_opt = lp_alternate_step(a_opt, max_iters=3)
        obj_final = compute_obj(a_opt)

        if obj_final < best_score - 1e-12:
            best_score = obj_final
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_final:.8f}")

    # === Phase 6: Re-optimize at best length with remaining time (200s) ===
    print(f"\n=== Phase 6: Final Optimization at Best Length ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    a_final = best_seq.copy()

    if remaining > 50:
        final_budget = min(200.0, remaining * 0.5)

        # Multiple short rounds
        for round_idx in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(final_budget / (6 - round_idx), remaining * 0.6)

            # Alternate between smooth max and LP
            if round_idx % 2 == 0:
                a_new = smooth_max_adam(a_final, time_budget=alloc,
                                        eps_start=30.0, eps_end=10.0,
                                        lr_init=0.2, verbose=False)
            else:
                a_new = lp_alternate_step(a_final, max_iters=max(2, int(alloc / 10)))

            obj_new = compute_obj(a_new)
            if obj_new < best_score - 1e-12:
                best_score = obj_new
                best_seq = a_new.copy()
                a_final = a_new.copy()
                print(f"  ** Final R{round_idx}: obj={best_score:.8f} **")
            else:
                # Perturb to escape
                scale = 0.01 + 0.005 * round_idx
                a_final = best_seq * np.exp(rng.normal(0, scale, n_final))
                a_final = np.maximum(a_final, 0)
                a_final = a_final / np.sum(a_final)
                print(f"  Final R{round_idx}: obj={obj_new:.8f} (perturbing)")

    # === Phase 7: Try a few completely different starting points ===
    print(f"\n=== Phase 7: Diversification ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Try from the second-best initialization
        for (obj_c, a_c, name) in all_candidates[1:4]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(30.0, remaining * 0.3)
            a_opt = smooth_max_adam(a_c, time_budget=alloc,
                                    eps_start=50.0, eps_end=15.0, lr_init=0.4)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score - 1e-12:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** From '{name}': obj={best_score:.8f} **")

    # === Phase 8: Very final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_polish = fast_local_polish(best_seq, time_budget=min(remaining * 0.6, 40.0))
        obj_polish = compute_obj(a_polish)
        if obj_polish < best_score - 1e-12:
            best_score = obj_polish
            best_seq = a_polish.copy()
            print(f"  ** Final polish: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
