"""
Step function optimization: minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth mirror descent, LP direction with golden-section search,
subgradient exploitation, peak shaving, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    p = a / s
    conv = compute_conv(p)
    return 2.0 * n * float(np.max(conv))


def smooth_max_grad(a, beta):
    """Gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    m = conv.max()
    shifted = beta * (conv - m)
    shifted -= shifted.max()
    weights = np.exp(shifted)
    weights /= weights.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-15:
            continue
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        j_start = k - i_end
        j_end = k - i_start
        grad[i_start:i_end + 1] += 2.0 * weights[k] * a[j_start:j_end + 1]
    return grad


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a_norm):
    """Build convolution matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a_norm)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    return A


def solve_lp_direction(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def golden_section_search(a, direction, obj_a, n_evals=20):
    """Golden section line search on t in [0,1] for (1-t)*a + t*direction."""
    gr = (np.sqrt(5) - 1) / 2
    t_lo, t_hi = 0.0, 1.0
    t1 = t_hi - gr * (t_hi - t_lo)
    t2 = t_lo + gr * (t_hi - t_lo)

    def eval_at(t):
        p = (1 - t) * a + t * direction
        p = np.maximum(p, 0)
        s = p.sum()
        if s < 1e-10:
            return np.inf
        p = p / s
        return compute_obj(p)

    f1 = eval_at(t1)
    f2 = eval_at(t2)
    best_t, best_f = t1, f1
    if f2 < f1:
        best_t, best_f = t2, f2

    for _ in range(n_evals):
        if f1 < f2:
            t_hi = t2
            t2 = t1
            f2 = f1
            t1 = t_hi - gr * (t_hi - t_lo)
            f1 = eval_at(t1)
        else:
            t_lo = t1
            t1 = t2
            f1 = f2
            t2 = t_lo + gr * (t_hi - t_lo)
            f2 = eval_at(t2)
        if f1 < best_f:
            best_t, best_f = t1, f1
        if f2 < best_f:
            best_t, best_f = t2, f2

    p = (1 - best_t) * a + best_t * direction
    p = np.maximum(p, 0)
    p = p / p.sum()
    return p, best_f


def smooth_mirror_optimize(a, time_budget=200.0, beta_start=20.0, beta_end=5000.0):
    """Mirror descent with log-sum-exp smoothing and adaptive beta."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    lr = 1.0 / np.sqrt(n)
    momentum = 0.0
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / max(1, int(time_budget)))
        beta = beta_start * (beta_end / beta_start) ** (frac ** 0.7)

        grad = smooth_max_grad(a, beta)

        # Momentum
        v = 0.9 * v + grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(10):
            log_a = np.log(np.maximum(a, 1e-30))
            log_a_new = log_a - step * v
            log_a_new -= log_a_new.max()
            a_new = np.exp(log_a_new)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                momentum = min(0.85, momentum + 0.003)
                break
            step *= 0.5

        if not improved:
            # Try smaller momentum
            v *= 0.5
            if np.linalg.norm(v) < 1e-10:
                # Restart
                rng = np.random.default_rng(it * 31 + n)
                noise = 0.02 * rng.normal(0, 1, n)
                a = best_a * (1.0 + noise)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                v = np.zeros(n)
                lr = max(0.01, lr * 0.7)

        lr = max(1e-6, lr * 0.999)
        it += 1

    return best_a


def subgradient_optimize(a, time_budget=200.0):

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        if max_val < 1e-15:
            break

        # Softmax weights over top peaks
        top_k = max(3, min(10, n // 10))
        sorted_idx = np.argsort(conv)[::-1]
        top_idx = sorted_idx[:top_k]
        top_vals = conv[top_idx]

        # Beta based on gap between top peaks
        gap = top_vals[0] - top_vals[-1] + 1e-10
        beta = 2.0 / max(gap, 1e-10)
        beta = min(beta, 200.0)

        weights = np.exp(beta * (top_vals - top_vals[0]))
        weights /= weights.sum()

        # Compute gradient: 2 * sum_k w_k * a[k-i]
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            j_start = k - i_end
            j_end = k - i_start
            grad[i_start:i_end + 1] += 2.0 * w * a[j_start:j_end + 1]

        # Mirror descent step with backtracking
        step = lr
        improved = False
        for _ in range(8):
            log_a = np.log(np.maximum(a, 1e-30))
            log_a_new = log_a - step * grad
            log_a_new -= log_a_new.max()
            a_new = np.exp(log_a_new)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            lr = max(1e-5, lr * 0.8)
            # Periodic perturbation
            if it % 50 == 49:
                rng = np.random.default_rng(it * 7 + n)
                a = best_a * np.exp(0.03 * rng.normal(0, 1, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
        else:
            lr = min(0.3, lr * 1.01)

        it += 1

    return best_a


def lp_anneal_optimize(a, time_budget=400.0):
    """LP alternating with golden section search and annealing."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    stall = 0
    patience = 25
    temperature = 0.08

    while time.monotonic() - start < time_budget:
        # Get LP direction
        direction, m_lp = solve_lp_direction(a)
        if direction is None:
            it += 1
            if it % 100 == 0:
                temperature = max(0.01, temperature * 0.9)
            continue

        # Check if direction is useful
        current_peak = compute_obj(a) / (2.0 * n)
        if m_lp >= current_peak * 0.999:
            stall += 2
        else:
            stall += 0

        # Golden section line search
        a_new, obj_new = golden_section_search(a, direction, best_obj, n_evals=15)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            temperature = max(0.01, temperature * 0.99)
        else:
            stall += 1
            if stall > patience:
                # Annealing restart
                rng = np.random.default_rng(it * 12345 + n)
                scale = temperature
                a = best_a * (1.0 + scale * rng.normal(0, 1, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()

                # Also try asymmetric perturbation
                if rng.random() < 0.4:
                    # Skew: boost one side
                    skew = rng.uniform(-0.3, 0.3)
                    x = np.linspace(0, 1, n)
                    a = a * (1.0 + skew * (x - 0.5))
                    a = np.maximum(a, 1e-15)
                    a = a / a.sum()

                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                temperature = min(0.25, temperature * 1.8)

        it += 1

    return best_a


def peak_shave(a, time_budget=150.0):
    """Direct peak shaving: move mass from peak contributors to non-contributors."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for _ in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        top_peaks = np.argsort(conv)[::-1][:5]

        # Find contributors to the top peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        # Find non-contributors (positions not contributing to any top peak)
        contrib_set = set()
        for k in top_peaks:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contrib_set.add(i)

        non_contrib = [i for i in range(n) if i not in contrib_set]
        if not non_contrib:
            non_contrib = [i for i in range(n) if a[i] < np.median(a)]

        improved = False
        for contrib, i, j in contributors[:20]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break

            src = i if a[i] >= a[j] else j if i != j else i
            for target in non_contrib[:10]:
                if target == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                    delta = frac * a[src]
                    if delta < 1e-12 or delta > a[src] * 0.5:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    obj_new = compute_obj(a)
                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
                if improved:
                    break

        if not improved:
            # Try smaller fracs or break
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-15)


def diverse_asymmetric_init(n, rng, mode=0):
    """Generate diverse asymmetric initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Skewed exponential
        a = np.exp(-2.0 * x) + 0.1
    elif mode == 1:
        # Power law
        a = x**2 + 0.05
    elif mode == 2:
        # Asymmetric Gaussian
        a = np.exp(-((x - 0.3) ** 2) / 0.05) + 0.05 * np.exp(-((x - 0.8) ** 2) / 0.02)
    elif mode == 3:
        # Cosine skew
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.sin(2 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif mode == 4:
        # Two humps asymmetric
        a = np.exp(-((x - 0.2) ** 2) / 0.04) + 0.6 * np.exp(-((x - 0.7) ** 2) / 0.06) + 0.05
    elif mode == 5:
        # Sigmoid-like
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5))) + 0.05
    elif mode == 6:
        # Random smooth
        a = np.ones(n)
        for _ in range(3):
            c = rng.uniform(0.1, 0.9)
            w = rng.uniform(0.05, 0.2)
            amp = rng.uniform(-0.5, 0.5)
            a += amp * np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines smooth mirror descent, subgradient exploitation,
    LP alternating with golden section, peak shaving, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = diverse_asymmetric_init(n0, rng, mode=7)  # uniform fallback
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    print(f"\nInitial: n={len(best_seq)}, obj={best_score:.8f}")

    # === Phase 1: Smooth mirror descent (quick convergence) ===
    print(f"\n=== Phase 1: Smooth Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p1 = min(remaining * 0.15, 180.0)
    if p1 > 30:
        a_opt = smooth_mirror_optimize(best_seq, time_budget=p1, beta_start=20, beta_end=3000)
        obj = compute_obj(a_opt)
        print(f"  Smooth: obj={obj:.8f}" + (" **" if obj < best_score else ""))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # === Phase 2: Subgradient optimization ===
    print(f"\n=== Phase 2: Subgradient Optimization ===")
    remaining = deadline - time.monotonic()
    p2 = min(remaining * 0.15, 180.0)
    if p2 > 30:
        a_opt = subgradient_optimize(best_seq, time_budget=p2)
        obj = compute_obj(a_opt)
        print(f"  SubGrad: obj={obj:.8f}" + (" **" if obj < best_score else ""))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # === Phase 3: LP Annealing (main workhorse) ===
    print(f"\n=== Phase 3: LP Annealing ===")
    remaining = deadline - time.monotonic()
    p3_total = min(remaining * 0.35, 450.0)
    n_rounds = 3
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(p3_total / (n_rounds - rnd), remaining * 0.35)

        if rnd > 0:
            rng_r = np.random.default_rng(rnd * 999 + len(best_seq))
            a_start = best_seq * (1.0 + 0.04 * rng_r.normal(0, 1, len(best_seq)))
            a_start = np.maximum(a_start, 1e-15)
            a_start = a_start / a_start.sum()
        else:
            a_start = best_seq.copy()

        a_opt = lp_anneal_optimize(a_start, time_budget=alloc)
        obj = compute_obj(a_opt)
        print(f"  LP rnd{rnd}: obj={obj:.8f}" + (" **" if obj < best_score else ""))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    p4 = min(remaining * 0.1, 100.0)
    if p4 > 20:
        a_opt = peak_shave(best_seq, time_budget=p4)
        obj = compute_obj(a_opt)
        print(f"  PeakShave: obj={obj:.8f}" + (" **" if obj < best_score else ""))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best, n_best + 50, n_best + 100,
        n_best + 200, n_best + 400, 600, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and n != n_best]

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 120.0)

        # Quick smooth + LP
        half = alloc * 0.4
        if half > 15:
            a_try = smooth_mirror_optimize(a_try, time_budget=half, beta_start=20, beta_end=1000)

        remaining = deadline - time.monotonic()
        lp_alloc = min(remaining * 0.08, 60.0)
        if lp_alloc > 20:
            a_opt = lp_anneal_optimize(a_try, time_budget=lp_alloc)
            obj = compute_obj(a_opt)
        else:
            obj = compute_obj(a_try)
            a_opt = a_try

        print(f"  n={n_try}: obj={obj:.8f}" + (" **" if obj < best_score else ""))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # === Phase 6: Diverse asymmetric restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = (r * 3 + 1) % 7
        a_r = diverse_asymmetric_init(n_use, rng, mode=mode)
        # Blend with best
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 1e-15)
        a_r = a_r / a_r.sum()

        budget_r = min(remaining * 0.1, 80.0)
        if budget_r > 30:
            a_r = smooth_mirror_optimize(a_r, time_budget=budget_r * 0.5, beta_start=15, beta_end=500)
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_r = lp_anneal_optimize(a_r, time_budget=min(remaining * 0.08, 30.0))

        obj_r = compute_obj(a_r)
        print(f"  Restart{r} (mode={mode}): obj={obj_r:.8f}" + (" **" if obj_r < best_score else ""))
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # Alternating: smooth -> subgrad -> LP -> peak_shave
    for phase_name, func, budget_frac in [
        ("Smooth", smooth_mirror_optimize, 0.2),
        ("SubGrad", subgradient_optimize, 0.15),
        ("LP", lp_anneal_optimize, 0.3),
        ("PeakShave", peak_shave, 0.1),
    ]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * budget_frac, 100.0)
        if alloc < 15:
            continue

        if phase_name == "Smooth":
            a_opt = func(best_seq, time_budget=alloc, beta_start=50, beta_end=5000)
        elif phase_name == "SubGrad":
            a_opt = func(best_seq, time_budget=alloc)
        elif phase_name == "LP":
            a_opt = func(best_seq, time_budget=alloc)
        else:
            a_opt = func(best_seq, time_budget=alloc)

        obj = compute_obj(a_opt)
        print(f"  {phase_name}: obj={obj:.8f}" + (" **" if obj < best_score else ""))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    # === Final validation ===
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
