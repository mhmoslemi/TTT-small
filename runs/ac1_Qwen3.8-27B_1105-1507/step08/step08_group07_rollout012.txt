Smooth optimization (log-sum-exp + L-BFGS) combined with LP polishing
to minimize the peak autocorrelation of non-negative sequences.
"""

from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

        size_fft <<= 1
    
    Fa = rfft(a, size_fft)
    Fal = rfft(alpha, size_fft)
    # We want h[i] = sum_j a[j] * alpha[i+j]
    # This is the "reversed" cross-correlation
    # h = ifft(Fa * conj(Fal)) won't directly give what we want
    # 
    # Let me think again. The standard cross-correlation:
    # (a ⋆ b)[k] = sum_j a[j+k] * conj(b[j])
    # 
    # In terms of FFT: ifft(fft(a, N) * conj(fft(b, N))) gives circular correlation
    # For linear correlation, we need to zero-pad.
    #
    # What I want: h[i] = sum_{j=0}^{n-1} a[j] * alpha[i+j]
    # Let me define b[m] = alpha[m] (length 2n-1)
    # h[i] = sum_j a[j] * b[i+j]
    # This is the cross-correlation (a ⋆ b) at lag i (with appropriate definition)
    # 
    # Standard definition: (a ⋆ b)[i] = sum_j a[j] * b[j+i]
    # 
    # Using FFT: ifft(conj(fft(a, N)) * fft(b, N)) gives (a ⋆ b) for circular correlation
    # With zero-padding to length >= n + (2n-1) - 1 = 3n - 2, we get linear correlation.
    
    Fa_full = rfft(a, size_fft)
    Fal_full = rfft(alpha, size_fft)
    # conj(fft(a)) * fft(alpha) gives the cross-correlation sum_j conj(a[j]) * alpha[j+i]
    # For real a: sum_j a[j] * alpha[j+i]
    # We want h[i] = sum_j a[j] * alpha[i+j] = sum_j a[j] * alpha[j+i]
    # Yes! This is exactly what we want.
    
    corr = irfft(np.conj(Fa_full) * Fal_full, size_fft)
    h = corr[:n]
    
    # Gradient
    grad = 4.0 * n * a * (h - C)
    
    return obj, grad


def smooth_optimize(a_init, betas=[10, 50, 200, 500, 1000], time_budget=100.0, verbose=False):
    """
    Run smooth optimization starting from a_init.
    a_init should be a non-negative vector (will be normalized).
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    # Convert to log-space
    x = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for beta in betas:
        if time.monotonic() - start > time_budget:
            break
        
        # Run L-BFGS for this beta
        def fun(x_curr):
            return smooth_obj_and_grad(x_curr, beta)
        
        try:
            result = minimize(
                fun, x, method='L-BFGS-B',
                options={'maxiter': 200, 'gtol': 1e-8, 'fun_tol': 1e-10, 'maxfun': 1000}
            )
            x = result.x
        except:
            pass
        
        # Convert back to a
        x_max = np.max(x)
        exp_x = np.exp(x - x_max)
        a_new = exp_x / np.sum(exp_x)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            
            if verbose:
                print(f"    beta={beta}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    
    return best_a


def solve_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    
    # Build the convolution matrix: A[k,j] = a[k-j] if 0<=k-j<n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, j + n):
            A[k, j] = a[k - j]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    return g_val, float(M.value)
        except:
            continue
    
    # Fallback
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                return g_val, float(res.x[n])
    except:
        pass
    
    return None, None


def lp_alternating(a, time_budget=60.0, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Try different mixing ratios
        best_cand_obj = np.inf
        best_cand = a.copy()
        for mix in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - mix) * a + mix * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_cand_obj:
                best_cand_obj = obj_try
                best_cand = a_try.copy()
        
        if best_cand_obj < best_obj - 1e-12:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            a = best_a
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
        else:
            # Stuck, try perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
            best_obj = compute_obj(a)
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)
