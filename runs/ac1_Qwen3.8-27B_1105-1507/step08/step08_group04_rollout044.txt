import numpy as np

def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp_fast(a, warm_start=None):

def subgradient_peak_proportional(a, top_k=30):
    """Subgradient with peak-proportional weighting."""
    n = len(a)
    conv = compute_conv(a)
    peak = np.max(conv)
    mean_conv = np.mean(conv)

    if peak <= mean_conv + 1e-15:
        return np.zeros(n)

    # Weight peaks by (val - mean)^2 normalized
    excess = np.maximum(conv - mean_conv, 0)
    weights = excess ** 2
    total_w = np.sum(weights)
    if total_w < 1e-30:
        return np.zeros(n)
    weights = weights / total_w

    # Gradient of max(conv(a,a)) w.r.t. a[j] = 2 * sum_k weights[k] * a[k-j]
    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1

    A_fft = rfft(a, n_fft)
    W_fft = rfft(weights, n_fft)
    R = irfft(np.conj(A_fft) * W_fft, n_fft)

    grad = np.zeros(n)
    for j in range(n):
        grad[j] = R[(n_fft - j) % n_fft]

    grad *= 4.0 * n
    # Center for simplex
    grad -= np.mean(grad)
    return grad


def subgradient_logsumexp(a, T=0.01):
    """Subgradient using log-sum-exp smooth approximation to max."""
    n = len(a)
    conv = compute_conv(a)

    # Soft max
    max_conv = np.max(conv)
    shifted = conv / T - max_conv / T
    exp_shifted = np.exp(np.minimum(shifted, 50))  # prevent overflow
    logsum = max_conv / T + np.log(np.sum(exp_shifted))
    weights = exp_shifted / np.sum(exp_shifted)

    # Same gradient computation as before
    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1

    A_fft = rfft(a, n_fft)
    W_fft = rfft(weights, n_fft)
    R = irfft(np.conj(A_fft) * W_fft, n_fft)

    grad = np.zeros(n)
    for j in range(n):
        grad[j] = R[(n_fft - j) % n_fft]

    grad *= 4.0 * n
    grad -= np.mean(grad)
    return grad


def subgradient_optimize(a, time_budget=60.0, top_k=30, verbose=False):
    """Polyak subgradient method with peak-proportional weighting."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Polyak step size parameters
    obj_upper = best_obj + 0.01  # optimistic upper bound
    lr_scale = 0.3
    momentum = np.zeros(n)
    beta = 0.85

    x = a.copy()
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break

        if it % 50 == 0:
            grad = subgradient_peak_proportional(x, top_k=top_k)
        else:
            grad = subgradient_logsumexp(x, T=0.01)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Polyak step size
        obj_current = compute_obj(x)
        polyak_gap = max(obj_upper - best_obj, 1e-10)
        lr = lr_scale * polyak_gap / (grad_norm ** 2 + 1e-15)
        lr = min(lr, 2.0 / np.sqrt(it + 1))

        momentum = beta * momentum - lr * grad / (grad_norm + 1e-15)
        a_new = x + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = x.copy()
        else:
            a_new = a_new / s_new

        x = a_new
        obj = compute_obj(x)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = x.copy()
            obj_upper = min(obj_upper, obj + 0.005)

        if it % 500 == 499:
            beta = max(0.7, beta * 0.99)

        if verbose and it % 5000 == 0:
            print(f"    SubGrad: it={it}, obj={best_obj:.8f}")

    return best_a

        ema = np.maximum(ema, 0)
        s_ema = float(np.sum(ema))
        if s_ema > 1e-10:
            ema = ema / s_ema

        # Evaluate both the EMA and the raw LP solution
        candidates = [g_lp, ema]

        # Also try blends of current a with ema
        for t in [0.3, 0.5, 0.7]:
            candidates.append((1 - t) * a + t * ema)

        best_local_obj = np.inf
        best_local_a = best_a.copy()

        for cand in candidates:
            cand = np.maximum(cand, 0)
            sc = float(np.sum(cand))
            if sc < 1e-10:
                continue
            cand = cand / sc
            obj_c = compute_obj(cand)
            if obj_c < best_local_obj:
                best_local_obj = obj_c
                best_local_a = cand.copy()

        if best_local_obj < best_obj - 1e-13:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP-EMA: it={ema_iters}, obj={best_obj:.8f}, time={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 10:
                # Perturbation to escape local minimum
                rng = np.random.default_rng((n * 7919 + int(time.monotonic() * 10000)) % (2**31))
                scale = min(0.03 + 0.01 * (stall // 10), 0.2)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
                ema = a.copy()
                ema_iters = 0

    return best_a


def peak_shaving(a, num_steps=500, time_budget=30.0):
    """Directly reduce the worst convolution peaks by adjusting responsible coordinates."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for step in range(num_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Find the peak position
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        # Find the coordinate j that contributes most to this peak
        # Contribution of j to conv[peak_k] is a[j] * a[peak_k - j] (if valid)
        j_low = max(0, peak_k - n + 1)
        j_high = min(n - 1, peak_k)
        if j_low > j_high:
            j_low, j_high = 0, min(n - 1, peak_k)

        contributions = np.zeros(n)
        for j in range(j_low, j_high + 1):
            k2 = peak_k - j
            if 0 <= k2 < n:
                contributions[j] = a[j] * a[k2]

        if np.max(contributions) < 1e-20:
            continue

        # Try reducing the top contributor and adding to a random coordinate
        top_contributors = np.argsort(contributions)[::-1][:5]

        improved = False
        for j_dec in top_contributors:
            if a[j_dec] < 1e-8:
                continue

            # Try different delta values
            for delta_frac in [0.05, 0.1, 0.2, 0.3]:
                delta = a[j_dec] * delta_frac

                # Find where to add (a coordinate not contributing to this peak)
                # Pick from low-contribution coordinates
                non_contrib = [j for j in range(n) if contributions[j] < np.max(contributions) * 0.1]
                if not non_contrib:
                    non_contrib = list(range(n))
                j_inc = non_contrib[rng.integers(0, len(non_contrib))]

                a_try = a.copy()
                a_try[j_dec] -= delta
                a_try[j_inc] += delta
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try

                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved and step % 100 == 99:
            # Random perturbation
            j_rand = rng.integers(0, n)
            a[j_rand] *= rng.uniform(0.7, 1.3)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s

    return best_a


def coordinate_local_search(a, num_iterations=2000, time_budget=60.0):
    """Coordinate-wise local search with renormalization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    for it in range(num_iterations):
        if time.monotonic() - start > time_budget:
            break

        # Pick a random coordinate
        j = rng.integers(0, n)

        # Try a few perturbations
        for delta_frac in [-0.2, -0.1, -0.05, 0.05, 0.1, 0.2]:
            delta = a[j] * delta_frac if delta_frac < 0 else np.mean(a) * delta_frac

            a_try = a.copy()
            a_try[j] += delta
            if a_try[j] < 0:
                continue
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                break

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 7)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-50 * (x - 0.5) ** 2)
    elif choice == 1:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 3 * x)
    elif choice == 2:
        a = np.exp(-80 * (x - 0.3) ** 2) + np.exp(-80 * (x - 0.7) ** 2)
    elif choice == 3:
        # Multi-frequency cosine
        a = 1.0
        for freq in [2, 5, 13]:
            a += rng.uniform(0.1, 0.5) * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)
    elif choice == 4:
        # Smooth random via Gaussian kernel
        a = np.ones(n)
        for _ in range(5):
            center = rng.uniform(0, 1)
            width = rng.uniform(0.05, 0.3)
            a *= 1.0 + rng.standard_normal() * np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a = np.maximum(a, 0.01)
    elif choice == 5:
        # Power law
        a = x ** rng.uniform(0.5, 3.0) + (1 - x) ** rng.uniform(0.5, 3.0)
    else:
        # Bump train
        a = np.ones(n) * 0.1
        num_bumps = rng.integers(2, 6)
        for _ in range(num_bumps):
            c = rng.uniform(0.1, 0.9)
            w = rng.uniform(0.02, 0.15)
            a += rng.uniform(0.5, 2.0) * np.exp(-((x - c) ** 2) / (2 * w ** 2))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining EMA-LP alternating, peak-proportional subgradient,
    peak shaving, coordinate local search, and adaptive multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        print(f"Parent loaded: n={len(a)}, obj={compute_obj(a):.8f}")
    except Exception:
        a = np.ones(800) / 800
        print(f"No parent, using uniform: n=800, obj={compute_obj(a):.8f}")

    best_a = a.copy()
    best_obj = compute_obj(a)
    n_base = len(best_a)

    # === Phase 1: Primary LP-EMA alternating ===
    print(f"\n=== Phase 1: LP-EMA Alternating (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 300.0)
    a_p1 = lp_alternate_ema(best_a, time_budget=p1_budget, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  Phase 1 result: obj={best_obj:.8f} **")
    else:
        print(f"  Phase 1 result: obj={obj_p1:.8f}")

    # === Phase 2: Peak shaving ===
    print(f"\n=== Phase 2: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = min(remaining * 0.10, 90.0)
        a_ps = peak_shaving(best_a, num_steps=2000, time_budget=p2_budget)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_obj:
            best_obj = obj_ps
            best_a = a_ps
            print(f"  Peak shaving: obj={best_obj:.8f} **")
        else:
            print(f"  Peak shaving: obj={obj_ps:.8f}")

    # === Phase 3: Subgradient polish ===
    print(f"\n=== Phase 3: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.15, 120.0)
        for top_k in [15, 40]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(p3_budget / 2.0, remaining * 0.45)
            a_sg = subgradient_optimize(best_a, time_budget=alloc, top_k=top_k)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg
                print(f"  SubGrad top_k={top_k}: obj={best_obj:.8f} **")
            else:
                print(f"  SubGrad top_k={top_k}: obj={obj_sg:.8f}")

    # === Phase 4: LP-EMA round 2 ===
    print(f"\n=== Phase 4: LP-EMA Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = min(remaining * 0.20, 180.0)
        a_p4 = lp_alternate_ema(best_a, time_budget=p4_budget, verbose=True)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_obj:
            best_obj = obj_p4
            best_a = a_p4
            print(f"  Phase 4 result: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 4 result: obj={obj_p4:.8f}")

    # === Phase 5: Coordinate local search ===
    print(f"\n=== Phase 5: Coordinate Local Search ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = min(remaining * 0.10, 80.0)
        a_cl = coordinate_local_search(best_a, num_iterations=5000, time_budget=p5_budget)
        obj_cl = compute_obj(a_cl)
        if obj_cl < best_obj:
            best_obj = obj_cl
            best_a = a_cl
            print(f"  Coord search: obj={best_obj:.8f} **")
        else:
            print(f"  Coord search: obj={obj_cl:.8f}")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    # Generate n candidates adaptively
    n_options = sorted(set([
        n_best,
        n_best - 100, n_best - 200,
        n_best + 100, n_best + 200, n_best + 400,
        max(200, n_best - 400),
        n_best + 600,
    ]))
    n_options = [x for x in n_options if 200 <= x <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_best:
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try

        # Quick LP optimization
        alloc = min(40.0, remaining * 0.2)
        a_opt = lp_alternate_ema(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse initializations at best n ===
    print(f"\n=== Phase 7: Diverse Initializations (n={n_best}) ===")
    remaining = deadline - time.monotonic()

    for d in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_div = diverse_initialization(n_best, rng)
        alloc = min(30.0, remaining * 0.15)
        a_opt = lp_alternate_ema(a_div, time_budget=alloc)
        obj_div = compute_obj(a_opt)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse {d}: obj={obj_div:.8f}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 20:
        alloc_lp = min(remaining * 0.35, 60.0)
        a_final = lp_alternate_ema(best_a, time_budget=alloc_lp, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_ps = min(remaining * 0.25, 40.0)
        a_ps2 = peak_shaving(best_a, num_steps=1000, time_budget=alloc_ps)
        obj_ps2 = compute_obj(a_ps2)
        if obj_ps2 < best_obj:
            best_obj = obj_ps2
            best_a = a_ps2
            print(f"  Final peak shave: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_cl = min(remaining * 0.25, 40.0)
        a_cl2 = coordinate_local_search(best_a, num_iterations=3000, time_budget=alloc_cl)
        obj_cl2 = compute_obj(a_cl2)
        if obj_cl2 < best_obj:
            best_obj = obj_cl2
            best_a = a_cl2
            print(f"  Final coord: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_sg = min(remaining * 0.3, 30.0)
        a_sg2 = subgradient_optimize(best_a, time_budget=alloc_sg, top_k=20)
        obj_sg2 = compute_obj(a_sg2)
        if obj_sg2 < best_obj:
            best_obj = obj_sg2
            best_a = a_sg2
            print(f"  Final SubGrad: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    try:
        val = evaluate_sequence(result)
        if val < np.inf:
            best_obj = val
        else:
            result = [1.0] * 100
            best_obj = 2.0
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
