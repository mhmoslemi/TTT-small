import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_response(a):
    """
    Solve LP: min M s.t. conv(a, g) <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)
    m = 2 * n - 1

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res.status == 0:

def golden_section_step(a, g, max_iter=30):
    """
    Find optimal step t in [0, 1.5] along direction g - a.
    Uses golden section search on the objective.
    Returns (best_t, best_a).
    """
    n = len(a)
    direction = g - a
    
    # Define objective along the line
    def obj_at_t(t):
        a_try = a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        return compute_obj(a_try)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~ 0.618
    lo, hi = -0.1, 1.5
    
    # Check if extending beyond [0,1] helps
    f_lo = obj_at_t(lo)
    f_0 = obj_at_t(0.0)
    f_1 = obj_at_t(1.0)
    f_hi = obj_at_t(hi)
    
    # Find the bracket
    if f_0 < f_lo and f_0 < f_1:
        lo, hi = 0.0, 1.0
    elif f_1 < f_0 and f_1 < f_hi:
        lo, hi = 0.0, 1.5
    else:
        lo, hi = -0.1, 1.0
    
    c = lo + gr * (hi - lo)
    d = hi - gr * (hi - lo)
    fc = obj_at_t(c)
    fd = obj_at_t(d)
    
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = hi - gr * (hi - lo)
            fd = obj_at_t(d)
    
    t_best = (lo + hi) / 2
    a_best = a + t_best * direction
    a_best = np.maximum(a_best, 0)
    s = float(np.sum(a_best))
    if s < 1e-10:
        return 0.0, a.copy()
    a_best = a_best / s
    return t_best, a_best


def lp_alternating_v2(a, time_budget=300.0, verbose=False):
    """
    Improved LP alternating with golden-section step and Polyak averaging.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    # Polyak averaging
    avg_a = a.copy()
    polyak_start = 0
    
    for it in range(2000):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        
        g_lp, M_lp = solve_lp_for_response(a)
        if g_lp is None:
            break
        
        # Golden section step
        t_best, a_new = golden_section_step(a, g_lp, max_iter=25)
        obj_new = compute_obj(a_new)
        
        # Also try direct jump to g_lp
        obj_direct = compute_obj(g_lp)
        
        if obj_direct < obj_new:
            a_new = g_lp.copy()
            obj_new = obj_direct
            t_best = 1.0
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            polyak_start = it
            # Polyak update
            avg_a = (avg_a * polyak_start + a_new) / (polyak_start + 1)
            if verbose and it % 20 == 0:
                print(f"    LP v2 it={it}: obj={best_obj:.8f}, t={t_best:.3f}, time={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 15:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                scale = 0.02 + 0.01 * min(stall // 15, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
    
    # Check if Polyak average is better
    s_avg = float(np.sum(avg_a))
    if s_avg > 1e-10:
        avg_a = avg_a / s_avg
        obj_avg = compute_obj(avg_a)
        if obj_avg < best_obj:
            best_obj = obj_avg
            best_a = avg_a.copy()
    
    return best_a


def newton_active_set_polish(a, time_budget=100.0, verbose=False):
    """
    Newton method on the active set to polish the solution.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for outer_it in range(50):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Find active set: peaks within 1% of max
        threshold = max_val * 0.99
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        # Limit active set size for computational tractability
        if len(active) > 50:
            active = np.argsort(conv)[::-1][:50]
        
        p = len(active)
        
        # Build the Newton system
        # F_i = (a*a)[k_i] - M = 0 for i in active
        # F_{p} = sum(a) - 1 = 0
        # Variables: a (n) + M (1) = n+1 unknowns
        # Equations: p+1
        
        # Current residuals
        F = np.zeros(p + 1)
        for idx, k in enumerate(active):
            F[idx] = conv[k] - max_val
        F[p] = float(np.sum(a)) - 1.0
        
        # Jacobian: (p+1) x (n+1)
        # J[idx, j] = d F_idx / d a_j = 2 * a[k - j] (for valid j, with adjustments)
        # J[idx, n] = -1 (derivative of F_idx w.r.t. M)
        # J[p, j] = 1 for all j
        # J[p, n] = 0
        
        J = np.zeros((p + 1, n + 1))
        for idx, k in enumerate(active):
            k = int(k)
            for j in range(n):
                ki = k - j
                if 0 <= ki < n:
                    coeff = a[ki]
                    if ki != j:
                        coeff = 2.0 * a[ki]
                    # Wait, this isn't right. Let me redo.
                    # d/d a[j] sum_i a[i]*a[k-i]:
                    # = a[k-j] (from i=j term) + a[k-j] (from k-i=j term, if k-j != j)
                    # = a[k-j] + a[k-j] if k-j != j and 0 <= k-j < n
                    # = a[k-j] if k-j == j (i.e., k=2j) or k-j out of range
                    pass
            
            # Let me just compute it properly
            for j in range(n):
                ki = k - j
                # Term 1: from i=j in sum, gives a[k-j] if 0<=k-j<n
                val = 0.0
                if 0 <= ki < n:
                    val += a[ki]
                # Term 2: from k-i=j, i.e., i=k-j, gives a[j] if 0<=j<n and i=k-j != j
                # Actually, the second term is: when i = k-j, the term a[i]*a[k-i] = a[k-j]*a[j]
                # and d/d a[j] of a[k-j]*a[j] = a[k-j] (if k-j != j) or 2*a[j] (if k-j=j)
                # But we already counted the case i=j in Term 1.
                # The full derivative is:
                # sum_i d/d a[j] [a[i] * a[k-i]]
                # = (da[j]/da[j]) * a[k-j] + a[j] * (da[k-j]/da[j])  [if both terms exist]
                # = a[k-j] * [0<=k-j<n] + a[j] * [0<=k-j<n and k-j != j]
                # = a[k-j] if 0<=k-j<n, plus a[j] if 0<=k-j<n and k-j!=j
                
                ki = k - j
                if 0 <= ki < n:
                    val = a[ki]  # from the first term (i=j)
                    if ki != j:
                        val += a[ki]  # from the second term (i=k-j=j... wait)
                # Hmm, I'm confusing myself. Let me be very explicit.
                # (a*a)[k] = sum_{i: valid} a[i] * a[k-i]
                # d/d a[j] = sum_{i: valid} [delta(i,j) * a[k-i] + a[i] * delta(k-i, j)]
                # = a[k-j] (if j is valid, i.e., 0<=j<n and 0<=k-j<n)
                #   + a[k-j] (if k-j is valid, i.e., 0<=k-j<n and 0<=j<n, AND k-j != j to avoid double counting)
                # 
                # Wait, delta(k-i, j) means k-i=j, i.e., i=k-j.
                # So the second sum is: a[k-j] * [0<=k-j<n] * [0<=j<n]
                # But we need i=k-j to be in the valid range, which means 0<=k-j<n.
                # And we also need j to be in [0,n-1] (which it always is since j is our index).
                # 
                # So: d/d a[j] (a*a)[k] = a[k-j] * [0<=k-j<n] + a[k-j] * [0<=k-j<n]
                # = 2*a[k-j] if 0<=k-j<n
                # But wait, if j = k-j (i.e., k=2j), then we're double-counting the same term.
                # In that case, the term is a[j]*a[j] = a[j]^2, and d/d a[j] = 2*a[j].
                # And 2*a[k-j] = 2*a[j]. So it's consistent!
                
                ki = k - j
                if 0 <= ki < n:
                    J[idx, j] = 2.0 * a[ki]
                else:
                    J[idx, j] = 0.0
            
            J[idx, n] = -1.0  # d F_idx / d M
        
        J[p, :n] = 1.0  # d sum(a) / d a_j
        J[p, n] = 0.0
        
        # Solve the underdetermined system J * delta = -F
        # Minimum norm solution: delta = J^T * (J J^T)^{-1} * (-F)
        try:
            JJT = J @ J.T  # (p+1) x (p+1)
            # Add small regularization for numerical stability
            JJT += 1e-10 * np.eye(p + 1)
            delta = J.T @ np.linalg.solve(JJT, -F)
        except np.linalg.LinAlgError:
            # Pseudoinverse fallback
            try:
                delta = np.linalg.pinv(J) @ (-F)
            except Exception:
                break
        
        # Line search
        step = 1.0
        for _ in range(20):
            a_trial = a + step * delta[:n]
            a_trial = np.maximum(a_trial, 0)
            s_trial = float(np.sum(a_trial))
            if s_trial < 1e-10:
                step *= 0.5
                continue
            a_trial = a_trial / s_trial
            obj_trial = compute_obj(a_trial)
            if obj_trial < best_obj:
                a = a_trial.copy()
                best_obj = obj_trial
                best_a = a_trial.copy()
                if verbose:
                    print(f"    Newton it={outer_it}: obj={best_obj:.8f}, active={p}")
                break
            step *= 0.5
        
        if step >= 0.5:
            break  # No improvement possible with this active set
    
    return best_a


def peak_shaving(a, time_budget=100.0, verbose=False):
    """
    Direct peak shaving: reduce the maximum peak by adjusting contributing elements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_val = float(conv[k_star])
        
        # Compute contributions to the peak
        contributions = np.zeros(n)
        for j in range(n):
            ki = k_star - j
            if 0 <= ki < n and ki != j:
                contributions[j] = 2.0 * a[j] * a[ki]
            elif 0 <= ki < n:
                contributions[j] = a[j] ** 2
        
        # Find the top contributors
        top_idx = np.argsort(contributions)[::-1][:30]
        
        improved = False
        for j in top_idx:
            if time.monotonic() - start > time_budget:
                break
            
            # Try reducing a[j] and adding to a "far" position
            j = int(j)
            if a[j] < 1e-8:
                continue
            
            # Target: position that minimizes contribution to the peak
            # Ideally far from both j and k_star-j in the convolution sense
            ki_star = k_star - j
            # Positions far from both j and ki_star
            distances = np.abs(np.arange(n) - j) + np.abs(np.arange(n) - ki_star)
            # Also avoid positions that create new peaks
            candidates = np.argsort(distances)[::-1][:20]
            
            for target in candidates:
                target = int(target)
                if target == j:
                    continue
                
                # Try moving a small amount
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    delta = frac * a[j]
                    if delta <= 0 or delta >= a[j]:
                        continue
                    
                    a[j] -= delta
                    a[target] += delta
                    s_new = float(np.sum(a))
                    a_norm = a / s_new
                    new_conv = compute_conv(a_norm)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a_norm.copy()
                        improved = True
                        break
                    else:
                        a[j] += delta
                        a[target] -= delta
                
                if improved:
                    break
            
            if improved:
                a = best_a.copy()
                break
    
        if not improved:
            # Try random perturbation
            if it % 20 == 19:
                rng = np.random.default_rng(int(time.monotonic() * 1e5) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
    
    return best_a


def spectral_flatten(a, time_budget=100.0, verbose=False):
    """
    Try to flatten the power spectrum |DFT(a)|^2 to reduce autocorrelation peaks.
    Uses iterative spectral projection.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Compute DFT
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute current DFT
        F = rfft(a, size)
        power = np.abs(F) ** 2
        
        # Target: flatten the power spectrum (but keep DC component)
        # Use a soft flattening: F_target = sign(F) * sqrt(target_power)
        # Target power: geometric mean of the power spectrum
        # To avoid making DC zero, keep it at its current value
        non_dc = power[1:]
        if len(non_dc) > 0 and np.all(non_dc > 0):
            target_power = np.ones_like(power) * np.median(non_dc)
            target_power[0] = power[0]  # Keep DC
        else:
            break
        
        # Interpolate between current and target spectrum
        alpha = 0.1  # Step size
        F_new = F * (1 - alpha) + F * np.sqrt(target_power / np.maximum(power, 1e-20)) * alpha
        # Wait, this isn't right. Let me think again.
        # We want to adjust F so that |F|^2 moves toward target_power.
        # F_new = F * sqrt(target_power / |F|^2) = F * (target_power)^0.5 / |F|
        # This preserves the phase and changes the magnitude.
        
        ratio = np.sqrt(target_power) / np.maximum(np.abs(F), 1e-10)
        F_new = F * (1 - alpha + alpha * ratio)
        
        # Inverse FFT
        a_new = irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose:
                print(f"    Spectral it={it}: obj={best_obj:.8f}")
    
    return best_a


def generate_initialization(kind, n, rng):
    """Generate a diverse initialization of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'blackman':
        a = np.blackman(n) + 0.01
        return a / np.sum(a)
    elif kind == 'kaiser':
        from scipy.signal import get_window
        a = get_window('kaiser', n, beta=4.0) + 0.01
        return a / np.sum(a)
    elif kind == 'hamming':
        a = np.hamming(n) + 0.01
        return a / np.sum(a)
    elif kind == 'banolax':
        # B-moment optimal window (approximation)
        x = np.linspace(0, 1, n)
        a = 1.0 - 2.0 * np.abs(x - 0.5)  # Triangle
        return a / np.sum(a)
    elif kind == 'two_peak':
        a = np.ones(n) * 0.3
        a[n//4] = 1.0
        a[3*n//4] = 1.0
        kernel = np.ones(max(3, n//20)) / max(3, n//20)
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if n == 2:
        x_old = np.array([0, 1])
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
        return np.maximum(a_new, 0)
    from scipy.interpolate import CubicSpline
    x_old = np.linspace(0, 1, n)
    cs = CubicSpline(x_old, a, bc_type='natural')
    x_new = np.linspace(0, 1, new_n)
    a_new = cs(x_new)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):

    Main search function combining LP-alternating, Newton polishing,
    peak shaving, spectral flattening, and adaptive length search.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Quick diverse initializations (40s) ===
    print(f"\n=== Phase 1: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    init_budget = min(40.0, remaining * 0.03)

    init_kinds = ['cosine', 'blackman', 'kaiser', 'hamming', 'gaussian', 'sine_taper', 'random', 'banolax']
    per_init = max(2.0, init_budget / len(init_kinds))

    for kind in init_kinds:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        a_init = generate_initialization(kind, n_parent, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init '{kind}': obj={obj_init:.8f} **")
        else:
            print(f"  Init '{kind}': obj={obj_init:.8f}")

    a = best_seq.copy()
    print(f"  Best after Phase 1: obj={best_score:.8f}")

    # === Phase 2: Improved LP Alternating (500s) ===
    print(f"\n=== Phase 2: LP Alternating v2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        lp_budget = min(500.0, remaining * 0.4)
        a_new = lp_alternating_v2(a, time_budget=lp_budget, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  ** LP v2: obj={best_score:.8f} **")
        else:
            print(f"  LP v2: obj={obj:.8f}")
    
    # Second round with perturbation
    remaining = deadline - time.monotonic()
    if remaining > 50 and best_score > 1.505:
        print(f"\n=== Phase 2b: LP Alternating (perturbed) ===")
        scale = 0.02
        a_pert = best_seq * np.exp(rng.normal(0, scale, n_parent))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / np.sum(a_pert)
        alloc = min(200.0, remaining * 0.2)
        a_new = lp_alternating_v2(a_pert, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  ** LP v2b: obj={best_score:.8f} **")

    # === Phase 3: Newton Active Set Polish (200s) ===
    print(f"\n=== Phase 3: Newton Active Set Polish ===")
    remaining = deadline - time.monotonic()

        if remaining > 60:

            a_opt = newton_active_set_polish(a_opt, time_budget=alloc_net, verbose=False)

"""
Optimized step function search: Multi-phase hybrid combining improved LP-alternating
with golden-section step selection, Newton active-set polishing, spectral flattening,
peak shaving, windowed initializations, and adaptive length search.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_response(a):
    """
    Solve LP: min M s.t. conv(a, g) <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)
    m = 2 * n - 1

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                return g_val, M_val
    except Exception:
        pass

    return None, None


def golden_section_step(a, g, max_iter=25):
    """
    Find optimal step t along direction g - a using golden section search.
    Returns (best_t, best_a).
    """
    n = len(a)
    direction = g - a

    def obj_at_t(t):
        a_try = a + t * direction
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        return compute_obj(a_try)

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    # Check endpoints to bracket
    f_minus = obj_at_t(-0.1)
    f_0 = obj_at_t(0.0)
    f_half = obj_at_t(0.5)
    f_1 = obj_at_t(1.0)
    f_15 = obj_at_t(1.5)

    # Determine bracket
    candidates_lo = [f_minus, f_0, f_half, f_1, f_15]
    vals_at = [-0.1, 0.0, 0.5, 1.0, 1.5]
    min_idx = int(np.argmin(candidates_lo))
    if min_idx == 0:
        lo, hi = -0.1, 0.5
    elif min_idx == 1:
        lo, hi = -0.1, 1.0
    elif min_idx == 2:
        lo, hi = 0.0, 1.5
    elif min_idx == 3:
        lo, hi = 0.5, 1.5
    else:
        lo, hi = 1.0, 1.5

    c = lo + gr * (hi - lo)
    d = hi - gr * (hi - lo)
    fc = obj_at_t(c)
    fd = obj_at_t(d)

    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gr * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = hi - gr * (hi - lo)
            fd = obj_at_t(d)
        if hi - lo < 1e-6:
            break

    t_best = (lo + hi) / 2.0
    a_best = a + t_best * direction
    a_best = np.maximum(a_best, 0)
    s = float(np.sum(a_best))
    if s < 1e-10:
        return 0.0, a.copy()
    a_best = a_best / s
    return t_best, a_best


def lp_alternating_v2(a, time_budget=300.0, verbose=False):
    """
    Improved LP alternating with golden-section step and Polyak averaging.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    avg_a = a.copy()
    count = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_for_response(a)
        if g_lp is None:
            break

        # Golden section step
        t_best, a_new = golden_section_step(a, g_lp, max_iter=22)
        obj_new = compute_obj(a_new)

        # Also try direct jump to g_lp
        obj_direct = compute_obj(g_lp)
        if obj_direct < obj_new:
            a_new = g_lp.copy()
            obj_new = obj_direct

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            count += 1
            avg_a = (avg_a * (count - 1) + a_new) / count
            if verbose and count <= 20:
                print(f"    LP v2 step={count}: obj={best_obj:.8f}, t={t_best:.3f}, "
                      f"time={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall >= 12:
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                scale = 0.015 + 0.005 * min(stall // 12, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    # Check Polyak average
    if count > 5:
        s_avg = float(np.sum(avg_a))
        if s_avg > 1e-10:
            avg_a = avg_a / s_avg
            obj_avg = compute_obj(avg_a)
            if obj_avg < best_obj:
                best_obj = obj_avg
                best_a = avg_a.copy()

    return best_a


def newton_active_set_polish(a, time_budget=100.0, verbose=False):
    """
    Newton method on the active set to polish the solution.
    Solves the underdetermined system for active peaks + sum constraint.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for outer_it in range(100):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))

        # Find active set: peaks within 1% of max
        threshold = max_val * 0.99
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 40:
            active = np.argsort(conv)[::-1][:40]

        p = len(active)

        # Build Jacobian: (p+1) x (n+1)
        # F_i = (a*a)[k_i] - M = 0
        # F_p = sum(a) - 1 = 0
        J = np.zeros((p + 1, n + 1))
        F = np.zeros(p + 1)

        for idx, k in enumerate(active):
            k = int(k)
            F[idx] = conv[k] - max_val
            for j in range(n):
                ki = k - j
                if 0 <= ki < n:
                    J[idx, j] = 2.0 * a[ki]
                else:
                    J[idx, j] = 0.0
            J[idx, n] = -1.0

        F[p] = float(np.sum(a)) - 1.0
        J[p, :n] = 1.0
        J[p, n] = 0.0

        # Minimum norm Newton step: delta = J^T (J J^T + eps I)^{-1} (-F)
        try:
            JJT = J @ J.T
            JJT += 1e-8 * np.eye(p + 1)
            delta = J.T @ np.linalg.solve(JJT, -F)
        except np.linalg.LinAlgError:
            try:
                delta = np.linalg.pinv(J) @ (-F)
            except Exception:
                break

        # Backtracking line search
        step = 1.0
        improved = False
        for _ in range(25):
            a_trial = a + step * delta[:n]
            a_trial = np.maximum(a_trial, 0)
            s_trial = float(np.sum(a_trial))
            if s_trial < 1e-10:
                step *= 0.5
                continue
            a_trial = a_trial / s_trial
            obj_trial = compute_obj(a_trial)
            if obj_trial < best_obj - 1e-14:
                a = a_trial.copy()
                best_obj = obj_trial
                best_a = a_trial.copy()
                improved = True
                if verbose:
                    print(f"    Newton it={outer_it}: obj={best_obj:.8f}, "
                          f"active={p}, step={step:.3f}")
                break
            step *= 0.5

        if not improved:
            if outer_it >= 3:
                break

    return best_a


def spectral_flatten(a, time_budget=100.0, verbose=False):
    """
    Iteratively flatten the power spectrum to reduce autocorrelation peaks.
    """
    n = len(a)

    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    alpha = 0.15
    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break

        F = rfft(a, size)
        power = np.abs(F) ** 2

        # Target: flatten power spectrum (keep DC)
        if len(power) > 1:
            non_dc = power[1:]
            positive = non_dc[non_dc > 1e-20]
            if len(positive) == 0:

            target_level = np.median(positive)
            target_power = np.full(len(power), target_level)
            target_power[0] = power[0]
        else:
            break

        ratio = np.sqrt(target_power) / np.maximum(np.abs(F), 1e-10)
        F_new = F * (1.0 - alpha + alpha * ratio)

        a_new = irfft(F_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose:
                print(f"    Spectral it={it}: obj={best_obj:.8f}")
        else:
            alpha *= 0.95
            if alpha < 0.01:
                break

    return best_a

                    if dst == src:

                    a[dst] += delta

def subgradient_polish(a, time_budget=100.0, verbose=False):
    """
    Nesterov-accelerated subgradient method targeting active peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    beta = 0.7
    lr = 0.08
    momentum = np.zeros(n)

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(y)
        max_val = float(np.max(conv))
        threshold = max_val * 0.9
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 30:
            active = np.argsort(conv)[::-1][:30]

        grad = np.zeros(n)
        for k in active:
            k = int(k)
            for j in range(n):
                ki = k - j
                if 0 <= ki < n:
                    grad[j] += 2.0 * y[ki]
        grad /= len(active)
        grad_centered = grad - np.mean(grad)

        momentum = beta * momentum - lr * grad_centered
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new

        y = y_new
        obj = compute_obj(y)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()
            if verbose and int(time.monotonic() - start) % 10 < 1:
                print(f"    Subgrad it={it}: obj={best_obj:.8f}")

        if it % 500 == 499:
            lr = max(lr * 0.9, 0.005)

    return best_a


def generate_initialization(kind, n, rng):
    """Generate a diverse initialization of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n, dtype=np.float64)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'blackman':
        a = np.blackman(n).astype(np.float64) + 0.01
        return a / np.sum(a)
    elif kind == 'kaiser':
        try:
            from scipy.signal import get_window
            a = get_window('kaiser', n, beta=4.0).astype(np.float64) + 0.01
        except Exception:
            a = np.ones(n) * 0.5
            edge = max(2, n // 10)
            a[:edge] = 0.1
            a[-edge:] = 0.1
        return a / np.sum(a)
    elif kind == 'hamming':
        a = np.hamming(n).astype(np.float64) + 0.01
        return a / np.sum(a)
    elif kind == 'triangle':
        x = np.linspace(0, 1, n)
        a = 1.0 - 2.0 * np.abs(x - 0.5)
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n) * 0.4
        a[n//6:5*n//6] = 1.0
        return a / np.sum(a)
    elif kind == 'smooth_plateau':
        a = np.ones(n) * 0.3
        center = n / 2.0
        x = np.arange(n, dtype=np.float64)
        width = n / 3.0
        a += np.exp(-0.5 * ((x - center) / width) ** 2)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    if n == 2:
        x_old = np.array([0.0, 1.0])
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
        return np.maximum(a_new, 0)
    try:
        from scipy.interpolate import CubicSpline
        x_old = np.linspace(0, 1, n)
        cs = CubicSpline(x_old, a, bc_type='natural')
        x_new = np.linspace(0, 1, new_n)
        a_new = cs(x_new)
    except Exception:
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
