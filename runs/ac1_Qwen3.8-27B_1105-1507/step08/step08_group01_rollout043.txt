Optimized search for minimum peak autocorrelation sequences.
Uses diverse initialization, adaptive smooth-max gradient descent,
LP alternating with adaptive active set, and multi-scale exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import random


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A

        A_use = A
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((A_use.shape[0], n + 1))
    A_ub[:, :n] = A_use
    A_ub[:, n] = -1.0
    b_ub = np.zeros(A_use.shape[0])
    
    A_eq = np.zeros((1, n + 1))

        # Top-k indices
        if top_k < 2 * n - 1:
            top_idx = np.argpartition(conv, -top_k)[-top_k:]
            # Renormalize weights over top-k only
            T_local = (conv[top_idx].max() - conv[top_idx].min()) / max(1, top_k) * 3
            T_local = max(T_local, 1e-10)
            exp_vals = np.exp((conv[top_idx] - conv[top_idx].max()) / T_local)
            weights_top = exp_vals / np.sum(exp_vals)
            # Full weight vector (sparse)
            w = np.zeros(2 * n - 1)
            w[top_idx] = weights_top
        else:
            shifted = (conv - max_c) / T
            shifted = np.clip(shifted, -500, 500)
            exp_vals = np.exp(shifted)
            w = exp_vals / np.sum(exp_vals)
        
        # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j] (for i+j in [0, 2n-2])
        # This is a correlation: grad = 2 * correlate(w, a)
        # Using FFT for efficiency
        L = len(w)
        size = 1
        while size < L + n - 1:
            size <<= 1
        fw = rfft(w, size)
        fa = rfft(a, size)
        corr = irfft(fw * np.conj(fa), size)[:L + n - 1]
        # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
        # This is corr[i] for i in [0, n-1] if we set it up right
        # Actually: corr[k] = sum_j w[k-j] * conj(a[j])... let me just do it directly
        # grad[i] = sum_j w[i+j] * a[j] = (w * a_padded)[i] where a_padded has a at the right position
        # Simpler: use np.convolve or direct
        # grad[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
        # This is the cross-correlation of w and a, evaluated at lags 0 to n-1
        # np.correlate(w, a, mode='valid') gives sum_j w[i+j]*a[j] for i=0..len(w)-len(a)
        # len(w) = 2n-1, len(a) = n, so valid gives n values. 
        grad_half = np.correlate(w, a, mode='valid')
        grad = 2.0 * grad_half
        
        # Center the gradient (to maintain sum constraint approximately)
        grad_centered = grad - np.mean(grad)
        
        # Nesterov momentum
        v = momentum * v - lr * grad_centered
        a_trial = simplex_project(a + v)
        
        obj_trial = compute_obj(a_trial)
        if obj_trial < best_obj - 1e-14:
            best_obj = obj_trial
            best_a = a_trial.copy()
            a = a_trial
            stall = 0
            # Slowly increase lr
            if lr < 0.05:
                lr *= 1.01
        else:
            stall += 1
            if stall > 30:
                # Backtrack
                lr *= 0.6
                if lr < 1e-7:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
                # Reheat
                T = max_c / 3.0
        
        # Decay temperature
        T = max(T_min, T * T_decay)
        
        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    SGD it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}, time={elapsed:.1f}s")
    
    return best_a


def lp_alternating_v2(a, time_budget=200.0, verbose=False):
    """
    LP alternating with adaptive active set and convex line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute active set: top peaks within adaptive threshold
        conv = compute_conv(a)
        max_c = np.max(conv)
        
        # Adaptive number of active constraints
        sorted_conv = np.sort(conv)[::-1]
        # Use all constraints within 5% of max
        threshold = max_c * 0.95
        active_mask = conv >= threshold
        active_idx = np.where(active_mask)[0]
        
        # If too many, take top 50
        if len(active_idx) > 50:
            top_50 = np.argpartition(conv, -50)[-50:]
            active_idx = top_50
        
        # Solve LP
        g = solve_lp_for_direction(a, active_idx, time_limit=min(5.0, max(1.0, (time_budget - (time.monotonic() - start)) / max(1, 100 - it))))
        if g is None:
            # Try full LP
            g = solve_lp_for_direction(a, None, time_limit=5.0)
            if g is None:
                break
        
        # Convex line search
        a_norm = a / np.sum(a)
        g_norm = g / np.sum(g)
        
        conv_aa = compute_conv(a_norm)
        conv_ag = compute_conv_cross(a_norm, g_norm)
        conv_gg = compute_conv(g_norm)
        
        def obj_t(t):
            val = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
            return 2.0 * n * float(np.max(val))
        
        # Ternary search
        lo, hi = 0.0, 1.0
        for _ in range(15):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if obj_t(m1) < obj_t(m2):
                hi = m2
            else:
                lo = m1
        t_opt = (lo + hi) / 2.0
        
        a_new = (1 - t_opt) * a_norm + t_opt * g_norm
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            stall += 1
        else:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                # Try smaller step
                for t_small in [0.1, 0.05, 0.01]:
                    a_try = (1 - t_small) * a_norm + t_small * g_norm
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-13:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            stall = 0
                            break
        
        if stall > 30:
            # Perturb and reset
            rng = np.random.default_rng(it * 7919 + n)
            scale = 0.01
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 1e-10:
                a = a_pert / s_p
            stall = 0
            if it > 500:
                break
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, |active|={len(active_idx)}, time={elapsed:.1f}s")
    
    return best_a


def coordinate_descent_v2(a, time_budget=60.0, verbose=False):
    """
    Fast coordinate descent using FFT-based evaluation.
    Tries pairs of coordinates for speed.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    delta = 1e-3
    rng = np.random.default_rng(12345 + n)
    
    pass_num = 0
    while time.monotonic() - start < time_budget:
        pass_num += 1
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            # Try small increase
            best_delta = 0
            best_obj_try = best_obj
            for d in [delta, delta * 0.1, delta * 3]:
                a_try = a.copy()
                a_try[i] += d
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj_try:
                    best_obj_try = obj_try
                    best_delta = d
            
            if best_delta != 0:
                a[i] += best_delta
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_obj = best_obj_try
                best_a = a.copy()
                improved = True
                continue
            
            # Try small decrease
            if a[i] > delta * 0.1:
