def golden_section_search(a, g, f_obj, a_lo=0.0, a_hi=1.0, tol=1e-4, max_iter=50):
    """Find t in [a_lo, a_hi] that minimizes f_obj((1-t)*a + t*g)."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    f_c = f_obj((1-c)*a + c*g)
    f_d = f_obj((1-d)*a + d*g)
    for _ in range(max_iter):
        if a_hi - a_lo < tol:
            break
        if f_c < f_d:
            a_hi = d
            d = c
            f_d = f_c

            f_c = f_obj((1-c)*a + c*g)

            c = d
            f_c = f_d
            d = a_lo + gr * (a_hi - a_lo)
            f_d = f_obj((1-d)*a + d*g)

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with golden-section line search, dual-informed subgradients,
smooth L-BFGS polishing, systematic n-exploration, and structured escape mechanisms.
Key improvements: proper 1D line search, faster LP via vectorized matrix build,
hard-update (t=1) priority, and more aggressive LP usage with periodic subgradient
corrections to escape local minima.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize as sp_minimize

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except Exception:
    HAVE_CVXPY = False


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g). Vectorized."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[diff[mask]]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    # Fallback to scipy
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def solve_lp_maxsum(a, M_target):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M_target, g >= 0."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M_target]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s
            except Exception:
                continue

    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M_target),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x, 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None


def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a[j] = 2*a[k-j] if valid."""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        m = k - j
        if 0 <= m < n:
            g[j] = 2.0 * a[m]
    return g


def weighted_subgradient(a, weights, peak_indices):
    """Compute weighted subgradient from multiple peaks."""
    n = len(a)
    g = np.zeros(n)
    for w, k in zip(weights, peak_indices):
        if w > 1e-14:
            g += w * grad_conv_at_peak(a, int(k))
    return g


def golden_section_search(a, g, a_lo=0.0, a_hi=1.0, tol=0.005, max_evals=40):
    """
    Find t in [a_lo, a_hi] minimizing compute_obj((1-t)*a + t*g).
    Uses golden section search. Returns (best_t, best_obj).
    """
    gr = (np.sqrt(5) - 1) / 2.0

    def f_of_t(t):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        return compute_obj(a_try)

    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    f_c = f_of_t(c)
    f_d = f_of_t(d)
    evals = 2

    for _ in range(max_evals):
        if a_hi - a_lo < tol or evals >= max_evals:
            break
        if f_c < f_d:
            a_hi = d
            d = c
            f_d = f_c
            c = a_hi - gr * (a_hi - a_lo)
            f_c = f_of_t(c)
        else:
            a_lo = c
            c = d
            f_c = f_d
            d = a_lo + gr * (a_hi - a_lo)
            f_d = f_of_t(d)
        evals += 2

    t_best = (a_lo + a_hi) / 2.0
    f_best = f_of_t(t_best)
    # Also check endpoints
    if f_of_t(0.0) < f_best:
        t_best = 0.0
        f_best = f_of_t(0.0)
    if f_of_t(1.0) < f_best:
        t_best = 1.0
        f_best = f_of_t(1.0)
    return t_best, f_best


def lp_alternate(a, time_budget=300.0, verbose=False):
    """
    LP alternating with golden-section line search.
    Prioritizes t=1 (hard update), falls back to line search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)
    it = 0
    patience = 25

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break

        # Solve LP for direction
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            conv_a = compute_conv(a)
            M_cur = float(np.max(conv_a))
            g_lp = solve_lp_maxsum(a, M_cur * 1.01)
            if g_lp is None:
                break

        it += 1
        improved = False

        # Try t=1 first (hard update)
        obj_t1 = compute_obj(g_lp)
        if obj_t1 < best_obj - 1e-12:
            best_obj = obj_t1
            best_a = g_lp.copy()
            a = g_lp.copy()
            stall = 0
            improved = True
            if verbose and it % 5 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f} (t=1), M={M_lp:.6f}, el={time.monotonic()-start:.1f}s")
        else:
            # Golden section search for optimal t
            t_best, obj_best = golden_section_search(a, g_lp, a_lo=0.05, a_hi=0.95,
                                                     tol=0.01, max_evals=25)
            if obj_best < best_obj - 1e-12:
                a_try = (1.0 - t_best) * a + t_best * g_lp
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                best_obj = obj_best
                best_a = a_try.copy()
                a = a_try
                stall = 0
                improved = True
                if verbose and it % 5 == 0:
                    print(f"    LP it{it}: obj={best_obj:.8f} (t={t_best:.3f}), M={M_lp:.6f}, el={time.monotonic()-start:.1f}s")
            else:
                stall += 1

        if not improved and stall > patience:
            # Escape: structured perturbation
            noise_scale = min(0.4, 0.08 + 0.05 * (stall // patience))
            a = best_a * np.exp(rng.normal(0, noise_scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            best_a = a.copy()
            best_obj = compute_obj(a)
            stall = 0
            patience = max(10, patience - 3)

    return best_a


def nesterov_subgrad(a, time_budget=100.0, K=5):
    """Nesterov-accelerated subgradient on simplex with adaptive K."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 100
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        # Adaptive K: use more peaks when spread is large
        top_K = min(K + int((stall // 50) * 2), 20)
        top_idx = np.argsort(conv)[::-1][:top_K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        # Softmax weights with adaptive temperature
        spread = (peak_vals[0] - peak_vals[-1]) / (max_val + 1e-15)
        temp = 5.0 / (1.0 + spread * 5.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / (max_val + 1e-15))
        weights /= weights.sum()

        g = weighted_subgradient(y, weights, top_idx)

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.2)

        a_prev = a_new.copy()
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        y = y_new.copy()
        t_k = t_new
        it += 1

    return best_a


def smooth_polish(a, time_budget=100.0, T=50.0):
    """Smooth optimization using log-sum-exp with L-BFGS-B."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_grad(x, T_inner):
        x = np.maximum(x, 1e-15)
        sv = x.sum()
        if sv < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / sv
        conv = compute_conv(x)
        max_c = conv.max()
        z = T_inner * (conv - max_c)
        z = np.clip(z, -500, 500)
        e = np.exp(z)
        sum_e = e.sum()
        if sum_e < 1e-300:
            sum_e = 1e-300
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T_inner)
        w = e / sum_e
        g = np.zeros(n)
        active = w > 1e-10
        for k in range(len(conv)):
            if active[k]:
                g += w[k] * grad_conv_at_peak(x, k)
        g *= 2.0 * n
        return obj_val, g

    try:
        # Multiple temperature runs
        for T_val in [T, T * 2, T * 0.5]:
            if time.monotonic() - start > time_budget:
                break
            res = sp_minimize(lambda x: obj_grad(x, T_val), a, method='L-BFGS-B',
                              bounds=[(1e-12, None)] * n,
                              options={'maxiter': 300, 'maxfun': 600, 'disp': False})
            a_opt = np.maximum(res.x, 0)
            s_opt = float(np.sum(a_opt))
            if s_opt > 1e-10:
                a_opt = a_opt / s_opt
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_obj:
                    best_obj = obj_opt
                    best_a = a_opt.copy()
                    a = a_opt.copy()
    except Exception:
        pass

    return best_a


def coordinate_refine(a, time_budget=30.0):
    """Coordinate descent: move mass away from peak-contributing positions."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Find contributing pairs
        contrib = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib.append((a[i] * a[j], i, j))
        contrib.sort(reverse=True)

        # Find non-contributing positions
        non_contrib = []
        for i in range(n):
            j = k_star - i
            if j < 0 or j >= n:
                non_contrib.append(i)

        if not non_contrib or len(contrib) == 0:
            break

        improved = False
        for ci in range(min(10, len(contrib))):
            val, i, j = contrib[ci]
            if val < 1e-12:
                break
            for target in non_contrib[:6]:
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[min(i, j)] if i != j else frac * a[i]
                    if delta < 1e-15:
                        continue
                    src = min(i, j)
                    a[src] -= delta
                    a[target] += delta
                    new_obj = 2.0 * n * float(np.max(compute_conv(a)))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        v = a[0]
        return np.full(new_n, v)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)**1.5
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.05))
        a = a / a.sum()
    elif mode == 4:
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x - np.pi))
        a = a / a.sum()
    else:
        # Gaussian center
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2 / (2 * 0.3**2))
        a = a / a.sum()
    return np.maximum(a, 0)
