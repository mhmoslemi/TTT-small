1. Multiplicative Weights Update Algorithm (MWUA) for fast initial optimization
2. LP alternating with dual-variable tracking for precise optimization
3. Mirror descent with adaptive learning rates

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv_fft(a, g=None):
    """Compute linear convolution a * g using FFT. If g is None, compute a * a."""
    if g is None:
        g = a
    n = len(a)
    m = len(g)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    return irfft(fa * fg, size)[:n + m - 1]

def compute_conv_self(a):
    """Compute a * a using FFT (exploiting symmetry)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:

    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    p = a / s
    conv = compute_conv_self(p)
    return 2.0 * n * float(np.max(conv))

    n = len(a)
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    # i.e., A[k, j] = a[k-j] if j <= k < j+n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        max_k = min(2 * n - 2, j + n - 1)
        A[j:max_k+1, j] = a[:max_k - j + 1]
    return A

def mwua_optimize(a, n_iter=5000, eta=None, verbose=False):
    """
    Multiplicative Weights Update Algorithm for the zero-sum game:
    min_g max_k (A g)_k  vs  max_lam min_j (A^T lam)_j
    where A is the convolution matrix.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s  # normalize
    
    # Build A^T explicitly for efficiency (or use FFT)
    # Actually, let's use FFT for both A@lam and A^T@lam (= cross-correlation)
    
    # For A @ g = conv(a, g): use FFT
    # For A^T @ lam: (A^T lam)_j = sum_k A[k,j] * lam_k = sum_k a[k-j] * lam_k (where valid)
    # This is cross-correlation of a and lam
    
    # Precompute FFT of a
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = rfft(a, size)
    fa_conj = np.conj(fa)
    
    # Initialize
    g = np.ones(n) / n
    lam = np.ones(2 * n - 1) / (2 * n - 1)
    
    if eta is None:
        eta = 1.0 / np.sqrt(n_iter)  # will adapt
    
    best_g = g.copy()
    best_obj = compute_obj(g)
    
    # Running average of game value
    value_sum = 0.0
    
    for t in range(n_iter):
        # Compute A @ lam = cross-correlation of a and lam
        # (A^T lam)_j = sum_k a[k-j] * lam_k where 0<=k<2n-1 and 0<=k-j<n
        # = sum_m a[m] * lam[m+j] where 0<=m<n and 0<=m+j<2n-1
        # This is the cross-correlation r[j] = sum_m a[m] * l[m+j]
        # r = IFFT(conj(FFT(a)) * FFT(l))  [verified above]
        flam = rfft(lam, size)
        r = irfft(fa_conj * flam, size)[:n]
        
        # Update g: g_j = exp(-eta * r_j) / Z
        # For numerical stability, subtract max
        r_shifted = r - r.max()
        g_new = np.exp(eta * r_shifted)  # Wait, sign...
        
        # Player 1 (g) minimizes: should put LESS weight on high-cost strategies
        # Cost for g at strategy j given lam: (A^T lam)_j = r[j]
        # So g_j should be proportional to exp(-eta * r[j])
        g_new = np.exp(-eta * (r - r.min()))  # shift for stability
        g_sum = g_new.sum()
        if g_sum < 1e-300:
            g_new = np.ones(n) / n
        else:
            g_new /= g_sum
        g = g_new
        
        # Compute A @ g = conv(a, g)
        conv_ag = compute_conv_fft(a, g)
        
        # Update lam: lam_k = exp(+eta * conv_ag_k) / Z
        # Player 2 (lam) maximizes: should put MORE weight on high-payoff strategies
        # Payoff for lam at strategy k given g: (A g)_k = conv_ag[k]
        # So lam_k should be proportional to exp(+eta * conv_ag[k])
        lam_new = np.exp(eta * (conv_ag - conv_ag.min()))
        lam_sum = lam_new.sum()
        if lam_sum < 1e-300:
            lam_new = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            lam_new /= lam_sum
        lam = lam_new
        
        # Track game value
        value = float(g @ conv_ag)  # = float(lam @ r) should be same
        value_sum += value
        
        if verbose and (t + 1) % 1000 == 0:
            obj = compute_obj(g)
            if obj < best_obj:
                best_obj = obj
                best_g = g.copy()
            print(f"    MWUA it{t+1}: avg_val={value_sum/(t+1):.8f}, obj={obj:.8f}, best={best_obj:.8f}")
        
        # Adaptive eta: decrease over time
        eta *= 0.9999
        
        # Track best
        obj = compute_obj(g)
        if obj < best_obj:
            best_obj = obj
            best_g = g.copy()
    
    return best_g

def mwua_optimize_v2(a, n_iter=10000, verbose=False):
    """
    Improved MWUA with:
    - Exponential annealing of eta
    - Softmax temperature scheduling
    - Better numerical stability
    """
    n = len(a)
    m = 2 * n - 1
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = rfft(a, size)
    fa_conj = np.conj(fa)
    
    g = np.ones(n) / n
    lam = np.ones(m) / m
    
    # Temperature schedule: start hot, cool down
    T_start = 1.0
    T_end = 0.01
    best_g = g.copy()
    best_obj = compute_obj(g)
    
    for t in range(n_iter):
        frac = t / max(1, n_iter - 1)
        T = T_start * (T_end / T_start) ** frac  # exponential cooling
        
        # Compute A^T @ lam
        flam = rfft(lam, size)
        r = irfft(fa_conj * flam, size)[:n]
        
        # Update g with temperature T
        r_norm = (r - r.min()) / (r.max() - r.min() + 1e-30)
        g_new = np.exp(-r_norm / T)
        g_sum = g_new.sum()
        if g_sum < 1e-300 or not np.isfinite(g_sum):
            g_new = np.ones(n) / n
        else:
            g_new /= g_sum
        g = g_new
        
        # Compute A @ g
        conv_ag = compute_conv_fft(a, g)
        
        # Update lam with temperature T
        c_norm = (conv_ag - conv_ag.min()) / (conv_ag.max() - conv_ag.min() + 1e-30)
        lam_new = np.exp(c_norm / T)
        lam_sum = lam_new.sum()
        if lam_sum < 1e-300 or not np.isfinite(lam_sum):
            lam_new = np.ones(m) / m
        else:
            lam_new /= lam_sum
        lam = lam_new
        
        # Track
        obj = compute_obj(g)
        if obj < best_obj:
            best_obj = obj
            best_g = g.copy()
        
        if verbose and (t + 1) % 2000 == 0:
            print(f"    MWUA2 it{t+1}: T={T:.4f}, obj={obj:.8f}, best={best_obj:.8f}")
    
    return best_g

def lp_solve_minpeak(a, warm_start_lam=None):
    """
    Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt, lam_dual).
    """
    n = len(a)
    m = 2 * n - 1
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)  # shape (m, n)
    
    # LP: min M
    # s.t. A @ g - M <= 0  (m constraints)
    #      sum(g) = 1
    #      g >= 0
    #      M >= 0 (implicit since A >= 0, g >= 0)
    
    # Variables: [g_0, ..., g_{n-1}, M]  (n+1 variables)
    # Objective: minimize M -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A @ g - M <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None, warm_start_lam
    
    g_opt = np.maximum(res.x[:n], 0)
    s = float(np.sum(g_opt))
    if s < 1e-10:
        return None, None, warm_start_lam
    g_opt /= s
    M_opt = float(res.x[n])
    
    # Extract dual variables for the inequality constraints
    # In HiGHS, res.ineqlin.marginals gives the dual variables
    lam_dual = None
    if hasattr(res, 'ineqlin') and res.ineqlin is not None:
        if hasattr(res.ineqlin, 'marginals'):
            lam_dual = np.maximum(res.ineqlin.marginals, 0)
            lam_sum = lam_dual.sum()
            if lam_sum > 1e-10:
                lam_dual /= lam_sum  # normalize to probability
    
    return g_opt, M_opt, lam_dual

def lp_alternating_v2(a, time_budget=100.0, verbose=False, max_iters=200):
    """
    LP alternating with:
    - Dual variable tracking for warm-starting
    - Adaptive line search
    - Momentum on the step
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)
    it = 0
    momentum_t = 0.5  # step size momentum
    
    while time.monotonic() - start < time_budget and it < max_iters:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break
        
        # Solve LP
        g_lp, M_lp, lam_dual = lp_solve_minpeak(a)
        if g_lp is None:
            break
        
        # Line search: try several step sizes
        ts = [momentum_t, 0.3, 0.5, 0.7, 0.9, 0.1, 0.2]
        ts = sorted(set([t for t in ts if 0.05 <= t <= 1.0]))
        
        best_t_obj = np.inf
        best_t_val = momentum_t
        best_a_new = a.copy()
        
        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_new = a_try.copy()
        
        it += 1
        
        if best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            a = best_a
            momentum_t = best_t_val
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP2 it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 10:
                # Escape: perturb and restart
                noise = 0.05 + 0.02 * (stall // 10)
                a = best_a * np.exp(rng.normal(0, noise, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                momentum_t = 0.5
    
    return best_a

def mirror_descent_v2(a, time_budget=60.0, verbose=False):
    """
    Mirror descent with:
    - Softmax smoothing of the max (log-sum-exp)
    - Adaptive temperature
    - Momentum
    """
    n = len(a)
    m = 2 * n - 1
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    T = 10.0  # temperature for log-sum-exp
    lr = 1.0 / np.sqrt(n)
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv_self(a)
        max_conv = conv.max()
        
        # Log-sum-exp approximation of max
        z = conv / T
        z -= z.max()
        log_sum_exp = z.max() + np.log(np.exp(z).sum())
        smooth_max = max_conv + T * np.log(np.exp((conv - max_conv) / T).sum())
        
        # Gradient of smooth_max w.r.t. a
        # d/d a_j of smooth_max = (1/T) * sum_k softmax_k * d(conv_k)/da_j
        softmax_w = np.exp((conv - max_conv) / T)
        softmax_w /= softmax_w.sum()
        
        # d(conv_k)/da_j = a[k-j] + a[j-k] (for valid indices)
        # Actually conv(a,a)[k] = sum_j a[j]*a[k-j], so d/da_j = a[k-j] (one term)
        # Wait: conv(a,a)[k] = sum_j a[j] * a[k-j]
        # d/da_j = a[k-j] + (if j != k-j) a[k-j]... 
        # No: d/da_j of sum_i a[i]*a[k-i] = a[k-j] + a[k-j] (if j != k-j) or 2*a[k-j] (if j = k-j)
        # Actually: d/da_j [sum_i a[i]*a[k-i]] = a[k-j] + a[k-j] only if both terms involve a_j
        # Term i=j: a[j]*a[k-j], derivative = a[k-j]
        # Term i=k-j: a[k-j]*a[j], derivative = a[j] (if k-j != j)
        # So if j != k-j: derivative = a[k-j] + a[j]
        # If j == k-j (i.e., k = 2j): derivative = 2*a[j]
        
        # For efficiency, use the fact that the gradient of conv(a,a)[k] w.r.t. a is:
        # [a[k-1], a[k-2], ..., a[k-(n-1)]] (padded appropriately)
        # Specifically, grad_k[j] = a[k-j] if 0 <= k-j < n, else 0
        # PLUS a[j-k] if 0 <= j-k < n, else 0
        # Wait no. Let me be more careful.
        
        # conv(a,a)[k] = sum_j a[j] * a[k-j]  (where 0<=j<n and 0<=k-j<n)
        # d/da_j = a[k-j] (from the term where the first factor is a[j])
        #         + a[k-j] (from the term where the second factor is a[j], i.e., i+j=k, so i=k-j)
        # Wait, that's the same term! Let me redo.
        
        # conv(a,a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
        # d/da_m = sum_{j: 0<=j<n, 0<=k-j<n} [delta(j,m) * a[k-j] + a[j] * delta(k-j,m)]
        #        = a[k-m] (if 0<=k-m<n) + a[m-k] (if 0<=m-k<n)
        
        # So grad_k[m] = a[k-m] (if 0<=k-m<n) + a[m-k] (if 0<=m-k<n)
        # Note: if k-m = m-k (i.e., k=2m), these are the same and we get 2*a[k-m]
        
        # For the weighted sum: grad = sum_k softmax_w[k] * grad_k
        # grad[m] = sum_k softmax_w[k] * (a[k-m] + a[m-k]) (where valid)
        #         = sum_k softmax_w[k] * a[k-m] + sum_k softmax_w[k] * a[m-k]
        #         = (softmax_w conv a)[m] + (softmax_w reversed conv a)[m]
        
        # Actually, let's just compute it directly for the weighted gradient:
        # grad[m] = sum_k w[k] * (a[k-m] if 0<=k-m<n else 0 + a[m-k] if 0<=m-k<n else 0)
        
        grad = np.zeros(n)
        for k in range(m):
            w = softmax_w[k]
            if w < 1e-10:
                continue
            # a[k-m] for m = max(0, k-n+1) to min(n-1, k)
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            if m_start <= m_end:
                grad[m_start:m_end+1] += w * a[k - m_end:k - m_start + 1][::-1]
            # a[m-k] for m = max(k, 0) to min(n-1, k+n-1)
            m_start2 = max(0, k)
            m_end2 = min(n - 1, k + n - 1)
            if m_start2 <= m_end2:
                grad[m_start2:m_end2+1] += w * a[:m_end2 - m_start2 + 1]
        
        # Update with momentum
        v = momentum * v + grad
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            T = max(1.0, T * 0.999)  # cool down
        else:
            stall += 1
            if stall > 50:
                T = min(50.0, T * 1.5)  # heat up
                lr = max(0.01, lr * 0.8)
                stall = 0
            if stall > 200:
                # Full restart with perturbation
                rng = np.random.default_rng(it + n)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                v = np.zeros(n)
                T = 10.0
                lr = 1.0 / np.sqrt(n)
                stall = 0
        
        it += 1
        if verbose and it % 200 == 0:
            print(f"    MD2 it{it}: obj={best_obj:.8f}, T={T:.2f}, lr={lr:.4f}, el={time.monotonic()-start:.1f}s")
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def coordinate_refine_fast(a, time_budget=30.0):
    """
    Fast coordinate descent: find the peak, then try moving mass from
    contributing positions to non-contributing positions.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv_self(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        
        # Find top contributing pairs for the peak
        # conv[k_star] = sum_j a[j] * a[k_star - j]
        pairs = []
        for j in range(n):
            k = k_star - j
            if 0 <= k < n:
                pairs.append((a[j] * a[k], j, k))
        pairs.sort(reverse=True)
        
        # Find non-contributing positions
        contrib_positions = set()
        for _, j, k in pairs:
            contrib_positions.add(j)
            contrib_positions.add(k)
        
        non_contrib = [i for i in range(n) if i not in contrib_positions]
        
        if not non_contrib:
            break
        
        improved = False
        for val, j, k in pairs[:10]:
            if val < 1e-12:
                break
            for target in non_contrib[:5]:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = frac * a[j]
                    if delta < 1e-15:
                        continue
                    a[j] -= delta
                    a[target] += delta
                    new_conv = compute_conv_self(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[j] += delta
                        a[target] -= delta
                        break
        
        if not improved:
            break
    
    return best_a

def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: MWUA pre-optimize -> LP alternating -> Mirror descent -> 
    Multi-n exploration -> Final polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 45.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")
    
    # === Phase 1: MWUA (fast, ~30s) ===
    print(f"\n=== Phase 1: MWUA (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.05, 60.0)
    if p1_budget > 10:
        n_iter_mwua = int(5000 * p1_budget / 30.0)
        a_mwua = mwua_optimize_v2(best_seq, n_iter=n_iter_mwua, verbose=True)
        obj_mwua = compute_obj(a_mwua)
        if obj_mwua < best_score:
            best_score = obj_mwua
            best_seq = a_mwua.copy()
            print(f"  MWUA: obj={best_score:.8f} **")
        else:
            print(f"  MWUA: obj={obj_mwua:.8f} (no improvement)")
    
    # === Phase 2: LP Alternating (main workhorse, ~60% of time) ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p2_total = min(remaining * 0.5, 700.0)
    
    n_rounds = 4
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p2_total / (n_rounds - rnd), remaining * 0.55)
        if alloc < 20:
            break
        
        if rnd > 0:
            noise_scale = 0.02 * rnd
            a_pert = best_seq * np.exp(rng.normal(0, noise_scale, len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        else:
            a_pert = best_seq.copy()
        
        a_lp = lp_alternating_v2(a_pert, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")
