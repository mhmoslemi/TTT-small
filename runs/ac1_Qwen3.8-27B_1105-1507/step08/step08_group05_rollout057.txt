The dual variables y give us a "measure" on the lags. The constraint A'y ≤ 1 means sum_k y[k]*a[k-j] ≤ 1 for all j.

At the optimum, the complementary slackness condition says: x[j] > 0 implies (A'y)[j] = 1.

This gives us a way to characterize the optimal solution: the optimal g has support only on indices j where sum_k y[k]*a[k-j] = 1.

"""
Optimizer using exact coordinate descent + LP alternating + multi-scale search.
Key innovation: exact 1D coordinate updates with incremental convolution tracking,
combined with LP-based direction finding and aggressive n-sweeping.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import cvxpy as cp
import time


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0. Returns (g_norm, M_opt) or (None, None)."""
    n = len(a)
    if n < 2:
        return None, None
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def coord_descent(a_init, time_budget=60.0, n_grid=25, verbose=False):
    """
    Exact coordinate descent with incremental convolution updates.
    For each coordinate, solve the 1D subproblem via grid + refinement.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a_init.copy()
    a = a / s
    s = 1.0
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv))
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    sweep = 0
    stall = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        sweep_improved = False
        # Randomize coordinate order periodically
        if sweep % 10 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            i = int(i)
            v_old = a[i]
            if v_old < 1e-15:
                continue

            # Find best delta for coordinate i
            # conv_new(k) = conv(k) + 2*delta*a[k-i] + delta^2*I(2i=k)
            # obj(delta) = 2n * max(conv_new) / (s+delta)^2, with s=1
            best_d = 0.0
            best_f = 2.0 * n * float(np.max(conv))

            # Precompute unaffected max
            if i > 0 and i + n < 2 * n - 1:
                unaffected_max = max(float(np.max(conv[:i])), float(np.max(conv[i + n:])))
            elif i > 0:
                unaffected_max = float(np.max(conv[:i]))
            elif i + n < 2 * n - 1:
                unaffected_max = float(np.max(conv[i + n:]))
            else:
                unaffected_max = 0.0

            # Grid search
            d_min = -v_old * 0.999
            d_max = v_old * 2.0
            grid = np.linspace(d_min, d_max, n_grid)
            for d in grid:
                s_new = s + d
                if s_new < 0.01:
                    continue
                affected = conv[i:i + n] + 2.0 * d * a
                if 2 * i < 2 * n - 1:
                    affected[i] += d * d
                max_aff = float(np.max(affected))
                max_cn = max(max_aff, unaffected_max)
                f = 2.0 * n * max_cn / (s_new * s_new)
                if f < best_f:
                    best_f = f
                    best_d = d

            # Refine around best_d if it's not on the boundary
            if best_d > d_min + 1e-6 and best_d < d_max - 1e-6:
                lo = best_d - (d_max - d_min) / n_grid
                hi = best_d + (d_max - d_min) / n_grid
                for d in np.linspace(lo, hi, 15):
                    s_new = s + d
                    if s_new < 0.01:
                        continue
                    affected = conv[i:i + n] + 2.0 * d * a
                    if 2 * i < 2 * n - 1:
                        affected[i] += d * d
                    max_aff = float(np.max(affected))
                    max_cn = max(max_aff, unaffected_max)
                    f = 2.0 * n * max_cn / (s_new * s_new)
                    if f < best_f:
                        best_f = f
                        best_d = d

            # Apply update
            v_new = max(0.0, v_old + best_d)
            actual_d = v_new - v_old
            if abs(actual_d) > 1e-15:
                # Update convolution incrementally
                conv[i:i + n] += 2.0 * actual_d * a
                if 2 * i < 2 * n - 1:
                    conv[2 * i] += actual_d * actual_d
                a[i] = v_new
                s += actual_d
                if best_f < best_obj:
                    best_obj = best_f
                    best_a = a.copy()
                    sweep_improved = True

        # Periodically recompute conv from scratch to avoid drift
        if sweep % 50 == 0:
            conv = compute_conv(a)
            s = float(np.sum(a))

        if not sweep_improved and sweep > 5:
            stall += 1
            if stall > 8:
                break
        else:
            stall = 0

        if verbose and sweep % 20 == 0:
            print(f"    CD sweep={sweep}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a

                        best_t = t
        except Exception:
            best_t = 0.5
            best_t_obj = best_obj

        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))

        else:

def resample_seq(a, new_n):
    """Cubic interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use numpy's interp (linear) - simple and robust
    result = np.interp(x_new, x_old, a)
    return np.maximum(result, 0)


def raised_cosine(n, power=1.0):
    """Raised cosine window: (0.5*(1-cos(2*pi*t)))^power, normalized."""
    t = np.linspace(0, 1, n)
    w = (0.5 * (1.0 - np.cos(2.0 * np.pi * t))) ** power
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def gaussian_window(n, sigma_frac=0.25):
    """Gaussian window centered at 0.5, normalized."""
    t = np.linspace(0, 1, n)
    w = np.exp(-0.5 * ((t - 0.5) / sigma_frac) ** 2)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def asymmetric_window(n, alpha=0.3):
    """Asymmetric window: rises slowly, drops quickly."""
    t = np.linspace(0, 1, n)
    w = np.exp(-((t - alpha) / (0.3 * (1.0 - alpha))) ** 2) * (t < 0.8)
    w = np.maximum(w, 1e-10)
    sw = np.sum(w)
    return w / sw


def mixed_window(n, frac=0.4):
    """Mix of raised cosine and asymmetric."""
    rc = raised_cosine(n, power=1.5)
    aw = asymmetric_window(n, alpha=0.35)
    w = (1 - frac) * rc + frac * aw
    return w / np.sum(w)

    n_base = max(len(best_seq), 800)

    inits = []
    inits.append(("parent", best_seq.copy()))
    inits.append(("hann1", raised_cosine(n_base, 1.0)))
    inits.append(("hann2", raised_cosine(n_base, 2.0)))
    inits.append(("hann3", raised_cosine(n_base, 3.0)))
    inits.append(("gauss", gaussian_window(n_base, 0.25)))
    inits.append(("gauss_wide", gaussian_window(n_base, 0.35)))
    inits.append(("asym", asymmetric_window(n_base, 0.35)))
    inits.append(("mixed", mixed_window(n_base, 0.3)))
    inits.append(("tri", None))  # triangular
    tri = np.minimum(np.arange(n_base, dtype=float) + 1, n_base - np.arange(n_base))
    inits[-1] = ("tri", tri / np.sum(tri))
    inits.append(("rand", rng.random(n_base) / (rng.random(n_base) + 1e-10)))
    inits[-1] = ("rand", inits[-1][1] / np.sum(inits[-1][1]))

    for name, init in inits:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        init = np.maximum(init, 0)
        si = float(np.sum(init))
        if si < 1e-10:
            continue
        init = init / si
        obj = compute_obj(init)
        if obj < best_score:
            best_score = obj
            best_seq = init.copy()
            print(f"  {name} (n={n_base}): obj={best_score:.8f} **")
        else:
            print(f"  {name} (n={n_base}): obj={obj:.8f}")
        if remaining > 40:
            alloc = min(15.0, remaining * 0.15)
            a_opt = coord_descent(init, time_budget=alloc)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  {name}+CD: obj={best_score:.8f} **")

    # === Phase 2: n-sweep (200s) ===
    print(f"\n=== Phase 2: n-Sweep ===")
    n_options = [400, 600, 800, 1000, 1200, 1500, 2000, 2500, 3000, 4000]
    n_options = sorted(set(n_options + [len(best_seq), max(200, len(best_seq) - 200),
                                         min(5000, len(best_seq) + 200)]))
    n_options = [n for n in n_options if 200 <= n <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        n_try = int(n_try)
        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
        # Also try different shapes at this n
        shapes = [a_try, raised_cosine(n_try, 1.5), raised_cosine(n_try, 2.5),
                  gaussian_window(n_try, 0.3)]
        best_at_n = a_try
        best_obj_n = compute_obj(a_try)
        for sh in shapes:
            sh = np.maximum(sh, 0)
            ss = float(np.sum(sh))
            if ss < 1e-10:
                continue
            sh = sh / ss
            o = compute_obj(sh)
            if o < best_obj_n:
                best_obj_n = o
                best_at_n = sh
        alloc = min(15.0, remaining * 0.2)
        a_opt = coord_descent(best_at_n, time_budget=alloc)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Main LP Alternating (300s) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    a = best_seq.copy()
    for round_idx in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(30.0, remaining * 0.25)
        a_new = lp_alternate(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new
            print(f"  LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj:.8f}")
            a = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
            a = np.maximum(a, 0)
            sa = float(np.sum(a))
            if sa > 1e-10:
                a = a / sa

    # === Phase 4: Coordinate Descent Main (400s) ===
    print(f"\n=== Phase 4: Coordinate Descent (main) ===")
    remaining = deadline - time.monotonic()
    alloc = min(350.0, remaining * 0.5)
    a_cd = coord_descent(best_seq, time_budget=alloc, n_grid=30, verbose=True)
    obj_cd = compute_obj(a_cd)
    if obj_cd < best_score:
        best_score = obj_cd
        best_seq = a_cd.copy()
        print(f"  CD main: obj={best_score:.8f} **")
    else:
        print(f"  CD main: obj={obj_cd:.8f}")

    # === Phase 5: LP + CD交替 (200s) ===
    print(f"\n=== Phase 5: LP + CD交替 ===")
    for cyc in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        # LP step
        alloc_lp = min(30.0, remaining * 0.2)
        a_lp = lp_alternate(best_seq, time_budget=alloc_lp)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Cyc{cyc} LP: obj={best_score:.8f} **")
        # CD step
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc_cd = min(40.0, remaining * 0.3)
        a_cd2 = coord_descent(best_seq, time_budget=alloc_cd, n_grid=30)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Cyc{cyc} CD: obj={best_score:.8f} **")

    # === Phase 6: Random restarts (remaining) ===
    print(f"\n=== Phase 6: Random Restarts ===")
    n_best = len(best_seq)
    for r in range(15):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        scale = 0.01 + 0.005 * r
        a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
        a_r = np.maximum(a_r, 0)
        sr = float(np.sum(a_r))
        if sr < 1e-10:
            continue
        a_r = a_r / sr
        alloc = min(25.0, remaining * 0.25)
        # Quick CD
        a_opt = coord_descent(a_r, time_budget=alloc * 0.6, n_grid=20)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  R{r} (scale={scale:.3f}): obj={best_score:.8f} **")
        else:
            # Try LP on it
            remaining = deadline - time.monotonic()
            if remaining > 20 and alloc * 0.3 > 5:
                a_lp_r = lp_alternate(a_opt, time_budget=min(15.0, remaining * 0.2))
                obj_lp_r = compute_obj(a_lp_r)
                if obj_lp_r < best_score:
                    best_score = obj_lp_r
                    best_seq = a_lp_r.copy()
                    print(f"  R{r}+LP: obj={best_score:.8f} **")

    # === Phase 7: Final CD polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.7, 100.0)
        a_final = coord_descent(best_seq, time_budget=alloc, n_grid=40, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # === Phase 8: Try one more LP on the final result ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 40.0)
        a_final_lp = lp_alternate(best_seq, time_budget=alloc)
        obj_flp = compute_obj(a_final_lp)
        if obj_flp < best_score:
            best_score = obj_flp
            best_seq = a_final_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation and return
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
