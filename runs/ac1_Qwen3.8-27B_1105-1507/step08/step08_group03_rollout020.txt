def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def smooth_lse_optimize(a, time_budget=120.0, T_init=50.0, T_final=5.0, verbose=True):
    """
    Smooth optimization using log-sum-exp with annealing temperature.
    Uses L-BFGS-B with multiple restarts at different temperatures.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_grad(x, T):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = compute_conv(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        z -= z.max()  # numerical stability
        e = np.exp(z)
        sum_e = e.sum()
        if sum_e < 1e-300:
            return 2.0 * n * max_c, np.zeros(len(x))
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)
        w = e / sum_e
        g = np.zeros(n)
        # Only compute gradient for significant weights
        threshold = 1e-8
        active = w > threshold
        for k in np.where(active)[0]:
            g += w[k] * grad_conv_at_peak(x, int(k))
        g *= 2.0 * n
        return obj_val, g

    # Try a range of temperatures
    for T_val in [T_init, T_init * 0.5, T_final, max(T_final * 0.5, 1.0)]:
        elapsed = time.monotonic() - start
        if elapsed > time_budget * 0.7:
            break

        try:
            res = sp_minimize(lambda x: obj_grad(x, T_val), a, method='L-BFGS-B',
                              bounds=[(1e-15, None)] * n,
                              options={'maxiter': 300, 'maxfun': 1000, 'disp': False})
            a_opt = np.maximum(res.x, 0)
            s_opt = float(np.sum(a_opt))
            if s_opt > 1e-10:
                a_opt = a_opt / s_opt
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_obj:
                    best_obj = obj_opt
                    best_a = a_opt.copy()
                    if verbose:
                        print(f"    LSE T={T_val:.1f}: obj={best_obj:.8f} **")
        except Exception:
            pass

    return best_a


def subgradient_accelerated(a, time_budget=120.0, K=7, verbose=True):
    """
    Accelerated subgradient method with adaptive step size and momentum.
    Uses multiple peaks with temperature-based weighting.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov momentum
    alpha = 2.0 / (2.0 + 1.0)
    v = np.zeros(n)
    momentum = 0.0
    stall = 0
    lr = 4.0 / np.sqrt(n)
    rng = np.random.default_rng(999)

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv[0]
        for k in range(1, len(conv)):
            if conv[k] > max_val:
                max_val = conv[k]

        if max_val < 1e-15:
            break

        # Temperature-based weighting over all peaks
        temp = 5.0 / (1.0 + it * 0.001)  # slowly decreasing
        weights = np.exp((conv - max_val) / (temp * max_val + 1e-15))
        total_w = weights.sum()
        if total_w < 1e-300:
            break
        weights /= total_w

        # Compute subgradient
        g = np.zeros(n)
        active = weights > 1e-6
        for k in np.where(active)[0]:
            g += weights[k] * grad_conv_at_peak(a, int(k))

        # Adaptive step size
        obj_gap = best_obj - 2.0 * n / (2.0 * n - 1.0)  # gap to theoretical lower bound
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq > 1e-20:
            lr_adapt = max(1e-4, min(lr, obj_gap / g_norm_sq * 0.1))
        else:
            lr_adapt = lr

        # Momentum update
        v = momentum * v + lr_adapt * g
        a_new = a - v
        a_new = project_simplex(a_new)

        # Nesterov extrapolation
        alpha_new = 2.0 / (2.0 + 1.0 + np.sqrt(1.0 + 4.0 * (alpha + it * 0.1) ** 2))
        a = np.maximum(a_new + alpha * (a_new - a), 0)
        s = float(np.sum(a))
        if s > 1e-10:
            a = a / s
        alpha = alpha_new

        obj_new = compute_obj(a)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            stall = 0
            momentum = min(0.85, momentum + 0.003)
            lr = max(1e-3, lr * 0.998)
        else:
            stall += 1
            if stall > 60:
                # Restart with perturbation
                a = best_a * np.exp(rng.normal(0, 0.1 + 0.02 * (stall // 60), n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = 0.0
                lr = 4.0 / np.sqrt(n)
                stall = 0

        it += 1

        if verbose and it % 100 == 0:
            print(f"    SubGrad it{it}: obj={best_obj:.8f}, lr={lr:.4f}, el={time.monotonic()-start:.1f}s")

    return best_a

                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
            if improved:
                break

        if not improved:
            # Try the second peak too
            conv_sorted = np.argsort(conv)[::-1]
            k_star2 = int(conv_sorted[1])
            if conv[k_star2] / peak_val > 0.9:
                contrib2 = []
                for i in range(n):
                    j = k_star2 - i
                    if 0 <= j < n:
                        val = a[i] * a[j]
                        if val > 1e-15:
                            contrib2.append((val, i, j))
                contrib2.sort(reverse=True)
                non_contrib2 = [i for i in range(n) if not (0 <= k_star2 - i < n)]
                for ci, (val, i, j) in enumerate(contrib2[:5]):
                    for target in non_contrib2[:4]:
                        for frac in [0.01, 0.03]:
                            src = i if a[i] >= a[j] else j
                            delta = frac * a[src]
                            if delta < 1e-15:
                                continue
                            a[src] -= delta
                            a[target] += delta
                            new_conv = compute_conv(a)
                            new_obj = 2.0 * n * float(np.max(new_conv))
                            if new_obj < best_obj - 1e-14:
                                best_obj = new_obj
                                best_a = a.copy()
                                improved = True
                                break
                            else:
                                a[src] += delta
                                a[target] -= delta
                                break
                    if improved:
                        break
            if not improved:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two-end concentration
        x = np.linspace(0, 1, n)
        a = x**2 + (1 - x)**2 + 0.01
        a = a / a.sum()
    elif mode == 3:
        # Periodic modulation
        x = np.linspace(0, 1, n)
        freq = rng.uniform(2, 8)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Gaussian concentration at random position
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(n / 6, n / 3)
        x = np.arange(n)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Alternating pattern
        a = np.ones(n)
        a[::2] = rng.uniform(0.3, 1.0)
        a = a / a.sum()
    elif mode == 6:
        # Triangle
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def spectral_init(n, rng, n_iter=200):
    """
    Initialize using iterative amplitude flattening in Fourier domain.
    Tries to make |A(w)|^2 as flat as possible (constant power spectrum).
    """
    a = np.ones(n) / n
    # Target: flat power spectrum
    # Iterate: a -> proj_pos(IDFT(sqrt(|DFT(a)|^2 / mean * phase)))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    for _ in range(n_iter):
        fa = rfft(a, size)
        power = np.abs(fa) ** 2
        target_amp = np.sqrt(power / (power.mean() + 1e-30))
        # Preserve phase
        phase = fa / (np.abs(fa) + 1e-30)
        fa_new = target_amp * phase
        a_new = irfft(fa_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a = a_new / s
        else:
            break
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid optimization combining primal-dual LP,
    accelerated subgradient, smooth LSE refinement, spectral init, and
    multi-n exploration with coordinate descent polish.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Spectral initialization + quick optimization ===
    print(f"\n=== Phase 1: Spectral Init + Quick Opt ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a_spec = spectral_init(n_target, rng, n_iter=300)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral init: obj={best_score:.8f} **")
        else:
            print(f"  Spectral init: obj={obj_spec:.8f} (no improvement)")
            # Blend spectral with best
            a_blend = 0.5 * best_seq + 0.5 * a_spec
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            obj_blend = compute_obj(a_blend)
            if obj_blend < best_score:
                best_score = obj_blend
                best_seq = a_blend.copy()
                print(f"  Blend: obj={best_score:.8f} **")

    # === Phase 2: Accelerated Subgradient ===
    print(f"\n=== Phase 2: Accelerated Subgradient ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.1, 100.0)
    if p2_budget > 20:
        a_sg = subgradient_accelerated(best_seq, time_budget=p2_budget, K=7, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SubGrad: obj={best_score:.8f} **")

    # === Phase 3: Primal-Dual LP Alternating (main workhorse) ===
    print(f"\n=== Phase 3: Primal-Dual LP Alternating ===")
    remaining = deadline - time.monotonic()
    p3_total = min(remaining * 0.45, 500.0)

    n_rounds = 3
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p3_total / (n_rounds - rnd), remaining * 0.6)
        if alloc < 20:
            break

        if rnd > 0:
            a_pert = best_seq * np.exp(rng.normal(0, 0.02 * (rnd + 1), len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        else:
            a_pert = best_seq.copy()

        a_lp = lp_primal_dual_alternating(a_pert, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")

    # === Phase 4: Smooth LSE Refinement ===
    print(f"\n=== Phase 4: Smooth LSE Refinement ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 120.0)
    if p4_budget > 20:
        a_lse = smooth_lse_optimize(best_seq, time_budget=p4_budget, T_init=50.0, T_final=5.0, verbose=True)
        obj_lse = compute_obj(a_lse)
        if obj_lse < best_score:
            best_score = obj_lse
            best_seq = a_lse.copy()
            print(f"  LSE: obj={best_score:.8f} **")

    # === Phase 5: Multi-n Exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(100, n_current - 200),
        max(100, n_current - 100),
        max(100, n_current - 50),
        n_current + 50,
        n_current + 100,
        n_current + 200,
        n_current + 400,
        n_current + 800,
    ]))
    n_options = [n for n in n_options if 100 <= n <= 3000]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= 8:
            break
        if n_try == n_current or n_try < 50:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        # Quick subgradient
        alloc_sg = min(remaining * 0.08, 30.0)
        if alloc_sg > 10:
            a_try = subgradient_accelerated(a_try, time_budget=alloc_sg, K=3, verbose=False)

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.08, 40.0)
        if alloc_lp > 10:
            a_try = lp_primal_dual_alternating(a_try, time_budget=alloc_lp, verbose=False)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse Restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = r % 7
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.15 + 0.08 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.12, 50.0)
        a_opt = lp_primal_dual_alternating(a_r, time_budget=restart_budget, verbose=False)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final Intensive Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 40:
        lp_alloc = min(remaining * 0.3, 100.0)
        a_lp = lp_primal_dual_alternating(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Subgradient polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        sg_alloc = min(remaining * 0.25, 60.0)
        a_sg = subgradient_accelerated(best_seq, time_budget=sg_alloc, K=10, verbose=False)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SubGrad: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd_alloc = min(remaining * 0.25, 40.0)
        a_cd = coordinate_descent_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_last = lp_primal_dual_alternating(best_seq, time_budget=min(remaining * 0.5, 60.0), verbose=True)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
