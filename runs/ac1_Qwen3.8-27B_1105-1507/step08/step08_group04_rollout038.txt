import numpy as np

def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a_fixed):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a_fixed, g)."""
    n = len(a_fixed)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        length = k_end - k_start + 1
        A[k_start:k_end + 1, j] = a_fixed[:length]
    return A


def solve_lp_with_cuts(cuts, verbose=False):
    """
    Solve LP: min M s.t. 2*conv(cut_t, a)[k] - conv(cut_t, cut_t)[k] <= M for all k, t
              sum(a) = 1, a >= 0, M >= 0
    Returns (a_normalized, M_value) or (None, None).
    """
    if not cuts:
        return None, None
    n = len(cuts[0])
    
    # Build combined constraint matrix
    all_A = []
    all_b = []
    for t_idx, (a_t, conv_t) in enumerate(cuts):
        A_t = build_conv_matrix(a_t)
        all_A.append(2.0 * A_t)
        all_b.append(conv_t)
    
    all_A = np.vstack(all_A)  # (m*(2n-1) x n)

    A_ub = np.zeros((num_constraints, n + 1))
    A_ub[:, :n] = all_A
    A_ub[:, n] = -1.0
    b_ub = all_b

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

def subgradient_step(a, top_k=None):
    """Compute subgradient of max(conv(a,a)) at a (sum(a)=1)."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    # Active set: positions where conv is close to max
    if top_k is not None and top_k < len(conv):
        k_indices = np.argsort(conv)[::-1][:top_k]
    else:
        threshold = M - 1e-5 * M
        k_indices = np.where(conv >= threshold)[0]
    
    if len(k_indices) == 0:
        k_indices = np.array([np.argmax(conv)])
    
    # Gradient of f_k(a) = conv(a,a)[k] is: grad[j] = 2*a[k-j] (if 0<=k-j<n)
    # Subgradient = average of gradients over active set
    grad = np.zeros(n)
    for k in k_indices:
        k_int = int(k)
        # grad[j] += 2*a[k-j] for valid j
        j_start = max(0, k_int - n + 1)
        j_end = min(n - 1, k_int)
        for j in range(j_start, j_end + 1):
            grad[j] += 2.0 * a[k_int - j]
    
    grad /= len(k_indices)
    
    # Center for simplex projection
    grad -= np.mean(grad)
    return grad, M

    best_obj = compute_obj(a)

    lr = 1.0 / np.sqrt(n)
    
    for it in range(100000):

            break

        grad, M = subgradient_step(x, top_k=top_k)

        x_new = np.maximum(x_new, 0)

            sx = 1.0
        x_new = x_new / sx

        a_new = x_new + (1 - t_k) * (x_new - a)
        a_new = np.maximum(a_new, 0)
        sa = float(np.sum(a_new))
        if sa < 1e-10:
            a_new = x_new
            sa = 1.0
        a_new = a_new / sa
        
        x = x_new
        a = a_new
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        
        # Decrease learning rate
        if it % 100 == 0 and it > 0:
            lr *= 0.95

        if verbose and it % 5000 == 0:
            print(f"    SubGrad: it={it}, obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def lp_alternate_with_cuts(a, time_budget=200.0, max_cuts=8, verbose=False):
    """LP alternating with cut accumulation (outer approximation)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    iteration = 0
    
    # Cut pool: list of (a_t, conv(a_t, a_t))
    cuts = []
    
    while time.monotonic() - start < time_budget:
        iteration += 1
        
        # Add current point as a cut
        conv_current = compute_conv(a)
        cuts.append((a.copy(), conv_current.copy()))
        
        # Prune old cuts (keep most recent max_cuts)
        if len(cuts) > max_cuts:
            cuts = cuts[-max_cuts:]
        
        # Solve LP with all cuts
        if len(cuts) == 1:
            g_lp, M_lp = solve_single_lp(cuts[-1][0])
        else:
            g_lp, M_lp = solve_lp_with_cuts(cuts)
        
        if g_lp is None:
            # Fallback to single LP
            g_lp, M_lp = solve_single_lp(a)
            if g_lp is None:
                break
        
        # Try different blending parameters
        best_local_obj = np.inf
        best_local_a = best_a.copy()
        best_t = 0.3
        
        t_values = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.8, 1.0]
        for t in t_values:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()
                best_t = t

        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            stall = 0
            if verbose and iteration % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP+cuts: it={iteration}, obj={best_obj:.8f}, t={best_t:.2f}, cuts={len(cuts)}, time={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 5:
                # Structured perturbation
                rng = np.random.default_rng(iteration * 7919 + 12345)
                
                # Try frequency-domain perturbation
                perturbation_type = rng.integers(0, 3)
                if perturbation_type == 0:
                    # Random phase perturbation in frequency domain
                    fa = rfft(a, 2 * n)
                    phase_noise = np.exp(1j * rng.normal(0, 0.1, len(fa)))
                    fa_new = fa * phase_noise
                    a_pert = irfft(fa_new, 2 * n)[:n]
                    a_pert = np.maximum(a_pert, 0)
                elif perturbation_type == 1:
                    # Smooth random perturbation
                    scale = min(0.03 + 0.01 * (stall // 5), 0.2)
                    noise = rng.normal(0, scale, n)
                    # Smooth the noise
                    kernel = np.ones(10) / 10
                    noise = np.convolve(noise, kernel, mode='same')
                    a_pert = a * np.exp(noise)
                    a_pert = np.maximum(a_pert, 0)
                else:
                    # Periodic perturbation
                    period = rng.integers(max(2, n // 20), max(3, n // 5))
                    phase = rng.uniform(0, 2 * np.pi)
                    perturbation = 0.05 * np.sin(2 * np.pi * np.arange(n) / period + phase)
                    a_pert = a * (1 + perturbation)
                    a_pert = np.maximum(a_pert, 0)
                
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    obj_p = compute_obj(a_pert)
                    if obj_p < best_obj:
                        best_obj = obj_p
                        best_a = a_pert.copy()
                        a = best_a.copy()
                        stall = 0
                    else:
                        # Keep best but clear cuts to explore new region
                        cuts = [(a_pert.copy(), compute_conv(a_pert).copy())]
                        a = a_pert.copy()
                        stall = 0
                else:
                    a = best_a.copy()
                    cuts = [(a.copy(), compute_conv(a).copy())]
                    stall = 0

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng):
    """Generate diverse structured initialization."""
    choice = rng.integers(0, 7)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-30 * (x - 0.5) ** 2)
    elif choice == 1:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * rng.integers(2, 8) * x)
    elif choice == 2:
        c1, c2 = rng.uniform(0.2, 0.4), rng.uniform(0.6, 0.8)
        a = np.exp(-60 * (x - c1) ** 2) + np.exp(-60 * (x - c2) ** 2)
    elif choice == 3:
        a = np.ones(n)
        for _ in range(4):
            center = rng.integers(0, n)
            width = rng.integers(n // 15, n // 4)
            for i in range(n):
                d = abs(i - center)
                a[i] += rng.standard_normal() * 0.5 * np.exp(-d / width)
        a = np.maximum(a, 0.01)
    elif choice == 4:
        p = rng.integers(3, 10)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * p * x + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        # Power law with oscillation
        freq = rng.integers(1, 5)
        a = (x + 0.1) ** rng.uniform(0.3, 2.0) * (1 + 0.2 * np.sin(2 * np.pi * freq * x))
    else:
        # Pure random (smoothed)
        a = rng.uniform(0.5, 1.5, n)
        kernel_size = max(3, n // 100)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def frequency_domain_optimize(a, time_budget=60.0):
    """Try to optimize in frequency domain by flattening the power spectrum."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Zero-pad to get linear convolution via circular convolution
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, size)
        power = np.abs(fa) ** 2
        target = np.sum(power) / size  # Flat target
        
        # Blend power spectrum towards flat
        alpha = 0.01
        power_new = (1 - alpha) * power + alpha * target
        
        # Keep the DC component fixed (to maintain sum)
        power_new[0] = power[0]
        if size % 2 == 0:
            power_new[size // 2] = power[size // 2]
        
        # Recover magnitude, keep phase
        mag_new = np.sqrt(power_new)
        phase = np.angle(fa)
        fa_new = mag_new * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = irfft(fa_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            # Decrease alpha as we converge
            alpha *= 0.999
        else:
            alpha *= 0.95
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function using cut-based outer approximation LP alternating,
    subgradient polishing, frequency-domain optimization, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        print(f"Parent loaded: n={len(a)}, obj={compute_obj(a):.8f}")
    except Exception:
        a = np.ones(800) / 800
        print(f"No parent, using uniform: n=800, obj={compute_obj(a):.8f}")

    best_a = a.copy()
    best_obj = compute_obj(a)
    n_base = len(best_a)

    # === Phase 1: Cut-based LP alternating (primary optimization) ===
    print(f"\n=== Phase 1: Cut-based LP Alternating (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 300.0)
    a_p1 = lp_alternate_with_cuts(best_a, time_budget=p1_budget, max_cuts=6, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  Phase 1 result: obj={best_obj:.8f} **")
    else:
        print(f"  Phase 1: no improvement (obj={obj_p1:.8f})")

    # === Phase 2: Subgradient polish ===
    print(f"\n=== Phase 2: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        for top_k in [10, 30, 50]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(80.0, remaining * 0.25)
            a_sg = subgradient_optimize(best_a, time_budget=alloc, top_k=top_k, verbose=False)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg
                print(f"  SubGrad top_k={top_k}: obj={best_obj:.8f} **")
            else:
                print(f"  SubGrad top_k={top_k}: obj={obj_sg:.8f}")

    # === Phase 3: Frequency domain optimization ===
    print(f"\n=== Phase 3: Frequency Domain Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc_fd = min(60.0, remaining * 0.15)
        a_fd = frequency_domain_optimize(best_a, time_budget=alloc_fd)
        obj_fd = compute_obj(a_fd)
        if obj_fd < best_obj:
            best_obj = obj_fd
            best_a = a_fd
            print(f"  Freq domain: obj={best_obj:.8f} **")
        else:
            print(f"  Freq domain: obj={obj_fd:.8f}")

    # === Phase 4: LP alternating round 2 (from improved point) ===
    print(f"\n=== Phase 4: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p4_budget = min(remaining * 0.25, 200.0)
        a_p4 = lp_alternate_with_cuts(best_a, time_budget=p4_budget, max_cuts=8, verbose=True)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_obj:
            best_obj = obj_p4
            best_a = a_p4
            print(f"  Phase 4 result: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 4: no improvement (obj={obj_p4:.8f})")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        max(200, n_base - 200), max(200, n_base - 100),
        n_base,
        n_base + 100, n_base + 200, n_base + 400, n_base + 800,
    ]))
    n_options = [x for x in n_options if 200 <= x <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            a_try = a_try / np.sum(a_try)

        alloc = min(40.0, remaining * 0.2)
        a_opt = lp_alternate_with_cuts(a_try, time_budget=alloc, max_cuts=4)
        obj_try = compute_obj(a_opt)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse initializations ===
    print(f"\n=== Phase 6: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    for d in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_div = diverse_initialization(n_best, rng)
        alloc = min(30.0, remaining * 0.15)
        a_opt = lp_alternate_with_cuts(a_div, time_budget=alloc, max_cuts=4)
        obj_div = compute_obj(a_opt)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse init {d}: obj={best_obj:.8f} **")
        else:
            if d < 3:
                print(f"  Diverse init {d}: obj={obj_div:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    if remaining > 20:
        alloc_lp = min(remaining * 0.35, 80.0)
        a_final = lp_alternate_with_cuts(best_a, time_budget=alloc_lp, max_cuts=10, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_sg = min(remaining * 0.3, 40.0)
        a_sg = subgradient_optimize(best_a, time_budget=alloc_sg, top_k=20)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  Final SubGrad: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_last = min(remaining * 0.5, 30.0)
        a_last = lp_alternate_with_cuts(best_a, time_budget=alloc_last, max_cuts=10)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
