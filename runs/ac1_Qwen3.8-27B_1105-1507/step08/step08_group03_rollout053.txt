def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    n_a = len(a)
    n_b = len(b)
    total = n_a + n_b - 1
    size = 1
    while size < total:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:total]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_direction(a):
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        g_val = g_val / s
                        M_opt = float(M.value)
                        # Get dual variables for lam
                        lam = np.zeros(2 * n - 1)
                        try:
                            dual_vals = constraints[0].dual_value
                            if dual_vals is not None:
                                lam = np.maximum(dual_vals, 0)
                        except Exception:
                            pass
                        return g_val, M_opt, lam
            except Exception:
                continue

    # Fallback to scipy HiGHS
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_val = g_val / s
            M_opt = float(res.x[n])
            lam = np.zeros(2 * n - 1)
            if res.ineqlin is not None and hasattr(res.ineqlin, 'mult'):
                try:
                    lam = np.maximum(res.ineqlin.mult, 0)
                except Exception:
                    pass
            return g_val, M_opt, lam
    return None, None, None


def quadratic_line_search(C_a, C_ag, C_g, a, g):
    n = len(a)
    best_t = 0.0
    best_val = np.max(C_a)

    # Coarse grid
    ts = np.linspace(0.05, 0.95, 100)
    best_coarse_t = 0.0
    best_coarse_val = best_val

    for t in ts:
        val = (1 - t)**2 * C_a + 2 * t * (1 - t) * C_ag + t**2 * C_g
        mv = float(np.max(val))
        if mv < best_coarse_val:
            best_coarse_val = mv
            best_coarse_t = t

    # Also check t=1
    val_at_1 = float(np.max(C_g))
    if val_at_1 < best_coarse_val:
        best_coarse_val = val_at_1
        best_coarse_t = 1.0

    # Refine with golden section around best
    lo = max(0.0, best_coarse_t - 0.1)
    hi = min(1.0, best_coarse_t + 0.1)
    gr = (np.sqrt(5) - 1) / 2

    for _ in range(30):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        v1 = (1 - m1)**2 * C_a + 2 * m1 * (1 - m1) * C_ag + m1**2 * C_g
        v2 = (1 - m2)**2 * C_a + 2 * m2 * (1 - m2) * C_ag + m2**2 * C_g
        if float(np.max(v1)) < float(np.max(v2)):
            hi = m2
        else:
            lo = m1

    best_t = (lo + hi) / 2
    val = (1 - best_t)**2 * C_a + 2 * best_t * (1 - best_t) * C_ag + best_t**2 * C_g
    best_val = float(np.max(val))

    return best_t, best_val

def spectral_equalize(a, time_budget=120.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    alpha = 0.5
    it = 0
    while time.monotonic() - start < time_budget:
        A = rfft(a, size)
        power = np.abs(A)**2
        # Target: flatten the power (but keep DC)
        avg_power = np.mean(power[1:']) if n > 1 else power[0]
        # Actually, for autocorrelation flattening, we want power to be flat
        # conv(a,a) = ifft(|A|^2), so flat conv means flat |A|^2
        target = np.mean(power)
        # Blend: new_power = power * (target/power)^alpha
        new_power = power * (target / (power + 1e-30))**alpha
        # Preserve phase
        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(new_power, 0)) * np.exp(1j * phase)
        # IFFT
        a_new = irfft(A_new, size)[:n]
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        it += 1
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 20 == 0:
                print(f"    Spec it{it}: obj={best_obj:.8f}, el={time.monotonic()-start:.1f}s")
        else:
            # Decrease alpha and try again
            alpha = max(0.1, alpha * 0.95)
            if obj_new > best_obj * (1 + 0.01) and it > 10:
                # Reset alpha
                alpha = 0.5

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c)**2) / (2 * w**2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.05))
        a = a / a.sum()
    elif mode == 4:
        # Alternating pattern (Barker-like)
        a = np.ones(n)
        for i in range(1, n):
            a[i] = 0.5 + 0.5 * np.cos(2 * np.pi * i / 5)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    elif mode == 5:
        # Two-tone
        a = np.ones(n)
        a[::2] = 1.5
        a[1::2] = 0.7
        a = a / a.sum()
    elif mode == 6:
        # Sine-modulated
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * 3 * x)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def nesterov_subgradient(a, time_budget=60.0, K=5):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 60

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        spread = (peak_vals[0] - peak_vals[-1]) / max_val
        temp = 3.0 / (1.0 + spread * 10.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                ki = int(k)
                i_start = max(0, ki - (n - 1))
                i_end = min(n - 1, ki)
                if i_start <= i_end:
                    j_start = ki - i_end
                    j_end = ki - i_start
                    g[i_start:i_end + 1] += w * y[j_start:j_end + 1][::-1]

        lr = lr0 / t_k
        a_new = y - lr * g
        # Project to simplex
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u)
        idx = np.arange(1, n + 1)
        cond = u * idx > (cssv - 1.0)
        rho = np.where(cond)[0]
        if len(rho) == 0:
            a_new = np.ones(n) / n
        else:
            rho = rho[-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_new - theta, 0.0)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.3)

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

    return best_a


def smooth_optimize(a, time_budget=60.0, T=50.0):
    from scipy.optimize import minimize as sp_minimize
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_grad(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = compute_conv(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        e = np.exp(z)
        sum_e = e.sum()
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)
        w = e / sum_e
        g = np.zeros(n)
        for k in range(len(conv)):
            if w[k] > 1e-8:
                ki = int(k)
                i_start = max(0, ki - (n - 1))
                i_end = min(n - 1, ki)
                if i_start <= i_end:
                    j_start = ki - i_end
                    j_end = ki - i_start
                    g[i_start:i_end + 1] += w[k] * x[j_start:j_end + 1][::-1]
        g *= 2.0 * n
        return obj_val, g

    max_iter = min(300, int((time_budget - (time.monotonic() - start)) * 5))
    if max_iter < 20:
        return best_a

    try:
        res = sp_minimize(obj_grad, a, method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 3, 'disp': False})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
    except Exception:
        pass

    return best_a

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_cur = len(best_seq)

    # === Phase 1: Spectral Equalization (new method, ~80s) ===
    print(f"\n=== Phase 1: Spectral Equalization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)
    if p1_budget > 10:
        a_spec = spectral_equalize(best_seq, time_budget=p1_budget, verbose=True)
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  Spectral: obj={obj_spec:.8f} (no improvement)")

    # === Phase 2: LP Alternating v2 (main workhorse, ~50-60% of time) ===
    print(f"\n=== Phase 2: LP Alternating v2 ===")
    remaining = deadline - time.monotonic()
    p2_total = remaining * 0.50

    n_rounds = 4
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p2_total / (n_rounds - rnd), remaining * 0.55)
        if alloc < 15:
            break

        if rnd == 0:
            a_start = best_seq.copy()
        elif rnd == 1:
            # Small perturbation
            a_start = best_seq * np.exp(rng.normal(0, 0.05, len(best_seq)))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        elif rnd == 2:
            # Try spectral-equalized version
            a_start = spectral_equalize(best_seq, time_budget=min(alloc * 0.2, 30.0))
        else:
            # Bigger perturbation
            a_start = best_seq * np.exp(rng.normal(0, 0.15, len(best_seq)))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        print(f"  LP round {rnd}: starting obj={compute_obj(a_start):.8f}")
        a_lp = lp_alternating_v2(a_start, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Nesterov + Smooth (refinement, ~15% of time) ===
    print(f"\n=== Phase 3: Nesterov + Smooth ===")
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.15

    if p3_budget > 10:
        a_ns = nesterov_subgradient(best_seq, time_budget=min(p3_budget * 0.4, 60.0), K=7)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} ** IMPROVED")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        for T_val in [30.0, 80.0]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            a_sm = smooth_optimize(best_seq, time_budget=min(remaining * 0.2, 40.0), T=T_val)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Smooth T={T_val}: obj={best_score:.8f} ** IMPROVED")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(100, n_best - 100),
        max(100, n_best - 50),
        n_best + 50,
        n_best + 100,
        n_best + 200,
        300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 100 <= n <= 2000]
    n_options.sort(key=lambda x: abs(x - n_best))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= 8:
            break
        if n_try == n_best:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        # Quick spectral equalization
        alloc_spec = min(remaining * 0.05, 20.0)
        if alloc_spec > 5:
            a_try = spectral_equalize(a_try, time_budget=alloc_spec)

        # Quick LP
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.08, 40.0)
        if alloc_lp > 10:
            a_try = lp_alternating_v2(a_try, time_budget=alloc_lp)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse Restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = r % 6
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.12, 50.0)
        if restart_budget < 15:
            break
        a_opt = lp_alternating_v2(a_r, time_budget=restart_budget)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} ** IMPROVED")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        lp_alloc = min(remaining * 0.4, 100.0)
        a_lp = lp_alternating_v2(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} ** IMPROVED")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_ns = nesterov_subgradient(best_seq, time_budget=min(remaining * 0.3, 40.0), K=10)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Final NS: obj={best_score:.8f} ** IMPROVED")

    # One more LP with fresh start
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_fresh = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
        a_fresh = np.maximum(a_fresh, 0)
        a_fresh = a_fresh / a_fresh.sum()
        a_last = lp_alternating_v2(a_fresh, time_budget=min(remaining * 0.4, 40.0))
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} ** IMPROVED")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
