import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def compute_obj(a):
    """Compute objective: 2n * max_conv / (sum_a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    # Water-filling: find tau such that sum(max(v_i - tau, 0)) = 1
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    # Find largest rho such that u[rho-1] > (cssv[rho-1] - 1) / rho
    cond = u > (cssv - 1.0) / rho
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = np.max(np.where(cond))
    tau = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - tau, 0)

def lp_alternating(a, time_budget=300.0, verbose=False, momentum_beta=0.5):
    """LP alternating with momentum and adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    momentum = np.zeros(n)
    last_t = 0.3
    active_set = None
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp, active = solve_lp_fast(a, active_set=active_set, prev_g=best_a)
        if g_lp is None:
            break
        active_set = active
        
        # Try multiple step sizes
        step_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if 0.05 < last_t < 1.0:
            step_candidates = sorted(set(step_candidates + [last_t * 0.8, last_t * 1.2]))
            step_candidates = [t for t in step_candidates if 0.03 <= t <= 1.5]
        
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = None
        
        for t in step_candidates:
            direction = g_lp - a
            momentum = momentum_beta * momentum + (1 - momentum_beta) * direction
            a_try = a + t * momentum
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_a_local is not None and best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local
            a = best_a_local
            last_t = best_t_val
            stall = 0
        else:
            stall += 1
            momentum *= 0.5
            if stall >= 8:
                # Perturbation
                scale = 0.02 + 0.005 * stall
                rng = np.random.default_rng(hash((stall, int(time.monotonic()*1e6))) % (2**31))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = project_simplex(a)
                stall = 0
                last_t = 0.5
                active_set = None
    
    return best_a


def nesterov_subgradient(a, time_budget=100.0, verbose=False):
    """Nesterov-accelerated subgradient with exact simplex projection."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()  # current point
    x = a.copy()  # Nesterov lookahead
    beta = 0.85
    lr = 0.03
    momentum = np.zeros(n)
    alpha = 1.0  # Nesterov parameter

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # Compute convolution at lookahead point
        x_eval = project_simplex(x)
        conv = compute_conv(x_eval)
        max_val = float(np.max(conv))
        
        # Get active peaks
        if max_val <= 0:
            break
        threshold = max_val * 0.85
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 50:
            active = np.argsort(conv)[::-1][:50]

        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += x_eval[k - indices]
        grad /= len(active)
        
        # Center gradient
        grad_centered = grad - np.mean(grad)

        # Nesterov step
        alpha = 2.0 / (np.sqrt(it + 1) + 2.0)
        x = y + (1 - alpha) * (y - x)
        x = project_simplex(x)
        
        # Gradient step from x
        y_new = x - lr * grad_centered
        y_new = project_simplex(y_new)
        y = y_new
        
        obj = compute_obj(y)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()
        
        # Adaptive learning rate
        if it % 500 == 499:
            lr = max(lr * 0.95, 0.005)

    return best_a


def peak_shaving(a, time_budget=100.0):
    """Iteratively reduce the peak of conv(a,a) by moving mass away from contributing positions."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    
    for iteration in range(20000):
        if time.monotonic() - start > time_budget:
            break
        
        k_star = int(np.argmax(conv))
        
        # Find contributing pairs
        contributions = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contributions[j] = a[j] * a[k_j]
        
        # Try to reduce the peak by moving mass from high-contributing positions
        sorted_idx = np.argsort(contributions)[::-1]
        improved = False
        
        for i in sorted_idx[:10]:
            if time.monotonic() - start > time_budget:
                break
            i = int(i)
            if a[i] < 1e-10:
                continue
            
            # Try moving small amount to a position that doesn't contribute to the peak
            delta = 0.05 * a[i]
            # Find target: position far from contributing to k_star
            # Position j contributes to k_star if 0 <= k_star - j < n
            # So we want j such that k_star - j < 0 or k_star - j >= n
            # i.e., j > k_star or j < k_star - n + 1 (but j < n always)
            # So j > k_star works (if k_star < n-1)
            candidates = []
            if k_star < n - 1:
                candidates = list(range(k_star + 1, n))
            if k_star >= n:
                candidates += list(range(0, k_star - n + 1))
            if not candidates:
                candidates = list(range(n))
            
            # Pick the candidate with smallest current value
            target = min(candidates, key=lambda j: a[j])
            
            a[i] -= delta
            a[target] += delta
            a = project_simplex(a)
            
            new_conv = compute_conv(a)
            new_obj = 2.0 * n * float(np.max(new_conv))
            
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()
                conv = new_conv
                improved = True
            else:
                a[i] += delta
                a[target] -= delta
                a = project_simplex(a)
        
        if not improved and iteration > 200:
            # Random perturbation
            rng = np.random.default_rng(hash((iteration, int(time.monotonic()*1e6))) % (2**31))
            for _ in range(5):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta = 0.02 * a[i]
                a[i] -= delta
                a[j] += delta
                a = project_simplex(a)
                new_obj = compute_obj(a)
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = compute_conv(a)
                    improved = True
                    break
        
        if not improved and iteration > 500:
            break

    return best_a


def generate_initialization(kind, n, rng):
    """Generate a diverse initialization of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 5.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'bilinear':
        a = np.linspace(0.5, 1.5, n)
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 8)
        a[:edge] = 0.2
        a[-edge:] = 0.2
        return a / np.sum(a)
    elif kind == 'chirp':
        # Chirp-like: oscillating amplitude
        x = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x / n)
        a = np.maximum(a, 0.1)
        return a / np.sum(a)
    elif kind == 'two_peak':
        a = np.ones(n) * 0.5
        a[n//4] = 2.0
        a[3*n//4] = 2.0
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'triangular':
        x = np.arange(n)
        center = n / 2.0
        a = 1.0 - np.abs(x - center) / center
        a = np.maximum(a, 0.01)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.001) / np.sum(np.maximum(a_new, 0.001))

import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def compute_obj(a):
    """Compute objective: 2n * max_conv / (sum_a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = np.max(np.where(cond))
    tau = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - tau, 0)


def solve_lp(a):
    """
    Solve LP: min M s.t. conv(a, g) <= M, sum(g)=1, g>=0.
    Returns (g, M, active_indices) or (None, None, None).

                    tol = max(M_val * 1e-5, 1e-8)
                    active = np.where(conv_val >= M_val - tol)[0]
                    # Also get dual variables if available (for y*)
                    return g_val, M_val, active
        except Exception:
            continue

def line_search_step(a, g_lp, best_obj, momentum=None, max_steps=15):
    """
    Line search: find best step size t for a + t*(g_lp - a) or with momentum.
    Returns (best_a, best_obj, best_t).
    """
    n = len(a)
    direction = g_lp - a
    if momentum is not None:
        direction = momentum

    best_a = None
    best_t_obj = np.inf
    best_t_val = 0.3

    # Golden section / bisection hybrid
    # First try a broad set of candidates
    candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]

    for t in candidates:
        a_try = a + t * direction
        a_try = project_simplex(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_t_obj:
            best_t_obj = obj_try
            best_t_val = t
            best_a = a_try.copy()

    # Refine around best
    if best_t_val > 0:
        lo = max(0.01, best_t_val * 0.6)
        hi = min(2.0, best_t_val * 1.5)
        for _ in range(6):
            mid = (lo + hi) / 2
            a_mid = a + mid * direction
            a_mid = project_simplex(a_mid)
            obj_mid = compute_obj(a_mid)
            # Also check lo and hi
            a_lo = project_simplex(a + lo * direction)
            obj_lo = compute_obj(a_lo)
            a_hi = project_simplex(a + hi * direction)
            obj_hi = compute_obj(a_hi)

            if obj_lo < best_t_obj:
                best_t_obj = obj_lo
                best_t_val = lo
                best_a = a_lo.copy()
            if obj_hi < best_t_obj:
                best_t_obj = obj_hi
                best_t_val = hi
                best_a = a_hi.copy()
            if obj_mid < best_t_obj:
                best_t_obj = obj_mid
                best_t_val = mid
                best_a = a_mid.copy()

            if obj_mid < obj_lo:
                hi = mid
            else:
                lo = mid

    return best_a, best_t_obj, best_t_val

def generate_initialization(kind, n, rng):
    """Generate a diverse initialization of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        a = np.maximum(a, 0.001)
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 5.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'ramp':
        a = np.linspace(0.3, 1.7, n)
        return a / np.sum(a)
    elif kind == 'chirp':
        x = np.arange(n)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * 2.7 * x / n)
        a = np.maximum(a, 0.1)
        return a / np.sum(a)
    elif kind == 'decay':
        x = np.arange(n)
        a = np.exp(-0.5 * x / (n / 3.0))
        return a / np.sum(a)
    elif kind == 'bump':
        center = n * 0.3
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - center) / (n / 6.0)) ** 2) + 0.05
        return a / np.sum(a)
    elif kind == 'log_uniform':
        a = np.exp(rng.uniform(0, 1, n))
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.001)
    return a_new / np.sum(a_new)
