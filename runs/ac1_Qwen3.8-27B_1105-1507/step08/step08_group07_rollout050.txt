```python
"""
Gradient-based optimizer using smooth-max (softmax) approximation of the 
max-convolution objective, combined with periodic LP refinements and 
multi-n exploration. Key insight: optimal sequences are highly asymmetric,
allowing the center convolution to be much less than 1/n.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_max_grad(a, tau):
    """
    Compute gradient of tau * log(sum(exp(conv/tau))) w.r.t. a.
    Returns (gradient, conv_values, soft_max_value).
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Softmax weights
    conv_shifted = conv - np.max(conv)
    weights = np.exp(conv_shifted / tau)
    w_sum = np.sum(weights)
    weights = weights / w_sum
    
    # soft_max value
    soft_max = tau * (np.max(conv) + np.log(w_sum))
    
    # Gradient: g[m] = 2 * sum_k weights[k] * a[k-m]
    # = 2 * conv[w_rev, a][m + 2n - 2]
    size = 1
    while size < 3 * n:
        size <<= 1
    w_rev = weights[::-1]
    W = rfft(w_rev, size)
    A = rfft(a, size)
    conv_wa = irfft(W * A, size)
    g = 2.0 * conv_wa[2*n-2:3*n-2]
    
    return g, conv, soft_max


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. A@a_conv(z) <= M, sum(z)=1, z>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    return z_val, float(M.value)
        except:
            continue
    
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n])

        pass
    
    return None, None


def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def adam_smooth_max(a, time_budget=200.0, verbose=False):
    """
    Optimize using Adam with smooth-max objective.
    Adaptive temperature schedule.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()

    conv0 = compute_conv(a)

    # Learning rate schedule
    lr = 0.05

    stall = 0
    patience = 500
    
    while time.monotonic() - start < time_budget:
        # Compute gradient
        g, conv, soft_max_val = smooth_max_grad(a, tau)
        
        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t_adam)
        v_hat = v / (1 - beta2**t_adam)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)

            best_a = a_new.copy()

            stall += 1
            if stall > patience:
                lr *= 0.5
                tau = max(tau * 0.9, tau_min)
                stall = 0
        
        # Decay temperature and learning rate
        tau = max(tau * tau_decay, tau_min)
        lr = max(lr * lr_decay, 1e-6)
        
        if verbose and int((time.monotonic() - start) * 2) % 20 == 0:

def lp_robust_alternating(a, time_budget=200.0, verbose=False):
    """LP alternating with cycle detection and momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))

        result = solve_lp(a)
        if result[0] is None:
            break

        g_momentum = np.maximum(g_momentum, 0)
        gs = float(np.sum(g_momentum))
        if gs > 1e-10:
            g_momentum = g_momentum / gs
        
        # Try multiple mixing ratios
        targets = [g_lp, g_momentum]
        best_cand_obj = np.inf
        best_cand = a.copy()
        
        for target in targets:
            for mix in [0.3, 0.5, 0.7, 1.0]:
                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj:
                    best_cand_obj = obj_try
                    best_cand = a_try.copy()
        
        if best_cand_obj < best_obj - 1e-12:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            prev_a = a.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Perturb to escape
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent(a, time_budget=30.0):
    """
    Coordinate descent polish: for each coordinate, find the optimal value
    given all others fixed (by 1D minimization).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    iter_count = 0
    while time.monotonic() - start < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                    break
            # Try a few perturbations of a[i]
            for scale in [0.8, 0.9, 1.1, 1.2]:
                a_try = a.copy()
                old_val = a_try[i]
                a_try[i] = old_val * scale
                s = float(np.sum(a_try))
                if s < 1e-10:
                    continue
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        iter_count += 1
        if not improved:
            break
    
    return best_a
```
