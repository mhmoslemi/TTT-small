Let me think about the mirror descent more carefully.

The update is:
`a_i(t+1) = a_i(t) * exp(-lr * g_i(t)) / Z`

where `g_i = 2 * sum_k w_k * a_{k-i}` and `Z = sum_i a_i * exp(-lr * g_i)`.

The effective step size `lr` controls how aggressive the update is. Too large and we oscillate; too small and we're slow.

For the temperature T:
- Large T: smooth approximation is poor, gradient points to "average" of all peaks
- Small T: gradient points to the actual max peak, but can be noisy

A good schedule: start with moderate T, decrease gradually.

Let me also think about **Nesterov acceleration** for the mirror descent:
- Momentum in the "exponential" domain: `m_i(t+1) = beta * m_i(t) + (1-beta) * (-lr * g_i(t))`
- Update: `a_i(t+1) = a_i(t) * exp(m_i(t+1)) / Z`

Actually, for mirror descent, the analog of Nesterov is "FISTA in the mirror space" which is more complex. Let me just use simple momentum.

One more idea: **alternating between different objective formulations**.

The objective `max(b_0, b_1, ..., b_{2n-2})` can be written as:
- The max (use smooth max approximation)
- The L_p norm for large p: `(sum b_k^p)^{1/p}`
- The sum of the top-k values

Using the L_p norm with p = 10 or 20 might give a different gradient structure that helps escape local minima.

"""
Optimized optimizer combining mirror descent (multiplicative weight updates),
smooth max gradient with Nesterov momentum, LP alternating, and coordinate descent
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_max_grad(a, T):
    """
    Compute smooth max gradient for the objective.
    Returns (smooth_max_value, gradient_wrt_a, weights).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    smooth_val = max_c + T * np.log(total)
    
    # Gradient: grad_i = 2 * sum_k weights_k * a_{k-i}
    # This is 2 * (weights * a but truncated)
    # weights has length 2n-1, a has length n
    # grad_i = 2 * sum_{k: 0<=k<n, 0<=k-i<2n-1} weights_k * a_{k-i}
    #         = 2 * (correlate-like operation)
    
    # Actually, grad_i = 2 * sum_k w_k * a_{k-i} where we need 0 <= k-i < n
    # This is like a cross-correlation of w and a, but truncated to length n
    
    # Let's compute it via FFT
    # grad = 2 * idft(fft(w) * conj(fft(a)))... no that's not right.
    # 
    # grad_i = 2 * sum_k w_k * a_{k-i} for 0 <= i < n
    # Let's define c_i = sum_k w_k * a_{k-i}
    # This is the "valid" part of the correlation of w and a
    # c_i = (w * flip(a))_i... hmm
    # 
    # Actually: c_i = sum_{k=i}^{i+n-1} w_k * a_{k-i} = sum_{j=0}^{n-1} w_{i+j} * a_j
    # This is a sliding dot product = cross-correlation of w with a
    # c = correlate(w, a, mode='valid')... but w has length 2n-1 and a has length n
    # correlate(w, a, 'valid') has length (2n-1) - n + 1 = n. 
    
    # Let me use np.correlate
    # np.correlate(w, a, 'valid') gives c_i = sum_j w_{i+j} * a_j for i=0,...,n-1
    # (assuming len(w) >= len(a))
    
    c = np.correlate(weights, a, mode='valid')
    grad = 2.0 * c
    
    return smooth_val, grad, weights


def mirror_descent_smooth_max(a, time_budget=100.0, T_init=0.001, T_min=0.0001, lr_init=0.01, verbose=False):
    """
    Mirror descent (multiplicative weight updates) with smooth max objective.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = lr_init
    momentum = 0.7
    m = np.zeros(n)  # momentum in log-space
    stall = 0

    max_iters = 20000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        _, grad, weights = smooth_max_grad(a, T)
        
        # Momentum update in log-space
        m = momentum * m + (1 - momentum) * (-lr * grad)
        
        # Multiplicative update
        log_update = m
        log_update -= np.max(log_update)  # for numerical stability
        exp_update = np.exp(log_update)
        a_new = a * exp_update
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a = np.ones(n) / n
            m = np.zeros(n)
            stall += 1
        else:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                if verbose and it % 500 == 0:
                    print(f"          MD it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}")
            else:
                stall += 1
                if stall > 30:
                    lr *= 0.8
                    m = np.zeros(n)
                    stall = 0
                    if lr < 1e-8:
                        break

        # Anneal T
        if it > 20:
            T = max(T_min, T_init * (0.997 ** it))

        # Occasionally reset momentum
        if it % 500 == 0 and it > 0:
            m *= 0.5

    return best_a


def nesterov_smooth_max(a, time_budget=100.0, T_init=0.001, T_min=0.00005, verbose=False):
    """
    Nesterov-accelerated gradient descent on smooth max (projected to simplex).
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.005
    beta = 0.9  # momentum
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    stall = 0

    max_iters = 20000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        _, grad, _ = smooth_max_grad(y, T)
        
        # Center gradient (since we're on the simplex)
        grad_centered = grad - np.sum(grad) / n
        
        # Nesterov update
        v = beta * v - lr * grad_centered
        a_new = a + v
        a_new = simplex_project(a_new)
        
        y = a_new + beta * (a_new - a)
        y = np.maximum(y, 0)
        y = y / max(np.sum(y), 1e-10)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 500 == 0:
                print(f"          NS it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}")
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                v = np.zeros(n)
                y = a.copy()
                stall = 0
                if lr < 1e-8:
                    break

        if it > 50:
            T = max(T_min, T_init * (0.998 ** it))

    return best_a


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp(a, time_limit=5.0):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None

    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in ['CLARABEL', 'ECOS', 'SCS', 'MOSEK', 'PDLP']:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    g_norm = g_val / s
                    # Compute true M
                    conv_check = compute_conv_cross(a, g_norm)
                    M_val = float(np.max(conv_check))
                    return g_norm, M_val
        except Exception:
            continue
    return None, None

        return a

def coordinate_descent_polish(a, time_budget=60.0, delta=1e-4, verbose=False):

        rng = np.random.default_rng(pass_count * 7919 + n)
        order = rng.permutation(n)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def block_optimization(a, n_blocks, time_budget=30.0, verbose=False):
    """
    Optimize block means first, then fine-tune within blocks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    block_size = n // n_blocks
    block_means = np.zeros(n_blocks)
    for b in range(n_blocks):
        start_idx = b * block_size
        end_idx = start_idx + block_size if b < n_blocks - 1 else n
        block_means[b] = np.mean(a[start_idx:end_idx])

    # Optimize block means via mirror descent on the reduced problem
    # Create a "coarse" sequence from block means
    T = 0.001
    lr = 0.05
    for it in range(500):
        if time.monotonic() - start > time_budget * 0.4:
            break
        coarse = np.repeat(block_means, block_size)
        if n_blocks > 1:
            coarse[-1] = coarse[-1]  # last block might be shorter
        # Actually let me just use a repeated sequence
        coarse = np.tile(block_means, n // n_blocks + 1)[:n]
        s_c = float(np.sum(coarse))
        if s_c < 1e-10:
            break
        coarse = coarse / s_c
        
        conv_c = compute_conv(coarse)
        max_c = np.max(conv_c)
        shifted = (conv_c - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)
        
        # Gradient for block means
        # grad_b = 2 * sum of weights[k] * (sum of coarse[k-i] for i in block b)
        # Simplified: just use the average gradient in each block
        c = np.correlate(weights, coarse, mode='valid')
        grad_block = np.zeros(n_blocks)
        for b in range(n_blocks):
            s_idx = b * block_size
            e_idx = min(s_idx + block_size, n)
            grad_block[b] = np.mean(c[s_idx:e_idx]) if e_idx > s_idx else 0
        
        # Mirror descent on block means
        grad_block_centered = grad_block - np.mean(grad_block)
        block_means = np.maximum(block_means * np.exp(-lr * grad_block_centered), 1e-10)
        block_means = block_means / np.sum(block_means)
        
        T = max(0.0001, T * 0.995)

    # Rebuild sequence from optimized block means with original within-block variation
    variation = a - np.repeat(np.array([np.mean(a[b*block_size:min((b+1)*block_size, n)]) for b in range(n_blocks)]), block_size)[:n]
    a_new = np.repeat(block_means, block_size)
    if n_blocks > 1:
        a_new[-(n - (n_blocks-1)*block_size):] = block_means[-1]  # handle last partial block
    a_new = a_new + variation * 0.5  # blend with original variation
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new > 1e-10:
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new

    return best_a

    remaining = deadline - time.monotonic()
    if remaining > 60:
        budget3 = min(200.0, remaining * 0.2)
        a_lp = lp_alternating(a, time_budget=budget3, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  LP done: obj={best_score:.8f}")
        else:
            print(f"  LP done: obj={obj_lp:.8f} (no improvement)")

    # === Phase 4: Mirror Descent round 2 (from improved point) ===
    print(f"\n=== Phase 4: MD Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        budget4 = min(150.0, remaining * 0.15)
        a_md2 = mirror_descent_smooth_max(a, time_budget=budget4, T_init=0.001, T_min=0.00005, lr_init=0.01, verbose=True)
        obj_md2 = compute_obj(a_md2)
        if obj_md2 < best_score:
            best_score = obj_md2
            best_seq = a_md2.copy()
            a = a_md2.copy()
            print(f"  MD2 done: obj={best_score:.8f}")
        else:
            print(f"  MD2 done: obj={obj_md2:.8f} (no improvement)")

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        budget5 = min(100.0, remaining * 0.15)
        a_cd = coordinate_descent_polish(a, time_budget=budget5, delta=1e-4, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a = a_cd.close()
            print(f"  CD done: obj={best_score:.8f}")
        else:
            print(f"  CD done: obj={obj_cd:.8f} (no improvement)")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 50, n_best - 25, n_best,
        n_best + 25, n_best + 50, n_best + 100,
        600, 700, 800, 900, 1000, 1100, 1200, 1300
    ]))
    n_options = [n for n in n_options if 500 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick MD
        alloc = min(remaining * 0.08, 20.0)
        a_opt = mirror_descent_smooth_max(a_try, time_budget=alloc, T_init=0.002, T_min=0.0002, lr_init=0.02)
        obj_try = compute_obj(a_opt)

        # Quick CD if promising
        remaining = deadline - time.monotonic()
        if remaining > 15 and obj_try < best_score + 0.003:
            a_opt = coordinate_descent_polish(a_opt, time_budget=min(15.0, remaining * 0.3), delta=1e-4)
            obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    a_final = best_seq.copy()

    if remaining > 40:
        # MD polish
        alloc = min(60.0, remaining * 0.3)
        a_p1 = mirror_descent_smooth_max(a_final, time_budget=alloc, T_init=0.0005, T_min=0.00005, lr_init=0.005)
        if compute_obj(a_p1) < best_score:
            best_score = compute_obj(a_p1)
            best_seq = a_p1.copy()
            a_final = a_p1.copy()
            print(f"  Final MD: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        # CD polish
        alloc = min(50.0, remaining * 0.4)
        a_p2 = coordinate_descent_polish(a_final, time_budget=alloc, delta=5e-5)
        if compute_obj(a_p2) < best_score:
            best_score = compute_obj(a_p2)
            best_seq = a_p2.copy()
            print(f"  Final CD: obj={best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Nesterov polish
        alloc = min(30.0, remaining * 0.4)
        a_p3 = nesterov_smooth_max(a_final, time_budget=alloc, T_init=0.0003, T_min=0.00003)
        if compute_obj(a_p3) < best_score:
            best_score = compute_obj(a_p3)
            best_seq = a_p3.copy()
            print(f"  Final Nesterov: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
