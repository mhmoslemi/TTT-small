    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()

    step_size = 1.0 / np.sqrt(n)
    beta_mom = 0.85
    v = np.zeros(n)
    stall = 0
    best_stall = 0

    for it in range(500000):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            grad += conv_grad(a, int(k))
        grad /= len(top_idx)
        grad -= np.mean(grad)

        gn = np.dot(grad, grad)
        if gn < 1e-20:
            break

        v = beta_mom * v - step_size * grad
        a_new = project_simplex(a + v)

        if it % 5 == 0:
            obj = obj_normalized(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                a = a_new
                stall += 1
                if stall > 100:
                    step_size = max(step_size * 0.7, 1e-4)
                    beta_mom = min(beta_mom + 0.005, 0.999)
                    stall = 0
                    if best_obj < best_stall:
                        best_stall = best_obj
                        a = best_a.copy()
                        v = np.zeros(n)
        if it % 1000 == 999:
            step_size = max(step_size * 0.98, 1e-5)
            beta_mom = min(beta_mom + 0.002, 0.999)

        if verbose and it % 5000 == 0:
            print(f"        SG it={it}: obj={best_obj:.8f}, step={step_size:.6f}, t={time.monotonic()-start:.1f}s")

    return best_a


def solve_lp_linearized(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    c = compute_conv(a_norm)
    A = build_conv_matrix_2a(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g - c <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_baseline(a):
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix_2a(a_norm)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= 2.0 * M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def line_search_blend(a, g, best_obj):
    def eval_blend(t):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            return np.inf
        return obj_normalized(a_try / st)

    try:
        res = minimize_scalar(eval_blend, bounds=(0.0, 1.0), method='bounded',
                              options={'maxiter': 40, 'xatol': 0.005})
        best_t = res.x
        best_val = eval_blend(best_t)
    except Exception:
        best_t = 0.5
        best_val = eval_blend(best_t)

    a_try = (1.0 - best_t) * a + best_t * g
    a_try = np.maximum(a_try, 0)
    st = float(np.sum(a_try))
    if st < 1e-10:
        return a, best_obj
    a_try = a_try / st
    return a_try, best_val


def lp_alternating(a0, time_budget=100.0, verbose=False):
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    stall = 0
    use_linearized = True

    while time.monotonic() - start < time_budget:
        if use_linearized:
            g_lp, _ = solve_lp_linearized(a)
        else:
            g_lp, _ = solve_lp_baseline(a)
        if g_lp is None:
            use_linearized = not use_linearized
            continue

        a_new, obj_new = line_search_blend(a, g_lp, best_obj)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose:
                print(f"        LP: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            use_linearized = not use_linearized
            if stall > 20:
                break

    return best_a


def coordinate_descent(a0, time_budget=60.0, verbose=False):
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    conv = compute_conv(a)
    top_lags = np.argsort(conv)[::-1][:10]
    top_lag_set = set(int(k) for k in top_lags)

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        i = int(rng.integers(0, n))
        if a[i] < 1e-8:
            continue
        for attempt in range(5):
            j = int(rng.integers(0, n))
            if i == j:
                continue
            for frac in [0.02, 0.05, 0.1, 0.2]:
                delta = frac * a[i]
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                st = float(np.sum(a))
                if st < 1e-10:
                    a[i] += delta
                    a[j] -= delta
                    continue
                a = a / st
                obj_try = obj_normalized(a)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a.copy()
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    a = a / st
        if it % 2000 == 1999:
            conv = compute_conv(best_a)
            top_lags = np.argsort(conv)[::-1][:10]
            if verbose:
                print(f"        CD it={it}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw
