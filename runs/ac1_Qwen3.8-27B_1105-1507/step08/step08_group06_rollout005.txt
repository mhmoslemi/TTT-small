"""
Optimized step function search using enhanced LP-alternating dynamics with FOMC-style
extrapolation, adaptive line search, diverse zero-boundary initializations,
multi-solver LP, and subgradient/coordinate polish.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv / sum^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_best_response(a):
    """
    Solve LP: min M s.t. conv(a, g) <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    # Try cvxpy solvers
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                return g_val, M_val
    except Exception:
        pass

    return None, None

        obj_best_local = best_obj

            best_a = a_best_local.copy()
            a = a_best_local

def subgradient_nesterov(a, time_budget=150.0, verbose=False):
    """
    Nesterov-accelerated subgradient method on the simplex.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    momentum = np.zeros(n)
    beta = 0.75
    lr = 0.03

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(y)
        max_val = float(np.max(conv))
        # Active set: peaks within 15% of max
        threshold = max_val * 0.85
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 40:
            active = np.argsort(conv)[::-1][:40]

        # Subgradient: average over active peaks
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += 2.0 * y[k - indices]
        grad /= len(active)

        # Project gradient to tangent space of simplex
        grad_centered = grad - np.mean(grad)

        # Nesterov momentum
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        momentum = beta * momentum - lr * grad_centered
        y_new = y + t_k * momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
        else:
            y_new = y_new / s_new

        y = y_new
        obj = compute_obj(y)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()

        # Decay learning rate
        if it % 500 == 499:
            lr *= 0.95
            if lr < 0.001:
                lr = 0.001

        if verbose and it % 2000 == 1999:
            elapsed = time.monotonic() - start
            print(f"    Subgrad: it={it}, obj={best_obj:.8f}, time={elapsed:.0f}s")

    return best_a


def coordinate_polish(a, time_budget=100.0):
    """
    Targeted coordinate moves to reduce the peak convolution value.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    for iteration in range(2000):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))

        # Find top contributors to the peak
        contributions = np.zeros(n)
        for j in range(min(n, k_star + 1)):
            k_j = k_star - j
            if 0 <= k_j < n:
                contributions[j] = a[j] * a[k_j]

        sorted_idx = np.argsort(contributions)[::-1]

        improved = False
        for frac in [0.003, 0.008, 0.015, 0.03, 0.05]:
            if improved:
                break
            for i in sorted_idx[:15]:
                i = int(i)
                if a[i] < 1e-8:
                    continue
                # Try moving to a position that's "far" from the peak structure
                # Choose target that minimizes its contribution to the peak
                target_candidates = [
                    (k_star - i + n // 3) % n,
                    (k_star - i + 2 * n // 3) % n,
                    (i + n // 2) % n,
                ]
                best_target = target_candidates[0]
                min_contrib = np.inf
                for tc in target_candidates:
                    k_tc = k_star - tc
                    if 0 <= k_tc < n:
                        c_val = a[tc] * a[k_tc]
                    else:
                        c_val = 0
                    if c_val < min_contrib:
                        min_contrib = c_val
                        best_target = tc

                target = int(best_target)
                if target == i:
                    target = (i + 1) % n

                delta = frac * a[i]
                if delta <= 0:
                    continue

                a[i] -= delta
                a[target] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[target] -= delta

        if not improved and iteration > 200:
            # Random exploration
            rng = np.random.default_rng(int(time.monotonic() * 1e4) % (2**31))
            for _ in range(10):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.005 * a[i]
                if delta <= 0 or delta > a[i]:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and iteration > 500:
            break

    return best_a


def generate_initialization(kind, n, rng=None):
    """Generate diverse initializations."""

    elif kind == 'gaussian_center':

        center = (n - 1) / 2.0
        sigma = n / 4.0

    elif kind == 'broad_gaussian':
        x = np.arange(n)
        center = (n - 1) / 2.0
        sigma = n / 2.5
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'power_law':
        x = np.arange(n)
        center = (n - 1) / 2.0
        d = np.abs(x - center) / (n / 2.0)
        a = (1 - d) ** 0.5 + 0.01

    elif kind == 'two_hump':
        x = np.arange(n)

    elif kind == 'random':

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            scale = 0.04
            a = best_seq * np.exp(rng.normal(0, scale, n_parent))
            a = np.maximum(a, 0)
            a = a / np.sum(a)

    # === Phase 3: Subgradient Nesterov ===
    print(f"\n=== Phase 3: Subgradient Nesterov ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        sg_budget = min(200.0, remaining * 0.18)
        a_sg = subgradient_nesterov(a, time_budget=sg_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 4: Coordinate Polish ===
    print(f"\n=== Phase 4: Coordinate Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        cd_budget = min(120.0, remaining * 0.12)
        a_cd = coordinate_polish(a, time_budget=cd_budget)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a = a_cd.copy()
            print(f"  Coord: obj={best_score:.8f} **")
        else:
            print(f"  Coord: obj={obj_cd:.8f}")

    # === Phase 5: Length Exploration ===
    print(f"\n=== Phase 5: Length Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Try a range of lengths
    n_options = sorted(set([
        max(200, n_best - 200),
        max(200, n_best - 100),
        max(200, n_best - 50),
        n_best,
        n_best + 50,
        n_best + 100,
        n_best + 200,
        n_best + 400,
    ]))
    n_options = [n for n in n_options if 200 <= n <= 4000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick LP + subgradient optimization
        alloc_lp = min(40.0, remaining * 0.35)
        alloc_sg = min(25.0, remaining * 0.2)

        a_opt = lp_alternating_enhanced(a_try, time_budget=alloc_lp)
        a_opt = subgradient_nesterov(a_opt, time_budget=alloc_sg)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final LP Polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    a_final = best_seq.copy()

    while remaining > 30:
        alloc = min(remaining * 0.3, 60.0)
        a_new = lp_alternating_enhanced(a_final, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_final = a_new.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            scale = 0.02
            a_final = best_seq * np.exp(rng.normal(0, scale, n_final))
            a_final = np.maximum(a_final, 0)
            a_final = a_final / np.sum(a_final)
            print(f"  Final LP: obj={obj:.8f} (perturbing)")

        remaining = deadline - time.monotonic()

    # === Phase 7: Final coordinate micro-polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_cd = coordinate_polish(best_seq, time_budget=min(remaining * 0.4, 30.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
