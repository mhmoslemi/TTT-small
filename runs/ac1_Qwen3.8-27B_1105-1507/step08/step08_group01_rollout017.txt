"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[:n] if j == 0 else np.concatenate([a[:n-j], np.zeros(j)])
    return A


def build_conv_matrix_fast(a):
    """Build convolution matrix efficiently using stride tricks."""
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n else 0
    # i.e., A is a banded/toeplitz-like matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        row_start = j
        row_end = min(2 * n - 1, j + n)
        A[row_start:row_end, j] = a[:row_end - row_start]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def accurate_line_search(a, g, n_points=12):
    """
    Accurate line search for a_new = (1-t)*a + t*g.

    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    # Initial grid
    ts = np.linspace(0, 1, n_points)
    objs = np.array([obj_at_t(t) for t in ts])
    best_idx = np.argmin(objs)

    # Refine around best point
    if best_idx > 0 and best_idx < n_points - 1:
        # Quadratic interpolation
        t0, t1, t2 = ts[best_idx - 1], ts[best_idx], ts[best_idx + 1]
        f0, f1, f2 = objs[best_idx - 1], objs[best_idx], objs[best_idx + 1]
        denom = (f0 - 2 * f1 + f2)
        if abs(denom) > 1e-15:
            t_quad = t1 + 0.5 * (f0 - f2) / (f0 - 2 * f1 + f2) * (t2 - t0)
            if ts[best_idx - 1] < t_quad < ts[best_idx + 1]:
                t_quad_obj = obj_at_t(t_quad)
                if t_quad_obj < objs[best_idx]:
                    ts = np.sort(np.concatenate([[t_quad], ts]))
                    objs = np.insert(objs, len(ts) - 1, t_quad_obj)

    # Final golden section in the best interval
    best_idx = np.argmin(objs)
    lo = ts[max(0, best_idx - 1)]
    hi = ts[min(len(ts) - 1, best_idx + 1)]

    for _ in range(20):
        m1 = lo + 0.382 * (hi - lo)
        m2 = lo + 0.618 * (hi - lo)
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    a_new = a_new / s
    return t_opt, a_new


def lp_alternating_v2(a, time_budget=200.0, verbose=False):
    """
    Improved LP alternating with adaptive active set and accurate line search.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    top_k = 10
    max_iters = 1000

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute current convolution
        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # Adaptive top-k: expand if peaks are spread out
        sorted_idx = np.argsort(conv)[::-1]
        top_vals = conv[sorted_idx[:min(20, len(sorted_idx))]]
        if len(top_vals) > top_k:
            spread = top_vals[0] - top_vals[top_k - 1] / max(top_vals[top_k - 1], 1e-15)
            if spread > 0.01:
                top_k = min(20, top_k + 2)
            elif top_k > 5 and top_vals[0] - top_vals[top_k - 1] < 1e-5:
                top_k = max(5, top_k - 1)

        # Get LP direction with current active set
        active = sorted_idx[:top_k]
        g_lp, M_lp = solve_lp_primal(a, indices=active)

        if g_lp is None:
            # Try full LP
            g_lp, M_lp = solve_lp_primal(a, indices=None)
            if g_lp is None:
                break

        # Line search
        t_opt, a_new = accurate_line_search(a, g_lp, n_points=10)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            improvement = best_obj - obj_new
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 10 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: obj={best_obj:.10f}, t={t_opt:.4f}, top_k={top_k}, time={elapsed:.1f}s")
        else:
            stall += 1
            if stall > 30:
                # Perturb to escape
                rng = np.random.default_rng(it * 7919 + n)
                scale = 0.002
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a = a_pert / s_p
                top_k = 15
                stall = 0
                if stall > 50:
                    break

    return best_a

def coordinate_descent_fine(a, time_budget=100.0, verbose=False):
    """
    Fine coordinate descent with adaptive step sizes and pair updates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv_cur = compute_conv(a)
    max_conv = float(np.max(conv_cur))

    delta = 1e-3
    pass_count = 0
    rng = np.random.default_rng(12345 + n)

    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            # Try pair update: increase a[i], decrease a[j] (preserving sum)
            j = rng.integers(n)
            if j == i:
                j = (i + 1) % n

            for d in [delta, delta * 0.1, delta * 5]:
                if a[j] < d * 0.5:
                    continue
                a_try = a.copy()
                a_try[i] += d
                a_try[j] -= d
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 0.01:
                    continue
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if not improved:
                # Single coordinate: increase
                for d in [delta, delta * 0.1]:
                    a_try = a.copy()
                    a_try[i] += d
                    s_try = float(np.sum(a_try))
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

                if not improved:
                    # Single coordinate: decrease
                    if a[i] > delta:
                        for d in [delta, delta * 0.1]:
                            a_try = a.copy()
                            a_try[i] -= d
                            a_try = np.maximum(a_try, 0)
                            s_try = float(np.sum(a_try))
                            if s_try < 0.01:
                                continue
                            a_try = a_try / s_try
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj - 1e-14:
                                best_obj = obj_try
                                best_a = a_try.copy()
                                a = a_try
                                improved = True
                                break

        if verbose and pass_count % 3 == 0:
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}, delta={delta:.2e}, time={time.monotonic()-start:.1f}s")

        if not improved:
            delta *= 0.5
            if delta < 1e-8:
                break

    return best_a


def block_optimize(a, n_blocks=8, block_size=50, time_budget=60.0, verbose=False):
    """
    Block coordinate optimization: optimize each block of coordinates while fixing others.
    Uses LP on the block variables.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999 + n)

    n_actual_blocks = min(n_blocks, n // block_size)
    if n_actual_blocks < 1:
        return best_a

    while time.monotonic() - start < time_budget:
        improved_any = False
        # Random block order
        block_starts = list(range(0, n - block_size + 1, max(1, (n - block_size) // n_actual_blocks)))
        rng.shuffle(block_starts)

        for bs in block_starts[:n_actual_blocks]:
            if time.monotonic() - start > time_budget:
                break

            be = min(bs + block_size, n)
            # Variables: a[bs:be], with sum(a[bs:be]) = s_block
            s_block = float(np.sum(a[bs:be]))
            if s_block < 1e-6:
                continue

            # The convolution contribution from this block:
            # We want to minimize max_k conv[k] by adjusting a[bs:be]
            # This is a sub-problem. Use a simple approach:
            # Try random perturbations of the block
            for trial in range(5):
                # Random direction in the block
                direction = rng.normal(0, 1, be - bs)
                direction -= np.mean(direction)  # Keep sum approximately constant
                direction = direction / (np.max(np.abs(direction)) + 1e-10)

                for scale in [0.01, 0.005, 0.001]:
                    a_try = a.copy()
                    a_try[bs:be] = np.maximum(a[bs:be] + scale * direction, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 0.01:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved_any = True
                        break

        if not improved_any and verbose:
            print(f"    Block: no improvement, obj={best_obj:.10f}, time={time.monotonic()-start:.1f}s")
            if time.monotonic() - start > time_budget * 0.5:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use cubic interpolation for smoother result
    try:
        from scipy.interpolate import CubicSpline
        cs = CubicSpline(x_old, a, bc_type='natural')
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def spectral_focusing(a, time_budget=60.0, verbose=False):
    """
    Frequency-domain approach: identify spectral components that cause
    high autocorrelation peaks and suppress them.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))

        # FFT of a
        size = 1
        while size < 2 * n:
            size <<= 1
        fa = rfft(a, size)

        # The autocorrelation in frequency domain:
        # |FFT(a)|^2 is the periodogram, and the autocorrelation is IFFT(|FFT|^2)
        # We want to reduce the peaks in the autocorrelation.
        # In frequency domain, this means making |FFT(a)|^2 more uniform.
        # But we need to keep a >= 0, so we can't directly modify the FFT.

        # Alternative: use the gradient information from the peaks
        # to design a frequency-domain filter
        # Find the top peak locations
        peak_idx = np.argmax(conv)
        # The peak at lag k corresponds to a specific "frequency" in some sense
        # For the main peak (k ≈ n-1), it's the DC component
        # For side peaks, they correspond to specific frequency bins

        # Simple approach: compute the "autocorrelation spectrum" and apply a Wiener-like filter
        # This is essentially what gradient descent does, but in frequency domain
        # Let's do a simple spectral reweighting:
        # Reduce the magnitude of FFT bins that contribute most to the peaks
        mag = np.abs(fa)
        phase = np.angle(fa)

        # Compute contribution of each bin to the peak
        # peak = sum_k w[k]*conv[k], gradient tells us which time-domain samples to change
        # In frequency domain: the gradient of |FFT(a)|^2 w.r.t. a is related to the phase

        # Simpler: just do a few iterations of "flatten the spectrum" while maintaining non-negativity
        # Target: make |FFT(a)| as flat as possible (white noise spectrum)
        target_mag = np.ones_like(mag) * np.mean(mag)
        # Blend: new_mag = (1-alpha)*mag + alpha*target_mag
        alpha = 0.05
        new_mag = (1 - alpha) * mag + alpha * target_mag
        fa_new = new_mag * np.exp(1j * phase)
        a_new = irfft(fa_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 0.01:
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert
            a = best_a.copy()

        if verbose and it % 20 == 0:
            print(f"    Spectral it{it}: obj={best_obj:.10f}, time={time.monotonic()-start:.1f}s")

    return best_a


def multi_direction_lp(a, n_dirs=3, time_budget=60.0):
    """
    Solve LP from multiple perturbed starting points to get diverse directions,
    then pick the best blend.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(777 + n)

    for d in range(n_dirs):
        if time.monotonic() - start > time_budget:
            break

        # Perturb a slightly
        if d == 0:
            a_pert = a.copy()
        else:
            perturbation = rng.normal(0, 0.005, n)
            a_pert = a * np.exp(perturbation)
            a_pert = np.maximum(a_pert, 0)
            s_p = float(np.sum(a_pert))
            if s_p > 1e-10:
                a_pert = a_pert / s_p

        # Get LP direction
        g_lp, M_lp = solve_lp_primal(a_pert)
        if g_lp is None:
            continue

        # Try blending
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a
