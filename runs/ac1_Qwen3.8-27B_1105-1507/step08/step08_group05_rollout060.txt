- At k=j+i and k=j-i (for j≠0): a_j*a_i → a_j*(a_i+δ), change = δ*a_j

So the change in conv at position k is:
- If k = i: 2*a_i*δ + δ^2

Actually, (a*a)[k] = sum_j a[j]*a[k-j]. If I change a[i] to a[i]+δ:

- The term a[k-i]*a[i] changes to a[k-i]*(a[i]+δ): change = δ*a[k-i]

(a_new * a_new)[k] = sum_j a_new[j] * a_new[k-j]

The terms that change are those where j = i or k-j = i:
- j = i: a_new[i] * a_new[k-i] = (a[i]+δ) * a_new[k-i] (if k-i ≠ i) or (a[i]+δ)^2 (if k-i = i)

For k = 2i:

This is useful for coordinate descent. The objective is:
J = 2n * max_k conv[k] / (sum(a))^2

If I increase a_i by δ, the sum changes from S to S+δ. The max might change too.

The max of a piecewise linear (or quadratic at k=2i) function divided by a quadratic is a nice 1D optimization. I can use golden section search or just evaluate at candidate points.

"""
Combined optimizer: Mirror descent with soft-max + LP alternating + coordinate descent.
Targets the 1.5027 bound through smooth gradient methods, dual-informed updates,
and careful local polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_weights(x, tau):
    x_max = np.max(x)
    exp_x = np.exp((x - x_max) / tau)
    return exp_x / np.sum(exp_x)


def grad_softmax_conv(a, tau):
    """Gradient of soft_max(conv(a,a), tau) w.r.t. a using FFT."""
    n = len(a)
    conv = compute_conv(a)
    w = softmax_weights(conv, tau)
    
    # c[m] = sum_{j=0}^{n-1} a[j] * w[j+m]
    # c[m] = conv_full[(2n-2) - m] where conv_full = a * w[::-1] (linear)
    w_rev = w[::-1]  # length 2n-1
    conv_size = n + (2*n - 1) - 1  # = 3n - 2
    size = 1
    while size < conv_size:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w_rev, size)
    conv_full = irfft(fa * fw, size)[:conv_size]
    
    # grad[m] = 2 * conv_full[2n-2-m] for m=0..n-1
    # i.e., grad = 2 * conv_full[n-1:2n-1][::-1]
    grad = 2.0 * conv_full[n-1:2*n-1][::-1]
    return grad


def mirror_descent(a0, time_budget=60.0, tau_start=0.05, tau_end=0.005, 
                   n_iters=None, verbose=False):

    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

        n_iters = 50000
    
    base_eta = 0.15
    patience = 0
    best_patience = 0
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal temperature
        frac = min(1.0, it / float(n_iters * 0.8))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Step size decay
        eta = base_eta / (1.0 + 0.05 * np.sqrt(it))
        
        # Compute gradient
        g = grad_softmax_conv(a, tau)
        g = g - np.mean(g)  # Center for simplex
        
        # Mirror descent update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-15)

                patience = 0

    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None

def coordinate_descent(a0, time_budget=60.0, n_sweeps=20, verbose=False):

    n = len(a0)

    if s < 1e-10:
        return a0.copy()

    best_obj = compute_obj(a)
    start = time.monotonic()
    
    rng = np.random.default_rng(123)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        # Randomize order
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                    break
            
            a_i = a[i]
            if a_i < 1e-12:
                continue
            
            # Try scaling a_i by various factors
            best_local = best_obj
            best_factor = 1.0
            for factor in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.1]:
                a[i] = a_i * factor
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a[i] = a_i
                    continue
                obj_try = compute_obj(a / s_new)
                if obj_try < best_local:
                    best_local = obj_try
                    best_factor = factor
            
            if best_factor != 1.0:
                a[i] = a_i * best_factor
                a = a / float(np.sum(a))
                if best_local < best_obj:
                    best_obj = best_local
                    best_a = a.copy()
            else:
                a[i] = a_i
    
    return best_a


def resample_seq(a, new_n, method='cubic'):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    if method == 'cubic' and n >= 4:
        from scipy.interpolate import CubicSpline
        try:
            cs = CubicSpline(x_old, a, bc_type='natural')
            result = cs(x_new)
            return np.maximum(result, 0)
        except Exception:
            pass
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def gaussian_window(n, sigma_frac=0.3):
    x = np.arange(n)
    center = (n - 1) / 2.0
    w = np.exp(-0.5 * ((x - center) / (n * sigma_frac)) ** 2)
    return w / np.sum(w)


def exponential_window(n, rate=0.5):
    x = np.arange(n)
    center = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(x - center) / center)
    return w / np.sum(w)

Combined optimizer: Mirror descent with soft-max + LP alternating + coordinate descent.
Uses multiplicative weights for natural simplex handling, annealed soft-max for smooth
gradients, LP dual information for informed peak selection, and coordinate descent for
final polishing. Exploits the non-symmetric structure to achieve sub-2 objectives.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar
from scipy.interpolate import CubicSpline


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_weights(x, tau):
    """Compute softmax weights for soft-max approximation."""
    x_max = np.max(x)
    exp_x = np.exp((x - x_max) / tau)
    return exp_x / np.sum(exp_x)


def grad_softmax_conv(a, tau):
    """
    Gradient of soft_max(conv(a,a), tau) w.r.t. a using FFT.
    Soft-max gives: sum_k w_k * conv[k] where w = softmax(conv/tau)
    grad[m] = 2 * sum_j w[j+m] * a[j]
    Computed via: grad = 2 * (a * w_rev)[2n-2-m] using linear convolution.
    """
    n = len(a)
    conv = compute_conv(a)
    w = softmax_weights(conv, tau)

    # c[m] = sum_{j=0}^{n-1} a[j] * w[j+m]
    # c[m] = conv_full[2n-2-m] where conv_full = a * w[::-1] (linear conv)
    w_rev = w[::-1]  # length 2n-1
    conv_size = n + (2 * n - 1) - 1  # = 3n - 2
    size = 1
    while size < conv_size:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w_rev, size)
    conv_full = irfft(fa * fw, size)[:conv_size]

    # grad = 2 * conv_full[n-1 : 2n-1][::-1]
    grad = 2.0 * conv_full[n - 1:2 * n - 1][::-1]
    return grad


def mirror_descent(a0, time_budget=60.0, tau_start=0.05, tau_end=0.005,
                   base_eta=0.12, verbose=False):
    """
    Mirror descent (multiplicative weights) with annealed soft-max gradient.
    Update: a_i <- a_i * exp(-eta * g_i) / Z
    Naturally maintains a >= 0.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    patience = 0
    cur_eta = base_eta
    max_iters = 200000

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Anneal temperature
        frac = min(1.0, it / 5000.0)
        tau = tau_start * (tau_end / tau_start) ** frac

        # Decaying step size
        eta = cur_eta / (1.0 + 0.03 * np.sqrt(float(it)))

        # Compute soft-max gradient
        g = grad_softmax_conv(a, tau)
        g = g - np.mean(g)  # Center for simplex compatibility

        # Mirror descent (multiplicative weights) update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)

                best_obj = obj
                best_a = a_new.copy()
                patience = 0
            else:
                patience += 1

            # Adaptive step size
            if patience > 80:
                cur_eta = max(cur_eta * 0.85, 0.003)
                patience = 0
            elif it > 0 and it % 500 == 0:
                cur_eta = max(cur_eta * 0.95, 0.005)

        a = a_new

        if verbose and it % 5000 == 0:
            print(f"      MD it={it}: best={best_obj:.8f}, eta={cur_eta:.4f}, t={time.monotonic()-start:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=60.0, verbose=False):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        # 1D line search
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            return compute_obj(a_try / st)

        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 30, 'xatol': 0.005})
            t_best = res.x
        except Exception:
            t_best = 0.5

        a_try = (1.0 - t_best) * a + t_best * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={t_best:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 12:
                break

    return best_a


def coordinate_descent(a0, time_budget=60.0, n_sweeps=10, verbose=False):
    """
    Coordinate descent: for each coordinate, try scaling by several factors.
    Also tries small local perturbations around the current value.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    factors = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 0.3, 2.0]

    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            a_i = a[i]
            if a_i < 1e-14:
                continue

            best_local_obj = best_obj
            best_f = 1.0

            for f in factors:
                a[i] = a_i * f
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    a[i] = a_i
                    continue
                obj_try = compute_obj(a / s_new)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_f = f

            if best_f != 1.0:
                a[i] = a_i * best_f
                a = a / float(np.sum(a))
                if best_local_obj < best_obj:
                    best_obj = best_local_obj
                    best_a = a.copy()
                    improved = True
            a[i] = a * float(np.sum(a)) / float(np.sum(a))  # ensure consistency
            # Actually, a is already normalized, just fix this coordinate
            a[i] = a[i]  # no-op, a is already correct

        if not improved and sweep > 3:
            break

    return best_a


def coordinate_descent_fine(a0, time_budget=60.0, n_sweeps=5, verbose=False):
    """Fine coordinate descent using 1D minimization for each coordinate."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(7)

    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            a_i = a[i]
            if a_i < 1e-14:
                continue

            def obj_1d(log_scale):
                new_val = a_i * np.exp(log_scale)
                a[i] = new_val
                s_new = float(np.sum(a))
                if s_new < 1e-10:
                    return np.inf
                obj = compute_obj(a / s_new)
                a[i] = a_i  # restore
                return obj

            # Restore a[i] before optimization
            a[i] = a_i
            try:
                res = minimize_scalar(obj_1d, bounds=(-1.5, 1.5), method='bounded',
                                      options={'maxiter': 15, 'xatol': 0.05})
                if res.success:
                    new_val = a_i * np.exp(res.x)
                    a[i] = new_val
                    s_new = float(np.sum(a))
                    if s_new > 1e-10:
                        obj_try = compute_obj(a / s_new)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a.copy()
                            a = a / s_new
                            improved = True
            except Exception:
                a[i] = a_i

        if not improved and sweep > 2:
            break

    return best_a


def resample_seq(a, new_n, method='cubic'):
    """Resample sequence to new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    if method == 'cubic' and n >= 4:
        try:
            cs = CubicSpline(x_old, a, bc_type='natural')
            result = cs(x_new)
            return np.maximum(result, 0)
        except Exception:
            pass
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def gaussian_window(n, sigma_frac=0.3):
    """Gaussian window centered."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    w = np.exp(-0.5 * ((x - center) / (n * sigma_frac)) ** 2)
    return w / np.sum(w)


def ramp_window(n):
    """Linear ramp (asymmetric)."""
    w = np.arange(1, n + 1, dtype=np.float64)
    return w / np.sum(w)


def exp_window(n, rate=1.0):
    """Exponential decay (asymmetric)."""
    x = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * x / n)
    return w / np.sum(w)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Mirror descent with soft-max + LP alternating + coordinate descent.
    Uses diverse (including asymmetric) initializations, n-sweeping, and
    multi-method alternating for robust convergence.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = hann_window(800)
        best_score = compute_obj(best_seq)
        print(f"Hann init (n=800): obj={best_score:.8f}")

    # === Phase 1: Diverse initialization (60s) ===
    print("\n=== Phase 1: Diverse Initialization ===")
    n_base = len(best_seq)

    inits = [
        ("parent", best_seq.copy()),
        ("hann", hann_window(n_base)),
        ("gauss025", gaussian_window(n_base, 0.25)),
        ("gauss035", gaussian_window(n_base, 0.35)),
        ("ramp", ramp_window(n_base)),
        ("exp1", exp_window(n_base, 1.0)),
        ("exp2", exp_window(n_base, 2.0)),
        ("half_gauss", None),  # asymmetric gaussian
    ]

    # Create half-gaussian (shifted)
    x = np.arange(n_base, dtype=np.float64)
    hg = np.exp(-0.5 * ((x - n_base * 0.4) / (n_base * 0.25)) ** 2)
    hg = np.maximum(hg, 0)
    inits[7] = ("half_gauss", hg / np.sum(hg))

    for name, init in inits:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        obj = compute_obj(init)
        if obj < best_score:
            best_score = obj
            best_seq = init.copy()
            print(f"  {name}: obj={best_score:.8f} **")
        else:
            print(f"  {name}: obj={obj:.8f}")

    # Quick mirror descent on best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_md = mirror_descent(best_seq, time_budget=min(30, remaining * 0.4),
                              tau_start=0.05, tau_end=0.01, base_eta=0.15)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD quick: obj={best_score:.8f} **")
        else:
            print(f"  MD quick: obj={obj_md:.8f}")

    # === Phase 2: Main optimization loop (800s) ===
    print(f"\n=== Phase 2: Main Optimization (budget ~{min(800, deadline - time.monotonic()):.0f}s) ===")
    a = best_seq.copy()
    n_cur = len(a)
    main_start = time.monotonic()
    main_budget = min(800.0, (deadline - main_start) - 200.0)

    iteration = 0
    while time.monotonic() - main_start < main_budget:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        phase = iteration % 7

        if phase == 0 or phase == 1:
            # Mirror descent with varying parameters
            alloc = min(70.0, remaining * 0.35)
            cycle = iteration // 7
            tau_s = max(0.008, 0.04 * (0.8 ** cycle))
            tau_e = max(0.002, tau_s * 0.2)
            eta_s = max(0.04, 0.12 * (0.9 ** cycle))
            a_new = mirror_descent(a, time_budget=alloc, tau_start=tau_s,
                                   tau_end=tau_e, base_eta=eta_s)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  [MD] iter {iteration}: obj={best_score:.8f} **",
                      f" (tau={tau_s:.4f}, eta={eta_s:.4f})")
            else:
                print(f"  [MD] iter {iteration}: obj={obj_new:.8f}")
                if iteration % 7 >= 2:
                    # Perturb to escape
                    a = best_seq * np.exp(rng.normal(0, 0.02, n_cur))
                    a = np.maximum(a, 1e-10)
                    a = a / np.sum(a)

        elif phase == 2 or phase == 3:
            # LP alternating
            alloc = min(90.0, remaining * 0.35)
            a_new = lp_alternate(a, time_budget=alloc, verbose=True)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()

            alloc = min(40.0, remaining * 0.2)
            a_new = coordinate_descent(a, time_budget=alloc, n_sweeps=8)

                best_seq = a_new.copy()
                a = a_new.copy()

        elif phase == 5:
            # Mirror descent with high temperature (exploration)
            alloc = min(40.0, remaining * 0.2)
            a_new = mirror_descent(a, time_budget=alloc, tau_start=0.1,
                                   tau_end=0.02, base_eta=0.2)

            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  [MD-ex] iter {iteration}: obj={best_score:.8f} **")
            else:
                print(f"  [MD-ex] iter {iteration}: obj={obj_new:.8f}")

        else:
            # Fine coordinate descent
            alloc = min(35.0, remaining * 0.2)
            a_new = coordinate_descent_fine(a, time_budget=alloc, n_sweeps=4)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  [CDf] iter {iteration}: obj={best_score:.8f} **")
            else:
                print(f"  [CDf] iter {iteration}: obj={obj_new:.8f}")

    # === Phase 3: n-sweeping (150s) ===
    print(f"\n=== Phase 3: n-Sweep ===")
    n_options = [400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600]
    n_options = sorted(set(n_options + [len(best_seq)]))
    n_options = [n for n in n_options if 300 <= n <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try, 'cubic')

        # Quick optimization at this n
        a_md = mirror_descent(a_try, time_budget=min(12, remaining * 0.15),
                              tau_start=0.03, tau_end=0.005, base_eta=0.1)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_md:.8f}")

    # === Phase 4: Final polish (200s) ===
    print(f"\n=== Phase 4: Final Polish ===")
    a = best_seq.copy()
    n_cur = len(a)

    # Aggressive fine coordinate descent
    remaining = deadline - time.monotonic()

        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a = a_cd.copy()
            print(f"  CD fine: obj={best_score:.8f} **")
        else:
            print(f"  CD fine: obj={obj_cd:.8f}")

    # LP final
    remaining = deadline - time.monotonic()
    if remaining > 25:
        a_lp = lp_alternate(a, time_budget=min(50, remaining * 0.3), verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP final: obj={best_score:.8f} **")

    return result
