`min 0.5 * ||g - a||^2 + (1/alpha) * M s.t. conv(a,g) <= M, sum(g)=1, g>=0`

This is a QP with linear constraints. MOSEK and CLARABEL can handle this.

But the benefit is modest. Let me stick with the LP approach.

**Another idea: Use the LP dual to get a better gradient**

After solving the LP, the dual solution y gives weights on the convolution lags. We can use these weights to compute a "weighted" gradient:
`grad_w = 2 * A^T @ y`

This gradient points in the direction that would increase the weighted sum of peaks, so we want to move against it. Combined with the simplex constraint, this gives a Frank-Wolfe step.

Actually, this is exactly what the LP is already doing! The LP solution g is the Frank-Wolfe direction. So there's no additional benefit.

Let me just write a clean, efficient implementation. Here's my final plan:

1. **Phase 1 (300s)**: Smoothed gradient descent to quickly improve the objective
2. **Phase 2 (1000s)**: LP polishing with line search (the main workhorse)
   - Solve LP, do line search
   - Every 30 iterations, add a small random perturbation
   - Track improvement; if stuck, increase perturbation
3. **Phase 3 (150s)**: Multi-length exploration
   - Try n-100, n+100, n+200
   - Quick LP polish for each
4. **Phase 4 (50s)**: Final polish on the best candidate

"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: FFT-smoothed gradient descent for initial improvement, then intensive
LP-based polishing with golden-section line search and adaptive perturbation.
Multi-length exploration at the end.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, time_limit=15.0):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0).copy()
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None


def golden_line_search(a, g, max_iter=40):
    """Golden section search for best t in [0,1] minimizing obj((1-t)*a + t*g)."""
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    
    def obj_at(t):
        x = (1 - t) * a + t * g
        x = np.maximum(x, 0)
        s = float(np.sum(x))
        if s < 1e-10:
            return np.inf
        x = x / s
        return compute_obj(x)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)
    
    for _ in range(max_iter):
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)
        if hi - lo < 1e-7:
            break
    
    best_t = (lo + hi) / 2.0
    best_x = (1 - best_t) * a + best_t * g
    best_x = np.maximum(best_x, 0)
    s = float(np.sum(best_x))
    if s > 1e-10:
        best_x = best_x / s
    best_obj = compute_obj(best_x)
    
    return best_x, best_obj, best_t


def smoothed_gradient(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # Cross-correlation: g[i] = 2 * sum_k p[k] * a[k-i]
    N = 1
    while N < 3 * n:
        N <<= 1
    from scipy.fft import fft, ifft
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    return 2.0 * g_full[:n], conv


def nesterov_descent(a, time_budget, tau_init=5.0, tau_end=500.0):
    """Nesterov accelerated smoothed gradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.6)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g, _ = smoothed_gradient(y, tau)
        
        # Barzilai-Borwein step
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 5.0 / np.sqrt(n))
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
        
        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-9, lr * 0.9999)
        
        if it - best_it > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it
    
    return best_a


def lp_polish(a, time_budget, rng, verbose=True):
    """LP-based polishing with golden-section line search and adaptive perturbation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    stall = 0
    perturbation_scale = 0.0
    last_improve_it = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5.0:
            break
        
        # Solve LP
        g_lp, M_lp = solve_lp(a, time_limit=min(15.0, remaining * 0.5))
        if g_lp is None:
            break
        
        # Line search
        a_new, obj_new, t_used = golden_line_search(a, g_lp, max_iter=40)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            last_improve_it = it
            # Decrease perturbation on success
            perturbation_scale = max(0.0, perturbation_scale * 0.5)
            if verbose and it % 20 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_used:.4f}, M={M_lp:.8f}")
        else:
            stall += 1
            a = best_a.copy()
            
            if stall > 10:
                # Increase perturbation
                perturbation_scale = min(0.1, perturbation_scale + 0.01)
                noise = rng.normal(0, perturbation_scale, n)
                a = best_a * np.exp(noise)
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                stall = 0
                if verbose:
                    print(f"    LP it{it}: stuck, perturbing (scale={perturbation_scale:.3f})")
            
            if stall > 50:
                # Larger restart
                perturbation_scale = 0.05
                noise = rng.normal(0, 0.05, n)
                a = best_a * np.exp(noise)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
        
        it += 1
    
    if verbose and it > 0:
        print(f"    LP polished: {it} iterations, obj={best_obj:.8f}")
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 1e-6))
        a = a / a.sum()
    elif mode == 3:
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    elif mode == 5:
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x * rng.uniform(1, 3))
        a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n))
    return a
