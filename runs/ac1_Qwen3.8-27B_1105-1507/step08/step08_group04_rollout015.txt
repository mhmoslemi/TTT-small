"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
- Edge-heavy structured initializations (theoretical insight)
- LP alternating with Nesterov acceleration and adaptive line search
- Smoothed gradient polish via log-sum-exp with FFT-based gradients
- Multi-n exploration with progressive resampling
- Sparse matrix LP for efficiency

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time
import math


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build dense matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        rows.extend(range(j, j + n))
        cols.extend([j] * n)
        vals.extend(a.tolist())
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A_sparse


def solve_lp_fast(a):
    """Solve: min M s.t. A@t - M<=0, sum(t)=1, t>=0, M free.
    Uses scipy linprog with HiGHS. Returns (g, M_val) or (None, None)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s

    A_sparse = build_conv_matrix(a)
    A_dense = A_sparse.toarray()

    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    # Inequality: A @ g - M <= 0  =>  [A, -1] @ [g; M] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A_dense
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds: g >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0.0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def line_search_blend(a, g, max_evals=15):

    best_t = 0.3
    best_obj = np.inf
    # Coarse search
    t_candidates = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 1.0]
    for t in t_candidates:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    # Fine search around best
    if max_evals > len(t_candidates):
        t_lo = max(0.0, best_t - 0.1)
        t_hi = min(1.0, best_t + 0.1)
        for t in np.linspace(t_lo, t_hi, 8):
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_t = t
    return best_t, best_obj


def lp_alternate_nesterov(a, time_budget=200.0, verbose=False):
    """LP alternating with Nesterov momentum on the step size."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()

    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    # Nesterov-like: maintain a "momentum" direction
    momentum_dir = np.zeros(n)
    momentum_alpha = 0.0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_fast(a)
        if g_lp is None:
            break

        # Try blending with momentum
        blended_g = g_lp.copy()
        if momentum_alpha > 0.05:

                best_a = a_new.copy()
                # Update momentum

                momentum_alpha = min(momentum_alpha + 0.05, 0.4)
                a = a_new.copy()
                stall = 0
                if verbose:
                    elapsed = time.monotonic() - start
                    print(f"    LP+nest: obj={best_obj:.8f}, t={best_t:.3f}, alpha={momentum_alpha:.2f}, time={elapsed:.0f}s")

                stall += 1

            stall += 1

    b = a / float(np.sum(a))
    conv = compute_conv(b)

    # Log-sum-exp weights
    shifted = conv / tau
    shifted = shifted - np.max(shifted)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / float(np.sum(exp_shifted))
    smooth_max = tau * (float(np.max(shifted)) + math.log(float(np.sum(exp_shifted))))

    obj = 2.0 * n * smooth_max

    # Gradient computation using FFT
    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1
    B_fft = rfft(b, n_fft)
    W_fft = rfft(w, n_fft)

    # R[j] = sum_k w[k] * b[k-j] for 0<=k-j<n (circular, but no wrap due to padding)
    R = irfft(np.conj(W_fft) * B_fft, n_fft)[:n]

    # S[j] = sum_{k=j}^{j+n-1} w[k]
    w_ext = np.zeros(3 * n)
    w_ext[:2 * n - 1] = w
    prefix = np.concatenate([[0.0], np.cumsum(w_ext)])
    S = prefix[n:2 * n] - prefix[:n]  # S[j] = prefix[j+n] - prefix[j] for j=0..n-1

    grad = 2.0 * n * (R + b * S)
    grad -= np.mean(grad)  # Center for simplex constraint

    return obj, grad


def smoothed_gradient_optimize(a, time_budget=60.0, verbose=False):
    """Optimize using smoothed gradient with decreasing temperature."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5 / math.sqrt(n)
    tau = 0.02  # Start with smooth approximation

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        obj_sm, grad = smoothed_gradient_obj(a, tau=tau)
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-10:
            grad = grad / gnorm

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Decrease tau and lr
        if it % 500 == 499:
            tau *= 0.9
            lr *= 0.97

        if verbose and it % 2000 == 0:
            print(f"    SmoothGrad: it={it}, obj={best_obj:.8f}, tau={tau:.5f}")

    return best_a


def edge_heavy_init(n, edge_ratio=3.0, rng=None):
    """Create edge-heavy initialization."""
    if rng is None:
        rng = np.random.default_rng(42)
    a = np.ones(n) * 0.5
    # Boost edges
    edge_weight = edge_ratio
    a[0] = edge_weight
    a[-1] = edge_weight
    # Also boost near-edges with decay
    decay = int(n * 0.1)
    for i in range(decay):
        w = edge_weight * np.exp(-i / (decay * 0.3))
        a[i + 1] = max(a[i + 1], w)
        a[n - 2 - i] = max(a[n - 2 - i], w)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    return a / s


def diverse_init(n, rng):
    """Generate diverse structured initialization."""
    choice = rng.integers(0, 7)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Edge-heavy cosine
        a = 0.3 + 0.7 * np.cos(np.pi * x) ** 2
    elif choice == 1:
        # Two bumps
        a = np.exp(-60 * (x - 0.2) ** 2) + np.exp(-60 * (x - 0.8) ** 2) + 0.1
    elif choice == 2:
        # Power law (edge-heavy)
        a = np.maximum(x, 0.01) ** (-1.0) + np.maximum(1 - x, 0.01) ** (-1.0) + 0.1
    elif choice == 3:
        # Oscillatory with edge boost
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 5 * x) + 0.5 * (np.exp(-20 * x) + np.exp(-20 * (x - 1)))
    elif choice == 4:
        # Beta-like (U-shaped)
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif choice == 5:
        # Random smooth
        a = np.ones(n)
        for _ in range(4):
            center = rng.integers(0, n)
            width = rng.integers(n // 8, n // 3)
            for i in range(n):
                d = min(abs(i - center), n - abs(i - center))
                a[i] += rng.uniform(0.5, 2.0) * np.exp(-d / (width * 0.4))
        a = np.maximum(a, 0.01)
    else:
        # Mixed: edge-heavy + oscillation
        a = np.exp(-15 * x) + np.exp(-15 * (x - 1)) + 0.2 * (1 + np.cos(2 * np.pi * 3 * x))
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_seq(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, kind='cubic')
    return np.maximum(a_new, 0.0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP alternating with Nesterov acceleration,
    smoothed gradient polish, edge-heavy initializations, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a_parent = parent / s
        n_base = len(a_parent)
        print(f"Parent loaded: n={n_base}, obj={compute_obj(a_parent):.8f}")
    except Exception:
        n_base = 800
        a_parent = edge_heavy_init(n_base, edge_ratio=4.0, rng=rng)
        print(f"No parent, using edge-heavy init: n={n_base}, obj={compute_obj(a_parent):.8f}")

    best_a = a_parent.copy()
    best_obj = compute_obj(a_parent)

    # === Generate multiple initializations ===
    inits = [best_a.copy()]

    # Edge-heavy variants
    for er in [2.0, 4.0, 8.0, 15.0]:
        inits.append(edge_heavy_init(n_base, edge_ratio=er, rng=rng))

    # Diverse inits
    for _ in range(4):
        inits.append(diverse_init(n_base, rng))

    # Evaluate all inits
    print("\n=== Initialization evaluation ===")
    init_objs = []
    for i, a_init in enumerate(inits):
        obj = compute_obj(a_init)
        init_objs.append(obj)
        if obj < best_obj:
            best_obj = obj
            best_a = a_init.copy()
    init_objs_sorted = np.argsort(init_objs)
    print(f"  Best init obj: {init_objs[init_objs_sorted[0]]:.8f} (index {init_objs_sorted[0]})")
    print(f"  Current best: {best_obj:.8f}")

    # === Phase 1: LP Alternating on best init ===
    print(f"\n=== Phase 1: LP Alternating (primary, n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 400.0)
    a_p1 = lp_alternate_nesterov(best_a, time_budget=p1_budget, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  ** Phase 1 result: obj={best_obj:.8f} **")
    else:
        print(f"  Phase 1 result: obj={obj_p1:.8f}")

    # === Phase 2: Smoothed gradient polish ===
    print(f"\n=== Phase 2: Smoothed Gradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p2_budget = min(remaining * 0.15, 150.0)
        a_sg = smoothed_gradient_optimize(best_a, time_budget=p2_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  ** SmoothGrad: obj={best_obj:.8f} **")
        else:
            print(f"  SmoothGrad: obj={obj_sg:.8f}")

    # === Phase 3: LP Alternating Round 2 ===
    print(f"\n=== Phase 3: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p3_budget = min(remaining * 0.25, 300.0)
        a_p3 = lp_alternate_nesterov(best_a, time_budget=p3_budget, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            print(f"  ** Phase 3 result: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 3 result: obj={obj_p3:.8f}")

    # === Phase 4: Try other initializations with LP ===
    print(f"\n=== Phase 4: Other Initializations ===")
    remaining = deadline - time.monotonic()
    # Try the top 3 other inits (excluding the one we already used)
    tried = set()
    for idx in init_objs_sorted:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        if idx == init_objs_sorted[0] and compute_obj(inits[idx]) > best_obj + 0.01:
            continue
        a_try = inits[idx].copy()
        alloc = min(60.0, remaining * 0.2)
        a_opt = lp_alternate_nesterov(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_opt)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** Init {idx}: obj={best_obj:.8f} **")
        else:
            print(f"  Init {idx}: obj={obj_try:.8f}")
        if len(tried) >= 3:
            break
        tried.add(idx)

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        max(100, n_best - 200),
        max(100, n_best - 100),
        n_best,
        n_best + 100,
        n_best + 200,
        n_best + 400,
        n_best + 800,
    ]))
    n_options = [x for x in n_options if 100 <= x <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            s = float(np.sum(a_try))
            if s > 1e-10:
                a_try = a_try / s

        # Also try edge-heavy at this n
        a_try2 = edge_heavy_init(n_try, edge_ratio=5.0, rng=rng)

        # Pick the better starting point
        if compute_obj(a_try2) < compute_obj(a_try):
            a_try = a_try2

        alloc = min(50.0, remaining * 0.25)
        a_opt = lp_alternate_nesterov(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_opt)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc_lp = min(remaining * 0.4, 100.0)
        a_final = lp_alternate_nesterov(best_a, time_budget=alloc_lp, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  ** Final LP: obj={best_obj:.8f} **")

    # Smoothed gradient micro-polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_sg = min(remaining * 0.3, 50.0)
        a_sg = smoothed_gradient_optimize(best_a, time_budget=alloc_sg, verbose=False)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  ** Final SmoothGrad: obj={best_obj:.8f} **")

    # One more LP if time allows
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_last = min(remaining * 0.5, 40.0)
        a_last = lp_alternate_nesterov(best_a, time_budget=alloc_last, verbose=False)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  ** Last LP: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
