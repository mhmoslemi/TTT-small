    k_indices = np.argsort(conv)[::-1][:top_k]
    
    A = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        k = int(k)
        # conv[k] = sum_i a[i] * a[k-i] for valid i
        # dconv/dg[i] = a[k-i] if 0 <= k-i < n, else 0
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_max >= i_min:
            A[idx, i_min:i_max+1] = a[k - i_max : k - i_min + 1]
    return A

def solve_lp_reduced(a, top_k=50):
    """Solve reduced LP: min M s.t. A_reduced @ g <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix_for_lp(a_norm, top_k)
    m = A.shape[0]
    
    # Variables: g (n) + M (1) = n+1
    # Objective: minimize M -> c = [0,...,0,1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A @ g - M <= 0 for each row
    # [A, -1] @ [g; M] <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs > 1e-10:
            return g / gs, float(res.x[n])
    return None, None

def multiplicative_weights(a, time_budget=30.0, eta_initial=0.01, verbose=False):
    """Multiplicative weights (mirror descent on simplex) for the objective."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)

    eta = eta_initial
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_val = np.max(conv)
        
        # Find active constraints (within 1% of max)
        active = conv >= max_val * 0.99
        n_active = np.sum(active)
        if n_active == 0:
            active = np.zeros(2*n-1, dtype=bool)
            active[int(np.argmax(conv))] = True
        
        # Compute subgradient: average of active constraint gradients
        # grad[i] = (1/|A|) * sum_{k in A} a[k-i] (for valid k-i)
        # This is like a filtered version of a
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_max >= i_min:
                grad[i_min:i_max+1] += a[k - i_max : k - i_min + 1]
        grad /= n_active
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)
        
        a = a_new
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Decay eta
        if it % 100 == 99:
            eta *= 0.995
    
    return best_a

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    n = len(a)
    size = 1

    fa = rfft(a, size)

"""
Optimized LP-alternating method for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses full LP solves, momentum, adaptive step size, multi-n exploration,
and multiplicative weights for fast pre-optimization.

import numpy as np

from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_full_conv_matrix(a):
    n = len(a)

        i_max = min(k, n - 1)
        if i_max >= i_min:
            A[k, i_min:i_max + 1] = a[k - i_max:k - i_min + 1]
    return A
