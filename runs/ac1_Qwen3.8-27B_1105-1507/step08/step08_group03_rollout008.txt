def grad_conv_k(a, k):
    """Gradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

The key insight is that this is NOT a subgradient method per se - it's a game-theoretic approach. The convergence guarantee is different.

For the MWU implementation:
- Row player (chooses k): updates y_k proportional to exp(-eta * payoff_k)
- Column player (chooses j): updates g_j proportional to exp(-eta * payoff_j)
- Payoff for column j given row distribution y: (B^T y)[j]
- Payoff for row k given column distribution g: (B g)[k] = conv(a, g)[k]

After T iterations of MWU, the averaged iterates converge to the game value at rate O(sqrt(log(K)/T)).

For our purposes, we want the column player's iterate g (which is our direction). So we run MWU for T iterations and use the averaged g (or the last g).

```python
def mwu_direction(a, n_iters=50, eta=None):
    """
    Find g that approximately minimizes max_k conv(a, g)
    using multiplicative weights.
    Returns g (normalized to sum 1).
    """
    n = len(a)
    a = a / float(np.sum(a))
    
    # Precompute FFT of a
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    
    if eta is None:
        eta = 1.0 / np.sqrt(n)
    
    y = 1.0 / (2 * n - 1)  # uniform over rows (2n-1 positions)
    g = np.ones(n) / n  # uniform initial column
    
    g_avg = np.zeros(n)
    
    for t in range(n_iters):
        # Compute row payoffs: (B @ g)[k] = conv(a, g)[k]
        fg = rfft(g, size)
        row_payoff = irfft(fa * fg, size)[:2 * n - 1]
        
        # Update y: y_k *= exp(-eta * row_payoff[k] / max_row_payoff)
        rp_max = row_payoff.max()
        if rp_max < 1e-15:
            break
        y = y * np.exp(-eta * row_payoff / rp_max)
        y_sum = y.sum()
        if y_sum < 1e-300:
            y = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            y = y / y_sum
        
        # Compute column payoffs: (B^T y)[j]
        # (B^T y)[j] = sum_m a[m] * y[j+m]
        # = np.correlate(y, a, 'valid')[j]
        col_payoff = np.correlate(y, a, mode='valid')
        
        # Update g: g_j *= exp(-eta * col_payoff[j] / max_col_payoff)
        cp_max = col_payoff.max()
        if cp_max < 1e-15:
            break
        g = g * np.exp(-eta * col_payoff / cp_max)
        g_sum = g.sum()
        if g_sum < 1e-300:
            g = np.ones(n) / n
        else:
            g = g / g_sum
        
        g_avg += g
    
    g_avg /= n_iters
    return g_avg
```
