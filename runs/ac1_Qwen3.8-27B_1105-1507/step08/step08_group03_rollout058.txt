Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategy: Multi-start LP alternating with anti-palindromic spectral initialization,
warm-started LP with adaptive backtracking line search, Nesterov acceleration,

and systematic local minimum escape via structured spectral perturbations.

import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize as sp_minimize

    import cvxpy as cp

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix C (2n-1 x n) such that C @ g = convolve(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        C[j:j + r, j] = a[:r]
    return C


def solve_lp_minpeak(a, use_cvxpy=True, timeout=10.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = normalize(a)
    C = build_conv_matrix(a)

    if use_cvxpy and HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [C @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.HIGHS, cp.ECOS, cp.SCS], :
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    # Dense LP via scipy
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2 * n - 1),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def lp_alternating(a, time_budget=200.0, verbose=False, seed=42):
    """
    LP alternating with adaptive backtracking line search and Nesterov momentum.
    """
    n = len(a)
    a = normalize(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    best_t = 0.5
    rng = np.random.default_rng(seed)
    it = 0
    momentum_v = np.zeros(n)
    momentum_beta = 0.0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break

        g_lp, M_lp = solve_lp_minpeak(a, use_cvxpy=(remaining > 5.0))
        if g_lp is None:
            conv_a = compute_conv(a)
            M_cur = float(np.max(conv_a))
            # Fallback: use subgradient direction
            k_star = int(np.argmax(conv_a))
            g_lp = np.zeros(n)
            for i in range(n):
                j1 = k_star - i
                j2 = i - k_star
                if 0 <= j1 < n:
                    g_lp[i] += a[j1]
                if 0 <= j2 < n:
                    g_lp[i] += a[j2]
            g_lp = np.maximum(g_lp, 0)
            s = float(np.sum(g_lp))
            if s < 1e-15:
                g_lp = np.ones(n) / n
            else:
                g_lp = g_lp / s

        # Backtracking line search
        ts = [best_t, best_t * 0.7, best_t * 1.3, 1.0, 0.5, 0.3, 0.8, 0.15]
        ts = sorted(set(t for t in ts if 0.01 < t <= 1.0), reverse=True)

        best_t_obj = np.inf
        best_t_val = best_t
        best_a_local = best_a.copy()

        for t in ts:
            a_try = (1.0 - t) * a + t * g_lp
            # Apply momentum
            if momentum_beta > 0.01:
                a_try = a_try + momentum_beta * momentum_v
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-14:
            # Update momentum
            momentum_v = momentum_beta * momentum_v + (best_a_local - a) / max(best_t_val, 1e-6)
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_t = best_t_val
            stall = 0
            momentum_beta = min(0.6, momentum_beta + 0.005)
            if verbose and it % 10 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_val:.3f}, M_lp={M_lp if M_lp is not None else 'N/A':.6f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            momentum_beta = max(0.0, momentum_beta - 0.02)
            if stall > 20:
                # Escape: structured perturbation
                noise_scale = min(0.4, 0.05 + 0.03 * (stall // 20))
                # Spectral perturbation
                fa = rfft(a, 2 * n)
                phase = rng.uniform(0, 2 * np.pi, size=len(fa))
                amp = rng.uniform(0, noise_scale, size=len(fa))
                fa_pert = fa * (1.0 + amp * np.exp(1j * phase))
                a_new = irfft(fa_pert, 2 * n)[:n]
                a_new = np.maximum(a_new, 0)
                s = float(np.sum(a_new))
                if s > 1e-10:
                    a = a_new / s
                else:
                    a = best_a * np.exp(rng.normal(0, noise_scale, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                best_t = 0.5
                momentum_v = np.zeros(n)
                momentum_beta = 0.0

    return best_a


def smooth_optimize(a, time_budget=60.0, T=30.0, verbose=False):
    """Smooth optimization using log-sum-exp with L-BFGS-B."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def obj_grad(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = compute_conv(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        e = np.exp(z)
        sum_e = e.sum()
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)
        w = e / sum_e
        g = np.zeros(n)
        for k in range(len(conv)):
            if w[k] > 1e-8:
                for i in range(n):
                    j1 = k - i
                    j2 = i - k
                    if 0 <= j1 < n:
                        g[i] += w[k] * x[j1]
                    if 0 <= j2 < n:
                        g[i] += w[k] * x[j2]
        g *= 2.0 * n
        return obj_val, g

    max_iter = 300
    try:
        res = sp_minimize(obj_grad, a, method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 3, 'disp': False,
                                   'gtol': 1e-8})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                if verbose:
                    print(f"    Smooth T={T}: obj={best_obj:.8f}")
    except Exception:
        pass

    return best_a


def nesterov_subgradient(a, time_budget=60.0, K=7, verbose=False):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a)
    a = normalize(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 100

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        spread = (peak_vals[0] - peak_vals[-1]) / max_val
        temp = 5.0 / (1.0 + spread * 5.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                k = int(k)
                for i in range(n):
                    j1 = k - i
                    j2 = i - k
                    if 0 <= j1 < n:
                        g[i] += w * y[j1]
                    if 0 <= j2 < n:
                        g[i] += w * y[j2]

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.5)

        a_prev = a_new.copy()
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        y = y_new.copy()
        t_k = t_new
        it += 1

        if verbose and it % 100 == 0:
            print(f"    NS it{it}: obj={best_obj:.8f}, t_k={t_k:.1f}, el={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_refine(a, time_budget=30.0):
    """Coordinate descent: move mass to reduce the dominant peak."""
    n = len(a)
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]

        # Find top contributors to the peak
        contrib = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib.append((a[i] * a[j], i, j))
        contrib.sort(reverse=True)

        # Find positions NOT contributing
        non_contrib = [i for i in range(n) if not (0 <= k_star - i < n)]
        if not non_contrib:
            non_contrib = list(range(n))

        improved = False
        for ci, (val, i, j) in enumerate(contrib[:10]):
            if val < 1e-12 or ci >= 10:
                break
            for target in non_contrib[:5]:
                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[i]
                    if delta < 1e-15:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break

        if not improved:
            # Try moving from peak position to random
            if it < 50:
                rng = np.random.default_rng(it)
                src = k_star % n
                tgt = rng.integers(0, n)
                if src != tgt:
                    delta = 0.02 * a[src]
                    a[src] -= delta
                    a[tgt] += delta
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    if s > 1e-10:
                        a = a / s
                        new_obj = compute_obj(a)
                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Anti-palindromic: first half high, second half low (with sine modulation)
        x = np.arange(n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * x / n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Two-hump
        centers = [n // 4, 3 * n // 4]
        widths = [n // 5, n // 7]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Chirp
        x = np.arange(n) / n
        a = 1.0 + 0.4 * np.cos(4 * np.pi * x ** 2)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Spectrally flat: inverse FFT of flat magnitude with random phases
        size = 1
        while size < 2 * n:
            size <<= 1
        mag = np.ones(size // 2 + 1)
        phase = rng.uniform(0, 2 * np.pi, size // 2 + 1)
        F = mag * np.exp(1j * phase)
        F[0] = mag[0]  # Fix DC
        a = irfft(F, size)[:n]
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Cosine window variant
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Random Fourier
        freqs = rng.integers(1, 8, size=4)
        amplitudes = rng.uniform(0.2, 0.5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, amp, p in zip(freqs, amplitudes, phases):
            a += amp * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Hybrid optimization combining LP alternating,
    Nesterov subgradient, smooth optimization, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # === Phase 0: Load parent ===
    n_target = 800
    parent_loaded = False
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                parent_loaded = True
                print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
    except Exception:
        print("No parent available, using default initialization.")

    if best_seq is None:
        # Try a few initializations and pick the best
        best_init = None
        best_init_obj = np.inf
        for mode in [0, 2, 3, 4, 5, 7]:
            a_try = diverse_init(n_target, rng, mode=mode)
            obj_try = compute_obj(a_try)
            if obj_try < best_init_obj:
                best_init_obj = obj_try
                best_init = a_try
        best_seq = best_init
        best_score = best_init_obj
        print(f"Best init (n={n_target}, mode={mode}): obj={best_score:.8f}")

    # === Phase 1: Quick Nesterov to warm up ===
    print(f"\n=== Phase 1: Quick Nesterov Warm-up ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.05, 50.0)
    if p1_budget > 10:
        a_ns = nesterov_subgradient(best_seq, time_budget=p1_budget, K=7, verbose=False)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov warm-up: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov warm-up: obj={obj_ns:.8f} (no improvement)")

    # === Phase 2: Main LP Alternating ===
    print(f"\n=== Phase 2: LP Alternating (main) ===")
    remaining = deadline - time.monotonic()
    p2_total = min(remaining * 0.40, 550.0)

    n_rounds = 4
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = p2_total / (n_rounds - rnd)
        alloc = min(alloc, remaining * 0.7)
        if alloc < 20:
            break

        # Perturb for later rounds
        if rnd > 0:
            pert_scale = 0.02 * rnd
            a_pert = best_seq * np.exp(rng.normal(0, pert_scale, len(best_seq)))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
        else:
            a_pert = best_seq.copy()

        a_lp = lp_alternating(a_pert, time_budget=alloc, verbose=(rnd == 0), seed=seed + rnd)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Smooth optimization (L-BFGS) ===
    print(f"\n=== Phase 3: Smooth Optimization ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 150.0)
    if p3_budget > 15:
        for T_val in [50.0, 100.0, 200.0]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            t_budget = min(p3_budget / 3.0, remaining * 0.5)
            if t_budget < 10:
                continue
            a_sm = smooth_optimize(best_seq, time_budget=t_budget, T=T_val, verbose=True)
            obj_sm = compute_obj(a_sm)
            if obj_sm < best_score:
                best_score = obj_sm
                best_seq = a_sm.copy()
                print(f"  Smooth T={T_val}: obj={best_score:.8f} **")
            else:
                print(f"  Smooth T={T_val}: obj={obj_sm:.8f} (no improvement)")

    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(200, n_current - 200),
        max(200, n_current - 100),
        n_current + 100,
        n_current + 200,
        n_current + 400,
        300, 400, 500, 600, 700, 800, 900, 1000, 1100, 1200, 1400
    ]))
    n_options = [nv for nv in n_options if 200 <= nv <= 1500]
    n_options.sort(key=lambda x: abs(x - n_current))

    n_tried = 0
    n_improved = set()
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= 8:
            break
        if n_try == n_current:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        # Quick Nesterov
        alloc_ns = min(remaining * 0.06, 30.0)
        if alloc_ns > 10:
            a_try = nesterov_subgradient(a_try, time_budget=alloc_ns, K=5)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.1, 60.0)
        if alloc_lp > 15:
            a_try = lp_alternating(a_try, time_budget=alloc_lp, seed=seed + n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_improved.add(n_try)
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts with LP ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r * 2 + 1) % 8  # modes 1, 3, 5, 7, 1
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.12, 60.0)
        a_opt = lp_alternating(a_r, time_budget=restart_budget, seed=seed + 100 + r)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 30:
        lp_alloc = min(remaining * 0.25, 80.0)
        a_lp = lp_alternating(best_seq, time_budget=lp_alloc, verbose=True, seed=seed + 999)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Nesterov polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        ns_alloc = min(remaining * 0.15, 40.0)
        a_ns = nesterov_subgradient(best_seq, time_budget=ns_alloc, K=9)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Final NS: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        cd_alloc = min(remaining * 0.15, 25.0)
        a_cd = coordinate_refine(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternating(best_seq, time_budget=min(remaining * 0.4, 40.0), seed=seed + 777)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final smooth refinement
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_sm = smooth_optimize(best_seq, time_budget=min(remaining * 0.3, 30.0), T=100.0)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  Final Smooth: obj={best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    if not np.isfinite(val):
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
