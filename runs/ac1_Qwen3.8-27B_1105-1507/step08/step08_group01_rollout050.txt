"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating, smooth-max L-BFGS, peak-targeted subgradient, and multi-n exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp_response(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

def smooth_max_obj_and_grad(a, T):
    """Compute smooth max objective and gradient for normalized a (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    
    # Smooth max value
    sm_val = float(np.dot(weights, conv))
    
    # Gradient: d/d a[i] of sum_k weights[k] * conv(a,a)[k]
    # conv(a,a)[k] = sum_j a[j] * a[k-j]
    # d/da[i] conv(a,a)[k] = a[k-i] + a[i-k] (when valid indices)
    # So grad[i] = sum_k weights[k] * (a[k-i] + a[i-k])
    # This is 2 * (weights * a) convolved appropriately
    # Actually: grad[i] = sum_k weights[k] * (a[k-i] + a[i-k])
    #           = sum_j weights[i+j] * a[j] + sum_j weights[i-j] * a[j]
    #           = conv(weights_centered, a)[i] + conv(weights_flipped, a)[i]
    # Since weights is symmetric (conv is symmetric), this is 2 * sum_j weights[i+j] * a[j]
    # But we need to be careful about bounds.
    
    # Actually let me compute it directly using the matrix approach
    # grad[i] = sum_k weights[k] * d(conv(a,a)[k])/d(a[i])
    # d(conv(a,a)[k])/d(a[i]) = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    # = A.T @ (2 * weights) where A is the convolution matrix... 
    # Actually it's simpler: the Jacobian of conv(a,a) w.r.t. a is 2*A where A is the conv matrix
    # But we need to be more careful.
    
    # Let's use the fact that:
    # conv(a,a)[k] = sum_j a[j]*a[k-j] (with valid index range)
    # d/da[i] conv(a,a)[k] = a[k-i] (if valid) + a[i-k] (if valid)
    # grad[i] = sum_k weights[k] * (a[k-i] + a[i-k])
    
    # Using FFT:
    # sum_k weights[k] * a[k-i] = (weights * a_flipped)[i] where * is correlation
    # Actually, sum_k weights[k]*a[k-i] for k from 0 to 2n-2, k-i from 0 to n-1
    # = sum_j weights[j+i]*a[j] (substituting j=k-i)
    # This is like a cross-correlation.
    
    # Let me just compute it as: grad = 2 * (matrix product)
    # Actually the simplest way:
    # grad[i] = sum_k weights[k] * (a[k-i] + a[i-k])
    #         = 2 * sum_j a[j] * (sum_k weights[k] * indicator(k = i+j or k = j-i))
    #         ... this is getting complicated. Let me just use the matrix.
    
    # Alternative: use the identity that conv(a,a) = ifft(fft(a)^2)
    # d/da conv(a,a) = 2 * A @ a where A is the "convolution by a" matrix
    # So grad = 2 * A.T @ weights where A is the conv matrix for a
    
    A_mat = build_conv_matrix(a)
    grad = 2.0 * (A_mat.T @ weights)
    
    return 2.0 * n * sm_val, grad

def peak_targeted_subgradient(a, time_budget=100.0, verbose=False):
    """
    Subgradient method targeting the current peaks.
    At each step, the subgradient of max(conv(a,a)) is 2*A.T @ e_{k*}
    where k* is argmax. We use Nesterov acceleration.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov accelerated subgradient
    y = a.copy()
    v = np.zeros(n)
    lr_init = 0.01
    lr = lr_init
    
    max_iters = 5000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Evaluate at y, get subgradient
        conv_y = compute_conv(y)
        max_k = np.argmax(conv_y)
        
        # Subgradient: 2 * A.T @ e_{k*}
        # A[k*, j] = a[max(0, k*-j) ... ] ... 
        # Actually A @ g = conv(a, g), so A.T @ e_k = column k of A.T = row k of A transposed
        # A[k, j] = a[k-j] if 0 <= k-j < n, else 0 (for the convolution matrix)
        # So (A.T @ e_k)[i] = A[k, i] = a[k-i] if 0 <= k-i < n
        
        # subgrad[i] = 2 * a[k* - i] (if valid) ... but we need to be more careful
        # conv(a,a)[k] = sum_j a[j]*a[k-j]
        # d/da[i] = a[k-i] + a[i-k] (if valid)
        
        subgrad = np.zeros(n)
        k_star = max_k
        for i in range(n):
            val = 0.0
            if 0 <= k_star - i < n:
                val += a[k_star - i]
            if 0 <= i - k_star < n:
                val += a[i - k_star]
            subgrad[i] = 2.0 * val
        
        # Nesterov update
        t_curr = 2.0 / (it + 2)
        v = (it / (it + 2)) * v
        y_new = simplex_project(y - lr * subgrad)
        a_new = (1 - t_curr) * a + t_curr * y_new  # This isn't quite right for Nesterov
        
        # Actually let me use a simpler approach:
        # Standard Nesterov: 
        # y_{k+1} = x_k + (t_k - 1)/(t_{k+1}) * (x_k - x_{k-1})
        # x_{k+1} = proj(x_{k+1} - lr * g(y_{k+1}))
        
        # Simpler: just use heavy ball momentum
        v = 0.9 * v - lr * subgrad
        a_candidate = simplex_project(a + v)
        
        obj_candidate = compute_obj(a_candidate)
        if obj_candidate < best_obj - 1e-12:
            best_obj = obj_candidate
            best_a = a_candidate.copy()
            a = a_candidate
        else:
            a = best_a.copy()
            v = np.zeros(n)
        
        # Decay learning rate
        lr = lr_init / (1 + 0.01 * lr_init * it)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    Subgrad it{it}: obj={best_obj:.8f}, lr={lr:.6f}, time={elapsed:.1f}s")

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""

Combines LP alternating with convex line search, L-BFGS on smooth max,

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_response(a, active_indices=None):
    """
    Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0.
    If active_indices given, only enforce those constraints.
    Returns (g, M_true) where M_true is the actual max conv(a,g).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return None, None
    a = a / s
    A_full = build_conv_matrix(a)

    if active_indices is not None and len(active_indices) < 2 * n - 1:
        A = A_full[active_indices, :]
    else:
        A = A_full

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((A.shape[0], n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(A.shape[0])

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_norm = g_val / s
            conv_check = A_full @ g_norm
            true_M = float(np.max(conv_check))
            return g_norm, true_M
    return None, None


def convex_line_search(a, g, n_iter=40):
    """
    Golden section search for optimal t in [0,1] minimizing
    obj((1-t)*a + t*g). Exploits that f(t) = max_k quadratic_k(t) is convex.
    Returns (t_opt, a_opt, obj_opt).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at_t(c)
    fd = obj_at_t(d)

    for _ in range(n_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at_t(d)

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm, obj_at_t(0.0)
    a_new = a_new / s
    return t_opt, a_new, obj_at_t(t_opt)


def lp_alternating(a, time_budget=200.0, verbose=False):
    """
    LP alternating with convex line search and adaptive active set.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    active_fraction = 0.5

    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        conv_cur = compute_conv(a)
        max_conv = float(np.max(conv_cur))

        # Adaptive active set: start wide, narrow as we converge
        threshold = max_conv * (1.0 - 0.02 * stall)
        threshold = max(threshold, max_conv * 0.85)
        active = np.where(conv_cur >= threshold)[0]

        if len(active) < 5:
            active = np.argsort(conv_cur)[::-1][:max(10, n // 10)]

        g_lp, M_lp = solve_lp_response(a, active)
        if g_lp is None:
            g_lp, M_lp = solve_lp_response(a, None)
        if g_lp is None:
            break

        # Convex line search for optimal blend
        t_opt, a_new, obj_new = convex_line_search(a, g_lp, n_iter=35)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, "
                      f"active={len(active)}, time={elapsed:.1f}s")
        else:
            # Try full LP (all constraints)
            g_full, _ = solve_lp_response(a, None)
            if g_full is not None:
                t2, a2, obj2 = convex_line_search(a, g_full, n_iter=25)
                if obj2 < best_obj - 1e-13:
                    best_obj = obj2
                    best_a = a2.copy()
                    a = a2
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

            if stall > 10:
                # Perturb to escape local minimum
                rng = np.random.default_rng(it * 13 + n)
                scale = 0.005 + 0.002 * stall
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0

        if stall > 60:
            break

    return best_a


def smooth_max_obj_grad(a, T):
    """
    Compute smooth max of 2n*conv(a,a) and its gradient.
    a should be normalized (sum=1).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        total = 1.0
    weights = exp_vals / total

    sm_val = float(np.dot(weights, conv))

    # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] / a[i-k]
    # Using matrix: A @ a = conv(a,a), so grad = 2 * A.T @ weights (approximate)
    A_mat = build_conv_matrix(a)
    grad = 2.0 * (A_mat.T @ weights)

    return 2.0 * n * sm_val, grad


def smooth_max_lbfgs(a, time_budget=120.0, T_init=0.002, T_min=0.00005, verbose=False):
    """
    L-BFGS optimization using softmax parameterization.
    Multiple rounds with decreasing temperature.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    round_num = 0

    while time.monotonic() - start < time_budget and T >= T_min:
        round_num += 1
        T_current = T

        # Use softmax parameterization: a = softmax(x)
        # This ensures a >= 0 and sum(a) = 1 automatically

        a_safe = np.maximum(a, 1e-15)

        def obj_fn(x):
            xm = x - np.max(x)
            a_s = np.exp(xm)
            ss = np.sum(a_s)
            if ss < 1e-300:
                return 1e10
            a_s = a_s / ss
            val, _ = smooth_max_obj_grad(a_s, T_current)
            return val

        def grad_fn(x):
            xm = x - np.max(x)
            a_s = np.exp(xm)
            ss = np.sum(a_s)
            a_s = a_s / ss
            _, g = smooth_max_obj_grad(a_s, T_current)
            # Softmax Jacobian: grad_x = (g - a^T g) * a
            grad_x = (g - np.dot(a_s, g)) * a_s
            return grad_x

        x_init = np.log(a_safe)

        result = minimize(
            obj_fn, x_init, jac=grad_fn,
            method='L-BFGS-B',
            options={'maxiter': 300, 'maxfun': 800, 'gtol': 1e-6, 'iprint': -1}
        )

        if result.fun < obj_fn(x_init) + 1e-10:
            x_opt = result.x
            xm = x_opt - np.max(x_opt)
            a_new = np.exp(xm)
            sm = np.sum(a_new)
            if sm > 1e-10:
                a_new = a_new / sm
                true_obj = compute_obj(a_new)
                if true_obj < best_obj:
                    best_obj = true_obj
                    best_a = a_new.copy()
                    a = a_new.copy()

        T = max(T_min, T * 0.4)

        if verbose and round_num % 2 == 0:
            elapsed = time.monotonic() - start
            print(f"    L-BFGS round {round_num}: T={T_current:.6f}, "
                  f"best={best_obj:.8f}, time={elapsed:.1f}s")

    return best_a


def peak_chasing(a, time_budget=80.0, verbose=False):
    """
    Iteratively target the top peaks with LP and reduce them.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = float(np.max(conv))

        # Top peaks (top 10)
        sort_idx = np.argsort(conv)[::-1]
        top_k = min(10, len(sort_idx))
        top_indices = sort_idx[:top_k]

        # Solve LP with only top constraints
        g_lp, M_lp = solve_lp_response(a, top_indices)
        if g_lp is None:
            break

        # Try blending
        best_t = 0
        best_obj_try = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_t = t

        if best_t > 0 and best_obj_try < best_obj - 1e-12:
            a_try = (1 - best_t) * a + best_t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            best_a = (a_try / st).copy()
            best_obj = best_obj_try
            a = best_a
            if verbose and it % 20 == 0:
                print(f"    PeakChase it{it}: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            # Try with wider active set
            wider = np.argsort(conv)[::-1][:min(50, len(sort_idx))]
            g2, _ = solve_lp_response(a, wider)
            if g2 is not None:
                t2, a2, obj2 = convex_line_search(a, g2, n_iter=20)
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a2.copy()
                    a = a2
            else:
                if it > 50:
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s


def coordinate_descent_polish(a, time_budget=60.0, delta=1e-4):
    """
    Randomized coordinate descent with adaptive step sizes.
    Updates pairs of coordinates to maintain the sum constraint more efficiently.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        improved = False
        rng = np.random.default_rng(int((time.monotonic() - start) * 1000) % (2**31))
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            # Try single coordinate updates
            for d in [delta, -delta, delta * 0.1, -delta * 0.1,
                      delta * 5, -delta * 5, delta * 20]:
                a_try = a.copy()
                a_try[i] = max(0.0, a_try[i] + d)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                continue

            # Try pair updates (transfer mass between two coordinates)
            if n > 1:
                j = int(rng.integers(0, n))
                if j == i:
                    continue
                for amount in [delta, delta * 0.1, -delta, -delta * 0.1]:
                    a_try = a.copy()
                    a_try[i] = max(0.0, a_try[i] + amount)
                    a_try[j] = max(0.0, a_try[j] - amount)
                    st = float(np.sum(a_try))
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if not improved:
            delta *= 0.6
            if delta < 1e-8:
                break

    return best_a

            best_score = obj
            best_seq = a_lb.copy()
            print(f"  Final L-BFGS: obj={best_score:.8f}")
        remaining = deadline - time.monotonic()

    # 7c: CD
    if remaining > 15:
        alloc = min(40.0, remaining * 0.3)
        a_cd = coordinate_descent_polish(best_seq, time_budget=alloc, delta=5e-5)
        obj = compute_obj(a_cd)
        if obj < best_score:
            best_score = obj
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
