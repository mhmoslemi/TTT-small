```python
def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    # Bisection on threshold
    # Find tau such that sum(max(v_i - tau, 0)) = s
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    # Find rho such that cumsum[rho] - rho * v_sorted[rho] = s
    rho = np.where(cumsum - np.arange(1, n+1) * v_sorted > s)[0]
    if len(rho) == 0:
        return np.zeros(n)  # shouldn't happen if sum(v) > 0
    rho = rho[0]
    tau = (cumsum[rho] - s) / (rho + 1)
    return np.maximum(v - tau, 0)
```

This is the standard O(n log n) simplex projection.

"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
1. Logsumexp smoothed optimization with Nesterov acceleration
2. LP alternating with 2-cycle breaking
3. Exact subgradient descent
4. Multi-n exploration
5. Pairwise exchange polish

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def softmax(x, t):
    """Compute exp(x/t) / sum(exp(x/t)) with numerical stability."""
    x_shifted = (x - np.max(x)) / t
    e = np.exp(x_shifted)
    return e / np.sum(e)

def logsumexp_obj_and_grad(a, t):
    """Compute logsumexp(conv(a,a)/t) and its gradient."""
    n = len(a)
    conv = compute_conv(a)
    # p_k = exp(conv_k / t) / sum_j exp(conv_j / t)
    p = softmax(conv, t)
    # grad[i] = 2 * sum_j a[j] * p[i+j]
    # = 2 * correlate(p, a, 'valid')[i]
    grad = 2.0 * np.correlate(p, a, mode='valid')
    obj = t * np.log(np.sum(np.exp((conv - np.max(conv)) / t)) + (2*n-1) * np.exp(np.max(conv)/t - np.max(conv)/t))
    # Simpler: obj = t * logsumexp(conv/t)
    max_conv = np.max(conv)
    obj = max_conv + t * np.log(np.sum(np.exp((conv - max_conv) / t)))
    return obj, grad

def proj_simplex(v, s=1.0):
    """Project v onto probability simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    idx = np.arange(1, n+1)
    # Find rho: largest rho such that cumsum[rho-1] - rho*v_sorted[rho-1] > s
    # i.e., cumsum[rho-1] - (rho-1+1)*v_sorted[rho-1] > s
    # i.e., cumsum[rho-1] - rho*v_sorted[rho-1] > s
    cond = cumsum - idx * v_sorted > s
    if not np.any(cond):
        # All zero - shouldn't happen
        return np.full(n, s/n)
    rho = np.max(np.where(cond)[0]) + 1
    tau = (cumsum[rho-1] - s) / rho
    return np.maximum(v - tau, 0)

"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
1. LP alternating with 2-cycle breaking and dual-guided perturbation
2. Logsumexp smoothed optimization with Nesterov momentum
3. Exact subgradient descent on active set
4. Multi-n exploration via resampling
5. Pairwise exchange coordinate polish
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. A@z <= M, sum(z)=1, z>=0.
    Returns (z_normalized, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except Exception:
                        pass
                    return z_val, float(M.value), dual_y
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except Exception:
        pass

    return None, None, None


def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_cond = u * idx + s - sv > 0
    if not np.any(rho_cond):
        return np.full(n, s / n)
    rho = int(np.max(np.where(rho_cond)[0])) + 1
    theta = (sv[rho - 1] - s) / rho
    return np.maximum(v - theta, 0)


def logsumexp_obj_grad(a, t):
    """Compute logsumexp(conv(a,a)/t) and gradient w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / t
    e = np.exp(shifted)
    p = e / float(np.sum(e))
    obj = max_c + t * float(np.log(np.sum(e)))
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return obj, grad


def logsumexp_optimize(a, time_budget=300.0, verbose=False):
    """
    Optimize using logsumexp smoothing with decreasing temperature
    and Nesterov-accelerated projected gradient.
    """
    n = len(a).astype(np.float64)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stop = start + time_budget

    # Temperature schedule: geometric decrease
    t_start = 0.05
    t_end = 0.001
    n_phases = 5
    temps = np.geomspace(t_start, t_end, n_phases)

    for phase_idx, t in enumerate(temps):
        if time.monotonic() > stop:
            break

        # Nesterov parameters
        momentum = 0.9
        lr = 0.1 / (n * t)
        v = np.zeros(n)
        x = a.copy()
        y = a.copy()
        best_phase_obj = compute_obj(a)
        best_phase_a = a.copy()
        stall = 0
        it = 0

        phase_stop = min(stop, time.monotonic() + time_budget / n_phases)

        while time.monotonic() < phase_stop:
            if time.monotonic() - start > time_budget - 5:
                break

            _, grad = logsumexp_obj_grad(y, t)

            # Nesterov momentum update
            v = momentum * v - lr * grad
            y_new = proj_simplex(y + v)

            # Check objective on original problem
            obj_new = compute_obj(y_new)
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = y_new.copy()
                stall = 0
            elif obj_new < best_phase_obj - 1e-10:
                best_phase_obj = obj_new
                best_phase_a = y_new.copy()
                stall = 0
            else:
                stall += 1

            x = y
            y = y_new

            # Adaptive learning rate
            if stall > 20:
                lr *= 0.5
                if lr < 1e-6:
                    break
                stall = 0

            it += 1
            if verbose and it % 200 == 0:
                print(f"    LS p{phase_idx} t={t:.4f} it={it} obj={best_obj:.8f}")

        a = best_phase_a.copy()

    return best_a


def subgradient_refine(a, time_budget=200.0, verbose=False):
    """
    Projected subgradient descent on the exact max objective.
    Averages subgradient over active (near-max) set.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = 0.7
    v = np.zeros(n)
    lr = 0.05
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 2:
            break

        conv = compute_conv(a)
        max_c = float(np.max(conv))
        active = conv > max_c * 0.95
        n_active = int(np.sum(active))

        # Compute subgradient: average of 2*shift_rev(a, k) over active k
        # shift_rev(a, k)[i] = a[k-i] if 0 <= k-i < n
        # grad[i] = 2 * sum_{k active} a[k-i] / n_active = 2 * correlate(a_ind, a, valid)[i] / n_active
        a_ind = np.zeros(2 * n - 1)
        a_ind[active] = 1.0 / n_active
        grad = 2.0 * np.correlate(a_ind, a, mode='valid')

        # Momentum update
        v = momentum * v - lr * grad
        a_new = proj_simplex(a + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-11:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                lr *= 0.7
                if lr < 1e-7:
                    break
                v *= 0.5
                a = best_a.copy()

        it += 1
        if verbose and it % 100 == 0:
            print(f"    SG it={it} obj={best_obj:.8f} lr={lr:.6f}")

    return best_a


def lp_robust_alternating(a, time_budget=200.0, verbose=False):

    best_obj = compute_obj(a)
    start = time.monotonic()

    prev_a = None
    prev_prev_a = None

    stall = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 5:
            break

        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp, dual_y = result

        # 2-cycle detection
        if prev_prev_a is not None:
            diff = float(np.max(np.abs(a - prev_prev_a)))
            if diff < 1e-7:
                a_break = 0.5 * (a + prev_a)
                a_break = np.maximum(a_break, 0)
                sb = float(np.sum(a_break))
                if sb > 1e-10:
                    a_break = a_break / sb
                obj_break = compute_obj(a_break)
                if obj_break < best_obj - 1e-12:
                    best_obj = obj_break
                    best_a = a_break.copy()
                a = a_break
                result = solve_lp(a)

                prev_a = None
                prev_prev_a = None
                g_momentum = a.copy()

        # Update momentum

                a_try = a_try / st
                obj_try = compute_obj(a_try)

            prev_a = a.copy()
            prev_prev_a = prev_a
            a = best_a
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start

        else:
            stall += 1
            if stall > 10:

                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                if dual_y is not None and np.max(dual_y) > 0:
                    A = build_conv_matrix(a)
                    active_idx = np.where(dual_y > 0.01 * np.max(dual_y))[0]
                    if len(active_idx) > 0:
                        col_imp = A[active_idx, :].sum(axis=0)
                        pert = -np.sign(col_imp - np.mean(col_imp)) * 0.01
                        pert += rng.normal(0, 0.015, n)
                    else:
                        pert = rng.normal(0, 0.02, n)
                else:
                    pert = rng.normal(0, 0.02, n)

                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                sn = float(np.sum(a))
                if sn > 1e-10:
                    a = a / sn
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev_prev_a = None

                stall = 0

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def pairwise_exchange(a, time_budget=60.0):
    """Pairwise exchange polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99999)

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 2:
            break
        improved = False
        for _ in range(30):
            if time.monotonic() - start > time_budget - 1:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            for frac in [0.03, 0.07, 0.15, 0.01]:
                delta = frac * min(a[i], a[j])
                if delta > 0.5 * min(a[i], a[j]):
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < -1e-15 or a_try[j] < -1e-15:
                    continue
                a_try = np.maximum(a_try, 0)
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        if not improved:
            # Try larger perturbations
            for _ in range(5):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                a_try = a.copy()
                a_try[i] *= rng.uniform(0.5, 1.5)
                a_try[j] *= rng.uniform(0.5, 1.5)
                a_try = np.maximum(a_try, 0)
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
        if not improved:
            break

    return best_a


def coordinate_descent(a, time_budget=40.0):
    """Cyclic coordinate descent: adjust one element at a time."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    order = np.arange(n)
    np.random.default_rng(42).shuffle(order)

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 2:
            break
        np.random.default_rng(int(time.monotonic() * 1e4) % 100000).shuffle(order)
        improved = False
        for i in order:
            if time.monotonic() - start > time_budget - 1:
                break
            for scale in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]:
                a_try = a.copy()
                a_try[i] *= scale
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple optimization strategies:
    1. LP robust alternating (primary)
    2. Logsumexp smoothed optimization (smooth complement)
    3. Subgradient refinement
    4. Multi-n exploration
    5. Coordinate and pairwise polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        for n_init in [400, 600, 800, 1000, 1200]:
            t = np.linspace(0, 1, n_init)
            windows = [
                0.5 + 0.5 * np.cos(2 * np.pi * t),
                np.sin(np.pi * t) ** 2,
                1.0 - 0.2 * np.cos(4 * np.pi * t),
                np.exp(-2.0 * (t - 0.5) ** 2),
            ]
            for w in windows:
                a_init = np.maximum(w, 0.01)
                a_init = a_init / np.sum(a_init)
                obj_init = compute_obj(a_init)
                if obj_init < best_score:
                    best_score = obj_init
                    best_seq = a_init.copy()

        if best_seq is None:
            best_seq = np.ones(800) / 800.0
            best_score = compute_obj(best_seq)
        print(f"Initial (n={len(best_seq)}): obj={best_score:.8f}")

    a = best_seq.copy()
    n_parent = len(a)

    # === Phase 1: LP Robust Alternating (40% of budget) ===
    print(f"\n=== Phase 1: LP Robust (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(500.0, remaining * 0.40)
    a_new = lp_robust_alternating(a, time_budget=alloc, verbose=True)
    obj = compute_obj(a_new)
    if obj < best_score:
        best_score = obj
        best_seq = a_new.copy()
        a = a_new.copy()
        print(f"  Phase 1 done: obj={best_score:.8f} **")
    else:
        print(f"  Phase 1 done: obj={obj:.8f} (no improvement)")

    # === Phase 2: Logsumexp Smoothed Optimization (25% of budget) ===
    print(f"\n=== Phase 2: Logsumexp (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(350.0, remaining * 0.25)
        a_new = logsumexp_optimize(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2 done: obj={obj:.8f}")

    # === Phase 3: Subgradient Refinement (15% of budget) ===
    print(f"\n=== Phase 3: Subgradient (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(200.0, remaining * 0.15)
        a_new = subgradient_refine(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Phase 3 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 done: obj={obj:.8f}")

    # === Phase 4: Multi-n Exploration (15% of budget) ===
    print(f"\n=== Phase 4: Multi-n ===")
    remaining = deadline - time.monotonic()
    n_candidates = sorted(set([
        n_parent - 50, n_parent - 100, n_parent + 50, n_parent + 100,
        n_parent + 200, n_parent + 400,
        500, 600, 700, 900, 1000, 1200, 1400,
    ]))
    n_candidates = [x for x in n_candidates if 300 <= x <= 2500]
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(40.0, remaining * 0.12)
        if alloc < 10:
            break
        a_opt = lp_robust_alternating(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Second LP round on best (10% of budget) ===
    print(f"\n=== Phase 5: LP Round 2 (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(200.0, remaining * 0.15)
        a_new = lp_robust_alternating(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 5: obj={best_score:.8f} **")
        else:
            print(f"  Phase 5: obj={obj:.8f}")

    # === Phase 6: Coordinate Descent Polish ===
    print(f"\n=== Phase 6: Coord Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(50.0, remaining * 0.10)
        a_new = coordinate_descent(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Coord: obj={best_score:.8f} **")
        else:
            print(f"  Coord: obj={obj:.8f}")

    # === Phase 7: Pairwise Exchange Polish ===
    print(f"\n=== Phase 7: Pairwise Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(40.0, remaining * 0.10)
        a_new = pairwise_exchange(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Pairwise: obj={best_score:.8f} **")
        else:
            print(f"  Pairwise: obj={obj:.8f}")

    # === Phase 8: Random Restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 8: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(5, int(remaining // 50))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if r % 3 == 0:
                a_r = best_seq * np.exp(rng.normal(0, 0.03 + 0.01 * r, n_best))
            elif r % 3 == 1:
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 100)
                block_size = n_best // n_blocks
                blocks = rng.choice(n_blocks, min(3, n_blocks), replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = si + block_size
                    a_r[si:ei] *= rng.uniform(0.6, 1.4)
            else:
                a_r = best_seq.copy()
                n_sparse = max(5, n_best // 20)
                idx = rng.choice(n_best, n_sparse, replace=False)
                a_r[idx] *= rng.uniform(0.4, 1.8, n_sparse)

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            alloc = min(remaining * 0.3, 30.0)
            if alloc < 10:
                break
            a_opt = lp_robust_alternating(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 9: Final LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 25.0)
        a_final = lp_robust_alternating(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
