        pass

    return None, None


def blend_search(a, g, n_points=25):
    """Find best blend t in [0,1] minimizing obj((1-t)*a + t*g).
    The objective as function of t is convex (max of quadratics),
    so a grid + refinement works well."""
    n = len(a)
    best_t = 0.0
    best_obj = compute_obj(a)
    
    # Coarse grid
    ts = np.linspace(0, 1, n_points)
    objs = []
    for t in ts:
        c = (1 - t) * a + t * g
        c = np.maximum(c, 0)
        s = float(np.sum(c))
        if s < 1e-10:
            objs.append(np.inf)
            continue
        c = c / s
        objs.append(compute_obj(c))
    objs = np.array(objs)
    
    # Find minimum on grid
    idx = int(np.argmin(objs))
    best_t = ts[idx]
    best_obj = objs[idx]
    
    # Refine around the minimum
    if 0 < idx < n_points - 1:
        lo = ts[max(0, idx - 1)]
        hi = ts[min(n_points - 1, idx + 1)]
        for _ in range(10):
            t1 = lo + 0.382 * (hi - lo)
            t2 = lo + 0.618 * (hi - lo)
            for t in [t1, t2]:
                c = (1 - t) * a + t * g
                c = np.maximum(c, 0)
                s = float(np.sum(c))
                if s < 1e-10:
                    continue
                c = c / s
                obj_t = compute_obj(c)
                if obj_t < best_obj:
                    best_obj = obj_t
                    best_t = t
            # Narrow interval
            c1 = (1 - t1) * a + t1 * g
            c1 = np.maximum(c1, 0)
            s1 = float(np.sum(c1))
            o1 = compute_obj(c1 / s1) if s1 > 1e-10 else np.inf
            c2 = (1 - t2) * a + t2 * g
            c2 = np.maximum(c2, 0)
            s2 = float(np.sum(c2))
            o2 = compute_obj(c2 / s2) if s2 > 1e-10 else np.inf
            if o1 < o2:
                hi = t2
            else:
                lo = t1
    
    c_best = (1 - best_t) * a + best_t * g
    c_best = np.maximum(c_best, 0)
    s_best = float(np.sum(c_best))
    if s_best > 1e-10:
        c_best = c_best / s_best
    return c_best, best_obj


def try_symmetrize(a):
    """Return (best_sequence, best_obj) comparing a vs symmetrized a."""
    sym_a = (a + a[::-1]) / 2.0
    sym_a = np.maximum(sym_a, 0)
    s = float(np.sum(sym_a))
    if s > 1e-10:
        sym_a = sym_a / s
    obj_orig = compute_obj(a)
    obj_sym = compute_obj(sym_a)
    if obj_sym < obj_orig:
        return sym_a, obj_sym
    return a, obj_orig


def lp_alternate(a, time_budget, verbose=False):
    """LP alternating with blend search and stall detection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if elapsed > time_budget * 0.95:
            break

        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        a_new, obj_new = blend_search(a, g_lp)
        
        # Try symmetrization
        a_new, obj_new = try_symmetrize(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.10f}, M={M_lp:.6f}, t={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall >= 4:
                # Structured perturbation based on current peak location
                conv = compute_conv(best_a / np.sum(best_a))
                peak_idx = int(np.argmax(conv))
                rng = np.random.default_rng(stall * 1000 + peak_idx)
                scale = 0.08 + 0.03 * min(stall, 8)
                
                # Perturb more around the peak-related indices
                perturb = rng.normal(0, 1, n)
                # Amplify perturbation near peak index
                for i in range(n):
                    dist = abs(i - peak_idx)
                    if dist < n // 4:
                        perturb[i] *= 2.0
                a = best_a * np.exp(scale * perturb)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def prolate_init(n, rng):
    """Initialize using a prolate-spheroidal-like sequence.
    The ideal sequence minimizes peak sidelobe; Slepian sequences
    concentrate energy in both time and frequency."""
    x = np.linspace(0, 1, n)
    # Use a raised cosine (Hann window) as base - good sidelobe properties
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    # Add small random perturbation to break symmetry
    a = a * np.exp(rng.normal(0, 0.05, n))
    a = np.maximum(a, 1e-6)
    a = a / np.sum(a)
    return a


def diverse_init(n, rng):
    """Generate diverse structured initialization."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-80 * (x - 0.5) ** 2)
    elif choice == 1:
        a = 0.3 + 0.7 * np.cos(2 * np.pi * 2 * x) ** 2
    elif choice == 2:
        a = np.exp(-60 * (x - 0.35) ** 2) + np.exp(-60 * (x - 0.65) ** 2)
    elif choice == 3:
        # Tabulate-like: periodic bumps
        a = np.ones(n)
        period = max(3, n // 8)
        for i in range(n):
            a[i] = 1.0 + 0.5 * np.cos(2 * np.pi * i / period)
        a = np.maximum(a, 0.1)
    elif choice == 4:
        # Power law
        x_safe = np.clip(x, 0.01, 0.99)
        a = x_safe ** rng.uniform(0.5, 2.0) * (1 - x_safe) ** rng.uniform(0.5, 2.0)
    else:
        a = prolate_init(n, rng)
    a = np.maximum(a, 1e-8)
    a = a / np.sum(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP alternating with blend search,
    multi-n exploration, diverse restarts, and structured perturbations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0  # 20s safety buffer

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        n_base = len(a)
        print(f"Parent loaded: n={n_base}, obj={compute_obj(a):.10f}")
    except Exception:
        a = prolate_init(800, rng)
        n_base = 800
        print(f"No parent. Prolate init: n={n_base}, obj={compute_obj(a):.10f}")

    best_a = a.copy()
    best_obj = compute_obj(a)

    # === Phase 1: Main LP alternation at parent n ===
    print(f"\n=== Phase 1: LP Alternation (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 300.0)
    if p1_budget > 20:
        a_p1 = lp_alternate(best_a, time_budget=p1_budget, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_obj:
            best_obj = obj_p1
            best_a = a_p1
            print(f"  Phase 1 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 1: no improvement (obj={obj_p1:.10f})")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    
    # Generate n options: around the best n, plus some spread
    n_options = set()
    for delta in [-300, -200, -100, 0, 100, 200, 300, 500, 700, 1000]:
        n_try = n_best + delta
        if 300 <= n_try <= 2500 and n_try % 2 == 0:
            n_options.add(n_try)
    # Add some absolute values
    for n_abs in [400, 600, 1000, 1200, 1600, 2000]:
        if 300 <= n_abs <= 2500:
            n_options.add(n_abs)
    n_options = sorted(n_options)
    
    # Sort by proximity to current best n (most likely to help first)
    n_options.sort(key=lambda x: abs(x - n_best))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == n_best:
            continue  # Already optimized at this n

        a_try = resample_seq(best_a, n_try)
        a_try = a_try / np.sum(a_try)
        obj_try = compute_obj(a_try)
        
        # Quick LP optimization
        alloc = min(40.0, remaining * 0.15)
        a_opt = lp_alternate(a_try, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.10f} (no improvement)")

    # === Phase 3: D diverse restarts at best n ===
    print(f"\n=== Phase 3: Diverse Restarts (n={n_best}) ===")
    remaining = deadline - time.monotonic()
    
    for d in range(12):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_div = diverse_init(n_best, rng)
        alloc = min(30.0, remaining * 0.12)
        a_opt = lp_alternate(a_div, time_budget=alloc)
        obj_div = compute_obj(a_opt)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.10f} **")
        else:
            print(f"  Diverse {d}: obj={obj_div:.10f}")

    # === Phase 4: Second LP round at best n ===
    print(f"\n=== Phase 4: LP Round 2 (n={len(best_a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = min(remaining * 0.35, 200.0)
        a_p4 = lp_alternate(best_a, time_budget=p4_budget, verbose=True)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_obj:
            best_obj = obj_p4
            best_a = a_p4
            print(f"  Phase 4 result: obj={best_obj:.10f} **")
        else:
            print(f"  Phase 4: no improvement (obj={obj_p4:.10f})")

    # === Phase 5: Final n sweep around best ===
    print(f"\n=== Phase 5: Fine n Sweep ===")
    remaining = deadline - time.monotonic()
    n_cur = len(best_a)
    for delta in [-50, -25, 25, 50, 75]:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        n_try = n_cur + delta
        if n_try < 200 or n_try > 3000:
            continue
        a_try = resample_seq(best_a, n_try)
        a_try = a_try / np.sum(a_try)
        alloc = min(20.0, remaining * 0.2)
        a_opt = lp_alternate(a_try, time_budget=alloc)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            n_cur = n_try
            print(f"  ** n={n_try}: obj={best_obj:.10f} **")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p6_budget = min(remaining * 0.5, 60.0)
        a_final = lp_alternate(best_a, time_budget=p6_budget)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final polish: obj={best_obj:.10f} **")

    # Try symmetrization one last time
    best_a, obj_sym = try_symmetrize(best_a)
    if obj_sym < best_obj:
        best_obj = obj_sym
        print(f"  Final symmetrize: obj={best_obj:.10f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_obj:
            best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.10f}, time={elapsed:.1f}s ===")

    return result
