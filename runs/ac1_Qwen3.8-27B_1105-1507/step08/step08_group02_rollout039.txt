```
def fast_lp_alternate(a, n_inner=100, lr=0.1):
    n = len(a)
    g = a.copy()
    for _ in range(n_inner):
        ag = convolve(a, g)  # FFT-based
        k_star = argmax(ag)
        sg = zeros(n)
        for j in range(n):
            k_minus_j = k_star - j
            if 0 <= k_minus_j < n:
                sg[j] = a[k_minus_j]
        g = g - lr * sg
        g = project_simplex(g)
    return g
```

```python
Optimized step function search using fast subgradient mirror descent
with periodic LP refinement, top-K peak averaging, multi-start,
and dimension exploration.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via real FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def subgradient_conv(a, k):
    """Subgradient of (a*a)_k w.r.t. a: g_j = 2*a_{k-j} for valid j."""
    n = len(a)
    g = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start > j_end:
        return g
    idx = k - np.arange(j_start, j_end + 1)
    g[j_start:j_end + 1] = 2.0 * a[idx]
    return g

def soft_subgradient(a, conv, topK=5, beta=100.0):
    """Weighted subgradient using top-K peaks with softmax weights."""
    n = len(a)
    # Get top-K indices
    top_idx = np.argsort(conv)[::-1][:topK]
    top_vals = conv[top_idx]
    # Softmax weights
    shifted = beta * (top_vals - top_vals[0])
    weights = np.exp(shifted)
    weights /= weights.sum()
    # Weighted sum of subgradients
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        g += w * subgradient_conv(a, int(k))
    return g

def mirror_step(a, g, lr):
    """Mirror descent step on simplex."""
    # a_new = a * exp(-lr * g) / norm
    scaled = lr * g
    scaled -= np.max(scaled)  # for numerical stability
    a_new = a * np.exp(-scaled)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new /= s
    return a_new

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. (a_norm*g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:

                s = float(g_val.sum())
                if s > 1e-10:

        except Exception:

    return None, None
```

"""
Optimized step function search using fast subgradient mirror descent with
top-K peak averaging, periodic LP refinement, multi-start, and dimension
exploration. Combines fast first-order methods for exploration with exact
LP solves for exploitation.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via real FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_norm(a):
    """Objective assuming a is already normalized to sum=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def subgradient_conv(a, k):
    """Subgradient of (a*a)_k w.r.t. a: g_j = 2*a_{k-j} for valid j."""
    n = len(a)
    g = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start > j_end:
        return g
    idx = k - np.arange(j_start, j_end + 1)
    g[j_start:j_end + 1] = 2.0 * a[idx]
    return g


def soft_subgradient(a, conv, topK=5, beta=100.0):
    """Weighted subgradient using top-K peaks with softmax weights."""
    n = len(a)
    top_idx = np.argsort(conv)[::-1][:topK]
    top_vals = conv[top_idx]
    shifted = beta * (top_vals - top_vals[0])
    weights = np.exp(shifted)
    weights /= weights.sum()
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        g += w * subgradient_conv(a, int(k))
    return g


def mirror_step(a, g, lr):
    """Mirror descent step on simplex: a_new ∝ a * exp(-lr * g)."""
    scaled = lr * g
    max_s = np.max(scaled)
    if max_s > 500:
        scaled = scaled - max_s
    a_new = a * np.exp(-scaled)
    s = float(a_new.sum())
    if s < 1e-30:
        return a.copy()
    a_new /= s
    return a_new


def project_simplex(v):
    """Project v onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. (a_norm*g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None)."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a_norm[:r]

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(g_val.sum())
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(g_val.sum())
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""

    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    rounds = 0
    while time.monotonic() - start < time_budget and rounds < 10:
        rounds += 1
        improved = False
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        contrib_set = set()
        for _, i, j in contributors:
            contrib_set.add(i)
            if i != j:
                contrib_set.add(j)
        non_contrib = [x for x in range(n) if x not in contrib_set]
        if not non_contrib:
            break

        for contrib_val, i, j in contributors[:20]:
            if time.monotonic() - start > time_budget:
                break
            if contrib_val < 1e-10:
                break
            src = i if a[i] >= a[j] else (j if i != j else i)
            for target in non_contrib[:5]:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = frac * a[src]
                    if delta <= 0 or delta >= a[src] * 0.9:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break
        if not improved:
            break

    return best_a
