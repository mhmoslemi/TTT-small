import numpy as np

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_lse(a, beta=50.0):
    """Smoothed objective using log-sum-exp."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    # log-sum-exp for smooth max
    max_c = np.max(conv)
    lse = max_c + np.log(np.sum(np.exp((conv - max_c) * beta))) / beta
    return 2.0 * n * lse


def grad_max_conv(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        m_start = k - i_end
        m_end = k - i_start
        grad[m_start:m_end + 1] = 2.0 * a[i_start:i_end + 1][::-1]
    return grad


def grad_lse_conv(a, beta=50.0):
    """Smoothed gradient using log-sum-exp weights."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    weights = np.exp((conv - max_c) * beta)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    # Only process top peaks for efficiency
    top_k = np.argsort(weights)[::-1][:min(20, n)]
    for k in top_k:
        grad += weights[int(k)] * grad_max_conv(a, int(k))
    return grad


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:min(j + n, m), j] = a[:min(n, m - j)]
    return A


def solve_lp_minpeak(a):
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None

            dg = grad - prev_grad
            num = np.dot(da, dg)

def spectral_init(n, rng):
    """Initialize using spectral considerations - design |A(f)|^2 to be flat."""
    # We want |DFT(a)|^2 to be as flat as possible (makes conv flat)
    # This means DFT(a) should have constant magnitude
    # A sequence with constant-magnitude DFT is related to Barker sequences
    # For non-negative sequences, we can't achieve perfectly flat DFT,
    # but we can try to make it as flat as possible.
    
    #approach: start with a random phase in frequency, take IFFT, take absolute value
    m = 2 * n - 1
    size = 1
    while size < m:
        size <<= 1
    
    # Random phases
    phases = rng.random(size // 2 + 1) * 2 * np.pi
    spec = np.exp(1j * phases)
    # Make it Hermitian
    if size > 1:
        spec_full = np.zeros(size, dtype=complex)
        spec_full[:size//2+1] = spec
        spec_full[size//2+1:] = np.conj(spec[:size//2-1][::-1])
    else:
        spec_full = spec
    
    sig = irfft(spec_full, size)[:n]
    a = np.abs(sig)
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw

"""

import numpy as np

def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def grad_max_conv_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a[m] = 2*a[k-m] for valid m."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        m_start = k - i_end
        m_end = k - i_start
        grad[m_start:m_end + 1] = 2.0 * a[i_start:i_end + 1][::-1]
    return grad


def grad_lse_conv(a, beta=30.0, top_k=15):
    """Smoothed gradient using log-sum-exp weights on convolution peaks."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    # Use only top peaks for efficiency
    w = np.zeros_like(conv)
    inds = np.argsort(conv)[::-1][:top_k]
    w[inds] = np.exp((conv[inds] - max_c) * beta)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        # Uniform weights over top peaks
        w[inds] = 1.0 / len(inds)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in inds:
        grad += w[int(k)] * grad_max_conv_at_k(a, int(k))
    return grad


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None

def spectral_init(n, rng):
    """Initialize using spectral shaping - try to flatten |DFT(a)|^2."""
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    # Random phases with low-frequency emphasis
    freqs = np.arange(size // 2 + 1)
    low_freq_weight = np.exp(-freqs / (size * 0.3))
    phases = rng.random(size // 2 + 1) * 2 * np.pi
    magnitudes = low_freq_weight + 0.1
    spec_half = magnitudes * np.exp(1j * phases)
    spec_full = np.zeros(size, dtype=complex)
    spec_full[:size // 2 + 1] = spec_half
    spec_full[size // 2 + 1:] = np.conj(spec_half[:size // 2 - 1][::-1])
    sig = irfft(spec_full, size)[:n]
    a = np.abs(sig)
    a = np.maximum(a, 1e-10)
    s = np.sum(a)
    return a / s


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw

    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(90.0, remaining * 0.25)
        a_bb = bb_gradient_descent(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_bb)
        if obj < best_score - 1e-13:
            best_score = obj
            best_seq = a_bb.copy()
            a = a_bb
            print(f"  BB R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  BB R{round_idx}: obj={obj:.8f}")
            a = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
            a = np.maximum(a, 0)
            a /= np.sum(a)
    
    # === Phase 4: Mirror descent ===
    print(f"\n=== Phase 4: Mirror Descent ===")
    for round_idx in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(50.0, remaining * 0.2)
        a_md = mirror_descent(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_md)
        if obj < best_score - 1e-13:
            best_score = obj
            best_seq = a_md.copy()
            print(f"  MD R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  MD R{round_idx}: obj={obj:.8f}")
            best_seq = best_seq * np.exp(rng.normal(0, 0.015, len(best_seq)))
            best_seq = np.maximum(best_seq, 0)
            best_seq /= np.sum(best_seq)
    
    # === Phase 5: LP Alternating ===
    print(f"\n=== Phase 5: LP Alternating ===")
    a = best_seq.copy()
    for round_idx in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        alloc = min(60.0, remaining * 0.25)
        a_lp = lp_alternate_v2(a, time_budget=alloc, verbose=True)
        obj = compute_obj(a_lp)
        if obj < best_score - 1e-13:
            best_score = obj
            best_seq = a_lp.copy()
            a = a_lp
            print(f"  LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj:.8f}")
            a = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
            a = np.maximum(a, 0)
            a /= np.sum(a)
    
    # === Phase 6: Combined BB + LP polish ===
    print(f"\n=== Phase 6: Combined Polish ===")
    for round_idx in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        alloc1 = min(35.0, remaining * 0.3)
        alloc2 = min(35.0, remaining * 0.3)
        a_bb = bb_gradient_descent(best_seq, time_budget=alloc1)
        a_md = mirror_descent(a_bb, time_budget=min(20.0, remaining * 0.15))
        a_lp = lp_alternate_v2(a_md, time_budget=alloc2)
        obj = compute_obj(a_lp)
        if obj < best_score - 1e-13:
            best_score = obj
            best_seq = a_lp.copy()
            print(f"  Combined R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  Combined R{round_idx}: obj={obj:.8f}")
    
    # === Phase 7: Random restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 7: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(40):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            scale = 0.008 + 0.004 * (r % 10)
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            # Also try structured perturbations
            if r % 5 == 0:
                # Smooth perturbation
                noise = np.cumsum(rng.normal(0, scale * 0.5, n_best))
                a_r = best_seq * np.exp(noise - np.mean(noise))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr
            alloc_total = min(remaining * 0.2, 12.0)
            if alloc_total < 5:
                break
            a_opt = bb_gradient_descent(a_r, time_budget=alloc_total * 0.5)
            a_opt = lp_alternate_v2(a_opt, time_budget=alloc_total * 0.4)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score - 1e-13:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  R{r}: obj={best_score:.8f} **")
            else:
                print(f"  R{r}: obj={obj_r:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
