```python
Hybrid LP-subgradient optimizer for minimum peak autocorrelation.
Uses: (1) fast subgradient pass, (2) LP alternating with dual-guided
perturbation, (3) multi-n exploration, (4) coordinate polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # a convolved with e_j gives a shifted copy of a
        # The shift is j, so rows j to j+n-1 get a[0] to a[n-1]
        # But we only have 2n-1 rows, so:
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp_full(a):
    """Solve LP: min M s.t. conv(a,z)_k <= M, sum(z)=1, z>=0.
    Returns (z, M, dual_y) or (None, None, None)."""
    n = len(a)
    A = conv_matrix(a)
    
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ z <= M, cp.sum(z) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    try:
                        dual_y = np.maximum(np.asarray(constraints[0].dual_value), 0)
                    except:
                        dual_y = None
                    return z_val, float(M.value), dual_y
        except:
            continue
    return None, None, None

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] * n / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

```python
"""
Hybrid LP-subgradient optimizer for minimum peak autocorrelation sequences.
Uses: (1) fast subgradient with FFT-based convolution, (2) LP alternating
with active-set reduction and dual-guided perturbation, (3) systematic
multi-n exploration, (4) coordinate polish. Targets improvement over 1.506330.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_matrix(a):
    """Build (2n-1) x n matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp_full(a):
    """Solve LP: min M s.t. conv(a,z)_k <= M for all k, sum(z)=1, z>=0.
    Returns (z, M, dual_y) or (None, None, None)."""
    n = len(a)
    A = conv_matrix(a)
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    try:
                        dual_y = np.maximum(np.asarray(c1.dual_value), 0)
                    except:
                        dual_y = None
                    return z_val, float(M.value), dual_y
        except:
            continue
    return None, None, None


def solve_lp_active(a, active_idx):
    """Solve LP with only active constraints (reduced problem)."""
    n = len(a)
    A = conv_matrix(a)
    A_act = A[active_idx, :]
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    prob = cp.Problem(cp.Minimize(M), [A_act @ z <= M, cp.sum(z) == 1])
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    return z_val, float(M.value)
        except:
            continue
    return None, None

    a = np.maximum(a, 0).astype(np.float64).copy()

    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = 0.3
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        current_obj = 2.0 * n * max_val

        # Bunch active lags (within 2% of max)
        threshold = max_val * 0.98
        active_lags = np.where(conv >= threshold)[0]
        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(conv)])
        # Limit to at most 10 active lags for efficiency
        if len(active_lags) > 10:
            top_idx = np.argsort(conv[active_lags])[-10:]
            active_lags = active_lags[top_idx]

        # Compute subgradient: for lag k, subgrad[j] = 2*a[k-j] if 0<=k-j<n
        subgrad = np.zeros(n)
        for k in active_lags:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                subgrad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]

        sg_norm = np.linalg.norm(subgrad)
        if sg_norm < 1e-15:
            break

        # Polyak-like step with lower bound 1.4
        lower_bound = 1.4
        eta = alpha * (current_obj - lower_bound) / (sg_norm * sg_norm)
        eta = min(eta, 0.3)
        eta = max(eta, 1e-8)

        a_new = a - eta * subgrad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            alpha = max(alpha * 0.998, 0.01)
        else:
            improved = False
            for frac in [0.5, 0.1, 0.01]:
                a_try = a - frac * eta * subgrad
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    alpha = max(alpha * 0.5, 0.005)
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall > 15:
                    rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                    a = best_a * np.exp(rng.normal(0, 0.02, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    stall = 0
                    alpha = 0.3

        it += 1
        if verbose and it % 500 == 0:
            print(f"    SG: it={it}, obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def lp_alt_improved(a, time_budget=200.0, verbose=False):
    """Improved LP alternating with active-set reduction, dual-guided perturb, 2-cycle break."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    prev_a = None
    prev2_a = None
    stall = 0
    momentum = a.copy()
    it = 0

    while time.monotonic() - start < time_budget:
        result = solve_lp_full(a)
        if result[0] is None:
            break
        g, M_opt, dual_y = result

        # 2-cycle detection
        if prev2_a is not None:
            if np.max(np.abs(a - prev2_a)) < 1e-8:
                a_avg = 0.5 * (a + prev_a)
                a_avg = np.maximum(a_avg, 0)
                a_avg = a_avg / np.sum(a_avg)
                obj_avg = compute_obj(a_avg)
                if obj_avg < best_obj:
                    best_obj = obj_avg
                    best_a = a_avg.copy()
                    a = a_avg
                prev_a = None
                prev2_a = None
                momentum = a.copy()
                it += 1
                continue

        # Update momentum with exponential weighting
        momentum = 0.6 * momentum + 0.4 * g
        momentum = np.maximum(momentum, 0)
        gs = float(np.sum(momentum))
        if gs > 1e-10:
            momentum = momentum / gs

        # Best-select over multiple targets and mixing ratios
        targets = [g, momentum]
        best_cand_obj = np.inf
        best_cand = a.copy()

        for target in targets:
            for mix in [0.3, 0.5, 0.7, 1.0]:
                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj:
                    best_cand_obj = obj_try
                    best_cand = a_try.copy()

        if best_cand_obj < best_obj - 1e-11:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            prev2_a = prev_a
            prev_a = a.copy()
            a = best_cand
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP-alt: it={it}, obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 8:
                # Dual-guided perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                if dual_y is not None and np.max(dual_y) > 0:
                    A = conv_matrix(a)
                    active = dual_y > 0.03 * np.max(dual_y)
                    n_act = int(np.sum(active))
                    if 0 < n_act:
                        col_imp = A[active, :].sum(axis=0)
                        imp_range = np.max(col_imp) - np.min(col_imp)
                        if imp_range > 1e-10:
                            normalized_imp = (col_imp - np.min(col_imp)) / imp_range
                            pert = -(normalized_imp - 0.5) * 0.04
                            pert += rng.normal(0, 0.008, n)
                        else:
                            pert = rng.normal(0, 0.03, n)
                    else:
                        pert = rng.normal(0, 0.03, n)
                else:
                    # Random block perturbation
                    pert = rng.normal(0, 0.03, n)

                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev2_a = None
                momentum = a.copy()
                stall = 0

        # Active set reduction every 4 iterations
        if it > 0 and it % 4 == 0 and dual_y is not None and np.max(dual_y) > 0:
            active_mask = dual_y > 0.005 * np.max(dual_y)
            n_active = int(np.sum(active_mask))
            if 0 < n_active < (2 * n - 1) // 2:
                z_red, M_red = solve_lp_active(a, active_mask)
                if z_red is not None:
                    conv_check = compute_conv(a * z_red)
                    if np.max(conv_check) <= M_opt * 1.02:
                        obj_red = compute_obj(z_red)
                        if obj_red < best_obj - 1e-11:
                            best_obj = obj_red
                            best_a = z_red.copy()
                            a = z_red
                            stall = 0

        it += 1

    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] * n / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def pairwise_polish(a, time_budget=40.0):
    """Pairwise mass exchange polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for _ in range(30):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for frac in [0.02, 0.05, 0.1, 0.01]:
                delta = frac * min(a[i], a[j]) * 0.5
                if delta < 1e-12:
                    continue
                for sign in [1, -1]:
                    a_try = a.copy()
                    a_try[i] -= sign * delta
                    a_try[j] += sign * delta
                    if a_try[i] < 0 or a_try[j] < 0:
                        continue
                    a_try = a_try / np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
        if not improved and it > 50:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid LP-subgradient with multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        for n_init in [600, 800, 1000, 1200]:
            t = np.linspace(0, 1, n_init)
            a_init = 0.5 + 0.5 * np.cos(2 * np.pi * t)
            a_init = np.maximum(a_init, 0.01)
            a_init = a_init / np.sum(a_init)
            obj_init = compute_obj(a_init)
            if obj_init < best_score:
                best_score = obj_init
                best_seq = a_init.copy()
        if best_seq is None:
            best_seq = np.ones(800) / 800
            best_score = compute_obj(best_seq)
        print(f"Initial (n={len(best_seq)}): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Quick subgradient pass (60s) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Phase 1: Subgradient (n={n_parent}) ===")
        alloc = min(60.0, remaining * 0.15)
        a_sg = subgradient_pass(best_seq, time_budget=alloc, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  SG done: obj={best_score:.8f} **")
        else:
            print(f"  SG done: obj={obj_sg:.8f}")

    # === Phase 2: LP alternating main (300s) ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n=== Phase 2: LP Alternating (n={len(best_seq)}) ===")
        alloc = min(300.0, remaining * 0.4)
        a_lp = lp_alt_improved(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP done: obj={best_score:.8f} **")
        else:
            print(f"  LP done: obj={obj_lp:.8f}")

    # === Phase 3: Multi-n exploration (350s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 3: Multi-n Exploration ===")
        n_best = len(best_seq)
        # Generate candidate n values centered around best
        n_candidates = set()
        for delta in [-200, -100, -50, 0, 50, 100, 200, 400, 600]:
            n_try = n_best + delta
            if 400 <= n_try <= 2500:
                n_candidates.add(n_try)
        # Also add some fixed values
        for n_fix in [500, 700, 900, 1000, 1200, 1500, 1800]:
            if 400 <= n_fix <= 2500:
                n_candidates.add(n_fix)

        n_candidates = sorted(n_candidates, key=lambda x: abs(x - n_best))
        n_tried = {n_best}

        for n_try in n_candidates:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            if n_try in n_tried:
                continue
            n_tried.add(n_try)

            a_try = resample(best_seq, n_try)
            # Quick subgradient + brief LP
            alloc_sg = min(15.0, remaining * 0.1)
            alloc_lp = min(45.0, remaining * 0.3)

            a_opt = subgradient_pass(a_try, time_budget=alloc_sg)
            a_opt = lp_alt_improved(a_opt, time_budget=alloc_lp)
            obj_try = compute_obj(a_opt)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                print(f"  ** n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Final LP refinement on best (250s) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 4: Final LP (n={len(best_seq)}) ===")
        alloc = min(250.0, remaining * 0.5)
        a_final = lp_alt_improved(best_seq, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # === Phase 5: Pairwise polish (50s) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 5: Pairwise Polish ===")
        alloc = min(50.0, remaining * 0.3)
        a_pol = pairwise_polish(best_seq, time_budget=alloc)
        obj_pol = compute_obj(a_pol)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Polish: obj={best_score:.8f} **")
        else:
            print(f"  Polish: obj={obj_pol:.8f}")

    # === Phase 6: One more LP round if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 6: Final LP Round ===")
        alloc = min(remaining * 0.6, 60.0)
        a_last = lp_alt_improved(best_seq, time_budget=alloc)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Last LP: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
