"""
Smoothed-gradient optimizer for step function convolution minimization.
Uses L-BFGS on a log-sum-exp smoothed objective in log-space parameterization,
combined with annealing, diverse initializations, and LP polish.
Fundamentally different from pure LP-alternating approaches.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def cross_correlate(a, p):
    """Compute cross-correlation (a star p)[i] = sum_k a[k] * p[k+i] for i=0..n-1.
    a has length n, p has length 2n-1."""
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    A = rfft(a, size)
    P = rfft(p, size)
    return irfft(np.conj(A) * P, size)[:n]


def smooth_obj_and_grad_a(a, T):
    """Compute smoothed max of conv(a,a) and its gradient w.r.t. a.
    Returns (smooth_value, grad_a)."""
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    logits = (conv - max_c) / T
    exp_logits = np.exp(logits)
    Z = float(np.sum(exp_logits))
    p = exp_logits / Z
    smooth_val = max_c + T * np.log(Z)
    grad_a = cross_correlate(a, p)
    return smooth_val, grad_a


def logspace_fn(w, T, n):
    """Objective in log-space: smooth_max(conv(a,a)) where a = softmax(w)."""
    a = np.exp(w - np.max(w))
    a = a / np.sum(a)
    val, grad_a = smooth_obj_and_grad_a(a, T)
    return val


def logspace_fn_and_grad(w, T, n):
    """Objective and gradient in log-space."""
    a = np.exp(w - np.max(w))
    a = a / np.sum(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    logits = (conv - max_c) / T
    exp_logits = np.exp(logits)
    Z = float(np.sum(exp_logits))
    p = exp_logits / Z
    val = max_c + T * np.log(Z)
    grad_a = cross_correlate(a, p)
    # Chain rule: dval/dw_j = a_j * (grad_a_j - sum_i a_i * grad_a_i)
    mean_g = float(np.sum(a * grad_a))
    grad_w = a * (grad_a - mean_g)
    return val, grad_w

def lbfgs_smooth_optimize(a_init, time_budget=100.0, verbose=False):

        if time.monotonic() - start > time_budget:

            if res.x is not None:

                        if verbose:

def exp_weights_descent(a_init, time_budget=100.0, verbose=False):
    """Exponential weights (multiplicative) descent with annealing."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    max_conv_init = float(np.max(compute_conv(a)))
    T_high = max_conv_init * 3.0
    T_low = max_conv_init * 0.02
    n_temps = 10
    temps = T_high * (T_low / T_high) ** (np.arange(n_temps) / max(n_temps - 1, 1))
    
    iters_per_temp = max(5, int(time_budget / (n_temps * 0.5)))
    
    for ti, T in enumerate(temps):
        for it in range(iters_per_temp):
            if time.monotonic() - start > time_budget:
                break
            
            _, grad_a = smooth_obj_and_grad_a(a, T)
            
            # Adaptive eta
            eta = n / (np.max(np.abs(grad_a)) + 1e-10) * 0.1
            
            # Exponential update
            log_a = np.log(a + 1e-30) - eta * grad_a
            log_a = log_a - np.max(log_a)
            a = np.exp(log_a)
            a = a / np.sum(a)
            
            # Periodic check of true objective
            if it % 3 == 0:
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,z)_k <= M, sum(z)=1, z>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ z <= M, cp.sum(z) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    return z_val, float(M.value)
        except:
            continue
    
    # Fallback
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n])
    except:
        pass
    
    return None, None


def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating with 2-cycle breaking."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    prev_a = None
    prev_prev_a = None
    stall = 0
    
    while time.monotonic() - start < time_budget:
        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp = result
        
        # 2-cycle detection
        if prev_prev_a is not None:
            if np.max(np.abs(a - prev_prev_a)) < 1e-7:
                a_break = 0.5 * (a + prev_a)
                a_break = np.maximum(a_break, 0)
                sb = float(np.sum(a_break))

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    try:
        from scipy.interpolate import interp1d
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
        a_new = f(x_new)
    except:
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:

                delta = frac * min(a[i], a[j])

                if obj_try < best_obj - 1e-14:

                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        if not improved and it > 20:
            break

"""
Smoothed-gradient optimizer for step function convolution minimization.
Primary approach: L-BFGS on log-sum-exp smoothed objective in log-space,
combined with exponential weights (multiplicative) descent, annealing,
diverse structured initializations, and LP polish as a secondary method.
This is fundamentally different from the pure LP-alternating approach.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def cross_correlate(a, p):
    """Compute (a *star* p)[i] = sum_k a[k]*p[k+i] for i=0..n-1.
    a has length n, p has length 2n-1. Returns array of length n."""
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    A = rfft(a, size)
    P = rfft(p, size)
    return irfft(np.conj(A) * P, size)[:n]


def smooth_obj_and_grad_a(a, T):
    """Compute smoothed max of conv(a,a) and gradient w.r.t. a.
    Returns (smooth_value, grad_a of same length as a)."""
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    logits = (conv - max_c) / T
    exp_logits = np.exp(logits)
    Z = float(np.sum(exp_logits))
    p = exp_logits / Z
    smooth_val = max_c + T * np.log(Z)
    grad_a = cross_correlate(a, p)
    return smooth_val, grad_a


def logspace_fn_and_grad(w, T, n):
    """Objective and gradient in log-space for L-BFGS.
    a = softmax(w), objective = smooth_max(conv(a,a))."""
    a = np.exp(w - np.max(w))
    a = a / np.sum(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    logits = (conv - max_c) / T
    exp_logits = np.exp(logits)
    Z = float(np.sum(exp_logits))
    p = exp_logits / Z
    val = max_c + T * np.log(Z)
    grad_a = cross_correlate(a, p)
    mean_g = float(np.sum(a * grad_a))
    grad_w = a * (grad_a - mean_g)
    return val, grad_w


def generate_initialization(n, kind, rng):
    """Generate a structured initialization of length n."""
    if kind == 'uniform':
        a = np.ones(n) / n
    elif kind == 'u_shape':
        t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = 0.3 + 0.7 * (2.0 * t - 1.0) ** 2
    elif kind == 'u_shape_strong':
        t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = 0.05 + 0.95 * (2.0 * t - 1.0) ** 4
    elif kind == 'cosine':
        t = np.arange(n, dtype=np.float64) / n
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 0.01)
    elif kind == 'sine_sq':
        t = np.arange(n, dtype=np.float64) / n
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 0.001)
    elif kind == 'fourier2':
        t = np.arange(n, dtype=np.float64) / n
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 2.0 * t)
        a = np.maximum(a, 0.01)
    elif kind == 'fourier3':
        t = np.arange(n, dtype=np.float64) / n
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 3.0 * t) + 0.1 * np.cos(2.0 * np.pi * 5.0 * t)
        a = np.maximum(a, 0.01)
    elif kind == 'bimodal':
        t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.8) ** 2) + 0.05
    elif kind == 'bimodal_wide':
        t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = np.exp(-20.0 * (t - 0.15) ** 2) + np.exp(-20.0 * (t - 0.85) ** 2) + 0.1
    elif kind == 'triangle':
        t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 0.01)
    elif kind == 'random':
        a = rng.uniform(0.1, 1.0, n)
    elif kind == 'random_smooth':
        a = rng.uniform(0.1, 1.0, n)
        # Smooth it
        kernel = np.ones(5) / 5.0
        a = np.convolve(a, kernel, mode='same')
    elif kind == 'cubic_u':
        t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = 0.2 + 0.8 * (2.0 * t - 1.0) ** 2 * (2.0 * t - 1.0) ** 2
        a += 0.1 * (2.0 * t - 1.0) ** 2
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 1e-10)
    a = a / np.sum(a)
    return a


def lbfgs_smooth_optimize(a_init, time_budget=100.0, verbose=False):
    """Run L-BFGS on smoothed objective in log-space with temperature annealing."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / np.sum(a_init)

    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    max_conv_init = float(np.max(compute_conv(a_init)))
    T_high = max_conv_init * 5.0
    T_low = max_conv_init * 0.005
    n_temps = 10
    temps = T_high * (T_low / T_high) ** (np.arange(n_temps, dtype=np.float64) / max(n_temps - 1, 1))

    w = np.log(a_init + 1e-30)

    for ti, T in enumerate(temps):
        if time.monotonic() - start > time_budget:
            break

        maxiter = max(10, int(200 * (time_budget / max(time.monotonic() - start + 1, 1) / n_temps)))
        try:
            res = minimize(logspace_fn_and_grad, w, args=(T, n),
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': maxiter, 'gtol': 1e-9,
                                    'maxls': 30, 'iprint': 0})
            if res.x is not None:
                a_new = np.exp(res.x - np.max(res.x))
                sn = float(np.sum(a_new))
                if sn > 1e-10:
                    a_new = a_new / sn
                    a_new = np.maximum(a_new, 0)
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        w = res.x.copy()
                        if verbose:
                            print(f"    L-BFGS T={ti}/{n_temps}, T={T:.8f}, obj={best_obj:.8f}")
        except Exception:
            pass

        # Also try a direct (non-annealed) step with current w
        if time.monotonic() - start > time_budget * 0.7:
            break

    return best_a, best_obj


def exp_weights_descent(a_init, time_budget=100.0, verbose=False):
    """Exponential weights (multiplicative) descent with annealed temperature."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    max_conv_init = float(np.max(compute_conv(a)))
    T_high = max_conv_init * 4.0
    T_low = max_conv_init * 0.01
    n_temps = 12
    temps = T_high * (T_low / T_high) ** (np.arange(n_temps, dtype=np.float64) / max(n_temps - 1, 1))

    for ti, T in enumerate(temps):
        # Adaptive iterations per temperature
        remaining_for_this = time_budget * 0.8 - (time.monotonic() - start)
        budget_per_temp = time_budget / n_temps * 0.8
        iters_per_temp = max(3, int(budget_per_temp / 0.3))

        for it in range(iters_per_temp):
            if time.monotonic() - start > time_budget:
                break
            try:
                _, grad_a = smooth_obj_and_grad_a(a, T)
                # Adaptive learning rate
                grad_mag = float(np.max(np.abs(grad_a)))
                if grad_mag < 1e-15:
                    break
                eta = 0.3 / grad_mag

                # Multiplicative update
                log_a = np.log(a + 1e-30) - eta * grad_a
                log_a = log_a - np.max(log_a)
                a_new = np.exp(log_a)
                sn = float(np.sum(a_new))
                if sn > 1e-10:
                    a = a_new / sn

                # Periodic check
                if it % 2 == 0:
                    obj = compute_obj(a)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                        if verbose and ti % 3 == 0:
                            print(f"    ExpW T={ti}/{n_temps}, obj={best_obj:.8f}")
            except Exception:
                break

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,z)_k <= M for all k, sum(z)=1, z>=0."""
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ z <= M, cp.sum(z) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    return z_val, float(M.value)
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating with 2-cycle breaking and adaptive mixing."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    prev_a = None
    prev_prev_a = None
    g_momentum = a.copy()
    stall = 0
    rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))

    while time.monotonic() - start < time_budget:
        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp = result

        # 2-cycle detection
        if prev_prev_a is not None:
            if np.max(np.abs(a - prev_prev_a)) < 1e-7:
                a_break = 0.5 * (a + prev_a)
                a_break = np.maximum(a_break, 0)
                sb = float(np.sum(a_break))
                if sb > 1e-10:
                    a_break = a_break / sb
                    obj_break = compute_obj(a_break)
                    if obj_break < best_obj:
                        best_obj = obj_break
                        best_a = a_break.copy()
                    a = a_break
                    prev_a = None
                    prev_prev_a = None
                    g_momentum = a.copy()
                    continue

        # Update momentum
        g_momentum = 0.5 * g_momentum + 0.5 * g_lp
        g_momentum = np.maximum(g_momentum, 0)
        gm = float(np.sum(g_momentum))
        if gm > 1e-10:
            g_momentum = g_momentum / gm

        # Try multiple targets and mixing ratios
        targets = [g_lp, g_momentum]
        best_cand_obj = np.inf
        best_cand = a.copy()

        for target in targets:
            for mix in [0.4, 0.6, 0.8, 0.3]:
                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj:
                    best_cand_obj = obj_try
                    best_cand = a_try.copy()

        if best_cand_obj < best_obj - 1e-13:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            prev_a = a.copy()
            prev_prev_a = prev_a
            a = best_a
            stall = 0
            if verbose:
                print(f"    LP-alt: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 8:
                # Perturb to escape
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev_prev_a = None
                g_momentum = a.copy()
                stall = 0

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    try:
        from scipy.interpolate import interp1d
        f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
        a_new = f(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-10)
    return a_new / np.sum(a_new)


def pairwise_polish(a, time_budget=30.0):
    """Pairwise exchange local search."""
    n = len(a)
    a = np.maximum(a, 1e-15).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for _ in range(15):
            if time.monotonic() - start > time_budget:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            for frac in [0.05, 0.1, 0.15, 0.02, 0.08]:
                delta = frac * min(a[i], a[j])
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0:
                    continue
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        if not improved and it > 30:
            break

    return best_a, best_obj


def nesterov_exp_weights(a_init, time_budget=100.0, verbose=False):
    """Nesterov-accelerated exponential weights descent."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / np.sum(a)
    a_prev = a.copy()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    max_conv_init = float(np.max(compute_conv(a)))
    T = max_conv_init * 1.0
    n_total_iters = int(time_budget / 0.2)

    for t in range(1, n_total_iters + 1):
        if time.monotonic() - start > time_budget:
            break

        # Anneal temperature
        T_cur = max_conv_init * 1.0 * (0.01) ** (t / max(n_total_iters, 1))
        T_cur = max(T_cur, max_conv_init * 0.005)

        # Nesterov extrapolation
        momentum = (t - 1) / (t + 1)
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 1e-15)
        y = y / np.sum(y)

        try:
            _, grad_y = smooth_obj_and_grad_a(y, T_cur)
            grad_mag = float(np.max(np.abs(grad_y)))
            if grad_mag < 1e-15:
                break
            eta = 0.5 / grad_mag

            # Multiplicative update from a (not y)
            log_a = np.log(a + 1e-30) - eta * grad_y
            log_a = log_a - np.max(log_a)
            a_new = np.exp(log_a)
            sn = float(np.sum(a_new))
            if sn > 1e-10:
                a_prev = a.copy()
                a = a_new / sn

            if t % 5 == 0:
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose and t % 50 == 0:
                        print(f"    NesterovExpW t={t}, obj={best_obj:.8f}")
        except Exception:
            break

    return best_a, best_obj
