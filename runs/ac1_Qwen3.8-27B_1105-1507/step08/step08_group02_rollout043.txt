Optimized step function search using binomial initialization, fast mirror descent
with Nesterov momentum, LP refinement, and multi-scale exploration.
Key innovation: binomial coefficient initialization exploits the generating function
identity (1-x)^(-1) = (1-x)^(-1/2)^2 to create a near-flat convolution.

import numpy as np
import time
from math import comb
from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    """Linear convolution via real FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n]

def compute_conv(a):
    """Compute a*a linear convolution."""
    return fft_convolve(a, a)

def compute_obj(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def binomial_init(n):
    """Initialize with binomial coefficients C(2i,i)/4^i, normalized."""
    b = np.ones(n)
    for i in range(1, n):
        b[i] = b[i-1] * (2*i) * (2*i-1) / (4*i*i)
    return b / b.sum()

def mirror_descend(a, time_budget=200.0, verbose=False):

    Mirror descent (exponential family) on the simplex with Nesterov momentum.
    Uses single-peak subgradient for efficiency.
    """

    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 3.0 / np.sqrt(n)
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    patience = 50
    it = 0

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        k_star = int(np.argmax(conv))

        j_start = max(0, k_star - (n - 1))

        if j_start <= j_end:
            # g[j] = 2 * a[k_star - j]
            # When j = j_start: a[k_star - j_start]
            # When j = j_end: a[k_star - j_end]
            idx = k_star - np.arange(j_start, j_end + 1)
            g[j_start:j_end+1] = 2.0 * a[idx]

        log_a_new -= np.max(log_a_new)  # numerical stability

        else:

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            # Slightly increase momentum and decrease lr
            momentum = min(0.85, momentum + 0.001)
            lr = max(0.1 / n, lr * 0.999)
        else:
            stall += 1
            lr = max(0.01 / n, lr * 0.9)
            momentum = max(0.3, momentum * 0.99)
            
            if stall > patience:
                # Annealing restart

                noise = rng.normal(0, 0.05 + 0.02 * (stall // patience), n)
                a = best_a * np.exp(noise)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = 0.5
                lr = max(0.1 / n, lr * 0.5)
                stall = 0
        
        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      MD it{it}: obj={best_obj:.8f}, lr={lr:.6f}, mom={momentum:.3f}, t={elapsed:.1f}s")
    
    return best_a

def soft_mirror_descend(a, time_budget=100.0, beta=5.0, verbose=False):
    """
    Mirror descent with soft (log-sum-exp) subgradient.
    Uses top-k peaks for a smoother gradient direction.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 2.0 / np.sqrt(n)
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    patience = 30
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        m = float(np.max(conv))
        
        # Soft weights (log-sum-exp)
        shifted = beta * (conv - m)
        w = np.exp(shifted)
        w_sum = float(np.sum(w))
        if w_sum < 1e-15:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum
        
        # Gradient via cross-correlation
        # g[j] = 2 * sum_k w[k] * a[k-j]
        # = 2 * np.correlate(w, a, 'full')[:n]
        g = 2.0 * np.correlate(w, a, mode='full')[:n]
        g = np.maximum(g, 0)  # gradient should be non-negative
        
        # Nesterov momentum
        v = momentum * v + g
        
        # Mirror descent
        log_a = np.log(a)
        log_a_new = log_a - lr * v
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            momentum = min(0.8, momentum + 0.002)
        else:
            stall += 1
            lr = max(0.05 / n, lr * 0.92)
            if stall > patience:
                rng = np.random.default_rng(it * 31337 + n)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = a / a.sum()
                v = np.zeros(n)
                stall = 0
                momentum = 0.5
                lr = 2.0 / np.sqrt(n)
        
        it += 1
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"      SoftMD it{it}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a

def lp_step(a, max_iters=20, verbose=False):
    """
    One step of alternating LP: fix a, find g that minimizes max(a*g).
    Approximate with projected subgradient on g.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    
    # Initialize g as uniform
    g = np.ones(n) / n
    
    lr = 0.1
    for it in range(max_iters):
        # Compute a*g convolution
        conv_ag = fft_convolve(a_norm, g)
        k_star = int(np.argmax(conv_ag))
        max_val = float(conv_ag[k_star])
        
        # Gradient of (a*g)[k] w.r.t. g[j] = a[k-j]
        grad = np.zeros(n)
        j_start = max(0, k_star - (n-1))
        j_end = min(n-1, k_star)
        if j_start <= j_end:
            idx = k_star - np.arange(j_start, j_end + 1)
            grad[j_start:j_end+1] = a_norm[idx]
        
        # Projected gradient descent on simplex
        g -= lr * grad
        # Project onto simplex
        g = project_simplex(g)
        
        # Adaptive lr
        conv_check = fft_convolve(a_norm, g)
        if np.max(conv_check) < max_val:
            lr = min(0.5, lr * 1.1)
        else:
            lr = max(0.01, lr * 0.7)
    
    return g

def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def alternating_optimize(a, time_budget=100.0, verbose=False):
    """
    Alternating optimization: optimize a with a fixed, then optimize the dual.
    Uses the LP formulation: min M s.t. a*g <= M, sum g = 1, g >= 0.
    We approximate the LP with fast projected gradient.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    stall = 0
    patience = 30
    
    while time.monotonic() - start < time_budget:
        # Step 1: Find g that minimizes max(a*g) (approximate LP)
        g = lp_step(a, max_iters=15)
        
        # Step 2: Blend a towards g
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                stall = 0
                break
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(int(time.monotonic() * 1000) + n)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
    
    return best_a

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        return binomial_init(n)
    elif mode == 1:
        a = np.ones(n) / n
        return a
    elif mode == 2:
        # Power law decay
        a = 1.0 / np.power(np.arange(1, n+1), rng.uniform(0.3, 1.5))
        return a / a.sum()
    elif mode == 3:
        # Exponential decay
        rate = rng.uniform(0.01, 0.2)
        a = np.exp(-rate * np.arange(n))
        return a / a.sum()
    elif mode == 4:
        # Bimodal: high at both ends
        x = np.arange(n) / n
        a = np.exp(-20 * (x - 0.1)**2) + np.exp(-20 * (x - 0.9)**2) + 0.01
        return a / a.sum()
    elif mode == 5:
        # Sinusoidal modulation
        freq = rng.uniform(1, 5)
        x = np.arange(n) / n
        a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 6:
        # S-curve (sigmoid)
        x = np.arange(n) / n
        steep = rng.uniform(5, 20)
        a = 1.0 / (1.0 + np.exp(-steep * (x - 0.5)))
        return a / a.sum()
    elif mode == 7:
        # Two-step: high first half, low second half
        a = np.ones(n)
        a[n//2:] = rng.uniform(0.1, 0.5)
        return a / a.sum()
    else:
        return binomial_init(n)
