Uses smooth approximation of max to get better gradient information from all peaks,
with adaptive temperature cooling and periodic LP-based refinement.
"""

from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def smoothed_grad_obj(a, tau):
    """
    Compute log-sum-exp smoothed objective and gradient.
    a should be normalized (sum=1).
    Returns (smoothed_max, gradient).
    """
    conv = compute_conv(a)
    conv_max = np.max(conv)
    # Numerically stable log-sum-exp

    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    log_sum_exp = conv_max + np.log(sum_exp) / tau
    
    # Softmax weights
    p = exp_shifted / sum_exp
    
    # Gradient: 2 * correlate(p, a, 'valid')
    grad = 2.0 * np.correlate(p, a, mode='valid')
    
    return log_sum_exp, grad

def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):

def coordinate_descent(a0, time_budget=60.0, verbose=False):
    """
    Coordinate descent: for each coordinate, find the optimal value 
    (given others fixed) that minimizes the objective.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute conv excluding each element
    # conv[k] = sum_i a[i]*a[k-i]
    # If we change a[m] to a[m]+delta:
    # conv[k] changes by: delta * (a[k-m] + a[m]) + delta^2 (if k=2m)
    # Actually: new_conv[k] = conv[k] + delta*(a[k-m] + a[m]) + delta^2 (if 2m==k)
    # Wait, more carefully:
    # conv[k] = sum_i a[i]*a[k-i]
    # new_conv[k] = sum_i (a[i]+delta*δ_{i,m})*(a[k-i]+delta*δ_{k-i,m})
    # = conv[k] + delta * (a[k-m] + a[m]) + delta^2 * δ_{k,2m}
    # (where a[k-m] is 0 if k-m not in [0,n-1])
    
    # So new_conv[k] = conv[k] + delta * (a[k-m] + a[m]) + delta^2 * (1 if k==2m else 0)
    # The objective is max_k new_conv[k] * 2n (since sum is unchanged if we renormalize)
    # Wait, if we change a[m], the sum changes too. Let's keep sum fixed.
    # If we increase a[m] by delta, we need to decrease something else by delta.
    # That's more complex. Let's just do: change a[m], then renormalize.
    # After renormalization, the effect is more complex.
    
    # Simpler approach: for each m, try a few values of a[m] (keeping others fixed),
    # renormalize, and check the objective.
    
    order = np.random.default_rng(42).permutation(n)
    
    for epoch in range(50):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for m in order:
            if time.monotonic() - start > time_budget:
                break
            a_save = a[m]
            # Try scaling a[m] by various factors
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a[m] = a_save * factor
                s_try = float(np.sum(a))
                if s_try < 1e-10:
                    a[m] = a_save
                    continue
                a_try = a / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
                elif factor < 1.0:
                    # If decreasing didn't help, stop trying smaller
                    if factor <= 0.5:
                        break
                else:
                    # If increasing didn't help, stop trying larger
                    if factor >= 1.5:
                        break
            if a[m] != a_save and obj_try >= best_obj:
                a[m] = best_a[m]  # Revert if not improving
            elif a[m] != a_save:
                a = best_a.copy()
        
        if not improved and epoch > 5:
            break
    
    return best_a


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

Hybrid optimizer: Log-sum-exp smoothed gradient descent (mirror descent) + LP refinement + coordinate descent.
Key innovation: uses smooth approximation of max via log-sum-exp to get gradient information
from all peaks simultaneously, combined with multiplicative weights update for stable simplex optimization.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def smoothed_grad(a, tau):
    """Compute log-sum-exp value and gradient for smoothed max of conv(a,a)."""
    conv = compute_conv(a)
    conv_max = np.max(conv)
    shifted = tau * (conv - conv_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    if sum_exp < 1e-300:
        # Fallback: uniform weights
        p = np.ones(len(conv)) / len(conv)
        log_sum_exp = conv_max
    else:
        log_sum_exp = conv_max + np.log(sum_exp) / tau
        p = exp_shifted / sum_exp
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return log_sum_exp, grad


def mirror_descent_step(a, grad, lr):
    """Multiplicative weights update on simplex."""
    # a_new[m] = a[m] * exp(-lr * grad[m]) / Z
    scaled = -lr * grad
    scaled = np.clip(scaled, -50, 50)
    weights = a * np.exp(scaled)
    Z = np.sum(weights)
    if Z < 1e-300:
        return a.copy()
    return weights / Z

                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_line_obj(t, a, g_lp):
    """Objective for 1D line search in LP alternating."""
    a_try = (1.0 - t) * a + t * g_lp
    a_try = np.maximum(a_try, 0)
    st = float(np.sum(a_try))
    if st < 1e-10:
        return np.inf
    a_try = a_try / st
    return compute_obj(a_try)


def lp_alternate(a0, time_budget=60.0, verbose=False):
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        try:
            res = minimize_scalar(lp_line_obj, bounds=(0.0, 1.0), method='bounded',
                                  args=(a, g_lp), options={'maxiter': 30, 'xatol': 0.01})
            best_t = res.x if res.success else 0.5
        except Exception:
            best_t = 0.5
        
        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall_count = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}")
        else:
            stall_count += 1
            if stall_count > 8:
                break
    
    return best_a


def coord_desc_obj(a, m, factor):
    """Objective for coordinate descent: scale a[m] by factor, renormalize."""
    a_try = a.copy()
    a_try[m] *= factor
    a_try = np.maximum(a_try, 0)
    st = float(np.sum(a_try))
    if st < 1e-10:
        return np.inf
    a_try = a_try / st
    return compute_obj(a_try), a_try


def coordinate_descent(a0, time_budget=60.0, verbose=False):
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    order = rng.permutation(n)
    
    for epoch in range(30):
        if time.monotonic() - start > time_budget:
            break
        epoch_improved = False
        for idx in range(n):
            m = int(order[idx])
            a_save = a[m]
            if a_save < 1e-12:
                continue
            # Try several factors
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                obj_try, a_try = coord_desc_obj(a, m, factor)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    epoch_improved = True
                    break
        if verbose and (epoch + 1) % 5 == 0:
            print(f"      CD epoch={epoch}: obj={best_obj:.8f}")
        if not epoch_improved and epoch > 3:
            break
    
    return best_a


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
