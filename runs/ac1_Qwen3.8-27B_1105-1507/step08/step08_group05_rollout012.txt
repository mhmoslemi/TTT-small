     p ≥ 0

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def smooth_conv_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum exp(beta*conv(a,a)[k])) w.r.t. a.
    Returns gradient vector of length n.
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    conv = compute_conv(a_norm)
    # Softmax weights
    max_c = np.max(conv)
    p = np.exp(beta * (conv - max_c))
    p_sum = np.sum(p)
    if p_sum < 1e-300:
        # Fallback: uniform weights
        p = np.ones(2 * n - 1) / (2 * n - 1)

        p = p / p_sum

    # We want conv_pa[2n-2-i] for i=0,...,n-1

    grad = 2.0 * conv_pa[n - 1: 2 * n - 1][::-1]
    return grad, p

    total_steps = 5000
    step_size = 0.1

        if time.monotonic() - start > time_budget:
            break

        frac = it / total_steps
        beta = 1.0 + 299.0 * frac

            a_try = project_simplex(a_try)

            if f_try < f_curr:
                improved = True
                break
            step *= 0.5

            # Check true objective

            if step_size <= 0.001:

                pass

    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, solver_pref=None):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    if solver_pref:
        solvers = [solver_pref] + [s for s in solvers if s != solver_pref]
    
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=100.0, verbose=False):

    n = len(a0)
    a = a0.astype(np.float64).copy()

        return a0.copy()
    a = a / s

    start = time.monotonic()
    stall_count = 0
    last_good_t = 0.5

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        # 1D line search
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)
        
        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 30, 'xatol': 0.01})
            best_t = res.x if res.success else last_good_t
        except Exception:
            best_t = last_good_t
        
        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            last_good_t = best_t
            stall_count = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 15:
                break

    return best_a


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def coordinate_descent(a0, time_budget=60.0, verbose=False):
    """
    Coordinate descent: for each coordinate, find optimal value.
    Uses the fact that conv(a,a)[k] is quadratic in a[i].
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    sweep = list(range(n))
    rng.shuffle(sweep)
    
    for sweep_idx in range(len(sweep)):
        if time.monotonic() - start > time_budget:
            break
        i = sweep[sweep_idx]
        
        # Find optimal a[i] keeping others fixed
        # conv[a][k] as function of t = a[i]:
        # conv[k] = base_k + linear_k * t + quad_k * t^2
        # where:
        #   quad_k = 1 if 2*i == k and i < n (i.e., k = 2i), else 0
        #   linear_k = 2*a[k-i] if 0 <= k-i < n and k-i != i, 
        #              + a[k-i] if 0 <= k-i < n and k-i == i... 
        #   Actually let me just compute the change directly
        
        old_val = a[i]
        a[i] = 0  # temporarily zero it out
        base_conv = compute_conv(a)  # convolution with a[i]=0
        
        # Now conv with a[i] = t:
        # The additional terms are:
        # For each k, the terms involving a[i]=t are:
        #   a[i]*a[k-i] + a[k-i]*a[i] = 2*t*a[k-i] (if i != k-i and both valid)
        #   a[i]*a[i] = t^2 (if i == k-i, i.e., k = 2i)
        #   t*a[k-i] (if only one of i, k-i is valid)
        
        # Let me compute the linear and quadratic coefficients
        quad_coeffs = np.zeros(2 * n - 1)
        linear_coeffs = np.zeros(2 * n - 1)
        
        for k in range(2 * n - 1):
            # Contribution of a[i]=t to conv[k]:
            # a[i] appears as a[k-i] when the "other index" is i
            # a[i] appears as a[i] when the "first index" is i (i.e., k-j = i, j = k-i)
            
            # Term 1: sum over j, a[j]*a[k-j]. When j=i: a[i]*a[k-i] = t*a[k-i]
            #   (valid if 0 <= k-i < n and k-i != i)
            #   (valid if 0 <= k-i < n and k-i == i: a[i]*a[i] = t^2)
            
            # Term 2: When k-j = i, i.e., j = k-i: a[k-i]*a[i] = a[k-i]*t
            #   (valid if 0 <= k-i < n and k-i != i)
            
            # So total linear coefficient (coefficient of t):
            #   If 0 <= k-i < n:
            #     If k-i != i: 2*a[k-i] (two terms: a[i]*a[k-i] and a[k-i]*a[i])
            #     If k-i == i: a[i] = t (one term: a[i]*a[i] = t^2, so it's quadratic)
            #   Wait, when k-i == i, the term is a[i]*a[i] = t^2, which is quadratic, not linear.
            #   And there's no separate "a[k-i]*a[i]" term because it's the same term.
            
            # Let me redo: conv[k] = sum_{j: valid} a[j] * a[k-j]
            # Terms involving a[i]:
            #   j = i: a[i]*a[k-i] = t * a[k-i] (if 0 <= k-i < n)
            #   k-j = i => j = k-i: a[k-i]*a[i] = a[k-i]*t (if 0 <= k-i < n and k-i != i)
            
            # So:
            # If 0 <= k-i < n:
            #   If k-i != i: linear coeff += a[k-i] (from j=i) + a[k-i] (from j=k-i) = 2*a[k-i]
            #     Wait, but a[k-i] here is the original value (before we set a[i]=0).
            #     Hmm, but we set a[i]=0 already. So a[k-i] in base_conv already excludes a[i].
            #     The linear coefficient should be: a_original[k-i] (the value of a at index k-i, 
            #     which is NOT i since k-i != i).
            #   If k-i == i (i.e., k = 2i): the term is a[i]*a[i] = t^2, so quad coeff += 1
            
            if k - i >= 0 and k - i < n:
                if k - i != i:
                    # Linear: 2 * a_original[k-i]
                    # But we need the original a[k-i], not the modified one
                    # Since we only modified a[i], and k-i != i, a[k-i] is unchanged
                    linear_coeffs[k] = 2.0 * a[k - i]  # a[k-i] is still the original value
                else:
                    # k = 2i: quadratic term t^2
                    quad_coeffs[k] = 1.0
            
            # Also, when j = k-i (the "second factor" is a[i]):
            # This gives a[k-i] * a[i] = a[k-i] * t
            # But we already counted this above (it's the same as the "j = k-i" case)
            # Wait, no. Let me be more careful.
            # conv[k] = sum_j a[j]*a[k-j]
            # The term with j=i is: a[i]*a[k-i] = t*a[k-i] (requires 0<=k-i<n)
            # The term with j=k-i is: a[k-i]*a[i] = a[k-i]*t (requires 0<=k-i<n AND k-i != i)
            # If k-i == i, then j=i and j=k-i are the SAME term, so we only count it once.
            # In that case, the term is a[i]*a[i] = t^2.
            
            # So my analysis above is correct. But I need to use the ORIGINAL a[k-i] values.
            # Since I set a[i]=0, a[k-i] for k-i != i is still the original.
            # For k-i == i, a[i] is now 0, but the original was old_val.
            # In that case, quad_coeff = 1 (t^2 term), which is correct.
        
        # Now find t > 0 that minimizes max_k (base_conv[k] + linear_coeffs[k]*t + quad_coeffs[k]*t^2)
        # This is a 1D minimization of a piecewise quadratic (max of quadratics).
        # For simplicity, use golden section or bisection on the derivative.
        
        # The range of t: [0, old_val * 3] (allow some expansion)
        t_low = 0.0
        t_high = old_val * 5 + 0.01
        
        def obj_at_t(t):
            conv_t = base_conv + linear_coeffs * t + quad_coeffs * t * t
            return float(np.max(conv_t))
        
        # Bounded minimization
        try:
            res = minimize_scalar(obj_at_t, bounds=(t_low, t_high), method='bounded',
                                  options={'maxiter': 50, 'xatol': 1e-6})
            t_opt = res.x
        except Exception:
            t_opt = old_val
        
        a[i] = max(0.0, t_opt)
        a = np.maximum(a, 0)
        st = float(np.sum(a))
        if st < 1e-10:
            a[i] = old_val
            continue
        a = a / st
        
        if time.monotonic() - start > time_budget:
            a = best_a.copy()
            break
        
        true_obj = compute_obj(a)
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a.copy()
            if verbose and sweep_idx % 100 == 0:
                print(f"      CoordDescent i={i}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    
    return best_a


def nesterov_subgrad(a0, time_budget=60.0, top_k=5, verbose=False):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.08
    patience = 0

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                grad[i_start:i_end + 1] += 2.0 * y[i_start:i_end + 1][::-1]
        grad /= len(top_idx)
        grad -= np.mean(grad)
        
        y_new = y - step_size * grad
        a_new = project_simplex(y_new)
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = project_simplex(y)
        a = a_new
        
        if it % 10 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if
