import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    size = 1 << (size - 1).bit_length()
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv(a_norm)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_obj_and_grad(z, n, tau):
    """Smooth objective and gradient using parameterization a = exp(z)/sum(exp(z))."""
    a = np.exp(z - z.max())
    s = a.sum()
    a = a / s
    conv = compute_conv(a)
    # Soft-max
    shifted = conv / tau
    shifted = shifted - shifted.max()
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    sm = tau * (np.log(sum_exp) + shifted.max())
    obj = 2.0 * n * sm / 1.0  # since a is normalized, sum=1, so obj = 2n*sm
    
    # Gradient of soft-max w.r.t. a
    weights = exp_shifted / sum_exp
    grad_a = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-12:
            continue
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        grad_a[i_min:i_max + 1] += weights[k] * a[k - np.arange(i_min, i_max + 1)]
    
    # Chain rule: grad_z = grad_a * a (from softmax parameterization, d a_i / d z_j = a_i (delta_ij - a_j))
    # Full gradient of obj w.r.t. z:
    # d obj / d z_j = sum_i (d obj / d a_i)(d a_i / d z_j)
    # = sum_i grad_a[i] * a[i] * (delta_ij - a[j])

    # Actually let's just do: grad_obj_da * (a * (I - a*a^T))

def projected_subgradient(a_init, time_budget=200.0):
    """Aggressive projected subgradient with momentum and adaptive step."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    y = a.copy()
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.1
    stall_count = 0
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(y)
        max_val = conv.max()
        k_star = int(np.argmax(conv))
        
        # Subgradient of max at k_star
        # d conv[k_star]/da[i] = a[k_star - i] if 0 <= k_star - i < n
        grad = np.zeros(n)
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        grad[i_min:i_max + 1] = a[k_star - np.arange(i_min, i_max + 1)]
        
        # Average with nearby peaks for smoother gradient
        threshold = max_val * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) > 1:
            grad_avg = np.zeros(n)
            for k in active[:20]:
                k = int(k)
                i_min = max(0, k - n + 1)
                i_max = min(n - 1, k)
                grad_avg[i_min:i_max + 1] += a[k - np.arange(i_min, i_max + 1)]
            grad = grad_avg / len(active)
        
        # Center gradient (maintain sum)
        grad_centered = grad - grad.mean()
        
        # Momentum step
        momentum = beta * momentum - lr * grad_centered
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = y_new.sum()
        if s_new < 1e-10:
            y_new = y.copy()
        else:
            y_new = y_new / s_new
        
        obj = compute_obj(y_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        y = y_new
        
        # Adaptive learning rate
        if it % 500 == 499:
            if stall_count > 200:
                lr = max(lr * 0.8, 0.001)
                beta = min(beta + 0.01, 0.99)
            elif stall_count > 50:
                # Perturb
                rng = np.random.default_rng(hash((it, n)) % (2**31))
                y = best_a * np.exp(rng.normal(0, 0.05, n))
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
                lr = 0.1
        
        # Reduce lr over time
        lr = max(lr * (1 - 1e-5), 0.001)
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def lp_alternating(a_init, time_budget=300.0, momentum_beta=0.5):
    """LP alternating with momentum."""
    n = len(a_init)
    a = np.maximum(a_init, 0).copy()
    s = a.sum()
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    last_t = 0.5
    
    while time.monotonic() - start < time_budget:
        # Solve LP: min M s.t. conv(a, g) <= M, sum(g)=1, g>=0
        A = build_conv_matrix(a)
        m = 2 * n - 1
        
        # Use scipy linprog (reliable)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'maxiter': 5000})
        except Exception:
            break
        
        if res.status != 0 or res.x is None:
            break
        
        g_lp = np.maximum(res.x[:n], 0)
        gs = g_lp.sum()
        if gs < 1e-10:
            break
        g_lp = g_lp / gs
        
        # Line search with momentum
        direction = g_lp - a
        momentum = momentum_beta * momentum + (1 - momentum_beta) * direction
        
        step_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = a.copy()
        
        for t in step_candidates:
            a_try = a + t * momentum
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
        else:
            momentum *= 0.5
            # Perturb
            if np.random.random() < 0.1:
                scale = 0.03
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
    
    return best_a


def parametric_shape(n, p=1.0, shift=0.0, width=1.0, center=0.5):
    """Parametric shape: ((j+1)/n)^p * gaussian_envelope."""
    j = np.arange(n, dtype=np.float64)
    base = ((j + 1) / n) ** p
    if width < 1.0:
        envelope = np.exp(-((j/n - center) / width) ** 2)
        base = base * (0.5 + 0.5 * envelope)
    base = base + 1e-6
    return base / base.sum()

                a = a / a.sum()
                obj = compute_obj(a)
                if obj < best_score:
                    best_score = obj
                    best_seq = a.copy()
                    print(f"  Ramp p={p} n={n}: obj={obj:.10f} **")

                a = a / a.sum()
                obj = compute_obj(a)
                if obj < best_score:
                    best_score = obj
                    best_seq = a.copy()
                    print(f"  Exp alpha={alpha} n={n}: obj={obj:.10f} **")

            a_rev = a_ramp[::-1].copy()
            # Already covered by symmetry
