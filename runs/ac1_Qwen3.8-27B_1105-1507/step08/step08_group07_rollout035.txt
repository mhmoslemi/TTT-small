```python
def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
def anderson_acceleration(X, F, m=5):
    """
    Anderson acceleration.
    X: list of previous iterates (list of arrays)
    F: list of previous F(X) values (list of arrays)
    m: memory size
    """
    k = len(X)
    if k < 2:
        return F[-1]
    
    # Use last m+1 points
    start = max(0, k - m - 1)
    X_sub = X[start:]
    F_sub = F[start:]
    n_pts = len(X_sub)
    
    if n_pts < 2:
        return F[-1]
    
    # Compute residuals
    G = [F_sub[i] - F_sub[i-1] for i in range(1, n_pts)]
    # Actually, for Anderson acceleration, we need:
    # F(x_k) - x_k for each k
    # Let me use the standard formulation
    
    # Standard Anderson acceleration:
    # Given x_0, x_1, ..., x_k and F(x_0), ..., F(x_k)
    # Compute beta coefficients to minimize ||sum beta_i (F(x_i) - x_i)||^2
    # subject to sum beta_i = 1
    # Then x_{k+1} = sum beta_i F(x_i)
    
    # Residuals: r_i = F(x_i) - x_i
    residuals = [F_sub[i] - X_sub[i] for i in range(n_pts)]
    # We want to find beta to minimize ||sum beta_i r_i||^2 s.t. sum beta_i = 1
    
    # Build the system
    # Using the last min(m, n_pts-1) residuals
    n_beta = min(m, n_pts - 1)
    r = residuals[-n_beta:]
    
    # Matrix Gamma[i,j] = <r_i, r_j>
    # Vector gamma[i] = <r_i, r_{n_beta}> (last residual)
    Gamma = np.zeros((n_beta, n_beta))
    gamma = np.zeros(n_beta)
    for i in range(n_beta):
        for j in range(n_beta):
            Gamma[i,j] = np.dot(r[i], r[j])
        gamma[i] = np.dot(r[i], r[n_beta-1])
    
    # Regularize
    Gamma += 1e-10 * np.eye(n_beta)
    
    try:
        beta = np.linalg.solve(Gamma, gamma)
    except:
        beta = np.zeros(n_beta)
        beta[-1] = 1.0
    
    # x_{k+1} = F(x_k) + sum_{i=k-n_beta+1}^{k-1} beta_i (F(x_i) - F(x_k))
    x_new = X_sub[-1].copy()
    # Actually: x_new = F(x_k) + sum beta_i (F(x_i) - F(x_k))
    x_new = F_sub[-1].copy()
    for i in range(n_beta):
        x_new += beta[i] * (F_sub[-1-i-1] - F_sub[-1])
    
    return x_new
```

```python
def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for i in range(n):
        end = min(i + n, m)
        A[i:end, i] = a[:end - i]
    return A


def solve_lp(a):
    """Solve: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M, dual)."""
    n = len(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ g <= M
    c2 = cp.sum(g) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    dual = None
                    try:
                        dual = np.array(c1.dual_value)
                        dual = np.maximum(dual, 0)
                    except:
                        pass
                    return g_val, float(M.value), dual
        except:
            continue
    
    # Fallback to linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)

        return a

    best_a = a.copy()

    history_f = [a.copy()]  # F(x) values
    anderson_m = 5
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        # Forward: fix a, find g
        g1, M1, dual1 = solve_lp(a)
        if g1 is None:
            break
        
        # Backward: fix g1, find a2

        else:
            x_new = g2

        if it >= 1:

def anderson_step(X, F, m):
    """Anderson acceleration step. Returns new point or None."""
    k = len(X)
    if k < 2:
        return None
    
    n_use = min(m + 1, k)
    X_sub = X[-n_use:]
    F_sub = F[-n_use:]
    
    # Compute residuals r_i = F(x_i) - x_i
    residuals = [F_sub[i] - X_sub[i] for i in range(n_use)]
    
    if n_use == 1:
        return F_sub[0]
    
    # Solve for beta: minimize ||sum beta_i r_i||^2 s.t. sum beta_i = 1
    # Using the last n_use - 1 residuals
    n_beta = n_use - 1
    r = residuals[-n_beta:]
    
    # Gamma[i,j] = <r_i, r_j>, gamma[i] = <r_i, r_last>
    Gamma = np.zeros((n_beta, n_beta))
    gamma = np.zeros(n_beta)
    r_last = residuals[-1]
    for i in range(n_beta):
        for j in range(n_beta):
            Gamma[i, j] = np.dot(r[i], r[j])
        gamma[i] = np.dot(r[i], r_last)
    
    Gamma += 1e-10 * np.eye(n_beta)
    
    try:
        beta = np.linalg.solve(Gamma, gamma)
    except:
        return F_sub[-1]
    
    # x_new = F(x_k) + sum_i beta_i * (F(x_i) - F(x_k))
    x_new = F_sub[-1].copy()
    for i in range(n_beta):
        x_new += beta[i] * (F_sub[-1 - i - 1] - F_sub[-1])
    
    return x_new


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
                best_cand_obj = obj_c
                best_cand = c.copy()
        if best_cand_obj < best_obj - 1e-14:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 15:
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                if dual1 is not None and np.max(dual1) > 0:
                    A = build_conv_matrix(best_a)
                    active_idx = np.where(dual1 > 0.01 * np.max(dual1))[0]
                    if len(active_idx) > 0:
                        col_imp = np.sum(A[active_idx, :], axis=0)
                        pert = -np.sign(col_imp - np.mean(col_imp)) * 0.03
                        pert += rng.normal(0, 0.01, n)
                    else:
                        pert = rng.normal(0, 0.03, n)
                else:
                    pert = rng.normal(0, 0.03, n)
                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                history_x = [a.copy()]
                history_f = [a.copy()]
                momentum = a.copy()
                stall = 0
                if verbose:
                    print(f"    [Perturbed] obj={best_obj:.8f}")
            else:
                a = best_a
        history_x.append(a.copy())
        history_f.append(g2.copy())
        if len(history_x) > anderson_m + 3:
            history_x = history_x[-(anderson_m + 3):]
            history_f = history_f[-(anderson_m + 3):]
        it += 1
        if verbose and it % 10 == 0:
            print(f"    Bidir iter {it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
    return best_a


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def logsumexp_polish(a, time_budget=100.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t_temp = 30.0
    lr = 0.5 / (t_temp * np.max(a) + 1e-8)
    x = a.copy()
    y = a.copy()
    x_prev = a.copy()
    fista_t = 1.0
    it = 0
    while time.monotonic() - start < time_budget:
        conv_y = compute_conv(y)
        tc = t_temp * conv_y
        tc_max = np.max(tc)
        w = np.exp(tc - tc_max)
        w_sum = np.sum(w)
        w = w / w_sum
        idx = np.add.outer(np.arange(n), np.arange(n))
        W_mat = w[idx]
        grad = 2.0 * (W_mat @ y)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        y_new = x + ((fista_t - 1.0) / (fista_t + 2.0)) * (x - x_prev)
        step = lr * grad
        step_norm = np.linalg.norm(step)
        if step_norm > 0.1:
            step = step * 0.1 / step_norm
        x_new = project_simplex(y_new - step)
        fista_t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * fista_t**2))
        x_prev = x
        x = x_new
        y = y_new
        obj = compute_obj(x)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = x.copy()
        it += 1
        if it % 500 == 0:
            t_temp *= 2.0
            lr = 0.5 / (t_temp * np.max(x) + 1e-8)
            fista_t = 1.0
            x_prev = x.copy()
            y = x.copy()
        if verbose and it % 200 == 0:
            print(f"    LSE iter {it}: obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def peak_targeted_refine(a, time_budget=60.0):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        conv_max = np.max(conv)
        peaks = np.where(conv > 0.95 * conv_max)[0]
        a_try = a.copy()
        n_perturb = min(len(peaks), 10)
        for _ in range(50):
            p = peaks[rng.integers(0, len(peaks))]
            if p < n:
                idx_range = range(max(0, p - n + 1), min(n, p + 1))
            else:
                idx_range = range(max(0, p - n + 1), min(n, p + 1))
            if len(idx_range) == 0:
                continue
            i = list(idx_range)[rng.integers(0, len(idx_range))]
            for delta in [0.01, -0.01, 0.02, -0.02]:
                a_try2 = a_try.copy()
                a_try2[i] = max(0, a_try2[i] + delta)
                s2 = np.sum(a_try2)
                if s2 < 1e-10:
                    continue
                a_try2 = a_try2 / s2
                obj_try = compute_obj(a_try2)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try2.copy()
                    a = a_try2
                    break
        if time.monotonic() - start > time_budget:
            break
    return best_a


def propose_candidate(seed=42, budget_s=1500):

            s = float(np.sum(parent_arr))

                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")

    a = best_seq.copy()

    return result
```
