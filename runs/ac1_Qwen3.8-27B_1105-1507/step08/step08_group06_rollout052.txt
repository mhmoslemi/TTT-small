import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] ** 2])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv / sum^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp_for_response(a):
    """
    Solve LP: min M s.t. conv(a, g) <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                return g_val, M_val
    except Exception:
        pass

    return None, None


def smooth_grad(a, T):
    """
    Compute gradient of logsumexp((a*a)/T) with respect to a using FFT.
    grad[i] = sum_k weights[k] * a[k-i] / T  (cross-correlation of weights and a)
    """
    n = len(a)
    conv = compute_conv(a)
    shifted = conv / T
    max_shifted = float(np.max(shifted))
    weights = np.exp(shifted - max_shifted)
    wsum = float(np.sum(weights))
    if wsum < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
        wsum = 1.0
    weights /= wsum

    # grad[i] = sum_j weights[i+j] * a[j] for j=0..n-1, (i+j < len(weights))
    # This is cross-correlation: (weights * a)[i] in the correlation sense
    # Use FFT: cross_corr = ifft(conj(fft(weights)) * fft(a))  -- need to be careful
    # Actually: grad[i] = sum_{j=0}^{n-1} weights[i+j] * a[j]

    # Define w_ext of length 2n-1, a_ext of length n (padded to 2n-1)
    # grad = ifft(fft(w_ext, size) * conj(fft(a_ext, size))) -- NO this gives correlation
    # Cross-correlation (f ⋆ g)[i] = sum_j f[i+j] * conj(g[j])
    # In FFT: (f ⋆ g) = ifft(conj(fft(f, size)) * fft(g, size))
    # Wait, let me be more careful.
    # (f ⋆ g)[i] = sum_j f[i+j] * conj(g[j])
    # FFT convention: F = fft(f), G = fft(g)
    # ifft(conj(F) * G) gives what?
    # Let's use: cross_corr via fft
    # (f ⋆ g)[i] = sum_j f[i+j] g[j] (real case)
    # = sum_j conj(f[i+j]) g[j]  (since real)
    # This is the same as: ifft(fft(f) * conj(fft(g))) with appropriate shifting
    # Actually the standard result is:
    # circular cross-correlation: ifft(conj(fft(f)) * fft(g))
    # For linear cross-correlation, we pad both to length >= len(f)+len(g)-1

    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1

    F = rfft(weights, size)
    G = rfft(a, size)
    # Cross-correlation: (f ⋆ g)[i] = sum_j f[i+j] * g[j]
    # Using the property: ifft(conj(FFT(f)) * FFT(g)) gives sum_j f[j-i] * g[j]
    # Hmm, let me just use direct computation for clarity and correctness
    # For n=800, direct is O(n^2) = 640000 which is fine

    # Direct computation (faster than I thought for n=800)
    grad = np.zeros(n)
    for i in range(n):
        j_start = 0
        j_end = n
        k_start = i
        k_end = min(i + n, len(weights))

        valid_j = np.arange(0, min(n, len(weights) - i))
        if len(valid_j) > 0:
            grad[i] = np.dot(weights[i:i + len(valid_j)], a[valid_j])
    return grad / T


def smooth_obj_value(a, T):
    """Compute logsumexp((a*a)/T) value."""
    conv = compute_conv(a)
    shifted = conv / T
    max_shifted = float(np.max(shifted))
    return T * (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted))))


def logsumexp_optimize(a, time_budget=100.0, T_start=1.0, T_end=0.01,
                       n_temps=8, verbose=False):
    """
    Multi-scale log-sum-exp optimization with Nesterov momentum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_temps)
    lr = 0.1
    beta = 0.9  # momentum
    y = a.copy()  # Nesterov lookahead
    x = a.copy()  # current point
    momentum = np.zeros(n)

    for T_idx, T in enumerate(temps):
        T_budget = (T_end / T_start) ** (T_idx / n_temps) * time_budget
        # For each temperature, spend proportional time
        it_budget = max(20, int(time_budget / n_temps * 3))

        for it in range(it_budget):
            if time.monotonic() - start > time_budget:
                return best_a

            # Compute gradient at Nesterov point
            grad = smooth_grad(y, T)

            # Project gradient to sum=0 hyperplane
            grad_centered = grad - np.mean(grad)

            # Momentum update
            momentum = beta * momentum - lr * grad_centered
            x_new = x + momentum
            x_new = np.maximum(x_new, 0)
            sx = float(np.sum(x_new))
            if sx < 1e-10:
                x_new = x.copy()
                sx = 1.0
            x_new = x_new / sx

            # Nesterov update
            t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
            y = x_new + (1 - t_k) * (x_new - x)
            y = np.maximum(y, 0)
            sy = float(np.sum(y))
            if sy > 1e-10:
                y = y / sy

            x = x_new

            # Check objective
            obj = compute_obj(x)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = x.copy()
                if verbose and (it % 50 == 0):
                    print(f"    LSE T={T:.4f}: obj={best_obj:.8f}, it={it}")

            # Adaptive lr
            if it % 100 == 99:
                lr = max(lr * 0.8, 0.005)

    return best_a

def subgradient_polish(a, time_budget=100.0, verbose=False):
    """
    Subgradient method targeting active peaks with Nesterov acceleration.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()  # Nesterov lookahead
    x = a.copy()
    beta_mom = 0.7
    lr = 0.03
    momentum = np.zeros(n)

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(y)
        max_val = float(np.max(conv))
        # Active peaks: within 5% of max
        threshold = max_val * 0.95
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 40:
            active = np.argsort(conv)[::-1][:40]

        # Subgradient: average of gradients of active peaks
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += a[k - indices]
        grad /= len(active)

        # Center (project to sum=0)
        grad_centered = grad - np.mean(grad)

        # Momentum step
        momentum = beta_mom * momentum - lr * grad_centered
        x_new = x + momentum
        x_new = np.maximum(x_new, 0)
        sx = float(np.sum(x_new))
        if sx < 1e-10:
            x_new = x.copy()
            sx = 1.0
        x_new = x_new / sx

        # Nesterov update
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        y = x_new + (1 - t_k) * (x_new - x)
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy > 1e-10:
            y = y / sy

        x = x_new
        obj = compute_obj(x)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = x.copy()
            if verbose and (it % 200 == 0):
                print(f"    Subgrad: obj={best_obj:.8f}, it={it}")

        if it % 300 == 299:

def coordinate_descent(a, time_budget=60.0):
    """
    Coordinate descent: adjust pairs of coordinates to reduce the peak convolution value.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    rng = np.random.default_rng(12345)
    for iteration in range(1000):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        improved = False

        # Compute contributions to the peak
        contributions = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                contributions[j] = a[j] * a[kj]

        # Try moving mass from high-contributing positions
        sorted_idx = np.argsort(contributions)[::-1]

        for frac in [0.003, 0.007, 0.015, 0.03]:
            done = False
            for i in sorted_idx[:15]:
                if done or time.monotonic() - start > time_budget:
                    break
                # Target: position far from the peak contribution
                target = (k_star - int(i) + n // 2) % n
                if target == i:
                    target = (i + n // 2) % n
                delta = frac * a[int(i)]
                if delta <= 0 or delta >= a[int(i)]:
                    continue
                a[int(i)] -= delta
                a[int(target)] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    done = True
                else:
                    a[int(i)] += delta
                    a[int(target)] -= delta
            if done:
                improved = True

        if not improved:
            # Random perturbation
            for _ in range(3):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta = 0.005 * min(a[i], a[j])
                if delta <= 0:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                a = a / s
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and iteration > 200:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_diverse_init(n, kind, rng):
    """Generate diverse initialization."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n, dtype=np.float64)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'bilinear':
        a = np.linspace(0.5, 1.5, n)
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 8)
        a[:edge] = 0.4
        a[-edge:] = 0.4
        return a / np.sum(a)
    elif kind == 'two_peaks':
        a = np.ones(n) * 0.3
        a[n // 4] = 1.5
        a[3 * n // 4] = 1.5
        kernel = np.ones(3) / 3
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'arch':
        x = np.linspace(-1, 1, n)
        a = 1.0 - 0.3 * x ** 2
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining multiple optimization strategies.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = generate_diverse_init(800, 'cosine', rng)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Quick diverse initialization check (30s) ===
    print(f"\n=== Phase 1: Diverse Inits (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    init_budget = min(30.0, remaining * 0.03)

    init_kinds = ['cosine', 'sine_taper', 'gaussian', 'random', 'plateau',
                  'bilinear', 'two_peaks', 'arch']
    for kind in init_kinds:
        remaining = deadline - time.monotonic()
        if remaining < 10 or time.monotonic() - start_time > init_budget + 5:
            break
        a_init = generate_diverse_init(n_parent, kind, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init '{kind}': obj={obj_init:.8f} **")

    print(f"  Best after Phase 1: obj={best_score:.8f}")

    # === Phase 2: Log-sum-exp smooth optimization (250s) ===
    print(f"\n=== Phase 2: LogSumExp Optimization ===")
    remaining = deadline - time.monotonic()
    lse_budget = min(250.0, remaining * 0.2)

    a_lse = logsumexp_optimize(best_seq.copy(), time_budget=lse_budget,
                                T_start=2.0, T_end=0.02, n_temps=6, verbose=True)
    obj_lse = compute_obj(a_lse)
    if obj_lse < best_score:
        best_score = obj_lse
        best_seq = a_lse.copy()
        print(f"  LSE: obj={best_score:.8f} **")
    else:
        print(f"  LSE: obj={obj_lse:.8f}")

    # Also try LSE from a different starting point
    remaining = deadline - time.monotonic()
    if remaining > 50:
        a_init2 = generate_diverse_init(n_parent, 'random', rng)
        a_lse2 = logsumexp_optimize(a_init2, time_budget=min(80.0, remaining * 0.1),
                                     T_start=2.0, T_end=0.02, n_temps=5)
        obj_lse2 = compute_obj(a_lse2)
        if obj_lse2 < best_score:
            best_score = obj_lse2
            best_seq = a_lse2.copy()
            print(f"  LSE random: obj={best_score:.8f} **")

    # === Phase 3: LP Alternating - Additive (350s) ===
    print(f"\n=== Phase 3: LP Alternating (Additive) ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(350.0, remaining * 0.3)

    a_lp1 = lp_alternate_improved(best_seq.copy(), time_budget=lp_budget,
                                   verbose=True, use_multiplicative=False,
                                   momentum_beta=0.4)
    obj_lp1 = compute_obj(a_lp1)
    if obj_lp1 < best_score:
        best_score = obj_lp1
        best_seq = a_lp1.copy()
        print(f"  LP-add: obj={best_score:.8f} **")
    else:
        print(f"  LP-add: obj={obj_lp1:.8f}")

    # === Phase 4: LP Alternating - Multiplicative (200s) ===
    print(f"\n=== Phase 4: LP Alternating (Multiplicative) ===")
    remaining = deadline - time.monotonic()
    lp_budget2 = min(200.0, remaining * 0.2)

    a_lp2 = lp_alternate_improved(best_seq.copy(), time_budget=lp_budget2,
                                   verbose=True, use_multiplicative=True,
                                   momentum_beta=0.5)
    obj_lp2 = compute_obj(a_lp2)
    if obj_lp2 < best_score:
        best_score = obj_lp2
        best_seq = a_lp2.copy()
        print(f"  LP-mult: obj={best_score:.8f} **")
    else:
        print(f"  LP-mult: obj={obj_lp2:.8f}")

    # === Phase 5: Subgradient polish (100s) ===
    print(f"\n=== Phase 5: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        sg_budget = min(100.0, remaining * 0.1)
        a_sg = subgradient_polish(best_seq.copy(), time_budget=sg_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 6: Coordinate descent (80s) ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        cd_budget = min(80.0, remaining * 0.08)
        a_cd = coordinate_descent(best_seq.copy(), time_budget=cd_budget)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CoordDesc: obj={best_score:.8f} **")
        else:
            print(f"  CoordDesc: obj={obj_cd:.8f}")

    # === Phase 7: Length sweep (250s) ===
    print(f"\n=== Phase 7: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(100, n_current - 200),
        max(100, n_current - 100),
        max(100, n_current - 50),
        n_current,
        n_current + 50,
        n_current + 100,
        n_current + 200,
        n_current + 400,
    ]))
    n_options = [n for n in n_options if 100 <= n <= 4000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_current:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization: short LP + subgradient
        alloc_lp = min(25.0, remaining * 0.3)
        alloc_sg = min(15.0, remaining * 0.15)

        a_opt = lp_alternate_improved(a_try, time_budget=alloc_lp, momentum_beta=0.5)
        a_opt = subgradient_polish(a_opt, time_budget=alloc_sg)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final LP rounds from best (remaining) ===
    print(f"\n=== Phase 8: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    a_final = best_seq.copy()
    n_final = len(a_final)
    round_count = 0

    while remaining > 25 and round_count < 8:
        alloc = min(remaining * 0.35, 60.0)
        if alloc < 10:
            break

        # Alternate between additive and multiplicative
        use_mult = (round_count % 2 == 1)
        a_new = lp_alternate_improved(a_final, time_budget=alloc,
                                       use_multiplicative=use_mult,
                                       momentum_beta=0.3 + 0.1 * (round_count % 3))
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_final = a_new.copy()
            print(f"  Final LP {round_count} (mult={use_mult}): obj={best_score:.8f} **")
        else:
            # Perturb and try
            scale = 0.01 + 0.005 * round_count
            a_final = best_seq * np.exp(rng.normal(0, scale, n_final))
            a_final = np.maximum(a_final, 0)
            a_final = a_final / np.sum(a_final)
            print(f"  Final LP {round_count} (mult={use_mult}): obj={obj:.8f} (no improvement)")

        round_count += 1
        remaining = deadline - time.monotonic()

    # === Phase 9: Final subgradient + coordinate polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_sg = subgradient_polish(best_seq.copy(), time_budget=min(remaining * 0.4, 30.0))
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final Subgrad: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_cd = coordinate_descent(best_seq.copy(), time_budget=min(remaining * 0.5, 20.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CoordDesc: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
