This is a system of quadratic inequalities, which is non-convex in general. BUT, if we relax it to a linear constraint (which is what the LP does by fixing one argument of the convolution), we get a tractable problem.

The LP relaxation: for fixed a, find g that minimizes max_k conv(a,g)[k]. This is valid because at the optimum of the full problem, by symmetry, the "best" g to pair with a should satisfy conv(a,g) ≈ conv(a,a).

                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if verbose and it % 25 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it{it}: best={best_obj:.8f}, curr={compute_obj(a):.8f}, "
                  f"t_ls={t_ls:.3f}, M={M_lp:.6f}, t={elapsed:.0f}s")

        # Escape strategy
        if stall > 15:
            # Geometric perturbation of best
            scale = 0.003 + 0.005 * min(stall / 50.0, 1.0)
            a = np.maximum(best_a * np.exp(rng.normal(0, scale, n)), 0)
            st = np.sum(a)
            if st > 1e-10:
                a /= st
            stall = 0
        elif stall > 5 and rng.random() < 0.1:
            # Small random kick
            a = np.maximum(a + 0.002 * rng.normal(0, 1, n), 0)
            st = np.sum(a)
            if st > 1e-10:
                a /= st
            stall = 2

        # Periodic re-anchor to best
        if stall > 40:
            a = best_a.copy()
            stall = 0

    return best_a


def coordinate_descent(a, time_budget, delta=1e-4, verbose=False):
    """Adaptive coordinate descent polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a /= s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    delta_cur = delta
    passes = 0
    no_improve_passes = 0

    while time.monotonic() - start < time_budget:
        passes += 1
        improved = False
        rng_p = np.random.default_rng(passes * 7 + n)
        order = rng_p.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            # Try increase
            for d in [delta_cur, delta_cur * 0.3, delta_cur * 3.0]:
                a_try = a.copy()
                a_try[i] += d
                st = np.sum(a_try)
                if st < 1e-10:
                    continue
                a_try = np.maximum(a_try, 0) / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if improved:
                continue

            # Try decrease
            if a[i] > delta_cur * 0.5:
                for d in [delta_cur, delta_cur * 0.3]:
                    a_try = a.copy()
                    a_try[i] -= d
                    a_try = np.maximum(a_try, 0)
                    st = np.sum(a_try)
                    if st < 1e-10:
                        continue
                    a_try /= st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if verbose and passes % 10 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.8f}, delta={delta_cur:.2e}, "
                  f"time={time.monotonic()-start:.0f}s")

        if not improved:
            no_improve_passes += 1
            delta_cur *= 0.6
            if delta_cur < 1e-9 or no_improve_passes > 8:
                break
        else:
            no_improve_passes = 0
            # Slowly grow delta if improving well
            if improved and delta_cur < delta * 0.5:
                delta_cur *= 1.2

    return best_a


def smooth_gradient_descent(a, time_budget, T_init=0.0001, T_min=0.00001):
    """Gradient descent on smooth max approximation with momentum."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a /= np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.002
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    T = T_init
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        remaining_t = time.monotonic() - start
        T = max(T_min, T_init * (0.997 ** it))

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = np.clip((conv - max_c) / T, -400, 400)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total

        # Gradient contribution: d/d a[i] of sum_k w_k * conv[a*a]_k
        # conv[a*a]_k = sum_j a[j]*a[k-j]
        # d/da[i] = sum_k w_k * (a[k-i] + a[i-k]) (where valid)
        # = sum_k w_k * a[k-i] + sum_k w_k * a[i-k]
        # First term: (w * a)[i] at valid indices (cross correlation w with a)
        # Actually: d/da[i] sum_k w_k sum_j a[j]a[k-j]
        # = sum_k w_k (a[k-i] + a[i-k]) for valid indices
        # = sum_j w[i+j] a[j] + sum_j w[i-j] a[j]  (with valid index constraints)
        # For full convolution indexed 0..2n-2:
        # d/da[i] = sum_{k: 0<=k-i<n, 0<=i-k<2n-1} ... 
        # Simpler: use the matrix form. grad[i] = (A.w)[i] where A is conv matrix of a
        # Actually: conv(a,a) = A_a @ a where A_a is the conv matrix.
        # d/da[i] of w^T A_a a = w^T (dA_a/da[i]) a + w^T A_a e_i
        # dA_a/da[i] is the matrix with A[k,i] perturbed... this is getting complex.
        # Use numerical: grad[i] ≈ sum_k w[k] * (a[|k-i|] if valid)
        # For simplicity, use the approximation:
        # grad[i] = sum_j w[i+j] * a[j] (valid j) + sum_j w[abs(i-j)] * a[j] (valid)
        # Even simpler: just use correlate
        grad = np.zeros(n)
        # conv(a,a)[k] derivative w.r.t a[i]:
        # = sum over valid j of: a[k-j] if j==i, else a[j] if k-j==i
        # = a[k-i] (when 0<=k-i<n) + a[i-k] (when 0<=i-k<n, i.e., k<=i and i-k<n)
        # Summing over k with weight w[k]:
        # = sum_{k: 0<=k-i<n} w[k] * a[k-i] + sum_{k: 0<=i-k<n} w[k] * a[i-k]
        # First: let j=k-i, j from max(0,-i) to min(n-1, 2n-2-i), but k=j+i must be in [0,2n-2]
        #   j from max(0,-i) to min(n-1, 2n-2-i)
        #   For 0<=i<n: j from 0 to min(n-1, 2n-2-i) = n-1 (since 2n-2-i >= n-1 for i<=n-1)
        #   So: sum_{j=0}^{n-1} w[j+i] * a[j] (where j+i < 2n-1)
        # Second: let j=i-k, j from max(0, i-(2n-2)) to min(i, n-1), but we need k=i-j in [0,2n-2]
        #   For 0<=i<n: j from 0 to min(i, n-1) = i (since i<n-1 typically)
        #   Actually k = i-j >= 0 means j <= i, and k <= 2n-2 always for j>=0, i<n
        #   Also a[i-k] = a[j], and we need 0<=j<n
        #   So: sum_{j=0}^{min(i, n-1)} w[i-j] * a[j]
        #   But j goes from 0 to min(i, n-1)

        # For efficiency, use FFT-based approach:
        # grad = w * a (elementwise, valid range) + flipped correlation
        # Actually let me just do it directly for n<=2000 (O(n^2) is fine)
        # n is at most ~2000, so O(n^2) = 4M which is fine

        # First sum: sum_j w[i+j] * a[j] for 0<=j<n and 0<=i+j<2n-1
        # Second sum: sum_j w[i-j] * a[j] for 0<=j<n and 0<=i-j<2n-1 (i.e., j<=i)
        
        # Use np.correlate for efficiency
        # First: (w padded) correlated with a
        # w has length 2n-1, a has length n
        # sum_j w[i+j]*a[j] for j=0..n-1 where i+j valid in w
        # = correlate(w, a)[i] in 'valid' mode? Let's check.
        # np.correlate(w, a, 'valid') gives out[k] = sum_j w[k+j]*a[n-1-j] ... not quite.
        
        # Let me just do it with the convolution matrix approach:
        A = build_conv_matrix(a)
        grad = A.T @ weights  # This gives d(conv(a,a))/da for linear case
        # But conv is quadratic, so we need: grad[i] = (A.T @ w)[i] where A depends on a
        # Actually d/da[i] (a^T Q_k a) ... 
        # For our purpose, the gradient of max_k conv_k(a,a) w.r.t a[i] is:
        # sum_k w_k * d(conv_k)/da[i] where conv_k = sum_j a[j]*a[k-j]
        # d(conv_k)/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        # So grad[i] = sum_k w[k]*(a[k-i]+a[i-k]) = 2 * (appearing in A^T @ w with the right a)
        # Actually A[k,i] = a[k-i] (if valid). So (A^T @ w)[i] = sum_k A[k,i]*w[k] = sum_k a[k-i]*w[k]
        # This is only the first term. The second term by symmetry gives the same.
        # So grad = 2 * (A.T @ w) ... but only if A is symmetric in the right way.
        # A[k,i] = a[k-i], and A[i,k] would be a[i-k] which is not the same entry.
        # Actually A is not symmetric. A[k,j] = a[k-j].
        # (A^T w)[i] = sum_k A[k,i] w[k] = sum_k a[k-i] w[k] (where 0<=k-i<n, i.e., i<=k<=i+n-1)
        # The second term: sum_k a[i-k] w[k] (where 0<=i-k<n, i.e., i-n+1<=k<=i)
        # = sum_j a[j] w[i-j] (substituting j=i-k)
        # = (w_reversed * a type correlation)
        
        # For now, use the approximation grad ≈ 2 * A.T @ w which is correct when 
        # a is symmetric. For non-symmetric a, it's an approximation but works in practice.
        grad = 2.0 * (A.T @ weights)
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered

        # Project onto simplex
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new /= s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if stall == 0 and it % 50 == 0 and lr < 0.01:
                lr *= 1.05
        else:
            stall += 1
            if stall > 30:
                lr *= 0.7
                if lr < 1e-7:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternating + n-exploration + CD polish + smooth gradient.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent n={len(parent)}: obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default n=800: obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Intensive LP Alternating ===
    print("\n=== Phase 1: LP Alternating ===")
    p1_budget = min(450.0, max(100.0, (deadline - time.monotonic()) * 0.35))
    a1 = lp_alternate(best_seq, p1_budget, verbose=True, seed=42)
    obj1 = compute_obj(a1)
    if obj1 < best_score:
        best_score = obj1
        best_seq = a1.copy()
    print(f"Phase 1: obj={best_score:.8f}, time={time.monotonic()-start_time:.0f}s")

    # === Phase 2: Second LP pass with different seed (escape) ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print("\n=== Phase 2: LP Re-optimization (different seed) ===")
        p2_budget = min(200.0, remaining * 0.3)
        # Perturb slightly to explore different basin
        rng_esc = np.random.default_rng(123)
        a_pert = np.maximum(best_seq * np.exp(0.002 * rng_esc.normal(0, 1, len(best_seq))), 0)
        a_pert /= np.sum(a_pert)
        a2 = lp_alternate(a_pert, p2_budget, verbose=True, seed=123)
        obj2 = compute_obj(a2)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  ** Improved: obj={best_score:.8f} **")
        else:
            print(f"  No improvement: obj2={obj2:.8f}")

    # === Phase 3: n Exploration ===
    print("\n=== Phase 3: n Exploration ===")
    n_base = len(best_seq)
    n_options = sorted(set([
        n_base - 200, n_base - 150, n_base - 100, n_base - 50,
        n_base + 50, n_base + 100, n_base + 150, n_base + 200, n_base + 300,
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1300, 1400, 1500, 1700
    ]))
    n_options = [n for n in n_options if 400 <= n <= 2000 and n != n_base]
    # Sort by proximity to current best n
    n_options.sort(key=lambda x: abs(x - n_base))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break
        a_try = resample_linear(best_seq, n_try)
        alloc = min(remaining * 0.06, 30.0)
        a_opt = lp_alternate(a_try, alloc, verbose=False, seed=n_try)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score - 1e-10:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Re-optimize at best n ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 4: Re-optimization at best n ===")
        p4_budget = min(200.0, remaining * 0.35)
        a4 = lp_alternate(best_seq, p4_budget, verbose=True, seed=777)
        obj4 = compute_obj(a4)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
        print(f"Phase 4: obj={best_score:.8f}, time={time.monotonic()-start_time:.0f}s")

    # === Phase 5: Coordinate Descent Polish ===
    print("\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p5_budget = min(remaining * 0.3, 150.0)
        a5 = coordinate_descent(best_seq, p5_budget, delta=1e-4, verbose=True)
        obj5 = compute_obj(a5)
        if obj5 < best_score:
            best_score = obj5
            best_seq = a5.copy()
        print(f"Phase 5: obj={best_score:.8f}")

    # === Phase 6: Smooth Gradient ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 6: Smooth Gradient ===")
        p6_budget = min(remaining * 0.3, 80.0)
        a6 = smooth_gradient_descent(best_seq, p6_budget, T_init=0.00008, T_min=0.000005)
        obj6 = compute_obj(a6)
        if obj6 < best_score:
            best_score = obj6
            best_seq = a6.copy()
        print(f"Phase 6: obj={best_score:.8f}")

    # === Phase 7: Final fine CD ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 7: Final CD ===")
        p7_budget = min(remaining * 0.5, 60.0)
        a7 = coordinate_descent(best_seq, p7_budget, delta=5e-5)
        obj7 = compute_obj(a7)
        if obj7 < best_score:
            best_score = obj7
            best_seq = a7.copy()
        print(f"Phase 7: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    return result
