            if a_new[i] < 0 or a_new[j] < 0:
                a_new[i] = a[i]
                a_new[j] = a[j]
                continue
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                a_new[i] = a[i]
                a_new[j] = a[j]
                continue
            a_new = a_new / s_new
            conv_new = compute_conv(a_new)
            obj_new = 2.0 * n * float(np.max(conv_new))
            if obj_new < best_obj_before:
                a[i] = a_new[i]
                a[j] = a_new[j]
                # Redo the rest from current state
                improved = True
                a_new = a.copy()
                break
        if improved:
            break
    
    if improved:
        return a, True
    return a, False


def lp_alternating_improved(a, time_budget=300.0, verbose=False, max_active=40, frac_thresh=0.97):
    """LP alternating with active set management and deflation escape."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        active = get_active_set(conv, num_max=max_active, frac=frac_thresh)
        
        g_lp = solve_lp_response(a, active)
        if g_lp is None:
            break

        best_t_obj = np.inf
        best_t = 0.3
        best_a_local = best_a.copy()

        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            stall = 0
            if verbose and stall % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.2f}, active={len(active)}, time={elapsed:.0f}s")
        else:
            stall += 1
            # Try deflation
            if stall >= 3:
                conv_cur = compute_conv(a)
                a_def, improved = deflation_step(a, conv_cur, num_top=6, num_bottom=6, frac=0.003)
                if improved:
                    obj_def = compute_obj(a_def)
                    if obj_def < best_obj:
                        best_obj = obj_def
                        best_a = a_def.copy()
                    a = a_def
                    stall = 0
                    continue
            
            if stall % 8 == 0:
                # Structured perturbation
                scale = 0.01 + 0.005 * min(stall // 8, 5)
                noise = rng.normal(0, scale, n)
                a = best_a * np.exp(noise)
                a = np.maximum(a, 0)
                s2 = float(np.sum(a))
                if s2 > 1e-10:
                    a = a / s2
                stall = 0

    return best_a


def subgradient_polish(a, time_budget=100.0, verbose=False):
    """Subgradient method targeting active peaks with momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    momentum = np.zeros(n)
    beta = 0.7
    lr = 0.03

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        x = y + (1.0 - 2.0 / (np.sqrt(it + 1) + 2.0)) * (y - y)  # x = y initially
        x = y.copy()
        conv = compute_conv(x)
        max_val = float(np.max(conv))
        threshold = max_val * 0.92
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 25:
            active = np.argsort(conv)[::-1][:25]

        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += a[k - indices]
        grad /= len(active)
        grad_centered = grad - np.mean(grad)

        momentum = beta * momentum - lr * grad_centered
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new

        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        y = y_new

        obj = compute_obj(y)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()

        if it % 500 == 499:
            lr = max(lr * 0.92, 0.002)

    return best_a


def coordinate_adjust(a, time_budget=60.0):
    """Pairwise coordinate adjustment targeting the peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    rng = np.random.default_rng(999)

    for iteration in range(2000):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        improved = False

        contributions = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contributions[j] = a[j] * a[k_j]

        sorted_idx = np.argsort(contributions)[::-1]

        for frac in [0.002, 0.005, 0.01, 0.02]:
            done = False
            for i in sorted_idx[:15]:
                if done or time.monotonic() - start > time_budget:
                    break
                target = (k_star - int(i) + n // 2) % n
                if target == i:
                    target = (i + n // 3) % n
                delta = frac * a[i]
                if delta <= 0 or delta > a[i] * 0.9:
                    continue
                a[i] -= delta
                a[target] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv))
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    improved = True
                    done = True
                else:
                    a[i] += delta
                    a[target] -= delta
            if done:
                improved = True

        if not improved:
            for _ in range(3):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                delta = 0.005 * min(a[i], a[j], 0.1)
                if delta <= 0:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s2 = float(np.sum(a))
                if s2 < 1e-10:
                    break
                a = a / s2
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and iteration > 200:
            break

    return best_a


def generate_initialization(kind, n, rng):
    """Generate a diverse initialization of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 12)
        a[:edge] = 0.4
        a[-edge:] = 0.4
        return a / np.sum(a)
    elif kind == 'bimodal':
        a = np.ones(n) * 0.5
        center1 = n // 4
        center2 = 3 * n // 4
        sigma = n // 10
        for i in range(n):
            a[i] += np.exp(-0.5 * ((i - center1) / sigma) ** 2)
            a[i] += np.exp(-0.5 * ((i - center2) / sigma) ** 2)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:

                obj = compute_obj(parent)

                best_seq = parent.copy()

    init_kinds = ['cosine', 'sine_taper', 'gaussian', 'plateau', 'bimodal']

    # === Phase 3: Length Sweep (500s) ===
    print(f"\n=== Phase 3: Length Sweep ===")
