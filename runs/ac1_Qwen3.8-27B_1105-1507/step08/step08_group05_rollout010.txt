import numpy as np

def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_gradient(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a: grad[m] = 2*a[k-m] if valid."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        m_start = k - i_end
        m_end = k - i_start
        grad[m_start:m_end + 1] = 2.0 * a[i_start:i_end + 1][::-1]
    return grad


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def build_conv_matrix_subset(a, k_indices):
    """Build (|k_indices|) x n matrix for conv(a, g) at selected positions."""
    n = len(a)
    p = len(k_indices)
    A = np.zeros((p, n))
    for idx, k in enumerate(k_indices):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            m = k - i
            A[idx, m] = a[i]
    return A


def solve_lp_active(a, top_p=50, max_M=None):
    """
    LP: min M s.t. conv(a,g)[k] <= M for k in top_p peaks, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    
    # Find top peaks in conv(a, a) to use as active constraints
    conv_aa = compute_conv(a_n)
    if max_M is not None:
        # Include all constraints within max_M of the peak
        threshold = max_M * 0.9
        active_k = np.where(conv_aa >= threshold)[0]
        if len(active_k) > top_p:
            active_k = np.argsort(conv_aa)[::-1][:top_p]
    else:
        active_k = np.argsort(conv_aa)[::-1][:top_p]
    
    A = build_conv_matrix_subset(a_n, active_k)
    p = len(active_k)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None

def multiplicative_weights(a0, time_budget=60.0, top_k=10, verbose=False):
    """
    Multiplicative weights / exponential descent on the simplex.
    a[m] *= exp(-eta * subgrad[m]), then renormalize.

    a = a0.astype(np.float64).copy()

        return a0.copy()

    start = time.monotonic()
    eta = 0.15
    stall = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:top_k]
        
        # Compute subgradient as weighted average over top peaks
        grad = np.zeros(n)
        weights = np.exp(-0.5 * np.arange(top_k))  # Decay weights
        weights /= np.sum(weights)
        for wi, k in zip(weights, top_idx):
            grad += wi * conv_gradient(a, int(k))
        
        # Center the gradient (remove mean for simplex)
        grad = grad - np.mean(grad)
        
        # Exponential update with clipping for stability
        log_update = -eta * grad
        log_update = np.clip(log_update, -5.0, 5.0)  # Prevent overflow
        a_new = a * np.exp(log_update)
        
        # Renormalize
        s_new = float(np.sum(a_new))
        if s_new < 1e-10 or not np.isfinite(s_new):
            eta *= 0.5
            continue
        a_new = a_new / s_new
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 50 == 0:
                print(f"      MW it={it}: obj={best_obj:.8f}, eta={eta:.4f}, t={time.monotonic()-start:.1f}s")

    return best_a

def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

        else:
            print(f"  Phase 1: obj={obj1:.8f} (no improvement)")

    # === Phase 2: Multiplicative Weights (150s) ===
    print(f"\n=== Phase 2: Multiplicative Weights ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(150.0, remaining * 0.3)
        a2 = multiplicative_weights(best_seq, time_budget=alloc, top_k=10, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_score - 1e-12:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase 2 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 2: obj={obj2:.8f}")

    # === Phase 3: Hybrid LP+MWU (300s) ===
    print(f"\n=== Phase 3: Hybrid LP+MWU ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(300.0, remaining * 0.5)
        a3 = hybrid_lp_mwu(best_seq, time_budget=alloc, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_score - 1e-12:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Phase 3 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3: obj={obj3:.8f}")

    # === Phase 4: Targeted Peak Reduction (150s) ===
    print(f"\n=== Phase 4: Targeted Peak Reduction ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(150.0, remaining * 0.3)
        a4 = targeted_peak_reduction(best_seq, time_budget=alloc, n_peaks=10, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_score - 1e-12:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Phase 4 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4: obj={obj4:.8f}")

    # === Phase 5: LP with more constraints (200s) ===
    print(f"\n=== Phase 5: LP Full (more constraints) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_cur = len(best_seq)
        a = best_seq.copy()
        for round_idx in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(40.0, remaining * 0.4)
            g_lp, M_lp = solve_lp_full(a, top_p=100)
            if g_lp is None:
                g_lp, M_lp = solve_lp_active(a, top_p=60)
            if g_lp is None:
                break
            # Line search
            best_t = 0.3
            best_line = compute_obj(a)
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_try = np.maximum((1.0 - t) * a + t * g_lp, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_line:
                    best_line = obj_try
                    best_t = t
            a_try = np.maximum((1.0 - best_t) * a + best_t * g_lp, 0)
            a_try = a_try / float(np.sum(a_try))
            obj_try = compute_obj(a_try)
            if obj_try < best_score - 1e-12:
                best_score = obj_try
                best_seq = a_try.copy()
                a = best_seq
                print(f"  LP Full R{round_idx}: obj={best_score:.8f} **")
            else:
                print(f"  LP Full R{round_idx}: obj={obj_try:.8f} (no improvement)")
                # Perturb
                a = best_seq * np.exp(rng.normal(0, 0.01, n_cur))
                a = np.maximum(a, 0)
                a = a / float(np.sum(a))

    # === Phase 6: Fine coordinate search (150s) ===
    print(f"\n=== Phase 6: Fine Coordinate Search ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(150.0, remaining * 0.4)
        a6 = fine_coordinate_search(best_seq, time_budget=alloc, verbose=True)
        obj6 = compute_obj(a6)
        if obj6 < best_score - 1e-12:
            best_score = obj6
            best_seq = a6.copy()
            print(f"  Phase 6 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 6: obj={obj6:.8f}")

    # === Phase 7: Final LP polish (100s) ===
    print(f"\n=== Phase 7: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(100.0, remaining * 0.5)
        a7 = lp_alternate(best_seq, time_budget=alloc, top_p=30, verbose=True)
        obj7 = compute_obj(a7)
        if obj7 < best_score - 1e-12:
            best_score = obj7
            best_seq = a7.copy()
            print(f"  Phase 7 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 7: obj={obj7:.8f}")

    # === Phase 8: Try n variations (remaining) ===
    print(f"\n=== Phase 8: n-Sweep ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_cur = len(best_seq)
        n_options = [n_cur - 50, n_cur, n_cur + 50, n_cur + 100, n_cur - 100]
        n_options = [n for n in n_options if 200 <= n <= 2000]
        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_try = resample_seq(best_seq, n_try)
            a_try = a_try / float(np.sum(a_try))
            # Quick LP
            alloc = min(remaining * 0.3, 25.0)
            a_opt = lp_alternate(a_try, time_budget=alloc, top_p=40)
            obj_try = compute_obj(a_opt)
            if obj_try < best_score - 1e-12:
                best_score = obj_try
                best_seq = a_opt.copy()
                print(f"  n={n_try}: obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 9: Final combined polish ===
    print(f"\n=== Phase 9: Final Combined ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_final = hybrid_lp_mwu(best_seq, time_budget=min(remaining * 0.6, 80.0), verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score - 1e-12:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final combined: obj={best_score:.8f} **")
        
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_coord = fine_coordinate_search(best_seq, time_budget=min(remaining * 0.5, 40.0))
            obj_coord = compute_obj(a_coord)
            if obj_coord < best_score - 1e-12:
                best_score = obj_coord
                best_seq = a_coord.copy()
                print(f"  Final coords: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
