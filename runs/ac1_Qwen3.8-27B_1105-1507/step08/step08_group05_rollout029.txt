**Algorithm: Smoothed Gradient Descent with Annealing**

1. Start with initial sequence `a` (normalized to sum=1).
2. Set initial temperature `T = T_max` (e.g., based on the spread of conv values).
3. Repeat:
   a. Compute `conv = convolve(a, a)`.
   b. Compute softmax weights `w = exp(conv/T) / sum(exp(conv/T))`.
   c. Compute gradient `grad = 2 * irfft(conj(rfft(a,N)) * rfft(w,N))[:n]`.
   d. Take a step: `a_new = a - lr * grad`.
   e. Project onto simplex {x >= 0, sum(x) = 1}.
   f. Optionally do line search.
   g. Update temperature: `T *= decay_factor`.
4. Return the best sequence found.

**Projection onto simplex**: The standard algorithm (sort, find threshold).

**Line search**: Backtracking to ensure decrease in the smoothed objective.

**Temperature schedule**: Exponential decay from T_max to T_min over the course of optimization.

Let me also think about what T values to use. The conv values range from 0 to ~max_conv. If we set T = max_conv / 10 initially, the softmax will be relatively flat. As T decreases, the softmax concentrates on the peaks.

Actually, a better schedule might be: start with T such that `T ≈ max_conv / 5` (quite smooth) and end with `T ≈ max_conv / 1000` (very peaked, essentially the true max).

"""
Smoothed gradient descent with log-sum-exp annealing for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses FFT-based gradient computation
and simplex projection. Combines with LP polishing and n-sweeping.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def lse_grad(a, T):
    """
    Gradient of 2*n*log(sum(exp(conv(a,a)/T))) w.r.t. a.
    Assumes a is already normalized (sum(a)=1).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    conv = compute_conv(a)
    # Numerically stable softmax
    max_c = np.max(conv)
    exp_c = np.exp((conv - max_c) / T)
    w = exp_c / np.sum(exp_c)
    # grad[m] = 2 * sum_j a[j] * w[j+m] (with w[k]=0 for k >= 2n-1)
    # = 2 * ifft(conj(FFT(a)) * FFT(w))[m] (with zero-padding)
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    A = rfft(a, N)
    W = rfft(w, N)
    corr = irfft(np.conj(A) * W, N)
    return 2.0 * corr[:n]


def smoothed_obj(a, T):
    """Smoothed objective value (for monitoring)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    lse = max_c / T + np.log(np.sum(np.exp((conv - max_c) / T)))
    return 2.0 * n * lse


def smooth_gradient_descent(a0, time_budget=100.0, T_max=None, T_min=None, 
                             n_steps=None, verbose=False):
    """
    Smoothed gradient descent on the simplex with log-sum-exp annealing.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    
    # Initial temperature
    conv0 = compute_conv(a)
    spread = np.max(conv0) - np.min(conv0) + 1e-10
    if T_max is None:
        T_max = spread / 5.0
    if T_min is None:
        T_min = spread / 1000.0
    if n_steps is None:
        n_steps = int(time_budget * 50)  # rough estimate
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1
    lr_min = 0.001
    patience = 0
    
    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break
        
        # Annealing schedule (geometric)
        frac = it / max(1, n_steps - 1)
        T = T_max * (T_min / T_max) ** frac
        
        # Compute gradient
        grad = lse_grad(a, T)
        
        # Center gradient (remove mean component since we're on simplex)
        grad = grad - np.mean(grad)
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-12:
            grad = grad / grad_norm
        
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        
        # Check improvement
        if it % 5 == 0:
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                patience = 0
            else:
                patience += 1
        
        a = a_new
        
        # Adaptive learning rate
        if patience > 50:
            lr = max(lr * 0.8, lr_min)
            patience = 0
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            obj_cur = compute_obj(a)
            print(f"    SGD it={it}: T={T:.4f}, obj={obj_cur:.8f}, best={best_obj:.8f}, lr={lr:.4f}, t={elapsed:.1f}s")
    
    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_norm, M_opt).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    if n > 2000:
        return None, None
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=60.0, verbose=False):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_good_t = 0.5

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        # 1D line search
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)
        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 30, 'xatol': 0.01})
            best_t = res.x if res.success else last_good_t
        except Exception:
            best_t = last_good_t
        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:

            best_a = a_try.copy()
            a = best_a
            last_good_t = best_t
            stall_count = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 10:
                break

    return best_a


def coordinate_descent(a0, time_budget=60.0, verbose=False):
    """
    Coordinate descent: try adjusting one or two elements at a time.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    no_improve = 0
    while time.monotonic() - start < time_budget:
        # Pick a random pair of indices
        if n > 1:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            # Try moving mass from i to j (and vice versa)
            for direction in [1, -1]:
                for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                    amt = frac * min(a[i], a[j]) if direction == 1 else frac * a[i]
                    if direction == 1:
                        a[i] -= amt
                        a[j] += amt
                    else:
                        a[i] += amt
                        a[j] -= amt
                    if a[i] < 0 or a[j] < 0:
                        a[i] += amt if direction == 1 else -amt
                        a[j] -= amt if direction == 1 else amt
                        continue
                    st = float(np.sum(a))
                    if st < 1e-10:
                        break
                    a = a / st
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-13:
                        best_obj = obj
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        # Revert
                        a = best_a.copy()
                        break
        
        no_improve += 1
        if no_improve > 500:
            break
    
    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw
