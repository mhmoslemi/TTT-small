    while size < 2*n - 1:

    return irfft(fa * fa, size)[:2*n-1]

    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    an = a / s
    conv = compute_conv(an)
    return 2.0 * n * float(np.max(conv))

    """Build matrix where A @ g = conv(a, g)."""
    n = len(a)
    m = 2*n - 1
    A = np.zeros((m, n))
    for i in range(n):
        start = i
        end = min(m, i + n)
        A[start:end, i] = a[:end-start]
    return A

def solve_lp(a):

    n = len(a)
    a = a / np.sum(a)
    A = build_A(a)
    m = 2*n - 1

    c[-1] = 1.0  # minimize M
    
    # A_ub @ x <= b_ub
    # conv(a,g)[k] - M <= 0 for each k
    A_ub = np.zeros((m, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    # A_eq @ x = b_eq
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        s = np.sum(g)
        if s > 1e-10:

from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    an = a / s
    conv = compute_conv(an)
    return 2.0 * n * float(np.max(conv))


def build_A(a):
    n = len(a)
    m = 2*n - 1
    A = np.zeros((m, n))
    for i in range(n):
        end = min(m, i + n)
        A[i:end, i] = a[:end-i]
    return A


def solve_lp(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return None
    a = a / s
    A = build_A(a)
    m = 2*n - 1
    
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g))
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_alternate(a, time_budget=300.0, beta=0.5, verbose=False):
    """LP alternating with momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    stall_count = 0
    
    while time.monotonic() - start < time_budget:
        g_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Try different step sizes with momentum
        direction = g_lp - a
        momentum = beta * momentum + (1.0 - beta) * direction
        
        best_t = 0.5
        best_t_obj = np.inf
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = a + t * momentum
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-15:
            a = a + best_t * momentum
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
            best_obj = best_t_obj
            stall_count = 0
            if verbose and (best_obj < best_obj + 1e-10):
                elapsed = time.monotonic() - start
                if elapsed > 10:
                    print(f"    LP: obj={best_obj:.8f}, t={best_t:.2f}, time={elapsed:.0f}s")
        else:
            stall_count += 1
            momentum *= 0.5
            if stall_count >= 5:
                # Perturb to escape
                rng = np.random.default_rng(int(time.monotonic()*1e6) % (2**31))
                scale = 0.02 + 0.01 * min(stall_count // 5, 5)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall_count = 0
                momentum = np.zeros(n)
    
    return best_a


def exp_gradient(a, time_budget=200.0, verbose=False):
    """Exponential gradient (multiplicative weight) method."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.1
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Active peaks (top fraction)
        threshold = max_val * 0.9
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # Compute gradient: grad[i] = sum_{k in active} a[k-i] / |active|
        # Using FFT for efficiency
        indicator = np.zeros(2*n-1)
        for k in active:
            indicator[k] = 1.0
        
        # grad = cross-correlation of indicator and a
        # grad[i] = sum_k indicator[k] * a[k-i]
        size = 1
        while size < 2*(2*n-1):
            size <<= 1
        I = rfft(indicator, size)
        A = rfft(a, size)
        grad_full = irfft(np.conj(I) * A, size)  # this gives cross-correlation
        # Actually, I need to be careful about the indexing
        # Let me just compute it directly for correctness
        grad = np.zeros(n)
        for i in range(n):
            k_min = i
            k_max = min(2*n-2, i + n - 1)
            if k_max >= k_min:
                ks = np.arange(k_min, k_max+1)
                ais = ks - i
                grad[i] = np.dot(indicator[ks], a[ais])
        grad /= len(active)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new  # still update, but keep best
            eta *= 0.95
        
        # Decay eta
        if it % 100 == 99:
            eta = max(eta * 0.9, 0.005)
    
    return best_a


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

"""
Step function optimization: LP alternating with FFT-based subgradient,
simulated annealing, diverse restarts, and coarse-to-fine refinement.
Minimizes 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    an = a / s
    conv = compute_conv(an)
    return 2.0 * n * float(np.max(conv))


def build_A(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for i in range(n):
        end = min(m, i + n)
        A[i:end, i] = a[:end - i]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return None
    a = a / s
    A = build_A(a)
    m = 2 * n - 1

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g))
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_alternate(a, time_budget=300.0, beta=0.5, verbose=False):
    """LP alternating with momentum and adaptive step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    stall = 0
    last_t = 0.5
    print_interval = 20

    iteration = 0
    while time.monotonic() - start < time_budget:
        iteration += 1
        g_lp = solve_lp(a)
        if g_lp is None:
            break

        direction = g_lp - a
        momentum = beta * momentum + (1.0 - beta) * direction

        best_t = 0.5
        best_t_obj = np.inf
        best_a_try = None

        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = a + t * momentum
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_try = a_try

        if best_a_try is not None and best_t_obj < best_obj - 1e-15:
            a = best_a_try
            best_a = a.copy()
            best_obj = best_t_obj
            stall = 0
            last_t = best_t
            if verbose and iteration % print_interval == 0:
                elapsed = time.monotonic() - start
                print(f"    LP iter {iteration}: obj={best_obj:.8f}, t={best_t:.2f}, time={elapsed:.0f}s")
        else:
            stall += 1
            momentum *= 0.5
            if stall >= 4:
                t_esc = 0.02 + 0.01 * min(stall // 4, 8)
                rng = np.random.default_rng(hash((iteration, stall)) % (2**31))
                a = best_a * np.exp(rng.normal(0, t_esc, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                momentum = np.zeros(n)

    return best_a


def fft_subgradient(a, time_budget=200.0, verbose=False):
    """Nesterov-accelerated subgradient with FFT-based gradient computation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()
    y = a.copy()
    momentum = np.zeros(n)
    beta = 0.8
    lr = 0.05

    # Precompute FFT size
    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(x)
        max_val = float(np.max(conv))
        threshold = max_val * 0.85
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 40:
            active = np.argsort(conv)[::-1][:40]

        # FFT-based gradient computation
        indicator = np.zeros(2 * n - 1)
        for k in active:
            indicator[k] = 1.0

        I = rfft(indicator, size)
        A = rfft(y, size)
        grad_full = irfft(I * np.conj(A), size).real
        grad = grad_full[:n].copy()
        grad /= len(active)

        # Center gradient (project to sum-zero hyperplane)
        grad_centered = grad - np.mean(grad)

        # Momentum + step
        momentum = beta * momentum - lr * grad_centered
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new

        # Nesterov lookahead
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx
        else:
            x = y_new.copy()

        y = y_new
        obj = compute_obj(y)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = y.copy()

        if it % 500 == 499:
            lr = max(lr * 0.9, 0.002)
            if verbose and best_obj < 100:
                elapsed = time.monotonic() - start
                print(f"    SG iter {it}: obj={best_obj:.8f}, lr={lr:.4f}, time={elapsed:.0f}s")

    return best_a


def simulated_annealing(a, time_budget=200.0, verbose=False):
    """Simulated annealing with FFT-based evaluation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = 0.1
    T_min = 0.001
    cool_rate = 0.9995
    current = a.copy()
    current_obj = best_obj

    rng = np.random.default_rng(12345)

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        if T < T_min:
            break

        # Random perturbation: pick a few elements and adjust
        n_perturb = max(1, n // 50)
        indices = rng.choice(n, size=n_perturb, replace=False)
        delta = rng.normal(0, 0.1, n_perturb) * np.abs(current[indices]) + 0.001

        new_a = current.copy()
        for idx, d in zip(indices, delta):
            new_a[idx] = max(0, current[idx] + d)

        s_new = float(np.sum(new_a))
        if s_new < 1e-10:
            continue
        new_a = new_a / s_new
        new_obj = compute_obj(new_a)

        if new_obj < current_obj or rng.random() < np.exp(-(new_obj - current_obj) / T):
            current = new_a
            current_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = new_a.copy()
                if verbose:
                    elapsed = time.monotonic() - start
                    print(f"    SA iter {it}: obj={best_obj:.8f}, T={T:.5f}, time={elapsed:.0f}s")

        T *= cool_rate

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
