import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):

    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def compute_smooth_grad(a, beta):
    """Compute gradient of smoothed objective F_beta(a) = (1/beta) log(sum exp(beta * conv_k))."""
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights (shift to avoid overflow)
    conv_shifted = conv - np.max(conv)
    p = np.exp(beta * conv_shifted)
    p /= np.sum(p)
    # Gradient: 2 * cross_correlation(p, a)
    grad = 2.0 * cross_correlation(p, a)
    return grad, conv, p

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def nesterov_smooth_optimize(a, time_budget=600.0, beta_init=500.0, beta_end=5000.0, 
                               lr_init=0.1, verbose=True):
    """
    Nesterov-accelerated smooth gradient descent on the simplex.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = a.copy()  # Nesterov lookahead
    y = a.copy()  # actual point
    velocity = np.zeros(n)
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Anneal beta
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        beta = beta_init * (beta_end / beta_init) ** progress
        
        # Compute gradient at lookahead point
        x_norm = x / np.sum(x)
        grad, conv, p = compute_smooth_grad(x_norm, beta)
        
        # Project gradient onto tangent space of simplex (remove mean component)
        grad_centered = grad - np.mean(grad)
        
        # Nesterov step
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        
        # Learning rate (adaptive)
        lr = lr_init / (1.0 + 0.01 * it)
        
        # Update
        velocity = 0.9 * velocity - lr * grad_centered
        y_new = y + t_k * velocity / (np.max(np.abs(velocity)) + 1e-10)
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
        else:
            y_new = y_new / s_new
        
        # Nesterov update
        x = y_new + ((t_k + 1) / 2.0 - 1.0) * (y_new - y)
        # Standard Nesterov: x_{k+1} = y_{k+1} + (t_{k+1}-1)/t_{k+1} * (y_{k+1} - y_k)
        t_next = 2.0 / (np.sqrt(it + 2) + 2.0)
        x = y_new + (t_next - 1.0) / t_next * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx
        
        y = y_new
        it += 1
        
        # Evaluate
        if it % 10 == 0:
            obj = compute_obj(y)
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = y.copy()
                if verbose and it % 100 == 0:
                    elapsed = time.monotonic() - start
                    print(f"    Nesterov: it={it}, obj={best_obj:.8f}, beta={beta:.0f}, "
                          f"lr={lr:.5f}, time={elapsed:.0f}s")
    
    return best_a

def lp_best_response(a):
    """

    Returns optimal g or None.
    """
    import cvxpy as cp
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None
    a = a / s
    
    # Build constraint matrix: (A @ g)[k] = conv(a, g)[k]
    # conv(a, g)[k] = sum_j a[j] * g[k-j]
    # For fixed k, this is sum_j a[j] * g[k-j]
    # Let's build it as: for each k, the coefficient of g[i] is a[k-i] (if valid)
    # So (A @ g)[k] = sum_i A[k,i] * g[i] where A[k,i] = a[k-i] if 0<=k-i<n else 0
    
    # This is a banded matrix. For efficiency, let's use the structure.
    # A[k, i] = a[max(0, k-i) ... ]... actually:
    # (A @ g)[k] = sum_i a[k-i] * g[i] for 0 <= k-i < n, i.e., k-n+1 <= i <= k
    # So A[k, i] = a[k-i] if k-n+1 <= i <= k and 0 <= i < n, else 0.
    
    # For n=800, A is (2n-1) x n = 1599 x 800. That's fine for cvxpy.
    
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            indices = np.arange(i_start, i_end + 1)
            A[k, indices] = a[k - indices]
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass
    
    return None

def lp_alternating(a, time_budget=200.0, verbose=True):
    """
    LP alternating: find best response and interpolate.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    last_improve = 0

    while time.monotonic() - start < time_budget:
        g = lp_best_response(a)
        if g is None:
            break
        
        # Try different interpolation factors
        best_t = 0.5
        best_t_obj = np.inf
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t_obj < best_obj - 1e-14:
            a = ((1 - best_t) * a + best_t * g)
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            best_a = a.copy()
            best_obj = best_t_obj
            last_improve = time.monotonic()
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP: obj={best_obj:.8f}, t={best_t:.2f}, time={elapsed:.0f}s")
        else:
            # No improvement, try perturbation
            if time.monotonic() - last_improve > 30:
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                last_improve = time.monotonic()
            else:
                time.sleep(0.1)
    
    return best_a

def multiplicative_update(a, p, lr=0.1):
    """
    Multiplicative (exponential) update: a_i <- a_i * exp(-lr * (grad_i - mean(grad)))
    where grad = 2 * cross_correlation(p, a).
    """
    n = len(a)
    grad = 2.0 * cross_correlation(p, a)
    grad_centered = grad - np.mean(grad)
    a_new = a * np.exp(-lr * grad_centered)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / np.sum(a_new)
    return a_new

def coordinate_descent(a, time_budget=100.0):
    """
    Coordinate descent: adjust individual coordinates to minimize the objective.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    
    rng = np.random.default_rng(12345)
    stall = 0
    
    while time.monotonic() - start < time_budget:
        improved = False
        
        # Try pairs of coordinates
        for _ in range(100):
            if time.monotonic() - start > time_budget:
                break
            
            # Pick two random coordinates
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            
            # Try moving mass from i to j
            for frac in [0.001, 0.005, 0.01, 0.02]:
                delta = frac * a[i]
                if delta <= 0:
                    continue
                a[i] -= delta
                a[j] += delta
                conv = compute_conv(a)
                obj = 2.0 * n * float(np.max(conv))
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
            
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                # Larger perturbation
                scale = 0.02
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
        else:
            stall = 0
    
    return best_a
