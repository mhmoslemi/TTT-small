Soft-max smoothed gradient descent with Nesterov acceleration.
Exploits the structure of the self-convolution gradient for O(n log n) per step.
Enforces symmetry, uses adaptive beta schedule, and polishes with LP alternating.

from scipy.fft import rfft, irfft
import time

from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

```python
"""
Soft-max smoothed Nesterov gradient descent + LP alternating polish.
Key innovation: O(n log n) gradient computation via FFT cross-correlation,
enabling many more iterations than subgradient methods.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=60.0, verbose=False):
    """LP alternating with line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_good_t = 0.5

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        # Line search
        best_t = last_good_t
        best_line_obj = best_obj
        for t in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = np.maximum((1.0 - t) * a + t * g_lp, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_line_obj:
                best_line_obj = obj_try
                best_t = t
        
        a_try = np.maximum((1.0 - best_t) * a + best_t * g_lp, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            last_good_t = best_t
            stall_count = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.2f}")
        else:
            stall_count += 1
            if stall_count > 10:
                break

    return best_a


def soft_max_nesterov(a0, time_budget=100.0, beta_start=5.0, beta_end=500.0, verbose=False):
    """
    Soft-max smoothed Nesterov gradient descent.
    Key: gradient computed via FFT cross-correlation in O(n log n).
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    a_prev = a.copy()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Precompute FFT sizes
    conv_size = 1
    while conv_size < 2 * n - 1:
        conv_size <<= 1
    corr_size = 1
    while corr_size < 3 * n - 2:
        corr_size <<= 1
    
    start = time.monotonic()
    step_base = 0.5  # base step size, will be divided by beta
    patience = 0
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Beta schedule: geometric from beta_start to beta_end
        progress = min(1.0, it / 5000.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute conv(y, y)
        fy = rfft(y, conv_size)
        conv_y = irfft(fy * fy, conv_size)[:2 * n - 1]
        
        # Softmax weights (numerically stable)
        conv_max = np.max(conv_y)
        exp_vals = np.exp(beta * (conv_y - conv_max))
        p = exp_vals / np.sum(exp_vals)
        
        # Gradient: 4n * cross_correlate(p, y)
        fp = rfft(p, corr_size)
        fy2 = rfft(y, corr_size)
        corr = irfft(fp * np.conj(fy2), corr_size)
        grad = 4.0 * n * corr[:n]
        
        # Center gradient
        grad -= np.mean(grad)
        
        # Step size (inverse proportional to beta)
        step = step_base / beta
        
        # Nesterov update
        a_new = project_simplex(y - step * grad)
        momentum = it / (it + 3.0)
        y = a_new + momentum * (a_new - a)
        y = project_simplex(y)
        a = a_new
        
        it += 1
        
        # Evaluate periodically
        if it % 100 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0

            else:
                patience += 1
                if patience > 50:
                    step_base = max(step_base * 0.7, 0.01)

def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw
