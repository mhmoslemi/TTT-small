"""
Hybrid Peak Flattening optimizer.
Combines weighted subgradient descent, LP alternating with warm-starting,
multi-scale approach, and coordinate descent for final polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_grad_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a[m]."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        m_start = k - i_end
        m_end = k - i_start
        grad[m_start:m_end + 1] = a[i_start:i_end + 1][::-1]
        if k % 2 == 0:
            m = k // 2
            if 0 <= m < n:
                grad[m] += a[m]
    return 2.0 * grad

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def weighted_peak_subgrad(a, top_k=10, power=2):
    """Compute weighted subgradient using top peaks."""
    conv = compute_conv(a)
    n = len(a)
    # Get top peaks
    top_idx = np.argsort(conv)[::-1][:top_k]
    weights = np.array([conv[k] for k in top_idx])
    weights = weights ** power
    w_sum = weights.sum()
    if w_sum < 1e-15:
        return np.zeros(n), conv
    weights = weights / w_sum
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        # grad[m] += 2 * w * a[k-m] for valid m
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            m_start = k - i_end
            m_end = k - i_start
            grad[m_start:m_end + 1] += 2.0 * w * a[i_start:i_end + 1][::-1]
    
    return grad, conv


def backtracking_line_search(a, direction, alpha=1.0, c=0.3, rho=0.5, max_iter=20):
    """Backtracking line search on the simplex."""
    # direction is in the tangent space of the simplex (sum = 0)
    # We project a + alpha * direction onto the simplex
    f0 = compute_obj(a)
    # Compute directional derivative (using the subgradient)
    grad, _ = weighted_peak_subgrad(a, top_k=5)
    grad = grad - np.mean(grad)  # center for simplex
    f0_grad = grad @ direction  # directional derivative
    
    for _ in range(max_iter):
        a_try = a + alpha * direction
        a_try = project_simplex(a_try)
        f_try = compute_obj(a_try)
        if f_try <= f0 + c * alpha * f0_grad:
            return a_try, alpha
        alpha *= rho
    return a, 0

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For fixed j, A[k, j] = a[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(m - 1, j + n - 1)
        a_idx_end = k_end - j + 1
        A[k_start:k_end + 1, j] = a[:a_idx_end]
    return A

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

"""
Hybrid Peak Flattening: Combines weighted multi-peak subgradient descent with
adaptive parameters, LP alternating with golden-section line search,
coordinate descent, and multi-scale exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def weighted_peak_subgrad(a, top_k=10, power=2.0):
    """Compute weighted subgradient of max conv using top peaks."""
    conv = compute_conv(a)
    n = len(a)
    top_idx = np.argsort(conv)[::-1][:top_k]
    weights = np.array([conv[k] for k in top_idx])
    weights = weights ** power
    w_sum = weights.sum()
    if w_sum < 1e-15:
        return np.zeros(n)
    weights = weights / w_sum

    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            m_start = k - i_end
            m_end = k - i_start
            grad[m_start:m_end + 1] += 2.0 * w * a[i_start:i_end + 1][::-1]
    return grad


def nesterov_weighted(a0, time_budget=60.0, top_k=10, power=2.0, verbose=False):
    """Nesterov-accelerated weighted subgradient on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.12
    patience = 0

    for it in range(500000):
        if time.monotonic() - start > time_budget:
            break
        grad = weighted_peak_subgrad(y, top_k=top_k, power=power)
        grad = grad - np.mean(grad)

        y_new = y - step_size * grad
        a_new = project_simplex(y_new)
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = project_simplex(y)
        a = a_new

        if (it + 1) % 20 == 0 or it < 30:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 500:
                step_size = max(step_size * 0.65, 0.002)
                patience = 0

        if (it + 1) % 2000 == 0:
            step_size = max(step_size * 0.96, 0.004)

        if verbose and (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"      NW it={it}: best={best_obj:.8f}, step={step_size:.4f}, t={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(m - 1, j + n - 1)
        a_len = k_end - j + 1
        A[k_start:k_end + 1, j] = a[:a_len]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate_v2(a0, time_budget=100.0, verbose=False):
    """LP alternating with bounded scalar minimization."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    last_t = 0.5

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)

        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 25, 'xatol': 0.005})
            best_t = res.x
        except Exception:
            best_t = last_t

        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            last_t = best_t
            stall_count = 0
            if verbose:
                print(f"      LP2: obj={best_obj:.8f}, t={best_t:.4f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 25:
                break

    return best_a


def coordinate_descent(a0, time_budget=60.0, verbose=False):
    """Coordinate descent with adaptive multiplier search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for outer in range(2000):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        # Cycle through coordinates
        for m in range(n):
            if time.monotonic() - start > time_budget:
                break
            if a[m] < 1e-12:
                # Try turning on this coordinate
                for val in [0.01 / n, 0.05 / n, 0.1 / n]:
                    a_try = a.copy()
                    a_try[m] = val
                    a_try = a_try / np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                continue

            # Try several multipliers (log-spaced around 1)
            for mult in [0.6, 0.8, 0.9, 1.1, 1.2, 1.5, 2.0]:
                a_try = a.copy()
                a_try[m] = a[m] * mult
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved and outer > 50:
            break
        if verbose and (outer + 1) % 50 == 0:
            print(f"      CD iter={outer}: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def peak_targeted_perturb(a, rng, n_peaks=5, strength=0.05):
    """Perturb the sequence by shifting mass from peak contributors."""
    n = len(a)
    a_p = a.copy()
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:n_peaks]
    for k in top_idx:
        k = int(k)
        # Find top contributors to this peak
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        contrib = np.zeros(n)
        for i in range(i_start, i_end + 1):
            m = k - i
            contrib[m] += a[i]
            if m != i:
                contrib[i] += a[m]
        # Move mass from top contributor to a random low contributor
        top_c = np.argsort(contrib)[::-1][0]
        low_candidates = np.where(contrib < np.percentile(contrib, 20))[0]
        if len(low_candidates) > 0:
            dst = rng.choice(low_candidates)
            delta = strength * a[top_c]
            if a_p[top_c] > delta:
                a_p[top_c] -= delta
                a_p[dst] += delta
    a_p = np.maximum(a_p, 0)
    s = float(np.sum(a_p))
    if s > 1e-10:
        a_p = a_p / s
    return a_p


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Multi-scale + adaptive weighted subgradient + LP alternating
    + coordinate descent + peak-targeted perturbation for escape.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = hann_window(800)
        best_score = compute_obj(best_seq)
        print(f"Hann init (n=800): obj={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: Multi-scale scan (150s) ===
    print(f"\n=== Phase 1: Multi-scale Scan ===")
    phase1_end = start_time + 150.0

    n_candidates = sorted(set([
        300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000,
        n_base, max(200, n_base - 150), min(2500, n_base + 150)
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 2500]

    for n_try in n_candidates:
        if time.monotonic() > phase1_end:
            break
        if n_try == n_base:
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
        alloc = min(10.0, (phase1_end - time.monotonic()) * 0.4)
        if alloc < 3:
            continue
        a_opt = nesterov_weighted(a_try, time_budget=alloc, top_k=10, power=2.0)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score - 1e-12:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 2: Adaptive weighted subgradient (250s) ===
    print(f"\n=== Phase 2: Adaptive Weighted Subgradient ===")
    phase2_end = start_time + 450.0

    # Vary power exponent
    for power_val in [1.0, 2.0, 3.0, 5.0]:
        if time.monotonic() > phase2_end:
            break
        alloc = min(40.0, (phase2_end - time.monotonic()) * 0.3)
        if alloc < 5:
            break
        a_ns = nesterov_weighted(best_seq, time_budget=alloc, top_k=10, power=power_val, verbose=False)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score - 1e-12:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  P={power_val}: obj={best_score:.8f} **")
        else:
            print(f"  P={power_val}: obj={obj_ns:.8f}")

    # Vary top_k
    for k_val in [3, 5, 10, 20, 40]:
        if time.monotonic() > phase2_end:
            break
        alloc = min(30.0, (phase2_end - time.monotonic()) * 0.3)
        if alloc < 5:
            break
        a_ns = nesterov_weighted(best_seq, time_budget=alloc, top_k=k_val, power=2.0, verbose=False)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score - 1e-12:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  K={k_val}: obj={best_score:.8f} **")
        else:
            print(f"  K={k_val}: obj={obj_ns:.8f}")

    # === Phase 3: LP Alternating (400s) ===
    print(f"\n=== Phase 3: LP Alternating ===")
    phase3_end = start_time + 900.0

    for round_idx in range(10):
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 30:
            break
        alloc = min(50.0, remaining * 0.3)
        a_new = lp_alternate_v2(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score - 1e-12:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj:.8f}")
            # Peak-targeted perturbation to escape
            a_pert = peak_targeted_perturb(best_seq, rng, n_peaks=5, strength=0.03)
            alloc2 = min(15.0, remaining * 0.15)
            if alloc2 > 5:
                a_q = nesterov_weighted(a_pert, time_budget=alloc2, top_k=10, power=2.0)
                obj_q = compute_obj(a_q)
                if obj_q < best_score - 1e-12:
                    best_score = obj_q
                    best_seq = a_q.copy()
                    print(f"  Perturb+NW R{round_idx}: obj={best_score:.8f} **")

    # === Phase 4: Coordinate Descent (150s) ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    phase4_end = start_time + 1080.0
    if time.monotonic() < phase4_end:
        alloc = min(120.0, phase4_end - time.monotonic())
        a_cd = coordinate_descent(best_seq, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score - 1e-12:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 5: Combined Polish (180s) ===
    print(f"\n=== Phase 5: Combined Polish ===")
    phase5_end = start_time + 1260.0

    # LP polish
    if time.monotonic() < phase5_end:
        alloc = min(50.0, phase5_end - time.monotonic())
        if alloc > 10:
            a_f = lp_alternate_v2(best_seq, time_budget=alloc, verbose=True)
            obj_f = compute_obj(a_f)
            if obj_f < best_score - 1e-12:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Polish LP: obj={best_score:.8f} **")

    # Nesterov polish with different params
    if time.monotonic() < phase5_end:
        alloc = min(40.0, phase5_end - time.monotonic())
        if alloc > 10:
            a_ns = nesterov_weighted(best_seq, time_budget=alloc, top_k=5, power=4.0, verbose=False)
            obj_ns = compute_obj(a_ns)
            if obj_ns < best_score - 1e-12:
                best_score = obj_ns
                best_seq = a_ns.copy()
                print(f"  Polish NW: obj={best_score:.8f} **")

    # CD polish
    if time.monotonic() < phase5_end:
        alloc = min(40.0, phase5_end - time.monotonic())
        if alloc > 5:
            a_cd = coordinate_descent(best_seq, time_budget=alloc, verbose=False)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score - 1e-12:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Polish CD: obj={best_score:.8f} **")

    # === Phase 6: Random restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 6: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(40):
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break
            # Different perturbation strategies
            if r % 3 == 0:
                scale = 0.015 + 0.005 * (r // 3)
                a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            elif r % 3 == 1:
                a_r = peak_targeted_perturb(best_seq, rng, n_peaks=r % 10 + 1,
                                            strength=0.02 + 0.01 * (r // 3))
            else:
                # Block perturbation
                a_r = best_seq.copy()
                block_size = max(1, n_best // 20)
                start_idx = int(rng.integers(0, max(1, n_best - block_size)))
                a_r[start_idx:start_idx + block_size] *= rng.uniform(0.5, 1.5)
                a_r = np.maximum(a_r, 0)

            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr

            alloc = min(remaining * 0.2, 10.0)
            if alloc < 4:
                continue
            a_opt = nesterov_weighted(a_r, time_budget=alloc * 0.6, top_k=10, power=2.0)
            a_opt = lp_alternate_v2(a_opt, time_budget=alloc * 0.4)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score - 1e-12:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  R{r}: obj={best_score:.8f} **")
            else:
                print(f"  R{r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
