   def convex_line_search(a, g, n_iter=30):
       n = len(a)
       a_n = a / a.sum()
       g_n = g / g.sum()
       
       conv_aa = compute_conv(a_n)
       conv_ag = compute_conv_cross(a_n, g_n)
       conv_gg = compute_conv(g_n)
       
       def obj_at_t(t):
           val = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
           return 2.0 * n * val.max()
       
       # Golden section search
       gr = (np.sqrt(5) - 1) / 2
       lo, hi = 0.0, 1.0
       m1 = hi - gr * (hi - lo)
       m2 = lo + gr * (hi - lo)
       f1 = obj_at_t(m1)
       f2 = obj_at_t(m2)
       for _ in range(n_iter):
           if f1 < f2:
               hi = m2
               m2 = m1
               f2 = f1
               m1 = hi - gr * (hi - lo)
               f1 = obj_at_t(m1)
           else:
               lo = m1
               m1 = m2
               f1 = f2
               m2 = lo + gr * (hi - lo)
               f2 = obj_at_t(m2)
       
       t_opt = (lo + hi) / 2
       a_new = (1 - t_opt) * a_n + t_opt * g_n
       a_new = np.maximum(a_new, 0)
       s = a_new.sum()
       if s < 1e-10:
           return 0.0, a_n
       return t_opt, a_new / s
