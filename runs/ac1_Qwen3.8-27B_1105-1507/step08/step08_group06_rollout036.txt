from scipy.fft import rfft, irfft
import time

def fft_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = fft_conv(a)
    return 2.0 * n * conv.max() / (s*s)

        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = fft_conv(a)
    max_conv = conv.max()
    
    it = 0

        # Try different orderings
        if it % 5 == 0:
            order = np.arange(n)
        elif it % 5 == 1:
            order = np.arange(n-1, -1, -1)  # reverse

            order = np.random.permutation(n)

            order = np.arange(n)

"""

convex subproblems via bisection) with LP-based alternating minimization.
Key innovation: each coordinate update solves the exact 1D problem
min_{x>=0} max_k (conv0[k] + L[k]*x + Q[k]*x^2) in O(n) time,
giving ~18000 full passes in the time budget for n=800.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fft_conv(a):
    """Compute full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    conv = fft_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def solve_1d_convex(C, L, Q):
    """
    Solve min_{x>=0} max_k (C[k] + L[k]*x + Q[k]*x^2).
    At most one Q[k] equals 1, rest are 0. Returns optimal x >= 0.
    Uses bisection on the subgradient (convex function).
    """
    f0 = float(np.max(C))
    tol0 = 1e-14 * max(1.0, abs(f0))
    active = C >= f0 - tol0
    if not active.any():
        return 0.0
    sg0 = L[active]
    if float(sg0.min()) >= -1e-10:
        return 0.0

        elif sg_max <= 0.0:
            lo = mid
        else:

def coordinate_descent(a, time_budget, verbose=False, rng=None,
                       skip_small=False, small_thresh=1e-8):
    """
    Cyclic coordinate descent with exact 1D convex subproblem solving.
    Minimizes max(conv(a,a)) for non-negative a (renormalized to sum=1 each cycle).
    """
    if rng is None:
        rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_maxc = float(fft_conv(a).max())
    start = time.monotonic()
    conv = fft_conv(a).copy()

    # Preallocate arrays for reuse
    ki_arr = np.empty(2 * n - 1, dtype=np.int64)

    cycle = 0
    no_improve = 0
    while time.monotonic() - start < time_budget:
        # Choose ordering based on cycle
        r = cycle % 9
        if r == 0:
            order = np.arange(n)
        elif r == 1:
            order = np.arange(n - 1, -1, -1)
        elif r in (2, 5, 8):
            order = rng.permutation(n)
        elif r == 3:
            order = np.argsort(a)
        elif r == 4:
            order = np.argsort(a)[::-1]
        elif r == 6:
            # Block random: random order within blocks of size 10
            blocks = [np.arange(i, min(i + 10, n)) for i in range(0, n, 10)]
            order = np.concatenate([b[rng.permutation(len(b))] for b in blocks])
        else:
            # Interleaved (zipper)
            evens = np.arange(0, n, 2)
            odds = np.arange(1, n, 2)
            order = np.empty(n, dtype=int)
            order[0::2] = evens
            order[1::2] = odds

        cycle_improved = False
        for i in order:
            if time.monotonic() - start > time_budget:
                break

            old = a[i]
            if old < small_thresh and skip_small and rng.random() < 0.7:
                continue

            # Compute conv without a[i]'s contribution
            np.subtract(np.arange(2 * n - 1), i, out=ki_arr)
            valid_k = (ki_arr >= 0) & (ki_arr < n)

            conv0 = conv.copy()
            if np.any(valid_k):
                conv0[valid_k] -= 2.0 * old * a[ki_arr[valid_k]]
                if 2 * i < 2 * n - 1:
                    conv0[2 * i] += old * old

            # Build L and Q for the 1D problem
            L = np.zeros(2 * n - 1)
            valid_L = valid_k & (ki_arr != i)
            if np.any(valid_L):
                L[valid_L] = 2.0 * a[ki_arr[valid_L]]
            Q = np.zeros(2 * n - 1)
            if 2 * i < 2 * n - 1:
                Q[2 * i] = 1.0

            # Solve 1D subproblem
            x_opt = solve_1d_convex(conv0, L, Q)

            if abs(x_opt - old) > 1e-15:
                a[i] = max(0.0, x_opt)
                # Update conv incrementally
                if np.any(valid_L):
                    conv[valid_L] = conv0[valid_L] + 2.0 * a[i] * a[ki_arr[valid_L]]
                if 2 * i < 2 * n - 1:
                    conv[2 * i] = conv0[2 * i] + a[i] * a[i]
                for k in range(2 * n - 1):
                    if not valid_k[k] and k != 2 * i:
                        conv[k] = conv0[k]
                cycle_improved = True

        # Renormalize
        s = a.sum()
        if s > 1e-12:
            a = a / s
            conv = fft_conv(a)

        maxc = float(conv.max())
        if maxc < best_maxc - 1e-12:
            best_maxc = maxc
            best_a = a.copy()
            no_improve = 0
            if verbose and cycle % 50 == 0:
                elapsed = time.monotonic() - start
                print(f"      CD: maxc={best_maxc:.10f}, obj={2*n*best_maxc:.8f}, "
                      f"cycle={cycle}, t={elapsed:.0f}s")

                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)

                    a = a / s
                conv = fft_conv(a)
                best_maxc = float(conv.max())
                best_a = a.copy()
                no_improve = 0

        cycle += 1

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_response(a):
    """
    Solve LP: min M s.t. conv(a, g) <= M (all k), sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None
    a = a / s
    A = build_conv_matrix(a)
    m = 2 * n - 1

    # Try cvxpy solvers
    g = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g <= M_var, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0.0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M_var.value) if M_var.value is not None else float(np.max(A @ g_val))
                    return g_val, M_val
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(m), A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0.0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                return g_val, M_val
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget, verbose=False, momentum_beta=0.5, rng=None):
    """LP alternating with momentum and adaptive step size."""
    if rng is None:
        rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_maxc = float(fft_conv(a).max())
    start = time.monotonic()
    momentum = np.zeros(n)
    last_t = 0.4
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_response(a)
        if g_lp is None:
            break

        # Try different step sizes
        step_candidates = [0.1, 0.2, 0.35, 0.5, 0.7, 1.0]
        best_t = 0.4
        best_a_local = a.copy()
        best_maxc_local = best_maxc

        for t in step_candidates:
            direction = g_lp - a
            momentum = momentum_beta * momentum + (1.0 - momentum_beta) * direction
            a_try = a + t * momentum
            a_try = np.maximum(a_try, 0.0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            maxc_try = float(fft_conv(a_try).max())
            if maxc_try < best_maxc_local:
                best_maxc_local = maxc_try
                best_t = t
                best_a_local = a_try.copy()

        if best_maxc_local < best_maxc - 1e-14:
            best_maxc = best_maxc_local
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                if elapsed > 3:
                    print(f"      LP: maxc={best_maxc:.10f}, obj={2*n*best_maxc:.8f}, "
                          f"t={best_t:.2f}, time={elapsed:.0f}s")
        else:
            stall += 1
            momentum *= 0.5
            if stall % 5 == 0:
                scale = 0.02 + 0.005 * min(stall // 5, 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-12:
                    a = a / s
                momentum = np.zeros(n)
                stall = 0

    return best_a


def generate_initialization(kind, n, rng):
    """Generate diverse initialization of length n, normalized to sum=1."""
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - center) / (n / 5.0)) ** 2)
    elif kind == 'u_shaped':
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 * 4 + 0.1
    elif kind == 'u_shaped_strong':
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 * 16 + 0.01
    elif kind == 'random':
        a = rng.random(n) + 0.01
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 8)
        a[:edge] = 0.4
        a[-edge:] = 0.4
    elif kind == 'bipolar':
        a = np.ones(n) * 0.3
        a[n // 4] = 2.0
        a[3 * n // 4] = 2.0
        a = np.convolve(a, np.ones(7) / 7, mode='same')
    elif kind == 'edges':
        a = np.ones(n) * 0.1
        a[0] = 3.0
        a[-1] = 3.0
        a[1] = 2.0
        a[-2] = 2.0
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0.0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: coordinate descent + LP alternating + length sweep.
    Returns list of non-negative floats minimizing 2n*max(conv)/(sum)^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        if parent.size > 0 and np.isfinite(parent).all():
            s = float(parent.sum())
            if s > 0.01:
                best_seq = parent / s
                best_score = compute_obj(best_seq)
                print(f"Parent (n={best_seq.size}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800.0
        best_score = compute_obj(best_seq)
        print(f"Default uniform (n=800): obj={best_score:.8f}")

    n = len(best_seq)
    a = best_seq.copy()

    # === Phase 1: Quick CD from parent (200s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 1: CD from parent (n={n}) ===")
    cd_budget = min(200.0, remaining * 0.15)
    a_cd = coordinate_descent(a, time_budget=cd_budget, verbose=True, rng=rng)
    obj_cd = compute_obj(a_cd)
    if obj_cd < best_score:
        best_score = obj_cd
        best_seq = a_cd.copy()
        a = a_cd.copy()
        print(f"  CD: obj={best_score:.8f} **")
    else:
        print(f"  CD: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: LP alternating (300s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 2: LP Alternating ===")
    if remaining > 60:
        lp_budget = min(300.0, remaining * 0.25)
        for beta in [0.4, 0.6, 0.8]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(lp_budget / 3.0, remaining * 0.5)
            a_lp = lp_alternating(a, time_budget=alloc, verbose=True,
                                  momentum_beta=beta, rng=rng)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a = a_lp.copy()
                print(f"  LP (beta={beta}): obj={best_score:.8f} **")
            else:
                print(f"  LP (beta={beta}): obj={obj_lp:.8f}")

    # === Phase 3: More CD (300s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 3: CD Polish ===")
    if remaining > 50:
        cd_budget = min(300.0, remaining * 0.25)
        a_cd2 = coordinate_descent(a, time_budget=cd_budget, verbose=True,
                                   rng=rng, skip_small=True)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            a = a_cd2.copy()
            print(f"  CD2: obj={best_score:.8f} **")
        else:
            print(f"  CD2: obj={obj_cd2:.8f}")

    # === Phase 4: LP alternating again (200s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 4: LP Polish ===")
    if remaining > 40:
        lp_budget = min(200.0, remaining * 0.18)
        a_lp2 = lp_alternating(a, time_budget=lp_budget, verbose=True,
                               momentum_beta=0.5, rng=rng)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            a = a_lp2.copy()
            print(f"  LP2: obj={best_score:.8f} **")
        else:
            print(f"  LP2: obj={obj_lp2:.8f}")

    # === Phase 5: Diverse initializations + quick CD (200s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Diverse Inits + CD ===")
    if remaining > 60:
        init_kinds = ['u_shaped', 'u_shaped_strong', 'edges', 'gaussian',
                      'cosine', 'random', 'bipolar']
        per_init = max(10.0, remaining * 0.12 / len(init_kinds))
        for kind in init_kinds:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_init = generate_initialization(kind, n, rng)
            a_opt = coordinate_descent(a_init, time_budget=per_init, rng=rng,
                                       skip_small=True)
            obj_try = compute_obj(a_opt)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                a = a_opt.copy()
                print(f"  Init '{kind}': obj={best_score:.8f} **")
            else:
                print(f"  Init '{kind}': obj={obj_try:.8f}")

    # === Phase 6: Length sweep (200s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 6: Length Sweep ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        max(100, n_best - 100), max(100, n_best - 50),
        max(100, n_best - 25), n_best,
        n_best + 25, n_best + 50, n_best + 100,
        n_best + 200, n_best + 400,
    ]))
    n_options = [x for x in n_options if 100 <= x <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
            a_try = a_try / a_try.sum()

        alloc_cd = min(25.0, remaining * 0.4)
        alloc_lp = min(15.0, remaining * 0.2)

        a_opt = coordinate_descent(a_try, time_budget=alloc_cd, rng=rng,
                                   skip_small=True)
        if remaining > alloc_cd + 10:
            a_opt = lp_alternating(a_opt, time_budget=alloc_lp, rng=rng,
                                   momentum_beta=0.5)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final CD polish (remaining) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 7: Final CD Polish (n={len(best_seq)}) ===")
    if remaining > 30:
        a_final = coordinate_descent(best_seq, time_budget=min(remaining * 0.6, 200.0),
                                     verbose=True, rng=rng, skip_small=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={best_score:.8f} **")
        else:
            print(f"  Final CD: obj={obj_final:.8f}")

    # === Phase 8: Final LP (if time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 60.0),
                                 verbose=True, momentum_beta=0.7, rng=rng)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
        else:
            print(f"  Final LP: obj={obj_final:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val
    elif not np.isfinite(val):
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
