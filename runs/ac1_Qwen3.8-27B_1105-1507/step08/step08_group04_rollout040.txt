def kkt_directed_perturbation(a, rng_seed=42):
    """Perturb a in a direction that targets the worst autocorrelation peaks."""
    n = len(a)
    rng = np.random.default_rng(rng_seed)
    conv = compute_conv(a)
    max_b = np.max(conv)
    
    # Compute subgradient direction
    # grad[j] = 2 * a[peak - j] for the peak(s)
    # Use top peaks with exponential weighting
    T = max_b * 0.1  # Temperature
    weights = np.exp((conv - max_b) / T)
    weights /= np.sum(weights)
    
    # Compute gradient
    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1
    A_fft = rfft(a, n_fft)
    W_fft = rfft(weights, n_fft)
    R = irfft(np.conj(A_fft) * W_fft, n_fft)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = R[(n_fft - j) % n_fft]
    
    # Perturb in the negative gradient direction (to reduce peaks)
    # Add some randomness
    scale = 0.05 + 0.03 * rng.random()
    perturbation = -scale * grad + 0.02 * rng.normal(0, 1, n)
    
    a_new = a * (1.0 + perturbation)
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s

def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
