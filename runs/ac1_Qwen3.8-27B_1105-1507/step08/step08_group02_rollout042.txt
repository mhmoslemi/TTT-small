"""
Optimized step function search using mirror descent with log-sum-exp smoothing,
LP-based best response, and multi-scale exploration.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Key insight: the max of linear convolution is NOT always at lag-0, enabling
asymmetric sequences to achieve objective < 2.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv(a))) / (s * s)


def smooth_grad_objective(a, beta):
    """
    Gradient of 2n * LSE_beta(conv(a)) where a is on the simplex.
    grad[i] = (4n/beta) * sum_m a[m] * w[i+m]
    where w = softmax(beta * (conv - max(conv)))
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = conv.max()
    shifted = beta * (conv - max_conv)
    max_shifted = np.max(shifted)
    w = np.exp(shifted - max_shifted)
    w /= w.sum()

    grad = np.zeros(n)
    for i in range(n):
        grad[i] = np.dot(a, w[i:i + n])

    return (4.0 * n / beta) * grad

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            # Slightly decrease learning rate
            lr = max(0.1 / np.sqrt(n), lr * 0.999)
        else:
            stall += 1
            if stall > patience:
                # Restart with perturbation
                noise_scale = min(0.3, 0.02 * (stall // patience))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = 2.0 / np.sqrt(n)
                stall = 0
            else:
                # Reduce learning rate
                lr = max(0.05 / np.sqrt(n), lr * 0.95)

        total_iters += 1

    return best_a


def build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_best_response(a, use_cvxpy=True):
    """
    Solve LP: min M s.t. (A@g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    if use_cvxpy:
        try:
            import cvxpy as cp
            g = cp.Variable(n, nonneg=True)
            M = cp.Variable()
            constraints = [A @ g <= M, cp.sum(g) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.SCS]:
                try:
                    prob.solve(solver=solver)
                    if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        g_val = np.maximum(g.value, 0)
                        s = float(np.sum(g_val))
                        if s > 1e-10:
                            return g_val / s, float(M.value)
                except Exception:
                    continue
        except ImportError:
            pass

    # Fallback to scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_refine(a, time_budget=100.0, verbose=False):
    """
    LP-based refinement: iteratively solve LP for best response and blend.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 10
    t_blend = 0.3

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_best_response(a)
        if g_lp is None:
            # Fallback: use perturbed version
            g_lp = a * (1 + 0.1 * np.random.randn(n))
            g_lp = np.maximum(g_lp, 0)
            g_lp = g_lp / g_lp.sum()

        # Try multiple blend factors
        best_t = t_blend
        best_t_obj = np.inf
        for t in [0.1, 0.2, t_blend, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t_obj < best_obj - 1e-12:
            a_new = (1 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            best_obj = best_t_obj
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            t_blend = best_t
        else:
            stall += 1
            if stall > patience:
                # Perturb and restart
                a = best_a * (1 + 0.05 * np.random.randn(n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                t_blend = 0.5
            else:
                t_blend = max(0.1, t_blend * 0.8)

        if verbose:
            elapsed = time.monotonic() - start
            print(f"      LP-Refine: obj={best_obj:.8f}, t={best_t:.2f}, stall={stall}, t_elapsed={elapsed:.1f}s")

    return best_a


def coordinate_descent_refine(a, time_budget=30.0):
    """
    Coordinate descent: try moving mass between pairs to reduce max convolution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Find the pair (i, k_star - i) that contributes most to conv[k_star]
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            contributions = np.array([a[i] * a[k_star - i] for i in range(i_start, i_end + 1)])
            idxs = np.arange(i_start, i_end + 1)
            top_idx = idxs[np.argmax(contributions)]
            j = k_star - top_idx

            # Try moving mass from the larger of a[top_idx], a[j] to a random other position
            if a[top_idx] >= a[j]:
                src = top_idx
            else:
                src = j

            for attempt in range(20):
                dst = rng.integers(0, n)
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2]:
                    delta = frac * a[src]
                    if delta < 1e-10 or delta > a[src]:
                        continue
                    a[src] -= delta
                    a[dst] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[dst] -= delta
                        break

        if not improved:
            # Try random pairwise moves
            for _ in range(50):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                for frac in [0.01, 0.03, 0.05]:
                    delta = frac * a[i]
                    if delta < 1e-10:
                        break
                    a[i] -= delta
                    a[j] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break
            if not improved:
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Linear ramp (increasing)
        a = np.arange(1, n + 1, dtype=np.float64)
        a = a / a.sum()
    elif mode == 3:
        # Quadratic ramp
        a = np.arange(1, n + 1, dtype=np.float64) ** 2
        a = a / a.sum()
    elif mode == 4:
        # Exponential ramp
        r = 1.0 + rng.uniform(0.01, 0.1)
        a = r ** np.arange(n, dtype=np.float64)
        a = a / a.sum()
    elif mode == 5:
        # Right-heavy: a[i] = (1 + i/n)^p
        p = rng.uniform(1, 4)
        a = (1.0 + np.arange(n, dtype=np.float64) / n) ** p
        a = a / a.sum()
    elif mode == 6:
        # Triangular shifted right
        center = n * rng.uniform(0.6, 0.9)
        width = n * rng.uniform(0.2, 0.4)
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Sine-weighted ramp
        x = np.linspace(0, 1, n)
        a = x ** 2 * (1 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 4) * x))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 8:
        # Two-peak (asymmetric)
        c1 = n * 0.3
        c2 = n * 0.8
        w1 = n * 0.1
        w2 = n * 0.15
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-((x - c1) ** 2) / (2 * w1 ** 2)) + 0.5 * np.exp(-((x - c2) ** 2) / (2 * w2 ** 2))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def hybrid_optimize(a, time_budget=100.0, seed=0):
    """
    Hybrid optimization: mirror descent + LP + coordinate descent.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Phase 1: Mirror descent (60% of budget)
    md_time = time_budget * 0.6
    a_md = mirror_descent_optimize(best_a, time_budget=md_time, beta_start=5, beta_end=1500, seed=seed)
    obj_md = compute_obj(a_md)
    if obj_md < best_obj:
        best_obj = obj_md
        best_a = a_md.copy()

    # Phase 2: LP refinement (25% of budget)
    lp_time = time_budget * 0.25
    remaining = time_budget - (time.time() - (time.time() - lp_time))  # approximate
    if lp_time > 10:
        a_lp = lp_refine(best_a, time_budget=lp_time)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 3: Coordinate descent (15% of budget)
    cd_time = time_budget * 0.15
    if cd_time > 5:
        a_cd = coordinate_descent_refine(best_a, time_budget=cd_time)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines diverse initializations, mirror descent,
    LP refinement, and multi-length exploration to minimize the objective.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Try to load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.arange(1, n0 + 1, dtype=np.float64)
        best_seq = best_seq / best_seq.sum()
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}, ramp): obj={best_score:.8f}")

    print(f"\nInitial best: n={len(best_seq)}, obj={best_score:.8f}")

    # --- Phase 1: Screen diverse initializations (60s) ---
    print("\n=== Phase 1: Diverse Initialization Screening ===")
    p1_deadline = start_time + 60.0
    n0 = len(best_seq)

    init_results = []
    # Always include current best
    init_results.append(("parent/best", best_seq.copy()))

    for mode in range(9):
        if time.monotonic() > p1_deadline:
            break
        a_init = diverse_init(n0, rng, mode=mode)
        obj = compute_obj(a_init)
        init_results.append((f"mode{mode}", a_init))
        if obj < best_score:
            best_score = obj
            best_seq = a_init.copy()
            print(f"  mode{mode} (n={n0}): obj={obj:.8f} **")
        else:
            print(f"  mode{mode} (n={n0}): obj={obj:.8f}")

    # Sort by score and keep top 4
    init_results.sort(key=lambda x: compute_obj(x[1]))
    top_inits = init_results[:4]
    print(f"  Top initializations: {[f'{name}:{compute_obj(a):.6f}' for name, a in top_inits]}")

    # --- Phase 2: Hybrid optimization on top candidates (700s) ---
    print("\n=== Phase 2: Hybrid Optimization ===")
    p2_deadline = start_time + 760.0
    n_candidates = len(top_inits)
    per_candidate_time = 700.0 / n_candidates

    for idx, (name, a_init) in enumerate(top_inits):
        if time.monotonic() > p2_deadline:
            break
        alloc = min(per_candidate_time, p2_deadline - time.monotonic())
        if alloc < 30:
            break
        print(f"  Optimizing {name} (budget={alloc:.0f}s)..."
              , end="", flush=True)
        a_opt = hybrid_optimize(a_init, time_budget=alloc, seed=seed + idx)
        obj_opt = compute_obj(a_opt)
        print(f" obj={obj_obj_str(obj_opt)}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"    ** New best: {best_score:.8f}")

    # --- Phase 3: LP refinement on best (300s) ---
    print(f"\n=== Phase 3: LP Refinement (best score: {best_score:.8f}) ===")
    p3_deadline = start_time + 1060.0

    for rnd in range(3):
        if time.monotonic() > p3_deadline:
            break
        alloc = min(100.0, p3_deadline - time.monotonic())
        if alloc < 20:
            break
        a_lp = lp_refine(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"    ** LP round {rnd}: {best_score:.8f}")
        else:
            print(f"    LP round {rnd}: {obj_lp:.8f} (no improvement)")

    # --- Phase 4: Multi-length exploration (250s) ---
    print(f"\n=== Phase 4: Multi-Length Exploration (best: {best_score:.8f}) ===")
    p4_deadline = start_time + 1310.0
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100),
        n_best + 100, n_best + 200,
        500, 700, 900, 1100, 1300
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500 and n != n_best]

    for n_try in n_options[:5]:
        if time.monotonic() > p4_deadline:
            break
        a_try = resample_sequence(best_seq, n_try)
        alloc = min(80.0, p4_deadline - time.monotonic())
        if alloc < 20:
            break

        # Quick mirror descent + LP
        md_alloc = alloc * 0.6
        lp_alloc = alloc * 0.35
        a_try = mirror_descent_optimize(a_try, time_budget=md_alloc, beta_start=5, beta_end=1000, seed=seed + n_try)
        if time.monotonic() < p4_deadline - 10:
            a_try = lp_refine(a_try, time_budget=lp_alloc)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f}")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # --- Phase 5: Final intensive polish (150s) ---
    print(f"\n=== Phase 5: Final Polish (best: {best_score:.8f}) ===")

    # Mirror descent with higher beta
    if time.monotonic() < deadline - 50:
        alloc = min(60.0, deadline - time.monotonic() - 50)
        a_md = mirror_descent_optimize(best_seq, time_budget=alloc, beta_start=50, beta_end=3000, seed=seed + 99)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            print(f"  MD polish: {best_score:.8f} **")

    # LP polish
    if time.monotonic() < deadline - 30:
        alloc = min(40.0, deadline - time.monotonic() - 30)
        a_lp = lp_refine(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP polish: {best_score:.8f} **")

    # Coordinate descent
    if time.monotonic() < deadline - 5:
        alloc = min(30.0, deadline - time.monotonic() - 5)
        a_cd = coordinate_descent_refine(best_seq, time_budget=alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD polish: {best_score:.8f} **")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result


def obj_obj_str(obj_val):
    """Helper to format objective value."""
    return f"{obj_val:.8f}"
