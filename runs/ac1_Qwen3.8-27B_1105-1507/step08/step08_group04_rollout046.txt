"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast LP alternating with adaptive blending, subgradient polishing,
Frank-Wolfe, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        for k in range(max(0, j), min(m, j + n)):
            A[k, j] = a[k - j]
    return A

def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None)."""
    n = len(a)
    m = 2 * n - 1
    a = a / float(np.sum(a))
    
    # Build A matrix (sparse would be better but dense is OK for n<=3000)
    A = np.zeros((m, n))
    for j in range(n):
        lo = max(0, j)
        hi = min(m, j + n)
        A[lo:hi, j] = a[:hi-lo]
    
    # LP: minimize M
    # Variables: g_0, ..., g_{n-1}, M
    # Objective: [0, 0, ..., 0, 1]
    # Inequality: A @ g - M <= 0  =>  [A, -I_m] @ [g; M] <= 0
    # Equality: sum(g) = 1  =>  [1, ..., 1, 0] @ [g; M] = 1
    # Bounds: g_i >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.eye(m)])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs > 1e-10:
            return g / gs, float(res.x[n])
    
    return None, None

    conv = compute_conv(a)
    if top_k >= len(conv):
        k_indices = np.arange(len(conv))
    else:
        k_indices = np.argsort(conv)[::-1][:top_k]
    w = np.zeros(2 * n - 1)
    for k in k_indices:
        w[int(k)] = 1.0 / len(k_indices)
    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1
    A_fft = rfft(a, n_fft)
    W_fft = rfft(w, n_fft)
    R = irfft(np.conj(A_fft) * W_fft, n_fft)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = R[(n_fft - j) % n_fft]
    grad *= 4.0 * n
    grad -= np.mean(grad)
    return grad

            if best_v != v_old:
                a[i] = best_v
                others_mask = np.arange(n) != i
                others_old_sum = float(np.sum(a[others_mask]))
                others_new_sum = 1.0 - best_v
                if others_old_sum > 1e-12 and others_new_sum > 1e-12:
                    a[others_mask] *= others_new_sum / others_old_sum
                a = np.maximum(a, 0)
                s_a = float(np.sum(a))
                if s_a > 1e-10:
                    a = a / s_a
                obj_new = compute_obj(a)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
        if verbose and pass_count % 2 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD pass {pass_count}: obj={best_obj:.8f}, time={elapsed:.0f}s")
        if not improved and pass_count > 2:
            break
    return best_a


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-30 * (x - 0.5) ** 2)
    elif choice == 1:
        a = 0.3 + 0.7 * (0.5 + 0.5 * np.cos(2 * np.pi * rng.integers(2, 8) * x))
    elif choice == 2:
        c1, c2 = rng.uniform(0.2, 0.4), rng.uniform(0.6, 0.8)
        a = np.exp(-50 * (x - c1) ** 2) + np.exp(-50 * (x - c2) ** 2)
    elif choice == 3:
        a = np.ones(n)
        for _ in range(5):
            center = rng.integers(0, n)
            width = rng.integers(n // 20, n // 5)
            for i in range(n):
                d = min(abs(i - center), n - abs(i - center))
                a[i] += rng.standard_normal() * np.exp(-d / (width * 0.3))
        a = np.maximum(a, 0.01)
    elif choice == 4:
        a = x ** rng.uniform(0.3, 2.0)
    else:
        a = np.ones(n)
        a[rng.integers(0, n // 2)] *= 0.3
        a[n // 2:] *= 1.5
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        n_base = len(a)
        print(f"Parent: n={n_base}, obj={compute_obj(a):.8f}")
    except Exception:
        n_base = 1000
        a = np.ones(n_base) / n_base
        print(f"Using uniform: n={n_base}, obj={compute_obj(a):.8f}")

    best_a = a.copy()
    best_obj = compute_obj(a)

    def remaining_time():
        return max(0.0, deadline - time.monotonic())

    # Phase 1: Primary LP alternating
    print(f"\n=== Phase 1: LP Alternating (n={n_base}) ===")
    p1_budget = min(remaining_time() * 0.40, 500.0)
    a_p1 = lp_alternate(best_a, time_budget=p1_budget, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  Phase 1: obj={best_obj:.8f} **")
    else:
        print(f"  Phase 1: obj={obj_p1:.8f}")

    # Phase 2: Coordinate descent polish
    print(f"\n=== Phase 2: Coordinate Descent ===")
    p2_budget = min(remaining_time() * 0.15, 200.0)
    if p2_budget > 30:
        a_cd = coordinate_descent(best_a, time_budget=p2_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd
            print(f"  CD: obj={best_obj:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # Phase 3: Subgradient polish
    print(f"\n=== Phase 3: Subgradient ===")
    p3_budget = min(remaining_time() * 0.10, 120.0)
    if p3_budget > 20:
        for tk in [10, 20]:
            if remaining_time() < 20:
                break
            alloc = min(p3_budget / 2.0, remaining_time() * 0.45)
            a_sg = subgradient_optimize(best_a, time_budget=alloc, top_k=tk)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg
                print(f"  SubGrad tk={tk}: obj={best_obj:.8f} **")
            else:
                print(f"  SubGrad tk={tk}: obj={obj_sg:.8f}")

    # Phase 4: LP round 2 from improved point
    print(f"\n=== Phase 4: LP Round 2 ===")
    p4_budget = min(remaining_time() * 0.20, 250.0)
    if p4_budget > 30:
        a_p4 = lp_alternate(best_a, time_budget=p4_budget, verbose=True)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_obj:
            best_obj = obj_p4
            best_a = a_p4
            print(f"  LP2: obj={best_obj:.8f} **")
        else:
            print(f"  LP2: obj={obj_p4:.8f}")

    # Phase 5: Multi-n exploration
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    n_best = len(best_a)
    n_candidates = sorted(set([
        max(400, n_best - 200), max(400, n_best - 100),
        n_best,
        n_best + 100, n_best + 300, n_best + 500,
        n_best + 800, n_best + 1200,
    ]))
    n_candidates = [x for x in n_candidates if 400 <= x <= 4000]
    for n_try in n_candidates:
        if remaining_time() < 30:
            break
        if n_try == n_best:
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
        alloc = min(40.0, remaining_time() * 0.20)
        a_opt = lp_alternate(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_opt)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # Phase 6: Diverse initializations at best n
    print(f"\n=== Phase 6: Diverse Inits (n={n_best}) ===")
    for d in range(6):
        if remaining_time() < 25:
            break
        a_div = diverse_init(n_best, rng)
        alloc = min(35.0, remaining_time() * 0.15)
        a_opt = lp_alternate(a_div, time_budget=alloc, verbose=False)
        obj_div = compute_obj(a_opt)
        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Div {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Div {d}: obj={obj_div:.8f}")

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Final Polish ===")
    if remaining_time() > 30:
        alloc_lp = min(remaining_time() * 0.35, 80.0)
        a_fin = lp_alternate(best_a, time_budget=alloc_lp, verbose=True)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin
            print(f"  Final LP: obj={best_obj:.8f} **")

    if remaining_time() > 20:
        alloc_cd = min(remaining_time() * 0.3, 50.0)
        a_cd2 = coordinate_descent(best_a, time_budget=alloc_cd, verbose=False)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_obj:
            best_obj = obj_cd2
            best_a = a_cd2
            print(f"  Final CD: obj={best_obj:.8f} **")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")
    return result
