def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0"""
    n = len(a)
    a = a / np.sum(a)
    
    # Build A matrix: A[k, j] such that (A @ g)[k] = conv(a, g)[k]
    # conv(a, g)[k] = sum_j a[k-j] * g[j] where 0<=k-j<=n-1
    # So A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    
    # Using scipy linprog: min c^T x s.t. A_ub @ x <= b_ub, A_eq @ x = b_eq, bounds
    # Variables: x = [g_0, ..., g_{n-1}, M]
    # c = [0, ..., 0, 1]
    
    # Inequalities: conv(a,g)[k] - M <= 0 for each k
    # So A_ub[:, :n] = A_conv, A_ub[:, n] = -1
    # b_ub = 0
    
    # Equality: sum(g) = 1
    # A_eq = [1, ..., 1, 0]
    # b_eq = [1]
    
    # Bounds: g_j >= 0, M free
    
    # Build A_conv efficiently (sparse)
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
        # M column
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_conv = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n+1))
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_conv, b_ub=np.zeros(2*n-1),
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x[:n]
        M = result.x[n]
        g = np.maximum(g, 0)
        s = float(np.sum(g))
        if s > 1e-10:
            g = g / s
        return g, float(M)
    return None, None

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):

def smoothed_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*conv(a,a)[k])) via FFT.
    Returns (gradient, conv).
    """
    n = len(a)
    conv = compute_conv(a)
    
    max_conv = float(np.max(conv))
    exp_vals = np.exp(tau * (conv - max_conv))
    p = exp_vals / float(np.sum(exp_vals))
    
    # Cross-correlation: g[i] = sum_k p[k] * a[k-i]
    # Use FFT: ifft(conj(fft(p, N)) * fft(a, N)).real[:n]
    N = 1
    while N < 3 * n:
        N <<= 1
    
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(np.conj(P) * A).real
    g = 2.0 * g_full[:n]
    
    return g, conv


def nesterov_smoothed_descent(a, time_budget=150.0, tau_init=5.0, tau_end=500.0):
    """Nesterov accelerated gradient descent on smoothed objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    beta = 0.9
    lr = 3.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)
        
        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y
        
        g, conv = smoothed_gradient(y, tau)
        
        # BB step size
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.5 * lr + 0.5 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))
        
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)

            a_new = a_new / s_new

        a = a_new
        it += 1

    print(f"    LP Polish done: obj={best_obj:.8f}, iters={iter_count}, time={time.monotonic()-start:.0f}s")
    return best_a


def subgradient_refine(a, time_budget=100.0, K=5):
    """Subgradient method with top-K peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 2.0 / np.sqrt(n)
    momentum = 0.7
    v = np.zeros(n)
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        
        peak_vals = conv[top_idx]
        max_pv = peak_vals[0]
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 5.0 / max_pv)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
        
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.995
        
        it += 1
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
