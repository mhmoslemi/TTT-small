def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

                    a[idx] = orig_val

                a_norm = a / st

                    best_local_val = cv

def nesterov_subgrad(a0, time_budget=60.0, top_k=5, verbose=False):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.1
    patience = 0

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                m_start = k - i_end
                m_end = k - i_start
                grad[m_start:m_end + 1] += 2.0 * y[i_start:i_end + 1][::-1]
        grad /= len(top_idx)
        grad = grad - np.mean(grad)
        y_new = y - step_size * grad
        a_new = project_simplex(y_new)
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = project_simplex(y)
        a = a_new
        if it % 5 == 0 or it < 10:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 400:
                step_size = max(step_size * 0.7, 0.005)
                patience = 0
        if it % 1000 == 999:
            step_size = max(step_size * 0.97, 0.003)
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"      Nesterov it={it}: best={best_obj:.8f}, step={step_size:.4f}, t={elapsed:.1f}s")

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def supergaussian(n, p=6.0):
    """Super-Gaussian: (1 + (x/sigma)^2)^(-p)"""
    x = np.linspace(-1, 1, n)
    sigma = 0.5
    a = (1.0 + (x / sigma) ** 2) ** (-p)
    return a / np.sum(a)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def blackman_harris(n):
    """Blackman-Harris window normalized to sum=1."""
    x = np.arange(n)
    a0, a1, a2, a3 = 0.35875, 0.48829, 0.14128, 0.01168
    w = a0 - a1 * np.cos(2 * np.pi * x / (n - 1)) + a2 * np.cos(4 * np.pi * x / (n - 1)) - a3 * np.cos(6 * np.pi * x / (n - 1))
    w = np.maximum(w, 0)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def bimodal(n, gap=0.2, width=0.15):
    """Bimodal distribution with two peaks."""
    x = np.linspace(0, 1, n)
    c1 = 0.5 - gap
    c2 = 0.5 + gap
    a = np.exp(-((x - c1) / width) ** 2) + np.exp(-((x - c2) / width) ** 2)
    a = np.maximum(a, 0)
    return a / np.sum(a)

def compute_conv(a):

    n = len(a)

def compute_obj(a):

    n = len(a)

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"      Nesterov it={it}: best={best_obj:.8f}, step={step_size:.4f}, t={elapsed:.1f}s")

    return best_a


def lp_alternate(a0, time_budget=60.0, verbose=False):
    """
    LP alternating with grid-assisted line search and stall detection.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    max_stall = 25

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Grid-assisted line search
        t_grid = np.linspace(0, 1, 15)
        obj_grid = np.inf
        best_t_grid = 0.0
        for t in t_grid:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                obj_try = np.inf
            else:
                a_try = a_try / st
                obj_try = compute_obj(a_try)
            if obj_try < obj_grid:
                obj_grid = obj_try
                best_t_grid = t

        # Refine around best grid point
        t_lo = max(0.0, best_t_grid - 0.2)
        t_hi = min(1.0, best_t_grid + 0.2)

        def line_eval(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)

        try:
            res = minimize_scalar(line_eval, bounds=(t_lo, t_hi), method='bounded',
                                  options={'maxiter': 40, 'xatol': 0.002})
            best_t = res.x
        except Exception:
            best_t = best_t_grid

        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall_count = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > max_stall:
                break

    return best_a


def coordinate_descent(a0, time_budget=30.0, verbose=False):
    """
    Coordinate descent: for each coordinate, try multiple candidate values
    and keep the best improvement.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    rng_local = np.random.default_rng(123)
    improved_any = True
    max_rounds = 50
    round_idx = 0

    while improved_any and round_idx < max_rounds:
        if time.monotonic() - start > time_budget:
            break
        improved_any = False
        round_idx += 1
        order = np.arange(n)
        rng_local.shuffle(order)

        for idx in order:
            if time.monotonic() - start > time_budget:
                break
            idx = int(idx)
            orig = a[idx]
            # Generate candidate values
            cands = []
            cands.append(0.0)
            if orig > 1e-6:
                cands.extend([orig * f for f in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 3.0]])
                cands.extend([orig * (1.0 + 0.1 * d) for d in range(-3, 4)])
            cands.extend([0.001, 0.01, 0.05, 0.1, 0.5])
            # Remove duplicates and sort
            cands = sorted(set(max(0.0, c) for c in cands))

            best_local = best_obj
            best_val = orig

            for cv in cands:
                a[idx] = cv
                st = float(np.sum(a))
                if st < 1e-10:
                    a[idx] = orig
                    continue
                a_norm = a / st
                obj_try = compute_obj(a_norm)
                if obj_try < best_local - 1e-14:
                    best_local = obj_try
                    best_val = cv

            a[idx] = best_val
            st = float(np.sum(a))
            if st > 1e-10:
                a = a / st
            if compute_obj(a) < best_obj - 1e-14:
                best_obj = compute_obj(a)
                best_a = a.copy()
                improved_any = True

        if verbose:
            elapsed = time.monotonic() - start
            print(f"      CD round={round_idx}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def blackman_harris(n):
    """Blackman-Harris window normalized to sum=1."""
    x = np.arange(n)
    a0, a1, a2, a3 = 0.35875, 0.48829, 0.14128, 0.01168
    denom = max(1, n - 1)
    w = a0 - a1 * np.cos(2 * np.pi * x / denom) + a2 * np.cos(4 * np.pi * x / denom) - a3 * np.cos(6 * np.pi * x / denom)
    w = np.maximum(w, 0)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def supergaussian(n, p=6.0):
    """Super-Gaussian: (1 + (x/sigma)^2)^(-p) normalized."""
    x = np.linspace(-1, 1, n)
    sigma = 0.5
    a = (1.0 + (x / sigma) ** 2) ** (-p)
    return a / np.sum(a)


def bimodal(n, gap=0.2, width=0.15):
    """Bimodal distribution with two peaks."""
    x = np.linspace(0, 1, n)
    c1 = 0.5 - gap
    c2 = 0.5 + gap
    a = np.exp(-((x - c1) / width) ** 2) + np.exp(-((x - c2) / width) ** 2)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining diverse initializations, LP alternating,
    coordinate descent, Nesterov subgradient, and n-sweeping.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = hann_window(800)
        best_score = compute_obj(best_seq)
        print(f"Hann init (n=800): obj={best_score:.8f}")

    n_base = len(best_seq)

    # === Phase 1: Diverse Initializations ===
    print(f"\n=== Phase 1: Diverse Initializations ===")
    inits = {}
    inits['parent'] = best_seq.copy()
    inits['hann'] = hann_window(n_base)
    inits['blackman_harris'] = blackman_harris(n_base)
    inits['super4'] = supergaussian(n_base, p=4.0)
    inits['super6'] = supergaussian(n_base, p=6.0)
    inits['super8'] = supergaussian(n_base, p=8.0)
    inits['super12'] = supergaussian(n_base, p=12.0)
    inits['super16'] = supergaussian(n_base, p=16.0)
    x = np.linspace(-1, 1, n_base)
    gauss = np.exp(-3.0 * x ** 2)
    inits['gauss'] = gauss / np.sum(gauss)
    inits['bimodal'] = bimodal(n_base, gap=0.2, width=0.15)
    inits['bimodal_wide'] = bimodal(n_base, gap=0.3, width=0.2)
    tri = np.minimum(np.arange(n_base) + 1.0, n_base - np.arange(n_base))
    inits['tri'] = tri / np.sum(tri)
    rand = np.maximum(rng.random(n_base), 0)
    inits['random'] = rand / np.sum(rand)

    for name, init in inits.items():
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        obj = compute_obj(init)
        if obj < best_score:
            best_score = obj
            best_seq = init.copy()
            print(f"  {name}: obj={best_score:.8f} **")
        else:
            print(f"  {name}: obj={obj:.8f}")

    # Quick Nesterov on top-3 initializations
    scored_inits = [(compute_obj(v), k, v) for k, v in inits.items()]
    scored_inits.sort(key=lambda x: x[0])
    for _, name, init in scored_inits[:3]:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(15.0, remaining * 0.25)
        a_opt = nesterov_subgrad(init, time_budget=alloc, top_k=5)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  {name}+Nest: obj={best_score:.8f} **")

    # === Phase 2: n-Sweep ===
    print(f"\n=== Phase 2: n-Sweep ===")
    n_options = [300, 400, 500, 600, 700, 800, 900, 1000, 1200, 1400, 1600, 2000]
    n_options = sorted(set(n_options + [n_base]))
    n_options = [nv for nv in n_options if 200 <= nv <= 2000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        if n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
        alloc = min(8.0, remaining * 0.12)
        a_opt = nesterov_subgrad(a_try, time_budget=alloc, top_k=5)
        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 3: Main LP Alternating ===
    print(f"\n=== Phase 3: Main LP Alternating ===")
    a = best_seq.copy()
    n_cur = len(a)

    for round_idx in range(15):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(50.0, remaining * 0.22)
        a_new = lp_alternate(a, time_budget=alloc, verbose=(round_idx % 2 == 0))
        obj = compute_obj(a_new)
        if obj < best_score - 1e-13:
            best_score = obj
            best_seq = a_new.copy()
            a = best_seq
            n_cur = len(a)
            print(f"  LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj:.8f}")
            # Perturb
            scale = 0.008 + 0.003 * round_idx
            a = best_seq * np.exp(rng.normal(0, scale, n_cur))
            a = np.maximum(a, 0)
            a /= np.sum(a)

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(50.0, remaining * 0.25)
        a_cd = coordinate_descent(best_seq, time_budget=alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score - 1e-13:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 5: Nesterov with various top_k ===
    print(f"\n=== Phase 5: Nesterov Polish ===")
    for k_val in [3, 5, 8, 15, 25]:
        remaining = deadline - time.monotonic()
        if remaining < 12:
            break
        alloc = min(30.0, remaining * 0.18)
        a_ns = nesterov_subgrad(best_seq, time_budget=alloc, top_k=k_val)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score - 1e-13:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nest k={k_val}: obj={best_score:.8f} **")
        else:
            print(f"  Nest k={k_val}: obj={obj_ns:.8f}")

    # === Phase 6: Another LP round after polish ===
    print(f"\n=== Phase 6: LP After Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        a = best_seq.copy()
        for round_idx in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(35.0, remaining * 0.25)
            a_new = lp_alternate(a, time_budget=alloc, verbose=False)
            obj = compute_obj(a_new)
            if obj < best_score - 1e-13:
                best_score = obj
                best_seq = a_new.copy()
                a = best_seq
                print(f"  LP R{round_idx}: obj={best_score:.8f} **")
            else:
                print(f"  LP R{round_idx}: obj={obj:.8f}")
                a = best_seq * np.exp(rng.normal(0, 0.008, len(a)))
                a = np.maximum(a, 0)
                a /= np.sum(a)

    # === Phase 7: Final Coordinate Descent ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(30.0, remaining * 0.4)
        a_cd2 = coordinate_descent(best_seq, time_budget=alloc)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score - 1e-13:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD2: obj={best_score:.8f} **")
        else:
            print(f"  CD2: obj={obj_cd2:.8f}")

    # === Phase 8: Random Restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 8: Random Restarts ===")
        n_best = len(best_seq)
        for r in range(12):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            scale = 0.008 + 0.004 * r
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr < 1e-10:
                continue
            a_r = a_r / sr
            alloc = min(remaining * 0.15, 10.0)
            if r % 3 == 0:
                a_opt = lp_alternate(a_r, time_budget=alloc * 0.6)
                a_opt = nesterov_subgrad(a_opt, time_budget=alloc * 0.4, top_k=5)
            elif r % 3 == 1:
                a_opt = nesterov_subgrad(a_r, time_budget=alloc * 0.4, top_k=5)
                a_opt = lp_alternate(a_opt, time_budget=alloc * 0.6)
            else:
                a_opt = lp_alternate(a_r, time_budget=alloc * 0.5)
                a_opt = coordinate_descent(a_opt, time_budget=alloc * 0.5)
            obj_r = compute_obj(a_opt)
            if obj_r < best_score - 1e-13:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  R{r}: obj={best_score:.8f} **")
            else:
                print(f"  R{r}: obj={obj_r:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
