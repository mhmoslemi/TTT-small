```python
    n = len(a_init)
    # Initialize x from a_init
    a = a_init.copy()
    a = np.maximum(a, 1e-15)  # avoid log(0)
    a = a / a.sum()
    x = np.log(a)  # so that softmax(x) ≈ a (up to normalization)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.01
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for t in range(max_iters):
        # Compute softmax
        x_centered = x - x.mean()  # fix gauge
        x_max = x_centered.max()
        exp_x = np.exp(x_centered - x_max)
        a = exp_x / exp_x.sum()
        
        # Compute smooth max gradient
        conv_a = compute_conv(a)
        T = T_schedule(t)
        shifted = (conv_a - conv_a.max()) / T
        p = np.exp(shifted)
        p = p / p.sum()
        
        # Gradient of smooth max w.r.t. a
        grad_a = 2 * cross_correlate(p, a)  # as derived above
        
        # Gradient w.r.t. x: chain rule
        # d(a_i)/d(x_j) = a_i * (delta_ij - a_j)
        # So dF/dx_j = sum_i (dF/da_i) * a_i * (delta_ij - a_j)
        # = a_j * (grad_a_j - sum_i grad_a_i * a_i)
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)
        
        # Project out the null direction (all-ones) to fix gauge
        grad_x = grad_x - grad_x.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x**2
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - x.mean()  # fix gauge
        
        # Evaluate
        a_new = softmax(x)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    return best_a
```

where `p` has length `2n-1` and `a` has length `n`.

Using FFT:
```python
def cross_correlate_peaks(p, a):
    """Compute grad[m] = sum_k p[k] * a[k-m] for m=0..n-1
    where p has length 2n-1 and a has length n."""
    n = len(a)
    # Use the formula: grad = 2 * conv_result[n-1:n-1+n]
    # where conv_result = conv(p, flip(a))
    size = 1
    while size < len(p) + n:
        size <<= 1
    P = rfft(p, size)
    A = rfft(a[::-1], size)
    conv_result = irfft(P * A, size)
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad
```

Let me verify this is correct with a small example.

`p = [p0, p1, p2]` (length 3, so n=2)
`a = [a0, a1]` (length 2)

`grad[0] = sum_k p[k] * a[k] = p0*a0 + p1*a1 + p2*a2(=0) = p0*a0 + p1*a1`
`grad[1] = sum_k p[k] * a[k-1] = p0*a(-1)(=0) + p1*a0 + p2*a1 = p1*a0 + p2*a1`

Using the formula: `flip(a) = [a1, a0]`
`conv(p, flip(a))[k] = sum_i p[i] * flip(a)[k-i]`
`conv[0] = p0*a1`
`conv[1] = p0*a0 + p1*a1`
`conv[2] = p1*a0 + p2*a1`
`conv[3] = p2*a0`

`grad = 2 * conv_result[n-1:n-1+n] = 2 * conv_result[1:3] = 2 * [p0*a0+p1*a1, p1*a0+p2*a1]`

That matches! Great.

Actually, let me reconsider. The LP alternating method is probably the strongest single method. Let me keep it but improve it, and add the Adam and simulated annealing as complementary methods.

Let me also think about **block updates** for coordinate descent. Instead of updating one coordinate at a time, update a block of B consecutive coordinates. For a block `a[i:i+B]`, we want to find the optimal values that minimize the objective. This is a small sub-problem that can be solved by:
- Grid search over the block (if B is small)
- Local LP (fixing the rest of the sequence)
- Gradient descent on the block

For B=2 or B=3, we can do a 2D or 3D local optimization.

Actually, for B=2: we fix all coordinates except `a[i]` and `a[i+1]`, and `a[i] + a[i+1] = c` (to maintain the sum). Then we minimize `max_k conv(a,a)[k]` over `a[i] ∈ [0, c]`. This is a 1D problem! We can solve it with golden section search or ternary search.

Wait, but `max_k conv(a,a)[k]` is the max of quadratic functions, so it's a convex function of `a[i]` (for fixed other coordinates). So we can use ternary search!

This is a key insight: **1D ternary search for pairwise coordinate updates**.

Let me formalize:
- Fix all coordinates except `a[i]` and `a[j]`
- Constraint: `a[i] + a[j] = c` (to maintain sum)
- Parameterize: `a[i] = t * c`, `a[j] = (1-t) * c` for `t ∈ [0,1]`
- The objective `max_k conv(a,a)[k]` is convex in `t` (max of convex functions, each of which is quadratic in `t`)
- Use ternary search on `t ∈ [0,1]` to find the optimal split

This is much more efficient than single-coordinate updates because it can move mass between two coordinates in one step.

"""
Optimized step function search using:
1. LP alternating method (two-block coordinate descent)
2. Adam optimizer on softmax parameterization with smooth max
3. Pairwise coordinate descent with ternary search (convex 1D subproblems)
4. Simulated annealing with exponential family updates
5. Multi-n exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_full(a):
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def lp_alternating(a, time_budget=200.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    
    max_iters = 1000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp_full(a)
        if g_lp is None:
            break
        
        # Convex line search
        a_norm = a / np.sum(a)
        g_norm = g_lp / np.sum(g_lp)
        
        conv_aa = compute_conv(a_norm)
        conv_ag = compute_conv_cross(a_norm, g_norm)
        conv_gg = compute_conv(g_norm)
        
        def obj_at_t(t):
            val = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
            return 2.0 * n * float(np.max(val))
        
        lo, hi = 0.0, 1.0
        for _ in range(12):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if obj_at_t(m1) < obj_at_t(m2):
                hi = m2
            else:
                lo = m1
        t_opt = (lo + hi) / 2.0
        
        a_new = (1 - t_opt) * a_norm + t_opt * g_norm
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 30:
                # Perturb
                rng = np.random.default_rng(it * 7919 + n)
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0
    
    return best_a


def cross_correlate_peaks(p, a):
    """Compute grad[m] = 2 * sum_k p[k] * a[k-m] for m=0..n-1."""
    n = len(a)
    size = 1
    while size < len(p) + n:
        size <<= 1
    P = rfft(p, size)
    A = rfft(a[::-1], size)
    conv_result = irfft(P * A, size)
    return 2.0 * conv_result[n-1:n-1+n]


def adam_softmax(a_init, time_budget=300.0, verbose=False):
    """Adam optimizer on softmax parameterization with smooth max."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy().astype(np.float64)
    a = a / a.sum()
    x = np.log(a)
    x = x - x.mean()
    
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = 0.05
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t_step = 0
    stall = 0
    
    T = 0.001  # smooth max temperature
    T_decay = 0.9995
    
    max_iters = 50000
    while t_step < max_iters and time.monotonic() - start < time_budget:
        # Compute a from x
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        a = exp_x / exp_x.sum()
        
        # Compute smooth max gradient
        conv_a = compute_conv(a)
        max_c = conv_a.max()
        shifted = (conv_a - max_c) / T
        shifted = np.clip(shifted, -50, 50)
        p = np.exp(shifted)
        p = p / p.sum()
        
        # Gradient w.r.t. a
        grad_a = cross_correlate_peaks(p, a)
        
        # Gradient w.r.t. x (chain rule through softmax)
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)
        grad_x = grad_x - grad_x.mean()  # fix gauge
        
        # Adam update
        t_step += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad_x
        v_adam = beta2 * v_adam + (1 - beta2) * grad_x**2
        m_hat = m_adam / (1 - beta1**t_step)
        v_hat = v_adam / (1 - beta2**t_step)
        
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - x.mean()
        
        # Evaluate
        x_shifted = x - x.max()
        a_new = np.exp(x_shifted)
        a_new = a_new / a_new.sum()
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.7
                stall = 0
                if lr < 1e-6:
                    break
        
        T = max(0.00005, T * T_decay)
        
        if verbose and t_step % 5000 == 0:
            print(f"    Adam step {t_step}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}")
    
    return best_a


def pairwise_cd(a, time_budget=120.0, verbose=False):
    """Pairwise coordinate descent with ternary search.
    
    For each pair (i,j), fix all other coordinates, and optimize
    the split of the mass between a[i] and a[j] using ternary search
    on the convex function max_k conv(a,a)[k].
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Precompute: for each pair (i,j), the contribution to conv(a,a)
    # from the cross terms involving a[i] and a[j]
    # conv(a,a)[k] = base_k + a[i]*contrib_i_k + a[j]*contrib_j_k + a[i]*a[j]*cross_ij_k
    # where base_k is conv without a[i] and a[j]
    
    # For efficiency, we'll use a random pair selection strategy
    rng = np.random.default_rng(42)
    
    max_passes = 200
    pass_num = 0
    while time.monotonic() - start < time_budget and pass_num < max_passes:
        pass_num += 1
        improved = False
        
        # Generate random pairs
        num_pairs = min(n, 500)
        for _ in range(num_pairs):
            if time.monotonic() - start > time_budget:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            c = a[i] + a[j]
            if c < 1e-12:
                continue
            
            # Ternary search on t in [0,1]: a[i] = t*c, a[j] = (1-t)*c
            lo, hi = 0.0, 1.0
            for _ in range(25):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                
                a1 = a.copy()
                a1[i] = m1 * c
                a1[j] = (1 - m1) * c
                obj1 = compute_obj(a1)
                
                a2 = a.copy()
                a2[i] = m2 * c
                a2[j] = (1 - m2) * c
                obj2 = compute_obj(a2)
                
                if obj1 < obj2:
                    hi = m2
                else:
                    lo = m1
            
            t_opt = (lo + hi) / 2.0
            a_new = a.copy()
            a_new[i] = t_opt * c
            a_new[j] = (1 - t_opt) * c
            obj_new = compute_obj(a_new)
            
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
        
        if verbose and pass_num % 10 == 0:
            print(f"    Pairwise CD pass {pass_num}: obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")
        
        if not improved and pass_num > 5:
            break
    
    return best_a


def simulated_annealing(a_init, time_budget=120.0, verbose=False):
    """Simulated annealing with multiplicative (exponential family) updates."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    T = 0.01  # initial temperature
    T_final = 0.0001
    cooling = 0.9995
    
    max_iters = 100000
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        if T < T_final:
            break
        
        # Generate a random multiplicative perturbation
        rng = np.random.default_rng(it + n)
        # Perturb a random subset of coordinates
        num_perturb = max(1, n // 100)
        indices = rng.integers(0, n, num_perturb)
        
        a_new = a.copy()
        for idx in indices:
            factor = np.exp(rng.normal(0, 0.1))
            a_new[idx] *= factor
        
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        delta = obj_new - best_obj
        
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            a = a_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        T *= cooling
        
        if verbose and it % 10000 == 0:
            print(f"    SA iter {it}: obj={best_obj:.8f}, T={T:.6f}")
    
    return best_a


def block_cd(a, time_budget=120.0, block_size=4, verbose=False):
    """Block coordinate descent: optimize blocks of consecutive coordinates."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        # Random block
        i = rng.integers(0, n - block_size)
        block = slice(i, i + block_size)
        c = a[block].sum()
        if c < 1e-12:
            continue
        
        # Optimize the block using small gradient steps
        # (for small blocks, we can do local search)
        a_tried = a.copy()
        for _ in range(20):
            # Random direction in the block
            direction = rng.normal(0, 0.1, block_size)
            direction = direction - direction.mean()  # maintain sum
            step = 0.01
            a_try = a_tried.copy()
            a_try[block] = a_tried[block] + step * direction
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < compute_obj(a_tried) - 1e-15:
                a_tried = a_try
            else:
                step *= 0.5
        
        obj_tried = compute_obj(a_tried)
        if obj_tried < best_obj - 1e-14:
            best_obj = obj_tried
            best_a = a_tried.copy()
            a = a_tried
    
    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
