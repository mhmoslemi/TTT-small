def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_lp_scipy(a):
    """LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    from scipy.optimize import linprog
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def convex_line_search(a, g, n_evals=12):
    """
    Find optimal t in [0,1] for a_new = (1-t)*a_norm + t*g_norm.
    Uses the quadratic structure of the convolution.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    s_g = float(np.sum(g))
    if s_g < 1e-10:
        return 0.0, a_norm
    g_norm = g / s_g

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2.0 * t * (1.0 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm
    a_new = a_new / s
    return t_opt, a_new

def subgradient_polish(a, time_budget=60.0, verbose=False):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diversify(a, rng, scale=0.05):
    """Create a diversified version of the sequence."""
    n = len(a)
    a_div = a * np.exp(rng.normal(0, scale, n))
    a_div = np.maximum(a_div, 0)
    s = float(np.sum(a_div))
    if s < 1e-10:
        return a.copy()
    return a_div / s

            print(f"  Phase 3 done: obj={obj_opt:.8f} (no improvement)")

    # === Phase 4: Exact CD again ===
    print(f"\n=== Phase 4: Exact CD round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        phase4_budget = min(200.0, remaining * 0.2)
        a_cd = coordinate_descent_exact(best_seq, time_budget=phase4_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Phase 4 done: obj={best_score:.8f}")
        else:
            print(f"  Phase 4 done: obj={obj_cd:.8f} (no improvement)")

    # === Phase 5: Subgradient polish ===
    print(f"\n=== Phase 5: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        phase5_budget = min(60.0, remaining * 0.1)
        a_sg = subgradient_polish(best_seq, time_budget=phase5_budget)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Phase 5 done: obj={best_score:.8f}")
        else:
            print(f"  Phase 5 done: obj={obj_sg:.8f} (no improvement)")

    # === Phase 6: Diversified restarts ===
    print(f"\n=== Phase 6: Diversified Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for restart_i in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        scale = 0.02 + 0.03 * restart_i
        a_div = diversify(best_seq, rng, scale=scale)

        # Quick LP + CD
        alloc_lp = min(remaining * 0.15, 60.0)
        alloc_cd = min(remaining * 0.2, 80.0)

        a_r = lp_alternating(a_div, time_budget=alloc_lp)
        obj_r = compute_obj(a_r)

        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_r = coordinate_descent_exact(a_r, time_budget=min(alloc_cd, remaining * 0.5))
            obj_r = compute_obj(a_r)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {restart_i}: obj={best_score:.8f} ** IMPROVED **")
        else:
            print(f"  Restart {restart_i}: obj={obj_r:.8f}")

    # === Phase 7: Multi-n Exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    n_options = []
    for dn in [-100, -50, -25, 0, 25, 50, 100, 150, 200]:
        n_try = n_best + dn
        if 400 <= n_try <= 2000:
            n_options.append(n_try)
    # Add some fixed values
    for n_fixed in [500, 600, 700, 800, 900, 1000, 1200, 1500]:
        if 400 <= n_fixed <= 2000 and n_fixed not in n_options:
            n_options.append(n_fixed)

    n_options = sorted(set(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # Quick optimization
        alloc_lp = min(remaining * 0.1, 30.0)
        alloc_cd = min(remaining * 0.15, 40.0)

        a_r = lp_alternating(a_try, time_budget=alloc_lp)
        obj_r = compute_obj(a_r)

        remaining = deadline - time.monotonic()
        if remaining > 20 and obj_r < best_score + 0.01:
            a_r = coordinate_descent_exact(a_r, time_budget=min(alloc_cd, remaining * 0.4))
            obj_r = compute_obj(a_r)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} ** IMPROVED **")
        else:
            print(f"  n={n_try}: obj={obj_r:.8f}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # LP polish
        if remaining > 40:
            alloc = min(60.0, remaining * 0.3)
            a_lp = lp_alternating(best_seq, time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: obj={best_score:.8f}")

        # CD polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(80.0, remaining * 0.4)
            a_cd = coordinate_descent_exact(best_seq, time_budget=alloc)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: obj={best_score:.8f}")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
