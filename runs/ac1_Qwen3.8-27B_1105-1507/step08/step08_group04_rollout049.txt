Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with convex 1D line search, coordinate descent polishing,

import numpy as np

def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A

def coordinate_descent_polish(a, time_budget=60.0, verbose=False):
    """Polish using coordinate descent: optimize one element at a time.
    For each element a[i], the objective is a max of convex quadratics in a[i],
    so we can find the optimal value by checking kink points."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # For coordinate descent on the simplex, we need to maintain sum(a) = 1
    # Strategy: optimize a[i] by shifting mass from the "average" other element
    # Or better: optimize a[i] and redistribute the difference uniformly
    
    for sweep in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            # Try changing a[i] by delta, and adjust a[(i+1)%n] by -delta
            # to maintain the sum
            j = (i + 1) % n
            
            # Binary search for optimal delta
            # The objective as a function of delta is convex (max of quadratics)
            # So we can use ternary search or just try a few values
            
            best_delta = 0.0
            best_local_obj = best_obj
            
            # Try a range of deltas
            scale = max(a[i], a[j], 0.01) * 0.1
            for delta in np.linspace(-scale, scale, 7):
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_local_obj - 1e-12:
                    best_local_obj = obj_try
                    best_delta = delta
            
            if best_delta != 0.0:
                a[i] += best_delta
                a[j] -= best_delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_obj = best_local_obj
                best_a = a.copy()
                improved = True
        
        if verbose and sweep % 10 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, improved={improved}")
        if not improved and sweep > 3:
            break
    
    return best_a

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def subgradient_step(a, top_k=20):
    """Compute subgradient of 2n*max(conv(a,a)) at a (sum(a)=1)."""
    n = len(a)
    conv = compute_conv(a)
    if top_k >= len(conv):
        k_indices = np.arange(len(conv))
    else:
        k_indices = np.argsort(conv)[::-1][:top_k]
    
    w = np.zeros(2 * n - 1)
    for k in k_indices:
        w[int(k)] = 1.0 / len(k_indices)
    
    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1
    
    A_fft = rfft(a, n_fft)
    W_fft = rfft(w, n_fft)
    R = irfft(np.conj(A_fft) * W_fft, n_fft)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = R[(n_fft - j) % n_fft]
    grad *= 4.0 * n
    grad -= np.mean(grad)
    return grad

def subgradient_optimize(a, time_budget=60.0, top_k=20, verbose=False):
    """Nesterov-accelerated subgradient method."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.5 / n
    x = a.copy()
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        grad = subgradient_step(x, top_k=top_k)
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        t_k = 2.0 / (it + 2.0)
        x = a_new + (1 - t_k) * (a_new - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx
        
        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
        
        if it % 200 == 199:
            lr *= 0.98
    
    return best_a

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with convex 1D line search, peak-equalization polishing,
adaptive multi-n exploration, and diverse restarts.
Key improvements over baseline:

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp(a):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g_normalized or None."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.SCIP]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > 180:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs
        except Exception:
            continue

    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:

    return None

    The function is convex in t when a,g >= 0 and sum(a)=sum(g)=1."""

        a_try = (1.0 - t) * a + t * g
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        val = compute_obj(a_try)
        if val < best_val:
            best_val = val
            best_t = t

    # Fine refinement (3 rounds)
    for _ in range(3):
        if time.monotonic() - start > max_time:
            break
        span = 0.05
        lo = max(0.0, best_t - span)
        hi = min(1.0, best_t + span)
        t_grid = np.linspace(lo, hi, 15)
        for t in t_grid:
            a_try = (1.0 - t) * a + t * g
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            val = compute_obj(a_try)
            if val < best_val:
                best_val = val
                best_t = t
        span *= 0.4

    return best_t, best_val


def lp_alternate(a, time_budget=200.0, verbose=False):
    """LP alternating with convex line search and adaptive perturbation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    iteration = 0

    while time.monotonic() - start < time_budget:
        g_lp = solve_lp(a)
        if g_lp is None:
            break

        best_t, best_val = convex_line_search(a, g_lp, max_time=3.0)

        if best_val < best_obj - 1e-13:
            a_new = (1.0 - best_t) * a + best_t * g_lp
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            best_obj = best_val
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            iteration += 1
            if verbose and (iteration <= 5 or iteration % 10 == 0):
                elapsed = time.monotonic() - start
                print(f"    LP it={iteration}: obj={best_obj:.8f}, t={best_t:.3f}, time={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 4:
                # Structured perturbation
                rng = np.random.default_rng((iteration * 104729 + int(time.monotonic() * 1000)) % (2**31))
                scale = min(0.03 + 0.01 * (stall - 4), 0.2)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    obj_p = compute_obj(a)
                    if obj_p < best_obj:
                        best_obj = obj_p
                        best_a = a.copy()
                stall = 0
                iteration += 1

    return best_a


def peak_equalization(a, time_budget=30.0, verbose=False):
    """Targeted optimization: try to reduce the top peaks of conv(a,a).
    Uses a subgradient approach focused specifically on the active peaks."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.02
    for it in range(2000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = conv.max()
        # Identify peaks above 95th percentile of the max
        threshold = max_val * 0.95
        active = np.where(conv >= threshold)[0]

        # Subgradient: for each active peak k, gradient contribution to a[j]
        # is proportional to a[k-j] (with valid indices)
        n_fft = 1
        while n_fft < 4 * n:
            n_fft <<= 1

        w = np.zeros(2 * n - 1)
        for k in active:
            w[int(k)] = 1.0 / len(active)

        A_fft = rfft(a, n_fft)
        W_fft = rfft(w, n_fft)
        R = irfft(np.conj(A_fft) * W_fft, n_fft)
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = R[(n_fft - j) % n_fft]
        grad *= 4.0 * n
        grad -= np.mean(grad)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new

        if it % 100 == 99:
            lr *= 0.97

    return best_a


def block_local_search(a, time_budget=30.0, block_size=16, verbose=False):
    """Local search: optimize small blocks of consecutive elements using scipy minimize."""
    from scipy.optimize import minimize

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Try optimizing each block
    for block_start in range(0, n - block_size, block_size // 2):
        if time.monotonic() - start > time_budget:
            break

        block_end = block_start + block_size
        a_fixed = a.copy()
        a_block = a[block_start:block_end].copy()
        block_sum = float(np.sum(a_block))

        def block_obj(x):
            a_try = a_fixed.copy()
            x_clipped = np.maximum(x, 0)
            # Preserve block sum to maintain total sum
            sx = float(np.sum(x_clipped))
            if sx < 1e-15:
                return best_obj
            x_clipped = x_clipped * (block_sum / sx)
            a_try[block_start:block_end] = x_clipped
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            a_try = a_try / s_try
            conv = compute_conv(a_try)
            return 2.0 * n * float(np.max(conv))

        res = minimize(block_obj, a_block, method='Nelder-Mead',
                       options={'maxiter': 200, 'xatol': 1e-8, 'fatol': 1e-10})

        if res.fun < best_obj - 1e-12:
            x_opt = np.maximum(res.x, 0)
            sx = float(np.sum(x_opt))
            if sx > 1e-15:
                x_opt = x_opt * (block_sum / sx)
                best_a[block_start:block_end] = x_opt
                s_try = float(np.sum(best_a))
                if s_try > 1e-10:
                    best_a = best_a / s_try
                    best_obj = res.fun
                    a = best_a.copy()

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng):
    """Generate a diverse structured initialization."""
    choice = rng.integers(0, 7)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * (x - 0.5) ** 2)
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 3 * x)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-80 * (x - 0.3) ** 2) + np.exp(-80 * (x - 0.7) ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-100 * (x - 0.25) ** 2) + 0.3 * np.exp(-100 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 2 * x) + 0.2 * np.sin(2 * np.pi * 5 * x) + 0.1 * np.sin(2 * np.pi * 11 * x)
        a = np.maximum(a, 0.01)
    elif choice == 5:
        # Smooth random
        a = np.ones(n)
        for _ in range(4):
            center = rng.integers(0, n)
            width = rng.integers(n // 10, n // 3)
            for i in range(n):
                d = abs(i - center)
                a[i] *= 1.0 + rng.standard_normal() * 0.5 * np.exp(-d / (width * 0.5))
        a = np.maximum(a, 0.01)
    else:
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(0.3, 4.0)
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function using LP alternating with convex line search,
    peak equalization, block local search, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        print(f"Parent loaded: n={len(a)}, obj={compute_obj(a):.8f}")
    except Exception:
        a = np.ones(800) / 800
        print(f"No parent, using uniform: n=800")

    best_a = a.copy()
    best_obj = compute_obj(a)
    n_base = len(best_a)

    # === Phase 1: Primary LP Alternating ===
    print(f"\n=== Phase 1: LP Alternating (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 400.0)
    a_p1 = lp_alternate(best_a, time_budget=p1_budget, verbose=True)
    obj_p1 = compute_obj(a_p1)
    if obj_p1 < best_obj:
        best_obj = obj_p1
        best_a = a_p1
        print(f"  Phase 1 result: obj={best_obj:.8f} **")

    # === Phase 2: Peak Equalization ===
    print(f"\n=== Phase 2: Peak Equalization ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p2_budget = min(remaining * 0.08, 60.0)
        a_pe = peak_equalization(best_a, time_budget=p2_budget)
        obj_pe = compute_obj(a_pe)
        if obj_pe < best_obj:
            best_obj = obj_pe
            best_a = a_pe
            print(f"  PeakEq result: obj={best_obj:.8f} **")
        else:
            print(f"  PeakEq result: obj={obj_pe:.8f}")

    # === Phase 3: LP Alternating Round 2 ===
    print(f"\n=== Phase 3: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p3_budget = min(remaining * 0.25, 250.0)
        a_p3 = lp_alternate(best_a, time_budget=p3_budget, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            print(f"  Phase 3 result: obj={best_obj:.8f} **")

    # === Phase 4: Block Local Search ===
    print(f"\n=== Phase 4: Block Local Search ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p4_budget = min(remaining * 0.12, 80.0)
        a_bl = block_local_search(best_a, time_budget=p4_budget, block_size=12)
        obj_bl = compute_obj(a_bl)
        if obj_bl < best_obj:
            best_obj = obj_bl
            best_a = a_bl
            print(f"  BlockSearch result: obj={best_obj:.8f} **")
        else:
            print(f"  BlockSearch result: obj={obj_bl:.8f}")

    # === Phase 5: LP + PeakEq from block search result ===
    print(f"\n=== Phase 5: LP + PeakEq Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = min(remaining * 0.15, 100.0)
        a_p5 = lp_alternate(best_a, time_budget=p5_budget * 0.7, verbose=True)
        obj_p5 = compute_obj(a_p5)
        if obj_p5 < best_obj:
            best_obj = obj_p5
            best_a = a_p5

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_pe2 = peak_equalization(best_a, time_budget=min(remaining * 0.3, 40.0))
            obj_pe2 = compute_obj(a_pe2)
            if obj_pe2 < best_obj:
                best_obj = obj_pe2
                best_a = a_pe2
                print(f"  Phase 5 result: obj={best_obj:.8f} **")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        max(200, n_base - 100),
        max(200, n_base - 50),
        n_base,
        n_base + 50, n_base + 100, n_base + 200,
    ]))
    n_options = [x for x in n_options if 100 <= x <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try

        alloc = min(max(15.0, remaining * 0.15), 50.0)
        a_opt = lp_alternate(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse Initializations ===
    print(f"\n=== Phase 7: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    for d in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_div = diverse_initialization(n_best, rng)
        alloc = min(max(10.0, remaining * 0.12), 40.0)
        a_opt = lp_alternate(a_div, time_budget=alloc)
        obj_div = compute_obj(a_opt)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse init {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse init {d}: obj={obj_div:.8f}")

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc_lp = min(remaining * 0.4, 60.0)
        a_final = lp_alternate(best_a, time_budget=alloc_lp, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc_pe = min(remaining * 0.3, 30.0)
        a_pe = peak_equalization(best_a, time_budget=alloc_pe)
        obj_pe = compute_obj(a_pe)
        if obj_pe < best_obj:
            best_obj = obj_pe
            best_a = a_pe
            print(f"  Final PeakEq: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 8:
        alloc_bl = min(remaining * 0.4, 25.0)
        a_bl = block_local_search(best_a, time_budget=alloc_bl, block_size=8)
        obj_bl = compute_obj(a_bl)
        if obj_bl < best_obj:
            best_obj = obj_bl
            best_a = a_bl
            print(f"  Final Block: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
