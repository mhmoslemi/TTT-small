import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def smooth_max_logsumexp(conv, T):
    """Smooth approximation to max using log-sum-exp."""
    scaled = T * conv
    shifted = scaled - scaled.max()
    return (1.0 / T) * (scaled.max() + np.log(np.sum(np.exp(shifted))))

def obj_smooth(x, T, n):
    """Objective as function of log-parameters x."""
    # Softmax parameterization
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a = a / a.sum()
    conv = compute_conv(a)
    return 2.0 * n * smooth_max_logsumexp(conv, T)

    # a_i = exp(x_i) / sum(exp(x))
    # x_i = log(a_i) (up to a constant)
    a_safe = np.maximum(a, 1e-15)
    x0 = np.log(a_safe / a_safe.sum())
    
    # L-BFGS-B on x with increasing temperature T
    for T in [10, 50, 100, 200, 500, 1000, 2000, 5000]:
        if time.monotonic() > deadline:
            break
        
        def obj_func(x):
            return obj_smooth(x, T, n)
        
        result = minimize(obj_func, x0, method='L-BFGS-B', 
                         options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10})
        
        if result.success or result.fun < obj_func(x0):
            x0 = result.x
            # Convert back
            x_shift = x0 - x0.max()
            new_a = np.exp(x_shift)
            new_a = new_a / new_a.sum()
            new_obj = 2.0 * n * compute_conv(new_a).max()
            print(f"T={T}: obj={new_obj:.8f}, result={result.message}")
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = new_a.copy()
    
    # Now do a "peaking" phase: try to reduce the max by direct gradient steps
    # on the true max (not smoothed)
    a = best_a.copy()
    for step in range(10000):
        if time.monotonic() > deadline:
            break
        conv = compute_conv(a)
        k_max = conv.argmax()
        # Gradient of (a*a)_{k_max} w.r.t. a
        # grad_m = 2 * a_{k_max - m} for valid m
        n_val = len(a)
        grad = np.zeros(n_val)
        for m in range(n_val):
            j = k_max - m
            if 0 <= j < n_val:
                grad[m] = 2.0 * a[j]
        # Subtract mean to maintain sum
        grad = grad - grad.mean()
        # Line search
        for alpha in [0.001, 0.005, 0.01, 0.02, 0.05]:
            a_new = a - alpha * grad
            if a_new.min() < 0:

            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a_new.copy()
                a = a_new
                break
    
    result = [float(x) for x in best_a]

from scipy.optimize import minimize
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def logsumexp_obj(x, beta, n):
    """Smooth objective using logsumexp on convolution values."""
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a = a / a.sum()
    conv = compute_conv(a)
    # logsumexp of beta * conv
    bc = beta * conv
    bc_max = bc.max()
    val = bc_max + np.log(np.sum(np.exp(bc - bc_max)))
    return val / beta

def logsumexp_grad(x, beta, n):
    """Gradient of the smooth objective w.r.t. x (softmax parameters)."""
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a = a / a.sum()
    conv = compute_conv(a)
    bc = beta * conv
    bc_max = bc.max()
    weights = np.exp(bc - bc_max)
    weights = weights / weights.sum()

    # = sum_k weights_k * 2 * a_{k-m}
    n_conv = 2 * n - 1
    grad_a = np.zeros(n)
    for k in range(n_conv):
        w = weights[k]
        if w < 1e-15:
            continue
        for m in range(n):
            j = k - m
            if 0 <= j < n:
                grad_a[m] += 2.0 * w * a[j]
    # Now gradient w.r.t. x via softmax Jacobian
    # da/dx = a * (I - a outer a) -- wait, that's for the case sum a = 1

"""
Optimizer using L-BFGS-B with adaptive logsumexp smoothing + subgradient + LP.
Key improvements over previous:
1. Smooth optimization via logsumexp (much faster than Nesterov subgradient)
2. FFT-based gradient computation
3. Adaptive temperature schedule
4. Multiple n values
5. LP alternating for final polish
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    return 2.0 * n * float(np.max(compute_conv(a_norm)))


def logsumexp_value_and_weights(a, beta):
    """Compute logsumexp(beta * conv(a,a)) and the softmax weights."""
    conv = compute_conv(a)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    logsumexp_val = bc_max + np.log(np.sum(exp_bc))
    weights = exp_bc / np.sum(exp_bc)
    return logsumexp_val / beta, weights, conv


def softmax_obj_and_grad(x, beta):
    """
    Objective and gradient for L-BFGS-B.
    x: log-parameters (a_i = exp(x_i) / sum(exp(x)))
    Returns (objective_value, gradient_wrt_x)
    """
    n = len(x)
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a = a / a.sum()
    
    ls_val, weights, conv = logsumexp_value_and_weights(a, beta)
    
    # grad_a[m] = 2 * sum_k weights[k] * a[k-m]
    # = 2 * conv(weights, a_flipped)[m + n - 1]
    a_flip = a[::-1]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    W = rfft(weights, size)
    Af = rfft(a_flip, size)
    corr = irfft(W * Af, size)
    grad_a = 2.0 * corr[n - 1 : 2 * n - 1]
    
    # Gradient w.r.t. x (softmax Jacobian)
    grad_x = grad_a * a - a * np.dot(grad_a, a)
    grad_x = grad_x / beta  # Wait, is this right?
    
    # The objective is (1/beta) * log(sum exp(beta * conv_k))
    # d/d a_m of this = (1/beta) * sum_k beta * weights_k * d(conv_k)/d(a_m)
    # = sum_k weights_k * 2 * a[k-m]
    # So grad_a (the gradient w.r.t. a) IS already the correct gradient of the objective
    # Then grad_x = J_softmax^T @ grad_a = grad_a * a - a * (grad_a @ a)
    
    return ls_val, grad_a * a - a * np.dot(grad_a, a)


def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 1e-15)


def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]

def subgradient_refine(a0, time_budget=60.0):

        return a0.copy()

    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.01

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        # Gradient of conv[k_max] w.r.t. a
        grad_a = np.zeros(n)
        for m in range(n):
            j = k_max - m
            if 0 <= j < n:
                grad_a[m] = 2.0 * a[j]
        grad_a = grad_a - np.mean(grad_a)
        
        # Backtracking line search
        alpha = step
        improved = False
        for _ in range(10):
            a_new = a - alpha * grad_a
            if a_new.min() >= 0:
                a_new = a_new / np.sum(a_new)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    break
            alpha *= 0.5
        if not improved:
            step *= 0.8
            if step < 1e-6:
                break

    return best_a

"""
Optimizer using L-BFGS-B with adaptive logsumexp smoothing + subgradient + LP alternating.
Key innovations over previous approach:
1. Smooth optimization via logsumexp surrogate (much faster convergence than Nesterov subgradient)
2. FFT-based gradient computation for O(n log n) per iteration
3. Softmax parameterization ensures non-negativity and enables unconstrained optimization
4. Adaptive temperature schedule for exploration → exploitation
5. Multiple sequence lengths and random restarts for escaping local minima
6. LP alternating for final polish
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar
import time
import cvxpy as cp


def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    return 2.0 * n * float(np.max(compute_conv(a_norm)))


def softmax_obj_and_grad(x, beta):
    """
    Compute objective and gradient for logsumexp-smoothed problem.
    a_i = exp(x_i) / sum(exp(x_j)), objective = (1/beta)*log(sum_k exp(beta*(a*a)_k))
    """
    n = len(x)
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a = a / a.sum()

    conv = compute_conv(a)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    sum_exp = np.sum(exp_bc)
    logsumexp_val = (bc_max + np.log(sum_exp)) / beta
    weights = exp_bc / sum_exp

    # grad_a[m] = 2 * sum_k weights[k] * a[k-m] (for valid k-m in [0,n-1])
    # This equals 2 * conv(weights, a[::-1])[m + n - 1]
    a_flip = a[::-1]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    W = rfft(weights, size)
    Af = rfft(a_flip, size)
    corr = irfft(W * Af, size)
    grad_a = 2.0 * corr[n - 1:2 * n - 1]

    # Gradient w.r.t. x via softmax Jacobian: grad_x = grad_a * a - a * (grad_a . a)
    grad_x = grad_a * a - a * np.dot(grad_a, a)

    return logsumexp_val, grad_x


def resample_seq(a, new_n):
    """Linear interpolation resampling preserving non-negativity."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    val = np.interp(x_new, x_old, a)
    return np.maximum(val, 1e-15)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=50.0):
    """LP alternating with 1D line search."""
    n = len(a0)

        return a0.copy()

    stall_count = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            return compute_obj(a_try / st)

        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 30, 'xatol': 0.01})
            best_t = res.x if res.success else 0.5
        except Exception:
            best_t = 0.5

        a_try = (1.0 - best_t) * a + best_t * g_lp

def subgradient_refine(a0, time_budget=60.0):
    """Direct subgradient on true max with backtracking line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.005
    stall = 0

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        # Use top-3 peaks for a better subgradient
        top_k = np.argsort(conv)[::-1][:3]
        grad_a = np.zeros(n)
        for k in top_k:
            k = int(k)
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    grad_a[m] += 2.0 * a[j]
        grad_a /= len(top_k)
        grad_a = grad_a - np.mean(grad_a)

        improved = False
        alpha = step
        for _ in range(15):
            a_new = a - alpha * grad_a
            if a_new.min() >= -1e-15:

                if sn > 1e-10:
                    a_new = a_new / sn
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        improved = True
                        stall = 0
                        break

        if not improved:
            stall += 1

def nesterov_subgrad(a0, time_budget=60.0, top_k=5):
    """Nesterov-accelerated subgradient on simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s

    def project_simplex(v):
        mv = np.sort(v)[::-1]
        cssv = np.cumsum(mv) - 1.0
        ind = np.arange(1, n + 1)
        cond = mv - cssv / ind > 0
        if not np.any(cond):
            return np.ones(n) / n
        i = np.max(ind[cond])
        theta = cssv[i - 1] / i
        return np.maximum(v - theta, 0)

    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05
    patience = 0

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    grad[m] += 2.0 * y[j]
        grad /= len(top_idx)
        grad = grad - np.mean(grad)
        y_new = y - step_size * grad
        a_new = project_simplex(y_new)
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = project_simplex(y)
        a = a_new
        if it % 20 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 150:
                step_size = max(step_size * 0.5, 0.001)
                patience = 0
        if it % 1000 == 999:
            step_size = max(step_size * 0.95, 0.002)

    return best_a


def propose_candidate():
    """
    Main search function. Uses L-BFGS-B with logsumexp smoothing as primary optimizer,
    combined with subgradient, Nesterov, and LP alternating methods.
    """
    start_time = time.monotonic()
    deadline = start_time + 1470.0
    rng = np.random.default_rng(2024)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        s = float(np.sum(parent_arr))
        if s > 0.01 and parent_arr.size > 0:
            parent = parent_arr / s
            n_base = len(parent)
        else:
            raise ValueError
    except Exception:
        n_base = 800
        parent = np.ones(n_base) / n_base

    best_a = parent.copy()
    best_obj = compute_obj(parent)
    print(f"Parent: n={n_base}, obj={best_obj:.10f}")

    # === Phase 1: L-BFGS-B with logsumexp (main phase) ===
    print("\n=== Phase 1: L-BFGS-B + LogSumExp ===")
    n_values = sorted(set([n_base,
                           max(300, n_base - 200),
                           n_base + 100,
                           n_base + 200,
                           600, 800, 1000, 1200, 1500]))
    n_values = [n for n in n_values if 300 <= n <= 2000]
    beta_schedule = [5, 10, 30, 100, 300, 1000, 3000, 10000, 30000]

    for n_try in n_values:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        if n_try == n_base:
            a0 = parent.copy()
        else:
            a0 = resample_seq(parent, n_try) / resample_seq(parent, n_try).sum()

        x0 = np.log(np.maximum(a0, 1e-15))
        x0 = x0 - x0.mean()

        for beta in beta_schedule:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            max_iter = min(500, max(50, int(remaining * 0.2)))

            def obj_func(x):
                val, _ = softmax_obj_and_grad(x, beta)
                return val

            def grad_func(x):
                _, g = softmax_obj_and_grad(x, beta)
                return g

            t0 = time.monotonic()
            try:
                result = minimize(
                    obj_func, x0, jac=grad_func,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16,
                             'gtol': 1e-12, 'maxfun': max_iter * 3}
                )
                elapsed = time.monotonic() - t0
                if result.x is not None:
                    xs = result.x - result.x.max()
                    new_a = np.exp(xs)
                    new_a = new_a / new_a.sum()
                    new_obj = compute_obj(new_a)
                    if new_obj < best_obj:
                        best_obj = new_obj
                        best_a = new_a.copy()
                        print(f"  ** n={n_try}, beta={beta}: obj={best_obj:.10f} (IMPROVED, {elapsed:.1f}s)")
                    else:
                        print(f"  n={n_try}, beta={beta}: obj={new_obj:.10f} ({elapsed:.1f}s)")
                    x0 = result.x
                else:
                    print(f"  n={n_try}, beta={beta}: no convergence")
            except Exception as e:
                print(f"  n={n_try}, beta={beta}: ERROR {str(e)[:50]}")

    # === Phase 2: Random restarts with L-BFGS-B ===
    print("\n=== Phase 2: Random Restarts ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_a)
        for r in range(15):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            scale = 0.03 + 0.01 * (r % 5)
            a_r = best_a * np.exp(rng.normal(0, scale, n_best))
            a_r = np.maximum(a_r, 1e-15)
            a_r = a_r / a_r.sum()
            x0 = np.log(a_r) - np.log(a_r).mean()

            for beta in [300, 1000, 3000]:
                remaining = deadline - time.monotonic()
                if remaining < 10:
                    break
                try:
                    result = minimize(
                        lambda x: softmax_obj_and_grad(x, beta)[0], x0,
                        jac=lambda x: softmax_obj_and_grad(x, beta)[1],
                        method='L-BFGS-B',
                        options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-11, 'maxfun': 400}
                    )
                    if result.x is not None:
                        xs = result.x - result.x.max()
                        new_a = np.exp(xs)
                        new_a = new_a / new_a.sum()
                        new_obj = compute_obj(new_a)
                        if new_obj < best_obj:
                            best_obj = new_obj
                            best_a = new_a.copy()
                            x0 = result.x
                            print(f"  Restart {r}, beta={beta}: obj={best_obj:.10f} **")
                        else:
                            x0 = result.x
                except Exception:
                    pass

    # === Phase 3: Subgradient refinement ===
    print("\n=== Phase 3: Subgradient Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(100.0, remaining * 0.25)
        a_sg = subgradient_refine(best_a, time_budget=alloc)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            print(f"  Subgrad: obj={best_obj:.10f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.10f}")

    # === Phase 4: Nesterov with different top_k ===
    print("\n=== Phase 4: Nesterov ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for k_val in [3, 5, 10]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(40.0, remaining * 0.2)
            a_ns = nesterov_subgrad(best_a, time_budget=alloc, top_k=k_val)
            obj_ns = compute_obj(a_ns)
            if obj_ns < best_obj:
                best_obj = obj_ns
                best_a = a_ns.copy()
                print(f"  Nesterov k={k_val}: obj={best_obj:.10f} **")
            else:
                print(f"  Nesterov k={k_val}: obj={obj_ns:.10f}")

    # === Phase 5: LP Alternating ===
    print("\n=== Phase 5: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for round_i in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(50.0, remaining * 0.2)
            a_lp = lp_alternate(best_a, time_budget=alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_obj - 1e-12:
                best_obj = obj_lp
                best_a = a_lp.copy()
                print(f"  LP R{round_i}: obj={best_obj:.10f} **")
                # Perturb for next round
                best_a = best_a * np.exp(rng.normal(0, 0.008, len(best_a)))
                best_a = np.maximum(best_a, 1e-15)
                best_a = best_a / best_a.sum()
            else:
                print(f"  LP R{round_i}: obj={obj_lp:.10f}")
                break

    # === Phase 6: Final L-BFGS-B polish with very high beta ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_cur = len(best_a)
        x0 = np.log(np.maximum(best_a, 1e-15))
        x0 = x0 - x0.mean()
        for beta in [10000, 30000, 100000, 300000]:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            try:
                result = minimize(
                    lambda x: softmax_obj_and_grad(x, beta)[0], x0,
                    jac=lambda x: softmax_obj_and_grad(x, beta)[1],
                    method='L-BFGS-B',
                    options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-13, 'maxfun': 800}
                )
                if result.x is not None:
                    xs = result.x - result.x.max()
                    new_a = np.exp(xs)
                    new_a = new_a / new_a.sum()
                    new_obj = compute_obj(new_a)
                    if new_obj < best_obj:
                        best_obj = new_obj
                        best_a = new_a.copy()
                        x0 = result.x
                        print(f"  Polish beta={beta}: obj={best_obj:.10f} **")
                    else:
                        print(f"  Polish beta={beta}: obj={new_obj:.10f}")
                        x0 = result.x
            except Exception as e:
                print(f"  Polish beta={beta}: ERROR")

    # === Phase 7: Final subgradient ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(40.0, remaining * 0.6)
        a_fin = subgradient_refine(best_a, time_budget=alloc)
        obj_fin = compute_obj(a_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
            print(f"  Final subgrad: obj={best_obj:.10f} **")

    # Validate and return
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    final_val = evaluate_sequence(result)
    if final_val == np.inf:
        # Fallback
        result = [1.0] * 200
        final_val = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={final_val:.10f}, time={elapsed:.1f}s ===")
    return result
