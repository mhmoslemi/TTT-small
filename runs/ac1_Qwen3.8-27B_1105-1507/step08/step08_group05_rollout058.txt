"""
Active-set QP optimizer for minimizing autocorrelation peak.
Uses CVXPY to solve the exact QP subproblem for the active set of peaks,
combined with smooth analytical initializations and multi-scale search.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy import sparse
from scipy.optimize import minimize_scalar

def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_D_k(n, k):
    """Build sparse matrix D_k where (D_k a)[i] = a[k-i] if valid."""
    rows = []
    cols = []
    vals = []
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            rows.append(i)
            cols.append(j)
            vals.append(1.0)
    if not rows:
        return sparse.csr_matrix((n, n))
    return sparse.csr_matrix((vals, (rows, cols)), shape=(n, n))

def qp_solve(active_indices, warm_start=None, n=None):
    """
    Solve: min t s.t. a^T D_k a <= t for k in active_indices,
           sum(a) = 1, a >= 0
    """
    a_var = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_indices:
        D_k = build_D_k(n, k)
        # conv[k] = a^T D_k a = sum_i a[i] * (D_k a)[i]
        # In CVXPY: cp.sum(cp.multiply(a_var, D_k @ a_var))
        # But D_k @ a_var is affine, so this is a quadratic form
        shifted = D_k @ a_var
        constraints.append(cp.sum(cp.multiply(a_var, shifted)) <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if a_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                result = np.maximum(a_var.value, 0)
                s = float(np.sum(result))
                if s > 1e-10:
                    return result / s
        except Exception:
            continue
    return None

def active_set_iterate(a, max_active=15, max_iter=20, time_budget=60.0):
    """
    Iteratively solve the active-set QP.
    Start with the top peaks, solve QP, check for new violations, repeat.
    """
    n = len(a)
    a = a / float(np.sum(a))
    start = time.monotonic()
    
    for iteration in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        # Find peaks that exceed the current minimum
        t_current = float(np.max(conv))
        
        # Active set: peaks that are close to the maximum
        threshold = t_current * 0.999
        active = np.where(conv >= threshold)[0]
        
        # Limit to max_active peaks (the largest ones)
        if len(active) > max_active:
            active = active[np.argsort(conv[active])[::-1][:max_active]]
        
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # Solve QP
        a_new = qp_solve(active.tolist(), n=n)
        
        if a_new is None:
            break
        
        # Check improvement
        conv_new = compute_conv(a_new)
        obj_new = 2.0 * n * float(np.max(conv_new))
        obj_old = 2.0 * n * t_current
        
        if obj_new < obj_old - 1e-10:
            a = a_new
        else:
            # No improvement; try with fewer active constraints
            if len(active) > 3:
                active = active[:3]
                a_new = qp_solve(active.tolist(), n=n)
                if a_new is not None:
                    conv_new = compute_conv(a_new)
                    obj_new = 2.0 * n * float(np.max(conv_new))
                    if obj_new < obj_old - 1e-10:
                        a = a_new
                    else:
                        break
            else:
                break
    
    return a

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def subgradient_of_max_conv(a, conv, num_peaks=None):
    """
    Compute subgradient of max_k (a*a)[k] at point a.
    Uses the top num_peaks values of conv to approximate the subgradient.
    """
    n = len(a)
    if num_peaks is None:
        num_peaks = 10
    
    # Get indices of top peaks
    sorted_idx = np.argsort(conv)[::-1][:num_peaks]
    
    grad = np.zeros(n)
    for k in sorted_idx:
        k = int(k)
        # grad[i] += 2 * a[k-i] if 0 <= k-i < n
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            # For i in [i_start, i_end], grad[i] += 2 * a[k-i]
            # a[k-i] for i in [i_start, i_end] is a[k-i_start] down to a[k-i_end]
            contrib = a[k - i_end : k - i_start + 1][::-1]
            grad[i_start:i_end + 1] += 2.0 * contrib
    
    grad /= num_peaks
    return grad

def hann_window(n):
    """Hann window normalized to sum=1."""
    if n == 1:
        return np.array([1.0])
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def raised_cosine(n, p):
    """Raised cosine to power p, normalized."""
    if n == 1:
        return np.array([1.0])
    x = np.arange(n) / (n - 1)
    w = np.maximum(0.5 * (1.0 - np.cos(2.0 * np.pi * (x - 0.5))), 0) ** p
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def gaussian_window(n, sigma_frac=0.3):
    """Gaussian window with specified sigma (fraction of n), normalized."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    w = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

                contrib = a[k - i_end : k - i_start + 1][::-1]

                if obj_try < best_obj:
                    best_obj = obj_try

            break

"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def subgradient_max_conv(a, conv, num_peaks=10):
    """Compute subgradient of max_k (a*a)[k] using top num_peaks."""
    n = len(a)
    sorted_idx = np.argsort(conv)[::-1][:num_peaks]
    grad = np.zeros(n)
    for k in sorted_idx:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            contrib = a[k - i_end:k - i_start + 1][::-1]
            grad[i_start:i_end + 1] += 2.0 * contrib
    grad /= float(num_peaks)
    return grad

def hann_window(n):
    """Hann window normalized to sum=1."""
    if n == 1:
        return np.array([1.0])
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def raised_cosine_p(n, p, width=1.0):
    """Raised cosine to power p with specified support width (fraction of n)."""
    if n == 1:
        return np.array([1.0])
    x = np.arange(n) / (n - 1)
    center = 0.5
    half_width = width / 2.0
    # Only non-zero within [center-half_width, center+half_width]
    t = (x - center) / half_width
    w = np.where(np.abs(t) <= 1.0, 0.5 * (1.0 + np.cos(np.pi * t)), 0.0)
    w = np.maximum(w, 0) ** p
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def gaussian_window(n, sigma_frac=0.3):
    """Gaussian window with specified sigma (fraction of n), normalized."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    w = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)

    w = np.ones(n)
    x = np.arange(n) / n
    for f, p in zip(freqs, phases):
        w += 0.2 * np.cos(2 * np.pi * f * x + p)
    w = np.maximum(w, 0.01)
    inits.append(w / np.sum(w))

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: FISTA + peak shaving + weighted gradient refinement.
    Uses diverse analytical initializations, n-sweeping, and multi-strategy restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        best_seq = hann_window(800)
        best_score = compute_obj(best_seq)
        print(f"No parent, Hann init (n=800): obj={best_score:.8f}")

    # === Phase 1: Diverse inits at multiple n (350s) ===
    print(f"\n=== Phase 1: Diverse Initializations + FISTA ===")
    n_options = [500, 600, 700, 800, 900, 1000, 1100, 1200, 1500, 1800, 2000]
    if best_seq is not None and len(best_seq) not in n_options:
        n_options.append(len(best_seq))
        n_options.sort()

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        inits = diverse_initializations(n_try, rng)

        # Add resampled best
        if best_seq is not None:
            resampled = resample_seq(best_seq, n_try)
            rs = float(np.sum(resampled))
            if rs > 1e-10:
                inits.append(resampled / rs)

        best_for_n = None
        best_obj_n = np.inf

        for i, init in enumerate(inits):
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break

            obj_init = compute_obj(init)
            if obj_init < best_obj_n:
                best_obj_n = obj_init
                best_for_n = init.copy()

            alloc = min(12.0, remaining * 0.6 / max(1, len(inits) - i))
            a_opt = fista_optimize(init, time_budget=alloc, verbose=False, seed=seed + i)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj_n:
                best_obj_n = obj_opt
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            if best_obj_n < best_score:
                best_score = best_obj_n
                best_seq = best_for_n.copy()
                print(f"  n={n_try}: obj={best_score:.8f} ** NEW BEST")
            else:
                print(f"  n={n_try}: best={best_obj_n:.8f}")

    # === Phase 2: Extended FISTA (200s) ===
    print(f"\n=== Phase 2: Extended FISTA ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        alloc = min(150.0, remaining * 0.5)
        a_ext = fista_optimize(best_seq, time_budget=alloc, verbose=True, seed=seed + 10)
        obj_ext = compute_obj(a_ext)
        if obj_ext < best_score:
            best_score = obj_ext
            best_seq = a_ext.copy()
            print(f"  Extended FISTA: obj={best_score:.8f} **")
        else:
            print(f"  Extended FISTA: obj={obj_ext:.8f}")

    # === Phase 3: Peak shaving (150s) ===
    print(f"\n=== Phase 3: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(100.0, remaining * 0.4)
        a_ps = peak_shave(best_seq, time_budget=alloc, rng=rng)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Peak Shave: obj={best_score:.8f} **")
        else:
            print(f"  Peak Shave: obj={obj_ps:.8f}")
            # Try weighted gradient instead
            a_wg = weighted_gradient_refine(best_seq, time_budget=min(60.0, alloc * 0.7), rng=rng)
            obj_wg = compute_obj(a_wg)
            if obj_wg < best_score:
                best_score = obj_wg
                best_seq = a_wg.copy()
                print(f"  Weighted Grad: obj={best_score:.8f} **")
            else:
                print(f"  Weighted Grad: obj={obj_wg:.8f}")

    # === Phase 4: Multi-strategy restarts (350s) ===
    print(f"\n=== Phase 4: Multi-Strategy Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    num_restarts = max(5, int(min(300.0, remaining * 0.5) / 12.0))

    for r in range(num_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        strategy = r % 5
        if strategy == 0:
            # Gaussian noise perturbation
            scale = 0.03 + 0.008 * (r // 5)
            a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
        elif strategy == 1:
            # Smooth cosine perturbation
            freq = rng.integers(1, 15)
            x = np.arange(n_best) / n_best
            phase = rng.uniform(0, 2 * np.pi)
            pert = np.cos(2 * np.pi * freq * x + phase)
            scale = 0.04
            a_r = best_seq * (1.0 + scale * pert)
        elif strategy == 2:
            # Blend with random analytical shape
            inits = diverse_initializations(n_best, rng)
            init = inits[rng.integers(len(inits))]
            blend = rng.uniform(0.05, 0.3)
            a_r = (1.0 - blend) * best_seq + blend * init
        elif strategy == 3:
            # Asymmetric perturbation
            x = np.arange(n_best) / n_best
            asym = np.exp(-((x - rng.uniform(0.3, 0.7)) * rng.uniform(3, 8)) ** 2)
            scale = 0.05
            a_r = best_seq * (1.0 + scale * (asym - 1.0))
        else:
            # Try different n
            delta_n = rng.integers(-100, 101)
            new_n = max(300, min(2500, n_best + delta_n))
            a_r = resample_seq(best_seq, new_n)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr
            a_r = resample_seq(a_r, n_best)  # Come back to same n for comparison
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr
            else:
                a_r = best_seq.copy()

        a_r = np.maximum(a_r, 0)
        sr = float(np.sum(a_r))
        if sr < 1e-10:
            continue
        a_r = a_r / sr

        alloc = min(remaining * 0.35, 12.0)
        a_opt = fista_optimize(a_r, time_budget=alloc * 0.6, verbose=False, seed=seed + 200 + r)
        # Follow with peak shaving
        a_opt = peak_shave(a_opt, time_budget=alloc * 0.3, rng=rng)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  R{r} (str={strategy}): obj={best_score:.8f} **")
        else:
            print(f"  R{r} (str={strategy}): obj={obj_r:.8f}")

    # === Phase 5: Final combined polish (remaining) ===
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # FISTA refinement
        alloc1 = min(60.0, remaining * 0.3)
        a_f1 = fista_optimize(best_seq, time_budget=alloc1, verbose=False, seed=seed + 999)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final FISTA: obj={best_score:.8f} **")

        # Peak shave
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc2 = min(40.0, remaining * 0.3)
            a_f2 = peak_shave(best_seq, time_budget=alloc2, rng=rng)
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final PeakShave: obj={best_score:.8f} **")

        # Weighted gradient
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc3 = min(40.0, remaining * 0.3)
            a_f3 = weighted_gradient_refine(best_seq, time_budget=alloc3, rng=rng)
            obj_f3 = compute_obj(a_f3)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Final WeightedGrad: obj={best_score:.8f} **")

    # Build result
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
