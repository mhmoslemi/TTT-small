def bregman_project(y, g):

"""
Multi-resolution optimizer with Bregman projected subgradient, LP alternating,
and spectral shaping for minimizing 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def conv_grad_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        m_start = k - i_end
        m_end = k - i_start
        grad[m_start:m_end + 1] = 2.0 * a[i_start:i_end + 1][::-1]
    return grad


def bregman_step(y, g, step_size):
    """Bregman (KL) proximal step: x = y * exp(-step_size * g) normalized."""
    log_x = np.log(np.maximum(y, 1e-300)) - step_size * g
    log_x -= np.max(log_x)
    x = np.exp(log_x)
    x_sum = np.sum(x)
    if x_sum < 1e-300:
        return np.ones(len(y)) / len(y)
    return x / x_sum


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)

    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adaptive step size
    step_size = 0.1
    prev_obj = best_obj
    patience = 0
    no_improve_count = 0
    
    for it in range(500000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute subgradient at y
        conv_y = compute_conv(y)
        # Active set: top peaks
        thresholds = np.sort(conv_y)[::-1][:top_k]
        active = np.where(conv_y >= thresholds[-1] * 0.95)[0]
        if len(active) == 0:
            active = np.argsort(conv_y)[::-1][:top_k]
        
        grad = np.zeros(n)
        for k in active:
            grad += conv_grad_at_k(y, int(k))
        grad /= len(active)
        
        # Center gradient
        grad = grad - np.mean(grad)
        
        # Bregman step
        y_new = bregman_step(y, grad, step_size)
        
        # Nesterov momentum
        t_k = 2.0 / (it + 2.0)
        a_new = y_new  # For KL, the momentum is different

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None

def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def spectral_shape(a, alpha=0.5):
    """Apply spectral shaping: reduce high-frequency content of |A|^2."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = rfft(a, size)
    # Apply a lowpass-like filter to |A|^2
    freqs = np.arange(size // 2 + 1)
    # Dampen frequencies where |A|^2 contributes to the peak
    conv = compute_conv(a)
    peak_k = int(np.argmax(conv))
    # The peak at position k is influenced by all frequencies
    # Simple approach: reduce |A|^2 at high frequencies
    damping = np.exp(-alpha * (freqs / (size // 2)) ** 2)
    A_modified = A * np.sqrt(damping)
    a_new = np.real(irfft(A_modified, size)[:n])
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)

def compute_grad_smooth(a, weights):
    """Compute grad[m] = 2 * sum_j a[j] * weights[j+m] for valid indices."""
    n = len(a)
    len_w = len(weights)
    L = 1
    while L < len_w + n - 1:
        L <<= 1
    P = np.fft.rfft(weights, L)
    Q = np.fft.rfft(a, L)
    # cross-correlation: sum_k p[k] * q[k-m] = ifft(conj(P) * Q)
    # But I want sum_k p[k+m] * q[k] = ifft(P * conj(Q))... 
    # Let me verify with a small example mentally.
    # p = [p0, p1, p2], q = [q0, q1]
    # result[0] = p0*q0 + p1*q1
    # result[1] = p1*q0 + p2*q1
    # 
    # ifft(P * conj(Q))[m] = sum_k P[k] conj(Q[k]) * exp(2πikm/L) / L
    # This is the circular cross-correlation which for zero-padded signals gives linear cross-correlation.
    # (p ⊛ q)[m] = sum_k p[k+m] q[k] = ifft(FFT(p) * conj(FFT(q)))[m] (circular, but linear with padding)
    
    corr = np.fft.irfft(P * np.conj(Q), L)
    # corr[m] for m = 0, ..., n-1 should give sum_k p[k+m] * q[k]
    # But the circular correlation might have wrap-around. With L >= len(p) + len(q) - 1, no wrap-around.
    grad = 2.0 * corr[:n]
    return grad
