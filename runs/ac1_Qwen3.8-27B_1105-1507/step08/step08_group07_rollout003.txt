import numpy as np

from scipy.optimize import linprog

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    """Build dense convolution matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_fast(a):
    """
    Solve LP: min M s.t. conv(a, z)_k <= M for all k, sum(z)=1, z>=0.
    Uses scipy linprog with HiGHS directly for speed.
    Returns (z_normalized, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix_sparse(a)

    # Variables: z_0, ..., z_{n-1}, M
    # c = [0, ..., 0, 1] (minimize M)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality: A @ z - M <= 0  =>  [A, -1] @ [z; M] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(z) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    if res.status == 0:
        z_val = np.maximum(res.x[:n], 0)
        zs = float(np.sum(z_val))
        if zs > 1e-10:
            z_val = z_val / zs
            return z_val, float(res.x[n])
    return None, None

def subgradient_step(a, eta, top_k=10):
    """
    One subgradient step using the clean gradient structure.
    ∂c_k/∂a_i = 2*a_{k-i} (if valid index)
    Uses top-k active peaks for a smoother subgradient.
    """
    n = len(a)
    conv = compute_conv(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    # Get top-k peaks
    k_vals = np.argsort(conv)[::-1][:top_k]
    max_val = conv[k_vals[0]]
    
    # Weight active peaks by their proximity to max
    weights = np.maximum(conv[k_vals] - 0.95 * max_val, 0)
    if np.sum(weights) < 1e-15:
        weights = np.ones(len(k_vals))
    weights = weights / np.sum(weights)

    # Compute subgradient: sum of w_k * 2*a_{k-i} for each i
    grad = np.zeros(n)
    for w, k in zip(weights, k_vals):
        # grad[i] += w * 2 * a[k-i] if 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        # For i in [lo, hi-1], k-i is in [0, n-1]
        # a[k-i] for i in [lo, hi-1] is a[k-lo] down to a[k-(hi-1)]
        idx = np.arange(lo, hi)
        rev_idx = k - idx
        grad[lo:hi] += 2.0 * w * a[rev_idx]

    # Mirror descent / exponential weights update
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return a
    return a_new / s_new


def subgradient_polish(a, time_budget=30.0, top_k=10):
    """Polish using subgradient steps with adaptive step size."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = 0.1
    stall = 0
    
    while time.monotonic() - start < time_budget:
        a_new = subgradient_step(a, eta, top_k)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            eta = min(eta * 1.05, 1.0)
        else:
            stall += 1
            eta = max(eta * 0.9, 0.001)
            if stall > 30:
                break
    
    return best_a

def spectral_perturb(a, rng, strength=0.1):
    """
    Perturb a in the frequency domain to create diverse starting points.
    """
    n = len(a)
    # Go to frequency domain
    fa = rfft(a)
    magnitude = np.abs(fa)
    phase = np.angle(fa)
    
    # Perturb magnitude
    mag_pert = magnitude * np.exp(rng.normal(0, strength, len(magnitude)))
    # Perturb phase
    phase_pert = phase + rng.normal(0, strength * 0.5, len(phase))
    
    # Reconstruct
    fa_pert = mag_pert * np.exp(1j * phase_pert)
    a_pert = irfft(fa_pert, 2 * n - 1)[:n]
    
    # Project to non-negative
    a_pert = np.maximum(a_pert, 0)
    s = float(np.sum(a_pert))
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(x_old, a)
        a_new = cs(x_new)
    except:
        a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_descent(a, time_budget=30.0):
    """
    1D coordinate descent: for each coordinate, find optimal value
    given all others fixed. Uses ternary search or analytical solution.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    sweep_count = 0
    while time.monotonic() - start < time_budget:
        improved = False
        # Random order of coordinates
        order = rng.permutation(n)
        
        for idx in order:
            if time.monotonic() - start > time_budget:
                    break
            
            # Try different values for a[idx]
            # Keep sum constant: if we change a[idx] by delta, adjust others
            # Simple approach: try a[idx] * factor for various factors
            a_save = a[idx]
            
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a[idx] = a_save * factor
                s = float(np.sum(a))
                if s < 1e-10:
                    a[idx] = a_save
                    continue
                a_norm = a / s
                obj_try = compute_obj(a_norm)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_norm.copy()
                    a = a_norm
                    a_save = a[idx]
                    improved = True
                    break
            
            a[idx] = a_save
        
        sweep_count += 1
        if not improved:
            break
    
    return best_a
