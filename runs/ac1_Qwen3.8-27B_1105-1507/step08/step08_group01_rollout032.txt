"""
Optimized search using L-BFGS with smooth max, multi-n exploration, and coordinate descent polish.
Key improvements over previous: L-BFGS quasi-Newton optimization captures curvature,
adaptive temperature scheduling, softmax parameterization for simplex constraint.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def smooth_max_obj_and_grad(b, T):
    """Compute smooth max objective and gradient in softmax parameterization.
    b is the parameter vector, a = softmax(b)."""
    a = np.exp(b - np.max(b))
    s = np.sum(a)
    a = a / s  # normalized, sum = 1
    n = len(a)
    
    conv = compute_conv(a)
    
    # Smooth max
    shifted = conv / T
    shifted = np.clip(shifted, -100, 100)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    logsum = np.log(np.sum(exp_shifted)) + max_shifted
    smooth_max_val = T * logsum
    
    # Weights
    w = exp_shifted / np.sum(exp_shifted)
    
    # Gradient in a-space: grad_a = 2 * correlate(w, a, 'valid')
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    
    # Objective: 2n * smooth_max
    obj = 2.0 * n * smooth_max_val
    
    # Gradient in a-space for full objective
    grad_a_full = 2.0 * n * grad_a
    
    # Convert to b-space: grad_b_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)
    mean_grad = np.dot(a, grad_a_full)
    grad_b = a * (grad_a_full - mean_grad)
    
    return obj, grad_b

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_lp_for_direction(a, time_budget=30.0):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': time_budget})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimized search using L-BFGS with smooth max, LP refinement, coordinate descent, and multi-n exploration.
Key improvements: L-BFGS quasi-Newton captures curvature for faster convergence,
adaptive temperature schedule, softmax parameterization for simplex constraint,
and systematic multi-n exploration.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_max_obj_grad(b, T):
    """Compute smooth max objective and gradient in softmax parameterization."""
    b_shifted = b - np.max(b)
    a = np.exp(b_shifted)
    s = np.sum(a)
    if s < 1e-300:
        return np.inf, np.zeros_like(b)
    a = a / s
    n = len(a)

    conv = compute_conv(a)

    # Smooth max: T * logsumexp(conv / T)
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / T
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    if sum_exp < 1e-300:
        return np.inf, np.zeros_like(b)
    log_sum_exp = max_conv / T + np.log(sum_exp)
    smooth_max_val = T * log_sum_exp

    # Softmax weights
    w = exp_shifted / sum_exp

    # Gradient in a-space: d/d(a_i) [2n * smooth_max]
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    grad_obj_a = 2.0 * n * grad_a

    # Convert to b-space via softmax Jacobian
    mean_grad = np.dot(a, grad_obj_a)
    grad_b = a * (grad_obj_a - mean_grad)

    obj = 2.0 * n * smooth_max_val
    return obj, grad_b

            return smooth_max_obj_grad(b, current_T)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_for_direction(a, time_budget=20.0):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': max(1, time_budget)})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None


def lp_refine(a, time_budget=60.0, verbose=False):
    """LP-based refinement with convex line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(100):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        per_lp = min(remaining * 0.25, 25.0)
        g = solve_lp_for_direction(a, time_budget=per_lp)
        if g is None:
            break

        # Convex line search on blend
        conv_aa = compute_conv(a)
        conv_gg = compute_conv(g)
        # Cross conv via FFT
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = rfft(a, size)
        fg = rfft(g, size)
        conv_ag = irfft(fa * fg, size)[:2 * n - 1]

        def obj_at_t(t):
            val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
            return 2.0 * n * float(np.max(val))

        # Ternary search
        lo, hi = 0.0, 1.0
        for _ in range(12):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if obj_at_t(m1) < obj_at_t(m2):
                hi = m2
            else:
                lo = m1
        t_opt = (lo + hi) / 2.0

        a_try = (1 - t_opt) * a + t_opt * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            break
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            if verbose and it % 5 == 0:
                print(f"          LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}")
        else:
            if verbose and it % 10 == 0:
                print(f"          LP it{it}: no improvement, obj={best_obj:.8f}")
            if it > 20:
                break

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            best_score = obj_opt
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  Phase 3 done: obj={best_score:.8f}")
        else:
            print(f"  Phase 3 done: obj={obj_opt:.8f} (no improvement)")

    # === Phase 4: Peak Reduction ===
    print(f"\n=== Phase 4: Peak Reduction ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        phase4_budget = min(60.0, remaining * 0.1)
        a_opt = peak_reduction(a, time_budget=phase4_budget)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  Phase 4 done: obj={best_score:.8f}")
        else:
            print(f"  Phase 4 done: obj={obj_opt:.8f} (no improvement)")

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        phase5_budget = min(100.0, remaining * 0.15)
        a_opt = coordinate_descent_polish(a, time_budget=phase5_budget, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  Phase 5 done: obj={best_score:.8f}")
        else:
            print(f"  Phase 5 done: obj={obj_opt:.8f} (no improvement)")

    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([
        n_best - 30, n_best - 15, n_best + 15, n_best + 30, n_best + 50,
        n_best + 100, n_best + 150, n_best + 200,
        550, 600, 650, 700, 750, 800, 850, 900, 950, 1000, 1100, 1200, 1400
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        # L-BFGS
        alloc = min(remaining * 0.1, 35.0)
        a_opt = lbfgs_smooth_max(a_try, time_budget=alloc, T_init=0.003, T_min=0.0001)

        remaining = deadline - time.monotonic()
        if remaining > 20:
            # Quick peak reduction
            a_opt = peak_reduction(a_opt, time_budget=min(10.0, remaining * 0.15))

        remaining = deadline - time.monotonic()
        if remaining > 15:
            # Quick CD
            a_opt = coordinate_descent_polish(a_opt, time_budget=min(10.0, remaining * 0.2))

        obj_try = compute_obj(a_opt)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Final Polish on Best ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # LP
        if remaining > 40:
            alloc = min(50.0, remaining * 0.25)
            a_opt = lp_refine(best_seq, time_budget=alloc)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Final LP: obj={best_score:.8f}")

        # L-BFGS
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(50.0, remaining * 0.25)
            a_opt = lbfgs_smooth_max(best_seq, time_budget=alloc, T_init=0.0005, T_min=0.00002)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Final L-BFGS: obj={best_score:.8f}")

        # Peak reduction
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(25.0, remaining * 0.2)
            a_opt = peak_reduction(best_seq, time_budget=alloc)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Final PeakRed: obj={best_score:.8f}")

        # CD
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(30.0, remaining * 0.4)
            a_opt = coordinate_descent_polish(best_seq, time_budget=alloc)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Final CD: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
