def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp_scipy(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0 using scipy HiGHS."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

        A_new = mag * np.exp(1j * phase_A)
        a_candidate = np.real(ifft(A_new))[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s_cand = float(np.sum(a_candidate))
        if s_cand < 1e-10:
            continue
        a_candidate = a_candidate / s_cand

        a_new = (1 - alpha) * a + alpha * a_candidate
        a_new = np.maximum(a_new, 0)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 50 == 0:
                print(f"          FFX-pen it{it}: obj={best_obj:.8f}")

        alpha *= 0.998

    return best_a


def subgradient_peak_method(a, time_budget=120.0, T=0.001, lr=0.005, verbose=False):
    """
    Subgradient method using exponential weighting over peaks.
    The subgradient of max_k conv(a,a)_k is conv(w, a) where w is
    a probability distribution over the peak locations.
    We use softmax weighting to get a smooth subgradient.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)  # momentum
    momentum = 0.8
    stall = 0

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = np.max(conv)

        # Softmax weights over convolution peaks
        shifted = (conv - max_conv) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        if total < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_vals / total

        # Subgradient of max_k conv(a,a)_k w.r.t. a:
        # d/d a[i] conv(a,a)[k] = a[k-i] (for valid k-i)
        # So subgradient = sum_k w[k] * (partial conv[k] / partial a[i])
        # = sum_k w[k] * a[k-i] (where valid)
        # This is the "valid" cross-correlation of w with a
        grad = np.correlate(w, a, mode='valid')
        # Center the gradient to preserve sum(a) = 1 constraint
        grad_centered = grad - np.mean(grad)

        # Update with momentum
        v = momentum * v - lr * grad_centered
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Adaptive learning rate
            if it % 50 == 0:
                lr = min(lr * 1.02, 0.05)
            if verbose and it % 100 == 0:
                print(f"          Subgrad it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.5f}")
        else:
            stall += 1
            a = best_a.copy()
            if stall % 30 == 0:
                lr *= 0.7
                T *= 1.5  # Make softmax sharper
                if lr < 1e-6:
                    break
                v = np.zeros(n)
                stall = 0

    return best_a


def coordinate_descent_line_search(a, time_budget=120.0, verbose=False):
    """
    Coordinate descent with bisection line search along each coordinate.
    For each coordinate i, find the optimal a[i] value (with others fixed)
    by bisection on the 1D objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    pass_count = 0
    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count * 7919 + n)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            # Bisection to find optimal a[i]
            # Objective as function of a[i] (with others fixed, after re-normalization)
            # a[i] can range from 0 to 1 (in practice, much less)
            lo = 0.0
            hi = max(a[i] * 3, 0.1)  # upper bound
            a_save = a[i]

            # Try to find the minimum via ternary search on [lo, hi]
            # f(x) = obj with a[i] = x (after renormalization)
            def obj_1d(x):
                a_try = a.copy()
                a_try[i] = x
                s_try = float(np.sum(a_try))
                if s_try < 0.01:
                    return np.inf
                a_try = a_try / s_try
                return compute_obj(a_try)

            # Coarse search first
            best_x = a_save
            best_f = obj_1d(a_save)
            for frac in [0.1, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 0.0]:
                x_try = a_save * (1 + (frac - 1.0)) if a_save > 0 else frac * 0.01
                x_try = max(0.0, min(hi, x_try))
                f_try = obj_1d(x_try)
                if f_try < best_f:
                    best_f = f_try
                    best_x = x_try

            if best_f < best_obj - 1e-13:
                best_obj = best_f
                best_a[i] = best_x
                a[i] = best_x
                s = float(np.sum(a))
                a = a / s
                improved = True

        if verbose and pass_count % 3 == 0:
            print(f"          CD-LS pass {pass_count}: obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")

        if not improved:
            break

    return best_a


def lp_alternating(a, time_budget=200.0, verbose=False):
    """
    LP alternating with adaptive active set and convex line search.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    active_set = None
    threshold_frac = 0.95

    max_iters = 300
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Choose active set
        conv_cur = compute_conv(a)
        max_conv = np.max(conv_cur)

        if active_set is None or it % 20 == 0:
            # Adaptive threshold: start wide, narrow down
            if stall_count > 10:
                threshold_frac = min(0.99, threshold_frac + 0.01)
            else:
                threshold_frac = max(0.85, threshold_frac - 0.005)

            threshold = max_conv * threshold_frac
            active_set = np.where(conv_cur >= threshold)[0]

            # Always include the top few peaks
            top_k = np.argsort(conv_cur)[::-1][:min(10, 2 * n - 1)]
            active_set = np.unique(np.concatenate([active_set, top_k]))

        # Solve LP
        if len(active_set) < 2 * n - 1:
            g_lp, M_lp = solve_lp_reduced(a, active_set)
            if g_lp is None:
                g_lp, M_lp = solve_lp_full(a)
        else:
            g_lp, M_lp = solve_lp_full(a)

        if g_lp is None:
            break

        # Convex line search using FFT-based evaluation
        a_norm = a / np.sum(a)
        g_norm = g_lp / np.sum(g_lp)

        conv_aa = compute_conv(a_norm)
        conv_ag = compute_conv_cross(a_norm, g_norm)
        conv_gg = compute_conv(g_norm)

        def obj_at_t(t):
            val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
            return 2.0 * n * float(np.max(val))

        # Ternary search
        lo, hi = 0.0, 1.0
        for _ in range(12):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = obj_at_t(m1)
            f2 = obj_at_t(m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1

        t_opt = (lo + hi) / 2.0
        a_new = (1 - t_opt) * a_norm + t_opt * g_norm
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose and it % 10 == 0:
                print(f"          LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, |AS|={len(active_set)}")
        else:
            stall_count += 1
            if stall_count > 15:
                # Perturb to escape
                rng = np.random.default_rng(hash((it, n)) % (2**32))
                a_pert = best_a * np.exp(rng.normal(0, 0.003, n))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / np.sum(a_pert)
                a = a_pert
                active_set = None
                stall_count = 0
                threshold_frac = 0.95

    return best_a


def multi_start_optimize(a_init, n, time_budget=200.0, verbose=False):
    """
    Try multiple perturbations of the initial sequence and optimize each.
    Return the best result.
    """
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    rng = np.random.default_rng(12345 + n)

    for attempt in range(5):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 30:
            break

        # Create perturbed version
        if attempt == 0:
            a_try = a_init.copy()
        elif attempt == 1:
            a_try = a_init * np.exp(rng.normal(0, 0.05, n))
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)
        elif attempt == 2:
            # Add sinusoidal perturbation
            freq = rng.integers(2, max(3, n // 10))
            a_try = a_init * (1 + 0.1 * np.sin(2 * np.pi * freq * np.arange(n) / n))
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)
        elif attempt == 3:
            # Random sparse perturbation
            a_try = a_init.copy()
            idx = rng.choice(n, size=n // 5, replace=False)
            a_try[idx] *= rng.uniform(0.5, 1.5, len(idx))
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)
        else:
            # Mix with random
            a_rand = rng.exponential(1, n)
            a_try = 0.7 * a_init + 0.3 * a_rand / np.sum(a_rand)
            a_try = a_try / np.sum(a_try)

        # Quick optimization
        alloc = min(remaining * 0.4, 40.0)
        a_opt = lp_alternating(a_try, time_budget=alloc, verbose=False)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"            Multi-start attempt {attempt}: obj={best_obj:.8f}")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, kind='cubic' if n > 3 else 'linear')
    return np.maximum(a_new, 0)
