"""
Optimized step function search: combines fast LP-alternating with dual-guided
directions, mirror descent, adaptive perturbations, polynomial parameterization,
and multi-scale refinement.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements over parent: dual-informed step direction, polynomial subspace
search, entropy-regularized smoothing, and adaptive active-set management.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Compute linear convolution a*b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_response(a):
    """
    Solve LP: min M s.t. conv(a, g) <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, active_indices, dual_lambda).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None, None, None
    a = a / s
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    conv_val = A @ g_val
                    active = np.where(conv_val >= M_val - 1e-5)[0]
                    # Extract dual variables (lambda on the convolution constraints)
                    dual_vals = None
                    try:
                        # The dual of the constraints A@g <= M
                        # cvxpy stores duals on the constraint objects
                        dual_list = []
                        for c in constraints:
                            if hasattr(c, 'dual_value') and c.dual_value is not None:
                                dual_list.append(np.asarray(c.dual_value).ravel())
                        # First constraint is A@g <= M (2n-1 values)
                        if len(dual_list) > 0 and len(dual_list[0]) == 2 * n - 1:
                            dual_vals = dual_list[0]
                    except Exception:
                        dual_vals = None
                    if dual_vals is not None:
                        dual_vals = np.maximum(dual_vals, 0)
                        ds = float(np.sum(dual_vals))
                        if ds > 1e-10:
                            dual_vals = dual_vals / ds
                    return g_val, M_val, active, dual_vals
        except Exception:
            continue

    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                conv_val = A @ g_val
                active = np.where(conv_val >= M_val - 1e-5)[0]
                return g_val, M_val, active, None
    except Exception:
        pass

    return None, None, None, None


def compute_pressure(a, dual_lambda):
    """
    Compute the 'pressure' on each position: (a * lambda)(j) for all j.
    This tells us which positions are 'pushed' by the active lags.
    """
    if dual_lambda is None:
        return None
    n = len(a)
    # pressure(j) = sum_k lambda(k) * A(k,j) = (a * lambda) correlation
    # p(j) = sum_k lambda(k) * a(k-j) where 0<=k-j<n
    # This is the cross-correlation of lambda with a (reversed)
    m = 2 * n - 1
    size = 1
    while size < m:
        size <<= 1
    fa = rfft(a, size)
    fl = rfft(dual_lambda, size)
    # correlation: sum_k lambda(k) * a(k-j) = (reverse(a) * lambda)(n-1-j)
    a_rev = a[::-1]
    fa_rev = rfft(a_rev, size)
    cross = irfft(fa_rev * fl, size)
    # We want p(j) = sum_k lambda(k) * a(k-j) for valid k, j
    # = (a_rev * lambda)(n-1-j)
    p = cross[n - 1: n - 1 + n][::-1]
    # Actually let me just do it directly for correctness
    p = np.zeros(n)
    for j in range(n):
        # p(j) = sum_k lambda(k) * a(k-j) where 0 <= k-j < n
        k_min = j
        k_max = min(m - 1, j + n - 1)
        if k_min <= k_max:
            ks = np.arange(k_min, k_max + 1)
            p[j] = np.sum(dual_lambda[ks] * a[ks - j])
    return p


def lp_alternating_dual_guided(a, time_budget=300.0, verbose=False, momentum_beta=0.5):
    """
    LP alternating with dual-guided direction and momentum.
    Uses the dual variables to compute a pressure-based step direction.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    momentum = np.zeros(n)
    last_t = 0.3

    while time.monotonic() - start < time_budget:
        g_lp, M_lp, active_idx, dual_lambda = solve_lp_for_response(a)
        if g_lp is None:
            break

        # Compute dual-guided pressure direction
        direction = g_lp - a
        if dual_lambda is not None:
            pressure = compute_pressure(a, dual_lambda)
            if pressure is not None:
                # Normalize pressure to have zero mean
                p_centered = pressure - np.mean(pressure)
                # Scale to be comparable to direction
                p_scale = np.sum(p_centered ** 2)
                if p_scale > 1e-12:
                    p_dir = p_centered / np.sqrt(p_scale)
                    d_scale = np.sum(direction ** 2)
                    if d_scale > 1e-12:
                        d_dir = direction / np.sqrt(d_scale)
                        # Blend: 70% LP direction, 30% pressure direction
                        direction = 0.7 * d_dir + 0.3 * (-p_dir)
                        direction = direction * np.sqrt(d_scale)

        # Apply momentum
        momentum = momentum_beta * momentum + (1 - momentum_beta) * direction

        # Adaptive step candidates
        step_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        if 0 < last_t < 1.0:
            extra = [last_t * 0.5, last_t * 0.8, last_t * 1.2, last_t * 1.5]
            step_candidates = sorted(set(step_candidates + [x for x in extra if 0.02 <= x <= 2.0]))

        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = a.copy()

        for t in step_candidates:
            a_try = a + t * momentum
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            stall = 0
            if verbose and (stall == 0 or stall % 20 == 0):
                elapsed = time.monotonic() - start
                print(f"    LP-dual: obj={best_obj:.8f}, t={best_t_val:.3f}, time={elapsed:.0f}s")
        else:
            stall += 1
            momentum *= 0.3
            if stall % 5 == 0:
                # Adaptive perturbation based on stall count
                scale = 0.01 + 0.005 * min(stall // 5, 20)
                rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2 ** 31))
                # Structured perturbation: target the high-pressure positions
                if dual_lambda is not None:
                    pressure = compute_pressure(a, dual_lambda)
                    if pressure is not None:
                        # Perturb more at high-pressure positions
                        p_norm = (pressure - pressure.min()) / (pressure.max() - pressure.min() + 1e-12)
                        noise = rng.normal(0, 1, n) * (0.5 + p_norm)
                        a = best_a * np.exp(scale * noise)
                else:
                    a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0
                last_t = 0.5

    return best_a


def mirror_descent_smooth(a, time_budget=150.0, verbose=False):
    """
    Mirror descent with entropic regularization on the simplex.
    Uses the log-sum-exp smooth approximation of the max.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        a = np.ones(n) / n
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Work in log space for mirror descent
    log_a = np.log(np.maximum(a, 1e-15))

    beta_lse = 50.0  # smoothness parameter for log-sum-exp
    lr = 0.02
    momentum = np.zeros(n)
    beta_mom = 0.8

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break

        # Recover a from log_a
        a_cur = np.exp(log_a - np.max(log_a))
        a_cur = a_cur / np.sum(a_cur)

        # Compute convolution
        conv = compute_conv(a_cur)
        max_val = float(np.max(conv))

        # Smooth approximation of max using log-sum-exp
        # Gradient of log-sum-exp(beta * conv) w.r.t. a
        weights = np.exp(beta_lse * (conv - max_val))
        weights = weights / np.sum(weights)

        # Gradient of smooth max w.r.t. a
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-10:
                continue
            k_int = int(k)
            i_max = min(n - 1, k_int)
            i_min = max(0, k_int - n + 1)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += w * a_cur[k_int - indices]
        grad *= 2.0 * n  # scale for the objective

        # Center gradient (project to tangent of simplex)
        grad_centered = grad - np.mean(grad)

        # Mirror descent step in log space
        momentum = beta_mom * momentum - lr * grad_centered
        log_a = log_a + momentum

        # Renormalize in log space
        log_a = log_a - np.max(log_a)
        a_new = np.exp(log_a)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        # Adaptive learning rate
        if it % 100 == 99:
            lr = max(lr * 0.95, 0.001)
            beta_lse = min(beta_lse * 1.05, 200.0)
        if verbose and it % 500 == 0:
            print(f"    Mirror: it={it}, obj={best_obj:.8f}, lr={lr:.4f}")

    return best_a


def subgradient_polish(a, time_budget=150.0, verbose=False):
    """
    Nesterov-accelerated subgradient method targeting active peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        a = np.ones(n) / n
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()
    y = a.copy()
    beta = 0.7
    lr = 0.04
    momentum = np.zeros(n)

    for it in range(8000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(x)
        max_val = float(np.max(conv))
        threshold = max_val * 0.85
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = [int(np.argmax(conv))]
        if len(active) > 25:
            active = np.argsort(conv)[::-1][:25]

        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += y[k - indices]
        grad /= len(active)

        grad_centered = grad - np.mean(grad)
        momentum = beta * momentum - lr * grad_centered
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new

        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        y = y_new
        obj = compute_obj(y)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()

        if it % 300 == 299:
            lr = max(lr * 0.92, 0.001)

    return best_a


def polynomial_search(a_best, time_budget=120.0, verbose=False):
    """
    Search in a polynomial subspace: a(j) = max(0, sum_i c_i * (j/(n-1))^i) / norm.
    This reduces the problem to optimizing a few coefficients.
    """
    n = len(a_best)
    start = time.monotonic()
    best_a = a_best.copy()
    best_obj = compute_obj(a_best)

    x = np.linspace(0, 1, n)

    # Try different polynomial degrees
    for degree in [2, 3, 4, 5]:
        if time.monotonic() - start > time_budget:
            break

        # Initialize coefficients from the current best (least squares fit)
        V = np.vander(x, degree + 1, increasing=True)
        coeffs, _, _, _ = np.linalg.lstsq(V, a_best, rcond=None)

        # Optimize coefficients with Nelder-Mead
        from scipy.optimize import minimize

        def poly_obj(c):
            vals = np.zeros(n)
            for i in range(len(c)):
                vals += c[i] * x ** i
            vals = np.maximum(vals, 0)
            s = np.sum(vals)
            if s < 1e-10:
                return np.inf
            vals = vals / s
            return compute_obj(vals)

        from scipy.optimize import minimize as sp_min
        result = sp_min(poly_obj, coeffs, method='Nelder-Mead',
                        options={'maxiter': 5000, 'xatol': 1e-8, 'fatol': 1e-10})

        if result.fun < best_obj:
            c = result.x
            vals = np.zeros(n)
            for i in range(len(c)):
                vals += c[i] * x ** i
            vals = np.maximum(vals, 0)
            s = np.sum(vals)
            if s > 1e-10:
                vals = vals / s
                obj = compute_obj(vals)
                if obj < best_obj:
                    best_obj = obj
                    best_a = vals.copy()
                    if verbose:
                        print(f"    Poly deg={degree}: obj={best_obj:.8f}")

    return best_a


def coordinate_polish(a, time_budget=80.0):
    """
    Focused coordinate adjustment targeting the top convolution peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for iteration in range(1000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_conv = float(conv[k_star])

        # Find positions that contribute most to the peak
        contributions = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contributions[j] = a[j] * a[k_j]

        sorted_idx = np.argsort(contributions)[::-1]
        top_k = sorted_idx[:15]
        bottom_k = np.argsort(contributions)[:15]

        improved = False

        # Try moving mass from top contributors to bottom contributors
        for i in top_k:
            if time.monotonic() - start > time_budget:
                break
            for target in bottom_k:
                if target == i:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    delta = frac * a[i]
                    if delta <= 0 or delta > a[i] * 0.9:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_max = float(np.max(new_conv))
                    if new_max < max_conv - 1e-10:
                        max_conv = new_max
                        conv = new_conv
                        obj = 2.0 * n * new_max
                        if obj < best_obj - 1e-14:
                            best_obj = obj
                            best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random pairs
            rng = np.random.default_rng(int(time.monotonic() * 1e4) % (2 ** 31))
            for _ in range(10):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta = 0.005 * min(a[i], a[j])
                if delta <= 0:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and iteration > 200:
            break

    return best_a


def generate_initialization(kind, n, rng):
    """Generate a diverse initialization of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'sine_taper':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.1
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'bilinear':
        a = np.linspace(1, 2, n)
        return a / np.sum(a)
    elif kind == 'plateau':
        a = np.ones(n)
        edge = max(1, n // 10)
        a[:edge] = 0.3
        a[-edge:] = 0.3
        return a / np.sum(a)
    elif kind == 'raised_cos':
        # (1 - cos(2*pi*x)) type
        x = np.linspace(0, 1, n)
        a = 1 - np.cos(2 * np.pi * x)
        return a / np.sum(a)
    elif kind == 'power_law':
        x = np.arange(1, n + 1)
        a = x ** (-0.5)
        return a / np.sum(a)
    elif kind == 'two_peak':
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n * 0.3) / (n * 0.1)) ** 2) + \
            np.exp(-0.5 * ((x - n * 0.7) / (n * 0.1)) ** 2)
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    from scipy.interpolate import CubicSpline
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    cs = CubicSpline(x_old, a, bc_type='natural')
    a_new = cs(x_new)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining multiple optimization strategies for the
    step function autocorrelation minimization problem.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    n_parent = len(best_seq)

    # === Phase 1: Diverse Initializations (80s) ===
    print(f"\n=== Phase 1: Diverse Initializations (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    init_budget = min(80.0, remaining * 0.05)

    init_kinds = ['cosine', 'sine_taper', 'gaussian', 'random', 'plateau',
                  'bilinear', 'raised_cos', 'power_law', 'two_peak']
    per_init = max(3.0, init_budget / len(init_kinds))

    for kind in init_kinds:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        a_init = generate_initialization(kind, n_parent, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init '{kind}': obj={obj_init:.8f} **")
        else:
            print(f"  Init '{kind}': obj={obj_init:.8f}")

    # Extra random inits
    for _ in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        a_init = generate_initialization('random', n_parent, rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Random init: obj={obj_init:.8f} **")

    a = best_seq.copy()
    print(f"  Best after Phase 1: obj={best_score:.8f}")

    # === Phase 2: LP Alternating with Dual Guidance (500s) ===
    print(f"\n=== Phase 2: LP Alternating (dual-guided) ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(500.0, remaining * 0.40)

    for round_idx in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        beta_vals = [0.3, 0.5, 0.7, 0.9, 0.4]
        beta = beta_vals[round_idx % len(beta_vals)]
        alloc = min(lp_budget / (5 - round_idx), remaining * 0.65)
        print(f"  LP Round {round_idx} (beta={beta}): ")
        a_new = lp_alternating_dual_guided(a, time_budget=alloc, verbose=True,
                                            momentum_beta=beta)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  ** LP R{round_idx}: obj={best_score:.8f} **")
        else:
            print(f"  LP R{round_idx}: obj={obj:.8f}")
            if round_idx > 0:
                scale = 0.02 + 0.01 * (round_idx - 1)
                a = best_seq * np.exp(rng.normal(0, scale, n_parent))
                a = np.maximum(a, 0)
                a = a / np.sum(a)

    # === Phase 3: Mirror Descent (150s) ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        md_budget = min(150.0, remaining * 0.15)
        a_md = mirror_descent_smooth(a, time_budget=md_budget, verbose=True)
        obj_md = compute_obj(a_md)
        if obj_md < best_score:
            best_score = obj_md
            best_seq = a_md.copy()
            a = a_md.copy()
            print(f"  Mirror: obj={best_score:.8f} **")
        else:
            print(f"  Mirror: obj={obj_md:.8f}")

    # === Phase 4: Subgradient Polish (120s) ===
    print(f"\n=== Phase 4: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        sg_budget = min(120.0, remaining * 0.12)
        a_sg = subgradient_polish(a, time_budget=sg_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            a = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f}")

    # === Phase 5: Polynomial Subspace Search (100s) ===
    print(f"\n=== Phase 5: Polynomial Subspace ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        poly_budget = min(100.0, remaining * 0.10)
        a_poly = polynomial_search(a, time_budget=poly_budget, verbose=True)
        obj_poly = compute_obj(a_poly)
        if obj_poly < best_score:
            best_score = obj_poly
            best_seq = a_poly.copy()
            a = a_poly.copy()
            print(f"  Poly: obj={best_score:.8f} **")
        else:
            print(f"  Poly: obj={obj_poly:.8f}")

    # === Phase 6: Coordinate Polish (80s) ===
    print(f"\n=== Phase 6: Coordinate Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        cd_budget = min(80.0, remaining * 0.08)
        a_cd = coordinate_polish(a, time_budget=cd_budget)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            a = a_cd.copy()
            print(f"  Coord: obj={best_score:.8f} **")
        else:
            print(f"  Coord: obj={obj_cd:.8f}")

    # === Phase 7: Length Sweep (200s) ===
    print(f"\n=== Phase 7: Length Sweep ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(100, n_best - 100),
        max(100, n_best - 50),
        max(100, n_best - 20),
        n_best,
        n_best + 20,
        n_best + 50,
        n_best + 100,
        n_best + 200,
    ]))
    n_options = [n for n in n_options if 100 <= n <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
            a_try = np.maximum(a_try, 0)
            a_try = a_try / np.sum(a_try)

        # Quick optimization
        alloc_lp = min(40.0, remaining * 0.35)
        alloc_sg = min(20.0, remaining * 0.15)

        a_opt = lp_alternating_dual_guided(a_try, time_budget=alloc_lp,
                                            momentum_beta=0.5)
        a_opt = subgradient_polish(a_opt, time_budget=alloc_sg)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            n_parent = len(best_seq)
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Final LP Polish (remaining time) ===
    print(f"\n=== Phase 8: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    n_final = len(best_seq)
    a_final = best_seq.copy()
    final_round = 0

    while remaining > 25:
        final_round += 1
        alloc = min(remaining * 0.35, 80.0)
        # Alternate between different momentum values
        beta = [0.4, 0.6, 0.8, 0.5, 0.7][final_round % 5]
        a_new = lp_alternating_dual_guided(a_final, time_budget=alloc,
                                            momentum_beta=beta)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_final = a_new.copy()
            print(f"  Final LP {final_round}: obj={best_score:.8f} **")
        else:
            # Try perturbations
            for scale in [0.01, 0.02, 0.03, 0.05]:
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_final))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / np.sum(a_pert)
                a_try2 = lp_alternating_dual_guided(a_pert, time_budget=min(alloc * 0.5, 30.0),
                                                     momentum_beta=0.5)
                obj2 = compute_obj(a_try2)
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a_try2.copy()
                    a_final = a_try2.copy()
                    print(f"  Final LP {final_round} (pert {scale}): obj={best_score:.8f} **")
                    break
            else:
                a_final = best_seq * np.exp(rng.normal(0, 0.02, n_final))
                a_final = np.maximum(a_final, 0)
                a_final = a_final / np.sum(a_final)
                print(f"  Final LP {final_round}: obj={obj:.8f} (no improvement)")

        remaining = deadline - time.monotonic()

    # === Phase 9: Final coordinate micro-polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd = coordinate_polish(best_seq, time_budget=min(remaining * 0.5, 25.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
