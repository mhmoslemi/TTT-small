"""
Hybrid optimizer: Fast coordinate descent + LP alternating + Nesterov subgradient.
Key innovation: exact 1D coordinate descent that exploits the quadratic structure
of the convolution objective, enabling very fast local optimization. Combined with
LP alternating for global progress and Nesterov acceleration for polynomial-time

import numpy as np
from scipy.fft import rfft, irfft
import time

from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_fast(a, conv=None):
    """Compute objective assuming a is already normalized to sum=1."""
    n = len(a)
    if conv is None:
        conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def nesterov_subgrad(a0, time_budget=60.0, top_k=5):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    step_size = 0.05
    patience = 0

    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                m_start = k - i_end
                m_end = k - i_start
                grad[m_start:m_end + 1] += 2.0 * y[i_start:i_end + 1][::-1]
        grad /= len(top_idx)
        grad = grad - np.mean(grad)
        y_new = y - step_size * grad
        # Project to simplex
        y_sorted = np.sort(y_new)[::-1]
        cssv = np.cumsum(y_sorted) - 1.0
        ind = np.arange(1, n + 1)
        cond = y_sorted - cssv / ind > 0
        if np.any(cond):
            i_star = np.max(ind[cond])
            theta = cssv[i_star - 1] / i_star
            a_new = np.maximum(y_new - theta, 0)
        else:
            a_new = np.ones(n) / n
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = np.maximum(y, 0)
        ys = float(np.sum(y))
        if ys > 0:
            y = y / ys
        a = a_new
        if it % 5 == 0 or it < 10:
            obj = compute_obj_fast(a)
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 300:
                step_size = max(step_size * 0.7, 0.002)
                patience = 0
        if it % 1000 == 999:
            step_size = max(step_size * 0.97, 0.003)

    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_norm, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=100.0):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    stall_count = 0
    last_good_t = 0.5

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        # 1D line search
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                return np.inf
            a_try = a_try / st
            return compute_obj_fast(a_try)
        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 25, 'xatol': 0.02})
            best_t = res.x if res.success else last_good_t
        except Exception:
            best_t = last_good_t
        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj_fast(a_try)
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            last_good_t = best_t
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 10:
                break

    return best_a


def coordinate_descent(a0, time_budget=60.0, n_sweeps=5):
    """
    Fast coordinate descent exploiting the quadratic structure.
    For each coordinate m, optimizes the 1D subproblem exactly.
    """

        return a0.copy()

        if time.monotonic() - start > time_budget:
            break
        # Randomize order for better convergence

        for m in order:
            if time.monotonic() - start > time_budget:
                break

            am = a[m]

                    a[m] = am
                    if obj_t < best_obj:
                        a = a.copy()  # will be overwritten
                        a[m] = t_trial
                        best_obj = obj_t
                        best_a = a.copy()
                        am = t_trial
                continue

            # Compute base convolution (excluding a[m]'s contribution)
            # conv[k] = base[k] + contribution
            # contribution of a[m] to conv[k]:

            # Subtract a[m]'s contribution

            valid_ks = range(max(0, m), min(2*n-1, 2*m + n))  # k-m in [0, n-1] means k in [m, m+n-1]

                if 0 <= k_minus_m < n:
                    if k_minus_m != m:
                        base[k] -= 2.0 * am * a[k_minus_m]
                    else:
                        base[k] -= am * am

                S_new = S_others + t
                if S_new < 1e-15:
                    return np.inf
                # Compute max of (base[k] + c_k * t + d_k * t^2) / still need to compute
                # For efficiency, precompute c_k and d_k
                # c_k = 2*a[k-m] if 0<=k-m<n and k-m!=m, else 0
                # d_k = 1 if k==2m, else 0
                conv_new = base.copy()
                for k in range(max(0, m), min(2*n-1, m + n)):
                    k_minus_m = k - m
                    if 0 <= k_minus_m < n:
                        if k_minus_m != m:
                            conv_new[k] += 2.0 * t * a[k_minus_m]
                        else:
                            conv_new[k] += t * t
                max_conv = np.max(conv_new)
                return 2.0 * n * max_conv / (S_new * S_new)
            
            # Minimize
            try:
                res = minimize_scalar(obj_1d, bounds=(0.0, am * 3 + 0.1), method='bounded',
                                      options={'maxiter': 30, 'xatol': 1e-5})
                t_opt = res.x
            except Exception:
                t_opt = am

            # Renormalize

            if s_new < 1e-10:
                a[m] = am
                continue
            a = a / s_new
            obj_new = compute_obj_fast(a)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()

    return best_a

"""
Hybrid optimizer: Fast coordinate descent + LP alternating + Nesterov subgradient.
Key innovation: exact 1D coordinate descent exploiting quadratic structure of
convolution, combined with LP alternating for global progress. Multiple n-values
and diverse initializations to avoid local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_obj_norm(a):
    """Compute objective assuming a is normalized to sum=1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def _cd_objective(t, base, c_vec, d_idx, S_others, n):
    """1D objective for coordinate descent at coordinate m."""
    S_new = S_others + t
    if S_new < 1e-15:
        return np.inf
    vals = base + c_vec * t
    if d_idx >= 0:
        vals = vals.copy()
        vals[d_idx] += t * t
    return 2.0 * n * float(np.max(vals)) / (S_new * S_new)


def coordinate_descent(a0, time_budget=60.0, max_sweeps=20):
    """
    Fast coordinate descent exploiting the quadratic structure of convolution.
    For each coordinate m, solves the exact 1D subproblem.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        order = np.arange(n) if sweep == 0 else rng.permutation(n)
        any_improve = False

        for m in order:
            if time.monotonic() - start > time_budget:
                break
            m = int(m)
            am = a[m]
            S_others = 1.0 - am

            # Precompute c_vec and d_idx
            c_vec = np.zeros(2 * n - 1)
            d_idx = -1
            k_start = max(0, m)
            k_end = min(2 * n - 2, m + n - 1)
            for k in range(k_start, k_end + 1):
                km = k - m
                if 0 <= km < n:
                    if km != m:
                        c_vec[k] = 2.0 * a[km]
                    else:
                        d_idx = k

            # Compute base (conv minus a[m]'s contribution)
            conv = compute_conv(a)
            base = conv - am * c_vec
            if d_idx >= 0:
                base = base - am * am * np.array([1.0 if i == d_idx else 0.0 for i in range(2 * n - 1)])

            # 1D minimization
            try:
                upper = max(am * 4.0 + 0.1, 0.05)
                res = minimize_scalar(
                    lambda t: _cd_objective(t, base, c_vec, d_idx, S_others, n),
                    bounds=(0.0, upper),
                    method='bounded', options={'maxiter': 20, 'xatol': 1e-6}
                )
                t_opt = max(0.0, res.x)
            except Exception:
                t_opt = am

            if abs(t_opt - am) < 1e-10:
                continue

            a[m] = t_opt
            sn = float(np.sum(a))
            if sn < 1e-10:
                a[m] = am
                continue
            a = a / sn
            obj_new = compute_obj_norm(a)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
                any_improve = True

        if not any_improve:
            break

    return best_a


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def _line_search_obj(t, a, g_lp):
    """Objective for LP alternating line search."""
    a_try = (1.0 - t) * a + t * g_lp
    a_try = np.maximum(a_try, 0)
    st = float(np.sum(a_try))
    if st < 1e-10:
        return np.inf
    a_try = a_try / st
    return compute_obj_norm(a_try)


def lp_alternate(a0, time_budget=100.0):
    """LP alternating with 1D line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    stall_count = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        try:
            res = minimize_scalar(
                lambda t: _line_search_obj(t, a, g_lp),
                bounds=(0.0, 1.0), method='bounded',
                options={'maxiter': 20, 'xatol': 0.02}
            )
            best_t = res.x if res.success else 0.3
        except Exception:
            best_t = 0.3
        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj_norm(a_try)
        if obj_try < best_obj - 1e-13:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 12:
                break

    return best_a


def nesterov_subgrad(a0, time_budget=60.0, top_k=5):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj_norm(a)
    start = time.monotonic()
    step_size = 0.04
    patience = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                m_start = k - i_end
                m_end = k - i_start
                grad[m_start:m_end + 1] += 2.0 * y[i_start:i_end + 1][::-1]
        grad /= len(top_idx)
        grad = grad - np.mean(grad)
        y_new = y - step_size * grad
        # Project to simplex
        y_sorted = np.sort(y_new)[::-1]
        cssv = np.cumsum(y_sorted) - 1.0
        ind = np.arange(1, n + 1)
        cond = y_sorted - cssv / ind > 0
        if np.any(cond):
            i_star = np.max(ind[cond])
            theta = cssv[i_star - 1] / i_star
            a_new = np.maximum(y_new - theta, 0)
        else:
            a_new = np.ones(n) / n
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = np.maximum(y, 0)
        ys = float(np.sum(y))
        if ys > 0:
            y = y / ys
        a = a_new
        if it % 5 == 0 or it < 10:
            obj = compute_obj_norm(a)
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 400:
                step_size = max(step_size * 0.7, 0.002)
                patience = 0
        if it % 2000 == 1999:
            step_size = max(step_size * 0.95, 0.003)

    return best_a


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw


def derichlet_window(n, order=3):
    """Derichlet (flat-top) window normalized to sum=1."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    ratio = np.abs(x - center) / max(1, center)
    w = np.ones(n)
    for k in range(1, order + 1):
        coeff = 1.0 - k / order
        w *= coeff + (1 - coeff) * np.cos(np.pi * ratio * (2 * k - 1) / (2.0 / (2 * k - 1)))
    # Simpler: use the standard flat-top formula
    w = 0.21557895 - 0.41663158 * np.cos(2 * np.pi * x / (n - 1)) + \
        0.277263158 * np.cos(4 * np.pi * x / (n - 1)) - \
        0.083578947 * np.cos(6 * np.pi * x / (n - 1)) + \
        0.006947368 * np.cos(8 * np.pi * x / (n - 1))
    w = np.maximum(w, 0)
    sw = np.sum(w)
    if sw < 1e-10:
        return np.ones(n) / n
    return w / sw
