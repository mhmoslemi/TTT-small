from scipy.optimize import minimize, linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

def build_conv_matrix(a):

    A_ub[:, :n] = A

def lp_alternate(a, time_budget=300.0, verbose=False):
    """
    LP alternating method with golden section line search and simulated annealing.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall = 0
    patience = 30
    temperature = 0.1
    iteration = 0
    
    # Golden section constant
    golden = (1.0 - np.sqrt(5.0)) / 2.0

    while time.monotonic() - start < time_budget:
        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        # Check similarity - if LP solution is too similar, we're at a fixed point
        diff_norm = np.linalg.norm(a - g_lp)
        if diff_norm < 1e-6:
            stall += 3
        else:
            # Evaluate objective at the LP solution
            obj_lp = compute_obj(g_lp)
            if obj_lp < best_obj * (1 - 1e-8):
                # LP solution is better than current best
                stall = 0
            else:
                stall += 1

        # Line search: find best t in [0, 1] for a_new = (1-t)*a + t*g_lp
        # Use golden section search
        t_lo, t_hi = 0.0, 1.0
        t1 = t_hi - golden * (t_hi - t_lo)
        t2 = t_lo + golden * (t_hi - t_lo)
        
        def eval_t(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                return np.inf
            return compute_obj(a_try / s_try)
        
        obj_t1 = eval_t(t1)
        obj_t2 = eval_t(t2)
        
        for _ in range(12):  # 12 bisections gives ~4000 possible t values
            if obj_t1 < obj_t2:
                t_hi = t2
                t2 = t1
                obj_t2 = obj_t1
                t1 = t_hi - golden * (t_hi - t_lo)
                obj_t1 = eval_t(t1)
            else:
                t_lo = t1
                t1 = t2
                obj_t1 = obj_t2
                t2 = t_lo + golden * (t_hi - t_lo)
                obj_t2 = eval_t(t2)
        
        t_best = (t_lo + t_hi) / 2.0
        obj_best = eval_t(t_best)
        
        # Also try t=1.0 (full LP solution)
        obj_t1_full = compute_obj(g_lp)
        if obj_t1_full < obj_best:
            t_best = 1.0
            obj_best = obj_t1_full
        
        # Accept/reject with simulated annealing
        if obj_best < best_obj - 1e-12:
            best_obj = obj_best
            best_a = ((1.0 - t_best) * a + t_best * g_lp)
            best_a = np.maximum(best_a, 0)
            best_a = best_a / best_a.sum()
            a = best_a.copy()
            stall = 0
            temperature = max(0.01, temperature * 0.99)
        else:
            # Annealing: accept with probability exp(-(obj_best - best_obj) / temperature)
            if np.random.random() < np.exp(-(obj_best - best_obj) / max(temperature, 1e-10)):
                a = (1.0 - t_best) * a + t_best * g_lp
                a = np.maximum(a, 0)
                a = a / a.sum()
            stall += 1

                a = np.maximum(a, 1e-15)

                # Occasionally try a random restart

                stall = 0
                temperature = min(0.3, temperature * 1.5)

def peak_shave(a, time_budget=60.0, verbose=False):
    """
    Peak shaving: identify top peaks in convolution and try to reduce them
    by moving mass from contributing positions to non-contributing ones.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(best_a)
        # Get top 5 peaks
        peak_indices = np.argsort(conv)[::-1][:5]
        
        improved = False
        for k in peak_indices:
            k = int(k)
            # Contributors to conv[k]: pairs (i, k-i) where 0<=i<n and 0<=k-i<n
            contributors = []
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                contributors.append((best_a[i] * best_a[j], i, j))
            contributors.sort(reverse=True)
            
            # Non-contributors (indices not involved in this peak)
            contrib_set = set()
            for _, i, j in contributors:
                contrib_set.add(i)
                contrib_set.add(j)
            non_contrib = [idx for idx in range(n) if idx not in contrib_set]
            
            if not non_contrib:
                continue
            
            # Try moving mass from top contributors to non-contributors
            for contrib_val, i, j in contributors[:10]:
                if best_a[i] * best_a[j] < 1e-10:
                    break
                src = i if best_a[i] >= best_a[j] else j
                for target in non_contrib[:5]:
                    for frac in [0.005, 0.01, 0.02, 0.05]:
                        delta = frac * best_a[src]
                        if delta < 1e-12 or delta >= best_a[src]:
                            continue
                        a_try = best_a.copy()
                        a_try[src] -= delta
                        a_try[target] += delta
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-12:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            improved = True
                            break
                        else:
                            break  # larger frac won't help if this didn't
                    
                    if improved:
                        break
            
            if improved:
                break
        
        if not improved:
            break
    
    if verbose:
        print(f"      PeakShave: obj={best_obj:.8f}, t={time.monotonic()-start:.1f}s")
    return best_a
