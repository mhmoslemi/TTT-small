import cvxpy as cp

from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    c = compute_conv(a_n)
    return 2.0 * n * float(np.max(c))


def compute_obj_normalized(a):
    """Assume a is already normalized (sum = 1)."""
    n = len(a)
    c = compute_conv(a)
    return 2.0 * n * float(np.max(c))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. A_a @ g <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)

def golden_section_search(a, g, num_evals=35):
    """Find t in [0,1] minimizing f((1-t)*a + t*g) using golden section."""
    phi = (np.sqrt(5) + 1) / 2  # golden ratio ≈ 1.618
    gamma = 1 - 1 / phi  # ≈ 0.382
    
    lo, hi = 0.0, 1.0
    c = lo + gamma * (hi - lo)  # ≈ 0.382
    d = hi - gamma * (hi - lo)  # ≈ 0.618
    
    fc = compute_obj_normalized((1 - c) * a + c * g)
    fd = compute_obj_normalized((1 - d) * a + d * g)
    
    for _ in range(num_evals - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + gamma * (hi - lo)
            fc = compute_obj_normalized((1 - c) * a + c * g)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gamma * (hi - lo)
            fd = compute_obj_normalized((1 - d) * a + d * g)
    
    t_opt = (lo + hi) / 2.0
    return t_opt, compute_obj_normalized((1 - t_opt) * a + t_opt * g)


def subgradient_step(a, num_active=5, lr=0.01):
    """One subgradient step on the max convolution objective."""
    n = len(a)
    conv = compute_conv(a)
    # Get top-k active lags
    top_k = np.argsort(conv)[::-1][:num_active]
    max_val = conv[top_k[0]]
    if max_val < 1e-15:
        return a.copy()
    
    # Weights: softmax over active values
    weights = np.exp((conv[top_k] - max_val) / (max_val * 0.1))
    weights /= weights.sum()
    
    # Compute subgradient
    grad = np.zeros(n)
    for w, k in zip(weights, top_k):
        k = int(k)
        # grad[m] += 2*w*a[k-m] for valid m
        m_start = max(0, k - (n - 1))
        m_end = min(n - 1, k)
        if m_start <= m_end:
            # a[k-m] for m in [m_start, m_end]
            indices = k - np.arange(m_start, m_end + 1)
            grad[m_start:m_end + 1] += 2.0 * w * a[indices]
    
    # Multiplicative update (mirror descent on simplex)
    a_new = a * np.exp(-lr * grad)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new


def smooth_optimize(a, time_budget=60.0, beta_start=50.0, beta_end=3000.0):
    """Smooth optimization using log-sum-exp with mirror descent."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / max(1, int(time_budget)))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        conv = compute_conv(a)
        m = conv.max()
        shifted = beta * (conv - m)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Gradient
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-10:
                continue
            m_start = max(0, k - (n - 1))
            m_end = min(n - 1, k)
            if m_start <= m_end:
                indices = k - np.arange(m_start, m_end + 1)
                grad[m_start:m_end + 1] += 2.0 * weights[k] * a[indices]
        
        lr = 1.0 / np.sqrt(beta * n)
        log_a = np.log(a)
        log_a_new = log_a - lr * grad
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        
        it += 1
    
    return best_a

"""
Step function optimization using LP alternating with golden section step size,
subgradient refinement, and smooth log-sum-exp polishing.
Combines fast LP solves with adaptive step sizing and stalling recovery
to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Target: score < 1.5027
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_n = a / s
    c = compute_conv(a_n)
    return 2.0 * n * float(np.max(c))


def compute_obj_norm(a):
    """Assume a is already normalized (sum=1, non-negative)."""
    n = len(a)
    c = compute_conv(a)
    return 2.0 * n * float(np.max(c))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A such that A@g = np.convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. A_a @ g <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_n = a / np.sum(a)
    A = build_conv_matrix(a_n)

    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    # Try solvers in order of expected performance
    solver_list = []
    for name in ['MOSEK', 'GUROBI', 'CLARABEL', 'PDLP', 'ECOS', 'SCS', 'XPRESS', 'GLOP']:
        try:
            attr = getattr(cp, name, None)
            if attr is not None:
                solver_list.append(attr)
        except Exception:
            pass

    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0.0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M_var.value)
        except Exception:
            continue

    # Fallback: scipy HiGHS
    try:
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def golden_section_search(a, g, num_evals=35):
    """
    Find t in [0,1] minimizing f((1-t)*a + t*g) using golden section search.
    Both a and g must be on the simplex (non-negative, sum=1).
    Returns (t_opt, f_opt).
    """
    gamma = (np.sqrt(5.0) - 1.0) / 2.0  # ≈ 0.618
    lo = 0.0
    hi = 1.0
    c = lo + (1.0 - gamma) * (hi - lo)
    d = lo + gamma * (hi - lo)

    fc = compute_obj_norm((1.0 - c) * a + c * g)
    fd = compute_obj_norm((1.0 - d) * a + d * g)

    for _ in range(num_evals - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1.0 - gamma) * (hi - lo)
            fc = compute_obj_norm((1.0 - c) * a + c * g)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gamma * (hi - lo)
            fd = compute_obj_norm((1.0 - d) * a + d * g)

    t_opt = (lo + hi) / 2.0
    f_opt = compute_obj_norm((1.0 - t_opt) * a + t_opt * g)
    return t_opt, f_opt


def subgradient_step(a, num_active=5, lr=0.01, temperature=0.1):
    """
    One mirror-descent subgradient step on max(conv(a,a)).
    Uses softmax-weighted top-k active lags.
    """
    n = len(a)
    conv = compute_conv(a)
    max_val = float(conv.max())
    if max_val < 1e-15:
        return a.copy()

    # Get top-k active lags
    top_k = np.argsort(conv)[::-1][:num_active]
    active_vals = conv[top_k]

    # Softmax weights
    w = np.exp((active_vals - active_vals[0]) / (max_val * temperature))
    w /= w.sum()

    # Compute subgradient: grad[m] = 2 * sum_k w_k * a[k-m] (for valid indices)
    grad = np.zeros(n)
    for weight, k in zip(w, top_k):
        k = int(k)
        m_start = max(0, k - (n - 1))
        m_end = min(n - 1, k)
        if m_start <= m_end:
            idx = k - np.arange(m_start, m_end + 1)
            grad[m_start:m_end + 1] += 2.0 * weight * a[idx]

    # Mirror descent (multiplicative) update
    a_new = a * np.exp(-lr * grad)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def smooth_optimize(a, time_budget=60.0, beta_start=100.0, beta_end=5000.0):
    """Smooth log-sum-exp optimization with mirror descent."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        frac = min(1.0, it / max(1, int(time_budget * 2)))
        beta = beta_start * (beta_end / beta_start) ** frac

        conv = compute_conv(a)
        m = conv.max()
        shifted = beta * (conv - m)
        weights = np.exp(shifted)
        weights /= weights.sum()

        # Gradient: grad[m] = 2 * sum_k weights[k] * a[k-m]
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-12:
                continue
            m_start = max(0, k - (n - 1))
            m_end = min(n - 1, k)
            if m_start <= m_end:
                idx = k - np.arange(m_start, m_end + 1)
                grad[m_start:m_end + 1] += 2.0 * weights[k] * a[idx]

        lr = 1.0 / np.sqrt(beta * n)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * grad
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample via linear interpolation, preserving non-negativity."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)

            break
        # Resample
        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # One LP iteration
        g_lp, M_lp = solve_lp_minpeak(a_try)
        if g_lp is not None:
            t_opt, obj_opt = golden_section_search(a_try, g_lp, num_evals=25)
            a_new = (1.0 - t_opt) * a_try + t_opt * g_lp
            a_new = a_new / a_new.sum()
            obj = compute_obj(a_new)
            print(f"  n={n_try}: obj={obj:.10f} (LP M={M_lp:.6f})")
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                n_current = n_try
                print(f"  ** Improved: n={n_try}, obj={best_score:.10f}")
        else:
            obj = compute_obj(a_try)
            print(f"  n={n_try}: obj={obj:.10f} (no LP)")

    n_current = len(best_seq)
    print(f"After exploration: n={n_current}, obj={best_score:.10f}")

    # ============ Phase 2: Main LP Alternating (core phase) ============
    print(f"\n=== Phase 2: Main LP Alternating ===")
    remaining = deadline - time.monotonic()
    lp_budget = min(900.0, remaining * 0.60)

    a_main, obj_main = lp_alternating_main(best_seq, time_budget=lp_budget, verbose=True)

    if obj_main < best_score:
        best_score = obj_main
        best_seq = a_main.copy()

    print(f"After main LP: obj={best_score:.10f}")

    # ============ Phase 3: Smooth optimization polish ============
    print(f"\n=== Phase 3: Smooth Optimization ===")
    remaining = deadline - time.monotonic()
    smooth_budget = min(remaining * 0.20, 150.0)

    if smooth_budget > 15:
        a_smooth = smooth_optimize(best_seq, time_budget=smooth_budget,
                                   beta_start=200.0, beta_end=8000.0)
        obj_s = compute_obj(a_smooth)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_smooth.copy()
            print(f"  Smooth: obj={best_score:.10f} **")
        else:
            print(f"  Smooth: obj={obj_s:.10f} (no improvement)")

    # ============ Phase 4: Intensive LP polish ============
    print(f"\n=== Phase 4: Intensive LP Polish ===")
    remaining = deadline - time.monotonic()
    polish_budget = min(remaining * 0.30, 150.0)

    if polish_budget > 15:
        a = best_seq.copy()
        start_p = time.monotonic()
        it = 0
        while time.monotonic() - start_p < polish_budget:
            g_lp, M_lp = solve_lp_minpeak(a)
            if g_lp is None:
                break
            t_opt, obj_opt = golden_section_search(a, g_lp, num_evals=35)
            obj_curr = compute_obj(a)
            if obj_opt < obj_curr - 1e-13:
                a = (1.0 - t_opt) * a + t_opt * g_lp
                a = a / a.sum()
            else:
                # Try perturbed LP direction
                rng = np.random.default_rng(it + 12345)
                eps = 0.05
                g_pert = (1.0 - eps) * g_lp + eps * (rng.random(n_current) ** 2)
                g_pert = np.maximum(g_pert, 0)
                g_pert = g_pert / g_pert.sum()
                t_pert, obj_pert = golden_section_search(a, g_pert, num_evals=20)
                if obj_pert < obj_curr - 1e-13:
                    a = (1.0 - t_pert) * a + t_pert * g_pert
                    a = a / a.sum()
                else:
                    break
            it += 1
            if it % 5 == 0:
                obj_now = compute_obj(a)
                print(f"    Polish it{it}: obj={obj_now:.10f}")

        obj_polish = compute_obj(a)
        if obj_polish < best_score:
            best_score = obj_polish
            best_seq = a.copy()
            print(f"  Polish: obj={best_score:.10f} **")

    # ============ Phase 5: Subgradient micro-refinement ============
    print(f"\n=== Phase 5: Subgradient Micro-Refinement ===")
    remaining = deadline - time.monotonic()
    sg_budget = min(remaining * 0.25, 80.0)

    if sg_budget > 5:
        a = best_seq.copy()
        start_sg = time.monotonic()
        lr = 0.005
        improved_count = 0
        while time.monotonic() - start_sg < sg_budget:
            a_new = subgradient_step(a, num_active=3, lr=lr, temperature=0.05)
            obj_new = compute_obj(a_new)
            obj_curr = compute_obj(a)
            if obj_new < obj_curr - 1e-14:
                a = a_new.copy()
                improved_count += 1
            else:
                lr *= 0.9
            if lr < 1e-4:
                break

        obj_sg = compute_obj(a)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a.copy()
            print(f"  Subgrad: obj={best_score:.10f} ** ({improved_count} improvements)")
        else:
            print(f"  Subgrad: obj={obj_sg:.10f} (no improvement)")

    # ============ Phase 6: Final diverse restarts ============
    print(f"\n=== Phase 6: Final Diversification ===")
    remaining = deadline - time.monotonic()

    if remaining > 30:
        rng_final = np.random.default_rng(seed + 99999)
        for r in range(3):
            if remaining < 10:
                break
            # Create perturbed version
            noise_scale = 0.02 + 0.01 * r
            a_div = best_seq * (1.0 + noise_scale * rng_final.normal(0, 1, len(best_seq)))
            a_div = np.maximum(a_div, 1e-10)
            a_div = a_div / a_div.sum()

            alloc = min(remaining * 0.3, 30.0)
            a_div_opt, obj_div = lp_alternating_main(a_div, time_budget=alloc, verbose=False)

            if obj_div < best_score:
                best_score = obj_div
                best_seq = a_div_opt.copy()
                print(f"  Diversify {r}: obj={best_score:.10f} **")
            remaining = deadline - time.monotonic()

    # ============ Final validation ============
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.10f}, time={elapsed:.1f}s ===")

    return result
