Multi-method optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Improvements over previous: constraint-generation LP, multi-start with diverse

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_row(a, k):
    """Build row k of the conv matrix: A[k, :] @ g = conv(a, g)[k]."""
    n = len(a)
    row = np.zeros(n)
    # conv(a,g)[k] = sum_i a[i] * g[k-i] where g indices in [0, n-1]
    # So g[j] appears with coefficient a[k-j] if k-j in [0,n-1]
    for j in range(n):
        idx = k - j
        if 0 <= idx < n:
            row[j] = a[idx]
    return row


def build_conv_matrix_subset(a, indices):
    """Build matrix for subset of convolution indices."""
    n = len(a)
    A = np.array([build_conv_matrix_row(a, k) for k in indices])
    return A

    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses constraint generation: start with top-K constraints, add violated ones.
    Returns (g, M) or (None, None).

    n = len(a)

    a = a / np.sum(a)

    # Get initial active set: top K peaks of conv(a,a)
    conv_aa = compute_conv(a)
    K = min(max_constraints, 2 * n - 1)
    active = np.argsort(conv_aa)[::-1][:K]
    active = sorted(set(active.tolist()))

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        A = build_conv_matrix_subset(a, active)

        # Variables: [g_0, ..., g_{n-1}, M]
        c = np.zeros(n + 1)
        c[-1] = 1.0

        A_ub = np.column_stack([A, -np.ones(len(active))])
        b_ub = np.zeros(len(active))

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(0, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        if res.status != 0:
            return None, None

        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s < 1e-10:
            return None, None
        g_val = g_val / s
        M_val = float(res.x[n])

        # Check all constraints

        max_viol = float(np.max(conv_check))

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.column_stack([A, -np.ones(len(active))])
    b_ub = np.zeros(len(active))
    A_eq = np.zeros((1, n + 1))

    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

            A_full = build_conv_matrix_subset(a, range(2 * n - 1))

def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def quadratic_line_search(a, g, n_evals=12):
    """
    Find optimal t in [0,1] for a_new = (1-t)*a + t*g.
    Uses the quadratic structure: conv(a+t*(g-a), a+t*(g-a)) is quadratic in t.
    """
    d = g - a
    conv_a = compute_conv(a)
    conv_ad = compute_conv_cross(a, d)
    conv_dd = compute_conv(d)

    n = len(a)

    def obj_at_t(t):
        val = conv_a + 2 * t * conv_ad + t * t * conv_dd
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = a + t_opt * d
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a
    return t_opt, a_new / s

    # grad_second[i] = sum_j a[j] * w[i-j]
    # 
    # grad_first: this is what you get from np.correlate(w, a, 'full')[n-1+n-1-i] or similar
    # Let me just use direct computation for clarity (n=800 is small enough)
    
    # Actually, let's use the efficient method:
    # Pad a to length 2n-1
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    w = weights  # length 2n-1
    
    # grad[i] = sum_j a[j] * (w[i+j] + w[abs(i-j)])  ... no this isn't right
    # Let me just do the direct sum for now (it's O(n^2) but n=800 is fine)
    # Actually for n=800, O(n^2) = 640000 operations per gradient, which is fine.
    
    grad = np.zeros(n)
    for i in range(n):
        # sum over k of w[k] * (a[k-i] if valid + a[i-k] if valid)
        # a[k-i] valid: 0 <= k-i < n => i <= k < i+n
        # a[i-k] valid: 0 <= i-k < n => i-n < k <= i => max(0, i-n+1) <= k <= i  (since k >= 0)
        # Wait, a[idx] valid means 0 <= idx < n
        # a[k-i]: valid if 0 <= k-i < n, i.e., i <= k < i+n. Combined with 0 <= k < 2n-1: max(i,0) <= k < min(i+n, 2n-1)
        # a[i-k]: valid if 0 <= i-k < n, i.e., i-n < k <= i. Combined: max(0, i-n+1) <= k <= min(i, 2n-2)
        lo1 = max(i, 0)
        hi1 = min(i + n, 2 * n - 1)
        lo2 = max(0, i - n + 1)
        hi2 = min(i + 1, 2 * n - 1)
        
        if hi1 > lo1:
            grad[i] += np.dot(w[lo1:hi1], a[:hi1 - lo1])  # a[k-i] for k in [lo1, hi1) => a[0] to a[hi1-lo1-1]
        if hi2 > lo2:
            # a[i-k] for k in [lo2, hi2) => a[i-lo2] to a[i-(hi2-1)] = a[i-hi2+1] to a[i-lo2]
            # That's a[i-hi2+1 : i-lo2+1]
            idx_lo = i - (hi2 - 1)
            idx_hi = i - lo2 + 1
            if idx_hi > idx_lo:
                grad[i] += np.dot(w[lo2:hi2], a[idx_lo:idx_hi])
    
    return smooth_val, grad


def smooth_max_gradient_descent(a, time_budget=60.0, T_init=0.001, T_min=0.00005, lr_init=0.01):
    """Gradient descent using smooth max with adaptive learning rate."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = lr_init
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    best_lr = lr

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        
        # Smooth max gradient
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 0)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total
        
        # Gradient computation (vectorized)
        # grad[i] = sum_k weights[k] * (a[k-i] + a[i-k]) with valid indices
        # Use the fact that this is a convolution/correlation
        # Trick: create a_padded and use FFT
        a_pad = np.zeros(4 * n)
        a_pad[:n] = a
        w_pad = np.zeros(4 * n)
        w_pad[:2 * n - 1] = weights
        
        size = 1
        while size < 4 * n:
            size <<= 1
        
        fa = rfft(a_pad, size)
        fw = rfft(w_pad, size)
        
        # cross-correlation: corr[i] = sum_j w[j] * a[i+j] (for valid j, i+j)
        # This is conv(w, a) at position... let me think
        # conv(w, a)[k] = sum_j w[j] * a[k-j]
        # We want sum_j w[j] * a[i+j] = conv(w[::-1], a)[i] (cross-correlation)
        # In FFT: ifft(FW * conj(FA)) gives cross-correlation
        corr1 = irfft(fw * np.conj(fa), size)  # sum_j w[j] * a[i-j+something]... 
        # Actually let me be more careful
        # np.correlate(w, a, 'full')[i] = sum_j w[j] * a[j + i - (n_w - 1)]
        # This is getting complicated. Let me use the matrix-free approach:
        # grad[i] = sum_k weights[k] * (a[k-i] if valid + a[i-k] if valid)
        # Term 1: sum_k weights[k] * a[k-i] = (weights conv a) but shifted
        #   = sum_j weights[j+i] * a[j]  (letting k=j+i)
        #   This is the cross-correlation of weights and a at lag i
        # Term 2: sum_k weights[k] * a[i-k] = sum_j weights[i-j] * a[j]  (letting k=i-j)  
        #   = sum_j weights[i-j] * a[j]
        #   This is also a cross-correlation-like thing
        
        # For term 1: t1[i] = sum_j a[j] * weights[i+j]
        #   = ifft(FA * FW) at appropriate positions? 
        #   conv(a, w)[i+j-1]... 
        #   Actually: (a * w)[m] = sum_j a[j] * w[m-j]
        #   t1[i] = sum_j a[j] * w[i+j] = (a * w_rev)[i] where w_rev[k] = w[2n-2-k]
        #   Hmm, let me just do it the simple way for n=800
        
        # Simple O(n^2) computation (fast enough for n=800)
        grad = np.zeros(n)
        for i in range(n):
            # Term: a[k-i] for k in [i, i+n-1] intersect [0, 2n-2]
            k_start = max(i, 0)
            k_end = min(i + n, 2 * n - 1)
            if k_end > k_start:
                # a[k-i] for k from k_start to k_end-1 => a[0] to a[k_end-k_start-1]
                grad[i] += np.dot(weights[k_start:k_end], a[:k_end - k_start])
            # Term: a[i-k] for k in [0, i] intersect [max(0, i-n+1), 2n-2]
            k_start2 = max(0, i - n + 1)
            k_end2 = i + 1
            if k_end2 > k_start2:
                # a[i-k] for k from k_start2 to k_end2-1 => a[i-k_end2+1] to a[i-k_start2]
                # = a[i-k_end2+1 : i-k_start2+1]
                ilo = i - k_end2 + 1
                ihi = i - k_start2 + 1
                if ihi > ilo:
                    grad[i] += np.dot(weights[k_start2:k_end2], a[ilo:ihi])
        
        # Center and apply momentum
        grad_centered = grad - np.mean(grad)
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            best_lr = lr
        else:
            stall += 1
            if stall > 30:
                lr *= 0.6
                if lr < 1e-7:
                    break
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
            elif stall > 10:
                lr *= 0.8

        if it > 20:
            T = max(T_min, T_init * (0.997 ** it))
        if it % 200 == 0 and stall == 0 and lr > best_lr * 0.5:
            lr = min(lr * 1.05, 0.05)

    return best_a


def peak_reduction_step(a, top_k=30, time_budget=2.0):
    """
    Direct peak reduction: for the top K peaks, compute the direction that
    most reduces them simultaneously, using a simplified LP (only top K constraints).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    conv = compute_conv(a)
    top_indices = np.argsort(conv)[::-1][:top_k]
    
    # Solve small LP: min M s.t. conv(a,g)[k] <= M for k in top_indices, sum(g)=1, g>=0
    A = np.array([build_conv_matrix_row(a, k) for k in top_indices])
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.column_stack([A, -np.ones(len(top_indices))])
    b_ub = np.zeros(len(top_indices))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return a

    g = np.maximum(res.x[:n], 0)
    s_g = np.sum(g)
    if s_g < 1e-10:
        return a
    g = g / s_g

    # Line search
    t_opt, a_new = quadratic_line_search(a, g, n_evals=8)
    obj_new = compute_obj(a_new)
    obj_old = compute_obj(a)
    if obj_new < obj_old:
        return a_new
    return a
