```python
def smooth_mirror_descent(a, tau, eta, n_iter):
    n = len(a)
    a = a / sum(a)
    for it in range(n_iter):
        conv = compute_conv(a)
        max_c = np.max(conv)
        exp_c = np.exp(tau * (conv - max_c))
        p = exp_c / np.sum(exp_c)
        
        # Gradient (before the 4n factor, since we'll absorb it into eta)
        N = 1
        while N < 4*n:
            N <<= 1
        P = rfft(p, N)
        A = rfft(a, N)
        cross_corr = irfft(P * np.conj(A), N)[:n]
        grad = 4.0 * n * cross_corr
        
        # Mirror descent update (multiplicative weights)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-30)  # prevent underflow
        a = a / np.sum(a)
    return a
```

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

    import cvxpy as cp

    HAS_CVXPY = False


def compute_conv(a):

    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_grad_smooth(a, tau):
    """Compute gradient of 2n/tau * log(sum exp(tau*conv(a,a))) at a (sum a=1).
    Returns (grad, p, conv) where p are softmax weights."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    exp_c = np.exp(tau * (conv - max_c))
    p = exp_c / float(np.sum(exp_c))

    N = 1
    while N < 4 * n:
        N <<= 1
    P = rfft(p, N)
    A = rfft(a, N)
    cross_corr = irfft(P * np.conj(A), N)[:n]
    grad = 4.0 * n * cross_corr
    return grad, p, conv


def mirror_descent_stage(a, tau, eta_init, max_iter, time_limit):

    n = len(a)
    a = np.maximum(a, 1e-30).astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    eta = eta_init
    start = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - start > time_limit:

        # Center gradient for numerical stability
        grad_c = grad - np.mean(grad)

        accepted = False
        eta_try = eta
        for bt in range(20):
            log_a = np.log(np.maximum(a, 1e-300))
            log_a_new = log_a - eta_try * grad_c
            log_a_new -= np.max(log_a_new)

            s_new = float(np.sum(a_new))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp(a):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    if HAS_CVXPY:
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        for solver in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g_var.value, 0)
                    gs = float(np.sum(g_val))
                    if gs > 1e-10:
                        return g_val / gs, float(M_var.value)
            except Exception:
                continue

    # Fallback to scipy HiGHS
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None

            a_wgeo = np.power(np.maximum(a, 1e-30), alpha) * np.power(np.maximum(g_lp, 1e-30), 1 - alpha)
            s_wg = float(np.sum(a_wgeo))
            if s_wg > 1e-10:
                a_wgeo = a_wgeo / s_wg
                obj_wg = compute_obj(a_wgeo)
                if obj_wg < best_local_obj:
                    best_local_obj = obj_wg
                    best_local_a = a_wgeo.copy()
                    best_update = f'wgeo_{alpha}'

        if best_local_obj < best_obj - 1e-13:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP: obj={best_obj:.8f}, update={best_update}, time={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 5:
                # Structured perturbation
                scale = min(0.05 + 0.03 * (stall // 5), 0.4)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s_p = float(np.sum(a))
                if s_p > 1e-10:
                    a = a / s_p
                    obj_p = compute_obj(a)
                    if obj_p < best_obj:
                        best_obj = obj_p
                        best_a = a.copy()
                stall = 0

    return best_a, best_obj


def resample_seq(a, new_n):
    """Resample sequence to new length via cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng):
    """Generate diverse structured initialization."""
    choice = int(rng.integers(0, 6))
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-50 * (x - 0.5) ** 2)
    elif choice == 1:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 3 * x)
    elif choice == 2:
        a = np.exp(-80 * (x - 0.3) ** 2) + np.exp(-80 * (x - 0.7) ** 2)
    elif choice == 3:
        a = np.ones(n)
        for _ in range(3):
            center = int(rng.integers(0, n))
            width = max(1, int(rng.integers(n // 10, n // 3)))
            for i in range(n):
                d = min(abs(i - center), n - abs(i - center))
                a[i] *= 1.0 + rng.standard_normal() * np.exp(-d / (width * 0.5))
        a = np.maximum(a, 0.01)
    elif choice == 4:
        xx = np.linspace(0.01, 1, n)
        a = xx ** float(rng.uniform(0.5, 3.0))
    else:
        # Asymmetric ramp with decay
        a = np.exp(-x * float(rng.uniform(2, 8))) + np.exp(-(1-x) * float(rng.uniform(2, 8)))
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def subgradient_polish(a, time_budget, top_k=20):
    """Nesterov-accelerated subgradient method for polishing."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a.copy(), compute_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.9
    lr = 2.0 / n
    x = a.copy()

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(x)
        k_indices = np.argsort(conv)[::-1][:top_k]

        # Build weight vector
        w = np.zeros(2 * n - 1)
        for k in k_indices:
            w[int(k)] = 1.0 / len(k_indices)

        # Compute gradient via FFT (same formula as smooth grad with p=w)
        N = 1
        while N < 4 * n:
            N <<= 1
        P = rfft(w, N)
        A = rfft(x, N)
        cross_corr = irfft(P * np.conj(A), N)[:n]
        grad = 4.0 * n * cross_corr
        grad -= np.mean(grad)

        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        t_k = 2.0 / (it + 2.0)
        x = a_new + (1.0 - t_k) * (a_new - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        a = a_new
        if it % 50 == 0:
            obj = compute_obj(a)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
        if it % 300 == 299:
            lr *= 0.97

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: smooth mirror descent with temperature annealing + LP alternating.
    The mirror descent provides a different optimization path than pure LP,
    potentially escaping local minima that trap the LP alternating method.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        n_base = len(a)
        print(f"Parent loaded: n={n_base}, obj={compute_obj(a):.8f}")
    except Exception:
        a = np.ones(800) / 800
        n_base = 800
        print(f"No parent, using uniform: n=800, obj={compute_obj(a):.8f}")

    best_a = a.copy()
    best_obj = compute_obj(a)

    # === Phase 1: Smooth Mirror Descent (main innovation) ===
    print(f"\n=== Phase 1: Smooth Mirror Descent (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.35, 400.0)
    a_sm, obj_sm = smooth_optimize(best_a, time_budget=p1_budget, verbose=True)
    if obj_sm < best_obj:
        best_obj = obj_sm
        best_a = a_sm
        print(f"  Phase 1 result: obj={best_obj:.8f} **")
    else:
        print(f"  Phase 1: no improvement (obj={obj_sm:.8f})")

    # === Phase 2: LP Alternating (polish the smooth result) ===
    print(f"\n=== Phase 2: LP Alternating ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.25, 250.0)
    a_lp, obj_lp = lp_alternate(best_a, time_budget=p2_budget, verbose=True)
    if obj_lp < best_obj:
        best_obj = obj_lp
        best_a = a_lp
        print(f"  Phase 2 result: obj={best_obj:.8f} **")
    else:
        print(f"  Phase 2: no improvement (obj={obj_lp:.8f})")

    # === Phase 3: Subgradient polish ===
    print(f"\n=== Phase 3: Subgradient Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.12, 100.0)
        for tk in [10, 30]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(p3_budget / 2.0, remaining * 0.4)
            a_sg, obj_sg = subgradient_polish(best_a, time_budget=alloc, top_k=tk)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg
                print(f"  SubGrad top_k={tk}: obj={best_obj:.8f} **")

    # === Phase 4: LP again from improved point ===
    print(f"\n=== Phase 4: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = min(remaining * 0.20, 200.0)
        a_lp2, obj_lp2 = lp_alternate(best_a, time_budget=p4_budget, verbose=True)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2
            print(f"  Phase 4 result: obj={best_obj:.8f} **")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        max(200, n_best - 200), max(200, n_best - 100),
        n_best,
        n_best + 100, n_best + 200, n_best + 400,
    ]))
    n_options = [x for x in n_options if 200 <= x <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            s_t = float(np.sum(a_try))
            if s_t > 1e-10:
                a_try = a_try / s_t

        # Quick smooth + LP
        alloc_smooth = min(20.0, remaining * 0.2)
        alloc_lp = min(40.0, remaining * 0.3)

        a_opt, obj_opt = smooth_optimize(a_try, time_budget=alloc_smooth)
        a_opt, obj_opt = lp_alternate(a_opt, time_budget=alloc_lp)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    for d in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_div = diverse_initialization(n_best, rng)
        alloc = min(30.0, remaining * 0.2)
        a_opt, obj_div = smooth_optimize(a_div, time_budget=alloc * 0.5)
        a_opt, obj_div = lp_alternate(a_opt, time_budget=alloc * 0.5)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse {d}: obj={obj_div:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        # LP polish
        alloc = min(remaining * 0.4, 60.0)
        a_f, obj_f = lp_alternate(best_a, time_budget=alloc, verbose=True)
        if obj_f < best_obj:
            best_obj = obj_f
            best_a = a_f
            print(f"  Final LP: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Subgradient micro-polish
        alloc = min(remaining * 0.3, 30.0)
        a_sg, obj_sg = subgradient_polish(best_a, time_budget=alloc, top_k=15)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  Final SubGrad: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        # One more smooth pass at high temperature
        alloc = min(remaining * 0.4, 25.0)
        a_sm2, obj_sm2 = smooth_optimize(best_a, time_budget=alloc)
        if obj_sm2 < best_obj:
            best_obj = obj_sm2
            best_a = a_sm2
            print(f"  Final Smooth: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.5, 20.0)
        a_last, obj_last = lp_alternate(best_a, time_budget=alloc)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
