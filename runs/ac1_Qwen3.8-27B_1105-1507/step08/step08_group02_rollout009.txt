"""
Optimized step function search using subgradient descent with peak tracking,
LP alternating, FFT-based smooth gradients, adaptive dimension exploration,
and simulated annealing restarts.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_peak(a, peaks):
    """
    Compute subgradient of max_k conv(a,a)[k] at given peak indices.
    g_j = 2 * sum_{k in peaks} a_{k-j} (where 0 <= k-j < n)
    O(n * len(peaks))
    """
    n = len(a)
    g = np.zeros(n)
    for k in peaks:
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        idx_start = k - j_end
        idx_end = k - j_start
        g[j_start:j_end + 1] += 2.0 * a[idx_start:idx_end + 1][::-1]
    return g


def smooth_logsumexp_grad(a, beta):
    """
    Smooth gradient using log-sum-exp weights over all convolution positions.
    g_j = 2 * sum_k w_k * a_{k-j} where w = softmax(beta * conv)
    O(n * (2n-1)) direct computation.
    """
    n = len(a)
    conv = compute_conv(a)
    m = conv.max()
    if m < 1e-30:
        return None
    shifted = beta * (conv - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    weights = np.exp(shifted)
    total = weights.sum()
    if total < 1e-30:
        return None
    weights /= total

    g = np.zeros(n)
    # Only process significant weights for speed
    threshold = 1.0 / (n * 10)
    significant = np.where(weights > threshold)[0]
    for k in significant:
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        idx_start = k - j_end
        idx_end = k - j_start
        g[j_start:j_end + 1] += 2.0 * weights[k] * a[idx_start:idx_end + 1][::-1]
    return g


def lp_solve(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized_to_sum1, M_opt) or (None, None).
    """
    n = len(a)
    a_s = float(np.sum(a))
    if a_s < 1e-10:
        return None, None
    a_norm = a / a_s

    # Build A matrix: A[k,j] = a_norm[k-j] if 0<=k-j<n else 0
    m_rows = 2 * n - 1
    A = np.zeros((m_rows, n))
    for j in range(n):
        k_max = min(m_rows - 1, j + n - 1)
        A[j:k_max + 1, j] = a_norm[:k_max + 1 - j]

    # Formulate: min M
    # s.t. A @ g - M <= 0  (m_rows constraints)
    #      sum(g) = 1
    #      g >= 0
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    A_ub = np.zeros((m_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m_rows)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def subgradient_optimize(a0, time_budget=300.0, verbose=False):
    """
    Subgradient method with adaptive step size and top-K peaks.
    Operates on the simplex (sum=1, a>=0).
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 1.0 / np.sqrt(n)
    patience = 0
    max_patience = 500
    topK = 3
    beta_smooth = 50.0
    use_smooth = False

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()

        # Get top-K peaks
        if topK >= len(conv):
            peaks = np.argsort(conv)[::-1]
        else:
            peaks = np.argsort(conv)[::-1][:topK]

        # Compute subgradient
        g = subgradient_peak(a, peaks)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Adaptive learning rate
        if patience > 200:
            use_smooth = True
            smooth_g = smooth_logsumexp_grad(a, beta_smooth)
            if smooth_g is not None and np.linalg.norm(smooth_g) > 1e-15:
                g = 0.5 * g + 0.5 * smooth_g
                g_norm = np.linalg.norm(g)
                if g_norm < 1e-15:
                    break

        # Step
        step = lr / max(g_norm, 1e-10) * g
        a_new = a - step
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            patience = 0
            lr = min(lr * 1.01, 2.0 / np.sqrt(n))
        else:
            patience += 1
            if patience > max_patience:
                # Restart with perturbation
                rng = np.random.default_rng(it * 31 + n)
                noise = 0.05 * rng.normal(0, 1, n)
                a = best_a * (1.0 + noise)
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                lr = 2.0 / np.sqrt(n)
                patience = 0
                topK = min(n, topK + 2)
                beta_smooth *= 1.5
            else:
                lr *= 0.995

        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"      Subgrad it{it}: obj={best_obj:.8f}, lr={lr:.5f}, "
                  f"peak={max_val:.6f}, t={elapsed:.1f}s")

    return best_a

        else:

        # Line search

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Cosine taper (Hann window)
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1 + 1e-10)))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 3:
        # Triangular
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Flat top
        a = np.ones(n)
        taper = max(1, n // 10)
        for i in range(taper):
            a[i] *= (i + 1) / taper
            a[n - 1 - i] *= (i + 1) / taper
        a = a / a.sum()
    elif mode == 5:
        # Gaussian bump
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
        a = a / a.sum()
    elif mode == 6:
        # Two bumps
        centers = [n // 4, 3 * n // 4]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * (n // 8) ** 2))
        a = a / a.sum()
    elif mode == 7:
        # Sine-modulated
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x)
        a = np.maximum(a, 0.05)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def coordinate_polish(a0, time_budget=30.0):
    """
    Coordinate descent polish: try adjusting pairs of coordinates
    to reduce the peak convolution value.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    for rnd in range(10):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        # Find top contributors to the peak
        contribs = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs.append((a[i] * a[j], i, j))
        contribs.sort(reverse=True)

        # Find positions NOT contributing to peak
        contributing_indices = set()
        for _, i, j in contribs:
            contributing_indices.add(i)
            if i != j:
                contributing_indices.add(j)
        non_contrib = [x for x in range(n) if x not in contributing_indices]

        if not non_contrib:
            break

        for contrib, i, j in contribs[:20]:
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break

            for target in non_contrib[:5]:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    src = i if a[i] >= (a[j] if i != j else a[i]) else (j if i != j else i)
                    delta = frac * a[src]
                    if delta >= a[src]:
                        continue
                    a[src] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        a[src] += delta
                        a[target] -= delta
                        break

        if not improved:
            break

    return best_a

            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Dimension exploration ===
    print(f"\n=== Phase 3: Dimension Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    # Try a range of dimensions
    n_options = []
    for delta in [-100, -50, -25, 25, 50, 75, 100, 150, 200]:
        n_try = n_best + delta
        if 300 <= n_try <= 2000:
            n_options.append(n_try)
    # Add some absolute values
    for n_abs in [500, 600, 700, 850, 900, 950, 1000, 1100, 1200]:
        if 300 <= n_abs <= 2000 and n_abs != n_best:
            n_options.append(n_abs)

    n_options = sorted(set(n_options))

    for n_try in n_options[:8]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)
        a_try = a_try / a_try.sum()

        # Quick subgradient polish
        if remaining > 80:
            a_try = subgradient_optimize(a_try, time_budget=min(remaining * 0.1, 40.0))
            remaining = deadline - time.monotonic()

        # Quick LP
        if remaining > 50:
            a_try = lp_alternating(a_try, time_budget=min(remaining * 0.1, 30.0))

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = len(best_seq)
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 4: Subgradient + LP combined on best ===
    print(f"\n=== Phase 4: Combined Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        a_s = subgradient_optimize(best_seq, time_budget=min(remaining * 0.25, 100.0), verbose=True)
        obj_s = compute_obj(a_s)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_s.copy()
            print(f"  Subgrad polish: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        a_l = lp_alternating(best_seq, time_budget=min(remaining * 0.3, 120.0), verbose=True)
        obj_l = compute_obj(a_l)
        if obj_l < best_score:
            best_score = obj_l
            best_seq = a_l.copy()
            print(f"  LP polish: obj={best_score:.8f} **")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_use = len(best_seq)

    for r in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = (r * 2 + 1) % 8
        a_r = diverse_init(n_use, rng, mode=mode)
        blend = 0.4 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        if remaining > 60:
            a_r = subgradient_optimize(a_r, time_budget=min(remaining * 0.15, 40.0))
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_r = lp_alternating(a_r, time_budget=min(remaining * 0.1, 30.0))

        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 6: Coordinate polish ===
    print(f"\n=== Phase 6: Coordinate Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_cd = coordinate_polish(best_seq, time_budget=min(remaining * 0.2, 40.0))
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord polish: obj={best_score:.8f} **")
        else:
            print(f"  Coord polish: obj={obj_cd:.8f}")

    # === Phase 7: Final LP ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = lp_alternating(best_seq, time_budget=min(remaining * 0.4, 50.0))
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
