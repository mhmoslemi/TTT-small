import numpy as np
import time
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)

def smooth_objective(a, T):
    """Smooth objective using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    return 2.0 * n * (max_c + T * np.log(np.sum(np.exp((conv - max_c) / T))))

def smooth_gradient(a, T):
    """Gradient of smooth objective w.r.t. a (on simplex)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = np.clip((conv - max_c) / T, -500, 500)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    # grad_i = 2n * sum_k w_k * a_{k-i} * 1(0 <= k-i < n)
    # = 2n * (correlate(weights, a, 'valid'))[i]
    grad = 2.0 * n * np.correlate(weights, a, mode='valid')
    return grad

def smooth_grad_ softmax(a, T):
    """Gradient of smooth objective w.r.t. z where a = softmax(z)."""
    n = len(a)
    grad_a = smooth_gradient(a, T)
    # df/dz_j = a_j * (grad_a_j - sum_i a_i * grad_a_i)
    inner = np.dot(a, grad_a)
    return a * (grad_a - inner)

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = a[:hi - lo]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None

                        a = a_try

def resample(a, new_n):

    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Hybrid LP-Adam optimizer for minimum peak autocorrelation step functions.
Combines:
- Smooth max (log-sum-exp) gradient descent with Adam optimizer

import numpy as np
import time
from scipy.fft import rfft, irfft


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_project(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1.0)
    return np.maximum(v - theta, 0)


def softmax_fn(z):
    """Numerically stable softmax."""
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    return e / np.sum(e)


def smooth_gradient_a(a, T):
    """
    Gradient of 2n * T * log(sum(exp(conv(a,a)/T))) w.r.t. a.
    Uses the identity: d(conv_k)/d(a_i) = 2*a_{k-i} * 1(0<=k-i<n, 0<=i<n)
    which gives grad = 2n * correlate(weights, a, 'valid').
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = np.clip((conv - max_c) / T, -500, 500)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    return 2.0 * n * np.correlate(weights, a, mode='valid')


def smooth_gradient_z(a, T):
    """
    Gradient of smooth objective w.r.t. z where a = softmax(z).
    Uses chain rule: df/dz_j = a_j * (df/da_j - sum_i a_i * df/da_i)
    """
    grad_a = smooth_gradient_a(a, T)
    inner = np.dot(a, grad_a)
    return a * (grad_a - inner)


def adam_softmax(z_init, time_budget, T_init=0.005, T_min=0.00005,
                 lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                 T_decay=0.999, verbose=False):
    """Adam optimization in softmax parameterization (unconstrained z-space)."""
    start = time.monotonic()
    z = z_init.copy().astype(np.float64)
    n = len(z)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = softmax_fn(z).copy()
    best_obj = compute_obj(best_a)
    T = T_init
    max_iters = 50000

    while time.monotonic() - start < time_budget and t < max_iters:
        t += 1
        a = softmax_fn(z)

        grad = smooth_gradient_z(a, T)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if t > 20:
            T = max(T_min, T * T_decay)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      Adam-softmax it{t}: obj={best_obj:.8f}, T={T:.6f}, time={elapsed:.1f}s")

    # Final check with exact softmax
    a_final = softmax_fn(z)
    obj_final = compute_obj(a_final)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a_final.copy()

    return best_a


def adam_projected(a_init, time_budget, T_init=0.003, T_min=0.00003,
                   lr=0.02, beta1=0.9, beta2=0.999, eps=1e-8,
                   T_decay=0.999, verbose=False):
    """Adam optimization with simplex projection."""
    start = time.monotonic()
    a = np.maximum(a_init, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = compute_obj(a)
    T = T_init
    max_iters = 50000

    while time.monotonic() - start < time_budget and t < max_iters:
        t += 1
        grad = smooth_gradient_a(a, T)
        grad_centered = grad - np.mean(grad)

        m = beta1 * m + (1 - beta1) * grad_centered
        v = beta2 * v + (1 - beta2) * (grad_centered * grad_centered)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = simplex_project(a)

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if t > 20:
            T = max(T_min, T * T_decay)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      Adam-proj it{t}: obj={best_obj:.8f}, T={T:.6f}, time={elapsed:.1f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns g or None."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None


def solve_lp_reduced(a, active_indices):
    """Solve LP with only active (near-binding) constraints for speed."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A_full = build_conv_matrix(a)
    A_sub = A_full[active_indices, :]
    m_constraints = len(active_indices)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((m_constraints, n + 1))
    A_ub[:, :n] = A_sub
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_val = g_val / s
            # Verify against all constraints
            conv_check = A_full @ g_val
            if np.max(conv_check) <= float(res.x[n]) + 1e-6:
                return g_val
            # Slightly infeasible, still use it
            return g_val
    return None


def lp_alternating(a, time_budget=100.0, use_reduced=True, verbose=False):
    """LP alternating with line search and active set reduction."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    max_iters = 500

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Determine active set
        if use_reduced:
            conv_cur = compute_conv(a)
            max_conv = np.max(conv_cur)
            threshold = max_conv * 0.95
            active_set = np.where(conv_cur >= threshold)[0]
            if len(active_set) < max(10, n // 4):
                g = solve_lp_reduced(a, active_set)
            else:
                g = solve_lp(a)
        else:
            g = solve_lp(a)

        if g is None:
            break

        # Convex line search for blend parameter
        conv_aa = compute_conv(a)
        conv_ag = compute_conv_cross(a, g)
        conv_gg = compute_conv(g)

        def obj_at_t(t):
            val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
            return 2.0 * n * float(np.max(val))

        # Ternary search
        lo, hi = 0.0, 1.0
        for _ in range(12):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if obj_at_t(m1) < obj_at_t(m2):
                hi = m2
            else:
                lo = m1
        t_opt = (lo + hi) / 2.0

        a_try = (1 - t_opt) * a + t_opt * g
        a_try = np.maximum(a_try, 0)
        s_try = float(np.sum(a_try))
        if s_try < 1e-10:
            break
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Try a small fixed step
                a_try = (1 - 0.05) * a + 0.05 * g
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try > 1e-10:
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stall = 5
                    else:
                        break

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, time={elapsed:.1f}s")

    return best_a


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def paired_coordinate_descent(a, time_budget=80.0, verbose=False):
    """Paired coordinate descent: update pairs (i,j) simultaneously."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    delta = 2e-3
    no_improve = 0
    max_no_improve = 8

    while time.monotonic() - start < time_budget:
        improved = False

        # Single coordinate updates
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            for d in [delta, delta * 0.3, delta * 3]:
                for sign in [1, -1]:
                    a_try = a.copy()
                    a_try[i] += sign * d
                    if a_try[i] < 0:
                        continue
                    s_try = float(np.sum(a_try))
                    if s_try < 0.01:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved and rng.random() < 0.3:
                continue  # Skip some after improvement for speed

        # Paired updates
        for _ in range(n // 4):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for d in [delta * 0.5, delta]:
                a_try = a.copy()
                a_try[i] += d
                a_try[j] -= d
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                s_try = float(np.sum(a_try))
                if s_try < 0.01:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                delta *= 0.5
                no_improve = 0
                if delta < 1e-7:
                    break
        else:
            no_improve = 0
