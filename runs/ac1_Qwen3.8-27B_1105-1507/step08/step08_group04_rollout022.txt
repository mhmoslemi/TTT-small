```

    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)  # (2n-1) x n
    
    # Primal: min M s.t. A @ g <= M, sum(g) = 1, g >= 0
    # Variables: [g (n), M (1)]
    # ...
    
    # For getting the dual, we can use cvxpy which gives us the dual variables
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve()
    
    g_val = np.maximum(g.value, 0)
    g_val /= g_val.sum()
    
    # Get dual variables for the inequality constraints A @ g <= M
    y = np.zeros(2*n-1)
    for i, con in enumerate(constraints):
        if hasattr(con, 'dual_value') and con.dual_value is not None:
            y[i] = abs(con.dual_value)  # dual values can be negative
    
    # Normalize y to sum to 1
    y_sum = y.sum()

        active = np.where(conv >= conv.max() - 1e-6)[0]

    return g_val, float(M.value), y

def dual_weighted_subgradient(a, y):
    """Compute subgradient weighted by dual variables y."""
    n = len(a)
    # grad[i] = 4n * sum_k y[k] * a[k-i] * 1[0 <= k-i < n]
    # This is a cross-correlation of a with y
    # Use FFT for efficiency
    n_fft = 1
    while n_fft < 4*n:
        n_fft <<= 1
    A_fft = rfft(a, n_fft)
    Y_fft = rfft(y, n_fft)
    R = irfft(np.conj(A_fft) * Y_fft, n_fft)
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = R[(n_fft - i) % n_fft]
    grad *= 4.0 * n
    grad -= grad.mean()  # Center for simplex
    return grad
```

Advanced optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternation with dual-weighted subgradient, line search,
mirror descent, and sophisticated perturbation strategies.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
from scipy.optimize import linprog
import time

def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)

        for l in range(n):
            j = k - l
            if 0 <= j < n:

    return A

def dual_weighted_subgradient(a, y):
    """Compute 4n * cross-correlation of a with y, centered."""
    n = len(a)
    # grad[i] = 4n * sum_k y[k] * a[k-i] * 1[0 <= k-i < n]
    # This is the cross-correlation R[i] = sum_k y[k] * a[k+i] (with appropriate indexing)
    # Using FFT:
    n_fft = 1
    while n_fft < 2 * (n + len(y)) :
        n_fft <<= 1
    
    A_fft = rfft(a, n_fft)
    Y_fft = rfft(y, n_fft)
    # We want R[i] = sum_k y[k] * a[k-i] for 0 <= k-i < n
    # = sum_j y[j+i] * a[j] (substituting j = k-i)
    # This is the cross-correlation of a with y, at lag i
    # crosscorr[i] = sum_j a[j] * y[j+i]
    # = sum_j a[j] * y[j+i]
    # Using FFT: if we compute IFFT(conj(FFT(a)) * FFT(y)), we get the circular cross-correlation
    R = irfft(np.conj(A_fft) * Y_fft, n_fft)
    # R[m] = sum_j a[j] * y[j+m] (circular)
    # We want R[i] for i = 0, ..., n-1
    grad = R[:n].copy()
    grad *= 4.0 * n
    grad -= np.mean(grad)
    return grad

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def line_search(a, direction, max_iter=50):
    """Golden section search for best t in [0, 1] to minimize f((1-t)*a + t*direction)."""
    n = len(a)
    a = a / np.sum(a)
    direction = direction / np.sum(direction) if np.sum(direction) > 0 else np.ones(n)/n
    
    def obj_at_t(t):
        a_t = (1 - t) * a + t * direction
        a_t = np.maximum(a_t, 0)
        s = a_t.sum()
        if s < 1e-10:
            return np.inf
        a_t = a_t / s
        return compute_obj(a_t)
    
    # Golden section search in [0, 1]
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    a_lo, a_hi = 0.0, 1.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = obj_at_t(c)
    fd = obj_at_t(d)
    
    for _ in range(max_iter):
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = obj_at_t(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = obj_at_t(d)
        if a_hi - a_lo < 1e-6:
            break
    
    t_best = (a_lo + a_hi) / 2
    return t_best, obj_at_t(t_best)

def mirror_descent_step(a, y, lr):
    """Exponential weights update using dual weights."""
    n = len(a)
    grad = dual_weighted_subgradient(a, y)
    # Mirror descent: a_new[i] proportional to a[i] * exp(-lr * grad[i])
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - lr * grad
    # Normalize
    log_a_new -= np.max(log_a_new)  # for numerical stability
    a_new = np.exp(log_a_new)
    a_new /= a_new.sum()
    return a_new

def diverse_initialization(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)

        freq = rng.uniform(1, 8)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        c1, c2 = 0.25 + 0.1*rng.standard_normal(), 0.75 + 0.1*rng.standard_normal()
        a = np.exp(-80 * (x - c1)**2) + np.exp(-80 * (x - c2)**2)
    elif choice == 3:
        a = np.ones(n)

            center = rng.integers(0, n)
            width = rng.integers(max(2, n//20), n//4)
            for i in range(n):
                d = abs(i - center)
                a[i] *= 1.0 + rng.standard_normal() * 0.5 * np.exp(-d**2 / (2*width**2))
        a = np.maximum(a, 0.01)
    elif choice == 4:
        a = x ** rng.uniform(0.3, 3.0)
    elif choice == 5:
        # Triangular
        a = 1 - np.abs(2*x - 1)
    elif choice == 6:
        # Random smooth (spline-like)
        k = rng.integers(3, 10)
        knots = np.sort(rng.uniform(0, 1, k))
        a = np.interp(x, knots, np.abs(rng.standard_normal(k)))
        a = a / (a.max() + 1e-10)
    else:

        blocks = rng.integers(2, 8)
        a = np.zeros(n)

    s = a.sum()
    if s < 1e-10:

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Advanced optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternation with dual-weighted subgradient, golden-section line search,
mirror descent, adaptive perturbation, and multi-n exploration.
Key innovation: uses LP dual variables to weight the subgradient, giving a much
better directional signal than uniform weighting over the active set.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    if n == 1:
        return np.array([a[0] ** 2])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_fast(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for l in range(n):
        k_start = l
        k_end = min(2 * n - 1, l + n)
        A[k_start:k_end, l] = a[:k_end - k_start]
    return A


def solve_lp_with_dual(a, timeout=30.0):
    """Solve LP and return (g, M, dual_y)."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix_fast(a)

    g_val = None
    M_val = None
    y = None

    # Try cvxpy first (gives us dual variables easily)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            ineq_con = A @ g_var <= M_var
            prob = cp.Problem(cp.Minimize(M_var), [ineq_con, cp.sum(g_var) == 1])

            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > timeout:
                continue

            if g_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g_var.value, 0).astype(np.float64)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M_var.value)
                    # Extract dual variables for inequality constraints
                    y = np.zeros(2 * n - 1)
                    try:
                        dv = ineq_con.dual_value
                        if dv is not None:
                            y = np.abs(np.asarray(dv, dtype=np.float64))
                    except Exception:
                        pass
                    break
        except Exception:
            continue

    if g_val is None:
        # Fallback to scipy linprog
        try:
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
            b_ub = np.zeros(2 * n - 1)
            A_eq = np.hstack([np.ones((1, n)), [[1.0]]])
            b_eq = np.array([1.0])
            bounds = [(0, None)] * n + [(0, None)]
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                g_val = np.maximum(res.x[:n], 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(res.x[n])
                    try:
                        if res.ineqlin is not None and res.ineqlin.marginals is not None:
                            y = np.abs(np.asarray(res.ineqlin.marginals, dtype=np.float64))
                    except Exception:
                        pass
        except Exception:
            pass

    if g_val is None:
        return None, None, None

    # Normalize y or compute fallback
    if y is not None and y.sum() > 1e-10:
        y = np.maximum(y, 0)
        y = y / y.sum()
    else:
        # Fallback: use active set from primal
        conv_ag = A @ g_val
        threshold = conv_ag.max() - 0.02 * conv_ag.max()
        y = (conv_ag >= threshold).astype(np.float64)
        if y.sum() > 1e-10:
            y = y / y.sum()
        else:
            y = np.ones(2 * n - 1) / (2 * n - 1)

    return g_val, M_val, y


def dual_weighted_subgradient(a, y):
    """Compute 4n * cross-correlation(a, y), centered for simplex projection."""
    n = len(a)
    # Need n_fft >= n + len(y) - 1 = n + 2n-2 = 3n-2
    n_fft = 1
    while n_fft < 3 * n:
        n_fft <<= 1

    a_padded = np.zeros(n_fft)
    a_padded[:n] = a
    y_padded = np.zeros(n_fft)
    y_padded[:len(y)] = y

    A_fft = rfft(a_padded)
    Y_fft = rfft(y_padded)
    # R[i] = sum_j a[j] * y[j+i] (circular cross-correlation)
    R = irfft(np.conj(A_fft) * Y_fft)
    grad = R[:n].copy()
    grad *= 4.0 * n
    grad -= np.mean(grad)
    return grad


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def line_search_blend(a, direction, max_iter=40):
    """Golden section search for best t in [0,1]: minimize f((1-t)*a + t*direction)."""
    n = len(a)
    a = a / float(np.sum(a))
    dir_sum = float(np.sum(direction))
    if dir_sum < 1e-10:
        return 0.0, compute_obj(a)
    direction = direction / dir_sum

    def obj_at_t(t):
        a_t = (1.0 - t) * a + t * direction
        a_t = np.maximum(a_t, 0)
        s = float(np.sum(a_t))
        if s < 1e-10:
            return np.inf
        a_t = a_t / s
        return compute_obj(a_t)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at_t(c)
    fd = obj_at_t(d)

    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at_t(d)
        if hi - lo < 1e-7:
            break

    t_best = (lo + hi) / 2.0
    return t_best, obj_at_t(t_best)


def lp_alternate_advanced(a, time_budget=200.0, verbose=False):
    """LP alternation with dual-weighted subgradient, line search, and adaptive perturbation."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    iter_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        if time_budget - elapsed < 5.0:
            break

        g_lp, M_lp, y_dual = solve_lp_with_dual(a)
        if g_lp is None:
            break

        iter_count += 1
        candidates = []

        # Candidate 1: Line search on LP direction
        t_best, obj_line = line_search_blend(a, g_lp, max_iter=30)
        a_line = (1.0 - t_best) * a + t_best * g_lp
        a_line = np.maximum(a_line, 0)
        sl = float(np.sum(a_line))
        if sl > 1e-10:
            a_line = a_line / sl
            candidates.append((a_line, obj_line, 'LP-blend'))

        # Candidate 2: Dual-weighted subgradient step
        if y_dual is not None:
            grad = dual_weighted_subgradient(a, y_dual)
            for lr in [0.02, 0.01, 0.005, 0.002]:
                a_sg = project_simplex(a - lr * grad)
                obj_sg = compute_obj(a_sg)
                candidates.append((a_sg, obj_sg, f'SG-lr{lr}'))

        # Candidate 3: Mirror descent step
        if y_dual is not None:
            grad = dual_weighted_subgradient(a, y_dual)
            for lr_md in [0.005, 0.002, 0.001]:
                log_a = np.log(np.maximum(a, 1e-300))
                log_a_new = log_a - lr_md * grad
                log_a_new -= np.max(log_a_new)
                a_md = np.exp(log_a_new)
                a_md = a_md / a_md.sum()
                obj_md = compute_obj(a_md)
                candidates.append((a_md, obj_md, f'MD-lr{lr_md}'))

        # Pick the best candidate
        candidates.sort(key=lambda x: x[1])
        best_cand_a, best_cand_obj, best_cand_name = candidates[0]

        if best_cand_obj < best_obj - 1e-13:
            improvement = best_obj - best_cand_obj
            best_obj = best_cand_obj
            best_a = best_cand_a.copy()
            a = best_cand_a.copy()
            stall = 0
            if verbose and (iter_count % 10 == 1 or improvement > 1e-5):
                print(f"    it={iter_count}: obj={best_obj:.10f} ({best_cand_name}), "
                      f"imp={improvement:.2e}, t={elapsed:.0f}s")
        else:
            stall += 1
            if stall >= 4:
                # Adaptive perturbation to escape local optimum
                rng = np.random.default_rng((n * 104729 + iter_count * 31337 +
                                            int(elapsed * 7919)) % (2 ** 31))
                # Mix of multiplicative and additive perturbation
                scale = min(0.02 + 0.008 * (stall // 4), 0.12)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert += rng.normal(0, scale * 0.3, n)
                a_pert = project_simplex(a_pert)
                obj_pert = compute_obj(a_pert)
                # Also try frequency-domain perturbation
                a_fft = rfft(best_a, 2 * n)
                a_fft[1:-1] *= (1.0 + rng.normal(0, scale * 0.5, len(a_fft[1:-1])))
                a_freq = irfft(a_fft, 2 * n)[:n]
                a_freq = project_simplex(a_freq)
                obj_freq = compute_obj(a_freq)

                if obj_pert <= obj_freq:
                    a = a_pert
                else:
                    a = a_freq
                stall = 0

    return best_a


def diverse_initialization(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-50 * (x - 0.5) ** 2)
    elif choice == 1:
        freq = rng.uniform(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x + phase)
    elif choice == 2:
        c1 = 0.25 + 0.15 * rng.standard_normal()
        c2 = 0.75 + 0.15 * rng.standard_normal()
        a = np.exp(-80 * (x - c1) ** 2) + np.exp(-80 * (x - c2) ** 2)
    elif choice == 3:
        a = np.ones(n)
        for _ in range(4):
            center = rng.integers(0, n)
            width = max(2, rng.integers(n // 25, n // 4))
            sig = rng.standard_normal() * 0.5
            for i in range(n):
                d = abs(i - center)
                a[i] *= 1.0 + sig * np.exp(-d ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0.01)
    elif choice == 4:
        a = x ** rng.uniform(0.3, 3.0)
    elif choice == 5:
        a = 1.0 - np.abs(2.0 * x - 1.0)
    elif choice == 6:
        k = rng.integers(3, 12)
        knots = np.sort(rng.uniform(0, 1, k))
        vals = np.abs(rng.standard_normal(k)) + 0.1
        a = np.interp(x, knots, vals)
    else:
        blocks = rng.integers(3, 10)
        a = np.zeros(n)
        for b in range(blocks):
            s_idx = int(b * n / blocks)
            e_idx = int((b + 1) * n / blocks)
            a[s_idx:e_idx] = rng.uniform(0.5, 2.0)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
