But we're dealing with linear convolution, not cyclic. The relationship is:
linear_conv(a,a)[k] = cyclic_conv(a_ext, a_ext)[k] for k = 0, ..., 2n-2

where a_ext is a zero-padded to length >= 2n-1.

So the "flat spectrum" approach might give a good initialization.

**Flat spectrum initialization**: Find a (length n, non-negative, sum=1) such that |FFT(a_padded)| is as constant as possible. This is equivalent to maximizing the entropy of |A(ω)|^2, or equivalently, minimizing the variance of |A(ω)|^2.

This can be done with iterative methods (similar to the Gerchberg-Saxton algorithm in optics).

Let me implement this as an initialization:

```python
def flat_spectrum_init(n, iters=1000):
    """Find non-negative sequence with flat power spectrum."""
    rng = np.random.default_rng(42)
    a = rng.random(n) + 0.1
    a = a / np.sum(a)
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for it in range(iters):
        A = rfft(a, size)
        power = A.real**2 + A.imag**2
        # Target: flat power spectrum
        target = np.full(size//2 + 1, 1.0 / (size//2 + 1))
        # Adjust: make |A| more flat
        new_mag = np.sqrt(target * power / np.mean(power))
        # Keep phase
        new_phase = np.angle(A)
        A_new = new_mag * np.exp(1j * new_phase)
        a_new = irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a = a_new / s
        else:
            a = a_new + 0.01
            a = np.maximum(a, 0)
            a = a / np.sum(a)
    
    return a
```

1. The previous solution used LP alternating with momentum. The main limitation is likely getting stuck in local minima.
2. The softmax-gradient approach provides a smoother landscape and might escape some local minima.
3. More diverse restarts might find better local minima.

2. **LP dual-guided updates** (use dual variables to weight the subgradient)

"""
Optimized step function search using softmax-gradient, LP-dual-guided updates,
and diverse initialization strategies.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def softmax_gradient_step(a, beta, lr, momentum_buf, beta_mom):
    """
    One step of softmax-gradient descent on max(a*a).
    Returns (new_a, new_momentum, obj).
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Softmax weights
    conv_shifted = conv - np.max(conv)
    p = np.exp(beta * conv_shifted)
    p /= np.sum(p)
    
    # Gradient: grad[j] = sum_k p[k] * a[k-j] (for valid k-j)
    # This is a Toeplitz matrix times a
    # M[j,k] contribution: if 0 <= k-j < n, add p[k] * a[k-j] to grad[j]
    # Equivalently: grad[j] = sum_{m=0}^{n-1} a[m] * p[j+m]
    
    # Efficient computation using the fact that this is a correlation
    # grad = correlation of a and p
    # Use direct computation (fast enough for n <= 3000)
    # grad[j] = sum_{m=0}^{n-1} a[m] * p[j+m]
    
    # Vectorized: create index matrix
    j_idx = np.arange(n)
    m_idx = np.arange(n)
    # grad[j] = sum_m a[m] * p[j+m]
    # = sum_m a[m] * p[j+m]
    
    # For efficiency, use the fact that p[j+m] for j,m in [0,n) 
    # accesses p[0:2n-1]
    # grad[j] = dot(a, p[j:j+n])
    
    # This is: grad = [dot(a, p[0:n]), dot(a, p[1:n+1]), ..., dot(a, p[n-1:2n-1])]
    # Which is the first n elements of the cross-correlation of p and a (with a reversed)
    
    # Use FFT for this:
    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1
    # cross_corr[j] = sum_m a[m] * p[j+m] = sum_m a[m] * p[j+m]
    # = (a_rev * p)[2n-2-j] where * is convolution
    # Actually: let's use ifft(fft(a) * conj(fft(p)))
    # corr[k] = sum_n a[n] * conj(p[n-k]) = sum_n a[n+k] * conj(p[n])
    # For real: corr[k] = sum_n a[n+k] * p[n]
    # We want: grad[j] = sum_m a[m] * p[j+m]
    # Let n=m, k=j: sum_m a[m] * p[m+j] = sum_n a[n] * p[n+j] = corr[-j] (with sign convention)
    
    # Using numpy/scipy: correlate(p, a) gives c[k] = sum_n p[n+k] * conj(a[n])
    # So correlate(p, a)[j] = sum_n p[n+j] * a[n] = sum_m a[m] * p[m+j] = grad[j] ✓
    
    from scipy.signal import correlate
    grad = correlate(p, a, mode='valid')  # This gives length (2n-1) - n + 1 = n
    
    # Center gradient (project to sum=0 hyperplane)
    grad_centered = grad - np.mean(grad)
    
    # Momentum update
    momentum_buf = beta_mom * momentum_buf - lr * grad_centered
    
    # Update
    a_new = a + momentum_buf
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-12:
        a_new = a.copy()
    else:
        a_new = a_new / s_new
    
    obj = compute_obj(a_new)
    return a_new, momentum_buf, obj


def softmax_gradient_optimize(a, time_budget, beta_start=50, beta_end=5, 
                                lr_start=0.05, beta_mom=0.7, verbose=False):
    """
    Optimize using softmax-gradient with adaptive temperature.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = np.zeros(n)
    lr = lr_start
    
    total_iters = 50000
    for it in range(total_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive beta: start high (focus on peak), decrease over time
        progress = min(1.0, it / total_iters)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        a, momentum, obj = softmax_gradient_step(a, beta, lr, momentum, beta_mom)
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        
        # Adaptive learning rate
        if it % 500 == 499:
            lr *= 0.95
        
        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SG iter {it}: obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.6f}, time={elapsed:.1f}s")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_response(a):
    """
    Solve LP: min M s.t. conv(a, g) <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, dual_vars) or (None, None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None, None
    a = a / s
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(M.value) if M.value is not None else float(np.max(A @ g_val))
                    # Get dual variables for the inequality constraints
                    # In cvxpy, dual values are accessed via constraint.dual_value
                    dual = np.array(constraints[0].dual_value) if constraints[0].dual_value is not None else np.zeros(2*n-1)
                    dual = np.maximum(dual, 0)
                    return g_val, M_val, dual
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                M_val = float(res.x[n])
                # Dual from linprog
                dual = np.maximum(res.ineqlin.marginals, 0) if hasattr(res.ineqlin, 'marginals') else np.zeros(m)
                return g_val, M_val, dual
    except Exception:
        pass
    
    return None, None, None


def lp_dual_guided_update(a, time_budget, verbose=False):
    """
    LP alternating with dual-guided subgradient updates.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        g_lp, M_lp, dual = solve_lp_for_response(a)
        if g_lp is None:
            break
        
        # Compute dual-guided subgradient
        # The dual variables tell us which peaks are most binding
        # Use them to weight the direction from a to g_lp
        if dual is not None and np.sum(dual) > 0:
            dual_norm = dual / np.sum(dual)
        else:
            # Fallback: use convolution peak distribution
            conv = compute_conv(a)
            dual_norm = conv
            dual_norm = dual_norm / np.sum(dual_norm)
        
        # Update: move towards g_lp, weighted by dual
        # Simple convex combination with adaptive step
        step_candidates = [0.1, 0.2, 0.3, 0.5, 0.8, 1.0]
        best_t_obj = np.inf
        best_a_local = a.copy()
        
        for t in step_candidates:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            if verbose and it % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP-dual iter {it}: obj={best_obj:.8f}, time={elapsed:.1f}s")
        else:
            # No improvement: try a perturbation
            # Use dual to guide perturbation: peaks with high dual need more reduction
            conv = compute_conv(a)
            target = np.full(2*n-1, best_obj / (2.0 * n))  # Target peak level
            excess = np.maximum(conv - target, 0)
            # Perturb in direction that reduces excess peaks
            rng = np.random.default_rng(hash(it) % (2**31))
            scale = 0.02
            perturbation = rng.normal(0, scale, n)
            a = a * np.exp(perturbation)
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
        
    return best_a


def coordinate_descent(a, time_budget, verbose=False):
    """
    Coordinate descent: adjust pairs of coordinates to reduce peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find top contributing pairs to the peak
        # (a*a)[k_star] = sum_j a[j] * a[k_star - j]
        contributions = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                contributions[j] = a[j] * a[k_j]
        
        sorted_idx = np.argsort(contributions)[::-1]
        
        improved = False
        for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
            if improved:
                break
            for i in sorted_idx[:15]:
                i = int(i)
                # Target: move mass to a position that doesn't contribute to peak
                # Position k_star - i contributes, so move to k_star - i + offset
                target = (k_star - i + n // 3) % n
                if target == i:
                    target = (i + n // 4) % n
                
                delta = frac * a[i]
                if delta <= 1e-10 or delta > 0.9 * a[i]:
                    continue
                
                a[i] -= delta
                a[target] += delta
                new_obj = compute_obj(a)
                
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[target] -= delta
        
        if not improved:
            # Random pair adjustment
            for _ in range(10):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                delta = 0.01 * a[i]
                if delta <= 1e-10:
                    continue
                a[i] -= delta
                a[j] += delta
                s = float(np.sum(a))
                a = a / s
                new_obj = compute_obj(a)
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
                    a = np.maximum(a, 0)
                    s = float(np.sum(a))
                    a = a / s
        
        if not improved and it > 500:
            break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation with smoothing."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    # Light smoothing
    if new_n > 10:
        kernel_size = min(5, new_n // 2 * 2 + 1)
        kernel = np.ones(kernel_size) / kernel_size
        a_new = np.convolve(a_new, kernel, mode='same')
    return np.maximum(a_new, 0)
