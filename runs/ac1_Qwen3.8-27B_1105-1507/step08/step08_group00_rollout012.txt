"""
Optimizes step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_aware_gradient(a, tau, K=10):
    """
    Compute gradient with peak-aware weighting.
    Top K peaks get full weight, others get reduced weight.
    This focuses the optimization on the most critical peaks.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    
    if max_conv < 1e-15:
        return np.zeros(n), conv
    
    # Identify top K peaks
    top_K_idx = np.argsort(conv)[::-1][:K]
    
    # Compute weights: top K get exp(tau * conv[k] / max_conv), others get reduced
    scaled = conv / max_conv
    exp_vals = np.exp(tau * (scaled - 1.0))
    
    # Reduce weight for non-top-K peaks
    weight_mask = np.ones(len(conv))
    weight_mask[top_K_idx] = 1.0
    weight_mask[np.setdiff1d(np.arange(len(conv)), top_K_idx)] = 0.01
    
    p = exp_vals * weight_mask
    p_sum = float(np.sum(p))
    if p_sum < 1e-15:
        p = np.exp(tau * (scaled - 1.0))
        p = p / p.sum()
    else:
        p = p / p_sum
    
    # Cross-correlation gradient
    N = 1
    while N < 3 * n:
        N <<= 1
    
    P = fft(p, N)
    A = fft(a, N)
    g_full = ifft(P * np.conj(A)).real
    g = 2.0 * g_full[:n]
    
    return g, conv


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_m(a, M_init=None, extra_constraints=None):
    """
    Solve: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Optionally with extra constraints.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable(initial_value=M_init if M_init else None)
    constraints = [cp.sum(g) == 1]
    
    if extra_constraints is not None:
        constraints.extend(extra_constraints)
    
    constraints.append(A @ g <= M)
    
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    return None, None


def solve_lp_feasibility(a, M_target):
    """
    Check: exists g >= 0, sum(g)=1, s.t. conv(a,g) <= M_target?
    Returns feasible g if yes, None if no.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(g) == 1, A @ g <= M_target]
    
    prob = cp.Problem(cp.Minimize(0), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate') and g.value is not None:
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s
        except Exception:
            continue

    return None

def smoothed_gradient_descent(a, time_budget=300.0, tau_init=20.0, tau_end=800.0, 
                               K=8, verbose=False):
    """
    Smoothed gradient descent with peak-aware gradient and adaptive tau.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 4.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Nonlinear tau schedule: slow start, fast middle, slow end

            tau = tau_init + (50 - tau_init) * (progress / 0.2)

        else:
            tau = tau_end
        
        if it % 5000 == 0 and verbose:
            print(f"    SGD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Barzilai-Borwein adaptive step
        if prev_g is not None and it % 3 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = max(1e-8, min(20.0 / np.sqrt(n), 0.7 * lr + 0.3 * bb_lr))

        # Update with momentum
        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it
            stall_count = 0
        else:
            stall_count += 1

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1

        # Decay learning rate
        lr = max(1e-8, lr * (1 - 0.5 / (it + 10)))
        
        # Restart if stuck
        if stall_count > 15000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 4.0 / np.sqrt(n) * 0.5
            stall_count = 0

    return best_a


def subgradient_peak_shave(a, time_budget=100.0, K=5, verbose=False):
    """
    Subgradient method that specifically shaves the top K peaks.
    Uses adaptive step size based on the gap between top peaks.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 3.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        max_val = conv[top_idx[0]]
        
        if max_val < 1e-15:
            break
        
        # Compute subgradient as weighted sum of peak contributions
        # Weight: exponential of (peak_value / max_value)
        weights = np.exp((conv[top_idx] - max_val) / (max_val * 0.1))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            # conv[k] = sum_j a[j] * a[k-j]
            # d(conv[k])/da[j] = a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n)
            # But for the convolution gradient: d(conv[k])/da[j] = a[k-j] (for valid range)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                contrib = a[j_start:j_end + 1][::-1]
                g[i_start:i_end + 1] += w * contrib * 2  # factor of 2 for symmetry
        
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 300:
                lr *= 0.5
                v *= 0.2
                stall = 0
                if lr < 1e-10:
                    break
            else:
                lr *= 0.998

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
