import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp(a, top_k=None):
    """
    Solve LP: min M s.t. (a*g)_k <= M for all k, sum(g)=1, g>=0.
    If top_k is given, only use top_k most active constraints.
    Returns (g_normalized, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    
    # Current max of a*a for reference
    conv_ref = compute_conv(a)
    M_ref = float(np.max(conv_ref))
    
    if top_k is not None and top_k < 2*n-1:
        # Use only top_k most active constraints
        active_idx = np.argsort(-conv_ref)[:top_k]
        A_use = A[active_idx, :]
    else:
        A_use = A
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A_use @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(constraints[0].dual_value), 0)
                    except:
                        pass
                    return g_val, float(M.value), dual_y
        except:
            continue
    
    return None, None, None

def subgradient_descent(a, time_budget=100.0, lr_init=0.01, verbose=False):
    """
    Subgradient descent with adaptive step size on the simplex.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        obj = 2.0 * n * M  # since sum(a)=1
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            lr = min(lr * 1.05, 0.1)
        else:
            lr = max(lr * 0.9, 1e-5)
        
        # Find all maximizing lags
        k_star = int(np.argmax(conv))
        
        # Subgradient: v_m = 2*a[k*-m] for valid m
        v = np.zeros(n)
        for m in range(n):
            idx = k_star - m
            if 0 <= idx < n:
                v[m] = 2.0 * a[idx]
        
        # Project gradient onto tangent space (sum v = 0)
        v = v - np.mean(v)
        
        # Update
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        a = a_new
        it += 1
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    Subgrad: it={it}, obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def coordinate_polish(a, time_budget=30.0):
    """Coordinate descent polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        improved = False
        # Shuffle coordinate order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > time_budget:
                    break
            for frac in [0.02, 0.05, 0.1, -0.05, -0.1]:
                delta = frac * a[i]
                if a[i] + delta < 0:
                    continue
                # Move mass from i to j (random j)
                j = rng.integers(0, n)
                if j == i:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                if a_try[j] < 0:
                    continue
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        if not improved:
            break
    
    return best_a

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, top_k=None):
    """
    Solve LP: min M s.t. (a*g)_k <= M for all k, sum(g)=1, g>=0.
    If top_k given, only enforce top_k most active constraints.
    Returns (g_normalized, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    if top_k is not None and top_k < 2 * n - 1:
        conv_ref = compute_conv(a)
        active_idx = np.argsort(-conv_ref)[:top_k]
        A_use = A[active_idx, :]
    else:
        A_use = A

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c_conv = A_use @ g <= M
    c_sum = cp.sum(g) == 1
    prob = cp.Problem(cp.Minimize(M), [c_conv, c_sum])

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(c_conv.dual_value), 0)
                    except Exception:
                        pass
                    return g_val, float(M.value), dual_y
        except Exception:
            continue

    return None, None, None


def nesterov_subgradient(a, time_budget=100.0, verbose=False):
    """
    Nesterov-accelerated projected subgradient descent on the simplex.
    Minimizes 2*n*max_k (a*a)_k subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov momentum
    y = a.copy()
    momentum = 1.0
    lr = 0.05
    lr_min = 1e-6
    lr_max = 0.5
    stall_count = 0
    it = 0

    while time.monotonic() - start < time_budget:
        # Compute subgradient at y
        conv_y = compute_conv(y)
        M_y = float(np.max(conv_y))

        # Find all maximizing lags (for subdifferential)
        threshold = M_y * (1.0 - 1e-6)
        active_lags = np.where(conv_y >= threshold)[0]
        k_star = active_lags[0]

        # Subgradient: v_m = 2*a_y[k*-m] for valid m (average over active lags)
        v = np.zeros(n)
        for k in active_lags:
            for m in range(n):
                idx = k - m
                if 0 <= idx < n:
                    v[m] += 2.0 * y[idx]
        v /= len(active_lags)

        # Project gradient onto tangent space of simplex
        v = v - np.mean(v)

        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        alpha = (momentum - 1.0) / t_k
        y_new = y + alpha * (y - a)
        a_new = y_new - lr * v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a
            s_new = 1.0
        a_new = a_new / s_new
        momentum = t_k

        # Evaluate
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_count = 0
            lr = min(lr * 1.02, lr_max)
        else:
            stall_count += 1
            if stall_count > 20:
                lr = max(lr * 0.85, lr_min)
                if lr <= lr_min * 1.1:
                    # Restart Nesterov with perturbation
                    rng = np.random.default_rng(hash((it, n)) % (2**31))
                    a = best_a * np.exp(rng.normal(0, 0.03, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    y = a.copy()
                    momentum = 1.0
                    lr = 0.05
                    stall_count = 0
                    a = best_a.copy()
                    best_obj = compute_obj(a)

        a = a_new
        y = a_new - alpha * (a_new - y)
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy > 1e-10:
            y = y / sy

        it += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    Nesterov: it={it}, obj={best_obj:.8f}, lr={lr:.6f}, t={elapsed:.1f}s")

    return best_a

                best_a = a.copy()

                prev_a = None
                prev_prev_a = None
                momentum = a.copy()
                stall = 0
                cycle_count = 0
                if verbose:
                    print(f"    [Perturbed] obj={best_obj:.8f}")

        it += 1
        if verbose and it % 15 == 0:
            elapsed = time.monotonic() - start

def resample_asymmetric(a, new_n):
    """
    Resample with optional asymmetric stretching.
    Try both uniform and asymmetric resampling.
    """
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])

    results = []
    # Uniform resampling
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    results.append(np.maximum(a_new, 0))

    # Asymmetric: stretch left side
    x_new_asym = np.linspace(0, 1, new_n) ** 0.9
    a_asym1 = np.interp(x_new_asym, x_old, a)
    results.append(np.maximum(a_asym1, 0))

    # Asymmetric: stretch right side
    x_new_asym2 = np.linspace(0, 1, new_n) ** 1.1
    a_asym2 = np.interp(x_new_asym2, x_old, a)
    results.append(np.maximum(a_asym2, 0))

    # Return the best one
    best = results[0]
    best_obj = compute_obj(best)
    for r in results[1:]:
        o = compute_obj(r)
        if o < best_obj:
            best_obj = o
            best = r
    return best


def coordinate_polish(a, time_budget=30.0):
    """Coordinate descent / pairwise exchange polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    while time.monotonic() - start < time_budget:
        improved = False
        for round_idx in range(5):
            if time.monotonic() - start > time_budget:
                break
            order = rng.permutation(n)
            for i in order:
                if time.monotonic() - start > time_budget:
                    break
                # Try moving mass from i to a few random j's
                js = rng.integers(0, n, size=min(5, n))
                for j in js:
                    if j == i:
                        continue
                    for frac in [0.03, 0.08, 0.15, -0.08, -0.15]:
                        delta = frac * a[i]
                        if a[i] + delta < 0 or a[j] - delta < 0:
                            continue
                        a_try = a.copy()
                        a_try[i] += delta
                        a_try[j] -= delta
                        a_try = a_try / np.sum(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

    return best_a

        if remaining > 40 and obj < best_score + 0.01:
            alloc2 = min(80.0, remaining * 0.1)
            a_new2 = lp_alternating_v2(a_new, time_budget=alloc2)
            obj2 = compute_obj(a_new2)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_new2.copy()
                print(f"  LP after Nesterov: obj={best_score:.8f} **")

    # === Phase 3: Multi-n Exploration (350s) ===
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_parent = len(best_seq)
    n_candidates = sorted(set([
        max(200, n_parent - 200), max(200, n_parent - 100),
        n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200, n_parent + 400,
        500, 600, 700, 800, 900, 1000, 1100, 1200, 1400, 1600,
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 2500 and n != n_parent]
    # Sort: try close to parent first, then spread out
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        a_try = resample_asymmetric(best_seq, n_try)
        alloc = min(45.0, remaining * 0.12)
        a_opt = lp_alternating_v2(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
            # Also try Nesterov on this resampled sequence if close
            if obj_try < best_score + 0.01:
                remaining = deadline - time.monotonic()
                if remaining > 30:
                    alloc_n = min(30.0, remaining * 0.08)
                    a_opt_n = nesterov_subgradient(a_try, time_budget=alloc_n)
                    obj_n = compute_obj(a_opt_n)
                    if obj_n < best_score:
                        best_score = obj_n
                        best_seq = a_opt_n.copy()
                        print(f"  ** n={n_try} (Nesterov): obj={best_score:.8f} **")

    # === Phase 4: Final LP refinement (200s) ===
    print(f"\n=== Phase 4: Final LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(200.0, remaining * 0.25)
        a_new = lp_alternating_v2(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 4 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 4 done: obj={obj:.8f}")

        # Nesterov polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc_n = min(80.0, remaining * 0.15)
            a_new = nesterov_subgradient(best_seq, time_budget=alloc_n)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Nesterov polish: obj={best_score:.8f} **")
            # Short LP after Nesterov
            remaining = deadline - time.monotonic()
            if remaining > 20:
                alloc_lp = min(40.0, remaining * 0.2)
                a_new2 = lp_alternating_v2(a_new, time_budget=alloc_lp)
                obj2 = compute_obj(a_new2)
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a_new2.copy()
                    print(f"  LP after Nesterov: obj={best_score:.8f} **")

    # === Phase 5: Coordinate Polish (60s) ===
    print(f"\n=== Phase 5: Coordinate Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(60.0, remaining * 0.2)
        a_new = coordinate_polish(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Polish: obj={best_score:.8f} **")
        else:
            print(f"  Polish: obj={obj:.8f}")

    # === Phase 6: Random Restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n=== Phase 6: Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(8, int(remaining // 35))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if r % 4 == 0:
                # Gaussian perturbation
                a_r = best_seq * np.exp(rng.normal(0, 0.03 + 0.005 * r, n_best))
            elif r % 4 == 1:
                # Asymmetric block perturbation
                a_r = best_seq.copy()
                n_blocks = max(5, n_best // 80)
                block_size = max(1, n_best // n_blocks)
                n_perturb = max(2, n_blocks // 4)
                blocks = rng.choice(n_blocks, n_perturb, replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = min(si + block_size, n_best)
                    a_r[si:ei] *= rng.uniform(0.4, 1.6)
            elif r % 4 == 2:
                # Asymmetric tilt
                tilt = np.linspace(0.7, 1.3, n_best)
                a_r = best_seq * tilt * np.exp(rng.normal(0, 0.02, n_best))
            else:
                # Sparse large perturbation
                a_r = best_seq.copy()
                n_sparse = max(5, n_best // 20)
                idx = rng.choice(n_best, n_sparse, replace=False)
                a_r[idx] *= rng.uniform(0.2, 3.0, n_sparse)

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            # Alternate between LP and Nesterov for restarts
            alloc = min(remaining * 0.4, 30.0)
            if r % 2 == 0:
                a_opt = lp_alternating_v2(a_r, time_budget=alloc)
            else:
                a_opt = nesterov_subgradient(a_r, time_budget=alloc)
                # Short LP follow-up
                remaining2 = deadline - time.monotonic()
                if remaining2 > 15:
                    a_opt2 = lp_alternating_v2(a_opt, time_budget=min(15.0, remaining2 * 0.3))
                    if compute_obj(a_opt2) < compute_obj(a_opt):
                        a_opt = a_opt2

            obj_r = compute_obj(a_opt)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 7: Final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 25.0)
        a_final = lp_alternating_v2(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

        # Very short coordinate polish
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_pol = coordinate_polish(best_seq, time_budget=min(remaining * 0.3, 10.0))
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  Final polish: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify with evaluator
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
