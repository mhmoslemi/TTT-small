"""
Hybrid optimizer: Adam with log-sum-exp + LP alternating + multi-n search + polishing.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_smooth_grad(a, t):
    """
    Compute gradient of (2/t) * logsumexp(t * conv(a,a)).
    Returns (gradient, conv, max_conv).
    """
    n = len(a)
    conv = compute_conv(a)  # length 2n-1
    
    # Softmax weights
    shifted = t * (conv - np.max(conv))
    w = np.exp(shifted)
    w /= np.sum(w)
    
    # c[i] = sum_k w[k] * a[k-i] for valid k-i
    # Using FFT cross-correlation
    w_rev = w[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    conv_full = irfft(rfft(a, size) * rfft(w_rev, size), size)
    c = conv_full[n-1:2*n-1][::-1]
    
    grad = (2.0 / t) * c
    return grad, conv, float(np.max(conv))

    """
    Adam optimization with log-sum-exp smooth approximation.
    Adaptive temperature: start high, decrease over time.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05  # base learning rate
    
    # Temperature schedule
    t_start = 5.0
    t_end = 150.0
    # We'll use a schedule: t increases over time
    
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget
        
        # Temperature: increase from t_start to t_end
        # Use exponential schedule for better behavior
        t = t_start * (t_end / t_start) ** progress
        
        # Compute gradient
        try:
            grad, conv, max_conv = compute_smooth_grad(a, t)
        except:
            break
        
        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)

        if s_new < 1e-10:
            break

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. A@g <= M (for all rows), sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)
    
    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except:
                        pass
                    return z_val, float(M.value), dual_y
        except:
            continue
    
    # Fallback
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except:
        pass
    
    return None, None, None

def coordinate_descent(a, time_budget=30.0):
    """
    Coordinate descent: try adjusting each coordinate individually.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    coords = list(range(n))
    
    while time.monotonic() - start < time_budget:
        improved = False
        rng.shuffle(coords)
        
        for i in coords:
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-10:
                continue
            
            for frac in [0.01, 0.05, 0.1, -0.01, -0.05, -0.1]:
                delta = frac * a[i]
                a_try = a.copy()
                a_try[i] += delta
                if a_try[i] < 0:
                    continue
                # Redistribute to maintain sum
                remaining = 1.0 - a_try[i]
                others_mask = np.ones(n, dtype=bool)
                others_mask[i] = False
                others_sum = np.sum(a[others_mask])
                if others_sum < 1e-10:
                    continue
                scale = remaining / others_sum
                a_try[others_mask] = a[others_mask] * scale
                a_try = np.maximum(a_try, 0)
                a_try = a_try / np.sum(a_try)
                
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)

"""
Hybrid optimizer combining Adam with log-sum-exp smooth approximation,
LP alternating with cycle detection, multi-n exploration, and coordinate polishing.
Key innovations: FFT-based smooth gradient, adaptive temperature schedule,
weighted LP guidance, and aggressive multi-n search near the parent length.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_smooth_gradient(a, t):
    """
    Compute gradient of (1/t)*log(sum_k exp(t*(a*a)[k])).
    Returns (grad, conv, max_conv).
    """
    n = len(a)
    conv = compute_conv(a)

    # Numerically stable softmax
    shifted = t * (conv - np.max(conv))
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum

    # Gradient: dF/da[i] = (2/t) * sum_k w[k] * a[k-i] (for 0<=k-i<n)
    # c[i] = sum_k w[k] * a[k-i]
    # Using FFT: c[i] = (a * w_rev)[2n-2-i]
    w_rev = w[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    conv_full = irfft(rfft(a, size) * rfft(w_rev, size), size)
    c = conv_full[n - 1:2 * n - 1][::-1]

    grad = (2.0 / t) * c
    return grad, conv, float(np.max(conv))


def adam_logsumexp(a, time_budget=200.0, t_start=3.0, t_end=200.0, verbose=False):
    """
    Adam optimizer with log-sum-exp smooth approximation and adaptive temperature.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.3

    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    start = time.monotonic()
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Exponential temperature schedule
        t = t_start * (t_end / t_start) ** progress

        try:
            grad, conv, max_conv = compute_smooth_gradient(a, t)
        except Exception:
            break

        step += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad ** 2
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # Adam update with gradient clipping
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1.0:
            grad = grad / grad_norm * 1.0
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad ** 2

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller steps
            found = False
            for shrink in [0.3, 0.1, 0.01]:
                a_try = a - lr * shrink * m_hat / (np.sqrt(v_hat) + eps)
                a_try = np.maximum(a_try, 0.0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    found = True
                    break
            if not found and no_improve > 200:
                # Perturb to escape
                rng = np.random.default_rng(int(time.monotonic() * 1e7))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                m = np.zeros(n)
                v = np.zeros(n)
                step = 0
                no_improve = 0

        if verbose and step % 1000 == 0:
            print(f"    Adam: step={step}, t={t:.1f}, obj={best_obj:.8f}, elapsed={elapsed:.0f}s")

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0.
    Returns (g, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except Exception:
                        pass
                    return z_val, float(M.value), dual_y
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except Exception:
        pass

    return None, None, None


def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating with momentum and 2-cycle detection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = a.copy()
    stall = 0
    prev_a = None
    prev_prev_a = None

    while time.monotonic() - start < time_budget:
        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp, dual_y = result

        # 2-cycle detection
        if prev_prev_a is not None:
            if np.max(np.abs(a - prev_prev_a)) < 1e-8:
                a_break = 0.5 * (a + prev_a)
                a_break = np.maximum(a_break, 0)
                a_break = a_break / np.sum(a_break)
                obj_break = compute_obj(a_break)
                if obj_break < best_obj:
                    best_obj = obj_break
                    best_a = a_break.copy()
                a = a_break
                prev_a = None
                prev_prev_a = None
                momentum = a.copy()
                continue

        # Momentum
        momentum = 0.5 * momentum + 0.5 * g_lp
        momentum = np.maximum(momentum, 0)
        ms = float(np.sum(momentum))
        if ms > 1e-10:
            momentum = momentum / ms

        # Try multiple targets and mixtures
        best_cand_obj = np.inf
        best_cand = a.copy()
        for target in [g_lp, momentum]:
            for mix in [0.4, 0.6, 0.8, 1.0]:
                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj:
                    best_cand_obj = obj_try
                    best_cand = a_try.copy()

        if best_cand_obj < best_obj - 1e-13:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            prev_a = a.copy()
            prev_prev_a = prev_a
            a = best_a
            stall = 0
        else:
            stall += 1
            if stall > 12:
                rng = np.random.default_rng(int(time.monotonic() * 1e6))
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev_prev_a = None
                momentum = a.copy()
                stall = 0

        if verbose:
            elapsed = time.monotonic() - start
            if int(elapsed) % 20 == 0 and int(elapsed) != getattr(lp_alternating, '_last_print', -1):
                lp_alternating._last_print = int(elapsed)
                print(f"    LP: obj={best_obj:.8f}, elapsed={elapsed:.0f}s")

    return best_a


def weighted_lp_alternating(a, time_budget=80.0, t_weight=50.0):
    """
    LP with weights based on current convolution peaks.
    Minimizes sum_k w_k * (a*g)[k] instead of max.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Softmax weights (heavier on peaks)
        shifted = t_weight * (conv - np.max(conv))
        w = np.exp(shifted)
        w /= np.sum(w)

        A = build_conv_matrix(a)
        # Weighted objective: minimize w^T (A@g)
        c = A.T @ w  # gradient of weighted sum w.r.t. g

        # Solve: min c^T g s.t. sum(g)=1, g>=0, (A@g) <= max_conv * 1.01
        max_conv = float(np.max(conv))

        z = cp.Variable(n, nonneg=True)
        prob = cp.Problem(
            cp.Minimize(c @ z),
            [cp.sum(z) == 1, A @ z <= max_conv * 1.005]
        )

        solved = False
        for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    z_val = np.maximum(z.value, 0)
                    zs = float(np.sum(z_val))
                    if zs > 1e-10:
                        z_val = z_val / zs
                        solved = True
                        break
            except Exception:
                continue

        if not solved:
            # Fallback to standard LP
            result = solve_lp(a)
            if result[0] is None:
                break
            g_lp = result[0]
        else:
            g_lp = z_val

        # Mix
        for mix in [0.5, 0.7, 1.0]:
            a_try = (1.0 - mix) * a + mix * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                break

    return best_a


def coordinate_polish(a, time_budget=40.0):
    """Coordinate descent with pairwise moves."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        improved = False

        # Random pairwise moves
        for _ in range(100):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-12 or a[j] < 1e-12:
                continue
            for frac in [0.01, 0.03, 0.1, 0.2, -0.01, -0.03]:
                delta = frac * min(a[i], a[j])
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        # Single coordinate moves
        if not improved:
            coords = rng.permutation(n)
            for i in coords:
                if time.monotonic() - start > time_budget:
                    break
                if a[i] < 1e-12:
                    continue
                for frac in [0.01, 0.05, -0.01, -0.05]:
                    delta = frac * a[i]
                    a_try = a.copy()
                    a_try[i] = a[i] + delta
                    if a_try[i] < 0:
                        continue
                    s_rest = 1.0 - a_try[i]
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    rest_sum = np.sum(a[mask])
                    if rest_sum < 1e-10:
                        continue
                    a_try[mask] = a[mask] * (s_rest / rest_sum)
                    a_try = a_try / np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    if remaining > 40:
        print(f"\n=== Phase 7: Restarts (remaining={remaining:.0f}s) ===")
        n_best = len(best_seq)

                idx = rng.choice(n_best, n_sparse, replace=False)
                a_r[idx] *= rng.uniform(0.5, 1.8, n_sparse)

    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
