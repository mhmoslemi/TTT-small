"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with adaptive blending, MWA-based fast approximate LP,
gradient-guided perturbation, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        length = k_end - k_start + 1
        A[k_start:k_end + 1, j] = a[:length]
    return A

def solve_lp_fast(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0 using scipy HiGHS."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.ones((1, n + 1))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            return g_val / gs, float(res.x[n])
    return None, None

def mwa_approx_lp(a, n_iters=200, eta=None):
    """Approximate LP solution using Multiplicative Weights Algorithm on the dual.
    Dual: max min_j (A^T lambda)_j s.t. sum(lambda)=1, lambda>=0
    Then recover primal from complementary slackness."""
    n = len(a)
    a = a / np.sum(a)
    
    if eta is None:
        eta = 1.0 / np.sqrt(np.log(2*n-1))
    
    # A has shape (2n-1, n), A[k,j] = a[k-j] if valid
    # We need to build A efficiently
    A = build_conv_matrix(a)
    
    # MWA: lambda is distribution over rows (lags)
    # At each step: find j* = argmin_j (A^T lambda)_j
    # Update: lambda_k *= exp(eta * A[k, j*])
    
    lam = np.ones(2 * n - 1) / (2 * n - 1)
    
    for it in range(n_iters):
        # Compute A^T lambda (n values)
        # (A^T lambda)_j = sum_k A[k,j] * lam[k]
        # = sum_k a[k-j] * lam[k] for valid k
        # This is convolution-like
        
        # Use FFT for efficiency
        # (A^T lam)_j = sum_k A[k,j] lam[k]
        # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
        # So (A^T lam)_j = sum_{m=0}^{n-1} a[m] * lam[m+j] (where lam is extended with zeros)
        
        # This is the cross-correlation of a and lam
        n_fft = 1
        while n_fft < n + (2*n-1):
            n_fft <<= 1
        A_fft = rfft(a, n_fft)
        L_fft = rfft(lam, n_fft)
        # corr[j] = sum_m a[m] * lam[m+j] = ifft(conj(A_fft) * L_fft)[n_fft - j] ... 
        # Actually: ifft(A_fft * conj(L_fft)) gives sum_m a[m] * conj(lam[m-j])
        # Let me just do it directly for clarity since we're already in a loop
        
        # Direct computation using FFT:
        # We want c[j] = sum_k A[k,j] * lam[k]
        # A[k,j] = a[k-j] if 0<=k-j<n
        # c[j] = sum_{j}^{j+n-1} a[k-j] * lam[k] (with bounds)
        # Let m = k-j: c[j] = sum_{m=0}^{min(n-1, 2n-2-j)} a[m] * lam[m+j]
        # This is the cross-correlation of a and lam at lag j
        
        # Using FFT: cross-correlation(a, lam) at lag j = sum_m a[m] * lam[m+j]
        # = ifft(conj(FFT(a)) * FFT(lam))[n_fft - j] (with appropriate indexing)
        
        R = irfft(np.conj(A_fft) * L_fft, n_fft)
        # R[0] = sum a[m] lam[m], R[1] = sum a[m] lam[m-1], ...
        # Actually R[l] = sum_m a[m] * lam[(m-l) mod n_fft]

            c[j] = R[(n_fft - j) % n_fft]
        
        j_star = int(np.argmin(c))

        # lambda_k *= exp(eta * A[k, j_star])
        # A[k, j_star] = a[k - j_star] if 0 <= k - j_star < n
        update = np.zeros(2 * n - 1)
        for k in range(j_star, min(2*n-1, j_star + n)):
            update[k] = a[k - j_star]
        
        # For numerical stability, use log-space
        log_lam = np.log(lam + 1e-300)
        log_lam += eta * update
        log_lam -= np.max(log_lam)  # for stability
        lam = np.exp(log_lam)
        lam /= np.sum(lam)
    
    # Recover primal from dual: g_j > 0 only if (A^T lam)_j = max_j (A^T lam)_j
    # Actually, at optimality: g_j > 0 => (A^T lam)_j = M* (the optimal value)
    # And lam_k > 0 => (A g)_k = M*
    
    # Simple approach: set g proportional to indicator of "active" columns
    c_final = np.zeros(n)
    for j in range(n):
        c_final[j] = R[(n_fft - j) % n_fft]
    
    # The optimal M* = max_j c_final[j] (by duality)
    # Active set: j where c_final[j] is close to M*
    M_star = float(np.max(c_final))
    # g_j > 0 where c_final[j] = M* (complementary slackness)
    # But this gives a vertex of the primal... 
    
    # Alternative: use the dual solution to get a better primal
    # g = argmin { M : A@g <= M, sum(g)=1, g>=0 } with hint from dual
    # Simple heuristic: g proportional to the "slack" in dual
    # g_j = 1 / (M* - c_final[j] + epsilon) for j where c_final[j] < M*
    # And 0 where c_final[j] = M*? No, that's backwards.
    
    # Actually, the correct recovery: at the dual optimum, 
    # the primal optimum has g_j > 0 iff (A^T lam*)_j = M*_dual
    # So: g_j = 1/|active_set| for j in active_set
    
    # But this is just one vertex. For a better solution, we can use:
    # g_j proportional to exp(gamma * c_final[j]) for some gamma
    gamma = 1.0
    g = np.exp(gamma * c_final)
    g = np.maximum(g, 0)
    gs = np.sum(g)
    if gs > 1e-10:
        g = g / gs
    else:
        g = np.ones(n) / n
    return g, M_star

def admissible_blend_search(a, g, n_trials=15):
    """Find best blending parameter t in (0,1) using golden section search."""
    n = len(a)
    
    def obj_at_t(t):
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        conv = compute_conv(a_try)
        return 2.0 * n * float(np.max(conv))
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # 0.618...
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at_t(c)
    fd = obj_at_t(d)
    
    for _ in range(n_trials):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at_t(d)
    
    best_t = (lo + hi) / 2
    best_obj = obj_at_t(best_t)
    return best_t, best_obj, best_t  # returning t, obj at t

    # Let me just do it directly for the gradient computation

    for j in range(n):

        for k in np.where(active)[0]:
            if 0 <= k - j < n:
                val += a[k - j]
            if 0 <= j - k < n:
                val += a[j - k]
        grad[j] = val
    
    # This is O(n * |active|) which could be slow. Let's use FFT.
    # grad[j] = sum_k w[k] * a[k-j] (first term, k-j in [0,n))
    #         + sum_k w[k] * a[j-k] (second term, j-k in [0,n))

    # (a ⋆ w)[j] = sum_k a[k] w[k+j] ... not quite
    # Actually: sum_k w[k] a[k-j] = sum_m w[m+j] a[m] = (w shifted) conv a
    # = ifft(FFT(w) * FFT(a))[j] ... that's circular correlation
    
    # Let me just use the convolution directly:
    # grad[j] ≈ 2 * (a * w)[j] where * denotes appropriate cross-correlation
    # For the purposes of perturbation, an approximate gradient is fine
    
    # Use the simpler approach: multiply by noise in the direction of -grad
    # Since exact grad is expensive, use a rough approximation
    noise = rng.standard_normal(n) * scale
    # Bias towards reducing peak contributions
    # Elements that contribute to peaks should be decreased
    peak_contribution = np.zeros(n)
    for j in range(n):
        # Contribution of a[j] to the peaks
        for k in np.where(active)[0]:
            if 0 <= k - j < n:
                peak_contribution[j] += a[k - j]
    peak_contribution /= (np.max(peak_contribution) + 1e-10)
    
    perturbed = a * (1 + noise - peak_contribution * scale * 0.5)
    perturbed = np.maximum(perturbed, 0)
    s = float(np.sum(perturbed))
    if s < 1e-10:
        return a.copy()
    return perturbed / s

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with golden-section blending, multi-scale optimization,
gradient-guided perturbation, and aggressive multi-n exploration.
Key improvements over previous: better time allocation, golden-section line search,
structured perturbation guided by peak contributions, and two-phase n search.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        length = k_end - k_start + 1
        A[k_start:k_end + 1, j] = a[:length]
    return A


def solve_lp(a):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.ones((1, n + 1))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g))
        if gs > 1e-10:
            return g / gs, float(res.x[n])
    return None, None


def blend_search(a, g, n_max=20):
    """Golden-section search for optimal blending parameter t in [0,1]."""
    n = len(a)

    def obj_at(t):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        conv = compute_conv(a_try)
        return 2.0 * n * float(np.max(conv))

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at(c)
    fd = obj_at(d)

    for _ in range(n_max):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at(d)

    best_t = (lo + hi) / 2.0
    best_obj = obj_at(best_t)
    return best_t, best_obj


def peak_guided_perturb(a, rng, strength=0.15):
    """Perturb guided by peak contributions: reduce elements contributing to peaks."""
    n = len(a)
    a = a / np.sum(a)
    conv = compute_conv(a)

    # Find active peaks (top 30%)
    threshold = np.percentile(conv, 70)
    active = conv > threshold
    n_active = int(np.sum(active))
    if n_active == 0:
        noise = rng.standard_normal(n) * strength
        p = a * np.exp(noise)
        p = np.maximum(p, 0)
        s = np.sum(p)
        return p / s if s > 1e-10 else a.copy()

    # Compute peak contribution for each element using FFT
    # contrib[j] = sum_{k active} a[k-j] for valid k-j
    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1
    a_fft = rfft(a, n_fft)
    w = active.astype(np.float64)
    w_fft = rfft(w, n_fft)

    # contrib[j] = sum_k w[k]*a[k-j] = cross-correlation
    R = irfft(np.conj(a_fft) * w_fft, n_fft)
    contrib = np.zeros(n)
    for j in range(n):
        contrib[j] = R[(n_fft - j) % n_fft]

    # Also add the symmetric part: sum_k w[k]*a[j-k]
    # This is the same as sum_k w[j-k]*a[k] = R_correlated at lag j
    R2 = irfft(np.conj(w_fft) * a_fft, n_fft)
    for j in range(n):
        contrib[j] += R2[j]

    # Normalize contribution
    max_c = np.max(contrib)
    if max_c > 1e-10:
        contrib = contrib / max_c

    # Apply perturbation: reduce high-contribution elements, add noise
    noise = rng.standard_normal(n) * strength
    log_perturb = -contrib * strength + noise * 0.5
    p = a * np.exp(log_perturb)
    p = np.maximum(p, 0)
    s = float(np.sum(p))
    if s < 1e-10:
        return a.copy()
    return p / s


def random_perturb(a, rng, strength=0.2):
    """Pure random log-perturbation."""
    noise = rng.standard_normal(len(a)) * strength
    p = a * np.exp(noise)
    p = np.maximum(p, 0)
    s = float(np.sum(p))
    if s < 1e-10:
        return a.copy()
    return p / s

    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:

    a = a / s

    best_a = a.copy()

    stall = 0
    rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))

    while time.monotonic() - start < time_budget:

            a = g_lp.copy()

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng):

                a[i] += rng.standard_normal() * np.exp(-d / (width * 0.7))

    a = np.maximum(a, 0)
    s = float(np.sum(a))

            print(f"  Final: obj={best_obj:.8f}")
        else:
            print(f"  Final: obj={obj_final:.8f}")

    # === Phase 6: Last-ditch micro improvements ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        # Try a few small perturbations + quick LP
        rng2 = np.random.default_rng(seed + 999)
        for attempt in range(min(5, int(remaining / 8))):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            a_pert = peak_guided_perturb(best_a, rng2, strength=0.05)
            alloc = min(8.0, remaining * 0.4)
            a_opt = lp_alternate(a_pert, time_budget=alloc, verbose=False, max_stall=3)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
                print(f"  Micro {attempt}: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf and val > 0:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")
    return result
