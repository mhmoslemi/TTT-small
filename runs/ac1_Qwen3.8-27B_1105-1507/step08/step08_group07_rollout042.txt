import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def peak_directed_moves(a, time_budget=60.0, verbose=False):
    """
    Directly identify peaks in the convolution and try to reduce them.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Find the top peaks
        peak_indices = np.argsort(conv)[-5:]
        
        improved = False
        for pk in peak_indices:
            if time.monotonic() - start > time_budget:
                break
            
            # The peak at position pk: conv[pk] = sum_i a[i]*a[pk-i]
            # Find which i values contribute most
            contributions = np.zeros(n)
            for i in range(n):
                j = pk - i
                if 0 <= j < n:
                    contributions[i] = a[i] * a[j]
            
            # Try reducing high-contribution pairs
            top_contrib = np.argsort(contributions)[-10:]
            
            for i in top_contrib:
                j = pk - i
                if not (0 <= j < n) or i == j:
                    continue
                
                # Try moving mass from i to j (or vice versa)
                for frac in [0.01, 0.02, 0.05]:
                    delta = frac * a[i]
                    a_try = a.copy()
                    a_try[i] -= delta
                    a_try[j] += delta
                    if a_try[i] < 0:
                        continue
                    a_try = a_try / np.sum(a_try)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Add small random perturbation
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
    
    return best_a


def lp_robust_alternating(a, time_budget=200.0, verbose=False):

    s = float(np.sum(a))

        return a

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    cycle_history = []
    g_momentum = a.copy()
    stall = 0

    while time.monotonic() - start < time_budget:

            break

        # Cycle detection: compare with history

                is_cyclic = True
                break
        cycle_history.append(a.copy())
        if len(cycle_history) > 10:
            cycle_history = cycle_history[-10:]

        if is_cyclic:
            # Average over the cycle
            a_break = np.mean(cycle_history, axis=0)
            a_break = np.maximum(a_break, 0)
            a_break = a_break / np.sum(a_break)
            obj_break = compute_obj(a_break)
            if obj_break < best_obj - 1e-12:

                best_a = a_break.copy()
            a = a_break
            cycle_history = []
            g_momentum = a.copy()
            continue

        # Update momentum
        g_momentum = 0.5 * g_momentum + 0.5 * g_lp
        g_momentum = np.maximum(g_momentum, 0)
        gs = float(np.sum(g_momentum))
        if gs > 1e-10:
            g_momentum = g_momentum / gs

        # Try multiple targets and mixing ratios with line search
        targets = [g_lp, g_momentum]

        best_cand = a.copy()

        for target in targets:

                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj:
                    best_cand_obj = obj_try

            if verbose:

        else:
            stall += 1
            if stall > 10:
                # Random perturbation to escape
                sigma = 0.02 + 0.01 * min(stall - 10, 5)
                a = best_a * np.exp(rng.normal(0, sigma, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)

                g_momentum = a.copy()

            else:

                a_try = a_try / np.sum(a_try)

def coordinate_polish(a, time_budget=60.0):

    n = len(a)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimized step function search using:
1. LP alternating with adaptive step sizes, long-cycle detection, and momentum
2. Log-sum-exp smoothed gradient descent with FFT-based gradient
3. Peak-directed pairwise moves
4. Multi-n exploration with interpolation
5. Coordinate polish for final refinement
6. Diversified random restarts

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. A@z <= M, sum(z)=1, z>=0.
    Returns (z_normalized, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ z <= M
    c2 = cp.sum(z) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    return z_val, float(M.value)
        except Exception:
            continue

    return None, None


def compute_smooth_grad(a, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    p = np.exp(tau * (conv - max_conv))
    p_sum = float(np.sum(p))
    if p_sum < 1e-300:
        return np.zeros(n)
    p = p / p_sum
    cross = np.correlate(p, a, mode='full')
    grad = 2.0 * cross[:n]
    return grad


def smooth_gradient_descent(a, time_budget=80.0, verbose=False):
    """
    Log-sum-exp smoothed gradient descent with adaptive temperature and step size.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    tau = 5.0
    lr = 0.005
    stall = 0

    while time.monotonic() - start < time_budget:
        grad = compute_smooth_grad(a, tau)
        grad_norm = float(np.sqrt(np.sum(grad * grad)))
        if grad_norm < 1e-15:
            break

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.05, 0.05)
            tau = max(tau * 0.98, 1.0)
        else:
            stall += 1
            lr *= 0.7
            if stall > 20:
                tau *= 1.3
                lr = 0.005
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0

        if verbose and int((time.monotonic() - start) * 2) % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    SGD: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.5f}, t={elapsed:.0f}s")

    return best_a

def peak_directed_moves(a, time_budget=50.0, verbose=False):
    """
    Identify peaks in convolution and try targeted pairwise mass transfers.
    """
    n = len(a)

            if time.monotonic() - start > time_budget:

            # Find which index pairs contribute most to this peak
            contributions = np.zeros(n)
            for i in range(n):
                j = int(pk) - i
                if 0 <= j < n:
                    contributions[i] = a[i] * a[j]

            top_idx = np.argsort(contributions)[-12:]
            for i in top_idx:
                j = int(pk) - i
                if not (0 <= j < n) or i == j:
                    continue
                for frac in [0.01, 0.02, 0.03, 0.05]:
                    delta = frac * min(a[i], a[j])

                        continue
                    a_try = a.copy()
                    # Try moving mass from i to the "other side"
                    a_try[i] -= delta
                    a_try[j] += delta
                    if a_try[i] < -1e-15:
                        continue
                    a_try = np.maximum(a_try, 0)
                    st = float(np.sum(a_try))
                    if st < 1e-10:
                        continue
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Small random perturbation
            a = best_a * np.exp(rng.normal(0, 0.008, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

            a_opt = lp_robust_alternating(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final LP + polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(20.0, remaining * 0.5)
        a_final = lp_robust_alternating(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            alloc = min(10.0, remaining * 0.5)
            a_final = coordinate_polish(best_seq, time_budget=alloc)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final polish: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
