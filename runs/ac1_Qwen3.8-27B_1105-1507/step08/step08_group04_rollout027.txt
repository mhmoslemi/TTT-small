"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternating, subgradient methods, multi-length exploration,
and Nelder-Mead local search.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import cvxpy as cp
import time
import math

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A

def solve_lp_cvxpy(a):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS, cp.MOSEK]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    return None, None

def solve_lp_scipy(a):
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.ones((1, n + 1))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            return g_val / gs, float(res.x[n])
    return None, None

def solve_lp(a):
    g, M = solve_lp_cvxpy(a)
    if g is not None:
        return g, M
    return solve_lp_scipy(a)

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with line search, subgradient polishing, multi-length
exploration, diverse restarts, and Nelder-Mead on reduced coordinates.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import cvxpy as cp
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp_cvxpy(a, timeout=30):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS, cp.MOSEK, cp.SCS]
    for solver in solvers:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > timeout:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    return None, None


def solve_lp_scipy(a):
    """Fallback LP solver using scipy."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.ones((1, n + 1))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 60})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp(a):
    """Try all LP solvers, return best result."""
    g, M = solve_lp_cvxpy(a)
    if g is not None:
        return g, M
    return solve_lp_scipy(a)

def subgradient_optimize(a, time_budget=60.0, top_k=15, verbose=False):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.9
    lr = 1.0 / n
    x = a.copy()

    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1

    A_fft = rfft(a, n_fft)

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # Recompute FFT of a periodically
        if it % 50 == 0:
            A_fft = rfft(a, n_fft)

        # Compute subgradient using top-k peaks
        conv = compute_conv(a)
        if top_k >= len(conv):
            k_indices = np.arange(len(conv))
        else:
            k_indices = np.argsort(conv)[::-1][:top_k]

        w = np.zeros(2 * n - 1)
        for k in k_indices:
            w[int(k)] = 1.0 / len(k_indices)

        W_fft = rfft(w, n_fft)
        R = irfft(np.conj(A_fft) * W_fft, n_fft)
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = R[(n_fft - j) % n_fft]
        grad *= 4.0 * n
        grad -= np.mean(grad)

        # Nesterov update
        momentum = beta * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        t_k = 2.0 / (it + 2.0)
        x = a_new + (1 - t_k) * (a_new - a)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-10:
            x = x / sx

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        # Adaptive learning rate
        if it % 100 == 99:
            lr *= 0.995

        if verbose and it % 5000 == 0:
            print(f"    SubGrad: it={it}, obj={best_obj:.8f}")

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_initialization(n, rng):

    elif choice == 3:

                a[i] *= 1.0 + rng.standard_normal() * np.exp(-d / (width * 0.5))

    elif choice == 5:

    else:

        a = np.cumsum(rng.standard_normal(n))

def nelder_mead_reduced(a, time_budget=30.0, num_coords=20):
    """Nelder-Mead on a random subset of coordinates (keeping others fixed)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    start = time.monotonic()
    rng = np.random.default_rng(999)

    # Select random coordinates to optimize
    coords = rng.choice(n, size=min(num_coords, n), replace=False)
    coords = np.sort(coords)
    fixed = np.ones(n, dtype=bool)
    fixed[coords] = False

    a_original = a.copy()

    def objective_reduced(x):
        full_a = a_original.copy()
        full_a[coords] = np.maximum(x, 0)
        s_full = float(np.sum(full_a))
        if s_full < 1e-10:
            return 1e10
        full_a = full_a / s_full
        return compute_obj(full_a)

    x0 = a_original[coords].copy()
    try:
        res = minimize(objective_reduced, x0, method='Nelder-Mead',
                       options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10,
                                'adaptive': True})
    except Exception:
        return a_original

    elapsed = time.monotonic() - start
    if elapsed > time_budget:
        return a_original

    best_a = a_original.copy()
    if res.fun < compute_obj(a_original):
        best_a[coords] = np.maximum(res.x, 0)
        s_best = float(np.sum(best_a))
        if s_best > 1e-10:
            best_a = best_a / s_best

    return best_a

            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diverse restarts ===
    print(f"\n=== Phase 5: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    for d in range(10):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_div = diverse_initialization(n_best, rng)
        alloc = min(35.0, remaining * 0.15)
        a_opt = lp_alternate(a_div, time_budget=alloc)
        obj_div = compute_obj(a_opt)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse init {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse init {d}: obj={obj_div:.8f}")

    # === Phase 6: Nelder-Mead local search on best ===
    print(f"\n=== Phase 6: Nelder-Mead Local Search ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Try several random coordinate subsets
        for trial in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(20.0, remaining * 0.3)
            a_nm = nelder_mead_reduced(best_a, time_budget=alloc, num_coords=30)
            obj_nm = compute_obj(a_nm)
            if obj_nm < best_obj:
                best_obj = obj_nm
                best_a = a_nm
                print(f"  ** NM trial {trial}: obj={best_obj:.8f} **")
            else:
                print(f"  NM trial {trial}: obj={obj_nm:.8f}")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc_lp = min(remaining * 0.5, 100.0)
        a_final = lp_alternate(best_a, time_budget=alloc_lp, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.8f} **")

    # === Phase 8: Subgradient micro-polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_sg = min(remaining * 0.4, 40.0)
        a_sg = subgradient_optimize(best_a, time_budget=alloc_sg, top_k=10)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  Final SubGrad: obj={best_obj:.8f} **")

    # === Phase 9: One more LP if time allows ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_last = min(remaining * 0.6, 30.0)
        a_last = lp_alternate(best_a, time_budget=alloc_last)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]

    # Validate using the actual evaluation function
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        # Fallback
        result = [1.0] * 800
        best_obj = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
