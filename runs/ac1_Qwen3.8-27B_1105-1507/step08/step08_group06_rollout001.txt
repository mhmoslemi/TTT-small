    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

        alpha = alpha + (1.0 - alpha) * 0.01
    
    return best_a


def projected_gradient_multipeak(a, time_budget=100.0, verbose=False):
    """
    Projected subgradient method using multiple active peaks simultaneously.
    Uses adaptive step size with backtracking.
    """
    n = len(a)
    a = normalize(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov momentum
    y = a.copy()
    velocity = np.zeros(n)
    beta = 0.9
    lr = 0.1
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute convolution at y (Nesterov point)
        conv = compute_conv(y)
        max_val = float(np.max(conv))
        
        # Get active peaks (top k peaks)
        k_peaks = min(10, 2 * n - 1)
        active = np.argsort(conv)[::-1][:k_peaks]
        # Weight by how close to max
        weights = np.array([conv[k] / max_val for k in active])
        weights = weights ** 2  # emphasize the top peaks
        weights = weights / weights.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for k, w in zip(active, weights):
            k = int(k)
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += w * y[k - indices]
        
        # Center gradient (project onto sum=0 hyperplane)
        grad_centered = grad - np.mean(grad)
        
        # Momentum step
        velocity = beta * velocity - lr * grad_centered
        y_new = y + velocity
        y_new = normalize(y_new)
        
        # Nesterov update
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        y = y_new + (1 - t_k) * (y_new - y)
        y = normalize(y)
        
        obj = compute_obj(y)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = y.copy()
        
        # Adaptive learning rate
        if it % 100 == 99:
            lr = max(lr * 0.95, 0.001)
            beta = min(0.99, beta + 0.001)
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_for_response(a):
    """
    Solve LP: min M s.t. A@g <= M for all k, sum(g)=1, g>=0.
    A = conv_matrix(a), so A@g = conv(a, g).
    """
    n = len(a)
    a = normalize(a)
    A = build_conv_matrix(a)
    m = 2 * n - 1
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, **{'max_time': 30})
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    return g_val
        except Exception:
            continue
    
    # Fallback: scipy linprog
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 30})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs
    except Exception:
        pass
    
    return None


def lp_alternating(a, time_budget=100.0, verbose=False):
    """
    LP alternating: for fixed a, find g minimizing max(conv(a,g)),
    then update a <- (1-t)*a + t*g.
    """
    n = len(a)
    a = normalize(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    momentum = np.zeros(n)
    beta_mom = 0.5
    
    while time.monotonic() - start < time_budget:
        g_lp = solve_lp_for_response(a)
        if g_lp is None:
            break
        
        # Try multiple step sizes
        best_t_obj = np.inf
        best_t_val = 0.3
        best_a_local = best_a.copy()
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            direction = g_lp - a
            momentum = beta_mom * momentum + (1 - beta_mom) * direction
            a_try = normalize(a + t * momentum)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            if verbose:
                print(f"    LP-alt: obj={best_obj:.8f}, t={best_t_val:.2f}")
        else:
            # Perturb to escape
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = normalize(a)
            momentum *= 0.5
    
    return best_a


def pairwise_local_search(a, time_budget=60.0):
    """
    Pairwise local search: try moving mass between pairs of positions.
    Focus on positions that contribute to the peak.
    """
    n = len(a)
    a = normalize(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        # Recompute peak periodically
        if it % 10 == 0:
            conv = compute_conv(a)
            k_star = int(np.argmax(conv))
        
        # Try a few random pairs
        for _ in range(5):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            for frac in [0.01, 0.02, 0.05, 0.1]:
                delta = frac * min(a[i], a[j])
                if delta <= 0:
                    continue
                a[i] -= delta
                a[j] += delta
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s < 1e-10:
                    a[i] += delta
                    a[j] -= delta
                    continue
                a_try = a / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def generate_initialization(kind, n, rng):
    """Generate diverse initialization."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
        return a / np.sum(a)
    elif kind == 'edge_concentrated':
        x = np.arange(n)
        center = (n - 1) / 2.0
        a = 1.0 + 0.5 * np.abs(x - center) / center
        return a / np.sum(a)
    elif kind == 'center_concentrated':
        x = np.arange(n)
        center = (n - 1) / 2.0
        a = 1.0 + 0.5 * (1 - np.abs(x - center) / center)
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = (n - 1) / 2.0
        x = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return a / np.sum(a)
    elif kind == 'random':
        a = rng.random(n) + 0.01
        return a / np.sum(a)
    elif kind == 'bimodal':
        a = np.ones(n) * 0.5
        a[n//4] = 2.0
        a[3*n//4] = 2.0
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'inverse_cos':
        # Concentrated at edges
        x = np.linspace(1e-6, np.pi - 1e-6, n)
        a = 1.0 / np.sqrt(np.sin(x))
        return a / np.sum(a)
    elif kind == 'quadratic_bowl':
        x = np.linspace(-1, 1, n)
        a = 1.0 + x**2
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def iterative_ratio_method(a, time_budget=100.0, verbose=False):
    """
    Iterative ratio method: adjust a_i based on how much it contributes
    to the peak of the convolution.
    """
    n = len(a)
    a = normalize(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 0.1
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        
        # Compute "excess" at each convolution position
        threshold = max_val * 0.95
        excess = np.maximum(conv - threshold, 0)
        
        # For each a_i, compute its weighted contribution to the excess
        # a_i contributes a_i * a_{k-i} to conv[k]
        # So the "pressure" on a_i is sum_k excess[k] * a_{k-i}
        pressure = np.zeros(n)
        active = np.where(excess > 0)[0]
        for k in active:
            k = int(k)
            i_max = min(n - 1, k)
            i_min = max(0, k - n + 1)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                pressure[indices] += excess[k] * a[k - indices]
        
        # Update: reduce a where pressure is high
        # a_i <- a_i * (1 - lr * pressure_i / (max_val * max_val))
        scale = max_val * max_val + 1e-10
        a_new = a * (1.0 - lr * pressure / scale)
        a_new = normalize(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        
        # Decay learning rate
        if it % 200 == 199:
            lr = max(lr * 0.9, 0.01)
    
    return best_a


def soft_max_smooth_obj(a, zeta=100.0):
    """
    Smooth approximation of max(conv(a,a)) using log-sum-exp.
    Returns (smooth_obj, grad) where grad is d/d a of the smooth max.
    """
    n = len(a)
    conv = compute_conv(a)
    # Soft max: (1/zeta) * log(sum exp(zeta * conv_k))
    # But for numerical stability:
    max_c = float(np.max(conv))
    shifted = conv - max_c
    log_sum_exp = max_c + np.log(np.sum(np.exp(zeta * shifted))) / zeta
    
    # Gradient: softmax weights on conv, then backprop through conv
    weights = np.exp(zeta * shifted) / np.sum(np.exp(zeta * shifted))
    
    grad = np.zeros(n)
    for k, w in enumerate(weights):
        if w < 1e-12:
            continue
        k = int(k)
        i_max = min(n - 1, k)
        i_min = max(0, k - n + 1)
        if i_max >= i_min:
            indices = np.arange(i_min, i_max + 1)
            grad[indices] += w * a[k - indices]
    
    return 2.0 * n * log_sum_exp, grad


def smooth_gradient_descent(a, time_budget=100.0, verbose=False):
    """
    Gradient descent on the smoothed objective using LSE approximation of max.
    """
    n = len(a)
    a = normalize(a.astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    zeta = 50.0  # sharpness of the soft max
    lr = 0.05
    momentum = np.zeros(n)
    beta = 0.8
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        obj, grad = soft_max_smooth_obj(a, zeta)
        
        # Center gradient
        grad_c = grad - np.mean(grad)
        
        # Momentum
        momentum = beta * momentum - lr * grad_c
        a_new = a + momentum
        a_new = normalize(a_new)
        
        true_obj = compute_obj(a_new)
        if true_obj < best_obj - 1e-15:
            best_obj = true_obj
            best_a = a_new.copy()
            a = a_new
        
        # Schedule
        if it % 100 == 99:
            lr = max(lr * 0.95, 0.001)
            zeta = min(zeta * 1.05, 500.0)  # increase sharpness
    
    return best_a
