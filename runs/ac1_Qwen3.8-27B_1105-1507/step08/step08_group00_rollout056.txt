def compute_conv(a):

    size = 1
    while size < 2 * n:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]


def compute_obj(a):

    conv = compute_conv(a)

def build_conv_matrix(a):

    n = len(a)

        A[j:j+r, j] = a[:r]
    return A

def exact_subgradient_step(a, K, lr, time_budget, verbose=False):
    """
    Exact subgradient descent using top-K peaks with momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        # Get top K peaks
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        
        # Weight by exponential of (peak - max)
        weights = np.exp((peak_vals - max_conv) * 10.0)
        weights /= weights.sum()
        
        # Compute subgradient: g[i] = 2 * sum_k w_k * (conv_k subgrad)_k[i]
        # For conv[k] = sum_j a[j]*a[k-j], d(conv[k])/da[i] = a[k-i] + a[i] (if valid) + ...
        # More precisely: d(conv[k])/da[i] = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j, i)]
        # = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n)... 
        # Actually for linear conv: conv[k] = sum_j a[j]*a[k-j]
        # d(conv[k])/da[i] = a[k-i] (from j=i term) + a[i] (from k-j=i, i.e., j=k-i term)
        # But if i = k-i (i.e., 2i=k), we only count once... actually no, in the sum over j,
        # j=i gives a[i]*a[k-i], and j=k-i gives a[k-i]*a[i]. These are the same term if 2i=k.
        # So d(conv[k])/da[i] = a[k-i] + a[i] if i != k-i, and a[i] if i = k-i.

        # If j=k-i: term is a[k-i]*a[i], derivative w.r.t. a[i] is a[k-i] (if 0<=k-i<n)
        # If i = k-i (i.e., 2i=k): these are the same term, derivative is 2*a[i]...

        # But in the sum, this term appears only once (j=i=k-i is a single value of j).
        # So d(conv[k])/da[i] = a[k-i] + a[k-i] = 2*a[k-i] if i != k-i and 0<=k-i<n
        #                            = 2*a[i] if 2i=k (same thing since k-i=i)

                v = np.zeros(n)
                lr *= 0.5
                stall = 0
                if lr < 1e-10:
                    break
        
        it += 1
    
    return best_a


def multiplicative_weights_step(a, K, time_budget, verbose=False):
    """
    Multiplicative weights (mirror descent) on the objective.
    a_j <- a_j * exp(-eta * g_j) / Z
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 1.0
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        weights = np.exp((peak_vals - max_conv) * 5.0)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end+1] += 2 * w * a[j_start:j_end+1][::-1]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-30)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            eta *= 0.95
            if stall > 50:
                break
        
        it += 1
    
    return best_a


def resample(a, new_n):
    """Resample to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
