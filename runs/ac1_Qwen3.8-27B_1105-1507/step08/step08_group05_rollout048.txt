import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def hann_window(n):
    """Hann window normalized to sum=1."""
    if n == 1:
        return np.array([1.0])
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    sw = np.sum(w)
    return w / sw


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)

    if n == 1:

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def build_conv_sparse(a_norm):
    """Build sparse matrix A where A @ g = conv(a_norm, g)."""
    n = len(a_norm)

    nnz = int(indptr[m])
    data = np.zeros(nnz)
    indices = np.zeros(nnz, dtype=np.int32)

    return csr_matrix((data, indices, indptr), shape=(m, n))


def solve_lp_conv(a, timeout=4.0):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""

    A = build_conv_sparse(a_norm)

    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:

            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):

                gs = float(np.sum(gv))
                if gs > 1e-10:

                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
                # Also try same-position scaling (not possible on simplex)
        
        if not improved and sweep > 20:
            break
    
    return best_a


def nesterov_subgrad(a0, time_budget=30.0, top_k=3, verbose=False):
    """Nesterov subgradient on the simplex."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.05
    patience = 0
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        conv_y = compute_conv(y)
        top_idx = np.argsort(conv_y)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                m_start = k - i_end
                m_end = k - i_start
                grad[m_start:m_end + 1] += 2.0 * y[i_start:i_end + 1][::-1]
        grad /= len(top_idx)
        grad = grad - np.mean(grad)
        
        y_new = y - step_size * grad
        a_new = project_simplex(y_new)
        t_k = 2.0 / (it + 2.0)
        y = a_new + (1.0 - t_k) * (a_new - a)
        y = project_simplex(y)
        a = a_new
        
        if it % 20 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 100:
                step_size = max(step_size * 0.5, 0.001)
                patience = 0
        
        if it % 200 == 199:
            step_size = max(step_size * 0.97, 0.002)
    
    return best_a

        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = hann_window(800)
        best_score = compute_obj(best_seq)

    print("\n=== Phase 1: Diverse Initializations ===")

    gauss = np.exp(-0.5 * ((np.arange(n_base) - n_base / 2.0) / (n_base / 4.0)) ** 2)
    gauss = np.maximum(gauss, 0)
    inits.append(gauss / np.sum(gauss))

    sine2 = np.sin(np.pi * np.arange(n_base) / n_base) ** 2
