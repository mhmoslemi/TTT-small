import numpy as np
import time

def compute_conv_fft(a):
    """Full convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa*fa, size)[:2*n-1]

def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    an = a / s
    conv = compute_conv_fft(an)
    return 2.0 * n * conv.max()

def coordinate_descent(a, time_budget=60.0, max_sweeps=50):
    """
    Coordinate descent on the objective 2*n*max(conv(a,a))/(sum(a))^2.
    Works on the unnormalized vector, normalizing at the end of each sweep.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        a = a / a.sum()  # normalize at start of sweep
        conv = compute_conv_fft(a)
        S = a.sum() - a  # S[i] = sum(a) - a[i] = (n-1)/n when normalized
        # Actually S is the same for all i when normalized: S = 1 - a[i]
        
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                    break
            
            ai = a[i]
            # Compute base convolutions (conv without the contribution of position i)
            base = conv.copy()
            ki = 2*i
            if ki < 2*n - 1:
                base[ki] -= ai * ai
            # For k where k-i is valid and k != 2i:
            j_start = max(0, i)
            j_end = min(n-1, 2*n-2-i)
            # k ranges from i to (n-1)+i, but k != 2i
            for k in range(max(0, i), min(2*n-1, n+i)):
                if k == 2*i:
                    continue
                j = k - i
                if 0 <= j < n:
                    base[k] -= 2*ai*a[j]
            
            # Golden section search for optimal x
            lo, hi = 0.0, max(ai * 5, 5.0/n)
            invphi = (np.sqrt(5) - 1) / 2
            for _ in range(40):
                m1 = hi - invphi*(hi - lo)
                m2 = lo + invphi*(hi - lo)
                
                obj1 = _coord_obj(m1, i, base, a, n)
                obj2 = _coord_obj(m2, i, base, a, n)
                
                if obj1 < obj2:
                    hi = m2
                else:
                    lo = m1
            
            x_opt = (lo + hi) / 2
            obj_opt = _coord_obj(x_opt, i, base, a, n)

                a[i] = x_opt

            a_try = a_try / st
            return compute_obj(a_try)

        lo, hi = 0.0, 1.0
        invphi = (np.sqrt(5.0) - 1.0) / 2.0
        best_t_local = best_t
        best_obj_local = obj_at_t(best_t)
        for _ in range(15):
            m1 = hi - invphi * (hi - lo)
            m2 = lo + invphi * (hi - lo)
            o1 = obj_at_t(m1)
            o2 = obj_at_t(m2)
            if o1 < o2:
                hi = m2
                if o1 < best_obj_local:
                    best_obj_local = o1
                    best_t_local = m1
            else:
                lo = m1
                if o2 < best_obj_local:
                    best_obj_local = o2
                    best_t_local = m2

        # Also try t = best_t_local and neighbors
        for t_try in [best_t_local, best_t_local * 0.7, best_t_local * 1.3, 0.3, 0.7]:
            if 0.01 <= t_try <= 1.0:
                o_try = obj_at_t(t_try)
                if o_try < best_obj_local:
                    best_obj_local = o_try
                    best_t_local = t_try

        it += 1
        if best_obj_local < best_obj - 1e-13:
            best_obj = best_obj_local
            a_try = (1.0 - best_t_local) * a + best_t_local * g_lp
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            best_a = a_try.copy()
            a = a_try
            best_t = best_t_local
            stall = 0
            if verbose and it % 3 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, t={best_t_local:.3f}, M={M_lp:.6f}, time={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 12:
                noise = min(0.25, 0.04 + 0.03 * (stall // 12))
                a = best_a * np.exp(rng.normal(0, noise, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                best_t = 0.4

    return best_a


def block_coordinate_descent(a, time_budget=60.0, block_size=8):
    """Block coordinate descent: optimize groups of adjacent coordinates together."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    invphi = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(100):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv_fft(a)
        a = a / a.sum()

        for start_idx in range(0, n, block_size):
            if time.monotonic() - start > time_budget:
                break
            end_idx = min(start_idx + block_size, n)
            block = a[start_idx:end_idx].copy()
            block_sum = block.sum()
            if block_sum < 1e-10:
                continue

            # Optimize the block using random directions
            for _ in range(5):
                direction = np.random._legacy.np.random.randn(end_idx - start_idx)
                # Project to simplex (keep sum = block_sum)
                direction = direction - direction.mean()
                if abs(direction.max()) < 1e-10:
                    continue
                # Line search along direction
                lo, hi = 0.0, 1.0
                for _ in range(10):
                    m1 = hi - invphi * (hi - lo)
                    m2 = lo + invphi * (hi - lo)

                    def obj_block_frac(frac):
                        new_block = block + frac * direction
                        new_block = np.maximum(new_block, 0)
                        new_block = new_block / new_block.sum() * block_sum
                        a_temp = a.copy()
                        a_temp[start_idx:end_idx] = new_block
                        a_temp = a_temp / a_temp.sum()
                        return compute_obj(a_temp)

                    o1 = obj_block_frac(m1)
                    o2 = obj_block_frac(m2)
                    if o1 < o2:
                        hi = m2
                    else:
                        lo = m1

                frac = (lo + hi) / 2.0
                a_temp = a.copy()
                a_temp[start_idx:end_idx] = np.maximum(block + frac * direction, 0)
                a_temp = a_temp / a_temp.sum()
                obj_temp = compute_obj(a_temp)
                if obj_temp < best_obj:
                    best_obj = obj_temp
                    best_a = a_temp.copy()
                    a = a_temp
                    a[start_idx:end_idx] = a[start_idx:end_idx] / a[start_idx:end_idx].sum() * block_sum
                    block = a[start_idx:end_idx].copy()

        a = a / a.sum()

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = 1.0 - np.abs(x)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x ** 2 + 0.01))
        a = a / a.sum()
    elif mode == 5:
        freqs = list(rng.integers(1, 5, size=3))
        phases = list(rng.uniform(0, 2 * np.pi, size=3))
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Triangle with perturbation
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)
        a += 0.1 * rng.uniform(0, 1, n)
        a = np.maximum(a, 0)
        a = a / a.sum()
    elif mode == 7:
        # Two narrow peaks
        a = np.zeros(n)
        c1, c2 = n // 3, 2 * n // 3
        a[c1:c1 + 5] = 1.0
        a[c2:c2 + 5] = 1.0
        a += 0.05 * rng.uniform(0, 1, n)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def nesterov_subgradient(a, time_budget=60.0, K=5):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 60

    while time.monotonic() - start < time_budget:
        conv = compute_conv_fft(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        spread = (peak_vals[0] - peak_vals[-1]) / max_val
        temp = 3.0 / (1.0 + spread * 10.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()

        # Gradient accumulation
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                k = int(k)
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                if i_start <= i_end:
                    j_start = k - i_end
                    j_end = k - i_start
                    g[i_start:i_end + 1] += w * y[j_start:j_end + 1][::-1]

        lr = lr0 / t_k
        a_new = y - lr * g
        # Project to simplex
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u)
        idx = np.arange(1, n + 1)
        cond = u * idx > (cssv - 1.0)
        rho = np.where(cond)[0]
        if len(rho) == 0:
            a_new = np.ones(n) / n
        else:
            rho = rho[-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_new - theta, 0.0)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                t_k = 1.0
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.4)

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines coordinate descent, LP alternating, Nesterov,
    and multi-scale n exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(parent_arr.sum())
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init: obj={best_score:.8f}")

    n_cur = len(best_seq)

    # === Phase 1: Coordinate Descent (fast, ~150s) ===
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.15, 150.0)
    if p1_budget > 15:
        print(f"\n=== Phase 1: Coordinate Descent (budget={p1_budget:.0f}s) ===")
        a_cd = coordinate_descent(best_seq, time_budget=p1_budget, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD result: obj={best_score:.8f} **")
        else:
            print(f"  CD result: obj={obj_cd:.8f} (no improvement)")

    # === Phase 2: LP Alternating (~250s) ===
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 250.0)
    if p2_budget > 20:
        print(f"\n=== Phase 2: LP Alternating (budget={p2_budget:.0f}s) ===")
        a_lp = lp_alternating(best_seq, time_budget=p2_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP result: obj={best_score:.8f} **")
        else:
            print(f"  LP result: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Coordinate Descent again (~100s) ===
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 15:
        print(f"\n=== Phase 3: CD Polish (budget={p3_budget:.0f}s) ===")
        a_cd2 = coordinate_descent(best_seq, time_budget=p3_budget, verbose=True)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  CD2 result: obj={best_score:.8f} **")
        else:
            print(f"  CD2 result: obj={obj_cd2:.8f} (no improvement)")

    # === Phase 4: Nesterov Subgradient (~80s) ===
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.08, 80.0)
    if p4_budget > 15:
        print(f"\n=== Phase 4: Nesterov (budget={p4_budget:.0f}s) ===")
        a_ns = nesterov_subgradient(best_seq, time_budget=p4_budget, K=7)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_ns:.8f} (no improvement)")

    # === Phase 5: Multi-n Exploration (~200s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Multi-n Exploration (budget={remaining:.0f}s) ===")

    n_options = sorted(set([
        max(300, n_cur - 100),
        max(300, n_cur - 50),
        n_cur + 50,
        n_cur + 100,
        n_cur + 200,
        600, 700, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500 and n != n_cur]
    n_options.sort(key=lambda x: abs(x - n_cur))

    n_tried = 0
    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        n_tried += 1

        a_try = resample_sequence(best_seq, n_try)
        # Quick CD + LP
        alloc_cd = min(remaining * 0.1, 40.0)
        if alloc_cd > 10:
            a_try = coordinate_descent(a_try, time_budget=alloc_cd)

        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.08, 40.0)
        if alloc_lp > 10:
            a_try = lp_alternating(a_try, time_budget=alloc_lp)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
        n_cur = len(best_seq)

    # === Phase 6: LP + CD Combined Polish (~150s) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 6: Final Polish (budget={remaining:.0f}s) ===")

    # LP
    lp_alloc = min(remaining * 0.3, 100.0)
    if lp_alloc > 15:
        a_lp_f = lp_alternating(best_seq, time_budget=lp_alloc, verbose=True)
        obj_lp_f = compute_obj(a_lp_f)
        if obj_lp_f < best_score:
            best_score = obj_lp_f
            best_seq = a_lp_f.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # CD
    remaining = deadline - time.monotonic()
    cd_alloc = min(remaining * 0.3, 80.0)
    if cd_alloc > 15:
        a_cd_f = coordinate_descent(best_seq, time_budget=cd_alloc, verbose=True)
        obj_cd_f = compute_obj(a_cd_f)
        if obj_cd_f < best_score:
            best_score = obj_cd_f
            best_seq = a_cd_f.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Another LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_lp_2 = lp_alternating(best_seq, time_budget=min(remaining * 0.5, 50.0))
        obj_lp_2 = compute_obj(a_lp_2)
        if obj_lp_2 < best_score:
            best_score = obj_lp_2
            best_seq = a_lp_2.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final CD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_cd_2 = coordinate_descent(best_seq, time_budget=min(remaining * 0.7, 30.0))
        obj_cd_2 = compute_obj(a_cd_2)
        if obj_cd_2 < best_score:
            best_score = obj_cd_2
            best_seq = a_cd_2.copy()
            print(f"  Final CD2: obj={best_score:.8f} **")

    # === Validation ===
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
