```python
"""
Hybrid optimizer combining:
1. Fast soft-max gradient descent (FFT-accelerated, O(n log n) per step)
2. LP alternating with 2-cycle detection
3. Subgradient polish with Nesterov acceleration
4. Multi-n exploration with resampling
5. Smart random restarts

Key innovation: Uses FFT-based cross-correlation for gradient computation,
enabling thousands of gradient steps per second. Combines smooth soft-max
optimization (fast exploration) with LP (targeted improvement) and
subgradient (final polish).
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_cross_corr(w, a):
    """Compute cross-correlation: result[j] = sum_k w[k]*a[k-j] for j=0..n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fw = rfft(w, size)
    fa = rfft(a, size)
    return irfft(fw * np.conj(fa), size)[:n]


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    c1 = A @ g <= M
    c2 = cp.sum(g) == 1
    prob = cp.Problem(cp.Minimize(M), [c1, c2])

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    dual_y = None
                    try:
                        dual_y = np.maximum(np.array(c1.dual_value), 0)
                    except:
                        pass
                    return g_val, float(M.value), dual_y
        except:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                g_val = g_val / gs
                return g_val, float(res.x[n]), None
    except:
        pass

    return None, None, None


def softmax_gradient_descent(a, time_budget=200.0, verbose=False):
    """
    Fast soft-max gradient descent with Nesterov acceleration.
    minimizes max(conv(a,a)) subject to sum(a)=1, a>=0.
    Uses FFT for O(n log n) gradient computation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov momentum
    momentum = np.zeros(n)
    beta = 50.0  # soft-max sharpness
    eta = 0.05  # base step size
    beta_growth = 1.0005
    beta_max = 500.0
    stall_count = 0

    while time.monotonic() - start < time_budget:
        # Compute a*a
        conv_aa = compute_conv(a)
        max_val = float(np.max(conv_aa))

        # Soft-max weights
        shifted = conv_aa - max_val
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / np.sum(exp_vals)

        # Gradient via FFT cross-correlation
        # grad[j] = beta * sum_k w[k] * a[k-j]
        cross = compute_cross_corr(w, a)
        grad = beta * cross

        # Nesterov step
        prev_a = a.copy()
        a = a + momentum
        a = np.maximum(a, 0)
        s_tmp = float(np.sum(a))
        if s_tmp > 1e-10:
            a = a / s_tmp
        else:
            a = prev_a.copy()

        # Recompute gradient at extrapolated point
        conv_aa2 = compute_conv(a)
        max_val2 = float(np.max(conv_aa2))
        shifted2 = conv_aa2 - max_val2
        exp_vals2 = np.exp(beta * shifted2)
        w2 = exp_vals2 / np.sum(exp_vals2)
        cross2 = compute_cross_corr(w2, a)
        grad2 = beta * cross2

        momentum = 0.9 * momentum - eta * grad2
        a_new = prev_a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = prev_a
        else:
            a_new = a_new / s_new

        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 50:
                # Anneal and reset momentum
                beta = min(beta * 1.5, beta_max)
                momentum = np.zeros(n)
                eta = max(eta * 0.7, 0.005)
                # Small random perturbation if very stuck
                if stall_count > 200:
                    rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                    a = best_a * np.exp(rng.normal(0, 0.02, n))
                    a = np.maximum(a, 0)
                    a = a / np.sum(a)
                    best_a = a.copy()
                    best_obj = compute_obj(a)
                    momentum = np.zeros(n)
                    stall_count = 0
                    eta = 0.05
                    beta = 50.0

        if verbose:
            elapsed = time.monotonic() - start
            if int(elapsed * 5) % 20 == 0:
                print(f"    SoftMax: obj={best_obj:.8f}, beta={beta:.1f}, t={elapsed:.0f}s")

    return best_a

    """LP alternating with 2-cycle detection and momentum."""
    n = len(a)

    start = time.monotonic()

    prev_a = None
    prev_prev_a = None
    g_momentum = a.copy()
    stall = 0

    while time.monotonic() - start < time_budget:
        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp, dual_y = result

        # 2-cycle detection
        if prev_prev_a is not None:
            diff = np.max(np.abs(a - prev_prev_a))
            if diff < 1e-8:
                a_break = 0.5 * (a + prev_a)
                a_break = np.maximum(a_break, 0)
                s_b = float(np.sum(a_break))
                if s_b > 1e-10:
                    a_break = a_break / s_b
                    obj_break = compute_obj(a_break)
                    if obj_break < best_obj - 1e-12:
                        best_obj = obj_break
                        best_a = a_break.copy()
                    a = a_break
                    result = solve_lp(a)
                    if result[0] is not None:
                        g_lp, M_lp, dual_y = result
                prev_a = None
                prev_prev_a = None
                g_momentum = a.copy()

        g_momentum = 0.5 * g_momentum + 0.5 * g_lp
        g_momentum = np.maximum(g_momentum, 0)
        gs = float(np.sum(g_momentum))
        if gs > 1e-10:
            g_momentum = g_momentum / gs

        targets = [g_lp, g_momentum]
        best_cand_obj = np.inf
        best_cand = a.copy()

        for target in targets:
            for mix in [0.3, 0.5, 0.7, 1.0]:
                a_try = (1.0 - mix) * a + mix * target
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_cand_obj:
                    best_cand_obj = obj_try
                    best_cand = a_try.copy()

        if best_cand_obj < best_obj - 1e-13:
            best_obj = best_cand_obj
            best_a = best_cand.copy()
            prev_a = a.copy()
            prev_prev_a = prev_a
            a = best_a
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                if int(elapsed * 3) % 15 == 0:
                    print(f"    LP-robust: obj={best_obj:.8f}, t={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 20:
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                if dual_y is not None and np.max(dual_y) > 0:
                    A = build_conv_matrix(a)
                    active_idx = np.where(dual_y > 0.01 * np.max(dual_y))[0]
                    if len(active_idx) > 0:
                        col_imp = np.sum(A[active_idx, :], axis=0)
                        pert = -np.sign(col_imp - np.mean(col_imp)) * 0.01
                        pert += rng.normal(0, 0.01, n)
                    else:
                        pert = rng.normal(0, 0.02, n)
                else:
                    pert = rng.normal(0, 0.02, n)

                a = best_a * np.exp(pert)
                a = np.maximum(a, 0)
                s_p = float(np.sum(a))
                if s_p > 1e-10:
                    a = a / s_p
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev_prev_a = None
                g_momentum = a.copy()
                stall = 0
                if verbose:
                    print(f"    [Perturbed] obj={best_obj:.8f}")
            else:
                a_try = a + 0.1 * (g_lp - a)
                a_try = np.maximum(a_try, 0)
                a_try = a_try / np.sum(a_try)
                if compute_obj(a_try) < best_obj:
                    a = a_try

    return best_a
