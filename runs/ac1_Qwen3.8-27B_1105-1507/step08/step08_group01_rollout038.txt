import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    from scipy.optimize import linprog
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_obj_and_grad(a, T):
    """
    Compute the smooth approximation of max(conv(a,a)) and its gradient.
    Uses log-sum-exp for numerical stability.
    Returns (smooth_max_value, gradient).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    conv = compute_conv(a_norm)
    
    # Log-sum-exp for stability
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    
    smooth_val = max_c + T * np.log(total)
    
    # Gradient: 2 * correlate(weights, a_norm, mode='valid')
    grad = 2.0 * np.correlate(weights, a_norm, mode='valid')
    
    # The objective is 2n * smooth_max_val
    # But we need to account for the normalization... 
    # Actually, since we're working with a_norm, the objective is 2n * smooth_max_val
    obj = 2.0 * n * smooth_val
    
    # The gradient w.r.t. a (not a_norm) needs the chain rule through normalization
    # If a_norm = a / sum(a), then d a_norm_i / d a_j = (delta_ij * sum(a) - a_i) / sum(a)^2
    # So grad_a_j = sum_i grad_norm_i * (delta_ij * s - a_i) / s^2
    # = (grad_norm_j * s - sum_i grad_norm_i * a_i) / s
    # where s = sum(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return obj, np.zeros(n)
    grad_norm_sum = float(np.sum(grad * a_norm))
    grad_a = (grad * s - grad_norm_sum * a) / s
    # Wait, let me redo this.
    # obj = 2n * smooth_max(a_norm) where a_norm = a/s, s = sum(a)
    # d obj / d a_j = 2n * sum_i (d smooth_max / d a_norm_i) * (d a_norm_i / d a_j)
    # d a_norm_i / d a_j = (delta_ij * s - a_i) / s^2
    # = 2n * sum_i grad_norm_i * (delta_ij * s - a_i) / s^2
    # = 2n / s^2 * (grad_norm_j * s - sum_i grad_norm_i * a_i)
    # = 2n / s * (grad_norm_j - grad_norm_sum * a_j / s)
    # = 2n / s * (grad_norm_j - grad_norm_sum * a_norm_j)
    
    grad_a = (2.0 * n / s) * (grad - grad_norm_sum * a_norm)
    
    return obj, grad_a


def smooth_max_gradient_descent(a, time_budget=100.0, verbose=False):
    """
    Smooth max gradient descent with adaptive temperature and momentum.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adaptive temperature
    T = 0.0005
    T_min = 0.00001
    T_max = 0.01
    lr = 0.005
    momentum = 0.9
    v = np.zeros(n)
    stall = 0
    best_stall_lr = lr
    
    max_iters = 10000
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        obj, grad = smooth_max_obj_and_grad(a, T)
        
        # Center the gradient (remove component in direction of 1, since we're on the simplex)
        grad_centered = grad - np.mean(grad)
        
        # Momentum update
        v = momentum * v - lr * grad_centered
        a_new = simplex_project(a + v)
        
        # Compute actual objective
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Reduce temperature gradually
            if it > 50:
                T = max(T_min, T * 0.9995)
        else:
            stall += 1
            if stall > 30:
                # Adaptive: increase temperature if weights too concentrated
                conv = compute_conv(a)
                max_c = np.max(conv)
                # Effective number of peaks
                shifted = (conv - max_c) / T
                shifted = np.clip(shifted, -500, 500)
                exp_vals = np.exp(shifted)
                weights = exp_vals / np.sum(exp_vals)
                eff_count = 1.0 / np.sum(weights**2)
                
                if eff_count < 5:
                    T = min(T_max, T * 1.5)  # Too concentrated, increase T
                elif eff_count > 50:
                    T = max(T_min, T * 0.7)  # Too spread, decrease T
                
                lr = max(1e-7, lr * 0.7)
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
        
        if verbose and it % 200 == 0:
            print(f"          GD it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}")
    
    return best_a


def lbfgs_smooth_max(a, time_budget=120.0, verbose=False):
    """
    Use L-BFGS-B on the smooth log-sum-exp approximation of the objective.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    T = 0.001
    
    def objective_and_grad(x):
        # x is on the simplex (we'll project)
        x_proj = simplex_project(x)
        conv = compute_conv(x_proj)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total
        smooth_val = max_c + T * np.log(total)
        obj = 2.0 * n * smooth_val
        grad = 2.0 * np.correlate(weights, x_proj, mode='valid')
        # Center gradient for simplex constraint
        grad = grad - np.mean(grad)
        return obj, grad
    
    # Run L-BFGS-B with bounds
    options = {'maxiter': 500, 'maxfun': 2000, 'ftol': 1e-12, 'gtol': 1e-8}
    
    try:
        res = scipy_minimize(
            lambda x: objective_and_grad(x)[0],
            a.copy(),
            jac=lambda x: objective_and_grad(x)[1],
            method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options=options
        )
        if res.x is not None:
            res_x = simplex_project(res.x)
            obj_res = compute_obj(res_x)
            if obj_res < best_obj:
                best_a = res_x.copy()
                best_obj = obj_res
                if verbose:
                    print(f"          L-BFGS: obj={best_obj:.8f}")
    except Exception as e:
        if verbose:
            print(f"          L-BFGS error: {e}")
    
    # Try with smaller T
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 30:
        T = 0.0003
        def objective_and_grad2(x):
            x_proj = simplex_project(x)
            conv = compute_conv(x_proj)
            max_c = np.max(conv)
            shifted = (conv - max_c) / T
            shifted = np.clip(shifted, -500, 500)
            exp_vals = np.exp(shifted)
            total = np.sum(exp_vals)
            weights = exp_vals / total
            smooth_val = max_c + T * np.log(total)
            obj = 2.0 * n * smooth_val
            grad = 2.0 * np.correlate(weights, x_proj, mode='valid')
            grad = grad - np.mean(grad)
            return obj, grad
        
        try:
            res = scipy_minimize(
                lambda x: objective_and_grad2(x)[0],
                best_a.copy(),
                jac=lambda x: objective_and_grad2(x)[1],
                method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options=options
            )
            if res.x is not None:
                res_x = simplex_project(res.x)
                obj_res = compute_obj(res_x)
                if obj_res < best_obj:
                    best_a = res_x.copy()
                    best_obj = obj_res
                    if verbose:
                        print(f"          L-BFGS (T={T}): obj={best_obj:.8f}")
        except Exception:
            pass
    
    return best_a


def pairwise_swap_optimize(a, time_budget=100.0, verbose=False):
    """
    Pairwise swap optimization: for pairs (i, j), try transferring mass from one to the other.
    More efficient than single-coordinate descent.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Precompute convolution for efficiency
    conv_cur = compute_conv(a)
    max_cur = np.max(conv_cur)
    
    iteration = 0
    while time.monotonic() - start < time_budget:
        iteration += 1
        improved = False
        
        # Random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try different transfer amounts
        max_transfer = min(a[i], a[j]) * 0.5
        for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
            transfer = max_transfer * frac
            if transfer < 1e-12:
                continue
            
            # Transfer from i to j
            a_try = a.copy()
            a_try[i] -= transfer
            a_try[j] += transfer
            if a_try[i] < 0:
                continue
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            
            # Transfer from j to i
            a_try = a.copy()
            a_try[j] -= transfer
            a_try[i] += transfer
            if a_try[j] < 0:
                continue
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved and iteration % 1000 == 0:
            if verbose:
                print(f"          Swap it{iteration}: obj={best_obj:.8f}")
        
        if not improved and iteration > 5000:
            break
    
    return best_a


def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating with convex line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Convex line search
        a_norm = a / np.sum(a)
        g_norm = g_lp / np.sum(g_lp)
        
        conv_aa = compute_conv(a_norm)
        # Cross convolution
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = rfft(a_norm, size)
        fg = rfft(g_norm, size)
        conv_ag = irfft(fa * fg, size)[:2 * n - 1]
        conv_gg = compute_conv(g_norm)
        
        def obj_at_t(t):
            val = (1 - t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
            return 2.0 * n * float(np.max(val))
        
        lo, hi = 0.0, 1.0
        for _ in range(15):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if obj_at_t(m1) < obj_at_t(m2):
                hi = m2
            else:
                lo = m1
        t_opt = (lo + hi) / 2.0
        
        a_new = (1 - t_opt) * a_norm + t_opt * g_norm
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 10 == 0:
                print(f"          LP it{it}: obj={best_obj:.8f}, t={t_opt:.4f}")
        else:
            # Try small step
            a_try = (1 - 0.05) * a_norm + 0.05 * g_norm
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st > 1e-10:
                a_try = a_try / st
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def structured_perturb(a, rng, scale=0.01):
    """
    Apply structured perturbations (low-frequency and high-frequency) to escape local minima.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    
    # Low-frequency perturbation (smooth wiggle)
    freq = rng.integers(1, 5)
    phase = rng.uniform(0, 2 * np.pi)
    t = np.linspace(0, 2 * np.pi * freq, n)
    wiggle = np.cos(t + phase)
    a_pert = a * (1 + scale * wiggle)
    a_pert = np.maximum(a_pert, 0)
    s = float(np.sum(a_pert))
    if s > 1e-10:
        a_pert = a_pert / s
    
    return a_pert

Actually, here's another idea: **The "dual certificate" approach**

At the optimum, we have:
- `sum_k (a*a)_k = 1` (for normalized a)
- `max_k (a*a)_k = M` (the value we're trying to minimize)
- There exist weights `w_k >= 0` (dual variables) such that:
  - `sum_k w_k = 1`
  - `sum_k w_k * (a*a)_k = M` (strong duality for the max)
  - For each i: `sum_k w_k * 2*a_{k-i} = constant` (stationarity)

The last condition means: `correlate(w, a, mode='valid')` is constant!

So at the optimum, there exists a probability vector `w` (over the 2n-1 autocorrelation lags) such that:
1. `w_k > 0` only where `(a*a)_k = M`
2. `correlate(w, a, mode='valid')` is constant

Condition 2 is a linear constraint on `a` (given `w`). Condition 1 determines which `w_k` are nonzero.

This gives us a way to verify if we're at the optimum: check if there exists `w` satisfying these conditions.

And it gives us a direction for improvement: if `correlate(w, a)` is not constant, we can move in the direction that makes it more constant.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def get_smooth_weights(conv, T):
    """Compute softmax weights for smoothing the max."""
    max_c = np.max(conv)
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    return weights, max_c


def compute_kkt_flatness_grad(a, w):
    """
    Compute gradient of var(correlate(w, a)) w.r.t. a.
    This measures how far we are from KKT stationarity.
    Returns the gradient vector (length n).
    """
    n = len(a)
    a_norm = a / np.sum(a)
    h = np.correlate(w, a_norm, mode='valid')
    h_centered = h - np.mean(h)
    grad = 2.0 * np.correlate(h_centered, w, mode='valid')
    return grad

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    ptype = rng.integers(0, 4)
    if ptype == 0:
        # Cosine wiggle
        freq = rng.integers(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.linspace(0, 2 * np.pi * freq, n)
        wiggle = np.cos(t + phase)
        a_pert = a * (1 + scale * wiggle)
    elif ptype == 1:
        # Gaussian blob
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(0.05, 0.2) * n
        t = np.arange(n)
        blob = np.exp(-0.5 * ((t - center) / width)**2)
        a_pert = a * (1 + scale * (blob - 1.0))
    elif ptype == 2:
        # Random smooth perturbation (low-pass filtered noise)
        noise = rng.normal(0, 1, n)
        # Simple low-pass: moving average
        k = max(2, n // 50)
        kernel = np.ones(k) / k
        noise_smooth = np.convolve(noise, kernel, mode='same')
        noise_smooth = noise_smooth - np.mean(noise_smooth)
        a_pert = a * (1 + scale * noise_smooth)
    else:
        # Sparse random perturbation
        n_sparse = max(3, n // 20)
        indices = rng.choice(n, n_sparse, replace=False)
        deltas = rng.normal(0, scale * 3, n_sparse)
        a_pert = a.copy()
        for idx, d in zip(indices, deltas):
            a_pert[idx] *= (1 + d)
    a_pert = np.maximum(a_pert, 0)
    s = float(np.sum(a_pert))
    if s > 1e-10:
        a_pert = a_pert / s
    else:
        a_pert = a / np.sum(a)
    return a_pert


def try_n_value(a_base, n_try, time_budget, verbose=False):
    """Optimize a sequence of length n_try starting from a_base (resampled)."""
    if n_try == len(a_base):
        a_try = a_base.copy()
    else:
        a_try = resample_sequence(a_base, n_try)

    # Quick GD
    a_opt = smooth_max_gradient_descent(a_try, time_budget=time_budget * 0.6, verbose=False)
    obj = compute_obj(a_opt)

    # If promising, do more
    remaining = time_budget - (time_budget * 0.6)
    if remaining > 10 and obj < np.inf:
        a_opt = smooth_max_gradient_descent(a_opt, time_budget=remaining * 0.5, verbose=False)
        obj = compute_obj(a_opt)

    return a_opt, obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: LP alternating + smooth max GD + KKT gradient + multi-n + diversification.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={len(parent)}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 800
        best_seq = np.ones(n_init) / n_init
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_init}): obj={best_score:.8f}")

    a = best_seq.copy()

    # === Phase 1: LP Alternating ===
    print(f"\n=== Phase 1: LP Alternating (n={len(a)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p1_budget = min(150.0, remaining * 0.2)
        a_lp = lp_alternating(a, time_budget=p1_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a = a_lp.copy()
            print(f"  Phase 1 done: obj={best_score:.8f}")
        else:
            print(f"  Phase 1 done: obj={obj_lp:.8f} (no improvement)")

    # === Phase 2: Smooth Max GD (main workhorse) ===
    print(f"\n=== Phase 2: Smooth Max GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = min(200.0, remaining * 0.25)
        a_gd = smooth_max_gradient_descent(a, time_budget=p2_budget, verbose=True)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_score:
            best_score = obj_gd
            best_seq = a_gd.copy()
            a = a_gd.copy()
            print(f"  Phase 2 done: obj={best_score:.8f}")
        else:
            print(f"  Phase 2 done: obj={obj_gd:.8f} (no improvement)")

    # === Phase 3: Aggressive multi-stage refinement ===
    print(f"\n=== Phase 3: Aggressive Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p3_budget = min(150.0, remaining * 0.2)
        a_ref = aggressive_refine(a, time_budget=p3_budget, verbose=True)
        obj_ref = compute_obj(a_ref)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            a = a_ref.copy()
            print(f"  Phase 3 done: obj={best_score:.8f}")
        else:
            print(f"  Phase 3 done: obj={obj_ref:.8f} (no improvement)")

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_best = len(best_seq)

    # Generate candidate n values - wider range than before
    n_candidates = set()
    for delta in range(-300, 301, 25):
        n_try = n_best + delta
        if 300 <= n_try <= 2000:
            n_candidates.add(n_try)
    # Add some round numbers
    for nr in range(400, 1601, 50):
        n_candidates.add(nr)

    n_candidates = sorted(n_candidates)

    # Prioritize n values close to current best
    n_candidates = sorted(n_candidates, key=lambda x: abs(x - n_best))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Skip if already tried this n with good results
        if n_try == n_best:
            continue

        # Budget per n: smaller for more distant n values
        dist = abs(n_try - n_best)
        if dist < 50:
            alloc = min(remaining * 0.06, 25.0)
        elif dist < 150:
            alloc = min(remaining * 0.04, 18.0)
        else:
            alloc = min(remaining * 0.03, 12.0)

        a_opt, obj_try = try_n_value(best_seq, n_try, alloc)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        elif n_try % 100 == 0:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: Diversification ===
    print(f"\n=== Phase 5: Diversification ===")
    remaining = deadline - time.monotonic()
    div_budget = min(remaining * 0.3, 200.0)
    div_start = time.monotonic()
    n_div_iters = 15

    for div_i in range(n_div_iters):
        if time.monotonic() - div_start > div_budget:
            break
        remaining_total = deadline - time.monotonic()
        if remaining_total < 20:
            break

        scale = 0.01 + 0.005 * div_i
        a_pert = structured_perturb(best_seq, rng, scale=scale)

        # Optimize perturbed version
        per_alloc = min(remaining_total * 0.12, 30.0)
        a_opt = smooth_max_gradient_descent(a_pert, time_budget=per_alloc, verbose=False)
        obj_try = compute_obj(a_opt)

        # If close, try LP
        remaining_total = deadline - time.monotonic()
        if remaining_total > 15 and obj_try < best_score + 0.003:
            if len(a_opt) <= 1000:
                a_opt = lp_alternating(a_opt, time_budget=min(20.0, remaining_total * 0.3))
                obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** Div {div_i} (scale={scale:.4f}): obj={best_score:.8f} **")
        else:
            if div_i % 3 == 0:
                print(f"  Div {div_i} (scale={scale:.4f}): obj={obj_try:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()
        n_final = len(a_final)

        # LP polish
        if remaining > 50 and n_final <= 1100:
            alloc = min(50.0, remaining * 0.3)
            a_lp = lp_alternating(a_final, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_final = a_lp.copy()
                print(f"  Final LP: obj={best_score:.8f}")

        # GD polish with very low temperature
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(40.0, remaining * 0.3)
            a_gd = smooth_max_gradient_descent(a_final, time_budget=alloc, verbose=False)
            obj_gd = compute_obj(a_gd)
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                a_final = a_gd.copy()
                print(f"  Final GD: obj={best_score:.8f}")

        # One more LP if time
        remaining = deadline - time.monotonic()
        if remaining > 30 and n_final <= 1000:
            alloc = min(30.0, remaining * 0.4)
            a_lp = lp_alternating(a_final, time_budget=alloc, verbose=False)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP2: obj={best_score:.8f}")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
