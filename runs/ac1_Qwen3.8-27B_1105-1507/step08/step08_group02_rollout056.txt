```python
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):

    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_A_sparse(a):
    """Build sparse constraint matrix A for the LP.
    A[k, j] = a[k-j] if 0 <= k-j < n else 0
    A has shape (2n-1, n)
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, min(j + n, 2 * n - 1)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp(a, a_prev_sol=None):

    n = len(a)
    a_norm = a / np.sum(a)
    
    A = build_A_sparse(a_norm)
    
    # Full constraint matrix for linprog: [A | -1] @ [g; M] <= 0
    # Using scipy's linprog with highs
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # A_ub @ [g; M] <= b_ub
    # [A | -ones] @ [g; M] <= 0
    import scipy.sparse as sp
    A_ub = sp.hstack([A, -sp.csr_matrix(np.ones((2 * n - 1, 1)))]).tocsc()
    b_ub = np.zeros(2 * n - 1)

    b_eq = np.array([1.0])

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:

    return None, None
```

```python
"""
Optimized step function search using LP alternating minimization with 
subgradient refinement, Polyak averaging, and multi-dimensional exploration.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix, hstack
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix_sparse(a):
    """Build sparse matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    
    idx = 0
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        length = k_end - k_start
        rows[idx:idx + length] = range(k_start, k_end)
        cols[idx:idx + length] = j
        vals[idx:idx + length] = a[:length]
        idx += length
    
    # Remove unused entries
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / np.sum(a)
    
    A = build_conv_matrix_sparse(a_norm)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A@g - M <= 0 for all rows
    # [A | -e] @ [g; M] <= 0
    neg_one_col = csr_matrix((-np.ones(2 * n - 1), (range(2 * n - 1), np.zeros(2 * n - 1, dtype=np.int32))), 
                              shape=(2 * n - 1, 1))
    A_ub = hstack([A, neg_one_col]).tocsc()
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        s = float(np.sum(g))
        if s > 1e-10:
            return g / s, float(res.x[n])
    return None, None


def subgradient_step(a, lr, topK=1):
    """Take a subgradient step to reduce max convolution."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:topK]
    
    # Subgradient: sum over active constraints
    subgrad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        # conv_k = sum_j a[j] * a[k-j]
        # d/d a[j] of conv_k = a[k-j] + a[j] (for the symmetric part)
        # Actually, d/d a[j] of sum_i a[i]*a[k-i] = a[k-j] + (a[j] if 2j=k else 0)
        # Wait, let me be more careful.
        # conv_k = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * a[k-i]
        # d conv_k / d a[j] = sum_{i: i=j or k-i=j} a[i]... 
        # Actually: d/d a[j] of sum_i a[i]*a[k-i]
        # = a[k-j] (from terms where i=j) + a[j] (from terms where k-i=j, i.e., i=k-j)
        # But if j = k-j (i.e., k = 2j), these are the same term and it's just a[j].
        # More carefully: sum_i a[i]*a[k-i] = ... + a[j]*a[k-j] + ... + a[k-j]*a[j] + ...
        # If j != k-j: both terms present, derivative = a[k-j] + a[j]
        # If j == k-j: one term a[j]^2, derivative = 2*a[j]
        # In both cases: derivative = a[k-j] + a[j] (since when j=k-j, this is 2*a[j])
        # Wait no: if j = k-j, then the term is a[j]*a[j] = a[j]^2, and d/d a[j] = 2*a[j].
        # And a[k-j] + a[j] = a[j] + a[j] = 2*a[j]. So the formula works!
        
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            subgrad[j_start:j_end + 1] += a[j_start:j_end + 1] + a[k - j_end:k - j_start + 1]
    
    # Project: a_new = a - lr * subgrad, then project to simplex
    a_new = a - lr * subgrad
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def parametric_init(n, mode, rng):
    """Generate parametric initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.ones(n)
    elif mode == 1:
        # Power law decay
        p = rng.uniform(0.5, 3.0)
        a = (1 - x + 0.1) ** p
    elif mode == 2:
        # Exponential decay
        lam = rng.uniform(0.5, 5.0)
        a = np.exp(-lam * x) + 0.05
    elif mode == 3:
        # Symmetric power law (high at edges)
        p = rng.uniform(1.0, 4.0)
        a = (1 - 2 * np.abs(x - 0.5) + 0.1) ** p
    elif mode == 4:
        # Linear tilt
        c = rng.uniform(0.5, 3.0)
        a = 1.0 + c * (x - 0.5)
        a = np.maximum(a, 0.01)
    elif mode == 5:
        # Quadratic
        c = rng.uniform(1.0, 5.0)
        a = 1.0 + c * (x - 0.5) ** 2
    elif mode == 6:
        # Sine-based
        f = rng.uniform(0.5, 3.0)
        a = np.ones(n) + 0.5 * np.sin(2 * np.pi * f * x)
        a = np.maximum(a, 0.01)
    elif mode == 7:
        # Two bumps
        c1, c2 = 0.25, 0.75
        w = rng.uniform(0.05, 0.2)
        a = np.exp(-((x - c1)**2) / (2*w**2)) + np.exp(-((x - c2)**2) / (2*w**2)) + 0.01
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a


def optimize_lp_subgrad(a, time_budget, verbose=True):
    """Combined LP alternating with subgradient refinement."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    iter_count = 0
    stall = 0
    patience = 50
    temp = 0.1
    lr = 1.0
    t_current = 0.3
    
    while time.monotonic() - start < time_budget:
        # Solve LP
        g, M_lp = solve_lp(a)
        if g is None:
            # Fallback to subgradient
            for _ in range(20):
                a = subgradient_step(a, lr)
                lr *= 0.99
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    stall = 0
            lr = 1.0
            continue
        
        iter_count += 1
        
        # Line search towards LP solution
        candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        best_t = t_current
        best_t_obj = np.inf
        best_t_a = a.copy()
        
        for t in candidates:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj = compute_obj(a_try)
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
                best_t_a = a_try.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a.copy()
            t_current = best_t
            stall = 0
            temp = max(0.01, temp * 0.99)
        else:
            # Try subgradient steps as intermediate
            a_init = a.copy()
            found = False
            for sg in range(10):
                a = subgradient_step(a, lr)
                obj = compute_obj(a)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    found = True
                    stall = 0
                    break
            if not found:
                a = a_init.copy()
                a = (1 - 0.3) * a + 0.3 * g
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall += 1
            
            if stall > patience:
                # Perturb and restart
                rng = np.random.default_rng(iter_count * 7919 + n)
                scale = temp
                a = best_a * (1.0 + scale * rng.normal(0, 1, n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                stall = 0
                temp = min(0.3, temp * 2.0)
                lr = 1.0
                t_current = 0.3
        
        # Decrease lr for subgradient
        lr = max(0.01, lr * 0.98)
        
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"      it{iter_count}: obj={best_obj:.8f}, M_lp={M_lp:.6f}, "
                  f"gap={compute_obj(a)-M_lp/n:.6f}, t={elapsed:.1f}s")
    
    return best_a
```

Wait, that's not right either. Let me be more careful.

`C_k = sum_i a[i] * a[k-i]` where the sum is over `i` such that `0 <= i < n` and `0 <= k-i < n`.

`dC_k / da[j] = sum_i [delta(i,j) * a[k-i] + a[i] * delta(k-i, j)]`

For the first term: `i = j`, so we need `0 <= j < n` (always true) and `0 <= k-j < n`.
If `0 <= k-j < n`: contribution is `a[k-j]`.

For the second term: `k-i = j`, so `i = k-j`. We need `0 <= k-j < n` and `0 <= j < n`.
If `0 <= k-j < n`: contribution is `a[k-j]`.

Wait, that gives `dC_k / da[j] = a[k-j] + a[k-j] = 2*a[k-j]` when `0 <= k-j < n`.

But wait, when `i = k-j` and `i = j` (i.e., `k = 2j`), the two terms are the same! In that case, the term in the sum is `a[j] * a[j] = a[j]^2`, and its derivative is `2*a[j]`. And `2*a[k-j] = 2*a[j]`. So the formula still works.

So: `dC_k / da[j] = 2 * a[k-j]` if `0 <= k-j < n`, else 0.

This means the subgradient of `max_k C_k` at lag `k*` is the vector `v` where `v[j] = 2*a[k*-j]` if `0 <= k*-j < n`, else 0.

In other words, `v = 2 * a_reversed_and_padded` where `a_reversed_and_padded[j] = a[k*-j]` if valid.

```python
def subgradient_step(a, lr, topK=1):
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:topK]
    
    subgrad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        # dC_k/da[j] = 2*a[k-j] if 0 <= k-j < n
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # a[k-j] for j in [j_start, j_end]
            # When j = j_start, k-j = k - j_start = k - max(0, k-n+1)
            # When j = j_end, k-j = k - j_end = k - min(n-1, k)
            subgrad[j_start:j_end+1] += 2.0 * a[k - j_end : k - j_start + 1][::-1]
    
    a_new = a - lr * subgrad
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new
```
