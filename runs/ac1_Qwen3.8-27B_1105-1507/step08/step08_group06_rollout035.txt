import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a (sum=1)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_response(a, solvers=None):
    """
    Solve LP: min M s.t. conv(a, g) <= M, sum(g)=1, g>=0.
    Returns (g, M, active_set) or (None, None, None).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        return None, None, None
    a = a / s
    A = compute_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    if solvers is None:
        solvers = [cp.CLARABEL, cp.PDLP, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    g_val = g_val / gs
                    M_val = float(np.max(A @ g_val))
                    conv_val = A @ g_val
                    active = np.where(conv_val >= M_val - 1e-7)[0]
                    return g_val, M_val, active
        except Exception:
            continue

    return None, None, None


def multi_peak_erosion(a, time_budget=100.0, top_k=5, eta_init=0.3, verbose=False):
    """
    Iteratively reduce peaks of conv(a,a) using multiplicative updates targeting
    the top-K peaks simultaneously.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_init
    stall = 0
    patience = 200

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        if max_val < 1e-15:
            break

        # Find top-K peaks
        if top_k >= len(conv):
            peak_indices = np.argsort(conv)[::-1]
        else:
            # Use partial sort for efficiency
            peak_indices = np.argsort(conv)[::-1][:top_k]

        # Compute combined attack vector
        # For peak at lag k, the contribution to a[j] is a[k-j] (if valid)
        attack = np.zeros(n)
        total_weight = 0.0
        for k in peak_indices:
            k = int(k)
            w = conv[k] / max_val
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    attack[j] += w * a[kj]
            total_weight += w

        if total_weight < 1e-15:
            break

        # Multiplicative update
        # a[j] *= (1 - eta * attack[j] / max_val)
        # But ensure we don't go negative
        factor = 1.0 - eta * attack / max_val
        factor = np.maximum(factor, 0.005)
        a_new = a * factor
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slowly decrease eta
            eta *= 0.9995
            if verbose and stall % 500 == 0:
                elapsed = time.monotonic() - start
                print(f"    MPE: obj={best_obj:.8f}, eta={eta:.4f}, t={elapsed:.1f}s")
        else:
            stall += 1
            if stall > patience:
                # Reset eta higher and perturb
                eta = min(eta * 2, 1.0)
                rng = np.random.default_rng(int(time.monotonic() * 1e9) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-12:
                    a = a / s
                stall = 0
            elif stall % 50 == 0:
                eta = max(eta * 0.9, 0.01)

    return best_a


def lp_overrelaxation(a, time_budget=200.0, verbose=False, omega=1.0, momentum=0.3):
    """
    LP-based alternating with over-relaxation and momentum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum_vec = np.zeros(n)
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp, active = solve_lp_response(a)
        if g_lp is None:
            break

        # LP gives a lower bound
        lp_bound = 2.0 * n * M_lp / 1.0  # since sum(g)=1
        if verbose and stall % 10 == 0:
            elapsed = time.monotonic() - start
            gap = best_obj - lp_bound
            print(f"    LP-OR: obj={best_obj:.8f}, LP_bound={lp_bound:.8f}, "
                  f"gap={gap:.6f}, t={elapsed:.1f}s")

        # Update with over-relaxation
        direction = g_lp - a
        momentum_vec = momentum * momentum_vec + (1 - momentum) * direction
        a_new = a + omega * momentum_vec
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 20 == 0:
                # Reduce over-relaxation
                omega = max(omega * 0.8, 0.3)
                momentum *= 0.5
            if stall > 100:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1e9) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-12:
                    a = a / s
                omega = 1.0
                momentum = 0.3
                stall = 0

    return best_a


def nesterov_subgradient(a, time_budget=100.0, verbose=False):
    """
    Nesterov-accelerated subgradient method targeting the top peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    x = a.copy()
    momentum = np.zeros(n)
    lr = 0.08
    beta = 0.75
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        conv = compute_conv(x)
        max_val = float(np.max(conv))
        if max_val < 1e-15:
            break

        # Find top peaks (within 15% of max)
        threshold = max_val * 0.85
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        if len(active) > 20:
            active = np.argsort(conv)[::-1][:20]

        # Compute subgradient (average over active peaks)
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_max >= i_min:
                indices = np.arange(i_min, i_max + 1)
                grad[indices] += a[k - indices]
                # Also the symmetric part: d/dx[i] of x[k-i]*x[i] includes x[i]*x[k-i]
                # conv[k] = sum_i a[i]*a[k-i], d/da[j] = a[k-j] + a[j-k] (if both valid)
                # But for our purpose, the subgradient of max_k conv[k] w.r.t. a[j] is:
                # a[k*-j] for the active peak k* (one-sided gradient at a)
        grad /= len(active)

        # Center the gradient (maintain sum constraint)
        grad_centered = grad - np.mean(grad)

        # Nesterov momentum
        momentum = beta * momentum - lr * grad_centered
        y_new = y + momentum
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-12:
            y_new = y.copy()
            s_new = 1.0
        y_new = y_new / s_new

        # Nesterov lookahead
        t_k = 2.0 / (np.sqrt(it + 1) + 2.0)
        x = y_new + (1 - t_k) * (y_new - y)
        x = np.maximum(x, 0)
        sx = float(np.sum(x))
        if sx > 1e-12:
            x = x / sx

        y = y_new
        obj = compute_obj(y)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = y.copy()

        # Adaptive learning rate
        if it % 300 == 299:
            lr = max(lr * 0.92, 0.002)

    return best_a


def smooth_peak_erosion(a, time_budget=60.0, verbose=False):
    """
    Smooth version of peak erosion: uses a soft target (all peaks above median)
    to avoid creating new peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.1

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(np.max(conv))
        med_val = float(np.median(conv))

        # Target all peaks above the median
        above = conv > med_val
        if not np.any(above):
            break

        attack = np.zeros(n)
        for k in range(len(conv)):
            if not above[k]:
                continue
            w = (conv[k] - med_val) / (max_val - med_val + 1e-15)
            k_int = int(k)
            for j in range(n):
                kj = k_int - j
                if 0 <= kj < n:
                    attack[j] += w * a[kj]

        # Attenuate
        factor = 1.0 - eta * attack / (max_val + 1e-15)
        factor = np.maximum(factor, 0.01)
        a_new = a * factor
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            eta = max(eta * 0.999, 0.005)
        else:
            eta = max(eta * 0.8, 0.001)
            if eta < 0.002:
                break

    return best_a


def generate_diverse_init(n, kind, rng):
    """Generate diverse initialization."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x)) + 0.01
        return a / np.sum(a)
    elif kind == 'sine_sq':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2 + 0.05
        return a / np.sum(a)
    elif kind == 'gaussian':
        center = n / 2.0
        x = np.arange(n)
        sigma = n / 3.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2) + 0.01
        return a / np.sum(a)
    elif kind == 'two_arc':
        # High in two separated regions
        a = np.ones(n) * 0.1
        a[n//4 - n//10:n//4 + n//10] = 1.0
        a[3*n//4 - n//10:3*n//4 + n//10] = 1.0
        # Smooth
        kernel_size = max(3, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'random_smooth':
        a = rng.random(n)
        kernel_size = max(3, n // 30)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same') + 0.01
        return a / np.sum(a)
    elif kind == 'bipolar_map':
        # Map a random ±1 sequence to non-negative
        s = rng.choice([-1, 1], size=n).astype(float)
        a = (1 + s) / 2 + 0.01
        kernel_size = max(3, n // 40)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        return a / np.sum(a)
    elif kind == 'raised_cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.02
        return a / np.sum(a)
    elif kind == 'plateau_edges':
        a = np.ones(n)
        edge = n // 8
        a[:edge] = 0.4
        a[-edge:] = 0.4
        return a / np.sum(a)
    else:
        return np.ones(n) / n


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        best_seq = a_new.copy()
        print(f"  SMOOTH: obj={best_score:.8f} **")
    else:
        print(f"  SMOOTH: obj={obj:.8f}")

    # === Phase 5: Diverse Initializations + Quick Optimize (300s) ===
    print(f"\n=== Phase 5: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(300.0, remaining * 0.25)
    n_init = len(best_seq)

    init_kinds = ['two_arc', 'bipolar_map', 'raised_cosine', 'random_smooth',
                  'gaussian', 'plateau_edges', 'sine_sq']

    per_init = max(10.0, p5_budget / (len(init_kinds) + 2))

    for kind in init_kinds:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_init = generate_diverse_init(n_init, kind, rng)
        obj_init = compute_obj(a_init)

        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Init '{kind}': obj={obj_init:.8f} **")

        # Quick optimization from this init
        alloc_mpe = min(15.0, remaining * 0.3)
        alloc_lp = min(15.0, remaining * 0.3)
        if remaining > 40:
            a_opt = multi_peak_erosion(a_init, time_budget=alloc_mpe, top_k=5, eta_init=0.3)
            a_opt = lp_overrelaxation(a_opt, time_budget=alloc_lp, omega=1.0, momentum=0.5)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Opt '{kind}': obj={obj_opt:.8f} **")
            else:
                print(f"  Opt '{kind}': obj={obj_opt:.8f} (init was {obj_init:.8f})")

    # Some random inits
    for ri in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        a_init = generate_diverse_init(n_init, 'random_smooth', rng)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
            print(f"  Random {ri}: obj={obj_init:.8f} **")
        else:
            alloc = min(20.0, remaining * 0.3)
            a_opt = multi_peak_erosion(a_init, time_budget=alloc, top_k=5, eta_init=0.3)
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Random {ri} opt: obj={obj_opt:.8f} **")

    # === Phase 6: Combined refinement on best (200s) ===
    print(f"\n=== Phase 6: Combined Refinement ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(200.0, remaining * 0.18)

    # Alternating: MPE -> LP -> Nesterov -> MPE -> ...
    a_curr = best_seq.copy()
    for round_i in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        alloc = min(p6_budget / 6, remaining * 0.4)

        method = round_i % 3
        if method == 0:
            a_curr = multi_peak_erosion(a_curr, time_budget=alloc, top_k=5, eta_init=0.2)
        elif method == 1:
            a_curr = lp_overrelaxation(a_curr, time_budget=alloc, omega=1.0, momentum=0.4)
        else:
            a_curr = nesterov_subgradient(a_curr, time_budget=alloc)

        obj = compute_obj(a_curr)
        if obj < best_score:
            best_score = obj
            best_seq = a_curr.copy()
            print(f"  R6-{method}: obj={best_score:.8f} **")

    # === Phase 7: Length Sweep (150s) ===
    print(f"\n=== Phase 7: Length Sweep ===")
    remaining = deadline - time.monotonic()
    p7_budget = min(150.0, remaining * 0.15)
    n_best = len(best_seq)

    n_options = sorted(set([
        max(100, n_best - 50),
        max(100, n_best - 25),
        n_best,
        n_best + 25,
        n_best + 50,
        n_best + 100,
    ]))
    n_options = [n for n in n_options if 100 <= n <= 3000]

    per_n = max(10.0, p7_budget / len(n_options))

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        if n_try == n_best:
            a_try = best_seq.copy()
        else:
            a_try = resample(best_seq, n_try)

        alloc_mpe = min(per_n * 0.5, remaining * 0.4)
        alloc_sg = min(per_n * 0.3, remaining * 0.3)

        a_opt = multi_peak_erosion(a_try, time_budget=alloc_mpe, top_k=5, eta_init=0.3)
        a_opt = nesterov_subgradient(a_opt, time_budget=alloc_sg)
        obj = compute_obj(a_opt)

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            n_best = n_try
            print(f"  n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj:.8f}")

    # === Phase 8: Final Squeeze (remaining time) ===
    print(f"\n=== Phase 8: Final Squeeze ===")
    remaining = deadline - time.monotonic()
    a_final = best_seq.copy()

    while remaining > 10:
        # Try a short burst of different methods
        alloc = min(15.0, remaining * 0.3)

        # MPE with small eta for fine tuning
        a_new = multi_peak_erosion(a_final, time_budget=alloc, top_k=3, eta_init=0.1)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_final = a_new.copy()
            print(f"  Final MPE: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        # Quick LP
        alloc = min(15.0, remaining * 0.3)
        a_new = lp_overrelaxation(a_final, time_budget=alloc, omega=0.8, momentum=0.3)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_final = a_new.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        # Nesterov
        alloc = min(10.0, remaining * 0.3)
        a_new = nesterov_subgradient(a_final, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_final = a_new.copy()
            print(f"  Final NS: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
