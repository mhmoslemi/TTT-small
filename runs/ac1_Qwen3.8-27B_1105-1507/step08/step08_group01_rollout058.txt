```python
"""
Multi-method optimizer for minimum peak autocorrelation step functions.
Combines SCA (Successive Convex Approximation) with non-uniform RHS,
Nesterov-accelerated smooth-max gradient, pair coordinate descent,
and multi-start diversification with Fourier-domain crossover.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]


def compute_conv_cross(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n+m-1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_sca_lp(a, active_indices=None):
    """
    SCA LP: min M s.t. 2*(a*g)[k] - (a*a)[k] <= M for all k (or active k),
    sum(g) = 1, g >= 0.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s

    conv_aa = compute_conv(a)
    A = build_conv_matrix(a)

    if active_indices is not None:
        A_use = A[active_indices, :]
        rhs = conv_aa[active_indices]
        n_cons = len(active_indices)
    else:
        A_use = A
        rhs = conv_aa
        n_cons = 2*n - 1

    # LP: min M
    # s.t. 2*A_use @ g - M <= rhs  =>  2*A_use @ g - M - rhs <= 0
    #      sum(g) = 1
    #      g >= 0, M >= 0

    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    A_ub = np.zeros((n_cons, n + 1))
    A_ub[:, :n] = 2.0 * A_use
    A_ub[:, n] = -1.0
    b_ub = rhs.copy()

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s_g = float(np.sum(g_val))
        if s_g > 1e-10:
            g_norm = g_val / s_g
            return g_norm, float(res.x[n])
    return None, None

    """

    else:
        A_use = A
        n_cons = 2*n - 1

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((n_cons, n + 1))
    A_ub[:, :n] = A_use
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_cons)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s_g = float(np.sum(g_val))
        if s_g > 1e-10:
            g_norm = g_val / s_g
            return g_norm, float(res.x[n])
    return None, None


def convex_line_search(a, g, n_evals=15):
    """Find optimal t for a_new = (1-t)*a + t*g."""
    n = len(a)
    a_norm = a / np.sum(a)
    g_norm = g / np.sum(g)

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1-t)**2 * conv_aa + 2*t*(1-t) * conv_ag + t**2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj_at_t(m1) < obj_at_t(m2):
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm.copy()
    a_new = a_new / s
    return t_opt, a_new


def sca_alternating(a, time_budget=200.0, verbose=False, use_sca=True):
    """SCA or standard LP alternating with line search."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        # Choose active set
        conv_cur = compute_conv(a)
        max_conv = np.max(conv_cur)
        # Use indices where conv is in top 20% of range or above 95% of max
        threshold = max_conv * 0.90
        active_set = np.where(conv_cur >= threshold)[0]

        # Solve LP
        if use_sca:
            g_lp, M_lp = solve_sca_lp(a, active_set)
            if g_lp is None:
                g_lp, M_lp = solve_sca_lp(a)
        else:
            g_lp, M_lp = solve_lp_standard(a, active_set)
            if g_lp is None:
                g_lp, M_lp = solve_lp_standard(a)

        if g_lp is None:
            break

        # Line search
        t_opt, a_new = convex_line_search(a, g_lp, n_evals=12)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 20 == 0:
                print(f"    SCA it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, time={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 15:
                # Perturb and try smaller step
                a_try = (1 - 0.03) * a + 0.03 * g_lp
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st > 1e-10:
                    a_try = a_try / st
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stall = 0
                    else:
                        rng = np.random.default_rng(it * 31 + n)
                        a_pert = best_a * np.exp(rng.normal(0, 0.003, n))
                        a_pert = np.maximum(a_pert, 0)
                        sp = float(np.sum(a_pert))
                        if sp > 1e-10:
                            a = a_pert / sp
                        stall = 0
                        if stall > 100:
                            break

    return best_a


def nesterov_smooth_max(a, time_budget=300.0, verbose=False, T_init=0.0005, T_min=0.00001):
    """
    Nesterov-accelerated gradient descent on smooth max approximation.
    Uses the softmax of the convolution peaks to get a smooth gradient.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.002
    momentum = 0.9
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    stall = 0
    best_lr = lr

    for it in range(20000):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov step
        a_prev = y.copy()
        y = simplex_project(y + (it / (it + 3.0)) * (y - a_prev))

        # Compute gradient at y
        conv_y = compute_conv(y)
        max_c = np.max(conv_y)
        shifted = (conv_y - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        if total < 1e-300:
            weights = np.zeros_like(conv_y)
            weights[np.argmax(conv_y)] = 1.0
        else:
            weights = exp_vals / total

        # Gradient of max_k conv(y)[k] w.r.t. y[i] = sum_k weights[k] * (conv gradient)
        # d/dy[i] conv(y)[k] = y[k-i] + y[i-k] (for valid indices)
        # Actually: conv(y)[k] = sum_j y[j] * y[k-j]
        # d/dy[i] conv(y)[k] = y[k-i] + y[i-k] (when both are in range)
        # = sum over the "partner" indices

        # More efficiently: grad[i] = sum_k weights[k] * (y[k-i] + y[i-k])
        # This is a weighted "autocorrelation" of y with weights
        # grad = conv(weights, y) [valid part] + conv(weights, y reversed) [valid part]
        # Actually: d/dy[i] (y*y)[k] = y[k-i] (if 0<=k-i<n) + y[i-k] (if 0<=i-k<n)
        # grad[i] = sum_k weights[k] * y[k-i] * [0<=k-i<n] + sum_k weights[k] * y[i-k] * [0<=i-k<n]
        # = (cross-correlation of weights with y) at lag i + (cross-correlation) at lag -i

        # Let's compute it directly using FFT
        # First term: sum_k weights[k] * y[k-i] for k such that 0<=k-i<n, i.e., i<=k<=i+n-1
        # This is the cross-correlation: it's the "valid" part of conv(weights, y) shifted
        # Actually, let's just use the convolution matrix approach for clarity (n is ~800, so O(n^2) is fine)
        # Or use FFT:
        # term1[i] = sum_k weights[k] * y[k-i] where 0 <= k < 2n-1 and 0 <= k-i < n
        # = sum_{j=0}^{n-1} weights[i+j] * y[j] (with bounds)
        # This is the valid cross-correlation of weights (length 2n-1) with y (length n)

        # Simpler approach: just compute using the relation
        # grad = A^T @ weights where A is the "gradient matrix"
        # A[k, i] = y[k-i] if 0<=k-i<n else y[i-k] if 0<=i-k<n else 0
        # Actually A[k,i] = y[k-i] + y[i-k] (circular-like)

        # Let me just do it with a loop for correctness first, then optimize
        # Actually for n=800, even O(n * (2n)) = O(2n^2) = 1.28M operations is fast

        grad = np.zeros(n)
        for i in range(n):
            # d/dy[i] conv(y)[k] = y[k-i] (if valid) + y[i-k] (if valid)
            # sum_k weights[k] * y[k-i] for valid k: k ranges from i to i+n-1 (but k < 2n-1)
            k_start = i
            k_end = min(2*n-1, i+n)
            if k_end > k_start:
                grad[i] += np.dot(weights[k_start:k_end], y[:k_end-k_start])
            # sum_k weights[k] * y[i-k] for valid k: i-k in [0,n-1], so k in [i-n+1, i]
            k_start2 = max(0, i-n+1)
            k_end2 = i + 1
            if k_end2 > k_start2:
                grad[i] += np.dot(weights[k_start2:k_end2], y[i-k_start2:i+1][::-1])

        # Normalize gradient (subtract mean to maintain sum=1 constraint)
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(y + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            if verbose and it % 500 == 0:
                print(f"    NSM it{it}: obj={best_obj:.8f}, T={T:.6f}, lr={lr:.6f}, time={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                best_a_calc = compute_obj(a)
                y = best_a.copy()
                v = np.zeros(n)
                stall = 0
                if lr < 1e-7:
                    break

        if it > 50:
            T = max(T_min, T_init * (0.999 ** it))

    return best_a
```

```python
    p2 = p2[:n]  # we need p2[i] for i=0..n-1
```

```python
"""
Multi-method optimizer for minimum peak autocorrelation step functions.
Uses SCA (Successive Convex Approximation) with non-uniform RHS LP,
FFT-accelerated Nesterov smooth-max gradient descent, pair coordinate
descent, and multi-start diversification with Fourier-domain operations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_sca_lp(a, active_indices=None, use_full_fallback=True):
    """
    SCA LP: min M s.t. 2*(a*g)[k] - (a*a)[k] <= M for active k,
    sum(g) = 1, g >= 0.
    The non-uniform RHS makes this different from standard LP alternating.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s

    conv_aa = compute_conv(a)
    A = build_conv_matrix(a)

    if active_indices is not None and len(active_indices) > 0:
        A_use = A[active_indices, :]
        rhs = conv_aa[active_indices]
        n_cons = len(active_indices)
    else:
        A_use = A
        rhs = conv_aa
        n_cons = 2 * n - 1

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((n_cons, n + 1))
    A_ub[:, :n] = 2.0 * A_use
    A_ub[:, n] = -1.0
    b_ub = rhs.copy()

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s_g = float(np.sum(g_val))
        if s_g > 1e-10:
            g_norm = g_val / s_g
            return g_norm, float(res.x[n])

    if use_full_fallback and active_indices is not None:
        return solve_sca_lp(a, None, False)
    return None, None


def solve_lp_standard(a, active_indices=None, use_full_fallback=True):
    """Standard LP: min M s.t. (a*g)[k] <= M for active k, sum(g) = 1, g >= 0."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))

    A_ub = np.zeros((n_cons, n + 1))
    A_ub[:, :n] = A_use
    A_ub[:, n] = -1.0
    b_ub = np.zeros(n_cons)

    A_eq = np.zeros((1, n + 1))

    b_eq = np.array([1.0])

    if use_full_fallback and active_indices is not None:

def convex_line_search(a, g, n_evals=15):
    """Find optimal t in [0,1] for a_new = (1-t)*a + t*g using ternary search."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    g_norm = g / float(np.sum(g))

    conv_aa = compute_conv(a_norm)
    conv_ag = compute_conv_cross(a_norm, g_norm)
    conv_gg = compute_conv(g_norm)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj_at_t(m1) < obj_at_t(m2):
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return 0.0, a_norm.copy()
    a_new = a_new / s
    return t_opt, a_new


def fft_smooth_grad(y, weights):
    """
    Compute gradient of sum_k weights[k] * conv(y,y)[k] w.r.t. y using FFT.
    grad[i] = sum_k weights[k] * (y[k-i] + y[i-k]) for valid indices.
    """
    n = len(y)
    size = 1
    while size < 4 * n:
        size <<= 1

    Wf = np.fft.rfft(weights, size)
    Yf = np.fft.rfft(y, size)

    # p1[i] = sum_j w[i+j] * y[j] = IFFT(W * conj(Y))[i]
    p1_full = np.fft.irfft(Wf * np.conj(Yf), size)
    p1 = p1_full[:n]

    # p2[i] = sum_j w[j] * y[i-j] = IFFT(W * Y)[i] (full conv, take first n)
    p2_full = np.fft.irfft(Wf * Yf, size)
    p2 = p2_full[:n]

    return p1 + p2


def nesterov_smooth_max(a, time_budget=300.0, verbose=False, T_init=0.0004, T_min=0.000005):
    """
    Nesterov-accelerated gradient descent on smooth max with FFT-based gradients.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_init
    lr = 0.003
    momentum = 0.92
    v = np.zeros(n)
    y = a.copy()
    stall = 0
    patience = 80

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov lookahead
        beta = (1.0 + np.sqrt(1.0 + 4.0 * ((it + 1) / 2.0) ** 2)) / 2.0
        a_prev = y.copy()
        y_new = y + (1.0 - 1.0 / beta) * (y - a_prev)
        y = simplex_project(y_new)

        # Compute smooth gradient
        conv_y = compute_conv(y)
        max_c = np.max(conv_y)
        shifted = (conv_y - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        if total < 1e-300:
            weights = np.zeros_like(conv_y)
            weights[np.argmax(conv_y)] = 1.0
        else:
            weights = exp_vals / total

        grad = fft_smooth_grad(y, weights)
        grad_centered = grad - np.mean(grad)

        v = momentum * v - lr * grad_centered
        a_new = simplex_project(y + v)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            if verbose and it % 1000 == 0:
                print(f"    NSM it{it}: obj={best_obj:.8f}, T={T:.2e}, lr={lr:.6f}, time={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > patience:
                lr *= 0.85
                y = best_a.copy()
                v = np.zeros(n)
                stall = 0
                if lr < 1e-8:
                    break

        # Temperature annealing
        if it > 100:
            T = max(T_min, T_init * (0.9995 ** it))

    return best_a

def pair_coordinate_descent(a, time_budget=200.0, verbose=False):
    """
    Coordinate descent that updates pairs of coordinates simultaneously.
    This provides more flexibility than single-coordinate updates.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    delta = 1e-4
    pass_count = 0

    while time.monotonic() - start < time_budget:
        pass_count += 1
        improved = False
        rng = np.random.default_rng(pass_count * 1000 + n)
        order_i = rng.permutation(n)
        order_j = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - start > time_budget:
                break

            i = order_i[idx]
            j = order_j[idx]
            if i == j:
                j = (i + 1) % n

            # Try increasing i, decreasing j (and vice versa)
            for di, dj in [(delta, -delta), (-delta, delta),
                           (delta * 0.3, -delta * 0.3), (-delta * 0.3, delta * 0.3)]:
                a_try = a.copy()
                a_try[i] += di
                a_try[j] += dj
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            # Also try single coordinate if pair didn't help
            if not improved:
                for di in [delta, delta * 0.1]:
                    a_try = a.copy()
                    a_try[i] += di
                    s_try = float(np.sum(a_try))
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-15:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break

        if verbose and pass_count % 3 == 0:
            print(f"    PairCD pass {pass_count}: obj={best_obj:.8f}, delta={delta:.2e}, time={time.monotonic()-start:.1f}s")

        if not improved:
            delta *= 0.6
            if delta < 1e-8:
                break

    return best_a


def fine_coordinate_descent(a, time_budget=100.0, verbose=False):
    """
    Very fine single-coordinate descent with binary search for optimal step.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)

    if s < 1e-10:
        return a

    for sweep in range(50):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        rng = np.random.default_rng(sweep * 7 + n)
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            # Binary search for best perturbation of a[i]
            current_val = a[i]
            lo = max(0.0, current_val - 5e-3)
            hi = current_val + 5e-3

            # Check if current is already optimal by checking neighbors
            for delta in [1e-5, 5e-5, 1e-4, 5e-4]:
                for sign in [1, -1]:
                    new_val = current_val + sign * delta
                    if new_val < 0:
                        continue
                    a_try = a.copy()
                    a_try[i] = new_val
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-16:
                        best_obj = obj_try
                        best_a = a_try.copy()

                        current_val = new_val
                        improved = True

        if verbose and sweep % 5 == 0:
            print(f"    FCD sweep {sweep}: obj={best_obj:.8f}, time={time.monotonic()-start:.1f}s")
        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def fourier_crossover(a1, a2, alpha):
    """
    Crossover in Fourier domain: mix amplitude spectra.
    This creates diverse candidates.
    """
    n = len(a1)
    F1 = rfft(a1)
    F2 = rfft(a2)
    # Mix amplitudes
    new_mag = alpha * np.abs(F1) + (1 - alpha) * np.abs(F2)
    new_phase = np.angle(F1)  # keep phase from a1
    new_F = new_mag * np.exp(1j * new_phase)
    result = irfft(new_F, 2 * n)[:n]
    result = np.maximum(result, 0)
    s = float(np.sum(result))
    if s < 1e-10:
        return a1.copy()
    return result / s

                                     T_init=0.0001, T_min=0.000001)
        obj_nsm2 = compute_obj(a_nsm2)
        if obj_nsm2 < best_score:
            best_score = obj_nsm2
            best_seq = a_nsm2.copy()
            a = a_nsm2.copy()
            print(f"  NSM2 done: obj={best_score:.8f}")
        else:
            print(f"  NSM2 done: obj={obj_nsm2:.8f} (no improvement)")

    # === Phase 6: Pair Coordinate Descent ===
    print(f"\n=== Phase 6: Pair Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        phase6_budget = min(120.0, remaining * 0.15)
        a_pcd = pair_coordinate_descent(a, time_budget=phase6_budget, verbose=True)
        obj_pcd = compute_obj(a_pcd)
        if obj_pcd < best_score:
            best_score = obj_pcd
            best_seq = a_pcd.copy()
            a = a_pcd.copy()
            print(f"  PairCD done: obj={best_score:.8f}")
        else:
            print(f"  PairCD done: obj={obj_pcd:.8f} (no improvement)")

    # === Phase 7: Multi-n Exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration ===")
    n_best = len(best_seq)
    n_options = sorted(set([n_best - 50, n_best - 25, n_best + 25, n_best + 50,
                            750, 800, 850, 900, 950, 1000]))
    n_options = [x for x in n_options if 500 <= x <= 1200 and x != n_best]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick SCA
        alloc = min(remaining * 0.08, 30.0)
        a_opt = sca_alternating(a_try, time_budget=alloc, method='sca')
        obj_try = compute_obj(a_opt)

        # Quick NSM if promising
        remaining = deadline - time.monotonic()
        if remaining > 20 and obj_try < best_score + 0.01:
            alloc2 = min(remaining * 0.08, 25.0)
            a_opt = nesterov_smooth_max(a_opt, time_budget=alloc2, T_init=0.0002, T_min=0.00001)
            obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 8: Fourier Crossover Diversification ===
    print(f"\n=== Phase 8: Fourier Crossover ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        # Create diverse candidates via Fourier crossover with itself (perturbed)
        for fork in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            rng_fork = np.random.default_rng(seed + fork * 1000)
            a_pert = best_seq * np.exp(rng_fork.normal(0, 0.005, n))
            a_pert = np.maximum(a_pert, 0)
            sp = float(np.sum(a_pert))
            if sp < 1e-10:
                continue
            a_pert = a_pert / sp
            a_cross = fourier_crossover(best_seq, a_pert, 0.5)

            # Quick optimize
            alloc = min(remaining * 0.06, 25.0)
            a_opt = sca_alternating(a_cross, time_budget=alloc, method='sca')
            obj_try = compute_obj(a_opt)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_opt.copy()
                a = a_opt.copy()
                print(f"  Fork {fork}: obj={best_score:.8f} **")
            else:
                print(f"  Fork {fork}: obj={obj_try:.8f}")

    # === Phase 9: Final Intensive Polish ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()

    # SCA final
    if remaining > 30:
        alloc = min(60.0, remaining * 0.25)
        a_final = sca_alternating(best_seq, time_budget=alloc, method='sca', verbose=False)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final SCA: obj={best_score:.8f}")

    # NSM final (very cool)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(60.0, remaining * 0.25)
        a_final = nesterov_smooth_max(best_seq, time_budget=alloc, T_init=0.00005, T_min=0.000001, verbose=False)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final NSM: obj={best_score:.8f}")

    # Pair CD final
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(50.0, remaining * 0.25)
        a_final = pair_coordinate_descent(best_seq, time_budget=alloc, verbose=False)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final PairCD: obj={best_score:.8f}")

    # Fine CD final
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(40.0, remaining * 0.3)
        a_final = fine_coordinate_descent(best_seq, time_budget=alloc, verbose=False)
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()
            print(f"  Final FCD: obj={best_score:.8f}")

    # === Final Validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
