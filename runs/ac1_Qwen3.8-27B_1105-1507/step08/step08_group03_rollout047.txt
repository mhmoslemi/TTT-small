"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses dual-guided LP alternating with geometric interpolation,
adaptive step sizes, multi-n exploration, and deflation-based
local minimum escape.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize as sp_minimize

try:
    import cvxpy as cp
    HAVE_CVXPY = True
    CVXPY_SOLVERS = []
    for _s in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            CVXPY_SOLVERS.append(_s)
        except Exception:
            pass
except Exception:
    HAVE_CVXPY = False
    CVXPY_SOLVERS = []


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, time_limit=5.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M, dual_y) or (None, None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in CVXPY_SOLVERS:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        g_val = g_val / s
                        return g_val, float(M.value), None
            except Exception:
                continue

    # Fallback to scipy linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_val = g_val / s
            return g_val, float(res.x[n]), None
    return None, None, None


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_at_peak(a, k):
    """Gradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def lp_alternating(a, time_budget=100.0, use_geometric=False, verbose=False):
    """
    LP alternating with adaptive step size and optional geometric interpolation.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    best_t = 0.3
    rng = np.random.default_rng(12345)
    it = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        g_lp, M_lp, _ = solve_lp(a)
        if g_lp is None:
            break

        # Line search
        ts = [best_t, best_t * 0.5, best_t * 1.5, 0.1, 0.3, 0.5, 0.7, 0.9, 1.0]
        ts = sorted(set(t for t in ts if 0.01 <= t <= 1.0))

        best_t_obj = np.inf
        best_t_val = best_t
        best_a_local = best_a.copy()

        for t in ts:
            if use_geometric:
                a_try = np.exp((1-t)*np.log(np.maximum(a, 1e-15)) + t*np.log(np.maximum(g_lp, 1e-15)))
            else:
                a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        it += 1
        if best_t_obj < best_obj - 1e-13:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            best_t = best_t_val
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Escape: add perturbation
                noise_scale = min(0.3, 0.05 + 0.03 * (stall // 10))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                best_t = 0.3

    return best_a


def nesterov_subgradient(a, time_budget=60.0, K=5):
    """Nesterov-accelerated subgradient method on simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 60

    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        spread = (peak_vals[0] - peak_vals[-1]) / (max_val + 1e-15)
        temp = 3.0 / (1.0 + spread * 5.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / (max_val + 1e-15))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_at_peak(y, int(k))

        lr = lr0 / t_k
        a_new = project_simplex(y - lr * g)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        stall += 1
        if stall > patience:
            y = best_a.copy()
            a_new = best_a.copy()
            a_prev = best_a.copy()
            y_new = best_a.copy()
            t_k = 1.0
            stall = 0
            patience = int(patience * 1.5)

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new

    return best_a


def smooth_optimize(a, time_budget=60.0, T=50.0):
    """Smooth optimization using log-sum-exp with SLSQP."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)

    def obj_and_grad(x):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = compute_conv(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        e = np.exp(z)
        sum_e = e.sum()
        obj_val = 2.0 * n * (max_c + np.log(sum_e) / T)
        w = e / sum_e
        g = np.zeros(n)
        for k in range(len(conv)):
            if w[k] > 1e-8:
                g += w[k] * grad_at_peak(x, int(k))
        g *= 2.0 * n
        return obj_val, g

    try:
        res = sp_minimize(obj_and_grad, a, method='L-BFGS-B',
                          bounds=[(1e-12, None)] * n,
                          options={'maxiter': 300, 'maxfun': 600, 'disp': False})
        a_opt = np.maximum(res.x, 0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
            obj_opt = compute_obj(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt.copy()
    except Exception:
        pass

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def coordinate_refine(a, time_budget=30.0):
    """Coordinate descent for final polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    improved = True
    while improved and time.monotonic() - start < time_budget:
        improved = False
        conv = compute_conv(a)
        max_c = np.max(conv)

        for i in range(n):
            if time.monotonic() - start >= time_budget:
                break
            # Try moving mass from i to other positions
            for delta_frac in [0.02, 0.05, 0.1]:
                delta = delta_frac * a[i]
                if delta < 1e-15:
                    continue
                # Try moving to position that doesn't contribute to the peak
                k_star = int(np.argmax(conv))
                # Find positions not contributing to peak
                for j in range(n):
                    if j == i:
                        continue
                    if (k_star - j < 0) or (k_star - j >= n):
                        # j doesn't contribute to peak at k_star
                        a[i] -= delta
                        a[j] += delta
                        new_conv = compute_conv(a)
                        new_obj = 2.0 * n * float(np.max(new_conv))
                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[i] += delta
                            a[j] -= delta
                        break  # only try one target

    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [max(1, n // 6)] * 2
        a = np.zeros(n)
        for c, w in zip(centers, widths):

        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = np.maximum(1.0 - np.abs(x), 0.01)

    elif mode == 4:

        a = 1.0 / (np.sqrt(1 - x**2 + 0.01))
        a = a / a.sum()

        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x**2 + (1-x)**2 + 0.05
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
