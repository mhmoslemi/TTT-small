```python
def solve_active_set_system(a, S):
    """
    Given current a and active set S (indices where conv(a,a) is at max),
    solve the linear system A_S @ z = M * 1, sum(z) = 1.
    
    Unknowns: z (n vars) and M (1 var)
    Equations: |S| + 1
    """
    n = len(a)
    A = build_conv_matrix(a)  # (2n-1) x n
    A_S = A[S, :]  # |S| x n
    
    # System: [A_S, -1; 1^T, 0] [z; M] = [0; 1]
    # Wait, A_S @ z - M * 1 = 0, sum(z) = 1
    # So: [A_S | -1] [z; M] = [0]  (|S| equations)
    #     [1^T |  0] [z; M] = [1]  (1 equation)
    
    sys_size = len(S) + 1
    mat = np.zeros((sys_size, n + 1))
    mat[:len(S), :n] = A_S
    mat[:len(S), n] = -1.0
    mat[len(S), :n] = 1.0
    rhs = np.zeros(sys_size)
    rhs[len(S)] = 1.0
    
    sol, residuals, rank, sv = np.linalg.lstsq(mat, rhs, rcond=None)
    z = np.maximum(sol[:n], 0)
    M = sol[n]
    zs = np.sum(z)
    if zs > 1e-10:
        z = z / zs
    return z, M
```

def compute_conv(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    c = compute_conv(a_norm)
    return 2.0 * n * np.max(c)

def compute_grad_smooth_max(a, t=0.01):
    """Compute gradient of t*log(sum(exp(conv(a,a)/t))) w.r.t. a."""
    n = len(a)
    c = compute_conv(a)
    c_shifted = c - c.max()
    w = np.exp(c_shifted / t)
    w_sum = w.sum()
    w /= w_sum
    a_flip = a[::-1].copy()
    L = 1
    while L < (2*n-1) + n - 1:
        L <<= 1
    W = rfft(w, L)
    AF = rfft(a_flip, L)
    conv_res = irfft(W * AF, L)
    grad = 2.0 * conv_res[n-1:n-1+n]
    obj_val = c.max() + t * np.log(w_sum)
    return grad, obj_val

    if s < 1e-10:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(max_iters):
        c = compute_conv(a)
        M = c.max()

        threshold = M - 1e-8 * max(M, 1)
        S = np.where(c >= threshold)[0]

            S = np.array([np.argmax(c)])

        # And sum_j z_j = 1
        
        # This is a (|S|+1) x (n+1) system.
        # If |S| + 1 == n + 1 (i.e., |S| == n), it's square.

        sys_rows = len(S) + 1

        for idx, k in enumerate(S):

                ki = k - j
                if 0 <= ki < n:
                    mat[idx, j] = a[ki]
            mat[idx, n] = -1.0  # coefficient of M
        mat[-1, :n] = 1.0  # sum(z) = 1
        mat[-1, n] = 0.0
        
        rhs = np.zeros(sys_rows)
        rhs[-1] = 1.0

        if sys_rows == n + 1:

The LP alternating method works as follows:

But the issue is that the active set changes with each iteration. The active-set method iterates:
1. Solve the LP (or the reduced system) with current active set S
2. Get solution z*
3. Compute conv(a, z*) and find new active set S_new
4. If S_new == S, we've converged
5. Otherwise, set S = S_new and go to step 1

The "reduced system" for a given active set S is:
`A_S @ z = M * 1, sum(z) = 1, z >= 0`

where A_S is the |S| x n submatrix of the convolution matrix.

If |S| = n, this is a square linear system (n+1 equations, n+1 unknowns including M).

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Fast linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    c = compute_conv(a_norm)
    return 2.0 * n * float(np.max(c))


def compute_grad_smooth_max(a, t):
    """
    Compute gradient and value of t*log(sum(exp(conv(a,a)/t))) w.r.t. a.
    Returns (grad, smooth_max_value).
    """
    n = len(a)
    c = compute_conv(a)
    c_max = float(np.max(c))
    c_shifted = c - c_max
    w = np.exp(c_shifted / t)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        w_sum = 1.0
    w /= w_sum
    # grad_j = 2 * sum_k w[k] * a[k-j] (a[i]=0 outside [0,n-1])
    # = 2 * conv(w, a_flip)[n-1+j]
    a_flip = a[::-1].copy()
    L = 1
    while L < (2 * n - 1) + n - 1:
        L <<= 1
    W = rfft(w, L)
    AF = rfft(a_flip, L)
    conv_res = irfft(W * AF, L)
    grad = 2.0 * conv_res[n - 1:2 * n - 1]
    obj_val = c_max + t * np.log(w_sum)
    return grad, obj_val


def active_set_solve(a, S):
    """
    Solve the linear system for a given active set S.
    System: A_S @ z = M * 1, sum(z) = 1.
    Returns (z_normalized, M) or (None, None).
    """
    n = len(a)
    sys_rows = len(S) + 1
    mat = np.zeros((sys_rows, n + 1))
    for idx, k in enumerate(S):
        # Row: sum_j a[k-j] * z_j - M = 0
        # a[k-j] is nonzero when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            mat[idx, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
        mat[idx, n] = -1.0
    # Last row: sum(z) = 1
    mat[-1, :n] = 1.0
    rhs = np.zeros(sys_rows)
    rhs[-1] = 1.0

    try:
        if sys_rows == n + 1:
            sol = np.linalg.solve(mat, rhs)
        else:
            sol, _, rank, _ = np.linalg.lstsq(mat, rhs, rcond=None)
        z = np.maximum(sol[:n], 0)
        M_val = float(sol[n])
        zs = float(np.sum(z))
        if zs < 1e-10 or M_val < 0:
            return None, None
        z = z / zs
        return z, M_val
    except np.linalg.LinAlgError:
        return None, None


def active_set_iteration(a, max_iters=5000, tol=1e-12):
    """
    Fast active-set iteration: solve linear system with current active set,
    then update the active set. Much faster than full LP.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()

    if s < 1e-10:
        return a
    a = a / s

        M = float(np.max(c))

        # Active set with tolerance
        if prev_M is not None:
            thresh = M - max(1e-10, 1e-8 * M)
        else:
            thresh = M - max(1e-10, 1e-6 * M)
        S = np.where(c >= thresh)[0]

        # Limit active set size to avoid ill-conditioning
        if len(S) > n + 5:
            # Keep the most "active" constraints
            S = S[np.argsort(c[S])[::-1][:n + 5]]

        z, M_new = active_set_solve(a, S)
        if z is None:
            # Fallback: use the argmax only
            S_min = np.array([int(np.argmax(c))])
            z, M_new = active_set_solve(a, S_min)
            if z is None:
                break

        obj_z = compute_obj(z)
        if obj_z < best_obj - 1e-13:
            best_obj = obj_z
            best_a = z.copy()

        # Check convergence
        a = z
        c = compute_conv(a)
        M_curr = float(np.max(c))
        if prev_M is not None and abs(M_curr - prev_M) < tol * max(prev_M, 1e-10):
            break
        prev_M = M_curr

    return best_a


def smooth_max_adam(a, time_budget=100.0, verbose=False):
    """
    Adam optimization in log-domain using smooth max (logsumexp) surrogate.

    n = len(a)

    v = np.zeros(n)

    lr = 0.05
    t_smooth = 0.05

    a_curr = np.exp(log_a)
    a_curr = a_curr / a_curr.sum()
    best_log_a = log_a.copy()
    best_obj = compute_obj(a_curr)

    t_iter = 0
    stall = 0
    best_obj_smooth = compute_obj(a_curr)

    while time.monotonic() - start < time_budget:

        a_curr = np.exp(log_a)
        a_curr = a_curr / a_curr.sum()

        grad_a, _ = compute_grad_smooth_max(a_curr, t_smooth)

        # Log-domain gradient: dL/dlog_a_i = a_i * (grad_a_i - sum(a*grad_a))
        lo = float(np.sum(a_curr * grad_a))
        grad_log = a_curr * (grad_a - lo)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * grad_log ** 2
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Anneal temperature
        t_smooth = max(t_smooth * 0.998, 0.001)
        lr = max(lr * 0.9995, 0.001)

            obj_check = compute_obj(a_check)

                best_obj = obj_check

                stall = 0

                    lr = min(0.1, lr * 1.5)
                    m[:] = 0
                    v[:] = 0
                    stall = 0

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-15)


def coordinate_descent(a, time_budget=30.0):
    """
    Coordinate descent: for each coordinate, optimize its value
    while keeping others fixed (with renormalization).
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    order = np.arange(n)
    while time.monotonic() - start < time_budget:
        np.random.default_rng().shuffle(order)
        # Use our rng
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            # Binary search for optimal a[i]
            # Fix all other coordinates, vary a[i]
            # Constraint: sum(a) = 1, a[i] >= 0
            # If a[i] = x, then others must sum to 1-x
            # Scale others: a_j' = a_j * (1-x) / (1-a[i]) for j != i
            # (only if 1-a[i] > 0)
            rest_sum = 1.0 - a[i]
            if rest_sum < 1e-10:
                continue

            best_x = a[i]
            best_local_obj = best_obj

            # Try a few values
            for frac in [0.5, 0.7, 0.85, 1.2, 1.5, 2.0, 0.3]:
                x_try = a[i] * frac
                if x_try < 0 or x_try >= 1.0 - 1e-10:
                    continue
                a_try = a.copy()
                a_try[i] = x_try
                # Renormalize others
                others_scale = (1.0 - x_try) / rest_sum
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_try[mask] *= others_scale
                if np.min(a_try) < -1e-10:
                    continue
                obj_try = compute_obj(a_try)
                if obj_try < best_local_obj:
                    best_local_obj = obj_try
                    best_x = x_try

            if best_x != a[i]:
                a[i] = best_x
                others_scale = (1.0 - best_x) / rest_sum
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a[mask] *= others_scale
                a = a / a.sum()
                if best_local_obj < best_obj - 1e-13:
                    best_obj = best_local_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a

        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
