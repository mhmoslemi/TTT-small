            if 0 < idx_best < grid_size - 1:
                spread = max(3, grid_size // 10)
                lo = max(0, idx_best - spread)
                hi = min(grid_size - 1, idx_best + spread)
                x_lo = x_grid[lo]
                x_hi = x_grid[hi]
                x_fine = np.linspace(x_lo, x_hi, 60)
                Xf = x_fine[:, None]
                vals_f = A_vec[None, :] * Xf * Xf + B_vec[None, :] * Xf + C_vec[None, :]
                obj_fine = np.max(vals_f, axis=1)
                idx_fine = int(np.argmin(obj_fine))
                x_best = x_fine[idx_fine]

            # Build new vector and verify
            if x_best >= 0.0 and x_best <= 1.0:
                a_new = (1.0 - x_best) * b
                a_new[i] = x_best
                a_new = np.maximum(a_new, 0.0)
                s_new = float(np.sum(a_new))
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    conv_new = compute_conv(a_new)
                    obj_new = 2.0 * n * float(np.max(conv_new))
                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        pass_improved = True
            no_change = True

        if verbose and p % 20 == 0:
            print(f"    CD pass {p}: obj={best_obj:.10f}, improved={pass_improved}, t={time.monotonic()-start:.1f}s")

        if not pass_improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start_k = j
        end_k = min(2 * n - 1, j + n)
        A[start_k:end_k, j] = a[:end_k - start_k]
    return A


def solve_lp(a, max_constr=None):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    from scipy.optimize import linprog
    n = len(a)
    a = a.astype(np.float64)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    if max_constr is not None and len(max_constr) < (2 * n - 1):
        A = A[max_constr, :]

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((A.shape[0], n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(A.shape[0])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s
    return None


def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating method."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        # Get active set
        conv_cur = compute_conv(a)
        max_conv = np.max(conv_cur)
        threshold = max_conv * 0.97
        active = np.where(conv_cur >= threshold)[0]

        g_lp = solve_lp(a, max_constr=active)
        if g_lp is None:
            g_lp = solve_lp(a)
        if g_lp is None:
            break

        # Line search: find best t for (1-t)*a + t*g
        conv_aa = compute_conv(a)
        conv_ag = compute_conv_cross(a, g_lp)
        conv_gg = compute_conv(g_lp)

        x_t = np.linspace(0, 1, 40)
        objs = []
        for t in x_t:
            val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
            objs.append(2.0 * n * float(np.max(val)))
        objs = np.array(objs)
        t_opt = x_t[int(np.argmin(objs))]

        # Refine
        if 0 < int(np.argmin(objs)) < len(x_t) - 1:
            lo = x_t[max(0, int(np.argmin(objs)) - 3)]
            hi = x_t[min(len(x_t) - 1, int(np.argmin(objs)) + 3)]
            x_fine = np.linspace(lo, hi, 20)
            objs_f = []
            for t in x_fine:
                val = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
                objs_f.append(2.0 * n * float(np.max(val)))
            t_opt = x_fine[int(np.argmin(objs_f))]

        a_new = (1 - t_opt) * a + t_opt * g_lp
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                if verbose and it % 10 == 0:
                    print(f"    LP it{it}: obj={best_obj:.10f}, t={t_opt:.4f}, time={time.monotonic()-start:.1f}s")

    return best_a


def smooth_max_grad_descent(a, time_budget=60.0, verbose=False):
    """Gradient descent with smooth max (Nesterov)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = 0.0002
    lr = 0.005
    momentum = 0.8
    v = np.zeros(n)
    stall = 0

    for it in range(3000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -400, 400)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        if total < 1e-300:
            break
        weights = exp_vals / total

        # grad[i] = sum_k weights[k] * (contribution of a_i to conv[k])
        # conv[k] = sum_j a_j * a_{k-j}, d(conv[k])/da_i = a_{k-i} + a_{k-i} (if valid)
        # So grad[i] = sum_k weights[k] * (a_{k-i} + a_{2i-k}) approximately
        # Use correlate for efficiency
        grad = np.zeros(n)
        for offset in range(-n + 1, n):
            k_range = np.arange(max(0, -offset), min(n, n - offset))
            a_part = a[k_range]
            w_part = weights[max(0, offset):min(2 * n - 1, n + offset)]
            grad += w_part * a_part if offset >= 0 else np.zeros(0) * 0
        # Simpler: grad[i] = sum_k weights[k] * (a[(k-i) mod valid] + a[(2i-k) mod valid])
        # Use FFT-based approach
        # Actually just do it directly with the weight vector
        # grad[i] = convolve(weights_padded, a)[i] + convolve(weights_padded, a)[2i]... 
        # Let's use a different approach: 
        # d(max smooth)/da_i = sum_k w_k * d(conv_k)/da_i
        # d(conv_k)/da_i = a_{k-i} (if 0<=k-i<n) + a_{k-i} (if 0<=k-i<n, from the other side)
        # Actually d(conv_k)/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i} if 0<=k-i<n, 
        # except when k-i=i (k=2i) where it's 2*a_i (but a_{k-i}=a_i so still 2*a_i)
        # Wait: conv_k = sum_j a_j * a_{k-j}. d/d a_i = a_{k-i} (from j=i term) + a_{k-i} (from k-j=i term)
        # But if k-i=i (k=2i), it's just 2*a_i (not a_i + a_i counted twice, it's the derivative of a_i*a_i = 2*a_i)
        # So: d(conv_k)/da_i = 2 * a_{k-i} if 0 <= k-i < n, else 0. (The k=2i case gives 2*a_i which is correct.)
        
        grad = np.zeros(n)
        for i in range(n):
            k_minus_i = np.arange(2 * n - 1) - i
            valid = (k_minus_i >= 0) & (k_minus_i < n)
            if np.any(valid):
                grad[i] = 2.0 * np.sum(weights[valid] * a[k_minus_i[valid]])
        
        # Center the gradient (invariance to adding constant)
        grad -= np.mean(grad)
        
        v = momentum * v - lr * grad
        a_new = np.maximum(a + v, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 30:
                lr *= 0.6
                a = best_a.copy()
                v = np.zeros(n)
                stall = 0
                if lr < 1e-7:
                    break
        
        T = max(0.00002, T * 0.999)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def pair_coordinate_descent(a, n_passes=200, time_budget=60.0, rng=None):
    """
    Pair coordinate descent: simultaneously update two coordinates.
    a_i = x, a_j = y, rest scaled to maintain sum=1.
    This is a 2D optimization but we use a simplified 1D path.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    if rng is None:
        rng = np.random.default_rng()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for p in range(n_passes):
        if time.monotonic() - start > time_budget:
            break

        pass_improved = False
        for _ in range(n // 2):
            if time.monotonic() - start > time_budget:
                break

            # Pick random pair
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue

            ai, aj = a[i], a[j]
            s_rest = 1.0 - ai - aj
            if s_rest < 1e-14:
                continue

            # Path: move from (ai, aj) toward (0, 0) or toward center
            # Parametrize: a_i = ai + t*(target_i - ai), a_j = aj + t*(target_j - aj)
            # Try several target points
            targets = [
                (0.0, 0.0),  # move both to 0
                (s_rest / 2, s_rest / 2),  # split rest equally
                (1.0, 0.0),  # concentrate on i
                (0.0, 1.0),  # concentrate on j
                (0.5 * s_rest, 0.0),
                (0.0, 0.5 * s_rest),
            ]

            # Also try the "current -> uniform-on-remaining" direction
            # and "current -> swap" direction
            targets.append((aj, ai))  # swap

            b_rest = a / s_rest
            b_rest[i] = 0.0
            b_rest[j] = 0.0

            best_x = (ai, aj)
            best_o = best_obj

            for ti, tj in targets:
                # Make sure ti + tj <= 1 and ti, tj >= 0
                if ti < 0 or tj < 0 or ti + tj > 1.0 + 1e-9:
                    continue
                # Grid along the path from (ai,aj) to (ti,tj)
                for t in np.linspace(0, 1, 15):
                    xi = ai + t * (ti - ai)
                    xj = aj + t * (tj - aj)
                    if xi < -1e-10 or xj < -1e-10 or xi + xj > 1 + 1e-9:
                        continue
                    xi = max(0.0, xi)
                    xj = max(0.0, xj)
                    s_rem = 1.0 - xi - xj
                    if s_rem < -1e-10:
                        continue
                    if s_rem < 1e-14:
                        a_try = np.zeros(n)
                        a_try[i] = xi
                        a_try[j] = xj
                    else:
                        a_try = (s_rem) * b_rest * (s_rest > 0)
                        # Wait, this is wrong. Let me redo.
                        a_try = a.copy()
                        a_try[i] = 0
                        a_try[j] = 0
                        a_try = a_try / (1 - ai - aj + 1e-300)  # normalize rest
                        a_try = s_rem * a_try
                        a_try[i] = xi
                        a_try[j] = xj
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_o - 1e-13:
                        best_o = obj_try
                        best_x = a_try.copy()

            if best_o < best_obj - 1e-13:
                best_obj = best_o
                best_a = best_x.copy()
                a = best_x.copy()
                pass_improved = True

        if not pass_improved and p > 20:
            break

    return best_a
