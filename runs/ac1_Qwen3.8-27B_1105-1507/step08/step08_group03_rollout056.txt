def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_minpeak(a, active_set=None, timeout=10.0):
    n = len(a)
    a = a / float(np.sum(a))
    
    if active_set is not None and len(active_set) < 2 * n - 1:

                if 0 <= jj < n:
                    A[idx, j] = a[jj]
    else:
        A = build_conv_matrix(a)
    
    if HAVE_CVXPY:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            try:

                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((A.shape[0], n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0

    A_eq = np.zeros((1, n + 1))

    top_k = np.argsort(conv_ag)[::-1][:max(len(active), 10)]
    active = np.unique(np.concatenate([active, top_k]))
    return list(active)

def mirror_descent_improved(a, time_budget=60.0, verbose=False):

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:

            j_end = k - i_start
            g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]

            w /= w.sum()
            all_w += w

        active_mask = all_w > threshold

        active_w /= active_w.sum()
        active_idx = np.where(active_mask)[0]

        g = np.zeros(n)
        for w, k in zip(active_w, active_idx):
            if w > 1e-10:
                g += w * grad_conv_at_peak(a, int(k))

        v = momentum * v + (1 - momentum) * g
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = max(1e-6, lr * 0.9995)
            momentum = min(0.7, momentum + 0.001)
        else:
            stall += 1
            if stall > 80:

                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = 0.0

        it += 1

def smooth_optimize_improved(a, time_budget=60.0, verbose=False):
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def grad_conv_at_peak(a, k):
        g = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
        return g

    def obj_grad(x, T):
        x = np.maximum(x, 1e-15)
        s = x.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        x = x / s
        conv = compute_conv(x)
        max_c = conv.max()
        z = T * (conv - max_c)
        z -= z.max()
        e = np.exp(z)
        sum_e = e.sum()
        smooth_max = max_c + np.log(sum_e) / T
        obj_val = 2.0 * n * smooth_max
        w = e / sum_e
        g = np.zeros(n)
        for k in range(len(conv)):
            if w[k] > 1e-7:
                g += w[k] * grad_conv_at_peak(x, int(k))
        g *= 2.0 * n
        return obj_val, g

    for T_val in [30.0, 60.0, 120.0]:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break
        try:
            res = sp_minimize(lambda x: obj_grad(x, T_val), a, method='L-BFGS-B',
                              bounds=[(1e-12, None)] * n,
                              options={'maxiter': 300, 'maxfun': 600, 'disp': False})
            a_opt = np.maximum(res.x, 0)
            s_opt = float(np.sum(a_opt))
            if s_opt > 1e-10:
                a_opt = a_opt / s_opt
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_obj:
                    best_obj = obj_opt
                    best_a = a_opt.copy()
                    a = a_opt
                    if verbose:
                        print(f"    Smooth T={T_val}: obj={best_obj:.8f}")
        except Exception:
            pass

    return best_a


def nesterov_accelerated(a, time_budget=60.0, K=5, verbose=False):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 100
    it = 0

    def grad_conv_at_peak(a, k):
        g = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
        return g

    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break

        spread = (peak_vals[0] - peak_vals[-1]) / max_val
        temp = 5.0 / (1.0 + spread * 15.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / (max_val + 1e-15))
        weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            if w > 1e-12:
                g += w * grad_conv_at_peak(y, int(k))

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 31 + n)
                y = best_a * np.exp(rng.normal(0, 0.08, n))
                y = np.maximum(y, 0)
                y = y / y.sum()
                a_new = y.copy()
                a_prev = y.copy()
                y_new = y.copy()
                t_k = 1.0
                stall = 0
                patience = int(patience * 1.2)

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    # Use symmetric interpolation to preserve structure
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Bell shape
        x = np.linspace(-1, 1, n)
        a = np.exp(-x**2 * 2)
        a = a / a.sum()
    elif mode == 3:
        # U-shape
        x = np.linspace(-1, 1, n)
        a = x**2 + 0.1
        a = a / a.sum()
    elif mode == 4:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        a = a / a.sum()
    elif mode == 5:
        # Hamming-like
        a = 0.54 + 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Two peaks
        centers = [n // 3, 2 * n // 3]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (n * 0.1))
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def coordinate_descent_refine(a, time_budget=30.0):
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Find top contributors to peak
        contrib_idx = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib_idx.append((a[i] * a[j], i, j))
        contrib_idx.sort(reverse=True)

        # Find non-contributors
        non_contrib = [i for i in range(n)
                      if (k_star - i < 0) or (k_star - i >= n)]

        if not non_contrib:
            break

        improved = False
        for val, i, j in contrib_idx[:10]:
            if val < 1e-14:
                break
            for target in non_contrib[:5]:
                for frac in [0.005, 0.01, 0.02, 0.04]:
                    delta = frac * a[i]
                    if delta < 1e-16:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_obj = compute_obj(a)
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(800) / 800
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Quick LP alternating on parent ===
    print(f"\n=== Phase 1: Quick LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.1, 120.0)
    if p1_budget > 20:
        a_p1 = lp_alternating_improved(best_seq, time_budget=p1_budget, verbose=True)
        obj_p1 = compute_obj(a_p1)
        if obj_p1 < best_score:
            best_score = obj_p1
            best_seq = a_p1.copy()
            print(f"  Phase 1: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1: obj={obj_p1:.8f} (no improvement)")

    # === Phase 2: Systematic n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    
    # Generate candidate n values
    n_candidates = sorted(set([
        400, 500, 600, 650, 700, 750, 800, 850, 900, 950, 1000,
        1100, 1200, 1300, 1400, 1500,
        max(300, n_current - 200),
        max(300, n_current - 100),
        n_current + 100,
        n_current + 200,
    ]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 1500]
    # Sort by proximity to current best n
    n_candidates.sort(key=lambda x: abs(x - n_current))
    
    n_explored = {}
    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == n_current and len(n_explored) > 0:
            continue
        if n_try in n_explored:
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.08, 60.0)
        if alloc < 20:
            alloc = min(remaining * 0.15, 40.0)
        
        a_try = lp_alternating_improved(a_try, time_budget=alloc, verbose=False)
        obj_try = compute_obj(a_try)
        n_explored[n_try] = obj_try
        
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

        if len(n_explored) >= 8:
            break

    # === Phase 3: Deep LP alternating on best ===
    print(f"\n=== Phase 3: Deep LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.35, 500.0)
    if p3_budget > 30:
        # Multiple rounds with increasing perturbation
        for rnd in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p3_budget / (3 - rnd), remaining * 0.5)
            if alloc < 20:
                break
            if rnd > 0:
                a_pert = best_seq * np.exp(rng.normal(0, 0.02 * rnd, len(best_seq)))
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
            else:
                a_pert = best_seq.copy()
            a_p3 = lp_alternating_improved(a_pert, time_budget=alloc, verbose=True)
            obj_p3 = compute_obj(a_p3)
            if obj_p3 < best_score:
                best_score = obj_p3
                best_seq = a_p3.copy()
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_p3:.8f}")

    # === Phase 4: Mirror descent ===
    print(f"\n=== Phase 4: Mirror Descent ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.12, 100.0)
    if p4_budget > 20:
        a_p4 = mirror_descent_improved(best_seq, time_budget=p4_budget, verbose=True)
        obj_p4 = compute_obj(a_p4)
        if obj_p4 < best_score:
            best_score = obj_p4
            best_seq = a_p4.copy()
            print(f"  Mirror: obj={best_score:.8f} **")
        else:
            print(f"  Mirror: obj={obj_p4:.8f}")

    # === Phase 5: Nesterov ===
    print(f"\n=== Phase 5: Nesterov ===")
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.08, 60.0)
    if p5_budget > 15:
        a_p5 = nesterov_accelerated(best_seq, time_budget=p5_budget, K=7, verbose=True)
        obj_p5 = compute_obj(a_p5)
        if obj_p5 < best_score:
            best_score = obj_p5
            best_seq = a_p5.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_p5:.8f}")

    # === Phase 6: Smooth optimization ===
    print(f"\n=== Phase 6: Smooth Optimization ===")
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.12, 100.0)
    if p6_budget > 15:
        a_p6 = smooth_optimize_improved(best_seq, time_budget=p6_budget, verbose=True)
        obj_p6 = compute_obj(a_p6)
        if obj_p6 < best_score:
            best_score = obj_p6
            best_seq = a_p6.copy()
            print(f"  Smooth: obj={best_score:.8f} **")
        else:
            print(f"  Smooth: obj={obj_p6:.8f}")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    restart_modes = [2, 3, 4, 5, 6]
    for r, mode in enumerate(restart_modes):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        
        restart_budget = min(remaining * 0.1, 50.0)
        if restart_budget < 15:
            break
        a_r = lp_alternating_improved(a_r, time_budget=restart_budget, verbose=False)
        obj_r = compute_obj(a_r)
        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_r.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # LP polish
    if remaining > 40:
        alloc = min(remaining * 0.25, 80.0)
        a_f1 = lp_alternating_improved(best_seq, time_budget=alloc, verbose=True)
        obj_f1 = compute_obj(a_f1)
        if obj_f1 < best_score:
            best_score = obj_f1
            best_seq = a_f1.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.15, 30.0)
        a_f2 = coordinate_descent_refine(best_seq, time_budget=alloc)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_score:
            best_score = obj_f2
            best_seq = a_f2.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.3, 40.0)
        a_f3 = lp_alternating_improved(best_seq, time_budget=alloc, verbose=False)
        obj_f3 = compute_obj(a_f3)
        if obj_f3 < best_score:
            best_score = obj_f3
            best_seq = a_f3.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Smooth polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.4, 30.0)
        a_f4 = smooth_optimize_improved(best_seq, time_budget=alloc, verbose=False)
        obj_f4 = compute_obj(a_f4)
        if obj_f4 < best_score:
            best_score = obj_f4
            best_seq = a_f4.copy()
            print(f"  Final Smooth: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
