def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a)). Assumes sum(a)=1, a>=0."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """
    n = len(a)
    # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # This is a sliding dot product = cross-correlation
    # conv[a, p_rev][m] = sum_j a[j]*p_rev[m-j] where p_rev[k] = p[2n-2-k]
    # And grad[i]/2 = conv[a, p_rev][2n-2-i]
    
    # Alternative: use FFT directly
    # cross[i] = sum_k p[k] * a[k-i] for k-i in [0,n-1]
    # = sum_j a[j] * p[i+j] (setting j = k-i, k = i+j)
    
    # This is the "anti-convolution": (p * a)[i] in the sense of sum_j a[j]*p[i+j]
    # We can compute this as: take the full convolution of a and reversed_p
    p_rev = p[::-1]  # p_rev[k] = p[2n-2-k], length 2n-1
    N = 1
    while N < n + (2 * n - 1) - 1:
        N <<= 1
    A = fft(a, N)
    PR = fft(p_rev, N)
    conv_full = ifft(A * PR).real  # length N, valid up to n + 2n-2 - 1 = 3n-3
    
    # conv_full[m] = sum_j a[j] * p_rev[m-j] = sum_j a[j] * p[2n-2-m+j]
    # We want grad[i]/2 = sum_j a[j] * p[i+j]
    # Set 2n-2-m+j = i+j => m = 2n-2-i
    # So grad[i]/2 = conv_full[2n-2-i]
    # i ranges from 0 to n-1, so 2n-2-i ranges from 2n-2 down to n-1
    grad_half = conv_full[n-1:3*n-3][::-1][:n]
    # Wait, let me recheck. i=0: m=2n-2. i=n-1: m=n-1.
    # So grad_half[i] = conv_full[2n-2-i], i from 0 to n-1
    # This is conv_full[2n-2], conv_full[2n-3], ..., conv_full[n-1]
    grad_half = conv_full[2*n-2 - np.arange(n)]
    return 2.0 * grad_half


def compute_gradient_direct(a, p):
    """Direct computation of gradient (slower but correct for verification)."""
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(p)):
        w = p[k]
        if w < 1e-15:
            continue
        # grad[i] += 2*w*a[k-i] for valid i where 0 <= k-i < n
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            # a[k-i] where i goes from i_start to i_end
            # k-i goes from k-i_start down to k-i_end
            idx = k - np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] += 2.0 * w * a[idx]
    return grad


def frank_wolfe(a, time_budget=100.0, verbose=False):
    """
    Frank-Wolfe (conditional gradient) for min max_k c[k] s.t. sum(a)=1, a>=0.
    Uses adaptive active-set weighting.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break

        # Soft-weighted active set: weight by proximity to peak
        # Use a temperature to control selectivity
        temp = max_conv * 0.05 + 1e-10
        weights = np.exp((conv - max_conv) / temp)
        total_w = float(weights.sum())
        if total_w < 1e-30:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        else:
            weights /= total_w

        # Compute gradient
        grad = compute_gradient_direct(a, weights)

        # Frank-Wolfe: pick vertex minimizing <grad, v> = min(grad)
        s_idx = int(np.argmin(grad))
        s_vertex = np.zeros(n)
        s_vertex[s_idx] = 1.0

        # Step size: 2/(t+2) (standard FW)
        gamma_t = 2.0 / (t + 2)
        a_new = (1.0 - gamma_t) * a + gamma_t * s_vertex

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        t += 1

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"    FW it{t}: obj={best_obj:.8f}, stall={stall}, t={elapsed:.1f}s")

        if stall > 50000:
            break

    return best_a


def smoothed_fista(a, time_budget=100.0, tau_init=5.0, tau_end=500.0, verbose=False):
    """
    FISTA on log-sum-exp smoothed max of autocorrelation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    t_k = 1.0
    iter_count = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.95))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.8)

        conv = compute_conv(y)
        max_c = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_c))
        p = exp_vals / float(np.sum(exp_vals))

        # Compute gradient using FFT cross-correlation
        grad = compute_gradient_smoothed(y, p)

        # Step size: conservative 1/L
        # L ~ 2 * max(p) * 2n (rough bound)
        lr = 1.0 / (4.0 * n * float(np.max(p)) * tau)
        lr = min(lr, 5.0 / np.sqrt(n))

        a_new = y - lr * grad
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        t_k = t_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        iter_count += 1

        if verbose and iter_count % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    FISTA it{iter_count}: obj={best_obj:.8f}, tau={tau:.1f}, t={elapsed:.1f}s")

    return best_a


def nesterov_smoothed(a, time_budget=100.0, tau_init=5.0, tau_end=400.0, verbose=False):
    """Nesterov accelerated gradient descent on smoothed objective over simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta = 0.9
    lr = 2.0 / np.sqrt(n)
    v = np.zeros(n)
    prev_a = a.copy()
    prev_g = None
    it = 0
    best_it = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (time_budget * 0.9))
        tau = tau_init * (tau_end / tau_init) ** (progress ** 0.7)

        if it % 5000 == 0 and verbose:
            print(f"    NSD it{it}: obj={best_obj:.8f}, tau={tau:.1f}, lr={lr:.6f}, t={elapsed:.1f}s")

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = best_a.copy()
        else:
            y = y / s_y

        # Compute gradient at y
        conv = compute_conv(y)
        max_c = float(np.max(conv))
        exp_vals = np.exp(tau * (conv - max_c))
        p = exp_vals / float(np.sum(exp_vals))
        g = compute_gradient_smoothed(y, p)

        # Barzilai-Borwein step size (alternate)
        if prev_g is not None and it % 2 == 0:
            s_vec = y - prev_a
            y_vec = g - prev_g
            ss = float(np.dot(s_vec, s_vec))
            sy = float(np.dot(s_vec, y_vec))
            if ss > 1e-15 and abs(sy) > 1e-12:
                bb_lr = ss / abs(sy)
                lr = 0.7 * lr + 0.3 * min(max(bb_lr, 1e-8), 10.0 / np.sqrt(n))

        v = beta * v - lr * g
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            best_it = it

        prev_a = a.copy()
        prev_g = g.copy()
        a = a_new
        it += 1
        lr = max(1e-8, lr * 0.9999)

        if it - best_it > 30000:
            a = best_a.copy()
            v = np.zeros(n)
            lr = 2.0 / np.sqrt(n) * 0.5
            best_it = it

    return best_a


def build_conv_matrix(a):
    """Build sparse-ish matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, max_constraints=2000):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Uses full constraint set for accuracy.
    """
    n = len(a)
    a_norm = a / np.sum(a)
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    return None, None


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing with adaptive blending."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    iter_count = 0
    last_t = 0.3
    rng = np.random.default_rng(99999)

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        candidates = sorted(set([
            0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8,
            max(0.05, last_t * 0.5), min(1.0, last_t * 1.5),
            max(0.05, last_t * 0.7), min(1.0, last_t * 1.3),
            max(0.05, last_t * 1.1), min(1.0, last_t / 1.1),
        ]))

        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()

        for t in candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        iter_count += 1
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            if verbose and iter_count % 2 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t_val:.3f}, M={M_lp:.6f}")
        else:
            scale = 0.015 + 0.005 * min(iter_count, 10)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            if iter_count > 40:
                break

    return best_a


def peak_shifting(a, time_budget=60.0, verbose=False):
    """
    Iterative peak reduction: identify dominant peaks and apply
    targeted modifications to reduce them.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.05
    stall = 0

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        peak_idx = int(np.argmax(conv))

        # Compute subgradient at the peak
        # grad_peak[i] = 2*a[peak_idx - i] (for valid indices)
        grad = np.zeros(n)
        k = peak_idx
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = k - np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] = 2.0 * a[idx]

        # Also include second peak for stability
        sorted_idx = np.argsort(conv)[::-1]
        if len(sorted_idx) > 1 and conv[sorted_idx[1]] > 0.8 * max_conv:
            k2 = int(sorted_idx[1])
            i_start2 = max(0, k2 - n + 1)
            i_end2 = min(n - 1, k2)
            if i_start2 <= i_end2:
                idx2 = k2 - np.arange(i_start2, i_end2 + 1)
                grad[i_start2:i_end2 + 1] += 2.0 * a[idx2]

        # Update with projection
        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr = min(lr * 1.02, 0.2)
        else:
            stall += 1
            a = best_a
            lr *= 0.8
            if stall > 300:
                break

        it += 1

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.01))
        a = a / a.sum()
    elif mode == 3:
        # Raised cosine
        x = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(x)
        a = a / a.sum()
    elif mode == 4:
        # Bimodal
        c1, c2 = n // 4, 3 * n // 4
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 5:
        # Spectral approach: flat PSD target
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * x)) ** 0.5
        a = a / a.sum()
    elif mode == 6:
        # Random Fourier features
        freqs = rng.uniform(0, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 7:
        # Triangular
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Frank-Wolfe + FISTA + Nesterov + LP polishing + multi-length.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    # === Phase 1: Frank-Wolfe (fast, simple) ===
    print(f"\n=== Phase 1: Frank-Wolfe ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.2, 200.0)
    if p1_budget > 30:
        a_fw = frank_wolfe(best_seq, time_budget=p1_budget, verbose=True)
        obj_fw = compute_obj(a_fw)
        if obj_fw < best_score:
            best_score = obj_fw
            best_seq = a_fw.copy()
            print(f"  FW: obj={best_score:.8f} **")
        else:
            print(f"  FW: obj={obj_fw:.8f}")

    # === Phase 2: Nesterov smoothed descent ===
    print(f"\n=== Phase 2: Nesterov Smoothed Descent ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.2, 200.0)
    if p2_budget > 30:
        a_nsd = nesterov_smoothed(best_seq, time_budget=p2_budget,
                                  tau_init=5.0, tau_end=400.0, verbose=True)
        obj_nsd = compute_obj(a_nsd)
        if obj_nsd < best_score:
            best_score = obj_nsd
            best_seq = a_nsd.copy()
            print(f"  NSD: obj={best_score:.8f} **")
        else:
            print(f"  NSD: obj={obj_nsd:.8f}")

    # === Phase 3: Peak shifting ===
    print(f"\n=== Phase 3: Peak Shifting ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.1, 100.0)
    if p3_budget > 20:
        a_ps = peak_shifting(best_seq, time_budget=p3_budget, verbose=True)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  PS: obj={best_score:.8f} **")
        else:
            print(f"  PS: obj={obj_ps:.8f}")

    # === Phase 4: LP Polishing (main refinement) ===
    print(f"\n=== Phase 4: LP Polishing ===")
    remaining = deadline - time.monotonic()
    for rnd in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.15, 80.0)
        if alloc < 20:
            break
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f}")
            if rnd < 5:
                scale = 0.015 * (rnd + 1)
                best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                best_seq = np.maximum(best_seq, 0)
                best_seq = best_seq / best_seq.sum()

    # === Phase 5: FISTA refinement ===
    print(f"\n=== Phase 5: FISTA ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        alloc = min(remaining * 0.15, 120.0)
        a_fista = smoothed_fista(best_seq, time_budget=alloc,
                                  tau_init=10.0, tau_end=600.0, verbose=True)
        obj_fista = compute_obj(a_fista)
        if obj_fista < best_score:
            best_score = obj_fista
            best_seq = a_fista.copy()
            print(f"  FISTA: obj={best_score:.8f} **")
        else:
            print(f"  FISTA: obj={obj_fista:.8f}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)

    n_options = sorted(set([
        max(400, n_current - 200), max(400, n_current - 100),
        n_current + 100, n_current + 200, n_current + 300,
        600, 700, 800, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1400 and abs(n - n_current) > 80]

    for n_try in n_options[:4]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        a_try = resample_sequence(best_seq, n_try)

        # Quick NSD
        alloc1 = min(remaining * 0.2, 60.0)
        if alloc1 > 15:
            a_try = nesterov_smoothed(a_try, time_budget=alloc1,
                                       tau_init=10.0, tau_end=300.0)

        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 50.0)
        if alloc2 > 15:
            a_try = lp_polish(a_try, time_budget=alloc2)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.3 + 0.1 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        alloc = min(remaining * 0.2, 80.0)
        if alloc > 30:
            half = alloc / 2
            a_r = nesterov_smoothed(a_r, time_budget=half,
                                     tau_init=10.0, tau_end=300.0)
            remaining = deadline - time.monotonic()
            lp_alloc = min(remaining * 0.1, 40.0)
            if lp_alloc > 10:
                a_r = lp_polish(a_r, time_budget=lp_alloc)

            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                n_best = len(a_r)
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    remaining = deadline - time.monotonic()

    # High-tau Nesterov
    if remaining > 40:
        alloc = min(remaining * 0.25, 100.0)
        a_sgd = nesterov_smoothed(best_seq, time_budget=alloc,
                                   tau_init=30.0, tau_end=800.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final NSD: obj={best_score:.8f} **")

    # LP
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining * 0.3, 100.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # Peak shifting
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 40.0)
        a_ps = peak_shifting(best_seq, time_budget=alloc)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_score:
            best_score = obj_ps
            best_seq = a_ps.copy()
            print(f"  Final PS: obj={best_score:.8f} **")

    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 50.0)
        a_lp = lp_polish(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
