    grad = conv[n-1:n-1+n] * 2.0
    return grad

Softmax mirror descent optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses exponentiated gradient with adaptive temperature for the soft-max approximation,
combined with periodic LP polishing and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def compute_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def cross_correlate(a, w):
    """Compute cross[j] = sum_l a[l] * w[j+l] for j=0..n-1 using FFT."""
    n = len(a)
    L = 1
    while L < n + len(w) - 1:
        L <<= 1
    a_rev = a[::-1]
    A = rfft(a_rev, L)
    W = rfft(w, L)
    conv = irfft(A * W, L)
    return conv[n - 1:n]

def softmax(x, t=1.0):
    """Stable softmax."""
    x = x - np.max(x)
    e = np.exp(t * x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

def mirror_descent_step(a, grad, lr):
    """Exponentiated gradient step: a_new[j] ∝ a[j] * exp(-lr * grad[j])."""
    # Clip to prevent underflow/overflow
    exponent = -lr * grad
    exponent = np.clip(exponent, -50, 50)
    a_new = a * np.exp(exponent)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s

def compute_softgrad(a, t):
    """Compute gradient of soft_max_t(conv(a,a)) w.r.t. a.
    soft_max_t(b) = (1/t) * log(sum exp(t*b))
    grad[j] = sum_k w[k] * db[k]/da[j] = 2 * sum_k w[k] * a[k-j] (valid)
    where w = softmax(t * b)
    """
    n = len(a)
    b = compute_conv(a)
    w = softmax(b, t)
    grad = 2.0 * cross_correlate(a, w)
    return grad, b, w

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A

def solve_lp(a):
    """Solve: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > 30:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except:
            continue
    
    # Fallback scipy
    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except:
        pass
    
    return None, None

"""
Softmax Mirror Descent optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses exponentiated gradient (mirror descent on simplex) with adaptive temperature
for soft-max approximation of the peak, combined with periodic LP polishing,

import numpy as np
from scipy.fft import rfft, irfft

def compute_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def cross_correlate(a, w):
    """Compute cross[j] = sum_l a[l] * w[j+l] for j=0..n-1 using FFT.
    a has length n, w has length 2n-1.
    """
    n = len(a)
    # Convolution of a_rev (length n) with w (length 2n-1) has length 3n-2
    L = 1
    while L < 3 * n - 1:
        L <<= 1
    a_rev = a[::-1]
    A = rfft(a_rev, L)
    W = rfft(w, L)
    conv = irfft(A * W, L)
    return conv[n - 1:n]


def stable_softmax(x, t):
    """Compute softmax(t * x) stably."""
    tx = t * (x - np.max(x))
    tx = np.clip(tx, -700, 700)
    e = np.exp(tx)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

    Returns (grad, b, w) where b=conv(a,a), w=softmax(t*b), grad = 2*cross(a,w).

def mirror_step(a, grad, lr):

    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

        grad, b, w = compute_softgrad(a, t)

        # Normalize gradient
        gmax = np.max(np.abs(grad))
        if gmax < 1e-15:
            break
        grad = grad / gmax

        # Mirror descent step
        a_new = mirror_step(a, grad, lr)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and (it % 500 == 0):
                print(f"    SoftMD: it={it}, obj={best_obj:.8f}, t={t:.2f}, lr={lr:.4f}")
        else:
            no_improve += 1
            if no_improve > 5000:
                # Perturb and restart
                rng_local = np.random.default_rng(it)
                a = best_a * np.exp(rng_local.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s2 = a.sum()
                if s2 > 1e-10:
                    a = a / s2
                no_improve = 0
                lr = min(lr * 1.5, 5.0)
                t = max(t * 0.8, 1.0)

        a = a_new

        # Adapt temperature
        t = min(t * t_growth, 1000.0)

        # Decay learning rate slowly
        if it % 20000 == 19999 and it > 0:
            lr *= 0.97

    return best_a


def softmd_optimize_v2(a, time_budget=300.0, verbose=False):
    """Alternative: use power-sum with adaptive p, Euclidean projection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    p = 4.0
    lr = 2.0 / n
    no_improve = 0

    for it in range(5000000):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        b = compute_conv(a)
        bp = np.maximum(b, 1e-30) ** (p - 1)
        total = (bp * b).sum()
        if total < 1e-30:
            break
        w = bp / total

        grad = 2.0 * cross_correlate(a, w)

        # Center gradient for simplex (tangent space)
        grad_centered = grad - np.mean(grad)

        # Gradient step + projection
        a_new = a - lr * grad_centered
        a_new = project_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and (it % 500 == 0):
                print(f"    SoftMDv2: it={it}, obj={best_obj:.8f}, p={p:.1f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                # Increase p, reduce lr
                p = min(p * 1.2, 200.0)
                lr *= 0.9
                no_improve = 0

        a = a_new
        p = min(p * 1.0005, 200.0)
        if it % 10000 == 9999:
            lr *= 0.95

    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp(a):
    """Solve: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > 45:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternate(a, time_budget=100.0, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break

        best_local_obj = np.inf
        best_local_a = best_a.copy()

        for tt in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - tt) * a + tt * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            stall = 0
            if verbose:
                print(f"    LP: obj={best_obj:.8f}, time={time.monotonic()-start:.0f}s")
        else:
            stall += 1
            if stall > 6:
                break

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng):
    """Generate a diverse structured initialization."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-40 * (x - 0.5) ** 2)
    elif choice == 1:
        a = 0.3 + 0.7 * np.cos(2 * np.pi * 3 * x) ** 2
    elif choice == 2:
        a = np.exp(-60 * (x - 0.3) ** 2) + np.exp(-60 * (x - 0.7) ** 2)
    elif choice == 3:
        a = np.ones(n)
        for _ in range(4):
            center = rng.integers(0, n)
            width = max(3, n // 8)
            for i in range(n):
                d = abs(i - center)
                a[i] += rng.standard_normal() * np.exp(-d / (width * 0.5))
        a = np.maximum(a, 0.01)
    elif choice == 4:
        # Beta-like distribution
        alpha = rng.uniform(1, 5)
        beta_p = rng.uniform(1, 5)
        a = x ** alpha * (1 - x) ** beta_p
        a = np.maximum(a, 0.001)
    else:
        # Low-frequency cosine mix
        a = np.ones(n)
        for k in range(3):
            freq = rng.integers(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            a += 0.3 * np.cos(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: softmax mirror descent (v1 and v2) + LP polishing +
    multi-n exploration + diverse restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        print(f"Parent: n={len(a)}, obj={compute_obj(a):.8f}")
    except Exception:
        a = np.ones(800) / 800
        print(f"Uniform init: n=800, obj={compute_obj(a):.8f}")

    best_a = a.copy()
    best_obj = compute_obj(a)
    n_base = len(best_a)

    # === Phase 1: Softmax Mirror Descent v1 (exponentiated gradient) ===
    print(f"\n=== Phase 1: SoftMD v1 (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 400.0)
    a1 = softmd_optimize(best_a, time_budget=p1_budget, verbose=True)
    obj1 = compute_obj(a1)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1
        print(f"  ** Phase 1: obj={best_obj:.8f} **")
    else:
        print(f"  Phase 1: obj={obj1:.8f}")

    # === Phase 2: Softmax Mirror Descent v2 (power-sum + projection) ===
    print(f"\n=== Phase 2: SoftMD v2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = min(remaining * 0.20, 250.0)
        a2 = softmd_optimize_v2(best_a, time_budget=p2_budget, verbose=True)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2
            print(f"  ** Phase 2: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 2: obj={obj2:.8f}")

    # === Phase 3: LP polishing ===
    print(f"\n=== Phase 3: LP Polishing ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.20, 200.0)
        a3 = lp_alternate(best_a, time_budget=p3_budget, verbose=True)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3
            print(f"  ** Phase 3: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 3: obj={obj3:.8f}")

    # === Phase 4: SoftMD v1 again from improved point ===
    print(f"\n=== Phase 4: SoftMD v1 Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = min(remaining * 0.20, 200.0)
        a4 = softmd_optimize(best_a, time_budget=p4_budget, verbose=True)
        obj4 = compute_obj(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4
            print(f"  ** Phase 4: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 4: obj={obj4:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        max(200, n_best - 200),
        max(200, n_best - 100),
        n_best,
        n_best + 100,
        n_best + 200,
        n_best + 400,
    ]))
    n_options = [x for x in n_options if 200 <= x <= 3000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        if n_try == n_best:
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

        # SoftMD optimization at this n
        alloc_md = min(80.0, remaining * 0.30)
        a_opt = softmd_optimize(a_try, time_budget=alloc_md)
        obj_try = compute_obj(a_opt)

        # Quick LP polish if promising
        if obj_try < best_obj + 0.01:
            remaining = deadline - time.monotonic()
            if remaining > 25:
                a_opt = lp_alternate(a_opt, time_budget=min(30.0, remaining * 0.3))
                obj_try = compute_obj(a_opt)

        if obj_try < best_obj - 1e-6:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse initializations ===
    print(f"\n=== Phase 6: Diverse Inits ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    for d in range(8):
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        a_div = diverse_init(n_best, rng)
        alloc = min(50.0, remaining * 0.20)
        a_opt = softmd_optimize(a_div, time_budget=alloc)
        obj_div = compute_obj(a_opt)

        if obj_div < best_obj + 0.01 and remaining > 30:
            # Quick LP polish
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_opt = lp_alternate(a_opt, time_budget=min(25.0, remaining * 0.3))
                obj_div = compute_obj(a_opt)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse {d}: obj={obj_div:.8f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.35, 60.0)
        a_f = softmd_optimize(best_a, time_budget=alloc)
        obj_f = compute_obj(a_f)
        if obj_f < best_obj:
            best_obj = obj_f
            best_a = a_f
            print(f"  Final SoftMD: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.40, 50.0)
        a_lp = lp_alternate(best_a, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp
            print(f"  Final LP: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.50, 30.0)
        a_last = softmd_optimize_v2(best_a, time_budget=alloc)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last SoftMDv2: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")
    return result
