"""
Advanced optimizer combining smooth log-sum-exp gradient descent with adaptive
temperature, LP alternating with dual-guided peak equalization, and parametric
initialization. Key innovation: uses the log-sum-exp smoothing of the max
function to get a smooth, differentiable objective that enables faster
convergence than subgradient methods. Combines with LP sub-solves for
global structure and iterative peak equalization for local refinement.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import cvxpy as cp

def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def smooth_max_and_weights(conv, t):
    """Compute smooth max and softmax weights of conv with temperature t."""
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / t
    # Clamp for numerical stability
    shifted = np.maximum(shifted, -50.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    if sum_exp < 1e-300:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
        return max_conv, weights
    weights = exp_shifted / sum_exp
    smooth_val = max_conv + t * np.log(sum_exp)
    return smooth_val, weights


def smooth_objective(a, t):
    """Compute smooth objective value."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    sm, _ = smooth_max_and_weights(conv, t)
    return 2.0 * n * sm


def smooth_gradient(a, t):

    Compute gradient of the smooth objective w.r.t. a (before normalization).
    Uses the formula: grad[i] = 2 * sum_j weights[i+j] * a[j]
    where weights are the softmax of conv/t.
    Returns gradient of 2n * smooth_max(conv(a/s, a/s), t) w.r.t. a.
    """
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.zeros(n)
    a_norm = a / s
    conv = compute_conv(a_norm)
    _, weights = smooth_max_and_weights(conv, t)
    # grad_norm[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a_norm[j]
    grad_norm = np.zeros(n)
    for i in range(n):
        j_end = min(n, 2 * n - 1 - i)
        if j_end > 0:
            grad_norm[i] = 2.0 * np.dot(weights[i:i + j_end], a_norm[:j_end])
    # Chain rule: d/d(a) of f(a/s) where a appears as a/s

    # But since we're on the simplex and projecting, we just use the gradient direction
    # normalized to account for the constraint sum(a) = 1
    # The effective gradient on the simplex is grad - mean(grad)
    grad = 2.0 * n * grad_norm / s
    grad = grad - np.mean(grad)
    return grad

def smooth_gradient_descent(a0, time_budget=100.0, t_init=0.05, t_min=0.005,
                            top_k_mode=False, top_k=5, verbose=False):

    patience_limit = 100

            grad = smooth_gradient(a, t)

        improved = False
        for backtrack in range(10):

            obj_try = compute_obj(a_try)
            if obj_try < obj_current - 1e-14:

                improved = True
                break

        if not improved:
            # Try small random perturbation

                best_obj = obj_try
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
        else:

            patience_limit = min(patience_limit * 2, 500)
            lr = max(lr * 0.7, 0.01)

            elapsed = time.monotonic() - start
            print(f"      SmoothGD it={it}: obj={best_obj:.8f}, t={t:.5f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_alternating(a, verbose=False):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_opt, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    y = cp.Variable(A.shape[0], nonneg=True)  # Dual variables for tracking
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    best_g = None
    best_M = np.inf

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    gv = gv / gs
                    Mv = float(M.value)
                    if Mv < best_M:
                        best_M = Mv
                        best_g = gv
        except Exception:
            continue

    if best_g is None:
        return None, None, None

    # Try to get dual variables
    dual_y = None
    try:
        # The dual of A@g <= M is y >= 0
        # Access via constraint dual values
        if prob.constraints[0].dual_value is not None:
            dual_y = -np.array(prob.constraints[0].dual_value).flatten()
            dual_y = np.maximum(dual_y, 0)
            if np.sum(dual_y) > 1e-10:
                dual_y = dual_y / np.sum(dual_y)
    except Exception:
        pass

    return best_g, best_M, dual_y


def lp_alternate_advanced(a0, time_budget=100.0, verbose=False):
    """LP alternating with 1D line search and dual-guided weighting."""
    n = len(a0)
    a = a0.astype(np.float64).copy()

    if s < 1e-10:
        return a0.copy()

    best_a = a.copy()

        if g_lp is None:
            break

        if dual_y is not None:
            # Weighted target: blend g_lp with a weighted average based on dual
            conv_a = compute_conv(a)
            # Peaks where dual has mass are the "active" ones
            # Create a modified target that emphasizes these peaks
            alpha = 0.3
            target = (1 - alpha) * g_lp + alpha * a
        else:
            target = g_lp

            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))

                return np.inf
            a_try = a_try / st
            return compute_obj(a_try)

        try:
            res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 30, 'xatol': 0.01})
            best_t = res.x if res.success else 0.5
        except Exception:
            best_t = 0.5

        a_try = (1.0 - best_t) * a + best_t * target
        a_try = np.maximum(a_try, 0)
        st = float(np.sum(a_try))
        if st < 1e-10:
            break
        a_try = a_try / st
        obj_try = compute_obj(a_try)

        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            a = best_a
            stall_count = 0
            if verbose:
                print(f"      LP: obj={best_obj:.8f}, t={best_t:.3f}, elapsed={time.monotonic()-start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 20:
                break

    return best_a


def peak_equalization(a0, time_budget=60.0, n_peaks=8, verbose=False):
    """
    Iterative peak equalization: identify top peaks, compute their contributors,
    and redistribute mass to reduce the maximum peak.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for outer in range(200):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        top_peaks = np.argsort(conv)[::-1][:n_peaks]
        max_val = conv[top_peaks[0]]
        mean_top = np.mean(conv[top_peaks])
        # If peaks are already well-equalized, stop
        if max_val - mean_top < 1e-6 * max_val:
            break

        improved = False
        for k in top_peaks[:4]:
            k = int(k)
            if conv[k] < max_val * 0.95:
                continue
            # Compute contribution of each a[i] to conv[k]
            # conv[k] = sum_j a[j] * a[k-j]
            # d(conv[k])/d(a[i]) = 2*a[k-i] (if valid)
            contributors = np.zeros(n)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                contributors[i] = a[j]
                if i != j:
                    contributors[j] += a[i]
                else:
                    contributors[i] = a[i]

            # Find positions with high contribution but that can be reduced
            # and positions with low contribution that can accept more mass
            threshold_med = np.median(contributors[contributors > 0])
            high = np.where(contributors > threshold_med * 1.5)[0]
            low = np.where(contributors > 0)
            low = low[np.argsort(contributors[low])[:len(low)//2]]

            if len(high) == 0 or len(low) == 0:
                continue

            # Try moving a small amount from high to low
            for src in high[:3]:
                for dst in low[:3]:
                    if src == dst or a[src] < 1e-10:
                        continue
                    for frac in [0.02, 0.05, 0.1]:
                        delta = frac * a[src]
                        a[src] -= delta
                        a[dst] += delta
                        a = np.maximum(a, 0)
                        st = float(np.sum(a))
                        if st < 1e-10:
                            a[src] += delta
                            a[dst] -= delta
                            continue
                        a = a / st
                        obj_try = compute_obj(a)
                        if obj_try < best_obj - 1e-13:
                            best_obj = obj_try
                            best_a = a.copy()
                            improved = True
                            break
                        else:
                            a[src] += delta
                            a[dst] -= delta
                            a = a / st
                            break

        if not improved and outer > 30:
            break
        a = best_a.copy()
        if verbose and outer % 10 == 0:
            print(f"      PEQ iter={outer}: obj={best_obj:.8f}")

    return best_a


def parametric_init(n, param_type='cos_power', exponent=1.5):
    """Generate a parametric initial sequence."""
    x = np.linspace(0, 1, n)
    if param_type == 'cos_power':
        a = np.cos(np.pi * x / 2) ** exponent
    elif param_type == 'gaussian':
        sigma = 1.0 / exponent
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif param_type == 'super_elliptic':
        p = exponent
        a = 1.0 - np.abs(2 * x - 1) ** p
    elif param_type == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif param_type == 'blackman':
        a = 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif param_type == 'raised_cosine':
        a = np.cos(np.pi * (x - 0.5)) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def combined_optimize(a0, time_budget=100.0, verbose=False):
    """Combined smooth gradient + LP + peak equalization cycle."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_idx = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Step 1: Smooth gradient
        alloc1 = min(20.0, remaining * 0.35)
        if alloc1 > 3:
            a1 = smooth_gradient_descent(best_a, time_budget=alloc1,
                                         t_init=0.02, t_min=0.003,
                                         top_k_mode=True, top_k=5)
            obj1 = compute_obj(a1)
            if obj1 < best_obj - 1e-13:
                best_obj = obj1
                best_a = a1.copy()
                if verbose:
                    print(f"      CG R{round_idx} smooth: obj={best_obj:.8f}")

        # Step 2: LP alternating
        alloc2 = min(25.0, remaining * 0.35)
        if alloc2 > 5:
            a2 = lp_alternate_advanced(best_a, time_budget=alloc2)
            obj2 = compute_obj(a2)
            if obj2 < best_obj - 1e-13:
                best_obj = obj2
                best_a = a2.copy()
                if verbose:
                    print(f"      CG R{round_idx} LP: obj={best_obj:.8f}")

        # Step 3: Peak equalization
        alloc3 = min(15.0, remaining * 0.25)
        if alloc3 > 3:
            a3 = peak_equalization(best_a, time_budget=alloc3, n_peaks=6)
            obj3 = compute_obj(a3)
            if obj3 < best_obj - 1e-13:
                best_obj = obj3
                best_a = a3.copy()
                if verbose:
                    print(f"      CG R{round_idx} PEQ: obj={best_obj:.8f}")

        round_idx += 1

    return best_a


def propose_candidate(budget_s=1500):
    """
    Main search function. Combines parametric initialization, smooth gradient
    descent (log-sum-exp), LP alternating, and peak equalization in a
    multi-phase strategy.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = parametric_init(800, 'cos_power', 1.5)
        best_score = compute_obj(best_seq)
        print(f"Default init (n=800): obj={best_score:.8f}")

    # === Phase 1: Parametric initialization sweep (80s) ===
    print(f"\n=== Phase 1: Parametric Initialization ===")
    n_base = len(best_seq)
    param_types = ['cos_power', 'gaussian', 'super_elliptic', 'hann', 'blackman', 'raised_cosine']
    exponents = [0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 2.5, 3.0]

    for pt in param_types:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        for exp in exponents:
            a_try = parametric_init(n_base, pt, exp)
            obj_try = compute_obj(a_try)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  {pt} e={exp}: obj={best_score:.8f} **")
            # Quick smooth GD on promising params
            if obj_try < best_score + 0.01:
                remaining = deadline - time.monotonic()
                if remaining > 8:
                    a_opt = smooth_gradient_descent(a_try, time_budget=min(5.0, remaining * 0.3),
                                                    t_init=0.05, t_min=0.01, top_k_mode=True, top_k=5)
                    obj_opt = compute_obj(a_opt)
                    if obj_opt < best_score:
                        best_score = obj_opt
                        best_seq = a_opt.copy()
                        print(f"  {pt} e={exp}+SGD: obj={best_score:.8f} **")

    # === Phase 2: n-sweep with smooth optimization (120s) ===
    print(f"\n=== Phase 2: n-Sweep ===")
    n_options = [400, 500, 600, 700, 800, 900, 1000, 1200, 1500, 2000]
    n_options = sorted(set(n_options + [n_base]))
    n_options = [n for n in n_options if 200 <= n <= 2500]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        if n_try == n_base:
            a_try = best_seq.copy()
        else:
            a_try = resample_seq(best_seq, n_try)
        a_opt = smooth_gradient_descent(a_try, time_budget=min(8.0, remaining * 0.2),
                                        t_init=0.05, t_min=0.008, top_k_mode=True, top_k=5)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 3: Main smooth gradient descent (250s) ===
    print(f"\n=== Phase 3: Smooth Gradient (main) ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        # Full log-sum-exp mode
        alloc = min(80.0, remaining * 0.35)
        a_full = smooth_gradient_descent(best_seq, time_budget=alloc,
                                         t_init=0.1, t_min=0.003,
                                         top_k_mode=False, verbose=True)
        obj_full = compute_obj(a_full)
        if obj_full < best_score:
            best_score = obj_full
            best_seq = a_full.copy()
            print(f"  Full SGD: obj={best_score:.8f} **")
        else:
            print(f"  Full SGD: obj={obj_full:.8f}")

        # Top-K mode
        remaining = deadline - time.monotonic()
        if remaining > 50:
            for k_val in [3, 5, 8]:
                if remaining < 20:
                    break
                alloc = min(50.0, remaining * 0.3)
                a_k = smooth_gradient_descent(best_seq, time_budget=alloc,
                                              t_init=0.05, t_min=0.003,
                                              top_k_mode=True, top_k=k_val)
                obj_k = compute_obj(a_k)
                if obj_k < best_score:
                    best_score = obj_k
                    best_seq = a_k.copy()
                    print(f"  TopK={k_val}: obj={best_score:.8f} **")
                else:
                    print(f"  TopK={k_val}: obj={obj_k:.8f}")
                remaining = deadline - time.monotonic()

    # === Phase 4: LP Alternating (300s) ===
    print(f"\n=== Phase 4: LP Alternating ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        for round_idx in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(50.0, remaining * 0.25)
            a_lp = lp_alternate_advanced(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score - 1e-12:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP R{round_idx}: obj={best_score:.8f} **")
            else:
                print(f"  LP R{round_idx}: obj={obj_lp:.8f}")
                # Perturb to escape
                a_pert = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
                a_pert = np.maximum(a_pert, 0)
                a_pert /= np.sum(a_pert)
                best_seq_for_next = a_pert
                # Use perturbed for next round
                a_lp2 = lp_alternate_advanced(a_pert, time_budget=min(20.0, remaining * 0.15))
                obj_lp2 = compute_obj(a_lp2)
                if obj_lp2 < best_score:
                    best_score = obj_lp2
                    best_seq = a_lp2.copy()
                    print(f"  LP R{round_idx} perturb: obj={best_score:.8f} **")

    # === Phase 5: Combined optimization (250s) ===
    print(f"\n=== Phase 5: Combined Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(200.0, remaining * 0.6)
        a_comb = combined_optimize(best_seq, time_budget=alloc, verbose=True)
        obj_comb = compute_obj(a_comb)
        if obj_comb < best_score:
            best_score = obj_comb
            best_seq = a_comb.copy()
            print(f"  Combined: obj={best_score:.8f} **")
        else:
            print(f"  Combined: obj={obj_comb:.8f}")

    # === Phase 6: Peak equalization polish (100s) ===
    print(f"\n=== Phase 6: Peak Equalization ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for n_peaks in [5, 8, 12]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(30.0, remaining * 0.3)
            a_peq = peak_equalization(best_seq, time_budget=alloc, n_peaks=n_peaks)
            obj_peq = compute_obj(a_peq)
            if obj_peq < best_score:
                best_score = obj_peq
                best_seq = a_peq.copy()
                print(f"  PEQ k={n_peaks}: obj={best_score:.8f} **")
            else:
                print(f"  PEQ k={n_peaks}: obj={obj_peq:.8f}")

    # === Phase 7: Random restarts with combined optimization (remaining) ===
    print(f"\n=== Phase 7: Random Restarts ===")
    n_best = len(best_seq)
    for r in range(30):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        scale = 0.01 + 0.003 * r
        a_r = best_seq * np.exp(rng.normal(0, scale, n_best))
        a_r = np.maximum(a_r, 0)
        sr = float(np.sum(a_r))
        if sr < 1e-10:
            continue
        a_r = a_r / sr
        # Quick combined optimization
        alloc = min(remaining * 0.3, 20.0)
        a_opt = combined_optimize(a_r, time_budget=alloc)
        obj_r = compute_obj(a_opt)
        if obj_r < best_score - 1e-12:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r}: obj={best_score:.8f} **")
        else:
            if r % 5 == 0:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Final: one more smooth GD pass ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.7, 60.0)
        a_final = smooth_gradient_descent(best_seq, time_budget=alloc,
                                          t_init=0.01, t_min=0.001,
                                          top_k_mode=True, top_k=3)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"\nFinal SGD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
