"""
Hybrid FFT-accelerated mirror descent + LP alternating optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses spectral initialization, adaptive learning rates,
fast gradient computation via FFT, and periodic LP refinement for escaping local minima.
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smooth_grad_fast(a, beta):
    """O(n log n) gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    m = conv.max()
    shifted = beta * (conv - m)
    weights = np.exp(shifted)
    total = weights.sum()
    if total < 1e-300:
        return np.ones(n) / n
    weights /= total

    # result[i] = sum_j a[j] * w[j+i] for valid j (0<=j<n, j+i<2n-1)
    # = (a * w_rev)[2n-2-i] where w_rev[m] = w[2n-2-m]
    w_rev = weights[::-1]
    conv_len = n + (2 * n - 1) - 1  # 3n-2
    size = 1
    while size < conv_len:
        size <<= 1
    A = rfft(a, size)
    W = rfft(w_rev, size)
    conv_resp = irfft(A * W, size)[:conv_len]

    # result[i] = conv_resp[2n-2-i] for i=0..n-1
    # result = conv_resp[n-1 : 2n-1][::-1]
    result = conv_resp[n - 1:2 * n - 1][::-1]
    return 2.0 * result


def topk_grad(a, K=5):
    """Gradient using top-K subgradients of max(conv)."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:K]
    max_val = conv[top_idx[0]]
    if max_val < 1e-30:
        return np.ones(n) / n
    weights = np.exp((conv[top_idx] - max_val) / max_val * K)
    weights /= weights.sum()

    result = np.zeros(n)
    for w_val, k in zip(weights, top_idx):
        k = int(k)
        # grad_k[i] = 2*a[k-i] for valid i (0 <= k-i < n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            js = k - np.arange(i_end + 1, i_start - 1, -1)
            result[i_start:i_end + 1] += w_val * a[js]
    return 2.0 * result


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build the convolution matrix A where A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a, timeout=5.0):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    if n < 2:
        return None
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def spectral_init(n, rng):
    """Initialize with flat power spectrum (IFFT of random complex with fixed magnitude)."""
    freqs = rng.integers(0, n, size=n)
    phases = rng.uniform(0, 2 * np.pi, size=n)
    spectrum = np.exp(1j * phases)
    a = np.real(np.fft.ifft(spectrum))
    a = np.abs(a) + 0.01
    a = a / a.sum()
    return a


def hann_init(n):
    """Hann window initialization."""
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    a = np.maximum(a, 0.01)

def triangle_init(n):
    """Triangle (Bartlett) window."""
    x = np.arange(n)
    mid = n / 2.0
    a = 1.0 - np.abs(x - mid + 0.5) / mid
    a = np.maximum(a, 0.01)
    return a / a.sum()


def raised_cosine_init(n, alpha=0.5):

    x = np.arange(n)
    t = x / n
    a = np.ones(n)
    edge = int(alpha * n)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.001)

    iteration = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1.0:
            break

        lp_result = solve_lp(a)
        if lp_result[0] is None:
            break
        g_lp = lp_result[0]

        # Line search
        best_t_obj = np.inf
        best_t_val = 0.5
        best_a_local = best_a.copy()

        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.7, 0.9]:
            a_try = (1 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()

        iteration += 1
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local.copy()
            stall = 0
            temperature = max(0.01, temperature * 0.95)
        else:
            stall += 1
            if stall > 20:
                rng = np.random.default_rng(iteration * 17 + n)
                a = best_a * (1.0 + temperature * rng.normal(0, 1, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                temperature = min(0.3, temperature * 1.5)
            else:
                temperature = max(0.02, temperature * 0.97)

        if verbose and iteration % 10 == 0:
            print(f"        LP it{iteration}: obj={best_obj:.8f}, t={best_t_val:.2f}, "
                  f"stall={stall}, temp={temperature:.3f}, t={time.monotonic()-start:.1f}s")

    return best_a


def coordinate_push(a, time_budget=10.0):
    """
    Coordinate descent: push mass from high-convolution-contributing positions
    to low-convolution positions.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    for rnd in range(20):
        if time.monotonic() - start > time_budget:
            break
        improved = False

        # Find top contributors to peak
        contributors = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributors.append((a[i] * a[j], i, j))
        contributors.sort(reverse=True)

        # Find low-convolution target positions
        target_scores = np.zeros(n)
        for i in range(n):
            for k in [0, n - 1, n]:
                j = k - i
                if 0 <= j < n:
                    target_scores[i] += conv[k] * a[j]
        sorted_targets = np.argsort(target_scores)

        for contrib, i, j in contributors[:20]:
            if contrib < 1e-10:
                break
            for target in sorted_targets[:10]:
                if target == i or (i != j and target == j):
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    if delta >= a[i] * 0.9:
                        continue
                    a[i] -= delta
                    if i != j:
                        pass  # we're only moving from one side
                    a[target] += delta
                    a = a / a.sum()
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[target] -= delta
                        a = a / a.sum()
                        break

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid FFT-accelerated mirror descent + LP alternating with
    spectral initialization and adaptive dimension exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = hann_init(n0)
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    n_current = len(best_seq)
    print(f"Initial best: n={n_current}, obj={best_score:.8f}")

    # --- Phase 1: Multi-initialization and quick screening ---
    print("\n=== Phase 1: Multi-Initialization ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.08, 80.0)

    inits = []
    inits.append(("parent", best_seq.copy()))
    inits.append(("hann", hann_init(n_current)))
    inits.append(("triangle", triangle_init(n_current)))
    inits.append(("spectral", spectral_init(n_current, rng)))
    inits.append(("raised_cos", raised_cosine_init(n_current, 0.3)))

    # Try a couple of other dimensions
    for n_alt in [n_current - 100, n_current + 100]:
        if 400 <= n_alt <= 1500:
            inits.append((f"hann_{n_alt}", resample_sequence(best_seq, n_alt)))

    for name, a_init in inits[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        alloc = p1_budget / len(inits[:6])
        a_opt = hybrid_optimize(a_init, time_budget=max(alloc, 2.0))
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  {name} (n={len(a_opt)}): obj={best_score:.8f} **")
        else:
            print(f"  {name} (n={len(a_opt)}): obj={obj:.8f}")

    # --- Phase 2: Main hybrid optimization ---
    print(f"\n=== Phase 2: Main Hybrid Optimization (best so far: {best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.45, 600.0)

    # Multiple rounds with increasing aggressiveness
    for rnd in range(4):
        remaining = deadline - time.monotonic()
        alloc = p2_budget / (4 - rnd) * (1.0 if rnd == 0 else 0.8)
        alloc = min(alloc, remaining * 0.4)
        if alloc < 10:
            break

        if rnd > 0:
            # Perturb for new round
            rng_r = np.random.default_rng(rnd * 99991 + n_current)
            scale = 0.03 + 0.02 * rnd
            a_start = best_seq * (1.0 + scale * rng_r.normal(0, 1, len(best_seq)))
            a_start = np.maximum(a_start, 1e-15)
            a_start = a_start / a_start.sum()
        else:
            a_start = best_seq.copy()

        a_opt = hybrid_optimize(a_start, time_budget=alloc, verbose=(rnd == 0),
                                lp_interval=max(20, 50 - rnd * 10))
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  Round {rnd}: obj={obj:.8f} (no improvement)")

    # --- Phase 3: Aggressive LP polish ---
    print(f"\n=== Phase 3: Aggressive LP Polish (best: {best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.25, 300.0)
    if p3_budget > 10:
        a_lp = aggressive_lp_optimize(best_seq, time_budget=p3_budget, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP polish: obj={best_score:.8f} **")
        else:
            print(f"  LP polish: obj={obj_lp:.8f}")

    # --- Phase 4: Dimension exploration ---
    print(f"\n=== Phase 4: Dimension Exploration (best: {best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        n_best - 150, n_best - 100, n_best - 50,
        n_best + 50, n_best + 100, n_best + 150,
        600, 700, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 400 <= n <= 1500 and n != n_best][:5]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(remaining * 0.12, 60.0)

        a_opt = hybrid_optimize(a_try, time_budget=alloc, lp_interval=40)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # --- Phase 5: Coordinate descent + final LP ---
    print(f"\n=== Phase 5: Final Polish (best: {best_score:.8f}) ===")
    remaining = deadline - time.monotonic()

    # Coordinate descent
    if remaining > 20:
        cd_alloc = min(remaining * 0.15, 40.0)
        a_cd = coordinate_push(best_seq, time_budget=cd_alloc)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  Coord descent: obj={best_score:.8f} **")

    # Final LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        lp_alloc = min(remaining * 0.3, 100.0)
        a_final = aggressive_lp_optimize(best_seq, time_budget=lp_alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    # One more hybrid round
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_h = hybrid_optimize(best_seq, time_budget=min(remaining * 0.4, 60.0), lp_interval=15)
        obj_h = compute_obj(a_h)
        if obj_h < best_score:
            best_score = obj_h
            best_seq = a_h.copy()
            print(f"  Final hybrid: obj={best_score:.8f} **")

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
