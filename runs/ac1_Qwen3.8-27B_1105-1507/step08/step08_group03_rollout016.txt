"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using
LP alternating method with adaptive M-tightening, Nesterov subgradient
polish, multi-n exploration, and KKT verification.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ b = conv(a, b)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_lp_maxsum(A, M):
    """Solve: max sum(b) s.t. A @ b <= M, b >= 0. Assumes A is (2n-1) x n."""
    n = A.shape[1]
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M), 
                  bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        b = np.maximum(res.x, 0)
        if b.sum() > 1e-10:
            return b / b.sum()
    return None

def solve_lp_minmax(a, warm_b=None):
    """Solve: min M s.t. conv(a,b)[k] <= M, sum(b)=1, b>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    # Variables: b[0..n-1], M
    # c = [0, ..., 0, 1] (minimize M)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # A_ub: A @ b - M <= 0 for each row
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)]*n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x[:n], 0)
        s = b.sum()
        if s > 1e-10:
            return b / s, float(res.x[n])
    return None, None

def nesterov_subgradient(a, time_budget=30.0, K=5):
    """Nesterov-accelerated subgradient on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 50
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        top_idx = np.argsort(conv)[::-1][:K]
        peak_vals = conv[top_idx]
        max_val = peak_vals[0]
        if max_val < 1e-15:
            break
        
        # Weights based on peak values
        spread = (peak_vals[0] - peak_vals[-1]) / (max_val + 1e-15)
        temp = 3.0 / (1.0 + spread * 10.0)
        weights = np.exp((peak_vals - peak_vals[0]) * temp / max_val)
        weights /= weights.sum()
        
        # Compute subgradient

        for w, k in zip(weights, top_idx):

                i_end = min(n - 1, k)
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end+1] += w * 2.0 * a[j_start:j_end+1][::-1]
        
        # Nesterov step

        a_new = y - lr * g
        a_new = project_simplex(a_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)
        
        obj_new = compute_obj(a_new)

            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:

                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                stall = 0
                patience = int(patience * 1.5)
        
        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_alternating(a, time_budget=200.0, verbose=False):
    """
    LP alternating method: alternate between solving LPs for a and b.
    At convergence, a = b is the global optimum.
    """
    n = len(a)

def solve_lp_minmax_from_A(A, n):
    """Solve LP: min M s.t. A@b <= M, sum(b)=1, b>=0. A is (2n-1 x n)."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((A.shape[0], n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(A.shape[0])
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)]*n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x[:n], 0)
        s = b.sum()
        if s > 1e-10:
            return b / s, float(res.x[n])
    return None, None

def solve_lp_maxsum_from_A(A, M):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = A.shape[1]
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M),
                  bounds=[(0, None)]*n, method='highs')
    if res.status == 0:
        b = np.maximum(res.x, 0)
        s = b.sum()
        if s > 1e-10:
            return b / s
    return None

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussian bumps
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((np.arange(n) - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangle
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1.0 - np.abs(x))
        a = a + 0.01
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / (np.sqrt(1 - x**2 + 0.001))
        a = a / a.sum()
    elif mode == 5:
        # Random Fourier
        freqs = rng.integers(1, 8, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a
