"""
Smoothed L-BFGS + LP Hybrid Optimizer for minimum peak autocorrelation sequences.

Approach:
1. Log-sum-exp smoothed objective optimized via L-BFGS in log-space
2. Continuation method (increasing temperature)
3. LP-based alternating refinement
4. Multi-n exploration
5. Subgradient polishing
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def smoothed_obj_grad(x, tau):
    """
    Compute the log-sum-exp smoothed objective and gradient in log-space.
    x: log-space parameters (a = softmax(x))
    tau: temperature
    Returns (f, grad_x)
    """
    n = len(x)
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    Z = np.sum(exp_x)
    a = exp_x / Z

    conv = compute_conv(a)

    # Numerically stable log-sum-exp
    max_c = np.max(conv)
    shifted = tau * (conv - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))
    f = (1.0 / tau) * log_sum + max_c

    # Attention weights p_k
    p = exp_shifted / np.sum(exp_shifted)

    # grad_a[j] = sum_k p[k] * a[k-j] for valid k-j
    # Using correlate: grad_a = correlate(p, a, 'full')[n-1 : n-1+n]
    corr = np.correlate(p, a, mode='full')
    grad_a = corr[n - 1: n - 1 + n]

    # Convert to x-space: grad_x[i] = a_i * (grad_a[i] - <grad_a, a>)
    inner = np.dot(grad_a, a)
    grad_x = a * (grad_a - inner)

    return f, grad_x


def lbfgs_smoothed_optimize(a_init, tau, maxiter=300, ftol=1e-12):
    """
    Run L-BFGS on the smoothed objective in log-space, starting from a_init.
    Returns the optimized sequence (normalized, non-negative).
    """
    n = len(a_init)
    # Initialize in log-space
    a_safe = np.maximum(a_init, 1e-10)
    a_safe = a_safe / np.sum(a_safe)
    x0 = np.log(a_safe)

    result = minimize(
        smoothed_obj_grad,
        x0,
        args=(tau,),
        method='L-BFGS-B',
        jac=True,
        options={
            'maxiter': maxiter,
            'ftol': ftol,
            'gtol': 1e-10,
            'maxcor': 20,
            'disp': False,
        }
    )

    x_opt = result.x
    x_max = np.max(x_opt)
    a_opt = np.exp(x_opt - x_max)
    a_opt = a_opt / np.sum(a_opt)
    return a_opt


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ z gives cross-conv a*z."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. (a*z)_k <= M for all k, sum(z)=1, z>=0.
    Returns (z_normalized, M_opt, dual_y) or (None, None, None).
    """
    n = len(a)
    if n > 3000:
        # Too large for LP, use direct approach
        return None, None, None

    A = build_conv_matrix(a)

    # Use scipy linprog directly (faster than cvxpy for this problem)
    from scipy.optimize import linprog
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n]), None
    except:
        pass

    # Try cvxpy
    try:
        z = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        prob = cp.Problem(cp.Minimize(M), [A @ z <= M, cp.sum(z) == 1])
        for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    z_val = np.maximum(z.value, 0)
                    zs = float(np.sum(z_val))
                    if zs > 1e-10:
                        z_val = z_val / zs
                        return z_val, float(M.value), None
            except:
                continue
    except:
        pass

    return None, None, None


def lp_alternating(a, time_budget=100.0, verbose=False):
    """LP alternating with 2-cycle detection."""
    n = len(a)
    if n > 3000:
        return a  # Too large for LP

    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    prev_a = None
    prev_prev_a = None
    stall = 0

    while time.monotonic() - start < time_budget:
        result = solve_lp(a)
        if result[0] is None:
            break
        g_lp, M_lp, _ = result

        # 2-cycle detection
        if prev_prev_a is not None:
            diff = np.max(np.abs(a - prev_prev_a))
            if diff < 1e-7:
                a_break = 0.5 * (a + prev_a)
                a_break = np.maximum(a_break, 0)
                a_break = a_break / np.sum(a_break)
                obj_break = compute_obj(a_break)
                if obj_break < best_obj:
                    best_obj = obj_break
                    best_a = a_break.copy()
                a = a_break
                prev_a = None
                prev_prev_a = None
                stall = 0
                continue

        # Try multiple mixing ratios
        improved = False
        for mix in [0.3, 0.5, 0.7, 1.0, 0.1, 0.2]:
            a_try = (1.0 - mix) * a + mix * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                prev_a = a.copy()
                prev_prev_a = prev_a
                a = a_try
                improved = True
                stall = 0
                if verbose:
                    print(f"    LP-alt: obj={best_obj:.8f} (mix={mix})")
                break

        if not improved:
            stall += 1
            if stall > 20:
                break
            # Small perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
            a = best_a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            stall = 0

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)


def subgradient_polish(a, time_budget=60.0):
    """
    Nesterov-accelerated mirror descent with exact subgradient.
    Works in the simplex (sum=1, a>=0) using multiplicative updates.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov momentum
    y = a.copy()  # extrapolated point
    beta = 0.9

    t = 1
    while time.monotonic() - start < time_budget:
        # Compute subgradient at y
        conv_y = compute_conv(y)
        max_val = np.max(conv_y)
        # Find all peak positions (within 1% of max)
        peak_mask = conv_y > max_val * 0.99
        peak_indices = np.where(peak_mask)[0]

        # Subgradient: average of shifted a's at peak positions
        # grad[j] = (1/|P|) * sum_{k in P} a[k-j] (valid)
        # Using correlate: for each peak k, the "shifted a" vector has
        # element j = a[k-j] if valid
        # This is equivalent to: corr = correlate(peak_indicator, a, 'full')
        
        # For efficiency, compute the weighted sum
        weights = np.zeros(2 * n - 1)
        for k in peak_indices:
            weights[k] = 1.0
        weights /= len(peak_indices)

        # grad_a[j] = sum_k weights[k] * a[k-j]
        corr = np.correlate(weights, a, mode='full')
        grad_a = corr[n - 1: n - 1 + n]

        # Mirror descent step in log-space (multiplicative weights)
        # Step size: eta = 1 / (t + 1) for subgradient methods
        eta = 1.0 / (t + 1)
        
        # Update: a <- a * exp(-eta * grad_a)
        # Then normalize
        a_new = a * np.exp(-eta * grad_a)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_new = (t - 1.0) / t_new
        y = a_new + beta_new * (a_new - a)
        y = np.maximum(y, 1e-15)
        y = y / np.sum(y)
        
        t = int(t_new)

        if compute_obj(a_new) < best_obj:
            best_obj = compute_obj(a_new)
            best_a = a_new.copy()

        a = a_new

    return best_a
