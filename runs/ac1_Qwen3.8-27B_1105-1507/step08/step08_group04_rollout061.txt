"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max FISTA, LP alternating with geometric blending,
peak shaving, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def soft_max_weights(conv, tau):
    """Compute softmax weights over convolution values."""
    max_c = np.max(conv)
    scaled = tau * (conv - max_c)
    exp_scaled = np.exp(scaled)
    return exp_scaled / np.sum(exp_scaled)

def compute_grad(a, tau):
    """Compute gradient of (2n/tau)*log(sum exp(tau*(a*a)[k])) at a."""
    n = len(a)
    conv = compute_conv(a)
    w = soft_max_weights(conv, tau)  # length 2n-1, sums to 1
    
    # Build even extension of w
    offset = 2 * n - 2
    w_even = np.zeros(4 * n - 3)
    w_even[offset:offset + 2*n-1] = w
    # Mirror: w_even[offset - k] = w[k] for k >= 1
    idx = np.arange(1, 2*n-1)
    w_even[offset - idx] = w[idx]
    # w_even[offset] = w[0] (already set)
    
    # convolve a with w_even, take elements [offset, offset+n)
    size = 1
    while size < n + 4*n - 3 - 1:
        size <<= 1
    A = rfft(a, size)
    W = rfft(w_even, size)
    result = irfft(A * W, size)
    grad = result[offset:offset + n] * (2.0 * n / tau)
    
    return grad

def fista_optimize(a, time_budget=100.0, tau_init=50.0, tau_final=500.0):
    """FISTA (accelerated projected gradient) with soft-max objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = a.copy()  # current iterate
    y = a.copy()  # FISTA momentum variable
    t_k = 1.0
    
    # Adaptive learning rate
    lr = 1.0 / (2.0 * n)  # initial
    
    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive temperature
        progress = min(1.0, it / 10000.0)
        tau = tau_init * (tau_final / tau_init) ** progress
        
        # Compute gradient at y
        grad = compute_grad(y, tau)
        
        # Gradient step
        v = y - lr * grad
        
        # Project onto simplex
        x_new = project_simplex(v)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_simplex(y)
        t_k = t_new
        
        x = x_new
        
        # Adaptive step size (backtracking)
        obj = compute_obj(x)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = x.copy()
        
        # Periodically adjust learning rate
        if it % 500 == 499:
            # Decrease lr slowly
            lr *= 0.995
        
        # Restart FISTA momentum periodically
        if it % 5000 == 4999:
            y = x.copy()
            t_k = 1.0
    
    return best_a

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A

def solve_lp(a):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    
    # Build constraint matrix
    A = build_conv_matrix(a)
    
    # LP: min c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize M => c = [0, ..., 0, 1]
    # Constraints: A@x_g - M <= 0 for each row => [A, -I] @ [g; M] <= 0
    # A_eq: sum(g) = 1 => [1, ..., 1, 0] @ [g; M] = 1
    # Bounds: g >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        gs = float(np.sum(g_val))
        if gs > 1e-10:
            return g_val / gs, float(res.x[n])
    
    return None, None

def geometric_blend(a, g, t):
    """Geometric (multiplicative) blending on the simplex."""
    # a^{(1-t)} * g^{t}
    # Handle zeros: if a[j] == 0, use g[j]; if g[j] == 0, use 0
    log_a = np.log(a + 1e-300)
    log_g = np.log(g + 1e-300)
    log_result = (1 - t) * log_a + t * log_g
    result = np.exp(log_result)
    s = float(np.sum(result))
    if s < 1e-10:
        return a.copy()
    return result / s

    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        
        # Try both geometric and linear blending with various t values
        best_local_obj = np.inf
        best_local_a = best_a.copy()
        
        t_values = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for t in t_values:

            a = best_a.copy()
            stall = 0
            if verbose:

        else:

            if stall > 5:

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max FISTA, LP alternating with geometric blending,
peak shaving with coordinate descent, and multi-n exploration.
Key improvements over previous: geometric (multiplicative) blending,
soft-max gradient FISTA, structured perturbations, and adaptive n-sweeping.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_soft_max_grad(a, tau):
    """Compute gradient of (2n/tau)*log(sum_k exp(tau*(a*a)[k])) at a (sum(a)=1)."""
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    scaled = tau * (conv - max_c)
    exp_scaled = np.exp(scaled)
    w = exp_scaled / np.sum(exp_scaled)

    # Build even extension of w (length 2n-1 -> effective range [-(2n-2), 2n-2])
    offset = 2 * n - 2
    w_even = np.zeros(4 * n - 3)
    w_even[offset:offset + 2 * n - 1] = w
    idx = np.arange(1, 2 * n - 1)
    w_even[offset - idx] = w[idx]

    # grad[j] = (2n/tau) * sum_l a[l] * w_even_at(j-l)
    # = (2n/tau) * convolve(a, w_even)[j + offset]
    size = 1
    while size < n + 4 * n - 3 - 1:
        size <<= 1
    A = rfft(a, size)
    W = rfft(w_even, size)
    result = irfft(A * W, size)
    grad = result[offset:offset + n] * (2.0 * n / tau)

    return grad


def fista_optimize(a, time_budget=100.0, tau_init=30.0, tau_final=300.0):
    """FISTA (accelerated projected gradient) with soft-max surrogate objective."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    x = a.copy()
    y = a.copy()
    t_k = 1.0
    lr = 1.0 / (3.0 * np.sqrt(n))

    total_iters = 0
    while time.monotonic() - start < time_budget:
        # Adaptive temperature
        progress = min(1.0, total_iters / max(1.0, time_budget * 5.0))
        tau = tau_init * (tau_final / tau_init) ** progress

        grad = compute_soft_max_grad(y, tau)
        v = y - lr * grad
        x_new = project_simplex(v)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = project_simplex(y)
        t_k = t_new
        x = x_new
        total_iters += 1

        obj = compute_obj(x)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = x.copy()

        # Periodic adjustments
        if total_iters % 300 == 299:
            lr *= 0.99
        if total_iters % 3000 == 2999:
            y = x.copy()
            t_k = 1.0

    return best_a


def geometric_blend(a, g, t):
    """Geometric (multiplicative) blending on the simplex: a^(1-t) * g^t."""
    eps = 1e-300
    log_a = np.log(a + eps)
    log_g = np.log(g + eps)
    log_result = (1.0 - t) * log_a + t * log_g
    result = np.exp(log_result)
    s = float(np.sum(result))
    if s < 1e-10:
        return a.copy()
    return result / s


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp(a):
    """Solve: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_value) or (None, None)."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternate(a, time_budget=200.0, verbose=False, max_n_lp=2500):
    """LP alternating with geometric and linear blending."""
    n = len(a)
    if n > max_n_lp:
        # Too large for LP, just return
        return a

    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    iteration = 0

    while time.monotonic() - start < time_budget:
        t0 = time.monotonic()
        g_lp, M_lp = solve_lp(a)
        if g_lp is None:
            break
        iteration += 1

        # Try both geometric and linear blending with various t values
        best_local_obj = np.inf
        best_local_a = best_a.copy()

        t_values = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.8, 1.0]
        for t in t_values:
            # Geometric blending
            a_geo = geometric_blend(a, g_lp, t)
            obj_geo = compute_obj(a_geo)
            if obj_geo < best_local_obj:
                best_local_obj = obj_geo
                best_local_a = a_geo.copy()

            # Linear blending
            a_lin = (1.0 - t) * a + t * g_lp
            a_lin = np.maximum(a_lin, 0)
            s_lin = float(np.sum(a_lin))
            if s_lin > 1e-10:
                a_lin = a_lin / s_lin
                obj_lin = compute_obj(a_lin)
                if obj_lin < best_local_obj:
                    best_local_obj = obj_lin
                    best_local_a = a_lin.copy()

        if best_local_obj < best_obj - 1e-13:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a.copy()
            stall = 0
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    LP iter {iteration}: obj={best_obj:.8f}, "
                      f"lp_time={time.monotonic()-t0:.1f}s, elapsed={elapsed:.0f}s")
        else:
            stall += 1
            if stall > 3:
                # Structured perturbation
                rng = np.random.default_rng(iteration * 7919 + int(time.monotonic() * 1000) % (2**31))
                pert_type = rng.integers(0, 4)
                if pert_type == 0:
                    # Gaussian multiplicative
                    scale = min(0.03 + 0.01 * stall, 0.15)
                    a = best_a * np.exp(rng.normal(0, scale, len(best_a)))
                elif pert_type == 1:
                    # Low-pass filter and re-randomize high freq
                    from scipy.fft import rfft as rfft2, irfft as irfft2
                    sz = len(best_a)
                    fa = rfft2(best_a, sz)
                    cutoff = sz // (4 + rng.integers(0, 4))
                    fa[cutoff:] *= rng.uniform(0.3, 0.7)
                    a = np.maximum(irfft2(fa, sz)[:sz], 0.001)
                elif pert_type == 2:
                    # Random shift (circular)
                    shift = rng.integers(1, max(2, len(best_a) // 10))
                    a = np.concatenate([best_a[-shift:], best_a[:-shift]])
                else:
                    # Mix with a smooth random function
                    x = np.linspace(0, 1, len(best_a))
                    noise = np.exp(-20 * (x - rng.uniform(0.2, 0.8))**2)
                    a = 0.7 * best_a + 0.3 * noise / np.sum(noise) * np.sum(best_a)

                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                    obj_p = compute_obj(a)
                    if obj_p < best_obj:
                        best_obj = obj_p
                        best_a = a.copy()
                stall = 0
                if verbose:
                    elapsed = time.monotonic() - start
                    print(f"    LP perturbed (type={pert_type}), obj={best_obj:.8f}, "
                          f"elapsed={elapsed:.0f}s")

    return best_a


def peak_shave(a, time_budget=60.0):
    """Targeted peak reduction via coordinate descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(98765)
    top_k = 5

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        peak_indices = np.argsort(conv)[::-1][:top_k]

        # Compute total contribution of each a[j] to top peaks
        total_contrib = np.zeros(n)
        for pk in peak_indices:
            for j in range(n):
                kj = int(pk) - j
                if 0 <= kj < n:
                    total_contrib[j] += a[j] * a[kj]

        # Try moving mass from high-contributing to low-contributing indices
        improved_this_round = False
        max_idx = int(np.argmax(total_contrib))
        min_candidates = np.argsort(total_contrib)[:max(10, n // 10)]

        for min_idx in min_candidates:
            if min_idx == max_idx:
                continue
            for eps in [0.002, 0.005, 0.01, 0.02]:
                if a[max_idx] < eps:
                    break
                a_try = a.copy()
                a_try[max_idx] -= eps
                a_try[int(min_idx)] += eps
                a_try = np.maximum(a_try, 0)
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved_this_round = True
                    break

        # Random pairwise swaps
        for _ in range(30):
            if time.monotonic() - start > time_budget:
                break
            j1 = int(rng.integers(0, n))
            j2 = int(rng.integers(0, n))
            if j1 == j2:
                continue
            eps = float(rng.uniform(0.001, 0.008))
            if a[j1] < eps:
                continue
            a_try = a.copy()
            a_try[j1] -= eps
            a_try[j2] += eps
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved_this_round = True

        if not improved_this_round and rng.random() < 0.3:
            # Small Gaussian perturbation
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 6))
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Gaussian bump at center
        a = np.exp(-40 * (x - 0.5) ** 2)
    elif choice == 1:
        # Cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 2:
        # Two bumps
        a = np.exp(-60 * (x - 0.3) ** 2) + np.exp(-60 * (x - 0.7) ** 2)
    elif choice == 3:
        # Power law
        a = np.maximum(x, 1e-6) ** rng.uniform(0.3, 3.0)
    elif choice == 4:
        # Smooth random
        a = np.ones(n)
        for _ in range(4):
            c = float(rng.uniform(0.1, 0.9))
            w = float(rng.uniform(0.05, 0.3))
            amp = float(rng.uniform(-0.5, 0.5))
            a = a + amp * np.exp(-((x - c) / w) ** 2)
        a = np.maximum(a, 0.01)
    else:
        # Oscillatory
        freq = float(rng.uniform(2, 8))
        a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP alternating with geometric blending,
    soft-max FISTA, peak shaving, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        s = float(np.sum(parent))
        if s < 0.01:
            raise ValueError("sum too small")
        a = parent / s
        print(f"Parent loaded: n={len(a)}, obj={compute_obj(a):.8f}")
    except Exception:
        a = np.ones(800) / 800
        print(f"No parent, using uniform: n=800, obj={compute_obj(a):.8f}")

    best_a = a.copy()
    best_obj = compute_obj(a)
    n_base = len(best_a)

    # === Phase 1: LP Alternating (main workhorse, ~45% time) ===
    print(f"\n=== Phase 1: LP Alternating (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.40, 550.0)
    a_lp1 = lp_alternate(best_a, time_budget=p1_budget, verbose=True)
    obj_lp1 = compute_obj(a_lp1)
    print(f"  LP1 result: obj={obj_lp1:.8f}")
    if obj_lp1 < best_obj:
        best_obj = obj_lp1
        best_a = a_lp1
        print(f"  ** New best: obj={best_obj:.8f} **")

    # === Phase 2: FISTA with soft-max (~15% time) ===
    print(f"\n=== Phase 2: Soft-Max FISTA ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = min(remaining * 0.15, 180.0)
        a_f1 = fista_optimize(best_a, time_budget=p2_budget)
        obj_f1 = compute_obj(a_f1)
        print(f"  FISTA result: obj={obj_f1:.8f}")
        if obj_f1 < best_obj:
            best_obj = obj_f1
            best_a = a_f1
            print(f"  ** New best: obj={best_obj:.8f} **")

    # === Phase 3: LP again from FISTA result (~20% time) ===
    print(f"\n=== Phase 3: LP Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.20, 250.0)
        a_lp2 = lp_alternate(best_a, time_budget=p3_budget, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        print(f"  LP2 result: obj={obj_lp2:.8f}")
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2
            print(f"  ** New best: obj={best_obj:.8f} **")

    # === Phase 4: Peak shaving (~10% time) ===
    print(f"\n=== Phase 4: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p4_budget = min(remaining * 0.10, 100.0)
        a_shave = peak_shave(best_a, time_budget=p4_budget)
        obj_shave = compute_obj(a_shave)
        print(f"  Peak shave result: obj={obj_shave:.8f}")
        if obj_shave < best_obj:
            best_obj = obj_shave
            best_a = a_shave
            print(f"  ** New best: obj={best_obj:.8f} **")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_options = sorted(set([
        max(200, n_best - 300), max(200, n_best - 150),
        n_best,
        n_best + 100, n_best + 200, n_best + 400, n_best + 600,
    ]))
    n_options = [x for x in n_options if 150 <= x <= 5000]

    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        if n_try == len(best_a):
            a_try = best_a.copy()
        else:
            a_try = resample_seq(best_a, n_try)
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
            else:
                a_try = np.ones(n_try) / n_try

        # Quick FISTA for this n
        alloc = min(35.0, remaining * 0.2)
        a_opt = fista_optimize(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        # Quick LP if n is small enough and time allows
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20 and n_try <= 2000:
            alloc_lp = min(20.0, remaining2 * 0.3)
            a_opt_lp = lp_alternate(a_opt, time_budget=alloc_lp)
            obj_try_lp = compute_obj(a_opt_lp)
            if obj_try_lp < obj_try:
                a_opt = a_opt_lp
                obj_try = obj_try_lp

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse initializations with quick optimization ===
    print(f"\n=== Phase 6: Diverse Starts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    for d in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_div = diverse_init(n_best, rng)
        alloc = min(25.0, remaining * 0.25)

        # Quick FISTA
        a_opt = fista_optimize(a_div, time_budget=alloc * 0.5)
        obj_try = compute_obj(a_opt)

        # Quick LP if time and n allow
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10 and n_best <= 2000:
            a_opt_lp = lp_alternate(a_opt, time_budget=alloc * 0.5)
            obj_try_lp = compute_obj(a_opt_lp)
            if obj_try_lp < obj_try:
                a_opt = a_opt_lp
                obj_try = obj_try_lp

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** Diverse {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse {d}: obj={obj_try:.8f}")

    # === Phase 7: Final refinement ===
    print(f"\n=== Phase 7: Final Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.5, 80.0)
        a_final = lp_alternate(best_a, time_budget=alloc, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.3, 40.0)
        a_sg = fista_optimize(best_a, time_budget=alloc)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  Final FISTA: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc = min(remaining * 0.4, 30.0)
        a_pv = peak_shave(best_a, time_budget=alloc)
        obj_pv = compute_obj(a_pv)
        if obj_pv < best_obj:
            best_obj = obj_pv
            best_a = a_pv
            print(f"  Final PeakShave: obj={best_obj:.8f} **")

    # One last LP if time permits
    remaining = deadline - time.monotonic()
    if remaining > 15 and len(best_a) <= 2000:
        alloc = min(remaining * 0.6, 40.0)
        a_last = lp_alternate(best_a, time_budget=alloc)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = compute_obj(best_a)

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
