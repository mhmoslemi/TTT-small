import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar, minimize


def compute_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0)


def conv_gradient_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a: grad[m] = 2*a[k-m] if valid."""
    n = len(a)
    grad = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        m_start = k - i_end
        m_end = k - i_start
        grad[m_start:m_end + 1] = 2.0 * a[i_start:i_end + 1][::-1]
    return grad


def subgradient_multi_peak(a, top_k=10):
    """Compute subgradient as average of gradients at top-k peaks."""
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:top_k]
    grad = np.zeros(len(a))
    for k in top_idx:
        grad += conv_gradient_at_k(a, int(k))
    grad /= len(top_idx)
    return grad, conv


def nesterov_subgrad(a0, time_budget=60.0, top_k=10, verbose=False):

    if s < 1e-10:

    a = a / s
    y = a.copy()
    best_a = a.copy()

    prev_a = a.copy()
    prev_grad = None
    patience = 0

    for it in range(500000):
        if time.monotonic() - start > time_budget:
            break

        grad = grad - np.mean(grad)  # Center for simplex

def pairwise_cd(a0, time_budget=100.0, n_pairs_per_iter=50, verbose=False):
    """
    Pairwise coordinate descent: for each pair (i,j), find optimal

    """

    a = a0.astype(np.float64).copy()

        return a0.copy()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    stall = 0

    while time.monotonic() - start < time_budget:
        # Randomly select pairs to update
        indices = rng.choice(n, size=n_pairs_per_iter, replace=False)
        improved = False
        for _ in range(2):  # Two passes per batch
            idx = rng.permutation(n)
            for i in idx:
                j = (i + 1 + rng.integers(0, n - 1)) % n
                if i == j:
                    continue
                # Find optimal δ: move mass from j to i
                # a[i] += δ, a[j] -= δ
                # δ in [0, a[j]]
                a_try = a.copy()
                # Objective as function of δ
                # conv changes: for each k, conv[k] += δ*(a[k-i] - a[k-j]) (approximately)
                # But this is approximate; for exact, we need full recompute
                # Use golden section on [0, a[j]]
                lo = 0.0
                hi = a[j]
                if hi < 1e-15:
                    continue
                
                def obj_of_delta(d):
                    a_try = a.copy()
                    a_try[i] += d
                    a_try[j] -= d
                    if a_try[j] < 0:
                        return np.inf
                    st = float(np.sum(a_try))
                    if st < 1e-10:
                        return np.inf
                    a_try = a_try / st
                    return compute_obj(a_try)
                
                # Bounded minimization
                try:
                    res = minimize_scalar(obj_of_delta, bounds=(0.0, hi), method='bounded',
                                          options={'maxiter': 25, 'xatol': 1e-8})
                    best_d = res.x
                except Exception:
                    continue
                
                if best_d > 1e-10:
                    a[i] += best_d
                    a[j] -= best_d
                    a[j] = max(0, a[j])
                    st = float(np.sum(a))
                    a = a / st
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                    else:
                        # Revert
                        a = best_a.copy()
        
        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0

        if verbose:
            elapsed = time.monotonic() - start
            print(f"      PCD: obj={best_obj:.8f}, t={elapsed:.1f}s")

def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        r = min(n, m - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a_n = a / float(np.sum(a))
    A = build_conv_matrix(a_n)
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                gv = np.maximum(g.value, 0)
                gs = float(np.sum(gv))
                if gs > 1e-10:
                    return gv / gs, float(M.value)
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_budget=100.0, verbose=False):

    n = len(a0)
    a = a0.astype(np.float64).copy()

        if g_lp is None:
            break
        # 1D line search
        def line_obj(t):
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:

            a_try = a_try / st
            return compute_obj(a_try)
        try:

            best_t = res.x if res.success else last_good_t
        except Exception:
            best_t = last_good_t
        a_try = (1.0 - best_t) * a + best_t * g_lp
        a_try = np.maximum(a_try, 0)

            if stall_count > 15:

def resample_seq(a, new_n):
    """Linear interpolation resampling."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def generate_shapes(n):
    """Generate diverse asymmetric shape initializations."""
    shapes = []
    x = np.arange(n) / max(1, n - 1)
    
    # Linear decreasing
    shapes.append(np.maximum(1 - x, 0.01))
    # Exponential decay (various rates)
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        shapes.append(np.exp(-alpha * x))
    # Power law decay
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        shapes.append(np.maximum(1 - x, 0.01) ** p)
    # Asymmetric Gaussian
    for center in [0.2, 0.3, 0.4, 0.5]:
        for width in [0.1, 0.15, 0.2, 0.3]:
            shapes.append(np.exp(-((x - center) / width) ** 2))
    # Linear increasing (mirror of decreasing)
    shapes.append(np.maximum(x, 0.01))
    # Triangle (peak at various positions)
    for peak in [0.2, 0.3, 0.5, 0.7, 0.8]:
        tri = 1 - np.abs(x - peak) / max(peak, 1 - peak)
        shapes.append(np.maximum(tri, 0.01))
    # Sigmoid-like
    for t in [5, 10, 20]:
        shapes.append(1 / (1 + np.exp(-t * (x - 0.5))))
    
    # Normalize all
    for i in range(len(shapes)):
        s = np.sum(shapes[i])
        if s > 1e-10:
            shapes[i] = shapes[i] / s
    return shapes

def pairwise_cd_fast(a0, time_budget=100.0, verbose=False):
    """Fast pairwise coordinate descent using first-order screening."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a0.copy()
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Find the peak
        k_peak = int(np.argmax(conv))
        # Also find top few peaks
        top_k = np.argsort(conv)[::-1][:5]
        
        # For the peak at k_peak, we want to reduce it
        # conv[k_peak] = sum_j a[j]*a[k_peak-j]
        # To reduce it, we want to decrease a[j] for j that contribute a lot
        # and increase a[m] for m that don't contribute
        
        # Identify contributors to the peak
        contrib = np.zeros(n)
        for j in range(max(0, k_peak - n + 1), min(n, k_peak + 1)):
            m = k_peak - j
            if 0 <= m < n:
                contrib[j] += a[m]
                contrib[m] += a[j]
        
        # Top contributors (to decrease) and low contributors (to increase)
        top_contrib = np.argsort(contrib)[::-1][:5]
        low_contrib = np.argsort(contrib)[:5]
        
        improved = False
        for src in top_contrib:
            for dst in low_contrib:
                if src == dst or a[src] < 1e-12:
                    continue
                # Try small transfers
                for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                    delta = frac * a[src]
                    a[src] -= delta
                    a[dst] += delta
                    a = np.maximum(a, 0)
                    st = float(np.sum(a))
                    a = a / st
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-13:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        # Revert
                        a = best_a.copy()
                        break
        
        if not improved:
            # Try random pairs
            for _ in range(20):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[j] < 1e-12:
                    continue
                for frac in [0.01, 0.05, 0.1]:
                    delta = frac * a[j]
                    a[j] -= delta
                    a[i] += delta
                    a = np.maximum(a, 0)
                    st = float(np.sum(a))
                    a = a / st
                    obj_try = compute_obj(a)

                        best_a = a.copy()
                        improved = True
                        break
                    else:

                        break
            if not improved:
                break
        
        if verbose:
            elapsed = time.monotonic() - start
            print(f"      PCD: obj={best_obj:.8f}, t={elapsed:.1f}s")

    return best_a
