from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_lp(a, time_limit=5.0):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

"""
Optimized search for minimum-peak autocorrelation sequences.
Uses aggressive LP alternating, multi-start, perturbation, and coordinate descent.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            sg = float(np.sum(g_val))
            if sg > 1e-10:
                return g_val / sg, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_reduced(a, active_indices):
    """Solve LP with only a subset of constraints."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return None, None
    a = a / s
    A_full = build_conv_matrix(a)
    A = A_full[active_indices, :]
    m = len(active_indices)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            sg = float(np.sum(g_val))
            if sg > 1e-10:
                g_norm = g_val / sg
                # Verify against all constraints
                conv_check = A_full @ g_norm
                true_M = float(np.max(conv_check))
                return g_norm, true_M
    except Exception:
        pass
    return None, None


def conv_line_search(a, g, n_evals=12):
    """
    Find optimal t in [0,1] for a_new = (1-t)*a + t*g.
    Uses the quadratic structure of conv.
    """
    n = len(a)
    sa = float(np.sum(a))
    sg = float(np.sum(g))
    if sa < 1e-10 or sg < 1e-10:
        return 0.0, a.copy()
    a_n = a / sa
    g_n = g / sg

    conv_aa = compute_conv(a_n)
    conv_ag = compute_conv_cross(a_n, g_n)
    conv_gg = compute_conv(g_n)

    def obj_at_t(t):
        val = (1 - t) ** 2 * conv_aa + 2.0 * t * (1 - t) * conv_ag + t ** 2 * conv_gg
        return 2.0 * n * float(np.max(val))

    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = obj_at_t(m1)
        f2 = obj_at_t(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    a_new = (1 - t_opt) * a_n + t_opt * g_n
    a_new = np.maximum(a_new, 0)
    s_new = float(np.sum(a_new))
    if s_new < 1e-10:
        return 0.0, a_n
    return t_opt, a_new / s_new

def coordinate_descent(a, time_budget=60.0, verbose=False):
    """

    n = len(a)

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, float(a[0]))
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def make_binary_init(n, rng):
    """Create a random binary initialization with good density."""
    p = rng.uniform(0.3, 0.7)
    a = (rng.random(n) < p).astype(np.float64)
    if np.sum(a) < 5:
        a = (rng.random(n) < 0.5).astype(np.float64)
    return a


def make_costas_like(n, rng):
    """Create a Costas-array-inspired initialization."""
    a = np.zeros(n)
    # Try a few prime-based constructions
    for _ in range(3):
        p = int(rng.integers(3, max(4, n // 3)))
        # Golay Costas array: x_i = i, y_i = i^2 mod p
        support = set()
        for i in range(1, p):
            pos = i ** 2 % p
            if pos < n:
                support.add(pos)
        # Also try the dual
        for i in range(1, p):
            pos = i
            if pos < n:
                support.add(pos)
        if len(support) > n // 4:
            for pos in support:
                a[pos] = 1.0
            break
    if np.sum(a) < 5:
        a = make_binary_init(n, rng)
    return a


def make_chirp_init(n, rng):
    """Create a chirp-like initialization."""
    x = np.linspace(0, 1, n)
    freq = rng.uniform(1, 5)
    a = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x)
    a = np.maximum(a, 0)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start LP alternating + perturbation + coordinate descent + multi-n.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    n_parent = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                n_parent = len(parent)
                best_score = compute_obj(parent)
                best_seq = parent.copy()
                print(f"Parent (n={n_parent}): obj={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        a = np.ones(n_parent) / n_parent
        best_score = compute_obj(a)
        best_seq = a.copy()
        print(f"Default uniform init (n={n_parent}): obj={best_score:.8f}")

    print(f"\nStarting search. Initial best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 1: Aggressive LP from parent ===
    remaining = deadline - time.monotonic()
    phase1_budget = min(350.0, remaining * 0.30)
    print(f"\n=== Phase 1: LP Alternating from parent (budget={phase1_budget:.0f}s) ===")
    a_opt = lp_alternating(best_seq.copy(), time_budget=phase1_budget,
                           perturbation_interval=25, verbose=True)
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  Phase 1 improved: {best_score:.8f}")
    else:
        print(f"  Phase 1 no improvement: {obj_opt:.8f}")

    # === Phase 2: Multi-start with diverse initializations ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        n_inits = min(6, max(3, int(remaining / 80)))
        phase2_budget = remaining * 0.35
        per_init_budget = phase2_budget / n_inits
        print(f"\n=== Phase 2: Multi-start ({n_inits} inits, {per_init_budget:.0f}s each) ===")

        inits = []
        # Diverse initializations at the best n
        n_cur = len(best_seq)
        inits.append(('parent', best_seq.copy()))
        inits.append(('binary1', make_binary_init(n_cur, rng)))
        inits.append(('binary2', make_binary_init(n_cur, np.random.default_rng(seed + 1))))
        inits.append(('costas', make_costas_like(n_cur, rng)))
        inits.append(('chirp', make_chirp_init(n_cur, rng)))
        inits.append(('random', np.maximum(rng.random(n_cur), 0)))

        for name, a_init in inits[:n_inits]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            t_alloc = min(per_init_budget, remaining * 0.3)
            if t_alloc < 10:
                break
            s_init = float(np.sum(a_init))
            if s_init < 0.01:
                continue
            a_init = a_init / s_init
            obj_init = compute_obj(a_init)
            print(f"  Init '{name}' (obj={obj_init:.6f}): running LP...")
            a_r = lp_alternating(a_init, time_budget=t_alloc, perturbation_interval=20)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"    ** Improved to {best_score:.8f} **")
            else:
                print(f"    Result: {obj_r:.8f}")

    # === Phase 3: Perturbation and re-optimization from best ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        phase3_budget = min(300.0, remaining * 0.30)
        n_perts = min(8, max(3, int(phase3_budget / 40)))
        per_pert = phase3_budget / n_perts
        print(f"\n=== Phase 3: Perturbation ({n_perts} attempts, {per_pert:.0f}s each) ===")

        for p in range(n_perts):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            t_alloc = min(per_pert, (remaining - 10) * 0.9)
            if t_alloc < 10:
                break

            # Different perturbation strategies
            a_pert = best_seq.copy()
            n_p = len(a_pert)
            strategy = p % 4

            if strategy == 0:
                # Zero out random block
                block_len = max(5, n_p // 8)
                start_idx = rng.integers(0, max(1, n_p - block_len))
                a_pert[start_idx:start_idx + block_len] = 0.0
            elif strategy == 1:
                # Multiply random subset by small factor
                n_sub = max(5, n_p // 6)
                idx = rng.choice(n_p, size=n_sub, replace=False)
                a_pert[idx] *= rng.uniform(0.01, 0.1, n_sub)
            elif strategy == 2:
                # Add sinusoidal perturbation
                a_pert = a_pert + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * np.linspace(0, 1, n_p))
                a_pert = np.maximum(a_pert, 0)
            else:
                # Random rescale of random half
                half = n_p // 2
                idx = rng.choice(n_p, size=half, replace=False)
                a_pert[idx] *= rng.uniform(0.5, 2.0)

            s_p = float(np.sum(a_pert))
            if s_p < 0.01:
                continue
            a_pert = a_pert / s_p
            obj_p = compute_obj(a_pert)

            a_r = lp_alternating(a_pert, time_budget=t_alloc, perturbation_interval=15)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Pert {p} (strategy={strategy}): ** {best_score:.8f} **")
            else:
                print(f"  Pert {p} (strategy={strategy}): {obj_r:.8f} (init was {obj_p:.6f})")

    # === Phase 4: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_seq)
        n_options = sorted(set([
            n_best - 100, n_best - 50, n_best - 25,
            n_best + 25, n_best + 50, n_best + 100,
            500, 600, 700, 800, 900, 1000, 1200
        ]))
        n_options = [n for n in n_options if 300 <= n <= 2000 and n != n_best]
        # Limit to at most 8 options
        n_options = n_options[:8]

        phase4_budget = min(200.0, remaining * 0.25)
        per_n = max(15, phase4_budget / max(1, len(n_options)))
        print(f"\n=== Phase 4: Multi-n exploration ({len(n_options)} values, ~{per_n:.0f}s each) ===")

        for n_try in n_options:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            t_alloc = min(per_n, (remaining - 10) * 0.8)
            if t_alloc < 10:
                break

            a_try = resample_sequence(best_seq, n_try)
            obj_init = compute_obj(a_try)
            a_r = lp_alternating(a_try, time_budget=t_alloc, perturbation_interval=15)
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  n={n_try}: ** {best_score:.8f} **")
            else:
                print(f"  n={n_try}: {obj_r:.8f} (init={obj_init:.6f})")

    # === Phase 5: Final polish ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Final Polish (remaining={remaining:.0f}s) ===")

    if remaining > 30:
        # Coordinate descent
        cd_budget = min(60.0, remaining * 0.4)
        a_cd = coordinate_descent(best_seq.copy(), time_budget=cd_budget, verbose=False)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: ** {best_score:.8f} **")
        else:
            print(f"  CD: {obj_cd:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        # Final LP
        lp_budget = min(40.0, remaining * 0.4)
        a_lp = lp_alternating(best_seq.copy(), time_budget=lp_budget, perturbation_interval=10)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: ** {best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        # Smooth max descent
        sm_budget = min(30.0, remaining * 0.5)
        a_sm = smooth_max_descent(best_seq.copy(), time_budget=sm_budget)
        obj_sm = compute_obj(a_sm)
        if obj_sm < best_score:
            best_score = obj_sm
            best_seq = a_sm.copy()
            print(f"  SmoothMax: ** {best_score:.8f} **")

    # === Final validation ===
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
