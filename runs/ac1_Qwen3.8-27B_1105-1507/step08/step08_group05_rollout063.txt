from scipy.fft import rfft, irfft
import time


def compute_conv_self(a):
    """conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """conv(a,b) using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

    n = len(a)

def build_conv_matrix(a):
    """Build (2n-1) x n matrix where row k is [a[k-i] for i in valid range]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for i in range(n):
        k_start = i
        k_end = min(i + n, m)
        A[k_start:k_end, i] = a[:k_end - k_start]

    """Project v onto {x >= 0, sum(x) = 1}."""

    i = np.max(np.where(cond)[0]) + 1

    return np.maximum(v - theta, 0)

def _fft_conv_self(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def _fft_conv_cross(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

    an = a / s

    return 2.0 * n * float(np.max(c))
