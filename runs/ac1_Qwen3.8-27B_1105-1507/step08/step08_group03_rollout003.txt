from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def grad_conv_at_peak(a, k):
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g

    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A

def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M, dual_vars) or (None, None, None)."""
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)
    
    # Use scipy linprog
    # Variables: g[0..n-1], M
    # Minimize M
    # s.t. A@g - M <= 0 (i.e., A@g <= M)
    #      sum(g) = 1
    #      g >= 0, M >= 0
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            g_val = g_val / s
            M_val = float(res.x[n])
            # Get dual variables if available
            duals = None
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                if hasattr(res.ineqlin, 'mul') and res.ineqlin.mul is not None:
                    duals = np.abs(res.ineqlin.mul)
            return g_val, M_val, duals
    return None, None, None

def golden_section_line_search(a, g, n, iters=30):
    """Find optimal t in [0,1] minimizing f((1-t)*a + t*g).
    Uses golden section search since f(t) is convex (max of quadratics)."""
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    lo = 0.0
    hi = 1.0
    
    # Initial points
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    
    a_c = (1 - c) * a + c * g
    a_d = (1 - d) * a + d * g
    
    fc = compute_obj(a_c)
    fd = compute_obj(a_d)
    
    for _ in range(iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            a_c = (1 - c) * a + c * g
            fc = compute_obj(a_c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            a_d = (1 - d) * a + d * g
            fd = compute_obj(a_d)
    
    t_opt = (lo + hi) / 2
    return t_opt

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Accelerated Frank-Wolfe (AFW) with adaptive line search,
softmax subgradient annealing, and multi-n exploration.
Key improvements over previous: AFW convergence (1/t^2 vs 1/t),
golden-section line search, and dual-informed restarts.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_conv_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] = a[j_start:j_end + 1][::-1]
    return g


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None


def line_search_two_stage(a, g, n_evals_coarse=15, n_evals_fine=10):
    """Two-stage line search: coarse grid then fine grid around best."""
    # Coarse grid
    ts_coarse = np.linspace(0.01, 1.0, n_evals_coarse)
    best_t = ts_coarse[0]
    best_obj = np.inf
    for t in ts_coarse:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    # Fine grid around best
    delta = 1.0 / n_evals_coarse
    ts_fine = np.linspace(max(0.01, best_t - delta), min(1.0, best_t + delta), n_evals_fine)
    for t in ts_fine:
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = float(np.sum(a_try))
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    return best_t


def accelerated_frank_wolfe(a, time_budget=200.0, verbose=False):
    """
    Accelerated Frank-Wolfe (AFW) for minimizing max(conv(a,a)) on the simplex.
    Uses extrapolation for 1/t^2 convergence rate.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = a.copy()
    y = a.copy()
    t_iter = 1.0
    stall = 0
    patience = 100
    
    it = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break
        
        # Solve LP at extrapolated point y
        s_dir, M_lp = solve_lp_minpeak(y)
        if s_dir is None:
            break

        t_opt = line_search_two_stage(x, s_dir)
        
        x_new = (1.0 - t_opt) * x + t_opt * s_dir
        x_new = np.maximum(x_new, 0)
        ss = float(np.sum(x_new))
        if ss < 1e-10:
            break
        x_new = x_new / ss
        
        obj_new = compute_obj(x_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Restart from best
                x = best_a.copy()
                y = best_a.copy()
                t_iter = 1.0
                stall = 0
                patience = int(patience * 1.5)
                it += 1
                continue
        
        # AFW extrapolation
        gamma = t_iter / (t_iter + 2.0)
        y_new = x_new + gamma * (x_new - x)
        y_new = project_simplex(y_new)
        
        x = x_new
        y = y_new
        t_iter += 1.0
        it += 1
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    AFW it{it}: obj={best_obj:.8f}, t={t_opt:.4f},"
                  f" M_lp={M_lp:.6f}, t_iter={t_iter:.0f}, el={elapsed:.1f}s")
    
    return best_a


def softmax_subgradient(a, time_budget=100.0, T_start=50.0, T_end=1.0,
                        lr_start=1.0, verbose=False):
    """
    Subgradient method with softmax weighting and temperature annealing.
    Uses multiplicative (mirror descent) updates for better conditioning.
    """
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    log_a = np.log(a)
    lr = lr_start / np.sqrt(n)
    momentum = 0.0
    v = np.zeros(n)
    stall = 0
    
    total_iters_estimate = 2000
    it = 0
    while time.monotonic() - start < time_budget:
        # Anneal temperature
        progress = min(1.0, it / total_iters_estimate)
        T = T_start * (T_end / T_start) ** progress
        
        conv = compute_conv(a)
        max_c = float(conv.max())
        
        if max_c < 1e-15:
            break
        
        # Softmax weights
        z = (conv - max_c) / (T * max_c + 1e-15)
        w = np.exp(z)
        w_sum = w.sum()
        w = w / w_sum
        
        # Compute weighted gradient
        g = np.zeros(n)
        # Only compute gradient for significant weights
        threshold = 1e-6
        active = w > threshold
        for k in np.where(active)[0]:
            g += w[k] * grad_conv_at_peak(a, int(k))
        
        # Momentum
        v = momentum * v + g
        momentum = min(0.7, momentum + 0.001)
        
        # Mirror descent step
        log_a_new = log_a - lr * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            log_a = np.log(np.maximum(a, 1e-15))
            lr *= 0.999
            stall = 0
        else:
            stall += 1
            lr *= 0.98
            if stall > 200:
                # Escape: perturbation
                rng = np.random.default_rng(it * 104729 + n)
                a = best_a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                log_a = np.log(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)
                momentum = 0.0
                lr = lr_start / np.sqrt(n)
                stall = 0
        
        it += 1
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    Softmax it{it}: obj={best_obj:.8f}, T={T:.2f},"
                  f" lr={lr:.6f}, el={elapsed:.1f}s")
    
    return best_a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Sinc-like (broadband)
        x = np.arange(n) - n / 2
        a = np.sinc(x / (n * 0.3))
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    elif mode == 3:
        # Chi distribution-like
        a = rng.chisquare(df=n, size=n)
        a = a / a.sum()
    elif mode == 4:
        # Exponential decay from center
        x = np.arange(n)
        c = n / 2
        a = np.exp(-np.abs(x - c) / (n * 0.2))
        a = a / a.sum()
    elif mode == 5:
        # Product of sines (Barker-like)
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * x) * np.sin(2 * np.pi * x) * np.sin(3 * np.pi * x)) + 0.01
        a = a / a.sum()
    elif mode == 6:
        # Random Fourier
        freqs = rng.uniform(0, 5, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.2 * np.cos(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
