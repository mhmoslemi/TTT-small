def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_conv_gradient(a, active_indices):
    """
    Compute gradient of sum_{k in active} b[k] with respect to a.
    db[k]/da[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    Returns gradient vector of length n.
    """
    n = len(a)
    grad = np.zeros(n)
    for k in active_indices:
        # db[k]/da[j] = 2*a[k-j] for j where 0 <= k-j < n
        # i.e., j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            grad[j_start:j_end+1] += 2.0 * a[k - np.arange(j_start, j_end+1)]
    return grad


def compute_conv_gradient_fft(a, weights):
    """
    Compute weighted gradient: sum_k weights[k] * grad b[k]
    using FFT for efficiency.
    weights has length 2n-1.
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    # grad_j = sum_k weights[k] * 2 * a[k-j] * I(0<=k-j<n)

    # 
    # Let's define w[k] = weights[k] for k=0..2n-2, 0 otherwise.
    # Then grad[j] = 2 * sum_{k: 0<=k-j<n} w[k] * a[k-j]
    # = 2 * sum_{m: 0<=m<n} w[j+m] * a[m]
    # = 2 * (w * a)_j (cross-correlation of w and a, evaluated at j)
    
    # This is a cross-correlation! We can compute it with FFT.
    fa = rfft(a, size)
    fw = rfft(weights, size)
    # Cross-correlation: sum_m w[j+m]*a[m] = corr(w, a) at lag j
    # corr(w,a)[j] = sum_m w[j+m]*a[m]
    # This is the same as the convolution of w(reversed) and a, or we can use the FFT trick.
    # 
    # Actually: sum_m w[j+m]*a[m] = (a * w_flipped)[j + n - 1] ... let me think differently.
    # 
    # Let's define: h[j] = sum_{m=0}^{n-1} a[m] * w[j+m]
    # This is the cross-correlation of a and w at lag... 
    # h[j] = sum_m a[m] * w[m+j] = (a ⋆ w)[j] (cross-correlation)
    # 
    # Cross-correlation can be computed as: ifft(conj(fft(a)) * fft(w))
    # But we need to be careful with the indexing.
    # 
    # Actually, let me just use the direct formula with FFT:
    # h = ifft(conj(fft(a, size)) * fft(w, size)) gives circular cross-correlation
    # But we need linear cross-correlation.
    
    # Simpler approach: use the fact that h[j] = sum_m a[m] * w[j+m]
    # This is the (j)th element of the cross-correlation of a and w.
    # Using FFT: cross_corr = ifft(conj(fft(a)) * fft(w))
    # The linear cross-correlation of a (length n) and w (length 2n-1) has length n + 2n-1 - 1 = 3n-2
    # But we only need h[0], ..., h[n-1].
    
    # Let me just use a different approach: 
    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[j+m]
    # where w has length 2n-1 (w[k] = weights[k]).
    # 
    # This is the convolution of a(reversed) with w, evaluated at specific positions.
    # Or equivalently, it's a matrix-vector product with a specific structure.
    
    # For simplicity and correctness, let me use the direct approach with numpy:
    # h[j] = sum_{m=0}^{n-1} a[m] * w[j+m] for j = 0, ..., n-1
    # where w has length 2n-1.

    # Pad a to length size, pad w to length size
    # h_circ = ifft(fft(a) * conj(fft(w_rev))) ... this is getting complicated.
    
    # Let me just use a Toeplitz matrix approach for moderate n:
    # Actually for n up to ~2000, the direct computation is O(n^2) which is fine.
    
    # Direct computation (O(n^2) but with good constant):
    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * weights[j+m] for j+m < 2n-1
    
    # Better: use the fact that this is a convolution-like operation
    # Let's define the "anti-diagonal" sum.
    # Create a (2n-1) x n matrix M where M[k,j] = 1 if k-j = m for some m... 
    # No, let me just do it directly with broadcasting for n <= 2000.
    
    # For large n, we can use FFT:
    # h[j] = sum_m a[m] * w[j+m] = (a_reversed * w)[n-1+j] (linear convolution)

    a_rev = a[::-1]
    conv_result = np.convolve(a_rev, weights, mode='full')

def lp_alternating_improved(a, time_budget=200.0, verbose=False):

        A[j:j + r, j] = a[:r]

    prev_a = None
    prev2_a = None
    stall = 0

    while time.monotonic() - start < time_budget:

                    z_val = np.maximum(z.value, 0)
                    zs = float(np.sum(z_val))

                        solved = True
                        break
            except:
                continue
        
        if not solved:

        best_mix = 0.5

                a = np.maximum(a, 0)
                a = a / a.sum()
                best_obj = compute_obj(a)
                best_a = a.copy()
                prev_a = None
                prev2_a = None

def power_sum_optimize(a, time_budget=200.0, p_start=10, p_end=100, verbose=False):
    """
    Minimize sum_k b[k]^p using gradient descent with momentum.
    p increases over time to approach the minimax problem.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    p = p_start
    momentum_v = np.zeros(n)
    lr = 0.1
    beta = 0.9  # momentum coefficient

    iteration = 0
    while time.monotonic() - start < time_budget:
        # Increase p over time
        fraction = min(1.0, (time.monotonic() - start) / time_budget)
        p = p_start + (p_end - p_start) * fraction

        # Compute convolution
        b = compute_conv(a)
        
        # Compute gradient of sum b[k]^p
        # db[k]/da[j] = 2*a[k-j] if 0<=k-j<n
        # grad[j] = sum_k p*b[k]^(p-1) * 2*a[k-j] * I(0<=k-j<n)
        # Use FFT-based computation
        weights = p * b ** (p - 1)
        grad = compute_conv_gradient_fft(a, weights)

        # Momentum update
        momentum_v = beta * momentum_v + (1 - beta) * grad
        a_new = a - lr * momentum_v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Decay learning rate slowly
            lr *= 0.999
        else:
            # Try smaller step
            a_try = a - 0.1 * lr * momentum_v
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            if compute_obj(a_try) < best_obj:
                a = a_try
                best_a = a.copy()
                best_obj = compute_obj(a)
            lr *= 0.95

        iteration += 1
        if verbose and iteration % 100 == 0:
            print(f"    Power-sum: p={p:.1f}, obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def nesterov_subgradient(a, time_budget=100.0, verbose=False):
    """
    Nesterov-accelerated subgradient method for the minimax problem.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()  # Nesterov look-ahead point
    v = np.zeros(n)  # momentum
    beta = 0.9
    lr = 0.05

    iteration = 0
    while time.monotonic() - start < time_budget:
        # Compute subgradient at y
        b_y = compute_conv(y)
        max_val = float(np.max(b_y))
        # Find active set (positions where b is close to max)
        active = np.where(b_y >= max_val * 0.999)[0]
        
        # Subgradient: average of gradients at active positions
        grad = compute_conv_gradient_fft(y, np.ones(len(b_y)) * (1.0 / len(active))[np.zeros(len(b_y), dtype=int)])
        # Actually, let me compute it properly
        # grad[j] = sum_{k in active} (1/|active|) * 2 * y[k-j] * I(0<=k-j<n)
        # = (2/|active|) * sum_{k in active} y[k-j] * I(0<=k-j<n)
        
        # Using the FFT trick:
        # h[j] = (1/|active|) * sum_{k in active} y[k-j] * I(0<=k-j<n)
        # Define w[k] = 1/|active| if k in active else 0
        w = np.zeros(2*n-1)
        w[active] = 1.0 / len(active)
        h = compute_conv_gradient_fft(y, w)  # This gives 2*h
        grad = h  # Already has the factor of 2

        # Nesterov update
        v = beta * v + grad
        y_new = y - lr * v
        y_new = np.maximum(y_new, 0)
        y_new = y_new / np.sum(y_new)

        # Weighted average (Nesterov)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (iteration+1)**2)) / 2.0
        alpha_k = (1.0 + t_k - 1) / (1.0 + t_k)  # = (t_k - 1 + 1) / (t_k + 1) ... let me use standard formula
        # Standard Nesterov: x_{k+1} = (t_k * x_k + (1-t_k) * y_{k+1}) / (t_k + 1)
        # where t_{k+1} = (1+sqrt(1+4*t_k^2))/2
        # Simpler: use the "heavy ball" version
        a_new = 0.5 * y_new + 0.5 * a  # Simple averaging as momentum
        a_new = np.maximum(a_new, 0)
        a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            y = a_new
        else:
            lr *= 0.98
            y = a  # Reset look-ahead

        # Decaying learning rate
        lr = 0.05 / (1.0 + 0.01 * iteration)
        iteration += 1

    return best_a


def coordinate_descent_polish(a, time_budget=50.0):
    """
    Coordinate descent: try adjusting each coordinate.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    max_passes = 200
    for it in range(max_passes):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        # Randomized coordinate order
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - start > time_budget:
                break
            # Try multiple step sizes
            for frac in [0.01, 0.02, 0.05, 0.1]:
                a_try = a.copy()
                delta = frac * a[j]
                # Move mass from j to a random other coordinate
                if rng.random() < 0.5:
                    # Decrease a[j], increase random other
                    a_try[j] -= delta
                    k = rng.integers(0, n)
                    a_try[k] += delta
                else:
                    # Increase a[j], decrease random other
                    k = rng.integers(0, n)
                    if k == j:
                        continue
                    delta2 = frac * a[k]
                    a_try[k] -= delta2
                    a_try[j] += delta2
                
                if np.any(a_try < 0):
                    continue
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        
        if not improved and it > 50:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def compute_weighted_gradient(a, weights):
    """
    Compute grad[j] = 2 * sum_k weights[k] * a[k-j] * I(0<=k-j<n)

        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n])
    except Exception:
        pass

    return None, None


def lp_alternating(a, time_budget=200.0, verbose=False):
    """LP alternating with line search and 2-cycle detection."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    prev_a = None
    prev2_a = None
    stall = 0

    while time.monotonic() - start < time_budget:
        z_val, M_lp = solve_lp_for_direction(a)
        if z_val is None:
            break

        # Line search over mixing ratios
        best_mix_obj = np.inf
        best_a_try = a.copy()
        for mix in [0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 1.0]:
            a_try = (1.0 - mix) * a + mix * z_val
            a_try = np.maximum(a_try, 0)
            st = float(np.sum(a_try))
            if st < 1e-10:
                continue
            a_try = a_try / st
            obj_try = compute_obj(a_try)
            if obj_try < best_mix_obj:
                best_mix_obj = obj_try
                best_a_try = a_try.copy()

        if best_mix_obj < best_obj - 1e-13:
            # 2-cycle detection
            if prev2_a is not None and np.max(np.abs(a - prev2_a)) < 5e-8:
                a = 0.5 * (a + prev_a)
                a = np.maximum(a, 0)
                a = a / a.sum()
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                prev_a = None
                prev2_a = None
                stall = 0
            else:
                prev2_a = prev_a
                prev_a = a.copy()
                a = best_a_try.copy()
                best_obj = best_mix_obj
                best_a = a.copy()
                stall = 0
        else:
            stall += 1
            if stall > 12:
                rng = np.random.default_rng(int(time.monotonic() * 1e7) % (2**31))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                prev_a = None
                prev2_a = None
                stall = 0

    return best_a


def power_sum_optimize(a, time_budget=200.0, p_start=10, p_end=200, verbose=False):
    """
    Minimize sum_k b[k]^p using gradient descent with momentum.
    p increases over time to approach the minimax objective.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum_v = np.zeros(n)
    lr = 0.05
    beta_mom = 0.85
    stall = 0

    while time.monotonic() - start < time_budget:
        # Increase p based on time fraction
        fraction = min(1.0, (time.monotonic() - start) / time_budget)
        p = p_start + (p_end - p_start) * fraction

        b = compute_conv(a)

        # Compute gradient of sum b[k]^p
        # Avoid numerical issues: clip b to [0, 1]
        b_clipped = np.clip(b, 0, 1.0)
        weights = p * np.power(b_clipped, p - 1)

        if np.max(weights) < 1e-15:
            break

        grad = compute_weighted_gradient(a, weights)

        # Momentum update
        momentum_v = beta_mom * momentum_v + (1.0 - beta_mom) * grad
        a_new = a - lr * momentum_v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            lr *= 0.9995
        else:
            stall += 1
            if stall > 20:
                lr *= 0.5
                momentum_v *= 0.5
                stall = 0
                if lr < 1e-6:
                    break
            else:
                # Try smaller step
                a_try = a - 0.1 * lr * momentum_v
                a_try = np.maximum(a_try, 0)
                st = float(np.sum(a_try))
                if st > 1e-10:
                    a_try = a_try / st
                    if compute_obj(a_try) < best_obj:
                        a = a_try
                        best_a = a.copy()
                        best_obj = compute_obj(a)

        if verbose and int((time.monotonic() - start) * 2) % 10 == 0:
            print(f"    PS: p={p:.0f}, obj={best_obj:.8f}, lr={lr:.6f}, t={time.monotonic()-start:.0f}s")

    return best_a


def nesterov_subgradient(a, time_budget=100.0, verbose=False):
    """
    Nesterov-accelerated subgradient method for max_k b[k].
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    v = np.zeros(n)
    beta_mom = 0.9
    iteration = 0

    while time.monotonic() - start < time_budget:
        b_y = compute_conv(y)
        max_val = float(np.max(b_y))
        if max_val < 1e-15:
            break

        # Active set: positions where b is within 1% of max
        threshold = max_val * 0.998
        active = np.where(b_y >= threshold)[0]
        if len(active) == 0:
            active = np.argmax(b_y)
            active = np.array([active])

        # Subgradient: weighted average of gradients at active positions
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        grad = compute_weighted_gradient(y, w)

        # Momentum
        v = beta_mom * v + (1.0 - beta_mom) * grad

        # Step size: diminishing
        lr = 0.03 / (1.0 + 0.005 * iteration)

        y_new = y - lr * v
        y_new = np.maximum(y_new, 0)
        s_new = float(np.sum(y_new))
        if s_new < 1e-10:
            y = a
            continue
        y_new = y_new / s_new

        # Nesterov combination
        t_k = np.sqrt(1.0 + 4.0 * (iteration + 1) ** 2) / 2.0 + 0.5
        alpha = (1.0 + t_k) / (2.0 + t_k)
        a_new = alpha * a + (1.0 - alpha) * y_new
        a_new = np.maximum(a_new, 0)
        s_new2 = float(np.sum(a_new))
        if s_new2 > 1e-10:
            a_new = a_new / s_new2

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            y = a_new
        else:
            y = 0.5 * y + 0.5 * a

        iteration += 1
        if verbose and iteration % 50 == 0:
            print(f"    NS: it={iteration}, obj={best_obj:.8f}, t={time.monotonic()-start:.0f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def pairwise_polish(a, time_budget=40.0):
    """Pairwise exchange polish."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    no_improve = 0
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        for _ in range(10):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for frac in [0.01, 0.03, 0.05, 0.1]:
                delta = frac * min(a[i], a[j])
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0 or a_try[j] < 0:
                    continue
                a_try = a_try / np.sum(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    return best_a


def peak_balancing_step(a, num_steps=50):
    """
    Targeted peak balancing: identify the largest peaks and apply
    corrections specifically designed to reduce them.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)

    for step in range(num_steps):
        b = compute_conv(a)
        max_val = float(np.max(b))
        # Find top peaks
        top_k = max(3, len(b) // 50)
        peak_indices = np.argsort(b)[-top_k:]

        # For each peak, compute the gradient and apply a small correction
        w = np.zeros(2 * n - 1)
        w[peak_indices] = 1.0 / len(peak_indices)
        grad = compute_weighted_gradient(a, w)

        # Apply correction
        eta = 0.01
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Try smaller step
                a_try = a - 0.001 * grad
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try

    return best_a
