def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v / target_sum)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]

    rho = rho[-1]

    return np.maximum(v - theta, 0.0)

        a = a_new
        it += 1
        
        # Adaptive learning rate
        if it % 100 == 0:
            lr = max(1e-10, lr * 0.998)
        
        # Stuck detection
        if it - best_it > patience:
            a = best_a.copy()
            v = np.zeros(n)
            lr = max(lr * 0.5, 1e-10)
            best_it = it
            patience = max(patience // 2, 5000)
    
    return best_a


def spectral_flatten(a, time_budget=60.0, flatten_strength=0.3, verbose=False):
    """
    Spectral flattening via iterative domain switching (Gerchberg-Saxton style).
    Aims to spread power spectrum energy away from DC to flatten autocorrelation.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget:
        # Go to frequency domain
        A = fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Flatten: reduce peak of power spectrum toward mean
        # Use a soft clipping approach
        target_mag = np.full(N, np.sqrt(np.mean(mag**2)))
        # Blend: push toward flat
        new_mag = (1.0 - flatten_strength) * mag + flatten_strength * target_mag
        # Ensure DC is not zero (maintain sum)
        new_mag[0] = mag[0] * 0.9 + target_mag[0] * 0.1
        
        # Also try a different flattening: reduce the top quantile
        top_threshold = np.percentile(mag[mag > 0], 70) if np.any(mag > 0) else 0
        mask = mag > top_threshold
        new_mag[mask] = (1.0 - flatten_strength * 0.5) * mag[mask] + flatten_strength * 0.5 * top_threshold
        
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, N))[:n]
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Don't update a, try different strength
            flatten_strength *= 0.95
        
        it += 1
        if it % 100 == 0 and verbose:
            print(f"    Spectral it{it}: obj={best_obj:.8f}, strength={flatten_strength:.3f}, t={time.monotonic()-start:.0f}s")
    
    return best_a


def peak_chase(a, time_budget=120.0, verbose=False):
    """
    Direct peak-chasing: iteratively identify the worst peak and reduce it
    by redistributing mass from contributing coordinates to non-contributing ones.
    """
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.05
    it = 0
    stall = 0
    
    rng = np.random.default_rng(777)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = conv[k_star]
        
        # Find contributing indices: i where i and k_star-i are both in [0, n-1]
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        
        if i_start > i_end:
            break
        
        contributing = slice(i_start, i_end + 1)
        contrib_a = a[contributing]
        
        # Reduce contributing elements proportionally
        if float(np.sum(contrib_a)) < 1e-15:
            break
        
        # How much to reduce
        reduction = lr * contrib_a
        
        # Where to put the mass: non-contributing positions
        non_contrib_mask = np.ones(n, dtype=bool)
        non_contrib_mask[i_start:i_end+1] = False
        
        if not np.any(non_contrib_mask):
            break
        
        # Distribute to non-contributing positions weighted by current values
        # (to avoid creating new peaks)
        non_contrib_a = a[non_contrib_mask]
        weights = non_contrib_a + 1e-10
        weights = weights / weights.sum()
        
        a_new = a.copy()
        a_new[contributing] -= reduction
        a_new = np.maximum(a_new, 0)
        
        # Add mass to non-contributing positions
        total_reduced = float(np.sum(reduction))
        a_new[non_contrib_mask] += total_reduced * weights
        
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.7
                # Add random perturbation
                a = a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                if lr < 1e-6:
                    break
            elif stall > 10:
                lr *= 0.9
        
        it += 1
        if it % 50 == 0 and verbose:
            print(f"    PeakChase it{it}: obj={best_obj:.8f}, k*={k_star}, peak={peak_val:.6f}, lr={lr:.4f}")
    
    return best_a


def subgradient_refine(a, time_budget=100.0, K=8, verbose=False):
    """Subgradient method with top-K peaks, momentum, and adaptive step on simplex."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    lr = 3.0 / np.sqrt(n)
    momentum = 0.8
    v = np.zeros(n)
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_idx = np.argsort(conv)[::-1][:K]
        
        peak_vals = conv[top_idx]
        max_pv = float(peak_vals[0])
        if max_pv < 1e-15:
            break
        weights = np.exp((peak_vals - max_pv) * 8.0 / max_pv)
        weights /= weights.sum()
        
        g = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j_start = k - i_end
                j_end = k - i_start
                g[i_start:i_end + 1] += w * a[j_start:j_end + 1][::-1]
        
        v = momentum * v + g
        a_new = a - lr * v
        a_new = project_simplex(a_new, target_sum=1.0)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            else:
                lr *= 0.99
        
        it += 1
        if it % 200 == 0 and verbose:
            print(f"    SG it{it}: obj={best_obj:.8f}, lr={lr:.6f}")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    A = build_conv_matrix(a_norm)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    solvers = [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                s = float(np.sum(g_val))
                if s > 1e-10:
                    return g_val / s, float(M.value)
        except Exception:
            continue
    
    return None, None


def lp_polish(a, time_budget=100.0, verbose=False):
    """LP-based polishing with adaptive blending and perturbation."""
    n = len(a)
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    iter_count = 0
    last_t = 0.3
    rng = np.random.default_rng(54321)
    
    while time.monotonic() - start < time_budget:
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break
        
        candidates = sorted(set([
            0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0,
            max(0.05, last_t * 0.5), min(1.0, last_t * 1.5),
            max(0.05, last_t * 0.7), min(1.0, last_t * 1.3),
            max(0.05, last_t * 1.1), min(1.0, last_t * 0.9)
        ]))
        
        best_t_obj = best_obj
        best_t_val = last_t
        best_a_local = best_a.copy()
        
        for t in candidates:
            a_try = (1.0 - t) * a + t * g_lp
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_val = t
                best_a_local = a_try.copy()
        
        iter_count += 1
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a
            last_t = best_t_val
            if verbose and iter_count % 2 == 0:
                print(f"    LP it{iter_count}: obj={best_obj:.8f}, t={best_t_val:.2f}, M={M_lp:.6f}")
        else:
            scale = 0.01 + 0.005 * min(iter_count, 15)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s < 1e-10:
                break
            a = a / s
            if iter_count > 25:
                break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Anti-symmetric: high on left, low on right
        x = np.arange(n) / n
        a = np.maximum(0, 1.0 - 2.0 * np.abs(x - 0.3)) + 0.1
        a = a / a.sum()
    elif mode == 3:
        # Two peaks
        c1 = n // 4
        c2 = 3 * n // 4
        w = max(1, n // 10)
        a = np.zeros(n)
        a += np.exp(-((np.arange(n) - c1) ** 2) / (2 * w ** 2))
        a += np.exp(-((np.arange(n) - c2) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 4:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** rng.uniform(-1.5, 1.5)
        a = a / a.sum()
    elif mode == 5:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = 0.3 + 0.7 * np.cos(x) ** 2
        a = a / a.sum()
    elif mode == 6:
        # Zigzag
        a = np.abs(np.sin(np.arange(n) * np.pi / n * rng.uniform(2, 8))) + 0.05
        a = a / a.sum()
    elif mode == 7:
        # Random smooth
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.5 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    else:
        a = rng.uniform(0.1, 1, n)
        a = a / a.sum()
    return a


def simulated_annealing_step(a, best_a, best_obj, temperature, rng):
    """One step of simulated annealing: perturb, evaluate, accept/reject."""
    n = len(a)
    scale = 0.05 * temperature
    a_perturbed = a * np.exp(rng.normal(0, scale, n))
    a_perturbed = np.maximum(a_perturbed, 0)
    s = float(np.sum(a_perturbed))
    if s < 1e-10:
        return best_a, best_obj, a
    a_perturbed = a_perturbed / s
    
    obj_perturbed = compute_obj(a_perturbed)
    delta = obj_perturbed - best_obj
    
    if delta < 0 or rng.random() < np.exp(-delta / max(temperature, 1e-8)):
        return a_perturbed, obj_perturbed if obj_perturbed < best_obj else best_obj, a_perturbed
    return a, best_obj, a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search combining multiple optimization strategies:
    1. FFT-smoothed Nesterov gradient descent (with adaptive tau up to 2000)
    2. Spectral flattening (Gerchberg-Saxton domain switching)
    3. Peak-chasing (direct peak reduction)
    4. LP-based block coordinate descent
    5. Subgradient refinement
    6. Multi-n exploration with resampling
    7. Simulated annealing for escape
    8. Diverse restarts
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)
    
    # Load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")
    
    n_best = len(best_seq)
    
    # === Phase 1: Smoothed GD with high tau ===
    print(f"\n=== Phase 1: High-tau Smoothed GD ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.25, 250.0)
    if p1_budget > 30:
        # Run with different tau ranges
        tau_configs = [
            (5.0, 500.0),
            (50.0, 2000.0),
            (100.0, 5000.0),
        ]
        for tau_i, tau_e in tau_configs:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p1_budget / len(tau_configs), remaining * 0.4)
            if alloc < 20:
                break
            a_sgd = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                              tau_init=tau_i, tau_end=tau_e, verbose=True)
            obj_sgd = compute_obj(a_sgd)
            if obj_sgd < best_score:
                best_score = obj_sgd
                best_seq = a_sgd.copy()
                print(f"  SGD (tau={tau_i}->{tau_e}): obj={best_score:.8f} **")
            else:
                print(f"  SGD (tau={tau_i}->{tau_e}): obj={obj_sgd:.8f}")
    
    # === Phase 2: Spectral Flattening ===
    print(f"\n=== Phase 2: Spectral Flattening ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        for strength in [0.2, 0.3, 0.4, 0.5]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(40.0, remaining * 0.2)
            if alloc < 15:
                break
            a_spec = spectral_flatten(best_seq, time_budget=alloc,
                                      flatten_strength=strength, verbose=True)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_score:
                best_score = obj_spec
                best_seq = a_spec.copy()
                n_best = len(best_seq)
                print(f"  Spectral (s={strength}): obj={best_score:.8f} **")
            else:
                print(f"  Spectral (s={strength}): obj={obj_spec:.8f}")
    
    # === Phase 3: Peak Chasing ===
    print(f"\n=== Phase 3: Peak Chasing ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(80.0, remaining * 0.15)
        if alloc > 20:
            a_pc = peak_chase(best_seq, time_budget=alloc, verbose=True)
            obj_pc = compute_obj(a_pc)
            if obj_pc < best_score:
                best_score = obj_pc
                best_seq = a_pc.copy()
                print(f"  PeakChase: obj={best_score:.8f} **")
            else:
                print(f"  PeakChase: obj={obj_pc:.8f}")
    
    # === Phase 4: LP Polishing (main refinement) ===
    print(f"\n=== Phase 4: LP Polishing ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.3, 350.0)
    if p4_budget > 30:
        for rnd in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(p4_budget / max(8 - rnd, 1), remaining * 0.5)
            if alloc < 20:
                break
            a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                n_best = len(best_seq)
                print(f"  LP round {rnd}: obj={best_score:.8f} **")
            else:
                print(f"  LP round {rnd}: obj={obj_lp:.8f}")
                # Perturb to escape
                scale = 0.015 * (rnd + 1)
                best_seq = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                best_seq = np.maximum(best_seq, 0)
                best_seq = best_seq / best_seq.sum()
    
    # === Phase 5: Subgradient Refinement ===
    print(f"\n=== Phase 5: Subgradient ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(60.0, remaining * 0.1)
        if alloc > 15:
            a_sg = subgradient_refine(best_seq, time_budget=alloc, K=8, verbose=True)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  SG: obj={best_score:.8f} **")
            else:
                print(f"  SG: obj={obj_sg:.8f}")
    
    # === Phase 6: Multi-n Exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_current = len(best_seq)
    
    n_options = sorted(set([
        max(300, n_current - 200), max(300, n_current - 100),
        n_current + 50, n_current + 100, n_current + 200,
        n_current + 400, 500, 600, 700, 800, 900, 1000, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500 and abs(n - n_current) > 30]
    
    for n_try in n_options[:6]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        a_try = resample_sequence(best_seq, n_try)
        
        # Quick smoothed GD
        alloc1 = min(remaining * 0.2, 50.0)
        if alloc1 > 15:
            a_sgd = nesterov_smoothed_descent(a_try, time_budget=alloc1,
                                              tau_init=20.0, tau_end=1000.0)
            if compute_obj(a_sgd) < compute_obj(a_try):
                a_try = a_sgd
        
        # LP polish
        remaining = deadline - time.monotonic()
        alloc2 = min(remaining * 0.15, 40.0)
        if alloc2 > 15:
            a_try = lp_polish(a_try, time_budget=alloc2)
        
        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_current = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")
    
    # === Phase 7: Simulated Annealing ===
    print(f"\n=== Phase 7: Simulated Annealing ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        n_sa = len(best_seq)
        a_sa = best_seq.copy()
        obj_sa = best_score
        temp = 0.1
        start_sa = time.monotonic()
        sa_budget = min(remaining * 0.15, 80.0)
        
        sa_rng = np.random.default_rng(999)
        sa_it = 0
        last_improve = 0
        
        while time.monotonic() - start_sa < sa_budget:
            a_sa, obj_candidate, a_sa = simulated_annealing_step(a_sa, best_seq, obj_sa, temp, sa_rng)
            
            if obj_candidate < best_score - 1e-12:
                best_score = obj_candidate
                best_seq = a_sa.copy()
                last_improve = sa_it
                temp = min(temp * 1.2, 0.5)  # Warm up slightly on improvement
            else:
                a_sa = a_sa  # keep perturbed for exploration
            
            if sa_it - last_improve > 200:
                a_sa = best_seq.copy()
                obj_sa = best_score
                temp = max(temp * 0.5, 0.01)
                last_improve = sa_it
            
            sa_it += 1
            if sa_it % 100 == 0:
                print(f"    SA it{sa_it}: temp={temp:.3f}, best={best_score:.8f}")
            if temp < 0.005:
                break
        
        print(f"  SA done: best={best_score:.8f}")
    
    # === Phase 8: Diverse Restarts ===
    print(f"\n=== Phase 8: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    for r in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.2 + 0.05 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()
        
        alloc = min(remaining * 0.2, 80.0)
        if alloc > 25:
            # Smoothed GD + LP
            half = alloc * 0.6
            a_r = nesterov_smoothed_descent(a_r, time_budget=half,
                                            tau_init=10.0, tau_end=500.0)
            remaining = deadline - time.monotonic()
            lp_alloc = min(remaining * 0.1, 30.0)
            if lp_alloc > 10:
                a_r = lp_polish(a_r, time_budget=lp_alloc)
            
            obj_r = compute_obj(a_r)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_r.copy()
                print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
            else:
                print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")
    
    # === Phase 9: Final Intensive Polish ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    
    # High-tau SGD
    if remaining > 25:
        alloc = min(remaining * 0.25, 80.0)
        a_sgd = nesterov_smoothed_descent(best_seq, time_budget=alloc,
                                          tau_init=100.0, tau_end=5000.0, verbose=True)
        obj_sgd = compute_obj(a_sgd)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  Final SGD: obj={best_score:.8f} **")
    
    # LP
    remaining = deadline - time.monotonic()
    if remaining > 25:
        alloc = min(remaining * 0.3, 100.0)
        a_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
    
    # Peak chase
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.2, 40.0)
        a_pc = peak_chase(best_seq, time_budget=alloc)
        obj_pc = compute_obj(a_pc)
        if obj_pc < best_score:
            best_score = obj_pc
            best_seq = a_pc.copy()
            print(f"  Final PC: obj={best_score:.8f} **")
    
    # One more LP
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining * 0.5, 40.0)
        a_lp = lp_polish(best_seq, time_budget=alloc)
        obj_lp = compute_obj(a_lp)
