import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = float(np.max(compute_conv(a)))
    start = time.monotonic()
    
    x = a.copy()  # Nesterov lookahead
    y = a.copy()  # current point
    v = np.zeros(n)  # momentum
    
    alpha = 0.1  # initial step size
    beta = 0.9  # momentum
    
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute convolution at lookahead point
        conv_x = compute_conv(x)
        max_val = float(np.max(conv_x))
        
        # Find active set (top-k lags)
        if top_k >= 2 * n - 1:
            active = np.arange(2 * n - 1)

            active = np.where(conv_x >= threshold)[0]
            if len(active) > top_k:

        grad = np.zeros(n)

"""
Optimized step function search using KKT-based LP solving,
Nesterov subgradient, and adaptive active set management.

import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_cross_conv(a, b):
    """Compute linear convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_obj(a):
    """Compute objective: 2n * max_conv for normalized a."""
    n = len(a)
    if n == 0:

    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf

    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
