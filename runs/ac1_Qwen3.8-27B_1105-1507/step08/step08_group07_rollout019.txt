import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fft_convolve(a, b):
    """Compute linear convolution of a and b using FFT."""
    n_out = len(a) + len(b) - 1
    size = 1
    while size < n_out:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n_out]


def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT."""
    return fft_convolve(a, a)


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g gives cross-conv a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp(a):
    """
    Solve LP: min M s.t. A(a)@z <= M, sum(z)=1, z>=0.
    Returns (z, M_opt) or (None, None).
    """
    n = len(a)
    A = build_conv_matrix(a)

    z = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    prob = cp.Problem(cp.Minimize(M), [A @ z <= M, cp.sum(z) == 1])

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if z.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                z_val = np.maximum(z.value, 0)
                zs = float(np.sum(z_val))
                if zs > 1e-10:
                    z_val = z_val / zs
                    return z_val, float(M.value)
        except:
            continue

    try:
        from scipy.optimize import linprog
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            z_val = np.maximum(res.x[:n], 0)
            zs = float(np.sum(z_val))
            if zs > 1e-10:
                z_val = z_val / zs
                return z_val, float(res.x[n])
    except:
        pass

    return None, None


def toeplitz_matvec(acf, v):
    """Apply symmetric Toeplitz matrix (from autocorrelation acf) to vector v.
    Q_{ij} = acf[|i-j|], so (Qv)_i = sum_j acf[|i-j|] v_j.
    """
    n = len(v)
    # r_full of length 2n-1: [acf[n-1], ..., acf[1], acf[0], acf[1], ..., acf[n-1]]
    r_full = np.concatenate([acf[n - 1:0:-1], acf])
    size = 1
    while size < 2 * n - 1:

    rv = rfft(r_full, size)
    vv = rfft(v, size)
    return irfft(rv * vv, size)[:n]


def solve_qp_l2(a, max_iter=500, tol=1e-10):
    """
    Solve min z^T Q z s.t. sum(z)=1, z>=0 using conjugate gradient.
    Q = A(a)^T A(a), which is symmetric Toeplitz with Q_{ij} = acf[|i-j|].
    Returns normalized z or None.
    """
    n = len(a)
    # Compute autocorrelation: acf[d] = sum_k a[k]*a[k+d]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    acf = irfft(fa * fa, size)[:n]

    # CG to solve Qz = 1
    b = np.ones(n)
    z = np.ones(n) / n
    Qz = toeplitz_matvec(acf, z)
    r_vec = b - Qz
    p = r_vec.copy()
    r_dot_p = np.dot(r_vec, p)

        Qp = toeplitz_matvec(acf, p)
        p_dot_Qp = np.dot(p, Qp)

        alpha = r_dot_p / p_dot_Qp
        z = z + alpha * p
        r_vec = r_vec - alpha * Qp

        p = r_vec + (r_new_dot_p / r_dot_p) * p

    s = np.sum(z)

    start = time.monotonic()

    # Use log parameterization: a = exp(x) / sum(exp(x))
    x = np.log(np.maximum(a, 1e-10))

    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    it = 0

    while time.monotonic() - start < time_budget:
        if it % 50 == 0 and tau > 0.003:
            tau *= 0.9  # Anneal the smoothing

        a_cur = np.exp(x - np.max(x))
        a_cur = a_cur / np.sum(a_cur)

        # Compute smoothed objective gradient
        conv = compute_conv(a_cur)
        # Log-sum-exp for stability
        max_c = np.max(conv)
        lse = max_c + tau * np.log(np.sum(np.exp((conv - max_c) / tau)))
        # Soft max weights
        p = np.exp((conv - max_c) / tau)
        p = p / np.sum(p)

        # Gradient of lse w.r.t. a_cur: sum_k p_k * d(conv_k)/d(a_cur)
        # d(conv_k)/d(a_i) = 2*a_cur[k-i] (if 0<=k-i<n)
        grad_lse = np.zeros(n)
        for i in range(n):
            # sum over k where 0<=k-i<n: p_k * 2*a_cur[k-i]
            k_min = i
            k_max = min(i + n, 2 * n - 1)
            if k_min < 2 * n - 1:
                grad_lse[i] = 2.0 * np.sum(p[k_min:k_max] * a_cur[0:k_max - k_min])

        # Full gradient of 2n * lse / (sum(a))^2 with sum(a)=1:
        # Since we use log param, the constraint is automatic
        # grad w.r.t. x: 2n * grad_lse - 2n*lse*1 (but sum constraint handled by softmax)
        # Actually with a = softmax(x), da_i/dx_j = a_i(δ_ij - a_j)
        # obj = 2n * lse (since sum=1)
        # d(obj)/dx_j = 2n * sum_i (da_i/dx_j) * grad_lse[i]
        # = 2n * (a_j * grad_lse[j] - a_j * sum_i a_i * grad_lse[i])
        grad_obj = 2.0 * n * (a_cur * grad_lse - a_cur * np.sum(a_cur * grad_lse))

        # Smoothed gradient w.r.t. x
        # The lse already incorporates the smoothing
        grad_x = 2.0 * n / tau * (a_cur * (tau * grad_lse) - a_cur * np.sum(a_cur * tau * grad_lse))
        # Simplify: grad_x = 2n * (a_cur * grad_lse - a_cur * mean_weighted)
        # Actually let's just use the direct subgradient approach
        # d lse / d a_i = sum_k p_k * d conv_k / d a_i = grad_lse[i]
        # d lse / d x_j = sum_i (d lse / d a_i) * (d a_i / d x_j)
        # = sum_i grad_lse[i] * a_i * (delta_ij - a_j)
        # = a_j * grad_lse[j] - a_j * sum_i a_i * grad_lse[i]
        dot = np.sum(a_cur * grad_lse)
        grad_x = a_cur * (grad_lse - dot)

        # Gradient descent with momentum
        v = momentum * v - lr * grad_x
        x = x + v

        # Check objective
        a_new = np.exp(x - np.max(x))
        a_new = a_new / np.sum(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        # Adaptive learning rate
        if it % 100 == 0:
            lr = max(0.001, lr * 0.95)

        it += 1

    return best_a


def coordinate_descent(a, time_budget=60.0):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    improved_total = 0
    while time.monotonic() - start < time_budget:
        improved = False
        # Random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            # Try a few values for coordinate i
            a_i_orig = a[i]
            # Remove a_i from sum, try new values
            rest_sum = 1.0 - a_i_orig  # assuming sum=1

            for scale in [0.5, 0.7, 1.3, 1.5, 0.3, 2.0]:
                a_try = a.copy()
                a_try[i] = a_i_orig * scale
                a_try = np.maximum(a_try, 0)
                s = np.sum(a_try)
                if s < 1e-10:
                    continue
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    a_i_orig = a_try[i]
                    improved = True
                    improved_total += 1

        if not improved and improved_total > 0:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    # Use scipy for cubic
    from scipy.interpolate import interp1d
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = f(x_new)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining LP/QP box-folding, smoothed GD,
    multi-n exploration, coordinate descent, and random restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        # Generate initial sequences
        for n_init in [400, 600, 800, 1000, 1200]:
            t = np.linspace(0, 1, n_init)
            windows = [
                0.5 + 0.5 * np.cos(2 * np.pi * t),
                np.sin(np.pi * t) ** 2,
                np.exp(-2 * (t - 0.5) ** 2 / 0.1),
                1.0 + 0.2 * np.cos(4 * np.pi * t),
            ]
            for w in windows:
                a_init = np.maximum(w, 0.01)
                a_init = a_init / np.sum(a_init)
                obj_init = compute_obj(a_init)
                if obj_init < best_score:
                    best_score = obj_init
                    best_seq = a_init.copy()

        if best_seq is None:
            best_seq = np.ones(800) / 800
            best_score = compute_obj(best_seq)
        print(f"Initial (n={len(best_seq)}): obj={best_score:.8f}")

    # === Phase 1: Quick LP refinement (150s) ===
    print(f"\n=== Phase 1: LP Refinement (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(150.0, remaining * 0.15)
        a_new = lp_qp_box_fold(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 1 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 1 done: obj={obj:.8f}, keeping best")

    # === Phase 2: QP-dominant exploration (200s) ===
    print(f"\n=== Phase 2: QP Exploration (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n = len(best_seq)
        a = best_seq.copy()
        start_p2 = time.monotonic()
        stall = 0
        while time.monotonic() - start_p2 < min(200.0, remaining * 0.2):
            # QP step
            z_qp = solve_qp_l2(a)
            if z_qp is None:
                break
            # LP step
            z_lp, _ = solve_lp(a)

            # Try combined direction
            if z_lp is not None:
                # Blend LP and QP
                for w_lp in [0.0, 0.3, 0.5, 0.3, 0.0]:
                    if w_lp == 0.0:
                        target = z_qp
                    elif w_lp == 1.0:
                        target = z_lp
                    else:
                        target = w_lp * z_lp + (1 - w_lp) * z_qp
                        target = target / np.sum(target)

                    for mix in [0.4, 0.6, 1.0]:
                        cand = (1 - mix) * a + mix * target
                        cand = np.maximum(cand, 0)
                        cs = float(np.sum(cand))
                        if cs < 1e-10:
                            continue
                        cand = cand / cs
                        obj_cand = compute_obj(cand)
                        if obj_cand < best_score - 1e-12:
                            best_score = obj_cand
                            best_seq = cand.copy()
                            a = cand.copy()
                            stall = 0
                        else:
                            stall += 1

            if stall > 20:
                rng_local = np.random.default_rng(int(time.monotonic() * 1e6) % (2**31))
                a = best_seq * np.exp(rng_local.normal(0, 0.04, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                best_seq = a.copy()
                best_score = compute_obj(a)
                stall = 0
                if time.monotonic() - start_p2 > 50:
                    print(f"  QP phase: obj={best_score:.8f}")

    # === Phase 3: Smoothed gradient descent (150s) ===
    print(f"\n=== Phase 3: Smoothed GD ===")
    remaining = deadline - time.monotonic()
    if remaining > 40:
        alloc = min(150.0, remaining * 0.15)
        a_new = smoothed_gradient_descent(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 3 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 3 done: obj={obj:.8f}")

    # === Phase 4: Multi-n exploration (300s) ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_parent = len(best_seq)
    n_candidates = sorted(set([
        max(200, n_parent - 200), max(200, n_parent - 100),
        n_parent, n_parent + 100, n_parent + 200,
        300, 400, 500, 600, 700, 900, 1000, 1100, 1300, 1500,
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 2000]
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        if n_try == len(best_seq):
            continue

        a_try = resample_sequence(best_seq, n_try)
        alloc = min(40.0, remaining * 0.12)
        a_opt = lp_qp_box_fold(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 5: LP/QP box-fold on best (200s) ===
    print(f"\n=== Phase 5: Box-fold on best (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alloc = min(200.0, remaining * 0.2)
        a_new = lp_qp_box_fold(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 5 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 5 done: obj={obj:.8f}")

    # === Phase 6: Coordinate descent polish (80s) ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(80.0, remaining * 0.1)
        a_new = coordinate_descent(best_seq, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            print(f"  Phase 6 done: obj={best_score:.8f} **")
        else:
            print(f"  Phase 6 done: obj={obj:.8f}")

    # === Phase 7: Random restarts (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 7: Random Restarts ===")
        n_best = len(best_seq)
        max_restarts = min(8, int(remaining // 35))

        for r in range(max_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            if r % 4 == 0:
                # Gaussian perturbation
                a_r = best_seq * np.exp(rng.normal(0, 0.03 + 0.01 * r, n_best))
            elif r % 4 == 1:
                # Block perturbation
                a_r = best_seq.copy()
                n_blocks = max(3, n_best // 80)
                block_size = n_best // n_blocks
                n_perturb = max(2, n_blocks // 4)
                blocks = rng.choice(n_blocks, n_perturb, replace=False)
                for b in blocks:
                    si = int(b) * block_size
                    ei = min(si + block_size, n_best)
                    a_r[si:ei] *= rng.uniform(0.4, 1.6)
            elif r % 4 == 2:
                # Frequency-domain perturbation
                fa = rfft(best_seq)
                mag = np.abs(fa) * np.exp(rng.normal(0, 0.1, len(fa)))
                phase = np.angle(fa)
                fa_new = mag * np.exp(1j * phase)
                a_r = irfft(fa_new, n_best)
            else:
                # Sparse modification
                a_r = best_seq.copy()
                n_sparse = max(5, n_best // 20)
                idx = rng.choice(n_best, n_sparse, replace=False)
                a_r[idx] *= rng.uniform(0.2, 3.0, n_sparse)

            a_r = np.maximum(a_r, 0)
            sr = float(np.sum(a_r))
            if sr > 1e-10:
                a_r = a_r / sr

            alloc = min(remaining * 0.3, 30.0)
            a_opt = lp_qp_box_fold(a_r, time_budget=alloc)
            obj_r = compute_obj(a_opt)

            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.8f} **")
            else:
                print(f"  Restart {r}: obj={obj_r:.8f}")

    # === Phase 8: Final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.4, 25.0)
        a_final = lp_qp_box_fold(best_seq, time_budget=alloc)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final: obj={best_score:.8f} **")

        # Quick coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_cd = coordinate_descent(best_seq, time_budget=min(15.0, remaining * 0.5))
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
