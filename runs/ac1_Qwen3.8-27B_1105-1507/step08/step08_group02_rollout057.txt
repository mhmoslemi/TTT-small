import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def solve_lp_fixed_a(a, timeout=10):
    """Solve: min M s.t. conv(a,g)[k] <= M forall k, sum(g)=1, g>=0."""
    n = len(a)
    a_norm = a / float(np.sum(a))
    # Build constraint matrix: conv(a,g)[k] = sum_j a_norm[k-j]*g[j] for valid j
    # A is (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(2 * n - 1, j + n)
        A[start:end, j] = a_norm[:end - j]
    
    # LP: min M
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

def coordinate_descent(a, time_budget, verbose=False):
    """Coordinate descent: try individual and pairwise adjustments."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    
    # Find peak contributors
    for rnd in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        # Get top contributors to peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        top_contrib = np.argsort(contrib)[::-1][:50]
        
        for i in top_contrib:
            if time.monotonic() - start > time_budget:
                break
            if contrib[i] < 1e-10:
                break
            
            # Try moving mass from i to other positions
            for j in range(n):
                if j == i:
                    continue
                if a[j] > 0.9 * a[i] / n:  # skip if target already has similar mass
                    continue
                
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    delta = frac * a[i]
                    if delta < 1e-15 or delta >= a[i]:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        # Undo
                        a[i] += delta
                        a[j] -= delta
                        if new_obj > best_obj:
                            break  # No point trying larger frac
        
        if not improved:
            break
    
    return best_a

def structured_init(n, mode, rng):
    """Generate structured initial sequences."""
    if mode == 0:
        return np.ones(n) / n
    elif mode == 1:
        # Chirp
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(2 * np.pi * (1 + 3 * x) * x)) + 0.01
        return a / a.sum()
    elif mode == 2:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 3:
        # Gaussian
        x = np.linspace(-3, 3, n)
        a = np.exp(-x**2 / 2)
        return a / a.sum()
    elif mode == 4:
        # Two bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.25)**2) / 0.01) + np.exp(-((x - 0.75)**2) / 0.01) + 0.01
        return a / a.sum()
    elif mode == 5:
        # Inverse chirp
        x = np.linspace(0, 1, n)
        a = 1.0 / (0.1 + x * (1 - x) * 10)
        return a / a.sum()
    elif mode == 6:
        # Random with smoothing
        a = rng.uniform(0, 1, n)
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif mode == 7:
        # Piecewise constant (Barker-like)
        a = np.zeros(n)
        seg = max(1, n // 13)
        signs = rng.choice([0.3, 1.0], size=n // seg + 1)
        for i in range(n // seg + 1):
            a[i*seg:(i+1)*seg] = signs[i]
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""
Hybrid optimization for minimum peak autocorrelation step functions.
Combines alternating LP, Frank-Wolfe conditional gradient, Nesterov subgradient,
log-parametrized gradient descent, coordinate descent, and dimension exploration.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def subgradient_of_peak(a, k):
    """Compute subgradient of conv(a,a)[k] w.r.t. a.
    Result: grad[i] = 2*a[k-i] if 0<=k-i<n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    for i in range(i_min, i_max + 1):
        grad[i] = 2.0 * a[k - i]
    return grad


def weighted_subgradient(a, topK=5):
    """Compute weighted subgradient using top-K peaks."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:topK]
    peak_vals = conv[top_idx]
    max_val = peak_vals[0]
    if max_val < 1e-15:
        return np.zeros(n), conv, top_idx[0]
    # Temperature-scaled weights
    temperature = max_val * 0.1
    weights = np.exp((peak_vals - max_val) / temperature)
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        grad += w * subgradient_of_peak(a, k)
    return grad, conv, int(top_idx[0])


def solve_lp_fixed(a):
    """Solve: min M s.t. conv(a,g)[k] <= M forall k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a_norm = a / float(np.sum(a))
    # Build A: A[k, j] = a_norm[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = a_norm[:end - j]
    # LP variables: [g_0,...,g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g_val = np.maximum(res.x[:n], 0)
        s = float(np.sum(g_val))
        if s > 1e-10:
            return g_val / s, float(res.x[n])
    return None, None

def frank_wolfe(a, time_budget, lower_bound=None, verbose=False):
    """Frank-Wolfe (conditional gradient) for min max_k conv(a,a)[k] on simplex."""
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    if lower_bound is None:
        lower_bound = best_obj * 0.9
    start = time.monotonic()
    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break
        # Subgradient
        grad, conv, k_star = weighted_subgradient(a, topK=3)
        # FW step: move towards e_{argmin grad}
        i_min = int(np.argmin(grad))
        # Frank-Wolfe step size: min of exact line search and 2/(it+2)
        obj_current = 2.0 * n * float(conv[k_star])
        gap = max(obj_current - lower_bound, 1e-12)
        grad_sq = float(np.sum(grad ** 2))
        polyak_t = gap / (grad_sq + 1e-15)
        fw_t = 2.0 / (it + 2)
        t = min(polyak_t, fw_t, 1.0)
        # Update: a_new = (1-t)*a + t*e_{i_min}
        a_new = (1.0 - t) * a
        a_new[i_min] += t
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        if verbose and it % 500 == 0:
            print(f"    FW it{it}: obj={best_obj:.8f}, t={t:.4f}, "
                  f"{time.monotonic()-start:.1f}s")
    return best_a

def logparam_gradient(a, time_budget, verbose=False):
    """Gradient descent in log-space (softmax parametrization) to stay in interior."""
    n = len(a)
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    log_a = np.log(a)
    lr = 1.0 / np.sqrt(n)
    momentum = 0.0
    v = np.zeros(n)
    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break
        a_curr = np.exp(log_a - np.max(log_a))
        a_curr = a_curr / a_curr.sum()
        grad, conv, k_star = weighted_subgradient(a_curr, topK=3)
        # Gradient in log-space: d f / d log_a[i] = a[i] * grad[i] - a[i] * sum(a*grad)
        a_grad = a_curr * grad
        a_grad -= a_curr * float(np.sum(a_curr * grad))
        v = momentum * v + a_grad
        log_a_new = log_a - lr * v
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            log_a = log_a_new.copy()
            momentum = min(0.8, momentum + 0.001)
            lr = max(1e-6, lr * 0.999)
        else:
            log_a = log_a_new.copy()  # still update y for exploration
            momentum = max(0.0, momentum * 0.95)
            if it % 500 == 0:
                lr = max(1e-6, lr * 0.9)
        if verbose and it % 1000 == 0:
            print(f"    LG it{it}: obj={best_obj:.8f}, lr={lr:.2e}, "
                  f"{time.monotonic()-start:.1f}s")
    return best_a


def coordinate_descent(a, time_budget, verbose=False):
    """Coordinate descent: move mass from peak contributors to reduce max conv."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    stalled_rounds = 0
    while time.monotonic() - start < time_budget:
        improved = False
        # Compute contributions to peak
        contrib = np.zeros(n)
        i_min = max(0, k_star - (n - 1))
        i_max = min(n - 1, k_star)
        for i in range(i_min, i_max + 1):
            contrib[i] = a[i] * a[k_star - i]
        top_idx = np.argsort(contrib)[::-1][:40]
        for i in top_idx:
            if time.monotonic() - start > time_budget:
                break
            if contrib[i] < 1e-10:
                break
            # Try moving mass from i to positions not contributing to peak
            j_min = max(0, k_star - (n - 1))
            j_max = min(n - 1, k_star)
            for j in range(n):
                if j == i or j_min <= j <= j_max:
                    continue
                if a[j] > 0.5 * a[i] / max(1, n // 10):
                    continue
                for frac in [0.003, 0.01, 0.03, 0.08]:
                    delta = frac * a[i]
                    if delta < 1e-15 or delta >= 0.95 * a[i]:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(new_conv))
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        if new_obj > best_obj * 1.001:
                            break
        if not improved:
            stalled_rounds += 1
            if stalled_rounds > 5:
                break
        else:
            stalled_rounds = 0
        if verbose and (it := int((time.monotonic() - start) * 2)) % 20 == 0:
            print(f"    CD: obj={best_obj:.8f}, {time.monotonic()-start:.1f}s")
    return best_a


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

    elif mode == 3:

        a = 1.0 / (0.05 + 5.0 * x * (1 - x))
        return a / a.sum()
    elif mode == 6:
        a = rng.uniform(0.1, 1.0, n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        return a / a.sum()
    elif mode == 7:
        x = np.linspace(0, 1, n)
        freqs = rng.integers(2, 8, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.4 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase hybrid search: alternating LP + Frank-Wolfe + Nesterov subgradient
    + log-param gradient + coordinate descent + dimension exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0
    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 800
        best_seq = np.ones(n0) / n0
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n0}): obj={best_score:.8f}")

    n_best = len(best_seq)
    print(f"\nStarting: n={n_best}, obj={best_score:.8f}")

    # === Phase 1: Alternating LP (primary optimizer) ===
    print(f"\n=== Phase 1: Alternating LP ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.30, 400.0)
    for restart in range(3):
        remaining = deadline - time.monotonic()
        alloc = p1_budget / (3 - restart)
        if alloc < 25:
            break
        if restart == 0:
            a_start = best_seq.copy()
        elif restart == 1:
            a_start = best_seq * (1.0 + 0.06 * rng.normal(0, 1, n_best))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        else:
            a_struct = structured_init(n_best, 4, rng)
            a_start = 0.75 * best_seq + 0.25 * a_struct
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        a_lp = alternating_lp(a_start, time_budget=alloc, verbose=(restart == 0))
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            n_best = len(best_seq)
            print(f"  AltLP restart {restart}: obj={best_score:.8f} **")
        else:
            print(f"  AltLP restart {restart}: obj={obj_lp:.8f}")

    # === Phase 2: Frank-Wolfe (fast, simple) ===
    print(f"\n=== Phase 2: Frank-Wolfe ===")
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.10, 120.0)
    if p2_budget > 15:
        a_fw = frank_wolfe(best_seq, time_budget=p2_budget,
                           lower_bound=best_score * 0.97, verbose=True)
        obj_fw = compute_obj(a_fw)
        if obj_fw < best_score:
            best_score = obj_fw
            best_seq = a_fw.copy()
            print(f"  FW: obj={best_score:.8f} **")
        else:
            print(f"  FW: obj={obj_fw:.8f}")

    # === Phase 3: Nesterov Subgradient ===
    print(f"\n=== Phase 3: Nesterov Subgradient ===")
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.12, 150.0)
    if p3_budget > 15:
        a_ns = nesterov_subgrad(best_seq, time_budget=p3_budget,
                                lower_bound=best_score * 0.97, verbose=True)
        obj_ns = compute_obj(a_ns)
        if obj_ns < best_score:
            best_score = obj_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov: obj={best_score:.8f} **")
        else:
            print(f"  Nesterov: obj={obj_ns:.8f}")

    # === Phase 4: Log-param gradient (interior method) ===
    print(f"\n=== Phase 4: Log-param Gradient ===")
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.10, 120.0)
    if p4_budget > 15:
        a_lg = logparam_gradient(best_seq, time_budget=p4_budget, verbose=True)
        obj_lg = compute_obj(a_lg)
        if obj_lg < best_score:
            best_score = obj_lg
            best_seq = a_lg.copy()
            print(f"  LogGrad: obj={best_score:.8f} **")
        else:
            print(f"  LogGrad: obj={obj_lg:.8f}")

    # === Phase 5: Dimension Exploration ===
    print(f"\n=== Phase 5: Dimension Exploration ===")
    remaining = deadline - time.monotonic()
    n_options = sorted(set([
        n_best - 100, n_best - 50, n_best + 50, n_best + 100,
        n_best + 200, 500, 600, 700, 900, 1000, 1100, 1200
    ]))
    n_options = [n for n in n_options if 300 <= n <= 1500 and n != n_best]
    n_options.sort(key=lambda x: abs(x - n_best))

    for n_try in n_options[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        a_try = resample(best_seq, n_try)
        alloc = min(remaining * 0.10, 80.0)
        # Quick LP + FW
        a_opt = alternating_lp(a_try, time_budget=alloc * 0.6)
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_fw = frank_wolfe(a_opt, time_budget=min(remaining * 0.08, 40.0),
                               lower_bound=best_score * 0.98)
            if compute_obj(a_fw) < compute_obj(a_opt):
                a_opt = a_fw
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 6: Combined refinement loop ===
    print(f"\n=== Phase 6: Combined Refinement ===")
    remaining = deadline - time.monotonic()
    for it in range(15):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break
        # LP
        lp_alloc = min(remaining * 0.12, 70.0)
        if lp_alloc > 12:
            a_lp = alternating_lp(best_seq, time_budget=lp_alloc)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Refine LP {it}: obj={best_score:.8f} **")
        # FW
        remaining = deadline - time.monotonic()
        fw_alloc = min(remaining * 0.06, 30.0)
        if fw_alloc > 8:
            a_fw2 = frank_wolfe(best_seq, time_budget=fw_alloc,
                                 lower_bound=best_score * 0.99)
            obj_fw2 = compute_obj(a_fw2)
            if obj_fw2 < best_score:
                best_score = obj_fw2
                best_seq = a_fw2.copy()
                print(f"  Refine FW {it}: obj={best_score:.8f} **")

    # === Phase 7: Coordinate Descent Polish ===
    print(f"\n=== Phase 7: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    cd_alloc = min(remaining * 0.20, 100.0)
    if cd_alloc > 10:
        a_cd = coordinate_descent(best_seq, time_budget=cd_alloc, verbose=True)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: obj={best_score:.8f} **")
        else:
            print(f"  CD: obj={obj_cd:.8f}")

    # === Phase 8: Final intensive LP ===
    print(f"\n=== Phase 8: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = alternating_lp(best_seq, time_budget=min(remaining * 0.4, 60.0))
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final LP: obj={best_score:.8f} **")
    # One more CD
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_cd2 = coordinate_descent(best_seq, time_budget=min(remaining * 0.4, 20.0))
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_score:
            best_score = obj_cd2
            best_seq = a_cd2.copy()
            print(f"  Final CD: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
