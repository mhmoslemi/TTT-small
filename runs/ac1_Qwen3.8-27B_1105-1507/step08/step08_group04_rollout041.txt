def peak_shave(a, n_steps=1000):
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    
    for step in range(n_steps):
        conv = compute_conv(a)
        max_val = conv.max()
        # Find all positions where conv is close to max
        threshold = max_val * (1 - 1e-6)
        peak_positions = np.where(conv >= threshold)[0]
        
        # For each peak position, find the contributing pairs
        for k in peak_positions[:5]:  # Top 5 peaks
            k = int(k)
            # Find j values that contribute to conv[k]
            # conv[k] = sum_j a[j] * a[k-j] for valid j
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            js = np.arange(j_start, j_end + 1)
            products = a[js] * a[k - js]
            
            # Find the j with maximum product
            j_max = js[np.argmax(products)]
            j_other = k - j_max
            
            # Reduce both, add to random other positions
            reduction = 0.1 * a[j_max]
            a[j_max] -= reduction
            a[j_other] -= reduction if j_other != j_max else 0  # careful if same
            
            # Distribute to random positions
            for _ in range(2):
                target = np.random.randint(0, n)
                a[target] += reduction
            
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return a

Optimizer using enhanced LP alternating with exact line search,
smoothed-max gradient descent, peak shaving, and multi-n exploration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize_scalar
import cvxpy as cp
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_conv_cross(a, b):
    """Compute cross-convolution a * b using FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A

def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)
    
    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    
    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > 30:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue
    
    # Fallback
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:

    except Exception:
        pass
    
    return None, None

def exact_line_search(a, g):
    """Find t in [0,1] that minimizes compute_obj((1-t)*a + t*g)."""
    def f(t):

def smoothed_max_gradient_descent(a, time_budget=60.0, beta=50.0, verbose=False):
    """Gradient descent on smoothed max convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.1 / n
    momentum = np.zeros(n)
    beta_mom = 0.9
    
    for it in range(100000):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        # Softmax weights
        max_c = conv.max()
        exp_c = np.exp(beta * (conv - max_c))
        sum_exp = exp_c.sum()
        weights = exp_c / sum_exp
        
        # Gradient of smoothed max wrt a
        # d/d a[j] of sum_k w_k * (a*a)[k] where w_k = softmax(beta*(a*a))_k
        # (a*a)[k] = sum_j a[j] * a[k-j]
        # d/d a[j] = sum_k w_k * (a[k-j] + a[j+k-n+1-...]) ... 
    
        # Using cross-correlation:
        # grad[j] = 2 * sum_k w_k * a[k-j] * 1[0<=k-j<n]
        # This is 2 * cross-correlation of a and w, evaluated at j
        n_fft = 1
        while n_fft < 4 * n:
            n_fft <<= 1
        
        A_fft = rfft(a, n_fft)
        W_fft = rfft(weights, n_fft)
        
        # We want R[j] = sum_k w[k] * a[k-j] for 0<=k-j<n
        # = sum_m a[m] * w[m+j] (substituting m = k-j, k = m+j)
        # This is the cross-correlation <a, shift_j(w)>
        # In FFT: ifft(conj(A) * W) gives sum_m a[m] * conj(w[m-j])... 
        # Since everything is real: ifft(conj(A) * W)[j] = sum_m a[m] * w[m-j] (circular)
        
        R = irfft(np.conj(A_fft) * W_fft, n_fft)
        # R[j] = sum_m a[m] * w[(m-j) mod n_fft]
        # We want sum_m a[m] * w[m+j] for valid m, m+j
        # = sum_k w[k] * a[k-j] where k = m+j, m = k-j
        # Hmm, let me think again.
        # 
        # (a*a)[k] = sum_j a[j] * a[k-j] (0<=j<n, 0<=k-j<n)
        # d/d a[j0] of (a*a)[k] = a[k-j0] + a[j0-k] (if both valid)... 
        # Wait, no. (a*a)[k] = sum_j a[j]*a[k-j]. 
        # d/d a[j0] = a[k-j0] (from the a[j] term) + a[j0] * d/d a[j0] of a[k-j]
        # The second term: a[k-j] depends on a[j0] when k-j = j0, i.e., j = k-j0.
        # So d/d a[j0] of (a*a)[k] = a[k-j0] + a[k-j0] = 2*a[k-j0] (if 0<=k-j0<n)
        # Plus the symmetric term... wait, (a*a)[k] = sum_j a[j]*a[k-j].
        # By symmetry, (a*a)[k] = (a*a)[2n-2-k]... no, that's not right for linear conv.
        # 
        # Actually (a*a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
        # d/d a[j0] = a[k-j0] (if 0<=k-j0<n) + a[j0-k+n-1]... 
        # No. Let me be more careful.
        # (a*a)[k] = sum_j a[j] * a[k-j]
        # The term a[j0] appears in two ways:
        # 1. When j = j0: contributes a[j0] * a[k-j0] (if 0<=k-j0<n)
        # 2. When k-j = j0, i.e., j = k-j0: contributes a[k-j0] * a[j0] (if 0<=k-j0<n)
        # These are the same! So d/d a[j0] of (a*a)[k] = 2*a[k-j0] if 0<=k-j0<n, else 0.
        # Wait, that's only if j0 != k-j0, i.e., j0 != k/2. If j0 = k/2, it's a[j0]^2, so derivative is 2*a[j0].
        # Either way, it's 2*a[k-j0] (which equals 2*a[j0] when j0 = k/2).
        
        # So grad[j0] = 2 * sum_k w[k] * a[k-j0] * 1[0<=k-j0<n]
        
        # This is 2 * (a circular-correlate w)[j0] ... let me think in terms of FFT.
        # sum_k w[k] * a[k-j0] for k in [0, 2n-2] and 0<=k-j0<n
        # = sum_{m=0}^{n-1} a[m] * w[m+j0] for m+j0 in [0, 2n-2]
        # = (circular cross-correlation of w and a)
        # 
        # Let's pad w to length n_fft and a to length n_fft.
        # circular xc: ifft(conj(FFT(a)) * FFT(w))
        # Result at position j: sum_m a[m] * w[m-j] (circular)
        # We want sum_m a[m] * w[m+j] = result at position -j (circular) = result at n_fft-j
        
        R = irfft(np.conj(A_fft) * W_fft, n_fft)
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = R[(n_fft - j) % n_fft]
        
        grad *= 2.0  # Factor of 2 from the derivative
        
        # Also need to account for the derivative of w (softmax) wrt a
        # But for large beta, the softmax is peaked, and the derivative of w is small
        # relative to the main term. So we ignore it for now.
        
        # Project gradient onto tangent space of simplex (sum=0)
        grad -= np.mean(grad)
        
        # Momentum update
        momentum = beta_mom * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
        
        if it % 500 == 499:
            lr *= 0.99
        
        if verbose and it % 1000 == 0:
            print(f"    SmoothedGD: it={it}, obj={best_obj:.8f}")
    
    return best_a

def peak_shave(a, time_budget=30.0):
    """Reduce peaks of auto-convolution by adjusting a."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = conv.max()
        # Find positions where conv is in top 1%
        threshold = np.percentile(conv, 99)
        peak_positions = np.where(conv >= threshold)[0]
        
        improved = False
        for k in peak_positions[:3]:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            js = np.arange(j_start, j_end + 1)
            if len(js) == 0:
                continue
            products = a[js] * a[k - js]
            
            # Find top contributing pairs
            top_idx = np.argsort(products)[::-1][:5]
            for idx in top_idx:
                j1 = int(js[idx])
                j2 = k - j1
                if j2 < 0 or j2 >= n or j2 == j1:
                    continue
                
                # Try reducing the pair
                reduction = 0.05 * min(a[j1], a[j2])
                a_try = a.copy()
                a_try[j1] -= reduction
                a_try[j2] -= reduction
                a_try = np.maximum(a_try, 0)
                
                # Redistribute to random positions (prefer low-convolution positions)
                redist = reduction  # total to redistribute (counting both reductions)
                for _ in range(2):
                    # Pick position with lowest convolution contribution
                    target = rng.integers(0, n)
                    a_try[target] += redist / 2
                
                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
        # If no improvement in a full pass, break
        # (Actually we should continue, just slower)
    
    return best_a

def resample_seq(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def diverse_initialization(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-50 * (x - 0.5) ** 2)
    elif choice == 1:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 3 * x)
    elif choice == 2:
        a = np.exp(-80 * (x - 0.3) ** 2) + np.exp(-80 * (x - 0.7) ** 2)
    elif choice == 3:
        a = np.ones(n)
        for _ in range(3):
            center = rng.integers(0, n)
            width = max(1, rng.integers(n // 10, n // 3))
            for i in range(n):
                d = min(abs(i - center), n - abs(i - center))
                a[i] *= 1.0 + rng.standard_normal() * np.exp(-d / (width * 0.5))
        a = np.maximum(a, 0.01)
    elif choice == 4:
        a = x ** rng.uniform(0.5, 3.0)
    else:
        # U-shaped
        a = (x - 0.5) ** 2 + 0.1
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

Wait, I need to double-check the gradient computation in the smoothed max method. Let me verify:

(a*a)[k] = sum_j a[j] * a[k-j] for valid j (0<=j<n and 0<=k-j<n)

d/d a[j0] of (a*a)[k]:
- a[j0] appears in the product a[j]*a[k-j] when:
  - j = j0: term is a[j0]*a[k-j0], derivative wrt a[j0] is a[k-j0] (if 0<=k-j0<n)
  - k-j = j0, i.e., j = k-j0: term is a[k-j0]*a[j0], derivative wrt a[j0] is a[k-j0] (if 0<=k-j0<n)
  
These are the same pair! (j=j0, k-j=k-j0) and (j=k-j0, k-j=j0). So the total derivative is 2*a[k-j0] (if 0<=k-j0<n), unless j0 = k-j0 (i.e., j0 = k/2), in which case it's 2*a[j0] (same thing).

So: d/d a[j0] of (a*a)[k] = 2*a[k-j0] * 1[0<=k-j0<n]

And the gradient of the smoothed max:
grad[j0] = sum_k w[k] * 2*a[k-j0] * 1[0<=k-j0<n]
         = 2 * sum_k w[k] * a[k-j0] * 1[0>=k-j0<n]
         = 2 * sum_{m: 0<=m<n} a[m] * w[m+j0] * 1[0<=m+j0<2n-1]

Since m ranges over [0,n-1] and j0 ranges over [0,n-1], m+j0 ranges over [0, 2n-2], which is always valid. So:
grad[j0] = 2 * sum_{m=0}^{n-1} a[m] * w[m+j0]

This is 2 times the cross-correlation of w with a, evaluated at lag j0.

In FFT terms: if we compute ifft(conj(FFT(a)) * FFT(w)), the result at position j is:
sum_m a[m] * w[(m-j) mod N]

We want sum_m a[m] * w[m+j0]. This is the result at position -j0 (mod N) = N - j0 (if j0 > 0) or 0 (if j0 = 0).

So grad[j0] = 2 * R[(N - j0) % N] where R = ifft(conj(FFT(a)) * FFT(w)).

This matches what I have in the code. Good.

Now, one issue: the gradient of the smoothed objective also includes the derivative of the weights w (since w depends on a through the convolution). But for large beta, the weights are concentrated on the peaks, and the derivative of w is of order 1/beta, so it's a small correction. For our purposes, ignoring it should be fine.

Actually wait, let me reconsider. The smoothed objective is:
F(a) = (1/beta) * log(sum_k exp(beta * (a*a)[k]))

dF/da[j0] = (1/beta) * (1/sum_k exp(beta*(a*a)[k])) * sum_k exp(beta*(a*a)[k]) * beta * d(a*a)[k]/da[j0]
= sum_k w[k] * d(a*a)[k]/da[j0]
= sum_k w[k] * 2*a[k-j0] * 1[0<=k-j0<n]
= 2 * sum_k w[k] * a[k-j0] * 1[0<=k-j0<n]

Where w[k] = exp(beta*(a*a)[k]) / sum_k' exp(beta*(a*a)[k']).

"""
Optimizer using enhanced LP alternating with exact line search,
smoothed-max gradient descent, peak shaving, and multi-n exploration.
Improves upon previous state-of-the-art by using exact 1D line search
in the alternating scheme, momentum acceleration, and broader n exploration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize_scalar
import cvxpy as cp
import time


def compute_conv(a):
    """FFT-based full linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        A[k_start:k_end + 1, j] = a[:k_end - k_start + 1]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    a = a / np.sum(a)
    A = build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [A @ g <= M, cp.sum(g) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)

    for solver in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            t0 = time.monotonic()
            prob.solve(solver=solver)
            if time.monotonic() - t0 > 60:
                continue
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.maximum(g.value, 0)
                gs = float(np.sum(g_val))
                if gs > 1e-10:
                    return g_val / gs, float(M.value)
        except Exception:
            continue

    # Fallback to scipy
    try:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            gs = float(np.sum(g_val))
            if gs > 1e-10:
                return g_val / gs, float(res.x[n])
    except Exception:
        pass

    return None, None


def exact_line_search(a, g):
    """Find t in [0,1] minimizing compute_obj((1-t)*a + t*g) via bounded scalar minimization."""
    n = len(a)

    def f(t):
        h = (1.0 - t) * a + t * g
        s = float(np.sum(h))
        if s < 1e-10:
            return 1e10
        h = h / s
        conv = compute_conv(h)
        return 2.0 * n * float(np.max(conv))

    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-8, 'maxiter': 80})
    return float(res.x), float(res.fun)

def smoothed_max_gradient_descent(a, time_budget=60.0, beta=50.0, verbose=False):
    """Gradient descent on smoothed (log-sum-exp) max of auto-convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.3 / n
    momentum_vec = np.zeros(n)
    beta_mom = 0.85

    n_fft = 1
    while n_fft < 4 * n:
        n_fft <<= 1

    for it in range(200000):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Softmax weights over convolution peaks
        max_c = float(conv.max())
        exp_c = np.exp(beta * (conv - max_c))
        sum_exp = float(np.sum(exp_c))
        if sum_exp < 1e-300:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        else:
            weights = exp_c / sum_exp

        # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
        A_fft = rfft(a, n_fft)
        W_fft = rfft(weights, n_fft)
        R = irfft(np.conj(A_fft) * W_fft, n_fft)
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = R[(n_fft - j) % n_fft]
        grad *= 2.0

        # Project to tangent of simplex
        grad -= np.mean(grad)

        # Momentum update
        momentum_vec = beta_mom * momentum_vec - lr * grad
        a_new = a + momentum_vec
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        if it % 500 == 499:
            lr *= 0.995
        if it % 2000 == 1999:
            beta_mom = min(0.95, beta_mom + 0.01)

        if verbose and it % 5000 == 0:
            print(f"    SmoothedGD: it={it}, obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a


def peak_shave(a, time_budget=30.0):
    """Reduce convolution peaks by local adjustments to a."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99999)

    attempt = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_val = float(conv.max())
        # Find top peak positions
        top_indices = np.argsort(conv)[::-1][:8]

        improved = False
        for k in top_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            js = np.arange(j_start, j_end + 1)
            products = a[js] * a[k - js]
            top_j = js[np.argmax(products)]
            j1 = int(top_j)
            j2 = k - j1
            if j2 < 0 or j2 >= n or j2 == j1:
                continue

            # Try several reduction amounts
            for frac in [0.03, 0.05, 0.08, 0.12]:
                reduction = frac * min(a[j1], a[j2])
                if reduction < 1e-12:
                    continue
                a_try = a.copy()
                a_try[j1] -= reduction
                a_try[j2] -= reduction
                a_try = np.maximum(a_try, 0)

                # Redistribute mass to random positions away from peaks
                for _ in range(2):
                    target = int(rng.integers(0, n))
                    a_try[target] += reduction

                a_try = a_try / a_try.sum()
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                break

        attempt += 1
        if attempt % 50 == 0:
            # Add noise to escape local minima
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                best_obj = obj_sg
                best_a = a_sg
                print(f"  SmoothedGD beta={beta_try}: obj={best_obj:.8f} **")
            else:
                print(f"  SmoothedGD beta={beta_try}: obj={obj_sg:.8f}")

    # === Phase 3: LP alternating round 2 ===
    print(f"\n=== Phase 3: LP Alternating Round 2 ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = min(remaining * 0.25, 200.0)
        a_p3 = lp_alternate_enhanced(best_a, time_budget=p3_budget, verbose=True)
        obj_p3 = compute_obj(a_p3)
        if obj_p3 < best_obj:
            best_obj = obj_p3
            best_a = a_p3
            print(f"  Phase 3 result: obj={best_obj:.8f} **")
        else:
            print(f"  Phase 3 result: obj={obj_p3:.8f}")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p4_budget = min(remaining * 0.08, 50.0)
        a_ps = peak_shave(best_a, time_budget=p4_budget)
        obj_ps = compute_obj(a_ps)
        if obj_ps < best_obj:
            best_obj = obj_ps
            best_a = a_ps
            print(f"  Peak shave: obj={best_obj:.8f} **")
        else:
            print(f"  Peak shave: obj={obj_ps:.8f}")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)
    n_candidates = sorted(set([
        max(300, n_best - 400), max(300, n_best - 200),
        n_best + 200, n_best + 400, n_best + 600, n_best + 1000,
        1000, 1500, 2000, 2400, 3000,
    ]))
    n_candidates = [x for x in n_candidates if 300 <= x <= 5000 and x != n_best]
    n_candidates.sort(key=lambda x: abs(x - n_best))

    for n_try in n_candidates[:5]:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_try = resample_seq(best_a, n_try)
        a_try = a_try / np.sum(a_try)

        alloc = min(45.0, remaining * 0.18)
        a_opt = lp_alternate_enhanced(a_try, time_budget=alloc)
        obj_try = compute_obj(a_opt)

        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_opt.copy()
            print(f"  ** n={n_try}: obj={best_obj:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse initializations ===
    print(f"\n=== Phase 6: Diverse Initializations ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_a)

    for d in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        a_div = diverse_initialization(n_best, rng)
        alloc = min(35.0, remaining * 0.15)
        a_opt = lp_alternate_enhanced(a_div, time_budget=alloc)
        obj_div = compute_obj(a_opt)

        if obj_div < best_obj:
            best_obj = obj_div
            best_a = a_opt.copy()
            print(f"  ** Diverse init {d}: obj={best_obj:.8f} **")
        else:
            print(f"  Diverse init {d}: obj={obj_div:.8f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_lp = min(remaining * 0.35, 50.0)
        a_final = lp_alternate_enhanced(best_a, time_budget=alloc_lp, verbose=True)
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final
            print(f"  Final LP: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc_sg = min(remaining * 0.25, 30.0)
        a_sg = smoothed_max_gradient_descent(best_a, time_budget=alloc_sg, beta=60.0)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg
            print(f"  Final SmoothedGD: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 12:
        alloc_ps = min(remaining * 0.2, 20.0)
        a_ps2 = peak_shave(best_a, time_budget=alloc_ps)
        obj_ps2 = compute_obj(a_ps2)
        if obj_ps2 < best_obj:
            best_obj = obj_ps2
            best_a = a_ps2
            print(f"  Final PeakShave: obj={best_obj:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 10:
        alloc_last = min(remaining * 0.5, 20.0)
        a_last = lp_alternate_enhanced(best_a, time_budget=alloc_last)
        obj_last = compute_obj(a_last)
        if obj_last < best_obj:
            best_obj = obj_last
            best_a = a_last
            print(f"  Last LP: obj={best_obj:.8f} **")

    # === Final validation ===
    if best_a is None or len(best_a) == 0:
        best_a = np.ones(100)
        best_obj = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_obj = val
    else:
        result = [1.0] * 100
        best_obj = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_obj:.8f}, time={elapsed:.1f}s ===")

    return result
