"""
Improved search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous:

- Dual subgradient refinement between LP steps

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

try:
    import cvxpy as cp
    HAVE_CVXPY = True
except Exception:
    HAVE_CVXPY = False


def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_cross(a, b):
    """Cross-correlation conv(a, b) via FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=3.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY and n <= 2000:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    # Fallback to scipy HiGHS
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def golden_section_line_search(a, g, f_obj, lo=0.0, hi=1.0, tol=1e-8, max_iter=50):
    """
    Golden section search for min over t in [lo, hi] of f_obj((1-t)*a + t*g).
    Both a and g should have sum 1.
    """
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
    
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    
    a_c = (1.0 - c) * a + c * g
    a_c = np.maximum(a_c, 0)
    s_c = a_c.sum()
    if s_c > 1e-10:
        a_c /= s_c
        f_c = f_obj(a_c)
    else:
        f_c = np.inf
    
    a_d = (1.0 - d) * a + d * g
    a_d = np.maximum(a_d, 0)
    s_d = a_d.sum()
    if s_d > 1e-10:
        a_d /= s_d
        f_d = f_obj(a_d)
    else:
        f_d = np.inf
    
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if f_c < f_d:
            hi = d
            d = c
            f_d = f_c
            c = hi - gr * (hi - lo)
            a_c = (1.0 - c) * a + c * g
            a_c = np.maximum(a_c, 0)
            s_c = a_c.sum()
            if s_c > 1e-10:
                a_c /= s_c
                f_c = f_obj(a_c)
            else:
                f_c = np.inf
        else:
            lo = c
            c = d
            f_c = f_d
            d = lo + gr * (hi - lo)
            a_d = (1.0 - d) * a + d * g
            a_d = np.maximum(a_d, 0)
            s_d = a_d.sum()
            if s_d > 1e-10:
                a_d /= s_d
                f_d = f_obj(a_d)
            else:
                f_d = np.inf
    
    t_opt = (lo + hi) / 2.0
    a_opt = (1.0 - t_opt) * a + t_opt * g
    a_opt = np.maximum(a_opt, 0)
    s_opt = a_opt.sum()
    if s_opt > 1e-10:
        a_opt /= s_opt
    return a_opt, t_opt


def enforce_symmetry(a):
    """Make sequence symmetric (palindromic)."""
    n = len(a)
    half = (n + 1) // 2
    sym = np.zeros(n)
    for i in range(half):
        sym[i] = (a[i] + a[n - 1 - i]) / 2.0
        sym[n - 1 - i] = sym[i]
    s = sym.sum()
    if s > 1e-10:
        sym /= s
    return sym


def lp_alternating_v2(a, time_budget=200.0, verbose=False, use_symmetry=False):
    """
    Improved LP alternating with golden section line search and dual-informed escapes.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)
    it = 0
    momentum = 0.0
    v = np.zeros(n)

    def quick_obj(x):
        conv = compute_conv(x)
        return 2.0 * n * float(np.max(conv))

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2.0:
            break

        # Solve LP
        g_lp, M_lp = solve_lp_minpeak(a)
        if g_lp is None:
            break

        if use_symmetry:
            g_lp = enforce_symmetry(g_lp)

        # Golden section line search
        a_try, t_opt = golden_section_line_search(a, g_lp, quick_obj, lo=0.01, hi=1.0, tol=1e-7, max_iter=30)

        obj_try = quick_obj(a_try)
        it += 1

        if obj_try < best_obj - 1e-12:
            # Momentum update
            v = momentum * v + (a_try - a)
            a = a_try.copy()
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()
            stall = 0
            momentum = min(0.5, momentum + 0.01)
            if verbose and it % 10 == 0:
                print(f"    LPv2 it{it}: obj={best_obj:.8f}, t={t_opt:.4f}, M={M_lp:.6f}, el={time.monotonic()-start:.1f}s")
        else:
            stall += 1
            if stall > 10:
                # Escape: perturbation + fresh direction
                noise_scale = min(0.2, 0.03 + 0.02 * (stall // 10))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                if use_symmetry:
                    a = enforce_symmetry(a)
                best_a = a.copy()
                best_obj = compute_obj(a)
                stall = 0
                momentum = 0.0
                v = np.zeros(n)
            elif stall > 5:
                # Smaller perturbation
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                if use_symmetry:
                    a = enforce_symmetry(a)

    return best_a


def subgradient_refine(a, time_budget=60.0, K=10, verbose=False):
    """
    Subgradient method with adaptive temperature weighting of active constraints.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 2.0 / np.sqrt(n)
    stall = 0
    patience = 100

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        max_val = float(np.max(conv))
        if max_val < 1e-15:
            break

        # Weight active constraints by softmax
        # Use all constraints that are within 5% of max
        threshold = max_val * 0.95
        active_mask = conv >= threshold
        if not active_mask.any():
            active_mask = np.zeros(len(conv), dtype=bool)
            active_mask[int(np.argmax(conv))] = True

        active_vals = conv[active_mask]
        # Temperature-adaptive weighting
        spread = (active_vals[0] - active_vals[-1]) / (max_val + 1e-15) if len(active_vals) > 1 else 0
        temp = 5.0 / (1.0 + spread * 20.0)
        weights = np.exp((active_vals - max_val) * temp / max_val)
        weights /= weights.sum()

        active_idx = np.where(active_mask)[0]

        # Compute subgradient
        g = np.zeros(n)
        for w, k in zip(weights, active_idx):
            if w > 1e-10:
                k_int = int(k)
                i_start = max(0, k_int - (n - 1))
                i_end = min(n - 1, k_int)
                j_start = k_int - i_end
                j_end = k_int - i_start
                g[i_start:i_end + 1] += w * y[j_start:j_end + 1][::-1]

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                t_k = 1.0
                stall = 0
                patience = int(patience * 1.5)

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

        if verbose and it % 200 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, t_k={t_k:.1f}, el={time.monotonic()-start:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] * n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a * n)  # preserve total mass
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        # Two Gaussians
        centers = [n // 4, 3 * n // 4]
        widths = [n // 6, n // 6]
        x = np.arange(n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        # Triangle
        x = np.linspace(-1, 1, n)
        a = np.maximum(1.0 - np.abs(x), 0.01)
        a = a / a.sum()
    elif mode == 4:
        # Arcsine-like
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 0.01)
        a = a / a.sum()
    elif mode == 5:
        # Sine mixture
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        # Polynomial
        x = np.linspace(0, 1, n)
        a = x**2 + (1-x)**2 + 0.05
        a = a / a.sum()
    elif mode == 7:
        # Random walks
        a = np.cumsum(rng.normal(0, 1, n))
        a = a - a.min() + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a

from scipy.fft import rfft, irfft

def compute_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_cross(a, b):
    """Cross-convolution conv(a, b) via FFT. Returns array of length len(a)+len(b)-1."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    conv = compute_conv(a_norm)
    return 2.0 * n * float(np.max(conv))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        r = min(n, 2 * n - 1 - j)
        A[j:j + r, j] = a[:r]
    return A


def solve_lp_minpeak(a, timeout=3.0):
    """
    Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    a = a / float(np.sum(a))
    A = build_conv_matrix(a)

    if HAVE_CVXPY and n <= 2500:
        g = cp.Variable(n, nonneg=True)
        M = cp.Variable()
        constraints = [A @ g <= M, cp.sum(g) == 1]
        prob = cp.Problem(cp.Minimize(M), constraints)
        for solver in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    g_val = np.maximum(g.value, 0)
                    s = float(np.sum(g_val))
                    if s > 1e-10:
                        return g_val / s, float(M.value)
            except Exception:
                continue

    # Fallback to scipy HiGHS
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout})
        if res.status == 0:
            g_val = np.maximum(res.x[:n], 0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s, float(res.x[n])
    except Exception:
        pass

    return None, None


def fast_line_search(a, g):
    """
    Fast 1D line search: minimize max_k f_k(t) for t in [0,1]
    where a_t = (1-t)*a + t*g, and f_k(t) = conv(a_t)[k].
    Precomputes quadratic coefficients for O(n) evaluation per t.
    Returns (a_opt, t_opt).
    """
    n = len(a)
    conv_a = compute_conv(a)
    cross = compute_cross(a, g)
    conv_g = compute_conv(g)

    # f_k(t) = (1-t)^2 * A[k] + 2t(1-t) * C[k] + t^2 * G[k]
    #        = A[k] + 2(C[k]-A[k])*t + (A[k]-2C[k]+G[k])*t^2
    A = conv_a
    C = cross
    G = conv_g

    a_lin = 2.0 * (C - A)
    b_quad = A - 2.0 * C + G

    def f_at(t):
        vals = A + a_lin * t + b_quad * t * t
        return float(np.max(vals))

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    f_c = f_at(c_pt)
    f_d = f_at(d_pt)

    for _ in range(50):
        if hi - lo < 1e-10:
            break
        if f_c <= f_d:
            hi = d_pt
            d_pt = c_pt
            f_d = f_c
            c_pt = hi - gr * (hi - lo)
            f_c = f_at(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            f_c = f_d
            d_pt = lo + gr * (hi - lo)
            f_d = f_at(d_pt)

    t_opt = (lo + hi) / 2.0
    a_opt = (1.0 - t_opt) * a + t_opt * g
    a_opt = np.maximum(a_opt, 0)
    s = a_opt.sum()
    if s > 1e-10:
        a_opt /= s
    return a_opt, t_opt


def enforce_symmetry(a):
    """Make sequence symmetric (palindromic): a[i] = a[n-1-i]."""
    n = len(a)
    half = (n + 1) // 2
    sym = np.zeros(n)
    for i in range(half):
        sym[i] = (a[i] + a[n - 1 - i]) / 2.0
        sym[n - 1 - i] = sym[i]
    s = sym.sum()
    if s > 1e-10:
        sym /= s
    return sym

def subgradient_refine(a, time_budget=60.0, verbose=False):
    """
    Nesterov-accelerated subgradient with adaptive active set weighting.
    """
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t_k = 1.0
    y = a.copy()
    a_prev = a.copy()
    lr0 = 3.0 / np.sqrt(n)
    stall = 0
    patience = 120

    it = 0
    while time.monotonic() - start < time_budget:
        conv = compute_conv(y)
        max_val = float(np.max(conv))
        if max_val < 1e-15:
            break

        # Adaptive active set: include constraints within adaptive fraction of max
        # Use multiple temperature levels
        active_mask1 = conv >= max_val * 0.90
        active_mask2 = conv >= max_val * 0.70

        # Weight by softmax with adaptive temperature
        spread1 = max_val - float(np.min(conv[active_mask1])) if active_mask1.any() else 0
        temp1 = 10.0 / (1.0 + (spread1 / max_val) * 50.0) if max_val > 0 else 1.0
        w1 = np.where(active_mask1, np.exp((conv - max_val) * temp1), 0.0)
        w1_sum = w1.sum()
        if w1_sum > 0:
            w1 /= w1_sum

        spread2 = max_val - float(np.min(conv[active_mask2])) if active_mask2.any() else 0
        temp2 = 3.0 / (1.0 + (spread2 / max_val) * 20.0) if max_val > 0 else 1.0
        w2 = np.where(active_mask2, np.exp((conv - max_val) * temp2), 0.0)
        w2_sum = w2.sum()
        if w2_sum > 0:
            w2 /= w2_sum

        # Blend weights: 70% tight, 30% broader
        weights = 0.7 * w1 + 0.3 * w2
        w_total = weights.sum()
        if w_total > 0:
            weights /= w_total
        else:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0

        # Compute subgradient
        g = np.zeros(n)
        active_idx = np.where(weights > 1e-10)[0]
        for k in active_idx:
            w_k = weights[k]
            if w_k < 1e-10:
                continue
            k_int = int(k)
            i_start = max(0, k_int - (n - 1))
            i_end = min(n - 1, k_int)
            j_start = k_int - i_end
            j_end = k_int - i_start
            g[i_start:i_end + 1] += w_k * y[j_start:j_end + 1][::-1]

        lr = lr0 / t_k
        a_new = y - lr * g
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = a_new + (t_k / t_new) * (a_new - a_prev)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                y = best_a.copy()
                a_new = best_a.copy()
                a_prev = best_a.copy()
                y_new = best_a.copy()
                t_k = 1.0
                stall = 0
                patience = int(patience * 1.4)

        a_prev = a_new.copy()
        y = y_new.copy()
        t_k = t_new
        it += 1

        if verbose and it % 300 == 0:
            print(f"    SG it{it}: obj={best_obj:.8f}, t_k={t_k:.1f}, el={time.monotonic()-start:.1f}s")

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length, preserving total mass."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] * n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a * n)
    return np.maximum(a_new, 0)


def diverse_init(n, rng, mode=0):

    if mode == 0:
        a = np.ones(n) / n
    elif mode == 1:
        a = rng.uniform(0, 1, n)
        a = a / a.sum()
    elif mode == 2:
        centers = [n // 4, 3 * n // 4]
        widths = [n // 8, n // 8]
        x = np.arange(n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
        a = a / a.sum()
    elif mode == 3:
        x = np.linspace(-1, 1, n)
        a = np.maximum(1.0 - np.abs(x), 0.01)
        a = a / a.sum()
    elif mode == 4:
        x = np.linspace(-1, 1, n)
        a = 1.0 / np.sqrt(1 - x**2 + 0.01)
        a = a / a.sum()
    elif mode == 5:
        freqs = rng.integers(1, 6, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.25 * np.sin(2 * np.pi * f * x + p)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = x**2 + (1 - x)**2 + 0.05
        a = a / a.sum()
    elif mode == 7:
        a = np.cumsum(rng.normal(0, 1, n))
        a = a - a.min() + 0.1
        a = a / a.sum()
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining:
    - Fast LP alternating with exact quadratic line search
    - Nesterov subgradient refinement
    - Symmetry exploitation
    - Multi-n exploration
    - Structured diverse restarts
    - Adaptive time allocation
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    n_target = 800
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.clip(parent_arr, 0, None)
        if parent_arr.size > 0 and np.isfinite(parent_arr).all():
            s = float(np.sum(parent_arr))
            if s > 0.01:
                parent = parent_arr / s
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                n_target = parent.size
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        best_seq = np.ones(n_target) / n_target
        best_score = compute_obj(best_seq)
        print(f"Default init (n={n_target}): obj={best_score:.8f}")

    n_current = len(best_seq)

    # === Phase 1: Subgradient warm-up (brief) ===
    print(f"\n=== Phase 1: Subgradient Warm-up (n={n_current}) ===")
    remaining = deadline - time.monotonic()
    p1_budget = min(remaining * 0.06, 60.0)
    if p1_budget > 10:
        a_sg = subgradient_refine(best_seq, time_budget=p1_budget, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad: obj={obj_sg:.8f} (no improvement)")

    # === Phase 2: Main LP Alternating (bulk of time) ===
    print(f"\n=== Phase 2: LP Alternating (main) ===")
    remaining = deadline - time.monotonic()
    p2_total = min(remaining * 0.55, 700.0)

    # Run in rounds with progressive perturbation
    n_rounds = 4
    for rnd in range(n_rounds):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(p2_total / (n_rounds - rnd), remaining * 0.55)
        if alloc < 20:
            break

        if rnd == 0:
            a_start = best_seq.copy()
        else:
            scale = 0.015 * (rnd + 1)
            a_start = best_seq * np.exp(rng.normal(0, scale, n_current))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        a_lp = lp_alternating_v2(a_start, time_budget=alloc, verbose=True,
                                 use_symmetry=False, seed=1000 + rnd)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            n_current = len(best_seq)
            print(f"  LP round {rnd}: obj={best_score:.8f} **")
        else:
            print(f"  LP round {rnd}: obj={obj_lp:.8f} (no improvement)")

    # === Phase 3: Symmetry-constrained optimization ===
    print(f"\n=== Phase 3: Symmetric LP ===")
    remaining = deadline - time.monotonic()
    if remaining > 80:
        a_sym = enforce_symmetry(best_seq)
        sym_budget = min(remaining * 0.12, 100.0)
        a_sym_opt = lp_alternating_v2(a_sym, time_budget=sym_budget, verbose=True,
                                      use_symmetry=True, seed=5555)
        obj_sym = compute_obj(a_sym_opt)
        if obj_sym < best_score:
            best_score = obj_sym
            best_seq = a_sym_opt.copy()
            print(f"  Symmetric: obj={best_score:.8f} **")
        else:
            print(f"  Symmetric: obj={obj_sym:.8f} (no improvement)")

    # === Phase 4: Subgradient refinement on current best ===
    print(f"\n=== Phase 4: Subgradient Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 50:
        sg_budget = min(remaining * 0.08, 60.0)
        a_sg2 = subgradient_refine(best_seq, time_budget=sg_budget, verbose=True)
        obj_sg2 = compute_obj(a_sg2)
        if obj_sg2 < best_score:
            best_score = obj_sg2
            best_seq = a_sg2.copy()
            print(f"  Subgrad2: obj={best_score:.8f} **")
        else:
            print(f"  Subgrad2: obj={obj_sg2:.8f} (no improvement)")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    n_options = sorted(set([
        max(200, n_best - 150),
        max(200, n_best - 50),
        n_best + 50,
        n_best + 100,
        n_best + 200,
        400, 600, 800, 1000, 1200, 1500
    ]))
    n_options = [n for n in n_options if 200 <= n <= 2000]
    n_options.sort(key=lambda x: abs(x - n_best))

    n_tried = 0
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 50 or n_tried >= 5:
            break
        if n_try == n_best:
            continue

        a_try = resample_sequence(best_seq, n_try)
        n_tried += 1

        alloc_lp = min(remaining * 0.1, 50.0)
        if alloc_lp > 15:
            a_try = lp_alternating_v2(a_try, time_budget=alloc_lp, seed=2000 + n_try)

        obj_try = compute_obj(a_try)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            n_best = n_try
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_try:.8f}")

    # === Phase 6: Diverse restarts ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)

    for r in range(6):
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        mode = r % 8
        a_r = diverse_init(n_best, rng, mode=mode)
        blend = 0.1 + 0.04 * r
        a_r = (1 - blend) * a_r + blend * best_seq
        a_r = np.maximum(a_r, 0)
        a_r = a_r / a_r.sum()

        restart_budget = min(remaining * 0.1, 45.0)
        a_opt = lp_alternating_v2(a_r, time_budget=restart_budget, seed=3000 + r)
        obj_r = compute_obj(a_opt)

        if obj_r < best_score:
            best_score = obj_r
            best_seq = a_opt.copy()
            print(f"  Restart {r} (mode={mode}): obj={best_score:.8f} **")
        else:
            print(f"  Restart {r} (mode={mode}): obj={obj_r:.8f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()

    if remaining > 40:
        lp_alloc = min(remaining * 0.35, 120.0)
        a_lp = lp_alternating_v2(best_seq, time_budget=lp_alloc, verbose=True, seed=9999)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Final LP: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 25:
        sg_alloc = min(remaining * 0.25, 50.0)
        a_sg = subgradient_refine(best_seq, time_budget=sg_alloc, verbose=True)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  Final SG: obj={best_score:.8f} **")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_last = lp_alternating_v2(best_seq, time_budget=min(remaining * 0.5, 40.0), seed=4242)
        obj_last = compute_obj(a_last)
        if obj_last < best_score:
            best_score = obj_last
            best_seq = a_last.copy()
            print(f"  Final LP2: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
