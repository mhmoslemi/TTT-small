    while size < 2 * n - 1:

def compute_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size=size)
    return irfft(fa * fa, n=size)[:2 * n - 1]


def compute_conv_cross(a, b):
    """Full linear cross-convolution via FFT."""
    n, m = len(a), len(b)
