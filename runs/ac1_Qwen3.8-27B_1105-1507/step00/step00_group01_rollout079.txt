def solve_lp(f, pattern, M, time_limit):
    rows, cols, idxs = pattern
    if M <= 0: return None
    fs = f / f.max()

    data = fs[ idxs ].astype(np.float64)

    A = csr_matrix((data, (rows, cols)), shape=(2*len(f)-1, len(f)))

```
def fast_score(seq):
    p = np.asarray(seq, dtype=np.float64)
    if p.size == 0: return np.inf, 0.0
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(np.sum(p))
    if s < 0.01:
        return np.inf, 0.0
    pn = p / s
    conv = np.convolve(pn, pn)
    m = float(np.max(conv))
    return float(2.0 * pn.size * m), m
```

```python
"""
Search for non-negative step-function coefficients that minimize
    2 * n * max(autocorrelation) / (sum(coeff))^2.

The program uses:
  - fast parametric/random initializations,
  - an LP best-response alternating method,
  - multiplicative subgradient refinement,
  - a light random multiplicative search at the end.

The exact evaluator is assumed to be available in the global namespace as
evaluate_sequence().
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

_PATTERNS = {}


def normalize_positive(a, floor=0.0):
    """Clip to non-negative finite values and normalize to sum one."""
    arr = np.asarray(a, dtype=np.float64)
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if floor > 0.0:
        arr = np.maximum(arr, float(floor))
    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(arr.size, 1.0 / max(1, arr.size), dtype=np.float64)
    return arr / s


def fast_score(seq):
    """Return (score, max_convolution) for a normalized copy of seq."""
    p = np.asarray(seq, dtype=np.float64)
    if p.size == 0:
        return np.inf, 0.0
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(np.sum(p))
    if s < 0.01:
        return np.inf, 0.0
    p = p / s
    conv = np.convolve(p, p)
    m = float(np.max(conv))
    return float(2.0 * p.size * m), m


def exact_score(seq):
    """Call the official evaluator safely."""
    try:
        return float(evaluate_sequence([float(x) for x in seq]))
    except Exception:
        return np.inf


def make_uniform(n):
    n = max(1, int(n))
    return np.ones(n, dtype=np.float64)


def make_linear(n, eps, high_left=True):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    eps = min(max(float(eps), 0.0), 0.999)
    if high_left:
        return 1.0 + eps * (1.0 - 2.0 * x)
    return 1.0 + eps * (2.0 * x - 1.0)


def make_parabola(n, eps, up=True):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    u = 2.0 * x - 1.0
    if up:
        eps = min(max(float(eps), 0.0), 0.999)
        return 1.0 + eps * (3.0 * u * u - 1.0)
    eps = min(max(float(eps), 0.0), 0.249)
    return 1.0 + eps * (1.0 - 3.0 * u * u)


def make_fourier(n, mode, amp, phase):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    amp = min(max(float(amp), 0.0), 0.98)
    return 1.0 + amp * np.cos(2.0 * np.pi * float(mode) * x + float(phase))


def make_reciprocal(n, beta, low_end=True):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    beta = max(float(beta), 1e-6)
    if low_end:
        return 1.0 / (1.0 + beta * x)
    return 1.0 / (1.0 + beta * (1.0 - x))


def make_flat_sqrt(n, edge_factor=1.0):
    """
    Truncated formal square-root of a nearly flat convolution profile.
    This is only a heuristic initializer.
    """
    n = max(2, int(n))
    q = np.full(2 * n - 1, 1.0 / float(2 * n - 1), dtype=np.float64)

    if edge_factor != 1.0:
        edge_factor = max(0.02, min(4.0, float(edge_factor)))
        L = max(1, n // 20)
        for i in range(L + 1):
            w = edge_factor + (1.0 - edge_factor) * (float(i) / float(L))
            q[i] = w * q[i]
            q[-1 - i] = w * q[-1 - i]

    p = np.zeros(n, dtype=np.float64)
    p0 = float(np.sqrt(max(q[0], 1e-300)))
    p[0] = p0

    for k in range(1, n):
        if k > 1:
            s = float(np.dot(p[1:k], p[k - 1:0:-1]))
        else:
            s = 0.0
        val = (q[k] - 2.0 * s) / (2.0 * p0) if p0 > 0.0 else 0.0
        p[k] = max(0.0, float(val))

    return p


def make_random_fourier(n, rng):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    base = np.ones(n, dtype=np.float64)
    r = int(rng.integers(1, 5))
    max_mode = max(2, n // 6)
    for _ in range(r):
        mode = int(rng.integers(1, max_mode + 1))
        amp = float(rng.uniform(0.04, 0.16))
        if rng.random() < 0.5:
            amp = -amp
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        base += amp * np.cos(2.0 * np.pi * mode * x + phase)

    if float(np.min(base)) <= 1e-6:
        for _ in range(8):
            base = 1.0 + 0.5 * (base - 1.0)
            if float(np.min(base)) > 1e-6:
                break
    if float(np.min(base)) <= 1e-9:
        base += 1.0
    return base


def resample_sequence(seq, n, floor=0.0):
    seq = np.asarray(seq, dtype=np.float64)
    n = max(1, int(n))
    if seq.size == 0:
        return normalize_positive(np.ones(n), floor)
    if seq.size == n:
        return normalize_positive(seq, floor)
    xold = np.linspace(0.0, 1.0, seq.size)
    xnew = np.linspace(0.0, 1.0, n)
    out = np.interp(xnew, xold, seq)
    return normalize_positive(out, floor)


def consider_candidate(seq, best, best_score, floor=1e-9):
    if seq is None:
        return best, best_score
    p = normalize_positive(seq, floor=floor)
    sc, _ = fast_score(p)
    if np.isfinite(sc) and sc < best_score:
        return p.copy(), sc
    return best, best_score


def get_start_for_length(n, global_seq, rng):
    n = int(n)
    best = None
    best_score = np.inf

    cands = []
    cands.append(make_uniform(n))

    for eps in (0.55, 0.75, 0.90):
        cands.append(make_linear(n, eps, True))
        cands.append(make_linear(n, eps, False))

    for eps in (0.40, 0.70, 0.95):
        cands.append(make_parabola(n, eps, True))
    for eps in (0.10, 0.18, 0.24):
        cands.append(make_parabola(n, eps, False))

    for mode, amp, phase in (
        (1, 0.50, np.pi / 2.0),
        (2, 0.50, 0.0),
        (2, 0.70, np.pi / 3.0),
        (3, 0.50, 0.0),
        (5, 0.70, 1.0),
        (8, 0.80, 2.0),
        (16, 0.90, 0.0),
    ):
        cands.append(make_fourier(n, mode, amp, phase))

    for beta in (1.0, 2.0, 4.0, 8.0):
        cands.append(make_reciprocal(n, beta, True))
        cands.append(make_reciprocal(n, beta, False))

    if n <= 768:
        for ef in (0.30, 0.60, 1.0):
            cands.append(make_flat_sqrt(n, ef))

    for _ in range(10):
        cands.append(make_random_fourier(n, rng))

    for cand in cands:
        best, best_score = consider_candidate(cand, best, best_score)

    if global_seq is not None:
        best, best_score = consider_candidate(resample_sequence(global_seq, n), best, best_score)

    if best is None:
        best = normalize_positive(np.ones(n), 1e-9)
        best_score = fast_score(best)[0]

    return best, best_score


def get_conv_pattern(n):
    n = int(n)
    if n in _PATTERNS:
        return _PATTERNS[n]

    rows_chunks = []
    cols_chunks = []
    idx_chunks = []

    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if hi < lo:
            continue
        cols = np.arange(lo, hi + 1, dtype=np.int32)
        rows_chunks.append(np.full(cols.shape, k, dtype=np.int32))
        cols_chunks.append(cols)
        idx_chunks.append((k - cols).astype(np.int32))

    if rows_chunks:
        rows = np.concatenate(rows_chunks)
        cols = np.concatenate(cols_chunks)
        idxs = np.concatenate(idx_chunks)
    else:
        rows = np.empty(0, dtype=np.int32)
        cols = np.empty(0, dtype=np.int32)
        idxs = np.empty(0, dtype=np.int32)

    _PATTERNS[n] = (rows, cols, idxs)
    return _PATTERNS[n]


def solve_lp_for_g(f, pattern, M_base, relax, time_limit):
    """
    solves:
       maximize sum(g)
       s.t.   f * g <= M_base * relax
              g >= 0
    where * is linear convolution.  Scaling is internal for numerical stability.
    """
    f = np.asarray(f, dtype=np.float64)
    n = int(f.size)
    if n == 0:
        return None

    fmax = float(np.max(f))
    if fmax <= 1e-300:
        return None
    M_base = float(M_base)
    relax = float(relax)
    if M_base <= 0.0 or relax <= 0.0:
        return None

    Ms = M_base / fmax * relax
    if Ms <= 1e-300:
        return None

    rows, cols, idxs = pattern
    fs = f / fmax
    data = fs[idxs].astype(np.float64)

    try:
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c = -np.ones(n, dtype=np.float64)
        b = np.full(2 * n - 1, Ms, dtype=np.float64)
        options = {"presolve": True, "time_limit": float(time_limit)}
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs", options=options)
    except Exception:
        return None

    x = None
    if res is not None and hasattr(res, "x") and res.x is not None:
        x = np.asarray(res.x, dtype=np.float64)

    if x is None or x.size != n or not np.all(np.isfinite(x)):
        return None

    x = np.clip(x, 0.0, None)
    if float(np.sum(x)) <= 1e-12:
        return None

    try:
        Ac = A.dot(x)
        if np.all(np.isfinite(Ac)):
            maxv = float(np.max(Ac))
            if maxv > Ms * (1.0 + 1e-5) + 1e-12 and maxv > 0.0:
                x = x * (Ms / maxv)
    except Exception:
        pass

    return x


def smooth_reciprocal(p):
    pmax = float(np.max(p))
    eps = 1e-8 + 0.1 * pmax
    return normalize_positive(1.0 / (p + eps), 1e-9)


def run_lp_trial(n, start, budget, deadline, rng, global_best_seq, global_best_score):

            local_best_sc = ts
            local_best_m = tm
            p = trial
            cur_sc = ts
            cur_m = tm

        global_best_score = local_best_sc

        pattern = get_conv_pattern(n)
    except Exception:
        return global_best_seq, global_best_score

    trial_count = 0

        if rem < 1.5:
            break

            cur_sc = local_best_sc
            cur_m = local_best_m

        if no_imp == 0 and trial_count > 2:
            relax = 1.0

        tl = min(20.0, max(3.0, rem / 12.0))
        t0 = time.monotonic()
        g = solve_lp_for_g(p, pattern, cur_m, relax, tl)
        solve_time = time.monotonic() - t0

        if g is None or not np.all(np.isfinite(g)) or float(np.sum(g)) <= 1e-8:
            g = smooth_reciprocal(p)

        gs = float(np.sum(g))
        if gs <= 1e-12:
            g = local_best_p.copy()
            gs = 1.0
        else:
            g = g / gs

        cands = []
        cands.append(p)
        cands.append(g)
        for t in (0.25, 0.50, 0.75, 0.90):
            cands.append((1.0 - t) * p + t * g)
        cands.append(smooth_reciprocal(p))
        cands.append(0.5 * local_best_p + 0.5 * g)

        best_cand = p.copy()
        best_cand_sc = cur_sc
        best_cand_m = cur_m

        for cand in cands:
            q = normalize_positive(cand, 1e-9)
            sc, mc = fast_score(q)
            if np.isfinite(sc) and sc < best_cand_sc:
                best_cand = q
                best_cand_sc = sc
                best_cand_m = mc

        improved = False
        if best_cand_sc < local_best_sc - 1e-12:
            local_best_p = best_cand.copy()
            local_best_sc = best_cand_sc
            local_best_m = best_cand_m
            if local_best_sc < global_best_score:
                global_best_seq = local_best_p.copy()
                global_best_score = local_best_sc

        if best_cand_sc < cur_sc - 1e-12:
            p = best_cand.copy()
            cur_sc = best_cand_sc
            cur_m = best_cand_m
            improved = True
            no_imp = 0
        elif no_imp >= 2 and rng.random() < 0.25:
            p = normalize_positive(0.5 * p + 0.5 * g, 1e-9)
            cur_sc, cur_m = fast_score(p)

        if improved:
            if trial_count % 10 == 0:
                print(
                    f"LP n={n} trial={trial_count} "
                    f"score={local_best_sc:.6f} relax={relax:.2f} "
                    f"solve={solve_time:.2f}s"
                )
        else:
            no_imp += 1
            if no_imp >= 8:
                if local_best_sc < np.inf:
                    pert = local_best_p * np.exp(0.20 * rng.normal(0.0, 1.0, n))
                    p = normalize_positive(pert, 1e-9)
                    cur_sc, cur_m = fast_score(p)
                no_imp = 0

        if trial_count % 25 == 0:
            print(
                f"LP n={n} trial={trial_count} "
                f"local={local_best_sc:.6f} current={cur_sc:.6f} "
                f"global={global_best_score:.6f}"
            )

    real = exact_score(local_best_p)
    if np.isfinite(real) and real < local_best_sc:
        local_best_sc = real

    if local_best_sc < global_best_score:
        global_best_seq = local_best_p.copy()
        global_best_score = local_best_sc

    return global_best_seq, global_best_score


def derive_lag_gradient(p, lag):
    n = int(p.size)
    deriv = np.zeros(n, dtype=np.float64)
    lo = max(0, lag - (n - 1))
    hi = min(n - 1, lag)
    if hi >= lo:
        i_arr = np.arange(lo, hi + 1, dtype=np.int64)
        j_arr = lag - i_arr
        deriv[i_arr] = 2.0 * p[j_arr]
        diag = lag // 2
        if (lag % 2 == 0) and (lo <= diag <= hi):
            deriv[diag] = p[diag]
    return deriv


def subgradient_refine(n, start, budget, deadline, rng, global_best_seq, global_best_score):
    n = int(n)
    end = min(float(deadline), time.monotonic() + max(0.0, float(budget)))

    p = normalize_positive(start, 1e-9)
    cur_sc, cur_m = fast_score(p)

    local_best_p = p.copy()
    local_best_sc = cur_sc
    local_best_m = cur_m

    if global_best_seq is not None:
        trial = resample_sequence(global_best_seq, n, 1e-9)
        ts, tm = fast_score(trial)
        if np.isfinite(ts) and ts < local_best_sc:
            local_best_p = trial
            local_best_sc = ts
            local_best_m = tm
            p = trial
            cur_sc = ts
            cur_m = tm

    if local_best_sc < global_best_score:
        global_best_seq = local_best_p.copy()
        global_best_score = local_best_sc

    etas = (0.002, 0.008, 0.02, 0.06, 0.15, 0.35)
    no_imp = 0
    it = 0

    while time.monotonic() < end - 0.75:
        it += 1
        if cur_sc > local_best_sc * 1.05 + 1e-9:
            p = local_best_p.copy()
            cur_sc = local_best_sc
            cur_m = local_best_m
            no_imp = 0

        conv = np.convolve(p, p)
        active_lags = np.argpartition(conv, -max(1, n // 32))[-max(1, n // 32):]
        k = int(active_lags[int(rng.integers(0, len(active_lags)))])

        best_cand = p.copy()
        best_cand_sc = cur_sc
        best_cand_m = cur_m

        for eta in etas:
            deriv = derive_lag_gradient(p, k)
            scale = float(np.max(deriv))
            if scale <= 1e-300:
                continue
            q = p * np.exp(-float(eta) * deriv / scale)
            q = normalize_positive(q, 1e-9)
            sc, mc = fast_score(q)
            if np.isfinite(sc) and sc < best_cand_sc:
                best_cand = q
                best_cand_sc = sc
                best_cand_m = mc

        improved = False
        if best_cand_sc < local_best_sc - 1e-12:
            local_best_p = best_cand.copy()
            local_best_sc = best_cand_sc
            local_best_m = best_cand_m
            if local_best_sc < global_best_score:
                global_best_seq = local_best_p.copy()
                global_best_score = local_best_sc

        if best_cand_sc < cur_sc - 1e-12:
            p = best_cand.copy()
            cur_sc = best_cand_sc
            cur_m = best_cand_m
            improved = True
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 5:
                p = normalize_positive(local_best_p * np.exp(0.10 * rng.normal(0.0, 1.0, n)), 1e-9)
                cur_sc, cur_m = fast_score(p)
                no_imp = 0

        if it % 20 == 0:
            print(
                f"SUBGRAD n={n} it={it} "
                f"local={local_best_sc:.6f} current={cur_sc:.6f} "
                f"global={global_best_score:.6f}"
            )

    real = exact_score(local_best_p)
    if np.isfinite(real) and real < local_best_sc:
        local_best_sc = real

    if local_best_sc < global_best_score:
        global_best_seq = local_best_p.copy()
        global_best_score = local_best_sc

    return global_best_seq, global_best_score


def random_mul_search(n, start, budget, deadline, rng, global_best_seq, global_best_score):
    n = int(n)
    end = min(float(deadline), time.monotonic() + max(0.0, float(budget)))

    p = normalize_positive(start, 1e-9)
    cur_sc, cur_m = fast_score(p)

    local_best_p = p.copy()
    local_best_sc = cur_sc
    local_best_m = cur_m

    if global_best_seq is not None:
        trial = resample_sequence(global_best_seq, n, 1e-9)
        ts, tm = fast_score(trial)
        if np.isfinite(ts) and ts < local_best_sc:
            local_best_p = trial
            local_best_sc = ts
            local_best_m = tm
            p = trial
            cur_sc = ts
            cur_m = tm

    if local_best_sc < global_best_score:
        global_best_seq = local_best_p.copy()
        global_best_score = local_best_sc

    sigmas = (0.02, 0.05, 0.10, 0.18, 0.32)
    it = 0

    while time.monotonic() < end - 0.75:
        it += 1
        if cur_sc > local_best_sc * 1.08 + 1e-9:
            p = local_best_p.copy()
            cur_sc = local_best_sc
            cur_m = local_best_m

        sigma = sigmas[int(rng.integers(0, len(sigmas)))]
        q = p * np.exp(float(sigma) * rng.normal(0.0, 1.0, n))
        q = normalize_positive(q, 1e-9)
        sc, mc = fast_score(q)

        if np.isfinite(sc):
            if sc < local_best_sc - 1e-12:
                local_best_p = q.copy()
                local_best_sc = sc
                local_best_m = mc
                if local_best_sc < global_best_score:
                    global_best_seq = local_best_p.copy()
                    global_best_score = local_best_sc

            if sc < cur_sc - 1e-12:
                p = q
                cur_sc = sc
                cur_m = mc
            elif rng.random() < 0.20:
                p = q
                cur_sc = sc
                cur_m = mc

        if it % 40 == 0:
            print(
                f"RMUL n={n} it={it} "
                f"local={local_best_sc:.6f} current={cur_sc:.6f} "
                f"global={global_best_score:.6f}"
            )

    real = exact_score(local_best_p)
    if np.isfinite(real) and real < local_best_sc:
        local_best_sc = real

    if local_best_sc < global_best_score:
        global_best_seq = local_best_p.copy()
        global_best_score = local_best_sc

    return global_best_seq, global_best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main entry point.  Returns a Python list of non-negative floats with the
    lowest official score found before the deadline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 8.0)

    best_seq = None
    best_score = np.inf

    # Use the supplied parent construction if it exists and is valid.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            p0 = normalize_positive(parent, 0.0)
            sc0, _ = fast_score(p0)
            if np.isfinite(sc0) and sc0 < best_score:
                best_seq = p0
                best_score = sc0
                print(f"parent fast score = {best_score:.6f}, length = {len(best_seq)}")
    except Exception:
        pass

    # Quick initialization over several lengths.
    quick_lengths = (64, 128, 256, 512)
    for n in quick_lengths:
        if time.monotonic() > deadline - 20.0:
            break
        seq, sc = get_start_for_length(n, best_seq, rng)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq
        print(f"quick n={n} best_score={best_score:.6f}")

    # LP alternating searches.
    lp_plan = (
        (128, 80),
        (256, 180),
        (512, 360),
        (1024, 360),
    )

    for n, nb in lp_plan:
        rem = deadline - time.monotonic()
        if rem < 80.0:
            break
        budget = min(float(nb), rem - 30.0)
        if budget < 25.0:
            continue

        if best_seq is None:
            start = make_linear(n, 0.85, True)
        else:
            start = resample_sequence(best_seq, n, 1e-9)

        best_seq, best_score = run_lp_trial(n, start, budget, deadline, rng, best_seq, best_score)
        print(f"after LP n={n} best_score={best_score:.6f} length={0 if best_seq is None else len(best_seq)}")

    # Refine the current best sequence.
    rem = deadline - time.monotonic()
    nbest = 0 if best_seq is None else int(len(best_seq))
    if best_seq is not None and 2 <= nbest <= 1024 and rem > 70.0:
        budget = min(120.0, rem - 30.0)
        best_seq, best_score = subgradient_refine(
            nbest, best_seq, budget, deadline, rng, best_seq, best_score
        )
        print(f"after subgrad best_score={best_score:.6f}")

    rem = deadline - time.monotonic()
    nbest = 0 if best_seq is None else int(len(best_seq))
    if best_seq is not None and 2 <= nbest <= 1024 and rem > 45.0:
        budget = min(60.0, rem - 15.0)
        best_seq, best_score = random_mul_search(
            nbest, best_seq, budget, deadline, rng, best_seq, best_score
        )
        print(f"after random multiplicative best_score={best_score:.6f}")

    # Final safety net.
    if best_seq is None or not np.isfinite(best_score):
        best_seq = normalize_positive(make_uniform(256), 1e-9)
        best_score = fast_score(best_seq)[0]

    real = exact_score(best_seq)
    if np.isfinite(real):
        best_score = real
    else:
        best_seq = normalize_positive(make_uniform(256), 1e-9)
        best_score = 2.0

    print(f"final best_score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
```
