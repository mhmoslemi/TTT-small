import numpy as np
import time

def compute_convolution(a):
    """Compute linear self-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(fa * fa, fft_size)[:size]
    np.maximum(b, 0.0, out=b)
    return b

def compute_gradient(a, w):
    """
    Compute grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices only).
    This is 2 * cross_correlation(w, a) at lags 0..n-1.
    Computed via FFT.
    """
    n = len(a)
    w_len = len(w)  # should be 2n-1
    # We need conv(a, w[::-1]) at specific indices
    # conv_full has length n + w_len - 1
    conv_len = n + w_len - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    fw = np.fft.rfft(w[::-1], fft_size)
    conv_full = np.fft.irfft(fa * fw, fft_size)[:conv_len]
    # Extract the right portion
    return 2.0 * conv_full[n-1 : 2*n-1][::-1]
