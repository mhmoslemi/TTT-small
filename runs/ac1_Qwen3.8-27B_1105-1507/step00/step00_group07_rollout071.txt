def compute_grad(a, m):
    """
    Compute gradient direction for peak at lag m.
    Returns (grad, M, S).
    grad[j] = a[m-j]*1_{valid} - M*a[j]/S  (up to positive constant)
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.zeros(n), 0.0, S
    lo = max(0, m - n + 1)
    hi = min(m + 1, n)

    """Compute the soft-max gradient efficiently using FFT."""
    n = len(a)

    # Compute softmax weights
    shifted = conv - conv.max()
    exp_vals = np.exp(beta * shifted)
    p = exp_vals / exp_vals.sum()

Gradient-based optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses soft-max smoothed gradient with FFT-accelerated correlation,
momentum, adaptive step size, and multi-length exploration with restarts.

import numpy as np

def compute_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def softmax_grad_normalized(a, beta):
    """
    Compute soft-max gradient for normalized sequence (sum(a)=1).
    Returns (grad, score_estimate, conv, p).
    """
    n = len(a)
    conv = np.convolve(a, a)
    conv_len = 2 * n - 1
    
    # Softmax weights
    shifted = conv - conv.max()
    exp_vals = np.exp(beta * shifted)
    p = exp_vals / exp_vals.sum()
    
    M_soft = float(np.dot(p, conv))
    
    # Correlation (p ⋆ a) via FFT
    total_len = conv_len + n - 1  # = 3n - 2
    P = 1
    while P < total_len:
        P <<= 1
    
    C = np.fft.rfft(p, P)
    B = np.fft.rfft(a, P)
    corr = np.fft.irfft(C * np.conj(B), P)[:n]
    
    # Gradient (for S=1): grad[j] = corr[j] - M_soft * a[j]
    grad = corr - M_soft * a
    
    score_est = 2.0 * n * M_soft  # since S=1, score = 2n * M_soft / 1 = 2n * M_soft
    return grad, score_est, conv, p


def normalize_nonneg(a):
    """Project onto non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    S = float(np.sum(a))
    if S < 1e-10:

    return a / S
