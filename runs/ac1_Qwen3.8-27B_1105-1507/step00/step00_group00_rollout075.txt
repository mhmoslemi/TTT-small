def _peak_shave(a, c, n, N, strength=0.1):
    """Reduce the peak of c by adjusting a."""
    k_star = int(np.argmax(c))
    # The terms contributing to c[k_star] are a[j]*a[k_star-j]
    # Find the heaviest pair
    # a[j]*a[k_star-j] for j=0..n-1 (valid k_star-j)
    
    # Compute the contribution of each a[i] to c[k_star]
    # a[i] contributes to c[k_star] through: a[i]*a[k_star-i] (if valid)
    # and a[k_star-i]*a[i] (if valid, and k_star-i != i)
    # So the "pressure" on a[i] from lag k_star is: a[k_star-i] (if valid)
    
    # The total pressure from the peak lag on position i is:
    # p[i] = a[k_star-i] if 0 <= k_star-i < n, else 0
    p = np.zeros(n)
    valid = (k_star >= 0) & (k_star < 2*n-1)  # always true
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            p[i] = a[j]
    
    # Reduce positions with high pressure, increase positions with low pressure
    p_norm = p / (p.max() + 1e-10)
    
    # Multiplicative update
    a_new = a * np.exp(-strength * p_norm)
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new

"""
Multi-phase optimization to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses smooth log-sum-exp subgradients with FFT-accelerated cross-correlation,
multiplicative (exponentiated gradient) updates, peak-targeted shaving steps,
and random restarts across multiple sequence lengths.
"""

import numpy as np
import time


def _next_pow2(x):
    """Smallest power of 2 >= x."""
    n = 1
    while n < x:
        n *= 2
    return n


def _fft_conv(a, N):
    """Linear convolution a*a via FFT."""
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)


def _fft_xcorr(w, a, N):
    """Cross-correlation: out[i] = sum_k w[k]*a[k-i] (zero-padded)."""
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(np.conj(W) * A, N)


def _quick_score(a, n, N):
    """Fast score computation (assumes sum(a)=1)."""
    c = _fft_conv(a, N)[:2 * n - 1]
    return float(2.0 * n * c.max())


def _optimize_inner(a, deadline, rng):
    """
    Optimize a single sequence (sum-normalized) until deadline.
    Combines smooth gradient descent with peak shaving.
    Returns (best_a, best_internal_score).
    """
    n = len(a)
    N = _next_pow2(3 * n)

    a = a.copy()
    s = a.sum()
    if s < 1e-12:
        return a, np.inf
    a = a / s

    c = _fft_conv(a, N)[:2 * n - 1]
    best_score = c.max()  # internal (sum=1 so score = 2n*max_c)
    best_a = a.copy()

    eta = 0.4
    beta = 80.0
    no_improve = 0
    it = 0
    t0 = time.monotonic()
    last_eval = t0

    while time.monotonic() < deadline:
        c = _fft_conv(a, N)[:2 * n - 1]
        M = c.max()
        if M < 1e-15:
            break

        # Soft-max weights (normalized)
        temp = M * (2.0 * n) / beta
        w = np.exp((c - M) / temp)
        w_sum = w.sum()
        if w_sum < 1e-15:
            break
        w = w / w_sum

        # Gradient via FFT cross-correlation

        # Multiplicative update based on relative pressure

                a = np.clip(a, 0.0, None)
                s = a.sum()
                if s < 1e-10:
                    a = np.ones(n) / n
                else:
                    a = a / s

        # Peak shaving step (every 5 iterations)
        if it % 5 == 0:
            k_star = int(np.argmax(c))
            p = np.zeros(n)
            valid_start = max(0, k_star - (n - 1))
            valid_end = min(n, k_star + 1)
            for i in range(valid_start, valid_end):
                j = k_star - i
                if 0 <= j < n:
                    p[i] = a[j]
            if p.max() > 1e-12:
                shave = 0.05
                a = a * np.exp(-shave * p / p.max())
                a = np.clip(a, 0.0, None)
                s = a.sum()
                if s > 1e-10:
                    a = a / s

        it += 1

        # Periodic score check (every 15 iterations)
        if it % 15 == 0:
            c = _fft_conv(a, N)[:2 * n - 1]
            sc = c.max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Adaptive parameters
            if no_improve > 30:
                eta *= 0.93
                beta *= 1.08
                no_improve = 0

            # Frank-Wolfe nudge (randomly)
            if rng.random() < 0.08:
                gi = int(np.argmax(grad))
                mi = int(np.argmin(grad))
                if gi != mi and a[gi] > a[mi] + 1e-8:
                    delta = min(a[gi] * 0.05, (a[gi] - a[mi]) * 0.3)
                    a[gi] -= delta
                    a[mi] += delta
                    a = np.clip(a, 0.0, None)
                    s = a.sum()
                    if s > 1e-10:
                        a = a / s

            # Sharpen over time
            if it % 200 == 0:
                beta = min(beta * 1.06, 5000.0)
                eta = max(eta * 0.98, 0.01)

        # Simulated annealing: occasionally accept a random perturbation
        if it % 100 == 0 and rng.random() < 0.3:
            idx = rng.integers(0, n)
            frac = rng.uniform(0.05, 0.25)
            a_pert = a.copy()
            a_pert[idx] *= (1.0 - frac)
            # Redistribute to random other positions

            targets = rng.choice(others, size=min(10, n - 1), replace=False)
            for t in targets:
                a_pert[t] += a_pert[idx] * frac / len(targets)
            a_pert = np.clip(a_pert, 0.0, None)
            sp = a_pert.sum()

                    a = a_pert

    best_score = 2.0 * n * best_score  # Convert to actual score (sum=1 normalization)
    return best_a, best_score
