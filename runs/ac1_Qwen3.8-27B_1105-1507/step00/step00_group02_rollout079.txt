def build_conv_matrix(f):
    """Build sparse convolution matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)

    cols = []

    for k in range(2 * n - 1):

        j_max = min(n - 1, k)

            rows.extend([k] * len(idx))
            cols.extend(idx.tolist())

    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(f, M):
    """Solve LP to find direction b that maximizes sum(b) subject to (f*b)[k] <= M."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',

    if res.success:
        return res.x
    return None


def gradient_step(f, eta):
