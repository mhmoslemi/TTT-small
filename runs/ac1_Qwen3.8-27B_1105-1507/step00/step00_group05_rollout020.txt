"""
LP-based optimization of non-negative step function to minimize
J = 2*n*max(autoconv) / (sum)^2.

Uses iterative linear programming: at each step, solve LP to find a
direction that increases the sum while keeping the auto-convolution bounded.
Rescale back to unit sum. Repeat until convergence.
"""

import numpy as np
import time
from scipy.optimize import linprog


def build_convolution_matrix(s, n):
    """Build the matrix A such that (A @ s')_k = (s * s')_k."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - (n-1))
        j_max = min(k, n-1)
        A[k, j_min:j_max+1] = s[k-j_max : k-j_min+1]

    A = build_convolution_matrix(s, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

        s_prime = result.x
        sum_prime = s_prime.sum()
        if sum_prime > 1.0 + 1e-9:
            return s_prime
    return None

        c_new = np.convolve(s_prime, s_prime)

            s = s_prime
            M = M_new
            best_M = M

"""
LP-based optimization of non-negative step functions to minimize
J = 2*n*max(autoconv) / (sum)^2.

Strategy: Since J is scale-invariant, normalize sum=1 and minimize n*max(autoconv).
Iteratively solve LPs to find directions that increase sum while bounding cross-convolution.
Try multiple sequence lengths and initializations to find the best result.
"""

import numpy as np
import time
from scipy.optimize import linprog


def build_conv_matrix(s):
    """Build matrix A where (A @ x)_k = (s * x)_k for linear convolution."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        A[k, j_lo:j_hi + 1] = s[k - j_hi:k - j_lo + 1]
    return A


def solve_lp(s, M):
    """Solve LP: maximize sum(x) subject to x >= 0 and (s*x)_k <= M for all k."""
    n = len(s)
    A = build_conv_matrix(s)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        return res.x
    return None


def make_initial(n, init_type):
    """Create initial sequence based on type."""
    if init_type == 'arith':
        s = np.arange(1, n + 1, dtype=np.float64)
    elif init_type == 'geom':
        s = np.exp(0.005 * np.arange(n, dtype=np.float64))
    elif init_type == 'cosine':
        t = np.arange(n, dtype=np.float64) / n
        s = np.sin(np.pi * t) ** 2 + 0.05
    elif init_type == 'ramp_plateau':
        n1 = n // 3
        s = np.concatenate([np.arange(1, n1 + 1), np.ones(n - 2 * n1) * n1,
                            np.arange(n1 - 1, 0, -1)])
        if len(s) > n:
            s = s[:n]
        elif len(s) < n:
            s = np.pad(s, (0, n - len(s)), 'edge')
    else:
        s = np.ones(n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    return s

    """Run iterative LP optimization for given n and init type."""
    s = make_initial(n, init_type)
    conv = np.convolve(s, s)
    M = conv.max()
    best_s = s.copy()
    best_J = 2.0 * n * M
    start_time = time.monotonic()
    print(f"  n={n}, init={init_type}: J0={best_J:.6f}, solving...")

    stall_count = 0
    for it in range(1000):
        if time.monotonic() > deadline:
            break

        x = solve_lp(s, M)
        if x is None:
            break

        sum_x = x.sum()
        if sum_x <= 1.0 + 1e-8:
            break

        x_rescaled = x / sum_x
        conv_new = np.convolve(x_rescaled, x_rescaled)
        M_new = conv_new.max()
        J_new = 2.0 * n * M_new

        if M_new < M * (1.0 - 1e-10):
            s = x_rescaled
            M = M_new
            if J_new < best_J:
                best_J = J_new
                best_s = s.copy()
            stall_count = 0
            if it % 20 == 0:
                print(f"  n={n}, {init_type}, iter {it}: J={best_J:.6f} (elapsed {time.monotonic()-start_time:.1f}s)")

            improved = False
            for t in [0.5, 0.3, 0.1, 0.05]:
