import numpy as np
import time
from scipy.optimize import minimize, linprog


def uniform_sequence(n, value=1.0):
    return np.full(n, value, dtype=np.float64)


def compute_score_fast(a):
    """Fast computation of the objective for a numpy array (no security checks)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    # Use FFT for convolution for speed on large n
    if n > 128:
        conv = np.fft.irfft(np.fft.rfft(a, 2*n-1) ** 2, 2*n-1)
    else:
        conv = np.convolve(a, a)
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (s * s))


def random_restart(n, rng):
    """Generate a random non-negative sequence."""
    method = rng.integers(0, 4)
    if method == 0:
        return rng.uniform(0.1, 1.0, n)
    elif method == 1:
        # Symmetric
        half = n // 2
        h = rng.uniform(0.1, 1.0, half + 1)
        a = np.concatenate([h, h[-2:0:-1]])
        return a[:n]
    elif method == 2:
        # Sine-based
        t = np.linspace(0, 2 * np.pi * rng.integers(1, 5), n)
        return np.abs(np.sin(t)) + 0.1
    else:
        # Exponential decay with random center
        center = rng.integers(0, n)
        dist = np.abs(np.arange(n) - center)
        return np.exp(-dist / rng.uniform(n/4, n)) + 0.01


def nlbfgs_optimize(a0, maxiter=500, time_limit=None, start_time=None):
    """Optimize using L-BFGS-B with bounds."""
    def objective(x):
        return compute_score_fast(x)
    
    bounds = [(0.0, 1000.0)] * len(a0)
    
    if time_limit is not None:
        def obj_with_check(x):
            if start_time is not None and time.monotonic() - start_time > time_limit:
                raise KeyboardInterrupt
            return objective(x)
        try:
            result = minimize(obj_with_check, a0, method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
        except KeyboardInterrupt:
            return a0, objective(a0)
    else:
        result = minimize(objective, a0, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
    
    a_opt = np.clip(result.x, 0.0, 1000.0)
    return a_opt, compute_score_fast(a_opt)


def nm_optimize(a0, time_limit=None, start_time=None):
    """Optimize using Nelder-Mead (works well for small dimensions)."""
    def objective(x):

        return compute_score_fast(x)
    
    try:

    a_opt = np.clip(result.x, 0.0, 1000.0)
    return a_opt, compute_score_fast(a_opt)


def lp_based_iteration(a0, n_steps=20, M_init=None):
    """LP-based iteration from the state of the art approach."""
    a = a0.copy()
    n = len(a)
    
    for step in range(n_steps):
        # Compute current max convolution
        conv = np.convolve(a, a)
        M = np.max(conv) if M_init is None else M_init
        if step > 0:
            M = np.max(conv)
        
        # LP: maximize sum(b) subject to (a*b)[k] <= M for all k, b >= 0
        # a*b has length 2n-1
        # Variables: b[0], ..., b[n-1]
        # Constraints: for each lag l from -(n-1) to (n-1): sum of a[i]*b[j] where i-j=l <= M
        # Build constraint matrix A such that A @ b <= M
        
        # For lag l (from -(n-1) to (n-1)), the constraint is:
        # sum_{i,j: i-j=l} a[i]*b[j] <= M
        # This is equivalent to: sum_i a[i+l]*b[i] (with appropriate index handling)
        
        # Build A matrix: A has (2n-1) rows and n columns
        A = np.zeros((2*n-1, n))
        for lag in range(-(n-1), n):
            row = lag + (n-1)
            for j in range(n):
                i = j + lag
                if 0 <= i < n:
                    A[row, j] = a[i]
        
        b_oks = -A  # minimize -sum(b) subject to A@b <= M
        b_bounds = [(0, None)] * n
        
        try:
            res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                         bounds=b_bounds, method='highs')
        except Exception:
            break
        
        if res.status != 0:
            break
        
        b_opt = res.x
        sum_b = np.sum(b_opt)
        if sum_b < 1e-10:
            break
        
        # Rescale to match normalization
        sum_a = np.sum(a)
        g = (sum_a / sum_b) * b_opt
        
        # Update with small step
        t = 0.1
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, 1000.0)
        
        new_score = compute_score_fast(a_new)
        old_score = compute_score_fast(a)
        
        if new_score < old_score:
            a = a_new
        else:
            # Try smaller step
            t = 0.01
            a_new = (1 - t) * a + t * g
            a_new = np.clip(a_new, 0, 1000.0)
            new_score = compute_score_fast(a_new)
            if new_score < old_score:
                a = a_new
            else:
                break
    
    return a, compute_score_fast(a)


def block_perturbation_search(a0, rng, n_trials=200, time_limit=None, start_time=None):
    """Random block perturbation local search."""
    best = a0.copy()
    best_score = compute_score_fast(best)
    n = len(best)
    trial = 0
    
    while True:
        if time_limit is not None and start_time is not None:
            if time.monotonic() - start_time > time_limit:
                break
            trial += 1
            if trial > n_trials:
                break
        
        candidate = best.copy()
        # Pick a random block and perturb
        width = rng.integers(max(2, n // 50), max(3, n // 5))
        start = rng.integers(0, max(1, n - width))
        
        # Multiplicative perturbation
        perturbation = np.exp(rng.normal(0, 0.2, width))
        candidate[start:start+width] *= perturbation
        
        # Sometimes symmetrize
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = np.clip(candidate, 0, 1000)
        score = compute_score_fast(candidate)
        
        if score < best_score - 1e-12:
            best = candidate
            best_score = score
    
    return best, best_score
