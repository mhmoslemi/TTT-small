def _fft_convolve(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def _simplex_project(v, target_sum=1.0):
    """Project vector v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = v.copy()
    # Sort in descending order
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cv - target_sum)
    if not np.any(cond):
        # All zero case, distribute evenly
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_gradient(a, k_star):
    """Compute gradient of conv(a,a)[k_star] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    # g[i] = 2 * a[k_star - i] where 0 <= k_star - i < n
    # i ranges from max(0, k_star - n + 1) to min(n-1, k_star)
    i_lo = max(0, k_star - n + 1)
    i_hi = min(n - 1, k_star)
    idx = np.arange(i_lo, i_hi + 1)
    jdx = k_star - idx
    g[idx] = 2.0 * a[jdx]
    return g

def _annealing_step(a, conv, k_star, temp, rng):
    """Simulated annealing: perturb and accept/reject."""
    n = len(a)
    a_new = a.copy()

    # Choose perturbation type
    r = rng.random()

    if r < 0.3:
        # Block multiplicative perturbation
        width = int(rng.integers(max(2, n // 50), max(3, n // 8)))
        start = int(rng.integers(0, max(1, n - width)))
        end = min(n, start + width)
        a_new[start:end] *= np.exp(rng.normal(0, 0.15, end - start))
    elif r < 0.5:
        # Swap-based: move mass from high-contribution to low-contribution
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i != j:
            delta = 0.01 * a_new[i]
            a_new[i] = max(0, a_new[i] - delta)
            a_new[j] += delta
    elif r < 0.7:
        # Add small noise
        a_new += rng.normal(0, 0.005, n)
    elif r < 0.85:
        # Local swap of adjacent blocks
        width = int(rng.integers(2, max(3, n // 100)))
        start = int(rng.integers(0, max(1, n - 2 * width)))
        block1 = a_new[start:start + width].copy()
        block2 = a_new[start + width:start + 2 * width].copy()
        a_new[start:start + width] = block2
        a_new[start + width:start + 2 * width] = block1
    else:
        # Smooth perturbation
        smooth_kernel = np.exp(-np.arange(-5, 6) ** 2 / 8.0)
        smooth_kernel /= np.sum(smooth_kernel)
        pert = rng.normal(0, 0.02 * temp, n)
        a_new += np.convolve(pert, smooth_kernel, mode='same')

    a_new = np.maximum(a_new, 0)
    a_new = a_new / np.sum(a_new)

    return a_new

    top_k = 5
    for rank in range(min(top_k, len(init_scores))):
        if time.monotonic() > phase1_deadline:
            break
        mc, init = init_scores[rank]
        print(f"Optimizing init {rank} (max_conv={mc:.8f})...")
        result = _projected_subgradient(init, max_iters=50000, deadline=phase1_deadline)
        conv_r = _fft_convolve(result)
        mc_r = np.max(conv_r)
        if mc_r < best_max_conv:
            best_max_conv = mc_r
            best_a = result.copy()
            print(f"  Improved! max_conv={mc_r:.8f}, score≈{2*n*mc_r:.6f}")

    # Phase 2: Simulated annealing from best
    print(f"\nPhase 2: Annealing from max_conv={best_max_conv:.8f}")
    a = best_a.copy()
    temp = 0.1
    anneal_deadline = total_deadline
    consecutive_rejects = 0

    for it in range(200000):
        if time.monotonic() > anneal_deadline:
            break
        if it % 5000 == 0:
            conv_check = _fft_convolve(a)
            print(f"  Anneal iter {it}, max_conv={np.max(conv_check):.8f}, temp={temp:.4f}")

        conv = _fft_convolve(a)
        k_star = int(np.argmax(conv))
        current_best = conv[k_star]

        a_new = _annealing_step(a, conv, k_star, temp, rng)
        conv_new = _fft_convolve(a_new)
        new_best = np.max(conv_new)

        if new_best < current_best:
            a = a_new
            if new_best < best_max_conv:
                best_max_conv = new_best
                best_a = a.copy()
            consecutive_rejects = 0
            temp = min(temp * 1.005, 0.5)
        else:
            # Metropolis acceptance
            delta = new_best - current_best
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
                a = a_new
                consecutive_rejects = 0
            else:
                consecutive_rejects += 1

            if consecutive_rejects > 200:
                temp *= 0.9
                consecutive_rejects = 0
            temp = max(temp, 0.001)

    # Phase 3: Fine subgradient polishing
    print(f"\nPhase 3: Polishing from max_conv={best_max_conv:.8f}")
    if time.monotonic() < total_deadline:
        result = _projected_subgradient(best_a, max_iters=50000, deadline=total_deadline)
        conv_r = _fft_convolve(result)
        mc_r = np.max(conv_r)
        if mc_r < best_max_conv:
            best_max_conv = mc_r
            best_a = result.copy()
            print(f"  Polished! max_conv={mc_r:.8f}")

    # Final: try a few random restarts if time remains
    if time.monotonic() < total_deadline - 10:
        for _ in range(5):
            if time.monotonic() > total_deadline - 10:
                break
            # Small random perturbation of best
            a_try = best_a + rng.normal(0, 0.01, n)
            a_try = _simplex_project(a_try, 1.0)
            result = _projected_subgradient(a_try, max_iters=5000, deadline=total_deadline)
            conv_r = _fft_convolve(result)
            mc_r = np.max(conv_r)
            if mc_r < best_max_conv:
                best_max_conv = mc_r
                best_a = result.copy()
                print(f"  Restart improved! max_conv={mc_r:.8f}")

    # Evaluate final
    final_score = evaluate_sequence(best_a.tolist())
    print(f"\nFinal score: {final_score}")
    print(f"Best max_conv: {best_max_conv:.8f}")

    # Scale to get good absolute values (score is scale-invariant)
    # Keep sum around sqrt(2n) for numerical stability
    target_sum = np.sqrt(2 * n)
    best_a = best_a * target_sum

    return [float(x) for x in best_a]
