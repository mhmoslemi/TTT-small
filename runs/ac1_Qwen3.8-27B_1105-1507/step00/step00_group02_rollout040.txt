    b = np.convolve(f, f)
    max_b = b.max()
    
    if max_b <= 0:
        return f, 2*n*max_b / (f.sum()**2)  # Should be 0, but handle edge case
    
    # Find active set (positions where b is close to max)
    tol = 0.005 * max_b
    active_mask = (b >= max_b - tol).astype(float)
    
    if active_mask.sum() == 0:

    grad = h[n-1 : 2*n-1][::-1] / active_mask.sum()

    sum_f = f.sum()
    # d/d f[i] [log(max_b) - 2*log(sum_f)] = grad[i]/max_b - 2/sum_f
    direction = grad / max_b - 2.0 / sum_f
    
    # Descent direction is -direction
    f_new = f - t * direction
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-10:

    f_new = f_new / s * np.sqrt(2 * n)
    
    return f_new, 2*n*max_b / (sum_f**2)

def compute_subgradient(f, active_mask):
    """Compute the subgradient of max(conv(f,f)) w.r.t. f."""
    n = len(f)
    # grad[i] = (1/|S|) * sum_j f[j] * active_mask[i+j]
    # Use convolution: grad[i] = h[2n-2-i] where h = conv(f, flip(active_mask))
    g = active_mask[::-1]
    h = np.convolve(f, g, mode='full')

    grad = h[n-1:2*n-1][::-1]

from scipy.optimize import linprog

def lp_direction(f, M):
    """Solve the LP to find the best direction g."""
    n = len(f)
    # Variables: g[0], ..., g[n-1] >= 0
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
    # For row k: the coefficient of g[i] is f[k-i] if 0<=k-i<n, else 0
    # i.e., A[k, i] = f[k-i] if 0 <= k-i <= n-1, else 0
    
    # Build the constraint matrix (sparse would be better, but let's try dense first)
    # For n=5000, the matrix is 9999 x 5000 = ~50 million entries. That's ~400 MB.
    # Might be too much for dense. Let's use sparse.
    
    from scipy.sparse import csr_matrix
    import scipy.sparse as sp
    
    # For row k (0 to 2n-2), the non-zero entries are in columns i where 0 <= k-i <= n-1
    # i.e., max(0, k-(n-1)) <= i <= min(k, n-1)
    # The value at (k, i) is f[k-i]
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Solve: min c^T g s.t. A g <= M (scalar), g >= 0
    # In scipy: minimize c^T x s.t. A_ub x <= b_ub, bounds
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return None

import numpy as np
import time

def subgradient_step(f, b, t):
    """One subgradient step. Returns new f and new b."""
    n = len(f)
    max_b = b.max()
    
    if max_b <= 0:
        return f, b
    
    # Active set
    tol = 0.001 * max_b
    active = (b >= max_b - tol)
    
    if active.sum() == 0:
        # Fallback: just the argmax
        idx = b.argmax()
        active = np.zeros(len(b), dtype=bool)
        active[idx] = True
    
    # Subgradient: grad[i] = (1/|S|) * sum_j f[j] * active[i+j]
    # = cross-correlation of f and active, divided by |S|
    # Using convolution: grad = conv(f, active_flipped)[n-1 : 2n-1][::-1] / |S|
    active_f = active.astype(np.float64)
    h = np.convolve(f, active_f[::-1])
    grad = h[n-1:2*n-1][::-1] / active.sum()
    
    # Direction
    sum_f = f.sum()
    direction = grad / max_b - 2.0 / sum_f
    
    # Step
    f_new = f - t * direction
    f_new[f_new < 0] = 0
    s = f_new.sum()
    if s < 1e-10:
        return f, b
    f_new = f_new / s * np.sqrt(2 * n)
    
    # Recompute b
    b_new = np.convolve(f_new, f_new)
    
    return f_new, b_new
