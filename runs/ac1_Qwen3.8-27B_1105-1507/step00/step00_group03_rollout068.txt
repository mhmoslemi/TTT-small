```python
        _try_update((1.0 - nn / n) ** 2 + 0.01, state)

    if state['best_seq'] is None:
        fallback = np.ones(256, dtype=np.float64)
        state['best_seq'] = fallback
        state['best_score'] = float(evaluate_sequence(fallback.tolist()))

    print(f"Phase 1 complete: best={state['best_score']:.6f}", flush=True)

    # Phase 2: Local search from best parametric solution
    print("Phase 2: Local search", flush=True)
    if time.monotonic() < deadline - 10:
        _local_search(state['best_seq'], state, deadline, rng)

    # Phase 3: Try a few different n values with fresh local search
    if time.monotonic() < deadline - 30:
        print("Phase 3: Additional lengths", flush=True)
        for n_try in [3000, 8000, 15000]:
            if time.monotonic() > deadline - 20:
                break
            idx = np.arange(1, n_try + 1, dtype=np.float64)
            seq = idx ** 1.0  # linear ramp
            _local_search(seq, state, deadline, rng)

    # Final
    best = state['best_seq']
    if best is None:
        best = np.ones(256, dtype=np.float64)
    final = [float(max(0.0, min(1000.0, x))) for x in best]
    fs = float(evaluate_sequence(final))
    print(f"FINAL: score={fs:.6f} n={len(final)} total_time={time.monotonic()-t0:.1f}s", flush=True)
    if not np.isfinite(fs):
        return [1.0] * 256
    return final
```
