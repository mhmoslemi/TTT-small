"""
Search for a non-negative step function minimizing
    2*n*max(convolve(a,a)) / (sum(a))**2
using a multi-phase strategy:
  1. L-BFGS-B on a smooth log-sum-exp surrogate of the max-convolution objective,
     tried over several lengths and several analytic initialisations.
  2. Refinement of the global best with larger surrogate sharpness (beta) and
     a linear-programming picket/peak-damping step (HiGHS via scipy.linprog).
  3. Short random local search to polish.
Characteristic gradient of the surrogate F = 2n*g(a)/Sa^2, g=(1/beta)logsumexp(beta*conv)
is 4n*(c*Sa - g)/Sa^3 with c = correlate(softmax(beta*conv), a, 'valid').
"""
import numpy as np
import time
import scipy.optimize as opt
from scipy.signal import fftconvolve

_deadline = 0.0
_cur_beta = 100.0
_cur_best = None
_cur_best_f = np.inf


def _remaining():
    return max(0.0, _deadline - time.monotonic())


def _real_score(a):
    s = evaluate_sequence([float(x) for x in a])
    return s if np.isfinite(s) else np.inf


def _smooth_obj(x, beta):
    n = x.size
    b = fftconvolve(x, x, mode='full')
    Sa = x.sum()
    if Sa < 1e-12:
        return 1e10
    mb = b.max()
    e = np.exp(beta * (b - mb))
    se = e.sum()
    g = mb + np.log(se) / beta
    return 2.0 * n * g / (Sa * Sa)


def _smooth_grad(x, beta):
    n = x.size
    b = fftconvolve(x, x, mode='full')
    Sa = x.sum()
    if Sa < 1e-12:
        return np.zeros(n)
    mb = b.max()
    e = np.exp(beta * (b - mb))
    se = e.sum()
    g = mb + np.log(se) / beta
    w = e / se
    c = np.correlate(w, x, mode='valid')  # c[j] = sum_m w[m+j] x[m]
    return 4.0 * n * (c * Sa - g) / (Sa * Sa * Sa)


def _record_cb(xk):
    global _cur_best, _cur_best_f
    try:
        f = _smooth_obj(xk, _cur_beta)
        if f < _cur_best_f:
            _cur_best_f = f
            _cur_best = np.array(xk, dtype=np.float64)
    except Exception:
        pass
    return False


def _lbfgsb(a0, beta, maxiter):
    global _cur_beta, _cur_best, _cur_best_f
    _cur_beta = beta
    _cur_best = np.array(a0, dtype=np.float64)
    _cur_best_f = _smooth_obj(a0, beta)
    n = a0.size
    bounds = [(0.0, 1000.0)] * n
    try:
        opt.minimize(_smooth_obj, a0, args=(beta,), jac=_smooth_grad,
                     method='L-BFGS-B', bounds=bounds, callback=_record_cb,
                     options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-7,
                              'maxfun': max(2 * maxiter, 200)})
    except Exception:
        pass
    return np.clip(_cur_best, 0.0, 1000.0)


def _build_A(a):
    n = a.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_refine(a, seconds):
    n = a.size
    if n < 8 or _remaining() < 2.0:
        return a
    b = fftconvolve(a, a, mode='full')
    M = b.max()
    Sa = a.sum()
    cur = 2.0 * n * M / (Sa * Sa)
    try:
        A = _build_A(a)
        res = opt.linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'time_limit': max(0.5, min(seconds, _remaining() - 1.0))})
    except Exception:
        return a
    if not getattr(res, 'success', False) or res.x is None:
        return a
    g0 = np.clip(res.x, 0.0, 1000.0)
    Sg = g0.sum()
    if Sg < 1e-10:
        return a
    best = a.copy()
    best_score = cur
    for t in (0.05, 0.1, 0.15, 0.2, 0.3, 0.5):
        a_new = (1.0 - t) * a + t * (Sa / Sg) * g0
        a_new = np.clip(a_new, 0.0, 1000.0)
        s = a_new.sum()
        if s < 0.01:
            continue
        bn = fftconvolve(a_new, a_new, mode='full')
        sc = 2.0 * n * bn.max() / (s * s)
        if sc < best_score:
            best = a_new
            best_score = sc
    return best


def _make_inits(n, rng):
    inits = []
    inits.append(np.ones(n))
    t = np.arange(n) / max(n - 1, 1)
    inits.append(0.5 * (1.0 + np.cos(2 * np.pi * t)))
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    if tri.sum() > 0:
        inits.append(tri)
    x = np.arange(n) - n / 2.0
    g = np.exp(-x * x / (2.0 * (n / 6.0) ** 2))
    inits.append(g)
    r = rng.random(n) + 0.1
    inits.append(r)
    cb = np.cos(np.pi * (np.arange(n) - n / 2.0 + 0.5) / n) ** 2
    inits.append(cb + 1e-6)
    return inits

def propose_candidate(seed=42, budget_s=1500, **kwargs):

    def _consider(a):
