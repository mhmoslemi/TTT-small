"""
Minimizes score = 2*n*max(f*f)/(sum f)^2 by multi-phase search over
non-negative step-like sequences.

Approach:
  * Use FFT-based self-convolution for fast, exact-enough scoring.
  * Phase 1: diverse initializations (parent + step/ramp sequences of several
    lengths) -> keep the best as the working base.
  * Phase 2: simulated-annealing local search with adaptive block
    multiplicative / Gaussian perturbations plus occasional reversal /
    peak-targeted reductions.
  * Phase 3: smoothed projected gradient descent on a soft-max surrogate of
    max(f*f) to refine the working base.
  * Periodically re-attempt at other lengths to avoid long sequences trapping
    us in weak basins; keep global best.
"""

import numpy as np
import time


def _next_pow2(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def fast_self_convolve(a):
    """Full linear self-convolution a*a via FFT, length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    needed = 2 * n - 1
    size = _next_pow2(needed)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:needed]


def fast_score(a):
    """Quick score matching evaluate_sequence for a in [0,1000], sum>=0.01."""
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = fast_self_convolve(a)
    return 2.0 * float(a.size) * float(np.max(conv)) / (s * s)


def normalize_sum1(a):
    s = float(np.sum(a))
    if s <= 0.0:
        return a.copy()
    return a / s


def block_perturb(base, rng, mode):
    """Return a non-negative perturbed copy of base using one of the modes."""
    f = base.copy()
    n = f.size
    r = rng.random()

    # block widths / count
    nb = int(rng.integers(1, 4))
    for _ in range(nb):
        width = int(rng.integers(max(1, n // 400), max(2, n // 8)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        seg = slice(start, min(n, start + width))

        if mode == 0:
            # multiplicative log-normal (local scaling)
            tau = float(rng.normal(0.0, 0.18))
            f[seg] = f[seg] * float(np.exp(tau))
        elif mode == 1:
            # small additive gauss, kept nonneg
            sigma = 0.05 * max(1.0 / n, np.mean(f))
            f[seg] += rng.normal(0.0, sigma, width)
        elif mode == 2:
            # swap-ish blend inside block (reorder mass)
            g = f[seg].copy()
            noise = rng.normal(0.0, 0.25 * (np.mean(g) + 1e-12), width)
            g = 0.7 * g + 0.3 * noise
            g = np.clip(g, 0.0, None)
            f[seg] = g
        else:
            # tooth: down some of the block
            f[seg] *= float(np.exp(float(rng.normal(-0.2, 0.2))))

    if r < 0.18:
        f = 0.5 * (f + f[::-1])
    f = np.clip(f, 0.0, 1000.0)
    return f


def peak_reduce(base, rng, strength):
    """Reduce elements contributing most to the current max convolution."""
    f = base.copy()

    conv = fast_self_convolve(f)

def softgrad_step(base, lr, temp):
    """Projected step on soft-max surrogate of max(f*f)."""
    f = base.copy()
    n = f.size
    conv = fast_self_convolve(f)
    z = conv - conv.max()
    w = np.exp(temp * z)
    w = w / w.sum()
    # H[j] = sum_k f[k] w[k+j]  via circular correlation with zero padding
    N = _next_pow2(3 * n)
    Ff = np.fft.rfft(f, N)
    Ww = np.fft.rfft(w, N)
    H = np.fft.irfft(Ff * np.conj(Ww), N)[:n]  # H[j] = sum_k f[k] w[k+j] (linear range)
    g = 2.0 * H
    g = g - np.mean(g)
    f = f - lr * g
    f = np.clip(f, 0.0, 1000.0)
    return f


def make_step_n(n, ratio, rng):
    """Step-like base: lower plateau then higher plateau, with light noise."""
    f = np.zeros(n)
    k = int(n * ratio)
    f[:k] = 1.0
    f[k:] = 1.7
    # gentle taper at the boundary to reduce sharp peak
    bw = max(1, n // 64)
    if k - bw > k and k + bw < n:
        pass
    noise = rng.normal(0.0, 0.03, n)
    f = f + noise
    return np.clip(f, 0.0, 1000.0)
