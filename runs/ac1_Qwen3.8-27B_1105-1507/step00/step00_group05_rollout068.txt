"""
Subgradient / backtracking search for low "reversal-convolution peak" sequences.

The score J = 2*n*max(convolve(a,a))/(sum a)^2 is scale-invariant.  The trivial
constant sequence sits at J=2 and is a *symmetric* local optimum; my earlier
runs got stuck there because random block moves were symmetrised back toward
the constant.  The key escape is to work on genuinely *asymmetric* seeds and
descend a smoothed sub-gradient of the max convolution cell (computed with an
FFT convolution, so every step is O(n log n)), with adaptive backtracking line
search and momentum.  Multiple diverse asymmetric seeds are evolved in parallel
under a shared time budget and the best (validated by evaluate_sequence) is
returned.
"""

import time
import numpy as np
from scipy.signal import fftconvolve


# ---------------------------------------------------------------------------
# Fast exact objective helpers (must match evaluate_sequence exactly)
# ---------------------------------------------------------------------------
def fast_conv(a):
    """Full linear convolution of a with itself (real, length 2n-1)."""
    n = a.size
    return np.real(fftconvolve(a, a, mode="full"))


def fast_J(a):
    """Exact 2*n*max(conv)/(sum)^2 with the same guards as evaluate_sequence."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    m = float(np.max(fast_conv(a)))
    return float(2.0 * a.size * m / (s * s))


# ---------------------------------------------------------------------------
# Sub-gradient of score w.r.t. the top-K maximising convolution cells
# ---------------------------------------------------------------------------
def _dcell(a, m, n):
    """d(conv[m])/da_i  =  a[m-i] + a[i-m]  (out-of-range treated as 0)."""
    idx = np.arange(n)
    p = m - idx
    q = idx - m
    tp = np.zeros(n, dtype=np.float64)
    tq = np.zeros(n, dtype=np.float64)
    mp = (p >= 0) & (p < n)
    mq = (q >= 0) & (q < n)
    tp[mp] = a[p[mp]]
    tq[mq] = a[q[mq]]
    return tp + tq


def subgrad(a, K=10, beta=8.0):
    """A sub-gradient of J(a) using a softmax average over the top-K cells."""
    n = a.size
    s = float(a.sum())
    conv = fast_conv(a)
    M = float(conv.max())
    # softmax weights over the top K cells (numerically stable)
    topk = np.argpartition(conv, -K)[-K:]
    cv = conv[topk]
    w = np.exp(beta * (cv - cv.max()))
    w /= w.sum()
    gradM = np.zeros(n, dtype=np.float64)
    for c, wt in zip(topk, w):
        gradM += wt * _dcell(a, c, n)
    # J = 2n M / s^2  ->  dJ/da = (2n/s^2)*dM - (4n M/s^3)
    return (2.0 * n / (s * s)) * gradM - (4.0 * n * M / (s ** 3))


# ---------------------------------------------------------------------------
# Asymmetric seed builders (all return sum-normalised, non-negative vectors)
# ---------------------------------------------------------------------------
def _norm1(v):
    v = np.clip(np.asarray(v, dtype=np.float64), 0.0, None)
    s = v.sum()
    return v / s if s > 0 else v


def seed_ramp_down(n):
    return _norm1(np.linspace(n, 1.0, n))


def seed_ramp_up(n):
    return _norm1(np.linspace(1.0, n, n))


def seed_asym_block(n, rng):
    a = np.ones(n)
    cut = int(n * rng.uniform(0.35, 0.65))
    a[:cut] *= rng.uniform(1.5, 4.0)
    a[cut:] *= rng.uniform(0.1, 0.6)
    return _norm1(a)


def seed_bump_asym(n, rng):
    ctr = n * rng.uniform(0.25, 0.45)
    width = n * rng.uniform(0.05, 0.2)
    x = np.arange(n)
    a = np.exp(-((x - ctr) ** 2) / (2.0 * width ** 2))
    # break the symmetry with a tilt
    a *= np.linspace(0.3, 2.0, n)
    return _norm1(a)


def seed_dual_bump(n, rng):
    c1 = n * 0.30
    c2 = n * 0.72
    w1 = n * 0.10
    w2 = n * 0.04
    x = np.arange(n)
    a = np.exp(-((x - c1) ** 2) / (2.0 * w1 ** 2)) + \
        0.7 * np.exp(-((x - c2) ** 2) / (2.0 * w2 ** 2))
    return _norm1(a)


def seed_random_asym(n, rng):
    a = rng.random(n)
    a *= np.linspace(3.0, 0.1, n)
    return _norm1(a)


# ---------------------------------------------------------------------------
# Adaptive backtracking + momentum optimiser on a single seed
# ---------------------------------------------------------------------------
def optimize_seed(seed, wall=None, rng=None, K=10, beta=8.0):
    a = _norm1(seed)
    n = a.size
    if rng is None:
        rng = np.random.default_rng()
    cur = fast_J(a)
    best = a.copy()
    bestJ = cur
    v = np.zeros(n, dtype=np.float64)
    lr = 0.02
    lr_min = 1e-5
    mu = 0.55
    step = 0
    t0 = time.monotonic()
    while True:
        if wall is not None and (time.monotonic() - t0) > wall:
            break
        g = subgrad(a, K=K, beta=beta)
        v = mu * v - lr * g
        cand = a + v
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s <= 0 or not np.isfinite(s):
            lr *= 0.5
            v *= 0.5
            continue
        cand = cand / s          # project back to sum=1 (J is scale invariant)
        nj = fast_J(cand)
        step += 1
        if np.isfinite(nj) and nj < cur:
            a = cand
            cur = nj
            lr = min(lr * 1.03, 0.25)
        else:
            # backtracking: shrink step, allow rare small excursions via momentum
            lr = max(lr * 0.6, lr_min)
            v *= 0.3
            if step % 4 == 0:
                # occasional asymmetric kick to leave flat regions
                c = int(rng.integers(0, n))
                width = max(2, n // 40)
                c2 = min(n, c + width)
                a[c:c2] *= np.exp(rng.normal(0.0, 0.4))
                a = _norm1(a)
                cur = fast_J(a)
        if np.isfinite(cur) and cur < bestJ:
            bestJ = cur
            best = a.copy()
        if lr < lr_min * 0.5:
            break
    return best, float(bestJ)

        _consider(parent, sc)

    # first pass: give every seed a short budget to escape 2.0
    per = max(5.0, min(40.0, (deadline - time.monotonic()) / max(1, len(seeds))))
    for i, sd in enumerate(seeds):
        if time.monotonic() > deadline:
            break
        wall = min(per, max(2.0, deadline - time.monotonic()))
        seq, sc = optimize_seed(sd, wall=wall, rng=rng, K=10, beta=8.0)
        print(f"pass1 seed{i} score={sc:.6f} t={time.monotonic()- (deadline- float(budget_s)+4.0):.0f}s")
        _consider(seq, sc)

    # second pass: pour remaining time into the current best (and a couple more
    # fresh asymmetric seeds) with heavier relaxation parameters
    floor = time.monotonic()
    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 4.0:
            break
        start = best_seq if best_seq is not None else seed_asym_block(n, rng)
        # fresh diversity is occasionally worth a slice of the budget
        if rng.random() < 0.25 and rem > 30.0:
            start = seed_bump_asym(n, rng) if rng.random() < 0.5 else seed_dual_bump(n, rng)
        wall = rem * 0.95
        # gradually increase K / beta for smoother landscape as we refine
        K = 16 if best_score < 1.95 else 10
        beta = 12.0 if best_score < 1.9 else 8.0
        seq, sc = optimize_seed(start, wall=wall, rng=rng, K=K, beta=beta)
        print(f"pass2 best_score={sc:.6f} (global {best_score:.6f}) "
              f"elapsed={time.monotonic()-floor:.0f}s")
        _consider(seq, sc)

    if best_seq is None:
        best_seq = seed_ramp_down(n)
        best_score = fast_J(best_seq)

    # final authoritative check with the provided evaluator on top candidates
    cands = [best_seq]
    if parent is not None:
        cands.append(parent)
    cands.append(seed_ramp_down(n))
    final = best_seq
    final_ev = np.inf
    for c in cands:
        ev = float(evaluate_sequence([float(x) for x in c]))
        if np.isfinite(ev) and ev < final_ev:
            final_ev = ev
            final = c
    print("FINAL evaluate_sequence:", final_ev, "fast_J:", fast_J(final))

    out = [float(x) for x in final]
    s = sum(out)
    if s < 0.01:  # guard: evaluator rejects near-zero sums
        out = [1.0 / len(out)] * len(out)
    return out
