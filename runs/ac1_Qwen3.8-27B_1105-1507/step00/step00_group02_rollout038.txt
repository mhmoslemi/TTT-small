import numpy as np
import time

def fast_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def compute_score(a):
    """Compute the score for a non-negative sequence."""
    n = len(a)
    b = fast_convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def gradient_descent(a, lr, max_iters, rng):
    """Subgradient descent with projection onto non-negative orthant."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(max_iters):
        b = fast_convolve(a, a)
        S = np.sum(a)
        M = np.max(b)
        if S < 0.01:
            break
        score = 2.0 * n * M / (S ** 2)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find active set (where b_k == M, within tolerance)
        active = np.where(b >= M * (1 - 1e-10))[0]
        
        # Compute subgradient: grad[i] = (2/|active|) * sum_{k in active} a[k-i] (with bounds)
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= len(active)
        
        # Full gradient of f
        g = 2.0 * n / (S ** 3) * (S * grad - M)
        
        # Update
        a -= lr * g
        a = np.maximum(a, 0)
        
        # Occasionally add noise
        if rng.random() < 0.01:
            a *= np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
    
    return best_a, best_score

def fast_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def project_simplex(v):
    n = len(v)
    if np.max(v) <= 0:
        return np.ones(n) / n
    lo = 0.0
    hi = float(np.max(v))
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.maximum(v - lam, 0)

For the uniform sequence, the convolution is triangular: [1, 2, 3, ..., n, ..., 3, 2, 1] * (1/n^2).
The peak is at the center with value n/n^2 = 1/n.
To reduce the peak, I need to reduce b[n-1] = sum a_i * a_{n-1-i}.

If I make a slight perturbation a = u + ε*v (where u is uniform and v is a perturbation with sum(v)=0), then:
b[n-1] = sum (u_i + ε*v_i)(u_{n-1-i} + ε*v_{n-1-i})
= sum u_i*u_{n-1-i} + ε*sum(u_i*v_{n-1-i} + v_i*u_{n-1-i}) + O(ε^2)
= 1/n + ε*sum(u_i*(v_{n-1-i} + v_i)) + O(ε^2)
= 1/n + ε/n * sum(v_{n-1-i} + v_i) + O(ε^2)
= 1/n + 2ε/n * sum(v_i) [by reindexing] + O(ε^2)
= 1/n + O(ε^2) [since sum(v)=0]

So the first-order change in b[n-1] is zero (due to the sum(v)=0 constraint). The second-order term is ε^2 * sum(v_i * v_{n-1-i}), which can be positive or negative.

For the b[0] = a[0]^2 = (u[0] + ε*v[0])^2 = u[0]^2 + 2ε*u[0]*v[0] + O(ε^2) = 1/n^2 + 2ε*v[0]/n + O(ε^2).

For the max to decrease, I need to decrease b[n-1] (the peak) while not increasing b[0] or other edges too much.

The second-order change in b[n-1] is ε^2 * sum(v_i * v_{n-1-i}). To make this negative, I need sum(v_i * v_{n-1-i}) < 0. This means v should be "anti-symmetric" in the sense that v_i and v_{n-1-i} tend to have opposite signs.

For example, v_i = sin(π*i/n) for i=0,...,n-1. This has v_i * v_{n-1-i} = sin(πi/n)*sin(π(n-1-i)/n) = sin(πi/n)*sin(π - π(n-1-i)/n) = sin(πi/n)*sin(π(n-1-i)/n).
Since sin(π(n-1-i)/n) = sin(π - π(i+1)/n) = sin(π(i+1)/n).
So v_i * v_{n-1-i} = sin(πi/n)
