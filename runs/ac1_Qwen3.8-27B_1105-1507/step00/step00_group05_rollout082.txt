```python
import numpy as np
import time
from scipy import optimize
from scipy.sparse import csr_matrix

def compute_conv_max_and_sum(f):
    """Compute max of convolution and sum of f."""
    conv = np.convolve(f, f)
    return np.max(conv), np.sum(f)

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. conv(f,g)[k] <= M, g >= 0."""
    n = len(f)
    # Build constraint matrix A (sparse)
    # A[k, j] = f[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max c^T g s.t. A g <= M, g >= 0
    # In min form: min -c^T g s.t. A g <= M, g >= 0
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return None

def lp_iteration(f, t=0.5):
    """One LP-based iteration."""
    n = len(f)
    # Normalize
    S = np.sum(f)
    if S < 1e-10:
        return f
    f_norm = f * (np.sqrt(2*n) / S)
    
    # Compute M
    M = np.max(np.convolve(f_norm, f_norm))
    
    # Solve LP
    g = solve_lp(f_norm, M)
    if g is None or np.sum(g) < 1e-10:
        return f_norm
    
    # Rescale g
    g = g * (np.sqrt(2*n) / np.sum(g))
    
    # Update
    f_new = (1 - t) * f_norm + t * g
    
    return f_new
```
