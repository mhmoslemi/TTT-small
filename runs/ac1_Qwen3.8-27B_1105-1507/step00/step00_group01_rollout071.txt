    return 2.0 * n * np.max(conv) / (s * s)


def smooth_objective_and_gradient(a, t):
    """
    Compute F(a) = (1/t)*log(sum exp(t*conv)) - 2*log(sum(a))
    and its gradient.
    """
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return 1e10, np.zeros(n)
    conv = np.convolve(a, a)
    tx = t * conv
    mx = np.max(tx)
    exp_vals = np.exp(tx - mx)
    total = np.sum(exp_vals)
    f = (mx + np.log(total)) / t - 2.0 * np.log(s)
    p = exp_vals / total
    g = np.convolve(a[::-1], p, mode='full')[n - 1:2 * n - 1]
    grad = 2.0 * g - 2.0 / s
    return f, grad

    """Objective only for scipy minimize."""
    f, _ = smooth_objective_and_gradient(a, t)
    return f


def smooth_gradient_fn(a, t):
    """Gradient only for scipy minimize."""
    _, g = smooth_objective_and_gradient(a, t)
    return g

    n = len(a)
    current = a.copy()
    t_schedule = [10, 30, 100, 300, 1000]
    max_iter = [100, 100, 100, 80, 60]
    for i, t in enumerate(t_schedule):

            break
        try:
            res = minimize(
                smooth_objective_fn, current, args=(t,),
                jac=smooth_gradient_fn,
                method='L-BFGS-B',
                bounds=[(0.0, None)] * n,
                options={'maxiter': max_iter[i], 'ftol': 1e-14, 'gtol': 1e-10}
            )
            if np.sum(res.x) > 0.01:
                current = np.maximum(res.x, 0.0)
        except Exception:
            pass
    return current


def run_lp_refinement(a, deadline, max_iters=5, step=0.05):
    """LP-based refinement: find direction b that maximizes sum(b) s.t. a*b <= M."""
    n = len(a)
    current = a.copy()
    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = np.max(np.convolve(current, current))
        # Build sparse constraint matrix A (2n-1) x n
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                indices = range(lo, hi + 1)
                for j in indices:
                    rows.append(k)
                    cols.append(j)
                    vals.append(current[k - j])
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M * 1.001),
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 60})

                b = res.x
                b_sum = np.sum(b)

                    b = b * (np.sum(current) / b_sum)
                    current = np.maximum((1 - step) * current + step * b, 0.0)
        except Exception:
            pass
    return current

Optimizes step function sequences to minimize 2n*max(autoconv)/(sum)^2.
Strategy: binomial sequence initialization, gradient-based optimization with

import numpy as np

def make_binomial_sequence(n):
    """Generate a[k] = C(2k,k)/4^k for k=0,...,n-1 via recursion."""
    a = np.ones(n, dtype=np.float64)
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return a


def fast_score(a):
    """Compute 2n*max(conv)/(sum)^2 directly."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


def smooth_obj_grad(a, t):
    """Return (objective, gradient) for F(a) = LSE(t*conv)/t - 2*log(sum(a))."""
    n = len(a)

        return 1e10, np.zeros(n)

    p = exp_vals / total
    g = np.convolve(a[::-1], p, mode='full')[n - 1:2 * n - 1]
    grad = 2.0 * g - 2.0 / s
    return f, grad

    f, _ = smooth_obj_grad(a, t)
