    inits.append(_norm1(4.0 * (t - 0.5) ** 2 + 0.1))

    inits.append(_norm1(1.0 - 4.0 * (t - 0.5) ** 2 + 0.1))
    # Bimodal
    inits.append(_norm1(np.exp(-((t - 0.3) / 0.08) ** 2) + np.exp(-((t - 0.7) / 0.08) ** 2) + 0.05))
    # Random exponential
    inits.append(_norm1(rng.exponential(1.0, n) + 0.01))
    # Smooth random
    freq = rng.integers(2, 6)
    inits.append(_norm1(0.5 + 0.4 * np.sin(2 * np.pi * freq * t + rng.uniform(0, 6.28)) + rng.uniform(0.01, 0.05, n)))
    # Two-block
    half = n // 2
    blk = np.ones(n)
    blk[:half] = 1.0
    blk[half:] = 0.4
    inits.append(_norm1(blk + rng.uniform(0, 0.05, n)))
    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.3
    inits.append(_norm1(alt + 0.05))

    # Phase 1: Mirror descent from each init
    n_inits = len(inits)
    time_per_init = max(20.0, (deadline - start_time) * 0.55 / n_inits)
    print(f"[Phase1] {n_inits} inits, ~{time_per_init:.0f}s each")

    for i, init in enumerate(inits):
        if time.monotonic() > deadline * 0.55:
            break
        remaining = deadline * 0.55 - time.monotonic()
        if remaining < 15:
            break
        t_budget = min(time_per_init, remaining)
        dl = start_time + time.monotonic() - start_time + t_budget
        # Adjust: compute absolute deadline
        dl_abs = time.monotonic() + t_budget
        a_res, M_res = _mirror_descent(init, n, dl_abs, eta0=0.8, momentum=0.85)
        T_res = 2.0 * n * M_res
        if T_res < best_score:
            best_a = a_res
            best_M = M_res
            best_score = T_res
            print(f"[MD-{i}] T = {best_score:.8f}")
        else:
            print(f"[MD-{i}] T = {T_res:.8f} (no improvement)")

    # Phase 2: Extended descent from best
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic() - 60
        print(f"[Phase2] Extended descent, {remaining:.0f}s")
        dl2 = time.monotonic() + remaining
        a_res, M_res = _mirror_descent(best_a, n, dl2, eta0=0.4, momentum=0.9, tol_frac=1e-8)
        T_res = 2.0 * n * M_res
        if T_res < best_score:
            best_a = a_res
            best_M = M_res
            best_score = T_res
            print(f"[Phase2] T = {best_score:.8f}")
        else:
            print(f"[Phase2] T = {T_res:.8f} (no improvement)")

    # Phase 3: Second extended descent with different params
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 30
        print(f"[Phase3] Second extended descent, {remaining:.0f}s")
        dl3 = time.monotonic() + remaining
        a_res, M_res = _mirror_descent(best_a, n, dl3, eta0=0.2, momentum=0.95, tol_frac=1e-9)
        T_res = 2.0 * n * M_res
        if T_res < best_score:
            best_a = a_res
            best_M = M_res
            best_score = T_res
            print(f"[Phase3] T = {best_score:.8f}")

    # Phase 4: Coordinate descent polish
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 10
        print(f"[Phase4] Coordinate descent, {remaining:.0f}s")
        a_cd, M_cd, _ = _coord_descent(best_a, n, time.monotonic() + remaining, step=max(1, n // 1000))
        T_cd = 2.0 * n * M_cd
        if T_cd < best_score:
            best_a = a_cd
            best_M = M_cd
            best_score = T_cd
            print(f"[Phase4] T = {best_score:.8f}")

    # Final: one more short mirror descent + coord descent
    if time.monotonic() < deadline - 5:
        dl_final = time.monotonic() + min(10, deadline - time.monotonic() - 2)
        a_f, M_f = _mirror_descent(best_a, n, dl_final, eta0=0.1, momentum=0.95)
        T_f = 2.0 * n * M_f
        if T_f < best_score:
            best_a = a_f
            best_M = M_f
            best_score = T_f
        # Quick coord descent
        if time.monotonic() < deadline - 2:
            a_cd2, M_cd2, _ = _coord_descent(best_a, n, deadline - 2, step=max(1, n // 500))
            T_cd2 = 2.0 * n * M_cd2
            if T_cd2 < best_score:
                best_a = a_cd2
                best_M = M_cd2
                best_score = T_cd2

    # Return
    result = best_a.tolist()
    final_score = float(evaluate_sequence(result))
    print(f"[Final] T = {final_score:.8f}, elapsed = {time.monotonic()-start_time:.1f}s")

    if not np.isfinite(final_score) or final_score > 2.0:
        print("[Fallback] Returning parent")
        return [float(x) for x in parent]

    return result
