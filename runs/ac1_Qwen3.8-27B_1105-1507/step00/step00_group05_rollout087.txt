def solve_lp(f, M, solver=None):
    """Solve the LP: maximize sum(b) s.t. (f*b)[k] <= M for all k, b >= 0."""
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    
    constraints = []
    for k in range(2*n - 1):
        # (f*b)[k] = sum_i f[i]*b[k-i] for valid i
        # valid i: max(0, k-(n-1)) <= i <= min(n-1, k)
        i_lo = max(0, k - (n - 1))

        # The terms are f[i] * b[k-i]

        j_lo = k - i_hi
        j_hi = k - i_lo
        term = f[i_lo:i_hi+1] @ b[j_lo:j_hi+1]
        constraints.append(term <= M)
    
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    
    try:
        if solver:
            prob.solve(solver=solver)
        else:
            prob.solve()
        return b.value
    except:
        return None

    # Normalize b to have the same sum as f
    S = np.sum(f)
    b_norm = b * (S / sum_b)
    return b_norm

def optimize_direct(f, n, budget, rng):
    """Direct subgradient optimization."""
    S = np.sqrt(2 * n)
    f = f.copy()
    f = np.clip(f, 0, 1000)
    total = f.sum()
    if total > 0:
        f = f * S / total
    else:
        f = np.ones(n) * S / n
    
    best_f = f.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    deadline = time.monotonic() + budget
    
    lr = 0.1  # initial learning rate
    lr_decay = 0.999
    
    while time.monotonic() < deadline:
        # Compute convolution
        c = np.convolve(f, f)
        k_star = np.argmax(c)
        
        # Compute subgradient
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n, k_star + 1)
        for j in range(j_start, j_end):
            idx = k_star - j
            if 0 <= idx < n:
                sg[j] = 2 * f[idx]
        
        # Gradient step
        f_new = f - lr * sg
        
        # Project
        f_new = np.maximum(f_new, 0)
        new_sum = f_new.sum()
        if new_sum > 0:
            f_new = f_new * S / new_sum
        else:
            f_new = np.ones(n) * S / n
        
        f = f_new
        lr *= lr_decay
        
        # Evaluate periodically
        if np.random.random() < 0.1:
            score = float(evaluate_sequence(f.tolist()))
            if score < best_score:
                best_score = score
                best_f = f.copy()
    
    return best_f, best_score

    """Normalize f to have the given target sum (or sqrt(2n) if not given)."""

def subgradient_step(f, lr):
    """One subgradient step to reduce max(f*f)."""
    n = len(f)

    sg = np.zeros(n)
