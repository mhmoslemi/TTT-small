def _score(a):
    """Quick internal score computation (avoids list conversion overhead)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sa = a.sum()
    if sa < 0.01:
        return np.inf
    bc = np.convolve(a, a)
    return float(2.0 * n * bc.max() / (sa * sa))


def _local_search_1d(n, rng, deadline, best_score):
    """Local search on the simplex for a fixed n."""
    best = None
    a = rng.dirichlet(np.ones(n, dtype=np.float64) * 2.0)
    cur = _score(a)

    best sequence found.

    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(seed)

    # --- Phase 1: Analytical optimum and quick verification ---
    best = [2.0 / 3.0, 1.0 / 3.0]
    best_score = float(evaluate_sequence(best))
    print(f"Analytical n=2 [2/3, 1/3]: score = {best_score:.10f}")

    # Verify a few edge cases
    for seq in [[1.0/3, 2.0/3], [0.5, 0.5], [1.0, 0.0],
                [1.0/3]*3, [1.0/4]*4, [1.0/5]*5]:
        s = float(evaluate_sequence(seq))
        if np.isfinite(s):
            print(f"  Verify {seq[:4]}{'...' if len(seq)>4 else ''}: {s:.6f}")
            if s < best_score:
                best = seq
                best_score = s

    # --- Phase 2: Systematic exploration for small n ---
    # For n=2, sweep the 1D problem finely
    xs = np.linspace(0.0, 1.0, 10001)
    for x in xs:
        y = 1.0 - x
        if x < 0 or y < 0:
            continue
        sc = _score([x, y])
        if sc < best_score:
            best = [float(x), float(y)]
            best_score = sc
    print(f"After n=2 sweep: score = {best_score:.10f}")

    # For n=3 to n=15, do more thorough random + local search
    for n in range(3, 16):
        if time.monotonic() > deadline:
            break
        # Try uniform
        sc = _score([1.0/n] * n)
        if sc < best_score:
            best = [1.0/n] * n
            best_score = sc

        # Random restarts
        best_a = None
        for restart in range(50):
            if time.monotonic() > deadline:
                break
            a = rng.dirichlet(np.ones(n) * rng.choice([0.5, 1.0, 2.0, 5.0]))
            sc = _score(a)
            if best_a is None or sc < _score(best_a):
                best_a = a.copy()
                if sc < best_score:
                    best = a.tolist()
                    best_score = sc

        if best_a is not None and time.monotonic() < deadline:
            # Short local search from best random
            perm = best_a > 0
            sub_budget = min(10.0, max(1.0, (deadline - time.monotonic()) / max(1, 16-n)))
            a_res, sc_res = _local_search_1d(n, rng, time.monotonic() + sub_budget, best_score)
            if sc_res < best_score:
                best = a_res.tolist()
                best_score = sc_res
                print(f"  n={n} local search improved: {sc_res:.6f}")

    # --- Phase 3: Try the LP-inspired approach from prior SOTA (adapted) ---
    # For the current best, try block perturbations (like previous approach)
    if time.monotonic() < deadline:
        best_arr = np.array(best, dtype=np.float64)
        na = len(best_arr)
        for _ in range(3000):
            if time.monotonic() > deadline:
                break
            cand = best_arr.copy()
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, na // 4))))
            start = int(rng.integers(0, max(1, na - width)))
            cand[start:start+width] *= np.exp(rng.normal(0.0, 0.1))
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best = cand.tolist()
                best_score = sc
                print(f"  Block perturbation: {sc:.6f}")
            # Sometimes try a completely different length
            if _ % 500 == 0:
                n_try = int(rng.integers(2, 30))
                a_try = rng.dirichlet(np.ones(n_try) * 2.0)
                sc_try = _score(a_try)
                if sc_try < best_score:
                    best = a_try.tolist()
                    best_score = sc_try
                    best_arr = a_try
                    na = n_try
                    print(f"  New length n={n_try}: {sc_try:.6f}")

    # --- Phase 4: Final verification with evaluate_sequence ---
    final_score = float(evaluate_sequence(best))
    if not np.isfinite(final_score):
        # Fallback to analytical
        best = [2.0/3.0, 1.0/3.0]
        final_score = float(evaluate_sequence(best))

    print(f"\n=== FINAL RESULT ===")
    print(f"Score: {final_score:.10f}")
    print(f"Length: {len(best)}")
    print(f"Theoretical 16/9 = {16.0/9.0:.10f}")
    return [float(x) for x in best]
