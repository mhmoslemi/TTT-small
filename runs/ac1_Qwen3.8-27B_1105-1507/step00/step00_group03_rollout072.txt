    n = len(a)
    # Constraint matrix: row k has A[k,j] = a[k-j] if valid
    # conv(a, g)[k] = sum_j A[k,j] * g[j] where A[k,j] = a[k-j] if 0<=k-j<n else 0

        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows_list.append(k)
            cols_list.append(j)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_conv_lp(a, n, rows, cols, index_diff, max_constraints=None):
    """
    Solve: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Uses only the top max_constraints if specified.
    Returns (g, M) or (None, M).
    """
    conv_aa = np.convolve(a, a)
    M = float(conv_aa.max())
    if M <= 0:
        return None, M

    if max_constraints is not None and max_constraints < 2 * n - 1:
        # Select top constraint positions
        top_k = np.argsort(conv_aa)[::-1][:max_constraints]
        top_k_sorted = np.sort(top_k)
        # Filter rows/cols to only these constraint positions
        mask = np.isin(rows, top_k_sorted)
        r_use = rows[mask]
        c_use = cols[mask]
        idx_use = index_diff[mask]
        vals = a[idx_use]
        A_ub = sparse.csr_matrix((vals, (r_use, c_use)), shape=(max_constraints, n))
        # Need to remap rows to 0..max_constraints-1
        row_map = np.searchsorted(top_k_sorted, r_use)
        A_ub = sparse.csr_matrix((vals, (row_map, c_use)), shape=(max_constraints, n))
        b_ub = np.full(max_constraints, M)
    else:
        vals = a[index_diff]
        A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M)

    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x, M
    except Exception:
        pass
    return None, M


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    candidates = []
    # All ones
    candidates.append(np.ones(n))
    # Tapered edges
    a = np.ones(n)
    taper = n // 6
    for i in range(taper):
        f = 0.3 + 0.7 * (i / taper)
        a[i] = f
        a[n - 1 - i] = f
    candidates.append(a)
    # Cosine bump
    a = 0.5 + 0.5 * np.cos(np.pi * (np.arange(n) - (n - 1) / 2) / ((n - 1) / 2 + 1))
    candidates.append(np.clip(a, 0.01, None))
    # Two-lobe
    x = np.arange(n)
    center = (n - 1) / 2.0
    a = np.exp(-((x - center * 0.3) ** 2) / (2 * (n * 0.15) ** 2)) + \
        np.exp(-((x - center * 1.7) ** 2) / (2 * (n * 0.15) ** 2))
    candidates.append(a)
    # Periodic
    for freq in [3, 5, 7]:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * np.arange(n) / n)
        candidates.append(np.clip(a, 0.01, None))
    # Random smooth
    a = np.exp(np.cumsum(rng.normal(0, 0.05, n)) * 0.3)
    candidates.append(a)
    return candidates


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Minimize 2n*max(conv(a,a))/sum(a)^2 using LP-based peak shaving.
    Combines structured random exploration with iterative LP optimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000)
    except:
        parent = np.ones(2535)

    if parent.size == 0:
        parent = np.ones(2535)
    n = parent.size
    print(f"Sequence length: {n}")

    # Precompute LP matrix structure
    rows, cols = build_conv_matrix_indices(n)
    index_diff = rows - cols  # k - j for each non-zero
    print(f"LP matrix: {2*n-1} x {n}, {len(rows)} non-zeros")

    # Phase 0: Try structured initializations
    best = None
    best_score = np.inf
    inits = structured_init(n, rng)
    # Add parent as an initialization
    s = parent.sum()
    if s > 0.01:
        inits.append(parent / s)

    for init in inits:
        init = np.clip(init, 0, None)
        s = init.sum()
        if s < 0.01:
            continue
        init_norm = init / s
        sc = compute_score_fast(init_norm)
        if sc < best_score:
            best_score = sc
            best = init_norm.copy()
            print(f"  Init score: {sc:.6f}")

    a = best.copy()
    print(f"Best after init: {best_score:.6f}")

    # Phase 1: Random exploration (45 seconds)
    phase1_end = time.monotonic() + 45
    n_iter = 0
    while time.monotonic() < min(phase1_end, deadline):
        strategy = rng.random()
        candidate = a.copy()

        if strategy < 0.2:
            # Block multiplicative perturbation (multiple blocks)
            n_blocks = int(rng.integers(1, 5))
            for _ in range(n_blocks):
                width = int(rng.integers(10, n // 8))
                start_idx = int(rng.integers(0, n - width))
                candidate[start_idx:start_idx + width] *= np.exp(rng.normal(0, 0.4, size=width))
        elif strategy < 0.4:
            # Edge tapering (varying strength)
            taper = int(rng.integers(20, n // 4))
            strength = rng.uniform(0.2, 0.8)
            for i in range(taper):
                f = 1.0 - strength * (1 - i / taper)
                candidate[i] *= max(f, 0.01)
                candidate[n - 1 - i] *= max(f, 0.01)
        elif strategy < 0.6:
            # Sinusoidal modulation
            freq = rng.uniform(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.2, 0.5)
            candidate *= (1.0 + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase))
        elif strategy < 0.8:
            # Smooth random (low-pass filtered noise)
            noise = rng.normal(0, 0.3, n)
            # Simple low-pass: cumulative average
            kernel_size = int(rng.integers(20, 100))
            kernel = np.ones(kernel_size) / kernel_size
            noise = np.convolve(noise, kernel, mode='same')
            candidate *= np.exp(noise)
        else:
            # Anti-peak: reduce values at high-convolution-contributing positions
            conv_aa = np.convolve(candidate, candidate)
            top_pos = np.argsort(conv_aa)[::-1][:5]
            for kp in top_pos:
                # Find which indices contribute to this peak
                j_start = max(0, kp - n + 1)
                j_end = min(kp, n - 1)
                for j in range(j_start, j_end + 1):
                    i = kp - j
                    # Reduce the smaller of the two to limit damage
                    if candidate[j] < candidate[i]:
                        candidate[j] *= 0.9
                    else:
                        candidate[i] *= 0.9

        candidate = np.clip(candidate, 0, None)
        cs = candidate.sum()
        if cs < 0.01:
            continue
        candidate /= cs

        sc = compute_score_fast(candidate)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = candidate.copy()
            a = candidate.copy()
            if n_iter % 100 == 0:
                print(f"  Phase1 iter {n_iter}: score={best_score:.6f}")
        n_iter += 1

    print(f"After Phase 1: score={best_score:.6f}")

    # Phase 2: LP-based optimization
    t_blend = 0.2
    momentum = np.zeros(n)
    stagnant = 0
    max_constraints = 300  # Use top 300 constraints for speed

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break

        # Solve LP
        lp_start = time.monotonic()
        g, M = solve_conv_lp(a, n, rows, cols, index_diff, max_constraints=max_constraints)
        lp_time = time.monotonic() - lp_start

        if g is None:
            # Fallback: gradient-like step using top peaks
            conv_aa = np.convolve(a, a)
            top_k = np.argsort(conv_aa)[::-1][:3]
            direction = np.zeros(n)
            for kp in top_k:
                j_start = max(0, kp - n + 1)
                j_end = min(kp, n - 1)
                for j in range(j_start, j_end + 1):
                    i = kp - j
                    direction[i] += a[j]
                    direction[j] += a[i]
            # Normalize direction
            dnorm = np.linalg.norm(direction)
            if dnorm > 0:
                direction /= dnorm
            eta = 0.01
            new_a = a - eta * direction * n
            new_a = np.clip(new_a, 0, None)
            ns = new_a.sum()
            if ns < 0.01:
                continue
            new_a /= ns
            sc = compute_score_fast(new_a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = new_a.copy()
                a = new_a.copy()
                stagnant = 0
            else:
                stagnant += 1
            n_iter += 1
            continue

        # Normalize g to same sum as a
        g_sum = g.sum()
        if g_sum < 1e-10:
            stagnant += 1
            n_iter += 1
            continue
        g = g * (a.sum() / g_sum)

        # Blend with momentum
        direction = g - a
        momentum = 0.6 * momentum + 0.4 * direction
        new_a = a + t_blend * momentum
        new_a = np.clip(new_a, 0, None)
        ns = new_a.sum()
        if ns < 0.01:
            t_blend *= 0.5
            n_iter += 1
            continue
        new_a /= ns

        sc = compute_score_fast(new_a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = new_a.copy()
            a = new_a.copy()
            stagnant = 0
            t_blend = min(0.7, t_blend * 1.02)
        else:
            stagnant += 1
            t_blend = max(0.05, t_blend * 0.9)

            # If stuck for a while, try a perturbation
            if stagnant > 20:
                stagnant = 0
                t_blend = 0.2
                # Perturb a to escape
                pert = a * np.exp(rng.normal(0, 0.15, n))
                pert = np.clip(pert, 0, None)
                ps = pert.sum()
                if ps > 0.01:
                    pert /= ps
                    sc_p = compute_score_fast(pert)
                    if np.isfinite(sc_p) and sc_p < best_score:
                        best_score = sc_p
                        best = pert.copy()
                    a = pert.copy()

        n_iter += 1
        if n_iter % 100 == 0:
            print(f"  Phase2 iter {n_iter}: score={best_score:.6f}, "
                  f"t={t_blend:.3f}, lp={lp_time:.2f}s")

    # Final verification with actual evaluator
    final = best
    fs = final.sum()
    if fs < 0.01:
        final = final * (0.02 / fs)
    final = np.clip(final, 0, 1000)
    final_score = evaluate_sequence([float(x) for x in final])
    print(f"FINAL: score={final_score:.6f}, iterations={n_iter}, time={time.monotonic()-deadline+budget_s-5:.0f}s")

    return [float(x) for x in final]
