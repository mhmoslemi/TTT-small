combined with simulated-annealing local search and multi-restart exploration.
The core idea: at each step, solve an LP to find the best direction b that keeps
conv(f,b) bounded, then move toward b. This monotonically reduces the peak.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _precompute_sparsity(n):
    """Build row/col index arrays for the (2n-1) x n convolution constraint matrix."""
    rows = np.empty(n * (n + 1) // 2 + n * (n - 1) // 2, dtype=np.int32)
    cols = np.empty_like(rows)
    idx = 0
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            rows[idx] = k
            cols[idx] = i
            idx += 1
    return rows[:idx], cols[:idx]


def _lp_direction(f, rows, cols, timeout=60.0):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0. Returns b or None."""
    n = f.shape[0]
    M = float(np.max(np.convolve(f, f)))
    if M < 1e-30:
        return None
    # Build sparse matrix values
    vals = f[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

def _make_msequence_init(bits, offset=1):
    """Generate a pseudo-random 0/1 sequence from a max-length LFSR."""
    # Use a known primitive polynomial for m-sequence
    # x^32 + x^22 + x^21 + x^7 + 1 (degree 32, period 2^32 - 1)
    # Simpler: use x^23 + x^18 + 1 for smaller sequences
    # We'll use a simple LFSR with a known tap
    reg = 1 << 23
    taps = (23, 18)  # x^23 + x^18
    seq = []
    period = (1 << 23) - 1
    for i in range(min(bits, period)):
        seq.append(1 if (reg & 1) else 0)
        feedback = ((reg >> (taps[0] - 1)) ^ (reg >> (taps[1] - 1))) & 1
        reg = ((reg >> 1) | (feedback << (taps[0] - 1))) & ((1 << taps[0]) - 1)
    # Pad or trim
    while len(seq) < bits:
        seq.extend(seq[:bits - len(seq)])
    return np.array(seq[:bits], dtype=np.float64)
