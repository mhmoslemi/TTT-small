import numpy as np
import time

def compute_score(a):
    """Compute the score for a sequence (normalized to sum=1)."""
    conv = np.convolve(a, a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2 * len(a) * np.max(conv) / (s * s)

def subgradient_step(a, lr, rng):
    """One subgradient descent step (unconstrained, then project)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_val = np.max(conv)
    # Find all positions where conv is close to max
    threshold = max_val - 1e-10
    max_positions = np.where(conv >= threshold)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in max_positions:
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            # a[i] * a[k-i] contributes to conv[k]
            # d conv[k] / d a[i] = a[k-i] (if k-i != i)
            # or 2 * a[i] (if k-i == i, i.e., k = 2i)
            j = k - i
            if 0 <= j < n:
                if j == i:
                    grad[i] += 2 * a[i]
                else:
                    grad[i] += a[j]
    
    # Average subgradient
    grad /= len(max_positions)
    
    # Update
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    
    # Normalize
    s = np.sum(a_new)
    if s > 0:
        a_new /= s
    
    return a_new, max_val

import numpy as np
import time

def fast_convolve(a):
    """Compute a * a (linear convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    result = np.fft.irfft(fa * fa, fsize)
    return result[:size]

def simplex_project(v, total=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = total}."""
    n = len(v)
    if total <= 0:
        return np.zeros(n)
    # Scale to projection onto sum=1 simplex
    v = v / total
    # Project onto {x >= 0, sum(x) = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w * total

def subgradient_descent(a, max_iters, deadline, rng):
    """Run subgradient descent to minimize max(conv(a,a)) with sum(a)=1, a>=0."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s  # normalize to sum = 1
    
    best_a = a.copy()
    best_conv_max = np.max(fast_convolve(a))
    
    # Precompute FFT size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    
    eta = 1.0 / np.sqrt(n)  # initial step size
    gamma = 0.999  # decay
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution via FFT
        result = np.fft.irfft(fa * fa, fsize)[:size]
        conv_max = np.max(result)
        
        if conv_max < best_conv_max:
            best_conv_max = conv_max
            best_a = a.copy()
        
        # Find argmax positions
        # Use a tolerance to find "near-max" positions
        threshold = conv_max * 0.9999
        max_pos = np.where(result >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in max_pos:

            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            # a[k-i] for i in [lo, hi) is a[k-hi+1 : k-lo+1] reversed
