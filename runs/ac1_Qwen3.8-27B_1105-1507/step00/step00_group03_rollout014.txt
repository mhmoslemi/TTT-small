def compute_conv_fft(x, L):
    """Compute linear convolution of x with itself using FFT."""
    X = np.fft.rfft(x, n=L)
    C = np.fft.irfft(X * X, n=L)
    return C

def compute_objective(x, n):
    """Compute the objective value."""
    c = np.convolve(x, x)
    return 2 * n * np.max(c)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    # Phase 2: Gradient-based refinement
    # Phase 3: Local search with random perturbations
    
    # Phase 1: Try different sinusoidal modulations
    for trial in range(20):
        if time.time() > deadline:
            break

            x = np.ones(n) / n
        else:
            # Random sinusoidal modulation
            f = rng.integers(1, n // 2 + 1)
            alpha = rng.uniform(0.1, 0.9)
            phi = rng.uniform(0, 2 * np.pi)
            x = 1 + alpha * np.cos(2 * np.pi * f * np.arange(n) / n + phi)
            x = np.maximum(x, 0)
            x = x / np.sum(x)

        for it in range(50):
            if time.time() > deadline:
                break
            
            c = np.convolve(x, x)
            max_c = np.max(c)
            obj = 2 * n * max_c
            
            if obj < best_obj:
                best_obj = obj
                best_x = x.copy()
                print(f"Trial {trial}, Iter {it}: obj = {obj:.6f}")
            
            # Flattening: reduce peaks
            c_mean = np.sum(c) / len(c)
            # Soft threshold: reduce values above mean, increase below
            flat_factor = 0.3  # how aggressively to flatten
            c_mod = c.copy()
            above = c_mod > c_mean
            c_mod[above] = c_mean + (c_mod[above] - c_mean) * (1 - flat_factor)
            # Don't change below-mean values (to maintain non-negativity of x)
            
            # Now we want to find x' such that conv(x', x') ≈ c_mod
            # Use FFT: X'^2 = FFT(c_mod), so X' = sqrt(FFT(c_mod))
            C_mod = np.fft.rfft(c_mod, n=L)
            # Handle negative real parts (shouldn't happen for non-negative c)
            C_mod = np.maximum(C_mod.real, 0) + 1j * C_mod.imag
            X_new = np.sqrt(C_mod)
            x_new = np.fft.irfft(X_new, n=L)[:n]
            x_new = np.maximum(x_new, 0)
            s_new = np.sum(x_new)
            if s_new < 1e-10:
                break
            x = x_new / s_new

    tau = 0.1  # softmax temperature (in units of max_c)
    
    for it in range(20000):

            break
        
        c = np.convolve(x, x, mode='full')  # length 2n-1
        max_c = np.max(c)

            best_obj = obj
            best_x = x.copy()
            if it % 100 == 0:

        c_scaled -= np.max(c_scaled)
        w = np.exp(c_scaled)
        w /= np.sum(w)

        # Use FFT for efficiency
        # G[i] = sum_j x[j] * w[i+j]
        # This is np.correlate(w, x, 'full')[:2*n-1]
        # But for large n, let's use FFT
        L2 = 1
        while L2 < 3 * n:
            L2 *= 2

        # FFT would be O(n log n) ≈ 30K ops. Let's use FFT.
        # G[i] = sum_j x[j] * w[i+j]
        # = cross-correlation
        
        # Using FFT: G = ifft(conj(fft(x, L2)) * fft(w, L2)) ... let me verify
        # corr[m] = ifft(conj(fft(a,L2)) * fft(b,L2))[m] = sum_n a[n+m] * b[n] (circular, with padding)
        # For linear: pad to L2 >= len(a) + len(b) - 1
        # So: G = ifft(conj(fft(w, L2)) * fft(x, L2))[:2*n-1]
        # Verify: G[m] = sum_n w[n+m] * x[n] for valid n. ✓
        
        Xw = np.fft.rfft(w, n=L2)
        Xx = np.fft.rfft(x, n=L2)
        G_full = np.fft.irfft(np.conj(Xw) * Xx, n=L2)
        # Hmm wait, I need to be careful with rfft/irfft.
        # Actually, let me just use full fft/iff
        Xw = np.fft.fft(w, n=L2)
        Xx = np.fft.fft(x, n=L2)
        G_full = np.fft.ifft(np.conj(Xw) * Xx, n=L2).real
        G = G_full[:2*n-1]
        
        # Gradient of soft_max wrt x[i]:
        # d(soft_max)/dx[i] = sum_k w[k] * x[k-i] (for valid k-i)
        # = sum_j w[i+j] * x[j] (j = k-i)
        # = G[i] (as computed above)
        # So grad[i] = G[i]
        
        # Projected gradient descent on simplex
        x_new = x - eta * G / max_c  # normalize by max_c for stable step
        x_new = project_simplex(x_new)
        x = x_new
        
        # Adaptive step size
        c_new = np.convolve(x, x, mode='full')
        if np.max(c_new) > max_c * 1.5:
            eta *= 0.7
        elif np.max(c_new) < max_c * 0.8:
            eta = min(eta * 1.1, 5.0)
    
    # Phase 3: Local search with random perturbations
    x = best_x.copy()
    obj = best_obj
    
    for it in range(50000):
        if time.time() > deadline:
            break
        
        # Random perturbation
        perturbation = rng.normal(0, 0.01, size=n)
        x_pert = x + perturbation
        x_pert = np.maximum(x_pert, 0)
        s_pert = np.sum(x_pert)
        if s_pert < 1e-10:
            continue
        x_pert = x_pert / s_pert
        
        c_pert = np.convolve(x_pert, x_pert, mode='full')
        obj_pert = 2 * n * np.max(c_pert)
        
        if obj_pert < obj:
            x = x_pert
            obj = obj_pert
            if obj < best_obj:
                best_obj = obj
                best_x = x.copy()
                print(f"Local search iter {it}: obj = {obj:.6f}")
    
    # Convert back to original scale (the evaluator handles scaling, so any scale works)
    # But we need sum >= 0.01. Let's set sum to 1.
    result = best_x  # already normalized to sum 1
    
    # Verify
    final_score = evaluate_sequence(result.tolist())
    print(f"Final score: {final_score:.6f}")
    
    return [float(v) for v in result]

Wait, I realize there's an issue. The evaluate_sequence function clips values to [0, 1000] and checks sum >= 0.01. Since our x has sum = 1 (after normalization), it's fine. And the values are all in [0, 1/n] ≈ [0, 0.0004], well within [0, 1000].

But actually, the objective is scale-invariant! `2n * max(conv(a,a)) / (sum a)^2` doesn't depend on the scale of `a`. So we can work with any normalization.

Let me also think about whether the gradient approach will actually work. The issue is that the "max" makes the objective non-smooth, and the softmax approximation might not be tight enough.

Let me also consider: what if I use a different parameterization? Instead of working directly with `x`, what if I work with `log(x)` or `exp(y)`?

`x[i] = exp(y[i]) / sum(exp(y))` (softmax parameterization). This automatically ensures `x >= 0` and `sum x = 1`.

The gradient would be different but might be more stable.

Actually, let me try a completely different approach: **simulated annealing** with a smart move set.

Moves:
1. Add a small amount to one position and subtract from another (maintains sum).
2. Shift a block of values.
3. Apply a sinusoidal modulation.
4. Mirror/symmetrize.
