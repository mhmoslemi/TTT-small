def my_convolve(a):
    """Full linear convolution using FFT for large arrays."""
    n = len(a)
    if n < 64:
        return np.convolve(a, a, mode='full')
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)[:size]
    result[result < 0] = 0  # Clean up tiny negative values from FFT
    return result

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v_sum = v.sum()
    if v_sum < 1e-15:
        return np.ones(n) / n
    if v.min() >= 0 and abs(v_sum - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[0]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    g_i = (2/|A|) * sum_{k in A} a_{k-i} for valid (k-i) in [0, n-1]
    Computed via FFT cross-correlation.
    """
    L = len(active_indices)
    if L == 0:
        return np.zeros(n)
    
    # Build indicator vector h of length 2n-1
    h = np.zeros(2 * n - 1)
    h[active_indices] = 1.0
    
    # Cross-correlation: corr[i] = sum_k h[k] * a[k-i]
    # Using FFT: corr = IFFT(FFT(h) * conj(FFT(a)))

    H = np.fft.rfft(h, fft_size)
    A = np.fft.rfft(a, fft_size)
    
    corr = np.fft.irfft(H * np.conj(A), fft_size)
    g = corr[:n] * (2.0 / L)
    
    return g
