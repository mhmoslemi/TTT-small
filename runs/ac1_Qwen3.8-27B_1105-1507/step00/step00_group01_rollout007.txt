2. Compute the gradient (which is a shifted/reversed s)

   d. Add small random noise for exploration

```python
import numpy as np
import time

def compute_peak_grad(s, k, n):
    """Compute gradient of conv[k] w.r.t. s."""
    grad = np.zeros(n)
    j_start = max(0, k - (n-1))
    j_end = min(n-1, k)
    if j_start <= j_end:
        m_vals = k - np.arange(j_start, j_end+1)
        grad[j_start:j_end+1] = 2.0 * s[m_vals]
    return grad

def propose_candidate(seed=42, budget_s=1500, **kwargs):

    print(f"Initial score: {best_score:.6f}, n={n}")
    
    # Normalize working copy
    s = s0 / max(s0.sum(), 1e-10)
    
    # Optimization loop
    lr = 0.1
    top_k = 5  # consider top 5 peaks
    
    iteration = 0
    while time.monotonic() < deadline:
        computation_start = time.monotonic()
        
        # Compute convolution
        conv = np.convolve(s, s)
        
        # Find top peaks
        # Use a weighted combination of top peaks for smoother gradient
        top_indices = np.argsort(conv)[-top_k:]
        top_values = conv[top_indices]
        weights = top_values / top_values.sum()
        
        # Compute combined gradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            grad += w * compute_peak_grad(s, int(k), n)
        
        # Center the gradient (to preserve sum)
        grad -= grad.mean()
        
        # Update
        s -= lr * grad
        
        # Clip and renormalize
        s = np.clip(s, 0, None)
        s_sum = s.sum()
        if s_sum < 1e-10:
            s = s0 / max(s0.sum(), 1e-10)
        else:
            s = s / s_sum
        
        # Occasionally add noise
        if iteration % 50 == 0:
            noise_scale = 0.01
            s += noise_scale * rng.normal(0, 1, n)
            s = np.clip(s, 0, None)
            s = s / s.sum()
        
        # Occasionally evaluate exactly
        if iteration % 100 == 0 or time.monotonic() - computation_start > 0.5:
            score = float(evaluate_sequence(s.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_s = s * best_s.sum()  # restore original scaling
                print(f"Iter {iteration}: score = {best_score:.6f}")
        
        iteration += 1
        
        # Adaptive learning rate
        if iteration % 200 == 0:
            lr *= 0.95
    
    return [float(x) for x in best_s]
```
