"""
Step function optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses parametric step sequences + LP-based local search + gradient descent.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

def fft_convolve(a, b):
    """Full linear convolution using FFT."""
    size = len(a) + len(b) - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:size]

def compute_score(a):
    """Compute the objective score (scale-invariant)."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    b = fft_convolve(a, a)
    return 2.0 * n * np.max(b) / (s * s)

def make_step_sequence(n, k, y):
    """Create a step sequence: first k entries are x, rest are y.
    x is determined by the constraint kx + (n-k)y = 1."""
    x = (1.0 - (n - k) * y) / k
    if x < 0 or y < 0:
        return None
    a = np.empty(n)
    a[:k] = x
    a[k:] = y
    return a

def step_sequence_score(n, k, y):
    """Compute score of step sequence efficiently."""
    x = (1.0 - (n - k) * y) / k
    if x < 0 or y < 0:
        return np.inf
    a = np.empty(n)
    a[:k] = x
    a[k:] = y
    b = fft_convolve(a, a)
    return 2.0 * n * np.max(b)  # sum(a) = 1

def lp_local_search(f, budget_s=60):
    """LP-based local search. Returns improved sequence."""
    n = len(f)
    f = f.copy()
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / s  # normalize
    best_f = f.copy()
    best_score = compute_score(f)
    deadline = time.monotonic() + budget_s
    
    while time.monotonic() < deadline:
        b = fft_convolve(f, f)
        M = np.max(b)
        # Build constraint matrix
        A = lil_matrix((2*n-1, n))
        for k in range(2*n-1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                A[k, j_start:j_end+1] = f[k - j_end : k - j_start + 1][::-1]
        A = A.tocsr()
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        t_start = time.monotonic()
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        t_end = time.monotonic()
        if not res.success:
            break
        g = res.x
        g_sum = g.sum()
        if g_sum <= 1e-15:
            break
        g = g / g_sum
        # Line search
        best_t = 0.0
        best_s = best_score
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            f_trial = (1-t)*f + t*g
            trial_score = compute_score(f_trial)
            if trial_score < best_s:
                best_s = trial_score
                best_t = t
        if best_t == 0:
            break
        f = (1-best_t)*f + best_t*g
        if best_s < best_score:
            best_score = best_s
            best_f = f.copy()
        print(f"  LP iter: score={best_score:.6f}, t={best_t}, LP_time={t_end-t_start:.2f}s")
    
    return best_f

def gradient_descent(f, budget_s=60):
    """Gradient descent on max(b_k) with sum(a)=1 constraint."""
    n = len(f)
    f = f.copy()
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / s
    best_f = f.copy()
    best_score = compute_score(f)
    deadline = time.monotonic() + budget_s
    lr = 0.05
    
    while time.monotonic() < deadline:
        b = fft_convolve(f, f)
        k_star = int(np.argmax(b))
        # Gradient
        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        grad[j_start:j_end+1] = 2.0 * f[k_star - j_end : k_star - j_start + 1][::-1]
        f_new = f - lr * grad
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new
        score_new = compute_score(f_new)
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
            f = f_new
        else:
            lr *= 0.95
            if lr < 1e-6:
                break
    return best_f
