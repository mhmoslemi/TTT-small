    """Compute linear convolution a * a using FFT."""

    A = sp_fft.fft(a, fft_size)
    C = sp_fft.ifft(A * A).real

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _convolve_pp(a):
    """Compute a * a (linear convolution) using FFT. Assumes a is 1D numpy array."""
    n = len(a)
    size = 2 * n - 1
    N = _next_pow2(size)
    A = sp_fft.rfft(a, N)  # real FFT for real input
    C = sp_fft.irfft(A * A, N)
    return C[:size]

def _smooth_grad(a, beta):
    """Compute smooth max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    c = _convolve_pp(a)
    c_max = c.max()
    w = np.exp(beta * (c - c_max))
    W = w.sum()
    wn = w / W
    
    # Gradient: g[i] = sum_j a[j] * wn[i+j]
    # Using FFT: g[i] = (a * wn_rev)[2n-2-i] where wn_rev = wn[::-1]
    N = _next_pow2(3 * n - 1)
    A = sp_fft.fft(a, N)
    Wf = sp_fft.fft(wn[::-1], N)
    conv = sp_fft.ifft(A * Wf).real

Let me think about whether I can adapt such an algorithm.

The PMAX algorithm:
1. Start with a sequence a (on the unit sphere/complex circle).
2. Compute the autocorrelation.
3. Threshold: reduce the peaks in the autocorrelation.
4. Update the sequence via inverse FFT.
5. Repeat.

This is in the frequency domain and works on the unit circle constraint. Our constraint is different (simplex instead of sphere), so we can't directly apply it. But the idea of "threshold the peaks and update" is similar to what we're doing.

I think my mirror descent approach is a reasonable adaptation. Let me finalize the code.

Let me also add a **subgradient step** as an alternative to mirror descent:
- g = subgradient of max(c) = a[k*-::-1] (reversed/shifted)
- a_new = a - eta * g
- Project onto simplex

For the subgradient, k* = argmax(c), and g[i] = a[k*-i] (for valid k*-i). This is just a shifted/reversed version of a.

The projected subgradient might be simpler and faster (no need for the full smooth gradient computation). Let me include both options.

Actually, the subgradient is a special case of the smooth gradient (when beta -> infinity, the smooth gradient concentrates on k*). So the mirror descent with large beta is essentially doing the same thing but with a smoother update.

Let me just use the mirror descent with a beta schedule and call it a day.

from scipy import fft as sp_fft
import time

def _next_pow2(x):
    """Return the smallest power of 2 >= x."""
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def _conv_pp(a):
    """Linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = _next_pow2(size)
    A = sp_fft.rfft(a, N)
    C = sp_fft.irfft(A * A, N)
    return C[:size]

def _smooth_max_grad(a, beta):
    """Smooth max of conv(a,a) and its gradient using FFT."""
    n = len(a)
    c = _conv_pp(a)
    c_max = float(c.max())
    
    # Numerically stable softmax-like weights
    w = np.exp(beta * (c - c_max))
    W = float(w.sum())
    wn = w / W  # probability vector over convolution indices
    
    # Gradient: g[i] = sum_j a[j] * wn[i+j]
    # = (a convolved with wn_reversed)[2n-2-i]
    N = _next_pow2(3 * n - 1)
    A = sp_fft.rfft(a, N)
    Wr = sp_fft.rfft(wn[::-1].copy(), N)
    conv = sp_fft.irfft(A * Wr, N)
    g = conv[n-1:2*n-1][::-1].copy()
    
    return c_max + np.log(W) / beta, g, c_max
