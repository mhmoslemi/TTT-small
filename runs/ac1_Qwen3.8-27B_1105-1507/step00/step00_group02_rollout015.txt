"""
Gradient-based simulated annealing for minimizing 2n*max(auto_conv)/(sum(a))^2.
Uses FFT for fast convolution, gradient descent on hot lags, and diverse move types.
Tries multiple initializations and keeps the best result.
"""

import numpy as np
import time


def _fft_size_for(n):
    """Find next power of 2 >= 2n-1."""
    size = 1
    target = 2 * n - 1
    while size < target:
        size *= 2
    return size


def _fast_convolve(a, fft_size):
    """Compute full auto-convolution using FFT."""
    b = np.fft.irfft(np.fft.rfft(a, size=fft_size) ** 2, size=fft_size)
    return b[:2 * len(a) - 1]


def _fast_score(a, fft_size):
    """Fast score computation using FFT convolution."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    b = _fast_convolve(a, fft_size)
    mb = np.max(b)
    return float(2.0 * n * mb / (s * s))


def _compute_hot_lag_gradient(current, b, fft_size, max_lags=30):
    """
    Compute gradient of the auto-convolution peak values w.r.t. current.
    For lag k: grad[m] += 2 * current[k-m] if 0<=k-m<n
    """
    n = len(current)
    sum_a = np.sum(current)
    mean_b = (sum_a * sum_a) / (2 * n - 1)
    threshold = mean_b * 1.01
    hot = b > threshold
    hot_lags = np.where(hot)[0]
    
    if len(hot_lags) == 0:
        return np.zeros(n)
    
    # Use top max_lags by value
    if len(hot_lags) > max_lags:
        vals = b[hot_lags]
        top = np.argpartition(vals, -max_lags)[-max_lags:]
        hot_lags = hot_lags[top]
    
    # Precompute reversed
    a_rev = current[::-1]
    grad = np.zeros(n)
    
    for k in hot_lags:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # current[k-m] for m in [lo, hi] = a_rev[n-1-(k-m)] = a_rev[n-1-k+m]
            # m=lo -> index n-1-k+lo; m=hi -> index n-1-k+hi
            start_idx = n - 1 - k + lo
            end_idx = n - 1 - k + hi + 1
            grad[lo:hi + 1] += 2.0 * a_rev[start_idx:end_idx]
    
    return grad

        try:
            exact = float(evaluate_sequence(local_best.tolist()))
            if np.isfinite(exact):
                local_score = exact
        except:
            pass
        
        best, best_score = _run_sa(seq, n, _fft_size_for(n), local_best, local_score, rng, duration, label)
        
        if best_score < global_best_score:
            global_best = best.copy()
            global_best_score = best_score
            print(f"  NEW GLOBAL BEST: {global_best_score:.6f} (n={n})")
        else:
            print(f"  Local best {best_score:.6f} not better than global {global_best_score:.6f}")
    
    # Final polish: run SA again on the global best for remaining time
    remaining_time = total_deadline - time.monotonic() - 10.0
    if remaining_time > 30:
        n_best = len(global_best)
        fft_best = _fft_size_for(n_best)
        print(f"\n--- Final polish on best (n={n_best}, {remaining_time:.0f}s) ---")
        rng = np.random.default_rng(seed + 999)
        cur = global_best + 0.01 * (np.sum(global_best) / n_best) * rng.standard_normal(n_best)
        cur = np.clip(cur, 0.0, 1000.0)
        cur_score = _fast_score(cur, fft_best)
        global_best, global_best_score = _run_sa(cur, n_best, fft_best, global_best, global_best_score, 
                                                   rng, remaining_time, "final")
    
    # Final verification
    final_score = float(evaluate_sequence(global_best.tolist()))
    print(f"\n=== FINAL SCORE: {final_score:.8f} (n={len(global_best)}) ===")
    
    if not np.isfinite(final_score):
        # Fallback to original
        print("Score is inf, falling back to original sequence.")
        return [float(x) for x in base]
    
    return [float(x) for x in global_best]
