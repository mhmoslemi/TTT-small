import numpy as np

def subgradient_step(a, n, S, f_star):
    """One subgradient step for minimizing max(conv(a,a)) subject to a>=0, sum(a)=S."""
    conv = np.convolve(a, a)
    max_b = conv.max()
    
    # Active set
    threshold = max_b - 1e-10  # small tolerance
    active = conv >= threshold
    active_set = np.where(active)[0]
    
    if len(active_set) == 0:
        return a, max_b
    
    # Indicator of active set (length 2n-1)
    indicator = np.zeros(2*n - 1)
    indicator[active_set] = 1.0
    
    # Subgradient: grad[k] = (2/|A|) * sum_{m in A} a[m-k]
    # = (2/|A|) * (indicator convolve a)[k] for k = 0, ..., n-1
    grad_full = np.convolve(indicator, a)
    grad = (2.0 / len(active_set)) * grad_full[:n]
    
    # Polyak step size
    f_current = max_b  # since sum(a) = S, the objective is max_b
    f_gap = f_current - f_star
    grad_norm_sq = np.dot(grad, grad)
    
    if grad_norm_sq < 1e-30:
        return a, max_b
    
    eta = f_gap / grad_norm_sq
    
    # Take step
    a_new = a - eta * grad
    
    # Project onto simplex {x >= 0, sum(x) = S}
    a_new = np.maximum(a_new, 0)
    a_new_sum = a_new.sum()
    if a_new_sum < 1e-15:
        a_new = np.ones(n) * S / n
    else:
        a_new = a_new * S / a_new_sum
    
    return a_new, max_b

"""
Subgradient method with Polyak step size for minimizing the convolution-based objective.
Uses FFT-based convolution for fast gradient computation, with LP polishing as a finishing step.
"""

import numpy as np
import time


def normalize_sequence(a, S):
    """Normalize a to be non-negative with sum S."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * S / len(a)
    return a * S / s


def subgradient_iteration(a, n, S, f_star, eta_scale=1.0):
    """
    Perform one subgradient step to minimize max(conv(a,a)) 
    subject to a >= 0, sum(a) = S.
    
    Returns new_a, max_b, active_count
    """
    conv = np.convolve(a, a)
    max_b = conv.max()
    
    # Find active set (with tolerance)
    tol = 1e-8 * max_b + 1e-12
    active = conv >= max_b - tol
    active_count = active.sum()
    
    if active_count == 0:
        return a, max_b, 0
    
    # Compute subgradient
    # grad[k] = (2/|A|) * sum_{m in A} a[m-k] (for valid m-k)
    # = (2/|A|) * (indicator_A * a)[k]
    indicator = np.zeros(2 * n - 1)
    indicator[active] = 1.0
    indicator /= active_count
    
    grad_full = np.convolve(indicator, a)
    grad = 2.0 * grad_full[:n]
    
    # Polyak step size
    f_gap = max_b - f_star
    grad_norm_sq = np.dot(grad, grad)
    
    if grad_norm_sq < 1e-30:
        return a, max_b, active_count
    
    eta = eta_scale * f_gap / grad_norm_sq
    
    # Take step and project
    a_new = a - eta * grad
    a_new = normalize_sequence(a_new, S)
    
    return a_new, max_b, active_count

"""
Subgradient + LP polishing method to minimize 2*n*max(conv(seq,seq))/sum(seq)^2.
Uses FFT-based subgradients for fast iterations, then switches to LP-based local optimization.
"""

import numpy as np
import time


def normalize_sequence(a, S):
    """Normalize a to be non-negative with sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * S / len(a)
    return a * S / s


def compute_obj_fast(a):
    """Compute max of self-convolution (objective when sum(a)=sqrt(2n))."""
    return float(np.convolve(a, a).max())


def subgradient_iteration(a, n, S, f_star, eta_multiplier=1.0, top_k=None):
    """
    One projected subgradient step to minimize max(conv(a,a))
    subject to a >= 0, sum(a) = S.
    Returns (new_a, current_max_b, active_count).
    """
    conv = np.convolve(a, a)
    max_b = float(conv.max())

    # Determine active set with tolerance
    tol = max(1e-8 * max_b, 1e-10)
    if top_k is not None and top_k < len(conv):
        # Use only top K positions
        top_indices = np.argpartition(conv, -top_k)[-top_k:]
        active_mask = np.zeros(len(conv), dtype=bool)
        active_mask[top_indices] = True
        active_count = top_k
    else:
        active_mask = conv >= (max_b - tol)
        active_count = int(active_mask.sum())

    if active_count == 0:
        return a, max_b, 0

    # Compute subgradient:
    # g[k] = (2/|A|) * sum_{m in A} a[m-k] for valid (m-k)
    # Using: g[k] = 2 * (indicator * a_rev)[n-1+k]
    a_rev = a[::-1]
    indicator = np.zeros(2 * n - 1)
    indicator[active_mask] = 1.0 / active_count
    conv_grad = np.convolve(indicator, a_rev)
    grad = 2.0 * conv_grad[n - 1: 2 * n - 1]

    # Polyak step size
    f_gap = max_b - f_star
    if f_gap < 1e-15:
        return a, max_b, active_count
    grad_norm_sq = float(np.dot(grad, grad))
    if grad_norm_sq < 1e-30:
        return a, max_b, active_count

    eta = eta_multiplier * f_gap / grad_norm_sq
    # Cap step size to avoid overshooting
    eta = min(eta, S / (np.max(np.abs(grad)) + 1e-15) * 0.9)

    a_new = a - eta * grad
    a_new = normalize_sequence(a_new, S)

    return a_new, max_b, active_count


def build_lp_structure(n):
    """Precompute (rows, cols) for the LP constraint matrix."""
    total_rows = 2 * n - 1
    rows_chunks = []
    cols_chunks = []

    for m in range(total_rows):
        k_start = max(0, m - n + 1)
        k_end = min(n - 1, m)
        if k_start <= k_end:
            rows_chunks.append(np.full(k_end - k_start + 1, m, dtype=np.int32))
            cols_chunks.append(np.arange(k_start, k_end + 1, dtype=np.int32))
        # t column
        rows_chunks.append(np.array([m], dtype=np.int32))
        cols_chunks.append(np.array([n], dtype=np.int32))

    rows = np.concatenate(rows_chunks)
    cols = np.concatenate(cols_chunks)

    # Precompute which entries are for 'a' values vs 't'
    a_mask = cols < n
    t_mask = ~a_mask

    # For a entries: m_idx - k_idx gives the index into a
    m_idx = rows[a_mask]
    k_idx = cols[a_mask]
    a_indices = m_idx - k_idx

    return rows, cols, a_mask, t_mask, a_indices


def lp_polish(a, n, S, deadline, verbose=True):
    """LP-based polishing: repeatedly solve min max(a*g) s.t. g>=0, sum(g)=S."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    current_a = a.copy()

    if verbose:
        print(f"LP polish starting: obj={best_obj:.6f}")

    # Build matrix structure (one-time cost)
    build_start = time.monotonic()
    rows, cols, a_mask, t_mask, a_indices = build_lp_structure(n)
    nnz = len(rows)

    b_ub = np.zeros(2 * n - 1)
    # A_eq: sum(g) = S
    eq_data = np.ones(n, dtype=np.float64)
    A_eq = csr_matrix((eq_data, (np.zeros(n, dtype=np.int32), np.arange(n, dtype=np.int32))),
                      shape=(1, n + 1))
    b_eq = np.array([S])
    bounds = [(0.0, None)] * (n + 1)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    if verbose:
        print(f"  LP matrix built in {time.monotonic()-build_start:.1f}s, nnz={nnz}")

    lp_iter = 0
    for lp_iter in range(200):
        if time.monotonic() > deadline:
            break

        # Build values
        vals = np.empty(nnz)
        vals[a_mask] = current_a[a_indices]
        vals[t_mask] = -1.0

        A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))

        # Solve LP
        solve_start = time.monotonic()
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': max(1.0, deadline - time.monotonic() - 2.0)})
        solve_time = time.monotonic() - solve_start

        if not result.success:
            if verbose:
                print(f"  LP iter {lp_iter}: failed ({result.message}) [{solve_time:.1f}s]")
            break

        g = result.x[:n].copy()
        g = normalize_sequence(g, S)
        new_obj = compute_obj_fast(g)

        if new_obj < best_obj - 1e-10:
            # Direct improvement
            best_obj = new_obj
            best_a = g.copy()
            current_a = g.copy()
            if verbose:
                print(f"  LP iter {lp_iter}: obj={new_obj:.6f} [{solve_time:.1f}s]")
        else:
            # Try blending
            improved = False
            for alpha in [0.5, 0.3, 0.7, 0.2]:
                g_blend = normalize_sequence((1 - alpha) * current_a + alpha * g, S)
                obj_blend = compute_obj_fast(g_blend)
                if obj_blend < best_obj - 1e-10:
                    best_obj = obj_blend
                    best_a = g_blend.copy()
                    current_a = g_blend.copy()
                    if verbose:
                        print(f"  LP iter {lp_iter} (alpha={alpha}): obj={obj_blend:.6f}")
                    improved = True
                    break
            if not improved:
                if verbose:
                    print(f"  LP iter {lp_iter}: no improvement (obj={new_obj:.6f}, best={best_obj:.6f})")
                break

    if verbose:
        print(f"  LP polish done after {lp_iter} iterations: obj={best_obj:.6f}")
    return best_a, best_obj

            # But reverting is expensive; instead, reduce step
            if no_improve > 50:
                eta_multiplier = max(eta_multiplier * 0.8, 0.1)
                no_improve = 0
            elif no_improve > 500:
                # Perturb to escape
                a = normalize_sequence(a + rng.normal(0, S / n / 10, size=n), S)
                eta_multiplier = 1.0
                no_improve = 0
            elif no_improve > 2000:
                # Bigger perturbation
                a = normalize_sequence(0.8 * a + 0.2 * rng.uniform(0, 2, size=n), S)
                eta_multiplier = 1.0
                no_improve = 0

        if verbose and iteration % 5000 == 0:
            print(f"  Subgradient iter {iteration}: best={best_obj:.6f}, "
                  f"current={new_obj:.6f}, eta={eta_multiplier:.2f}, active={active_count}")

    if verbose:
        print(f"  Subgradient done: {iteration} iters, best_obj={best_obj:.6f}")
    return best_a, best_obj


def run_optimization(a_init, n, seed=42, time_budget=100.0, verbose=True):
    """Run the full optimization pipeline for a given initialization."""
    S = np.sqrt(2.0 * n)
    f_star = S ** 2 / (2 * n - 1)
    a = normalize_sequence(a_init.copy(), S)

    if verbose:
        print(f"\n=== Optimization run (n={n}, seed={seed}, budget={time_budget:.0f}s) ===")

    # Phase 1: Subgradient (60% of time)
    sg_time = 0.6 * time_budget
    deadline_sg = time.monotonic() + sg_time - 2.0
    a, obj = run_subgradient_phase(a, n, S, f_star, sg_time, seed=seed, verbose=verbose)

    # Phase 2: LP polishing (remaining time minus buffer)
    remaining = time_budget - (time.monotonic() - (deadline_sg - sg_time)) - 10.0
    if remaining > 30:
        if verbose:
            print(f"  Starting LP polish with {remaining:.0f}s remaining")
        deadline_lp = time.monotonic() + remaining
        a, obj = lp_polish(a, n, S, deadline_lp, verbose=verbose)

    if verbose:
        print(f"  Final obj: {obj:.6f}")
    return a, obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function: minimize 2*n*max(conv(seq,seq))/sum(seq)^2.
    Uses subgradient method followed by LP-based local optimization.
    Tries multiple initializations to avoid local optima.
    """
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    # Get initial sequence
    a_given = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n_given = len(a_given)

    if n_given < 10 or not np.isfinite(a_given).all():
        n_given = 256
        a_given = np.ones(n_given)

    best_a = None
    best_obj = np.inf

    # Strategy: try the given sequence first (it's likely a good starting point)
    # Then try a few other sizes/initializations if time permits

    configs = [
        # (sequence, n, seed, time_fraction)
        (a_given, n_given, seed, 0.45),
        (a_given, n_given, seed + 100, 0.20),
    ]

    # Try a different size if we have a lot of time
    if budget_s > 600:
        n_alt = min(3000, max(1000, n_given // 3))
        rng = np.random.default_rng(seed + 200)
        a_alt = np.ones(n_alt) + 0.3 * np.sin(np.linspace(0, 6 * np.pi, n_alt))
        a_alt = np.maximum(a_alt, 0.1)
        configs.append((a_alt, n_alt, seed + 200, 0.15))

    # Try a smooth structured initialization at the given size
    n_smooth = n_given
    rng_s = np.random.default_rng(seed + 300)
    freqs = rng_s.uniform(0.5, 8.0, size=8)
    phases = rng_s.uniform(0, 2 * np.pi, size=8)
    x = np.linspace(0, 1, n_smooth)
    a_smooth = np.ones(n_smooth)
    for f, p in zip(freqs, phases):
        a_smooth += 0.3 * np.sin(2 * np.pi * f * x + p)
    a_smooth = np.maximum(a_smooth, 0.05)
    configs.append((a_smooth, n_smooth, seed + 300, 0.20))

    total_time = sum(c[3] for c in configs)
    # Scale to fit budget
    scale = (float(budget_s) - 30.0) / total_time

    for a_init, n, s, t_frac in configs:
        if time.monotonic() > total_deadline:
            break

        t_budget = max(20.0, t_frac * scale)
        # Don't exceed remaining time
        remaining = total_deadline - time.monotonic()
        t_budget = min(t_budget, remaining - 10.0)
        if t_budget < 20.0:
            break

        try:
            a_result, obj = run_optimization(a_init, n, seed=s, time_budget=t_budget, verbose=True)
            if obj < best_obj:
                best_obj = obj
                best_a = a_result.copy()
                print(f">> New best: {obj:.6f} (n={n})")
        except Exception as e:
            print(f"  Error in config (n={n}): {e}")
            continue

    # Final verification
    if best_a is not None:
        final_obj = float(evaluate_sequence([float(x) for x in best_a]))
        if np.isfinite(final_obj):
            print(f"Final verified objective: {final_obj:.6f}")
            if final_obj != best_obj:
                # The evaluate_sequence might give a slightly different value
                # due to clipping; use the actual value
                best_obj = final_obj

    if best_a is None:
        # Fallback: return the given sequence
        best_a = a_given
        best_obj = float(evaluate_sequence(best_a.tolist()))

    return [float(x) for x in best_a]
