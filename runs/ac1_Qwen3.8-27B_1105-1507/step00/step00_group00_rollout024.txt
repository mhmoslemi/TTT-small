    n = len(f)
    grad = np.zeros(n)
    for i in range(n):
        # sum_j f[j] * p[j+i] where 0 <= j < n and 0 <= j+i < 2n-1
        hi = min(n, 2 * n - 1 - i)
        if hi > 0:
            grad[i] = 2.0 * np.dot(f[:hi], p[i:i + hi])
    return grad


def run_smoothed_adam(f, n, time_budget, beta_start=5.0, beta_end=100.0,
                      lr=0.01, verbose=True):
    """Run Adam optimization with log-sum-exp smoothed objective."""
    f = np.maximum(f, 0).copy()
    f = normalize(f)
    best_f = f.copy()
    best_score = do_score(f)
    initial_score = best_score

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    start = time.monotonic()
    last_improve = start
    stall_count = 0
    best_beta = beta_start

    while time.monotonic() - start < time_budget:
        t += 1
        # Adaptive beta: increase over time
        progress = min(1.0, (time.monotonic() - start) / (time_budget * 0.8))
        beta = beta_start * (beta_end / beta_start) ** progress

        c = np.convolve(f, f)
        m_c = c.max()
        # Compute softmax weights (numerically stable)
        scaled = beta * (c - m_c)
        p = np.exp(scaled - scaled.max())
        p = p / p.sum()

        # Compute gradient
        grad = compute_gradient(f, p)

        # Project gradient to sum=0 subspace (to maintain normalization)
        grad = grad - grad.mean()

        # Adam update
        m_buf = b1 * m_buf + (1 - b1) * grad
        v_buf = b2 * v_buf + (1 - b2) * grad ** 2
        m_hat = m_buf / (1 - b1 ** t)
        v_hat = v_buf / (1 - b2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / s

        score = do_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            last_improve = time.monotonic()
            stall_count = 0
            if verbose and (t % 200 == 0 or score < best_score - 1e-4):
                print(f"    Adam t={t}: score={best_score:.6f} (beta={beta:.1f})")
        else:
            stall_count += 1
            # If stalling, increase learning rate temporarily
            if stall_count > 500:
                lr = min(lr * 1.1, 0.1)
                stall_count = 0

        f = f_new

        # Early stop check: if we've been stuck for a long time
        if time.monotonic() - last_improve > time_budget * 0.3 and t > 1000:
            break

    if verbose:
        print(f"    Adam done: t={t}, score={best_score:.6f}, time={time.monotonic()-start:.1f}s")

    return best_f, best_score


def run_fw_polish(f, n, time_budget, max_iters=50, verbose=True):
    """Run Frank-Wolfe LP-based polishing."""
    f = np.maximum(f, 0).copy()
    f = normalize(f)
    best_f = f.copy()
    best_score = do_score(f)

    start = time.monotonic()
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        c = np.convolve(f, f)
        M = c.max()

        # Build constraint matrix
        A = build_conv_matrix_sparse(f, n)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': max(1, time_budget - (time.monotonic() - start))})
        except Exception:
            break

        if not res.success:
            break

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-10:
            break

        g = g / g_sum

        # Frank-Wolfe step size
        t = 2.0 / (it + 2)
        t = min(t, 0.4)

        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new = normalize(f_new)

        score = do_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            if verbose and it % 5 == 0:
                print(f"    FW it={it}: score={best_score:.6f}")

        f = f_new

    if verbose:
        print(f"    FW done: best_score={best_score:.6f}")

    return best_f, best_score


def make_initialization(n, method, rng, seed):
    """Create an initial sequence of length n using the given method."""
    if method == 'uniform':
        f = np.ones(n)
    elif method == 'random_exp':
        f = rng.exponential(1.0, n)
    elif method == 'random_uniform':
        f = rng.uniform(0.1, 1.0, n)
    elif method == 'u_shaped':
        i = np.arange(n, dtype=np.float64)
        mid = (n - 1) / 2.0
        f = 1.0 / (1.0 + ((i - mid) / (n * 0.25)) ** 2)
    elif method == 'bimodal':
        i = np.arange(n, dtype=np.float64)
        f = np.exp(-((i - n * 0.15) ** 2) / (n * 0.04)) + \
            np.exp(-((i - n * 0.85) ** 2) / (n * 0.04))
        f = f + 0.05
    elif method == 'one_sided':
        i = np.arange(n, dtype=np.float64)
        f = np.exp(-i / (n * 0.3))
        f = f + 0.01
    elif method == 'resample':
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if len(base) >= 2:
                indices = np.linspace(0, len(base) - 1, n).astype(int)
                f = base[indices].copy()
            else:
                f = np.ones(n)
        except (NameError, Exception):
            f = np.ones(n)
    elif method == 'triangular':
        i = np.arange(n, dtype=np.float64)
        mid = (n - 1) / 2.0
        f = 1.0 - np.abs(i - mid) / mid
        f = np.maximum(f, 0.1)
    elif method == 'inverse_triangular':
        i = np.arange(n, dtype=np.float64)
        mid = (n - 1) / 2.0
        f = 0.3 + np.abs(i - mid) / mid
    else:
        f = np.ones(n)

    f = np.maximum(f, 0.0)
    return normalize(f)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a sequence minimizing 2n*max(convolve(a,a))/sum(a)^2.
    Uses smoothed gradient descent (Adam with log-sum-exp) as the main
    optimizer, with periodic Frank-Wolfe LP polishing.
    Tries multiple lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0, float(budget_s) - 5.0)

    best_sequence = None
    best_score = float('inf')

    # Define search configurations: (length, init_method)
    configs = [
        (1000, 'uniform'),
        (1000, 'u_shaped'),
        (1000, 'random_exp'),
        (2000, 'uniform'),
        (2000, 'bimodal'),
        (1500, 'resample'),
        (2000, 'u_shaped'),
        (1000, 'bimodal'),
        (3000, 'uniform'),
        (1500, 'random_uniform'),
        (2000, 'random_exp'),
        (1000, 'one_sided'),
    ]

    # Time allocation: give more time to early configs, less to later
    # Total budget minus safety margin
    total_time = budget_s - 10.0
    # Weight configs: earlier ones get more time
    weights = np.linspace(2.0, 0.5, len(configs))
    weights = weights / weights.sum()
    times = weights * total_time

    for idx, (n, method) in enumerate(configs):
        if time.monotonic() > deadline - 30:
            break

        time_budget = min(times[idx], deadline - time.monotonic() - 10)
        if time_budget < 5:
            continue

        print(f"\nConfig {idx+1}/{len(configs)}: n={n}, method={method}, budget={time_budget:.0f}s")
        print(f"  Current best: {best_score:.6f}")

        # Phase 1: Smoothed Adam (70% of time)
        adam_time = time_budget * 0.65
        f_init = make_initialization(n, method, rng, seed + idx)
        s_init = do_score(f_init)
        if s_init < best_score:
            best_score = s_init
            best_sequence = f_init.copy()

        f_opt, s_opt = run_smoothed_adam(
            f_init, n, adam_time,
            beta_start=5.0, beta_end=80.0,
            lr=0.005, verbose=True
        )
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()

        # Phase 2: FW Polish (30% of time)
        fw_time = min(time_budget * 0.30, deadline - time.monotonic() - 10)
        if fw_time > 5:
            f_polished, s_polished = run_fw_polish(
                f_opt, n, fw_time,
                max_iters=30, verbose=True
            )
            if s_polished < best_score:
                best_score = s_polished
                best_sequence = f_polished.copy()

        # Phase 3: Quick perturbation restarts (remaining time)
        remaining = deadline - time.monotonic() - 10
        if remaining > 20 and best_sequence is not None and len(best_sequence) == n:
            # Try a few random perturbations of the best
            for restart in range(3):
                if time.monotonic() > deadline - 10:
                    break
                f_restart = best_sequence.copy()
                f_restart *= np.exp(rng.normal(0, 0.15, n))
                f_restart = np.maximum(f_restart, 0)
                f_restart = normalize(f_restart)
                s_r = do_score(f_restart)
                if s_r < best_score:
                    best_score = s_r
                # Quick Adam from here
                f_r_opt, s_r_opt = run_smoothed_adam(
                    f_restart, n, min(remaining / 3, 30),
                    beta_start=20.0, beta_end=60.0,
                    lr=0.003, verbose=False
                )
                if s_r_opt < best_score:
                    best_score = s_r_opt
                    best_sequence = f_r_opt.copy()
                remaining = deadline - time.monotonic() - 10

        print(f"  After config {idx+1}: best={best_score:.6f}")

    # Final safety: if we somehow have no valid sequence
    if best_sequence is None:
        best_sequence = np.ones(1000) / 1000.0

    print(f"\nFinal best score: {best_score:.6f}, length: {len(best_sequence)}")

    # Ensure all values are valid (non-negative, finite, bounded)
    result = np.clip(best_sequence, 0.0, 1000.0)
    return [float(x) for x in result]
