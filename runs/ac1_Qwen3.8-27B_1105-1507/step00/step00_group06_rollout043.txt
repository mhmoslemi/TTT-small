    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()  # next power of 2

    fb = np.fft.rfft(b, fft_size)

import time

def fast_convolve(a, b=None):
    """Compute linear convolution a * a (or a * b) using FFT."""
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size)[:n]

def compute_score(a):
    """Compute the evaluation score without calling evaluate_sequence."""
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf, conv
    return float(2.0 * n * max_b / (sum_a ** 2)), conv

def subgradient(a, active_set):
    """Compute subgradient of max conv at active positions."""
    n = len(a)
    sg = np.zeros(n)
    for k in active_set:
        for j in range(n):
            if 0 <= k - j < n:
                sg[j] += a[k - j]
            if 0 <= j - k < n:
                sg[j] += a[j - k]
    return sg

def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp_step(f, M, active_indices, n):
    """Solve max sum(b) s.t. (f*b)[k] <= M for k in active, b >= 0."""
    b = cp.Variable(n)
    constraints = [b >= 0]
    
    # Build constraint matrix for active indices only
    # (f*b)[k] = sum_j A[k,j] * b[j] where A[k,j] = f[k-j] if 0<=k-j<n
    for k in active_indices:
        # (f*b)[k] = sum over valid j of f[k-j] * b[j]
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        # A for this row: f[k-j] for j in [j_lo, j_hi]
        # = f[k-j_lo], f[k-j_lo-1], ..., f[k-j_hi]
        # = f[k-j_lo : k-j_hi-1 : -1]
        row = f[k - j_lo : k - j_hi - 1 : -1]
        # Constraint: row @ b[j_lo:j_hi+1] <= M
        # But we need the full b vector, so pad with zeros
        # Actually, we can just use the full b and only constrain the relevant entries
        full_row = np.zeros(n)
        full_row[j_lo:j_hi+1] = row
        constraints.append(full_row @ b <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS, max_iter=5000)
    
    if b.value is None:
        return None
    return np.maximum(b.value, 0)

import cvxpy as cp

def fast_self_convolve(a):
    n = len(a)
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:conv_len]

def compute_score(a):
    n = len(a)
    conv = fast_self_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf, conv
    return float(2.0 * n * max_b / (sum_a ** 2)), conv

def get_active_set(conv, rel_thresh=0.95):
    max_b = conv.max()
    if max_b <= 0:
        return np.array([len(conv) // 2])
    thresh = rel_thresh * max_b
    return np.where(conv >= thresh)[0]

def build_constraint_row(f, k, n):
    """Build the row of the constraint matrix for (f*b)[k] <= M."""
    j_lo = max(0, k - n + 1)
    j_hi = min(k, n - 1)
    row = np.zeros(n)
    # (f*b)[k] = sum_{j=j_lo}^{j_hi} f[k-j] * b[j]
    # f[k-j] for j = j_lo, ..., j_hi
    # = f[k-j_lo], f[k-j_lo-1], ..., f[k-j_hi]
    indices = k - np.arange(j_lo, j_hi + 1)
    row[j_lo:j_hi+1] = f[indices]
    return row

def lp_direction(f, M, active_set, n):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M for k in active, b >= 0."""
    b = cp.Variable(n)
    constraints = [b >= 0]
    
    for k in active_set:
        row = build_constraint_row(f, int(k), n)
        constraints.append(row @ b <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    
    # Try multiple solvers
    for solver in [cp.ECOS, cp.SCS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver, max_iter=10000)
            if b.value is not None and np.all(np.isfinite(b.value)):
                return np.maximum(b.value, 0)
        except:
            continue
    
    return None

def line_search(f, g, score_f, max_iter=20):
    """Find t in (0,1] that minimizes score of (1-t)*f + t*g."""
    best_t = 0
    best_score = score_f
    sum_f = f.sum()
    sum_g = g.sum()
    if sum_g < 0.01:
        return f, score_f
    
    # Normalize g to have same sum as f
    g_norm = g * (sum_f / sum_g)

            best_score = score
            best_t = t
    
    if best_t == 0:
        return f, score_f

```python
def compute_subgradient(a, active_indices):
    n = len(a)
    sg = np.zeros(n)
    for k in active_indices:
        k = int(k)
        # Term 1: a[k-j] for j in [max(0, k-n+1), min(k, n-1)]
        lo1 = max(0, k - n + 1)
        hi1 = min(n, k + 1)
        # a[k-j] for j = lo1, ..., hi1-1
        # = a[k-lo1], a[k-lo1-1], ..., a[k-(hi1-1)]
        sg[lo1:hi1] += a[k-lo1 : k-hi1+1 : -1]
        
        # Term 2: a[j-k] for j in [max(0, k), min(n, k+n-1)]
        lo2 = max(0, k)
        hi2 = min(n, k + n)
        # a[j-k] for j = lo2, ..., hi2-1
        # = a[lo2-k], a[lo2-k+1], ..., a[hi2-1-k]
        sg[lo2:hi2] += a[lo2-k : hi2-k]
    
    return sg / len(active_indices)
```

This is O(K*n) with small constant thanks to numpy vectorization.

The simplex projection: the standard algorithm projects a vector onto {x >= 0, sum(x) = 1}. Here's the O(n log n) algorithm:

```python
def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

This is the standard Dykstra's algorithm / Euclidean projection onto the simplex.
