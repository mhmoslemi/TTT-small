from scipy import sparse

def build_conv_matrix(f, n):
    """Build sparse matrix A where (A @ g)[k] = (f*g)[k] for full convolution."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        js = np.arange(j_lo, j_hi + 1)
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[k - js].tolist())
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n - 1, n))
    return A

def solve_lp_step(f, n):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= max(f*f) for all k.
    Returns the optimal g (normalized to sum 1) or None if LP fails."""
    c_conv = np.convolve(f, f)
    M = c_conv.max()
    if M <= 0:
        return None
    
    # Build constraint matrix
    A = build_conv_matrix(f, n)
    
    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: A @ g <= M (all 2n-1 constraints)
    b_ub = np.full(2 * n - 1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

        return None
    
    g = res.x
    g = np.clip(g, 0, None)
    s = g.sum()
    if s < 1e-10:
        return None
    g = g / s  # normalize to sum 1
    return g

    """Iteratively apply LP projection to minimize max(convolve(f,f))."""

    if s < 1e-10:

def gradient_optimize(f, n, max_iter=10000, lr_init=0.01):
    """Projected subgradient descent (using all near-active constraints)."""
    f = f.copy().astype(np.float64)

            best_f = f.copy()

"""
LP-based and smoothed-subgradient optimization for minimizing
score = 2n * max(convolve(a,a)) / sum(a)^2.
Uses the state-of-the-art LP projection method with sparse constraints,
combined with softmax-smoothed subgradient descent for longer sequences.
Multiple initializations (Hann, triangle, random, provided) at various lengths.

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def build_conv_matrix(f, n):
    """Build sparse matrix A where (A @ g)[k] = (f*g)[k] for full convolution."""
    n_rows = 2 * n - 1
    total_nnz = n * n
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    idx = 0
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        count = j_hi - j_lo + 1
        js = np.arange(j_lo, j_hi + 1, dtype=np.int32)
        rows[idx:idx + count] = k
        cols[idx:idx + count] = js
        vals[idx:idx + count] = f[k - js]
        idx += count
    A = sparse.csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(n_rows, n))
    return A


def solve_lp_step(f, n, rows_pre, cols_pre):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= max(f*f) for all k.
    Returns optimal g normalized to sum 1, or None."""
    c_conv = np.convolve(f, f)
    M = c_conv.max()
    if M <= 1e-15:
        return None

    k_arr = rows_pre.astype(np.int64)
    j_arr = cols_pre.astype(np.int64)
    vals = f[k_arr - j_arr]
    A = sparse.csr_matrix((vals, (rows_pre, cols_pre)), shape=(2 * n - 1, n))

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None

    g = np.clip(res.x, 0, None)
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s


def lp_optimize(f, n, max_iter=100, deadline=None):
    """Iterative LP projection to minimize max(convolve(f,f))."""
    f = np.clip(f.astype(np.float64), 0, None)
    s = f.sum()
    if s < 1e-10:
        return f
    f = f / s

    total_nnz = n * n
    rows_pre = np.empty(total_nnz, dtype=np.int32)
    cols_pre = np.empty(total_nnz, dtype=np.int32)

        j_hi = min(n - 1, k)
        count = j_hi - j_lo + 1
        rows_pre[idx:idx + count] = k
        cols_pre[idx:idx + count] = np.arange(j_lo, j_hi + 1, dtype=np.int32)
        idx += count
    rows_pre = rows_pre[:idx]
    cols_pre = cols_pre[:idx]

    best_f = f.copy()
    best_M = np.convolve(f, f).max()
    t = 0.4

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        g = solve_lp_step(f, n, rows_pre, cols_pre)
        if g is None:

        # Line search over a few step sizes
        best_t = t
        best_M_new = best_M
        for t_try in [t * 0.5, t, t * 1.5, min(1.0, t * 2.0)]:

            best_f = f_new.copy()
            f = f_new
            t = max(0.1, best_t * 0.95)
