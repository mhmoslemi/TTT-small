import numpy as np
import time

"""
Minimizes score = 2*n*max(convolve(a,a))/sum(a)^2 using smooth gradient descent
(soft-max approximation) with multiple parametric initializations, momentum,
and adaptive step sizing. Uses FFT-based convolution for O(n log n) steps.
"""

def _setup_fft(n):
    """Return the FFT size for computing linear convolution of two length-n arrays."""
    conv_len = 2 * n - 1
    fft_size = 1
    while fft_size < conv_len:
        fft_size *= 2
    return fft_size


def _compute_conv(a, fft_size):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    A = np.fft.rfft(a, fft_size)
    C = A * A
    c = np.fft.irfft(C, fft_size)
    return c[:2 * len(a) - 1]


def _smooth_grad(a, fft_size, tau):
    """
    Compute soft-max score and its gradient w.r.t. a.
    score = (1/tau) * log(sum_k exp(tau*c[k]/S^2))
    Returns (score, grad) where grad has same shape as a.
    """
    c = _compute_conv(a, fft_size)
    S = a.sum()
    z = tau * c / (S * S)
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    Z = exp_z.sum()
    p = exp_z / Z  # softmax weights, length 2n-1
    score = (z_max + np.log(Z)) / tau  # = (1/tau)*log(sum exp(z_k))

    # Compute corr[m] = sum_{j=0}^{n-1} p[m+j] * a[j] via FFT
    n = len(a)
    N = fft_size
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(np.conj(P) * A, N)
    corr = R[:n]  # corr[m] = sum_j p[m+j]*a[j] for m=0..n-1

    # sum_k p[k]*c[k]
    c_avg = np.dot(p, c)

    # grad[m] = 2*corr[m]/S^2 - 2*c_avg*a[m]/S^3
    grad = 2.0 * corr / (S * S) - 2.0 * c_avg * a / (S * S * S)
    return score, grad


def _hard_subgrad(a, fft_size):
    """Compute hard max score and subgradient. Returns (score, subgrad)."""
    c = _compute_conv(a, fft_size)
    S = a.sum()
    M = c.max()
    k_star = int(c.argmax())
    # Subgrad: g[m] = 2*a[k_star - m] if 0 <= k_star - m < n
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        # g[lo:hi+1] = 2*a[k_star-hi : k_star-lo+1] reversed
        g[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]
    # Full gradient of M/S^2: g/S^2 - 2*M*a/S^3
    grad = g / (S * S) - 2.0 * M * a / (S * S * S)
    score = M / (S * S)
    return score, grad


def _optimize(a_init, fft_size, time_limit, tau=10.0):

    Returns the best sequence found.
    """

    n = len(a)

    step_iter = 0
    stagnant = 0

    tau_current = tau

    while True:

        # Adapt tau: increase if peaks are too spread (low concentration),

        S = a.sum()
        peak_ratio = c.max() / c.mean()
        if peak_ratio > 5.0:

            tau_current = max(tau_current * 0.8, 3.0)

        # Momentum update

        # Adaptive step size (Polyak-inspired)
        gap = max(score - lower_bound, 1e-10)
        gnorm_sq = np.dot(momentum, momentum) + 1e-15

        lr = min(lr, 0.05)
        lr = max(lr, 1e-8)

        if S_new < 0.01:

        else:

        a = a_new

                stagnant = 0

                        if S2 > 0.01:

    # Final eval

    S_final = best_local.sum()
    final_score = c_final.max() / (S_final * S_final)
    if final_score < best_score:
        best_score = final_score
    return best_local, 2.0 * n * best_score
