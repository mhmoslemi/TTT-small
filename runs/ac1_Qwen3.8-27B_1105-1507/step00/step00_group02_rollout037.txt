"""
Minimizes 2*n*max(convolve(seq,seq))/sum(seq)^2 using power ramp initialization
and subgradient-based local optimization with FFT convolution.
"""

import numpy as np
import time


def make_power_ramp(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    return np.power(idx, p)


def make_bell_ramp(n, p, alpha):
    """a_i = (i+1)^p * exp(-alpha * i/(n-1))"""
    idx = np.arange(1, n + 1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    return np.power(idx, p) * np.exp(-alpha * t)


def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    result = np.fft.irfft(A * B, size)
    return result[:n]


def compute_pressure(a, active_positions):
    """Compute the pressure vector for subgradient step."""
    n = len(a)
    pressure = np.zeros(n)
    for k in active_positions:
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        # For position k in convolution: c[k] = sum_{j=j_min}^{j_max} a[j] * a[k-j]
        # k-j ranges from k-j_max to k-j_min
        k_j_min = k - j_max
        k_j_max = k - j_min
        # partner[j] = a[k-j] for j in [j_min, j_max]
        # = a[k_j_max], a[k_j_max-1], ..., a[k_j_min] (reversed)
        partner = a[k_j_min:k_j_max + 1][::-1]
        pressure[j_min:j_max + 1] += partner
    return pressure


def subgradient_iter(a, lr, tol=0.02):
    """One subgradient iteration. Returns updated a and the max convolution value."""
    n = len(a)
    c = fft_convolve(a, a)
    max_c = c.max()
    if max_c <= 0:
        return a, max_c
    threshold = (1.0 - tol) * max_c
    active = np.where(c >= threshold)[0]
    pressure = compute_pressure(a, active)
    # Update: decrease elements proportional to their pressure
    a_new = a * np.exp(-lr * pressure)
    a_new = np.clip(a_new, 0.0, None)
    # Normalize to max = 1
    max_a = a_new.max()
    if max_a > 0:
        a_new = a_new / max_a
    return a_new, max_c


def random_perturb(a, rng, strength=0.3):
    """Apply random multiplicative perturbation."""
    n = len(a)
    log_pert = rng.normal(0, strength, size=n)
    a_new = a * np.exp(log_pert)
    return np.clip(a_new, 0.0, None)


def block_perturb(a, rng, num_blocks=3, strength=0.5):
    """Apply random block perturbation."""
    n = len(a)
    a_new = a.copy()
    for _ in range(num_blocks):
        width = max(2, int(n * rng.uniform(0.01, 0.1)))
        start = int(rng.integers(0, n - width + 1))
        a_new[start:start + width] *= np.exp(rng.normal(0, strength, size=width))
    return np.clip(a_new, 0.0, None)

"""
Minimizes 2*n*max(convolve(seq,seq))/sum(seq)^2 using power ramp initialization
followed by subgradient-based local optimization with FFT convolution,
adaptive learning rates, and periodic random perturbations for diversity.
"""

import numpy as np
import time


def make_power_ramp(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    return np.power(idx, p)


def make_bell_ramp(n, p, alpha):
    idx = np.arange(1, n + 1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    return np.power(idx, p) * np.exp(-alpha * t)


def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    result = np.fft.irfft(A * B, size)
    return result[:n]


def compute_pressure(a, active_positions):
    n = len(a)
    pressure = np.zeros(n)
    for k in active_positions:
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        k_j_min = k - j_max
        k_j_max = k - j_min
        partner = a[k_j_min:k_j_max + 1][::-1]
        pressure[j_min:j_max + 1] += partner
    return pressure


def subgradient_iter(a, lr, tol=0.02):
    n = len(a)
    c = fft_convolve(a, a)
    c = np.maximum(c, 0.0)
    max_c = c.max()
    if max_c <= 0:
        return a, max_c
    threshold = (1.0 - tol) * max_c
    active = np.where(c >= threshold)[0]
    pressure = compute_pressure(a, active)
    a_new = a * np.exp(-lr * pressure)
    a_new = np.clip(a_new, 0.0, None)
    max_a = a_new.max()
    if max_a > 0:
        a_new = a_new / max_a
    return a_new, max_c


def block_perturb(a, rng, num_blocks=3, strength=0.5):

        a_new[start:start + width] *= np.exp(rng.normal(0, strength, size=width))
    return np.clip(a_new, 0.0, None)


def random_perturb(a, rng, strength=0.3):
    n = len(a)
    a_new = a * np.exp(rng.normal(0, strength, size=n))
    return np.clip(a_new, 0.0, None)
