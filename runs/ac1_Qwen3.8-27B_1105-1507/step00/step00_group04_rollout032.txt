def compute_convolution(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    L = 1
    while L < 2*n - 1:
        L *= 2
    fa = np.fft.rfft(a, L)
    b = np.fft.irfft(fa * fa, L)
    return b[:2*n-1]

def normalize_simplex(a):
    """Project a onto the simplex {x >= 0, sum(x) = 1}."""
    # This is the Euclidean projection onto the probability simplex
    n = len(a)
    if np.sum(a) <= 0:
        return np.ones(n) / n
    # Use the water-filling algorithm
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cssv - 1) / rho > 0
    if not np.any(cond):
        idx = -1
    else:
        idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1) / (idx + 1)
    return np.maximum(a - theta, 0)

def gradient_step(a, lr):
    """One gradient descent step with projection."""
    n = len(a)
    b = compute_convolution(a)
    k_star = np.argmax(b)
    max_b = b[k_star]
    sum_a = np.sum(a)

    a_new = a - lr * grad

    a = project_simplex(a_init.copy())
    current_score = score_fast(a)
    best_a = a.copy()
    best_score = current_score
    T = 0.1  # initial temperature

    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        T *= 0.999

        # Mix gradient direction with random noise
        grad = compute_gradient(a)
        noise_scale = T * 0.1
        pert = rng.normal(0, noise_scale, a.shape) + 0.005 * grad / (np.linalg.norm(grad) + 1e-15)

        a_new = project_simplex(a + pert)
        new_score = score_fast(a_new)

        # Accept with Metropolis criterion
        delta = new_score - current_score
        if delta < 0 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
            a = a_new
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()

    return best_a, best_score


def generate_starting_points():
    """Generate diverse starting points for the optimization."""
    starts = []

    # Given sequence
    try:
        init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if len(init) > 0 and np.sum(init) > 0.01:
            starts.append(("given", init / np.sum(init)))
    except Exception:
        pass

    # Uniform sequences at various lengths
    for n in [128, 256, 512, 1024, 2048, 4096]:
        starts.append((f"uniform_{n}", np.ones(n) / n))

    # Linearly skewed sequences
    for n in [256, 512, 1024, 2048]:
        x = np.arange(n, dtype=np.float64) / (n - 1)
        for c in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a = 1.0 + c * (x - 0.5)
            a = np.maximum(a, 0.01)
            starts.append((f"linear_{n}_{c}", a / np.sum(a)))

    # Exponential skew
    for n in [256, 512, 1024]:
        x = np.arange(n, dtype=np.float64) / (n - 1)
        for c in [0.5, 1.0, 1.5, 2.0, 3.0]:
            a = np.exp(c * (x - 0.5))
            starts.append((f"exp_{n}_{c}", a / np.sum(a)))

    # Quadratic skew
    for n in [256, 512, 1024]:
        x = np.arange(n, dtype=np.float64) / (n - 1)
        xm = x - 0.5
        for c1, c2 in [(0.5, 0.3), (0.3, -0.5), (0.7, 0.5), (0.5, -0.3)]:
            a = 1.0 + c1 * xm + c2 * xm * xm
            a = np.maximum(a, 0.01)
            if np.sum(a) > 0:
                starts.append((f"quad_{n}_{c1}_{c2}", a / np.sum(a)))

    # Sine modulated
    for n in [256, 512, 1024]:
        x = np.arange(n, dtype=np.float64) / (n - 1)
        for freq, amp in [(1, 0.3), (1, 0.5), (2, 0.3), (3, 0.3), (1, 0.7)]:
            a = 1.0 + amp * np.sin(2 * np.pi * freq * x)
            a = np.maximum(a, 0.01)
            if np.sum(a) > 0:
                starts.append((f"sine_{n}_{freq}_{amp}", a / np.sum(a)))

    # Combination: skew + modulation
    for n in [512, 1024]:
        x = np.arange(n, dtype=np.float64) / (n - 1)
        a = 1.0 + 0.4 * (x - 0.5) + 0.2 * np.sin(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        starts.append((f"combo_{n}", a / np.sum(a)))

    return starts


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Combines gradient descent, simulated annealing,
    and multiple starting points to find sequences with low J values.
    """
    deadline = time.monotonic() + budget_s - 5.0
    rng = np.random.default_rng(seed)

    best_sequence = None
    best_score = np.inf

    # Generate and score starting points
    starts = generate_starting_points()
    scored = []
    for name, s in starts:
        sc = score_fast(s)
        if np.isfinite(sc) and sc < 2.05:  # Only keep promising starts
            scored.append((sc, name, s))

    scored.sort(key=lambda x: x[0])
    print(f"Found {len(scored)} promising starting points. Best: {scored[0][0]:.6f} ({scored[0][1]})")

    # Phase 1: Gradient descent from top starting points
    top_k = min(20, len(scored))
    for i in range(top_k):
        if time.monotonic() > deadline - 30:
            break
        sc, name, s = scored[i]
        n = len(s)
        max_iters = max(1000, min(10000, 5000000 // n))
        lr_init = 0.01 * np.sqrt(n / 512)

        best_a, best_sc = gradient_descent_optimize(
            s, max_iters=max_iters, lr_init=lr_init, deadline=deadline - 10
        )

        # Verify with exact evaluation
        exact_sc = evaluate_sequence(best_a.tolist())
        if np.isfinite(exact_sc) and exact_sc < best_score:
            best_score = exact_sc
            best_sequence = best_a.copy()

        print(f"  [{i+1}/{top_k}] {name} (n={n}): init={sc:.6f} -> GD={best_sc:.6f} (exact={exact_sc:.6f})")

    print(f"After Phase 1: best_score = {best_score:.6f}")

    # Phase 2: Annealing from the best solution found so far
    if best_sequence is not None and time.monotonic() < deadline - 20:
        a_best, sc_best = anneal_from(best_sequence, n_steps=5000, deadline=deadline - 5)
        exact_sc = evaluate_sequence(a_best.tolist())
        if np.isfinite(exact_sc) and exact_sc < best_score:
            best_score = exact_sc
            best_sequence = a_best.copy()
        print(f"After Phase 2 (annealing): best_score = {best_score:.6f}")

    # Phase 3: Continue gradient descent from the global best
    if best_sequence is not None and time.monotonic() < deadline - 5:
        n = len(best_sequence)
        max_iters = max(2000, min(20000, 10000000 // n))
        best_a, best_sc = gradient_descent_optimize(
            best_sequence, max_iters=max_iters, lr_init=0.005, deadline=deadline - 2
        )
        exact_sc = evaluate_sequence(best_a.tolist())
        if np.isfinite(exact_sc) and exact_sc < best_score:
            best_score = exact_sc
            best_sequence = best_a.copy()
        print(f"After Phase 3 (refined GD): best_score = {best_score:.6f}")

    # Phase 4: Try different length near best
    if best_sequence is not None and time.monotonic() < deadline - 10:
        n_best = len(best_sequence)
        # Try a few nearby lengths by interpolating
        for n_try in [int(n_best * 0.75), int(n_best * 1.25)]:
            if time.monotonic() > deadline - 5:
                break
            if n_try < 32 or n_try > 10000:
                continue
            # Resample the best sequence to new length
            x_orig = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, n_try)
            a_resampled = np.interp(x_new, x_orig, best_sequence)
            a_resampled = np.maximum(a_resampled, 0)
            a_resampled = project_simplex(a_resampled)

            best_a, best_sc = gradient_descent_optimize(
                a_resampled, max_iters=2000, lr_init=0.01, deadline=deadline - 2
            )
            exact_sc = evaluate_sequence(best_a.tolist())
            if np.isfinite(exact_sc) and exact_sc < best_score:
                best_score = exact_sc
                best_sequence = best_a.copy()
            print(f"  Resampled to n={n_try}: score={exact_sc:.6f}")

    # Final verification and return
    if best_sequence is not None:
        final_sc = evaluate_sequence(best_sequence.tolist())
        if np.isfinite(final_sc) and final_sc < best_score:
            best_score = final_sc
        print(f"FINAL: best_score = {best_score:.6f}, length = {len(best_sequence)}")
        return [float(x) for x in best_sequence]
    else:
        # Fallback: uniform sequence
        fallback = [1.0] * 256
        print(f"FALLBACK: score = {evaluate_sequence(fallback):.6f}")
        return fallback
