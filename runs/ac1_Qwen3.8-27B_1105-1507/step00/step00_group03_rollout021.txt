```python
def optimize(start, rng, deadline):
    n = len(start)
    a = start.copy()
    # Normalize to sum = n
    if np.sum(a) > 0:
        a = a * n / np.sum(a)
    a = np.clip(a, 0, 1000)
    
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    
    step_size = 0.1
    alpha = 5.0  # temperature for soft-max
    k_peaks = 10  # number of peaks to consider
    
    iterations = 0
    patience = 500
    no_improve = 0
    
    while time.monotonic() < deadline and no_improve < patience:
        iterations += 1
        
        # Compute convolution
        c = np.convolve(a, a)
        
        # Find top peaks
        top_idx = np.argsort(c)[-k_peaks:]
        c_max = c[top_idx[0]]  # actually c[top_idx[-1]] is the max
        c_max = np.max(c)
        
        # Compute soft weights
        weights = np.zeros(len(c))
        weights[top_idx] = np.exp(alpha * (c[top_idx] - c_max))
        weights /= np.sum(weights)
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in top_idx:
            w = weights[k]
            # dc_k / da_i for valid i
            i_min = max(0, k - (n-1))
            i_max = min(n-1, k)
            for i in range(i_min, i_max+1):
                j = k - i
                if i == j:
                    grad[i] += w * 2 * a[i]
                else:
                    grad[i] += w * a[j]
        
        # Adjust for the denominator S^2
        S = np.sum(a)
        # Objective = 2n * max_c / S^2
        # d(obj)/da_i = 2n * (S * grad[i] - 2*max_c) / S^3
        obj_grad = 2*n * (S * grad - 2*c_max) / S**3
        
        # Take a step
        a = a - step_size * obj_grad
        a = np.clip(a, 0, 1000)
        
        # Normalize
        new_sum = np.sum(a)
        if new_sum < 0.01:
            a = best_a.copy() * rng.normal(1, 0.05)
```

```python
def compute_grad(a, top_idx, weights, n):
    """Compute the subgradient for the smooth max objective."""
    grad = np.zeros(n)
    for k, w in zip(top_idx, weights):
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # For i in [i_min, i_max]: grad[i] += 2*w*a[k-i]
        # a[k-i] for i in [i_min, i_max] is a[k-i_max] to a[k-i_min] (reversed)
```

```python
def optimize_sequence(a, rng, deadline, step_size_init=0.05, top_k=50, alpha=5.0):

        # Re-normalize

            a = np.clip(a, 0, 1000)
            S_new = np.sum(a)
        a = a * n / S_new
        
        # Periodic evaluation

            score = float(evaluate_sequence(a.tolist()))
            if score < best_score:
                best_score = score
```

```python
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using subgradient descent
with multiplicative updates, soft-max peak selection, and multiple restarts.
"""

import time


def compute_peak_gradient(a, c, top_k=50, alpha=5.0):
    """Compute weighted subgradient from top-k peaks."""
    n = len(a)
    c_max = np.max(c)
    sorted_idx = np.argsort(c)[::-1][:top_k]
    w = np.exp(alpha * (c[sorted_idx] - c_max))
    w = w / w.sum()
    
    grad = np.zeros(n)
    for idx in range(top_k):
        k = sorted_idx[idx]
        wi = w[idx]
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        start = k - i_max
```

def compute_gradient(a, c, top_k, alpha):
    """Compute weighted subgradient from top-k peaks of convolution c."""
    n = len(a)
    c_max = np.max(c)
    
    # Get top-k indices
    sorted_idx = np.argsort(c)[::-1][:top_k]
    
    # Soft weights
    w = np.exp(alpha * (c[sorted_idx] - c_max))
    w = w / w.sum()
    
    # Accumulate gradient
    grad = np.zeros(n)
    for idx in range(top_k):
        k = int(sorted_idx[idx])

        partner_slice = a[k - i_max : k - i_min + 1][::-1]
        grad[i_min : i_max + 1] += 2.0 * wi * partner_slice
    
    return grad, c_max
