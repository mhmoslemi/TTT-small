```python
def compute_gradient(a, c, temperature):
    """Compute smoothed gradient of max(c) w.r.t. a."""
    n = len(a)
    c_max = np.max(c)
    
    # Softmax weights (shifted for numerical stability)
    log_weights = temperature * (c - c_max)
    # Only compute for significant weights
    threshold = -50  # exp(-50) ≈ 2e-22
    significant = log_weights > threshold
    if not np.any(significant):
        return np.zeros(n)
    
    sig_idx = np.where(significant)[0]
    w = np.exp(log_weights[sig_idx])
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum
    
    g = np.zeros(n)
    for idx in range(len(sig_idx)):
        k = sig_idx[idx]
        wi = w[idx]
        
        # First term: a[k-j] for valid j (0 <= k-j < n)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:

        j_hi2 = min(n - 1, k + n - 1)
        if j_lo2 <= j_hi2:
            g[j_lo2:j_hi2 + 1] += wi * a[np.arange(j_lo2, j_hi2 + 1) - k]
    
    return g
```

def compute_smoothed_gradient(a, c, temperature):

def lp_descent_direction(a, n):
    """
    Solve LP to find descent direction.
    Maximize sum(g) subject to g >= 0 and (a*g)_k <= M for all k.
    Returns the direction g (normalized to sum=1) or None.
    """
    c = np.convolve(a, a)
    M = np.max(c)
    
    # Build constraint matrix: A[k, j] = a[k-j] if 0 <= k-j < n
    # Use sparse for efficiency
    from scipy.sparse import lil_matrix, csr_matrix
    
    A = lil_matrix((2*n-1, n))
    for j in range(n):
        end = min(n, 2*n-1-j)
        if end > 0:
            A[j:j+end, j] = a[:end]
    A = A.tocsr()
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, 
                        method='highs', options={'presolve': True, 'time_limit': 30})
        if result.success and result.x is not None:
            s = np.sum(result.x)
            if s > 1e-10:
                return result.x / s
    except Exception:
        pass
    return None

```python
def apply_block_perturbation(a, rng, strength=0.15):
    """Apply multiplicative block perturbation for escape."""
    n = len(a)
    candidate = a.copy()
    n_blocks = rng.integers(3, max(4, n // 200))
    block_size = max(1, n // n_blocks)
    for _ in range(n_blocks):
        start = rng.integers(0, max(1, n - block_size))
        end = min(n, start + block_size)
        candidate[start:end] *= np.exp(rng.normal(0, strength))
    candidate = np.maximum(candidate, 0)
    s = np.sum(candidate)
    if s < 1e-10:
        return a.copy()
    return candidate / s
```
