def solve_lp(f, M):
    """Solve the LP to find the best direction b."""
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    
    # Constraints: (f*b)[k] <= M for all k = 0, ..., 2n-2
    # (f*b)[k] = sum_i f[i] * b[k-i] where 0 <= i < n and 0 <= k-i < n
    constraints = []
    f_arr = np.array(f)
    for k in range(2*n - 1):
        # i ranges from max(0, k-n+1) to min(k, n-1)
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        # (f*b)[k] = sum_{i=i_min}^{i_max} f[i] * b[k-i]
        # b[k-i] where k-i ranges from k-i_max to k-i_min
        # Let j = k-i, so j ranges from k-i_max to k-i_min
        j_min = k - i_max
        j_max = k - i_min
        # constraint: sum_{i=i_min}^{i_max} f[i] * b[k-i] <= M
        # = sum_{j=j_min}^{j_max} f[k-j] * b[j] <= M
        coeffs = f_arr[k - j_max : k - j_min + 1][::-1]
        constraints.append(coeffs @ b <= M)
    
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLOP)  # or CLP, or CBC
    
    if b.value is None:
        return None
    return b.value

def lp_direction(f, M):
    """Solve LP to find direction b that maximizes sum(b) subject to f*b <= M and b >= 0."""
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    
    constraints = []
    f_arr = np.asarray(f, dtype=np.float64)
    
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        # coefficients for b[j] where j = k - i
        # i goes from i_min to i_max, so j = k-i goes from k-i_max to k-i_min
        j_lo = k - i_max
        j_hi = k - i_min
        # coeff for b[j_hi], b[j_hi-1], ..., b[j_lo] is f[i_lo], f[i_lo+1], ..., f[i_hi]

"""
Minimizes score = 2*n*max(convolve(a,a))/(sum(a))^2 using LP-based direction finding,
subgradient steps, and local search. Exploits asymmetry to break the score=2 barrier.
"""

import numpy as np

def lp_find_direction(f, M, max_iters_lp=100):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0. Returns b or None."""
    n = len(f)
    f = np.asarray(f, dtype=np.float64)
    b = cp.Variable(n, nonneg=True)
    constraints = []
    
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        # (f*b)[k] = sum_{i=i_min}^{i_max} f[i] * b[k-i]
        # Let j = k-i, j ranges from k-i_max to k-i_min
        # Coeff of b[j] is f[k-j]
        j_lo = k - i_max
        j_hi = k - i_min
        # b[j_lo], ..., b[j_hi] have coeffs f[k-j_hi], ..., f[k-j_lo] = f[i_min], ..., f[i_max]
        # Wait: k - j_hi = k - (k-i_min) = i_min, k - j_lo = k - (k-i_max) = i_max
        # So coeff of b[j_lo+m] is f[k-(j_lo+m)] = f[i_max-m]
        # coeffs = f[i_max], f[i_max-1], ..., f[i_min]
        coeffs = f[i_min:i_max+1][::-1]
        constraints.append(coeffs @ b[j_lo:j_hi+1] <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.CLP)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK)  
        except Exception:
            return None
    
    if b.value is None or not np.all(np.isfinite(b.value)):
        return None
    return np.asarray(b.value, dtype=np.float64)


def lp_optimize(f, n_lp_iters=30, t_init=1.0, t_decay=0.95):
    """Run LP-based iterations. Returns (best_f, best_score)."""
    f = np.asarray(f, dtype=np.float64).copy()
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, float('inf')
    
    best_f = f.copy()
    best_score = float('inf')
    t = t_init
    
    for it in range(n_lp_iters):
        C = np.convolve(f, f)
        M = float(np.max(C))
        S = float(np.sum(f))
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        b = lp_find_direction(f, M)
        if b is None:
            break
        Sb = float(np.sum(b))
        if Sb < 1e-10:
            break
        g = b * (S / Sb)
        f = (1.0 - t) * f + t * g
        f = np.clip(f, 0.0, None)
        t *= t_decay
        t = max(t, 0.01)
    
    C = np.convolve(f, f)
    M = float(np.max(C))
    S = float(np.sum(f))
    score = 2.0 * n * M / (S * S)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def subgradient_descent(f, n_steps=2000, step_init=0.1):
