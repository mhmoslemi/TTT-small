                seqs.insert(0, parent)
        except:
            pass

        # Evaluate all initial sequences
        local_best_seq = None
        local_best_score = float('inf')

        for s in seqs:
            s = np.clip(np.asarray(s, dtype=np.float64), 0.0, 1000.0)
            if s.size != n or s.sum() < 0.01:
                continue
            score = evaluate_sequence(s.tolist())
            if np.isfinite(score) and score < local_best_score:
                local_best_score = score
                local_best_seq = s.copy()

        if local_best_seq is None:
            continue

        print(f"n={n}: initial best score = {local_best_score:.6f}")

        # Gradient descent phase
        a = local_best_seq.copy()
        current_score = local_best_score
        no_improve_count = 0
        max_no_improve = 500

        # Adaptive beta schedule
        phase = 0
        betas = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

        while time.monotonic() < deadline - 30 and no_improve_count < max_no_improve:
            conv = np.convolve(a, a)
            M = float(conv.max())
            S = float(a.sum())

            if S < 0.01 or M < 1e-15:
                break

            # Choose gradient mode
            r = rng.random()
            if phase < 3:
                mode = 'max'  # Start with pure max gradient
            elif r < 0.4:
                mode = 'max'
            else:
                mode = 'softmax'

            beta = betas[min(phase, len(betas) - 1)]
            grad = _compute_gradient(a, conv, mode=mode, beta=beta)

            if np.all(grad == 0):
                no_improve_count += 1
                continue

            # Normalize gradient for consistent step sizing
            grad_norm = np.max(np.abs(grad))
            if grad_norm < 1e-15:
                no_improve_count += 1
                continue
            grad = grad / grad_norm

            # Line search with multiple trial step sizes
            best_trial_score = current_score
            best_trial = None

            # Step sizes scaled to be meaningful relative to a
            a_scale = np.max(a)
            for lr_mult in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                lr = lr_mult * a_scale
                trial = a - lr * grad
                trial = np.maximum(trial, 0.0)
                ts = trial.sum()
                if ts < 0.01:
                    continue
                trial = trial / ts
                trial_score = evaluate_sequence(trial.tolist())
                if np.isfinite(trial_score) and trial_score < best_trial_score:
                    best_trial_score = trial_score
                    best_trial = trial.copy()

            if best_trial is not None and best_trial_score < current_score:
                a = best_trial
                current_score = best_trial_score
                no_improve_count = 0

                if current_score < local_best_score:
                    local_best_score = current_score
                    local_best_seq = a.copy()

                if current_score < best_score:
                    best_score = current_score
                    best_seq = a.copy()
                    print(f"  n={n}: new best = {best_score:.6f} (iter, no_improve={no_improve_count})")
            else:
                no_improve_count += 1

                # Diversification: random perturbation
                if no_improve_count % 50 == 0:
                    strength = min(0.1, 0.01 * no_improve_count / 50)
                    a = a * (1.0 + strength * rng.standard_normal(n))
                    a = np.maximum(a, 0.0)
                    a_sum = a.sum()
                    if a_sum > 0.01:
                        a = a / a_sum
                    else:
                        a = np.ones(n) / n

                # Gradient ascent on min (to move away from saddle)
                if no_improve_count % 200 == 0:
                    a = a * (1.0 + 0.2 * rng.standard_normal(n))
                    a = np.maximum(a, 0.0)
                    a_sum = a.sum()
                    if a_sum > 0.01:
                        a = a / a_sum

            # Advance phase
            if no_improve_count > 200 and phase < len(betas) - 1:
                phase += 1

        # Coordinate descent polish
        if time.monotonic() < deadline - 10:
            a = local_best_seq.copy()
            a = _coordinate_descent_polish(a, max_sweeps=100)
            pol_score = evaluate_sequence(a.tolist())
            if np.isfinite(pol_score) and pol_score < local_best_score:
                local_best_score = pol_score
                local_best_seq = a
                print(f"  n={n}: after CD polish = {pol_score:.6f}")

            if local_best_score < best_score:
                best_score = local_best_score
                best_seq = local_best_seq.copy()

    # Final: try a few more random restarts if time remains
    if best_seq is None:
        best_seq = np.ones(4806) / 4806
        best_score = 2.0

    if time.monotonic() < deadline - 10:
        n = len(best_seq)
        for _ in range(5):
            if time.monotonic() > deadline - 10:
                break
            a = 0.1 + 0.9 * rng.random(n)
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"  Random restart improved to {best_score:.6f}")
                # Do a few gradient steps from here
                for _ in range(200):
                    if time.monotonic() > deadline - 10:
                        break
                    conv = np.convolve(a, a)
                    M = conv.max()
                    grad = _compute_gradient(a, conv, mode='max')
                    if np.all(grad == 0):
                        break
                    grad_norm = np.max(np.abs(grad))
                    grad = grad / grad_norm
                    a_scale = np.max(a)
                    improved = False
                    for lr_mult in [0.05, 0.1, 0.2, 0.3, 0.5]:
                        trial = a - lr_mult * a_scale * grad
                        trial = np.maximum(trial, 0.0)
                        ts = trial.sum()
                        if ts < 0.01:
                            continue
                        trial = trial / ts
                        ts_score = evaluate_sequence(trial.tolist())
                        if np.isfinite(ts_score) and ts_score < score:
                            score = ts_score
                            a = trial
                            improved = True
                            break
                    if improved and score < best_score:
                        best_score = score
                        best_seq = a.copy()

    print(f"FINAL: best_score = {best_score:.6f}, length = {len(best_seq)}")
    return [float(x) for x in best_seq]
