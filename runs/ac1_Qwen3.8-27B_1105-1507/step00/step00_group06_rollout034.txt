def _conv_fft(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2 * n - 1]
    return conv
