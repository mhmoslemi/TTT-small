```python
"""
Search for low-peaked non-negative step-function coefficients.

Strategy:
  * Work primarily with piecewise-constant block profiles of the final
    sequence, which reduces the search dimension while preserving the
    important shape of the coefficient sequence.
  * Generate several asymmetric initial profiles (two-level steps,
    exponentials, powers, smooth random profiles).
  * Optimize block values with a soft-max cardinality contour:
        minimize  sum_k w_k (f*f)_k
    where w_k is concentrated near peaks of the convolution.  Updates are
    made in log-block-value space, projected onto the non-negative simplex
    by normalization.
  * Refine the best full-length sequence with peak-shaving multiplicative
    updates and random local perturbations.
  * Use the exact evaluate_sequence() call only periodically to keep the
    authoritative best sequence.
"""

import numpy as np
import time


def _pow2(m: int) -> int:
    p = 1
    while p < m:
        p <<= 1
    return p


def _fft_conv_self(a: np.ndarray) -> np.ndarray:
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    ca = np.fft.irfft(fa * fa, N)
    ca = ca[: 2 * n - 1]
    # The true convolution of non-negative data is non-negative.
    ca[ca < 0.0] = 0.0
    return ca


def _score_fft(f: np.ndarray) -> float:
    f = np.asarray(f, dtype=np.float64).ravel()
    if f.size == 0:
        return np.inf
    if not np.all(np.isfinite(f)):
        return np.inf
    f = np.clip(f, 0.0, 1000.0)
    s = float(f.sum())
    if s < 0.01:
        return np.inf
    conv = _fft_conv_self(f)
    if conv.size == 0:
        return np.inf
    m = float(np.max(conv))
    if not np.isfinite(m) or m < 0.0:
        return np.inf
    return float(2.0 * f.size * m / (s * s))


def _exact_score(seq) -> float:
    try:
        val = evaluate_sequence(seq)
    except Exception:
        return np.inf
    try:
        return float(val)
    except Exception:
        return np.inf


def _normalize_seq(f: np.ndarray):
    f = np.asarray(f, dtype=np.float64).ravel().copy()
    if f.size == 0:
        return None
    f = np.clip(f, 0.0, 1000.0)
    s = float(f.sum())
    if s <= 0.0:
        return None
    return f / s


def _make_lengths(n: int, K: int) -> np.ndarray:
    n = int(n)
    K = int(K)
    if n <= 0:
        return np.array([1], dtype=np.int64)
    if K < 1:
        K = 1
    if K > n:
        K = n
    base = n // K
    rem = n % K
    if base < 1:
        return np.ones(n, dtype=np.int64)
    lengths = np.full(K, base, dtype=np.int64)
    lengths[:rem] += 1
    return lengths


def _starts(lengths: np.ndarray) -> np.ndarray:
    if lengths.size == 0:
        return np.array([], dtype=np.int64)
    return np.concatenate((np.array([0], dtype=np.int64), np.cumsum(lengths[:-1], dtype=np.int64)))


def _build_blocks(v: np.ndarray, lengths: np.ndarray) -> np.ndarray:
    return np.repeat(np.asarray(v, dtype=np.float64), np.asarray(lengths, dtype=np.int64))


def _score_blocks(v: np.ndarray, lengths: np.ndarray) -> float:
    return _score_fft(_build_blocks(v, lengths))


def _normalize_blocks(v, lengths):
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    if v.size != lengths.size:
        return None
    if not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, 0.0, 1000.0)
    total = float(np.dot(v, lengths))
    if total <= 0.0:
        v = np.ones(v.size, dtype=np.float64)
        total = float(np.sum(lengths))
    v = v / total
    v = np.maximum(v, 1e-15)
    total = float(np.dot(v, lengths))
    if total <= 0.0:
        return None
    return v / total


def _smooth_noise(rng, K: int, scale: float) -> np.ndarray:
    raw = rng.normal(0.0, 1.0, K)
    out = raw
    if K > 1:
        out = 0.5 * raw + 0.25 * np.roll(raw, 1) + 0.25 * np.roll(raw, -1)
    return out * float(scale)


def _make_two_step(K: int, alpha: float, ratio: float) -> np.ndarray:
    v = np.ones(K, dtype=np.float64)
    hi = int(round(alpha * K))
    hi = max(1, min(K - 1, hi))
    v[:hi] = float(ratio)
    return v


def _generate_inits(K: int, lengths: np.ndarray, seed: int, parent):
    rng = np.random.default_rng(seed)
    K = int(len(lengths))
    inits = []

    # Robust asymmetric two-level seeds.
    two_step_cases = [
        (0.50, 2.0),
        (0.50, 1.8),
        (0.50, 2.4),
        (0.45, 2.4),
        (0.40, 2.8),
        (0.35, 3.2),
        (0.30, 4.0),
        (0.60, 1.6),
        (0.20, 6.0),
    ]
    for alpha, ratio in two_step_cases:
        v = _make_two_step(K, alpha, ratio)
        nv = _normalize_blocks(v, lengths)
        if nv is not None:
            inits.append(nv)

    idx = np.arange(K, dtype=np.float64)
    if K > 1:
        t = idx / float(K - 1)
    else:
        t = np.zeros(K, dtype=np.float64)

    # Smooth one-sided profiles.
    for a in (0.5, 0.8, 1.0, 1.3, 1.8, 2.5, 3.5, 5.0):
        v = np.exp(-a * t) + 1e-8
        nv = _normalize_blocks(v, lengths)
        if nv is not None:
            inits.append(nv)

    # Power-law decreasing profiles.
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        v = (1.0 - t) ** p + 1e-8
        nv = _normalize_blocks(v, lengths)
        if nv is not None:
            inits.append(nv)

    # Smooth random asymmetric profiles.
    for _ in range(5):
        amp = rng.uniform(0.25, 0.75)
        freq = int(rng.integers(1, 4))
        phase = rng.uniform(0.0, 2.0 * np.pi)
        trend = rng.uniform(-0.35, 0.35) * t
        v = 1.0 + amp * np.cos(2.0 * np.pi * freq * t + phase) + trend
        v = np.maximum(v, 0.05)
        nv = _normalize_blocks(v, lengths)
        if nv is not None:
            inits.append(nv)

    # Use the parent sequence if it matches the current target length.
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if p.size == int(np.sum(lengths)) and np.all(np.isfinite(p)):
            starts = _starts(lengths)
            v = np.zeros(K, dtype=np.float64)
            for j in range(K):
                a = int(starts[j])
                b = a + int(lengths[j])
                if b > a:
                    v[j] = float(np.mean(p[a:b]))
            nv = _normalize_blocks(v, lengths)
            if nv is not None:
                inits.append(nv)

    out = []
    for v in inits:
        if v is not None and np.all(np.isfinite(v)):
            out.append(v)
    return out


def _average_to_blocks(seq: np.ndarray, lengths: np.ndarray) -> np.ndarray:
    seq = np.asarray(seq, dtype=np.float64).ravel()
    starts = _starts(lengths)
    v = np.zeros(len(lengths), dtype=np.float64)
    L = int(seq.size)
    for j in range(len(lengths)):
        a = int(starts[j])
        b = int(starts[j] + lengths[j])
        if b > L:
            b = L
        if b > a:
            v[j] = float(np.mean(seq[a:b]))
        else:
            v[j] = 0.0
    return v


def _interp_profile(v_from, K_to: int) -> np.ndarray:
    v_from = np.asarray(v_from, dtype=np.float64).ravel()
    if v_from.size <= 1:
        return np.ones(K_to, dtype=np.float64)
    x_from = np.arange(v_from.size, dtype=np.float64)
    x_to = np.linspace(0.0, float(v_from.size - 1), int(K_to))
    return np.interp(x_to, x_from, v_from)


def _block_gradient_logs(v: np.ndarray, lengths: np.ndarray, gamma: float):
    f = _build_blocks(v, lengths)
    conv = _fft_conv_self(f)
    if conv.size == 0:
        return None, conv
    maxc = float(np.max(conv))
    if not np.isfinite(maxc) or maxc <= 0.0:
        return None, conv

    z = conv / maxc
    # Soft peak weights: w_k proportional to exp(-gamma * (1 - z_k)).
    w = np.exp(-float(gamma) * (1.0 - z))
    ws = float(w.sum())
    if not np.isfinite(ws) or ws <= 0.0:
        w = np.zeros_like(conv)
        max_idx = np.where(np.isclose(conv, maxc, rtol=1e-6, atol=0.0))[0]
        if max_idx.size > 0:
            w[max_idx] = 1.0 / float(max_idx.size)
        else:
            w[int(np.argmax(conv))] = 1.0
    else:
        w = w / ws

    n = int(f.size)
    # Need r[l] = sum_m f[m] w[l+m]. Use convolution with reversed w.
    N = _pow2(3 * n - 2)
    wr = w[::-1]
    F = np.fft.rfft(wr, N)
    G = np.fft.rfft(f, N)
    c = np.fft.irfft(F * G, N)[: 3 * n - 2]
    g = 2.0 * c[n - 1: 2 * n - 1][::-1]

    # g should be non-negative; clean numerical noise.
    g = np.asarray(g, dtype=np.float64)
    g[~np.isfinite(g)] = 0.0
    g = np.maximum(g, 0.0)

    starts = _starts(lengths)
    grad = np.empty(v.size, dtype=np.float64)
    for j in range(v.size):
        a = int(starts[j])
        b = a + int(lengths[j])
        s = float(g[a:b].sum())
        grad[j] = float(v[j]) * s
    return grad, conv


def _optimize_blocks(v_in, lengths: np.ndarray, deadline: float, seed: int):
    rng = np.random.default_rng(seed)
    K = int(len(lengths))

    v = _normalize_blocks(v_in, lengths)
    if v is None:
        v = np.full(K, 1.0 / float(np.sum(lengths)), dtype=np.float64)

    f = _build_blocks(v, lengths)
    current_score = _score_fft(f)
    best_v = v.copy()
    best_score = current_score

    logv = np.log(np.maximum(v, 1e-300))
    base_step = 1.0
    stale = 0
    it = 0

    while time.monotonic() < deadline:
        gamma = min(60.0, 3.0 + 0.05 * it)
        grad, conv = _block_gradient_logs(v, lengths, gamma)
        if grad is None:
            break

        pos = grad[grad > 0.0]
        if pos.size == 0:
            break
        scale = float(np.max(pos))
        if not np.isfinite(scale) or scale <= 0.0:
            break

        dirv = -grad / scale
        # Remove the mass-weighted mean so only relative changes are searched.
        dirv = dirv - float(np.dot(lengths, dirv)) / float(np.sum(lengths))

        cand_score = current_score
        cand_v = v
        for mult in (1.0, 0.5, 0.25, 0.1):
            st = base_step * mult
            if st <= 0.0:
                continue
            lv = logv + st * dirv
            lv = lv - float(np.max(lv))
            vn = np.exp(lv)
            vn = np.maximum(vn, 1e-15)
            total = float(np.dot(vn, lengths))
            if total <= 0.0:
                continue
            vn = vn / total
            sc = _score_fft(_build_blocks(vn, lengths))
            if np.isfinite(sc) and sc < cand_score - 1e-12:
                cand_score = sc
                cand_v = vn

        if cand_score < current_score - 1e-12:
            current_score = cand_score
            v = cand_v.copy()
            logv = np.log(v)
            if current_score < best_score - 1e-12:
                best_score = current_score
                best_v = v.copy()
            stale = 0
            base_step = min(3.0, base_step * 1.05)
        else:
            stale += 1
            base_step = max(0.02, base_step * 0.85)

            if stale > 60:
                noise = _smooth_noise(rng, K, 0.35)
                trend = 0.20 * rng.uniform(-1.0, 1.0) * np.linspace(-1.0, 1.0, K)
                lv = logv + 0.5 * (noise + trend)
                lv = lv - float(np.max(lv))
                vn = np.exp(lv)
                vn = np.maximum(vn, 1e-15)
                total = float(np.dot(vn, lengths))
                if total > 0.0:
                    vn = vn / total
                    sc = _score_fft(_build_blocks(vn, lengths))
                    if np.isfinite(sc) and sc < current_score + 0.10 * max(1.0, current_score):
                        v = vn
                        logv = np.log(v)
                        current_score = sc
                    else:
                        v = best_v.copy()
                        logv = np.log(v)
                        current_score = best_score
                else:
                    v = best_v.copy()
                    logv = np.log(v)
                    current_score = best_score
                base_step = 0.7
                stale = 0

        if stale > 400:
            break
        it += 1

    best_v = _normalize_blocks(best_v, lengths)
    if best_v is None:
        best_v = np.full(K, 1.0 / float(np.sum(lengths)), dtype=np.float64)
        best_score = _score_fft(_build_blocks(best_v, lengths))
    return best_v, float(best_score)


def _full_refine(f_in: np.ndarray, deadline: float, seed: int):
    f = np.asarray(f_in, dtype=np.float64).ravel().copy()
    n = int(f.size)
    if n == 0:
        return np.array([], dtype=np.float64), np.inf
    f = np.clip(f, 0.0, 1000.0)
    s = float(f.sum())
    if not np.isfinite(s) or s <= 0.0:
        f = np.ones(n, dtype=np.float64) / float(n)
    else:
        f = f / s

    best = f.copy()
    best_sc = _score_fft(best)
    if not np.isfinite(best_sc):
        return best, best_sc

    f_work = np.maximum(best, 1e-15)
    logf = np.log(f_work)
    rng = np.random.default_rng(seed)
    step_scale = 1.0
    stale = 0
    it = 0
    P = 8

    while time.monotonic() < deadline:
        improved = False
        phase = it % 6

        if phase < 3:
            # Peak shaving by soft active-set gradient.
            f = np.exp(logf - float(np.max(logf)))
            f = np.maximum(f, 1e-15)
            f = f / float(f.sum())

            conv = _fft_conv_self(f)
            if conv.size == 0:
                break
            maxc = float(np.max(conv))
            if not np.isfinite(maxc) or maxc <= 0.0:
                break

            count = min(P, conv.size)
            if count > 0:
                top = np.argpartition(conv, -count)[-count:]
                top = top[np.argsort(conv[top])[::-1]]

                gs = np.zeros(n, dtype=np.float64)
                for kk in top:
                    kk = int(kk)
                    ww = float(conv[kk]) / maxc
                    ist = max(0, kk - n + 1)
                    ied = min(n - 1, kk)
                    if ist <= ied:
                        ii = np.arange(ist, ied + 1)
                        gs[ii] += 2.0 * ww * f[kk - ii]

                smax = float(np.max(gs))
                if np.isfinite(smax) and smax > 0.0:
                    direction = -gs / smax
                    direction -= float(np.mean(direction))
                    for mult in (1.0, 0.5, 0.2, 0.1):
                        st = step_scale * mult
                        if st <= 0.0:
                            continue
                        lv = logf + st * direction
                        lv = lv - float(np.max(lv))
                        vn = np.exp(lv)
                        vn = np.maximum(vn, 1e-15)
                        vn = vn / float(vn.sum())
                        sc = _score_fft(vn)
                        if np.isfinite(sc) and sc < best_sc - 1e-12:
                            best_sc = sc
                            best = vn.copy()
                            logf = np.log(vn)
                            improved = True
                            break

        elif phase == 3:
            # Local multiplicative block perturbations.
            f = np.exp(logf - float(np.max(logf)))
            f = np.maximum(f, 1e-15)
            f = f / float(f.sum())

            cand = f.copy()
            nblocks = int(rng.integers(8, max(9, n // 250)))
            width = int(max(1, n // 350))
            for _ in range(nblocks):
                start = int(rng.integers(0, max(1, n - width + 1)))
                mult = float(np.exp(rng.normal(0.0, 0.15)))
                cand[start:start + width] *= mult
            cand = np.clip(cand, 1e-15, 1000.0)
            cand = cand / float(cand.sum())
            sc = _score_fft(cand)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best = cand.copy()
                logf = np.log(best)
                improved = True

        else:
            # Smooth global multiplicative noise.
            f = np.exp(logf - float(np.max(logf)))
            f = np.maximum(f, 1e-15)
            f = f / float(f.sum())

            noise = rng.normal(0.0, 0.06, n)
            if n > 3:
                noise = 0.6 * noise + 0.2 * np.roll(noise, 1) + 0.2 * np.roll(noise, -1)
            cand = f * np.exp(noise)
            cand = np.maximum(cand, 1e-15)
            cand = cand / float(cand.sum())
            sc = _score_fft(cand)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best = cand.copy()
                logf = np.log(best)
                improved = True

        if improved:
            stale = 0
            step_scale = min(2.5, step_scale * 1.02)
        else:
            stale += 1
            step_scale = max(0.05, step_scale * 0.90)
            if stale > 80:
                noise = _smooth_noise(rng, n, 0.22)
                lv = logf + 0.30 * noise
                lv = lv - float(np.max(lv))
                vn = np.exp(lv)
                vn = np.maximum(vn, 1e-15)
                vn = vn / float(vn.sum())
                sc = _score_fft(vn)
                if np.isfinite(sc) and sc < best_sc + 0.50 * max(1.0, best_sc):
                    logf = np.log(vn)
                    if sc < best_sc:
                        best_sc = sc
                        best = vn.copy()
                else:
                    logf = np.log(np.maximum(best, 1e-15))
                stale = 0

        if stale > 300:
            break
        it += 1

    best = np.clip(best, 0.0, 1000.0)
    s = float(best.sum())
    if s > 0.0:
        best = best / s
    return best, float(best_sc)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    start = time.monotonic()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 1500.0
    deadline = start + max(60.0, total_budget - 60.0)

    # Recover a parent sequence if available.
    parent = None
    try:
        p = height_sequence_1
        p = np.asarray(p, dtype=np.float64).ravel()
        if p.size > 0 and np.all(np.isfinite(p)):
            parent = p
    except Exception:
        parent = None

    if parent is not None and 100 <= parent.size <= 50000:
        n = int(parent.size)
    else:
        n = 5582

    # Fall back to the all-ones sequence, whose score is exactly 2.
    best_seq = None
    best_exact = np.inf

    if parent is not None and parent.size == n:
        sc = _exact_score(parent.tolist())
        if np.isfinite(sc) and sc < best_exact:
            best_exact = sc
            best_seq = parent.copy()

    ones = np.ones(n, dtype=np.float64) / float(n)
    if best_seq is None or best_exact > 2.0 + 1e-12:
        best_exact = min(best_exact if np.isfinite(best_exact) else np.inf, 2.0)
        best_seq = ones.copy()

    best_internal_seq = best_seq.copy()
    best_internal_score = _score_fft(best_internal_seq)

    rng = np.random.default_rng(seed)

    # Block-profile search. K is kept much smaller than n for speed.
    # The first two sizes are explored with several seeds; K=256 is used
    # if time remains to refine the best shape.
    block_plans = [
        (64, 2),
        (128, 3),
    ]

    for K, topc in block_plans:
        if time.monotonic() > deadline - 380.0:
            break
        K = min(int(K), max(2, n))
        lengths = _make_lengths(n, K)
        inits = _generate_inits(K, lengths, int(seed) + K, parent)

        scored = []
        for vv in inits:
            sc = _score_blocks(vv, lengths)
            if np.isfinite(sc):
                scored.append((float(sc), len(scored), vv))
        scored.sort()
        if not scored:
            continue

        for rank in range(min(topc, len(scored))):
            if time.monotonic() > deadline - 380.0:
                break
            sc0, _, v0 = scored[rank]
            rem = deadline - time.monotonic()
            tl = min(180.0, max(45.0, rem * 0.25))
            v_best, sc_best = _optimize_blocks(v0, lengths, time.monotonic() + tl, int(seed) + 100000 * K + rank)
            seq = _build_blocks(v_best, lengths)
            sc_int = _score_fft(seq)
            if np.isfinite(sc_int) and sc_int < best_internal_score - 1e-12:
                best_internal_score = sc_int
                best_internal_seq = seq.copy()
            sc_exact = _exact_score(seq.tolist())
            if np.isfinite(sc_exact) and sc_exact < best_exact - 1e-12:
                best_exact = sc_exact
                best_seq = seq.copy()
            print(f"block K={K} rank={rank} internal={sc_int:.6f} exact={sc_exact:.6f} best_exact={best_exact:.6f}")

    # Refine at a higher block resolution if time remains.
    if time.monotonic() < deadline - 250.0:
        K = min(256, max(2, n))
        lengths = _make_lengths(n, K)
        candidates = []

        if best_internal_seq is not None and best_internal_seq.size == n:
            vavg = _average_to_blocks(best_internal_seq, lengths)
            nv = _normalize_blocks(vavg, lengths)
            if nv is not None:
                candidates.append(nv)

            vnoise = _normalize_blocks(nv * np.exp(_smooth_noise(rng, K, 0.05)), lengths)
            if vnoise is not None:
                candidates.append(vnoise)

        inits = _generate_inits(K, lengths, int(seed) + 256, parent)
        # Keep only a few structured seeds for speed.
        for vv in inits[:3]:
            candidates.append(vv)

        scored = []
        for vv in candidates:
            sc = _score_blocks(vv, lengths)
            if np.isfinite(sc):
                scored.append((float(sc), len(scored), vv))
        scored.sort()

        for rank in range(min(2, len(scored))):
            if time.monotonic() > deadline - 120.0:
                break
            sc0, _, v0 = scored[rank]
            rem = deadline - time.monotonic()
            tl = min(220.0, max(60.0, rem * 0.45))
            v_best, sc_best = _optimize_blocks(v0, lengths, time.monotonic() + tl, int(seed) + 726372637 + rank)
            seq = _build_blocks(v_best, lengths)
            sc_int = _score_fft(seq)
            if np.isfinite(sc_int) and sc_int < best_internal_score - 1e-12:
                best_internal_score = sc_int
                best_internal_seq = seq.copy()
            sc_exact = _exact_score(seq.tolist())
            if np.isfinite(sc_exact) and sc_exact < best_exact - 1e-12:
                best_exact = sc_exact
                best_seq = seq.copy()
            print(f"block K={K} rank={rank} internal={sc_int:.6f} exact={sc_exact:.6f} best_exact={best_exact:.6f}")

    # Make sure the best internal candidate has been exactly scored.
    if best_internal_seq is not None and best_internal_seq.size == n:
        sc_exact = _exact_score(best_internal_seq.tolist())
        if np.isfinite(sc_exact) and sc_exact < best_exact - 1e-12:
            best_exact = sc_exact
            best_seq = best_internal_seq.copy()
        print(f"before full refine: exact={best_exact:.6f}")

    # Full-sequence polishing.
    if time.monotonic() < deadline - 35.0 and best_seq is not None:
        refined = _full_refine(best_seq, deadline - 10.0, int(seed) + 987654321)
        sc_int = _score_fft(refined)
        sc_exact = _exact_score(refined.tolist())
        if np.isfinite(sc_exact) and sc_exact < best_exact - 1e-12:
            best_exact = sc_exact
            best_seq = refined.copy()
        print(f"after full refine: internal={sc_int:.6f} exact={sc_exact:.6f} best_exact={best_exact:.6f}")

    if best_seq is None:
        best_seq = ones.copy()

    # Final sanitation.
    best_seq = np.asarray(best_seq, dtype=np.float64).ravel()
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s > 0.0:
        best_seq = best_seq / s

    print(f"final best_exact={best_exact:.6f} n={best_seq.size}")
    return [float(x) for x in best_seq]
```
