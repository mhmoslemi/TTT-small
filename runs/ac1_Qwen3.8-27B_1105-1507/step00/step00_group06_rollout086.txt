```python
"""
Search for low-score non-negative coefficient sequences.

The score is scale invariant:
    score(a) = 2 * n * max(convolve(a, a)) / sum(a)^2

We work with normalized vectors p with sum(p)=1 and minimize
    score(p) = 2 * n * max(convolve(p, p)).

The code uses:
  - fast FFT-based convolution,
  - a multiplicative minimax / active-set style iterative reducer,
  - several analytic and random positive initial shapes,
  - coarse-to-fine lengths (2048, 4096, and optionally 8192).
"""

import time
import numpy as np


def next_pow2(m):
    """Smallest power of two >= m."""
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def sanitize_nonnegative(arr):
    """Convert to 1D float, replace NaN/inf by 0, and clip below zero."""
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    return arr


def normalize_p(p):
    """Normalize a non-negative vector to sum 1."""
    arr = sanitize_nonnegative(p)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(max(1, arr.size), dtype=np.float64) / max(1, arr.size)
    arr = arr / s
    return arr


def quick_score(p):
    """FFT score for a normalized non-negative sequence."""
    p = normalize_p(p)
    n = p.size
    if n == 1:
        return 2.0
    N = next_pow2(2 * n - 1)
    P = np.fft.rfft(p, N)
    h = np.fft.irfft(P * P, N)[:2 * n - 1]
    h = np.maximum(h, 0.0)
    mx = float(np.max(h))
    if not np.isfinite(mx):
        return np.inf
    return 2.0 * float(n) * mx


def score_key(item):
    """Sorting key for (score, vector) tuples."""
    return item[0]


def make_uniform(n):
    return np.ones(n, dtype=np.float64) / float(n)


def make_sine2(n, amp):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    f = 1.0 + amp * np.sin(2.0 * np.pi * x)
    return normalize_p(np.maximum(f, 1e-12))


def make_sine4(n, amp):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    f = 1.0 + amp * np.sin(4.0 * np.pi * x)
    return normalize_p(np.maximum(f, 1e-12))


def make_mixed(n, a2, a4, a6):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    f = (
        1.0
        + a2 * np.sin(2.0 * np.pi * x)
        + a4 * np.sin(4.0 * np.pi * x)
        + a6 * np.sin(6.0 * np.pi * x)
    )
    f = np.maximum(f, 1e-8)
    return normalize_p(f)


def make_clipped_sine2(n, amp):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    f = np.maximum(0.0, 1.0 + amp * np.sin(2.0 * np.pi * x))
    if f.sum() <= 0:
        return make_uniform(n)
    return normalize_p(f)


def make_random_exp(n, rng, harmonics=7):
    """Random smooth positive shape, guaranteed positive."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    f = 1.0 + 0.35 * np.sin(2.0 * np.pi * x)
    for m in range(1, harmonics + 1):
        amp = rng.normal(0.0, 0.22)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        f = f + amp * np.sin(2.0 * np.pi * m * x + ph)
    f = np.maximum(f, 1e-7)
    return normalize_p(f)


def resample_p(p, n):
    """
    Rescale a normalized discrete shape p to length n while preserving its
    continuous-limit density profile.
    """
    p = normalize_p(p)
    m = p.size
    if m == n:
        return p.copy()
    if m <= 0:
        return make_uniform(n)
    f_old = float(m) * p
    x_old = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
    x_new = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    f_new = np.interp(x_new, x_old, f_old, left=f_old[0], right=f_old[-1])
    f_new = np.maximum(f_new, 1e-15)
    q = f_new / float(n)
    return normalize_p(q)


def build_pool(n, rng, extra_init=None):
    """
    Build a pool of initial normalized sequences for a given length.
    """
    pool = []
    pool.append(make_uniform(n))

    for a in (0.20, 0.30, 0.40, 0.50, 0.60, 0.70, 0.80, 0.90):
        pool.append(make_sine2(n, a))

    for a in (0.20, 0.30, 0.40, 0.50, 0.60, 0.70):
        pool.append(make_sine4(n, a))

    mixed_params = [
        (0.40, 0.00, 0.00),
        (0.30, 0.30, 0.00),
        (0.30, -0.30, 0.00),
        (0.20, 0.50, 0.20),
        (0.20, -0.50, -0.20),
        (0.50, 0.30, -0.10),
        (0.50, -0.30, 0.10),
        (0.10, 0.60, 0.30),
        (0.60, 0.20, -0.20),
        (0.40, 0.40, 0.00),
        (0.40, 0.00, 0.40),
        (0.40, -0.40, 0.00),
        (0.25, 0.55, -0.15),
        (0.35, -0.45, 0.25),
        (0.15, 0.40, 0.50),
        (0.55, -0.25, -0.35),
    ]
    for a2, a4, a6 in mixed_params:
        pool.append(make_mixed(n, a2, a4, a6))

    for a in (1.10, 1.30, 1.60, 2.00):
        pool.append(make_clipped_sine2(n, a))

    for _ in range(18):
        pool.append(make_random_exp(n, rng, harmonics=8))

    if extra_init is not None:
        arr = sanitize_nonnegative(extra_init)
        if arr.size > 0 and float(arr.sum()) > 0:
            pool.append(resample_p(arr, n))

    return pool


def optimize_sequence(n, init_p, rng, time_limit, tag=""):
    """
    Multiplicative minimax-style optimizer for a fixed length n.

    The update is based on the stationarity condition for minimizing max_k
    h_k with h = p * p and sum(p)=1.  We softly weight the active lags k
    and multiplicatively move p away from components that contribute to
    those active lags.
    """
    p = normalize_p(init_p)
    if p.size != n:
        p = resample_p(p, n)

    tiny = 1e-30 / float(n)
    N = next_pow2(3 * n - 2)
    start = int(2 * n - 2)
    stop = int(n - 2)

    best = p.copy()
    best_score = np.inf

    # Initial score
    P0 = np.fft.rfft(p, N)
    h0 = np.fft.irfft(P0 * P0, N)[:2 * n - 1]
    h0 = np.maximum(h0, 0.0)
    best_score = 2.0 * float(n) * float(np.max(h0))

    eta = 0.32
    step = 0
    no_improve = 0
    floor_log = np.log(max(1e-300, tiny))

    if tag:
        print(f"[optimize] started {tag} n={n} initial_score={best_score:.10f}")

    while time.monotonic() < time_limit:
        P = np.fft.rfft(p, N)
        h = np.fft.irfft(P * P, N)[:2 * n - 1]
        h = np.maximum(h, 0.0)
        maxh = float(np.max(h))
        if not np.isfinite(maxh):
            break
        score = 2.0 * float(n) * maxh

        improved = False
        if score < best_score - 1e-12:
            best_score = score
            best = p.copy()
            no_improve = 0
            improved = True
        else:
            no_improve += 1

        if step == 0 or step % 5000 == 0 or (improved and step % 2000 == 0):
            print(
                f"[optimize] {tag} iter={step:6d} score={score:.10f} "
                f"best={best_score:.10f}"
            )

        if no_improve > 250:
            eta = max(0.035, eta * 0.80)
            no_improve = 80

        if no_improve > 0 and no_improve % 2000 == 0:
            p = p * np.exp(rng.normal(0.0, 0.055, n))
            p = normalize_p(p)

        # Soft active-set weights.
        # Cycle through several sharpness levels to avoid sticking.
        m_exp = 16 + 16 * (step % 3)  # 16, 32, 48
        rel = h / maxh
        logrel = np.full_like(rel, -25.0, dtype=np.float64)
        pos = rel > 1e-14
        logrel[pos] = np.log(rel[pos])
        W = np.exp(m_exp * logrel)
        W = W + np.maximum(0.0, rel - 0.88)
        W = W + 1e-9
        wsum = float(W.sum())
        if not np.isfinite(wsum) or wsum <= 0.0:
            W = np.ones_like(h)
            wsum = float(W.size)
        W = W / wsum

        # q_i = sum_t W[i+t] p[t], i=0..n-1.
        WR = np.fft.rfft(W[::-1], N)
        q = np.fft.irfft(P * WR, N)[start:stop:-1]
        q = np.maximum(q, 1e-300)

        logq = np.log(q)
        center = float(np.mean(logq))
        delta = eta * (center - logq)
        delta = np.clip(delta, -3.0, 3.0)
        factor = np.exp(delta)

        p = p * factor
        p = np.maximum(p, tiny)
        s = float(p.sum())
        if not np.isfinite(s) or s <= 0.0:
            p = make_uniform(n)
        else:
            p = p / s

        # Keep log-scale sane after very rare extreme excursions.
        minp = float(np.min(p))
        if minp <= tiny * 1.0001:
            p = np.maximum(p, tiny)
            p = p / float(p.sum())

        step += 1

    if tag:
        print(f"[optimize] finished {tag} n={n} best_score={best_score:.10f}")

    return best, best_score


def run_stage(n, rng, top_k, per_sec, deadline, best_seq, best_score):
    """
    Run one length stage: build pool, quick-rank, then optimize top few.
    """
    local_candidates = []

    if time.monotonic() > deadline - 20.0:
        return best_seq, best_score, local_candidates

    pool = build_pool(n, rng, extra_init=best_seq)
    ranked = []
    for p in pool:
        p = normalize_p(p)
        sc = quick_score(p)
        if np.isfinite(sc):
            ranked.append((sc, p))

    ranked.sort(key=score_key)

    print(
        f"[stage] n={n} pool_size={len(ranked)} "
        f"best_pool_score={ranked[0][0] if ranked else float('inf'):.10f}"
    )

    for idx, (est_score, p_init) in enumerate(ranked[:top_k]):
        if time.monotonic() > deadline - 10.0:
            break
        trial_limit = min(time.monotonic() + float(per_sec), deadline - 8.0)
        if trial_limit < time.monotonic() + 1.0:
            continue

        tag = f"stage{n}_r{idx}"
        p_best, sc_best = optimize_sequence(
            n, p_init, rng, trial_limit, tag=tag
        )

        if np.isfinite(sc_best) and sc_best < best_score + 0.05:
            local_candidates.append((float(sc_best), np.asarray(p_best, dtype=np.float64)))

        if np.isfinite(sc_best) and sc_best < best_score:
            best_score = float(sc_best)
            best_seq = np.asarray(p_best, dtype=np.float64)
            print(f"[stage] new global best n={n} score={best_score:.10f}")

    print(f"[stage] done n={n} current_best={best_score:.10f}")
    return best_seq, best_score, local_candidates


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main entry point.  Running time is bounded by budget_s (default 1500s).
    Returns the best list of non-negative coefficients found.
    """
    total_budget = 1500.0
    if budget_s is not None:
        try:
            total_budget = float(budget_s)
        except Exception:
            total_budget = 1500.0
    total_budget = max(60.0, total_budget)

    start_time = time.monotonic()
    opt_deadline = start_time + max(30.0, total_budget - 35.0)
    exact_deadline = start_time + max(10.0, total_budget - 2.0)

    rng = np.random.default_rng(seed)

    # A safe non-uniform guard shape.  This is already better than the
    # all-ones parent score 2 in most moderate lengths.
    best_seq = make_sine2(2048, 0.50)
    best_score = quick_score(best_seq)
    candidates = [(float(best_score), best_seq.copy())]

    # If a parent construction exists, consider it too.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = None

    if parent is not None:
        parent = sanitize_nonnegative(parent)
        if parent.size > 0 and float(parent.sum()) > 0:
            ppar = normalize_p(parent)
            spar = quick_score(ppar)
            if np.isfinite(spar):
                candidates.append((float(spar), ppar.copy()))
                if spar < best_score:
                    best_score = float(spar)
                    best_seq = ppar.copy()

    print(f"[init] guard_score={best_score:.10f}")

    # Coarse stage: faster length to find useful shapes.
    if time.monotonic() < opt_deadline - 520.0:
        best_seq, best_score, new_cands = run_stage(
            n=2048,
            rng=rng,
            top_k=4,
            per_sec=28,
            deadline=opt_deadline,
            best_seq=best_seq,
            best_score=best_score,
        )
        if len(candidates) < 24:
            candidates.extend(new_cands)

    # Main stage.
    if time.monotonic() < opt_deadline - 260.0:
        best_seq, best_score, new_cands = run_stage(
            n=4096,
            rng=rng,
            top_k=6,
            per_sec=78,
            deadline=opt_deadline,
            best_seq=best_seq,
            best_score=best_score,
        )
        if len(candidates) < 24:
            candidates.extend(new_cands)

    # Fine stage if there is enough time.
    if time.monotonic() < opt_deadline - 160.0:
        best_seq, best_score, new_cands = run_stage(
            n=8192,
            rng=rng,
            top_k=3,
            per_sec=130,
            deadline=opt_deadline,
            best_seq=best_seq,
            best_score=best_score,
        )
        if len(candidates) < 24:
            candidates.extend(new_cands)

    # If time remains, do a few extra randomized top-shape refinements at 4096.
    extra_deadline = opt_deadline - 12.0
    extra_budget = extra_deadline - time.monotonic()
    if np.isfinite(extra_budget) and extra_budget > 90.0:
        n_extra = 4096
        pool = build_pool(n_extra, rng, extra_init=best_seq)
        ranked = []
        for p in pool:
            sc = quick_score(p)
            if np.isfinite(sc):
                ranked.append((sc, p))
        ranked.sort(key=score_key)
        per_extra = max(25.0, min(70.0, extra_budget / 3.0))
        for idx, (est_score, p_init) in enumerate(ranked[:3]):
            if time.monotonic() > extra_deadline - 8.0:
                break
            trial_limit = min(time.monotonic() + per_extra, extra_deadline - 6.0)
            if trial_limit < time.monotonic() + 1.0:
                continue
            p_best, sc_best = optimize_sequence(
                n_extra,
                p_init,
                rng,
                trial_limit,
                tag=f"extra4096_{idx}",
            )
            if np.isfinite(sc_best):
                candidates.append((float(sc_best), np.asarray(p_best, dtype=np.float64)))
                if sc_best < best_score:
                    best_score = float(sc_best)
                    best_seq = np.asarray(p_best, dtype=np.float64)
                    print(f"[extra] new global best score={best_score:.10f}")

    # Exact evaluation of a limited candidate set to choose the true best.
    exact_seq = [float(x) for x in np.asarray(best_seq, dtype=np.float64)]
    exact_score = evaluate_sequence(exact_seq)
    if not np.isfinite(exact_score):
        exact_seq = [float(x) for x in make_sine2(4096, 0.50)]
        exact_score = evaluate_sequence(exact_seq)

    # Evaluate more candidates if time remains.
    seen_sizes = set()
    for sc_est, arr in candidates:
        if time.monotonic() > exact_deadline:
            break
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size <= 0:
            continue
        key = arr.size
        if key in seen_sizes:
            pass  # still allow exact eval, but avoid too many duplicates
        sc_est = float(sc_est)
        # Skip candidates much worse than current exact best.
        if np.isfinite(exact_score) and sc_est > exact_score + 0.08:
            continue
        seq = [float(x) for x in arr]
        if time.monotonic() > exact_deadline:
            break
        sc_exact = evaluate_sequence(seq)
        if np.isfinite(sc_exact) and sc_exact < exact_score:
            exact_score = float(sc_exact)
            exact_seq = seq
            print(f"[exact] improved exact score to {exact_score:.10f} len={len(seq)}")
        seen_sizes.add(key)

    # Final fallback: if exact evaluation somehow failed, return a valid
    # non-uniform guard sequence.
    if not np.isfinite(exact_score) or exact_seq is None or len(exact_seq) == 0:
        exact_seq = [float(x) for x in make_sine2(4096, 0.50)]
        exact_score = evaluate_sequence(exact_seq)

    # Ensure the returned sequence is finite and non-negative.
    final = []
    for x in exact_seq:
        if np.isfinite(x) and x > 0.0:
            final.append(float(min(1000.0, x)))
        else:
            final.append(0.0)

    if not final or not all(np.isfinite(x) for x in final) or sum(final) <= 0:
        final = [float(x) for x in make_sine2(2048, 0.50)]

    print(f"[final] returned length={len(final)} exact_score={exact_score:.10f}")
    return final
```
