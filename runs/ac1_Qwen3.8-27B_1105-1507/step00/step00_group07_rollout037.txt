```python
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def build_conv_matrix(a):
    """Build the sparse convolution matrix A where A[k,i] = a[k-i] if valid."""
    n = len(a)
    diagonals = [np.full(n, a[d]) for d in range(n)]
    return sp.diags(diagonals, offsets=list(range(n)), shape=(2*n-1, n), format='csr')

def solve_lp(a):
    """Solve the LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    C = np.convolve(a, a)
    M = C.max()
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x, M
    return None, M

def score_normalized(a):
```

```python
def compute_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def solve_lp(a, rng, n_constraints=None, time_limit=60):
    """Solve the LP to find optimal g."""
    n = len(a)
    C = np.convolve(a, a)
    M = C.max()
    
    if n <= 4096:
        A = build_conv_matrix(a)
        indices = None
    else:
        # Subsampled LP
        if n_constraints is None:
            n_constraints = min(2*n - 1, 5000)
        # Pick the most important constraints (where C is largest)
        active_thresh = 0.3 * M
        important = np.where(C > active_thresh)[0]
        if len(important) >= n_constraints:
            indices = important[:n_constraints]
        else:
            rest = np.setdiff1d(np.arange(2*n-1), important)
            extra = rng.choice(rest, size=min(n_constraints - len(important), len(rest)), replace=False)
            indices = np.concatenate([important, extra])
        A = build_conv_matrix_sub2(a, indices)
    
    # Random bias to break degeneracy
    bias = rng.normal(0, 1e-4, n)
    c = -np.ones(n) - bias
    
    b = np.full(A.shape[0], M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return res.x, M
    return None, M
```
