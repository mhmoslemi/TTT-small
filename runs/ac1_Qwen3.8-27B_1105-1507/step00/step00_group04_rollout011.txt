from scipy.optimize import linprog

def solve_lp_for_g(f, M):
    """Solve: max sum(g) s.t. g >= 0, (f*g)_i <= M for all i"""
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)_i <= M for i = 0, ..., 2n-2
    # (f*g)_i = sum_j f_j * g_{i-j} where 0 <= j < n and 0 <= i-j < n
    # So for a given i, the valid j range is max(0, i-n+1) <= j <= min(i, n-1)
    
    A_ub = []
    b_ub = []
    for i in range(2 * n - 1):
        row = np.zeros(n)
        j_min = max(0, i - n + 1)
        j_max = min(i, n - 1)
        for j in range(j_min, j_max + 1):
            row[j] = f[i - j]  # (f*g)_i = sum_j f_{i-j} * g_j
        A_ub.append(row)
        b_ub.append(M)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds: g_j >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x if result.success else None

        j_max = min(k, n - 1)

"""
LP-based iterative optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
At each step, solve an LP to find direction g maximizing sum(g) subject to
convolve(f,g)_k <= max(convolve(f,f)) and g >= 0, then blend f toward g.
Tries multiple sequence lengths and initializations with adaptive step sizes.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    # Trim if we over-allocated
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_iterate(f, deadline, step_size_init=0.15, verbose_prefix=""):
    """Run LP-based iterations on f. Returns (best_f, best_score)."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    step_size = step_size_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        start_t = time.monotonic()
        g = solve_lp_direction(f, M)
        lp_time = time.monotonic() - start_t

        if g is None:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue

        # Rescale g to have same sum as f
        g = g * (float(np.sum(f)) / sum_g)

        # Blend
        new_f = (1.0 - step_size) * f + step_size * g
        new_f = np.clip(new_f, 0.0, 1000.0)

        new_score = evaluate_sequence(new_f.tolist())
        it += 1

        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            if it % 100 == 1:
                print(f"    {verbose_prefix}iter {it}: score={best_score:.6f} (lp={lp_time:.2f}s, step={step_size:.3f})")
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.003 or no_improve > 60:
                break

    return best_f, best_score

    Simple subgradient polish: at each step, compute which lag has max
