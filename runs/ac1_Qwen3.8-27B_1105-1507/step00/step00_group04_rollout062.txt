```python
    active_set = {n-1}  # position of max in conv(a,a)

        # Add the most violated constraint
        active_set.add(max(violated, key=lambda k: conv_ab[k]))
```

```python
"""
Step function optimization using LP-based Frank-Wolfe method.
Minimizes J = 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import linprog

def compute_conv(a):
    """Compute linear auto-convolution of a."""
    return np.convolve(a, a)

def objective(a):
    """Compute J = 2*n*max(c)/(sum)^2."""
    n = len(a)
    c = np.convolve(a, a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * np.max(c) / (s * s)

def solve_lp_direction(a, M, max_constraints=200):
    """
    Solve: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0.
    Uses active set method.
    """
    n = len(a)
    
    # Start with the most binding constraint
    conv_aa = np.convolve(a, a)
    active = {int(np.argmax(conv_aa))}
    
    b_best = None
    for iteration in range(max_constraints):
        # Build constraint matrix
        constraint_list = []
        for k in sorted(active):
            row = np.zeros(n)
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            for i in range(i_min, i_max + 1):
                row[k - i] += a[i]
            constraint_list.append(row)
        
        A_ub = np.array(constraint_list)
        b_ub = np.full(len(active), M)
        
        # Solve LP
        try:
            res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs',
                         options={'presolve': True})
        except Exception:
            break
        
        if not res.success:
            break
        
        b = res.x
        b_best = b
        
        # Check which constraints are violated
        conv_ab = np.convolve(a, b)
        violated = np.where(conv_ab > M + 1e-8)[0]
        
        if len(violated) == 0:
            return b
        
        # Add the most violated constraint
        new_k = int(violated[np.argmax(conv_ab[violated])])
        if new_k not in active:
            active.add(new_k)
        else:
            # All violated are already active; add another
            new_k = int(violated[0])
    
    return b_best

def frank_wolfe_step(a, t=0.5):
    """One step of the FP algorithm."""
    M = np.max(np.convolve(a, a))
    b = solve_lp_direction(a, M)
    if b is None:
        return a
```
