   if n <= 1:
      return 0.0, np.zeros(0)
   v=np.empty(n,dtype=np.float64)
   v[0]=0.0
   vr=np.asarray(vr,dtype=np.float64).ravel()

       m=min(vr.size,n-1); tmp[:m]=vr[:m]; vr=tmp

"""
Algorithms used:
- Softmax reparameterization of the nonnegative simplex.
- Smoothed-max objective (log-sum-exp of the full convolution) with FFT-based
  gradient.
- Deterministic line-search on the sorted normalized gradient direction, with
  stochastic jumps from the current best when stagnated.
- Short LBFGS-B warm starts on a reduced softmax parameterization.
- Random/block mutation polish near the end.
- Multiple sequence lengths are explored and the best internal candidate is
  verified with the authoritative evaluate_sequence().
"""

import time
import math
import numpy as np

try:
    from scipy.optimize import minimize
    _HAVE_SCIPY = True
except Exception:
    _HAVE_SCIPY = False

_LBFGS_N = 512
_LBFGS_TAU = 1e-3


def _pow2(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()


def _normalize(a):
    b = np.asarray(a, dtype=np.float64).ravel()
    if b.size == 0:
        return None
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.clip(b, 0.0, 1000.0)
    s = float(b.sum())
    if s <= 1e-15 or not math.isfinite(s):
        return None
    return b / s


def _conv_full(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a)
    m = _pow2(2 * n - 1)
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)[:2 * n - 1]
    c = np.where((c < 0.0) & (c > -1e-12), 0.0, c)
    return c


def _score_internal(a, n=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    if n is None:
        n = a.size
    else:
        n = int(n)
        if a.size != n:
            n = a.size
    s = float(a.sum())
    if s <= 1e-15 or not math.isfinite(s):
        return float("inf")
    c = _conv_full(a)
    if c.size == 0:
        return float("inf")
    cm = float(c.max())
    if not math.isfinite(cm):
        return float("inf")
    return float(2.0 * n * cm / (s * s))


def _grad_a_from_weights(w, a):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    if n == 1:
        return np.array([2.0 * float(w[0]) * float(a[0])], dtype=np.float64)
    N = _pow2(3 * n - 2)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    gh = np.fft.irfft(fw * np.conj(fa), N)[:n]
    return 2.0 * gh


def _grad_v_from_c(c, a, tau_frac):
    a = np.asarray(a, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    n = a.size
    if c.size == 0:
        return np.zeros(n, dtype=np.float64)
    cmax = float(c.max())
    if not math.isfinite(cmax) or cmax <= 0.0:
        return np.zeros(n, dtype=np.float64)
    tau = max(1e-15, float(tau_frac) * cmax)
    z = np.exp((c - cmax) / tau)
    s = float(z.sum())
    if s <= 1e-300 or not math.isfinite(s):
        w = np.zeros(c.shape, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = z / s
    ga = _grad_a_from_weights(w, a)
    dot = float(np.dot(a, ga))
    g = a * (ga - dot)
    g = np.where(np.isfinite(g), g, 0.0)
    return g


def _v_to_a(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size == 0:
        return np.empty(0, dtype=np.float64)
    v = np.where(np.isfinite(v), v, -1.0e6)
    v = v - float(np.max(v))
    e = np.exp(np.clip(v, -745.0, 0.0))
    s = float(e.sum())
    if s <= 1e-300 or not math.isfinite(s):
        return np.full(v.size, 1.0 / v.size, dtype=np.float64)
    return e / s


def _a_to_v(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.empty(0, dtype=np.float64)
    a = np.maximum(a, 1e-300)
    v = np.log(a)
    v = v - float(v[0])
    v = np.where(np.isfinite(v), v, -1.0e6)
    return v


def _resample(a, n):
    a = _normalize(a)
    n = int(n)
    if a is None:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if a.size == n:
        return a.copy()
    if a.size == 1:
        return np.full(n, 1.0 / n, dtype=np.float64)
    x_old = np.arange(a.size, dtype=np.float64)
    x_new = np.linspace(0.0, a.size - 1.0, n, dtype=np.float64)
    b = np.interp(x_new, x_old, a)
    b = b + 1e-12
    return _normalize(b)


def _block_initial(n, rng):
    n = int(n)
    a = np.zeros(n, dtype=np.float64)
    q = int(rng.integers(3, 9))
    for _ in range(q):
        low = max(1, n // (3 * q))
        high = max(low + 1, n // max(1, q))
        b = int(rng.integers(low, high))
        b = min(b, max(1, n - 1))
        if b <= 0:
            continue
        s = int(rng.integers(0, max(1, n - b)))
        w = float(rng.uniform(0.5, 2.0))
        a[s:s + b] += w
    a = np.maximum(a, 1e-3)
    if n > 5:
        kern = np.array([1.0, 1.8, 1.0], dtype=np.float64)
        a = np.convolve(a, kern, mode="same")
    a = _normalize(a)
    if a is None:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a


def _initial_a(n, kind, rng, parent=None):
    n = int(n)
    i = np.arange(n, dtype=np.float64)
    t = i / max(1, n - 1)
    kind = int(kind)

    if kind == 8 and parent is not None:
        a = _resample(parent, n)
    elif kind == 0:
        a = np.exp(rng.normal(0.0, 0.4, n))
    elif kind == 1:
        f = float(rng.uniform(0.15, 0.85))
        f2 = f + 0.111
        p1 = float(rng.uniform(0.0, 2.0 * math.pi))
        p2 = float(rng.uniform(0.0, 2.0 * math.pi))
        a = (
            1.0
            + 0.45 * np.sin(2.0 * math.pi * f * i + p1)
            + 0.25 * np.sin(2.0 * math.pi * f2 * i + p2)
        )
    elif kind == 2:
        f = 0.6180339887498949
        a = (
            1.0
            + 0.45 * np.sin(2.0 * math.pi * f * i + 0.7)
            + 0.25 * np.sin(2.0 * math.pi * (f + 0.251966) * i + 1.9)
        )
    elif kind == 3:
        alpha = float(rng.uniform(0.1, 0.9))
        beta = float(rng.uniform(0.2, 0.8))
        sigma = float(rng.uniform(0.05, 0.18))
        x = (alpha * i) % 1.0
        d = np.abs(x - beta)
        d = np.minimum(d, 1.0 - d)
        a = np.exp(-(d * d) / (2.0 * sigma * sigma)) + 0.05
    elif kind == 4:
        a = _block_initial(n, rng)
    elif kind == 5:
        a = 1.0 + 0.6 * t + 0.3 * np.sin(2.0 * math.pi * 3.0 * t + 0.5)
    elif kind == 6:
        a = 1.0 + 0.6 * (t - 0.5) * (t - 0.5) + 0.25 * np.sin(2.0 * math.pi * 0.7 * i)
    elif kind == 7:
        f = float(rng.uniform(0.3, 0.7))
        p = float(rng.uniform(0.0, 2.0 * math.pi))
        a = 1.0 + 0.45 * np.cos(2.0 * math.pi * f * i + p)
    elif kind == 9:
        a = np.ones(n, dtype=np.float64)
        amp = 0.22
        for _ in range(4):
            f = float(rng.uniform(0.1, 0.9))
            p = float(rng.uniform(0.0, 2.0 * math.pi))
            a += amp * np.sin(2.0 * math.pi * f * i + p)
    else:
        a = np.ones(n, dtype=np.float64)

    a = np.maximum(a, 1e-4)
    a = a * np.exp(rng.normal(0.0, 0.05, n))
    return _normalize(a)


def _add_candidate(best_state, saved, a, score):
    try:
        score = float(score)
    except Exception:
        return
    if not math.isfinite(score):
        return
    a = _normalize(a)
    if a is None:
        return
    if score < best_state["score"] - 1e-12:
        best_state["score"] = score
        best_state["a"] = a.copy()
        if len(saved) == 0 or score < saved[-1]["score"] - 1e-5:
            saved.append({"a": a.copy(), "score": score, "n": int(a.size)})
            if len(saved) > 12:
                saved.pop(0)


def _saved_score(item):
    return item["score"]


def _try_truncations(a, best_state, saved):
    a = _normalize(a)
    if a is None or a.size < 64:
        return
    fracs = (0.4, 0.5, 0.65, 0.75, 0.9)
    mlist = []
    for frac in fracs:
        m = int(a.size * frac)
        if m >= 32 and (not mlist or mlist[-1] != m):
            mlist.append(m)
    for m in mlist:
        if m >= a.size:
            continue
        starts = (0, max(0, (a.size - m) // 2))
        for st in starts:
            b = _normalize(a[st:st + m])
            if b is None:
                continue
            sc = _score_internal(b, b.size)
            _add_candidate(best_state, saved, b, sc)


def _run_lbfgs(a, n, tau, best_state, saved, max_iter=40, max_fun=700):
    if not _HAVE_SCIPY:
        return
    a = _normalize(a)
    if a is None:
        return
    n = int(a.size)
    if n <= 2:
        return

    v = _a_to_v(a)
    vr = v[1:] - v[0]

    global _LBFGS_N, _LBFGS_TAU
    _LBFGS_N = n
    _LBFGS_TAU = float(tau)

    def _lbfgs_fun(x):
        n_local = int(_LBFGS_N)
        if n_local <= 2:
            return 0.0, np.zeros(0, dtype=np.float64)
        vfull = np.empty(n_local, dtype=np.float64)
        vfull[0] = 0.0
        xl = np.asarray(x, dtype=np.float64).ravel()
        vfull[1:] = xl[:n_local - 1]
        vfull = np.where(np.isfinite(vfull), vfull, -1.0e6)
        aa = _v_to_a(vfull)
        c = _conv_full(aa)
        cmax = float(c.max()) if c.size else 0.0
        if not math.isfinite(cmax) or cmax <= 0.0:
            return 0.0, np.zeros(n_local - 1, dtype=np.float64)
        tau_local = max(1e-15, float(_LBFGS_TAU))
        z = np.exp((c - cmax) / tau_local)
        s = float(z.sum())
        if s <= 1e-300 or not math.isfinite(s):
            return cmax, np.zeros(n_local - 1, dtype=np.float64)
        F = cmax + tau_local * math.log(s)
        w = z / s
        ga = _grad_a_from_weights(w, aa)
        dot = float(np.dot(aa, ga))
        g = aa * (ga - dot)
        g = np.where(np.isfinite(g), g, 0.0)
        return float(F), g[1:].copy()

    try:
        res = minimize(
            _lbfgs_fun,
            vr,
            method="L-BFGS-B",
            options={
                "maxiter": int(max_iter),
                "maxfun": int(max_fun),
                "ftol": 1e-15,
                "gtol": 1e-10,
                "maxls": 20,
            },
        )
        x = res.x
        vfull = np.empty(n, dtype=np.float64)
        vfull[0] = 0.0
        vfull[1:] = np.asarray(x, dtype=np.float64).ravel()[:n - 1]
        a2 = _normalize(_v_to_a(vfull))
        if a2 is not None:
            sc = _score_internal(a2, n)
            _add_candidate(best_state, saved, a2, sc)
    except Exception:
        pass


def _run_custom(a, n, deadline, rng, best_state, saved, tau_init=0.15, tau_min=0.025):
    a = _normalize(a)
    if a is None:
        return
    n = int(a.size)
    deadline = float(deadline)

    v = _a_to_v(a)
    run_start = time.monotonic()
    step_base = min(2.0, max(0.1, 0.02 * math.sqrt(float(n))))
    step = step_base

    local_best_score = float("inf")
    local_best_a = None
    last_improve = run_start
    stag = 0
    it = 0

    a = _normalize(_v_to_a(v))
    score = _score_internal(a, n)
    _add_candidate(best_state, saved, a, score)
    if math.isfinite(score) and score < local_best_score:
        local_best_score = score
        local_best_a = a.copy()
        last_improve = run_start

    print(f"custom start n={n} score={score:.6f}", flush=True)
    imp_tol = 1e-10

    while time.monotonic() < deadline:
        now = time.monotonic()
        a = _normalize(_v_to_a(v))
        if a is None:
            v = np.zeros(n, dtype=np.float64)
            continue

        c = _conv_full(a)
        s = float(a.sum())
        cm = float(c.max()) if c.size else float("inf")
        if not math.isfinite(cm) or s <= 0.0:
            v = np.zeros(n, dtype=np.float64)
            continue
        current_score = float(2.0 * n * cm / (s * s))

        if math.isfinite(current_score):
            _add_candidate(best_state, saved, a, current_score)
            if current_score < local_best_score:
                local_best_score = current_score
                local_best_a = a.copy()
                last_improve = now
                stag = 0

        tau_frac = max(tau_min, tau_init * (0.9993 ** it))
        g = _grad_v_from_c(c, a, tau_frac)
        nrm2 = float(np.dot(g, g))

        if not math.isfinite(nrm2) or nrm2 < 1e-24:
            v = np.clip(v + rng.normal(0.0, 0.25, n), -100.0, 100.0)
            stag += 10
        else:
            nrm = math.sqrt(nrm2)
            if nrm > 0.0:
                g = g + rng.normal(0.0, 1e-5 * nrm, n)
                nrm2 = float(np.dot(g, g))
                if not math.isfinite(nrm2) or nrm2 < 1e-24:
                    v = np.clip(v + rng.normal(0.0, 0.2, n), -100.0, 100.0)
                    stag += 5
                else:
                    direction = -g / math.sqrt(nrm2)
                    best_lam = 0.0
                    best_score = current_score
                    best_v = v
                    lams = (step, step * 0.5, step * 0.25, step * 0.1)
                    for lam in lams:
                        v_new = np.clip(v + lam * direction, -100.0, 100.0)
                        a_new = _normalize(_v_to_a(v_new))
                        if a_new is None:
                            continue
                        sc = _score_internal(a_new, n)
                        if math.isfinite(sc) and sc < best_score - imp_tol:
                            best_score = sc
                            best_v = v_new
                            best_lam = lam

                    if best_lam != 0.0:
                        v = best_v
                        stag = 0
                        step = min(step_base * 2.0, step * 1.05)
                    else:
                        stag += 1
                        step = max(0.05, step * 0.8)
                        if stag > 80 or (now - last_improve) > 2.5:
                            if local_best_a is not None:
                                base = local_best_a
                            else:
                                base = a
                            elapsed = max(1e-9, now - run_start)
                            total = max(elapsed, deadline - run_start)
                            frac = min(1.0, max(0.0, elapsed / total))
                            jump_std = 0.35 + 0.55 * (1.0 - frac)
                            v = _a_to_v(base) + rng.normal(0.0, jump_std, n)
                            v = np.clip(v, -100.0, 100.0)
                            step = step_base
                            stag = 0
                        else:
                            v = v + lams[-1] * direction + rng.normal(0.0, 0.05, n)
                            v = np.clip(v, -100.0, 100.0)

        it += 1

    a = _normalize(_v_to_a(v))
    if a is not None:
        sc = _score_internal(a, n)
        _add_candidate(best_state, saved, a, sc)
    print(f"custom end n={n} local={local_best_score:.6f} global={best_state['score']:.6f}", flush=True)


def _block_search(a, n, deadline, rng, best_state, saved):
    a = _normalize(a)
    if a is None:
        return
    n = int(a.size)
    deadline = float(deadline)

    score = _score_internal(a, n)
    _add_candidate(best_state, saved, a, score)
    local_best_score = score
    local_best_a = a.copy()
    last_improve = time.monotonic()

    width_min = max(2, int(n * 0.002))
    width_max = max(width_min + 2, int(n * 0.25))
    print(f"block start n={n} score={score:.6f}", flush=True)

    while time.monotonic() < deadline:
        b = a.copy()
        w = int(rng.integers(width_min, max(width_min + 1, width_max)))
        st = int(rng.integers(0, max(1, n - w)))
        sig = float(rng.uniform(0.04, 0.28))
        mult = np.exp(rng.normal(0.0, sig, w))
        b[st:st + w] *= mult
        b = _normalize(b)
        if b is None:
            continue
        sc = _score_internal(b, n)
        if math.isfinite(sc) and sc < score - 1e-10:
            a = b
            score = sc
            local_best_score = sc
            local_best_a = b.copy()
            last_improve = time.monotonic()
            _add_candidate(best_state, saved, a, score)
        elif (time.monotonic() - last_improve) > 2.0:
            a = _normalize(local_best_a * np.exp(rng.normal(0.0, 0.18, n)))
            if a is not None:
                score = _score_internal(a, n)
                if math.isfinite(score) and score < local_best_score:
                    local_best_score = score
                    local_best_a = a.copy()
                last_improve = time.monotonic()

    a = _normalize(_v_to_a(_a_to_v(a)))
    if a is not None:
        sc = _score_internal(a, n)
        _add_candidate(best_state, saved, a, sc)
    print(f"block end n={n} best={best_state['score']:.6f}", flush=True)


def _get_parent():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        return None
    if p.size < 4:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 1e-15:
        return None
    return p / s


def _safe_evaluate(seq):
    try:
        vals = []
        for x in seq:
            vals.append(float(x))
        sc = evaluate_sequence(vals)
        if np.isfinite(sc):
            return float(sc)
    except Exception:
        pass
    return float("inf")


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    total = max(1.0, float(budget_s))
    start = time.monotonic()
    reserve = max(5.0, min(30.0, total * 0.05))
    final_reserve = max(5.0, min(90.0, total * 0.06))
    search_deadline = start + max(1.0, total - reserve)
    main_deadline = search_deadline - final_reserve
    if main_deadline <= start:
        main_deadline = start + max(1.0, (search_deadline - start) * 0.5)
    if main_deadline > search_deadline:
        main_deadline = search_deadline

    rng = np.random.default_rng(seed)
    parent = _get_parent()
    best_state = {"score": float("inf"), "a": None}
    saved = []

    uniform = np.ones(256, dtype=np.float64)
    _add_candidate(best_state, saved, uniform, _score_internal(uniform, 256))

    if parent is not None:
        for n in (256, 512):
            if time.monotonic() >= main_deadline:
                break
            a = _resample(parent, n)
            _add_candidate(best_state, saved, a, _score_internal(a, n))

    if main_deadline - time.monotonic() > 35.0:
        a256 = _initial_a(256, 0, rng, parent)
        end = min(main_deadline, time.monotonic() + 30.0)
        _block_search(a256, 256, end, rng, best_state, saved)

    if _HAVE_SCIPY and (main_deadline - time.monotonic()) > 45.0:
        lb_end = min(main_deadline, time.monotonic() + 45.0)
        for kind in (4, 1, 9):
            if time.monotonic() >= lb_end:
                break
            a = _initial_a(512, kind, rng, parent)
            _add_candidate(best_state, saved, a, _score_internal(a, 512))
            _run_lbfgs(a, 512, tau=0.12 / 512.0, best_state=best_state, saved=saved, max_iter=35, max_fun=550)
        if time.monotonic() < lb_end - 15.0 and best_state["a"] is not None:
            a = _resample(best_state["a"], 1024)
            a = _normalize(a * np.exp(rng.normal(0.0, 0.12, 1024)))
            if a is not None:
                _add_candidate(best_state, saved, a, _score_internal(a, 1024))
                _run_lbfgs(a, 1024, tau=0.10 / 1024.0, best_state=best_state, saved=saved, max_iter=25, max_fun=350)

    B = max(1.0, main_deadline - start)
    phase_defs = (
        (512, 0.07, 2),
        (1024, 0.15, 3),
        (2048, 0.22, 3),
        (4096, 0.33, 3),
        (8192, 1.00, 2),
    )

    for idx, (n, frac, starts) in enumerate(phase_defs):
        now = time.monotonic()
        if now >= main_deadline - 10.0:
            break
        if n >= 8192 and best_state["score"] < 1.52:
            continue

        if n >= 8192:
            phase_budget = max(0.0, main_deadline - now)
        else:
            alloc = B * frac
            phase_budget = min(alloc, max(0.0, main_deadline - now - 15.0))
        if phase_budget < 18.0:
            continue

        seeds = []
        if best_state["a"] is not None:
            a_seed = _resample(best_state["a"], n)
            a_seed = _normalize(a_seed * np.exp(rng.normal(0.0, 0.12, n)))
            if a_seed is not None:
                seeds.append(a_seed)
        if parent is not None:
            seeds.append(_initial_a(n, 8, rng, parent))
        for k in (0, 1, 4, 9, 3):
            seeds.append(_initial_a(n, k, rng, parent))

        clean = []
        for s in seeds:
            if s is not None:
                clean.append(s)
        seeds = clean
        if not seeds:
            seeds.append(np.ones(n, dtype=np.float64))
        if len(seeds) > starts:
            seeds = seeds[:starts]

        per_budget = phase_budget / max(1, len(seeds))
        for si, a_seed in enumerate(seeds):
            now = time.monotonic()
            if now >= main_deadline - 5.0:
                break
            if si == len(seeds) - 1:
                nb = main_deadline
            else:
                nb = min(main_deadline, now + per_budget)
            if nb - now < 8.0 and si != len(seeds) - 1:
                continue
            _run_custom(a_seed, n, nb, rng, best_state, saved)

    if best_state["a"] is not None and search_deadline - time.monotonic() > 25.0:
        fc_end = main_deadline + 0.35 * max(0.0, search_deadline - main_deadline)
        if fc_end > time.monotonic() + 10.0:
            _run_custom(best_state["a"], int(best_state["a"].size), fc_end, rng, best_state, saved)

    if best_state["a"] is not None and search_deadline - time.monotonic() > 8.0:
        _try_truncations(best_state["a"], best_state, saved)
        _block_search(best_state["a"], int(best_state["a"].size), search_deadline, rng, best_state, saved)

    eval_candidates = []
    if best_state["a"] is not None:
        eval_candidates.append(best_state["a"])
    for item in sorted(saved, key=_saved_score):
        if len(eval_candidates) >= 6:
            break
        a = item["a"]
        if best_state["a"] is not None and a.size == best_state["a"].size and abs(item["score"] - best_state["score"]) < 1e-9:
            continue
        eval_candidates.append(a)

    best_seq = None
    best_actual = float("inf")
    for seq in eval_candidates:
        if seq is None:
            continue
        if int(seq.size) > 20000:
            continue
        if time.monotonic() > start + total - 3.0:
            break
        sc = _safe_evaluate(seq)
        if math.isfinite(sc) and sc < best_actual:
            best_actual = sc
            best_seq = seq

    if best_seq is None:
        fallback = np.ones(512, dtype=np.float64)
        sc = _safe_evaluate(fallback)
        if math.isfinite(sc):
            best_seq = fallback
        else:
            best_seq = fallback

    out = []
    for x in best_seq:
        y = float(x)
        if not math.isfinite(y):
            y = 0.0
        if y < 0.0:
            y = 0.0
        if y > 1000.0:
            y = 1000.0
        out.append(y)

    s = sum(out)
    if s < 0.01 or not math.isfinite(s):
        out = [0.25] * 256

    return out
