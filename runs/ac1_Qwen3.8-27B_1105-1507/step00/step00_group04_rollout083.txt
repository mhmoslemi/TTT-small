```python
"""
Multi-phase spectral projective search for non-negative low-convolution sequences.

The objective is scale invariant.  We normalize sequences to sum 1 and use a
fast FFT score.  The search emphasizes anti-palindromic asymmetry, because
breaking the symmetry of the uniform starting point is what can reduce the
maximum full-convolution peak below the uniform value.  Gradient directions are
derived from a log-sum-exp smoothing of the peak, with pair-preserving antisymmetric
step projection to keep non-negativity and sum simple.  Multiple starts, beta
schedules, random kicks, and a final polishing phase are used.
"""

import time
import numpy as np


def _fft_len(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a[np.logical_not(np.isfinite(a))] = 0.0
    a[a < 0.0] = 0.0
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    m = _fft_len(L)
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)[:L]
    c[c < 0.0] = 0.0
    c[np.logical_not(np.isfinite(c))] = 0.0
    return c


def _fast_z_a_conv(z, a):
    a = np.asarray(a, dtype=np.float64)
    z = np.asarray(z, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    full_len = n + z.size - 1
    m = _fft_len(full_len)
    fa = np.fft.rfft(a, m)
    fz = np.fft.rfft(z[::-1], m)
    full = np.fft.irfft(fa * fz, m)
    return full[n - 1:2 * n - 1][::-1]


def _fast_score_c(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf"), np.array([], dtype=np.float64)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return float("inf"), np.zeros(2 * n - 1, dtype=np.float64)
    c = _fast_self_conv(a)
    if c.size == 0:
        return float("inf"), c
    mx = float(np.max(c))
    score = 2.0 * float(n) * mx / (s * s)
    return float(score), c


def _weights_from_c(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.array([], dtype=np.float64)
    n = (c.size + 1) // 2
    p = c * float(n)
    mx = float(np.max(p))
    if not np.isfinite(mx):
        return np.full(c.size, 1.0 / float(c.size), dtype=np.float64)
    z = np.exp(float(beta) * (p - mx))
    zsum = float(z.sum())
    if not np.isfinite(zsum) or zsum <= 0.0:
        z = (p == mx).astype(np.float64)
        zsum = float(z.sum())
        if zsum <= 0.0:
            z = np.full(c.size, 1.0 / float(c.size), dtype=np.float64)
            return z
    return z / zsum


def _gradient_from_weights(z, a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if z.size != 2 * n - 1:
        _, c = _fast_score_c(a)
        z = _weights_from_c(c, 10.0)
    cc = _fast_z_a_conv(z, a)
    g = 2.0 * cc
    g[np.logical_not(np.isfinite(g))] = 0.0
    return g


def _apply_antisym_step(a, d, step):
    a = np.asarray(a, dtype=np.float64)
    d = np.asarray(d, dtype=np.float64)
    n = a.size
    if d.size != n:
        d = np.zeros(n, dtype=np.float64)
    d = 0.5 * (d - d[::-1])
    d[np.logical_not(np.isfinite(d))] = 0.0
    cand = a + float(step) * d
    cand[np.logical_not(np.isfinite(cand))] = 0.0
    m = n // 2
    if m > 0:
        i = np.arange(m)
        j = n - 1 - i
        s_pair = a[i] + a[j]
        ci = np.clip(cand[i], 0.0, s_pair)
        cand[i] = ci
        cand[j] = s_pair - ci
    return cand


def _apply_direct_step(a, d, step):
    a = np.asarray(a, dtype=np.float64)
    d = np.asarray(d, dtype=np.float64)
    d[np.logical_not(np.isfinite(d))] = 0.0
    return _normalize_nonneg(a + float(step) * d)


def _random_antisym(n, rng):
    d = np.zeros(int(n), dtype=np.float64)
    m = int(n) // 2
    if m > 0:
        vals = rng.normal(0.0, 1.0, size=m)
        idx = np.arange(m)
        d[idx] = vals
        d[int(n) - 1 - idx] = -vals
    return d


def _random_antisym_start(n, rng):
    n = int(n)
    a = np.ones(n, dtype=np.float64) / float(n)
    m = n // 2
    if m <= 0:
        return a
    idx = np.arange(m)
    mid = float(m)
    shape = int(rng.integers(0, 4))
    t = float(rng.uniform(0.2, 1.05))
    sign = float(rng.choice((-1.0, 1.0)))
    if shape == 0:
        u = t * (1.0 - idx / mid)
    elif shape == 1:
        u = t * np.ones(m, dtype=np.float64)
    elif shape == 2:
        u = t * np.cos(np.pi * idx / (2.0 * mid))
    else:
        u = t * (1.0 - (idx / mid) * (idx / mid))
    u = np.clip(u, -1.0, 1.0)
    a[idx] = (1.0 + sign * u) / float(n)
    a[n - 1 - idx] = (1.0 - sign * u) / float(n)
    return _normalize_nonneg(a)


def _make_start_candidates(n, rng, parent=None):
    n = int(n)
    if n < 4:
        return [np.ones(4, dtype=np.float64) / 4.0]

    starts = []
    base = np.ones(n, dtype=np.float64) / float(n)
    starts.append(base.copy())

    if parent is not None:
        parent = np.asarray(parent, dtype=np.float64)
        parent = np.where(np.isfinite(parent), parent, 0.0)
        parent = np.maximum(parent, 0.0)
        if parent.size > 0 and parent.sum() > 0.0:
            if parent.size == n:
                starts.append(_normalize_nonneg(parent))
            else:
                x_old = np.linspace(0.0, 1.0, parent.size)
                x_new = np.linspace(0.0, 1.0, n)
                res = np.interp(x_new, x_old, parent)
                starts.append(_normalize_nonneg(res))

    m = n // 2
    if m > 0:
        idx = np.arange(m)
        mid = float(m)
        for shape in (0, 1, 2, 3):
            for sign in (1.0, -1.0):
                for t in (0.2, 0.4, 0.6, 0.8, 0.95, 1.1):
                    if shape == 0:
                        u = t * (1.0 - idx / mid)
                    elif shape == 1:
                        u = t * (1.0 - (idx / mid) * (idx / mid))
                    elif shape == 2:
                        u = t * np.cos(np.pi * idx / (2.0 * mid))
                    else:
                        u = t * np.ones(m, dtype=np.float64)
                    u = np.clip(u, -1.0, 1.0)
                    a = np.ones(n, dtype=np.float64) / float(n)
                    a[idx] = (1.0 + sign * u) / float(n)
                    a[n - 1 - idx] = (1.0 - sign * u) / float(n)
                    starts.append(_normalize_nonneg(a))

    for scale in (0.1, 0.3, 0.6, 1.0):
        a = np.ones(n, dtype=np.float64) / float(n)
        d = _random_antisym(n, rng) * (float(scale) / float(n))
        starts.append(_normalize_nonneg(a + d))

    for scale in (0.05, 0.15):
        a = np.ones(n, dtype=np.float64) / float(n)
        a = a + rng.normal(0.0, 1.0, size=n) * (float(scale) / float(n))
        starts.append(_normalize_nonneg(a))

    return starts


def _get_parent():
    try:
        obj = height_sequence_1
    except Exception:
        return None
    try:
        arr = np.asarray(obj, dtype=np.float64)
    except Exception:
        return None
    if arr.ndim != 1 or arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.sum() <= 0.0:
        return None
    return arr


def _exact_score(seq):
    try:
        val = evaluate_sequence([float(x) for x in seq])
        return float(val)
    except Exception:
        return float("inf")


def _optimize_direct(a_init, budget_s, rng, initial_beta=10.0, run_id=0):
    current = _normalize_nonneg(np.asarray(a_init, dtype=np.float64))
    n = current.size
    if n < 2:
        return current, float("inf")

    a_scale = 1.0 / float(n)
    current_score, c = _fast_score_c(current)
    best = current.copy()
    best_score = current_score
    if not np.isfinite(current_score):
        return best, best_score

    deadline = time.monotonic() + max(0.0, float(budget_s)) - 2.0
    beta = float(initial_beta)
    no_improve = 0
    it = 0

    factors = (0.25, 0.10, 0.04, 0.015, 0.005)
    kick_factors = (0.40, 0.15, 0.05, 0.02)

    while time.monotonic() < deadline:
        it += 1
        if not np.isfinite(current_score):
            break

        if it % 2000 == 0:
            print(f"iter run={run_id} it={it} cur={current_score:.6f} best={best_score:.6f} beta={beta:.2f}")

        z = _weights_from_c(c, beta)
        g = _gradient_from_weights(z, current)

        r = rng.random()
        if r < 0.80:
            asym = True
            d = -0.5 * (g - g[::-1])
        elif r < 0.90:
            asym = True
            d = _random_antisym(n, rng)
        elif r < 0.95:
            asym = False
            d = -0.5 * (g + g[::-1])
            d = d - float(d.mean())
        else:
            asym = False
            d = rng.normal(0.0, 1.0, size=n)
            d = d - float(d.mean())

        if d.size != n:
            d = _random_antisym(n, rng)
            asym = True
        d[np.logical_not(np.isfinite(d))] = 0.0

        dmax = float(np.max(np.abs(d))) if d.size else 0.0
        if not np.isfinite(dmax) or dmax <= 1e-300:
            if asym:
                d = _random_antisym(n, rng)
            else:
                d = rng.normal(0.0, 1.0, size=n)
                d = d - float(d.mean())
            dmax = float(np.max(np.abs(d))) if d.size else 0.0
            if not np.isfinite(dmax) or dmax <= 1e-300:
                no_improve += 1
                continue
        d = d / dmax

        accepted = False
        for factor in factors:
            step = float(factor) * a_scale
            if asym:
                cand = _apply_antisym_step(current, d, step)
            else:
                cand = _apply_direct_step(current, d, step)
            sc, cand_c = _fast_score_c(cand)
            if not np.isfinite(sc):
                continue
            if sc + 1e-12 < current_score:
                current = cand
                current_score = sc
                c = cand_c
                accepted = True
                break
            if no_improve > 25 and sc < current_score + 0.01 and rng.random() < 0.12:
                current = cand
                current_score = sc
                c = cand_c
                accepted = True
                break

        if accepted:
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            no_improve += 1
            if no_improve >= 35:
                kick_asym = rng.random() < 0.75
                if kick_asym:
                    kd = _random_antisym(n, rng)
                else:
                    kd = rng.normal(0.0, 1.0, size=n)
                    kd = kd - float(kd.mean())
                kd[np.logical_not(np.isfinite(kd))] = 0.0
                kmax = float(np.max(np.abs(kd))) if kd.size else 0.0
                if not np.isfinite(kmax) or kmax <= 1e-300:
                    kd = _random_antisym(n, rng)
                    kmax = float(np.max(np.abs(kd))) if kd.size else 0.0
                if np.isfinite(kmax) and kmax > 1e-300:
                    kd = kd / kmax

                kicked = False
                for factor in kick_factors:
                    step = float(factor) * a_scale
                    if kick_asym:
                        cand = _apply_antisym_step(current, kd, step)
                    else:
                        cand = _apply_direct_step(current, kd, step)
                    sc, cand_c = _fast_score_c(cand)
                    if np.isfinite(sc) and (sc < current_score or sc < best_score * 1.02):
                        current = cand
                        current_score = sc
                        c = cand_c
                        kicked = True
                        break

                if not kicked:
                    if rng.random() < 0.35:
                        current = _random_antisym_start(n, rng)
                    else:
                        rd = _random_antisym(n, rng)
                        rmax = float(np.max(np.abs(rd))) if rd.size else 0.0
                        if np.isfinite(rmax) and rmax > 1e-300:
                            rd = rd / rmax
                        step = (0.05 + 0.35 * float(rng.random())) * a_scale
                        current = _apply_antisym_step(best, rd, step)
                    current_score, c = _fast_score_c(current)

                no_improve = 0
                beta *= 1.6
                if beta > 250.0:
                    beta = 10.0

            if no_improve >= 800:
                rd = _random_antisym(n, rng)
                rmax = float(np.max(np.abs(rd))) if rd.size else 0.0
                if np.isfinite(rmax) and rmax > 1e-300:
                    rd = rd / rmax
                step = (0.20 + 0.80 * float(rng.random())) * a_scale
                current = _apply_antisym_step(best, rd, step)
                current_score, c = _fast_score_c(current)
                no_improve = 0
                beta = 10.0

    return best, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    t0 = time.monotonic()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 1500.0
    if not np.isfinite(total_budget) or total_budget <= 0.0:
        total_budget = 1500.0

    hard_deadline = t0 + total_budget - 20.0
    rng = np.random.default_rng(seed)

    parent = _get_parent()
    if parent is not None and 256 <= parent.size <= 5000:
        n_main = int(parent.size)
    else:
        n_main = 3809
    if n_main < 256:
        n_main = 512
    if n_main > 5000:
        n_main = 3809

    print(f"main length {n_main}, parent length {0 if parent is None else int(parent.size)}")

    starts = _make_start_candidates(n_main, rng, parent)
    scored = []
    for idx, a in enumerate(starts):
        sc, _ = _fast_score_c(a)
        if np.isfinite(sc):
            scored.append((sc, idx, a))
    if not scored:
        a = np.ones(n_main, dtype=np.float64) / float(n_main)
        scored = [(2.0, 0, a)]
    scored.sort()

    best_seq = None
    best_exact = float("inf")

    if parent is not None and parent.size <= 6000:
        es = _exact_score(parent)
        if np.isfinite(es):
            best_exact = es
            best_seq = parent.copy()
            print(f"parent exact {best_exact:.8f}")

    for rec in scored[:3]:
        if time.monotonic() >= hard_deadline - 30.0:
            break
        es = _exact_score(rec[2])
        if np.isfinite(es):
            if best_seq is None or es < best_exact:
                best_exact = es
                best_seq = rec[2].copy()
    if best_seq is None:
        best_seq = scored[0][2].copy()
        best_exact = float("inf")

    print(f"initial best exact {best_exact:.8f}")

    top_count = min(5, len(scored))
    reserve_final = 300.0 if (hard_deadline - time.monotonic()) > 900.0 else 120.0
    run_budget = max(0.0, hard_deadline - time.monotonic() - reserve_final - 30.0)
    weights = (0.35, 0.18, 0.14, 0.12, 0.10, 0.11)
    betas = (10.0, 20.0, 5.0, 30.0, 10.0)

    for k in range(top_count):
        if run_budget <= 5.0:
            break
        if time.monotonic() >= hard_deadline - reserve_final - 20.0:
            break
        b = float(run_budget) * float(weights[k % len(weights)])
        b = min(b, max(0.0, hard_deadline - time.monotonic() - reserve_final - 15.0))
        if b < 15.0:
            break
        a_run, s_run = _optimize_direct(scored[k][2], b, rng, betas[k % len(betas)], k + 1)
        print(f"main run {k + 1} fast {s_run:.8f}")
        es = _exact_score(a_run)
        if np.isfinite(es):
            print(f"main run {k + 1} exact {es:.8f}")
            if es < best_exact:
                best_exact = es
                best_seq = a_run.copy()

    if time.monotonic() < hard_deadline - 180.0:
        n2 = 2048 if n_main != 2048 else 1024
        if 256 <= n2 <= 5000:
            starts2 = _make_start_candidates(n2, rng, parent)
            scored2 = []
            for idx, a in enumerate(starts2):
                sc, _ = _fast_score_c(a)
                if np.isfinite(sc):
                    scored2.append((sc, idx, a))
            if scored2:
                scored2.sort()
                remaining_to_secondary = hard_deadline - time.monotonic() - 60.0
                b2 = min(320.0, max(0.0, remaining_to_secondary * 0.30))
                if b2 > 30.0:
                    print(f"secondary length {n2}, budget {b2:.1f}")
                    a_run2, s_run2 = _optimize_direct(scored2[0][2], b2, rng, 15.0, 500)
                    print(f"secondary fast {s_run2:.8f}")
                    es2 = _exact_score(a_run2)
                    if np.isfinite(es2):
                        print(f"secondary exact {es2:.8f}")
                        if es2 < best_exact:
                            best_exact = es2
                            best_seq = a_run2.copy()

    if time.monotonic() < hard_deadline - 20.0:
        final_budget = hard_deadline - time.monotonic() - 10.0
        if final_budget > 25.0 and best_seq is not None:
            final_arr = np.asarray(best_seq, dtype=np.float64)
            if final_arr.size <= 6000:
                final_arr, s_final = _optimize_direct(final_arr, final_budget, rng, 20.0, 1000)
                print(f"final fast {s_final:.8f}")
                es_final = _exact_score(final_arr)
                if np.isfinite(es_final):
                    print(f"final exact {es_final:.8f}")
                    if es_final < best_exact:
                        best_exact = es_final
                        best_seq = final_arr.copy()

    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64) / 256.0

    best_seq = np.asarray(best_seq, dtype=np.float64)
    if best_seq.size <= 6000:
        best_seq = _normalize_nonneg(best_seq)
        fin = _exact_score(best_seq)
        if not np.isfinite(fin):
            best_seq = np.ones(256, dtype=np.float64) / 256.0
            fin = _exact_score(best_seq)
        print(f"FINAL exact {fin:.10f}")
    else:
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        print("FINAL length too large for final exact check")

    out = [float(x) for x in best_seq]
    return out
```
