multiple non-uniform initializations, and periodic restarts to escape local minima.
"""

import numpy as np
import time


def _next_pow2(n):
    p = 1
    while p < n:
        p <<= 1
    return p


def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    N = _next_pow2(n)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n]


def _fft_pressure(a, W):
    n = len(a)
    m = len(W)
    conv_len = n + m - 1
    N = _next_pow2(conv_len)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    W_rev = W[::-1]
    W_pad = np.zeros(N)
    W_pad[:m] = W_rev
    result = np.fft.irfft(np.fft.rfft(a_pad) * np.fft.rfft(W_pad), N)[:conv_len]
    return result[n - 1:2 * n - 1][::-1]


def _optimize_normalized(a_init, deadline, eta_init=0.5, local_rng=None):
    a = a_init.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S
    n = len(a)
    if local_rng is None:
        local_rng = np.random.default_rng(0)

    best_a = a.copy()
    c = _fft_convolve(a, a)
    best_score = 2.0 * n * c.max()
    eta = eta_init
    no_improve = 0
    target_frac = 1.0

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        M = c.max()
        score = 2.0 * n * M

        if score < best_score * (1.0 - 1e-7):
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        theo_min = 1.0 / (2.0 * n - 1.0)
        target = theo_min + (1.0 - target_frac) * (M - theo_min)
        if no_improve > 200:
            target_frac = max(0.5, target_frac - 0.1)

        W = np.maximum(0.0, c - target)
        Wmax = W.max()
        if Wmax < 1e-14:
            target_frac = max(0.3, target_frac - 0.2)
            target = theo_min + (1.0 - target_frac) * (M - theo_min)
            W = np.maximum(0.0, c - target)
            Wmax = W.max()
            if Wmax < 1e-14:
                break

        g = _fft_pressure(a, W)
        grad = 4.0 * g

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-10:
            eta *= 0.5
            if eta < 1e-8:
                break
            continue
        a_new = a_new / S_new

        new_c = _fft_convolve(a_new, a_new)
        new_M = new_c.max()
        new_score = 2.0 * n * new_M

        if new_score < score:
            a = a_new
            eta = min(eta * 1.04, 50.0)
        else:
            eta *= 0.6
            if eta < 1e-8:
                if no_improve > 100:
                    kick_scale = 0.05 / n
                    a = a + kick_scale * local_rng.random(n)
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    eta = eta_init * 0.5
                    no_improve = 0
                else:
                    break

    return best_a, best_score
