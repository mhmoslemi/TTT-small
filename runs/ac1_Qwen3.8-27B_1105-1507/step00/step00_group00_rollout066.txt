def _fft_convolve_full(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1

    F = np.fft.rfft(f, size)
    h = np.fft.irfft(F * F, size)
    return h[:2 * n - 1]

def _score_raw(f):
    """Fast score computation assuming f >= 0, sum(f) > 0.01."""
    n = len(f)
    h = _fft_convolve_full(f)
    M = float(np.max(h))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def _normalize(f):
    """Normalize f to sum to 1, clip negatives. Returns None if degenerate."""
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-30:
        return None
    return f / s

def _pressure_vector(f, active_lags):
    """
    Compute p[j] = (1/|K|) * sum_{k in K} 2*f[k-j] for valid 0<=k-j<n.
    This is the average gradient component of max_k (f*f)_k.
    """
    n = len(f)
    p = np.zeros(n, dtype=np.float64)
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            lo = k - j_end
            hi = k - j_start
            p[j_start:j_end + 1] += 2.0 * f[lo:hi + 1][::-1]
    p /= float(len(active_lags))
    return p

def _mirror_descent(f_init, n_iter, eta_scale, threshold_frac, label=""):
    """
    Mirror descent with multiplicative updates for min 2n*max(conv(f,f)) s.t. sum(f)=1, f>=0.
    Uses adaptive step size eta = eta_scale / M and active-set subgradient.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)

    print(f"    MD start: n={n}, score={best_score:.6f}")

    for it in range(n_iter):
        h = _fft_convolve_full(f)
        M = float(np.max(h))
        if M <= 0:
            break

        threshold = threshold_frac * M + 1e-15
        active = np.where(h >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(h))], dtype=np.int64)

        p = _pressure_vector(f, active)

        eta = eta_scale / M
        log_ratio = -eta * p
        log_ratio = np.clip(log_ratio, -25.0, 25.0)

        f = f * np.exp(log_ratio)
        f_norm = _normalize(f)
        if f_norm is None:
            f = np.ones(n) / n
        else:
            f = f_norm

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        if (it + 1) % 1000 == 0:
            print(f"    MD it={it + 1}: best={best_score:.6f}, M={M:.8f}, |K|={len(active)}")

    return best_f, best_score

def _lp_frank_wolfe(f_init, max_iters, time_budget):
    """LP-based Frank-Wolfe for the normalized problem."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix

    f = np.asarray(f_init, dtype=np.float64).copy()
    f = _normalize(f)
    if f is None:
        return f_init.copy(), float('inf')
    n = len(f)
    best_f = f.copy()
    best_score = _score_raw(f)
    start_time = time.monotonic()

    print(f"    LP start: n={n}, score={best_score:.6f}")

    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break

        h = np.convolve(f, f)
        M = float(np.max(h))

        # Build sparse constraint matrix C: C[k, j] = f[k-j] if 0<=k-j<n
        row_idx = []
        col_idx = []
        vals = []
        for j in range(n):
            rows_j = np.arange(j, j + n)
            row_idx.extend(rows_j.tolist())
            col_idx.extend([j] * n)
            vals.extend(f.tolist())

        C = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))

        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=C, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')

        if not res.success or res.x is None:
            print(f"    LP failed at iter {it}: {res.message}")
            break

        g = res.x
        Sg = float(np.sum(g))
        if Sg < 1e-10:
            break
        g = g / Sg

        t = 2.0 / (it + 2)
        f_new = (1.0 - t) * f + t * g
        f_n = _normalize(f_new)
        if f_n is None:
            break
        f = f_n

        score = _score_raw(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

        if (it + 1) % 10 == 0:
            print(f"    LP it={it + 1}: score={best_score:.6f}")

    return best_f, best_score

def _cosine_init(n):
    j = np.arange(n, dtype=np.float64)
    f = np.cos(np.pi * (j + 0.5) / float(n)) ** 2
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _gaussian_init(n, sigma_frac):
    j = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    sigma = sigma_frac * float(n)
    f = np.exp(-j * j / (2.0 * sigma * sigma))
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _tapered_init(n, power):
    j = np.arange(n, dtype=np.float64)
    x = np.sin(np.pi * (j + 0.5) / float(n))
    f = x ** power
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s > 0:
        f /= s
    return f

def _additive_perturb(f, rng, frac):
    """Add a small random perturbation and re-normalize."""
    n = len(f)
    noise = rng.random(n, dtype=np.float64) * frac
    f_p = f + noise
    return _normalize(f_p)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for non-negative step function heights minimizing the evaluation score.
    Strategy: multi-start mirror descent with LP-based Frank-Wolfe refinement.
    """
    deadline = time.monotonic() + float(budget_s) - 5.0
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(1024, dtype=np.float64)

    best_f = parent.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print(f"Parent: n={len(best_f)}, score={best_score:.6f}")

    # Build candidate initializations
    candidates = []

    p = _normalize(parent)
    if p is not None:
        candidates.append(("parent", p))

    for n in [512, 1024, 2048, 4096]:
        candidates.append(("cos_%d" % n, _cosine_init(n)))

    for n in [1024, 2048]:
        for sf in [0.25, 0.35]:
            candidates.append(("gauss_%d_%d" % (n, int(sf * 100)), _gaussian_init(n, sf)))

    for n in [1024, 2048]:
        for pw in [1.0, 2.0, 3.0]:
            candidates.append(("tap_%d_%d" % (n, int(pw)), _tapered_init(n, pw)))

    for n in [1024, 2048]:
        f_r = rng.random(n, dtype=np.float64)
        f_rn = _normalize(f_r)
        if f_rn is not None:
            candidates.append(("rand_%d" % n, f_rn))

    print(f"Total candidates: {len(candidates)}")

    # Phase 1: Mirror descent on each candidate
    seqs = []
    scores = []
    names = []

    remaining = deadline - time.monotonic()
    time_per = max(8.0, min(45.0, remaining / max(1, len(candidates)) * 0.65))

    for idx, (name, f0) in enumerate(candidates):
        if time.monotonic() > deadline - 30:
            break

        n_it = int(max(500, min(4000, time_per * 400)))
        t0 = time.monotonic()

        f_md, s_md = _mirror_descent(f0, n_it, 0.5, 0.7, label=name)
        elapsed = time.monotonic() - t0
        print(f"  [{idx + 1}/{len(candidates)}] {name}: score={s_md:.6f} ({elapsed:.1f}s)")

        seqs.append(f_md)
        scores.append(s_md)
        names.append(name)

        if s_md < best_score:
            best_score = s_md
            best_f = f_md.copy()
            print(f"  ** New best: {best_score:.6f} from {name} **")

    # Phase 2: Refine top candidates
    # Sort by score (tuples of float sort correctly)
    order = []
    for i in range(len(scores)):
        order.append((scores[i], i))
    order.sort()

    top_k = min(4, len(order))
    for r in range(top_k):
        if time.monotonic() > deadline - 30:
            break
        s_cur, i_cur = order[r]
        name = names[i_cur]
        f_cur = seqs[i_cur]
        n = len(f_cur)
        print(f"\n  Refining {name} (n={n}, score={s_cur:.6f})")

        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            # More mirror descent with smaller eta
            n_it2 = int(min(5000, remaining * 80))
            f_ref, s_ref = _mirror_descent(f_cur, n_it2, 0.2, 0.85, label=name + "_ref")
            if s_ref < best_score:
                best_score = s_ref
                best_f = f_ref.copy()
                print(f"  ** New best: {best_score:.6f} **")
            f_cur = f_ref
            s_cur = s_ref

            # LP refinement for small n
            n_eff = len(f_cur)
            if n_eff <= 1200 and (deadline - time.monotonic()) > 25:
                lp_budget = min(60.0, deadline - time.monotonic() - 8)
                if lp_budget > 10:
                    f_lp, s_lp = _lp_frank_wolfe(f_cur, max_iters=25, time_budget=lp_budget)
                    if s_lp < best_score:
                        best_score = s_lp
                        best_f = f_lp.copy()
                        print(f"  ** New best from LP: {best_score:.6f} **")

            # Additive perturbation restart
            if (deadline - time.monotonic()) > 20:
                for pert_frac in [0.01, 0.05]:
                    f_pert = _additive_perturb(f_cur, rng, pert_frac)
                    if f_pert is None:
                        continue
                    n_it3 = int(min(3000, (deadline - time.monotonic() - 5) * 60))
                    f_p, s_p = _mirror_descent(f_pert, n_it3, 0.3, 0.8, label=name + "_pert")
                    if s_p < best_score:
                        best_score = s_p
                        best_f = f_p.copy()
                        print(f"  ** New best from perturb({pert_frac}): {best_score:.6f} **")

    # Phase 3: Try different threshold fractions on the current best
    if (deadline - time.monotonic()) > 20:
        n_it4 = int(min(3000, (deadline - time.monotonic() - 5) * 60))
        for tf in [0.5, 0.9, 0.95]:
            if (deadline - time.monotonic()) < 5:
                break
            f_t, s_t = _mirror_descent(best_f, n_it4, 0.15, tf, label="thresh_%d" % int(tf * 100))
            if s_t < best_score:
                best_score = s_t
                best_f = f_t.copy()
                print(f"  ** New best (thresh={tf}): {best_score:.6f} **")

    # Final verification
    final_list = [float(x) for x in best_f]
    final_score = float(evaluate_sequence(final_list))
    print(f"\nFinal verified score: {final_score:.6f} (n={len(final_list)})")

    if not np.isfinite(final_score) or final_score > 2.0:
        print("  Falling back to parent.")
        return [float(x) for x in parent]

    return final_list
