Minimize 2n * max(autocorrelation) over non-negative sequences normalized to sum=1.
Uses a hybrid of subgradient descent, LP-based corrections, and parametric initialization.

import numpy as np

from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_peak(p):
    """Compute max of full autocorrelation."""
    corr = np.convolve(p, p)
    return corr.max(), corr

def lp_solve(p, M, time_limit=10):
    """Solve the LP: max sum(g) s.t. (p*g)_k <= M, g >= 0."""
    n = len(p)
    c_obj = -np.ones(n)
    
    # Build sparse constraint matrix
    # A[k, j] = p[k-j] if 0 <= k-j <= n-1, else 0
    # Using the structure: column j has nonzeros at rows j, j+1, ..., j+n-1 (clamped)
    # A[j+offset, j] = p[offset] for offset = 0, ..., n-1 (when j+offset < 2n-1)
    
    indptr = np.zeros(n+1, dtype=np.int64)
    for j in range(n):
        maxlen = min(n, 2*n-1-j)
        indptr[j+1] = indptr[j] + maxlen
    
    data = np.empty(indptr[n])
    indices = np.empty(indptr[n], dtype=np.int64)
    
    for j in range(n):
        maxlen = min(n, 2*n-1-j)
        data[indptr[j]:indptr[j+1]] = p[:maxlen]
        indices[indptr[j]:indptr[j+1]] = np.arange(j, j+maxlen)
    
    A_ub = csr_matrix((data, indices, indptr), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                 bounds=[(0, None)]*n, method='highs',
                 options={'time_limit': time_limit, 'presolve': True})
    
    if res.success:
        return res.x
    return None

    """One subgradient descent step to minimize max autocorrelation."""

    M = corr.max()
    # Get active lags (within tolerance)
    tol = 1e-10
    active = np.where(corr >= M - tol)[0]
    
    # Subgradient: average gradient over active lags
    # grad[j] = (1/|K*|) * sum_{k in K*} 2*p[k-j] (for valid k-j)
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        # For j in [lo, hi-1]: k-j is in [k-hi+1, k-lo+1] = [max(0,k-n+1), min(n-1,k)]

        js = np.arange(lo, hi)
