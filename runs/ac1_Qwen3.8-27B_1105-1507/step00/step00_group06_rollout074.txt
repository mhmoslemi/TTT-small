        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except (NameError, TypeError, ValueError):
        parent = np.ones(5582, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(5582, dtype=np.float64)

    n = parent.size
    print(f"Sequence length: {n}")

    best_a = normalize_nonneg(parent.copy())
    best_score = float(evaluate_sequence(best_a.tolist()))
    print(f"Initial parent score: {best_score:.6f}")

    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)

    # Build diverse initializations
    inits = []

    # Parent
    inits.append(("parent", best_a.copy()))

    # Uniform
    inits.append(("uniform", np.ones(n) / n))

    # Random
    r = rng.random(n)
    inits.append(("random", r / r.sum()))

    # Bimodal (peaks at edges)
    bimodal = 1.0 + 2.0 * np.cos(np.pi * x / (n - 1))
    bimodal = np.clip(bimodal, 0.01, None)
    inits.append(("bimodal", bimodal / bimodal.sum()))

    # Unimodal (peak in center)
    sig = n / 4.0
    unimodal = 1.0 + 3.0 * np.exp(-0.5 * ((x - n / 2) / sig) ** 2)
    inits.append(("unimodal", unimodal / unimodal.sum()))

    # Cosine waves at different frequencies
    for freq in [2, 3, 5, 7]:
        cw = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * x / n)
        cw = np.clip(cw, 0.01, None)
        inits.append((f"cos{freq}", cw / cw.sum()))

    # Structured random (low-frequency random)
    for i in range(4):
        m = max(3, n // 80)
        base = 0.5 + rng.random(m)
        sm = np.interp(x, np.linspace(0, n - 1, m), base)
        sm = np.clip(sm, 0.05, None)
        inits.append((f"lowfreq{i}", sm / sm.sum()))

    # Alternating pattern
    alt = np.where(x % 2 == 0, 1.0, 0.3)
    inits.append(("alt", alt / alt.sum()))

    # Power-law
    pl = 1.0 + 0.5 * np.cos(2.0 * np.pi * 3.0 * x / n) + 0.3 * np.cos(2.0 * np.pi * 7.0 * x / n)
    pl = np.clip(pl, 0.05, None)
    inits.append(("multiwave", pl / pl.sum()))

    num_inits = len(inits)
    time_per_init = max(20.0, (budget_s - 80.0) / num_inits)
    time_per_init = min(time_per_init, 150.0)

    print(f"Running {num_inits} initializations, ~{time_per_init:.0f}s each")

    for idx, (name, init_a) in enumerate(inits):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        alloc = min(time_per_init, remaining)
        if alloc < 10.0:
            break

        print(f"[{idx + 1}/{num_inits}] '{name}', budget={alloc:.0f}s")
        a, score = optimize_gradient(init_a, n, alloc, seed + idx * 7919, verbose=True)
        if score < best_score:
            best_a = a.copy()
            best_score = score
            print(f"  *** NEW BEST: {best_score:.6f} (from '{name}')")
        time.sleep(0.01)

    # Final refinement phase from best found
    if time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 15.0:
            print(f"Final refinement phase, budget={remaining:.0f}s")
            a, score = optimize_gradient(best_a, n, remaining, seed + 54321, verbose=True)
            if score < best_score:
                best_a = a.copy()
                best_score = score
                print(f"  *** FINAL BEST: {best_score:.6f}")

    # Safety: ensure valid output
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    result = [float(v) for v in best_a]

    # Verify
    final_score = float(evaluate_sequence(result))
    if not np.isfinite(final_score) or final_score > best_score + 0.01:
        # Fallback to parent
        result = [float(max(0.0, min(1000.0, v))) for v in parent]
        best_score = float(evaluate_sequence(result))

    print(f"Final returned score: {best_score:.6f}")
    return result
