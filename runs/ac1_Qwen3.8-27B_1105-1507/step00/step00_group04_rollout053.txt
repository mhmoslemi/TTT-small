def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def fast_score(a):

        elif n <= 2048:
            weight = 1.5
        else:
            weight = 1.0
        config_times.append(weight * len(inits))

    total_weight = sum(config_times)

    for idx, ((n, inits), wt) in enumerate(zip(configs, config_times)):
        time_remaining = deadline - time.monotonic()
        if time_remaining < 5.0:
            break
        # Time for this config
        future_weight = sum(config_times[idx + 1:])
        if future_weight == 0:
            config_time = time_remaining * 0.9
        else:
            config_time = time_remaining * wt / (wt + future_weight) * 0.85
        config_time = max(3.0, min(config_time, 300.0))

        for init_type in inits:
            time_remaining = deadline - time.monotonic()
            if time_remaining < 3.0:
                break

            # Create initialization
            if init_type == 'parent' and parent_seq is not None:
                if len(parent_seq) == n:
                    a = parent_seq.copy()
                elif len(parent_seq) > n:
                    # Downsample
                    step = len(parent_seq) / n
                    indices = (np.arange(n) * step).astype(int)
                    indices = np.minimum(indices, len(parent_seq) - 1)
                    a = parent_seq[indices]
                else:
                    # Upsample
                    from numpy import interp
                    x_old = np.linspace(0, 1, len(parent_seq))
                    x_new = np.linspace(0, 1, n)
                    a = interp(x_new, x_old, parent_seq)
                a = np.maximum(a, 0)
            else:
                a = make_initialization(n, init_type, rng)

            a = a / a.sum()
            init_score = fast_score(a)

            # Time per init
            per_init_time = config_time / len(inits)
            per_init_time = min(per_init_time, time_remaining - 2.0)
            if per_init_time < 1.0:
                per_init_time = 1.0

            # Run gradient descent
            a_best, score = gradient_descent(a, per_init_time, rng, verbose=True)

            # Try symmetrization
            a_sym = symmetrize_and_check(a_best)
            sym_score = fast_score(a_sym)
            if sym_score < score:
                a_best = a_sym
                score = sym_score

            if score < best_score:
                best_score = score
                best_seq = a_best.copy()
                elapsed = time.monotonic() - t_global_start
                print(f"  [t={elapsed:.0f}s] n={n}, init={init_type}: score={best_score:.6f}")

        # LP refinement on current best after each config
        if best_seq is not None and best_score < 2.0:
            time_remaining = deadline - time.monotonic()
            if time_remaining > 5.0 and len(best_seq) <= 2500:
                lp_result = lp_refine_step(best_seq, max_time=min(3.0, time_remaining * 0.1))
                if lp_result is not None:
                    a_lp, score_lp = lp_result
                    if score_lp < best_score:
                        best_score = score_lp
                        best_seq = a_lp.copy()
                        print(f"  [t={time.monotonic()-t_global_start:.0f}s] LP refinement: score={best_score:.6f}")
                        # Continue GD from LP result
                        if time.monotonic() < deadline - 10:
                            a_post, score_post = gradient_descent(a_lp, min(30.0, deadline - time.monotonic() - 5), rng, verbose=False)
                            if score_post < best_score:
                                best_score = score_post
                                best_seq = a_post.copy()
                                print(f"  Post-LP GD: score={best_score:.6f}")

    # Final: try LP on the overall best one more time
    if best_seq is not None and len(best_seq) <= 2500 and time.monotonic() < deadline - 5:
        lp_result = lp_refine_step(best_seq, max_time=min(5.0, deadline - time.monotonic() - 2))
        if lp_result is not None:
            a_lp, score_lp = lp_result
            if score_lp < best_score:
                best_score = score_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: score={best_score:.6f}")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(256) / 256.0

    # Ensure valid output format
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(final_seq)
    print(f"  FINAL: n={len(final_seq)}, evaluate_sequence score={final_score:.6f}")

    if not np.isfinite(final_score):
        print("  Fallback to uniform-256")
        return [1.0] * 256

    return final_seq
