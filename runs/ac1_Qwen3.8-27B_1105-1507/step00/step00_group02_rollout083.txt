```python
"""
Searches for a low-score nonnegative sequence for

    score = 2 * n * max(a * a) / (sum(a))**2

where * is ordinary linear convolution.

The method combines:
1. a smooth low-order Fourier parametrization,
2. FFT-based fast scoring,
3. multi-restart local search over Fourier parameters,
4. a final light raw-coordinate local search in sequence space,
5. exact evaluator verification when available.

A strong starting family is 1 + a*sin(2*pi*x), whose optimal amplitude is
around 0.74-0.75.  The search refines this and optionally adds harmonics.
"""

import numpy as np
import time

K_HARMONICS = 4
SEARCH_N = 4096
FINAL_N = 8192


def _make_basis(n):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    c_list = []
    s_list = []
    for m in range(1, K_HARMONICS + 1):
        ang = 2.0 * np.pi * float(m) * x
        c_list.append(np.cos(ang))
        s_list.append(np.sin(ang))
    return c_list, s_list


_C_BASIS, _S_BASIS = _make_basis(SEARCH_N)


def _next_power_of_two(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _fft_convolve_self(a):
    n = int(a.size)
    size = 2 * n - 1
    fn = _next_power_of_two(size)
    fa = np.fft.rfft(a, fn)
    b = np.fft.irfft(fa * fa, fn)
    return b[:size]


def _score_array(a):
    """Fast FFT approximation to evaluate_sequence, with the same semantics."""
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return float("inf")
    if not np.all(np.isfinite(a)):
        return float("inf")

    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")

    n = int(a.size)
    b = _fft_convolve_self(a)
    mb = float(np.max(b))
    if mb < 0.0:
        mb = 0.0
    return float(2.0 * float(n) * mb / (s * s))


def _shape_score(params):
    """Score of the smooth Fourier shape at SEARCH_N."""
    params = np.asarray(params, dtype=np.float64)
    if params.size != 2 * K_HARMONICS:
        return float("inf")
    if not np.all(np.isfinite(params)):
        return float("inf")

    f = np.ones(SEARCH_N, dtype=np.float64)
    for k in range(K_HARMONICS):
        c = float(params[2 * k])
        s = float(params[2 * k + 1])
        if c != 0.0:
            f += c * _C_BASIS[k]
        if s != 0.0:
            f += s * _S_BASIS[k]

    if not np.all(np.isfinite(f)):
        return float("inf")

    mn = float(np.min(f))
    if mn < -1e-6:
        return float("inf")
    if mn < 0.0:
        f = np.maximum(f, 0.0)

    total = float(np.sum(f))
    if total < 0.01:
        return float("inf")

    return _score_array(f)


def _make_shape(n, params):
    """Construct a length-n nonnegative sequence from Fourier parameters."""
    params = np.asarray(params, dtype=np.float64)
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    f = np.ones(n, dtype=np.float64)

    for k in range(K_HARMONICS):
        c = float(params[2 * k])
        s = float(params[2 * k + 1])
        if c != 0.0:
            f += c * np.cos(2.0 * np.pi * float(k + 1) * x)
        if s != 0.0:
            f += s * np.sin(2.0 * np.pi * float(k + 1) * x)

    f = np.clip(f, 0.0, 1000.0)
    return f


def _random_params(rng, bounds):
    p = np.zeros(2 * K_HARMONICS, dtype=np.float64)
    for k in range(K_HARMONICS):
        b = float(bounds[k])
        p[2 * k] = float(rng.uniform(-b, b))
        p[2 * k + 1] = float(rng.uniform(-b, b))
    return p


def _param_local_search(p, score, stop_time, rng):
    """
    Low-dimensional local search over Fourier coefficients.
    One coefficient is perturbed at a time.
    """
    p = np.asarray(p, dtype=np.float64).copy()
    score = float(score)
    npar = int(p.size)
    step = 0.05

    for _ in range(1000000):
        if time.monotonic() >= stop_time:
            break

        cand = p.copy()
        dim = int(rng.integers(0, npar))

        if rng.random() < 0.75:
            delta = step * float(rng.normal())
        else:
            delta = step * (0.5 if rng.random() < 0.5 else -0.5)

        cand[dim] += delta
        sc = _shape_score(cand)

        if np.isfinite(sc) and sc < score - 1e-12:
            p = cand
            score = float(sc)
            step = min(0.25, step * 1.03)
        else:
            step = max(0.003, step * 0.98)

    return p, score


def _raw_local_search(a, score, stop_time, rng):
    """
    Light blockwise multiplicative coordinate search in the actual sequence.
    Used only as a final polish; the exact evaluator remains authoritative.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = int(a.size)
    score = float(score)

    if n < 16:
        return a, score

    width_min = max(8, n // 2048)
    width_max = max(width_min + 1, n // 32)
    sigma = 0.02

    for _ in range(1000000):
        if time.monotonic() >= stop_time:
            break

        cand = a.copy()
        w = int(rng.integers(width_min, width_max + 1))
        st = int(rng.integers(0, n - w))

        noise = np.exp(rng.normal(0.0, sigma, size=w))
        cand[st:st + w] *= noise
        cand = np.clip(cand, 0.0, 1000.0)

        sc = _score_array(cand)
        if np.isfinite(sc) and sc < score - 1e-12:
            a = cand
            score = float(sc)
            sigma = min(0.08, sigma * 1.02)
        else:
            sigma = max(0.003, sigma * 0.995)

    return a, score


def _safe_exact(seq_list):
    """Call the authoritative evaluator without allowing exceptions to stop search."""
    try:
        val = evaluate_sequence(seq_list)
        val = float(val)
        if np.isnan(val) or np.isinf(val):
            return float("inf")
        return val
    except Exception:
        return float("inf")


def _sort_candidates(cands):
    out = []
    for idx, item in enumerate(cands):
        sc, p = item
        if np.isfinite(sc):
            out.append((float(sc), int(idx), np.asarray(p, dtype=np.float64).copy()))
    out.sort()
    return out

        if np.isfinite(sc1):
            candidates.append((sc1, p1))
            if sc1 < scored[0][0]:
                scored = _sort_candidates(candidates)
                top = scored[:min(3, len(scored))]
                print(
                    f"param local restart {i}: fast={sc1:.10f}, "
                    f"elapsed={time.monotonic() - start:.1f}s"
                )

    scored = _sort_candidates(candidates)
    if scored:
        print(f"best Fourier parameter fast={scored[0][0]:.10f}")

    # Verify a few best parameter candidates with the exact evaluator.
    exact_stop = min(hard_return, start + max(5.0, total - 45.0))
    for i, item in enumerate(scored[:3]):
        if time.monotonic() >= exact_stop:
            break
        sc, _, p = item
        seq = _make_shape(FINAL_N, p)
        ex = _safe_exact(seq.tolist())
        if np.isfinite(ex) and ex < best_score:
            best_seq = seq
            best_score = ex
            print(f"exact param candidate {i}: score={ex:.10f}, n={FINAL_N}")
        elif np.isfinite(ex):
            print(f"exact param candidate {i}: score={ex:.10f} (not better)")

    # Optional raw polish.  It uses FFT scores internally, then verifies exact.
    if time.monotonic() < work_end:
        raw_start = best_seq.copy()
        raw_score = _score_array(raw_start)
        raw_stop = min(work_end, hard_return - max(5.0, min(15.0, 0.01 * total)))

        if time.monotonic() < raw_stop:
            raw_seq, raw_score = _raw_local_search(raw_start, raw_score, raw_stop, rng)
            ex = _safe_exact(raw_seq.tolist())
            if np.isfinite(ex) and ex < best_score:
                best_seq = raw_seq
                best_score = ex
                print(f"exact raw-polished score={ex:.10f}, n={raw_seq.size}")
            elif np.isfinite(ex):
                print(f"raw polished exact={ex:.10f} (not better), fast≈{raw_score:.10f}")

    print(f"FINAL n={best_seq.size}, exact_best={best_score:.10f}")
    return [float(x) for x in best_seq]
```
