    f = np.fft.rfft(a, N)

    n = len(a)

    N = 1 << (2 * n - 1).bit_length()
    f = np.fft.rfft(a, N)

def _gradient_at_k(a, k):
    """Compute gradient of b[k] w.r.t. a for full linear auto-convolution."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k - idx]
    return g

    if best_a is not None and time.monotonic() < total_deadline:
        remaining = total_deadline - time.monotonic()
        print(f"Phase 2: refining best (score={best_score:.6f}), time remaining={remaining:.1f}s")
        a_opt, _ = _optimize_sequence(best_a, rng, total_deadline, max_peaks=7)
        update_best(a_opt, "refined")

    # Phase 3: Try a few fresh starts with more time each
    if time.monotonic() < total_deadline - 60.0:
        for trial in range(3):
            if time.monotonic() > total_deadline - 30.0:
                break
            n = int(rng.choice([128, 256, 512]))
            a0 = rng.uniform(0.1, 1.0, n)
            a_opt, _ = _optimize_sequence(a0, rng, total_deadline - 30.0, max_peaks=5)
            update_best(a_opt, f"phase3_trial{trial}_n={n}")

    # Fallback
    if best_a is None:
        best_a = np.ones(128)
        print("Falling back to uniform sequence")
        best_score = evaluate_sequence(best_a.tolist())

    print(f"Final score: {best_score:.6f}, length: {len(best_a)}")
    return [float(x) for x in best_a]
