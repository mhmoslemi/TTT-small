import time
import math

def convolve_full(a):
    """Full convolution of a with itself."""
    return np.convolve(a, a, mode='full')

def compute_score_fast(a):
    """Compute score without calling evaluate_sequence."""
    c = np.convolve(a, a, mode='full')
    max_b = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * len(a) * max_b / (S * S)

def subgradient_update(a, lr, top_k=5):
    """Subgradient descent step using top-k peaks."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    S = np.sum(a)
    max_b = np.max(c)
    
    # Get top-k peak positions
    k = min(top_k, 2*n - 1)
    peak_indices = np.argsort(c)[-k:]
    
    # Compute subgradient (average over top-k peaks)
    g = np.zeros(n)
    for ki in peak_indices:
        # d(c[ki])/d(a[j]) = 2*a[ki-j] if 0 <= ki-j < n
        valid = np.arange(max(0, ki - n + 1), min(n, ki + 1))
        idx = ki - valid

    grad = 2.0 * n * g / (S * S) - 4.0 * n * max_b / (S * S * S)
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() > 0:
        a_new /= a_new.sum()
    return a_new, c, max_b

So the maximum of d is at the largest k where the formula holds, which is around k ≈ n (where it transitions to the "center" behavior). At k = n-1: d[n-1] = c[n-1] ≈ n^5/30.
