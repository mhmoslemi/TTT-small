import time

def compute_greedy(n):
    """Compute the greedy non-increasing sequence with f(0)=1, (f*f)(k)<=1 for k<n."""
    f = np.zeros(n)
    f[0] = 1.0
    for k in range(1, n):
        if k >= 2:
            interference = np.dot(f[1:k], f[k-1:0:-1])
        else:
            interference = 0.0
        f[k] = max(0.0, (1.0 - interference) / 2.0)
    return f

def compute_conv_max(f):
    """Compute max of full linear convolution."""
    conv = np.convolve(f, f)
    return conv.max(), conv

def subgradient_step(f, eta):
    """
    One subgradient step to reduce max convolution.
    Returns updated f and the max value.
    """
    n = len(f)
    conv = np.convolve(f, f)
    max_val = conv.max()
    # Get all positions where max is achieved (for robustness)
    max_indices = np.where(conv >= max_val * (1 - 1e-12))[0]
    n_max = len(max_indices)
    
    # Subgradient: for each i, sum over k in max_indices of 2*f(k-i) if valid
    subgrad = np.zeros(n)
    for k in max_indices:
        i_range = np.arange(n)
        j = k - i_range
        valid = (j >= 0) & (j < n)
        subgrad[valid] += 2.0 * f[j[valid]]
    subgrad /= n_max
    
    # Normalized step
    f_norm = f / f.sum()
    step = eta * f_norm  # scale step by typical magnitude
    f_new = f_norm - step * (subgrad / (subgrad.max() + 1e-30))
    f_new = np.clip(f_new, 0.0, None)
    if f_new.sum() > 1e-15:
        f_new = f_new / f_new.sum()
    else:
        f_new = f_norm.copy()
    return f_new, max_val

def multiplicative_update(f, eta):
    """
    Multiplicative (exponentiated gradient) update.
    Reduces elements that contribute to the max convolution.
    """
    n = len(f)
    conv = np.convolve(f, f)
    max_val = conv.max()
    max_indices = np.where(conv >= max_val * (1 - 1e-12))[0]
    
    # Compute "pressure" on each element
    pressure = np.zeros(n)
    for k in max_indices:
        i_range = np.arange(n)
        j = k - i_range
        valid = (j >= 0) & (j < n)
        pressure[valid] += f[j[valid]]
    pressure /= len(max_indices)

    for j in range(n):
        k_min = 0 if j > 0 else 0
        k_max = min(2*n-2, j + n - 1)
        # A[k, j] = f[k-j] for k in [max(0,j), min(2n-2, j+n-1)]
        valid_k = np.arange(max(0, j), min(2*n-1, j+n))
        valid_f = valid_k - j
        A[valid_k, j] = f[valid_f]
    
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL)
    except:
        try:
            prob.solve(solver=cp.ECOS)
        except:
            return None
    
    elapsed = time.monotonic() - start_time
    if elapsed > max_time:
        return None
    
    if b.value is None:
        return None
    
    b_val = np.clip(b.value, 0, 1000.0)
    if b_val.sum() < 1e-10:
        return None
    
    # Rescale to match sum of f
    b_val = b_val * (f.sum() / b_val.sum())
    return b_val

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Greedy initialization for multiple n values
    Phase 2: Subgradient/multiplicative descent
    Phase 3: Simulated annealing with diverse perturbations
    Phase 4: LP-based refinement (if feasible)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    
    # Get parent sequence if available
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("Invalid parent")
    except:
        parent = np.ones(256, dtype=np.float64)
    
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial: n={len(best)}, score={best_score:.6f}", flush=True)
    
    # Phase 1: Try greedy sequences for various n
    greedy_candidates = []
    n_values = [256, 512, 768, 1024, 1536, 2048, 3072, 4096, 6144, 8192]
    
    for n_try in n_values:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        f_greedy = compute_greedy(n_try)
        total = f_greedy.sum()
        if total < 0.01:
            continue
        # Normalize to sum = sqrt(2n)
        f_greedy = f_greedy * (np.sqrt(2*n_try) / total)
        score = float(evaluate_sequence(f_greedy.tolist()))
        greedy_candidates.append((score, n_try, f_greedy.copy()))
        if np.isfinite(score) and score < best_score:
            best = f_greedy.copy()
            best_score = score
            print(f"Phase 1 - Greedy n={n_try}: score={best_score:.6f} (sum={total:.2f}, took {time.monotonic()-t0:.1f}s)", flush=True)
    
    # Sort greedy candidates by score
    greedy_candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best greedy: score={greedy_candidates[0][0]:.6f}, n={greedy_candidates[0][1]}", flush=True)
    
    # Phase 2: Subgradient/multiplicative descent on best candidate
    if best_score > 1.4 and time.monotonic() < deadline:
        n_best = len(best)
        f = best.copy()
        f_norm = f / f.sum()  # work with sum=1 for numerical stability
        
        # Adaptive learning rate
        eta = 0.5
        no_improve = 0
        phase2_best = f_norm.copy()
        phase2_best_score = best_score
        
        for it in range(2000):
            if time.monotonic() > deadline:
                break
            
            # Alternate between subgradient and multiplicative updates
            if it % 3 == 0:
                f_new, max_val = subgradient_step(f_norm, eta)
            elif it % 3 == 1:
                f_new = multiplicative_update(f_norm, eta)
                _, max_val = compute_conv_max(f_new)
            else:
                f_new = multiplicative_update(f_norm, eta * 0.5)
                _, max_val = compute_conv_max(f_new)
            
            # Evaluate
            f_scaled = f_new * np.sqrt(2 * n_best)
            score = float(evaluate_sequence(f_scaled.tolist()))
            
            if np.isfinite(score) and score < phase2_best_score:
                phase2_best = f_new.copy()
                phase2_best_score = score
                no_improve = 0
                if it % 200 == 0:
                    print(f"Phase 2 iter {it}: score={phase2_best_score:.6f}", flush=True)
            else:
                no_improve += 1
            
            f_norm = f_new
            
            # Adaptive eta
            if no_improve > 100:
                eta *= 0.7
                no_improve = 0
            elif no_improve == 50 and it > 100:
                eta = min(eta * 1.5, 2.0)
        
        if phase2_best_score < best_score:
            best = phase2_best * np.sqrt(2 * n_best)
            best_score = phase2_best_score
            print(f"Phase 2 done: score={best_score:.6f}", flush=True)
    
    # Phase 3: Simulated annealing
    if time.monotonic() < deadline:
        n_sa = len(best)
        f_sa = best / best.sum()  # normalize to sum=1
        sa_score = best_score
        T = 0.005
        T_min = 1e-9
        cooling = 0.9995
        
        for it in range(100000):
            if time.monotonic() > deadline:
                break
            
            # Choose perturbation type
            u = rng.random()
            if u < 0.3:
                candidate = block_perturb(f_sa, rng, n_blocks=rng.integers(1, 4), width_frac=rng.uniform(0.02, 0.15))
            elif u < 0.5:
                candidate = smooth_perturb(f_sa, rng, sigma=rng.uniform(0.01, 0.05))
            elif u < 0.7:
                # Tail scaling
                split = int(rng.integers(n_sa//4, 3*n_sa//4))
                candidate = f_sa.copy()
                candidate[split:] *= float(np.exp(rng.normal(0, 0.2)))
                candidate = np.clip(candidate, 0, None)
            elif u < 0.8:
                # Head scaling  
                split = int(rng.integers(1, n_sa//3))
                candidate = f_sa.copy()
                candidate[:split] *= float(np.exp(rng.normal(0, 0.1)))
                candidate = np.clip(candidate, 0, None)
            else:
                # Random element perturbation
                candidate = f_sa.copy()
                idx = rng.integers(0, n_sa)
                candidate[idx] *= float(np.exp(rng.normal(0, 0.3)))
                candidate[idx] = max(0, candidate[idx])
            
            if candidate.sum() < 1e-10:
                continue
            
            candidate = candidate / candidate.sum()
            candidate_scaled = candidate * np.sqrt(2 * n_sa)
            new_score = float(evaluate_sequence(candidate_scaled.tolist()))
            
            if not np.isfinite(new_score):
                continue
            
            delta = new_score - sa_score
            if delta < 0 or rng.random() < np.exp(-delta / T):
                f_sa = candidate
                sa_score = new_score
                if sa_score < best_score:
                    best = candidate * np.sqrt(2 * n_sa)
                    best_score = sa_score
                    if it % 500 == 0:
                        print(f"Phase 3 iter {it}: score={best_score:.6f}, T={T:.2e}", flush=True)
            
            T = max(T * cooling, T_min)
        
        print(f"Phase 3 done: score={best_score:.6f}", flush=True)
    
    # Phase 4: Try LP-based update on a moderate-sized sequence
    if time.monotonic() < deadline and best_score > 1.45:
        # Use a moderate n for LP
        n_lp = min(len(best), 2000)
        f_lp = best.copy()
        if len(f_lp) > n_lp:
            f_lp = f_lp[:n_lp]
        
        # Normalize
        total = f_lp.sum()
        f_lp_norm = f_lp / total
        
        conv = np.convolve(f_lp_norm, f_lp_norm)
        M = conv.max()
        
        # Try LP
        b_new = try_lp_update(f_lp_norm, M * 1.001, max_time=60)
        if b_new is not None:
            b_scaled = b_new * np.sqrt(2 * len(b_new))
            score = float(evaluate_sequence(b_scaled.tolist()))
            if np.isfinite(score) and score < best_score:
                best = b_scaled
                best_score = score
                print(f"Phase 4 LP: score={best_score:.6f}", flush=True)
        
        # Also try LP starting from greedy
        f_greedy_lp = compute_greedy(n_lp)
        total_g = f_greedy_lp.sum()
        if total_g > 0.01:
            f_greedy_norm = f_greedy_lp / total_g
            conv_g = np.convolve(f_greedy_norm, f_greedy_norm)
            M_g = conv_g.max()
            b_g = try_lp_update(f_greedy_norm, M_g * 1.001, max_time=60)
            if b_g is not None:
                b_g_scaled = b_g * np.sqrt(2 * len(b_g))
                score = float(evaluate_sequence(b_g_scaled.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = b_g_scaled
                    best_score = score
                    print(f"Phase 4 LP (greedy start): score={best_score:.6f}", flush=True)
    
    # Final: try a few more n values with improved initializations
    if time.monotonic() < deadline:
        # Use the best found pattern to initialize new lengths
        for n_extra in [1500, 2500, 3500, 5000, 7000]:
            if time.monotonic() > deadline:
                break
            if n_extra == len(best):
                continue
            f_new = compute_greedy(n_extra)
            total = f_new.sum()
            if total < 0.01:
                continue
            f_new = f_new * (np.sqrt(2 * n_extra) / total)
            score = float(evaluate_sequence(f_new.tolist()))
            if np.isfinite(score) and score < best_score:
                best = f_new
                best_score = score
                print(f"Final - Greedy n={n_extra}: score={best_score:.6f}", flush=True)
            
            # Quick SA on this
            if np.isfinite(score) and score < best_score + 0.1:
                f_sa2 = f_new / f_new.sum()
                sa2_score = score
                T2 = 0.003
                for it2 in range(5000):
                    if time.monotonic() > deadline:
                        break
                    candidate = block_perturb(f_sa2, rng, n_blocks=2, width_frac=0.05)
                    if candidate.sum() < 1e-10:
                        continue
                    candidate = candidate / candidate.sum()
                    candidate_s = candidate * np.sqrt(2 * n_extra)
                    new_s = float(evaluate_sequence(candidate_s.tolist()))
                    if np.isfinite(new_s):
                        if new_s < sa2_score or rng.random() < np.exp(-(new_s - sa2_score) / T2):
                            f_sa2 = candidate
                            sa2_score = new_s
                    T2 *= 0.999
                    if sa2_score < best_score:
                        best = f_sa2 * np.sqrt(2 * n_extra)
                        best_score = sa2_score
    
    print(f"Final best: n={len(best)}, score={best_score:.6f}", flush=True)
    
    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best]
    return result
