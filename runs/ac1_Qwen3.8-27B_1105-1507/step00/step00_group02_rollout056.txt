This is a symmetric matrix with 0s and 1s, which is NOT positive semidefinite in general. So the problem is non-convex.

def _fft_score(a, N, n):
    """Compute score = 2*n*max(conv(a,a)) assuming sum(a)=1, using FFT."""

    c = np.fft.irfft(A * A, N)[:2 * n - 1]

def _gradient_force(a, c, N, n, T):
    """Compute the gradient force using FFT adjoint of convolution."""
    A = np.fft.rfft(a, N)

    c_range = c_max - c.min()

        return np.ones(n) / n

    force = np.maximum(force, 0.0)
