"""
Minimizes the step function peak-to-mean ratio using subgradient descent
on the autocorrelation peak, with multiple length explorations and 
random restarts. Exploits FFT-based convolution for efficiency.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fast_self_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    return np.real(ifft(A * A))[:size]


def compute_objective_value(a):
    """Compute the objective: 2*n*max(conv)/sum^2"""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fast_self_convolve(a)
    M = np.max(conv)
    return 2.0 * n * M / (S * S)


def subgradient_step(a, top_fraction=0.1, max_active=20):
    """
    Compute descent direction for minimizing max_k C_k (with S=1 fixed).
    Returns (direction, M, conv)
    """
    n = len(a)
    conv = fast_self_convolve(a)
    M = np.max(conv)
    
    # Find active set: indices where C_k is within top_fraction of max
    threshold = M * (1.0 - top_fraction)
    active = np.where(conv >= threshold)[0]
    
    # Limit to max_active for computational efficiency
    if len(active) > max_active:
        # Keep the largest ones
        sorted_active = active[np.argsort(conv[active])[::-1]]
        active = sorted_active[:max_active]
    
    # Compute dM/da[m] = (1/|A|) * sum_{k in A} 2*a[k-m]
    # For each k in active, a[k-m] is valid when 0 <= k-m <= n-1
    g = np.zeros(n)
    for k in active:
        k = int(k)
        # a[k-m] for m in [0, n-1]: valid when 0 <= k-m <= n-1
        # i.e., k-n+1 <= m <= k, clipped to [0, n-1]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # g[m] += a[k-m] for m in [lo, hi]
            # When m=lo: k-m = k-lo
            # When m=hi: k-m = k-hi
            # So we need a[k-hi : k-lo+1] reversed
            g[lo:hi+1] += 2.0 * a[k-hi : k-lo+1][::-1]
    
    g /= len(active)
    
    # Descent direction is -g (we want to decrease M)
    direction = -g
    return direction, M, conv


def optimize_sequence(a0, n_iters, step0, step_min, rng, top_fraction=0.1, max_active=15):
    """
    Optimize sequence using subgradient descent with annealing.
    Input: a0 normalized to sum=1, all non-negative.
    Returns: best sequence found (normalized to sum=1).
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = np.sum(a)
    if S < 0.01:
        return a
    a = a / S
    
    best_a = a.copy()
    best_M = np.max(fast_self_convolve(a))
    
    stagnation = 0
    patience = 100
    
    for it in range(n_iters):
        direction, M, conv = subgradient_step(a, top_fraction, max_active)
        
        if M < best_M - 1e-10:
            best_M = M
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Annealed step size
        frac = float(it) / max(1, n_iters)
        step = step0 * (step_min / step0) ** frac
        
        # Take step
        a_new = a + step * direction  # direction is already -grad
        a_new = np.maximum(a_new, 0.0)
        
        S_new = np.sum(a_new)
        if S_new < 0.01:
            a_new = a.copy()
            S_new = S
        else:
            a = a_new / S_new
        
        # Random perturbation to escape local minima
        if stagnation > patience and rng.random() < 0.1:
            # Add multiplicative noise
            noise = np.exp(rng.normal(0, 0.2, n))
            a = a * noise
            a = np.maximum(a, 0)
            S = np.sum(a)
            if S > 0.01:
                a = a / S
            stagnation = 0
        
        # Periodic random restart (small)
        if it % 200 == 199:
            idx = rng.integers(0, n)
            a[idx] *= rng.uniform(0.3, 3.0)
            a = np.maximum(a, 0)
            S = np.sum(a)
            if S > 0.01:

"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using subgradient descent on the
autocorrelation peak with FFT-accelerated convolution, multiple length
explorations, diverse initializations, momentum, and optional LP polishing.

import numpy as np

from scipy.fft import fft, ifft


def fast_self_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    return np.real(ifft(A * A))[:size]


def compute_score(a):
    """Compute objective value: 2*n*max(conv)/sum^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fast_self_convolve(a)
    M = np.max(conv)
    return 2.0 * n * M / (S * S)


def active_subgradient(a, top_frac=0.05, max_active=15):
    """
    Compute descent direction for minimizing max_k C_k (assumes sum(a)=1).
    Returns (direction, M, conv).
    dM/da[m] = (1/|A|) * sum_{k in A} 2*a[k-m] (out-of-range = 0).
    """
    n = len(a)
    conv = fast_self_convolve(a)
    M = np.max(conv)
    threshold = M * (1.0 - top_frac)
    active = np.where(conv >= threshold)[0]
    if len(active) > max_active:
        active = active[np.argsort(conv[active])[::-1]][:max_active]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])

    g = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(active)
    return -g, M, conv


def optimize_subgrad(a0, n_iters, step0, step_end, rng, top_frac, max_active, momentum=0.7):
    """Subgradient descent with momentum and annealing for minimizing peak."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    S = np.sum(a)
    if S < 0.01:
        return a
    a = a / S

    best_a = a.copy()
    best_M = np.max(fast_self_convolve(a))
    vel = np.zeros(n)
    stagnation = 0

    for it in range(n_iters):
        frac = float(it) / max(1, n_iters)
        step = step0 * (step_end / step0) ** frac

        direction, M, conv = active_subgradient(a, top_frac, max_active)
        if M < best_M - 1e-9:
            best_M = M
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Momentum update
        vel = momentum * vel + direction
        a_new = a + step * vel
        a_new = np.maximum(a_new, 0.0)
        S_new = np.sum(a_new)
        if S_new < 0.01:
            a_new = a.copy()
        else:
            a = a_new / S_new

        # Escape local minima
        if stagnation > 80:
            r = rng.random()
            if r < 0.15:
                # Multiplicative noise
                noise = np.exp(rng.normal(0, 0.3, n))
                a = a * noise
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 0.01:
                    a = a / s
                    vel = np.zeros(n)
                stagnation = 0
            elif r < 0.25:
                # Block scaling
                w = max(2, n // 50)
                start = int(rng.integers(0, n - w))
                a[start:start + w] *= rng.uniform(0.2, 5.0)
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 0.01:
                    a = a / s
                    vel = np.zeros(n)
                stagnation = 0

    return best_a

    LP-based direction: find g>=0 maximizing sum(g) s.t. max_k (convolve(f,g))<=1.
    Then blend f and normalized g.

        import cvxpy as cp
    except ImportError:
        return None

    n = len(f)
    f = np.asarray(f, dtype=np.float64)
    f = f / np.sum(f)

    constraints = []

        lo = max(0, k - n + 1)

            idx = k - np.arange(lo, hi + 1)
