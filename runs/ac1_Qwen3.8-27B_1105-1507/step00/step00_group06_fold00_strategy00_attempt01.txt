def evaluate_sequence(sequence):
    """Protect against negativity and normalizes on the fly."""
    sequence = np.asarray(sequence, dtype=np.float64)

    if sequence.ndim != 1 or n == 0:

    sum_a = np.sum(sequence_clipped)
