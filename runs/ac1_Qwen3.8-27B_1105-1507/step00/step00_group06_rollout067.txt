from scipy.fft import fft, ifft

def _next_pow2(n):
    """Return smallest power of 2 >= n."""
    p = 1
    while p < n:
        p <<= 1
    return p


def _conv_sq(p, size):
    """Compute linear convolution p*p using FFT of given size."""
    P = fft(p, n=size)
    return ifft(P * P).real[:2 * len(p) - 1]


def _simplex_proj(v):
    """Project v onto the probability simplex {p: sum p = 1, p >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rhs = (cssv - 1.0) / np.arange(1, n + 1)
    cond = u > rhs
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, 1.0 / n)
    r = idx[-1]
    theta = (cssv[r] - 1.0) / (r + 1)
    p = np.maximum(v - theta, 0.0)
    s = p.sum()
    if s > 0:
        p /= s
    else:
        p = np.full(n, 1.0 / n)
    return p


def _topk_gradient(p, c, k, grad_size):
    """Compute average gradient of top-k convolution values w.r.t. p.

    grad_j = 2 * sum_i p[i] * w[j+i] where w is uniform over top-k indices.
    Computed via FFT cross-correlation.
    """

    mc = len(c)
    top_idx = np.argpartition(c, -k)[-k:]

    w[top_idx] = 1.0 / k
    P_ft = fft(p, n=grad_size)
    W_ft = fft(w, n=grad_size)
    corr = ifft(np.conj(P_ft) * W_ft).real
