Now let me think about the overall algorithm more carefully.

**Algorithm: Projected Gradient Descent on the Simplex**

Problem: $\min_a M(a) = \max_k (a*a)_k$ s.t. $a \ge 0, \sum a = 1$.

Smooth approximation: $\tilde{M}(a) = \tau \log \sum_k \exp(b_k / \tau)$ where $b = a*a$.

Gradient: $\nabla \tilde{M}(a)_j = \sum_k \frac{\exp(b_k/\tau)}{\sum_l \exp(b_l/\tau)} a_{k-j} = \sum_k p_k a_{k-j} = g_j$.

Update:
1. Compute $b = a*a$ (via FFT).
2. Compute $p = \text{softmax}(b/\tau)$ (shift by max for stability).
3. Compute $g = \nabla \tilde{M}(a)$ (via FFT).
4. $v = a - \eta \cdot g$.
5. $a \leftarrow \text{project\_simplex}(v)$.

The projected gradient method converges at rate $O(1/\sqrt{T})$ for smooth functions, or $O(1/T)$ with accelerated methods. For non-smooth (the true max), it's $O(1/\sqrt{T})$.

With the softmax smoothing, we get a smooth approximation, and the convergence rate depends on the smoothness (which is $1/\tau$ in a sense).

For practical purposes, let's just run it for a fixed number of iterations or until the time budget is exhausted.

**Step size selection:**
- The Lipschitz constant of the gradient is bounded by $\|p\|_1 \cdot L$ where $L$ is related to the autocorrelation operator. Roughly, $\|g\| \le \|a\|_2^2 + \ldots$
- A safe step size: $\eta = 1 / (\|g\|_\infty + \epsilon)$.
- Or use backtracking line search.

"""
Projected gradient descent (with Nesterov acceleration) to minimize the
peak autocorrelation of a non-negative sequence, subject to the simplex
constraint. Uses FFT for efficient autocorrelation and gradient computation.
"""

import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _autocorr_fft(a):
    """Compute linear autocorrelation of a via FFT."""
    n = len(a)
    size = _next_pow2(2 * n - 1)
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F.conjugate(), size)[:2*n-1]
    return b


def _smooth_max_gradient(a, b, tau):
    """Compute gradient of softmax-smoothed max of autocorrelation b.
    
    Returns g where g_j = sum_k p_k * a_{k-j} (valid indices only).
    """
    n = len(a)
    # Numerically stable softmax
    b_shifted = (b - b.max()) / tau
    exp_b = np.exp(b_shifted)
    p = exp_b / exp_b.sum()
    
    # g_j = sum_{m=0}^{n-1} p_{j+m} * a_m
    # = (p * flip(a))[n-1+j]
    a_flip = a[::-1].copy()
    size = _next_pow2(len(p) + n - 1)
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a_flip, size)
    conv = np.fft.irfft(P * A, size)
    g = conv[n-1 : 2*n-1]
    return g


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u, _ = np.sort(v), None
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (cssv - 1)
    if not cond.any():
        # All components are 0 or negative after threshold
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _run_gradient_descent(a0, n_iters, tau0, eta, use_nesterov=True, verbose=False):
    """Run projected gradient descent (optionally with Nesterov acceleration)."""
    a = _project_simplex(a0.astype(np.float64).copy())
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
        return a, 2.0 * len(a)
    a = a / S
    
    n = len(a)
    best_score = float('inf')
    best_a = a.copy()
    
    y = a.copy() if use_nesterov else None
    x_prev = a.copy() if use_nesterov else None
    t = 1.0
    
    for it in range(n_iters):
        # Compute autocorrelation at current point
        b = _autocorr_fft(y)
        M = b.max()
        score = 2.0 * n * M
        
        if score < best_score:
            best_score = score
            best_a = y.copy()
        
        if verbose and it % 1000 == 0:
            print(f"  iter {it}: score={score:.6f}, M={M:.6f}, tau={tau0:.2e}")
        
        # Compute gradient
        g = _smooth_max_gradient(y, b, tau0)
        
        if use_nesterov:
            # FISTA update
            x_new = _project_simplex(y - eta * g)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = x_new + ((t - 1.0) / t_new) * (x_new - x_prev)
            y = np.maximum(y, 0.0)  # Maintain non-negativity
            # Renormalize to stay on simplex (approximate)

            if y_sum > 1e-15:

            else:
                y = x_new
            x_prev = x_new
            t = t_new
        else:
            a = _project_simplex(a - eta * g)
            y = a
        
        # Update tau (anneal)
        # tau0 = tau0 * 0.999  # Very slow annealing
    
    return best_a, best_score

    if best_seq is not None and time.monotonic() < deadline:
        print(f"Refining best sequence (score={best_score:.6f})...")
        a0 = best_seq.copy()
        n = len(a0)
        a0 = a0 / a0.sum()
        b_init = _autocorr_fft(a0)
        M_init = b_init.max()
        tau = max(M_init / 200.0, 1e-14)
        eta = 0.1 / (M_init + 1e-12)
        
        best_a, local_score = _run_gradient_descent(
            a0, 500000, tau, eta, use_nesterov=True, verbose=True
        )

        if actual_score < best_score:
            best_score = actual_score
            best_seq = best_a.copy()
