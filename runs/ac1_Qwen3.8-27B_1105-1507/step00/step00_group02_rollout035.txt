    """Heuristic: find g that approximately maximizes sum(g) s.t. conv(f,g) <= M."""
    q = fftconvolve(f, f)
    # Peak mask: where q is close to M
    peak_mask = q >= peak_threshold * M
    peak_count = peak_mask.sum()
    
    if peak_count == 0:
        return np.ones(n)
    
    # For each position j, compute how much it would contribute to peaks
    # interference_j = sum_{k: peak_mask[k]} f_{k-j} (where f_{k-j} is valid)
    # This is the cross-correlation of f with peak_mask
    interference = np.correlate(peak_mask.astype(float), f, mode='valid')  # length n
    
    # g should be small where interference is high
    # Normalize: sum(g) should be as large as possible
    # Simple choice: g_j = 1 / (1 + c * interference_j)
    c = 1.0
    g = 1.0 / (1.0 + c * interference)
    g = np.clip(g, 0, None)
    
    return g

**Stochastic smoothed gradient**:
1. Sample K random indices from the 2n-1 convolution positions, weighted by exp(tau * q_k).
2. Compute the gradient as a weighted average of grad q_k for the sampled k's.

Each grad q_k is just 2 * a shifted by k, which is O(n) to compute. So the stochastic gradient is O(K*n). For K=10, that's O(10n) = O(70K) operations, which is very fast.

```python
def stochastic_gradient(a, tau, n, K, rng):
    """Compute a stochastic approximation of the smoothed gradient."""
    q = fftconvolve(a, a)  # O(n log n)
    q_max = q.max()
    
    # Sample K indices with softmax weights
    tau_q = tau * (q - q_max)
    weights = np.exp(tau_q)
    total = weights.sum()
    probs = weights / total
    
    # Sample K indices (with replacement)
    indices = rng.choice(2*n-1, size=K, replace=True, p=probs)
    
    # Compute gradient as average
    grad = np.zeros(n)
    for idx in indices:
        # grad q_idx: (grad q_idx)[j] = 2 * a[idx-j] if 0 <= idx-j <= n-1
        # So grad j ranges from max(0, idx-n+1) to min(idx, n-1)
        j_start = max(0, idx - n + 1)
        j_end = min(idx, n - 1) + 1
        m_start = idx - (j_end - 1)  # = idx - j_end + 1
        m_end = idx - j_start + 1
        grad[j_start:j_end] += 2.0 * a[m_start:m_end]
    
    grad /= K
    return grad, q_max
```

```python
class AdamState:
    def __init__(self, n, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros(n)
        self.v = np.zeros(n)
        self.t = 0
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
    
    def update(self, grad, lr):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad**2
        m_hat = self.m / (1 - self.beta1**self.t)
        v_hat = self.v / (1 - self.beta2**self.t)
        return lr * m_hat / (np.sqrt(v_hat) + self.eps)
```

Minimizes 2n*max(conv(a,a))/(sum(a))^2 using stochastic smoothed gradient
descent with Adam optimizer, followed by fine-tuning with perturbations.

import numpy as np

from scipy.signal import fftconvolve

def project_simplex(a, S):
    """Project a onto {x >= 0, sum(x) = S}."""
    n = a.size
    if n == 0:
        return a
    lo = 0.0
    hi = float(a.max()) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.maximum(a - mid, 0.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.maximum(a - (lo + hi) * 0.5, 0.0)
