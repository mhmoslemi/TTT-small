Let me parameterize: $a_i = c \cdot (s + (1-s) \cdot i/(n-1))$ where $s \in [0, 1]$ controls the slope.
- $s = 1$: uniform.
- $s = 0$: linear ramp from 0 to max.
- $s < 1$: partial ramp.

For $s = 0$ (linear ramp from 0 to $2c$): $a_i = 2c \cdot i/(n-1)$ (approximately).
Sum: $2c \cdot (n-1)/2 \cdot n/(n-1) = cn$. So $c = 1/n$.
$a_i = 2i/(n(n-1))$.

$(a*a)_{n-1} = \sum_i a_i a_{n-1-i} = \sum_i \frac{2i}{n(n-1)} \cdot \frac{2(n-1-i)}{n(n-1)} = \frac{4}{n^2(n-1)^2} \sum_i i(n-1-i)$

$\sum_i i(n-1-i) = (n-1)\frac{n(n-1)}{2} - \frac{(n-1)n(2n-1)}{6} = \frac{n(n-1)}{6}[3(n-1) - (2n-1)] = \frac{n(n-1)}{6}(2n-2) = \frac{n(n-1)^2}{3}$

So $(a*a)_{n-1} = \frac{4n(n-1)^2}{3n^2(n-1)^2} = \frac{4}{3n}$.

Objective: $2n \cdot \frac{4}{3n} = 8/3 \approx 2.67$. Worse than uniform!

What about a step function? $a_i = c_1$ for $i < m$, $a_i = c_2$ for $i \geq m$, with $m \cdot c_1 + (n-m) \cdot c_2 = 1$.

$(a*a)_{n-1} = \sum_i a_i a_{n-1-i}$.

This is complicated. Let me just go with the computational approach and let the optimizer figure it out.

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using subgradient descent
with multiple initializations and adaptive sequence lengths.
"""
import numpy as np
from scipy.signal import fftconvolve
import time

def _normalize(a):
    """Normalize to sum=1, clip to [0, 1000]."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones_like(a) / len(a)
    return a / s

def _compute_objective(a):
    """Compute the objective value without calling evaluate_sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fftconvolve(a, a)
    return float(2 * n * b.max() / (s * s))

def _subgradient_step(a, b, step_size):
    """Perform one subgradient step to reduce max(a*a)."""
    n = len(a)
    M = b.max()
    # Find active set (elements close to max)
    threshold = M * 0.95
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    # Compute subgradient
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            g[i] += a[k - i]
    g /= len(active)
    
    # Step
    a_new = a - step_size * g
    a_new = np.clip(a_new, 0.0, 1000.0)
    a_new = _normalize(a_new)
    return a_new

def _optimize_from_init(a0, n, time_limit):
    """Optimize starting from a0 using subgradient descent."""
    deadline = time.monotonic() + time_limit
    a = a0.copy()
    a = _normalize(a)
    
    best_a = a.copy()
    best_obj = _compute_objective(a)
    
    step_size = 1.0 / n  # initial step size
    patience = 0
    max_patience = 200
    
    while time.monotonic() < deadline:
        b = fftconvolve(a, a)
        M = b.max()
        obj = 2 * n * M / (a.sum() ** 2)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            # Reset with smaller step
            a = best_a.copy()
            step_size *= 0.5
            patience = 0
            if step_size < 1e-10 / n:
                break
        
        a = _subgradient_step(a, b, step_size)
    
    return best_a, best_obj

        # Exponential (decaying right)
        a_exp2 = np.exp(-beta * np.arange(n) / n)

    """Compute g[i] = sum_j a[j] * w[j+i] for i = 0, ..., len(a)-1."""
    na = len(a)
