    n = len(a)

    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:size]

    """
    Compute gradient of sum_k weights[k] * (a*a)[k] w.r.t. a.
    grad[j] = 2 * sum_k weights[k] * a[k-j] (with a[m]=0 outside [0,n-1])

    Actually: grad[j] = 2 * sum_m a[m] * weights[m+j]

    v = beta2 * v + (1 - beta2) * grad * grad
    
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    
    params['m'] = m
    params['v'] = v

    n = len(a)
    wlen = len(w)
