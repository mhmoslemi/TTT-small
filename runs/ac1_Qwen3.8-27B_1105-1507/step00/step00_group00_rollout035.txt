    n = len(a)

def top_lag_gradient(a, b, K=30):
    """Compute gradient of max(b) approximated by top K lags."""
    n = len(a)
    # Get top K lags
    # Use argpartition for efficiency
    if K >= len(b):
        idx = np.arange(len(b))
    else:
        part_idx = np.argpartition(b, -K)[-K:]
        idx = part_idx[np.argsort(b[part_idx])[::-1]]
    
    # Weights proportional to b values (or all 1)
    w = b[idx]
    w_sum = w.sum()
    if w_sum == 0:
        return a.copy()
    w = w / w_sum
    
    g = a.copy()  # a_j * sum(w_k) = a_j
    for k, wi in zip(idx, w):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            a_idx = k - np.arange(j_lo, j_hi + 1)
            g[j_lo:j_hi+1] += wi * a[a_idx]
    return g

    S = a.sum()

def top_lag_gradient(a, b, K=50):

    else:
