import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_matrix_sparse(f):
    """Build sparse (2n-1)xn matrix A such that A@g = f*g."""
    n = len(f)
    offsets = [-d for d in range(n)]
    diags = [np.full(n - d, f[d]) for d in range(n)]
    A = csr_matrix(diags, offsets, shape=(2*n-1, n))
    return A

def fast_score(a):
    """Compute score without calling evaluate_sequence (for speed)."""
    n = len(a)
    conv = np.convolve(a, a, mode='full')
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def lp_refine(f, deadline):
    """Apply one LP-based refinement step to f. Returns improved f or None."""
    n = len(f)
    # Normalize to sum = 1
    f = f / np.sum(f)
    conv = np.convolve(f, f, mode='full')
    M = np.max(conv)
    
    if time.monotonic() > deadline:
        return None
    
    # Build constraint matrix
    A = build_conv_matrix_sparse(f)
    
    # Solve LP
    c = -np.ones(n)
    b = np.full(2*n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs',
                  options={'time_limit': 60})
    
    if not res.success or res.x is None:
        return None
    
    g = res.x
    if np.sum(g) < 1e-10:
        return None
    
    # Line search
    best_t = 0
    best_score = fast_score(f)
    for t in np.arange(0.05, 1.01, 0.05):
        h = (1 - t) * f + t * g
        s = fast_score(h)
        if s < best_score:
            best_score = s
            best_t = t
    
    if best_t == 0:
        return None
    
    return best_t * g + (1 - best_t) * f

        phase = rng.uniform(0, 2 * np.pi)
        pert = eps * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
        h = f + pert
        h = np.clip(h, 0, None)
        h = h / np.sum(h)

        s_h = fast_score(h)
        if s_h < current_s:
            f = h
            current_s = s_h
            ls_count += 1
            ls_stagnant = 0
            if ls_count % 100 == 0:
                print(f"  LS[{ls_count}]: {current_s:.6f}")
        else:
            ls_stagnant += 1
            if ls_stagnant > 500:
                break

    # Final evaluation
    candidate_final = f * np.sum(best)
    s_final = evaluate_sequence(candidate_final.tolist())
    if s_final < best_score:
        best = candidate_final.copy()
        best_score = s_final
        print(f"  Phase 3 final: {s_final:.6f}")
    else:
        print(f"  Phase 3 did not improve. Best remains: {best_score:.6f}")

    # Ensure validity
    best = np.clip(best, 0.0, 1000.0)
    if np.sum(best) < 0.01:
        best = fallback

    final_score = evaluate_sequence(best.tolist())
    print(f"FINAL: score={final_score:.6f}, n={np.sum(best > 1e-10)}, total_items={len(best)}")

    return [float(x) for x in best]
