"""
Minimize J(a) = 2n * max_k convolve(a,a)[k] / (sum a)^2  over non-negative a.

Key insight exploited here: convolve(a,a) is NOT the autocorrelation unless a is
palindromic.  For a NON-palindromic "single-step" profile (low block then high
block) the self-convolution develops a flat plateau whose peak is well below the
uniform value, giving J < 2 (e.g. [1,1,2,2] -> J ~ 1.778).  Symmetrizing the
vector is therefore forbidden (it forces a palindromic shape and pins J at 2).

Algorithm:
  * Multi-start: for several lengths n and several non-symmetric structured
    initializers (single steps, ramps, trapezoids, ramped parent), run a local
    optimizer.
  * Local optimizer: projected (clip + renormalize to sum=1) gradient steps on
    max_k C[k], using the subgradient averaged over the active set of lag
    positions that (nearly) attain the max; tuned step size with backtracking;
    interleaved with non-symmetric random perturbations (block rescale, smooth
    bump, mass-shift) to escape local minima.
  * FFT-based self-convolution makes each objective evaluation O(n log n), so
    very long sequences ("attack surface") are affordable.
  * The global best (by the authoritative evaluate_sequence) is kept and a final
    polish is run on it before returning.
"""

import time
import math
import numpy as np

_DEADLINE_EXTRA = 4.0  # safety margin before the hard 1500 s stop


def _conv_sq(a):
    """Full self-convolution of a (== np.convolve(a,a)) via FFT."""
    n = a.size
    size = 2 * n - 1
    m = 1
    while m < size:
        m <<= 1
    F = np.fft.rfft(a, m)
    out = np.fft.irfft(F * F, m)[:size]
    out = np.maximum(out, 0.0)  # kill tiny negative round-off
    return out


def _fast_obj(a):
    """Objective for a vector already normalized to sum=1: 2*n*max(C)."""
    n = a.size
    C = _conv_sq(a)
    return float(2.0 * n * np.max(C))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.ones_like(a)
        return a / a.size
    return a / s


def _active_gradient(a, thr=0.985):
    """Subgradient of max_k C[k], averaged over active lags (within thr of max)."""
    n = a.size
    C = _conv_sq(a)
    mx = C.max()
    active = np.nonzero(C >= thr * mx)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(C))])
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        grad[js] += 2.0 * a[k - js]
    grad /= float(active.size)
    return grad, int(active.size)


def _perturb(a, rng):
    """Non-symmetric random perturbation of a (always renormalized)."""
    n = a.size
    b = a.copy()
    mode = rng.integers(0, 4)
    if mode == 0:
        # rescale a random contiguous block
        width = int(rng.integers(max(2, n // 64), max(3, n // 4)))
        s = int(rng.integers(0, max(1, n - width + 1)))
        b[s:s + width] *= float(math.exp(rng.normal(0.0, 0.35)))
    elif mode == 1:
        # smooth bump (positive or negative)
        c = float(rng.uniform(0, n))
        w = float(rng.uniform(n / 16.0, n / 4.0))
        amp = float(rng.uniform(-0.5, 0.5)) * a.mean()
        ii = np.arange(n, dtype=float)
        b = a + amp * np.exp(-((ii - c) ** 2) / (2.0 * w * w))
    elif mode == 2:
        # shift mass between two random points
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        while j == i:
            j = int(rng.integers(0, n))
        alpha = float(rng.uniform(-0.5, 0.5)) * a.mean()
        b[i] -= alpha
        b[j] += alpha
    else:
        # gentle global noise
        b = a + rng.normal(0.0, 0.1, n) * a.mean()
    return _normalize(b)


def _local_opt(a0, deadline, rng, verbose_tag=""):
    """Refine a0 (sum normalized) until deadline; return (best_a, best_f)."""
    a = _normalize(a0.copy())
    best = a.copy()
    best_f = _fast_obj(best)
    lr = 0.05
    stall = 0
    t0 = time.monotonic()
    steps = 0
    while time.monotonic() < deadline:
        # --- gradient step (subgradient averaged over active lags) -----------
        grad, nact = _active_gradient(a)
        grad = grad - grad.mean()          # stay on the sum=1 slice
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            step = lr * grad / gn
            a_new = _normalize(a - step)
            f_new = _fast_obj(a_new)
            if f_new < best_f:
                best = a_new
                best_f = f_new
                a = a_new
                lr = min(lr * 1.10, 1.0)
                stall = 0
            else:
                lr *= 0.7
                stall += 1
            steps += 1
        # --- occasional perturbation to escape local minima -----------------
        do_perturb = (stall >= 6) or (steps % 25 == 0)
        if do_perturb:
            p = _perturb(best, rng)
            fp = _fast_obj(p)
            if fp < best_f:
                a = p
                best = p
                best_f = fp
                lr = min(lr * 1.2, 1.0)
                stall = 0
            else:
                a = best
                stall += 1
        if steps > 0 and (steps % 2000 == 0):
            print("[%s] steps=%d  local_f=%.6f" % (verbose_tag, steps, best_f))
        if lr < 1e-5 and stall > 30:
            break
    return best, best_f


def _make_inits(n, rng, parent):
    """Yield several NON-symmetric structured initializers of length n."""
    inits = []
    base = np.ones(n)
    # single steps: low then high
    for frac in (0.2, 0.33, 0.5, 0.66, 0.8):
        m = max(1, min(n - 1, int(round(frac * n))))
        for r in (0.7, 1.0, 1.5, 2.0, 3.0, 5.0):
            v = np.ones(n)
            v[m:] = r
            inits.append(v)
            v2 = np.ones(n)
            v2[:m] = r
            inits.append(v2)
    # ramps
    for r in (0.5, 0.8, 1.5, 3.0):
        inits.append(np.linspace(1.0, r, n))
        inits.append(np.linspace(r, 1.0, n))
    # trapezoid: low middle, high edges (non-palindromic bias via random cut)
    m1 = max(1, int(n * 0.25))
    m2 = min(n - 1, int(n * 0.75))
    trap = np.ones(n)
    trap[m1:m2] = 0.4
    inits.append(trap)
    # ramped parent (breaks any symmetry of the supplied parent)
    if parent is not None:
        p = np.clip(np.asarray(parent, dtype=float).reshape(-1), 0.0, 1000.0)
        if p.size == n and p.sum() > 0 and np.isfinite(p).all():
            ramp = np.linspace(0.4, 1.8, n)
            inits.append(p * ramp)
            inits.append(p * (2.0 - ramp))
        elif p.size > 1:
            # resample parent to length n and ramp it
            idx = np.linspace(0, p.size - 1, n)
            lo = np.floor(idx).astype(int)
            hi = np.minimum(lo + 1, p.size - 1)
            fr = idx - lo
            res = p[lo] * (1.0 - fr) + p[hi] * fr
            res = np.clip(res, 0.0, 1000.0)
            ramp = np.linspace(0.4, 1.8, n)
            inits.append(np.maximum(res * ramp, 1e-4))
    # a couple of purely random non-symmetric starts
    for _ in range(2):
        inits.append(np.maximum(rng.gamma(2.0, 1.0, n), 1e-4))
    return inits
