```python
def multiplicative_weight_update(f, eta=0.1, t=0.5, top_k=1):
    n = len(f)
    c = np.convolve(f, f)
    M = c.max()
    k_star = c.argmax()
    
    # Compute pressure: pressure_i = f[k_star - i] for valid i
    # f[k_star - i] is valid when 0 <= k_star - i < n
    # i.e., k_star - n + 1 <= i <= k_star
    pressure = np.zeros(n)
    i_start = max(0, k_star - n + 1)
    i_end = min(n - 1, k_star)
    # For i in [i_start, i_end]: pressure[i] = f[k_star - i]
    # k_star - i ranges from k_star - i_end to k_star - i_start
    for i in range(i_start, i_end + 1):
        pressure[i] = f[k_star - i]
    
    # g_i = exp(-eta * pressure_i)
    g = np.exp(-eta * pressure)
    
    # Normalize to same sum
    S = f.sum()
    g = g * S / g.sum()
    
    # Update
    f_new = (1 - t) * f + t * g
    return f_new
```
