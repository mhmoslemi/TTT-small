    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()  # next power of 2
    A = np.fft.rfft(a, fft_size)

    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2 * n * max_b / (sum_a ** 2)

def fast_score(a):
    """Fast score computation using FFT."""
    n = len(a)
    if n == 0:
        return np.inf
    fft_size = 1 << (2*n-1).bit_length()

def coordinate_descent(a, rng, max_passes=1000, deadline=None):
    """Coordinate descent optimization."""
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0, 1000)
    # Normalize to sum=1 for stability
    s = np.sum(a)
    if s < 1e-10:
        a[:] = 1.0/n
        s = 1.0
    else:
        a /= s
    
    best_score = fast_score(a)
    print(f"  CD start: score={best_score:.6f}, n={n}")
    
    for p in range(max_passes):
        if deadline and time.monotonic() > deadline:
            break
        improved = False
        # Random order of coordinates
        order = rng.permutation(n)
        for j in order:
            # Minimize over a[j] with others fixed
            base = a.copy()
            base[j] = 0
            s_base = np.sum(base)
            
            def obj(x):
                base[j] = x
                sc = fast_score(base)
                base[j] = 0
                return sc
            
            res = minimize_scalar(obj, bounds=(0, 1000), method='bounded',
                                 options={'xatol': 1e-10})
            new_val = res.x
            if new_val != a[j]:
                a[j] = new_val
                new_score = res.fun
                if new_score < best_score:
                    best_score = new_score
                    improved = True
        
        if not improved:

            idx = rng.integers(0, n)
            a[idx] *= np.exp(rng.normal(0, 0.5))
            a[idx] = min(1000.0, max(0.0, a[idx]))
        
        if p % 50 == 0:
            print(f"  CD pass {p}: score={best_score:.6f}")
    
    return a, best_score

    """

            return np.inf
        size = 2 * n - 1
        fs = 1 << (size - 1).bit_length()
        A = np.fft.rfft(a, fs)
        conv = np.fft.irfft(A * A, fs)[:size]
