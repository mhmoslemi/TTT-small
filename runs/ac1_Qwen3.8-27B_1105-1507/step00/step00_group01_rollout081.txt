def lp_refine(a, M):
    n = len(a)
    # Variables: g_0, ..., g_{n-1}
    # Maximize sum(g) → minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (a*g)_k <= M for k = 0, ..., 2n-2
    # (a*g)_k = sum_j a[j] * g[k-j] for valid j
    # = sum_j A[k,j] * g[j]
    # where A[k,j] = a[k-j] if 0 <= k-j < n
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A_ub[k, j] = a[k-j]
    
    b_ub = np.full(2*n-1, M)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return result.x

def fast_convolve(a):
    """Compute a * a using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def gradient_step(a, b, peak_frac=0.95, eta=0.1):
    """One step of gradient descent on max(b)."""
    n = len(a)
    max_b = np.max(b)
    threshold = max_b * peak_frac
    peak_mask = b >= threshold
    n_peaks = np.sum(peak_mask)
    if n_peaks == 0:
        return a, max_b
    
    # Compute subgradient
    subgrad = np.zeros(n)
    peak_indices = np.where(peak_mask)[0]
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        subgrad[lo:hi+1] += a[lo+hi-k:hi-lo+1][::-1]
    subgrad /= n_peaks
    
    # Gradient step
    a_new = a - eta * subgrad
    a_new = np.maximum(0.0, a_new)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new /= s
    return a_new, max_b

    for _ in range(max_iter_lp):
