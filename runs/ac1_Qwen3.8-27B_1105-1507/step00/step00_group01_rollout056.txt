```python
import numpy as np
import time
from numpy.lib.stride_tricks import sliding_window_view

def compute_objective(a):
    """Compute the objective function value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    max_c = c.max()
    return 2.0 * n * max_c / (S * S)

def gradient_smooth_max(a, beta):
    """Compute the gradient of the smooth max approximation of max(convolve(a,a))."""
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1
    # Numerical stability
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    weights = exp_c / exp_c.sum()  # (2n-1,)
    
    # smooth max value
    smooth_max_val = c_max + np.log(exp_c.sum()) / beta
    
    # Gradient computation
    # grad_i = sum_k weights[k] * dc_k/da_i
    # = sum_k weights[k] * a[k-i] * (1 + delta_{k,2i})
    
    # First part: sum_k weights[k] * a[k-i] (for valid k-i)
    # = (sliding window of weights) @ a
    # M[i, j] = weights[i+j], shape (n, n)
    # We need weights to have length at least 2n-1, which it does.
    
    M = sliding_window_view(weights, n)  # shape (n, n), M[i,j] = weights[i+j]
    grad_main = M @ a  # shape (n,)
    
    # Second part: a_i * weights[2i] (if 2i < 2n-1)
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < (2 * n - 1)
    grad_diag = np.zeros(n)
    grad_diag[valid] = a[valid] * weights[diag_idx[valid]]
    
    grad = grad_main + grad_diag
    return grad, smooth_max_val

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 0:
        return np.ones(n) / n
    # Algorithm from "Euclidean distance to a simplex"
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_max_descent(a_init, beta_start, beta_end, n_iter, lr, deadline):
    """Run smooth max gradient descent."""
    a = a_init.copy().astype(float)
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_objective(a)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        # Adaptive beta
        beta = beta_start + (beta_end - beta_start) * (it / max(1, n_iter - 1))
        beta = max(beta, 1.0)
        
        grad, smooth_val = gradient_smooth_max(a, beta)
        
        # Gradient step
        a_new = a - lr * grad
        
        # Project onto simplex
        a_new = project_simplex(a_new)
        
        # Check improvement
        score = compute_objective(a_new)
        if score < best_score:
            best_a = a_new.copy()
            best_score = score
        
        a = a_new
        
        # Diminishing learning rate
        lr = lr / (1 + it * 1e-4)
    
    return best_a, best_score
```

```python
def lp_best_response(f, M):
    """Find g that maximizes sum(g) subject to f*g <= M, g >= 0."""
    n = len(f)
    # Build constraint matrix
    # C[k, i] = f[k-i] if 0 <= k-i < n else 0
    # Use broadcasting
    k = np.arange(2*n-1)[:, None]
    i = np.arange(n)[None, :]
    idx = k - i
    mask = (idx >= 0) & (idx < n)
    idx_clipped = np.clip(idx, 0, n-1)
    C = np.where(mask, f[idx_clipped], 0.0)
    
    # LP: minimize -sum(g) s.t. C @ g <= M, g >= 0
    from scipy.optimize import linprog
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=C, b_ub=np.full(2*n-1, M), 
                  bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```
