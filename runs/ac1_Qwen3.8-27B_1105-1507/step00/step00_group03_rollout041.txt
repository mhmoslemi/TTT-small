    """Compute gradient of (sum conv_k^p)^(1/p) with respect to a."""
    n = len(a)
    conv = np.convolve(a, a)
    conv_p = conv ** (p - 1)
    f_p = (np.sum(conv ** p)) ** (0.5 / p)  # (1/p) * (sum conv^p)^(1/p) = ... 
    # Actually f_p = (sum conv^p)^(1/p), df_p/da_i = f_p^(1-p)/p * p * sum_k conv_k^(p-1) * d(conv_k)/da_i
    # = f_p^(1-p) * sum_k conv_k^(p-1) * d(conv_k)/da_i
    
    # d(conv_k)/da_i = a[k-i] (if 0 <= k-i < n) + a[i] (if 0 <= k-i < n, for the other term)
    # Actually: conv_k = sum_t a[t]*a[k-t]
    # d(conv_k)/da_i = a[k-i] (from the first factor, when t=i) + a[k-i] (from the second factor, when k-t=i, i.e., t=k-i)
    # = 2*a[k-i] if i != k-i, or a[i] if i == k-i

power-sum smoothing, simulated annealing, and multiple initializations.
The objective is scale-invariant, so we work with sum(a)=1.
"""


def compute_conv(a):
    return np.convolve(a, a)


def obj_from_conv(conv, n):
    return 2.0 * n * float(np.max(conv))


def mirror_descent_step(a, p, eta, beta, v):
    """One mirror descent step with momentum using power-sum smoothing."""
    n = len(a)
    conv = compute_conv(a)
    # Avoid zeros causing issues with power
    conv_safe = np.maximum(conv, 1e-30)
    conv_p = conv_safe ** (p - 1)
    # Normalize to avoid overflow
    max_cp = conv_p.max()
    if max_cp > 0:
        conv_p = conv_p / max_cp
    grad = np.correlate(conv_p, a, mode='valid')[:n]
    # Update momentum
    v = beta * v + (1.0 - beta) * grad
    # Mirror descent update
    scaled = eta * v
    # Clip to avoid underflow
    scaled = np.clip(scaled, -50.0, 50.0)
    a_new = a * np.exp(-scaled)
    s = a_new.sum()
    if s < 1e-30:
        a_new = np.ones(n) / n
    else:
        a_new = a_new / s
    return a_new, v, conv


def targeted_perturbation(a, conv, rng, strength):
    """Perturb a to reduce the peak of conv."""
    n = len(a)
    k_star = int(np.argmax(conv))
    # Sensitivity: a[i] contributes a[i]*a[k_star-i] to conv[k_star]
    sens = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for i in range(lo, hi + 1):
        sens[i] = a[k_star - i]
    # Donors: high contribution to peak
    donor_score = sens * a
    # Receivers: low sensitivity
    # Select subsets
    n_sel = max(3, n // 15)
    top_d = np.argsort(donor_score)[::-1][:n_sel]
    bot_r = np.argsort(sens * a)[:n_sel]
    # Move mass

        d = top_d[i]
        r = bot_r[i]
        move = min(amt, a[d] * 0.5)
        a[d] -= move
        a[r] += move
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-30:
        a = a / s
    return a
