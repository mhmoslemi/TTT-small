    n = len(a)
    a = a.copy().astype(float)
    a = a / a.sum() * S  # normalize
    
    for it in range(max_iter):
        # Compute convolution
        conv = np.convolve(a, a)  # length 2n-1
        max_val = conv.max()
        
        # Active set: positions where conv is within 1% of max
        threshold = max_val * 0.99
        active = conv >= threshold
        
        # Gradient of max (approximated by average over active set)
        # grad[i] = (1/|A|) * sum_{k in A} 2*a[k-i] * 1[0<=k-i<n]
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            # Add contribution of this k to the gradient
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for i in range(lo, hi):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2 * a[j]
        grad /= active.sum()
        
        # Direction: we want to DECREASE the max, so move opposite to grad
        # But we also want to keep sum constant
        # Project: update = -lr * (grad - mean(grad))
        direction = -(grad - grad.mean())
        
        # Scale to maintain sum
        # New a = a + lr * direction
        # Sum change = lr * sum(direction) = lr * (n*(-mean(grad)) + n*mean(grad)) = 0
        # So sum is automatically preserved!
        
        a_new = a + lr * direction
        a_new = np.clip(a_new, 0, None)
        a_new = a_new / a_new.sum() * S  # renormalize
        
        if a_new.sum() < 1e-10:
            break
        a = a_new
    
    return a

import time

def compute_conv_fft(a):
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2*n-1]

def smooth_max_and_weights(x, tau):
    x_max = x.max()
    shifted = (x - x_max) / tau
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth_max = x_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth_max, weights

    """Compute grad[i] = 2 * sum_j a[j] * w[i+j] using FFT."""
    n = len(a)
    ra = a[::-1]
    f_ra = np.fft.rfft(ra, N_grad)
    f_w = np.fft.rfft(w, N_grad)
    c = np.fft.irfft(f_ra * f_w, N_grad)
    return 2.0 * c[n-1:2*n-1]

from scipy.optimize import linprog

def lp_step(a, M, K=50):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for top K positions, g >= 0."""
    n = len(a)
    conv = compute_conv_fft(a)  # This is a*a, but we need a*g
    
    # Actually, we need the convolution of a with g, evaluated at all positions.
    # (a*g)[k] = sum_i a[i]*g[k-i]
    # This is linear in g, so the constraint matrix is:
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # for k = 0, ..., 2n-2
    
    # But we only want the top K constraints.
    # First, find which positions are "active" (near the max of a*a).
    # At the LP step, the active set is determined by a*a (the current convolution).
    
    # Find top K positions of a*a
    top_k_indices = np.argsort(conv)[-K:]
    
    # Build the constraint matrix A (K x n)
    # A[k_local, j] = a[k_global - j] if 0 <= k_global - j < n
    A = np.zeros((K, n))
    for k_local, k_global in enumerate(top_k_indices):
        for j in range(n):
            idx = k_global - j
            if 0 <= idx < n:
                A[k_local, j] = a[idx]
    
    # LP: max c^T g s.t. A g <= M*e, g >= 0
    # c = ones(n)
    # Using scipy linprog: minimize -c^T g s.t. A g <= M*e, g >= 0
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        if g.sum() > 0:
            return g / g.sum()  # Normalize to sum = 1
    return a  # Return original if LP fails
