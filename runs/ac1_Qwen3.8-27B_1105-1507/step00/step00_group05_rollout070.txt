    """Compute convolution, find max location, and return subgradient."""
    n = len(a)

    b = np.convolve(a, a)
    k_star = np.argmax(b)
    b_max = b[k_star]
    
    # Subgradient of J = 2*n*b[k*]/S^2
    # dJ/da[i] = (2n/S^2) * 2*a[k*-i] (if valid) - (4n*b_max/S^3)

    return g, b_max, S
