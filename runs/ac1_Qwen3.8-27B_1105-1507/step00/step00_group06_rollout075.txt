"""
Minimum-peak-aperiodic-autocorrelation search for non-negative step functions.

Objective: 2 * n * max_k (a*a)[k] / (sum a)^2  (see evaluate_sequence).

The peak of the full linear convolution is identical (as a value) to the peak
of the aperiodic autocorrelation, so the inner loop minimises that peak using a
fast O(n log n) FFT autocorrelation instead of the O(n^2) exact evaluator.

Strategy (multi-strategy, time-budgeted, single CPU):
  1. Build a bank of smooth, *asymmetric* initialisers (ramps, skewed Gaussians,
     tilted raised-cosines, polynomial windows, smooth randoms).  Asymmetry makes
     <a, a_rev> smaller than sum a^2, which is the loophole that lets the
     convolution peak drop below (sum a)^2 / n and hence the score below 2.
  2. Iterative "lag shrink": repeatedly find the dominant lag of the
     autocorrelation and remove its largest contributing mass pair, relocating
     that mass to a low element so the sum (and hence the scale-invariant score)
     is preserved.  When the zero lag (main lobe) dominates, smooth the sequence
     to spread its energy.
  3. Simulated-annealing acceptance of smooth block / global perturbations to
     escape local minima, with a cooling schedule.
  4. A second phase upsamples the best sequence to a larger length (more attack
     surface) and repeats.
The exact evaluate_sequence is used only to seed from the parent and to
periodically / finally confirm the best true score.
"""

import time
import numpy as np


# --------------------------------------------------------------------------- #
# Fast autocorrelation / objective                                            #
# --------------------------------------------------------------------------- #
def _autocorr(a):
    """Aperiodic autocorrelation of 1-D array a via zero-padded FFT."""
    n = a.shape[0]
    N = 2 * n
    A = np.fft.rfft(a, N)
    r = np.fft.irfft(A * np.conj(A), N)
    return r


def _fast_objective(a):
    """Fast (scale-invariant) value of 2*n*max(a*a)/sum(a)^2."""
    S = a.sum()
    if S <= 0.0:
        return np.inf
    r = _autocorr(a)
    return float(2.0 * a.size * r.max() / (S * S))


def _exact_objective(a):

def _clamp(a):
    return np.clip(a, 0.0, 1000.0)


def _normalise(a):
    s = a.sum()
    if s <= 0.0:
        return a
    return a / s


def _init_constant(n, rng):
    return np.ones(n, dtype=np.float64)


def _init_ramp(n, rng, slope):
    x = np.linspace(0.0, 1.0, n)
    return _clamp(1.0 + slope * (2.0 * x - 1.0))


def _init_skew_gauss(n, rng, mu, sigma):
    x = np.linspace(0.0, 1.0, n)
    return _clamp(np.exp(-0.5 * ((x - mu) / sigma) ** 2) + 0.2)


def _init_tilted_cosine(n, rng, tilt):
    x = np.linspace(0.0, 1.0, n)
    base = 1.0 + 0.3 * np.cos(np.pi * (2.0 * x - 1.0))
    return _clamp(base * (1.0 + tilt * (2.0 * x - 1.0)))


def _init_poly_window(n, rng, deg, coeff):
    x = np.linspace(0.0, 1.0, n)
    t = 1.0 - (2.0 * x - 1.0) ** 2          # 0 at edges, 1 in middle
    return _clamp(1.0 + coeff * np.power(t, deg) + 0.1 * x)


def _init_smooth_random(n, rng, nfreq=4):
    x = np.arange(n, dtype=np.float64)
    p = np.zeros(n)
    for _ in range(nfreq):
        f = rng.uniform(0.5, 6.0)
        phi = rng.uniform(0.0, 2.0 * np.pi)
        p += rng.normal(0.0, 0.3) * np.sin(2.0 * np.pi * f * x / n + phi)
    return _clamp(1.0 + p)


def _make_initialisers(rng, n):
    inits = [_init_constant(n, rng)]
    for s in (-0.4, -0.2, 0.0, 0.2, 0.4, 0.6):
        inits.append(_init_ramp(n, rng, s))
    for mu in (0.3, 0.4, 0.5, 0.6, 0.7):
        for sg in (0.15, 0.25, 0.4):
            inits.append(_init_skew_gauss(n, rng, mu, sg))
    for t in (-0.5, -0.3, 0.0, 0.3, 0.5):
        inits.append(_init_tilted_cosine(n, rng, t))
    for d in (1, 2, 3):
        for c in (-0.3, 0.0, 0.3, 0.5):
            inits.append(_init_poly_window(n, rng, d, c))
    for _ in range(8):
        inits.append(_init_smooth_random(n, rng))
    # drop any all-zero / invalid
    out = []
    for a in inits:
        a = _clamp(np.asarray(a, dtype=np.float64))
        if a.size == n and np.isfinite(a).all() and a.sum() > 1e-9:
            out.append(a)
    return out


# --------------------------------------------------------------------------- #
# Moves                                                                       #
# --------------------------------------------------------------------------- #
def _lag_shrink(a, rng, step):
    """Reduce the dominant autocorrelation lag by relocating its biggest mass
    pair to a low element, preserving the sum."""
    r = _autocorr(a)
    n = a.size
    idx = int(np.argmax(r))
    a = a.copy()
    if idx == 0:
        # main lobe dominates -> smooth to spread sum a^2
        k = max(3, n // 500)
        kern = np.exp(-0.5 * (np.arange(-k, k + 1) / max(1, k // 3)) ** 2)
        kern /= kern.sum()
        a = np.convolve(a, kern, mode="same")
    else:
        L = idx if idx <= n else (2 * n - idx)
        if L <= 0 or L >= n:
            kern = np.exp(-0.5 * (np.arange(-(n // 500), n // 500 + 1) / 1.0) ** 2)
            kern /= kern.sum()
            a = np.convolve(a, kern, mode="same")
        else:
            pair = a[:n - L] * a[L:]
            i = int(np.argmax(pair))
            j = i + L
            x, y = a[i], a[j]
            hi, lo = (i, j) if x >= y else (j, i)
            hub = a[hi]
            q = int(np.argmin(a))
            if q == hi:
                q = int(rng.integers(0, n))
            delta = step * hub
            if delta >= hub:
                delta = 0.95 * hub
            a[hi] = hub - delta
            a[q] = a[q] + delta
    a = _clamp(a)
    return a


def _block_perturb(a, rng, sigma):
    c = a.copy()
    n = c.size
    w = int(rng.integers(max(2, n // 80), max(3, n // 4)))
    s = int(rng.integers(0, max(1, n - w)))
    c[s:s + w] *= np.exp(rng.normal(0.0, sigma, w))
    return _clamp(c)


def _global_perturb(a, rng, sigma):
    n = a.size
    x = np.arange(n, dtype=np.float64)
    p = np.zeros(n)
    nf = int(rng.integers(1, 5))
    for _ in range(nf):
        f = rng.uniform(0.5, 5.0)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        p += np.sin(2.0 * np.pi * f * x / n + ph)
    p /= max(1.0, np.abs(p).max())
    return _clamp(a * np.exp(sigma * p))

def _optimise_length(n, start, budget_s, seed):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 1.0

    best = _clamp(np.asarray(start, dtype=np.float64).copy())
    best = best if best.size == n else np.ones(n)
    best_fast = _fast_objective(best)
    best_true = _exact_objective(best)
    print(f"  [n={n}] start: fast={best_fast:.5f} true={best_true:.5f}")

    # seed with a sweep of initialisers
    for init in _make_initialisers(rng, n):
        fs = _fast_objective(init)
        if np.isfinite(fs) and fs < best_fast:
            best_fast = fs
            best = init.copy()
        if time.monotonic() > deadline:
            break

    true_best = best.copy()
    true_best_score = best_true
    last_true_check = time.monotonic()
    T = 0.05
    T_min = 1e-4
    cool = 0.9995

    shrink_iters = 0
    while time.monotonic() < deadline:
        # --- deterministic-ish peak shaving (several inner steps) ----------- #
        for _ in range(3):
            step = 0.05 + 0.05 * np.exp(-shrink_iters / 400.0)
            cand = _lag_shrink(best, rng, step)
            fs = _fast_objective(cand)
            if np.isfinite(fs) and fs < best_fast:
                best, best_fast = cand, fs
            shrink_iters += 1
        # --- stochastic smooth move with annealed acceptance --------------- #
        move = _block_perturb(best, rng, 0.10) if rng.random() < 0.5 \
            else _global_perturb(best, rng, 0.12)
        fs = _fast_objective(move)
        if np.isfinite(fs):
            accept = (fs < best_fast) or \
                (rng.random() < np.exp(-(fs - best_fast) / max(T, T_min)))
            if accept:
                best, best_fast = move, fs
            T = max(T_min, T * cool)

        # --- periodic exact confirmation ----------------------------------- #
        if time.monotonic() - last_true_check > 3.0 or best_fast < best_true - 1e-6:
            ts = _exact_objective(best)
            if np.isfinite(ts) and ts < true_best_score:
                true_best_score = ts
                true_best = best.copy()
                print(f"    n={n} improved true -> {true_best_score:.5f} (fast {best_fast:.5f})")
            last_true_check = time.monotonic()

    ts = _exact_objective(best)
    if np.isfinite(ts) and ts < true_best_score:
        true_best_score = ts
        true_best = best.copy()
    print(f"  [n={n}] done: best_true={true_best_score:.5f}")
    return true_best, true_best_score
