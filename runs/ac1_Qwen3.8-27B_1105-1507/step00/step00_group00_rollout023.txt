import numpy as np
from scipy.optimize import minimize
import time

def fast_score(a):
    """Compute the score without calling evaluate_sequence (faster)."""
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)

def bowl_initial(n, depth=1.0):
    """Generate a bowl-shaped initial sequence."""
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / (n - 1) - 1.0  # -1 to 1
    return 1.0 + depth * t * t

def optimize_sequence(n, init, max_iter=1000, time_limit=100):
    """Optimize a sequence of length n using Powell's method."""
    def objective(x):
        a = np.maximum(0, x)
        s = np.sum(a)
        if s < 0.01:
            return 1e10
        conv = np.convolve(a, a)
        return np.max(conv) / (s * s)
    
    result = minimize(objective, init, method='Powell',
                      options={'maxiter': max_iter, 'xtol': 1e-10, 'ftol': 1e-10})
    best = np.maximum(0, result.x)
    return best, fast_score(best)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    deadline = time.monotonic() + budget_s - 10.0
    rng = np.random.default_rng(seed)
    
    best_score = float(evaluate_sequence(height_sequence_1))
