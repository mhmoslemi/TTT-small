Actually, let me reconsider the problem structure. The key insight is:

For a sequence a of length n with sum 1, the convolution a*a has sum 1 and length 2n-1. We want to minimize its maximum.

The "ideal" convolution would be 1/(2n-1) at every position. The minimum possible maximum is 1/(2n-1) (by averaging), and the objective would be 2n/(2n-1) ≈ 1.

This is a well-studied problem in signal processing. Sequences with flat autocorrelation are called "low-autocorrelation sequences" or "Zadoff-Chu sequences" (though those are complex-valued).

For real non-negative sequences, the problem is more constrained but still has good solutions.

import numpy as np
import time

def compute_conv_max_and_grad(a):
    """Compute max of convolution and the gradient at that position."""
    conv = np.convolve(a, a)
    k_max = np.argmax(conv)
    max_val = conv[k_max]
    # Gradient of conv[k_max] w.r.t. a
    # grad[m] = 2 * a[k_max - m] if 0 <= k_max - m < len(a), else 0
    n = len(a)
    grad = np.zeros(n)
    valid = np.arange(max(0, k_max - n + 1), min(n, k_max + 1))
    grad[valid] = 2 * a[k_max - valid]
    return max_val, k_max, grad

def local_search(a, deadline, verbose=False):
    """Gradient-guided local search to minimize max(conv) subject to sum=1, a>=0."""
    a = a.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s  # Normalize to sum = 1
    
    best_a = a.copy()
    best_conv_max = np.max(np.convolve(a, a))
    
    lr = 0.1  # Learning rate
    patience = 0
    max_patience = 100
    
    while time.time() < deadline:
        max_val, k_max, grad = compute_conv_max_and_grad(a)
        
        # Gradient step (negate gradient since we want to minimize)
        # But we need to be careful: the gradient points in the direction of increasing conv[k_max]
        # So we want to go in the opposite direction
        step = lr * grad
        a_new = a - step
        a_new = np.clip(a_new, 0, None)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
        
        # Evaluate
        new_conv_max = np.max(np.convolve(a_new, a_new))
        
        if new_conv_max < best_conv_max:
            best_a = a_new.copy()
            best_conv_max = new_conv_max
            a = a_new
            patience = 0
        else:
            a = a_new  # Still accept to explore, or reject?
            patience += 1
            if patience > max_patience:
                # Restart with perturbation
                a = best_a + 0.1 * np.random.randn(len(a))
                a = np.clip(a, 0, None)
                a = a / a.sum()
                patience = 0
            lr *= 0.95
        
        # Periodically try random perturbations
        if np.random.random() < 0.1:
            perturbation = np.random.randn(len(a)) * 0.05
            a_pert = a + perturbation
            a_pert = np.clip(a_pert, 0, None)
            if a_pert.sum() > 1e-10:
                a_pert = a_pert / a_pert.sum()
                if np.max(np.convolve(a_pert, a_pert)) < best_conv_max:
                    best_a = a_pert.copy()
                    best_conv_max = np.max(np.convolve(a_pert, a_pert))
    
    return best_a, best_conv_max

            lr = max(lr * 0.5, 0.01)
            stagnation = 0

        # Choose action: gradient step, water-fill, or random
        r = np.random.random()
        if r < 0.5:
            # Multi-peak gradient step
            grad, _ = _multipeak_gradient(a, top_m=min(20, n // 4 + 1))
            a = a - lr * grad
            a = _normalize(a)
        elif r < 0.75:
            # Water-filling
            a = _water_fill_step(a, strength=0.2)
        else:
            # Random multiplicative perturbation
            a = a * np.exp(0.05 * np.random.randn(n))
            a = _normalize(a)

        # Adaptive learning rate
        if stagnation % 30 == 0 and stagnation > 0:
            lr *= 0.92
        if lr < 0.001:
            lr = 0.001

    return best_a, best_val


def _coordinate_descent(a, deadline, verbose=False):
    """Coordinate descent: optimize one element at a time."""
    a = _normalize(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    conv = np.convolve(a, a)
    best_val = conv.max()

    while time.time() < deadline:
        improved = False
        for m in range(n):
            if time.time() > deadline:
                break
            # Try a few values for a[m]
            old_val = a[m]
            # The optimal value for a[m] given fixed k_max:
            conv = np.convolve(a, a)
            k = np.argmax(conv)
            km = k - m
            if 0 <= km < n and km != m:
                b = a[km]
                # C = conv[k] - 2*a[m]*b (remove contribution of a[m])
                C = conv[k] - 2.0 * a[m] * b
                S = 1.0 - a[m]  # sum of others
                x_opt = S - C / b if b > 1e-15 else 0.0
            elif km == m:
                C = conv[k] - a[m] ** 2
                S = 1.0 - a[m]
                x_opt = C / S if S > 1e-15 else 0.0
            else:
                x_opt = old_val

            x_opt = max(0.0, min(x_opt, 10.0))
            # Try candidates around x_opt
            candidates_x = [max(0.0, x_opt * 0.5), max(0.0, x_opt), max(0.0, x_opt * 1.5),
                            max(0.0, old_val * 0.5), max(0.0, old_val * 1.5)]
            best_x = old_val
            best_c = _conv_max(a)[0]
            for xc in candidates_x:
                a[m] = xc
                a = _normalize(a)
                c = _conv_max(a)[0]
                if c < best_c:
                    best_c = c
                    best_x = xc
            a[m] = best_x
            a = _normalize(a)

            if _conv_max(a)[0] < best_val:
                best_val = _conv_max(a)[0]
                best_a = a.copy()
                improved = True

        if not improved:
            # Try random coordinate perturbation
            m = np.random.randint(n)
            a[m] *= np.exp(np.random.randn() * 0.3)
            a = _normalize(a)

    return best_a, best_val


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses multi-start gradient-guided local search with water-filling and
    coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 15

    best_score = np.inf
    best_seq = None

    def try_sequence(seq):
        nonlocal best_score, best_seq
        try:
            score = evaluate_sequence(seq.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = np.array(seq, dtype=np.float64).copy()
                elapsed = time.time() - start_time
                print(f"  New best: {score:.6f} (n={len(seq)}, t={elapsed:.0f}s)")
                return True
        except Exception:
            pass
        return False

    # Get parent sequence if available
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0)
        else:
            parent = None
    except Exception:
        parent = None

    # Generate diverse initial sequences
    initial_seqs = []

    # From parent (subsampled to various lengths)
    if parent is not None and len(parent) > 200:
        for target_len in [512, 1024, 2048]:
            if len(parent) >= target_len:
                # Interpolate subsample
                indices = np.linspace(0, len(parent) - 1, target_len).astype(int)
                initial_seqs.append(parent[indices])
            elif len(parent) > 100:
                # Stretch
                from numpy import interp
                x_old = np.arange(len(parent))
                x_new = np.linspace(0, len(parent) - 1, target_len)
                initial_seqs.append(interp(x_new, x_old, parent))

    # Standard diverse initializations
    for n in [512, 1024, 2048]:
        x = np.arange(n) / n
        initial_seqs.append(np.ones(n))
        initial_seqs.append(np.random.dirichlet(np.ones(n) * 0.5))
        initial_seqs.append(np.maximum(np.sin(np.pi * x) ** 0.5, 0.01))
        initial_seqs.append(np.maximum(np.cos(np.pi * x) ** 2 + 0.3, 0.01))
        initial_seqs.append(np.exp(-2.0 * (x - 0.5) ** 2 / 0.05))
        # Asymmetric
        initial_seqs.append(np.exp(-3.0 * x))
        initial_seqs.append(np.exp(-3.0 * (1.0 - x)))
        # Two-hump
        initial_seqs.append(np.maximum(
            np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02), 0.01))
        # Geometric with noise
        initial_seqs.append(np.exp(-0.5 * np.arange(n) * 0.01) + 0.1 * np.random.rand(n))

    print(f"Total initial sequences: {len(initial_seqs)}")
    print(f"Time budget: {budget_s - 15}s, deadline in {deadline - time.time():.0f}s")

    # Phase 1: Quick evaluation of all initial sequences (find good starting region)
    quick_deadline = min(time.time() + 0.5, deadline)
    for seq in initial_seqs:
        if time.time() > quick_deadline:
            break
        seq = _normalize(seq)
        try_sequence(seq)

    # Phase 2: Local optimization on top candidates
    # Sort initial sequences by their convolution max
    scored = []
    for seq in initial_seqs:
        seq = _normalize(seq)
        cm = _conv_max(seq)[0]
        scored.append((cm, seq))
    scored.sort(key=lambda x: x[0])

    # Optimize top candidates
    top_k = min(8, len(scored))
    time_per_candidate = (deadline - time.time() - 60) / max(1, top_k)
    time_per_candidate = min(time_per_candidate, 200)

    for i in range(top_k):
        if time.time() > deadline - 30:
            break
        cand = scored[i][1].copy()
        cand_deadline = min(time.time() + time_per_candidate, deadline - 30)
        print(f"Optimizing candidate {i+1}/{top_k} (n={len(cand)}, alloc={time_per_candidate:.0f}s)...")
        opt_seq, opt_val = _local_optimize(cand, cand_deadline)
        try_sequence(opt_seq)
        # Also try coordinate descent on the result
        if time.time() < deadline - 30:
            cd_deadline = min(time.time() + 30, deadline - 15)
            cd_seq, cd_val = _coordinate_descent(opt_seq, cd_deadline)
            try_sequence(cd_seq)

    # Phase 3: Intensive optimization on the best found
    if best_seq is not None and time.time() < deadline - 20:
        print(f"\nPhase 3: Intensive optimization of best (score={best_score:.6f})...")
        int_deadline = min(time.time() + (deadline - time.time()) * 0.6, deadline - 20)
        opt_seq, opt_val = _local_optimize(best_seq, int_deadline)
        try_sequence(opt_seq)

        # Final coordinate descent polish
        if time.time() < deadline - 15:
            cd_deadline = deadline - 10
            cd_seq, cd_val = _coordinate_descent(opt_seq, cd_deadline)
            try_sequence(cd_seq)

    # Phase 4: Try additional longer sequences if time remains
    if time.time() < deadline - 30:
        for n in [4096, 8192]:
            if time.time() > deadline - 30:
                break
            seq = _normalize(np.random.dirichlet(np.ones(n) * 0.3))
            seq_deadline = min(time.time() + 60, deadline - 20)
            opt_seq, _ = _local_optimize(seq, seq_deadline)
            try_sequence(opt_seq)

    # Final verification
    if best_seq is not None:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"\nFinal result: score={final_score:.6f}, n={len(best_seq)}")
        if np.isfinite(final_score):
            return [float(x) for x in best_seq]

    # Fallback: uniform sequence
    fallback = np.ones(512)
    print("Using fallback uniform sequence")
    return [float(x) for x in fallback]
