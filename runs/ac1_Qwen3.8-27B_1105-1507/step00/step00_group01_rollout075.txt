        cn = optimize_sequence(cn, n, s, state, phase3_deadline, rng, max_iters=50000)

    print(f"Phase 3 done in {time.monotonic()-t0:.1f}s. Best={state['best_score']:.6f}")

    # --- Phase 4: Resample best shape to different lengths ---
    print("Phase 4: Length resampling...")
    t0 = time.monotonic()

    if state['best_c'] is not None:
        best_shape = state['best_c'] / state['best_c'].max()
        for n_new in [2048, 3072, 4096, 5120, 6121, 7168, 8192, 10240]:
            if time.monotonic() > total_deadline - 10:
                break
            x_old = np.linspace(0, 1, len(best_shape))
            x_new = np.linspace(0, 1, n_new)
            new_shape = np.interp(x_new, x_old, best_shape)
            new_shape = new_shape * (state['best_c'].sum() / new_shape.sum())
            try_evaluate(new_shape, state)

    print(f"Phase 4 done in {time.monotonic()-t0:.1f}s. Best={state['best_score']:.6f}")

    # --- Phase 5: Final fine-tuning ---
    print("Phase 5: Fine-tuning...")
    t0 = time.monotonic()

    if state['best_c'] is not None:
        cn = state['best_c'] / state['best_c'].sum()
        n = len(cn)
        s = state['best_c'].sum()
        cn = fine_tune(cn, n, s, state, total_deadline, rng, max_iters=100000)

    print(f"Phase 5 done in {time.monotonic()-t0:.1f}s. Best={state['best_score']:.6f}")

    print(f"FINAL: best_score = {state['best_score']:.6f}, n = {len(state['best_c'])}")

    if state['best_c'] is None:
        return [0.5] * 64  # fallback

    return [float(x) for x in state['best_c']]
