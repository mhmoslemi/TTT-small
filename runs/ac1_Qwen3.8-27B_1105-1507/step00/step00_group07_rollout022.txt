def _fast_score(a):
    """Compute the score without security checks (faster than evaluate_sequence)."""
    n = a.size
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S*S)

def _local_search(seq, deadline, rng):
    """Multi-scale local search with subgradient guidance."""
    n = seq.size
    score = _fast_score(seq)
    best = seq.copy()
    
    # Determine block sizes (from large to small)
    block_sizes = []
    bs = max(1, n // 4)
    while bs >= 1:
        block_sizes.append(bs)
        bs = max(1, bs // 2)
    
    for bs in block_sizes:
        std = 0.2
        patience = 0
        max_patience = 500 + 100 * (n // 1024)
        
        while patience < max_patience:
            if time.monotonic() >= deadline:
                return best, _fast_score(best)
            
            # Choose perturbation type
            r = rng.random()
            candidate = best.copy()
            
            if r < 0.5:
                # Block multiplicative perturbation
                start = rng.integers(0, max(1, n - bs + 1))
                end = min(n, start + bs)
                candidate[start:end] *= np.exp(rng.normal(0, std, end - start))
            elif r < 0.7:
                # Subgradient-guided: reduce values contributing to peak
                c = np.convolve(best, best)
                peak_idx = c.argmax()
                # The peak is at convolution index peak_idx
                # c[peak_idx] = sum_i best[i] * best[peak_idx - i]
                # Reduce the largest contributors
                contribs = []
                for i in range(max(0, peak_idx - n + 1), min(n, peak_idx + 1)):
                    j = peak_idx - i
                    if 0 <= j < n:

                    i, _ = contribs[rng.integers(len(contribs))]
