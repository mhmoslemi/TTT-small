def full_conv(s):
    """Full linear convolution of s with itself."""
    n = len(s)
    N = 2 * n
    S = np.fft.rfft(s, N)
    return np.fft.irfft(S * S, N)[:2*n-1]

def simplex_project(s, total):
    """Project s onto {x >= 0, sum x = total}."""
    n = len(s)
    v = np.sort(s)[::-1]
    cssv = np.cumsum(v)
    rho = np.searchsorted(cssv, total)  # find where cssv exceeds total
    if rho >= n:
        rho = n - 1
    theta = (cssv[rho] - total) / (rho + 1)
