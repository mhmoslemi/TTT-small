import numpy as np
import time

def fast_convolve(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)[:size]
    return c

def project_simplex(a):
    """Project a onto the probability simplex."""
    n = len(a)
    # Bisection on tau
    lo, hi = a.min(), a.max()
    for _ in range(60):
        tau = (lo + hi) / 2
        s = np.sum(np.maximum(a - tau, 0))
        if s > 1:
            lo = tau
        else:
            hi = tau
    return np.maximum(a - (lo + hi) / 2, 0)

def subgradient_step(a, c, k_star, eta):
    """Take a subgradient step to reduce c[k_star]."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    g[lo:hi+1] = 2 * a[k_star - lo : k_star - hi : -1]

    # Let me just compute it directly
    g = np.zeros(n)
    for i in range(lo, hi + 1):
        g[i] = 2 * a[k_star - i]
    return a - eta * g

Wait, there's an issue. The subgradient `g[i] = 2*a[k_star - i]` is the gradient of `c[k_star]` with respect to `a[i]`. But we want to MINIMIZE `c[k_star]`, so we should subtract the gradient. That's what we're doing: `a = a - eta * g`.

But the issue is that reducing `c[k_star]` might increase other `c[k']` values. This is the fundamental challenge of non-convex optimization.

A better approach might be to use the **maximum subgradient**:
`g[i] = sum_{k: c[k] = c_max} 2*a[k-i] / (number of maximizing k's)`

Or even better, use a **smoothed maximum**:
`f_smooth(a) = (1/T) * log(sum_k exp(T * c[k] / (sum a)^2))`

For large T, this approximates the max, and it's smooth.

The gradient of `f_smooth` is:
`df/da[i] = sum_k p_k * dc[k]/da[i]` where `p_k = exp(T*c[k]) / sum_j exp(T*c[j])`

This gives a "soft" subgradient that accounts for all near-maximal lags.

For T = 10 or 20, this should be a good approximation.

Let me implement this:

```python
def smooth_obj_grad(a, T=10):
    """Compute the gradient of the smoothed objective."""
    n = len(a)
    c = fast_convolve(a)  # length 2n-1
    s = a.sum()
    
    # Softmax weights
    c_scaled = T * c / (s * s)  # Scale by T/s^2
    c_scaled -= c_scaled.max()  # For numerical stability
    p = np.exp(c_scaled)
    p /= p.sum()
    
    # Gradient: sum_k p[k] * dc[k]/da[i]
    # dc[k]/da[i] = 2*a[k-i] if 0 <= k-i < n
    # So df/da[i] = 2 * sum_k p[k] * a[k-i] * 1[0<=k-i<n]
    # = 2 * (p * a)_circular... no, it's a partial correlation
    
    # Let's compute it as: for each i, df/da[i] = 2 * sum_{k: 0<=k-i<n} p[k] * a[k-i]
    # = 2 * sum_{j: 0<=j<n} p[i+j] * a[j]  (substituting j = k-i)
    
    # This is a correlation of p and a!
    # df/da[i] = 2 * sum_j p[i+j] * a[j] for j = 0..n-1, where p is defined for 0..2n-2
    
    g = np.zeros(n)
    for i in range(n):
        # Sum over j: p[i+j] * a[j]
        # i+j ranges from i to i+n-1, which is within [0, 2n-2] since i < n
        g[i] = 2 * np.dot(p[i:i+n], a)
    
    return g
```
