import numpy as np
import time
from scipy.optimize import minimize_scalar

"""
Minimizes 2n*max(convolve(a,a))/sum(a)^2 using subgradient descent from a linear ramp initialization.
The linear ramp a[j]=j+1 achieves score 4(n+2)/(3(n+1)) ≈ 4/3 for large n, well below the target.
We refine with subgradient descent using active-set averaging and adaptive step sizes.
"""


def compute_score_fast(a):
    """Compute the evaluation score without calling evaluate_sequence."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    B = np.max(conv)
    return float(2.0 * n * B / (S * S))


def subgradient_step(a, lr, active_frac=0.98):
    """
    One subgradient descent step on score = 2n*max(conv)/S^2.
    Returns updated a and the score.
    """
    n = len(a)
    conv = np.convolve(a, a)
    S = float(np.sum(a))
    B = float(np.max(conv))
    k_star = int(np.argmax(conv))
    score = 2.0 * n * B / (S * S)

    # Determine active set
    threshold = active_frac * B
    active_indices = np.where(conv >= threshold)[0]
    active_count = len(active_indices)

    if active_count == 0:
        active_indices = np.array([k_star])
        active_count = 1

    # Compute average subgradient of B w.r.t. a
    # For each active k: d(conv[k])/d(a[j]) = 2*a[k-j] for valid j
    # We need: grad_conv[j] = (1/|A|) * sum_{k in A} 2*a[k-j]
    grad_conv = np.zeros(n)

    if active_count == 1:
        # Fast path: single active constraint
        k = k_star
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        idx = k - np.arange(lo, hi + 1)
        grad_conv[lo:hi + 1] = 2.0 * a[idx]
    else:
        # Accumulate over active set (vectorized per k)
        for k in active_indices:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            idx = k - np.arange(lo, hi + 1)
            grad_conv[lo:hi + 1] += 2.0 * a[idx]
        grad_conv /= active_count

    # score = 2nB/S^2
    # d(score)/d(a_j) = (2n/S^2)*grad_conv[j] - 4nB/S^3
    # Descent: a_j -= lr * d(score)/d(a_j)
    #         = a_j + lr * (4nB/S^3 - 2n*grad_conv[j]/S^2)

    neg_grad = (4.0 * n * B / (S ** 3)) - (2.0 * n / (S * S)) * grad_conv

    a_new = a + lr * neg_grad
    a_new = np.clip(a_new, 0.0, None)
    new_S = a_new.sum()
    if new_S < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new / new_S  # renormalize to sum=1

    return a_new, score


def linear_ramp(n, scale=1.0):
    """Create a linear ramp sequence normalized to sum=1."""
    a = np.arange(1, n + 1, dtype=np.float64)
    a = a / a.sum()
    return a * scale


def power_ramp(n, p, scale=1.0):
    """Create a power-law ramp a[j] = (j+1)^p normalized to sum=1."""
    a = np.arange(1, n + 1, dtype=np.float64) ** p
    a = a / a.sum()
    return a * scale


def optimized_ramp(n, scale=1.0):
    """
    Try to find a better shape by optimizing a parametric family.
    a[j] = (j+1) * (1 + alpha * (2*(j+0.5)/n - 1))
    This 'bends' the linear ramp.
    """
    def param_score(alpha):
        j = np.arange(n, dtype=np.float64) + 1
        center_correction = 2.0 * (j - 0.5) / n - 1.0
        a = j * (1.0 + alpha * center_correction)
        a = np.clip(a, 0.0, None)
        if a.sum() < 1e-10:
            return np.inf
        a = a / a.sum()
        conv = np.convolve(a, a)
        B = np.max(conv)
        return 2.0 * n * B / (a.sum() ** 2)

    result = minimize_scalar(param_score, bounds=(-0.5, 0.5), method='bounded',
                             options={'xatol': 1e-6, 'maxiter': 200})
    alpha_opt = result.x
    j = np.arange(n, dtype=np.float64) + 1
    center_correction = 2.0 * (j - 0.5) / n - 1.0
    a = j * (1.0 + alpha_opt * center_correction)
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-10:
        a = np.arange(1, n + 1, dtype=np.float64)
    a = a / a.sum()
    return a * scale


def run_gradient_descent(a_init, budget_s, rng, active_frac=0.98):
    """Run subgradient descent from a given initialization."""
    n = len(a_init)
    a = a_init.copy()
    start_time = time.monotonic()
    deadline = start_time + budget_s

    best_a = a.copy()
    best_score = compute_score_fast(a)

    lr = 0.005
    no_improve = 0
    consecutive_bad = 0

    for it in range(500000):
        if time.monotonic() > deadline:
            break

        a, score = subgradient_step(a, lr, active_frac)

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            consecutive_bad = 0
            if it % 2000 == 0:
                print(f"  GD iter {it}: score = {best_score:.6f}")
        else:
            no_improve += 1
            if score > best_score * 1.001:
                consecutive_bad += 1
            else:
                consecutive_bad = 0

        # Adaptive learning rate
        if consecutive_bad > 100:
            lr = max(lr * 0.7, 1e-8)
            consecutive_bad = 0
        elif it > 0 and it % 500 == 0 and no_improve < 500:
            lr = min(lr * 1.05, 1.0)

        # Reset if stuck for too long
        if no_improve > 5000:
            a = best_a.copy()
            # Add perturbation to escape
            noise = rng.normal(0, 0.005, n)
            a = a + noise
            a = np.clip(a, 0.0, None)
            if a.sum() > 1e-10:
                a = a / a.sum()
            else:
                a = best_a.copy()
            lr = max(lr * 2, 0.001)
            no_improve = 0
            consecutive_bad = 0
