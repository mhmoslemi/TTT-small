"""
Minimizes 2n*max(a*a)/sum(a)^2 over non-negative sequences using
smooth (log-sum-exp) approximation of the max with projected gradient
descent on the probability simplex, multiple diverse initializations,
and adaptive annealing of the smoothing parameter.

import numpy as np
import time


def simplex_project(v):
    """Project vector v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    sorted_v = np.sort(v)[::-1]
    j = np.arange(1, n + 1, dtype=np.float64)
    mask = sorted_v + 1.0 / j > 0
    rho = int(np.sum(mask))
    if rho == 0:
        return np.ones(n) / n
    tau = (np.sum(sorted_v[:rho]) - 1.0) / rho
    return np.maximum(v - tau, 0.0)


def compute_smooth_obj_and_grad(a, beta):
    """
    Compute smooth objective 2n/beta * ln(sum_k exp(beta*c_k)) and its gradient.
    a is on the simplex (sum=1, a>=0).
    Returns (obj_value, gradient_vector).
    """
    n = len(a)
    c = np.convolve(a, a)
    c_max = np.max(c)
    beta_c = beta * (c - c_max)
    exp_bc = np.exp(beta_c)
    sum_exp = np.sum(exp_bc)
    weights = exp_bc / sum_exp

    # g[i] = sum_k weights[k] * a[k-i] (zero-padded)
    # Computed via: g[i] = (a * w_rev)[2n-2-i]
    w_rev = weights[::-1]
    conv_full = np.convolve(a, w_rev)
    g = conv_full[2 * n - 2 : n - 2 : -1]

    grad = (4.0 * n / beta) * g
    obj = 2.0 * n * (c_max + np.log(sum_exp) / beta)
    return obj, grad


def run_smooth_descent(a0, n, beta_start, beta_end, total_iters, time_limit):
    """
    Run projected gradient descent with annealed beta.
    Returns (best_a, best_exact_score).
    """
    a = simplex_project(np.maximum(a0, 0.0))
    t_start = time.monotonic()

    # Evaluate initial
    init_score = float(evaluate_sequence(a.tolist()))
    best_a = a.copy()
    best_score = init_score

    # Adaptive step size
    step_scale = 0.5

    for it in range(total_iters):
        elapsed = time.monotonic() - t_start
        if elapsed > time_limit:
            break

        # Anneal beta
        frac = it / max(1, total_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        obj, grad = compute_smooth_obj_and_grad(a, beta)

        # Gradient norm for adaptive step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            if (it + 1) % 200 == 0:
                sc = float(evaluate_sequence(a.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
            continue

        # Backtracking line search (a few tries)
        lr = step_scale / (grad_norm + 1e-15)
        improved = False
        for trial in range(8):
            v = a - lr * grad
            a_new = simplex_project(v)
            if np.allclose(a_new, a, atol=1e-15):
                break
            # Smooth objective check (cheap)
            obj_new, _ = compute_smooth_obj_and_grad(a_new, beta)
            if obj_new < obj - 1e-10:
                a = a_new
                improved = True
                break
            lr *= 0.5
        if not improved:
            step_scale *= 0.9
        else:
            step_scale = min(step_scale * 1.05, 5.0)

    sc = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(sc):

    return best_a, best_score

def structured_start(n, kind, param, rng=None):
    """Generate a structured starting sequence on the simplex."""
    x = np.zeros(n)
    if kind == 'uniform':
        x[:] = 1.0
    elif kind == 'linear':
        x[:] = np.arange(n, 0, -1, dtype=np.float64)
        x[:] = np.power(x, param)
    elif kind == 'exponential':
        x[:] = np.exp(-param * np.arange(n, dtype=np.float64) / n)
    elif kind == 'cosine':
        x[:] = np.cos(np.pi * np.arange(n) / (2.0 * n)) ** param
        x[:] = np.maximum(x, 0.0)
    elif kind == 'two_block':
        m = n // 2
        x[:m] = param
        x[m:] = 1.0
    elif kind == 'three_block':
        m1 = n // 3
        m2 = 2 * n // 3
        x[:m1] = param
        x[m1:m2] = 1.0
        x[m2:] = 0.3
    elif kind == 'ramp_down':
        x[:] = 1.0 - (np.arange(n) / max(1, n - 1)) ** param

    elif kind == 'bump':

        x[:] = 1.0

    x = np.maximum(x, 0.0)
    s = np.sum(x)
    if s < 1e-10:
        return np.ones(n) / n

    """
    Searches for a non-negative sequence minimizing 2n*max(a*a)/sum(a)^2.
    Uses smooth approximation gradient descent with multiple initializations.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(0.0, float(budget_s) - 10.0)
    t_global_start = time.monotonic()

    # Determine sequence length from parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    except (NameError, AttributeError):
        n = 2535
        parent = np.ones(n)

    # Ensure parent is valid
    parent = np.maximum(np.asarray(parent, dtype=np.float64), 0.0)
    if parent.size != n:
        n = parent.size
    S = np.sum(parent)
    if S < 1e-10:
        a_parent = np.ones(n) / n
    else:
        a_parent = parent / S

    # Global best tracking
    best_a = a_parent.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    print(f"n={n}, initial parent score: {best_score:.6f}", flush=True)

    # Try uniform
    a_uni = np.ones(n) / n
    sc = float(evaluate_sequence(a_uni.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_uni.copy()
        print(f"Uniform: {best_score:.6f}", flush=True)

    # Define a set of structured starts
    start_configs = [
        ('linear', 0.5), ('linear', 1.0), ('linear', 2.0), ('linear', 3.0), ('linear', 5.0),
        ('exponential', 1.0), ('exponential', 2.0), ('exponential', 5.0), ('exponential', 10.0), ('exponential', 20.0),
        ('cosine', 1.0), ('cosine', 2.0), ('cosine', 4.0),
        ('two_block', 1.5), ('two_block', 2.0), ('two_block', 3.0), ('two_block', 5.0), ('two_block', 10.0),
        ('three_block', 2.0), ('three_block', 4.0),
        ('ramp_down', 1.0), ('ramp_down', 2.0), ('ramp_down', 3.0),
        ('bump', 2.0), ('bump', 4.0),
    ]

    # Add some random smooth starts
    for _ in range(6):
        pass  # handled below

    total_starts = len(start_configs) + 6  # structured + random
    per_start_budget = total_budget / total_starts
    # Give more budget to first few (likely better starts)
    per_start_budget = max(2.0, per_start_budget)

    start_idx = 0
    for kind, param in start_configs:
        if time.monotonic() - t_global_start > total_budget - 30.0:
            break
        a0 = structured_start(n, kind, param, rng)
        start_idx += 1
        print(f"[{start_idx}/{total_starts}] Trying {kind}({param}), budget={per_start_budget:.1f}s", flush=True)

        a_opt, score = run_smooth_descent(a0, n, beta_start=10.0, beta_end=200.0,
                                          total_iters=300, time_limit=per_start_budget)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  IMPROVED: {best_score:.6f}", flush=True)

    # Random smooth starts
    for r in range(6):
        if time.monotonic() - t_global_start > total_budget - 30.0:
            break
        a0 = random_smooth_start(n, rng, smoothness=30 + 20 * r)
        start_idx += 1
        print(f"[{start_idx}/{total_starts}] Random smooth {r+1}, budget={per_start_budget:.1f}s", flush=True)

        a_opt, score = run_smooth_descent(a0, n, beta_start=10.0, beta_end=200.0,
                                          total_iters=300, time_limit=per_start_budget)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  IMPROVED: {best_score:.6f}", flush=True)

    # Intensive refinement from the best solution found
    remaining = total_budget - (time.monotonic() - t_global_start)
    print(f"\nRefinement phase: {remaining:.1f}s remaining, current best={best_score:.6f}", flush=True)

    if remaining > 20.0:
        # Phase 1: fine-tune from best with high beta
        a_opt, score = run_smooth_descent(best_a.copy(), n, beta_start=100.0, beta_end=5000.0,
                                          total_iters=500, time_limit=remaining * 0.4)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  Refine 1: {best_score:.6f}", flush=True)

        # Phase 2: perturb and re-optimize
        remaining = total_budget - (time.monotonic() - t_global_start)
        if remaining > 20.0:
            n_pert = max(1, int(remaining / 30.0))
            for p in range(n_pert):
                if time.monotonic() - t_global_start > total_budget - 10.0:
                    break
                a_pert = perturb_and_reoptimize(best_a, n, rng, 10.0)
                a_opt, score = run_smooth_descent(a_pert, n, beta_start=20.0, beta_end=1000.0,
                                                  total_iters=200, time_limit=20.0)
                if score < best_score:
                    best_score = score
                    best_a = a_opt.copy()
                    print(f"  Perturb {p+1}: {best_score:.6f}", flush=True)

        # Phase 3: final polish
        remaining = total_budget - (time.monotonic() - t_global_start)
        if remaining > 10.0:
            a_opt, score = run_smooth_descent(best_a.copy(), n, beta_start=200.0, beta_end=10000.0,
                                              total_iters=300, time_limit=remaining * 0.8)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  Final polish: {best_score:.6f}", flush=True)

    # Prepare final result
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    final_score = float(evaluate_sequence(result))
    print(f"\n=== FINAL: score={final_score:.6f}, n={n} ===", flush=True)

    return result
