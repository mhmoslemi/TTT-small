def _evaluate_fast(a):
    """Fast evaluation without the overhead of list conversion."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def _linear_ramp(n, scale=1000.0):
    """Generate a linear ramp from 0 to scale."""
    return scale * np.arange(n, dtype=np.float64) / float(n - 1)


def _power_ramp(n, p, scale=1000.0):
    """Generate a power ramp: (i/(n-1))^p * scale."""
    x = np.arange(n, dtype=np.float64) / float(n - 1)
    return scale * x ** p


def _shifted_ramp(n, shift_frac, scale=1000.0):
    """Ramp that starts at shift_frac*n and goes to 1000 at the end."""

    if start < n - 1:
        a[start:] = scale * np.arange(n - start, dtype=np.float64) / float(n - 1 - start)
    return a


def _subgradient_step(a, lr, top_k=20):
    """One subgradient step to reduce max(conv(a,a))."""
    n = len(a)
    b = np.convolve(a, a)

    g = np.zeros(n, dtype=np.float64)
