import numpy as np

from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)[:2*n-1]
    return result


def fast_score(a):
    """Fast computation of the score."""
    n = len(a)
    c = fast_autocorr(a)
    max_c = c.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def build_conv_index_arrays(n):
    """Precompute sparse index arrays for the convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        rows_list.extend([k] * (j_hi - j_lo + 1))
        cols_list.extend(range(j_lo, j_hi + 1))
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp_direction(f, M, rows, cols, n, time_limit=20):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    data = f[rows - cols].astype(np.float64)
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return res.x
    return None
