    """Solve the LP: max 1^T g s.t. (f*g)[m] <= M for all m, g >= 0"""
    n = len(f)
    # The constraint matrix A has shape (2n-1, n)
    # A[m, j] = f[m-j] if 0 <= m-j < n, else 0

    rows = []
    cols = []
    vals = []

                cols.append(j)

def _smoothed_gradient(f, temperature):
    """Compute smoothed gradient of max(f*f) w.r.t. f."""
    n = len(f)
    conv = np.convolve(f, f)
    
    # Compute softmax weights (numerically stable)
    max_conv = conv.max()
    exp_conv = np.exp((conv - max_conv) / temperature)
    softmax = exp_conv / exp_conv.sum()
    
    # Gradient: sum_m softmax[m] * 2*f[m-i] (for valid m-i)
    grad = np.zeros(n)

    # grad[i] = 2 * sum_m softmax[m] * f[m-i] * I(0 <= m-i < n)
    # This is 2 * (softmax * f)[i] in a circular sense, but with truncation
    # For now, use the direct computation which is O(n^2) but with small constants

    # The operation is: for each m, add softmax[m] * f[m-i] to grad[i] for valid i
    # This is a convolution-like operation

    # grad[i] = 2 * sum_{m: 0<=m-i<2n-1, 0<=m<2n-1} softmax[m] * f[m-i] * I(0<=m-i<n)
    # = 2 * sum_{m: 0<=m<2n-1, 0<=m-i<n} softmax[m] * f[m-i]
