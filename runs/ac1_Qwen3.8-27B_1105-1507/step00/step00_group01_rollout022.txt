    """Compute gradient of log-sum-exp of convolution values."""
    c = np.convolve(a, a)
    weights = np.exp(c / T - c.max() / T)  # normalize for stability
    weights = weights / weights.sum()
    # grad_i = sum_k weights[k] * a[k-i] (with bounds)
    # This is a weighted convolution: grad = convolve(weights, a) but with appropriate alignment
    # Actually: d c[k] / d a[i] = a[k-i] if 0 <= k-i < n
    # So grad[i] = sum_k weights[k] * a[k-i] (valid k-i)
    # This is the cross-correlation of weights with a
    grad = np.correlate(weights, a, mode='full')
    # But we need to be careful about alignment
    # Actually, let's just compute it directly
    # c[k] = sum_i a[i] * a[k-i] (valid i)
    # d c[k] / d a[i] = a[k-i] (if k-i is valid)
    # grad[i] = sum_k weights[k] * a[k-i] (if k-i is valid)
    
    n = len(a)

import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def peak_reduction_step(a, eta=0.05, top_k=5):
    """One step of peak reduction."""
    n = len(a)
    c = fft_convolve(a, a)
    
    # Find top-k peak positions
    top_positions = np.argsort(c)[-top_k:]

    pressure = np.zeros(n)
