"""
Minimize J(a) = 2n * max(a*a) / (sum a)^2  over non-negative a.

Method
------
Because J is scale invariant we normalize sum(a)=1, so J = 2n * max(conv)
with conv = linear self-convolution.  We descend on a smooth max
(log-sum-exp) of conv via a projected (non-negative, sum=1) gradient step.
Convolution and the gradient (a cross-correlation) are computed with FFTs.
Many diverse initial points are tried; the best is refined by projected
gradient descent with adaptive step / temperature, multiple restarts, and a
time-limited top-K LP polish if a suitable solver is available.  The exact
evaluate_sequence is the arbiter of "best".
"""

import time
import numpy as np


def _next_pow2(m):
    N = 1
    while N < m:
        N *= 2
    return N


def compute_conv(a):
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def smooth_max_and_weights(x, tau):
    xm = float(x.max())
    sh = (x - xm) / tau
    sh = np.clip(sh, -100.0, 100.0)
    es = np.exp(sh)
    s = es.sum()
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(x)
        im = int(np.argmax(x))
        w[im] = 1.0
        return xm, w
    w = es / s
    return xm + tau * np.log(s), w


def compute_grad(a, w, N_grad):
    # grad[i] = 2 * sum_j a[j] * w[i+j];  via c = conv(reverse(a), w)
    n = a.size
    ra = a[::-1]
    f_ra = np.fft.rfft(ra, N_grad)
    f_w = np.fft.rfft(w, N_grad)
    c = np.fft.irfft(f_ra * f_w, N_grad)
    return 2.0 * c[n - 1:2 * n - 1]


def grad_step(a, tau, lr, N_fft, N_grad):
    conv = compute_conv(a)
    sm, w = smooth_max_and_weights(conv, tau)
    g = compute_grad(a, w, N_grad)
    g -= g.mean()            # keeps the sum unchanged
    a_new = a - lr * g
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s < 1e-12:
        a_new = a.copy()
    else:
        a_new = a_new / s
    return a_new, sm, conv


def lp_polish(a, deadline, topK=120):
    # Time-limited LP direction finding (state-of-the-art inspired).
    try:
        from scipy.optimize import linprog
    except Exception:
        return a
    n = a.size
    a = a.copy()
    a = a / max(a.sum(), 1e-12)
    t = 0.2
    while time.monotonic() < deadline:
        conv = compute_conv(a)
        M = float(conv.max())
        idx = np.argsort(conv)[-topK:]
        Arows = []
        for kg in idx:
            row = np.zeros(n)
            lo = max(0, kg - (n - 1))
            hi = min(n, kg + 1)
            for j in range(lo, hi):
                row[j] = a[kg - j]
            Arows.append(row)
        A = np.array(Arows, dtype=np.float64)
        try:
            res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(topK, M),
                          bounds=[(0.0, None)] * n, method="highs")
        except Exception:
            break
        if res is None or not res.success or res.x is None:
            break
        g = np.clip(res.x, 0.0, None)
        if g.sum() <= 1e-12:
            break
        g = g / g.sum()
        improved = False
        for _ in range(6):
            an = (1.0 - t) * a + t * g
            an = np.clip(an, 0.0, None)
            s = an.sum()
            if s < 1e-12:

            an = an / s
            if compute_conv(an).max() < M - 1e-12:
                a = an
                improved = True
                break
            t *= 0.5
        if not improved:
            break
    return a


def _normalize(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-9:
        return None
    return a / s

    base = _normalize(a0)

    x = np.arange(n, dtype=np.float64)

    for k in range(24):
        r = rng.random(n) + 1e-3
        starts.append(("rand%d" % k, r / r.sum()))
    return starts
