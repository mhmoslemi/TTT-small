"""
Minimizes score = 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses alternating LP minimization (fix one factor, optimize the other) 
combined with subgradient steps and local search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    return float(2.0 * n * np.max(b) / (S * S))


def build_lp_matrix(f, n):
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        for m in range(m_lo, m_hi + 1):
            rows.append(k)
            cols.append(m)
            data.append(f[k - m])
        rows.append(k)
        cols.append(n)
        data.append(-1.0)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n + 1))


def solve_lp(f, n, S):
    A_ub = build_lp_matrix(f, n)
    b_ub = np.zeros(2 * n - 1)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    bounds = [(0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x[:n], 0.0)
    return None


def lp_iteration(f, n, S):
    g = solve_lp(f, n, S)
    if g is None:
        return None
    # Renormalize to sum = S
    gsum = np.sum(g)
    if gsum > 0:
        g = g * (S / gsum)
    return g
