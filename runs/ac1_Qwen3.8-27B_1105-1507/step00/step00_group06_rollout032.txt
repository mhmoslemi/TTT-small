Minimizes 2n*max(autocorrelation)/(sum)^2 using subgradient descent on
the sum of top-K autocorrelation peaks. Uses multiple sequence lengths,
diverse initializations (central binomial, power-law, exponential, random),
and adaptive learning rates with momentum.

import time


def compute_grad_topK(a, topK_idx):
    """Compute gradient of sum_{m in topK} c[m] where c = convolve(a,a)."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    for m in topK_idx:

        hi = min(n, m + 1)
        if lo < hi:
            grad[lo:hi] += 2.0 * a[m - hi + 1:m - lo + 1][::-1]

    """Fast evaluation: 2n*max(conv)/(sum^2). Assumes non-negative."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * len(a) * c.max() / (s * s)

    """Central binomial coefficients: C(k) = (2k choose k) / 4^k."""
    a = np.zeros(n, dtype=np.float64)
