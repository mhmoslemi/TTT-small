def compute_conv(a):
    """Compute full convolution a * a using FFT."""
    n = len(a)
    size = 2 * n  # need at least 2n-1, use 2n for power of 2 efficiency
    f = np.fft.rfft(a, size)
    g = np.fft.irfft(f * f, size)
    return g[:2*n-1]

def compute_J(a):
    """Compute J(a) = 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)

    S = np.sum(a)
    if S < 1e-10:

    return 2.0 * n * np.max(conv) / (S * S)

    M = np.max(conv)
    
    # Active set
    active = np.where(np.abs(conv - M) < 1e-10 * M)[0]
    
    # Compute g[i] = (1/|A|) * sum_{k in A} 2*a[k-i] (if 0 <= k-i < n)
    g = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += 2.0 * a[j]
    g /= len(active)
    
    # Full subgradient
    grad = 2.0 * n / (S * S) * g - 4.0 * n * M / (S ** 3)
    return grad, M, S

    a = np.array(a0, dtype=np.float64)

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft

def fast_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2*n-1:
        size *= 2
    f = rfft(a, size)
    return irfft(f * f, size)[:2*n-1]

def objective_and_grad(x, n):
    """Objective function and gradient for optimization."""
    conv = fast_conv(x)
    S = np.sum(x)
    M = np.max(conv)
    J = 2.0 * n * M / (S * S) if S > 1e-10 else 1e10
    
    # Subgradient
    active = np.where(conv >= M - 1e-10)[0]
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += 2.0 * x[k-hi:k-lo+1][::-1]
    g /= len(active)
    
    grad = 2.0 * n * g / (S*S) - 4.0 * n * M / (S**3)
