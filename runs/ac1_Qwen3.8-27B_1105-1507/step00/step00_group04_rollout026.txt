def _fft_convolve(a):
    """Compute linear convolution of a with a using FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2*n-1]

def _project_nonneg_sum_one(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    # Bisection for the water-filling level
    lo = a.min() - 1
    hi = a.max() + 1
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.maximum(0, a - mid).sum()
        if s > 1:
            lo = mid
        else:
            hi = mid

    result = np.maximum(0, a - lam)
    total = result.sum()
    if total < 1e-15:
        return np.ones_like(a) / len(a)
    return result / total

def _gs_step(a, N, relaxation=1.0):

    n = len(a)

    num_bins = len(A)  # = N//2 + 1

    # so it contributes 2*|A[k]|^2 to the total power (except Nyquist which contributes 1*|A[N//2]|^2)

    # If we set |A[k]| = c for all k >= 1:
    # |A[0]|^2 + (num_bins - 1)*c^2 + extra for Nyquist... 
    # Actually for rfft: total power = |A[0]|^2 + 2*sum_{k=1}^{N//2-1}|A[k]|^2 + |A[N//2]|^2
    # If we set all non-DC to c: |A[0]|^2 + 2*(N//2-1)*c^2 + c^2 = N*S2

    p = 1
    while p < n:
        p <<= 1
    return p

def _fft_autocorr(a):
    """Compute np.convolve(a, a) using FFT."""
    n = len(a)
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

def _project(a):
    """Project onto {x >= 0, sum(x) = 1} via water-filling."""
    n = len(a)
    sorted_a = np.sort(a)[::-1]  # descending
    cumsum = np.cumsum(sorted_a)
    # Find rho such that sum(max(0, a - rho)) = 1
    # For the k-th largest element a_(k): if rho is between a_(k+1) and a_(k),
    # then sum = (a_(1) + ... + a_(k)) - k*rho = 1
    # rho = (cumsum[k-1] - 1) / k
    # Valid when a_(k) >= rho >= a_(k+1)
    k = np.arange(1, n + 1)
    rho_candidates = (cumsum - 1.0) / k
    # Find the largest k such that a_(k) >= rho_candidate[k]
    # and (k == n or a_(k+1) <= rho_candidate[k])
    valid = np.zeros(n, dtype=bool)
    valid[0] = sorted_a[0] >= rho_candidates[0]
    for i in range(1, n):
        valid[i] = (sorted_a[i] >= rho_candidates[i])
    # We want the largest k where valid[k-1] is True
    # Actually, let's just use bisection - it's simpler and fast enough
    lo = a.min()
    hi = a.max()
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        s = np.maximum(0.0, a - mid).sum()
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    rho = 0.5 * (lo + hi)
    result = np.maximum(0.0, a - rho)
    total = result.sum()
    if total > 1e-15:
        result /= total
    else:
        result = np.full(n, 1.0 / n)
    return result

def _gs_iteration(a, N, relax=1.0):
    """Single GS iteration. Returns new a and the projected (unrelaxed) version."""
    n = len(a)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    
    # Compute target magnitude
    S2 = float(np.dot(a, a))
    # Parseval: sum_k |A[k]|^2 = N * S2 (for the zero-padded DFT)
    # For rfft: |A[0]|^2 + 2*sum_{k=1}^{N/2-1}|A[k]|^2 + |A[N/2]|^2 = N*S2
    # If we set |A[k]| = c for all k >= 1:
    # |A[0]|^2 + (N-1)*c^2 = N*S2  (since there are N-1 non-DC "slots")
    # c^2 = (N*S2 - |A[0]|^2) / (N-1)
    c2 = (N * S2 - mag[0]**2) / (N - 1)
    c = np.sqrt(max(c2, 0.0))
    
    # Construct target spectrum
    phase = np.ones_like(A)
    nonzero = mag > 1e-30
    phase[nonzero] = A[nonzero] / mag[nonzero]
    
    A_target = np.zeros_like(A)
    A_target[0] = A[0]
    A_target[1:] = c * phase[1:]
    
    # Inverse transform
    a_padded = np.fft.irfft(A_target, N)
    a_proj = np.clip(a_padded[:n], 0.0, None)
    total = a_proj.sum()
    if total > 1e-15:
        a_proj = a_proj / total
    else:
        a_proj = _project(a)
    
    a_out = (1.0 - relax) * a + relax * a_proj
    return a_out

def _run_gs(a_init, N, max_iters=3000, relax=1.0, deadline=None):
    """Run GS algorithm for max_iters or until deadline."""
    a = _project(a_init.copy())
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        a = _gs_iteration(a, N, relax)
        
        if (it + 1) % 200 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    score = float(evaluate_sequence(a.tolist()))
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

def _smoothed_grad(a, t):
    """Gradient of (1/t)*log(sum_k exp(t*C[k])) where C = convolve(a,a)."""
    n = len(a)
    C = _fft_autocorr(a)
    tc = t * C
    tc -= tc.max()
    exp_tc = np.exp(tc)
    p = exp_tc / exp_tc.sum()
    obj = (tc.max() + np.log(exp_tc.sum())) / t
    
    # grad[i] = 2 * sum_k p[k] * a[k-i] (linear cross-correlation)
    # Compute via FFT
    size = _next_pow2(len(p) + n - 1)
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    # Cross-correlation: ifft(P * conj(A))
    # Result[m] = sum_k p[k] * a[k-m] (linear, with zero-padding)
    cross = np.fft.irfft(P * np.conj(A), size)
    grad = 2.0 * cross[:n]
    return grad, obj

def _run_gradient(a_init, deadline, max_iters=5000, t_start=20.0, t_end=200.0):
    """Smoothed gradient descent with adaptive learning rate."""
    a = _project(a_init.copy())
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.toList() if hasattr(a, 'toList') else a.tolist()))
    
    n = len(a)
    best_lr = 0.05
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max_iters)
        t = t_start + (t_end - t_start) * frac
        lr = best_lr * (1.0 - 0.5 * frac)
        
        grad, obj = _smoothed_grad(a, t)
        
        # Line search: try a few step sizes
        best_a_trial = a
        best_obj = obj
        for lr_try in [lr, lr * 0.5, lr * 2.0]:
            a_trial = a - lr_try * grad
            a_trial = _project(a_trial)
            # Use the smoothed objective as a proxy
            _, obj_trial = _smoothed_grad(a_trial, t)
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a_trial = a_trial
        
        a = best_a_trial
        
        if (it + 1) % 100 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    score = float(evaluate_sequence(a.tolist()))
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

def _make_initializations(n, rng, parent=None):
    """Create various initializations for length n."""
    inits = []
    i = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n, dtype=np.float64))
    
    # Linear ramp
    ramp = i + 1
    inits.append(ramp)
    
    # Inverted ramp
    inits.append(n - i)
    
    # Raised cosine
    if n > 2:
        rc = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1)))
        inits.append(rc + 1e-6)
    
    # Gaussian (centered)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    g = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    inits.append(g)
    
    # Gaussian (off-center)
    center2 = n / 3.0
    g2 = np.exp(-0.5 * ((i - center2) / sigma) ** 2)
    inits.append(g2)
    
    # Two bumps
    c1, c2 = n * 0.25, n * 0.75
    b = np.exp(-0.5 * ((i - c1) / (n/10))**2) + np.exp(-0.5 * ((i - c2) / (n/10))**2)
    inits.append(b)
    
    # Sine wave (non-negative)
    freq = rng.integers(1, max(2, n // 10))
    s = 0.5 + 0.5 * np.sin(2 * np.pi * freq * i / n)
    inits.append(s + 1e-6)
    
    # Random
    r = rng.random(n)
    inits.append(r + 0.1)
    
    # Random smooth (low-pass filtered)
    r2 = rng.random(n)
    # Simple moving average as low-pass
    if n > 10:
        kernel = np.ones(min(n, 32)) / min(n, 32)
        r2 = np.convolve(r2, kernel, mode='same')
    inits.append(r2 + 0.1)
    
    # Parent sequence (if applicable)
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
    
    # Two-level random (split into groups)
    for _ in range(2):
        levels = rng.random(n) > 0.5
        two_level = levels.astype(np.float64) * 0.8 + 0.2
        inits.append(two_level)
    
    return inits
