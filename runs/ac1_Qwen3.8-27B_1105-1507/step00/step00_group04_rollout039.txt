    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def build_constraint_matrix(a):
    """Build A where A[k,j] = a[k-j] for valid indices."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def build_A(a):
    """Build Toeplitz-like constraint matrix A where A[k,j] = a[k-j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A(a)
    r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
