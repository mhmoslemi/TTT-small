"""
LP-based optimization for minimizing 2n*max(f*f)/(sum(f))^2.
Uses the linear programming approach to find descent directions,
combined with subgradient steps and random restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def _build_conv_matrix(f):
    """Build sparse matrix A where (A@g)_k = sum_j f[k-j]*g[j] (convolution)."""
    n = len(f)
    # Diagonal d (offset d) has constant value f[d], length n
    diags = [np.full(n, f[d]) for d in range(n)]
    offsets = list(range(n))
    return sparse.diags(diags, offsets=offsets, shape=(2*n-1, n), format='csr')


def _lp_direction(f, M):
    """Find g maximizing sum(g) subject to g>=0 and (f*g)_k <= M for all k."""
    n = len(f)
    A = _build_conv_matrix(f)
    b = np.full(2*n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def _score(f):
    """Compute the evaluation score (without calling evaluate_sequence)."""
    n = len(f)
    S = f.sum()
    if S < 0.01:
        return float('inf')
    c = np.convolve(f, f)
    return 2.0 * n * c.max() / (S * S)


def _optimize_lp(f0, max_time, rng):
    """Run LP-based optimization from initial point f0."""
    f = f0.astype(np.float64).copy()
    n = len(f)
    S0 = f.sum()
    f = f / S0  # Normalize to probability
    
    best_f = f.copy()
    best_score = _score(best_f)
    deadline = time.monotonic() + max_time
    
    t = 0.5  # step size for convex combination
    no_improve_count = 0
    
    while time.monotonic() < deadline:
        M = np.convolve(f, f).max()
        g = _lp_direction(f, M)
        
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum <= 1.0 + 1e-10:
            # LP didn't find improvement; try subgradient
            f = _subgrad_step(f, 0.05)
            continue
        
        # Rescale g to be a probability distribution
        g = g / g_sum
        
        # Convex combination with line search
        improved = False
        for t_try in [t, 0.7*t, 0.5*t, 0.3*t, 0.1*t]:
            f_new = (1 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            sc = _score(f_new)
            if sc < best_score:
                best_f = f_new.copy()
                best_score = sc
                f = f_new
                improved = True
                break
        
        if not improved:
            no_improve_count += 1
            if no_improve_count > 50:
                # Try smaller step or restart
                t = max(0.01, t * 0.5)
                if t < 0.01:
                    break
        else:
            no_improve_count = 0
    
    # Denormalize (the score is invariant to scaling, so this doesn't matter)
    return best_f, best_score


def _subgrad_step(f, step_size):
    """One subgradient step to reduce peak of f*f."""
    n = len(f)
    c = np.convolve(f, f)
    k_star = np.argmax(c)
    
    # Subgradient of max(p*p) w.r.t. p at the peak
    sg = np.zeros(n)
    j_min = max(0, k_star - n + 1)
    j_max = min(k_star, n - 1)
    for j in range(j_min, j_max + 1):
        sg[j] = 2.0 * f[k_star - j]
    
    f_new = f - step_size * sg
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s

        f_hump[:n//4] = 1.0
        f_hump[3*n//4:] = 1.0

        configs.append((f'hump_{n}', f_hump))

"""
LP-based optimization for minimizing 2n*max(f*f)/(sum(f))^2.
Uses linear programming to find descent directions in the space of
non-negative sequences, combined with subgradient steps, adaptive
step sizes, and multiple restarts with different initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def _build_conv_matrix(f):
    """Build sparse convolution matrix A where (A@g)[k] = sum_j f[k-j]*g[j]."""
    n = len(f)
    diags = [np.full(n, f[d]) for d in range(n)]
    offsets = list(range(n))
    return sparse.diags(diags, offsets=offsets, shape=(2 * n - 1, n), format='csr')


def _lp_direction(f, M):
    """Solve LP: maximize sum(g) subject to g>=0 and (f*g)_k <= M for all k."""
    n = len(f)
    try:
        A = _build_conv_matrix(f)
        b = np.full(2 * n - 1, M)
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _score_local(f):
    """Compute 2n*max(f*f)/(sum(f))^2 locally."""
    n = len(f)
    S = f.sum()
    if S < 1e-10:
        return float('inf')
    c = np.convolve(f, f)
    return 2.0 * n * c.max() / (S * S)


def _subgrad_step(p, step_size):
    """One projected subgradient step on probability vector p to reduce peak of p*p."""
    n = len(p)
    c = np.convolve(p, p)
    k_star = int(np.argmax(c))
    sg = np.zeros(n)
    j_min = max(0, k_star - n + 1)
    j_max = min(k_star, n - 1)
    sg[j_min:j_max + 1] = 2.0 * p[k_star - j_min:k_star - j_min + 1][::-1]
    p_new = p - step_size * sg
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-10:
        return p
    return p_new / s


def _multi_peak_subgrad(p, step_size, n_peaks=5):
    """Subgradient step using top peaks for more robust descent."""
    n = len(p)
    c = np.convolve(p, p)
    top_idx = np.argsort(c)[::-1][:n_peaks]
    weights = c[top_idx]
    weights = weights / weights.sum()
    sg = np.zeros(n)
    for w, k_star in zip(weights, top_idx):
        j_min = max(0, k_star - n + 1)
        j_max = min(k_star, n - 1)
        sg[j_min:j_max + 1] += w * 2.0 * p[k_star - j_min:k_star - j_min + 1][::-1]
    p_new = p - step_size * sg
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-10:
        return p
    return p_new / s


def _optimize_sequence(p0, max_time, use_lp=True, seed=0):
    """Optimize a probability vector to minimize max(p*p). Returns (best_p, best_score)."""
    p = np.array(p0, dtype=np.float64).copy()
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return p, float('inf')
    p = p / s
    n = len(p)
    rng = np.random.default_rng(seed)

    best_p = p.copy()
    best_score = _score_local(best_p)
    deadline = time.monotonic() + max_time

    t = 0.5
    lp_cooldown = 0
    subgrad_count = 0
    stall_count = 0
    last_best = best_score

    while time.monotonic() < deadline:
        phase = rng.random()

        if use_lp and phase < 0.6 and lp_cooldown <= 0:
            # LP-based direction
            M = np.convolve(p, p).max()
            g = _lp_direction(p, M)
            lp_cooldown = 3

            if g is not None and g.sum() > 1.0 + 1e-8:
                g = g / g.sum()
                improved = False
                for t_try in [t, 0.7 * t, 0.5 * t, 0.3 * t, 0.1 * t]:
                    p_new = (1.0 - t_try) * p + t_try * g
                    p_new = np.maximum(p_new, 0.0)
                    sn = p_new.sum()
                    if sn < 1e-10:
                        continue
                    p_new = p_new / sn
                    sc = _score_local(p_new)
                    if sc < best_score:
                        best_p = p_new.copy()
                        best_score = sc
                        p = p_new
                        improved = True
                        stall_count = 0
                        break
                if not improved:
                    stall_count += 1
                    if stall_count > 30:
                        t = max(0.005, t * 0.7)
            else:
                # LP degenerate, use subgradient
                step = 0.02 / (n ** 0.3)
                p = _multi_peak_subgrad(p, step, n_peaks=3)
                subgrad_count += 1
        else:
            lp_cooldown -= 1
            # Subgradient step
            step = 0.01 / (n ** 0.2)
            if subgrad_count % 10 == 0:
                p = _multi_peak_subgrad(p, step, n_peaks=5)
            else:
                p = _subgrad_step(p, step)
            subgrad_count += 1

            sc = _score_local(p)
            if sc < best_score:
                best_p = p.copy()
                best_score = sc
                stall_count = 0
            else:
                stall_count += 1

        # Escape mechanism: if stalled, perturb
        if stall_count > 50:
            idx = rng.integers(n)
            width = max(1, n // 50)
            start = max(0, idx - width // 2)
            end = min(n, start + width)
            p[start:end] *= np.exp(rng.normal(0, 0.3, end - start))
            p = np.maximum(p, 0)
            sn = p.sum()
            if sn > 1e-10:
                p = p / sn
            stall_count = 0

        # Periodic check on best
        if time.monotonic() > deadline:
            break

    return best_p, best_score
