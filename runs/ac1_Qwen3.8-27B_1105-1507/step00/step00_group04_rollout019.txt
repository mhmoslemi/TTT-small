```python
import cvxpy as cp

def solve_lp(a, M):
    """Solve the LP: maximize sum(b) s.t. (a*b)[k] <= M, b >= 0"""
    n = len(a)
    b = cp.Variable(n)
    
    # Build the constraint matrix: A[k, i] = a[k-i] if valid
    # A is (2n-1) x n
    # Instead of building the full matrix, we can express the constraints directly
    
    # (a*b)[k] = sum_i a[i] * b[k-i] for valid k-i
    # This is a linear function of b
    
    # Using cvxpy, we can write:
    # for k in range(2*n-1):
    #     cp.sum(a[i] * b[k-i] for i in range(max(0, k-n+1), min(n, k+1))) <= M
    
    # But this would be 2n-1 constraints with up to n terms each.
    # Let's use the matrix form instead.
    
    # Actually, cvxpy has a more efficient way using the expression directly.
    # Let's use the convolution property:
    # np.convolve(a, b) gives the full convolution
    # But cvxpy doesn't have a built-in convolution for variable b.
    
    # Let's build the A matrix explicitly for moderate n.
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            A[k, i] = a[i]
    
    constraints = [b >= 0, A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLOP)  # or another solver
    
    return b.value
```

```python
import numpy as np
import time

def compute_score(a):
    """Compute the evaluation score without calling evaluate_sequence."""
    n = len(a)
    c = np.convolve(a, a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * np.max(c) / (S * S)

def refine_sequence(a, deadline, base_step=0.5, top_k_frac=0.1, anneal_rate=100):
    """Iteratively refine a sequence to reduce the peak of its auto-convolution."""
    n = len(a)
    a = np.clip(a.astype(np.float64), 0, None).copy()
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S  # normalize to sum = 1
    
    best_a = a.copy()
    best_score = compute_score(best_a)
    
    it = 0
    no_improve = 0
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        current_score = 2.0 * n * np.max(c)
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            # Restart with perturbation
            a = best_a * np.exp(np.random.normal(0, 0.3, n))
            a = np.clip(a, 0, None)
            a = a / a.sum()
            no_improve = 0
            it = 0
        
        # Compute gradient
        mean_c = np.mean(c)
        e = c - mean_c
        
        # Focus on top peaks
        k = max(1, int(top_k_frac * len(c)))
        threshold = np.partition(e, -k)[-k]
        e_focused = np.where(e > threshold, e, 0)
        
        # g[m] = sum_{j=0}^{n-1} e_focused[m+j] * a[j]
        g = np.zeros(n)
        for j in range(n):
            g[j] = np.dot(e_focused[j:j+n], a)
        
        # Adaptive step size
        step = base_step / (1.0 + it / anneal_rate)
        
        a = a - step * g
        a = np.clip(a, 0, None)
        S = a.sum()
        if S < 1e-10:
            a = best_a.copy()
            break
        a = a / S
        it += 1
    
    return best_a
```

The score is 2*n*max(c)/S^2 where c = a*a (full convolution).

For a uniform sequence of length n: score = 2.
For the target: score = 1.5027.

What if I try a very specific construction? Consider a sequence where a[i] = 1 for i in S and 0 otherwise, for some subset S of {0, ..., n-1}. Then:
- sum(a) = |S| = w
- c[k] = number of pairs (i, j) in S with i + j = k
- score = 2*n*max(c) / w^2

For the score to be 1.5: max(c) = 1.5 * w^2 / (2n) = 0.75 * w^2 / n

The average of c[k] over k is w^2 / (2n-1) ≈ w^2/(2n).
So max(c) ≈ 1.5 * average, which means the convolution is only 1.5x the average at its peak.

This is achievable with certain combinatorial designs. For example, a "Sumset" where each sum k has approximately the same number of representations.

import numpy as np
import time

def compute_score_np(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def refine_gradient(a, deadline, base_step=1.0, top_k_frac=0.1, anneal_rate=100, restart_after=200, seed=0):
    n = len(a)
    a = np.clip(a.astype(np.float64), 0.0, None).copy()
    S = float(a.sum())
    if S < 1e-10:
        return a
    a = a / S
    
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_score = 2.0 * n * float(np.max(np.convolve(a, a)))
    
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        current_score = 2.0 * n * float(np.max(c))
        
        if current_score < best_score - 1e-10:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > restart_after:
            a = best_a * np.exp(rng.normal(0, 0.5, n))
            a = np.clip(a, 0.0, None)
            S = float(a.sum())
            if S < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / S
            no_improve = 0
            it = 0
            continue
        
        # Top-K peak weights
        k = max(1, int(top_k_frac * len(c)))
        threshold = np.partition(c, len(c) - k)[len(c) - k]
        w = np.maximum(c - threshold, 0.0)
        w_sum = float(w.sum())
        if w_sum < 1e-15:
            w = np.ones(len(c)) / len(c)
        else:
            w = w / w_sum
        
        # Gradient computation
        g = np.empty(n)
        for j in range(n):
            g[j] = 2.0 * float(np.dot(w[j:j+n], a))
        
        step = base_step / (1.0 + it / anneal_rate)
        a = a - step * g
        a = np.clip(a, 0.0, None)
        S = float(a.sum())
        if S < 1e-10:
            a = best_a.copy()
            break
        a = a / S
        it += 1
    
    return best_a
