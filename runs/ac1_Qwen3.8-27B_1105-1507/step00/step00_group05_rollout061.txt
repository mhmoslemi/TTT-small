    n = len(a)

def block_perturb(a, rng, sigma=0.1):
    """Multiplicative block perturbation."""
    n = len(a)
    w = max(2, int(rng.integers(n // 100, n // 5)))
    start = int(rng.integers(0, n - w))
    a_new = a.copy()
    a_new[start:start+w] *= np.exp(rng.normal(0, sigma, w))
    np.clip(a_new, 0, 1000, out=a_new)
    return a_new

def lag_targeted_perturb(a, C, rng, strength=0.1):
    """Target the hottest lag for reduction."""
    n = len(a)
    k_hot = C.argmax()
    a_new = a.copy()
    # For each i contributing to C[k_hot], reduce a_i proportionally
    # C[k_hot] = sum_i a[i] * a[k_hot - i] (valid indices)
    # Contribution of a[i] is a[i] * a[k_hot-i]
    # We reduce a[i] by strength * a[k_hot-i] / max(a)
    max_a = a.max()
    if max_a > 0:
        for i in range(max(0, k_hot - n + 1), min(n, k_hot + 1)):
            j = k_hot - i
            if 0 <= j < n:
                a_new[i] *= (1.0 - strength * a[j] / max_a)
                a_new[j] *= (1.0 - strength * a[i] / max_a)
    np.clip(a_new, 0, 1000, out=a_new)
    return a_new

def spectral_flatten(a, rng, alpha=0.3):
    """Flatten the power spectrum."""
    n = len(a)
    # Use zero-padded FFT for linear convolution
    N = 1
    while N < 2 * n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    # Target magnitude: geometric mean or some flat target
    target = np.mean(mag[mag > 0]) if np.any(mag > 0) else 1.0
    # Blend current and target
    A_new = A / mag * (mag * (1 - alpha) + alpha * target)
    # Actually simpler: A_new = A * (target / mag)^alpha
    A_new = A * (target / (mag + 1e-10))**alpha
    a_new = np.fft.irfft(A_new, N)[:n]
    np.clip(a_new, 0, 1000, out=a_new)
    return a_new

    # = ifft(fft(w) * conj(fft(a)))
    grad = np.fft.irfft(np.fft.rfft(w_full) * np.conj(np.fft.rfft(a_full)), N)[:n]
    
    # Add diagonal correction: for k=2i, the term a_i^2 contributes 2*a_i
    # This means grad_i should have an extra w[2i]*a_i for even k=2i
    even_lags = 2 * np.arange(n)  # 0, 2, 4, ..., 2n-2
    valid = even_lags < 2 * n - 1
    even_lags = even_lags[valid]
    grad[:len(even_lags)] += w[even_lags] * a[:len(even_lags)]
    
    return grad


def gradient_descent_step(a, lr, beta, N):
    """One step of gradient descent on max(C)."""
    n = len(a)
    S = a.sum()
    grad = compute_subgradient(a, N, beta)
    a_new = a - lr * grad / S  # normalize by S for scale invariance
    a_new = np.clip(a_new, 0.0, 1000.0)
    s_new = a_new.sum()
    if s_new < 0.01:
        return a, S
    a_new = a_new * S / s_new
    return a_new, S


def random_perturb(a, rng, sigma=0.05):
    """Additive random perturbation."""
    n = len(a)
    S = a.sum()
    noise = rng.normal(0, sigma * S / n, n)
    a_new = a + noise
    a_new = np.clip(a_new, 0.0, 1000.0)
    s_new = a_new.sum()
    if s_new > 0.01:
        a_new = a_new * S / s_new
    return a_new


def block_perturb(a, rng, n_blocks=1):
    """Multiplicative block perturbation."""
    n = len(a)
    a_new = a.copy()
    for _ in range(n_blocks):
        w = max(2, int(rng.integers(n // 200, n // 10)))
        start = int(rng.integers(0, max(1, n - w)))
        factor = np.exp(rng.normal(0, 0.15))
        a_new[start:start + w] *= factor
    a_new = np.clip(a_new, 0.0, 1000.0)
    S = a.sum()
    s_new = a_new.sum()
    if s_new > 0.01:
        a_new = a_new * S / s_new
    return a_new


def initialize_sequence(n, rng, method='constant'):
    """Create initial sequence."""
    if method == 'constant':
        return np.ones(n)
    elif method == 'cosine':
        t = np.arange(n) / (n - 1)
        return 0.5 + 0.5 * np.cos(np.pi * t)
    elif method == 'sinusoidal':
        t = np.arange(n)
        return 1.0 + 0.3 * np.cos(2 * np.pi * t / n)
    elif method == 'random_smooth':
        # Generate smooth random sequence
        freq = rng.integers(1, 5)
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
        a += 0.2 * np.cos(4 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
        return np.clip(a, 0.1, 1000)
    else:
        return np.ones(n)


def optimize_from_start(a, deadline, rng, initial_lr=1.0):
    """Run gradient descent optimization from a starting sequence."""
    n = len(a)
    N = compute_fft_size(n)
    a = a.copy()
    S = a.sum()
    if S < 0.01:
        return a, np.inf
    a = a * S / a.sum()  # ensure sum is consistent
    
    best = a.copy()
    best_score = fast_j(a)
    lr = initial_lr
    beta = 5.0
    temperature = 1.0
    cooldown = 0.999
    
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        
        # Compute gradient and take step
        a_cand, S = gradient_descent_step(a, lr, beta, N)
        score = fast_j(a_cand)
        
        # Accept or reject
        if score < best_score:
            best = a_cand.copy()
            best_score = score
            a = a_cand
            lr = min(lr * 1.05, 5.0)  # increase learning rate slightly
        else:
            # Simulated annealing: accept with probability exp(-delta/T)
            delta = score - best_score
            if delta < temperature:
                a = a_cand
            else:
                # Try alternative perturbations
                alt = rng.random()
                if alt < 0.3:
                    a = random_perturb(a, rng)
                elif alt < 0.6:
                    a = block_perturb(a, rng)
                else:
                    a = a_cand  # still move in gradient direction
                    lr = lr * 0.9  # decrease learning rate
            beta = min(beta * 1.02, 50.0)  # increase concentration
            temperature *= cooldown
        
        # Periodic random restart (mild)
        if iteration % 500 == 0:
            a = random_perturb(a, rng, sigma=0.1)
            beta = 5.0
        
        # Update temperature
        temperature = max(temperature, 0.01)
    
    return best, best_score
