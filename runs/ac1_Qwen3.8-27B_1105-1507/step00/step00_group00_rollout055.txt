```python
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def propose_candidate():
    # Choose sequence length
    n = 3000  # Start with a moderate size
    
    # Initial: uniform
    f = np.ones(n) / n
    best_f = f.copy()
    best_score = float(evaluate_sequence(f.tolist()))
    
    deadline = time.monotonic() + 1490  # Leave 10s buffer
    
    beta = 50  # Softmax parameter (not used in LP approach)
    t = 0.1  # Blend parameter
    
    while time.monotonic() < deadline:
        # Compute current peak
        b = np.convolve(f, f)
        M = float(np.max(b))
        
        # Build LP constraint matrix (sparse)
        # Row k, col i: A[k,i] = f[k-i] if 0 <= k-i < n
        rows = []
        cols = []
        data = []
        for k in range(2*n - 1):
            # i ranges from max(0, k-n+1) to min(n-1, k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                rows.append(k)
                cols.append(i)
                data.append(f[k - i])
        
        A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        
        # LP: max sum(g) s.t. A @ g <= M, g >= 0
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                      bounds=[(0, None)]*n, method='highs')
        
        if res.success:
            g = res.x
            sum_g = float(np.sum(g))
            
            if sum_g > 0:
                # Blend
                f_new = (1 - t) * f + t * g / sum_g
                # Normalize
                f_new = f_new / np.sum(f_new)
                
                # Check
                score = float(evaluate_sequence(f_new.tolist()))
                if score < best_score:
                    best_f = f_new.copy()
                    best_score = score
                
                f = f_new
        
        # Progressively decrease t
        # (optional: adapt t based on progress)
    
    return [float(x) for x in best_f]
```
