import scipy.sparse as sp

# Global LP state
_glp_g_var = None
_glp_rows = None

_glp_n = 0

def _init_lp(n):
    global _glp_g_var, _glp_rows, _glp_cols, _glp_n
    _glp_n = n
    _glp_g_var = cp.Variable(n, nonneg=True)
    cols = np.repeat(np.arange(n), n)
    row_chunks = [np.arange(j, j + n) for j in range(n)]
    rows = np.concatenate(row_chunks)
    _glp_rows = rows
    _glp_cols = cols

def _solve_lp(f, M):
    n = _glp_n
    data = np.tile(f, n)
    A = sp.csr_matrix((data, _glp_rows, _glp_cols), shape=(2 * n - 1, n))
    g = _glp_g_var
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except Exception:
                return None

        return None

- conv(a,a)[0] = sum(a_i^2) = sum(a_i) = n/2 (since a_i is 0 or 1)
- conv(a,a)[k] for k≠0 ≈ n/4 (expected value of a_i * a_{i+k})
- max ≈ n/2 (at lag 0)
- score ≈ 2n * n/2 / (n/2)^2 = n^2 / (n^2/4) = 4

That's bad. The issue is that for binary sequences, conv[0] = sum(a) (since a_i^2 = a_i), which makes the ratio bad.

- max = n/4
- score = 2n * n/4 / (n/4)^2 = 2n * n/4 * 16/n^2 = 8

"""
Minimize 2n*max(convolve(a,a))/sum(a)^2 using LP-based inflation-direction
iteration followed by local block-perturbation search.

The LP at each step finds the direction g (nonnegative, max sum) such that
the cross-correlation conv(f,g) is bounded by the current peak M. Moving
toward g flattens the autocorrelation, reducing the score.
"""

import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp

# Global LP state
_glp_var = None
_glp_rows = None
_glp_cols = None
_glp_n = 0


def _init_lp(n):
    """Precompute sparse structure for the convolution constraint matrix."""
    global _glp_var, _glp_rows, _glp_cols, _glp_n
    _glp_n = n
    _glp_var = cp.Variable(n, nonneg=True)
    cols = np.repeat(np.arange(n), n)
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    _glp_rows = rows
    _glp_cols = cols


def _solve_lp(f, M):
    """
    Solve: maximize sum(g) subject to A @ g <= M, g >= 0.
    A is the (2n-1) x n convolution matrix with A[k,j] = f[k-j].
    """
    n = _glp_n
    data = np.tile(f, n)
    A = sp.csr_matrix((data, _glp_rows, _glp_cols), shape=(2 * n - 1, n))
    g = _glp_var
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.GLOP, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except Exception:
                return None
    if g.value is None:
        return None
    return np.clip(np.asarray(g.value, dtype=np.float64), 0.0, None)


def _score(s):
    """Compute 2n*max(conv(s,s))/sum(s)^2."""
    s = np.asarray(s, dtype=np.float64)
    ss = float(np.sum(s))
    if ss < 0.01:
        return np.inf
    n = len(s)
    conv = np.convolve(s, s)
    return 2.0 * n * float(np.max(conv)) / (ss * ss)


def _line_search(f, g, target_sum, current_best):
    """Find best step t in (0,1] such that score of (1-t)f+tg improves."""
    best_t = 0.0
    for t in (0.5, 0.4, 0.35, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.01):
        cand = (1.0 - t) * f + t * g
        cs = float(np.sum(cand))
        if cs < 0.01:
            continue
        cand = cand * (target_sum / cs)
        sc = _score(cand)
        if sc < current_best:
            best_t = t
    return best_t

    Search for a nonnegative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    
    Phase 1: LP-based direction search (inflation method).
    Phase 2: Local search with random block perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 8.0

    # Choose optimization length
    n = 1000
    target_sum = np.sqrt(2.0 * n)

    _init_lp(n)

    # Initialize from parent with perturbation to break symmetry
    try:

        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size != n:
            f = np.interp(np.linspace(0, parent.size - 1, n),
                          np.arange(parent.size), parent)
        else:
            f = parent.copy()
    except Exception:
        f = np.ones(n)

    # Add random perturbation to break all-ones symmetry
    f = f * np.exp(rng.normal(0.0, 0.3, n))
    f = np.clip(f, 0.0, 1000.0)
    fs = float(np.sum(f))
    if fs < 0.01:
        f = np.ones(n) * target_sum / n
        fs = target_sum
    f = f * (target_sum / fs)

    best_f = f.copy()
    best_score = _score(f)
    print(f"n={n}, initial score = {best_score:.6f}")

    # ===== Phase 1: LP iterations =====
    phase1_frac = 0.65
    phase1_end = time.monotonic() + (budget_s - 8) * phase1_frac

        if sg < 1e-12:

        if cs < 0.01:
            continue

    print(f"Total local search steps: {ls_count}")
