```python
import numpy as np
import time
from scipy.optimize import linprog

def solve_lp(f, M, n):
    """Solve the LP: max sum(g) s.t. f*g <= M, g >= 0."""
    c = np.ones(n)  # objective: maximize sum(g_i)
    # Constraints: for each k = 0..2n-2, sum_i f_i * g_{k-i} <= M
    # The constraint matrix A has rows for each k.
    # A[k, j] = f[j] if 0 <= k-j <= n-1, else 0
    # But this is a (2n-1) x n matrix, which is large.
    
    # Instead, use the structure: the k-th row of A is f reversed and shifted.
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    
    # For efficiency, we can note that each row of A is a shifted/reversed version of f.
    # But for linprog, we need to provide A explicitly (or as a sparse matrix).
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = f[k-j]
    
    b = np.full(2*n-1, M)
    
    # Minimize -sum(g) subject to A @ g <= b, g >= 0
    res = linprog(-c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        return None
```
