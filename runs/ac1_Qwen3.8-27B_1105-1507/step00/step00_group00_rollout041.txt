        return 0.0
    scaled = np.clip(alpha * (conv - m) / m, -60.0, 0.0)
    return m * (1.0 + np.log(np.mean(np.exp(scaled))) / alpha)


def gradient_step(a, step_size):
    """One step of projected gradient descent on the score."""
    n = len(a)
    sum_a = np.sum(a)
    conv = fft_convolve_auto(a)
    k_star = int(np.argmax(conv))
    max_b = conv[k_star]
    score = 2.0 * n * max_b / (sum_a ** 2)

    # Gradient of score w.r.t. a[i]:
    # d(score)/da[i] = (2n/sum^2)*(a[i] + a[k*-i if valid]) - 4n*max_b/sum^3
    grad = np.empty(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = a[i] + a[j]
        else:
            grad[i] = a[i]

    grad = (2.0 * n / (sum_a ** 2)) * grad - (4.0 * n * max_b / (sum_a ** 3))

    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-15:
        a_new = a
        return a, score, 0.0
    a_new = a_new / s
    new_score = compute_score(a_new)
    return a_new, score, new_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search: geometric init + LBFGS + projected gradient."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Phase 1: Quick scan over geometric parameters
    best_a = None
    best_score = np.inf
    best_n = 0

    lengths = [512, 1024, 2048, 4096]
    c_values = [0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 2.5, 3.0, 4.0, 6.0]

    for n in lengths:
        for c in c_values:
            a = make_geometric(n, c)
            s = evaluate_sequence(a.tolist())
            if s < best_score:
                best_a = a.copy()
                best_score = s
                best_n = n

    print(f"Phase 1 best: n={best_n}, score={best_score:.6f}")

    # Also try the provided starting sequence
    if 'height_sequence_1' in dir() or True:
        try:
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size > 0 and np.isfinite(hs).all():
                hs = np.clip(hs, 0, 1000.0)
                s_hs = evaluate_sequence(hs.tolist())
                print(f"height_sequence_1: score={s_hs:.6f}")
        except Exception:
            pass

    # Phase 2: L-BFGS-B optimization from best initial
    n_opt = best_n
    a_current = best_a.copy()

    # Try multiple alpha values for soft-max
    for alpha in [50.0, 200.0, 500.0]:
        if time.monotonic() > deadline - 60:
            break
        log_params = np.log(a_current + 1e-30)
        t0 = time.monotonic()
        try:
            res = minimize(
                lambda p: soft_max_objective(p, n_opt, alpha),
                log_params,
                method='L-BFGS-B',
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30}
            )
            a_new = np.exp(res.x - np.max(res.x))
            a_new = a_new / np.sum(a_new)
            s_new = evaluate_sequence(a_new.tolist())
            elapsed = time.monotonic() - t0
            print(f"  L-BFGS (alpha={alpha:.0f}): score={s_new:.6f}, "
                  f"iters={res.nit}, time={elapsed:.1f}s")
            if s_new < best_score:
                best_a = a_new
                best_score = s_new
                a_current = a_new.copy()
        except Exception as e:
            print(f"  L-BFGS error (alpha={alpha}): {e}")

    # Phase 3: Projected gradient descent refinement
    if time.monotonic() < deadline - 10:
        a_pg = best_a.copy()
        n_pg = len(a_pg)
        step_size = 0.05
        best_pg_score = best_score
        no_improve = 0

        for it in range(3000):
            if time.monotonic() > deadline - 5:
                break

            a_new, score_now, new_score = gradient_step(a_pg, step_size)

            if new_score < score_now - 1e-12:
                a_pg = a_new
                step_size = min(step_size * 1.1, 1.0)
                no_improve = 0
            else:
                step_size *= 0.7

            if new_score < best_pg_score:
                best_pg_score = new_score
                best_a = a_pg.copy()
                best_score = new_score
                no_improve = 0
            else:
                no_improve += 1

            if it % 200 == 0:
                s_check = evaluate_sequence(a_pg.tolist())
                print(f"  PG iter {it}: score={s_check:.6f}, "
                      f"step={step_size:.5f}, no_improve={no_improve}")

            if no_improve > 200 and step_size < 0.001:
                break

    # Phase 4: Try random restarts and different shapes if time remains
    if time.monotonic() < deadline - 30:
        for restart in range(20):
            if time.monotonic() > deadline - 10:
                break
            n_try = int(rng.integers(256, 6000))
            # Random smooth initialization
            x = np.arange(n_try, dtype=np.float64) / n_try
            c1 = rng.uniform(0.5, 3.0)
            x0 = rng.uniform(0.0, 1.0)
            a_try = np.exp(-c1 * np.abs(x - x0))
            a_try = a_try / np.sum(a_try)
            s_try = evaluate_sequence(a_try.tolist())
            if s_try < best_score:
                best_a = a_try.copy()
                best_score = s_try
                print(f"  Restart {restart}: n={n_try}, score={s_try:.6f}")

                # Quick PG refinement
                for it2 in range(200):
                    if time.monotonic() > deadline - 5:
                        break
                    a_try, _, s2 = gradient_step(a_try, 0.05)
                    if s2 < best_score:
                        best_a = a_try.copy()
                        best_score = s2

    # Final verification
    final_score = evaluate_sequence([float(x) for x in best_a])
    print(f"Final score: {final_score:.6f}, length={len(best_a)}")
    return [float(x) for x in best_a]
