"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
- SLSQP direct optimization for small n (3-8)
- LP-based iterative method (SOTA-inspired) for medium n (5-150)
- Smooth max gradient descent with simplex projection for polish
- Multiple n values, random restarts, and adaptive step sizes
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def max_conv_obj(x, *args):
    c = np.convolve(x, x)
    return np.max(c)


def sum_con_fun(x, *args):
    return np.sum(x) - 1.0


def sum_con_jac(x, *args):
    return np.ones(len(x))


def get_score(b):
    n = len(b)
    if n == 0:
        return float('inf')
    conv = np.convolve(b, b)
    return 2.0 * n * np.max(conv)


def build_A(b):
    n = len(b)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = max(0, j)
        k_end = min(2 * n - 2, j + n - 1)
        for k in range(k_start, k_end + 1):
            A[k, j] = b[k - j]
    return A


def lp_step(b, step):
    n = len(b)
    if n < 2:
        return b
    conv = np.convolve(b, b)
    M = np.max(conv)
    if M < 1e-15:
        return b
    A = build_A(b)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0.0, None)] * n, method='highs')
    except Exception:
        return b
    if not res.success:
        return b
    g = res.x
    sg = g.sum()
    if sg < 1e-15:
        return b
    g_norm = g / sg
    b_new = (1.0 - step) * b + step * g_norm
    b_new = np.clip(b_new, 0, None)
    s = b_new.sum()
    if s < 1e-15:
        return b
    return b_new / s


def solve_slsqp(n, n_trials, seed):
    rng = np.random.default_rng(seed)
    best_b = np.ones(n) / n
    best_val = get_score(best_b)
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}]
    bounds = [(0.0, 1.0)] * n

    for t in range(n_trials):
        if t == 0:
            x0 = np.ones(n) / n
        elif t == 1:
            x0 = np.linspace(0.05, 1.0, n)
            x0 /= x0.sum()
        elif t == 2:
            x0 = np.exp(-np.linspace(0, 3, n))
            x0 /= x0.sum()
        elif t == 3:
            x0 = 1.0 / (1.0 + np.arange(n) * 0.5)
            x0 /= x0.sum()
        elif t == 4:
            x0 = np.abs(np.sin(np.linspace(0, 2 * np.pi, n)))
            x0 = np.maximum(x0, 0.01)
            x0 /= x0.sum()
        else:
            x0 = rng.dirichlet(np.ones(n))

        try:
            res = minimize(max_conv_obj, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14})
        except Exception:
            continue
        res_x = np.clip(res.x, 0, None)
        rs = res_x.sum()
        if rs < 1e-12:
            continue
        res_x = res_x / rs
        sc = get_score(res_x)
        if sc < best_val:
            best_val = sc
            best_b = res_x.copy()

    return best_b, best_val


def smooth_max_grad(b, beta):
    n = len(b)
    conv = np.convolve(b, b)
    max_c = np.max(conv)
    exp_vals = np.exp(beta * (conv - max_c))
    es = exp_vals.sum()
    if es < 1e-300:
        return np.zeros(n)
    w = exp_vals / es
    grad = np.zeros(n)
    for m in range(n):
        g = 0.0
        for k in range(max(0, m), min(2 * n - 1, m + n)):
            j = k - m
            if 0 <= j < n:
                g += w[k] * 2.0 * b[j]
        grad[m] = g
    return grad


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cssv - 1.0) / rho_arr > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(rho_arr[cond]))
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def gradient_descent(b, n_iter, beta, lr_init, seed, deadline):
    current = b.copy()
    current_score = get_score(current)
    lr = lr_init

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        grad = smooth_max_grad(current, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        candidate = current - lr * grad
        candidate = simplex_project(candidate)
        sc = get_score(candidate)
        if sc < current_score - 1e-14:
            current = candidate
            current_score = sc
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break

    return current, current_score


def lp_search(b, n_iters, deadline, seed):
    rng = np.random.default_rng(seed)
    current = b.copy()
    current_score = get_score(current)
    step_sizes = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]
    stall = 0

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        improved = False
        for step in step_sizes:
            b_new = lp_step(current, step)
            sc = get_score(b_new)
            if sc < current_score - 1e-12:
                current = b_new
                current_score = sc
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 20:
                if rng.random() < 0.5:
                    pert = current + 0.02 * rng.normal(size=len(current))
                    pert = np.clip(pert, 0, None)
                    ps = pert.sum()
                    if ps > 1e-12:
                        pert /= ps
                        psc = get_score(pert)
                        if psc < current_score:
                            current = pert
                            current_score = psc
                            stall = 0
                else:
                    break

    return current, current_score
