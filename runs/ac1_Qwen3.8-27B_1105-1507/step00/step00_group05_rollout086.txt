"""
Subgradient optimization with top-K hot lags to minimize J = 2n*max(conv(p))/(sum(p))^2.
Uses FFT-based convolution, softmax-weighted subgradients, momentum, and adaptive restarts.
"""

import numpy as np
import time

def _next_pow2(x):
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()

def _fft_conv(p, N):
    F = np.fft.rfft(p, N)
    return np.fft.irfft(F * F, N)

def _subgrad_topk(p, top_lags, top_weights, n):
    """Compute weighted subgradient for given lags."""
    g = np.zeros(n)
    for m, w in zip(top_lags, top_weights):
        if m < n:
            g[:m + 1] += 2.0 * w * p[m::-1]
        else:
            start = m - n + 1
            g[start:] += 2.0 * w * p[n - 1:start - 1:-1]
    return g

def _normalize_nonneg(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-15:
        return np.ones_like(p) / len(p)
    return p / s

def _get_initial_seq(n, rng):
    """Generate diverse initial sequences."""
    choice = rng.integers(0, 5)
    if choice == 0:
        # Uniform with small noise
        p = np.ones(n) + 0.1 * rng.normal(0, 1, n)
    elif choice == 1:
        # Linear ramp (asymmetric)
        t = np.linspace(0, 1, n)

"""
Subgradient optimization with top-K hot lags to minimize J = 2n*max(conv(p))/(sum(p))^2.
Uses FFT-based convolution, softmax-weighted subgradients, momentum, adaptive learning rate,
and multiple restarts with diverse initializations. Also tries a 'flatten' objective
(max - min) to spread autocorrelation energy more evenly.
"""

import numpy as np
import time


def _next_pow2(x):
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _fft_conv(p, N):
    F = np.fft.rfft(p, N)
    return np.fft.irfft(F * F, N)


def _subgrad_single(p, m, n):
    """Compute gradient of conv[m] w.r.t. p (a vector of length n)."""
    g = np.zeros(n)
    if m < n:
        g[:m + 1] = 2.0 * p[m::-1]
    else:
        start = m - n + 1
        g[start:] = 2.0 * p[n - 1:start - 1:-1]
    return g


def _subgrad_weighted(p, lags, weights, n):
    """Compute weighted sum of gradients for given lags."""
    g = np.zeros(n)
    for m, w in zip(lags, weights):
        if m < n:
            g[:m + 1] += 2.0 * w * p[m::-1]
        else:
            start = m - n + 1
            g[start:] += 2.0 * w * p[n - 1:start - 1:-1]
    return g


def _normalize_nonneg(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-15:
        return np.ones_like(p) / len(p)
    return p / s


def _make_initial(n, kind, rng):
    """Generate diverse initial sequences."""
    if kind == 0:
        p = np.ones(n) + 0.15 * rng.normal(0, 1, n)
    elif kind == 1:

        p = np.maximum(p, 0.01)
