```python
def fft_convolve(a):
    n = len(a)
    size = 2*n - 1
    # Pad to next power of 2 for FFT efficiency
    fsize = 1
    while fsize < size:
        fsize *= 2
    f = np.fft.rfft(a, fsize)
    return np.fft.irfft(f * f, fsize)[:size]

def compute_score(a):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = fft_convolve(a)
    max_b = np.max(b)
    return 2.0 * n * max_b / (S * S)

def gradient_step(a, k):
```

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/sum(a)^2`. 

Key insight: For symmetric (palindromic) sequences, `b[n-1] = sum(a_i^2) >= S^2/n`, giving score >= 2. But **asymmetric** sequences like linear ramps avoid this, achieving score ≈ 4/3 since the center convolution `b[n-1] = sum a_i * a[n-1-i]` pairs large values with small ones.

I'll use gradient descent with FFT-computed convolutions, starting from ramp-like shapes, with backtracking line search and multiple restarts.

```python
"""
Minimize score = 2*n*max(conv(a,a))/sum(a)^2 using FFT-accelerated gradient descent.
Key insight: asymmetric sequences (linear ramps) achieve score ~4/3 by pairing
large and small values at the convolution center. We start from ramp initializations
and refine with gradient descent + backtracking line search.
"""
import numpy as np
import time


def fft_convolve(a):
    """Full convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize *= 2
    f = np.fft.rfft(a, fsize)
    return np.fft.irfft(f * f, fsize)[:size]


def compute_score(a):
    """Compute score = 2*n*max_b/S^2 using FFT."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = fft_convolve(a)
    return 2.0 * n * float(np.max(b)) / (S * S)


def compute_grad(a, k, max_b, S):
    """
    Gradient of 2n*b[k]/S^2 w.r.t. a[j].
    db[k]/da[j] = 2*a[k-j] (0 if k-j outside [0,n-1]).
    df/da[j] = (4n/S^2)*(a[k-j] - max_b/S)
    """
    n = len(a)
    a_rev = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        a_rev[lo:hi] = a[k - np.arange(lo, hi)]
    return (4.0 * n / (S * S)) * (a_rev - max_b / S)


def optimize_sequence(a_init, deadline, max_iter=3000, lr_init=0.5, rng=None):
    """
    Gradient descent to minimize peak convolution.
    Returns (best_a, best_score).
    """
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0, 1000).copy()
    S = np.sum(a)
    if S < 0.01:
        return a, np.inf
    a = a / S  # normalize (score is scale-invariant)

    score = compute_score(a)
    best_a = a.copy()
    best_score = score
    lr = lr_init
    stall = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        b = fft_convolve(a)
        k = int(np.argmax(b))
        max_b = b[k]
        S = np.sum(a)

        grad = compute_grad(a, k, max_b, S)

        # Backtracking line search
        improved = False
        for t in range(8):
            step = lr * (0.6 ** t)
            a_try = a - step * grad
            a_try = np.clip(a_try, 0, 1000)
            S_try = np.sum(a_try)
            if S_try < 0.01:
                continue
            a_try = a_try / S_try
            score_try = compute_score(a_try)
            if score_try < score * (1.0 - 1e-12):
                a = a_try
                score = score_try
                improved = True
                break

        if improved:
            stall = 0
            if score < best_score:
                best_score = score
                best_a = a.copy()
        else:
            stall += 1
            lr *= 0.6
            if stall > 150:
                # Escape: random perturbation
                noise_scale = 0.05 + 0.05 * (stall // 150)
                a = a + noise_scale * rng.standard_normal(n)
                a = np.clip(a, 0, 1000)
                S = np.sum(a)
                if S > 0.01:
                    a = a / S
                score = compute_score(a)
                stall = 0
                lr = lr_init

    return best_a, best_score
```
