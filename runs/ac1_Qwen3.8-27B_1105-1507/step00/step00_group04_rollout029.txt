1. FFT-based gradient descent on log-sum-exp smoothed objective
2. LP-based direction finding for refinement
3. Multiple initializations with adaptive parameters

import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def _fft_convolve(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize *= 2
    A = fft(a, fsize)
    return ifft(A * A).real[:size]


def _compute_gradient(a, T):
    """Compute gradient of T*logsumexp(conv(a,a)/T) w.r.t. a.
    Returns (gradient, b, M)."""
    n = len(a)
    b = _fft_convolve(a)
    M = np.max(b)
    # Softmax weights
    w = np.exp(b / T - M / T)
    w /= np.sum(w)
    # Gradient via FFT cross-correlation
    fsize = 1
    while fsize < (2 * n - 1) + n - 1:
        fsize *= 2
    W = fft(w, fsize)
    Af = fft(a[::-1], fsize)
    grad_raw = ifft(W * Af).real[n - 1:n - 1 + n]
    grad = (2.0 / T) * grad_raw
    return grad, b, M

    """Fast evaluation without type checking."""
    n = len(a)

    M = np.max(b)
    s = np.sum(a)
    if s < 0.01:

    return 2.0 * n * M / (s * s)

            vals.append(a[j])

Wait, that's not quite right. Cauchy-Schwarz says `(sum a_j * b_j)² <= (sum a_j²)(sum b_j²)`. Here `b_j = a_{n-1-j}`, so `sum b_j² = sum a_j²`. So `(a*a)_{n-1} <= sum a_j²`. And `sum a_j² >= (sum a_j)²/n = 1/n`. So `(a*a)_{n-1} >= 1/n`... no wait, that's a lower bound on `sum a_j²`, not on `(a*a)_{n-1}`.

Actually, `(a*a)_{n-1} = sum_j a_j a_{n-1-j}`. If the sequence is a palindrome (a_j = a_{n-1-j}), then `(a*a)_{n-1} = sum_j a_j² >= 1/n`. So for palindromic sequences, the peak (at center) is at least 1/n, and we can't beat the uniform sequence.

But for NON-palindromic sequences, the peak might be at a different position, and the center value could be less than 1/n!

For example, consider `a = [1, 0, 0, ..., 0]` (all mass at one end). Then `convolve(a,a) = [1, 0, 0, ..., 0]` (a delta). The peak is 1, and sum = 1, so J = 2n * 1 = 2n. That's terrible.

What about `a = [1/2, 1/2, 0, ..., 0]`? Then `convolve(a,a) = [1/4, 1/2, 1/4, 0, ...]`. Peak = 1/2, sum = 1, J = 2n * 1/2 = n. Still bad.

The issue is that concentrating mass at one end makes the convolution peak large relative to the sum.

What about a "two-sided" distribution? `a = [1/2, 0, ..., 0, 1/2]`? Then `convolve(a,a) = [1/4, 0, ..., 0, 1/2, 0, ..., 0, 1/4]`. Peak = 1/2, J = n. Still bad.

What about `a` uniformly distributed but with a dip in the middle? Like `a_j = 1` except `a_{n/2} = 0`? Then the convolution would have a modified center.

I think the key insight is: we need an asymmetric sequence where the mass is distributed in a way that the convolution peak is reduced. The LP approach naturally finds such distributions.

Let me also think about this from the LP perspective. The LP finds `g` that maximizes `sum(g)` subject to `a*g <= M`. If `a` is uniform, then `a*g <= M` means `(1/n) * sum_j g_j * ... <= M` for each position of the cross-correlation. This favors `g` being spread out, but the constraint at the center is the tightest (since it involves the most terms).

"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient descent on log-sum-exp smoothed objective
2. LP-based direction finding for refinement  
3. Multiple diverse initializations with adaptive parameters
4. Simulated annealing restarts to escape local minima
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _fft_self_conv(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize *= 2
    A = fft(a, fsize)
    return ifft(A * A).real[:size]


def _fft_cross_corr(w, a):
    """Compute R[k] = sum_j a[j]*w[j+k] for k=0..n-1 using FFT.
    w has length 2n-1, a has length n."""
    n = len(a)
    fsize = 1
    while fsize < (2 * n - 1) + n - 1:
        fsize *= 2
    W = fft(w, fsize)
    Af = fft(a[::-1], fsize)
    result = ifft(W * Af).real
    return result[n - 1:n - 1 + n]


def _eval_fast(a):
    """Fast score computation (scale-invariant, assumes sum normalizable)."""
    n = len(a)
    b = _fft_self_conv(a)
    M = np.max(b)
    s = np.sum(a)
    if s < 1e-10:
        return float('inf')
    return 2.0 * n * M / (s * s)


def _gradient_step(a, T):
    """One gradient step for T*logsumexp(conv(a,a)/T). Returns (new_a, score, M)."""
    n = len(a)
    b = _fft_self_conv(a)
    M = np.max(b)
    if M < 1e-15:
        return a.copy(), 0.0, 0.0
    w = np.exp(b / T - M / T)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        return a.copy(), 2.0 * n * M / (np.sum(a) ** 2), M
    w /= w_sum
    grad_raw = _fft_cross_corr(w, a)
    grad = (2.0 / T) * grad_raw
    return grad, b, M


def _gd_optimize(a, deadline, max_iters=200000):
    """Gradient descent with log-sum-exp smoothing and adaptive parameters."""
    n = len(a)
    a = np.clip(a.copy(), 0.0, 1e6)
    s = np.sum(a)
    if s < 1e-10:
        return a, float('inf')
    a = a / s

    best_score = _eval_fast(a)
    best_a = a.copy()

    # Initial temperature: fraction of peak
    b0 = _fft_self_conv(a)
    M0 = np.max(b0)
    T = M0 * 0.3
    lr = 1.0 / n
    best_lr = lr
    patience = 0
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, M = _gradient_step(a, T)
        if not np.all(np.isfinite(grad)):
            break

        # Adaptive temperature: anneal toward the peak scale
        T = max(M * 0.02, M0 * 0.3 * np.exp(-it / 3000.0))

        # Try the step
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, 1e6)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            lr *= 0.5
            patience += 1
            if patience > 50:
                break
            continue
        a_new = a_new / s_new

        score = _eval_fast(a_new)
        if np.isfinite(score) and score < best_score:
            a = a_new
            best_score = score
            best_a = a.copy()
            best_lr = lr
            lr = min(lr * 1.05, 10.0 / n)
            patience = 0
            stall_count = 0
        else:
            lr *= 0.85
            patience += 1
            stall_count += 1
            if patience > 80:
                # Restart with different temperature
                T = M * (0.05 + 0.5 * np.random.random())
                lr = best_lr
                patience = 0
                stall_count += 1
                if stall_count > 5:
                    break

    return best_a, best_score


def _solve_lp(a, M, time_limit=60):
    """Solve LP: max 1^T g s.t. A*g <= M, g >= 0. Returns g or None."""
    n = len(a)
    if n > 3000:
        return None

    # Build sparse convolution matrix
    rows = []
    cols = []
    vals = []
    for i in range(2 * n - 1):
        j_start = max(0, i - n + 1)
        j_end = min(n - 1, i)
        if j_end >= j_start:
            j_range = range(j_start, j_end + 1)
            rows.extend([i] * len(j_range))
            cols.extend(list(j_range))
            vals.extend(a[j_start:j_end + 1].tolist())

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            return np.clip(result.x, 0, None)
    except Exception:
        pass
    return None


def _lp_refine(a, deadline, max_iters=30):
    """Iterative LP-based refinement."""
    n = len(a)
    a = np.clip(a.copy(), 0.0, 1e6)
    s = np.sum(a)
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    best_score = _eval_fast(a)
    best_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = _fft_self_conv(a)
        M = np.max(b)

        g = _solve_lp(a, M, time_limit=90)
        if g is None or np.sum(g) < 1e-10:
            break

        # Rescale g to match sum of a (which is 1)
        g = g / np.sum(g)

        # Blend with decreasing step
        t = 0.4
        improved = False
        for _ in range(15):
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, 1e6)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                t *= 0.5
                continue
            a_new = a_new / s_new
            score = _eval_fast(a_new)
            if np.isfinite(score) and score < best_score:
                a = a_new
                best_score = score
                best_a = a.copy()
                improved = True
                break
            t *= 0.5

        if not improved:
            break

        if it % 5 == 0:
            print(f"      LP iter {it}: score={best_score:.8f}")

    return best_a, best_score


def _anneal_restart(a, rng, strength=0.3):
    """Create a perturbed version of a for restart."""
    n = len(a)
    a_new = a.copy()
    # Random multiplicative perturbation
    noise = np.exp(rng.normal(0, strength, n))
    a_new *= noise
    # Random block perturbation
    width = max(1, n // 50)
    start = rng.integers(0, n - width)
    a_new[start:start + width] *= np.exp(rng.normal(0, 0.5))
    a_new = np.clip(a_new, 0, 1e6)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new /= s
    return a_new
