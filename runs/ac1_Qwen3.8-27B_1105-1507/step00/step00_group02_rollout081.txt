"""
Search for non-negative step-function heights minimizing
    2*n*max(convolve(a,a)) / sum(a)^2

The code works primarily in the scale-invariant normalization sum(a)=1.
It uses FFT-based exact gradients of log-sum-exp smoothings of the linear
autocorrelation, low-dimensional smooth initialization, full softmax-Adam
refinement, and a short L-BFGS polish.
"""

import numpy as np
import time
import operator

    from scipy.optimize import minimize

    minimize = None

def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _fft_len_3n(n):
    return _next_pow2(max(16, 3 * int(n) - 2))


def _fft_len_2n(n):
    return _next_pow2(max(16, 2 * int(n) - 1))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones_like(a)
        s = float(a.sum())
    return a / s


def _softmax_y(y):
    y = np.asarray(y, dtype=np.float64)
    y = np.nan_to_num(y, nan=0.0, posinf=500.0, neginf=-500.0)
    y = np.clip(y, -500.0, 500.0)
    y = y - float(np.max(y))
    a = np.exp(y)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(y.size, dtype=np.float64) / max(1, y.size)
    return a / s


def _smooth_f_and_grad(a, n, beta):
    """
    Return f(a)=logsumexp(beta*c), c=linear conv(a,a), and grad wrt a.
    """
    n = int(n)
    beta = float(beta)
    N = _fft_len_3n(n)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    maxc = float(np.max(c))
    if beta <= 0.0 or maxc <= 0.0:
        return 0.0, c, np.zeros(n, dtype=np.float64)

    z = beta * (c - maxc)
    expz = np.exp(z)
    s = float(expz.sum())
    if not np.isfinite(s) or s <= 0.0:
        expz = np.zeros(c.size, dtype=np.float64)
        expz[int(np.argmax(c))] = 1.0
        s = 1.0

    f = beta * maxc + float(np.log(s))
    w = expz / s

    if n == 1:
        grad = beta * 2.0 * float(w[0]) * a.copy()
    else:
        fw = np.fft.rfft(w[::-1], N)
        r_all = np.fft.irfft(fa * fw, N)
        r = r_all[2 * n - 2 : n - 2 : -1]
        grad = beta * 2.0 * r

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(f), c, grad


def _conv_score_fft(a, n):
    n = int(n)
    N = _fft_len_2n(n)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return 2.0 * float(n) * float(np.max(c))


def _uniform(n):
    return np.ones(int(n), dtype=np.float64) / float(n)


def _power_ramp(n, r):
    x = np.arange(1, int(n) + 1, dtype=np.float64) ** float(r)
    s = x.sum()
    if s <= 0:
        return _uniform(n)
    return x / s


def _geometric(n, lam):
    x = np.linspace(0.0, 1.0, int(n), dtype=np.float64)
    a = np.exp(float(lam) * x)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _two_level(n, alpha, ratio):
    n = int(n)
    a = np.ones(n, dtype=np.float64)
    k = max(1, min(n - 1, int(round(float(alpha) * n))))
    a[k:] = float(ratio)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _bump(n, center, sigma, base=0.2):
    x = np.linspace(0.0, 1.0, int(n), dtype=np.float64)
    a = float(base) + np.exp(-((x - float(center)) ** 2) / (2.0 * float(sigma) ** 2))
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s


def _smooth_random(n, rng, slope, noise, cutoff, gamma):
    x = np.linspace(0.0, 1.0, int(n), dtype=np.float64)
    z = rng.standard_normal(int(n))
    Z = np.fft.rfft(z)
    kmax = min(len(Z), max(2, int(cutoff)))
    Z[kmax:] = 0.0
    z = np.fft.irfft(Z, int(n))
    sd = float(np.std(z))
    if sd <= 1e-12:
        sd = 1.0
    z = (z - float(np.mean(z))) / sd
    a = 1.0 + float(slope) * x + float(noise) * z
    a = np.clip(a, 1e-4, None)
    if abs(float(gamma) - 1.0) > 1e-12:
        a = np.power(a, float(gamma))
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s

def _lowdim_basis(n, p, rng):
    n = int(n)
    p = int(min(p, max(4, n)))
    B = np.zeros((n, p), dtype=np.float64)
    x = (np.arange(n, dtype=np.float64) / max(1, n - 1)) - 0.5

    B[:, 0] = x
    if p > 1:
        B[:, 1] = x * x - 1.0 / 3.0
    idx = 2

    maxf = max(2, n // 16)
    while idx < p:
        f = int(rng.integers(1, maxf + 1))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.4, 1.5))
        if idx < p:
            B[:, idx] = amp * np.cos(2.0 * np.pi * f * x + ph)
            idx += 1
        if idx < p:
            ph2 = float(rng.uniform(0.0, 2.0 * np.pi))
            amp2 = float(rng.uniform(0.4, 1.5))
            B[:, idx] = amp2 * np.sin(2.0 * np.pi * f * x + ph2)
            idx += 1

    while idx < p:
        c = float(rng.uniform(0.15, 0.85))
        k = float(rng.uniform(5.0, 45.0))
        col = 1.0 / (1.0 + np.exp(-k * (x + 0.5 - c)))
        B[:, idx] = col - float(np.mean(col))
        idx += 1

    for j in range(p):
        norm = float(np.linalg.norm(B[:, j]))
        if norm > 1e-12:
            B[:, j] /= norm
    return B


def _lowdim_optimize(a0, n, p, betas, time_limit, deadline, rng):
    n = int(n)
    a = _normalize(a0, n)
    best = a.copy()
    best_score = _conv_score_fft(a, n)

    B = _lowdim_basis(n, p, rng)
    y0 = np.log(a + 1e-300)
    theta = B.T @ y0
    theta = np.nan_to_num(theta, nan=0.0, posinf=0.0, neginf=0.0)
    theta = np.clip(theta, -60.0, 60.0)

    m = np.zeros(theta.size, dtype=np.float64)
    v = np.zeros(theta.size, dtype=np.float64)
    t = 0
    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    lr = 0.20
    no_improve = 0

    start_time = time.monotonic()
    end = min(float(deadline), start_time + float(time_limit))

    remaining = len(betas)
    for beta in betas:
        now = time.monotonic()
        if now >= end:
            break
        phase_len = (end - now) / float(max(1, remaining))
        phase_end = now + phase_len
        remaining -= 1

        while time.monotonic() < min(phase_end, deadline):
            y = B @ theta
            a = _softmax_y(y)
            _, c, grad_a = _smooth_f_and_grad(a, n, beta)
            score = 2.0 * float(n) * float(np.max(c))

            if np.isfinite(score) and score < best_score - 1e-10:
                best_score = float(score)
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 4000:
                theta += 0.25 * (rng.random(theta.size) - 0.5)
                theta = np.clip(theta, -60.0, 60.0)
                no_improve = 0

            m_dot = float(np.dot(a, grad_a))
            grad_y = a * (grad_a - m_dot)
            grad_theta = B.T @ grad_y
            grad_theta = np.nan_to_num(grad_theta, nan=0.0, posinf=0.0, neginf=0.0)

            t += 1
            m = b1 * m + (1.0 - b1) * grad_theta
            v = b2 * v + (1.0 - b2) * grad_theta * grad_theta
            mhat = m / (1.0 - b1 ** t)
            vhat = v / (1.0 - b2 ** t)
            step = (lr / (1.0 + 0.05 * np.sqrt(float(t)))) * mhat / (np.sqrt(vhat) + eps)
            step = np.clip(step, -1.0, 1.0)
            theta = theta - step
            theta = np.nan_to_num(theta, nan=0.0, posinf=0.0, neginf=0.0)
            theta = np.clip(theta, -80.0, 80.0)

    final_a = _softmax_y(B @ theta)
    final_s = _conv_score_fft(final_a, n)
    if np.isfinite(final_s) and final_s < best_score:
        best = final_a
        best_score = float(final_s)
    return best, best_score

                no_improve = 0

            m_dot = float(np.dot(a, grad_a))
            grad_y = a * (grad_a - m_dot)
            grad_y = np.nan_to_num(grad_y, nan=0.0, posinf=0.0, neginf=0.0)

            t += 1
            m = b1 * m + (1.0 - b1) * grad_y
            v = b2 * v + (1.0 - b2) * grad_y * grad_y
            mhat = m / (1.0 - b1 ** t)
            vhat = v / (1.0 - b2 ** t)
            step = (lr / (1.0 + 0.10 * np.sqrt(float(t)))) * mhat / (np.sqrt(vhat) + eps)
            step = np.clip(step, -1.0, 1.0)

            y = y - step
            y -= float(np.mean(y))
            y = np.nan_to_num(y, nan=0.0, posinf=80.0, neginf=-80.0)
            y = np.clip(y, -80.0, 80.0)
            a = _softmax_y(y)

    final_s = _conv_score_fft(a, n)
    if np.isfinite(final_s) and final_s < best_score:
        best = a
        best_score = float(final_s)
    return best, best_score


def _lbfgs_objective(y):
    n = int(_LB_N)
    beta = float(_LB_BETA)
    y = np.asarray(y, dtype=np.float64)
    y = np.nan_to_num(y, nan=0.0, posinf=500.0, neginf=-500.0)
    y = np.clip(y, -500.0, 500.0)
    a = _softmax_y(y)
    f, c, grad_a = _smooth_f_and_grad(a, n, beta)
    if not np.isfinite(f):
        return 1e100, np.zeros_like(y)
    m_dot = float(np.dot(a, grad_a))
    grad_y = a * (grad_a - m_dot)
    grad_y = np.nan_to_num(grad_y, nan=0.0, posinf=0.0, neginf=0.0)
    return float(f), grad_y


def _lbfgs_polish(a0, n, betas, deadline):
    global _LB_N, _LB_BETA

    if minimize is None:
        a = _normalize(a0, n)
        return a, _conv_score_fft(a, n)

    n = int(n)
    a = _normalize(a0, n)
    score = _conv_score_fft(a, n)

    for beta in betas:
        now = time.monotonic()
        if now > float(deadline) - 2.0:
            break

        _LB_N = n
        _LB_BETA = float(beta)

        y0 = np.log(a + 1e-300)
        y0 -= float(np.mean(y0))
        y0 = np.clip(y0, -80.0, 80.0)

        t0 = time.monotonic()
        try:
            f0, g0 = _lbfgs_objective(y0)
        except Exception:
            f0, g0 = np.inf, None
        est = max(1e-4, time.monotonic() - t0)

        allowed = max(1.0, float(deadline) - time.monotonic() - 2.0)
        if n <= 1024:
            cap = 70
        elif n <= 2500:
            cap = 50
        else:
            cap = 30
        maxiter = int(min(cap, max(15, allowed / max(0.03, est * 8.0))))

        try:
            res = minimize(
                _lbfgs_objective,
                y0,
                jac=True,
                method="L-BFGS-B",
                options={
                    "maxiter": int(maxiter),
                    "maxfun": int(max(60, maxiter * 4)),
                    "maxcor": 10,
                    "ftol": 1e-12,
                    "gtol": 1e-8,
                },
            )
            y_new = res.x
        except Exception:
            y_new = y0

        a_new = _softmax_y(y_new)
        s_new = _conv_score_fft(a_new, n)
        if np.isfinite(s_new) and s_new < score:
            a = a_new
            score = float(s_new)

    return a, score


def _exact_update(candidate, best_exact, best_list, deadline):
    if candidate is None:
        return best_exact, best_list
    if time.monotonic() > float(deadline) - 0.5:
        return best_exact, best_list
    try:
        lst = [float(x) for x in candidate]
    except Exception:
        return best_exact, best_list
    try:
        score = evaluate_sequence(lst)
    except Exception:
        return best_exact, best_list
    if np.isfinite(score) and (best_list is None or float(score) < float(best_exact)):
        return float(score), lst
    return best_exact, best_list


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_candidate = None
    best_fft = np.inf
    candidates = []

    lengths = [1024, 2048, 4096, 512]

    for n in lengths:
        now = time.monotonic()
        if now > deadline - 90.0:
            break

        starts = _generate_starts(n, rng)

        lowdims = [(0.0, None)]
        for a0 in starts[:2]:
            if time.monotonic() > deadline - 120.0:
                break
            atime = min(35.0, max(6.0, deadline - time.monotonic() - 180.0))
            betas = (2.0 * n, 8.0 * n, 30.0 * n, 100.0 * n)
            a, s = _lowdim_optimize(a0, n, 16, betas, atime, deadline, rng)
            if np.isfinite(s):
                lowdims.append((s, a))
                if s < best_fft:
                    best_fft = float(s)
                    best_candidate = a.copy()
                candidates.append((s, a, n))

        full_starts = []
        for _, a in lowdims[1:]:
            full_starts.append(a)
        for a0 in starts[:5]:
            full_starts.append(a0)

        full_betas = (5.0 * n, 20.0 * n, 80.0 * n, 500.0 * n)
        for a0 in full_starts:
            if time.monotonic() > deadline - 90.0:
                break
            atime = min(28.0, max(6.0, deadline - time.monotonic() - 150.0))
            a, s = _adam_full(a0, n, full_betas, atime, deadline, rng)
            if np.isfinite(s):
                if s < best_fft:
                    best_fft = float(s)
                    best_candidate = a.copy()
                candidates.append((s, a, n))

    if candidates:
        candidates = [(s, a, n) for (s, a, n) in candidates if np.isfinite(s)]
    if best_candidate is None:
        best_candidate = _uniform(1024)
        best_fft = _conv_score_fft(best_candidate, 1024)

    candidates.sort(key=operator.itemgetter(0))

    seen = 0
    for s, a, n in candidates:
        if time.monotonic() > deadline - 25.0:
            break
        if seen >= 10:
            break
        seen += 1
        betas = (100.0 * n, 500.0 * n)
        a2, s2 = _lbfgs_polish(a, n, betas, deadline)
        if np.isfinite(s2):
            if s2 < best_fft:
                best_fft = float(s2)
                best_candidate = a2.copy()
            candidates.append((s2, a2, n))

    candidates.sort(key=operator.itemgetter(0))

    best_exact = np.inf
    best_list = None

    for s, a, n in candidates[:6]:
        best_exact, best_list = _exact_update(a, best_exact, best_list, deadline + 1.0)
        if best_list is not None and best_exact < 1.50:
            break

    if best_list is None or best_exact > best_fft + 1e-6:
        best_exact, best_list = _exact_update(best_candidate, best_exact, best_list, deadline + 1.0)

    if best_list is None:
        best_list = [float(x) for x in best_candidate]

    arr = np.asarray(best_list, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 0.01:
        arr = _uniform(1024)
        s = 1.0
    else:
        arr = arr / s

    return [float(x) for x in arr]
