import numpy as np
from scipy import sparse
from scipy.optimize import linprog
import time

def compute_convolution(a):
    """Compute full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use FFT for speed
    a_ext = np.zeros(2 * n, dtype=np.float64)
    a_ext[:n] = a
    c = np.fft.ifft(np.fft.fft(a_ext) ** 2).real[:size]
    return c

def solve_lp(a, M):
    """Solve the LP: max sum(b) s.t. b >= 0, a*b <= M (componentwise)."""
    n = len(a)
    size = 2 * n - 1
    
    # Build constraint matrix A (size x n), A[k, j] = a[k-j] if 0 <= k-j < n
    # This is a Toeplitz-like matrix
    rows = []
    cols = []
    vals = []
    for k in range(size):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(size, n))
    b_vec = np.full(size, M)
    
    # Minimize -sum(b)
    c_vec = -np.ones(n)
    
    res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=[(0, None)] * n, method='highs')
    
    if res.status != 0:
        return None
    
    return res.x

But we can also use a "smoothed" version: replace $\max_k c[k]$ with $\log \sum_k \exp(c[k]/\tau)$ (log-sum-exp) for small $\tau$. This gives a smooth approximation whose gradient is:
$\nabla = \frac{1}{\tau} \frac{\sum_k c'[k] \exp(c[k]/\tau)}{\sum_k \exp(c[k]/\tau)}$

where $c'[k] = 2a[k-j]$ for each $j$. This is the "soft" subgradient.

For the implementation, I'll use the top-$m$ peaks approximation (which is simpler and faster than the full log-sum-exp).

"""
Hybrid optimization combining subgradient descent with LP-based updates
to minimize the self-convolution peak ratio of a non-negative sequence.
"""

def _fft_convolve(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for FFT efficiency
    m = 1
    while m < 2 * n:
        m *= 2
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)[:size]
    # Clip tiny negative values from FFT numerical error
    c = np.maximum(c, 0.0)
    return c

def _subgradient_update(a, c, eta, top_m=10):
    """Compute multiplicative subgradient update."""
    n = len(a)
    m = min(top_m, len(c))
    # Find top m positions
    if m == len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -m)[-m:]
    weights = c[idx]
    wsum = weights.sum()
    if wsum < 1e-30:
        return a.copy()
    weights = weights / wsum
    
    g = np.zeros(n)
    for w, k in zip(weights, idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # g[j] += 2*w*a[k-j] for j in [lo, hi]
        # a[k-j] for j in [lo,hi] = a[k-hi : k-lo+1] reversed
        g[lo:hi+1] += 2.0 * w * a[k-hi:k-lo+1][::-1]
    
    # Multiplicative update
    a_new = a * np.exp(-eta * g)
    total = a_new.sum()
    if total < 1e-30:
        return np.ones(n) / n
    a_new /= total
    return a_new

def _lp_update(a, M, max_constraints=None):
    """Solve LP to find a direction that keeps cross-convolution bounded."""
    n = len(a)
    size = 2 * n - 1
    
    # For large n, only use a subset of constraints
    if max_constraints is not None and size > max_constraints:
        # Use the most binding constraints
        c = _fft_convolve(a)
        idx = np.argpartition(c, -max_constraints)[-max_constraints:]
        idx = np.sort(idx)
    else:
        idx = np.arange(size)
    
    # Build constraint matrix (sparse)
    rows = []
    cols = []
    vals = []
    for k in idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)  # Will remap
            cols.append(j)
            vals.append(a[k - j])
    
    # Remap row indices
    row_map = {k: i for i, k in enumerate(idx)}
    rows_mapped = [row_map[r] for r in rows]
    
    A = sparse.csr_matrix((vals, (rows_mapped, cols)), shape=(len(idx), n))
    b_vec = np.full(len(idx), M)
    c_vec = -np.ones(n)
    
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b_vec, 
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30})
        if res.status == 0 and res.x is not None:
            b = res.x
            bsum = b.sum()
            if bsum > 1e-10:
                return b / bsum
    except:
        pass
    return None
