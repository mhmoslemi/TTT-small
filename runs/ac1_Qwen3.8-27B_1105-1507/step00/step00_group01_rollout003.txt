- $(a*a)_k = m - |k|$ for $|k| < m$, 0 otherwise
- $\max(a*a) = m$
- Score = $2n \cdot m / m^2 = 2n/m > 2$ (since $m < n$)

def compute_direct_score(a):

    n = len(a)
    S = np.sum(a)

def lp_based_search(a, M, max_iters=50):
    """
    LP-based direction finding: maximize sum(b) s.t. b>=0, (a*b)_k <= M.
    Uses iterative projection.
    """
    n = len(a)
    b = np.ones(n) * 0.001

    for _ in range(max_iters):
        # Check feasibility
        ab = np.convolve(a, b)
        max_ab = np.max(ab)

        if max_ab <= M * (1 + 1e-9):
            # Feasible: try to increase b
            b += 0.005
        else:
            # Infeasible: reduce b at positions contributing to the max
            k_max = int(np.argmax(ab))
            # Reduce b proportionally to contribution

                j = k_max - ci
                if 0 <= j < n:
                    b[ci] = max(0.0, b[ci] - 0.002 * a[j])
