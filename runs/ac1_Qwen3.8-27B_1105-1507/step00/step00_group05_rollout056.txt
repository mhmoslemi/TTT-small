import numpy as np
import time

def compute_smooth_max_gradient(a, T):
    """Compute gradient of T*log(sum(exp(c/T))) w.r.t. a"""
    n = len(a)
    c = np.convolve(a, a)
    # Softmax weights
    c_max = np.max(c)
    w = np.exp((c - c_max) / T)
    w /= np.sum(w)
    # Gradient: grad_i = 2 * sum_k w_k * a_{k-i}
    # Using convolution: result[j] = sum_k w[k] * a[n-1-j+k]
    # We want sum_k w[k] * a[k-i], which is result[n-1+i]
    result = np.convolve(w, a[::-1], mode='full')
    grad = 2.0 * result[n-1:2*n-1]
    return grad, c

def optimize_sequence(a, T_init, T_min, lr, momentum, max_iters, time_budget):
    """Gradient descent on smooth max of autocorrelation"""
    n = len(a)
    S = np.sum(a)
    a = a / S * np.sqrt(2*n)  # Normalize
    best_a = a.copy()
    best_val = np.max(np.convolve(a, a))  # Track best actual max
    
    v = np.zeros(n)  # Momentum
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive temperature
        T = max(T_min, T_init * (T_min / T_init) ** (it / max_iters))
        
        grad, c = compute_smooth_max_gradient(a, T)
        
        # Update with momentum
        v = momentum * v - lr * grad
        a_new = a + v
        
        # Project: non-negative and fixed sum
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * np.sqrt(2*n) / s
        
        # Check improvement
        c_new = np.convolve(a_new, a_new)
        max_new = np.max(c_new)
        if max_new < best_val:
            best_val = max_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_val

import numpy as np
import time

def smooth_max_and_grad(a, T):
    """Compute smooth max of autocorrelation and its gradient."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - np.max(c)
    w = np.exp(c_shifted / T)
    w_sum = np.sum(w)
    w = w / w_sum
    # Gradient
    # grad_i = 2 * sum_k w_k * a_{k-i}
    # Using: result[j] = sum_k w[k] * a[n-1-j+k]
    # We need: sum_k w[k] * a[k-i] = result[n-1+i]
    if n > 256:
        # Use FFT for efficiency
        L = 1
        while L < 2*n + 2*n - 1:  # Need at least len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
            L *= 2
        W = np.fft.rfft(w, L)
        A = np.fft.rfft(a[::-1], L)
        result = np.fft.irfft(W * A, L)
    else:
        result = np.convolve(w, a[::-1])
    grad = 2.0 * result[n-1:2*n-1]
    smooth_val = np.max(c) + T * np.log(w_sum)
    return smooth_val, grad, np.max(c)

def normalize_seq(a):
    """Normalize sequence to have sum = sqrt(2n)."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) * np.sqrt(2.0/n)
    return a * np.sqrt(2*n) / s

def gd_optimize(a_init, time_budget, T_init=2.0, T_min=0.1, lr_init=0.01, momentum=0.9, 
                adapt_T=True, jump_prob=0.05, jump_scale=0.1):
    """Gradient descent on smooth max of autocorrelation."""
    a = normalize_seq(a_init.copy())
    n = len(a)
    best_a = a.copy()
    best_max = np.max(np.convolve(a, a))
    
    v = np.zeros(n)
    start = time.monotonic()
    lr = lr_init
    T = T_init
    stuck_count = 0
    
    while time.monotonic() - start < time_budget:
        smooth_val, grad, actual_max = smooth_max_and_grad(a, T)
        
        # Check improvement
        if actual_max < best_max - 1e-10:
            best_max = actual_max
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            # Normalize gradient to unit direction
            grad_dir = grad / grad_norm
        else:
            break
        
        # Adaptive temperature
        if adapt_T:
            elapsed = time.monotonic() - start
            progress = min(1.0, elapsed / time_budget)
            T = T_init * (T_min / T_init) ** progress
        
        # Update with momentum
        v = momentum * v - lr * grad_dir
        a_new = a + v
        
        # Project
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * s_target / s  # where s_target = sqrt(2n)
        
        # Sometimes jump to escape local minima
        if stuck_count > 50 and np.random.random() < jump_prob:
            a_new = a + np.random.normal(0, jump_scale, n)
            a_new = np.maximum(a_new, 0)
            a_new = normalize_seq(a_new)
            v = np.zeros(n)
            stuck_count = 0
        
        a = a_new
    
    return best_a, best_max

def _compute_grad(a, w):
    """Compute grad_i = 2 * sum_k w_k * a_{k-i}."""
    n = len(a)
    L = 1
    while L < 3 * n - 1:
        L <<= 1
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a[::-1].copy(), L)
    result = np.fft.irfft(W * A, L)
    return 2.0 * result[n-1 : 2*n-1]

def _smooth_max_grad(a, T):
    """Return (smooth_max, gradient, actual_max) of autocorrelation."""
    c = np.convolve(a, a)
    c_max = float(c.max())
    z = (c - c_max) / T
    z = np.minimum(z, 0.0)  # clip upper (already done by subtracting max)
    z = np.maximum(z, -50.0)  # clip lower to avoid underflow
    e = np.exp(z)
    s = float(e.sum())
    w = e / s
    grad = _compute_grad(a, w)
    smooth = c_max + T * np.log(s)
    return smooth, grad, c_max

def _normalize(a, S=None):
    """Project to non-negative with sum S."""
    a = np.maximum(a, 0.0)
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = float(a.sum())
    if s < 1e-14:
        return np.full(len(a), S / len(a))
    return a * (S / s)

An alternative is to use **additive gradient descent** with projection:

The advantage of additive is that it can increase some elements and decrease others (within the same step, before projection). But since grad >= 0, the additive update also only decreases all elements (before projection). So it's similar to mirror descent in that sense.

The difference is in how the "redistribution" happens after normalization:
- Mirror: a_i * exp(-lr * g_i) / sum_j (a_j * exp(-lr * g_j)) * S
- Additive: (a_i - lr * g_i)_+ / sum_j (a_j - lr * g_j)_+ * S

Mirror descent is more natural for the non-negativity constraint and tends to be more stable.
