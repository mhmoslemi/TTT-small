"""
Search for non-negative step heights minimizing

    2 * n * max(convolve(a, a)) / sum(a)^2

using:
- centered binary / formal-square-root initializations,
- LP Frank-Wolfe style mixing (the previous state-of-the-art idea),
- smoothed mirror-descent polish,
- light randomized local search.

The function propose_candidate(seed, budget_s) is the required entry point.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


LP_MAX_N = 2048


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    a = a / s
    return a


def _make_binomial(n):
    """a_k = C(2k,k)/4^k, normalized. Its first n convolution coefficients are equal."""
    if n <= 0:
        return None
    a = np.zeros(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(n - 1):
        a[k + 1] = a[k] * (2.0 * k + 1.0) / (2.0 * k + 2.0)
    return _normalize(a)


def _convolve_full(x, y):
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    if x.size == 0 or y.size == 0:
        return np.array([], dtype=np.float64)
    L = x.size + y.size - 1
    if L <= 4000:
        return np.convolve(x, y)
    N = 1 << (L - 1).bit_length()
    fx = np.fft.rfft(x, N)
    fy = np.fft.rfft(y, N)
    z = np.fft.irfft(fx * fy, N)[:L]
    return z


def _internal_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return float("inf")
    b = _convolve_full(a, a)
    if b.size == 0:
        return float("inf")
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return float("inf")
    if mb < 0.0:
        mb = 0.0
    return float(2.0 * a.size * mb / (s * s))


def _smooth_b_grad(a, lam):
    """
    Gradient of f(a) = (1/lam) * log(sum_k exp(lam * (a*a)_k)) with respect to a.
    Also returns b = a*a.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    L = 2 * n - 1
    fullL = 3 * n - 2

    if n <= 900:
        b = np.convolve(a, a)
        b = np.maximum(b, 0.0)
        max_b = float(np.max(b)) if b.size else 0.0
        if max_b <= 0.0:
            p = np.full(L, 1.0 / L, dtype=np.float64)
        else:
            p = np.exp(lam * (b - max_b))
            ps = float(p.sum())
            if ps <= 0.0 or not np.isfinite(ps):
                p = np.full(L, 1.0 / L, dtype=np.float64)
            else:
                p = p / ps

        conv_pa = np.convolve(p, a)
        p_rev = p[::-1]
        conv_apr = np.convolve(a, p_rev)
        idx = L - 1 - np.arange(n, dtype=np.int64)
        g = conv_pa[:n] + conv_apr[idx]
    else:
        N = 1 << (fullL - 1).bit_length()
        fa = np.fft.rfft(a, N)

        b = np.fft.irfft(fa * fa, N)[:L]
        b = np.maximum(b, 0.0)
        max_b = float(np.max(b)) if b.size else 0.0
        if max_b <= 0.0:
            p = np.full(L, 1.0 / L, dtype=np.float64)
        else:
            p = np.exp(lam * (b - max_b))
            ps = float(p.sum())
            if ps <= 0.0 or not np.isfinite(ps):
                p = np.full(L, 1.0 / L, dtype=np.float64)
            else:
                p = p / ps

        fp = np.fft.rfft(p, N)
        p_rev = p[::-1]
        fr = np.fft.rfft(p_rev, N)

        conv_pa = np.fft.irfft(fa * fp, N)[:n]
        conv_apr = np.fft.irfft(fa * fr, N)
        idx = L - 1 - np.arange(n, dtype=np.int64)
        g = conv_pa + conv_apr[idx]

    b = np.asarray(b, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    if not np.isfinite(g).all():
        g = np.where(np.isfinite(g), g, 0.0)
    return b, g


def _linprog_direction(f, time_limit=None):
    """
    Solve LP:
        max 1^T g
        s.t. f*g <= M, g >= 0
    where M = max(f*f). Returns (g, f_work). If infeasible/unavailable returns (None, None).
    """
    if linprog is None:
        return None, None

    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n < 2 or n > LP_MAX_N:
        return None, None

    # Keep strictly positive so LP is bounded.
    f_work = np.maximum(f, 1e-12)
    sf = float(f_work.sum())
    if not np.isfinite(sf) or sf <= 0.0:
        return None, None
    f_work = f_work / sf

    b = _convolve_full(f_work, f_work)
    M = float(np.max(np.maximum(b, 0.0)))
    if not np.isfinite(M) or M <= 0.0:
        return None, None

    # Constraint matrix A: row k, col j = f[k-j] if 0 <= k-j < n.
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f_work

    # Scale constraints to b_ub = 1 for better numerical conditioning.
    A = A / M

    c = -np.ones(n, dtype=np.float64)
    b_ub = np.ones(2 * n - 1, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = None
    opts = {"presolve": True, "threads": 1}
    if time_limit is not None and time_limit > 10.0:
        opts["time_limit"] = float(min(240.0, max(10.0, time_limit)))

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options=opts)
    except Exception:
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
        except Exception:
            try:
                res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds)
            except Exception:
                return None, None

    if res is None or not hasattr(res, "x") or res.x is None:
        return None, None

    g = np.asarray(res.x, dtype=np.float64)
    if g.size != n:
        return None, None
    g = np.maximum(g, 0.0)
    g = np.where(np.isfinite(g), g, 0.0)

    sg = float(g.sum())
    if not np.isfinite(sg) or sg <= 0.0:
        return None, None

    status = getattr(res, "status", 0)
    if status != 0:
        # Accept time-limited/iteration-limited points if they are feasible.
        try:
            ag = A @ g
            viol = float(np.max(ag - 1.0))
        except Exception:
            return None, None
        if not np.isfinite(viol) or viol > 1e-4:
            return None, None

    return g, f_work


def _best_mix(f, g, ts):
    """Evaluate candidate mixtures (1-t) f + t g and return the best one."""
    g = np.asarray(g, dtype=np.float64)
    f = np.asarray(f, dtype=np.float64)
    sg = float(g.sum())
    sf = float(f.sum())
    if not np.isfinite(sg) or not np.isfinite(sf) or sg <= 0.0 or sf <= 0.0:
        return None, float("inf")

    g_scaled = g * (sf / sg)
    best = None
    best_sc = float("inf")

    for t in ts:
        tt = float(t)
        if tt <= 0.0:
            continue
        if tt > 1.0:
            tt = 1.0
        h = (1.0 - tt) * f + tt * g_scaled
        h = np.maximum(h, 0.0)
        s = float(h.sum())
        if not np.isfinite(s) or s <= 0.0:
            continue
        h = h / s
        sc = _internal_score(h)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = h.copy()

    return best, best_sc


def _lp_refine(start, deadline, max_iters=100):
    """Iterate the LP mixing step until no improvement or deadline."""
    f = _normalize(start)
    if f is None:
        return start, float("inf")

    cur = _internal_score(f)
    best_f = f.copy()
    best_sc = cur
    if not np.isfinite(best_sc):
        return best_f, best_sc

    ts = [0.02, 0.05, 0.1, 0.2, 0.35, 0.6, 1.0]

    for it in range(max_iters):
        now = time.monotonic()
        if now >= deadline:
            break
        rem = deadline - now
        if rem < 20.0:
            break

        tl = max(10.0, min(240.0, rem - 10.0))
        g, f_work = _linprog_direction(f, tl)
        if g is None or f_work is None:
            break

        sc_work = _internal_score(f_work)
        if not np.isfinite(sc_work):
            break
        f = f_work
        cur = sc_work
        if sc_work < best_sc:
            best_f = f.copy()
            best_sc = sc_work

        cand, sc = _best_mix(f, g, ts)
        if cand is None or not np.isfinite(sc):
            break

        if sc < cur - 1e-12:
            f = cand
            cur = sc
            if sc < best_sc:
                best_f = cand.copy()
                best_sc = sc
                print(f"  LP improve len={f.size} score={sc:.6f} it={it}")
        else:
            break

    return best_f, best_sc


def _mirror_polish(start, deadline, lam, max_iters=20000, lr=0.1):
    """
    Smooth-objective mirror descent on the simplex.
    Parameterization: update log(a) in the direction a*(grad - <a,grad>).
    """
    a = _normalize(start)
    if a is None:
        return start, float("inf")

    cur = _internal_score(a)
    best = a.copy()
    best_sc = cur
    if not np.isfinite(best_sc):
        return best, best_sc

    step = lr
    no_improve = 0

    for it in range(max_iters):
        now = time.monotonic()
        if now >= deadline:
            break
        if step < 1e-4:
            break

        _, g = _smooth_b_grad(a, lam)
        if g.size != a.size:
            break

        H = float(np.dot(a, g))
        lg = a * (g - H)
        lg -= float(np.mean(lg))

        scale = float(np.max(np.abs(lg)))
        if not np.isfinite(scale) or scale <= 1e-300:
            break

        u = -step * lg / scale
        u = np.clip(u, -10.0, 10.0)
        new_a = a * np.exp(u)
        new_a = np.maximum(new_a, 1e-300)
        s = float(new_a.sum())
        if not np.isfinite(s) or s <= 0.0:
            step *= 0.5
            continue
        new_a = new_a / s

        sc = _internal_score(new_a)
        if np.isfinite(sc) and sc < cur - 1e-11:
            a = new_a
            cur = sc
            no_improve = 0
            if sc < best_sc:
                best = a.copy()
                best_sc = sc
                if it < 10 or it % 50 == 0 or sc < best_sc - 0.001:
                    print(f"  mirror polish len={a.size} score={sc:.6f} it={it}")
            step = min(1.0, step * 1.05)
        else:
            step *= 0.7
            no_improve += 1
            if no_improve > 250:
                break

    return best, best_sc


def _local_search(start, deadline, max_iters=50000, sigma=0.08):
    """Small randomized multiplicative block search as a final escape mechanism."""
    f = _normalize(start)
    if f is None:
        return start, float("inf")

    cur = _internal_score(f)
    best = f.copy()
    best_sc = cur
    if not np.isfinite(best_sc):
        return best, best_sc

    n = f.size
    if n <= 2:
        return best, best_sc

    rng = np.random.default_rng(24681357)

    for it in range(max_iters):
        now = time.monotonic()
        if now >= deadline:
            break

        base = np.maximum(f, 1e-10)
        sb = float(base.sum())
        if sb <= 0.0:
            base = f.copy()
        else:
            base = base / sb

        lo = max(1, n // 500)
        hi = max(lo + 1, min(n, n // 10 + 1))
        width = int(rng.integers(lo, hi))
        start_hi = max(1, n - width + 1)
        pos = int(rng.integers(0, start_hi))

        cand = base.copy()
        cand[pos:pos + width] *= np.exp(rng.normal(0.0, sigma, width))

        if rng.random() < 0.35:
            idx = int(rng.integers(0, n))
            cand[idx] = max(0.0, cand[idx] * np.exp(rng.normal(0.0, sigma)))

        sc = _internal_score(cand)
        if np.isfinite(sc) and sc < cur - 1e-11:
            f = cand
            cur = sc
            if sc < best_sc:
                best = f.copy()
                best_sc = sc
                if it < 10 or it % 100 == 0:
                    print(f"  local search len={f.size} score={sc:.6f} it={it}")
            sigma = min(0.30, sigma * 1.05)
        else:
            sigma *= 0.8
            if sigma < 0.004:
                break

    return best, best_sc


def _best_initial(n, rng):
    """Build a small pool of non-symmetric starters and keep the best."""
    b = _make_binomial(n)
    if b is None:
        return None, float("inf")

    rev = b[::-1]
    candidates = [b]

    for p in (0.85, 0.70):
        a = p * b + (1.0 - p) * rev
        a = a * np.exp(rng.normal(0.0, 0.03, n))
        candidates.append(a)

    x = np.linspace(0.0, 1.0, n)
    for c in (0.5, 2.0):
        candidates.append(np.exp(-c * x))
        candidates.append(np.exp(-c * (1.0 - x)))

    best = None
    best_sc = float("inf")
    for c in candidates:
        cc = _normalize(c)
        if cc is None:
            continue
        sc = _internal_score(cc)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cc
    return best, best_sc


def _update_best(best, best_score, cand, score):
    """Validating update of the global best."""
    if cand is None or not np.isfinite(score):
        return best, best_score
    cn = _normalize(cand)
    if cn is None:
        return best, best_score
    scn = _internal_score(cn)
    if np.isfinite(scn) and scn < best_score:
        return cn, scn
    return best, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main entry point: search within budget_s and return a list of floats."""
    total = float(budget_s if budget_s is not None else 1500.0)
    if not np.isfinite(total) or total <= 0.0:
        total = 1500.0

    t0 = time.monotonic()
    final_deadline = t0 + max(10.0, total) - 5.0

    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    best = None
    best_score = float("inf")

    # Use the supplied parent construction if it exists and is useful.
    parent = globals().get("height_sequence_1", None)
    if parent is not None:
        try:
            p = _normalize(parent)
            if p is not None:
                ps = _internal_score(p)
                best, best_score = _update_best(
                    best, best_score, p, ps
                )
                if np.isfinite(best_score):
                    print(f"parent len={p.size} score={best_score:.6f}")
        except Exception as e:
            print("parent processing failed:", e)

    init_cache = {}

    # Broad initial survey.
    init_lengths = [256, 384, 512, 768, 1024, 1536]
    for n in init_lengths:
        if time.monotonic() > final_deadline - 120.0:
            break
        cand, sc = _best_initial(n, rng)
        init_cache[n] = (cand, sc)
        if cand is not None and np.isfinite(sc):
            best, best_score = _update_best(best, best_score, cand, sc)
            print(f"init n={n} score={best_score:.6f}")

    # LP phase. Smaller and larger lengths get allocated portions of a shared deadine.
    lp_start = time.monotonic()
    lp_budget = max(0.0, 0.55 * max(0.0, total - 30.0))
    if lp_budget > 30.0:
        lp_deadline = min(lp_start + lp_budget, final_deadline - 60.0)
        if lp_deadline > lp_start + 20.0:
            lp_total = max(1.0, lp_deadline - lp_start)
            plans = [
                (384, 0.15),
                (512, 0.30),
                (1024, 0.25),
                (1536, 0.15),
                (2048, 0.15),
            ]
            for n, frac in plans:
                if time.monotonic() >= lp_deadline:
                    break
                alloc = max(20.0, frac * lp_total)
                n_deadline = min(lp_deadline, time.monotonic() + alloc)
                if n_deadline - time.monotonic() < 20.0:
                    continue

                if n in init_cache and init_cache[n][0] is not None:
                    start = init_cache[n][0]
                else:
                    if n <= 1536:
                        start, sc = _best_initial(n, rng)
                    else:
                        start = _make_binomial(n)
                        sc = _internal_score(start) if start is not None else float("inf")
                    init_cache[n] = (start, sc)

                if start is None:
                    continue

                cand, sc = _lp_refine(start, n_deadline, max_iters=120)
                best, best_score = _update_best(best, best_score, cand, sc)
                if np.isfinite(best_score):
                    print(f"LP phase n={n} global best score={best_score:.6f}")

    # Mirror polish the current best.
    mirror_deadline = final_deadline - 10.0
    if best is not None and time.monotonic() < mirror_deadline - 30.0:
        nbest = best.size
        for lam_mult in (20.0, 50.0, 100.0):
            if time.monotonic() >= mirror_deadline - 30.0:
                break
            stage_deadline = min(mirror_deadline - 30.0, time.monotonic() + 60.0)
            lam = max(100.0, lam_mult * float(nbest))
            cand, sc = _mirror_polish(best, stage_deadline, lam, max_iters=25000, lr=0.1)
            best, best_score = _update_best(best, best_score, cand, sc)
            if np.isfinite(best_score):
                print(f"mirror phase mult={lam_mult:.0f} score={best_score:.6f}")

    # A short final LP polish can exploit the mirror-polished base.
    if best is not None and best.size <= LP_MAX_N and time.monotonic() < final_deadline - 60.0:
        stage_deadline = min(final_deadline - 20.0, time.monotonic() + 180.0)
        cand, sc = _lp_refine(best, stage_deadline, max_iters=50)
        best, best_score = _update_best(best, best_score, cand, sc)
        if np.isfinite(best_score):
            print(f"LP polish score={best_score:.6f}")

    # Try one or two longer mirror-polished binary starts if time remains.
    if time.monotonic() < final_deadline - 60.0:
        for n in (4096, 2048):
            if best is not None and n == best.size:
                continue
            if time.monotonic() >= final_deadline - 60.0:
                break
            start = _make_binomial(n)
            if start is None:
                continue
            stage_deadline = min(final_deadline - 30.0, time.monotonic() + 120.0)
            lam = max(100.0, 50.0 * float(n))
            cand, sc = _mirror_polish(start, stage_deadline, lam, max_iters=25000, lr=0.1)
            best, best_score = _update_best(best, best_score, cand, sc)
            if np.isfinite(sc):
                print(f"long mirror n={n} score={sc:.6f} global={best_score:.6f}")

    # Final local search.
    if best is not None and time.monotonic() < final_deadline - 5.0:
        max_iters = 20000 if best.size <= 1500 else 6000
        cand, sc = _local_search(best, final_deadline - 2.0, max_iters=max_iters, sigma=0.08)
        best, best_score = _update_best(best, best_score, cand, sc)
        if np.isfinite(best_score):
            print(f"local final score={best_score:.6f}")

    if best is None:
        best = _make_binomial(512)
        if best is None:
            best = np.ones(512, dtype=np.float64)

    best = np.asarray(best, dtype=np.float64)
    best = np.where(np.isfinite(best), best, 0.0)
    best = np.clip(best, 0.0, 1000.0)
    s = float(best.sum())
    if not np.isfinite(s) or s <= 0.0:
        best = np.ones(512, dtype=np.float64)
        s = float(best.sum())
    best = best / s
    best = np.clip(best, 0.0, 1000.0)

    out = [float(x) for x in best]

    # Final sanity check with the official evaluator if available.
    try:
        fs = evaluate_sequence(out)
        if not np.isfinite(fs):
            out = [float(1.0 / 512)] * 512
            print(f"final score before fallback={fs:.6f}")
        else:
            print(f"final official score={fs:.6f} len={len(out)}")
    except Exception as e:
        print("final evaluator failed:", e)

    return out
