```python
    rng = np.random.default_rng(seed)
    a = a0.astype(float).copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-12:

    a = a/s

    bestJ = _eval_local(a)

    avg = a.copy()
```

```python
"""
Minimize J(a) = 2*n*max(conv(a,a)) / (sum(a))^2 for a non-negative step function.

Key idea (breaks the palindromic J>=2 barrier): for a palindromic sequence the
center autocorrelation equals sum(a^2) >= (sum a)^2 / n, forcing J >= 2.
For NON-palindromic sequences the peak of the convolution can be made strictly
smaller than sum(a^2), and the 2n-1 convolution entries (which sum to (sum a)^2)
can be made flatter, pushing J below 2 toward the 2n/(2n-1) lower bound.

We run a subgradient descent (active-set / max subgradient) with FFT-based
convolution, scale-normalization (J is scale invariant, so we keep sum(a)=1),
non-negativity projection, adaptive step size, plus stochastic block
multiplicative perturbations to escape local (palindromic) minima. We sweep many
lengths and diverse asymmetric initializations and keep the global best, then
polish the best configuration with a block hill-climb.
"""

import numpy as np
import time


# --------------------------------------------------------------------------- #
# Fast convolution / objective / subgradient                                  #
# --------------------------------------------------------------------------- #
def _full_conv(a):
    """Full convolution a*a (length 2n-1) via real FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(a, size)
    out = np.fft.irfft(F * F, size)
    return out[:2 * n - 1]


def _j_from(conv, S):
    return float(2.0 * len(conv) + 0.0)  # placeholder, replaced below


def _objective(a):
    """J(a) = 2n*max(conv)/(sum a)^2  (a assumed finite, non-negative)."""
    n = len(a)
    conv = _full_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return float("inf"), conv, 0.0, S
    M = float(conv.max())
    J = 2.0 * n * M / (S * S)
    return J, conv, M, S


def _subgrad_from(conv, a, M, S):
    """Subgradient of J(a) at a, given conv(a,a), its max M and sum S."""
    n = len(a)
    tol = 1e-9 * M + 1e-15
    active = np.where(conv >= M - tol)[0]
    A = max(1, active.size)
    g0 = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g0[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g0 /= A
    grad = (2.0 * n / (S * S)) * (g0 - 2.0 * M / S)
    return grad


# --------------------------------------------------------------------------- #
# Stochastic block perturbation (escape / polish)                             #
# --------------------------------------------------------------------------- #
def _block_perturb(a, rng, strength=0.12):
    n = len(a)
    c = a.astype(float).copy()
    w = max(1, n // 60)
    s = int(rng.integers(0, max(1, n - w + 1)))
    c[s:s + w] *= float(np.exp(rng.normal(0.0, strength)))
    if rng.random() < 0.25:
        c *= float(np.exp(rng.normal(0.0, strength * 0.4)))
    c = np.clip(c, 0.0, 1000.0)
    sc = c.sum()
    if sc < 1e-12:
        return a
    return c / sc


# --------------------------------------------------------------------------- #
# Core optimizer: subgradient hill-descent + stochastic block escapes         #
# --------------------------------------------------------------------------- #
def _optimize(a0, budget, seed=0):
    rng = np.random.default_rng(seed)
    a = np.clip(np.asarray(a0, dtype=float), 0.0, 1000.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones_like(a)
        S = a.sum()
    a = a / S
    n = len(a)
    Jlow = 2.0 * n / (2.0 * n - 1.0)

    Jcur, conv, Mcur, Scur = _objective(a)
    best = a.copy()
    bestJ = Jcur

    scale = 0.5
    it = 0
    t0 = time.monotonic()
    while time.monotonic() - t0 < budget:
        # --- stochastic escape: block-perturb the running best -------------
        if it % 400 == 0:
            cand = _block_perturb(best, rng, 0.12)
            Jc, _, _, _ = _objective(cand)
            if Jc < bestJ:
                bestJ = Jc
                best = cand
                a = cand.copy()
                Jcur, conv, Mcur, Scur = _objective(a)
        # --- subgradient step ---------------------------------------------
        grad = _subgrad_from(conv, a, Mcur, Scur)
        g2 = float(np.dot(grad, grad))
        if g2 < 1e-18:
            # stall (e.g. exact palindromic point): force asymmetry
            cand = _block_perturb(a, rng, 0.2)
            Jc, conv, Mcur, Scur = _objective(cand)
            if Jc < bestJ:
                bestJ = Jc; best = cand
            a = cand
            Jcur = Jc
            it += 1
            continue
        lr = max(1e-9, (bestJ - Jlow)) / (g2 + 1e-12)
        lr = min(lr, 0.5) * scale
        a2 = a - lr * grad
        a2 = np.clip(a2, 0.0, 1000.0)
        s2 = a2.sum()
        if s2 < 1e-12:
            scale *= 0.5
            continue
        a2 = a2 / s2
        Jnew, conv, Mnew, Snew = _objective(a2)
        if np.isfinite(Jnew) and Jnew < Jcur:
            scale = min(scale * 1.15, 1.0)
        else:
            scale = max(scale * 0.6, 1e-4)
        a = a2
        Jcur = Jnew
        Mcur = Mnew
        Scur = Snew
        if Jcur < bestJ:
            bestJ = Jcur
            best = a.copy()
        it += 1
    return best, bestJ


# --------------------------------------------------------------------------- #
# Standalone block hill-climb (final polish)                                  #
# --------------------------------------------------------------------------- #
def _block_hillclimb(a0, budget, seed=0):
    rng = np.random.default_rng(seed)
    a = np.clip(np.asarray(a0, dtype=float), 0.0, 1000.0)
    S = a.sum()
    if S < 1e-12:
        return a, float("inf")
    a = a / S
    Jcur = _objective(a)[0]
    best = a.copy()
    bestJ = Jcur
    t0 = time.monotonic()
    while time.monotonic() - t0 < budget:
        cand = _block_perturb(best, rng, 0.06)
        Jc = _objective(cand)[0]
        if Jc < bestJ:
            bestJ = Jc
            best = cand
    return best, bestJ

# Initializers (all asymmetric to avoid the palindromic J>=2 saddle)          #
# --------------------------------------------------------------------------- #
def _make_init(name, n, rng):
    i = np.arange(n, dtype=float)
    if name == "ramp2":
        a = np.linspace(1.0, 2.0, n)
    elif name == "ramp35":
        a = np.linspace(1.0, 3.5, n)
    elif name == "step":
        a = np.ones(n)
        a[n // 2:] = 1.6
    elif name == "sine3":
        a = 0.5 + 0.5 * np.sin(2.0 * np.pi * 3.0 * i / n)
    elif name == "sine5":
        a = 0.5 + 0.5 * np.sin(2.0 * np.pi * 5.0 * i / n + 0.7)
    elif name == "power":
        a = ((i + 1) / n) ** 0.5 + 0.3
    elif name == "noise":
        a = 1.0 + 0.05 * rng.normal(0.0, 1.0, n)
    else:  # random
        a = 0.2 + rng.random(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n)
        s = a.sum()
    return a / s

# Main entry point                                                            #
# --------------------------------------------------------------------------- #
def propose_candidate(budget_s=1500, **kwargs):
    """Search for a non-negative step function minimizing the evaluation."""
    if not np.isfinite(budget_s) or budget_s is None:
        budget_s = 1500
    budget_s = float(budget_s)
    if kwargs.get("time_s") is not None:
        try:
            budget_s = min(budget_s, float(kwargs["time_s"]))
        except Exception:
            pass
    deadline = time.monotonic() + max(10.0, budget_s - 8.0)
    print("propose_candidate: total budget %.1fs" % budget_s, flush=True)

    # ---- safety-net starting point: previous construction -----------------
    best_seq = None
    best_score = float("inf")
    try:
        parent = np.asarray(height_sequence_1, dtype=float)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size >= 2 and np.isfinite(parent).all():
            ps = float(evaluate_sequence([float(x) for x in parent]))
            if np.isfinite(ps):
                best_score = ps
                best_seq = [float(x) for x in parent]
                print("parent score = %.6f (len=%d)" % (ps, parent.size), flush=True)
    except Exception as e:
        print("could not load parent:", e, flush=True)
    if best_seq is None:
        u = np.ones(64)
        ps = float(evaluate_sequence([float(x) for x in u]))
        best_score = ps
        best_seq = [float(x) for x in u]

    lengths = [32, 64, 128, 256, 512, 1024, 2048, 4096]
    inits = ["ramp2", "ramp35", "step", "sine3", "sine5", "power", "noise", "random"]

    configs = []
    for nidx, n in enumerate(lengths):
        for iname in inits:
            configs.append((n, iname, nidx + hash(iname)))
    # add the parent as its own config if its length is in a sane range
    try:
        pl = len(np.asarray(height_sequence_1, dtype=float))
        if 16 <= pl <= 8192:
            configs.append((pl, "parent", 12345))
    except Exception:
        pass

    per = max(2.0, min(22.0, (deadline - time.monotonic()) / max(1, len(configs))))
    print("num configs = %d, per-config time ~ %.1fs" % (len(configs), per), flush=True)

    for ci, (n, iname, seed) in enumerate(configs):
        if time.monotonic() > deadline - 4.0:
            break
        cfg_budget = min(per, deadline - time.monotonic() - 4.0)
        if cfg_budget < 1.5:
            continue
        try:
            if iname == "parent":
                arr = np.asarray(height_sequence_1, dtype=float)
                sc = arr.sum()
                a0 = np.clip(arr, 0.0, None) / sc
            else:
                a0 = _make_init(iname, n, np.random.default_rng(seed))
            cand, cJ = _optimize(a0, cfg_budget, seed=1000 + seed)
            if np.isfinite(cJ) and cJ < best_score - 1e-9:
                # confirm with the authoritative evaluator
                ls = [float(x) for x in cand]
                real = float(evaluate_sequence(ls))
                if np.isfinite(real) and real < best_score:
                    best_score = real
                    best_seq = ls
                    print("  [%d/%d] n=%d init=%s  J=%.6f  (global best)"
                          % (ci + 1, len(configs), n, iname, real), flush=True)
            elif int(ci) % 8 == 0:
                print("  [%d/%d] n=%d init=%s  localJ=%.6f  best=%.6f  t=%.0fs"
                      % (ci + 1, len(configs), n, iname, cJ, best_score,
                         time.monotonic() - (deadline - budget_s + 8.0)), flush=True)
        except Exception as e:
            print("  config n=%d init=%s error: %s" % (n, iname, e), flush=True)

    # ---- final polish of the global best ---------------------------------
    remaining = deadline - time.monotonic()
    if best_seq is not None and remaining > 15.0:
        b = np.asarray(best_seq, dtype=float)
        pb, pJ = _block_hillclimb(b, min(120.0, remaining - 2.0), seed=777)
        ls = [float(x) for x in pb]
        real = float(evaluate_sequence(ls))
        if np.isfinite(real) and real < best_score:
            best_score = real
            best_seq = ls
            print("polish improved to %.6f" % real, flush=True)
        print("final best score = %.6f (len=%d)" % (best_score, len(best_seq)), flush=True)

    # ---- guarantee a valid, safe return ----------------------------------
    if best_seq is None:
        best_seq = [1.0] * 64
    out = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    s = sum(out)
    if s < 0.05:
        out = [x / s for x in out] if s > 0 else [1.0] * 64
    return out
```
