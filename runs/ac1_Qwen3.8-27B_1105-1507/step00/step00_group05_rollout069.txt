            if new_score < best_score:
                best_score = new_score
                if count % 1000 == 0 or new_score < best_score - 0.01:
                    print(f"    P3: new best = {best_score:.6f} at t={time.monotonic()-t0:.0f}s")
        else:
            delta = new_score - cur_score
            if delta < 10.0 * T and rng.random() < np.exp(-delta / max(T, 1e-10)):
                a = candidate
                cur_score = new_score

        T = max(T * 0.9995, T_min)
        count += 1

        if count % 50000 == 0:
            print(f"    P3: count={count}, cur={cur_score:.6f}, best={best_score:.6f}, "
                  f"T={T:.2e}, t={time.monotonic()-t0:.0f}s")

    print(f"  Phase 3: {count} moves, best={best_score:.6f} in {time.monotonic()-t0:.1f}s")
    return a, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Phase 1: Structured asymmetric initializations.
    Phase 2: Subgradient descent on autocorrelation peak.
    Phase 3: Simulated annealing with diverse mutation moves.
    Returns the best non-negative sequence found.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(10.0, float(budget_s) - 10.0)

    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad parent")
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    print(f"Sequence length n = {n}")

    # Phase 1: Structured initialization (reserve 30s)
    p1_deadline = min(time.monotonic() + 40.0, total_deadline - 400.0)
    best_a, best_score = _phase1_search(n, parent, rng, p1_deadline)

    # Phase 2: Subgradient optimization (reserve 200s for phase 3)
    p2_deadline = min(total_deadline - 200.0, time.monotonic() + 800.0)
    best_a, best_score = _phase2_subgradient(best_a, n, p2_deadline, rng)

    # Phase 3: Simulated annealing (remaining time)
    p3_deadline = total_deadline - 2.0
    if time.monotonic() < p3_deadline:
        best_a, best_score = _phase3_anneal(best_a, n, best_score, best_score, p3_deadline, rng)

    # Verify with actual evaluate_sequence
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    print(f"Final verified score: {final_score:.6f}")

    # Compare with parent
    parent_score = float(evaluate_sequence(parent.tolist()))
    if parent_score < final_score and np.isfinite(parent_score):
        print(f"Parent better: {parent_score:.6f} vs {final_score:.6f}")
        return [float(x) for x in parent]

    return final_list
