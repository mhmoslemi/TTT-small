"""
Minimizer for  2 * n * max(convolve(a,a)) / (sum(a))**2   with a >= 0.

Combines Gerchberg-Saxton iterative time/frequency projection (drives the
periodogram toward flat, which flattens the autocorrelation) with a smoothed
log-sum-exp gradient descent stage for fine polishing. Explores several
sequence lengths and diverse initializations, keeping the global best.
"""

import time
import numpy as np


def _next_pow2(n):
    p = 1
    while p < int(n):
        p <<= 1
    return p


def _fft_autocorr(a):
    """Linear convolution of a with itself via FFT (equals np.convolve(a,a))."""
    n = len(a)
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def _project(a):
    """Project onto {x >= 0, sum(x) = 1} by water-filling (bisection)."""
    n = len(a)
    lo = float(a.min()) - 1e-9
    hi = float(a.max()) + 1e-9
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.maximum(0.0, a - mid).sum() > 1.0:
            lo = mid
        else:
            hi = mid
    rho = 0.5 * (lo + hi)
    res = np.maximum(0.0, a - rho)
    tot = res.sum()
    if tot > 1e-15:
        res /= tot
    else:
        res = np.full(n, 1.0 / n, dtype=np.float64)
    return res


def _gs_iteration(a, N, relax):
    """One GS step: force flat non-DC magnitude, keep phase, project back."""
    n = len(a)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    s2 = float(np.dot(a, a))
    c2 = (N * s2 - mag[0] ** 2) / (N - 1.0)
    c = np.sqrt(max(c2, 0.0))
    phase = np.ones_like(A)
    nz = mag > 1e-30
    phase[nz] = A[nz] / mag[nz]
    At = np.zeros_like(A)
    At[0] = A[0]
    At[1:] = c * phase[1:]
    ap = np.fft.irfft(At, N)
    proj = np.clip(ap[:n], 0.0, None)
    tot = proj.sum()
    if tot > 1e-15:
        proj = proj / tot
    else:
        proj = _project(a)
    return (1.0 - relax) * a + relax * proj

def _smoothed_grad(a, t):
    """Gradient of (1/t) log sum exp(t C[k]) with C = convolve(a,a)."""
    n = len(a)
    C = _fft_autocorr(a)
    tc = t * C
    tc = tc - tc.max()
    e = np.exp(tc)
    p = e / e.sum()
    obj = (tc.max() + np.log(e.sum())) / t
    size = _next_pow2((2 * n - 1) + n - 1)
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    cross = np.fft.irfft(P * np.conj(A), size)
    return 2.0 * cross[:n], obj


def _run_gradient(a_init, deadline, max_iters, t_start, t_end):
