```python
def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    cond = u * np.arange(1, n+1) - cssv > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
import numpy as np
import time

def project_simplex(v):
    """Project v onto the probability simplex {x: x_i >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) - cv > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cv[rho] / (rho + 1)
    return np.maximum(v - theta, 0)

def compute_subgradient(a, n_active=5):
    """Compute subgradient of max_k (convolve(a,a))_k."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    # Find active set
    active = np.where(c >= max_c * 0.999)[0]  # 99.9% of max
    if len(active) > n_active:
        # Keep only the top n_active
        top = np.argsort(c)[::-1][:n_active]
        active = top
    # Compute subgradient
    n = len(a)
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += 2 * a[k-hi:k-lo+1][::-1]
    grad /= len(active)
    return grad, c, max_c

def gs_initialization(n, n_iter=100, seed=42):

        # Project to non-negative
```
