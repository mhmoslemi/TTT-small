import numpy as np
import time

def self_convolve(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_n = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_n)
    return np.fft.irfft(fa * fa, fft_n)[:size]

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1} using sort-based method."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1) / rho
    rho_star = int(np.max(np.where(cond, rho, 0)))
    if rho_star == 0:
        return np.ones(n) / n
    theta = (cssv[rho_star - 1] - 1) / rho_star
    return np.maximum(v - theta, 0)

def compute_smoothed_grad(a, conv, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv_k))."""
    n = len(a)
    max_c = np.max(conv)
    weights = np.exp(tau * (conv - max_c))
    weights /= np.sum(weights)
    # grad[j] = 2 * sum_k weights[k] * a[k-j] (with a[m]=0 for m out of [0,n-1])
    # = 2 * cross_correlation(weights, a)[j] for j=0..n-1
    total = len(weights) + n - 1
    fft_n = 1 << (total - 1).bit_length()
    W = np.fft.rfft(weights, fft_n)
    A = np.fft.rfft(a, fft_n)
    cross = np.fft.irfft(W * np.conj(A), fft_n)
    grad = 2.0 * cross[n - 1:n - 1 + n]
    return grad

def self_convolve(a):

            s_sum = a_new.sum()
            if s_sum < 1e-12:
                a_new = a
            else:
                a_new = a_new / s_sum

            conv_new = self_convolve(a_new)
            obj_new = 2.0 * n * np.max(conv_new)

            if obj_new < cur_obj:
                a = a_new
                conv = conv_new
                cur_obj = obj_new
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
                    if best_obj < 1.51 or it % 2000 == 0:
                        print(f"  Start {si}, iter {it}, t={time.monotonic()-t0:.1f}s: obj={best_obj:.6f}")
                # Occasionally increase eta
                if it % 500 == 0:
                    eta = min(eta * 1.01, 2.0 / np.sqrt(n))
            else:
                eta *= 0.85
                momentum_v *= 0.5

    print(f"After Phase 1: best_obj = {best_obj:.6f}")

    # Phase 2: Focused optimization on best with higher tau
    a = best_a.copy()
    conv = self_convolve(a)
    cur_obj = 2.0 * n * np.max(conv)

    momentum_v = np.zeros(n)
    beta = 0.9
    eta = 0.3 / np.sqrt(n)
    tau = 30.0

    while time.monotonic() < phase2_end:
        grad = compute_smoothed_grad(a, conv, tau)
        momentum_v = beta * momentum_v + eta * grad
        a_new = a - momentum_v
        a_new = np.maximum(a_new, 0.0)
        s_sum = a_new.sum()
        if s_sum < 1e-12:
            break
        a_new = a_new / s_sum

        conv_new = self_convolve(a_new)
        obj_new = 2.0 * n * np.max(conv_new)

        if obj_new < cur_obj:
            a = a_new
            conv = conv_new
            cur_obj = obj_new
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if best_obj < 1.51:
                    print(f"  Phase2: obj={best_obj:.6f}")
            if best_obj < 1.45:
                tau = min(tau * 1.05, 200.0)
        else:
            eta *= 0.93
            tau = min(tau * 1.1, 200.0)

    print(f"After Phase 2: best_obj = {best_obj:.6f}")

    # Phase 3: Coordinate descent polish
    a = best_a.copy()
    conv = self_convolve(a)
    cur_obj = 2.0 * n * np.max(conv)

    while time.monotonic() < deadline:
        # Random coordinate scaling
        idx = rng.integers(0, n)
        a[idx] *= np.exp(rng.normal(0, 0.05))
        a[idx] = max(0.0, a[idx])
        a = a / a.sum()
        conv_new = self_convolve(a)
        obj_new = 2.0 * n * np.max(conv_new)
        if obj_new < cur_obj:
            conv = conv_new
            cur_obj = obj_new
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                if best_obj < 1.50:
                    print(f"  Phase3: obj={best_obj:.6f}")
        else:
            # Revert
            a[idx] /= np.exp(rng.normal(0, 0.05))  # Won't exactly revert, but re-normalize
            a = np.maximum(a, 0.0)
            ss = a.sum()
            if ss > 1e-12:
                a = a / ss
            # Try small positive step instead
            a2 = best_a.copy()
            idx2 = rng.integers(0, n)
            a2[idx2] *= (1.0 + rng.normal(0, 0.02))
            a2 = np.maximum(a2, 0.0)
            a2 = a2 / a2.sum()
            conv2 = self_convolve(a2)
            obj2 = 2.0 * n * np.max(conv2)
            if obj2 < best_obj:
                best_obj = obj2
                best_a = a2
                a = a2
                conv = conv2
                cur_obj = obj2

    # Verify with actual evaluator
    final_seq = [float(x) for x in best_a]
    final_score = float(evaluate_sequence(final_seq))
    print(f"Final verified objective: {final_score:.6f}")

    return final_seq
