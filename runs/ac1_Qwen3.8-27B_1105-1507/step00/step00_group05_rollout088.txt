"""
Minimizes J = 2n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Uses smooth-max gradient descent with FFT-based convolution, initialized
with Pascal (central binomial) sequences that achieve J ~ pi/2 ~ 1.57.
Iteratively flattens the autocorrelation to push J below 1.5.
"""

import numpy as np
import time

def pascal_fast(n):
    """Generate central binomial sequence C(2i,i)/4^i for i=0..n-1 using recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2.0*i - 1.0) / (2.0*i)
    return a

def fft_convolve(a, b):
    """Full convolution of 1D arrays using FFT."""
    na, nb = len(a), len(b)
    n = na + nb - 1
    size = 1 << (n - 1).bit_length()
    if size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def compute_objective(a):
    """Compute 2n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def smooth_max_weights(c, tau):
    """Compute softmax weights for smooth max approximation."""
    shifted = c / tau - np.max(c / tau)
    w = np.exp(shifted)
    total = w.sum()
    if total > 0:
        w /= total
    return w

def composite_gradient(a, w, n):
    """Compute composite gradient of smooth max w.r.t. a.
    grad[j] = 2 * sum_k a[k] * w[j+k] for valid k, j.
    Uses FFT: 2 * convolve(a[::-1], w)[n-1 : 2n-2]
    """
    a_rev = a[::-1]
    conv = fft_convolve(a_rev, w)
    return 2.0 * conv[n-1:2*n-2]

        c = fft_convolve(a, a)
        c_max = c.max()

        tau = tau_frac * c_max

"""
Minimizes J = 2n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Uses smooth-max gradient descent with FFT-based convolution.
Initialized with Pascal (central binomial) sequences achieving J ~ pi/2,
then iteratively flattens the autocorrelation to push J toward 1.5.
"""

import numpy as np
import time


def pascal_fast(n):
    """Generate C(2i,i)/4^i for i=0..n-1 using recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def fft_convolve(a, b):
    """Full linear convolution of two 1D real arrays via FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def smooth_max_weights(c, tau):
    """Numerically stable softmax of c/tau."""
    shifted = c / tau - (c.max() / tau)
    w = np.exp(shifted)
    s = w.sum()
    if s > 0:
        w /= s
    return w


def composite_gradient(a, w, n):
    """Gradient of smooth-max w.r.t. a.
    grad[j] = 2 * sum_k a[k] * w[j+k] for valid (k, j+k).
    Computed as 2 * convolve(a[::-1], w)[n-1 : 2n-2].
    """
    conv = fft_convolve(a[::-1], w)
    return 2.0 * conv[n - 1:2 * n - 1]


def descend(a, deadline, rng, max_iters=500000):
    """Smooth-max multiplicative gradient descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_j = 2.0 * n * fft_convolve(a, a).max()
    lr = 0.4
    tau_frac = 0.01
    stuck = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fft_convolve(a, a)
        c_max = c.max()
        j = 2.0 * n * c_max

        if j < best_j:
            best_j = j
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Smooth max weights
        tau = tau_frac * c_max
        if tau < 1e-18:
            break
        w = smooth_max_weights(c, tau)

        # Gradient
        grad = composite_gradient(a, w, n)
        mg = grad.max()
        if mg <= 1e-18:
            break

        # Multiplicative update
        a = a * np.exp(-lr * grad / mg)
        ns = a.sum()
        if ns < 1e-10:
            a = best_a.copy()
        else:
            a = a / ns

        # Schedule
        lr = max(0.03, lr * 0.99985)
        if it % 2000 == 1000:
            tau_frac = min(0.2, tau_frac * 1.03)

        # Reheat if stuck

            a = np.maximum(a, 0.0)
            a = a / a.sum()
            lr = min(0.6, lr * 3.0)
            stuck = 0

        if it % 2000 == 0:
            print(f"      iter={it} J={j:.6f} best={best_j:.6f} lr={lr:.4f} tau={tau_frac:.4f}")

    # Final exact check on best
    fs = evaluate_sequence(best_a.tolist())
    if fs < best_j:
        best_j = fs
    return best_a, best_j

"""
Minimizes J = 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Uses smooth-max multiplicative gradient descent with FFT-based convolution.
Initialized with Pascal (central binomial) sequences achieving J ~ pi/2 ~ 1.57,
then iteratively flattens the autocorrelation to push J below 1.5.
Multiple initializations (one-sided, symmetric, tilted Pascal) are tried.
"""

import numpy as np
import time


def pascal_fast(n):
    """Generate C(2i,i)/4^i for i=0..n-1 using recurrence (avoids overflow)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def fft_convolve(a, b):
    """Full linear convolution of two 1D real arrays via FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def smooth_max_weights(c, tau):
    """Numerically stable softmax of c/tau."""
    shifted = c / tau - (c.max() / tau)
    w = np.exp(shifted)
    s = w.sum()
    if s > 0:
        w /= s
    return w


def composite_gradient(a, w, n):
    """
    Gradient of smooth-max objective w.r.t. a.
    grad[j] = 2 * sum_k a[k] * w[j+k] for valid indices.
    Computed as 2 * convolve(a[::-1], w)[n-1 : 2n-1].
    """
    conv = fft_convolve(a[::-1], w)
    return 2.0 * conv[n - 1:2 * n - 1]


def descend(a, deadline, rng, max_iters=2000000):
    """
    Smooth-max multiplicative gradient descent to minimize max(convolve(a,a))
    subject to sum(a)=1, a>=0.
    Returns (best_a, best_score).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_j = 2.0 * n * fft_convolve(a, a).max()
    lr = 0.4
    stuck = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Learning rate schedule: slowly decaying
        lr = max(0.02, 0.4 * np.exp(-it / 150000.0))
        # Temperature: slowly increasing to explore broader lags
        tau_frac = 0.01 + 0.00002 * min(it, 100000)

        # Compute convolution
        c = fft_convolve(a, a)
        c_max = c.max()
        j = 2.0 * n * c_max

        if j < best_j:
            best_j = j
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Smooth max weights
        tau = tau_frac * c_max
        if tau < 1e-18:
            break
        w = smooth_max_weights(c, tau)

        # Composite gradient
        grad = composite_gradient(a, w, n)
        mg = grad.max()
        if mg <= 1e-18:
            break

        # Multiplicative update (preserves non-negativity)
        a = a * np.exp(-lr * grad / mg)
        ns = a.sum()
        if ns < 1e-10:
            a = best_a.copy()
        else:
            a = a / ns

            a = a * np.exp(rng.normal(0.0, 0.04, n))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            stuck = 0

    # Final check
    fs = evaluate_sequence(best_a.tolist())
    if fs < best_j:
        best_j = fs
    return best_a, best_j
