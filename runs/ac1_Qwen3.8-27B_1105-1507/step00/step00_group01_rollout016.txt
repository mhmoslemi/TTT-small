def compute_obj(a):

    n = len(a)

"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses alternating LP: fix one convolution factor, solve LP for the other.
Tries multiple sequence lengths and random restarts to escape local optima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix for LP. A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_step(a, upper_bound=1000.0):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, 0 <= g_i <= ub."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M


def run_lp_optimize(n, seed=0, max_iter=300, t=0.5, verbose=False):
    """Run LP-based iterations to minimize the objective for sequence of length n."""
    rng = np.random.default_rng(seed)

                a = a / s
                stall = 0
            continue
        stall = 0

        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)

            break

                    best_obj = obj2
                    a = a_new2
                else:
                    a = a_new  # still advance
            else:

def random_refine(a, rng, max_iter=5000):
    """Random block perturbation local search."""
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    n = len(best_a)
    
    for it in range(max_iter):
        width = max(1, n // 40)
        start = int(rng.integers(0, n - width + 1))

        candidate = np.clip(candidate, 0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        obj_new = compute_obj(candidate)
        if obj_new < best_obj:
            best_a = candidate
            best_obj = obj_new
    
    return best_a, best_obj
