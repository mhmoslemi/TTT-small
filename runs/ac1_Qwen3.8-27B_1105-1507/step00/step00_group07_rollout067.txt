def compute_score(a):

def fft_cross_corr(a, s, n):
    """Compute cross[i] = sum_j a[j] * s[j+i] for i=0..n-1 using FFT."""
    N = 1
    while N < 3 * n:
        N *= 2
    A = np.fft.fft(a, N)
    S = np.fft.fft(s, N)
    return np.fft.ifft(A * np.conj(S)).real[:n]


def gradient_direction(a, tau):
    """Compute gradient direction and diagnostic values."""
    n = len(a)
    b = np.convolve(a, a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.zeros(n), b, sum_a, 0.0

    b_scaled = b / tau
    b_max = np.max(b_scaled)
    exp_b = np.exp(b_scaled - b_max)
    Z = np.sum(exp_b)
    if Z < 1e-300:
        softmax_b = np.zeros_like(b)
        softmax_b[np.argmax(b)] = 1.0
        logsumexp_val = b_max
    else:
        softmax_b = exp_b / Z
        logsumexp_val = b_max + np.log(Z)

    cross = fft_cross_corr(a, softmax_b, n)
    grad = cross - logsumexp_val * tau / sum_a
    return grad, b, sum_a, logsumexp_val

def run_lp_step(a, M, solver='CBC'):
    """
    One LP step: maximize sum(b) subject to b>=0 and (a*b)_k <= M for all k.
    Returns the optimal b rescaled, or None if LP fails.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a)
    b = cp.Variable(n, nonnegative=True)
    objective = cp.Maximize(cp.sum(b))

    constraints = []
    # Convolution constraints: (a * b)_k <= M for k = 0..2n-2
    # (a*b)[k] = sum_i a[i] * b[k-i] for valid i, k-i
    # Build as: for each k, sum of a[i]*b[k-i] <= M
    # To be efficient, only add constraints where sum(a[i] for valid i) > 0
    # This is dense, so let's just build the constraint matrix.
    # For n up to ~2000, this is feasible. For larger n, might be slow.
    if n > 3000:
        return None

    # Build constraint matrix: row k has b[j] coefficient = a[k-j] (if 0<=k-j<n)
    # Actually (a*b)[k] = sum_j a[k-j]*b[j] where 0<=k-j<n and 0<=j<n
    # So coefficient of b[j] in row k is a[k-j] if 0 <= k-j < n, else 0.
    # We'll build this as a sparse-like dense matrix for small n.

    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        # b[j] appears in (a*b)[k] for k = j, j+1, ..., j+n-1
        # with coefficient a[k-j] = a[0], a[1], ..., a[n-1]
        for i in range(n):
            A_mat[j + i, j] = a[i]

    for k in range(2 * n - 1):
        row_sum = np.sum(A_mat[k])
        if row_sum > 1e-10:
            constraints.append(A_mat[k] @ b <= M)

    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=getattr(cp, solver, cp.CBC), max_time=30)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK, max_time=30)
        except Exception:
            return None

    if b.value is None:
        return None

    b_val = np.clip(np.asarray(b.value, dtype=np.float64), 0, None)
    if np.sum(b_val) < 1e-10:
        return None
    return b_val
