        rs = rs_new

def hankel_matvec(Q, a):
    """Compute (Ma)[i] = sum_j a[j] * Q[i+j] for i = 0,...,n-1 using FFT."""
    n = len(a)
    # We need convolve(a[::-1], Q)[n-1 : 2n-1]
    # Use FFT
    m = 2 * n - 1  # length of Q
    size = 1
    while size < n + m - 1:
        size <<= 1
    Af = np.fft.rfft(a[::-1], size)
    Qf = np.fft.rfft(Q, size)
    result = np.fft.irfft(Af * Qf)
    return result[n-1 : 2*n-1]

def hankel_matvec(Q, a):
    """Compute (Ma)[i] = sum_j a[j] * Q[i+j] for i=0..n-1 using FFT."""
    n = len(a)
    m = len(Q)  # should be 2n-1
    # convolve(a[::-1], Q) gives result of length n + m - 1 = 3n - 2
    # We want indices n-1 to 2n-2

        a = a_new
        f_prev = f_new
    
    return a


def run_fictitious_play(n, a_init, max_iters, deadline, label=""):
    """Run fictitious play iterations from a_init."""
    a = np.maximum(0.0, a_init.copy())
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n

    # Initialize Q as triangular distribution (autocorrelation of uniform sequence)
    k = np.arange(2 * n - 1, dtype=np.float64)
    Q = np.minimum(k + 1.0, 2.0 * n - 1.0 - k)
    Q /= Q.sum()

    local_best_a = a.copy()
    local_best_score = 2.0 * n * fft_convolve(a, a).max()
    
    print(f"  {label} start: score={local_best_score:.6f}")

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Solve QP subproblem
        a = solve_subproblem(Q, n, a, n_iters=40)

        # Evaluate
        c = fft_convolve(a, a)
        score = 2.0 * n * c.max()

        if score < local_best_score:
            local_best_score = score
            local_best_a = a.copy()
            if it % 20 == 0 or score < 1.6:
                print(f"    {label} iter {it}: score={score:.6f}")

        # Update Q with exponential tilting
        # Use normalized autocorrelation as new Q
        Q_new = c / c.sum()
        Q = 0.15 * Q + 0.85 * Q_new

    return local_best_a, local_best_score


def local_search_polish(a, n, deadline, rng, max_iters=300):
    """Polish with subgradient-based local search on the peak autocorrelation."""
    a = a.copy()
    s = a.sum()
    if s > 0:
        a /= s
    cur_score = 2.0 * n * fft_convolve(a, a).max()
    best_a = a.copy()
    best_score = cur_score
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fft_convolve(a, a)
        # Find top peaks
        top_k = np.argsort(c)[-8:]
        weights = c[top_k]
        weights = weights / weights.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for pk, w in zip(top_k, weights):
            pk = int(pk)
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            if hi >= lo:
                # grad[i] += w * a[pk - i] for i in [lo, hi]
                # pk - i ranges from pk-lo down to pk-hi
                grad[lo:hi + 1] += w * a[pk - lo:pk - hi - 1:-1] if pk - lo != pk - hi - 1 else w * a[pk - lo:hi - 1:-1]

        # Actually let me redo this more carefully
        grad = np.zeros(n)
        for pk, w in zip(top_k, weights):
            pk = int(pk)
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            for i in range(lo, hi + 1):
                grad[i] += w * a[pk - i]

        # Try multiple step sizes
        improved = False
        for eta in [0.02, 0.05, 0.1, 0.01, 0.2]:
            candidate = a - eta * grad
            candidate = np.maximum(0.0, candidate)
            sc = candidate.sum()
            if sc < 1e-30:
                continue
            candidate /= sc
            new_score = 2.0 * n * fft_convolve(candidate, candidate).max()
            if new_score < cur_score - 1e-12:
                a = candidate
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                improved = True
                break

        if not improved and it % 50 == 0 and it > 0:
            # Random perturbation to escape
            noise = rng.normal(0, 0.02, n)
            candidate = a + noise
            candidate = np.maximum(0.0, candidate)
            sc = candidate.sum()
            if sc > 1e-30:
                candidate /= sc
                new_score = 2.0 * n * fft_convolve(candidate, candidate).max()
                if new_score < cur_score:
                    a = candidate
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()

        if it % 100 == 0:
            print(f"    Polish iter {it}: score={best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Fictitious play + QP subproblem + local search for peak autocorrelation minimization.
    
    The objective 2n*max(f*f)/(sum f)^2 >= 2n/(2n-1) by averaging argument.
    We use the zero-sum game formulation to approach this bound.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0

    # Get and validate parent
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad parent")
    except Exception:
        parent = np.ones(1000, dtype=np.float64)

    n = len(parent)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial: n={n}, score={best_score:.6f}, theoretical lower bound={2.0*n/(2*n-1):.6f}")

    # Build multiple starting points
    starts = {}
    parent_sum = parent.sum()
    if parent_sum > 0:
        starts['parent'] = parent / parent_sum
    starts['uniform'] = np.ones(n) / n

    t = np.linspace(0, 1, n)
    # Edge-weighted: U-shape
    edge_w = 0.3 + 0.7 * (2 * t - 1) ** 2
    starts['edge'] = edge_w / edge_w.sum()

    # "Arch" shape: high in middle, low at edges (inverse of edge)
    arch_w = 1.0 - 0.5 * (2 * t - 1) ** 4
    arch_w = np.maximum(0.05, arch_w)
    starts['arch'] = arch_w / arch_w.sum()

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(np.pi * t))
    cos_w = np.maximum(0.01, cos_w)
    starts['cosine'] = cos_w / cos_w.sum()

    # Two peaks
    two_peak = 0.3 + 0.7 * np.exp(-((t - 0.25) / 0.1) ** 2) + 0.7 * np.exp(-((t - 0.75) / 0.1) ** 2)
    starts['two_peak'] = two_peak / two_peak.sum()

    # Run fictitious play from each starting point
    time_per_start = (deadline - time.monotonic()) / len(starts)
    for name, a0 in starts.items():
        if time.monotonic() > deadline:
            break
        print(f"\n=== Fictitious play from '{name}' (time budget: {time_per_start:.0f}s) ===")
        local_deadline = min(deadline, time.monotonic() + time_per_start * 1.5)
        max_iters = max(10, int(time_per_start / 0.5))  # estimate ~0.5s per iter
        a_opt, score = run_fictitious_play(n, a0, max_iters, local_deadline, label=name)

        if score < best_score:
            # Scale back to original magnitude
            candidate = a_opt * parent_sum
            actual = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best = candidate.copy()
                print(f"  >>> New best: {best_score:.6f}")
            elif np.isfinite(actual):
                # Still track it
                if score < best_score:
                    best_score = score
                    best = candidate.copy()

    # Phase 2: Local search polish on the best found
    if time.monotonic() < deadline - 5:
        print(f"\n=== Local search polish (current best: {best_score:.6f}) ===")
        a_polish = best / best.sum() if best.sum() > 0 else np.ones(n) / n
        a_polished, polish_score = local_search_polish(
            a_polish, n, deadline, rng,
            max_iters=max(50, int((deadline - time.monotonic()) / 0.1))
        )
        if polish_score < best_score:
            candidate = a_polished * parent_sum
            actual = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best = candidate.copy()
                print(f"  >>> Polished best: {best_score:.6f}")

    # Phase 3: Quick random restarts if time remains
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Random restarts (remaining: {remaining:.0f}s) ===")
        for restart in range(5):
            if time.monotonic() > deadline:
                break
            a_rand = np.abs(rng.normal(0, 1, n)) + 0.1
            a_rand /= a_rand.sum()
            # Quick fictitious play (fewer iterations)
            max_iters_quick = max(5, int(remaining / 5 / 0.5))
            a_opt, score = run_fictitious_play(n, a_rand, max_iters_quick, deadline, label=f"rand{restart}")
            if score < best_score:
                candidate = a_opt * parent_sum
                actual = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(actual) and actual < best_score:
                    best_score = actual
                    best = candidate.copy()
                    print(f"  >>> Restart best: {best_score:.6f}")

    # Final check
    final_score = float(evaluate_sequence(best.tolist()))
    print(f"\nFinal result: score={final_score:.6f} (tracked: {best_score:.6f})")

    return [float(x) for x in best]
