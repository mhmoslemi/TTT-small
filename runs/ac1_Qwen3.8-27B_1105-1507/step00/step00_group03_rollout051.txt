def softmax_weighted_gradient(b, T):
    """Compute softmax-weighted subgradient of max(b)."""
    b_max = np.max(b)
    w = np.exp((b - b_max) / T)
    w = w / np.sum(w)
    return w

def correlate(w, a):
    """Compute cross-correlation: result[j] = sum_i w[i+j] * a[i] for j=0..len(a)-1."""
    # w has length 2n-1, a has length n
    # result has length n
    return np.correlate(w, a, 'valid')
