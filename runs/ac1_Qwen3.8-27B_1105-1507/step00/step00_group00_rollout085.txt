    return b / b.sum()

    return 2.0 * n * max_b / (sum_a * sum_a)

We want to minimize F(a) = max_k (a*a)[k] subject to a ≥ 0, sum(a) = 1.

- Zero-pad w to size N
- Compute Ifft(fft(a_rev_padded) * fft(w_padded))
- Take elements at indices n-1 to 2n-2

import numpy as np
import time
import math

"""
Search for optimal non-negative sequences minimizing 2n*max(conv)/(sum)^2.

Key insight: Truncated central binomial coefficient sequences (coefficients of
(1-x)^(-1/2)) have exactly flat auto-convolution in the interior (lags 0 to n-1),
giving score = 2n/S(n)^2 where S(n) = sum_{k=0}^{n-1} C(2k,k)/4^k. This
approaches pi/2 from below for large n but has a finite-n minimum. We sweep over
lengths and apply gradient-based local optimization using softmax approximation
of the max convolution to further reduce the score.
"""


def central_binomial(n):
    """Generate normalized central binomial coefficient sequence of length n."""
    if n <= 0:
        return np.array([])
    b = np.empty(n, dtype=np.float64)
    b[0] = 1.0
    for i in range(1, n):
        b[i] = b[i - 1] * (2 * i - 1) / (2 * i)
    s = b.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return b / s


def compute_score_internal(a):
    """Fast internal score computation (same as evaluate_sequence but no validation)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_gradient(a, beta=50.0):
    """Compute gradient of log-sum-exp approximation of max(conv)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = conv.max()

    # Softmax weights (shifted for numerical stability)
    exp_vals = np.exp(beta * (conv - max_conv))
    w = exp_vals / exp_vals.sum()

    # Gradient: g_i = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # Computed as (a_rev * w)[n-1+i] using FFT
    size = 1
    while size < 3 * n:
        size *= 2

    a_rev = a[::-1]
    A = np.zeros(size, dtype=np.float64)
    A[:n] = a_rev
    W = np.zeros(size, dtype=np.float64)
    W[:2 * n - 1] = w

    Conv = np.fft.ifft(np.fft.fft(A) * np.fft.fft(W)).real
    g = 2.0 * Conv[n - 1:2 * n - 2]

    return g, conv, max_conv


def optimize_gradient(a, max_time=60.0, beta_start=30.0, verbose=False):
    """Gradient-based optimization of the central binomial sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return a
    a = a / s

    best = a.copy()
    best_score = compute_score_internal(a)
    lr = 0.1
    beta = beta_start
    patience = 0
    start_time = time.monotonic()
    iteration = 0

    while time.monotonic() - start_time < max_time:
        g, conv, max_conv = compute_gradient(a, beta)

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Step size: scale by max_conv / g_norm for invariance
        step = lr * max_conv / g_norm
        a_new = a - step * g / g_norm * max_conv
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        new_score = compute_score_internal(a_new)

        if new_score < best_score:
            if verbose and iteration % 100 == 0:
                print(f"    iter {iteration}: score={new_score:.6f}")
            best = a_new.copy()
            best_score = new_score
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience % 20 == 0:
                lr *= 0.7
            if patience % 50 == 0:
                beta = min(beta * 1.4, 500.0)

        if patience > 300:
            break

        iteration += 1

    return best


def block_perturb(base, rng):
    """Apply a random block perturbation to the sequence."""
    n = len(base)
    a = base.copy()
    n_blocks = rng.integers(1, max(2, n // 100))
    for _ in range(n_blocks):
        width = max(2, int(n * rng.uniform(0.005, 0.1)))
        start = int(rng.integers(0, max(1, n - width)))
        end = min(n, start + width)
        a[start:end] *= np.exp(rng.normal(0.0, rng.uniform(0.05, 0.2)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-12:
        a = a / s
    return a
