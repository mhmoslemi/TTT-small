import numpy as np

def compute_conv(f):
    """Compute full convolution of f with itself using FFT for speed."""
    n = len(f)
    size = 2 * n  # Next power of 2 >= 2n-1
    while size < 2 * n - 1:
        size *= 2
    F = np.fft.fft(f, size)
    return np.real(np.fft.ifft(F * F)[:2*n-1])

"""
Optimizes non-negative step function to minimize 2n*max(conv(f,f))/(sum(f))^2.
Uses mirror descent (multiplicative updates) followed by LP-based Frank-Wolfe
refinement. Tries multiple initializations and sizes.
"""

import numpy as np
import time

def _fft_convolve(f):
    """Fast convolution using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2*n-1]

def _score(f):
    """Compute evaluation score (same as evaluate_sequence but faster)."""
    n = len(f)
    h = _fft_convolve(f)
    M = np.max(h)
    S = np.sum(f)
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)

def _mirror_descent(f, n_iter=5000, eta_init=0.01, eta_min=0.0001):
    """Mirror descent with multiplicative updates."""
    f = f.astype(np.float64).copy()
    S = np.sum(f)
    f = f / S
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    eta = eta_init
    
    for it in range(n_iter):
        h = _fft_convolve(f)
        M = np.max(h)
        S = np.sum(f)
        
        # Active set: lags where h is within 20% of max
        threshold = 0.8 * M
        active = np.where(h >= threshold)[0]
        
        # Compute pressure vector p
        p = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # p[js:je+1] += 2 * f[k-je : k-js+1][::-1] = 2 * f[js:k-js+1]... 
            # f[k-j] for j in [js, je] = f[k-je] to f[k-js]
            # = f[k-je : k-js+1] (in order from j=je to j=js, i.e., reversed)
            # For p[j], the contribution from lag k is 2*f[k-j]
            # j=js: 2*f[k-js], j=js+1: 2*f[k-js-1], ..., j=je: 2*f[k-je]
            # So p[js:je+1] += 2 * f[k-js : k-je-1 : -1] = 2 * f[k-je : k-js+1][::-1]
            # k-js >= 0 and k-je < n (by construction)
            p[js:je+1] += 2.0 * f[k-je:k-js+1][::-1]
        
        p /= len(active)
        
        # Update
        grad = (p * S - 2.0 * M) / (S * S)
        f = f * np.exp(-eta * grad)
        f = np.maximum(f, 1e-30)
        S = np.sum(f)
        f = f / S
        
        # Adaptive eta
        eta = max(eta_min, eta_init * (0.9995 ** it))
        
        score = _score(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        if it % 1000 == 0 and it > 0:
            print(f"  MD iter {it}: score={best_score:.6f}, eta={eta:.6f}")
    
    return best_f, best_score

def _lp_refine(f, time_budget=60):
    """LP-based Frank-Wolfe refinement."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    f = f.astype(np.float64).copy()
    S = np.sum(f)
    f = f / S
    n = len(f)
    best_f = f.copy()
    best_score = _score(f)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget:
            break
        
        h = np.convolve(f, f)
        M = np.max(h)
        
        # Build constraint matrix
        rows = []
        cols = []
        vals = []
        for j in range(n):
            for i in range(n):
                rows.append(j + i)
                cols.append(j)
                vals.append(f[i])
        
        C = csr_matrix((np.array(vals), (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))), shape=(2*n-1, n))
        
        res = linprog(-np.ones(n), A_ub=C, b_ub=np.full(2*n-1, M), method='highs')
        
        if not res.success:
            break
        
        g = res.x
        Sg = np.sum(g)
        if Sg < 1e-10:
            break
        g = g / Sg
        
        t = 2.0 / (it + 2)
        f_new = np.maximum((1-t)*f + t*g, 0)
        Sn = np.sum(f_new)
        if Sn < 1e-10:
            break
        f_new = f_new / Sn
        
        score = _score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        
        f = f_new
        if it % 50 == 0 and it > 0:
            print(f"  LP iter {it}: score={best_score:.6f}")
    
    return best_f, best_score

def _cosine_init(n):
    j = np.arange(n)
    f = np.cos(np.pi * j / n) ** 2
    f = np.maximum(f, 0)
    f /= np.sum(f)
    return f

    sigma = sigma_frac * n
    f = np.exp(-j**2 / (2*sigma**2))
    f /= np.sum(f)
    return f
