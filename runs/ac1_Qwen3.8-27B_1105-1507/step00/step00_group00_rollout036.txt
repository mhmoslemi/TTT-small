"""
Iterative Convolution Flattening via Mirror Descent.

Minimizes 2n*max(autoconvolve(a,a))/(sum(a))^2 by iteratively reducing
elements that contribute most to convolution peaks. Uses the analytical
gradient (cross-correlation) computed via np.correlate for efficiency.
Adaptive peak-sharpening parameter and step size with periodic restarts.
"""

import numpy as np
import time


def flip_step(a, p, alpha, epsilon=1e-12):
    """One multiplicative mirror-descent step to flatten auto-convolution."""
    n = len(a)
    c = np.convolve(a, a)
    peak = c.max()
    if peak < 1e-15:
        return a
    
    # Soft weights emphasizing peaks
    w = (c / peak) ** p
    
    # Pressure: h[j] = sum_m a[m] * w[m+j] = np.correlate(w, a, 'full')[j]
    h = np.correlate(w, a, 'full')[:n]
    h = np.maximum(h, 1e-30)
    
    # Multiplicative update (mirror descent)
    geo_mean = np.exp(np.mean(np.log(h)))
    ratio = geo_mean / h
    a_new = a * np.power(ratio, alpha)
    
    # Floor to prevent permanent zeros
    floor = epsilon * (a.sum() / n)
    a_new = np.maximum(a_new, floor)
    
    # Renormalize to preserve sum (score is scale-invariant but this helps stability)
    s = a.sum()
    s_new = a_new.sum()
    if s_new > 0:
        a_new *= s / s_new
    
    return a_new
