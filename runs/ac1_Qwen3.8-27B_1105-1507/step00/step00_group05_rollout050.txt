def compute_grad_l2(a):
    """Compute gradient of ||conv(a,a)||^2 with respect to a."""
    c = np.convolve(a, a)  # length 2n-1
    n = len(a)
    # grad[i] = 4 * sum_k c[k] * a[k-i] for valid k-i
    # This is 4 * correlate(c, a, 'valid')
    v = np.correlate(c, a, mode='valid')  # length n
    return 4.0 * v

def compute_grad_max(a, T):
    """Compute gradient of softmax(T * conv(a,a)) with respect to a."""
    c = np.convolve(a, a)  # length 2n-1
    n = len(a)
    # Softmax weights
    c_max = np.max(c)
    w = np.exp(T * (c - c_max))
    w /= np.sum(w)
    # Gradient: 2 * sum_k w[k] * a[k-i] for valid k-i
    v = np.correlate(w, a, mode='valid')  # length n
    return 2.0 * v

def normalize(a, target_sum):
    s = np.sum(a)
    if s < 1e-10:
        return np.full(len(a), target_sum / len(a))
    return a * target_sum / s

    """Optimize ||conv(a,a)||^2 using gradient descent."""
    rng = np.random.default_rng(seed)
    n = len(a)

        obj = np.sum(c**2)
        if obj < best_obj:

            best_a = a.copy()

        a = np.maximum(a, 0)
        a = normalize(a, target_sum)

"""
Multi-phase gradient optimization to minimize J = 2n*max(conv(a,a))/(sum(a))^2.
Uses continuation method: L2 -> L4 -> explicit max, with non-negativity projection.
"""

import numpy as np
import time


def compute_conv(a):
    """Compute full linear convolution of a with itself."""
    return np.convolve(a, a)


def normalize_sequence(a, target_sum):
    """Normalize a to have the given sum, maintaining non-negativity."""
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    return a * (target_sum / s)


def grad_power(a, p):
    """
    Gradient of sum_k (conv(a,a)[k])^p with respect to a.
    = 2p * sum_k c[k]^(p-1) * a[k-i] for valid indices.
    """
    c = compute_conv(a)
    n = len(a)
    if p == 1:
        weights = np.ones_like(c)
    elif p == 2:
        weights = c
    elif p == 4:
        c_clipped = np.maximum(c, 0)
        weights = c_clipped ** 3
    elif p == 8:
        c_clipped = np.maximum(c, 0)
        weights = c_clipped ** 7
    else:
        c_clipped = np.maximum(c, 0)
        weights = c_clipped ** (p - 1)
    v = np.correlate(weights, a, mode='valid')
    return 2.0 * p * v


def grad_max_explicit(a):
    """
    Subgradient of max(conv(a,a)) with respect to a.
    Average over all positions achieving the maximum.
    """
    c = compute_conv(a)
    n = len(a)
    max_val = np.max(c)
    # Find all positions achieving (near) the max
    threshold = max_val * (1.0 - 1e-10)
    mask = c >= threshold
    if np.sum(mask) == 0:
        mask[np.argmax(c)] = True
    # Average the subgradients
    v = np.zeros(n)
    for k in np.where(mask)[0]:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        # c[k] = sum_{j=lo}^{hi-1} a[j] * a[k-j]
        # d c[k] / d a[i] = 2 * a[k-i] if 0 <= k-i < n
        idx = k - np.arange(lo, hi)
        valid = (idx >= 0) & (idx < n)
        v[lo:hi][valid] += a[idx[valid]]
    v /= np.sum(mask)
    return 2.0 * v


def grad_max_softmax(a, T):
    """
    Gradient of log-sum-exp(T * conv(a,a)) as a smooth max approximation.
    Returns gradient of sum_k w[k] * c[k] where w = softmax(T*c).
    (Ignoring the derivative of w w.r.t. a, which is lower order for large T.)
    """
    c = compute_conv(a)
    n = len(a)
    c_shifted = c - np.max(c)
    w = np.exp(T * c_shifted)
    w_sum = np.sum(w)
    w = w / w_sum
    v = np.correlate(w, a, mode='valid')
    return 2.0 * v

    """
    Run an optimization phase.

    Returns best a found during this phase and its score.
    """
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = a.copy()
    a = np.maximum(a, 0)
    a = normalize_sequence(a, target_sum)

    best_a = a.copy()
    best_score = float('inf')
    
    for it in range(num_iters):

        max_c = np.max(c)
        sum_a = np.sum(a)
        score = 2.0 * n * max_c / (sum_a ** 2)
        if score < best_score:

            best_a = a.copy()

        # Compute gradient

        elif phase_type == 4:

        elif phase_type == 8:

        elif phase_type == 'max_softmax':
