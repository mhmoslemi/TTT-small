import numpy as np

def fft_convolve_square(f):
    """Compute linear convolution f*f using FFT. Returns array of length 2n-1."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    return np.fft.irfft(F * F, fft_size)[:size]


def topk_objective_and_gradient(f, K=5):
    """
    Compute the objective (2n * mean_topK(C) / S^2) and its gradient.
    Top-K smoothing makes the max differentiable.
    """
    n = len(f)
    conv = fft_convolve_square(f)
    S = np.sum(f)
    
    if S < 1e-15:
        return 1e10, np.zeros(n)
    
    # Get top K indices
    if K >= len(conv):
        top_k_idx = np.arange(len(conv))
    else:
        # Use argpartition for efficiency, then sort just the top K
        part_idx = np.argpartition(conv, -K)[-K:]
        top_k_idx = part_idx[np.argsort(conv[part_idx])[::-1]]
    
    top_vals = conv[top_k_idx]
    M = np.mean(top_vals)
    
    # Value
    value = 2.0 * n * M / (S * S)
    
    # Gradient of M = mean of top C_k w.r.t. f_m:
    # dC_k/df_m = 2*f_{k-m} if 0 <= k-m < n
    # dM/df_m = (1/K) * sum_{k in topK} 2*f_{k-m} * [0<=k-m<n]
    g_M = np.zeros(n)
    for k in top_k_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            sl_start = k - hi
            sl_end = k - lo + 1
            g_M[lo:hi + 1] += f[sl_start:sl_end][::-1]
    g_M *= 2.0 / K
    
    # d(obj)/df_m = (2n/S^3) * (dM/df_m * S - 2*M)
    grad = (2.0 * n / (S ** 3)) * (g_M * S - 2.0 * M)
    
    return value, grad

"""
Minimizes 2n*max(f*f)/(sum(f))^2 using L-BFGS-B with progressive top-K smoothing,
FFT-accelerated convolution, and multiple diverse restarts.
The objective is scale-invariant, so only shape optimization is needed.
Progressive K reduction (30->1) transitions from smooth to sharp objective.

import numpy as np

def fft_convolve_square(f):
    """Compute linear convolution f*f using FFT. Returns array of length 2n-1."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    return np.fft.irfft(F * F, fft_size)[:size]


def topk_objective_and_gradient(f, K=5):
    """
    Compute the objective (2n * mean_topK(C) / S^2) and its gradient.
    Top-K smoothing makes the max differentiable.
    """
    n = len(f)
    conv = fft_convolve_square(f)
    S = np.sum(f)
    
    if S < 1e-15:

        top_k_idx = np.arange(len(conv))
    else:
        part_idx = np.argpartition(conv, -K)[-K:]
        top_k_idx = part_idx[np.argsort(conv[part_idx])[::-1]]
    
    top_vals = conv[top_k_idx]
    M = np.mean(top_vals)
    
    value = 2.0 * n * M / (S * S)
    
    # Gradient of M = mean of top C_k w.r.t. f_m
    g_M = np.zeros(n)
    for k in top_k_idx:
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo <= hi:

    grad = (2.0 * n / (S ** 3)) * (g_M * S - 2.0 * M)
    
    return value, grad
