def water_filling(x, target_sum=1.0):
    """Project x onto {v >= 0, sum(v) = target_sum} via bisection on tau."""
    n = len(x)
    if target_sum <= 0:
        return np.zeros(n)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.maximum(x - mid, 0.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    result = np.maximum(x - tau, 0.0)
    s = float(np.sum(result))
    if s > 1e-15:
        result *= target_sum / s
    return result


def build_conv_matrix(a):
    """Build sparse matrix A where (A @ b)[m] = sum_j a[j]*b[m-j] (full conv)."""
    n = len(a)
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    rows = cols + np.tile(np.arange(n, dtype=np.int32), n)
    vals = np.repeat(a, n)
    mat = sp.coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return mat.tocsr()

    i_end = min(m, n - 1)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[m - idx]
    return g
