"""
Minimize J = 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
  - Linearg ramps are an analytic near-optimal family (J -> 4/3 as n grows),
    already beating the 1.5027 target and the previous 2.0 best.
  - Seed with several ramp lengths, then run a projected / annealed local
    search that (a) uses the exact gradient of the peak convolution term
    (holding the argmax position fixed) to push mass off the dominant
    autocorrelation lag, and (b) multiplical block perturbations to escape
    local minima. Scale invariance is exploited by normalizing sum to 1.
  - Fast convolution via FFT for long sequences, direct for short ones.
  - Robust time budget with valid fallback.
"""
import time
import numpy as np

try:
    from scipy.signal import fftconvolve as _fftconv
except Exception:
    _fftconv = None

_FFT_CUTOFF = 1024


def _conv_a(a):
    a = np.asarray(a, dtype=np.float64)
    if _fftconv is not None and a.size >= _FFT_CUTOFF:
        return np.asarray(_fftconv(a, a), dtype=np.float64)
    return np.convolve(a, a)


def _raw_score(a):
    """Internal fast score identical to evaluate_sequence for valid inputs."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0 or not np.isfinite(a).all():
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    b = _conv_a(a)
    max_b = float(np.max(b))
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _normalize(a):
    s = float(np.sum(a))
    if s <= 0.0:
        a = np.ones_like(a)
        s = a.size
    return a / s


def _ramp(n):
    return _normalize(np.arange(1, n + 1, dtype=np.float64))


def _peak_grad(a, max_b):
    """Gradient of the peak convolution term b[k*] w.r.t. a (sum fixed)."""
    b = _conv_a(a)
    k = int(np.argmax(b))
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    # b[k] = sum_i a[i]*a[k-i]  ->  db[k]/da[i] = a[k-i] (0<=k-i<n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    for i in range(lo, hi + 1):
        g[i] = a[k - i]
    return g, k


def _grad_step(a, lr):
    g, k = _peak_grad(a, None)
    a = a - lr * g
    a = np.clip(a, 0.0, None)
    return a, k


def _block_perturb(a, rng, width_range, sigma, sym_prob=0.3):
    n = a.size
    cand = a.copy()
    wlo, whi = width_range
    if whi <= 1:
        whi = max(2, n // 50)
    w = int(rng.integers(wlo, max(wlo + 1, whi)))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = min(n, start + w)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, sigma)))
    if rng.random() < sym_prob:
        cand = 0.5 * (cand + cand[::-1])
    return cand


def _anneal_accept(cand_score, cur_score, temp, rng):
    if cand_score < cur_score:
        return True
    if temp <= 0.0:
        return False
    return bool(rng.random() < min(1.0, np.exp(-(cand_score - cur_score) / (temp + 1e-12))))


def _local_search(base, budget_s, seed):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    a = _normalize(base.copy())
    cur = _raw_score(a)
    best = a.copy()
    best_score = cur

    lr_base = 0.02
    temp = 1e-3
    no_improve = 0
    max_no_improve = 400

    while time.monotonic() < deadline:
        # Attempt a gradient step (reduces the dominant lag).
        cand, k = _grad_step(a, lr_base * (0.5 + rng.random()))
        cand = _normalize(cand)
        cs = _raw_score(cand)
        if np.isfinite(cs) and _anneal_accept(cs, cur, temp, rng):
            a = cand
            cur = cs
        # Occasionally a multiplicative block move to escape.
        if rng.random() < 0.35:
            cand2 = _block_perturb(a, rng, (max(2, a.size // 200), max(3, a.size // 20)),
                                   sigma=0.10 + 0.3 * temp, sym_prob=0.25)
            cand2 = _normalize(cand2)
            cs2 = _raw_score(cand2)
            if np.isfinite(cs2) and _anneal_accept(cs2, cur, temp, rng):
                a = cand2
                cur = cs2

        if cur < best_score - 1e-12:
            best_score = cur
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # annealing schedule
        temp *= 0.995
        if no_improve > max_no_improve:
            no_improve = 0
            temp = max(temp, 1e-3)
            lr_base *= 0.7

    return best, best_score
