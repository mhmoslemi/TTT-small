```
def local_conv_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0: return float('inf')
    a = np.clip(a,0.0,1000.0)
    s = float(np.sum(a))
```

```
def make_candidate(n, seed, kind=None, parent=None):
    rng=np.random.default_rng(seed)
    m=(n+1)//2
    if kind is None: kind=int(rng.integers(0,6))
    if kind==0:
       h = rng.uniform(0.05,1.0,m)
    elif kind==1:
       # smoothed random
       h = rng.normal(size=m)
       # simple moving average window 3-10
       w = int(rng.integers(3, min(11,m+1)))
       kernel = np.ones(w)/w
       h = np.convolve(h, kernel, mode='same')
       h = np.clip(h,0,None)
    elif kind==2:
       # two-level from half x
       x = rng.choice([-1.0,1.0], size=m)
       alpha = float(rng.choice([0.3,0.5,0.7,1.0]))
       h = (1.0 + alpha*x)/2.0
    elif kind==3:
       # blocks
       nb = int(rng.integers(max(2,m//16), max(3,m//4)))
       h = np.repeat(rng.uniform(0.1,1.0,nb), m//nb + 1)[:m]
       # smooth slightly
       if m>3:
           h = 0.5*h +0.5*np.convolve(h, np.ones(3)/3, mode='same')
    elif kind==4:

           if p.size != m:
               # linear interpolate
```

```python
"""
Search for low-score non-negative step-function coefficients.

The objective is
    score = 2*n*max(convolve(a,a)) / sum(a)^2.

This program uses:
  - scale normalization to sum 1,
  - multiplicative subgradient/perturbation updates on active convolution peaks,
  - an LP relaxation step inspired by the previous state-of-the-art,
  - both full and symmetry-reduced variables to reduce LP size,
  - several randomized starts and lengths, with the best found sequence returned.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def normalize_to_sum1(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return None
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def force_sym_norm(a):
    a = normalize_to_sum1(a)
    if a is None:
        return None
    a = 0.5 * (a + a[::-1])
    return normalize_to_sum1(a)


def local_score(a):
    """Exact same objective as evaluate_sequence, assuming finite nonnegative values."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    if not np.isfinite(a).all():
        return float('inf')
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float('inf')
    return float(2.0 * len(a) * m / (s * s))


def sym_expand(h, n):
    """Expand symmetric half h to full length n."""
    h = np.asarray(h, dtype=np.float64)
    m = (n + 1) // 2
    if h.size < m:
        pad = np.zeros(m - h.size, dtype=np.float64)
        h = np.concatenate((h, pad))
    else:
        h = h[:m]
    if n == 1:
        return h.copy()
    if n % 2 == 0:
        return np.concatenate((h, h[::-1]))
    else:
        if m == 1:
            return np.array([h[0]], dtype=np.float64)
        return np.concatenate((h, h[-2::-1]))


def make_candidate(n, seed, kind=None, parent=None, force_sym=False):
    rng = np.random.default_rng(seed + 1000003 * (kind if kind is not None else 0))
    m = (n + 1) // 2
    L = m if force_sym else n

    if kind is None:
        kind = int(rng.integers(0, 6))
    kind = int(kind) % 6

    val = None

    if kind == 0:
        val = rng.uniform(0.05, 1.0, L)

    elif kind == 1:
        val = rng.normal(size=L)
        w = int(rng.integers(3, max(4, min(11, L + 1))))
        kern = np.ones(w, dtype=np.float64)
        kern /= float(w)
        val = np.convolve(val, kern, mode='same')
        val = np.clip(val, 0.0, None)

    elif kind == 2:
        x = rng.choice(np.array([-1.0, 1.0], dtype=np.float64), size=L)
        alpha = float(rng.choice(np.array([0.25, 0.4, 0.5, 0.65, 0.8, 1.0], dtype=np.float64)))
        val = 0.5 * (1.0 + alpha * x)

    elif kind == 3:
        nb = int(rng.integers(max(2, L // 24), max(3, L // 6) + 1))
        nb = min(nb, max(1, L))
        bv = rng.uniform(0.1, 1.0, nb)
        val = np.repeat(bv, max(1, L // nb + 1))[:L]
        if L > 4:
            kern = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            val = 0.5 * val + 0.5 * np.convolve(val, kern, mode='same')
        val = np.clip(val, 0.0, None)

    elif kind == 4:
        p = parent
        if p is not None:
            try:
                p = np.asarray(p, dtype=np.float64)
                p = np.clip(p, 0.0, 1000.0)
            except Exception:
                p = None
        if p is None or p.size == 0:
            val = rng.uniform(0.1, 1.0, L)
        elif p.size == 1:
            val = np.full(L, float(p.flat[0]), dtype=np.float64)
        else:
            old = np.linspace(0.0, 1.0, p.size)
            new = np.linspace(0.0, 1.0, L)
            val = np.interp(new, old, p)
            val = np.clip(val, 0.0, 1000.0)
        val = val * np.exp(rng.normal(0.0, 0.35, L))
        val = np.clip(val, 0.01, 1000.0)

    else:
        val = rng.beta(2.0, 2.0, L)

    val = np.asarray(val, dtype=np.float64)
    if force_sym:
        a = sym_expand(val, n)
        a = force_sym_norm(a)
    else:
        a = normalize_to_sum1(val)

    if a is None:
        a = np.ones(n, dtype=np.float64) / float(n)
        if force_sym:
            a = force_sym_norm(a)
    return a


def multiplicative_update(a, eta, K, force_sym=False):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a
    c = np.convolve(a, a)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return a.copy()

    if K is None or K <= 0:
        K = max(1, n // 4)
    K = int(K)
    if K >= c.size:
        active = np.nonzero(c > 1e-15)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(c))], dtype=np.int64)
    else:
        active = np.argpartition(c, -K)[-K:]

    w = c[active] / M
    h = np.zeros(n, dtype=np.float64)

    for idx in range(active.size):
        k = int(active[idx])
        wk = float(w[idx])
        s = k - (n - 1)
        if s < 0:
            s = 0
        e = k
        if e >= n:
            e = n - 1
        if s > e:
            continue
        i = np.arange(s, e + 1, dtype=np.int64)
        j = k - i
        h[i] += wk * a[j]
        h[j] += wk * a[i]

    maxh = float(np.max(h))
    if not np.isfinite(maxh) or maxh <= 0.0:
        return a.copy()

    meanh = float(np.mean(h))
    fac = eta * (meanh - h) / maxh
    fac = np.clip(fac, -5.0, 5.0)
    a2 = a * np.exp(fac)
    a2 = np.clip(a2, 0.0, 1000.0)

    if force_sym:
        a2 = 0.5 * (a2 + a2[::-1])
        a2 = normalize_to_sum1(a2)
    else:
        a2 = normalize_to_sum1(a2)

    if a2 is None:
        return a.copy()
    return a2


def multiplicative_optimize(a, max_time, seed, force_sym=False):
    a = normalize_to_sum1(a)
    if force_sym:
        a = force_sym_norm(a)
    if a is None:
        return None, float('inf')

    best = a.copy()
    bestsc = local_score(best)
    if max_time <= 0.0:
        return best, bestsc

    t0 = time.monotonic()
    deadline = t0 + float(max_time)
    rng = np.random.default_rng(seed)
    n = best.size

    K = int(np.clip(max(20, n // 5), 8, 160))
    eta = 0.35
    stuck = 0
    it = 0

    while time.monotonic() < deadline:
        if it % 250 == 0:
            mult = 0.75 + 0.5 * float(rng.random())
            K = int(np.clip(int(K * mult), 8, min(240, max(8, n))))

        a = multiplicative_update(best if stuck > 120 else a, eta, K, force_sym)
        sc = local_score(a)

        if np.isfinite(sc) and sc < bestsc:
            best = a.copy()
            bestsc = float(sc)
            stuck = 0
            if eta > 0.15:
                eta = float(np.clip(eta * 1.02, 0.05, 0.85))
        else:
            stuck += 1

        if stuck > 100:
            if eta > 0.05:
                eta *= 0.75
            else:
                noise = np.exp(rng.normal(0.0, 0.18, n))
                a = best * noise
                if force_sym:
                    a = force_sym_norm(a)
                else:
                    a = normalize_to_sum1(a)
                if a is None:
                    a = best.copy()
                eta = 0.35
                stuck = 0

        it += 1

    return best, bestsc


def build_A_full(f, n):
    """Dense constraint matrix for LP step with full variables."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        if s < 0:
            s = 0
        e = k
        if e >= n:
            e = n - 1
        if s <= e:
            j = np.arange(s, e + 1, dtype=np.int64)
            A[k, j] = f[k - j]
    return A


def build_A_sym(f, n):
    """Dense constraint matrix for LP step with symmetry variables."""
    m = (n + 1) // 2
    A = np.zeros((n, m), dtype=np.float64)
    for v in range(m):
        A[v:n, v] += f[:n - v]
        j2 = n - 1 - v
        if j2 != v:
            A[j2:n, v] += f[:v + 1]
    return A


def lp_solve_step(f, n, force_sym, seed):
    if not HAVE_SCIPY:
        return None

    f = np.asarray(f, dtype=np.float64)
    if force_sym:
        f = force_sym_norm(f)
    else:
        f = normalize_to_sum1(f)
    if f is None:
        return None

    c = np.convolve(f, f)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return None

    rng = np.random.default_rng(seed + 987654321)

    try:
        if force_sym:
            m = (n + 1) // 2
            A = build_A_sym(f, n)
            rows = n
        else:
            m = n
            A = build_A_full(f, n)
            rows = 2 * n - 1

        cvec = -np.ones(m, dtype=np.float64)
        # Tiny objective noise helps avoid pathological vertices when the LP optimum
        # set is large.
        cvec += 1e-8 * rng.uniform(-1.0, 1.0, m)
        bvec = np.full(rows, M * (1.0 + 1e-7), dtype=np.float64)
        bounds = [(0.0, 1.0)] * m

        res = linprog(
            cvec,
            A_ub=A,
            b_ub=bvec,
            bounds=bounds,
            method='highs',
            options={'presolve': True}
        )
    except MemoryError:
        return None
    except Exception:
        return None

    if res is None or not res.success:
        return None

    x = np.asarray(res.x, dtype=np.float64)
    x = np.clip(x, 0.0, 1.0)
    if x.size == 0 or float(np.sum(x)) <= 0.0:
        return None

    if force_sym:
        g = sym_expand(x, n)
        g = force_sym_norm(g)
    else:
        g = normalize_to_sum1(x)

    return g


def lp_optimize(a, max_time, seed, force_sym=False):
    f = normalize_to_sum1(a)
    if force_sym:
        f = force_sym_norm(f)
    if f is None:
        return None, float('inf')

    best = f.copy()
    bestsc = local_score(best)
    if max_time <= 0.0:
        return best, bestsc

    t0 = time.monotonic()
    deadline = t0 + float(max_time)
    ts = (0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7, 1.0)
    stall = 0

    while time.monotonic() < deadline:
        if not HAVE_SCIPY:
            break

        rem = deadline - time.monotonic()
        if rem < 3.0:
            break

        g = lp_solve_step(best, len(best), force_sym, seed)
        if g is None:
            break

        new_best = best
        new_sc = bestsc

        for t in ts:
            new = (1.0 - t) * best + t * g
            if force_sym:
                new = force_sym_norm(new)
            else:
                new = normalize_to_sum1(new)
            if new is None:
                continue
            sc = local_score(new)
            if np.isfinite(sc) and sc < new_sc:
                new_sc = float(sc)
                new_best = new.copy()

        if new_sc < bestsc:
            best = new_best
            bestsc = new_sc
            stall = 0
        else:
            stall += 1
            if stall > 10:
                break

            # Small multiplicative polish between LP calls.
            K = max(20, min(80, len(best) // 4))
            for _ in range(8):
                if time.monotonic() >= deadline:
                    break
                cand = multiplicative_update(best, 0.22, K, force_sym)
                sc = local_score(cand)
                if np.isfinite(sc) and sc < bestsc:
                    best = cand
                    bestsc = float(sc)
                    break

    return best, bestsc


def block_perturb(a, rng):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return a
    out = a.copy()
    wmin = max(1, n // 40)
    wmax = max(wmin + 1, n // 8)
    w = int(rng.integers(wmin, min(wmax, n) + 1))
    start = int(rng.integers(0, max(1, n - w + 1)))
    out[start:start + w] *= float(np.exp(rng.normal(0.0, 0.18)))
    if rng.random() < 0.35:
        out = 0.5 * (out + out[::-1])
    return out


def block_local_search(a, max_time, seed):
    a = normalize_to_sum1(a)
    if a is None:
        return None, float('inf')
    best = a.copy()
    bestsc = local_score(best)
    if max_time <= 0.0:
        return best, bestsc

    t0 = time.monotonic()
    deadline = t0 + float(max_time)
    rng = np.random.default_rng(seed)
    n = best.size
    stuck = 0

    while time.monotonic() < deadline:
        cand = block_perturb(best, rng)
        cand = normalize_to_sum1(cand)
        if cand is None:
            cand = best.copy()

        sc = local_score(cand)
        if np.isfinite(sc) and sc < bestsc:
            best = cand.copy()
            bestsc = float(sc)
            stuck = 0
        else:
            stuck += 1

        if stuck > 250:
            noise = np.exp(rng.normal(0.0, 0.08, n))
            cand = best * noise
            cand = normalize_to_sum1(cand)
            if cand is None:
                cand = best.copy()
            sc = local_score(cand)
            if np.isfinite(sc) and sc < bestsc:
                best = cand.copy()
                bestsc = float(sc)
            stuck = 0

    return best, bestsc


def optimize_length(n, budget, seed, force_sym, parent):
    if n <= 1 or budget <= 0.0:
        return None, float('inf')

    t0 = time.monotonic()
    deadline = t0 + float(budget)
    best = None
    bestsc = float('inf')
    kinds = (0, 1, 2, 3, 4, 5)

    warmup = min(max(8.0, 0.15 * float(budget)), 90.0)
    warm_deadline = t0 + warmup

    trial = 0
    while time.monotonic() < warm_deadline and trial < 8:
        kind = kinds[trial % len(kinds)]
        cand = make_candidate(n, seed + 101 * trial + kind, kind, parent, force_sym)
        sc = local_score(cand)

        mb = min(18.0, max(1.5, 0.5 * (warm_deadline - time.monotonic())))
        if mb > 0.5:
            c2, sc2 = multiplicative_optimize(cand, mb, seed + 777 * trial, force_sym)
            if c2 is not None and sc2 < sc:
                cand, sc = c2, sc2

        if np.isfinite(sc) and sc < bestsc:
            best = cand.copy()
            bestsc = float(sc)

        trial += 1

    stall = 0
    last_update = time.monotonic()

    while time.monotonic() < deadline:
        if best is None:
            best = make_candidate(n, seed, 0, parent, force_sym)
            bestsc = local_score(best)

        rem = deadline - time.monotonic()
        if rem < 4.0:
            break

        if HAVE_SCIPY:
            lp_b = min(max(15.0, 0.55 * rem), 180.0)
            c_lp, sc_lp = lp_optimize(best, lp_b, seed + 1234567, force_sym)
            if c_lp is not None and sc_lp < bestsc:
                best = c_lp.copy()
                bestsc = float(sc_lp)
                last_update = time.monotonic()
                stall = 0
            else:
                stall += 1

        rem = deadline - time.monotonic()
        if rem > 3.0:
            mul_b = min(max(5.0, 0.25 * rem), 80.0)
            c_mu, sc_mu = multiplicative_optimize(best, mul_b, seed + 2345678, force_sym)
            if c_mu is not None and sc_mu < bestsc:

                stall = 0
            else:
                stall += 1

        if stall >= 3 and (time.monotonic() - last_update) > max(45.0, 0.2 * float(budget)):
            cand = make_candidate(n, seed + 3456789 + 17 * stall, int(stall) % 6, parent, force_sym)
            sc = local_score(cand)
            if np.isfinite(sc) and sc <= bestsc:
                best = cand.copy()
                bestsc = float(sc)
            else:
                blend = 0.5 * best + 0.5 * cand
                if force_sym:
                    blend = force_sym_norm(blend)
                else:
                    blend = normalize_to_sum1(blend)
                if blend is not None:
                    scb = local_score(blend)
                    if np.isfinite(scb) and scb < bestsc:
                        best = blend.copy()
                        bestsc = float(scb)
            stall = 0
            last_update = time.monotonic()

    return best, bestsc


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1500.0
    deadline = start_time + max(10.0, budget) - 10.0

    parent = None
    try:
        parent = globals().get('height_sequence_1', None)
    except Exception:
        parent = None

    parent_arr = None
    if parent is not None:
        try:
            parent_arr = np.asarray(parent, dtype=np.float64)
            parent_arr = np.clip(parent_arr, 0.0, 1000.0)
            if parent_arr.size == 0 or not np.isfinite(parent_arr).all():
                parent_arr = None
        except Exception:
            parent_arr = None

    best_seq = None
    best_score = float('inf')

    if parent_arr is not None:
        p = normalize_to_sum1(parent_arr)
        if p is not None:
            best_seq = p.copy()
            best_score = local_score(p)
            print(f"baseline parent length={p.size} score={best_score:.6f}", flush=True)

    # Length plan.  Shorter lengths are faster to search; longer lengths may improve.
    if budget >= 1200.0:
        plan = (
            (128, 20.0, False),
            (256, 30.0, False),
            (512, 60.0, False),
            (1024, 320.0, False),
            (2048, 720.0, True),
            (4096, 260.0, True),
        )
    elif budget >= 600.0:
        plan = (
            (256, 30.0, False),
            (512, 100.0, False),
            (1024, 320.0, False),
            (2048, 450.0, True),
        )
    else:
        plan = (
            (128, 20.0, False),
            (256, 50.0, False),
            (512, 160.0, False),
            (1024, 250.0, False),
        )

    seedcur = int(seed)
    for n, alloc, sym in plan:
        rem = deadline - time.monotonic()
        if rem < 20.0:
            break
        b = min(float(alloc), rem - 10.0)
        if b <= 5.0:
            b = max(5.0, 0.5 * rem)

        try:
            cand, sc = optimize_length(n, b, seedcur, bool(sym), parent_arr)
            if cand is not None and np.isfinite(sc) and sc < best_score:
                best_seq = cand.copy()
                best_score = float(sc)
                print(f"improved length={n} sym={sym} score={best_score:.6f}", flush=True)
            else:
                print(f"length={n} sym={sym} best_here={sc:.6f}", flush=True)
        except Exception as exc:
            print(f"length={n} sym={sym} failed: {exc}", flush=True)

        seedcur += 101237

    # Final polishing on the globally best sequence.
    rem = deadline - time.monotonic()
    if best_seq is None and rem > 5.0:
        best_seq = make_candidate(256, seedcur, 0, parent_arr, False)
        best_score = local_score(best_seq)

    if best_seq is not None:
        if rem > 15.0:
            try:
                c1, s1 = multiplicative_optimize(best_seq, min(0.45 * rem, 120.0), seedcur + 991, False)
                if c1 is not None and s1 < best_score:
                    best_seq = c1.copy()
                    best_score = float(s1)
                    print(f"final multiplicative score={best_score:.6f}", flush=True)
            except Exception as exc:
                print(f"final multiplicative failed: {exc}", flush=True)

        rem2 = deadline - time.monotonic()
        if rem2 > 10.0:
            try:
                c2, s2 = block_local_search(best_seq, min(0.6 * rem2, 120.0), seedcur + 997)
                if c2 is not None and s2 < best_score:
                    best_seq = c2.copy()
                    best_score = float(s2)
                    print(f"final block search score={best_score:.6f}", flush=True)
            except Exception as exc:
                print(f"final block search failed: {exc}", flush=True)

    # Final validation against the authoritative evaluator when possible.
    if best_seq is None:
        if parent_arr is not None:
            p = normalize_to_sum1(parent_arr)
            if p is not None:
                best_seq = p.copy()
        if best_seq is None:
            best_seq = np.ones(64, dtype=np.float64) / 64.0

    final = normalize_to_sum1(best_seq)
    if final is None:
        final = np.ones(64, dtype=np.float64) / 64.0

    try:
        official = float(evaluate_sequence([float(x) for x in final]))
        if np.isfinite(official):
            best_score = official
        else:
            # Fallback to parent if possible.
            fp = None
            if parent_arr is not None:
                fp = normalize_to_sum1(parent_arr)
            if fp is not None:
                try:
                    official2 = float(evaluate_sequence([float(x) for x in fp]))
                    if np.isfinite(official2):
                        final = fp
                        best_score = official2
                except Exception:
                    pass
            if final is None:
                final = np.ones(64, dtype=np.float64) / 64.0
    except Exception:
        pass

    print(f"FINAL score_local={best_score:.6f} length={len(final)}", flush=True)
    return [float(x) for x in np.clip(np.asarray(final, dtype=np.float64), 0.0, 1000.0)]
```
