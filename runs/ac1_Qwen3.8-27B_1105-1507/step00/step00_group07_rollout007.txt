    # A[k,j] = f[k-j] if 0 <= k-j <= n-1, else 0

"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses alternating LP method to find sequences with flat self-convolution.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix(f):
    n = len(f)
    j_vals = np.repeat(np.arange(n), n)
    i_vals = np.tile(np.arange(n), n)
    row_vals = j_vals + i_vals
    col_vals = j_vals
    val_vals = f[i_vals]
    A = csr_matrix((val_vals, (row_vals, col_vals)), shape=(2*n-1, n))
    return A


def lp_solve(f):
    n = len(f)
    conv_ff = np.convolve(f, f)
    M = float(conv_ff.max())
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None


def lp_iteration(f, step=1.0):
    g = lp_solve(f)
    if g is None:
        return f.copy()
    sum_f = float(f.sum())
    sum_g = float(g.sum())
    if sum_g < 1e-12:
        return f.copy()
    g = g * sum_f / sum_g
    f_new = (1.0 - step) * f + step * g
    return np.maximum(f_new, 0.0)

    return float(evaluate_sequence(seq.tolist() if hasattr(seq, 'tolist') else list(seq)))

    f = init_fn(n, rng)
    f = np.maximum(f, 0.0)
    if f.sum() < 1e-6:
        f = np.ones(n)
    best_f = f.copy()
    best_score = objective(f)

        f = np.maximum(f, 0.0)

        score = objective(f)
        if score < best_score:
            best_f = f.copy()
            best_score = score

                best_score = parent_score
                best_seq = parent.copy()
                print(f"Parent: score={parent_score:.6f}")

                f, score = local_search(parent, n_iter=1000, seed=456)
                if score < best_score:
                    best_score = score
                    best_seq = f.copy()
                    print(f"Parent after local search: score={score:.6f}")
    except Exception as e:
        print(f"Parent sequence error: {e}")
    
    # Final: ensure we return a valid sequence
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = objective(best_seq)
    
    print(f"Final score: {best_score:.6f}, length={len(best_seq)}, time={time.monotonic()-start_time:.1f}s")
    return [float(x) for x in best_seq]
