def solve_lp_cross(a):
    """Given sequence a (sum=1, non-neg), solve LP to find b minimizing max (a*b)[k]."""
    n = len(a)

    # c = [0, ..., 0, 1] (only minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # For each k = 0, ..., 2n-2: sum_j a[k-j]*b[j] - t <= 0

    # Equality: sum(b) = 1 -> A_eq @ x = b_eq
    # Inequality: (a*b)[k] - t <= 0 for each k
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # For inequality constraints, we need (2n-1) rows
    A_ub = np.zeros((2*n - 1, n + 1))
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A_ub[k, j] = a[k - j]
        A_ub[k, n] = -1.0  # coefficient of t
    b_ub = np.zeros(2*n - 1)

    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        b = result.x[:n]
        return np.maximum(b, 0)  # ensure non-negative
    return None

import numpy as np
import time
from scipy.optimize import linprog

def _solve_lp_for_b(a):
    """Solve LP: min t s.t. (a*b)[k] <= t, sum(b)=1, b>=0."""
    n = len(a)
    # Variables: [b_0, ..., b_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    # Inequalities: (a*b)[k] - t <= 0
    A_ub = np.zeros((2*n - 1, n + 1))
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A_ub[k, j] = a[k - j]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(2*n - 1)
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x[:n], 0)
    return None

def _subgradient_descent(a, max_iter=5000, lr_init=0.01):
    """Subgradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())

    for it in range(max_iter):
        conv = np.convolve(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - 1e-12)[0]
        
        # Gradient of max_k f_k at a
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    grad[i] += 2 * a[ki]
        
        # Projected subgradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        if a_new.sum() > 1e-10:
            a_new = a_new / a_new.sum()
        else:
            a_new = a.copy()
        
        s = evaluate_sequence(a_new.tolist())
        if np.isfinite(s) and s < best_score:
            best = a_new.copy()
            best_score = s
        
        a = a_new
        lr = lr_init / (1 + 0.001 * it)
    
    return best, best_score

_DEADLINE = time.monotonic() + 1490  # Leave 10s buffer

def _time_left():
    return _DEADLINE - time.monotonic()

def _build_lp_matrix(a):
    """Build the LP constraint matrix for the cross-correlation LP."""
    n = len(a)
    # A_ub has (2n-1) rows, n+1 columns
    A_ub = np.zeros((2*n - 1, n + 1))
    # For each k, the coefficient of b[j] is a[k-j] if valid
    # We can vectorize this
    for k in range(2*n - 1):

        j_max = min(n - 1, k)
        # b[j] for j in [j_min, j_max] has coefficient a[k-j]
        # a[k-j] for j in [j_min, j_max] = a[k-j_max : k-j_min+1] reversed
        A_ub[k, j_min:j_max+1] = a[k-j_max:k-j_min+1][::-1]
        A_ub[k, n] = -1.0

"""
LP-iteration based optimizer for minimum peak autocorrelation non-negative sequences.
Uses the cross-correlation LP to generate descent directions, with line search
and multiple restarts across different sequence lengths and initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar

def _solve_lp_cross(a):
    """Solve LP: min t s.t. (a*b)[k] <= t for all k, sum(b)=1, b>=0.
    Returns optimal b or None."""
    n = len(a)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    
    # Build A_ub efficiently
    A_ub = np.zeros((2*n - 1, n + 1))
    for k in range(2*n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        # For j in [j_lo, j_hi], coefficient is a[k-j]
        # k-j ranges from k-j_hi to k-j_lo
        A_ub[k, j_lo:j_hi+1] = a[k-j_hi:k-j_lo+1][::-1]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(2*n - 1)
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        b = np.clip(res.x[:n], 0, None)
        if b.sum() > 1e-12:
            return b / b.sum()
    return None


def _lp_iterate(a, max_rounds=200):
    """LP-based iterative optimization with line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    if a.sum() < 1e-12:
        a = np.ones(n) / n
    a = a / a.sum()
    
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    
    for r in range(max_rounds):
        if time.monotonic() > _DEADLINE - 10:
            break
        b = _solve_lp_cross(a)
        if b is None:
            break
        
        # Check convergence
        if np.max(np.abs(b - a)) < 1e-9:
            break
        
        # Line search: try decreasing step sizes
        improved = False
        for step in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            a_new = (1.0 - step) * a + step * b
            a_new = np.maximum(a_new, 0)
            s_sum = a_new.sum()
            if s_sum < 1e-12:
                continue
            a_new = a_new / s_sum
            score = evaluate_sequence(a_new.tolist())
            if np.isfinite(score) and score < best_score - 1e-12:
                best = a_new.copy()
                best_score = score
                a = a_new
                improved = True
                break
        if not improved:

    return best, best_score
