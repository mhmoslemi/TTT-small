def fft_convolve(p):
    """Compute linear convolution of p with itself using FFT."""
    n = len(p)
    m = 2 * n - 1
    # Next power of 2 >= m
    size = 1
    while size < m:
        size *= 2
    F = fft(p, size)
    conv = np.real(ifft(F * F, size))[:m]
    return conv

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

    n = len(p)
    # Compute gradient of the "excess" function
    # F(p) = sum_s max(0, conv[s] - t)^2
    # grad F / grad p[i] = sum_s 2*max(0, conv[s]-t) * ∂conv[s]/∂p[i]

def solve_lp(f, n):
    """Solve: minimize t s.t. conv(f,g)[k] <= t, sum(g)=1, g>=0."""
    # Build constraint matrix
    # A_conv[k, j] = f[k-j] for valid (k,j)
    
    # Using the structure: for column j, non-zeros are in rows j to j+n-1
    # with values f[0], f[1], ..., f[n-1]
    
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        for d in range(n):
            k = j + d
            if k < 2*n - 1:
                rows.append(k)
                cols.append(j)
                vals.append(f[d])
    
    A_conv = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Full A_ub: [A_conv | -I_{2n-1 x 1}]
    # Shape: (2n-1) x (n+1)
    A_ub = A_conv.copy()
    # Add the -t column: for each row k, the coefficient of t is -1
    # In sparse form, this is a column of -1's
    neg_ones = sp.csc_matrix(-np.ones((2*n-1, 1)))
    A_ub = sp.hstack([A_conv, neg_ones], format='csr')
    
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    A_eq[0, n] = 0.0  # t doesn't appear in the sum constraint
    b_eq = np.array([1.0])
    
    c = np.zeros(n+1)
    c[n] = 1.0  # Minimize t
    
    bounds = [(0, None)] * n + [(0, None)]  # g >= 0, t >= 0
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.status == 0:
        g = result.x[:n]
        t = result.x[n]
        return g, t
    else:
        return None, None
