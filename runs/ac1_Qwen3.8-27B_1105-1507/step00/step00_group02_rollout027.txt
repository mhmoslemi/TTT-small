```python
import numpy as np
import time

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_descent(seq, n_iterations=1000, initial_lr=0.1):
    """Run subgradient descent to minimize max autocorrelation."""
    seq = project_simplex(seq.copy())
    n = len(seq)
    best = seq.copy()
    best_obj = 2.0 * n * np.max(np.convolve(seq, seq))
    
    lr = initial_lr
    for it in range(n_iterations):
        b = np.convolve(seq, seq)
        max_b = np.max(b)
        obj = 2.0 * n * max_b
        if obj < best_obj:
            best_obj = obj
            best = seq.copy()
        
        # Subgradient (using top few for stability)
        k_star = np.argmax(b)
        grad = np.zeros(n)
        
        # Main subgradient from argmax
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            grad[indices] = 4.0 * n * seq[k_star - indices]
        
        # Step
        seq = seq - lr * grad
        seq = project_simplex(seq)
        
        # Decay learning rate
        lr *= 0.999
    
    return best, best_obj
```

```python
def compute_subgradient(seq, K=5):
    """Compute weighted subgradient using top-K autocorrelation peaks."""
    n = len(seq)
    b = np.convolve(seq, seq)
    
    # Find top K peaks
    top_k_indices = np.argsort(b)[-K:]
    top_k_values = b[top_k_indices]
    
    # Softmax weights
    weights = np.exp((top_k_values - top_k_values.max()) / max(1e-10, top_k_values.max() * 0.1))
    weights /= weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for i, k in enumerate(top_k_indices):
        w = weights[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            grad[indices] += w * 2.0 * seq[k - indices]
    
    return grad, b.max()
```

This is O(K*n) per iteration, which is fine.

```python
import numpy as np
import time

def project_prob_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Duchi et al. (2011)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def run_subgradient_descent(seq, n_iter=20000, initial_lr=None, max_time=600, seed=42):
    """
    Run projected subgradient descent on the probability simplex
    to minimize max autocorrelation.
    """
    rng = np.random.default_rng(seed)
    n = len(seq)
    seq = project_prob_simplex(seq.copy())
    
    if initial_lr is None:
        initial_lr = 1.0 / np.sqrt(n)
    
    best = seq.copy()
    best_val = np.max(np.convolve(best, best))
    lr = initial_lr
    
    start = time.monotonic()
    for it in range(n_iter):
        if time.monotonic() - start > max_time:
            break
        
        b = np.convolve(seq, seq)
        max_b = b.max()
        
        if max_b < best_val:
            best_val = max_b
            best = seq.copy()
        
        # Subgradient
        K = 3
        top_idx = np.argpartition(b, -K)[-K:]
        top_vals = b[top_idx]
        # Softmax weights
        w = np.exp((top_vals - top_vals.max()) / (top_vals.max() * 0.05 + 1e-10))
        w /= w.sum()
        
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += w[i] * 2.0 * seq[k - np.arange(lo, hi+1)]
        
        # Step
        seq = seq - lr * grad
        seq = project_prob_simplex(seq)
        
        # Decay learning rate
        lr *= 0.9995
        
        # Periodic random perturbation (annealing)
        if it % 500 == 0 and it > 0:
            noise_scale = 0.01 * (1.0 - it / n_iter)
            seq = project_prob_simplex(seq + rng.normal(0, noise_scale, n))
    
    return best, best_val
```

```python
def spectral_flatten(seq, n_iter=100):
    """Try to flatten the power spectrum of seq."""
    n = len(seq)
    seq = seq.copy()
    
    for it in range(n_iter):
        F = np.fft.rfft(seq)
        # Target: uniform magnitude for non-DC components
        target_mag = np.mean(np.abs(F[1:])) if n > 1 else 1.0
        # Modify spectrum
        for k in range(1, len(F)):
            if abs(F[k]) > 0:
                F[k] = F[k] / abs(F[k]) * target_mag
        # Keep DC component
        # Inverse FFT
        seq_new = np.fft.irfft(F, n=n)
        # Project to simplex
        seq = project_prob_simplex(seq_new)
    
    return seq
```

```python
import numpy as np
import time

def project_prob_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def sgd_step(seq, K_top=5):
    """One subgradient descent step. Returns (new_seq, max_b)."""
    n = len(seq)
    C = np.convolve(seq, seq)
    max_b = C.max()
    
    # Get top K peaks
    if K_top == 1:
        top_idx = np.array([np.argmax(C)])
        top_vals = C[top_idx]
    else:
        top_idx = np.argpartition(C, -K_top)[-K_top:]
        top_vals = C[top_idx]
    
    # Softmax weights (temperature relative to max)
    temp = max_b * 0.1 + 1e-10
    w = np.exp((top_vals - top_vals.max()) / temp)
    w /= w.sum()
    
    # Subgradient
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += w[i] * 2.0 * seq[k - np.arange(lo, hi+1)]
    
    return grad, max_b

def run_sgd(seq_init, n_steps, max_time, seed=42, lr_decay=0.999, noise_decay=0.999):
    """Run projected subgradient descent."""
    rng = np.random.default_rng(seed)
    n = len(seq_init)
    seq = project_prob_simplex(seq_init.copy())
    
    best = seq.copy()
    best_val = np.convolve(best, best).max()
    
    lr = 1.0 / np.sqrt(n)
    noise_scale = 0.1
    
    start = time.monotonic()
    for it in range(n_steps):
        if time.monotonic() - start > max_time:
            break
        
        grad, max_b = sgd_step(seq)
        
        if max_b < best_val:
            best_val = max_b
            best = seq.copy()
        
        seq = seq - lr * grad
        seq += rng.normal(0, noise_scale, n)
        seq = project_prob_simplex(seq)
        
        lr *= lr_decay
        noise_scale *= noise_decay
    
    return best, best_val
```

```python
def sgd_step_fft(seq):
    """Subgradient step using FFT for efficient cross-correlation."""
    n = len(seq)
    C = np.convolve(seq, seq)  # length 2n-1
    max_b = C.max()
    
    # Weight: use C itself as the weight (this gives a "smooth" subgradient)
    # Or use a power of C to emphasize peaks more
    # Let's use C^2 to emphasize the peaks
    weights = C ** 2
    weights /= weights.sum()
    
    # grad[j] = 2 * sum_k weights[k] * seq[k-j] (for valid k-j)
    # This is a cross-correlation, computable via FFT
    N = 2 * n  # FFT size (need at least 2n-1 + n - 1 = 3n-2 for linear, but let's try 2n first)
    # Actually for correctness we need N >= (2n-1) + n - 1 = 3n-2
    N = 1 << (3*n - 2).bit_length()  # next power of 2
    
    W = np.fft.rfft(weights, N)
    S = np.fft.rfft(seq, N)
    corr = np.fft.irfft(W * np.conj(S), N)
    
    # corr[j] = sum_k weights[k] * seq[(k-j) mod N]
    # For 0 <= j < n and 0 <= k < 2n-1, we need k-j in [0, n-1]
    # With N >= 3n-2, there's no circular aliasing for the range we care about
    # The relevant part of corr is corr[0:n] (or shifted, depending on convention)
    
    # Let me verify the convention:
    # irfft(rfft(w) * conj(rfft(s))) gives the cross-correlation
    # corr[j] = sum_k w[k] * s[(k-j) mod N]
    # So for j=0: corr[0] = sum_k w[k] * s[k] (circular)
    # For our linear cross-correlation, we need:
    # grad[j] = sum_{k: 0<=k<2n-1, 0<=k-j<n} w[k] * s[k-j]
    # = sum_{k=j}^{j+2n-2} w[k] * s[k-j] (with w outside [0,2n-2] being 0)
    # This is corr[j] in the circular sense, if N is large enough
    
    grad = 2.0 * corr[:n]
    
    return grad, max_b, C
```

```python
import time
```

```python
Minimizes 2*n*max(autocorrelation)/(sum)^2 using projected subgradient descent
on the probability simplex, with multiple restarts, adaptive step sizes,
spectral initialization, and simulated annealing.
"""


def project_prob_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_sgd_grad(seq, C, K_top=5):
    """Compute subgradient of max(C) using top-K peaks."""
    n = len(seq)
    K = min(K_top, 2 * n - 1)
    if K == 1:
        top_idx = np.array([np.argmax(C)])
    else:

    top_vals = C[top_idx]
    temp = max(top_vals.max() * 0.1, 1e-10)
    w = np.exp((top_vals - top_vals.max()) / temp)
    w /= w.sum()

    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w[i] * 2.0 * seq[k - np.arange(lo, hi + 1)]
    return grad

    C = np.convolve(seq, seq)

    noise_scale = noise_scale_init / np.sqrt(n)

    else:
        seq = np.ones(n) / n
    return project_prob_simplex(seq)


def random_init(n, seed):
    """Random initialization on the simplex."""
    rng = np.random.defaultrng(seed)
    seq = rng.exponential(1.0, n)
    return project_prob_simplex(seq)


def structured_init(n, seed):
    """Structured initialization using cosine modulation."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    f1 = rng.uniform(0.1, 0.4)
    f2 = rng.uniform(0.3, 0.7)
    a1 = rng.uniform(0.3, 0.8)
    a2 = rng.uniform(0.1, 0.5)
    seq = 1.0 + a1 * np.cos(2 * np.pi * f1 * x / n) + a2 * np.cos(2 * np.pi * f2 * x / n)
    seq = np.abs(seq)
    s = seq.sum()
    if s > 0:
        seq = seq / s
    else:
        seq = np.ones(n) / n
    return project_prob_simplex(seq)


def fine_tune_local(seq_init, max_time, seed):
    """Local refinement with small perturbations."""
    rng = np.random.default_rng(seed)
    n = len(seq_init)
    seq = project_prob_simplex(seq_init.astype(np.float64).copy())
    C = np.convolve(seq, seq)
    best_val = C.max()
    best = seq.copy()

    start = time.monotonic()
    perturb_scale = 0.02 / np.sqrt(n)

    while time.monotonic() - start < max_time:
        # Random block perturbation
        width = max(1, int(rng.integers(n // 50, n // 10)))
        start_idx = int(rng.integers(0, max(1, n - width)))
        coupon = rng.normal(1.0, perturb_scale, width)
        candidate = seq.copy()
        candidate[start_idx:start_idx + width] *= coupon
        candidate = project_prob_simplex(candidate)

        C = np.convolve(candidate, candidate)
        if C.max() < best_val:
            best_val = C.max()
            best = candidate.copy()
            seq = candidate.copy()
        elif rng.random() < 0.01:
            # Occasional larger jump
            jump_width = max(1, int(rng.integers(n // 50, n // 5)))
            jump_start = int(rng.integers(0, max(1, n - jump_width)))
            candidate = seq.copy()
            candidate[jump_start:jump_start + jump_width] *= rng.normal(1.0, 0.1, jump_width)
            candidate = project_prob_simplex(candidate)
            C2 = np.convolve(candidate, candidate)
            if C2.max() < best_val:
                best_val = C2.max()
                best = candidate.copy()
                seq = candidate.copy()

    return best, best_val


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for the best non-negative sequence minimizing
    2*n*max(convolve(seq,seq))/(sum(seq))^2.

    Uses projected subgradient descent on the probability simplex with
    multiple initializations, adaptive step sizes, and local refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 10.0  # safety margin

    # Get initial sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(1024, dtype=np.float64)

    n = len(parent)
    S = parent.sum()
    if S < 0.01:
        raise ValueError("Parent sequence sum too small")

    # Normalize to simplex
    seq_simplex = parent / S

    # Evaluate parent
    best_score = float(evaluate_sequence(parent.tolist()))
    best_seq = parent.copy()
    best_simplex_val = np.convolve(seq_simplex, seq_simplex).max()
    print(f"Initial: n={n}, score={best_score:.6f}, max_C={best_simplex_val:.8f}")

    # Track best on simplex
    global_best_simplex = seq_simplex.copy()
    global_best_val = best_simplex_val

    # Phase 1: Subgradient descent from parent (main workhorse)
    time_left = deadline - time.monotonic()
    t1 = max(60.0, 0.4 * time_left)
    print(f"Phase 1: SGD from parent, time={t1:.0f}s")
    best1, val1 = run_sgd_inner(seq_simplex, t1, seed=seed, lr_scale=1.0,
                                 noise_scale_init=0.05, K_top=5, use_anneal=True)
    if val1 < global_best_val:
        global_best_val = val1
        global_best_simplex = best1.copy()
        best_score = 2.0 * n * val1
        best_seq = best1 * S  # back to original scale (doesn't matter, obj is scale-invariant)
        # Actually just use the simplex version scaled to have reasonable values
        best_seq = best1 * S  # keep original scale
        print(f"  Phase 1 result: max_C={val1:.8f}, score~{2*n*val1:.6f}")

    # Update from evaluation
    eval_candidate = global_best_simplex * S
    real_score = float(evaluate_sequence(eval_candidate.tolist()))
    if np.isfinite(real_score) and real_score < best_score:
        best_score = real_score
        best_seq = eval_candidate.copy()
    elif np.isfinite(real_score):
        # Use simplex representation; scale doesn't matter for objective
        pass

    time_left = deadline - time.monotonic()
    if time_left < 60:
        print("Time running out, returning best so far")
        return [float(max(0, x)) for x in best_seq]

    # Phase 2: SGD from different initializations
    inits = []
    inits.append(("spectral", spectral_flat_init(n, seed + 1)))
    inits.append(("random", random_init(n, seed + 2)))
    inits.append(("structured", structured_init(n, seed + 3)))
    # Also try a "spiky" initialization
    rng2 = np.random.default_rng(seed + 4)
    spiky = np.zeros(n)
    spiky[rng2.choice(n, size=n // 4, replace=False)] = 1.0
    inits.append(("spiky", project_prob_simplex(spiky)))

    time_per_init = max(30.0, time_left * 0.35 / len(inits))
    for idx, (name, init_seq) in enumerate(inits):
        if time.monotonic() > deadline:
            break
        remaining_for_this = min(time_per_init, deadline - time.monotonic())
        if remaining_for_this < 10:
            break
        print(f"Phase 2 ({name}): time={remaining_for_this:.0f}s")
        best_i, val_i = run_sgd_inner(init_seq, remaining_for_this, seed=seed + 100 + idx,
                                        lr_scale=1.0, noise_scale_init=0.03,
                                        K_top=5, use_anneal=True)
        if val_i < global_best_val:
            global_best_val = val_i
            global_best_simplex = best_i.copy()
            best_score = 2.0 * n * val_i
            best_seq = best_i * S
            print(f"  New best: max_C={val_i:.8f}, score~{2*n*val_i:.6f}")

    time_left = deadline - time.monotonic()
    if time_left < 30:
        # Final evaluation
        final = global_best_simplex * S
        fs = float(evaluate_sequence(final.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
            best_seq = final.copy()
        return [float(max(0, x)) for x in best_seq]

    # Phase 3: Local refinement
    print(f"Phase 3: Local refinement, time={time_left:.0f}s")
    t3 = max(30.0, 0.5 * time_left)
    best3, val3 = fine_tune_local(global_best_simplex, t3, seed=seed + 1000)
    if val3 < global_best_val:
        global_best_val = val3
        global_best_simplex = best3.copy()
        best_score = 2.0 * n * val3
        best_seq = best3 * S
        print(f"  Phase 3 result: max_C={val3:.8f}, score~{2*n*val3:.6f}")

    # Phase 4: Final SGD run with smaller steps
    time_left = deadline - time.monotonic()
    if time_left > 20:
        t4 = time_left - 10
        print(f"Phase 4: Final SGD, time={t4:.0f}s")
        best4, val4 = run_sgd_inner(global_best_simplex, t4, seed=seed + 2000,
                                      lr_scale=0.5, noise_scale_init=0.01,
                                      K_top=10, use_anneal=True)
        if val4 < global_best_val:
            global_best_val = val4
            global_best_simplex = best4.copy()
            best_score = 2.0 * n * val4
            best_seq = best4 * S
            print(f"  Phase 4 result: max_C={val4:.8f}, score~{2*n*val4:.6f}")

    # Final evaluation with the actual evaluate_sequence
    final_candidate = global_best_simplex * S
    # Make sure all values are non-negative and finite
    final_candidate = np.clip(final_candidate, 0.0, 1000.0)
    final_score = float(evaluate_sequence(final_candidate.tolist()))

    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
        best_seq = final_candidate.copy()

    # Also try the original parent as fallback
    parent_score = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(parent_score) and parent_score < best_score:
        best_score = parent_score
        best_seq = parent.copy()

    print(f"Final: score={best_score:.6f}, n={len(best_seq)}")
    return [float(max(0.0, x)) for x in best_seq]
```
