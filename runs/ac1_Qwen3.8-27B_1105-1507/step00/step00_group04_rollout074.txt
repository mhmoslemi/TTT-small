Minimizes 2n*max(convolve(a,a))/(sum(a))^2 using LP-based iterative optimization.
For each iteration, solves an LP to find a direction that reduces the autocorrelation peak.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

BEST_SEQ = None
BEST_SCORE = float('inf')

def build_A(f, n):
    """Build sparse constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    rows_list = []
    cols_list = []
    vals_list = []
    for j in range(n):
        L = min(n, 2*n-1-j)
        rows_list.append(np.arange(j, j+L, dtype=np.int32))
        cols_list.append(np.full(L, j, dtype=np.int32))
        vals_list.append(f[:L])
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_step(f, n, time_limit=30):
    """One LP iteration: find g maximizing sum(g) s.t. f*g <= M, g >= 0."""
    conv = np.convolve(f, f)
    M = np.max(conv)
    
    A = build_A(f, n)
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': time_limit, 'dual_feasibility_tolerance': 1e-8})
        if result.status == 0:
            return result.x, M
    except Exception as e:
        pass
    return None, M

        if time.monotonic() > deadline:
            break
        
        t = t_start + (t_end - t_start) * it / max(max_iters - 1, 1)
        
        g, M = lp_step(f, n, time_limit=20)

            # Try smaller time limit or skip
            if it > 5:
                break
            continue
        
        sum_g = np.sum(g)
        if sum_g < 1e-12:

        sum_f = np.sum(f)
        g = g * sum_f / sum_g
        
        # Try the update
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, 1000)

Minimizes 2n*max(convolve(a,a))/(sum(a))^2 using LP-based iterative optimization
and smooth refinement. The LP finds directions that reduce autocorrelation peaks,
and a smooth flatness objective polishes the result.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize

_best_seq = None
_best_score = float('inf')


def build_A(f, n):
    """Build sparse constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    rows_list = []
    cols_list = []
    vals_list = []
    for j in range(n):
        L = min(n, 2 * n - 1 - j)
        rows_list.append(np.arange(j, j + L, dtype=np.int32))
        cols_list.append(np.full(L, j, dtype=np.int32))
        vals_list.append(f[:L])
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_step(f, n, time_limit=30):
    """One LP iteration: find g maximizing sum(g) s.t. f*g <= M, g >= 0."""
    conv = np.convolve(f, f)
    M = np.max(conv)
    A = build_A(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': time_limit})
        if result.status == 0:
            return result.x, M
    except Exception:
        pass
    return None, M

            f_new2 = (1 - t2) * f + t2 * g
            f_new2 = np.clip(f_new2, 0, 1000.0)
            score_new2 = evaluate_sequence(f_new2.tolist())
            if np.isfinite(score_new2) and score_new2 < score:
                f = f_new2
                score = score_new2
                update_best(f, f"{tag}it{it}small")
            else:
                # Perturb to escape local minimum
                f = f + 0.03 * rng.standard_normal(n)
                f = np.clip(f, 0, 1000.0)
    return f, score


def flatness_obj_grad(a):
    """Objective: sum of squared deviations of autocorrelation from mean, plus gradient."""
    n = len(a)
    s = np.sum(a)
    conv = np.convolve(a, a)
    target = s * s / (2 * n - 1)
    diff = conv - target
    obj = np.sum(diff * diff)
    # Gradient: 4 * cross-corr(diff, a) at valid lags
    corr = np.correlate(diff, a, mode='valid')
    grad = 4.0 * corr
    return obj, grad
