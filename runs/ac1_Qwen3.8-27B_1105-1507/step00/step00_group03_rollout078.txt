```python
"""
Step-function score minimizer.

The score is score(a) = 2*n*max_k C_k / (sum a)^2 with C = convolve(a, a)
("full" autocorrelation).  It is scale invariant, so the search is really
about shaping a non-negative vector so that its full autocorrelation as flat
as possible (the uniform vector has score exactly 2.0).

We combine several proven low-autocorrelation operators:
  * an LP operator (maximise sum(b) s.t. (a*b)_k <= M, b >= 0) - the standard
    max-autocorrelation reduction operator,
  * a multiplicative "peak suppression" flow that subtracts the responsibility
    of the hottest lags,
  * normalised ping-pong (a / (1 + |a*a|^p)),
  * diverse random / simulated-annealing perturbations to escape basins.

All candidates are scored with an FFT-based full autocorrelation (identical to
np.convolve used by the reference evaluator) so the inner loop is fast and we
are limited only by the LP wall-clock time.  A few different lengths and a few
different initialisations are tried and the best valid sequence is returned.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


# ----------------------------------------------------------------------------
# Fast primitives (match the reference evaluator exactly for valid inputs)
# ----------------------------------------------------------------------------
def fast_full_convolve(a):
    """Full linear autocorrelation of a, equal to np.convolve(a, a)."""
    n = len(a)
    m = 2 * n - 1
    size = 1
    while size < m:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:m]


def fast_score(a):
    """2*n*max(c)/(sum a)^2 with c = convolve(a,a); inf when sum < 0.01."""
    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")
    c = fast_full_convolve(a)
    mx = float(np.max(c))
    return float(2.0 * len(a) * mx / (s * s))


def normalise(a, target_sum):
    """Clip to [0, 1000] and rescale to the requested (positive) sum."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return None
    return a * (target_sum / s)


def build_lp_index(n):
    """Sparse (row, col) structure of A with A[k, j] = a[k-j] (in range)."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        rows.extend([k] * (hi - lo + 1))
        cols.extend(range(lo, hi + 1))
    return np.asarray(rows, dtype=np.int64), np.asarray(cols, dtype=np.int64)


# ----------------------------------------------------------------------------
# Operators (each takes a normalised non-negative vector and returns a new one)
# ----------------------------------------------------------------------------
def op_lp(a, rows, cols, n, time_limit=25.0):
    """LP operator: b = argmax sum(b) s.t. (a*b)_k <= M, b >= 0."""
    c = fast_full_convolve(a)
    M = float(np.max(c))
    data = a[rows - cols].astype(np.float64)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                  options={"presolve": True, "time_limit": time_limit})
    if res is not None and str(res.status).strip() == "0" and res.x is not None:
        return np.clip(res.x, 0.0, 1000.0)
    return None


def op_suppress(a, topk=6, eta=0.6):
    """Subtract responsibility of the hottest lags, keep non-neg, preserve sum."""
    n = len(a)
    c = fast_full_convolve(a)
    ks = np.argsort(-c)[:topk]
    a2 = a.astype(np.float64).copy()
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        seg = a[lo:hi + 1] * a[k - hi:k - lo + 1][::-1]
        a2[lo:hi + 1] -= eta * seg
    return a2


def op_pingpong(a, p=1.0, alpha=1.0):
    """Normalised ping-pong: a / (1 + alpha*|a*a|^p), symmetrised for safety."""
    z = fast_full_convolve(a)
    # map the full autocorrelation back onto the n-length support in a symmetric
    # way so every coordinate is damped by a non-negative 'local conflict'.
    n = len(a)
    work = np.zeros(n, dtype=np.float64)
    for i in range(n):
        # lag that gives full-ish overlap contributing to coordinate i
        work[i] = float(c_gather(z, i, n))
    d = 1.0 + alpha * (np.abs(work) ** p)
    return a / d


def c_gather(z, i, n):
    """A cheap non-negative 'conflict' level at coordinate i from z = a*a."""
    # combine the central lag (full overlap) with the two edge lags that
    # heavily involve a[i]; all values of z are non-negative for non-neg a.
    centre = z[n - 1]
    e0 = z[min(i, 2 * n - 2)]
    e1 = z[min((2 * n - 2) - i, 2 * n - 2)]
    return max(centre * 0.25, e0, e1)


# ----------------------------------------------------------------------------
# Main search
# ----------------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Search for a non-negative sequence minimising the step-function score.

    Runs a time-bounded meta-heuristic (LP + peak-suppression + ping-pong +
    annealing over several starts and lengths) and returns the best valid
    sequence found as a list of floats.
    """
    rng = np.random.default_rng(int(seed))
    total = float(max(0.0, float(budget_s)))
    hard_deadline = time.monotonic() + total

    # ---- choose a small set of lengths to explore -------------------------
    try:
        n_parent = len(height_sequence_1)
    except Exception:
        n_parent = 2535
    lengths = sorted({int(max(256, n_parent)), int(n_parent),
                      int(0.8 * n_parent), int(1.2 * n_parent)})
    lengths = [max(256, x) for x in lengths]

    global_best = None
    global_score = float("inf")

    def consider(a):
        nonlocal global_best, global_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or not np.all(np.isfinite(a)) or a.sum() < 0.01:
            return
        sc = fast_score(a)
        if np.isfinite(sc) and sc < global_score:
            global_score = sc
            global_best = a.copy()

    def fresh_starts(n, rng):
        out = []
        out.append(np.ones(n, dtype=np.float64))
        out.append(np.clip(1.0 + rng.normal(0, 0.15, n), 0.05, None))
        out.append(np.clip(1.0 + 0.3 * rng.random(n) * rng.choice([-1, 1], n), 0.05, None))
        out.append(np.linspace(1.0, 0.4, n))                      # ramp down
        out.append(np.linspace(0.4, 1.0, n))                      # ramp up
        two = np.ones(n)
        half = n // 2
        two[:half] = 1.4
        two[half:] = 0.7
        out.append(two)
        out.append(1.0 + 0.2 * np.cos(2 * np.pi * np.arange(n) / n))
        out.append(np.clip(np.abs(rng.standard_t(3.0, n)) + 0.2, 0.0, None) /
                   max(1e-6, np.abs(rng.standard_t(3.0, n)).mean()))
        return out

    # Give each length a fair slice of the budget.
    per_len = (hard_deadline - time.monotonic()) / float(len(lengths))

    for n in lengths:
        if time.monotonic() >= hard_deadline:
            break
        len_deadline = min(hard_deadline, time.monotonic() + per_len)
        rows, cols = build_lp_index(n)

        st = time.monotonic()
        for a0 in fresh_starts(n, rng):
            a = normalise(a0, float(n))
            if a is None:
                continue
            consider(a)
            if time.monotonic() >= len_deadline:
                break

            stuck = 0
            it = 0
            best_local = a.copy()
            best_local_sc = fast_score(a)
            while time.monotonic() < len_deadline:
                it += 1
                roll = rng.random()
                if roll < 0.45:
                    b = op_lp(a, rows, cols, n, time_limit=20.0)
                    if b is None:
                        stuck += 1
                        continue
                    cand_list = []
                    for t in (0.5, 0.35, 0.2, 0.65):
                        cand_list.append((1 - t) * a + t * b)
                    # try also a near-pure b
                    cand_list.append(b)
                    for cand in cand_list:
                        cn = normalise(cand, float(n))
                        if cn is None:
                            continue
                        consider(cn)
                        sc = fast_score(cn)
                        if sc < best_local_sc:
                            best_local_sc = sc
                            best_local = cn
                            a = cn
                            stuck = 0
                            break
                elif roll < 0.70:
                    # peak suppression (a few sub-steps)
                    cur = a
                    for _ in range(3):
                        cur = op_suppress(cur, topk=int(rng.integers(4, 10)),
                                          eta=float(rng.uniform(0.3, 0.8)))
                        cn = normalise(cur, float(n))
                        if cn is None:
                            break
                        consider(cn)
                        sc = fast_score(cn)
                        if sc < best_local_sc:
                            best_local_sc = sc
                            best_local = cn
                            a = cn
                            stuck = 0
                            break
                elif roll < 0.85:
                    # ping-pong
                    cur = op_pingpong(a, p=float(rng.uniform(0.5, 1.5)),
                                      alpha=float(rng.uniform(0.5, 2.0)))
                    cn = normalise(cur, float(n))
                    if cn is not None:
                        consider(cn)
                        sc = fast_score(cn)
                        if sc < best_local_sc:
                            best_local_sc = sc
                            best_local = cn
                            a = cn
                            stuck = 0
                else:
                    # annealing / random perturbation
                    mode = rng.random()
                    if mode < 0.4:
                        cur = a * (1.0 + rng.normal(0, 0.10, n))
                    else:
                        cur = a.copy()
                        w = int(rng.integers(max(2, n // 60), max(3, n // 6)))
                        s0 = int(rng.integers(0, max(1, n - w)))
                        cur[s0:s0 + w] *= float(np.exp(rng.normal(0, 0.30)))
                    # accept with annealing
                    cn = normalise(cur, float(n))
                    if cn is not None:
                        sc = fast_score(cn)
                        consider(cn)
                        improve = (sc < best_local_sc)
                        if not improve and rng.random() < np.exp(-max(0.0, sc - best_local_sc) / 0.05):
                            a = cn
                            continue
                        if improve:
                            best_local_sc = sc
                            best_local = cn
                            a = cn
                            stuck = 0
                        else:
                            stuck += 1
                if stuck > 60:
                    # restart basin from current best_local with a kick
                    a = normalise(best_local * (1.0 + rng.normal(0, 0.20, n)), float(n))
                    if a is None:
                        a = best_local.copy()
                    stuck = 0
                if it % 25 == 0:
                    print("  n=%d it=%d local=%.5f global=%.5f elapsed=%.0fs"
                          % (n, it, best_local_sc, global_score,
                             time.monotonic() - st))

    # ---- final polish on the global best ---------------------------------
    if global_best is not None:
        n = len(global_best)
        rows, cols = build_lp_index(n)
        a = global_best.copy()
        polish_deadline = hard_deadline
        for _ in range(4000):
            if time.monotonic() >= polish_deadline:
                break
            roll = rng.random()
            if roll < 0.5:
                b = op_lp(a, rows, cols, n, time_limit=15.0)
                if b is not None:
                    for t in (0.3, 0.2, 0.4):
                        cn = normalise((1 - t) * a + t * b, float(n))
                        if cn is not None:
                            consider(cn)
                            if fast_score(cn) < fast_score(a):
                                a = cn
                                break
            elif roll < 0.8:
                cn = normalise(op_suppress(a, topk=6, eta=0.4), float(n))
                if cn is not None:
                    consider(cn)
                    if fast_score(cn) < fast_score(a):
                        a = cn
            else:
                cn = normalise(a * (1.0 + rng.normal(0, 0.05, n)), float(n))
                if cn is not None:
                    consider(cn)
                    if fast_score(cn) < fast_score(a):
                        a = cn

    if global_best is None:
        # absolute fallback: uniform (score 2.0, always valid)
        fb = np.ones(max(256, int(n_parent)), dtype=np.float64)
        global_best = fb
        global_score = fast_score(fb)

    final = np.clip(global_best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.ones(256, dtype=np.float64)
    print("FINAL n=%d score=%.6f" % (len(final), fast_score(final)))
    return [float(x) for x in final]
```
