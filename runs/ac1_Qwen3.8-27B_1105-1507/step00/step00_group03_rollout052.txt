def compute_conv(a):
    """Compute full convolution using FFT for efficiency."""
    n = len(a)
    size = 2 * n - 1
    # Use FFT-based convolution
    fsize = 1
    while fsize < size:
        fsize *= 2
    A = np.fft.rfft(a, fsize)
    C = np.fft.irfft(A * A, fsize)[:size]
    return C

def compute_objective(a):
    """Compute the objective function value."""
    n = len(a)
    c = compute_conv(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2 * n * np.max(c) / (s * s)
