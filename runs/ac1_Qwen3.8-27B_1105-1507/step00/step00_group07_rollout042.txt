```python
import numpy as np
from scipy.optimize import linprog
import time

def build_convolution_matrix(f):
    """Build matrix A where A @ b = f * b (full convolution)."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    return A

def lp_step(f, M):
    """Solve one LP step."""
    n = len(f)
    A = build_convolution_matrix(f)
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

def optimize_sequence(f, max_iters=100, t=0.1):
    """Iterate the LP-based optimization."""
    n = len(f)
    f = f.astype(np.float64).copy()
    for it in range(max_iters):
        conv = np.convolve(f, f)
        M = conv.max()
        S = f.sum()
        score = 2 * n * M / S**2
        
        b = lp_step(f, M)
        if b is None:
            break
        
        # Rescale b to have same sum as f
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        b = b * (S / b_sum)
        
        # Update
        f_new = (1 - t) * f + t * b
        f_new = np.clip(f_new, 0, 1000)
        
        new_conv = np.convolve(f_new, f_new)
        new_M = new_conv.max()
        new_S = f_new.sum()
        new_score = 2 * n * new_M / new_S**2 if new_S > 0.01 else np.inf
        
        if new_score < score:
            f = f_new
        else:
            # Try smaller step
            t = t * 0.5
            if t < 1e-6:
                break
    
    return f
```

The LP is:

     b >= 0

The optimal solution b* will have sum(b*) = B_max. If B_max > M (which it should, since b = 0 is feasible with sum = 0, and we're maximizing), then the rescaled g = b*/B_max has sum = 1.

The key question is: does the update f_new = (1-t)f + tg reduce the score?

The score of f_new is 2n * max_k((f_new * f_new)_k) (assuming sum = 1).

(f_new * f_new)_k = (1-t)^2 (f*f)_k + 2t(1-t) (f*g)_k + t^2 (g*g)_k

We know (f*f)_k <= M and (f*g)_k <= M (by the LP constraint). But we don't have a bound on (g*g)_k.

If (g*g)_k is also <= M for all k, then (f_new * f_new)_k <= M for all k, and the max doesn't increase. But the sum is still 1, so the score doesn't increase.

However, (g*g)_k might exceed M. In that case, the max might increase.

The LP only constrains (f*g)_k, not (g*g)_k. So the LP direction is a "safe" direction in the sense that the cross-term is bounded, but the quadratic term might blow up.

This is why the step size t needs to be small. For small t, the t^2 term is negligible, and the update is approximately:
(f_new * f_new)_k ≈ M + 2t(1-t)((f*g)_k - M)

Since (f*g)_k <= M, this is <= M. So for small enough t, the max doesn't increase.

But we also need the max to decrease (not just stay the same). The decrease comes from the fact that (f*g)_{k*} < M at the critical lag k* (where (f*f)_{k*} = M), which pulls the value down.

```python
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A
```
