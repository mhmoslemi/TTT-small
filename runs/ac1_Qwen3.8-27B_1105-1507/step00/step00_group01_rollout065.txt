Minimizes 2*n*max(conv(a,a))/sum(a)^2 using gradient-based optimization
with a log-sum-exp smooth approximation of max, a continuation method
(annealing alpha from small to large), multiple random restarts,
peak-directed local search, and random block perturbations.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize
from scipy.fft import rfft, irfft

# Module-level globals for smooth objective (avoids closures/lambdas)
_g_L = 0
_g_n = 0
_g_alpha = 0.0


def _get_p(x):
    """Compute softmax(x)."""
    xc = x - x.max()
    e = np.exp(xc)
    return e / e.sum()


def _smooth_obj(x):
    """Smooth max of conv(p,p) via log-sum-exp, where p=softmax(x)."""
    p = _get_p(x)
    P = rfft(p, _g_L)
    s = irfft(P * P, _g_L).real[:2 * _g_n - 1]
    sm = s.max()
    ev = np.exp(_g_alpha * (s - sm))
    return (_g_alpha * sm + np.log(ev.sum())) / _g_alpha


def _smooth_grad(x):
    """Gradient of smooth max objective w.r.t. x. Returns same-shape vector."""
    p = _get_p(x)
    P = rfft(p, _g_L)
    s = irfft(P * P, _g_L).real[:2 * _g_n - 1]
    sm = s.max()
    ev = np.exp(_g_alpha * (s - sm))
    w = ev / ev.sum()
    W = rfft(w, _g_L)
    c = irfft(W * np.conj(P), _g_L).real[:_g_n]
    S = np.dot(w, s)
    return 2.0 * p * (c - S)

def _run_lbfgs(x0, alpha, L, n, maxiter):
    """Run L-BFGS optimization with given smoothness parameters."""
    global _g_L, _g_n, _g_alpha

def _random_block_perturb(base, rng):
    """Apply random multiplicative perturbations to a few blocks."""
    n = len(base)
    candidate = base.copy()
    n_perturb = rng.integers(2, 8)
    for _ in range(n_perturb):
        width = rng.integers(max(2, n // 100), max(3, n // 10))
        start = int(rng.integers(0, max(1, n - width)))
        factor = float(np.exp(rng.normal(0.0, 0.15)))
        candidate[start:start + width] *= factor
    candidate = np.clip(candidate, 0.0, 1000.0)
    if candidate.sum() < 1e-10:
        return base.copy()
    return candidate

            maxiter = min(100, int(remaining))
        if maxiter < 5:
            break
        x = _run_lbfgs(x, alpha, L, n, maxiter)
        # Re-center to prevent numerical drift
        x = x - x.mean()
    return x
