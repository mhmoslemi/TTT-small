This is just adding a scaled version of a (shifted) to c. I can precompute the "influence vector" for each i.

Let me define: for coordinate i, the influence on c is:

So infl_i is just a shifted and possibly modified version of a. I can precompute this for all i.

S_new = S - a[i] + x
f(x) = 2n * max(c_new) / S_new^2

The max(c_new) = max(c + (x - a[i]) * infl_i)

For x > a[i]: c_new[k] = c[k] + (x-a[i]) * infl_i[k] >= c[k] for all k (since infl_i[k] >= 0)
So max(c_new) = max_k (c[k] + (x-a[i]) * infl_i[k])
This is achieved at the k that maximizes c[k] + (x-a[i]) * infl_i[k], which depends on x.

For x < a[i]: c_new[k] = c[k] - (a[i]-x) * infl_i[k] <= c[k] for all k
So max(c_new) = max_k (c[k] - (a[i]-x) * infl_i[k])
This is achieved at the k that maximizes c[k] - (a[i]-x) * infl_i[k].

In both cases, the active k changes as x varies. The function f(x) is piecewise smooth with breakpoints where the argmax changes.

```python
import numpy as np
import time
import cvxpy as cp

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def lp_direction(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
    except Exception:
        try:
            prob.solve(verbose=False)
        except Exception:
            return None
    if b.value is None:
        return None
    return np.maximum(b.value, 0.0)

def compute_score(a):
    c = np.convolve(a, a)
    M = np.max(c)
    S = np.sum(a)
    if S < 0.01:
        return float('inf')
    n = len(a)
    return 2.0 * n * M / (S * S)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    deadline = time.monotonic() + budget_s - 10.0
    rng = np.random.default_rng(seed)
    
    best_score = float('inf')
    best_seq = None
    
    def try_sequence(a):
        nonlocal best_score, best_seq
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  New best: n={len(a)}, score={score:.6f}, time={time.monotonic():.1f}")
    
    # Try different lengths and initializations
    lengths = [256, 512, 1024]
    
    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        print(f"Starting n={n}")
        
        # Initializations
        inits = [
            np.ones(n),
            0.5 + 0.5 * np.cos(2*np.pi*np.arange(n)/n),
            rng.uniform(0.1, 2.0, n),
        ]
        
        for a0 in inits:
            if time.monotonic() > deadline - 30:
                break
            
            a = np.clip(a0.astype(np.float64), 0.01, 1000.0)
            score = compute_score(a)
            print(f"  Init: score={score:.6f}")
            try_sequence(a)
            
            # LP iterations
            for it in range(500):
                if time.monotonic() > deadline - 20:
                    break
                
                try_sequence(a)
                
                c = np.convolve(a, a)
                M = np.max(c)
                
                b = lp_direction(a, M)
                if b is None or np.sum(b) < 1e-10:
                    break
                
                sum_a = np.sum(a)
                sum_b = np.sum(b)
                
                # Try different step sizes
                improved = False
                for t in [0.5, 0.3, 0.1, 0.05]:
                    a_new = (1-t)*a + t*b*(sum_a/sum_b)
                    a_new = np.clip(a_new, 0, 1000)
                    if np.sum(a_new) < 0.01:
                        continue
                    new_score = compute_score(a_new)
                    if new_score < score:
                        a = a_new
                        score = new_score
                        improved = True
                        break
                
                if not improved:
                    # Try adding noise
                    noise = rng.normal(0, 0.05, n)
                    a_pert = np.clip(a + noise, 0.01, 1000)
                    pert_score = compute_score(a_pert)
                    if pert_score < score:
                        a = a_pert
                        score = pert_score
                    else:
                        break  # Local minimum, move on
        
        # Verify best
        if best_seq is not None:
            verify = evaluate_sequence(best_seq.tolist())
            if verify != best_score:
                print(f"  Warning: score mismatch {best_score} vs {verify}")
    
    if best_seq is None:
        best_seq = np.ones(256)
    
    final = evaluate_sequence(best_seq.tolist())
    print(f"Final: n={len(best_seq)}, score={final}")
    return best_seq.tolist()
```
