```python
import numpy as np
import time

def _convolve_max(a):
    """Compute max of autocorrelation and the argmax index."""
    c = np.convolve(a, a)
    return c.max(), np.argmax(c), c

def _gradient_c_k(a, k):
    """Compute gradient of c[k] = (a*a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo > j_hi:
        return grad
    indices = np.arange(j_lo, j_hi + 1)
    grad[indices] = 2.0 * a[k - indices]
    # Diagonal correction: where k = 2*j
    if k % 2 == 0:
        j_diag = k // 2
        if j_lo <= j_diag <= j_hi:
            grad[j_diag] = a[j_diag]
    return grad

def _mirror_descent_step(a, c, num_active=20, tau=None, eta=0.1):
    """One step of mirror descent with multiplicative update."""
    n = len(a)
    max_c = c.max()
    
    # Select active constraints
    top_k = np.argsort(c)[-num_active:]
    top_vals = c[top_k]
    
    if tau is None:
        tau = max(1e-6, (max_c - top_vals.min()) * 0.3)
    
    # Softmax weights
    weights = np.exp(-(max_c - top_vals) / tau)
    weights /= weights.sum()
    
    # Compute combined gradient (pressure)
    pressure = np.zeros(n)
    for w, k in zip(weights, top_k):
        pressure += w * _gradient_c_k(a, int(k))
    
    # Multiplicative update
    a_new = a * np.exp(-eta * pressure)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new, pressure

def _local_search_step(a, k_star, n, rng, num_trials=10):
    """Try to improve by moving mass from high-gradient to low-gradient coords."""
```

    n = len(a)
    max_c = c.max()
    w = np.exp((c - max_c) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        return np.zeros(n)
    w /= w_sum
    
    # Cross-correlation: (a ⊛ w)[j] = sum_i a[i] * w[i+j]
    n_out = n + len(w) - 1  # = n + 2n-2 = 3n-2
    size = 1 << (n_out - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    xcorr = np.fft.irfft(np.conj(fa) * fw, size)[:n]
    
    pressure = 2.0 * xcorr
    
    # Diagonal correction: where k = 2j, the gradient is a[j] not 2*a[j]
    # correction[j] = -a[j] * w[2j] for 2j < len(w) = 2n-1
    diag_indices = np.arange(n)
    w_indices = 2 * diag_indices  # = 0, 2, 4, ..., 2n-2
    valid = w_indices < len(w)  # always true since 2n-2 < 2n-1
    pressure[valid] -= a[diag_indices[valid]] * w[w_indices[valid]]
    
    return pressure

def _mirror_step_fft(a, c, tau, eta):
    """One mirror descent step using FFT-accelerated pressure."""
    pressure = _pressure_fft(a, c, tau)
    a_new = a * np.exp(-eta * pressure)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new, pressure

def _obj(a, n):
    """Objective value (assuming sum(a) = 1)."""
    return 2.0 * n * np.max(_autocorr(a))

    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.

    best_a = parent.copy()

    else:
