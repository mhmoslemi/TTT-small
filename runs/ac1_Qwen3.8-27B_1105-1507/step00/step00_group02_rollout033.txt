import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog

def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ g) = convolve(a, g) (full convolution)."""
    n = len(a)
    num_rows = 2 * n - 1
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Equivalently: for column j, A[k, j] = a[k-j] for k = j, j+1, ..., min(j+n-1, 2n-2)
    # For column j (0-indexed), rows j to j+n-1 (since j <= n-1, j+n-1 <= 2n-2)
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    data = np.repeat(a, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(num_rows, n))
    return A

def solve_lp(a, M):
    """Solve LP: maximize sum(g) s.t. g >= 0, convolve(a,g) <= M."""
    n = len(a)
    A = build_conv_matrix(a)
    b = np.full(2 * n - 1, M)

    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog
import sys

def build_conv_pattern(n):
    """Precompute sparsity pattern for convolution matrix."""
    # For matrix A of shape (2n-1, n): A[k,j] is non-zero iff 0 <= k-j < n
    # Column j: rows j to j+n-1
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    return rows, cols

def make_conv_mat(a, rows, cols):
    """Create sparse convolution matrix."""
    n = len(a)
    data = np.repeat(a, n)
    return sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def simplex_project(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_score_normalized(a):
    """Compute score for a (assumed sum=1, non-negative)."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * max(b)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    LP-based iterative optimization to minimize the peak of self-convolution.
    Uses alternating LP relaxations: fix a, solve LP for g, mix, repeat.
    Also tries a mirror descent / subgradient approach for comparison.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    
    best_a = None
    best_score = float('inf')
    
    # Try different dimension
    n = 2500
    print(f"Using n={n}")
    
    # Start with uniform
    a = np.ones(n) / n
    score = compute_score_normalized(a)
    best_score = score
    best_a = a.copy()
    print(f"Initial score: {score:.6f}")
    
    # Precompute pattern
    rows, cols = build_conv_pattern(n)
    
    t = 0.4  # mixing parameter
    patience = 0
    best_m = max(np.convolve(a, a))
    
    for iteration in range(10000):
        if time.monotonic() > deadline:
            break
        
        # Compute current M
        b = np.convolve(a, a)
        M = float(max(b))
        
        # Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0
        A = make_conv_mat(a, rows, cols)
        b_ub = np.full(2*n-1, M)
        c_obj = -np.ones(n)
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            print(f"LP failed at iteration {iteration}: {res.message}")
            break
        
        g = res.x
        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            print(f"g_sum too small at iteration {iteration}")
            break
        g = g / g_sum
        
        # Mix
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        # Check score
        score = compute_score_normalized(a_new)
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            if iteration % 20 == 0:
                print(f"Iter {iteration}: score = {best_score:.6f}, M = {max(np.convolve(a_new, a_new)):.8f}")
        
        # Adapt mixing parameter
        new_M = max(np.convolve(a_new, a_new))
        if new_M < best_M:
            best_M = new_M
            patience = 0
            t = min(0.5, t * 1.02)
        else:
            patience += 1
            if patience > 10:
                t = max(0.05, t * 0.7)
                patience = 0
        
        a = a_new
    
    print(f"Final best score: {best_score:.6f}")
    return [float(x) for x in best_a]

    return sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def compute_score(a):
    n = len(a)
    b = np.convolve(a, a)
    max_b = max(b)
    sum_a = np.sum(a)
    return 2.0 * n * max_b / (sum_a * sum_a)

    """
    LP-based iterative optimization for flat self-convolution.
    Starts from a cosine-tapered distribution and uses alternating
    LP relaxations to minimize the peak of the self-convolution.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    
    best_a = None
    best_score = float('inf')
    
    n = 2500
    print(f"Starting with n={n}")
    
    # Cosine-tapered initial distribution
    i = np.arange(n)
    a = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    a = np.clip(a, 0, None)
    a = a / a.sum()
    
    # Make sure it's not uniform (add small random perturbation)
    a = a * (1.0 + 0.1 * rng.standard_normal(n))
    a = np.clip(a, 0, None)
    a = a / a.sum()
    
    score = compute_score(a)
    best_score = score
    best_a = a.copy()
    print(f"Initial score: {score:.6f}")
    
    rows, cols = build_conv_pattern(n)
    t = 0.3
    
    for iteration in range(5000):
        if time.monotonic() > deadline:
            break
        
        t0 = time.monotonic()
        
        b = np.convolve(a, a)
        M = float(np.max(b))
        
        A = make_conv_mat(a, rows, cols)
        b_ub = np.full(2*n-1, M)
        c_obj = -np.ones(n)
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            break
        
        g = res.x
        g_sum = float(np.sum(g))
        if g_sum < 1e-8:
            # Increase M
            M *= 1.1
            continue
        g = g / g_sum
        
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        a_new = a_new / a_new.sum()
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            if iteration % 10 == 0:
                elapsed = time.monotonic() - t0
                print(f"Iter {iteration}: score={best_score:.6f}, M={M:.6f}, time/iter={elapsed:.2f}s")
        
        # Adapt t
        new_M = float(np.max(np.convolve(a_new, a_new)))
        if new_M < M * 0.99:
            t = min(0.5, t * 1.01)
        else:
            t = max(0.05, t * 0.9)
        
        a = a_new
    
    print(f"Final best score: {best_score:.6f}")
    return [float(x) for x in best_a]
