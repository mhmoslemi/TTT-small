    n = len(f)

LP-based iterative optimization for minimizing autocorrelation peak of step functions.
Uses active-set linear programming to find descent directions, combined with
adaptive step sizes and occasional random perturbations for diversity.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_convolution(f):
    """Compute the autocorrelation (linear convolution) of f with itself."""
    return np.convolve(f, f)


def solve_lp_active_set(f, M, max_constraints=3000, tol=1e-8, max_iter=50):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
    Using an active-set / cutting-plane approach.
    """
    n = len(f)
    
    # Initialize active set with the most violated constraints
    conv_ff = np.convolve(f, f)
    active_indices = list(np.argsort(-conv_ff)[:min(max_constraints, 2*n-1)])
    
    g = None
    for iteration in range(max_iter):
        # Build constraint matrix for active constraints
        k_array = np.array(active_indices)
        A = np.zeros((len(k_array), n))
        for i, k in enumerate(k_array):
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_max >= j_min:
                # A[i, j] = f[k-j] for j in [j_min, j_max]
                # k-j ranges from k-j_max to k-j_min
                A[i, j_min:j_max+1] = f[k-j_max:k-j_min+1][::-1]
        
        # Solve LP
        c = -np.ones(n)
        b_ub = np.full(len(k_array), M)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        except Exception:
            break
        
        if not res.success:
            break
        
        g = res.x.copy()
        
        # Check all constraints
        conv_fg = np.convolve(f, g)
        violations = np.where(conv_fg > M * (1 + tol) + tol)[0]
        
        if len(violations) == 0:
            return g
        
        # Add most violated constraints
        if len(violations) > 0:
            # Sort by violation severity
            violation_amounts = conv_fg[violations] - M
            worst = violations[np.argsort(-violation_amounts)][:min(200, len(violations))]
            for v in worst:
                active_indices.append(int(v))
            
            # Keep only the most relevant constraints
            if len(active_indices) > max_constraints * 2:
                # Re-select based on current violation
                all_indices = list(set(active_indices))
                conv_vals = [conv_fg[i] for i in all_indices]
                active_indices = [all_indices[i] for i in np.argsort(-conv_vals)[:max_constraints]]
    
    return g

"""
LP-based iterative optimization for minimizing autocorrelation peak of step functions.
Uses active-set linear programming to find descent directions that flatten the
autocorrelation profile, combined with adaptive step sizes, FFT-based convolution,
and local polishing. The objective 2*n*max(conv(f))/sum(f)^2 is scale-invariant,
so we normalize sum(f)=1 and minimize 2*n*max(conv(f)).
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_autocorr(f):
    """Compute linear autocorrelation using FFT (O(n log n))."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(f, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def solve_lp_direction(f, M, max_constraints=2500, max_iter=80, time_limit=15.0):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
    Using active-set / cutting-plane method.
    Returns optimal g (normalized to sum=1) or None on failure.
    """
    n = len(f)
    t_start = time.monotonic()

    # Initialize active set with most violated constraints (highest conv values)
    conv_ff = fast_autocorr(f)
    active_indices = list(np.argsort(-conv_ff)[:min(max_constraints, 2 * n - 1)])

    best_g = None
    best_obj = 0.0

    for iteration in range(max_iter):
        if time.monotonic() - t_start > time_limit:
            break

        k_array = np.array(active_indices, dtype=int)
        # Build constraint matrix: A[i, j] = f[k_i - j] if valid
        A = np.zeros((len(k_array), n))
        for i in range(len(k_array)):
            k = int(k_array[i])
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_max >= j_min:
                A[i, j_min:j_max + 1] = f[k - j_max:k - j_min + 1][::-1]

        c = -np.ones(n)
        b_ub = np.full(len(k_array), M)

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': max(1.0, time_limit - (time.monotonic() - t_start))})
        except Exception:
            break

        if not res.success:
            break

        g = res.x.copy()
        obj = g.sum()

        if obj > best_obj:
            best_obj = obj
            best_g = g.copy()

        # Check all constraints via FFT
        conv_fg = fast_cross_corr(f, g)
        max_violation = conv_fg.max() - M

        if max_violation < 1e-10:
            return g / g.sum() if g.sum() > 0 else None

        # Add most violated constraints
        violations = np.where(conv_fg > M + 1e-12)[0]
        if len(violations) > 0:
            severity = conv_fg[violations] - M
            worst = violations[np.argsort(-severity)][:min(300, len(violations))]
            for v in worst:
                active_indices.append(int(v))

            # Prune if too large
            if len(active_indices) > max_constraints * 3:
                # Keep the most violated among current active set
                conv_at_active = conv_fg[np.array(active_indices)]
                order = np.argsort(-conv_at_active)[:max_constraints]
                active_indices = [active_indices[i] for i in order]

    if best_g is not None and best_g.sum() > 0:
        return best_g / best_g.sum()
    return None


def fast_cross_corr(f, g):
    """Compute linear cross-correlation (f * g) using FFT."""
    n = len(f)
    m = len(g)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(f, size)
    fg = np.fft.rfft(g, size)
    return np.fft.irfft(fa * fg, size)[:n + m - 1]
