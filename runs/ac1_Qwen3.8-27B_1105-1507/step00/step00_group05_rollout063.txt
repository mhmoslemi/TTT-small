def proj_simplex_simple(v):
    """Simple projected gradient: clip to 0 and renormalize to sum 1."""
    v = np.clip(v, 0, None)
    s = v.sum()
    if s < 1e-10:
        return np.ones(len(v)) / len(v)
    return v / s

import numpy as np
import time

def compute_convolution_fft(a):
    """Compute linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for FFT
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)
    return C[:size]

def gradient_at_lag(a, k, n):
    """Gradient of c[k] with respect to a."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    indices = np.arange(lo, hi + 1)
    grad[indices] = 2 * a[k - indices]
    return grad

def logsumexp_smooth_max(c, T):
    """Compute smooth approximation of max using log-sum-exp."""
    c_max = c.max()
    return c_max + np.log(np.sum(np.exp(T * (c - c_max)))) / T

def smooth_gradient(a, c, T, n):
    """Gradient of the smooth max approximation."""
    # p_k = exp(T*c[k]) / sum_j exp(T*c[j])
    c_max = c.max()
    weights = np.exp(T * (c - c_max))
    weights /= weights.sum()
    # Gradient is weighted sum of individual gradients
    grad = np.zeros(n)
    for k in range(len(c)):
        if weights[k] > 1e-10:
            grad += weights[k] * gradient_at_lag(a, k, n)
    return grad
