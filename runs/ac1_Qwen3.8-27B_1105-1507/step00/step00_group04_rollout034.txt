```python
import numpy as np
import time

def projected_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    # Algorithm from Duchi et al. (2011)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_convolution(a, n_fft):
    """Compute full convolution of a with itself using FFT."""
    fa = np.fft.rfft(a, n=n_fft)
    result = np.fft.irfft(fa * fa)
    return result[:2*len(a)-1]

def compute_gradient(p, a, n_fft):
    """Compute 2 * cross_corr(p, a)."""
    n = len(a)
    b = a[::-1]
    N = n_fft
```

```python
def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if v.sum() <= 1e-30:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

                print(f"L={n} FFT={n_fft} init={init_name} starting...")

            a_opt, score = _optimize_sequence(init_a, n_fft, segment_deadline, rng, verbose=True)

            if np.isfinite(score) and score < local_best_score:
                local_best_score = score
                local_best_a = a_opt.copy()

            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"*** New global best: {score:.6f} (n={n}, init={init_name})")

        time_spent += (time.monotonic() - (deadline - alloc + 2.0))

    # If we found something, do a final polish on the best
    if best_seq is not None and time.monotonic() < deadline - 30:
        n = len(best_seq)
        n_fft = 1
        while n_fft < 3 * n - 2:
            n_fft *= 2
        polish_deadline = min(time.monotonic() + 60.0, deadline)
        a_polished, score = _optimize_sequence(best_seq, n_fft, polish_deadline, rng, verbose=True)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_polished
            print(f"*** Polished best: {score:.6f}")

    # Fallback
    if best_seq is None:
        n_fb = parent.size if parent is not None else 1024
        best_seq = np.ones(n_fb) / n_fb
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"Using uniform fallback: score={best_score:.6f}")

    # Final verification
    final = evaluate_sequence(best_seq.tolist())
    print(f"FINAL: score={final:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
```
