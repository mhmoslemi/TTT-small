def project_nonneg_fixed_sum(x, target_sum):
    """Project x onto {a >= 0, sum(a) = target_sum} using water-filling."""
    x = x.copy()
    n = len(x)
    if np.sum(x) <= target_sum:
        # Need to add to make sum = target_sum
        # Just add evenly to non-zero elements, or to all
        x = np.maximum(0, x)
        s = np.sum(x)
        if s < target_sum:
            x += (target_sum - s) / n
        return x
    # Need to subtract
    # Water-filling: find lambda such that sum(max(0, x_j - lambda)) = target_sum
    # Sort x in decreasing order
    sorted_x = np.sort(x)[::-1]
    # Try different lambda values
    cumsum = np.cumsum(sorted_x)
    # lambda_k = (cumsum[k] - target_sum) / (k+1) for k = 0, ..., n-1
    lambdas = (cumsum - target_sum) / np.arange(1, n+1)
    # Find the right lambda
    # We want the largest k such that sorted_x[k] >= lambda_k
    valid = sorted_x >= lambdas
    if not np.any(valid):
        lambda_val = cumsum[-1] / n  # uniform
    else:
        k = np.max(np.where(valid)[0])
        lambda_val = lambdas[k]
    return np.maximum(0, x - lambda_val)

    """Project x onto {v >= 0, sum(v) = target_sum}."""
    n = len(x)

def lp_direction_heuristic(a, b, top_frac=0.05):
    """Compute a descent direction that reduces the top convolution values."""
    n = len(a)
    threshold = np.percentile(b, 100 * (1 - top_frac))
    critical = np.where(b > threshold)[0]
    
    # For each j, compute total contribution to critical b_k's
    contrib = np.zeros(n)
    for k in critical:
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            k_minus_j = k - j
            if 0 <= k_minus_j < n:
                contrib[j] += a[k_minus_j]
    
    # Direction: reduce high-contrib positions, increase low-contrib positions
    # Normalize
    total = a.sum()
    target_reduction = 0.1 * total  # reduce 10% of total mass from high-contrib
    top_idx = np.argsort(contrib)[::-1][:n // 4]
    bottom_idx = np.argsort(contrib)[:n // 4]
    
    reduction = np.zeros(n)
    for j in top_idx:
        reduction[j] = contrib[j] / max(1, np.sum(contrib[top_idx]))
    reduction *= target_reduction
    
    increase = np.zeros(n)
    for j in bottom_idx:
        increase[j] = 1.0 / max(1, len(bottom_idx))
    increase *= target_reduction

"""
Multi-phase optimizer for minimum peak-self-convolution step function.
Combines spectral/Hanning/uniform initialization, projected gradient descent
with adaptive soft-peak threshold, simulated annealing with diverse moves,
and periodic restarts. Exploits scale invariance of the objective.

import numpy as np
import time
import math


def fft_self_convolve(a):
    """Compute a * a (full convolution) using FFT for speed."""
    n = len(a)
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    A = np.fft.rfft(a, size)
    result = np.fft.irfft(A * A)
    return result[:out_len]


def project_simplex(x, target_sum):
    """Project x onto {v >= 0, sum(v) = target_sum} via water-filling."""
    n = len(x)
    if target_sum <= 0:
        return np.zeros(n)
    x = np.asarray(x, dtype=np.float64)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (cssv - target_sum) / rho
    if np.any(cond):
        rho_hat = int(np.max(np.where(cond)[0])) + 1
        theta = (cssv[rho_hat - 1] - target_sum) / rho_hat
    else:
        theta = (np.sum(x) - target_sum) / n
    return np.maximum(x - theta, 0.0)


def compute_grad(a, threshold):
    """Gradient of sum_k max(0, b_k - T)^2 w.r.t. a, where b = a*a."""
    b = fft_self_convolve(a)
    excess = np.maximum(0.0, b - threshold)
    if np.sum(excess) < 1e-20:
        return np.zeros(len(a))
    n = len(a)
    g = np.correlate(excess, a, mode='valid')[:n]
    return 2.0 * g
