import numpy as np

def simplex_projection(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_candidates = idx * u - (cssv - 1)
    rho = np.max(idx[rho_candidates > 0])
    theta = (cssv[rho - 1] - 1) / rho
    return np.maximum(v - theta, 0)


def compute_smoothed_gradient(a, t):
    """Compute gradient of (1/t)*log(sum exp(t*c[k])) where c=conv(a,a)."""
    n = len(a)
    c = np.convolve(a, a)
    
    # Softmax weights with numerical stability
    c_max = c.max()
    log_w = t * (c - c_max)
    # Further shift for stability if needed
    log_w_max = log_w.max()
    if log_w_max > 500:
        log_w -= log_w_max
    w = np.exp(log_w)
    w_sum = w.sum()
    w /= w_sum
    
    # Gradient: grad[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]
    # Using FFT: g[i] = (w * a_flip)[n-1+i]
    a_flip = a[::-1]

    conv_result = np.fft.irfft(W * AF, N)

    exponents -= exponents.max()
    w = np.exp(exponents)
    w /= w.sum()
    
    # grad[i] = 2 * sum_m w[i+m] * a[m]
    # = 2 * (w * a_flipped)[n-1+i]
    a_flip = a[::-1]
    conv_len = 3 * n - 2
    N = 1 << (conv_len - 1).bit_length()
    
    W = np.fft.rfft(w, N)
    AF = np.fft.rfft(a_flip, N)
    result = np.fft.irfft(W * AF, N)[:conv_len]
    
    g = result[n-1 : 2*n-1]
    return 2.0 * g, c

Wait, I have a concern about the simplex projection. Let me double-check the formulation.

The probability simplex is {x : x >= 0, sum(x) = 1}.

The Euclidean projection onto the simplex: given v, find x = argmin ||x - v||^2 s.t. x >= 0, sum(x) = 1.

By KKT conditions: x_i = max(v_i - theta, 0) where theta is chosen so that sum(x) = 1.

To find theta: sort v in decreasing order: v_(1) >= v_(2) >= ... >= v_(n).
For a given rho, if the solution has exactly rho positive components, then:
theta = (sum_{i=1}^{rho} v_(i) - 1) / rho
and we need v_(rho) > theta >= v_(rho+1).

The standard algorithm (Duchi et al.) sorts once and finds rho in O(1) after the sort.

Let me verify my implementation:

def simplex_project(v):
    n = v.size
    u = -np.sort(-v)  # sort descending

    rho = rho_candidates[-1] + 1  # 1-indexed
    theta = (cssv[rho-1] - 1) / rho

"""
Gradient-based optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothed max, FFT-based gradient, Adam optimizer,
simplex projection, and multiple diverse restarts.
"""

import numpy as np
import time


def simplex_project(v):
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n <= 1:
        return np.ones(max(n,1)) / max(n,1)
    u = -np.sort(-v)
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > cssv - 1
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0]) + 1
    theta = (cssv[rho-1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def compute_grad(a, t):
    """
    Compute gradient of (1/t)*log(sum_k exp(t*c[k])) where c=conv(a,a).
    Returns (gradient, c, w) where w are softmax weights.
    """
    n = a.size
    c = np.convolve(a, a)
    
    # Stable softmax
    c_shift = c - c.max()
    exponents = t * c_shift
    exponents -= exponents.max()
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum <= 0 or not np.isfinite(w_sum):
        # Fallback: uniform weights
        w = np.ones_like(c) / c.size
    else:
        w /= w_sum
    
    # grad[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]
    # Computed via FFT: grad[i] = 2 * (w * a_flipped)[n-1+i]
    a_flip = a[::-1]
    conv_len = 3 * n - 2
    N = 1 << (conv_len - 1).bit_length()
    
    W = np.fft.rfft(w, N)
    AF = np.fft.rfft(a_flip, N)
    conv = np.fft.irfft(W * AF, N)[:conv_len]
    
    g = conv[n-1 : 2*n-1]
    grad = 2.0 * g
    return grad, c, w
