    print("Phase 1: Spectral compression")
    best, best_score = _run_spectral_phase(best, n_fft, p1_deadline, rng)
    print(f"After Phase 1: score={best_score:.6f}")
    
    # Phase 2: Subgradient descent (20% of budget)
    remaining = total_deadline - time.monotonic()
    if remaining > 30:
        p2_deadline = time.monotonic() + remaining * 0.35
        print("Phase 2: Subgradient descent")
        best, best_score = _subgrad_loop(best, p2_deadline, eta_init=0.02)
        print(f"After Phase 2: score={best_score:.6f}")
    
    # Phase 3: Spectral refinement (15% of budget)
    remaining = total_deadline - time.monotonic()
    if remaining > 30:
        p3_deadline = time.monotonic() + remaining * 0.35
        print("Phase 3: Spectral refinement")
        best, best_score = _run_spectral_phase(best, n_fft, p3_deadline, rng)
        print(f"After Phase 3: score={best_score:.6f}")
    
    # Phase 4: Subgradient fine-tuning (10%)
    remaining = total_deadline - time.monotonic()
    if remaining > 20:
        p4_deadline = time.monotonic() + remaining * 0.4
        print("Phase 4: Subgradient fine-tuning")
        best, best_score = _subgrad_loop(best, p4_deadline, eta_init=0.005)
        print(f"After Phase 4: score={best_score:.6f}")
    
    # Phase 5: Block perturbation (remaining time)
    remaining = total_deadline - time.monotonic()
    if remaining > 10:
        print("Phase 5: Block perturbation search")
        best, best_score = _block_search(best, total_deadline - 5, rng)
        print(f"After Phase 5: score={best_score:.6f}")
    
    print(f"Final: n={n}, score={best_score:.6f}")
    
    # Safety: verify and sanitize
    result = [float(x) for x in np.maximum(best, 0.0)]
    for i, x in enumerate(result):
        if not np.isfinite(x):
            result[i] = 0.0
        result[i] = min(1000.0, max(0.0, result[i]))
    
    # Verify with actual evaluator
    try:
        verified = float(evaluate_sequence(result))
        print(f"Verified score via evaluate_sequence: {verified:.6f}")
    except Exception as e:
        print(f"Verification error: {e}")
    
    return result
