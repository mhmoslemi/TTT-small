def make_seed(n, kind, eps, rng):
    i=np.arange(n,dtype=np.float64)
    if kind==0:
       a=np.ones(n)
    elif kind==1:
       a=1.0/np.sqrt(i+1.0)
    elif kind==2:
       a=1.0/np.sqrt(n - i)
    elif kind==3:

    elif kind==4:

```python
"""
Search for nonnegative step-function coefficients minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2.

The code uses several ideas:
  * FFT-based fast convolution and fast gradient of a smooth log-sum-exp
    approximation of the maximum convolution coefficient.
  * Multiplicative gradient steps with adaptive step size and restarts.
  * Spectral Newton / flattened-square-root moves toward lower target peaks.
  * Multiple good seeds, especially one-sided (i+1)^(-1/2)-type sequences.
"""

import numpy as np
import time

PLAN_CACHE = {}


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def get_plan(n):
    if n in PLAN_CACHE:
        return PLAN_CACHE[n]
    Nc = next_pow2(2 * n - 1)
    Ng = next_pow2(3 * n - 2)
    idx = make_fft_offset_indices(Ng, n)
    plan = (Nc, Ng, idx)
    PLAN_CACHE[n] = plan
    return plan


def make_fft_offset_indices(Ng, n):
    if n <= 0:
        return np.empty(0, dtype=int)
    if n == 1:
        return np.array([0], dtype=int)
    # We need cross-correlation values h[j] = sum_k p[k] a[k-j]
    # from ifft(conj(FFT(p)) * FFT(a)) at indices 0, -1, -2, ..., -(n-1).
    neg = np.arange(Ng - 1, Ng - n, -1, dtype=int)
    idx = np.empty(n, dtype=int)
    idx[0] = 0
    idx[1:] = neg
    return idx


def normalize_seq(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return None
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a, dtype=np.float64))
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return a / s


def convolve_fft(a, N):
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    L = 2 * a.size - 1
    if c.size < L:
        c = np.pad(c, (0, L - c.size), mode="constant")
    else:
        c = c[:L]
    return np.maximum(c, 0.0)


def score_candidate(a, plan):
    Nc = plan[0]
    c = convolve_fft(a, Nc)
    return 2.0 * a.size * float(np.max(c))


def smooth_value(c, beta):
    bc = beta * c
    m = float(np.max(bc))
    e = np.exp(bc - m)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return m / beta
    return m / beta + float(np.log(s)) / beta


def compute_gradient(a, Nc, Ng, idx, beta_factor):
    c = convolve_fft(a, Nc)
    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax):
        return None

    n = a.size
    score = 2.0 * n * cmax
    beta = beta_factor / cmax

    bc = beta * c
    m = beta * cmax
    e = np.exp(bc - m)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return None

    p = e / s
    f = smooth_value(c, beta)

    fp = np.fft.rfft(p, Ng)
    fa = np.fft.rfft(a, Ng)
    out = np.fft.irfft(np.conj(fp) * fa, Ng)
    h = out[idx]

    g = 2.0 * h
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)

    d = g - float(np.mean(g))
    d = np.nan_to_num(d, nan=0.0, posinf=0.0, neginf=0.0)
    dn = float(np.linalg.norm(d))
    if dn < 1e-18:
        d = None
    else:
        d = d / dn

    return {
        "c": c,
        "cmax": cmax,
        "score": score,
        "beta": beta,
        "f": f,
        "d": d,
    }


def apply_log_step(a, d, base):
    if d is None:
        return None
    logs = -base * d
    logs = np.nan_to_num(logs, nan=0.0, posinf=0.0, neginf=0.0)
    logs = logs - float(np.max(logs))
    t = a * np.exp(logs)
    return normalize_seq(t)


def try_gradient(a, Nc, Ng, idx, beta_factor, step_factor):
    stats = compute_gradient(a, Nc, Ng, idx, beta_factor)
    if stats is None or stats["d"] is None:
        return None, None

    f = stats["f"]
    score = stats["score"]
    beta = stats["beta"]
    d = stats["d"]

    n = a.size
    base = step_factor / float(np.sqrt(n))

    best_score_cand = None
    best_score_val = np.inf
    best_f_cand = None
    best_f_val = np.inf

    for factor in (1.0, 0.5, 2.0):
        t = apply_log_step(a, d, base * factor)
        if t is None:
            continue

        ct = convolve_fft(t, Nc)
        st = 2.0 * n * float(np.max(ct))
        ft = smooth_value(ct, beta)

        if st < best_score_val:
            best_score_val = st
            best_score_cand = t
        if ft < best_f_val:
            best_f_val = ft
            best_f_cand = t

    if best_score_cand is not None and best_score_val < score * (1.0 - 1e-9):
        return best_score_cand, best_score_val

    if best_f_cand is not None and best_f_val < f * (1.0 - 1e-7):
        ct = convolve_fft(best_f_cand, Nc)
        st = 2.0 * n * float(np.max(ct))
        return best_f_cand, st

    return None, None


def perturb_log(a, rng, strength):
    n = a.size
    loga = np.log(np.maximum(a, 1e-300))
    loga = loga - float(np.mean(loga))
    noise = rng.normal(0.0, 1.0, n)
    noise = 0.5 * (noise + np.roll(noise, 1))
    noise = 0.5 * (noise + np.roll(noise, -1))
    t = np.exp(loga + strength * noise)
    return normalize_seq(t)


def block_perturb(a, rng, strength):
    t = np.array(a, dtype=np.float64, copy=True)
    n = t.size
    if n < 2:
        return normalize_seq(t)

    m = int(rng.integers(1, 4))
    for _ in range(m):
        q = int(rng.integers(5, 80))
        w = max(2, int(n // max(1, q)))
        if w >= n:
            w = max(2, n // 2)
        s = int(rng.integers(0, max(1, n - w + 1)))
        mult = float(np.exp(rng.normal(0.0, strength)))
        t[s:s + w] *= mult

    return normalize_seq(t)


def make_seed(n, kind, eps, rng):
    i = np.arange(n, dtype=np.float64)

    if n <= 1:
        a = np.ones(max(1, n), dtype=np.float64)
        return normalize_seq(a)

    denom = float(n - 1)

    if kind == 0:
        a = np.ones(n, dtype=np.float64)
    elif kind == 1:
        a = 1.0 / np.sqrt(i + 1.0)
    elif kind == 2:
        a = 1.0 / np.sqrt(n - i)
    elif kind == 3:
        left = 1.0 / np.sqrt(i + 1.0)
        r = 1.0 - i / denom
        mask = r * r
        ms = float(np.sum(mask))
        if ms <= 0.0:
            mask = np.ones(n, dtype=np.float64)
            ms = float(n)
        a = left + eps * float(np.sum(left)) * mask / ms
    elif kind == 4:
        lam = float(rng.uniform(0.5, 6.0))
        a = np.exp(-lam * i / denom)
    elif kind == 5:
        z = rng.normal(0.0, 1.0, n)
        zf = np.fft.rfft(z)
        cutoff = max(2, int(n // 5))
        if cutoff < zf.size:
            zf[cutoff:] = 0.0
        z = np.fft.irfft(zf, n)[:n]
        a = np.exp(0.15 * z)
    elif kind == 6:
        left = 1.0 / np.sqrt(i + 1.0)
        a = left + eps * float(np.sum(left)) / float(n)
    elif kind == 7:
        # Often useful compromise: left singular plus a weaker right tail.
        left = 1.0 / np.sqrt(i + 1.0)
        r = 1.0 - i / denom
        mask = np.exp(-3.0 * r * r)
        ms = float(np.sum(mask))
        if ms <= 0.0:
            mask = np.ones(n, dtype=np.float64)
            ms = float(n)
        a = left + eps * float(np.sum(left)) * mask / ms
    else:
        a = np.ones(n, dtype=np.float64)

    return normalize_seq(a)


def spectral_step(a, target, lam, gamma, Nc):
    A = np.fft.rfft(a, Nc)
    T = np.fft.rfft(target, Nc)
    maxA = float(np.max(np.abs(A)))
    if maxA <= 0.0 or not np.isfinite(maxA):
        return None

    denom = A + lam * maxA
    An = 0.5 * (A + T / denom)
    raw = np.fft.irfft(An, Nc)[:a.size]
    raw = np.nan_to_num(raw, nan=0.0, posinf=0.0, neginf=0.0)

    t = a + gamma * (raw - a)
    return normalize_seq(t)


def try_spectral(a, Nc, base_score):
    n = a.size
    c = convolve_fft(a, Nc)
    cmax = float(np.max(c))
    score = 2.0 * n * cmax
    L = c.size
    avg = 1.0 / float(L)

    best = None
    best_sc = np.inf

    reductions = (0.005, 0.01, 0.02, 0.04)
    for red in reductions:
        ts = base_score - red
        if ts <= 1.0:
            continue

        T = ts / (2.0 * n)
        Tmin = avg * 1.001
        if T < Tmin:
            T = Tmin
        if T >= cmax:
            continue

        target = np.minimum(c, T)
        s = float(np.sum(target))
        if s <= 0.0 or not np.isfinite(s):
            continue
        target = target / s

        for lam in (1e-3, 1e-2):
            for gamma in (1.0, 0.5):
                t = spectral_step(a, target, lam, gamma, Nc)
                if t is None:
                    continue
                st = 2.0 * n * float(np.max(convolve_fft(t, Nc)))
                if st < best_sc:
                    best_sc = st
                    best = t

    if best is not None and best_sc < score * (1.0 - 1e-9) and best_sc < base_score * (1.0 - 1e-9):
        return best, best_sc
    return None, None


def optimize_sequence(a, plan, rng, deadline):
    n = a.size
    Nc, Ng, idx = plan

    a = normalize_seq(a)
    if a is None:
        return None, np.inf

    local_best = a.copy()
    local_score = score_candidate(local_best, plan)

    step_factor = 1.0
    beta_factor = 10.0
    stuck = 0
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # Periodic larger jumps by spectral flattening.
        if iteration % 45 == 0:
            cand, sc = try_spectral(a, Nc, local_score)
            if cand is not None and sc < local_score * (1.0 - 1e-8):
                local_best = cand.copy()
                local_score = sc
                a = cand.copy()
                stuck = 0
                if iteration < 50 or (iteration % 100 == 0):
                    print("    spectral improve:", local_score)

        # Restart / kick if stuck or step too small.
        if step_factor < 0.03 or stuck > 35 or (iteration % 160 == 0):
            base = local_best if rng.random() < 0.8 else a
            strength = 0.03 + 0.12 * float(rng.random())
            a = perturb_log(base, rng, strength)
            if rng.random() < 0.35:
                a = block_perturb(a, rng, 0.05 + 0.10 * float(rng.random()))
            a = normalize_seq(a)
            step_factor = 1.0
            stuck = 0
            continue

        t, sc = try_gradient(a, Nc, Ng, idx, beta_factor, step_factor)

        if t is not None:
            a = t
            if sc < local_score * (1.0 - 1e-9):
                local_best = a.copy()
                local_score = sc
                stuck = 0
                step_factor = min(2.0, step_factor * 1.05)
            else:
                stuck += 1
                step_factor = min(2.0, step_factor * 1.02)
        else:
            stuck += 1
            step_factor = max(0.03, step_factor * 0.8)

        if time.monotonic() > deadline:
            break

    return local_best, local_score


def task_key(task):
    return task[0]


def generate_tasks(rng):
    tasks = []

    n_mains = [8192, 4096]
    specs = [
        (1, 0.0),
        (3, 0.02),
        (3, 0.05),
        (3, 0.10),
        (3, 0.20),
        (7, 0.03),
        (7, 0.08),
        (6, 0.05),
        (6, 0.15),
        (2, 0.0),
        (4, 0.0),
        (5, 0.0),
    ]

    for n in n_mains:
        plan = get_plan(n)
        for kind, eps in specs:
            a = make_seed(n, kind, eps, rng)
            if a is None:
                continue
            sc = score_candidate(a, plan)
            if np.isfinite(sc):
                tasks.append((sc, a, n, plan))

    try:
        hs = height_sequence_1
    except NameError:
        hs = None

    if hs is not None:
        a = normalize_seq(hs)
        if a is not None and 1024 <= a.size <= 16384:
            plan = get_plan(a.size)
            sc = score_candidate(a, plan)
            if np.isfinite(sc):
                tasks.append((sc, a, a.size, plan))

                # A gentle perturbed copy of the provided state-of-the-art sequence.
                a2 = block_perturb(a, rng, 0.08)
                if a2 is not None:
                    sc2 = score_candidate(a2, plan)
                    if np.isfinite(sc2):
                        tasks.append((sc2, a2, a.size, plan))

    return tasks


def propose_candidate(seed=42, budget_s=1500):
    """Main entry point. Runs FFT/local-spectral search within the time budget."""
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(10.0, budget_s - 5.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Initialize from the provided previous construction if available.
    try:
        hs = height_sequence_1
    except NameError:
        hs = None

    if hs is not None:
        a = normalize_seq(hs)
        if a is not None:
            plan = get_plan(a.size)
            sc = score_candidate(a, plan)
            if np.isfinite(sc) and sc < best_score:
                best_seq = a.copy()
                best_score = sc
                print("previous construction score:", float(best_score), "len:", a.size)

    tasks = generate_tasks(rng)
    valid = []
    for t in tasks:
        if np.isfinite(t[0]):
            valid.append(t)

    valid.sort(key=task_key)
    valid = valid[:12]

    # Reserve time for final polishing of the global best.
    reserve = 120.0
    base_budget = max(0.0, deadline - time.monotonic() - reserve)
    if base_budget > 60.0:
        weights = [0.30, 0.18, 0.12, 0.09, 0.07, 0.06, 0.05, 0.04, 0.04, 0.03, 0.02, 0.00]

        for i, (sc, a, n, plan) in enumerate(valid):
            if time.monotonic() >= deadline - reserve:
                break

            # Skip clearly poor initializations.
            if np.isfinite(best_score) and sc > best_score + 0.15:
                continue

            w = weights[i] if i < len(weights) else 0.0
            if w <= 0.0:
                continue
            sub_deadline = time.monotonic() + w * base_budget
            sub_deadline = min(sub_deadline, deadline - reserve)

            if sub_deadline - time.monotonic() < 5.0:
                continue

            print("task", i, "n=", n, "initial=", round(float(sc), 6), "time=", round(float(w * base_budget), 1), "s")

            b, bs = optimize_sequence(a, plan, rng, sub_deadline)
            if b is not None and np.isfinite(bs) and bs < best_score:
                best_score = bs
                best_seq = b.copy()
                print("global update:", float(best_score), "len:", b.size)

    # Final polish using remaining time.
    if best_seq is not None:
        final_deadline = deadline - 2.0
        if final_deadline > time.monotonic() + 10.0:
            plan = get_plan(best_seq.size)
            b, bs = optimize_sequence(best_seq, plan, rng, final_deadline)
            if b is not None and np.isfinite(bs) and bs < best_score:
                best_score = bs
                best_seq = b.copy()
                print("final polish update:", float(best_score), "len:", b.size)
        else:
            # Even a short finalimprovement window can help.
            final_deadline = deadline - 0.5
            if final_deadline > time.monotonic() + 1.0:
                plan = get_plan(best_seq.size)
                b, bs = optimize_sequence(best_seq, plan, rng, final_deadline)
                if b is not None and np.isfinite(bs) and bs < best_score:
                    best_score = bs
                    best_seq = b.copy()

    # Fallback if something went wrong: use a cheap one-sided singular seed.
    if best_seq is None:
        n = 2048
        best_seq = make_seed(n, 1, 0.0, rng)

    best_seq = normalize_seq(best_seq)
    if best_seq is None:
        best_seq = np.ones(2048, dtype=np.float64)
        best_seq /= 2048.0

    out_list = [float(x) for x in best_seq]

    # Official final evaluation is scale-invariant and clip-safe for our outputs.
    try:
        official = evaluate_sequence(out_list)
    except Exception:
        official = np.inf

    if np.isfinite(official):
        print("official score:", float(official), "len:", len(out_list))
    else:
        print("official score not finite; fast estimate:", float(best_score), "len:", len(out_list))

    return out_list
```
