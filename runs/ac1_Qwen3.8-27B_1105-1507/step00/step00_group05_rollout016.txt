```python
import numpy as np
import time

def simplex_project(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    # Use the standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_score_normalized(a):
    """Compute score assuming sum(a) = 1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * b.max()

def optimize_sequence(a0, lr=0.001, t=0.01, max_iter=10000, verbose=False):
    """Optimize sequence using projected gradient descent on soft-max objective."""
    a = a0.copy()
    s = a.sum()
    a = a / s  # normalize to sum = 1
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    
    for it in range(max_iter):
        b = np.convolve(a, a)
        # Softmax weights
        b_max = b.max()
        w = np.exp((b - b_max) / t)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.zeros_like(b)
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        
        # Gradient
        a_rev = a[::-1]
        conv = np.convolve(w, a_rev)
        grad = 2.0 * conv[len(a)-1:len(a)-1+len(a)]
        
        # Update
        a_new = a - lr * grad
        a_new = simplex_project(a_new, 1.0)
        
        score = compute_score_normalized(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            if verbose and it % 1000 == 0:
                print(f"Iter {it}: score = {best_score:.6f}")
        a = a_new
    
    return best_a, best_score
```

```python
import numpy as np
import time

def simplex_project(v, s=1.0):
    """Project v onto the probability simplex scaled by s."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - s)
    if not np.any(cond):
        return np.full(n, s/n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def make_u_shape(n, alpha=1.0):
    """Create a U-shaped sequence of length n."""
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2*x - 1)**2
    a = np.maximum(a, 0.0)
    return a / a.sum()

def make_cosine(n, num_harmonics=5, amplitudes=None):
    """Create a cosine series sequence."""
    if amplitudes is None:
        amplitudes = np.random.default_rng(0).uniform(-0.3, 0.3, num_harmonics)
        amplitudes[0] = 0
    a = np.ones(n)
    for j in range(1, num_harmonics+1):
        a += amplitudes[j-1] * np.cos(2*np.pi*j*np.arange(n)/n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    return a / a.sum()

def compute_b(a):
    """Compute the linear convolution a*a."""
    return np.convolve(a, a)

def compute_score(a):
    """Compute the evaluation score (scale-invariant)."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = compute_b(a)
    return 2.0 * n * b.max() / (s*s)

def soft_gradient(a, K=10, t=0.05):
    """Compute gradient of soft-top-K of convolve(a,a)."""
    b = compute_b(a)
    n = len(a)
    # Get top K indices
    if K < len(b):
        top_idx = np.argsort(b)[::-1][:K]
        # Soft weights among top K
        b_top = b[top_idx]
        w_top = np.exp((b_top - b_top.max()) / t)
        w_top /= w_top.sum()
        # Build full weight vector
        w = np.zeros_like(b)
        w[top_idx] = w_top
    else:
        w = np.exp((b - b.max()) / t)
        w /= w.sum()

    a_rev = a[::-1]
    conv = np.convolve(w, a_rev)
    grad = 2.0 * conv[n-1:n-1+n]

def optimize_one(a0, time_budget=60.0, K=10, t=0.05, lr_init=0.01, verbose=False):
    """Run projected gradient descent on one initial sequence."""
    a = a0.copy()

    if s < 1e-10:
        a = np.ones_like(a) / len(a)
        s = 1.0

    no_improve = 0

        grad = soft_gradient(a, K=K, t=t)
        a_new = a - lr * grad
        a_new = simplex_project(a_new, 1.0)
        
        score = compute_score(a_new)
        if score < best_score:

                no_improve = 0
                a = best_a.copy()

        it += 1

def block_perturb(a, rng, scale=0.1):
    """Apply a random block perturbation."""
    n = len(a)
    a_new = a.copy()
    width = max(1, n // rng.integers(50, 200))
    start = rng.integers(0, n - width)
    # Multiplicative perturbation
    factor = np.exp(rng.normal(0, scale, size=width))
    a_new[start:start+width] *= factor
    # Clip to non-negative
    a_new = np.maximum(a_new, 0)
    # Renormalize
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new
```
