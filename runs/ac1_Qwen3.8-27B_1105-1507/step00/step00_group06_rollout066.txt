"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over
non-negative sequences.

Strategy:
  Phase 1: Quickly explore several lengths and smooth patterns to escape
           the supplied parent's local minimum.
  Phase 2: LP-based direction finding.  At each step solve
             max sum(b)  s.t.  b>=0, (p*b)[k] <= M  for active k,
           where M = max(conv(p,p)).  The solution b* is the most
           "spread" direction compatible with the current peak; mixing
           towards it reduces the autocorrelation peak.
  Phase 3: Random local search with peak-guided and block perturbations,
           interleaved with cheap LP steps to escape stalls.

The objective is scale-invariant, so we work with sum=1 internally and
rely on evaluate_sequence as the authoritative scorer.
"""

import numpy as np

def _project_simplex(v):
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v.astype(float)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _score_raw(p):
    """Scale-invariant objective for a non-negative sequence p."""
    n = len(p)
    conv = np.convolve(p, p)
    return 2.0 * n * conv.max()


def _solve_lp_direction(p, M, max_rows, rng_solver):
    """
    Solve  max sum(b) s.t. b>=0 and (p*b)[k] <= M for a subset of active
    constraints.  Returns normalized b (sum=1) or None on failure.
    """
    import cvxpy as cp

    n = len(p)
    conv = np.convolve(p, p)
    threshold = M * 0.95
    active = np.where(conv >= threshold)[0]
    if len(active) > max_rows:
        active = np.argsort(conv)[-max_rows:]
    if len(active) == 0:
        active = np.array([int(conv.argmax())])

    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active:
        k = int(k)
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        if j_min > j_max:
            continue
        js = np.arange(j_min, j_max + 1)
        coeffs = p[k - js]
        row = np.zeros(n)
        row[j_min:j_max + 1] = coeffs
        constraints.append(row @ b_var <= M)

    if not constraints:
        return None

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    for solver in (cp.ECOS, cp.SCS, cp.CLARABEL, cp.GLPK):
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status in ("optimal", "optimal_inaccurate") and b_var.value is not None:
            b = np.asarray(b_var.value, dtype=np.float64)
            b = np.clip(b, 0, None)
            s = b.sum()
            if s > 1e-10:
                return b / s
    return None


def _make_patterns(n, rng):
    """A small library of smooth non-negative prototype patterns of length n."""
    x = np.arange(n, dtype=np.float64)
    cx = (n - 1) / 2.0
    patterns = [
        np.ones(n, dtype=np.float64),
        np.exp(-0.5 * ((x - cx) / max(1.0, n / 6.0)) ** 2),
        np.exp(-((x - cx) / max(1.0, n / 4.0)) ** 4),
        0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1))),
        np.abs(np.sin(np.pi * x / float(n))),
        1.0 / (1.0 + ((x - cx) / max(1.0, n / 3.0)) ** 2),
    ]
    r = rng.random(n)

    # a two-hump pattern

                          0.3 * np.cos(4 * np.pi * x / float(n))) + 0.05)
    return [np.clip(np.asarray(p, dtype=np.float64), 0, None) for p in patterns]
