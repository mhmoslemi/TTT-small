```python
from scipy.optimize import linprog

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. f*g <= M (componentwise), g >= 0"""
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)_k <= M for k = 0, ..., 2n-2
    # (f*g)_k = sum_{i=0}^{n-1} f_i * g_{k-i} where g_j = 0 for j outside [0, n-1]

    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):

            j = k - i
            if 0 <= j < n:

    b_ub = np.full(2*n-1, M)
```

```python
def simplex_project(y):
    """Project y onto the simplex {x >= 0, sum(x) = 1}"""
    n = len(y)
    # Sort in decreasing order
    sorted_y = np.sort(y)[::-1]
    # Find tau
    cv = np.cumsum(sorted_y)
    rho = np.nonzero(sorted_y * np.arange(1, n+1) > (cv[-1] - 1))[0][-1]
    tau = (cv[rho] - 1) / (rho + 1)
    return np.maximum(y - tau, 0)
```

```python
        return np.ones(n) / n
    u = np.sort(y)[::-1]
```
