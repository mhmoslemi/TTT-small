"""
Subgradient optimization for minimizing the peak-to-sum ratio
of the autocorrelation of a non-negative sequence.
Uses FFT-based convolution, adaptive learning rates, and
simulated annealing for exploration.
"""

import numpy as np
import time

def compute_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2*n - 1
    # Use next power of 2 for efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    fa2 = fa * fa
    result = np.fft.irfft(fa2, fft_size)[:size]
    # Small numerical errors can produce tiny negative values
    result = np.maximum(result, 0)
    return result

def subgradient_optimize(a_init, time_budget, rng, target_sum=1.0):
    """
    Optimize a non-negative sequence to minimize max(conv(a,a)) / sum(a)^2.
    Uses subgradient method with simulated annealing.
    """
    n = len(a_init)
    a = np.array(a_init, dtype=np.float64)
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S  # normalize to sum=1
    
    # Initial evaluation
    C = compute_conv(a)
    M = C.max()
    best_a = a.copy()
    best_M = M
    best_J = 2 * n * M  # since sum=1, J = 2n*M
    
    lr = 0.005
    temp = 0.01
    no_improve = 0
    start_time = time.monotonic()
    deadline = start_time + time_budget
    iter_count = 0
    
    while time.monotonic() < deadline:
        iter_count += 1
        
        # Compute convolution
        C = compute_conv(a)
        M = C.max()
        
        # Check improvement
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            best_J = 2 * n * M
            no_improve = 0
        
        if no_improve > 0:
            no_improve += 1
        
        # Find peak positions (within 0.999 of max)
        threshold = M * 0.999
        peak_indices = np.where(C >= threshold)[0]
        
        # Limit to top 50 peaks for efficiency
        if len(peak_indices) > 50:
            peak_vals = C[peak_indices]
            top_idx = peak_indices[np.argsort(peak_vals)[-50:]]
            peak_indices = top_idx
        
        # Compute subgradient
        g = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-lo : k-hi-1 : -1]
        
        g /= len(peak_indices)
        
        # Update
        a = np.maximum(0.0, a - lr * g + temp * rng.standard_normal(n) * 0.01)
        
        # Normalize
        S = a.sum()
        if S < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / S
        
        # Adaptive parameters
        if no_improve > 20000:
            lr *= 0.5
            temp *= 0.5
            no_improve = 0
            # Big random restart
            if rng.random() < 0.3:
                mask = rng.random(n) < 0.1
                a[mask] = rng.random(mask.sum()) * 0.1
                a = np.maximum(0, a)
                S = a.sum()
                if S > 1e-10:
                    a = a / S
        
        if no_improve > 5000:
            lr *= 0.9
            no_improve = 0
        
        # Decay temperature
        if iter_count % 1000 == 0:
            temp *= 0.995
    
    return best_a, best_J

"""
Subgradient optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2
over non-negative sequences. Uses FFT-based convolution, adaptive
subgradient updates with simulated annealing for exploration, and
multiple initialization strategies (parent, random, geometric, windowed).
"""

import numpy as np
import time


def fft_convolve(a):
    """Compute linear autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]


def subgradient_step(a, C, n, lr, temp, rng):
    """Perform one subgradient step. Returns updated sequence."""
    M = C.max()
    if M < 1e-15:
        return a
    # Find peak positions (within 0.999 of max)
    threshold = M * 0.999
    peak_mask = C >= threshold
    peak_indices = np.where(peak_mask)[0]
    # Limit to top 30 for efficiency
    if len(peak_indices) > 30:
        pk_vals = C[peak_indices]
        top = np.argsort(pk_vals)[-30:]
        peak_indices = peak_indices[top]
    # Compute subgradient
    g = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - lo:k - hi - 1:-1]
    g /= len(peak_indices)
    # Update with annealing noise
    noise = temp * rng.standard_normal(n) * 0.005
    a_new = a - lr * g + noise
    a_new = np.maximum(0.0, a_new)
    S = a_new.sum()
    if S < 1e-12:
        a_new = np.ones(n) / n
    else:
        a_new /= S
    return a_new


def optimize_sequence(a_init, time_budget, rng):
    """
    Run subgradient optimization on a_init for time_budget seconds.
    Returns (best_sequence, best_J) where J = 2*n*max(conv)/sum^2.
    """
    n = len(a_init)
    a = np.array(a_init, dtype=np.float64)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a /= S

    C = fft_convolve(a)
    M = C.max()
    best_a = a.copy()
    best_M = M
    best_J = 2.0 * n * M

    lr = 0.008 / np.sqrt(n / 256.0)
    temp = 0.05
    no_improve = 0
    start_t = time.monotonic()
    deadline = start_t + time_budget
    iters = 0

    while time.monotonic() < deadline:
        iters += 1
        C = fft_convolve(a)
        M = C.max()

        if M < best_M * (1.0 - 1e-8):
            best_M = M
            best_a = a.copy()
            best_J = 2.0 * n * M
            no_improve = 0
            # Reduce lr slightly on improvement
            lr *= 0.999
        else:
            no_improve += 1

        a = subgradient_step(a, C, n, lr, temp, rng)

        # Adaptive scheduling
        if no_improve > 30000:
            lr *= 0.5
            temp *= 0.5
            no_improve = 0
            # Random restart perturbation
            if rng.random() < 0.4:
                n_perturb = max(1, n // 10)
                idx = rng.choice(n, size=n_perturb, replace=False)
                a[idx] += rng.random(n_perturb) * 0.1
                a = np.maximum(0, a)
                s = a.sum()
                if s > 1e-12:
                    a /= s

        elif no_improve > 5000:
            lr *= 0.85
            no_improve = 0

        # Slowly decay temperature
        if iters % 500 == 0:
            temp *= 0.99

        # Floor on lr
        lr = max(lr, 1e-6 / np.sqrt(n))

    return best_a, best_J


def make_windowed_init(n, rng, window_type='hann'):
    """Create a windowed initialization."""
    x = np.linspace(0, 1, n)
    if window_type == 'hann':
        w = 0.5 * (1 - np.cos(2 * np.pi * x))
    elif window_type == 'hamming':
        w = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    elif window_type == 'inverted_hann':
        w = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        w = np.ones(n)
    # Add small random perturbation
    w *= (1.0 + 0.1 * rng.standard_normal(n))
    w = np.maximum(0, w)
    s = w.sum()
    if s > 1e-12:
        w /= s
    return w
