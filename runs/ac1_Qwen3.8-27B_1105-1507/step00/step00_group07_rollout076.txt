import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

"""
Iterative LP-based optimization to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
At each step, solves an LP to find direction g maximizing sum(g) subject to
convolve(f,g) <= current_peak, then blends f toward g.
Tries multiple sequence lengths (512, 1024, 2048) and initializations.
"""


def _build_lp_matrix(f, n):
    """Build sparse (2n-1 x n) matrix A where A@G = convolve(f, G)."""
    n_rows = 2 * n - 1
    row_list = []
    col_list = []
    val_list = []
    for k in range(n_rows):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        row_list.extend([k] * length)
        col_list.extend(range(j_start, j_end + 1))
        val_list.extend((f[k - j_start:k - j_start + length]).tolist())
    return csr_matrix((val_list, (row_list, col_list)), shape=(n_rows, n))


def _score(a):
    """Compute 2*n*max(convolve(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _solve_step(f, n, M):
    """Solve LP: max sum(g) s.t. A@G <= M, G >= 0. Returns g or None."""
    A = _build_lp_matrix(f, n)
    c_obj = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    return np.clip(res.x, 0.0, None)


def _optimize_lp(n, budget_s, rng, init=None):
    """Iterative LP optimization for fixed length n."""
    deadline = time.monotonic() + budget_s

    if init is not None:
        f = np.array(init, dtype=np.float64)
    else:
        f = np.ones(n, dtype=np.float64)
    f = np.clip(f, 0.0, None)
    if f.sum() < 1e-10:
        f = np.ones(n, dtype=np.float64)
    f = f / f.sum()

    best_f = f.copy()
    best_score = _score(f)
    print(f"    init: n={n}, score={best_score:.6f}")

    it = 0
    stall = 0
    t_values = [0.4, 0.6, 0.8, 1.0]

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(conv.max())

        g = _solve_step(f, n, M)
        if g is None or g.sum() < 1e-15:
            break

        f_sum = f.sum()
        g_scaled = g * (f_sum / g.sum())

        best_step_score = best_score
        best_step_f = None
        for t in t_values:
            f_new = (1.0 - t) * f + t * g_scaled
            f_new = np.clip(f_new, 0.0, None)
            if f_new.sum() < 1e-10:
                continue
            s = _score(f_new)
            if s < best_step_score:
                best_step_score = s
                best_step_f = f_new

        if best_step_f is not None:
            stall = 0
            best_f = best_step_f.copy()
            best_score = best_step_score
            f = best_step_f
        else:
            stall += 1
            if stall > 30:
                f = f + rng.normal(0, 0.1, n)
                f = np.clip(f, 0.0, None)

                f = f / f.sum()
                stall = 0

            print(f"    n={n}, it={it}, score={best_score:.6f}, M={M:.4f}, elapsed={elapsed:.0f}s")

    return best_f, best_score
