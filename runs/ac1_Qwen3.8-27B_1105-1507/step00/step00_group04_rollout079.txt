Wait, I need to be careful about the FFT convention. In numpy:

- IFFT: x[n] = (1/L) * sum_k X[k] * exp(2πikn/L)

The cross-correlation theorem: if R = IFFT(FFT(x, L) * conj(FFT(y, L))), then

So R[k] = sum_n x[n+k] * y[n] (for real signals) (where the sum is over n where both are defined, with circular wraparound if L is too small).

def compute_gradient_fft(p, w):
    """Compute g[j] = 2 * sum_m p[m] * w[j+m] for j=0..n-1 using FFT."""
    n = len(p)
    L = 1
    while L < 3 * n:
        L *= 2
    W_fft = np.fft.rfft(w, L)
    P_fft = np.fft.rfft(p, L)

def compute_grad_fft(p, w):
    """Compute g[j] = 2 * sum_m p[m] * w[j+m] for j=0..n-1 using FFT."""
    n = len(p)
    # Need L >= len(w) + len(p) - 1 = 3n - 2
    L = 1
    while L < 3 * n:
        L <<= 1
    Wf = np.fft.rfft(w, L)
    Pf = np.fft.rfft(p, L)
    R = np.fft.irfft(Wf * np.conj(Pf), L)

            alpha = 0.02
            p_opt, max_val = _run_descent(best_seq, 50.0, alpha, deadline,
                                          int(time_left * 300))
            score = 2.0 * n * max_val
            if score < best_score:
                best_score = score
                best_seq = p_opt
                print(f"Polished: J={score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = 2.0

    result = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final: n={len(result)}, J={best_score:.6f}")
    return [float(x) for x in result]
