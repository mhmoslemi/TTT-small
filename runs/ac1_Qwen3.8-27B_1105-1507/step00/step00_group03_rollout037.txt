```python
def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if v.sum() < 1e-10:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(ind[cond])
    theta = cssv[rho-1] / rho
    return np.maximum(v - theta, 0)
```

```python
Minimize 2n * max(self_conv) / (sum_a)^2 using LP-based iterative refinement.
The objective is scale-invariant, so we normalize to sum=1.
Main algorithm: iteratively solve an LP to find the best g minimizing max(f*g),
then line-search along the direction f -> (1-t)*f + t*g.
Fallback: subgradient descent on the piecewise-convex objective.

def project_simplex(v):
    """Project vector onto probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    v = np.asarray(v, dtype=np.float64)
    if not np.isfinite(v).all():
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(ind[cond]))
    theta = cssv[rho - 1] / rho
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    else:
        result = np.ones(n) / n
    return result


def fast_self_conv(a):
    """Compute full self-convolution using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    return conv

    """Build the (2n-1) x n convolution constraint matrix.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):

    return A

    A_ub[:, n] = -1.0
    
    c = np.zeros(n + 1)
    c[n] = 1.0

    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    b_ub = np.zeros(2 * n - 1)
    bounds = [(0.0, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 60})

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < float('inf'):
            return candidate, score
    return None, float('inf')


def multi_k_gradient_step(f, n, lr_list):
    """Use gradient from top-k peaks for more robust descent."""
    conv = fast_self_conv(f)
    top_k = min(5, len(conv))
    peak_indices = np.argsort(conv)[::-1][:top_k]
    
    grad = np.zeros(n)
    for k_star in peak_indices:
        k_star = int(k_star)
        weight = conv[k_star]
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] += weight * 2.0 * f[idx]
    
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return None, float('inf')
    grad /= grad_norm
    
    for lr in lr_list:
        candidate = f - lr * grad
        candidate = project_simplex(candidate)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score):
            return candidate, score
    return None, float('inf')


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Minimize 2n * max(self_conv) / (sum_a)^2 via LP-based iterative refinement.
    
    The objective is scale-invariant, so we work with sum(a)=1.
    Each iteration solves an LP to find the best direction, then line-searches.
    Falls back to subgradient descent when LP stalls.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 10.0
    
    # Initialize from parent or uniform
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        parent = np.ones(2535, dtype=np.float64)
    
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        n = 2535
        parent = np.ones(n, dtype=np.float64)
    else:
        n = parent.size
    
    parent = np.clip(parent, 0.0, 1000.0)
    s = parent.sum()
    if s < 1e-10:
        best = np.ones(n) / n
    else:
        best = parent / s
    
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Init: n={n}, score={best_score:.8f}", flush=True)
    
    if not np.isfinite(best_score):
        best = np.ones(n) / n
        best_score = float(evaluate_sequence(best.tolist()))
        print(f"Reset to uniform: score={best_score:.8f}", flush=True)
    
    stalled = 0
    max_stall = 30
    last_print = 0
    phase = 'lp'  # 'lp', 'grad', 'perturb'
    
    for iteration in range(500000):
        if time.monotonic() > deadline:
            break
        elapsed = time.monotonic() - (deadline - total_budget + 10.0)
        
        improved = False
        
        if phase == 'lp':
            # LP-based step
            t0 = time.monotonic()
            result = solve_lp(best, n)
            lp_time = time.monotonic() - t0
            
            if result.success and result.x is not None:
                g = np.clip(result.x[:n], 0.0, 1000.0)
                gs = g.sum()
                if gs < 1e-10:
                    g = np.ones(n) / n
                else:
                    g = g / gs
                
                # Line search
                for t in [1.0, 0.9, 0.7, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
                    candidate = (1.0 - t) * best + t * g
                    candidate = np.maximum(candidate, 0.0)
                    cs = candidate.sum()
                    if cs < 1e-10:
                        continue
                    candidate = candidate / cs
                    score = float(evaluate_sequence(candidate.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best = candidate
                        best_score = score
                        improved = True
                        if iteration - last_print >= 20 or best_score < 2.0:
                            print(f"Iter {iteration} [{phase}] t={t:.2f} score={best_score:.8f} lp_t={lp_time:.1f}s elapsed={elapsed:.0f}s", flush=True)
                            last_print = iteration
                        break
                
                if not improved:
                    # Try pure LP solution
                    score_g = float(evaluate_sequence(g.tolist()))
                    if np.isfinite(score_g) and score_g < best_score:
                        best = g
                        best_score = score_g
                        improved = True
                        print(f"Iter {iteration} [{phase}] pure-LP score={best_score:.8f}", flush=True)
                        last_print = iteration
            else:
                # LP failed, switch to gradient
                phase = 'grad'
        
        if not improved and (phase == 'grad' or phase == 'lp'):
            # Gradient descent fallback
            lr_list = [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002]
            candidate, score = gradient_descent_step(best, n, lr_list)
            if candidate is not None and np.isfinite(score) and score < best_score:
                best = candidate
                best_score = score
                improved = True
                if iteration - last_print >= 50 or best_score < 2.0:
                    print(f"Iter {iteration} [grad] score={best_score:.8f} elapsed={elapsed:.0f}s", flush=True)
                    last_print = iteration
            
            if not improved:
                # Try multi-k gradient
                lr_list2 = [0.5, 0.3, 0.1, 0.05, 0.02, 0.01]
                candidate, score = multi_k_gradient_step(best, n, lr_list2)
                if candidate is not None and np.isfinite(score) and score < best_score:
                    best = candidate
                    best_score = score
                    improved = True
        
        if improved:
            stalled = 0
            # If we've been improving well, switch back to LP
            if stalled == 0 and phase == 'grad' and iteration % 10 == 0:
                phase = 'lp'
        else:
            stalled += 1
            if stalled >= max_stall:
                # Escape: random perturbation
                print(f"Stalled {stalled} iters, perturbing...", flush=True)
                for attempt in range(20):
                    sigma = 0.1 + 0.1 * rng.random()
                    perturbed = best * np.exp(rng.normal(0, sigma, n))
                    perturbed = project_simplex(perturbed)
                    score_p = float(evaluate_sequence(perturbed.tolist()))
                    if np.isfinite(score_p) and score_p < best_score:
                        best = perturbed
                        best_score = score_p
                        improved = True
                        print(f"  Escaped: score={best_score:.8f} sigma={sigma:.3f}", flush=True)
                        break
                
                if not improved:
                    # More aggressive: try a completely different structure
                    for trial in range(5):
                        # Try front-loaded, back-loaded, or alternating
                        mode = rng.integers(0, 4)
                        if mode == 0:
                            # Linear ramp
                            t = np.linspace(0.1, 2.0, n)
                        elif mode == 1:
                            # Sine
                            t = 1.0 + 0.5 * np.sin(np.pi * np.arange(n) / n)
                        elif mode == 2:
                            # Exponential decay
                            t = np.exp(-np.linspace(0, 2, n))
                        else:
                            # Random
                            t = np.ones(n) + 0.3 * rng.standard_normal(n)
                        t = np.maximum(t, 0.0)
                        ts = t.sum()
                        if ts > 1e-10:
                            t = t / ts
                            score_t = float(evaluate_sequence(t.tolist()))
                            if np.isfinite(score_t) and score_t < best_score:
                                best = t
                                best_score = score_t
                                improved = True
                                print(f"  New structure (mode={mode}): score={best_score:.8f}", flush=True)
                                break
                
                stalled = 0
                max_stall = min(max_stall + 10, 100)
    
    # Final polish: try very small gradient steps
    if time.monotonic() < deadline:
        for _ in range(50):
            if time.monotonic() > deadline:
                break
            conv = fast_self_conv(best)
            k_star = int(np.argmax(conv))
            grad = np.zeros(n)
            for j in range(n):
                idx = k_star - j
                if 0 <= idx < n:
                    grad[j] = 2.0 * best[idx]
            for lr in [0.01, 0.005, 0.002, 0.001]:
                candidate = best - lr * grad
                candidate = project_simplex(candidate)
                score = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = candidate
                    best_score = score
                    break
        print(f"Polish done: score={best_score:.8f}", flush=True)
    
    print(f"FINAL: n={n}, score={best_score:.8f}", flush=True)
    
    # Ensure output is valid: non-negative, sum >= 0.01
    result = np.clip(best, 0.0, 1000.0)
    rs = result.sum()
    if rs < 0.01:
        result = np.ones(n)
    return [float(x) for x in result]
```
