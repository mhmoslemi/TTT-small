```
import numpy as np
import time

def fft_len_needed(m):
    m=int(m)
    p=1
    while p < m:
        p <<= 1
    return p

def normalize_nonnegative(a):
    a=np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a=np.maximum(a, 0.0)
    s=float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        a=np.ones(a.size, dtype=np.float64)
        return a / float(a.size)
    a = a / s
    return a

def fft_conv_square(a, m=None):
    n=a.size
    need=2*n-1
    if m is None:
       m=fft_len_needed(need)
    fa=np.fft.rfft(a, m)
    c=np.fft.irfft(fa*fa, m)
    if c.size > need:
       c=c[:need]
    c=np.maximum(c,0.0)
    return c

def smooth_gradient(a,c,tau,m=None):
    n=a.size
    if m is None:
       m=fft_len_needed(3*n-1)
    M=float(c.max())
    if M <= 0 or not np.isfinite(M):
       return np.zeros(n, dtype=np.float64)
    z=(c-M)/tau
    # clip
    z=np.clip(z, -50.0, 0.0)
    w=np.exp(z)
    s=float(np.sum(w))
    if not np.isfinite(s) or s <= 0.0:
       w=np.zeros(c.shape, dtype=np.float64)
       w[int(np.argmax(c))]=1.0
    else:
       w=w/s
    b=a[::-1]
    fb=np.fft.rfft(b,m)
    fw=np.fft.rfft(w,m)
    conv=np.fft.irfft(fb*fw,m)
    g=2.0*conv[n-1:2*n-1]
    g=np.maximum(g,0.0)
    return g

def mirror_update(a,g,step):
    gm=float(np.max(g))
    if gm <= 1e-300 or not np.isfinite(gm):
       return a.copy()
    u=g/gm
    # step clip to avoid exp overflow (u>=0 so -step*u <=0)
    if step > 50.0:
       step = 50.0
    ex=np.exp(-step*u)
    out=a*ex
    out=np.maximum(out,0.0)
    s=float(np.sum(out))
    if not np.isfinite(s) or s <= 1e-300:
       return a.copy()
    out=out/s
    return out

def make_modulated(n, alpha, right_alpha, r, phase, freq):
    x=np.arange(n, dtype=np.float64)
    base=(x+1.0)**(-alpha)
    if right_alpha > 0.0:
        right=n-x
        right=np.maximum(right,1.0)
        base=base*(right**(-right_alpha))
    t=x/float(n)
    mod=1.0 + r*np.sin(2.0*np.pi*float(freq)*t + phase)
    base=base*np.maximum(mod,0.0)
    return normalize_nonnegative(base)

def make_structured_candidate(n, rng):
    alpha=float(rng.uniform(0.35, 0.65))
    right_alpha=0.0

    base=(x+1.0)**(-alpha)

    t=x/float(n)
    r=float(rng.uniform(0.0, 1.0))
    phase=float(rng.uniform(0.0, 2.0*np.pi))

    mod=1.0 + r*np.sin(2.0*np.pi*float(freq)*t + phase)

def add_pool_score(pool, a):

    if a.ndim !=1 or a.size ==0 or a.size > 16000:
        return

    c=fft_conv_square(a, None)
    s=float(c.max())
    if not np.isfinite(s):

    if np.isfinite(s):

def generate_pool(lengths, rng, max_seconds):
    pool=[]
    start=time.monotonic()
    # deterministic pure/sym powers
    for n in lengths:
       for alpha in (0.4,0.45,0.5,0.55,0.6):
          x=np.arange(n,dtype=np.float64)
          a=(x+1.0)**(-alpha)
          add_pool_score(pool,a)
       for ra in (0.1,0.25,0.4):
          x=np.arange(n,dtype=np.float64)
          a=(x+1.0)**(-0.5)
          right=np.maximum(n-x,1.0)
          a=a*(right**(-ra))
          add_pool_score(pool,a)
       for r in (0.4,0.7,1.0):
          for freq in (1,2):
             for phase in (0.0, 0.5*np.pi, float(np.pi), 1.5*np.pi):
                a=make_modulated(n,0.5,0.0,r,phase,freq)
                add_pool_score(pool,a)
    # random structured
    for n in lengths:
       if time.monotonic() - start > max_seconds*0.7:
          break
       if n <= 1024:
          count=16
       elif n <= 2048:
          count=12
       elif n <= 4096:
          count=8
       else:
          count=5
       for _ in range(count):
          if time.monotonic() - start > max_seconds:
             break
          a=make_structured_candidate(n,rng)
          add_pool_score(pool,a)
    # height_sequence if present
    try:
       h=height_sequence_1
       if h is not None:
          arr=np.asarray(h, dtype=np.float64)
          if arr.ndim == 1 and arr.size > 0 and arr.size <= 16000:
             arr=normalize_nonnegative(arr)
             add_pool_score(pool, arr)
             if np.all(np.isfinite(arr)):

                x=np.arange(n,dtype=np.float64)
                t=x/float(n)

                      a=arr*np.maximum(mod,0.0)
                      add_pool_score(pool,a)
    except Exception:
       pass
    return pool

def pair_score(p):
    return p[0]

def is_already_selected(selected, p):
    target_id=id(p[1])
    for q in selected:
       if id(q[1]) == target_id:
          return True
    return False

def select_candidates(pool, lengths, max_candidates):
    selected=[]
    best_by_n={}
    for p in pool:
       n=p[1].size
       if n not in best_by_n:
          best_by_n[n]=p
       else:
          if p[0] < best_by_n[n][0]:
             best_by_n[n]=p
    for n in lengths:
       if n in best_by_n and len(selected) < max_candidates:
          p=best_by_n[n]
          if not is_already_selected(selected,p):
             selected.append(p)
    for p in pool:
       if len(selected) >= max_candidates:
          break
       if not is_already_selected(selected,p):
          selected.append(p)
    return selected

def local_optimize(a0, max_seconds, rng):
    a=normalize_nonnegative(a0)
    n=a.size
    if n == 0:
       a=np.ones(1,dtype=np.float64)
       n=1
    m_conv=fft_len_needed(2*n-1)
    m_grad=fft_len_needed(3*n-1)
    c=fft_conv_square(a, m_conv)
    score=2.0*float(n)*float(c.max())
    best=a.copy()
    best_score=score
    base_step=0.25
    tau_frac=0.3
    no_improve=0
    start=time.monotonic()
    if max_seconds <= 0.0:
       return best, best_score
    while time.monotonic() - start < max_seconds:
       M=float(c.max())
       score=2.0*float(n)*M
       if not np.isfinite(score):
          break
       if score < best_score - 1e-13:
          best=a.copy()
          best_score=score
          no_improve=0
       else:
          no_improve += 1
       tau=tau_frac*M
       if not np.isfinite(tau) or tau <= 1e-18:
          tau=1e-18
       g=smooth_gradient(a, c, tau, m_grad)
       accepted=False
       step=base_step
       for attempt in range(5):
          a_new=mirror_update(a,g,step)
          c_new=fft_conv_square(a_new,m_conv)
          score_new=2.0*float(n)*float(c_new.max())
          if np.isfinite(score_new) and score_new < score - 1e-13:
             a=a_new
             c=c_new
             score=score_new
             if score < best_score - 1e-13:
                best=a.copy()
                best_score=score
             no_improve=0
             accepted=True
             if attempt == 0:
                base_step = min(1.5, base_step * 1.05)
             elif attempt == 1:
                base_step = max(0.05, base_step * 0.95)
             else:
                base_step = max(0.03, base_step * 0.85)
             break
          step *= 0.5
       if not accepted:
          no_improve += 1
          if no_improve < 20:
             base_step = max(0.02, base_step * 0.85)
          elif no_improve < 80:
             base_step = min(1.0, base_step * 1.2)
             tau_frac = float(rng.uniform(0.05, 1.0))
          else:
             a=best.copy()
             sig=0.05 + 0.3*min(1.0, float(no_improve - 80)/1000.0)
             a=a*np.exp(rng.normal(0.0, sig, n))
             a=normalize_nonnegative(a)
             c=fft_conv_square(a,m_conv)
             score=2.0*float(n)*float(c.max())
             base_step=0.25
             tau_frac=float(rng.uniform(0.1,0.8))
             no_improve=0
    a=best
    a=normalize_nonnegative(a)
    return a, best_score
```

```python
"""
Search for nonnegative step-function heights minimizing
    2*n*max(conv(a,a)) / (sum(a))^2
using randomized structured initializations, active-lag multiplicative
mirror descent, time-budgeted multi-start local polishing, and final
verification with the supplied evaluate_sequence.
"""

import numpy as np
import time


def fft_len_needed(m):
    m = int(m)
    p = 1
    while p < m:
        p <<= 1
    return p


def normalize_nonnegative(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        a = np.ones(a.size, dtype=np.float64)
        return a / float(a.size)
    return a / s


def fft_conv_square(a, m=None):
    n = a.size
    need = 2 * n - 1
    if m is None:
        m = fft_len_needed(need)
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    if c.size > need:
        c = c[:need]
    return np.maximum(c, 0.0)


def active_gradient_by_tau(a, c, tau, k_count):
    n = a.size
    M = float(c.max())
    if M <= 0.0 or not np.isfinite(M):
        return np.zeros(n, dtype=np.float64)

    thresh = M - tau
    idx = np.where(c >= thresh)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))], dtype=int)

    if idx.size > k_count:
        if k_count <= 1:
            idx = np.array([int(np.argmax(c))], dtype=int)
        else:
            idx = np.linspace(idx[0], idx[-1], int(k_count), dtype=int)

    g = np.zeros(n, dtype=np.float64)
    for kk in idx:
        k = int(kk)
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        if hi >= lo:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]

    denom = float(idx.size)
    if denom > 0.0:
        g /= denom
    return g


def mirror_update(a, g, step):
    gm = float(np.max(g))
    if gm <= 1e-300 or not np.isfinite(gm):
        return a.copy()

    if step > 50.0:
        step = 50.0

    u = g / gm
    ex = np.exp(-step * u)
    out = a * ex
    out = np.maximum(out, 0.0)

    floor = 1e-14 / float(out.size)
    out = np.maximum(out, floor)

    s = float(np.sum(out))
    if not np.isfinite(s) or s <= 1e-300:
        return a.copy()
    return out / s


def make_modulated(n, alpha, right_alpha, r, phase, freq):
    x = np.arange(n, dtype=np.float64)
    base = (x + 1.0) ** (-alpha)
    if right_alpha > 0.0:
        right = n - x
        right = np.maximum(right, 1.0)
        base = base * (right ** (-right_alpha))

    t = x / float(n)
    mod = 1.0 + r * np.sin(2.0 * np.pi * float(freq) * t + phase)
    base = base * np.maximum(mod, 0.0)
    return normalize_nonnegative(base)


def make_structured_candidate(n, rng):
    alpha = float(rng.uniform(0.35, 0.65))
    right_alpha = 0.0
    if float(rng.random()) < 0.4:
        right_alpha = float(rng.uniform(0.0, 0.4))

    x = np.arange(n, dtype=np.float64)
    base = (x + 1.0) ** (-alpha)
    if right_alpha > 0.0:
        right = np.maximum(n - x, 1.0)
        base = base * (right ** (-right_alpha))

    t = x / float(n)

    r = float(rng.uniform(0.0, 1.0))
    phase = float(rng.uniform(0.0, 2.0 * np.pi))
    freq = int(rng.integers(1, 4))
    mod = 1.0 + r * np.sin(2.0 * np.pi * float(freq) * t + phase)
    base = base * np.maximum(mod, 0.0)

    for _ in range(2):
        sigma = float(rng.uniform(0.05, 0.30))
        freq2 = int(rng.integers(1, 8))
        phase2 = float(rng.uniform(0.0, 2.0 * np.pi))
        base = base * np.exp(sigma * np.sin(2.0 * np.pi * float(freq2) * t + phase2))

    return normalize_nonnegative(base)


def add_pool_score(pool, a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0 or a.size > 16000:
        return

    a = normalize_nonnegative(a)
    c = fft_conv_square(a, None)
    s = 2.0 * float(a.size) * float(c.max())
    if np.isfinite(s):
        pool.append((s, a))


def generate_pool(lengths, rng, max_seconds):
    pool = []
    start = time.monotonic()

    for n in lengths:
        for alpha in (0.40, 0.45, 0.50, 0.55, 0.60):
            x = np.arange(n, dtype=np.float64)
            a = (x + 1.0) ** (-alpha)
            add_pool_score(pool, a)

        for ra in (0.10, 0.25, 0.40):
            x = np.arange(n, dtype=np.float64)
            a = (x + 1.0) ** (-0.5)
            right = np.maximum(n - x, 1.0)
            a = a * (right ** (-ra))
            add_pool_score(pool, a)

        for r in (0.4, 0.7, 1.0):
            for freq in (1, 2):
                for phase in (0.0, 0.5 * np.pi, float(np.pi), 1.5 * np.pi):
                    a = make_modulated(n, 0.5, 0.0, r, phase, freq)
                    add_pool_score(pool, a)

    for n in lengths:
        if time.monotonic() - start > max_seconds * 0.7:
            break

        if n <= 1024:
            count = 16
        elif n <= 2048:
            count = 12
        elif n <= 4096:
            count = 8
        else:
            count = 5

        for _ in range(count):
            if time.monotonic() - start > max_seconds:
                break
            a = make_structured_candidate(n, rng)
            add_pool_score(pool, a)

    try:
        h = height_sequence_1
        if h is not None:
            arr = np.asarray(h, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and arr.size <= 16000:
                arr = normalize_nonnegative(arr)
                add_pool_score(pool, arr)

                n = arr.size
                x = np.arange(n, dtype=np.float64)
                env = (x + 1.0) ** (-0.5)
                add_pool_score(pool, arr * env)

                t = x / float(n)
                for r in (0.3, 0.6, 0.9):
                    for phase in (0.0, 0.5 * np.pi, float(np.pi)):
                        mod = 1.0 + r * np.sin(2.0 * np.pi * t + phase)
                        add_pool_score(pool, arr * np.maximum(mod, 0.0))
                        add_pool_score(pool, arr * env * np.maximum(mod, 0.0))
    except Exception:
        pass

    return pool


def pair_score(p):
    return p[0]


def is_already_selected(selected, p):
    target_id = id(p[1])
    for q in selected:
        if id(q[1]) == target_id:
            return True
    return False


def select_candidates(pool, lengths, max_candidates):
    selected = []
    best_by_n = {}

    for p in pool:
        n = p[1].size
        if n not in best_by_n:
            best_by_n[n] = p
        else:
            if p[0] < best_by_n[n][0]:
                best_by_n[n] = p

    for n in lengths:
        if n in best_by_n and len(selected) < max_candidates:
            p = best_by_n[n]
            if not is_already_selected(selected, p):
                selected.append(p)

    for p in pool:
        if len(selected) >= max_candidates:
            break
        if not is_already_selected(selected, p):
            selected.append(p)

    return selected


def local_optimize(a0, max_seconds, rng):
    a = normalize_nonnegative(a0)
    n = a.size
    if n == 0:
        a = np.ones(1, dtype=np.float64)
        n = 1

    m_conv = fft_len_needed(2 * n - 1)
    c = fft_conv_square(a, m_conv)
    score = 2.0 * float(n) * float(c.max())

    best = a.copy()
    best_score = score

    if n <= 16:
        k_active = max(1, int(n))
    else:
        k_active = int(min(128, max(16, int(n // 64))))

    base_step = 0.25
    tau_frac = 0.3
    no_improve = 0

    start = time.monotonic()
    if max_seconds <= 0.0:
        return best, best_score

    while time.monotonic() - start < max_seconds:
        M = float(c.max())
        score = 2.0 * float(n) * M
        if not np.isfinite(score):
            break

        if score < best_score - 1e-13:
            best = a.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        tau = tau_frac * M
        if not np.isfinite(tau) or tau <= 1e-18:
            tau = 1e-18

        g = active_gradient_by_tau(a, c, tau, k_active)

        accepted = False
        step = base_step
        for attempt in range(5):
            a_new = mirror_update(a, g, step)
            c_new = fft_conv_square(a_new, m_conv)
            score_new = 2.0 * float(n) * float(c_new.max())

            if np.isfinite(score_new) and score_new < score - 1e-13:
                a = a_new
                c = c_new
                score = score_new

                if score < best_score - 1e-13:
                    best = a.copy()
                    best_score = score

                no_improve = 0
                accepted = True

                if attempt == 0:
                    base_step = min(1.5, base_step * 1.05)
                elif attempt == 1:
                    base_step = max(0.05, base_step * 0.95)
                else:
                    base_step = max(0.03, base_step * 0.85)
                break

            step *= 0.5

        if not accepted:
            no_improve += 1
            if no_improve < 20:
                base_step = max(0.02, base_step * 0.85)
            elif no_improve < 80:
                base_step = min(1.0, base_step * 1.2)
                tau_frac = float(rng.uniform(0.05, 1.0))
            else:
                a = best.copy()
                sig = 0.05 + 0.30 * min(1.0, float(no_improve - 80) / 1000.0)
                a = a * np.exp(rng.normal(0.0, sig, n))
                a = normalize_nonnegative(a)
                c = fft_conv_square(a, m_conv)
                score = 2.0 * float(n) * float(c.max())
                base_step = 0.25
                tau_frac = float(rng.uniform(0.1, 0.8))
                no_improve = 0

    a = best
    a = normalize_nonnegative(a)
    return a, best_score


def add_final_candidate(candidates, arr):
    target_id = id(arr)
    for c in candidates:
        if id(c) == target_id:
            return
    candidates.append(arr.copy())
    if len(candidates) > 6:
        candidates.pop(0)


def propose_candidate(seed=42, budget_s=1500.0, **kwargs):
    rng = np.random.default_rng(seed)
    start_all = time.monotonic()
    budget = float(budget_s)
    deadline = start_all + max(1.0, budget - 10.0)

    if budget < 120.0:
        lengths = [512, 1024]
    elif budget < 400.0:
        lengths = [512, 1024, 2048]
    elif budget < 900.0:
        lengths = [512, 1024, 2048, 4096]
    else:
        lengths = [512, 1024, 2048, 4096, 8192]

    gen_budget = min(45.0, max(8.0, budget * 0.04))
    pool = generate_pool(lengths, rng, gen_budget)

    if len(pool) == 0:
        a = make_structured_candidate(1024, rng)
        c = fft_conv_square(a)
        pool = [(2.0 * float(a.size) * float(c.max()), a)]

    pool.sort(key=pair_score)

    global_best_score = pool[0][0]
    global_best_a = pool[0][1].copy()
    final_candidates = [global_best_a.copy()]

    print("initial internal best score=", global_best_score, "n=", global_best_a.size)

    max_selected = 6 if budget < 900.0 else 8
    selected = select_candidates(pool, lengths, max_selected)

    for idx in range(len(selected)):
        now = time.monotonic()
        remaining = deadline - now - 10.0
        if remaining <= 30.0:
            break

        p = selected[idx]
        chunks = max(1, len(selected) - idx)
        chunk = remaining / float(chunks)
        chunk = min(120.0, max(20.0, chunk))
        if remaining < 60.0:
            chunk = min(chunk, remaining * 0.8)

        a2, s2 = local_optimize(p[1], chunk, rng)
        if s2 < global_best_score:
            global_best_score = s2
            global_best_a = a2.copy()
            add_final_candidate(final_candidates, a2)
            print("local selected improved:", idx, s2, "n=", a2.size)

    while True:
        now = time.monotonic()
        remaining = deadline - now - 10.0
        if remaining <= 40.0:
            break

        chunk = min(60.0, max(20.0, remaining * 0.3))
        a2, s2 = local_optimize(global_best_a, chunk, rng)
        if s2 < global_best_score:
            global_best_score = s2
            global_best_a = a2.copy()
            add_final_candidate(final_candidates, a2)
            print("polish improved:", s2, "n=", a2.size)

        now = time.monotonic()
        remaining = deadline - now - 10.0
        if remaining <= 40.0:
            break

        if float(rng.random()) < 0.5:
            a = np.array(global_best_a, copy=True)
            sig = float(rng.uniform(0.10, 0.35))
            a = a * np.exp(rng.normal(0.0, sig, a.size))
            a = normalize_nonnegative(a)
            c = fft_conv_square(a)
            cand_score = 2.0 * float(a.size) * float(c.max())
        else:
            if float(rng.random()) < 0.6 or int(global_best_a.size) in lengths:
                n_choice = int(global_best_a.size)
            else:
                n_choice = int(rng.choice(lengths))
            a = make_structured_candidate(n_choice, rng)
            c = fft_conv_square(a)
            cand_score = 2.0 * float(a.size) * float(c.max())

        if np.isfinite(cand_score) and cand_score < global_best_score + 0.05:
            chunk = min(45.0, max(15.0, (deadline - time.monotonic() - 10.0) * 0.3))
            a2, s2 = local_optimize(a, chunk, rng)
            if s2 < global_best_score:
                global_best_score = s2
                global_best_a = a2.copy()
                add_final_candidate(final_candidates, a2)
                print("exploration improved:", s2, "n=", a2.size)

    best_seq = None
    best_val = np.inf

    for arr in final_candidates:
        arr = normalize_nonnegative(arr)
        if arr.size == 0:
            arr = np.ones(1, dtype=np.float64)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(np.sum(arr))
        if s <= 0.0:
            arr = np.ones(arr.size, dtype=np.float64) / float(arr.size)
        else:
            arr = arr / s

        seq = arr.tolist()
        try:
            val = evaluate_sequence(seq)
        except Exception:
            val = np.inf

        if np.isfinite(val) and val < best_val:
            best_val = val
            best_seq = seq

    if best_seq is None:
        a = make_structured_candidate(1024, rng)
        best_seq = a.tolist()

    print("final evaluated score=", best_val)
    return best_seq
```
