def _compute_convolution(a):
    """Compute full convolution of a with itself using FFT for efficiency."""
    n = len(a)
    size = 2 * n - 1
    # Use FFT for O(n log n) convolution
    fft_size = 1 << (size - 1).bit_length()  # Next power of 2
    fa = np.fft.rfft(a, fft_size)
    fa2 = fa * fa
    b = np.fft.irfft(fa2, fft_size)[:size]
    return b

    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):

    A_rows = []
    A_cols = []

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _convolve_fft(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(fa * fa, fft_size)[:size]
    return b

def _lse_objective_and_grad(a, tau):
    """Compute LSE of convolution and its gradient."""
    n = len(a)
    b = _convolve_fft(a)
    b_max = b.max()
    b_shifted = tau * (b - b_max)
    exp_b = np.exp(b_shifted)
    lse_val = b_max + np.log(exp_b.sum()) / tau
    w = exp_b / exp_b.sum()
    # Gradient: 2 * correlate(w, a, 'full')[:n]
    grad = 2.0 * np.correlate(w, a, 'full')[:n]
    return lse_val, grad

def _solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    num_con = 2 * n - 1
    
    # Build sparse constraint matrix
    A_data = []
    A_row = []
    A_col = []
    
    for k in range(num_con):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A_data.append(a[k - j])
            A_row.append(k)
            A_col.append(j)
    
    A = csr_matrix((A_data, (A_row, A_col)), shape=(num_con, n))
    
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(num_con, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.status == 0 and result.fun is not None:
        g = result.x
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None
