```python
def compute_pressure(f, peak_idx):
    """Compute the pressure vector c[j] = f[j]*f[peak_idx-j] for valid j."""

    if j_min <= j_max:

            t = min(0.3, t * 1.01)
            if iteration % 50 == 0:
                print(f"  P1 iter {iteration}: score={best_score:.8f}, t={t:.4f}", flush=True)
        else:
            stall_count += 1
            t = max(0.03, t * 0.97)
    
    print(f"Phase 1 done after {iteration} iters, score={best_score:.8f}", flush=True)
    
    # === Phase 2: Diverse local search ===
    f = best.copy()
    local_improve_count = 0
    
    while time.monotonic() < deadline:
        candidate = make_local_perturbation(f, rng, n)
        
        if candidate.sum() < 0.01:
            continue
        
        score = float(evaluate_sequence(candidate.tolist()))
        
        if np.isfinite(score) and score < best_score:
            best = candidate.copy()
            f = candidate.copy()
            best_score = score
            local_improve_count += 1
            if local_improve_count % 20 == 1:
                print(f"  P2 improve #{local_improve_count}: score={best_score:.8f}", flush=True)
    
    print(f"Final: n={n}, score={best_score:.8f}", flush=True)
    return [float(x) for x in best]
```
