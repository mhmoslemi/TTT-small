from scipy.signal import fftconvolve

def _fast_score(a):
    n = len(a)
    s = np.sum(a)

    return 2.0 * n * np.max(b) / (s * s)

    """Subgradient using all peaks above (1-frac)*max."""
    n = len(a)
    threshold = np.max(b) * (1.0 - frac)
    active = np.where(b >= threshold)[0]

        k = int(k)
        idx = k - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g[valid] += 2.0 * a[idx[valid]]
    return g

def _fast_score(a):
    """Compute score without calling evaluate_sequence."""
    n = len(a)

        return np.inf

    return 2.0 * n * np.max(b) / (s * s)


def _subgrad_top(b, a, top_k=5):
    """Compute subgradient from top-k peaks of b."""
    n = len(a)
    idx = np.argpartition(b, -top_k)[-top_k:]
    g = np.zeros(n)
    ar = np.arange(n)
    for k in idx:
        k = int(k)
        j = k - ar
        valid = (j >= 0) & (j < n)
        g[valid] += 2.0 * a[j[valid]]
    return g
