"""
Flattening search for low-peak self-convolution sequences.

Objective (as in evaluator): F(a) = 2*n*max(conv(a,a)) / sum(a)^2  (scale invariant).
We initialise with non-uniform tapered windows (Hann / raised-power / trapezoid /
resampled parent) and perform multiplicative sub-gradient descents on the L_inf
of the self-convolution. At each step the "pressure" on element i is the weighted
amount of convolution energy it feeds, p_i = sum_k W_k a_{k-i} with peak-emphasised
weights W_k ∝ c_k^q; we update a_i *= exp(-eta * p_i) and rescale (scale is
invariant). Several (length, initialiser, hyperparam) tracks run in parallel and
the global best is kept. A short optional SDP/... LP refinement (max sum(g) s.t.
conv(a,g)<=M, g>=0) is applied to the best track if time and size allow.
"""

import time

def _local_objective(a):
    """Mirror of evaluate_sequence for in-range arrays (fast, no list cast)."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    mc = float(np.max(np.convolve(a, a, mode="full")))
    return float(2.0 * n * mc / (s * s))


def _hann(n):
    if n == 1:
        return np.ones(1)
    t = np.arange(n) / (n - 1)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * t))


def _power_sine(n, p):
    if n == 1:
        return np.ones(1)
    t = np.arange(n) / (n - 1)
    return np.sin(np.pi * t) ** p


def _trap(n, flat_frac):
    if n == 1:
        return np.ones(1)
    half = max(1, int(flat_frac * n // 2))
    a = np.ones(n)
    m = n - half
    if m > 1:
        edge = np.linspace(0.0, 1.0, m)
        a[:m] = edge
        a[n - m:] = edge[::-1]
    return a


def _smooth_random(n, rng, win):
    base = rng.random(n)
    k = max(1, win)
    ker = np.exp(-0.5 * (np.arange(-k, k + 1) / max(1, k // 4)) ** 2)
    ker /= ker.sum()
    base = np.convolve(base, ker, mode="same")
    base = np.clip(base, 0.0, None)
    if base.max() > 0:
        base /= base.max()
    if base.sum() < 0.01:
        base = _hann(n)
    return base


def _resample(x, n, rng):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return _hann(n)
    x = np.clip(x, 0.0, 1000.0)
    if x.size == n:
        return x.copy()
    old = np.linspace(0.0, 1.0, x.size)
    new = np.linspace(0.0, 1.0, n)
    y = np.interp(new, old, np.maximum(x, 0.0))
    y = np.clip(y, 0.0, None)
    if y.max() > 0:
        y /= y.max()
    if y.sum() < 0.01:
        y = _hann(n)
    return y


def _pressure(a, c, q):
    """p_i = sum_k W_k a_{k-i},  W ∝ c^q,  returned normalised to max 1."""
    n = a.size
    W = np.maximum(c, 0.0) ** q
    ws = W.sum()
    if ws <= 0.0:
        W = np.ones_like(c)
    else:
        W /= ws
    p = np.convolve(W, a[::-1], mode="full")[n - 1: n - 1 + n]
    pm = float(p.max())
    if pm <= 0.0:
        return np.zeros_like(a)
    return p / pm


def _step(a, c, q, eta):
    p = _pressure(a, c, q)
    a = a * np.exp(-eta * p)
    am = float(a.max())
    if am <= 0.0 or not np.isfinite(am):
        return np.ones_like(a)
    return a / am


def _build_conv_matrix(a):
    """Sparse A with (A g)_k = (a*g)_k  (linear convolution, full)."""
    from scipy.sparse import diags
    n = a.size
    offs = list(range(n))
    data = []
    for d in offs:
        data.append(np.full(n - d, a[d]))
    return diags(data, offsets=offs, shape=(2 * n - 1, n), format="csr")


def _lp_refine(a, max_solves, budget_from, deadline):
    """Refine a by alternating with LP direction max sum(g) s.t. conv(a,g)<=M."""
    from scipy.optimize import linprog
    n = a.size
    a = a.copy()
    s = float(a.sum())
    best = a.copy()
    best_obj = _local_objective(a)
    for _ in range(max_solves):
        if time.monotonic() >= deadline:
            break
        c = np.convolve(a, a, mode="full")
        M = float(c.max())
        A = _build_conv_matrix(a)
        res = linprog(
            -np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
            bounds=[(0.0, None)] * n, method="highs",
            options={"presolve": True},
        )
        if not res.success or res.x is None:
            break
        g = np.clip(res.x, 0.0, None)
        gs = float(g.sum())
        if gs <= 0.0:
            break
        g = g * (s / gs)
        for t in (0.5, 0.3, 0.15):
            cand = (1 - t) * a + t * g
            cand = np.clip(cand, 0.0, 1000.0)
            if cand.sum() < 0.01:
                continue
            obj = _local_objective(cand)
            if np.isfinite(obj) and obj < best_obj:
                best_obj = obj
                best = cand
                a = cand
                break
    return best, best_obj


def _run_track(n, init, q, eta, etamin, rng, slice_deadline):
    a = np.asarray(init, dtype=np.float64)
    if a.size != n:
        a = _resample(a, n, rng)
    if a.max() > 0:
        a = a / a.max()
    obj = _local_objective(a)
    best = a.copy()
    best_obj = obj
    nsteps = 0
    cur_eta = eta
    while time.monotonic() < slice_deadline:
        c = np.convolve(a, a, mode="full")
        # track improvement
        s = float(a.sum())
        loc = float(2.0 * n * c.max() / (s * s)) if s >= 0.01 else np.inf
        if np.isfinite(loc) and loc < best_obj:
            best_obj = loc
            best = a.copy()
        a = _step(a, c, q, cur_eta)
        nsteps += 1
        if nsteps % 200 == 0 and cur_eta > etamin:
            cur_eta = max(etamin, cur_eta * 0.85)
    return best, best_obj, nsteps


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Driver: parallel tapered-window + multiplicative-flattening search."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s) - 4.0)

    # optional parent seed
    parent = None
    try:
        p = globals().get("height_sequence_1", None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64)
            if p.ndim == 1 and p.size > 1 and np.all(np.isfinite(p)):
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        parent = None

    # ---- build track pool (length, initialiser, q, eta, etamin) ----------
    pool = []
    lengths = [1024, 2048, 3072, 4096, 5120, 6144]
    inits_makers = [
        ("hann", lambda n: _hann(n)),
        ("psin1", lambda n: _power_sine(n, 1.0)),
        ("psin2", lambda n: _power_sine(n, 2.0)),
        ("psin4", lambda n: _power_sine(n, 4.0)),
        ("trap50", lambda n: _trap(n, 0.5)),
        ("trap30", lambda n: _trap(n, 0.3)),
    ]
    if parent is not None:
        inits_makers.append(("parent", lambda n: _resample(parent, n, rng)))
    for n in lengths:
        for name, mk in inits_makers:
            q = float(rng.choice([1.0, 1.5, 2.0, 3.0]))
            eta = float(rng.uniform(0.03, 0.10))
            pool.append((n, mk(n), q, eta, max(0.01, eta * 0.2)))

    # ---- phase 1: parallel multiplicative search ------------------------
    global_best = None
    global_obj = np.inf
    n_tracks = len(pool)
    phase1_deadline = t0 + 0.70 * max(10.0, float(budget_s) - 4.0)
    per = (phase1_deadline - t0) / max(1, n_tracks)
    for idx, (n, init, q, eta, etamin) in enumerate(pool):
        entry_dead = min(phase1_deadline, t0 + per * (idx + 1))
        if entry_dead <= time.monotonic() + 1.0:
            continue
        a, obj, ns = _run_track(n, init, q, eta, etamin, rng, entry_dead)
        if np.isfinite(obj) and obj < global_obj:
            global_obj = obj
            global_best = (a, n, q, eta, etamin)
            print("new best  obj=%.6f  n=%d  q=%.2f  eta=%.3f  steps=%d  t=%.0fs"
                  % (global_obj, n, q, eta, ns, time.monotonic() - t0))
        if time.monotonic() > phase1_deadline:
            break

    if global_best is None:
        a = _hann(1024)
        global_obj = _local_objective(a)
        global_best = (a, a.size, 2.0, 0.05, 0.01)

    # ---- phase 2: grind the best track longer ---------------------------
    a, n, q, eta, etamin = global_best
    a, obj, ns = _run_track(n, a, q, min(eta, 0.06), etamin, rng,
                            t0 + 0.88 * max(10.0, float(budget_s) - 4.0))
    if np.isfinite(obj) and obj < global_obj:
        global_obj = obj
        global_best = (a, n, q, eta, etamin)
        print("grind best  obj=%.6f  n=%d  t=%.0fs"
              % (global_obj, n, time.monotonic() - t0))

    # ---- phase 3: LP refinement if size & time allow --------------------
    a, n, q, eta, etamin = global_best
    if n <= 4096 and time.monotonic() < deadline - 30.0:
        try:
            b, bobj = _lp_refine(a, max_solves=4,
                                 budget_from=time.monotonic(), deadline=deadline)
            if np.isfinite(bobj) and bobj < global_obj:
                global_obj = bobj
                global_best = (b, n, q, eta, etamin)
                print("LP best     obj=%.6f  n=%d  t=%.0fs"
                      % (global_obj, n, time.monotonic() - t0))
        except Exception as e:
            print("LP refine skipped:", e)

    a, n, q, eta, etamin = global_best
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    final = _local_objective(a)
    # hard confirm with the official evaluator for the returned sequence
    try:
        off = float(evaluate_sequence([float(x) for x in a]))
        if np.isfinite(off):
            final = min(final, off)
            # if official is worse (clipping/rounding), keep local-best anyway
    except Exception:
        pass

    print("FINAL   obj=%.6f  n=%d  t=%.0fs" % (final, n, time.monotonic() - t0))
    return [float(x) for x in a]
