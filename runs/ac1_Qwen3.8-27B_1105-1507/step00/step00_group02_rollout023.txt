```python
import numpy as np
import time

def _compute_autocorr(p):
    """Compute full autocorrelation of p (length 2n-1)."""
    n = len(p)
    # Use FFT for efficiency
    L = 2 * (2 * n - 1)
    P = np.fft.rfft(p, L)
    return np.fft.irfft(P * P, L)[:2*n-1]

def _softmax_smooth_max(c, T):
    """Compute softmax weights for smooth max approximation."""
    # c has length 2n-1
    shifted = c / T - np.max(c) / T  # for numerical stability
    w = np.exp(shifted)
    w /= w.sum()
    return w

def _compute_grad_fft(p, w):
    """Compute gradient of smooth max using FFT."""
    n = len(p)
    L = 2 * (2 * n - 1)
    P = np.fft.rfft(p, L)
    W = np.fft.rfft(w, L)
    # grad[i] = sum_k w[k] * 2 * p[k-i] (for valid k-i)
    # = 2 * correlation(w, p)[i]
    # correlation(w, p)[i] = sum_k w[k] * p[k-i]
    # = ifft(FFT(w) * conj(FFT(p)))
    grad = 2.0 * np.fft.irfft(W * np.conj(P), L)[:n]
    return grad

def _project_simplex(v):
    """Project v onto the probability simplex {p >= 0, sum p = 1}."""
    n = len(v)
    # Simple projection algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _gradient_descent(p, T_list, steps_per_T, deadline):
    """Run gradient descent with annealing schedule."""
    n = len(p)
    best_p = p.copy()
    best_score = np.inf
    
    for T in T_list:
        for step in range(steps_per_T):
            if time.monotonic() > deadline:
                break
            
            c = _compute_autocorr(p)
            w = _softmax_smooth_max(c, T)
            grad = _compute_grad_fft(p, w)
            
            # Project gradient onto tangent space of simplex
            grad = grad - np.mean(grad)
            
            # Step size
            alpha = 1.0 / (2.0 * np.max(w) * np.max(p) + 1e-10)
            
            # Update
            p_new = p - alpha * grad
            p_new = _project_simplex(p_new)
            
            # Check score
            score = 2 * n * np.max(_compute_autocorr(p_new))
            if score < best_score:
                best_score = score
                best_p = p_new.copy()
            
            p = p_new
        
        if time.monotonic() > deadline:
            break
    
    return best_p, best_score
```
