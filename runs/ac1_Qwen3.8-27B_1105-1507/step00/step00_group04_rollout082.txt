def gs_step(a, N_fft, sigma, target_power=None):
    ap=np.zeros(N_fft); ap[:n]=a
    F=np.fft.rfft(ap)
    mag=np.abs(F)
    # target maybe geometric mean over all bins or windows? Keep DC.
    # To minimize max time domain, target flat amplitude equals sqrt(total power / number bins)? Since F[0]=sum a=(for zero pad same) magnitude; if force all bins same as DC, time domain delta impossible. Better use geometric mean to not change energy too much.
    if target_power is None:
       non_dc=mag[1:]
       # use mean of mag? geometric if positive
       target=np.exp(np.mean(np.log(non_dc+1e-300)))
    new_mag = np.empty_like(mag)
    new_mag[0]=mag[0]
    # Mix in linear log: log_new=(1-sigma)*log mag + sigma*log target
    log_mag=np.log(np.maximum(mag[1:],1e-300))
    log_target=np.log(max(target,1e-300))
    new_mag[1:]=np.exp((1-sigma)*log_mag + sigma*log_target)
    phase=np.angle(F)
    F2=new_mag*np.exp(1j*phase)
    # ensure Hermitian? rfft handles.
    aN=np.fft.irfft(F2, N_fft)[:n]
    aN=np.clip(aN,0,None)
    s=aN.sum()
    if s<=0: return a.copy()
    aN/=s
    return aN

"""
Search for a non-negative step function minimizing
J = 2*n*max(convolve(a,a))/(sum(a))^2.

The method combines:
- deterministic shape scanning (U-shaped / end-heavy / cosine / random),
- Gerchberg-Saxton spectral flattening,
- mirrored subgradient descent with FFT-accelerated gradients,
- and smooth-max L-BFGS-B optimization in log-softmax coordinates.
"""

import numpy as np
import time

try:
    import scipy.optimize as _scipy_opt
except Exception:
    _scipy_opt = None

_CAND_ID = [0]


def _clean_sequence(a, normalize=True):
    try:
        arr = np.asarray(a, dtype=np.float64).reshape(-1)
    except Exception:
        return np.ones(1, dtype=np.float64)
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if not normalize:
        return arr
    s = float(arr.sum())
    if s <= 1e-300:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def _fast_J(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(a.size)
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if s <= 1e-12:
        return float("inf")
    c = np.convolve(a, a)
    mc = float(c.max())
    return float(2.0 * n * mc / (s * s))


def _fft_corr(a, w, N):
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    fa = np.fft.rfft(a, int(N))
    fw = np.fft.rfft(w, int(N))
    g = np.fft.irfft(fa * np.conj(fw), int(N))
    g = g[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    g[g < 0.0] = 0.0
    return g


def _softmax(y, floor=1e-300):
    y = np.asarray(y, dtype=np.float64).reshape(-1)
    if y.size == 0:
        return np.ones(1, dtype=np.float64)
    m = float(np.max(y))
    if not np.isfinite(m):
        return _clean_sequence(np.ones(y.size))
    z = y - m
    a = np.exp(z)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, floor)
    s = float(a.sum())
    if s <= 1e-300:
        return _clean_sequence(np.ones(y.size))
    return a / s


def _edge_sequence(n, gamma=5.0, power=1.0, beta=0.0, baseline=0.2, noise=0.0, rng=None):
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    idx = np.arange(int(n), dtype=np.float64)
    if n > 1:
        edge = 2.0 * np.minimum(idx, float(n - 1) - idx) / float(n - 1)
    else:
        edge = np.zeros(1, dtype=np.float64)
    a = np.full(n, float(baseline), dtype=np.float64)
    if gamma != 0.0:
        a += 1.0 / (1.0 + float(gamma) * (edge ** float(power)))
    if beta != 0.0:
        a += np.exp(float(beta) * (1.0 - edge))
    a = np.maximum(a, 0.0)
    if noise > 0.0 and rng is not None:
        a = a * np.exp(np.clip(rng.normal(0.0, float(noise), size=n), -2.0, 2.0))
    return _clean_sequence(a)


def _uniform_noise(n, scale, rng):
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    a = np.ones(int(n), dtype=np.float64)
    if scale > 0.0 and rng is not None:
        a = a * np.exp(np.clip(rng.normal(0.0, float(scale), size=n), -2.0, 2.0))
    return _clean_sequence(a)


def _downsample_sequence(seq, n):
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    if seq.size == 0:
        return np.ones(int(n), dtype=np.float64) / float(n)
    if seq.size == int(n):
        return _clean_sequence(seq)
    x = np.linspace(0.0, float(seq.size - 1), int(n))
    arr = np.interp(x, np.arange(seq.size, dtype=np.float64), seq)
    return _clean_sequence(arr)


def _smooth_score_grad(y, beta_coeff, N):
    a = _softmax(y)
    n = int(a.size)
    c = np.convolve(a, a)
    maxc = float(c.max())

    beta = float(beta_coeff) * float(max(1, n))
    if beta > 0.0 and maxc > 0.0:
        z = beta * (c - maxc)
        z = np.where(np.isfinite(z), z, -745.0)
        z = np.minimum(z, 0.0)
        z = np.maximum(z, -745.0)
        weights = np.exp(z)
        ws = float(weights.sum())
        if ws > 0.0 and np.isfinite(ws):
            weights = weights / ws
            val = float(2.0 * n * maxc + (2.0 * n * np.log(ws) / beta))
        else:
            weights = np.ones(c.size, dtype=np.float64) / float(c.size)
            val = float(2.0 * n * maxc)
    else:
        weights = np.ones(c.size, dtype=np.float64) / float(c.size)
        val = float(2.0 * n * maxc)

    corr = _fft_corr(a, weights, int(N))
    ga = 2.0 * corr
    s = float(np.dot(a, ga))
    gy = a * ga - a * s
    gy = (2.0 * float(n)) * gy
    gy = np.where(np.isfinite(gy), gy, 0.0)
    gy = np.clip(gy, -1e4, 1e4)
    return float(val), gy


def _run_lbfgs(a, beta_coeffs, time_limit, rng, label, max_n_iters=800):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(time_limit))
    a0 = _clean_sequence(a)
    n = int(a0.size)
    best_a = a0.copy()
    bestJ = _fast_J(a0)
    if n == 0 or _scipy_opt is None or deadline <= t0 + 3.0:
        return best_a, bestJ

    N = 1 << (3 * n - 2).bit_length()
    a_log = np.maximum(a0, 1e-12 / float(n))
    a_log = a_log / a_log.sum()
    y = np.log(a_log + 1e-15)
    lin = (np.arange(n, dtype=np.float64) - 0.5 * float(n)) / float(max(1, n))
    y = y + 1e-3 * lin
    if rng is not None:
        y = y + 1e-3 * rng.standard_normal(n)
    y = np.where(np.isfinite(y), y, 0.0)
    y = np.clip(y, -1000.0, 1000.0)

    beta_coeffs = tuple(beta_coeffs)
    if len(beta_coeffs) == 0:
        return best_a, bestJ

    base_time = max(0.0, deadline - t0) / float(len(beta_coeffs))

    if n < 1024:
        per_iter_est = 0.02
    elif n < 2048:
        per_iter_est = 0.05
    elif n < 8192:
        per_iter_est = 0.08
    else:
        per_iter_est = 0.16

    for bi, b in enumerate(beta_coeffs):
        if time.monotonic() >= deadline - 3.0:
            break
        remain = max(0.0, deadline - time.monotonic())
        spend = min(remain, max(3.0, base_time))
        if spend < 3.0:
            break
        maxiter = int(min(max_n_iters, max(10, (spend / per_iter_est) / 2.0)))
        try:
            res = _scipy_opt.minimize(
                _smooth_score_grad,
                y,
                method="L-BFGS-B",
                args=(float(b), int(N)),
                jac=True,
                options={
                    "maxiter": maxiter,
                    "maxfun": max(50, 2 * maxiter),
                    "ftol": 1e-12,
                    "gtol": 1e-10,
                    "maxcor": 20,
                    "maxls": 25,
                    "disp": 0,
                },
            )
            if res is not None and hasattr(res, "x") and res.x is not None:
                a_new = _softmax(res.x)
                J_new = _fast_J(a_new)
                if np.isfinite(J_new) and J_new < bestJ:
                    bestJ = float(J_new)
                    best_a = a_new.copy()
                if np.all(np.isfinite(res.x)):
                    y = res.x
        except Exception as e:
            print(label, "minimize error:", e, flush=True)

        y = y + 1e-5 * (np.arange(n, dtype=np.float64) - 0.5 * float(n))
        if rng is not None:
            y = y + 1e-4 * rng.standard_normal(n)
        y = np.where(np.isfinite(y), y, 0.0)
        y = np.clip(y, -1000.0, 1000.0)

        print(label, f"beta={b}", f"fast={bestJ:.6f}", f"wall={time.monotonic()-t0:.1f}s", flush=True)

    return best_a, bestJ


def _run_smooth_mirror(a, N, time_limit, temp_cycle, eta0, noise_prob, noise_scale, rng, label):

    a = _clean_sequence(a)
    n = int(a.size)
    best_a = a.copy()
    bestJ = _fast_J(a)
    temp_cycle = tuple(temp_cycle)
    if len(temp_cycle) == 0:
        temp_cycle = (0.2,)

    t = 0
    last_improve = t0
    last_log = t0

    while time.monotonic() < deadline - 0.05:
        c = np.convolve(a, a)
        maxc = float(c.max())
        J = float(2.0 * n * maxc)
        if np.isfinite(J) and J < bestJ:
            bestJ = J
            best_a = a.copy()
            last_improve = time.monotonic()

        if time.monotonic() - last_log > 8.0:
            print(label, "iter", t, f"best={bestJ:.6f}", f"cur={J:.6f}", flush=True)
            last_log = time.monotonic()

        if time.monotonic() - last_improve > 25.0:
            bias = _edge_sequence(n, gamma=5.0, power=1.0, beta=0.7, baseline=1.0, noise=0.0, rng=None)
            a = 0.8 * a + 0.2 * bias
            if rng is not None:
                a = a * np.exp(np.clip(rng.normal(0.0, 0.25, size=n), -2.0, 2.0))
            a = _clean_sequence(a)
            last_improve = time.monotonic()

        temp = float(temp_cycle[t % len(temp_cycle)])
        denom = maxc * temp
        if denom > 0.0 and np.isfinite(denom):
            z = (c - maxc) / denom
            z = np.where(np.isfinite(z), z, -745.0)
            z = np.minimum(z, 0.0)
            z = np.maximum(z, -745.0)
            w = np.exp(z)
            ws = float(w.sum())
            if ws > 0.0:
                w = w / ws
            else:
                w = np.ones(c.size, dtype=np.float64) / float(c.size)
        else:
            w = np.ones(c.size, dtype=np.float64) / float(c.size)

        step = _fft_corr(a, w, int(N))
        m = float(step.max())
        if m > 0.0:
            step = step / m
        else:
            step = np.zeros_like(a)
        step_range = float(step.max() - step.min()) if step.size else 0.0

        structured = step_range < 0.005
        if m == 0.0 or structured or (rng is not None and rng.random() < float(noise_prob)):
            if structured:
                bias = _edge_sequence(n, gamma=5.0, power=1.0, beta=0.7, baseline=1.0, noise=0.0, rng=None)
                a = 0.8 * a + 0.2 * bias

            if rng is not None:

            a = _clean_sequence(a)

        logmul = -eta * step
        logmul = np.where(np.isfinite(logmul), logmul, 0.0)
        logmul = np.clip(logmul, -50.0, 0.0)
        a2 = a * np.exp(logmul)
        a2 = np.maximum(a2, 1e-300)
        s = float(a2.sum())
        if s > 0.0:
            a = a2 / s
        else:
            a = _clean_sequence(a)

        t += 1

    return best_a, bestJ


def _run_gs(a, N, time_limit, sigma0, sigma1, mix, label, rng):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(time_limit))
    a = _clean_sequence(a)
    n = int(a.size)
    best_a = a.copy()
    bestJ = _fast_J(a)
    sigma = float(sigma0)
    last_improve = t0
    last_log = t0

    while time.monotonic() < deadline - 0.05:
        F = np.fft.rfft(a, int(N))
        mag = np.abs(F)
        if mag.size > 1:
            md = np.maximum(mag[1:], 1e-300)
            target = float(np.sqrt(np.mean(md * md)))
            if target <= 0.0 or not np.isfinite(target):
                target = 1e-300
            logmag = np.log(np.maximum(mag, 1e-300))
            logtarget = np.log(max(target, 1e-300))
            lognew = logmag.copy()
            lognew[1:] = (1.0 - sigma) * logmag[1:] + sigma * logtarget
            Fnew = np.exp(lognew) * np.exp(1j * np.angle(F))
            aN = np.fft.irfft(Fnew, int(N))
            a2 = np.clip(aN[:n], 0.0, None)
            a2 = np.where(np.isfinite(a2), a2, 0.0)
        else:
            a2 = a.copy()

        s = float(a2.sum())
        if s <= 1e-300:
            break
        a2 = a2 / s
        blend = (1.0 - float(mix)) * a + float(mix) * a2
        blend = _clean_sequence(blend)
        J = _fast_J(blend)
        if np.isfinite(J) and J < bestJ:
            bestJ = J
            best_a = blend.copy()
            last_improve = time.monotonic()
        a = blend

        if time.monotonic() - last_improve > 30.0:
            if rng is not None:
                a = best_a * np.exp(np.clip(rng.normal(0.0, 0.20, size=n), -2.0, 2.0))
            a = _clean_sequence(a)
            last_improve = time.monotonic()

        sigma = max(float(sigma1), sigma * 0.999)

        if time.monotonic() - last_log > 10.0:
            print(label, f"best={bestJ:.6f}", f"wall={time.monotonic()-t0:.1f}s", flush=True)
            last_log = time.monotonic()

    return best_a, bestJ


def _topk_update(cands, J, a, k=10):
    if not np.isfinite(J):
        return
    _CAND_ID[0] += 1
    cands.append([float(J), _CAND_ID[0], np.array(a, dtype=np.float64)])
    if len(cands) > k:
        worst = 0
        for i in range(1, len(cands)):
            if cands[i][0] > cands[worst][0]:
                worst = i
        del cands[worst]


def _scan_initial_shapes(n, parent, rng, max_time, k=10):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(max_time))
    if n <= 0:
        n = 128
    n = int(n)
    cands = []

    p = _downsample_sequence(parent, n)
    _topk_update(cands, _fast_J(p), p, k)
    _topk_update(cands, _fast_J(np.ones(n)), np.ones(n), k)

    gammas = (0.5, 1.0, 2.0, 5.0, 10.0, 25.0)
    powers = (1.0, 2.0)
    betas = (0.0, 1.0, 3.0)
    baselines = (0.0, 0.2, 1.0)

    for g in gammas:
        if time.monotonic() >= deadline - 1.0:
            break
        for pw in powers:
            if time.monotonic() >= deadline - 1.0:
                break
            for bt in betas:
                if time.monotonic() >= deadline - 1.0:
                    break
                for ba in baselines:
                    a = _edge_sequence(n, gamma=g, power=pw, beta=bt, baseline=ba, noise=0.0, rng=None)
                    _topk_update(cands, _fast_J(a), a, k)
                    if time.monotonic() >= deadline - 1.0:
                        break

    if n > 1:
        idx = np.arange(n, dtype=np.float64)
        phi = np.pi * idx / float(n - 1)
        for alpha in (0.2, 0.5, 0.8, 0.95):
            a = 1.0 + alpha * np.cos(phi)
            _topk_update(cands, _fast_J(_clean_sequence(a)), _clean_sequence(a), k)
            a = 1.0 - alpha * np.cos(phi)
            _topk_update(cands, _fast_J(_clean_sequence(a)), _clean_sequence(a), k)
            a = 1.0 + alpha * np.sin(phi)
            _topk_update(cands, _fast_J(_clean_sequence(a)), _clean_sequence(a), k)
            if time.monotonic() >= deadline - 1.0:
                break

    dist = np.minimum(np.arange(n, dtype=np.float64), float(n - 1) - np.arange(n, dtype=np.float64)) + 1.0
    for pw in (0.25, 0.5, 0.75, 1.0):
        for ba in (0.0, 0.3, 2.0):
            a = dist ** (-float(pw)) + float(ba)
            _topk_update(cands, _fast_J(_clean_sequence(a)), _clean_sequence(a), k)
            if time.monotonic() >= deadline - 1.0:
                break

    if rng is not None:
        for sc in (0.1, 0.3, 0.6):
            a = _uniform_noise(n, sc, rng)
            _topk_update(cands, _fast_J(a), a, k)
            if time.monotonic() >= deadline - 1.0:
                break

        for sc in (0.05, 0.15, 0.3):
            a = p * np.exp(np.clip(rng.normal(0.0, sc, size=n), -2.0, 2.0))
            a = _clean_sequence(a)
            _topk_update(cands, _fast_J(a), a, k)
            if time.monotonic() >= deadline - 1.0:
                break

    cands.sort()
    return [c[2] for c in cands]

        gs_limit = min(0.10 * float(time_limit), max(0.0, deadline - time.monotonic()))
        if gs_limit > 3.0:
            a_gs, J_gs = _run_gs(a0, Ngs, gs_limit, gs_sigma0, gs_sigma1, gs_mix, label + " GS", rng)
            if np.isfinite(J_gs) and J_gs < bestJ:
                bestJ = float(J_gs)
                best_a = a_gs.copy()

    current = best_a.copy()
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 5.0:
        if use_lbfgs and float(time_limit) > 45.0:
            reserve = min(0.30 * float(time_limit), remaining * 0.35)
            mirror_limit = min(0.55 * float(time_limit), max(0.0, remaining - reserve - 1.0))
        else:
            mirror_limit = max(0.0, remaining - 1.0)
        if mirror_limit > 2.0:
            a_m, J_m = _run_smooth_mirror(
                current, Ngrad, mirror_limit, temp_cycle, eta0, noise_prob, noise_scale, rng, label + " MD"
            )
            if np.isfinite(J_m) and J_m < bestJ:
                bestJ = float(J_m)
                best_a = a_m.copy()

    if use_lbfgs:
        remaining = max(0.0, deadline - time.monotonic())
        if remaining > 8.0:
            a_l, J_l = _run_lbfgs(best_a, beta_coeffs, remaining - 1.0, rng, label + " LBFGS", lbfgs_max_iters)
            if np.isfinite(J_l) and J_l < bestJ:
                bestJ = float(J_l)
                best_a = a_l.copy()

    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 8.0 and bestJ > 1.02:
        polish_limit = min(remaining - 1.0, max(3.0, 0.2 * float(time_limit)))
        if polish_limit > 3.0:
            reversed_temp = tuple(reversed(temp_cycle))
            a_p, J_p = _run_smooth_mirror(
                best_a, Ngrad, polish_limit, reversed_temp,
                max(0.2, float(eta0) * 0.4),
                float(noise_prob) * 0.5,
                float(noise_scale) * 0.5,
                rng,
                label + " polish"
            )
            if np.isfinite(J_p) and J_p < bestJ:
                bestJ = float(J_p)
                best_a = a_p.copy()

    return best_a, bestJ


def _get_parent():
    raw = None
    try:
        raw = height_sequence_1
    except Exception:
        raw = None
    try:
        arr = np.asarray(raw, dtype=np.float64).reshape(-1) if raw is not None else None
    except Exception:
        arr = None
    if arr is None or arr.size == 0:
        arr = np.ones(1024, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.size > 15000:
        arr = np.interp(np.linspace(0.0, float(arr.size - 1), 15000), np.arange(arr.size, dtype=np.float64), arr)
    if arr.size < 16:
        arr = np.interp(np.linspace(0.0, float(arr.size - 1), 16), np.arange(arr.size, dtype=np.float64), arr)
    return arr


def propose_candidate(seed=42, budget_s=1500.0, **kwargs):
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if budget <= 0.0:
        budget = 1.0

    t_start = time.monotonic()
    available = max(0.0, budget - 10.0)
    deadline = t_start + max(1.0, available)
    rng = np.random.default_rng(seed)

    parent = _get_parent()
    best_a = _clean_sequence(parent)
    best_fast = _fast_J(best_a)
    best_exact = _exact_J(best_a)

    if not np.isfinite(best_exact):
        best_a = _clean_sequence(np.ones(max(128, int(best_a.size)), dtype=np.float64))
        best_fast = _fast_J(best_a)
        best_exact = _exact_J(best_a)

    print("initial exact", f"{best_exact:.8f}", "n", int(best_a.size), flush=True)

    total_budget = max(1.0, budget - 15.0)
    plans = []
    scan_limit = min(35.0, max(0.0, (budget - 15.0) * 0.035))
    scan_cands = []
    if available > 25.0:
        try:
            scan_cands = _scan_initial_shapes(int(best_a.size), parent, rng, scan_limit, k=8)
        except Exception as e:
            print("scan error:", e, flush=True)
            scan_cands = []

    n_parent = int(best_a.size)

    if scan_cands:
        s0 = _clean_sequence(scan_cands[0])
        sJ = _fast_J(s0)
        s0_exact = _exact_J(s0)
        if np.isfinite(s0_exact) and s0_exact < best_exact:
            best_exact = float(s0_exact)
            best_a = s0.copy()
            best_fast = float(sJ)
            print("scan candidate exact", f"{best_exact:.8f}", "n", int(best_a.size), flush=True)
        if np.isfinite(sJ) and sJ < best_fast:
            plans.append(("scan0", s0, 0.22))
            plans.append(("parent", best_a, 0.26))
        else:
            plans.append(("parent", best_a, 0.30))
            plans.append(("scan0", s0, 0.10))
        if len(scan_cands) > 1:
            s1 = _clean_sequence(scan_cands[1])
            plans.append(("scan1", s1, 0.07))
    else:
        plans.append(("parent", best_a, 0.34))

    edge1 = _edge_sequence(n_parent, gamma=5.0, power=1.0, beta=1.0, baseline=0.2, noise=0.10, rng=rng)
    plans.append(("edge_parent", edge1, 0.09))

    if n_parent > 1200:
        n_mid = int(max(1024, min(2048, n_parent // 2)))
        mid_parent = _downsample_sequence(parent, n_mid)
        plans.append(("mid_parent", mid_parent, 0.08))
        mid_edge = _edge_sequence(n_mid, gamma=8.0, power=1.5, beta=2.0, baseline=0.2, noise=0.14, rng=rng)
        plans.append(("mid_edge", mid_edge, 0.08))
    else:
        n_mid = int(max(1024, n_parent))
        mid_parent = _downsample_sequence(parent, n_mid)
        plans.append(("mid_parent", mid_parent, 0.08))
        mid_edge = _edge_sequence(n_mid, gamma=8.0, power=1.5, beta=2.0, baseline=0.2, noise=0.14, rng=rng)
        plans.append(("mid_edge", mid_edge, 0.08))

    short_edge = _edge_sequence(1024, gamma=10.0, power=1.5, beta=2.5, baseline=0.2, noise=0.15, rng=rng)
    plans.append(("short_edge", short_edge, 0.08))

    short_uni = _uniform_noise(1024, 0.25, rng)
    plans.append(("short_uniform", short_uni, 0.07))

    allocated_frac = 0.0
    for label, cand, frac in plans:
        if time.monotonic() >= deadline - 30.0:
            break
        max_frac = max(0.0, 1.0 - allocated_frac - 0.10)
        f = min(float(frac), max_frac)
        if f <= 0.01:
            continue
        remaining = max(0.0, deadline - time.monotonic())
        alloc = min(f * total_budget, max(0.0, remaining - 15.0))
        if alloc < 20.0:
            continue
        allocated_frac += f

        cand = _clean_sequence(cand)
        cand_fast = _fast_J(cand)
        cand_exact = _exact_J(cand)
        if np.isfinite(cand_fast) and cand_fast < best_fast:
            best_fast = cand_fast
        if np.isfinite(cand_exact) and cand_exact < best_exact:
            best_exact = cand_exact
            best_a = cand.copy()
            print("candidate exact improved", label, f"{best_exact:.8f}", "n", int(best_a.size), flush=True)

        try:
            res_a, resJ = _run_pipeline(cand, alloc, label, rng)
        except Exception as e:
            print("pipeline error", label, e, flush=True)
            res_a, resJ = None, float("inf")

        if res_a is not None and np.isfinite(resJ) and resJ < best_fast:
            best_fast = float(resJ)
        if res_a is not None:
            res_exact = _exact_J(res_a)
            if np.isfinite(res_exact) and res_exact < best_exact:
                best_exact = float(res_exact)
                best_a = _clean_sequence(res_a)
                print("pipeline exact improved", label, f"{best_exact:.8f}", "n", int(best_a.size), flush=True)

    remaining = max(0.0, deadline - time.monotonic() - 2.0)
    if remaining > 30.0 and np.isfinite(best_exact) and best_exact > 1.01:
        polish_alloc = min(remaining, max(30.0, 0.15 * total_budget))
        try:
            res_a, resJ = _run_pipeline(
                best_a,
                polish_alloc,
                "final_polish",
                rng,
                temp_cycle=(0.1, 0.2, 0.05, 0.3, 0.5),
                eta0=0.5,
                noise_prob=0.05,
                noise_scale=0.10,
                use_gs=False,
                use_lbfgs=True,
                beta_coeffs=(10.0, 50.0, 300.0),
                lbfgs_max_iters=500,
            )
        except Exception as e:
            print("final polish error:", e, flush=True)
            res_a, resJ = None, float("inf")
        if res_a is not None:
            res_exact = _exact_J(res_a)
            if np.isfinite(res_exact) and res_exact < best_exact:
                best_exact = float(res_exact)
                best_a = _clean_sequence(res_a)
                print("final polish exact improved", f"{best_exact:.8f}", "n", int(best_a.size), flush=True)

    out = _clean_sequence(best_a)
    final_exact = _exact_J(out)
    if not np.isfinite(final_exact):
        out = _clean_sequence(np.ones(max(128, n_parent), dtype=np.float64))
        final_exact = _exact_J(out)

    print("final exact", f"{final_exact:.8f}", "n", int(out.size), flush=True)
    return [float(x) for x in out]
