from scipy.signal import fftconvolve
from scipy.optimize import minimize
import time

def softmax(x):
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    return exp_x / np.sum(exp_x)

def compute_c(a):
    """Compute full convolution of a with itself."""
    return np.convolve(a, a, mode='full')

def objective_and_gradient(x, alpha):
    a = softmax(x)
    c = compute_c(a)  # length 2n-1
    n = len(a)
    
    # Soft-max: (1/alpha) * log(sum exp(alpha * c))
    ac = alpha * c
    max_ac = np.max(ac)
    log_sum_exp = max_ac + np.log(np.sum(np.exp(ac - max_ac)))
    obj = log_sum_exp / alpha
    
    # p_k = exp(alpha * c_k) / sum(exp(alpha * c_k))
    p = np.exp(ac - max_ac)
    p /= np.sum(p)

    conv_pa = fftconvolve(p, a_flip)
    g = 2.0 * conv_pa[n-1 : n-1+n]
    
    const = 2.0 * np.dot(p, c)
    grad_x = a * (g - const)
    
    return obj, grad_x

Gradient-based optimization using softmax parameterization and soft-max objective

Uses continuation (increasing alpha) and multiple random restarts.

from scipy.signal import fftconvolve
from scipy.optimize import minimize
import time


def _softmax(x):
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    return exp_x / np.sum(exp_x)


def _objective_and_gradient(x, alpha, reg=0.0):
    a = _softmax(x)
    n = len(a)
    c = fftconvolve(a, a)  # length 2n-1
    
    ac = alpha * c
    max_ac = np.max(ac)
    exp_ac = np.exp(ac - max_ac)
    sum_exp = np.sum(exp_ac)
    log_sum_exp = max_ac + np.log(sum_exp)
    obj = log_sum_exp / alpha
    
    p = exp_ac / sum_exp
    
    # g_j = 2 * sum_k p_k * a_{k-j}

    grad_x = a * (g - const)
    
    if reg > 0:
        obj += reg * np.dot(x, x)
        grad_x += 2.0 * reg * x
    
    return obj, grad_x
