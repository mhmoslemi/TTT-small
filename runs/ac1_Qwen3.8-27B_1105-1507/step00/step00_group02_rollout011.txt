            if eta < 1e-6:

        final_score = evaluate_sequence(a.tolist())

            best_score = final_score
            best = a.copy()

    if best is None:

        best_score = evaluate_sequence(best.tolist())
    
    print(f"Final best score: {best_score:.6f}")
