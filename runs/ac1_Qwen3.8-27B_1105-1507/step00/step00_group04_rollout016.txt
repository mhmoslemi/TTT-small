def gradient_refine(a, deadline=None, max_time=120, c_init=0.1):
    """Refine using L-BFGS-B with smoothed objective."""
    n = len(a)
    
    start_time = time.monotonic()
    
    def obj_grad(x):
        x = np.clip(x, 0, None)
        # Smoothed objective with log-sum-exp
        conv = np.convolve(x, x)
        S = np.sum(x)
        if S < 1e-10:
            return 1e10, np.zeros_like(x)
        
        conv_max = np.max(conv)
        shifted = (conv - conv_max) / c_init
        exp_shifted = np.exp(shifted)
        sum_exp = np.sum(exp_shifted)
        lse = conv_max + c_init * np.log(sum_exp)
        weights = exp_shifted / sum_exp

        grad_lse = np.zeros(n)
        for i in range(n):
            grad_lse[i] = np.dot(weights[i:i+n], x)
        
        obj = 2.0 * n * lse / (S * S)
        grad = (2.0 * n / (S * S)) * grad_lse - (2.0 * n * lse / (S ** 3))
        
        return obj, grad

                      bounds=[(0, None)]*n,

                              'gtol': 1e-6, 'iprint': 0})

            if np.sum(result) > 1e-10:

        pass
