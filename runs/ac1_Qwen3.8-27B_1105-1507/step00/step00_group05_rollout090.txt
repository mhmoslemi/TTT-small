```
def project_simplex(v, s=1.0):
    v=np.asarray(v,float)
    n=v.size
    if n==0: return v
    if not np.all(np.isfinite(v)):
       v=np.where(np.isfinite(v), v, 0.0)
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    idx=np.nonzero(u * np.arange(1,n+1) > (cssv - s))[0]
    if idx.size == 0:
       return np.full(n, s/n)
    rho=idx[-1]
    theta=(cssv[rho]-s)/(rho+1.0)
    w=np.maximum(v-theta,0.0)
    tmp=w.sum()
    if tmp <=0 or not finite:
       w=np.full(n,s/n)
    else:
       w *= s/tmp
    return w
```

```python
"""
Search non-negative sequences minimizing J = 2*n*max(conv(a,a))/sum(a)**2.

Method overview:
  * Work mostly in the scale-invariant normalization sum(a) = 1.
  * Use a fast FFT self-convolution for candidate scoring.
  * Initialize from:
      - mixed anti-symmetric / symmetric basis arms (sines, cosines, linear term),
        refined by coordinate descent;
      - a bank of random structured templates (two-block, linear, exponential, etc.);
      - a binary bit-flip local search.
  * Refine promising starts with a projected sub-gradient/mini line search:
      - project onto the simplex {a >= 0, sum(a) = 1};
      - take a weighted average of a small set of currently maximal convolution lags;
      - backtrack over a few step sizes.
  * Final answer is verified with the supplied exact evaluate_sequence function.
"""

import numpy as np
import time


def get_nfft(n):
    """Smallest power of two >= 2n-1 for full convolution via FFT."""
    need = 2 * int(n) - 1
    out = 1
    while out < need:
        out <<= 1
    return out


def fast_conv(a):
    """Full self-convolution using FFT. Suitable for search scoring."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = get_nfft(n)
    fa = np.fft.rfft(a, n=N)
    c = np.fft.irfft(fa * fa, n=N)
    return c[: 2 * n - 1]


def fast_score(a):
    """Fast approximate objective (same form as evaluate_sequence)."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-12:
        return np.inf
    c = fast_conv(a)
    mc = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(mc):
        return np.inf
    return float(2.0 * a.size * mc / (s * s))


def normalize_positive(a, target_sum=1.0):
    """Clip to non-negative and rescale to target sum."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        a = np.ones_like(a, dtype=np.float64)
        return a * (float(target_sum) / a.size)
    return a * (float(target_sum) / s)


def project_simplex(v, s=1.0):
    """Euclidean projection onto {x >= 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    nz = np.nonzero(u * np.arange(1, n + 1) > (cssv - float(s)))[0]
    if nz.size == 0:
        return np.full(n, float(s) / n, dtype=np.float64)
    rho = int(nz[-1])
    theta = (cssv[rho] - float(s)) / float(rho + 1)
    w = np.maximum(v - theta, 0.0)
    sw = float(w.sum())
    if sw <= 0.0 or not np.isfinite(sw):
        w = np.full(n, float(s) / n, dtype=np.float64)
    else:
        w *= float(s) / sw
    return w


def add_candidate(top, score, vec, max_size=12):
    """Maintain a small list of the best (score, vector) candidates."""
    if not np.isfinite(score):
        return
    vec = np.asarray(vec, dtype=np.float64)
    top.append((float(score), vec))
    if len(top) > max_size:
        worst_idx = 0
        worst_score = top[0][0]
        for i in range(1, len(top)):
            if top[i][0] > worst_score:
                worst_score = top[i][0]
                worst_idx = i
        del top[worst_idx]


def get_top_vectors(top, k):
    """Return up to k vectors from top sorted by ascending score."""
    items = list(top)
    for i in range(len(items)):
        best_i = i
        for j in range(i + 1, len(items)):
            if items[j][0] < items[best_i][0]:
                best_i = j
        if best_i != i:
            items[i], items[best_i] = items[best_i], items[i]
    return [items[i][1] for i in range(min(k, len(items)))]


def two_block_vector(n, ratio=2.0, frac=0.5):
    """Simple strong baseline: first frac part high, second part low."""
    a = np.ones(int(n), dtype=np.float64)
    p = max(1, min(int(n) - 1, int(frac * int(n))))
    a[:p] = float(ratio)
    return normalize_positive(a, 1.0)


def build_mixed_basis(x):
    """
    Basis:
      - sine modes (anti-palindromic around x=0.5)
      - a couple of cosine modes
      - linear anti-palindromic term x-0.5
    """
    cols = []
    for k in range(1, 6):
        cols.append(np.sin(2.0 * np.pi * k * x))
    for k in range(1, 3):
        cols.append(np.cos(2.0 * np.pi * k * x))
    cols.append(x - 0.5)
    basis = np.column_stack(cols).astype(np.float64)
    bounds = np.array([1.0] * 5 + [0.7] * 2 + [1.0], dtype=np.float64)
    return basis, bounds


def make_from_basis(params, basis):
    """raw = 1 + basis @ params, clipped non-negative and normalized."""
    raw = 1.0 + basis.dot(params)
    raw = np.maximum(raw, 0.0)
    # Avoid pathological all-zero raw.
    s = float(raw.sum())
    if s <= 1e-15:
        raw = np.ones_like(raw, dtype=np.float64)
        s = float(raw.sum())
    return raw * (1.0 / s)


def optimize_basis(x, rng, deadline, top):
    """Coordinate descent over a mixed trigonometric basis."""
    basis, bounds = build_mixed_basis(x)
    K = basis.shape[1]
    starts = 18

    for _ in range(starts):
        if time.monotonic() > deadline:
            break
        q = rng.uniform(-0.75, 0.75, size=K)
        params = q * bounds
        local_params = params.copy()
        local_best = make_from_basis(local_params, basis)
        local_score = fast_score(local_best)

        step = 0.12
        passes = 0
        improved = True
        while improved and passes < 10 and time.monotonic() < deadline:
            improved = False
            passes += 1
            for j in range(K):
                for mult in (-1.0, -0.5, -0.2, 0.2, 0.5, 1.0):
                    p = local_params.copy()
                    p[j] += step * mult
                    if p[j] < -1.25 * bounds[j] or p[j] > 1.25 * bounds[j]:
                        continue
                    cand = make_from_basis(p, basis)
                    cs = fast_score(cand)
                    if np.isfinite(cs) and cs < local_score * (1.0 - 1e-10):
                        local_params = p
                        local_best = cand
                        local_score = cs
                        improved = True
            if improved:
                step = min(0.45, step * 1.15)
            else:
                step *= 0.62
                if step < 0.015:
                    break
        add_candidate(top, local_score, local_best, 12)


def random_family_vector(x, rng):
    """Random structured template vector."""
    n = x.size
    kind = int(rng.integers(0, 7))

    if kind == 0:
        # two-block
        f = float(rng.uniform(0.25, 0.75))
        ratio = 10.0 ** float(rng.uniform(-1.0, 1.0))
        p = max(1, min(n - 1, int(f * n)))
        a = np.ones(n, dtype=np.float64)
        a[:p] = ratio
        if rng.random() < 0.5:
            a = a[::-1].copy()
    elif kind == 1:
        # linear
        lo = 10.0 ** float(rng.uniform(-1.0, 1.0))
        hi = 10.0 ** float(rng.uniform(-1.0, 1.0))
        a = np.linspace(hi, lo, n, dtype=np.float64)
    elif kind == 2:
        # low-order sinusoid + linear term
        A = float(rng.uniform(-0.9, 0.9))
        B = float(rng.uniform(-0.5, 0.5))
        C = float(rng.uniform(-0.9, 0.9))
        ph = float(rng.uniform(-0.5, 0.5))
        a = (
            1.0
            + A * np.sin(2.0 * np.pi * (x + ph))
            + B * np.cos(2.0 * np.pi * (x + ph))
            + C * (x - 0.5)
        )
    elif kind == 3:
        # two-sided exponential
        k1 = float(rng.uniform(-3.0, 3.0))
        k2 = float(rng.uniform(-3.0, 3.0))
        c = float(rng.uniform(0.0, 2.0))
        a = np.exp(k1 * x) + c * np.exp(k2 * (1.0 - x)) + 0.1
    elif kind == 4:
        # power functions
        p = float(rng.uniform(0.3, 3.0))
        q = float(rng.uniform(0.3, 3.0))
        c = float(rng.uniform(0.1, 3.0))
        e = 0.01
        a = (x + e) ** p + c * ((1.0 - x) + e) ** q
    elif kind == 5:
        # three-block
        f1 = float(rng.uniform(0.1, 0.45))
        f2 = float(rng.uniform(0.1, 0.45))
        if 1.0 - f1 - f2 < 0.05:
            f1 = f2 = 0.4
            f3 = 0.2
        else:
            f3 = 1.0 - f1 - f2
        levels = 10.0 ** rng.uniform(-1.0, 1.0, size=3)
        p1 = max(1, min(n - 2, int(f1 * n)))
        p2 = max(1, min(n - p1 - 1, int(f2 * n)))
        a = np.empty(n, dtype=np.float64)
        a[:p1] = levels[0]
        a[p1:p1 + p2] = levels[1]
        a[p1 + p2:] = levels[2]
        if rng.random() < 0.5:
            a = a[::-1].copy()
    else:
        # random small basis vector
        a = np.ones(n, dtype=np.float64)
        for k in range(1, 5):
            a += float(rng.uniform(-0.8, 0.8)) * np.sin(2.0 * np.pi * k * x)
        a += float(rng.uniform(-0.4, 0.4)) * np.cos(2.0 * np.pi * x)

    return normalize_positive(a, 1.0)


def random_families(x, rng, count, deadline, top):
    """Some random structured initializations."""
    for _ in range(count):
        if time.monotonic() > deadline:
            break
        a = random_family_vector(x, rng)
        sc = fast_score(a)
        add_candidate(top, sc, a, 12)


def flip_apply(c, a, pos, w, n):
    """Flip a[pos] and update convolution c and weight w."""
    if a[pos] == 0:
        seg = c[pos:pos + n]
        seg += a
        seg += a
        c[2 * pos] += 1.0
        a[pos] = 1.0
        return w + 1
    else:
        seg = c[pos:pos + n]
        seg -= a
        seg -= a
        c[2 * pos] += 1.0
        a[pos] = 0.0
        return w - 1


def adjust_weight(a, n, lo_frac, hi_frac, rng):
    """Adjust binary weight into a reasonable density range."""
    w = int(np.sum(a))
    lo = max(1, int(lo_frac * n))
    hi = min(n - 1, int(hi_frac * n))
    if w < lo or w > hi:
        idx = np.arange(n)
        rng.shuffle(idx)
        if w < lo:
            for i in idx:
                if a[i] == 0:
                    a[i] = 1.0
                    w += 1
                    if w >= lo:
                        break
        else:
            for i in idx:
                if a[i] == 1:
                    a[i] = 0.0
                    w -= 1
                    if w <= hi:
                        break
    return a, w


def bit_search(n, rng, deadline, top, initial=None):
    """
    Binary local search over 0/1 sequences.
    Uses incremental convolution updates and periodic FFT re-sync.
    """
    if initial is None:
        a = (rng.integers(0, 2, size=n) > 0).astype(np.float64)
    else:
        base = np.asarray(np.asarray(initial, dtype=np.float64), dtype=np.float64).copy()
        base = np.where(np.isfinite(base), base, 0.0)
        if base.size == 0:
            a = (rng.integers(0, 2, size=n) > 0).astype(np.float64)
        else:
            q = float(np.median(base))
            a = (base > q).astype(np.float64)
            if float(a.sum()) <= 0:
                a[n // 2] = 1.0

    a, w = adjust_weight(a, n, 0.30, 0.70, rng)
    c = fast_conv(a)
    local_score = float(2 * n * np.max(c) / (w * w))
    best_a = a.copy()
    best_score = local_score

    min_w = max(1, int(0.22 * n))
    max_w = int(0.78 * n)
    no_improve = 0
    flips = 0

    while time.monotonic() < deadline:
        pos = int(rng.integers(0, n))
        neww = flip_apply(c, a, pos, w, n)
        if neww < min_w or neww > max_w:
            w = flip_apply(c, a, pos, w, n)
            continue

        mc = float(np.max(c))
        new_score = float(2 * n * mc / (neww * neww))

        if np.isfinite(new_score) and new_score < local_score * (1.0 - 1e-12):
            w = neww
            local_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            w = flip_apply(c, a, pos, w, n)
            no_improve += 1

        flips += 1
        if flips % 2500 == 0:
            c = fast_conv(a)
            mc = float(np.max(c))
            local_score = float(2 * n * mc / (w * w))
            if local_score < best_score:
                best_score = local_score
                best_a = a.copy()

        if no_improve > 20000:
            for _ in range(20):
                rp = int(rng.integers(0, n))
                w = flip_apply(c, a, rp, w, n)
            a, w = adjust_weight(a, n, 0.30, 0.70, rng)
            c = fast_conv(a)
            mc = float(np.max(c))
            local_score = float(2 * n * mc / (w * w))
            if local_score < best_score:
                best_score = local_score
                best_a = a.copy()
            no_improve = 0

    final_a = normalize_positive(best_a, 1.0)
    final_score = fast_score(final_a)
    add_candidate(top, final_score, final_a, 12)


def add_shifted(a, idx, weight, out):
    """
    Add weight * shifted a to out for one convolution lag idx:
    out[i] += weight * a[idx-i] whenever 0 <= idx-i < len(a).
    """
    n = a.size
    start = idx - (n - 1)
    if start < 0:
        start = 0
    end = idx
    if end >= n:
        end = n - 1
    if start > end or weight == 0:
        return
    left = idx - end
    right = idx - start
    out[start:end + 1] += weight * a[left:right + 1][::-1]


def gradient_top(a, c, K=32, hot=0.05):
    """
    Weighted subgradient from top K lags of c.
    Used to minimize max(c) in the sum=1 formulation.
    """
    n = a.size
    L = c.size
    maxc = float(np.max(c)) if c.size else 0.0
    grad = np.zeros(n, dtype=np.float64)
    if maxc <= 0.0 or not np.isfinite(maxc):
        return grad

    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -K)[-K:]

    vals = c[idx]
    temp = max(1e-300, maxc * float(hot))
    w = np.exp((vals - maxc) / temp)
    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.full(idx.shape, 1.0 / max(1, idx.size), dtype=np.float64)
    else:
        w = w / sw

    for i in range(idx.size):
        add_shifted(a, int(idx[i]), float(w[i]), grad)
    grad *= 2.0
    return grad


def perturb_vector(a, rng, mode=0):
    """Small random perturbation, then simplex projection."""
    n = a.size
    mean = float(a.mean())
    if mean <= 0:
        mean = 1.0 / n

    if mode == 0:
        cand = a + rng.normal(0.0, 0.05 * mean, size=n)
    else:
        cand = a.copy()
        width = max(8, n // 64)
        hi = max(width + 1, n)
        st = int(rng.integers(0, hi - width + 1))
        cand[st:st + width] *= np.exp(rng.normal(0.0, 0.25, size=width))
        if rng.random() < 0.35:
            st2 = int(rng.integers(0, hi - width + 1))
            cand[st2:st2 + width] *= np.exp(rng.normal(0.0, 0.25, size=width))

    return project_simplex(np.maximum(cand, 0.0), 1.0)


def refine_gradient(a, deadline, rng):
    """
    Projected top-lag gradient refinement.
    The variable is kept on the simplex sum=1.
    """
    a = normalize_positive(a, 1.0)
    best = a.copy()
    best_score = fast_score(a)
    n = a.size
    if n == 0:
        return best, np.inf

    moment = np.zeros(n, dtype=np.float64)
    mu = 0.55
    hot = 0.05
    no_improve = 0
    K = 32

    while time.monotonic() < deadline:
        c = fast_conv(a)
        if c.size == 0:
            break
        maxc = float(np.max(c))
        if not np.isfinite(maxc) or maxc <= 0:
            break
        s = float(a.sum())
        score = float(2 * n * maxc / (s * s))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()

        grad = gradient_top(a, c, K, hot)
        direction = grad + mu * moment
        dmax = float(np.max(np.abs(direction)))
        if not np.isfinite(dmax) or dmax <= 1e-30:
            break

        # Scale so that a basic step changes typical entries by O(1/n).
        scale = (1.0 / float(n)) / dmax
        improved = False
        for base in (1.0, 0.5, 0.2, 0.1, 0.05):
            cand = a - (base * scale) * direction
            cand = project_simplex(cand, 1.0)
            cs = fast_score(cand)
            if np.isfinite(cs) and cs < best_score * (1.0 - 1e-12):
                a = cand
                best_score = cs
                best = a.copy()
                moment = direction.copy()
                improved = True
                break

        if improved:
            no_improve = 0
            hot = max(0.005, hot * 0.985)
        else:
            # Try a couple of smaller steps before declaring stagnation.
            for base in (0.02, 0.01):
                cand = a - (base * scale) * direction
                cand = project_simplex(cand, 1.0)
                cs = fast_score(cand)
                if np.isfinite(cs) and cs < best_score * (1.0 - 1e-12):
                    a = cand
                    best_score = cs
                    best = a.copy()
                    moment = direction.copy()
                    improved = True
                    break

            if not improved:
                no_improve += 1
                moment *= 0.5
                if no_improve % 8 == 0:
                    cand = perturb_vector(a, rng, 0)
                    cs = fast_score(cand)
                    if np.isfinite(cs) and cs < best_score * (1.0 - 1e-12):
                        a = cand
                        best_score = cs
                        best = a.copy()
                        moment *= 0.0
                        no_improve = 0
                if no_improve % 12 == 0:
                    cand = perturb_vector(a, rng, 1)
                    cs = fast_score(cand)
                    if np.isfinite(cs) and cs < best_score * (1.0 - 1e-12):
                        a = cand
                        best_score = cs
                        best = a.copy()
                        moment *= 0.0
                        no_improve = 0
                if no_improve % 15 == 0:
                    hot = max(0.005, hot * 0.7)
                if no_improve > 110:
                    break

    return best, best_score


def exact_best_candidates(candidates, hard_deadline):
    """
    Verify top candidates with the authoritative evaluate_sequence function
    (if available) and return the best valid Python list.
    """
    best_list = None
    best_score = np.inf
    seen = set()

    for a in candidates:
        if time.monotonic() > hard_deadline - 2.0:
            break
        arr = np.asarray(a, dtype=np.float64)
        if arr.size == 0:
            continue
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if s <= 0.0:
            continue
        sig = (arr.size, round(s, 12), round(float(np.max(arr)), 12))
        if sig in seen:
            continue
        seen.add(sig)
        lst = [float(v) for v in arr.tolist()]
        try:
            sc = float(evaluate_sequence(lst))
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_list = lst

    if best_list is None:
        arr = np.clip(two_block_vector(4096), 0.0, 1000.0)
        best_list = [float(v) for v in arr.tolist()]
    return best_list


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main entry point: return best non-negative sequence found."""
    t0 = time.monotonic()
    budget = max(60.0, float(budget_s))
    hard_deadline = t0 + budget
    search_deadline = hard_deadline - 25.0  # reserve final exact verification

    rng = np.random.default_rng(int(seed))
    n = 4096
    x = (np.arange(n) + 0.5) / float(n)

    top = []
    add_candidate(top, fast_score(two_block_vector(n)), two_block_vector(n), 12)

    # Phase 1: basis coordinate search.
    d1 = min(t0 + max(50.0, 0.10 * budget), search_deadline - 120.0)
    optimize_basis(x, rng, d1, top)

    # Phase 2: random structured templates.
    d2 = min(t0 + max(80.0, 0.16 * budget), search_deadline - 120.0)
    random_families(x, rng, 2500, d2, top)

    # Phase 3: binary local search from random and from current best template.
    d3 = min(t0 + max(130.0, 0.22 * budget), search_deadline - 80.0)
    bit_search(n, rng, d3, top, None)

    if time.monotonic() < search_deadline - 80.0:
        best_template = top[0][1] if top else two_block_vector(n)
        bit_search(n, rng, min(time.monotonic() + 70.0, search_deadline - 80.0), top, best_template)

    # Phase 4: gradient refinement of several promising starts.
    if time.monotonic() < search_deadline - 30.0:
        seeds = get_top_vectors(top, 3)
        if not seeds:
            seeds = [two_block_vector(n)]

        now = time.monotonic()
        remaining = max(20.0, (search_deadline - now) - 10.0)
        num_seeds = len(seeds)
        chunk = remaining / float(num_seeds + 1)

        global_best = seeds[0].copy()
        global_score = fast_score(global_best)

        for i, s in enumerate(seeds):
            end = now + chunk * (i + 1)
            if time.monotonic() >= end or time.monotonic() >= search_deadline - 25.0:
                break
            a, sc = refine_gradient(s, min(end, search_deadline - 25.0), rng)
            if np.isfinite(sc) and sc < global_score:
                global_score = sc
                global_best = a.copy()

        # Final deep refinement of the global leader.
        if time.monotonic() < search_deadline - 12.0:
            a, sc = refine_gradient(global_best, search_deadline - 12.0, rng)
            if np.isfinite(sc) and sc < global_score:
                global_score = sc
                global_best = a.copy()
            add_candidate(top, global_score, global_best, 12)
    else:
        global_best = top[0][1].copy() if top else two_block_vector(n)
        global_score = fast_score(global_best)
        add_candidate(top, global_score, global_best, 12)

    # Final exact verification of a few top candidates.
    final_candidates = [global_best]
    for v in get_top_vectors(top, 5):
        final_candidates.append(v)

    return exact_best_candidates(final_candidates, hard_deadline - 2.0)
```
