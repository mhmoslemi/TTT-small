    import numpy as np
    
    deadline = time.monotonic() + budget_s - 5
    
    # Get initial sequence
    try:
        base = np.array(height_sequence_1, dtype=np.float64)
    except:
        base = np.ones(1000)
    
    n = len(base)
    best = base.copy()
    best_score = evaluate_sequence(best.tolist())
    
    # Normalize for optimization (score is scale-invariant)
    # Work with sum(f) = 1
    f = best.copy()
    f_sum = f.sum()
    if f_sum > 0:
        f = f / f_sum
    
    N = 1
    while N < 2*n - 1:
        N *= 2
    
    def compute_autocorr(f):
        F = np.fft.rfft(f, N)
        return np.fft.irfft(F * F, N)[:2*n-1]
    
    def project_simplex(v):
        # Project v onto {x >= 0, sum(x) = 1}
        n = len(v)
        # Binary search for threshold
        v_sorted = np.sort(v)[::-1]
        cv = np.cumsum(v_sorted)
        rho = np.nonzero(v_sorted * np.arange(1, n+1) > (cv - 1))[0][-1]
        theta = (cv[rho] - 1) / (rho + 1)
        return np.maximum(v - theta, 0)
    
    # Optimization loop
    for t in range(100000):
        if time.monotonic() > deadline:
            break
        
        h = compute_autocorr(f)
        k_star = np.argmax(h)
        
        # Subgradient
        sg = np.zeros(n)
        # sg[i] = f[k_star - i] if 0 <= k_star - i < n
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            sg[i] = f[k_star - i]
        
        # Step size
        alpha = 1.0 / (np.linalg.norm(sg, 2) ** 2 * (t + 1) ** 0.5)
        
        f_new = f - alpha * sg
        f = project_simplex(f_new)
        
        # Check improvement
        score = evaluate_sequence(f.tolist())
        if score < best_score:
            best_score = score

import numpy as np
import time

def _compute_autocorr(f, N):
    """Compute linear autocorrelation (convolution with itself) using FFT."""
    F = np.fft.rfft(f, N)
    return np.fft.irfft(F * F, N)

def _project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _smoothed_gradient(f, h, beta, N2, n):
    """Compute the gradient of the log-sum-exp smoothed objective."""
    bh = beta * h
    bh -= bh.max()
    p = np.exp(bh)
    p_sum = p.sum()
    if p_sum < 1e-300:
        return np.zeros(n)
    p /= p_sum
    # g[i] = sum_k p[k] * f[k-i] for valid indices
    # = conv(p, f_reversed)[n-1 : n-1+n]
    P = np.fft.rfft(p, N2)
    Fr = np.fft.rfft(f[::-1], N2)
    conv = np.fft.irfft(P * Fr, N2)
    return conv[n - 1 : n - 1 + n]

def _peak_subgradient(f, h, n):
    """Compute subgradient using only the peak location."""
    k_star = np.argmax(h)
    sg = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for i in range(lo, hi + 1):
        sg[i] = f[k_star - i]
    return sg

def _multi_peak_subgradient(f, h, n, top_k=5):
    """Compute subgradient using top-k peaks weighted by their values."""
    h_copy = h.copy()
    indices = np.argsort(h_copy)[::-1][:top_k]
    weights = h_copy[indices]
    total = weights.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights /= total
    sg = np.zeros(n)
    for k_star, w in zip(indices, weights):
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            sg[i] += w * f[k_star - i]
    return sg

def _local_polish(f, n, N, max_iter=200):
    """Local polish: iteratively reduce the peak by adjusting contributing variables."""
    f = f.copy()
    for _ in range(max_iter):
        h = _compute_autocorr(f, N)[:2*n-1]
        h_max = h.max()
        k_star = np.argmax(h)
        # Find all i that contribute to h_k_star
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contributors = list(range(lo, hi + 1))
        # Reduce the largest contributors
        # The contribution of f_i to h_k_star is f_i * f[k_star - i]
        contribs = np.array([f[i] * f[k_star - i] for i in contributors])
        # Reduce proportionally
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break
        target = h_max * 0.9  # Reduce peak by 10%
        reduction = h_max - target
        if reduction < 1e-15:
            break

        scale = 1.0 - reduction / (total_contrib + 1e-15)
        for idx, i in enumerate(contributors):

"""
Minimize the autocorrelation peak of a non-negative sequence using
FFT-accelerated smoothed gradient descent with adaptive step size,
multiple restarts, and local polishing.
"""

import numpy as np
import time


def _autocorr_fft(f, N):
    """Compute linear autocorrelation (full convolution) using FFT."""
    F = np.fft.rfft(f, N)
    return np.fft.irfft(F * F, N)


def _project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _grad_smoothed(f, h, beta, N2, n):
    """Gradient of log-sum-exp smoothed max(autocorrelation)."""
    bh = beta * h
    bh -= bh.max()
    p = np.exp(bh)
    ps = p.sum()
    if ps < 1e-300:
        return np.zeros(n)
    p /= ps

def _grad_multi_peak(f, h, n, top_k=5):
    """Subgradient using top-k peaks, weighted by their values."""
    idx = np.argsort(h)[::-1][:top_k]
    w = h[idx].copy()
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    sg = np.zeros(n)
    for ki in range(len(idx)):
        k = idx[ki]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            sg[i] += w[ki] * f[k - i]
    return sg


def _local_polish(f, n, N, max_iter=300, time_limit_s=None):
    """Reduce peak by iteratively scaling down peak-contributing variables."""
    f = f.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        return f
    f = f / s
    t_end = time.monotonic() + (time_limit_s if time_limit_s else 1e9)
    
    for it in range(max_iter):
        if time.monotonic() > t_end:
            break
        h = _autocorr_fft(f, N)[:2 * n - 1]
        h_max = h.max()
        k_star = int(np.argmax(h))
        
        # Find contributors to h[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        idx = np.arange(lo, hi + 1)
        # Contribution of f[i] to h[k_star] is f[i] * f[k_star - i]
        contribs = f[idx] * f[k_star - idx]
        total = contribs.sum()
        if total < 1e-15:
            break
        
        # Target: reduce peak by a small fraction
        target = h_max * 0.95
        reduction = h_max - target
        if reduction < 1e-15:
            break
        
        # Scale down contributors proportionally
        # f[i] -> f[i] * (1 - delta * contribs[i] / total)
        # This reduces h[k_star] by approximately delta * h_max
        delta = min(1.0, reduction / (h_max + 1e-15))
        scale = 1.0 - delta * contribs / total
        f[idx] *= scale
        
        # Project back to simplex
        f = _project_simplex(f)
    
    return f

```python
"""
Minimize 2n*max(f*f)/(sum(f))^2 using FFT-accelerated soft-peak gradient
descent on the probability simplex, with multi-restart and local polish.
"""

import numpy as np
import time


def _next_pow2(m):
    """Smallest power of 2 >= m."""
    n = 1
    while n < m:
        n <<= 1
    return n


def _autocorr(f, N):
    """Linear autocorrelation via FFT."""
    F = np.fft.rfft(f, N)
    return np.fft.irfft(F * F, N)


def _proj_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _grad_soft(f, h, N2, n, gamma):
    """Soft-peak weighted gradient: g[i] = sum_k w[k]*f[k-i]."""
    h_max = h.max()
    if h_max < 1e-15:
        return np.zeros(n)
    w = np.exp(h / h_max * gamma - gamma)  # subtract max for stability
    w /= w.sum()
    W = np
