        for alpha in [0.3, 0.5, 0.7, 0.9]:
            for func in [np.cos, np.sin]:
                a = 1.0 + alpha * func(2.0 * np.pi * x / n)
                a = np.clip(a, 0.01, None)
                candidates.append(a / np.sum(a))
        # Off-center Gaussians
        for center in [0.3, 0.4, 0.5, 0.6, 0.7]:
            for sigma in [0.1, 0.15, 0.2, 0.3, 0.4]:
                a = np.exp(-0.5 * ((x - center * (n - 1)) / (sigma * n)) ** 2)
                candidates.append(a / np.sum(a))
        # Power laws
        for p in [0.5, 1.0, 2.0, 3.0]:
            candidates.append((x + 1) ** p / np.sum((x + 1) ** p))
            candidates.append((n - x + 1) ** p / np.sum((n - x + 1) ** p))
        # Asymmetric bumps (steep rise, gentle fall)
        mid = max(1, n // 2)
        for rise in [1.5, 2.0, 3.0, 4.0]:
            for fall in [0.5, 1.0, 1.5]:
                left_frac = (x + 1) / (mid + 1)
                right_frac = (n - x) / (n - mid + 1)
                a = np.where(x < mid, left_frac ** rise, right_frac ** fall)
                a = np.clip(a, 0.01, None)
                candidates.append(a / np.sum(a))
        # Quadratic shapes
        for c0 in [0.2, 0.3, 0.5, 0.7]:
            a = 1.0 - ((x / max(n - 1, 1)) - c0) ** 2
            a = np.clip(a, 0.01, None)
            candidates.append(a / np.sum(a))
        # From parent
        if parent is not None:
            if len(parent) >= n:
                candidates.append(parent[:n] / np.sum(parent[:n]))
                ms = (len(parent) - n) // 2
                candidates.append(parent[ms:ms + n] / np.sum(parent[ms:ms + n]))
            else:
                xp = np.linspace(0, 1, len(parent))
                xn = np.linspace(0, 1, n)
                ai = np.interp(xn, xp, parent)
                ai = np.clip(ai, 0.01, None)
                candidates.append(ai / np.sum(ai))

        # Evaluate and sort
        scored = []
        for a in candidates:
            sc = compute_score(a)
            if np.isfinite(sc):
                scored.append((sc, a.copy()))
        scored.sort(key=lambda t: t[0])

        if scored and scored[0][0] < best_score:
            best_score = scored[0][0]
            best_seq = scored[0][1].copy()
            print(f"n={n}: best init = {best_score:.6f}")

        # Optimize top candidates
        num_opt = min(4, len(scored))
        for i in range(num_opt):
            remaining = deadline - time.monotonic()
            if remaining < 3.0:
                break
            t_alloc = min(alloc / max(num_opt, 1) - 1.0, remaining - 2.0)
            if t_alloc < 1.0:
                break
            opt_a, opt_score = optimize_sequence(scored[i][1], n, t_alloc)
            if opt_score < best_score:
                best_score = opt_score
                best_seq = opt_a.copy()
                print(f"  n={n} init[{i}]: {best_score:.6f}")

        print(f"n={n} done, best={best_score:.6f}, time={time.monotonic()-global_start:.0f}s")

    # Try parent if time remains
    if parent is not None and len(parent) > 0:
        remaining = deadline - time.monotonic()
        if remaining > 15.0:
            n_p = len(parent)
            a_p = parent / np.sum(parent)
            sc_p = compute_score(a_p)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_p.copy()
                print(f"parent n={n_p}: {best_score:.6f}")
            t_opt = min(remaining - 5.0, 400.0)
            opt_a, opt_score = optimize_sequence(a_p, n_p, t_opt)
            if opt_score < best_score:
                best_score = opt_score
                best_seq = opt_a.copy()
                print(f"parent opt n={n_p}: {best_score:.6f}")

    # Final: validate with actual evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if np.sum(best_seq) < 0.01:
        best_seq = best_seq * 100.0
    final = evaluate_sequence([float(v) for v in best_seq])
    print(f"FINAL: score={final:.6f}, len={len(best_seq)}")
    return [float(v) for v in best_seq]
