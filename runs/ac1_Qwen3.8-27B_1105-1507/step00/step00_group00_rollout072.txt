    n = len(f)
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # A_ub @ g <= b_ub
    # (f*g)[k] = sum_{i: 0<=i<n, 0<=k-i<n} f[i] * g[k-i]

    else:
        threshold = M * (1.0 - frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        if len(active) > max_active:
            order = np.argsort(b[active])[::-1][:max_active]
            active = active[order]

    A_rows_parts = []
    A_cols_parts = []
    A_vals_parts = []

    for i, k in enumerate(active):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        A_rows_parts.append(np.full(len(js), i, dtype=np.int32))
        A_cols_parts.append(js.astype(np.int32))
        A_vals_parts.append(f[k - js].astype(np.float64))

    if not A_rows_parts:
        return None

    rows = np.concatenate(A_rows_parts)
    cols = np.concatenate(A_cols_parts)
    vals = np.concatenate(A_vals_parts)

    A = coo_matrix((vals, (rows, cols)), shape=(len(active), n)).tocsr()
    b_ub = np.full(len(active), M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A, b_ub, c_obj, bounds


def _lp_step(f, frac, max_active=300, time_limit=60):
    """One LP step: find g maximizing sum(g) s.t. (f*g)[k]<=M for active k, g>=0."""
    n = len(f)
    b = np.convolve(f, f)
    M = float(b.max())
    if M <= 1e-15:
        return None, M, b

    data = _build_active_lp(f, b, M, frac, max_active)
    if data is None:
        return None, M, b

    A, b_ub, c_obj, bounds = data
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            g_sum = g.sum()
            if g_sum > 1e-12:
                g = g / g_sum
                return g, M, b
    except Exception:
        pass
    return None, M, b


def _try_length_search(f_current, best_score, deadline, rng):
    """Try a few alternative lengths to see if a shorter/longer sequence helps."""
    best_seq = f_current.copy()
    results = []
    for n_try in [200, 500, 1000, 2000]:
        if time.monotonic() > deadline:
            break
        f_t = np.ones(n_try) / n_try
        for it in range(30):
            if time.monotonic() > deadline:
                break
            g, M, b = _lp_step(f_t, 0.5, max_active=100, time_limit=10)
            if g is None:
                break
            f_t = 0.7 * f_t + 0.3 * g
            f_t = np.clip(f_t, 0, None)
            s = f_t.sum()
            if s < 1e-10:
                break
            f_t = f_t / s
        score = float(evaluate_sequence(f_t.tolist()))
        results.append((score, n_try, f_t.copy()))
        print(f"  Length {n_try}: score={score:.6f}")

    if results:
        results.sort(key=lambda x: x[0])
        if results[0][0] < best_score:
            best_seq = results[0][2]
    return best_seq


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    LP-based iterative search minimizing 2*n*max(f*f)/(sum(f))^2.
    At each step, solves an LP over active (near-peak) convolution constraints
    to find a mixing target g, then updates f = (1-t)*f + t*g.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    # Initialize from the provided construction
    f = np.array(height_sequence_1, dtype=np.float64)
    f = np.clip(f, 0, None)
    if f.sum() <= 0:
        f = np.ones(4806)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print(f"Start: score={best_score:.6f}, n={n}")

    t = 0.1
    frac = 0.4
    max_active = 200
    no_improve = 0
    iter_count = 0
    eval_interval = 3

    while time.monotonic() < deadline:
        iter_count += 1

        # LP step
        g, M, b = _lp_step(f, frac, max_active=max_active, time_limit=90)

        if g is None:
            # Fallback: small multiplicative perturbation
            noise = 1.0 + 0.02 * rng.standard_normal(n)
            f_new = f * noise
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                f_new = f.copy()
                s = 1.0
            f = f_new / s
            no_improve += 1
            if no_improve > 20:
                frac = min(0.9, frac * 1.15)
                max_active = min(500, max_active + 20)
                no_improve = 0
            continue

        # Mix and normalize
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-12:
            t *= 0.5
            continue
        f_new = f_new / s
        f = f_new

        # Evaluate periodically
        if iter_count % eval_interval == 0:
            score = float(evaluate_sequence(f.tolist()))
            if np.isfinite(score) and score < best_score:
                best_f = f.copy()
                best_score = score
                no_improve = 0
                print(f"Iter {iter_count}: score={best_score:.6f} t={t:.4f} frac={frac:.2f}")
            else:
                no_improve += 1
        else:
            # Quick proxy: check if peak decreased
            b_check = np.convolve(f, f).max()
            if b_check < M * 0.999:
                no_improve = 0
            else:
                no_improve += 1

        # Adapt hyperparameters
        if no_improve > 40:
            frac = min(0.85, frac + 0.04)
            max_active = min(500, max_active + 30)
            no_improve = 0
        if no_improve > 100:
            t = max(0.02, t * 0.85)
            no_improve = 0
        if no_improve > 250:
            # Escape local minimum with perturbation
            f = f + 0.03 * rng.standard_normal(n)
            f = np.clip(f, 0, None)
            f = f / f.sum()
            frac = 0.3
            t = 0.1
            max_active = 150
            no_improve = 0
            print(f"  Escaped at iter {iter_count}")

    # Polish phase: use all constraints with very small step
    if time.monotonic() < deadline - 30:
        print("Polish phase...")
        t_polish = 0.02
        for _ in range(20):
            if time.monotonic() > deadline - 30:
                break
            g, M, b = _lp_step(f, 0.0, max_active=500, time_limit=60)
            if g is None:
                break
            f_new = (1.0 - t_polish) * f + t_polish * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-12:
                break
            f_new = f_new / s
            score = float(evaluate_sequence(f_new.tolist()))
            if np.isfinite(score) and score < best_score:
                best_f = f_new.copy()
                best_score = score
                f = f_new
                print(f"  Polish: score={best_score:.6f}")
            else:
                f = f_new

    final_score = float(evaluate_sequence(best_f.tolist()))
    print(f"Done: score={final_score:.6f}, n={len(best_f)}, iterations={iter_count}")
    return [float(x) for x in best_f]
