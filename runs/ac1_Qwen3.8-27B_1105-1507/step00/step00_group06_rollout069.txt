def _soft_max_weights(b, T):
    """Compute soft-max weights for autocorrelation values."""
    b_max = np.max(b)
    shifted = T * (b - b_max)
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        idx = np.argmax(b)
        w = np.zeros(len(b))
        w[idx] = 1.0
    else:
        w /= w_sum
    return w
