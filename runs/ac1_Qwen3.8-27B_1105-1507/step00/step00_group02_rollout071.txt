"""
Minimise  2*n*max(conv(a,a)) / (sum a)**2  over non-negative real sequences a.

Approach
--------
The objective is scale invariant, so only the *shape* of a matters.  We
parameterise a = exp(z) (guarantees a > 0) and minimise a smooth surrogate
of the arg‑max:  soft = soft­max of the 2n-1 taps of the linear self‑convolution
c = a*a (computed with a real FFT, O(n log n)).  The gradient of soft w.r.t. a
has the closed form  d soft / da[i] = 2 * sum_k  p[k] * a[k-i]  (p = softmax(c)),
which is a cross‑correlation evaluated with one more FFT.  This gives a cheap
exact gradient for L‑BFGS‑B.

We run many short L‑BFGS‑B attempts from diverse structured starts (exponential /
linear ramps, smooth oscillators, the provided reference sequence) at several
lengths, then polish each with a few natural‑gradient multiplicative steps that
drive the *true* max(c) down, and keep the globally best sequence (verified with
the authoritative evaluate_sequence when it improves).  The best shape is scaled
into [0, 1000] and returned.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# --- module level scratch (no closures allowed) ---------------------------
_N = 1
_n = 1
_TEMP_FACT = 0.10
_REG_SCALE = 1.0
_DEADLINE = 0.0
_STOP = False


def _next_pow2(m):
    return 1 << max(1, (m).bit_length())


def _conv_fft(a):
    """Linear self‑convolution of 1‑d a via real FFT (length 2n-1)."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    F = np.fft.rfft(a, N)
    c = np.fft.irfft(F * F, N)[: 2 * n - 1]
    return np.maximum(c, 0.0)


def fast_score(a):
    """Exact (FFT) value of the evaluation objective for a non‑negative a."""
    a = np.ascontiguousarray(a, dtype=np.float64)
    S = a.sum()
    if S <= 1e-12:
        return float("inf")
    c = _conv_fft(a)
    return float(2.0 * a.size * c.max() / (S * S))


def _init_exp_ramp(n, alpha, rng):
    R = np.linspace(0.0, 1.0, n)
    return np.exp(alpha * R) + 1e-3


def _init_lin_ramp(n, rng):
    R = np.linspace(0.0, 1.0, n)
    return (1e-3 + R)


def _init_smooth(osc, rng):
    f0 = int(rng.integers(2, 9))
    ph = float(rng.uniform(0, 2 * math.pi))
    amp = float(rng.uniform(0.3, 0.9))
    R = np.linspace(0.0, 1.0, osc.size)
    a = 1.0 + amp * np.sin(2 * math.pi * f0 * R + ph)
    a += float(rng.uniform(0.0, 0.15)) * rng.standard_normal(osc.size)
    return np.maximum(a, 1e-3)


def _make_init(kind, n, base, rng):
    if kind == "ramp":
        return _init_exp_ramp(n, float(rng.uniform(2.0, 8.0)), rng)
    if kind == "lin":
        return _init_lin_ramp(n, rng)
    if kind == "base":
        if base is not None and base.size:
            return np.maximum(np.resize(base, n), 1e-3)
        return _init_exp_ramp(n, 4.0, rng)
    # smooth
    return _init_smooth(np.ones(n), rng)


def _lbgf_fun(z):
    """Surrogate objective + exact gradient w.r.t. z (a = exp z)."""
    a = np.exp(np.clip(z, -30.0, 30.0))
    n = _n
    N = _N
    c = _conv_fft(a)
    cmax = c.max()
    t = _TEMP_FACT * cmax + 1e-12
    m = cmax
    ex = np.exp((c - m) / t)
    ssum = ex.sum()
    p = ex / ssum
    soft = m + t * math.log(ssum)

    S = a.sum()
    den = S * S
    K = 2.0 * n
    obj = K * soft / den + _REG_SCALE * (z.mean() ** 2)
    if not math.isfinite(obj):
        return obj, np.zeros_like(z)

    # GA[i] = d soft / da[i] = 2 * sum_k p[k] a[k-i]   (cross correlation)
    A = np.fft.rfft(a, N)
    Pf = np.fft.rfft(np.concatenate([p, np.zeros(N - c.size)]), N)
    # w[x] = sum_k a[k + x] p[k]   (circular, padded)  ->  h[i] = w[(N-i) mod N]
    W = np.fft.irfft(A * np.conj(Pf), N).real
    idx = np.empty(n, dtype=int)
    idx[0] = 0
    if n > 1:
        idx[1:] = N - np.arange(1, n)
    h = W[np.mod(idx, N)]
    GA = 2.0 * h

    grad_a = (K / den) * GA - (2.0 * K * soft * S) / (den * den)
    grad_z = grad_a * a + (2.0 * _REG_SCALE / n)
    return obj, grad_z


def _optimize_once(z0, maxiter):
    global _STOP
    _STOP = False
    try:
        res = minimize(
            _lbgf_fun,
            z0,
            method="L-BFGS-B",
            jac=True,
            bounds=[(-20.0, 20.0)] * z0.size,
            options={"maxiter": maxiter, "maxfun": maxiter * 2,
                     "ftol": 1e-10, "gtol": 1e-7, "iprint": 0},
        )
    except Exception:
        return z0.copy()
    if res is None or not hasattr(res, "x"):
        return z0.copy()
    return res.x.astype(np.float64)


def _polish(a, steps=40, eta=0.15, tfact=0.06):
    """Multiplicative natural‑gradient steps reducing the true max(c)."""
    a = a.astype(np.float64).copy()
    n = a.size
    N = _next_pow2(2 * n - 1)
    for _ in range(steps):
        S = a.sum()
        if S <= 1e-12:
            break
        an = a / S
        c = _conv_fft(an)
        cmax = c.max() + 1e-12
        t = tfact * cmax
        m = cmax
        ex = np.exp((c - m) / t)
        p = ex / ex.sum()
        A = np.fft.rfft(an, N)
        Pf = np.fft.rfft(np.concatenate([p, np.zeros(N - c.size)]), N)
        W = np.fft.irfft(A * np.conj(Pf), N).real
        idx = np.empty(n, dtype=int)
        idx[0] = 0
        if n > 1:
            idx[1:] = N - np.arange(1, n)
        h = W[np.mod(idx, N)]
        logit = eta * (2.0 * h - np.mean(2.0 * h))
        a = a * np.exp(logit)
        a = np.maximum(a, 1e-6)
    return a


def _scale_to_range(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    m = a.max()
    if m > 1000.0 and m > 0:
        a = a * (1000.0 / m)
    elif m == 0.0:
        a = np.ones_like(a)
    return a
