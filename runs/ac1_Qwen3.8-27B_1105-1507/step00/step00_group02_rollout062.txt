import numpy as np
import time
from scipy.optimize import minimize

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu) - 1
    ind = np.arange(1, n+1)
    cond = mu - cv/ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(ind[cond])
    theta = (mu[:rho].sum() - 1) / rho
    return np.maximum(v - theta, 0)

def smooth_max(values, t=1.0):
    """Smooth approximation of max using log-sum-exp."""
    m = np.max(values)

from scipy.optimize import linprog

def lp_step(f, M):
    """Solve the LP to find the best update direction."""
    n = len(f)
    # Variables: b_0, ..., b_{n-1} (n variables)
    # Objective: maximize sum(b_j) → minimize -sum(b_j)
    c = -np.ones(n)
    
    # Constraints: (f*b)[k] ≤ M for k = 0, ..., 2n-2
    # (f*b)[k] = sum_{i: 0≤i<n, 0≤k-i<n} f[i]*b[k-i]
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, j] = f[i]
    
    b_ub = np.full(2*n-1, M)
    
    # b ≥ 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

import numpy as np
import time
from scipy.optimize import linprog

def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu) - 1
    ind = np.arange(1, n+1)
    cond = mu - cv/ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(ind[cond])
    theta = (mu[:rho].sum() - 1) / rho
    return np.maximum(v - theta, 0)

def compute_score(f):
    """Compute the score 2*n*max(conv(f,f))/(sum(f))^2."""
    n = len(f)
    s = np.sum(f)
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    M = np.max(conv)
    return 2.0 * n * M / (s * s)
