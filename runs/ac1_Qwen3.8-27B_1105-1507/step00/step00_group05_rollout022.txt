def subgradient_step(a, b, active_positions, lr):
    """
    Compute a subgradient step to reduce the peak of the convolution.
    a: current sequence (normalized, non-negative)
    b: convolution of a with itself
    active_positions: positions where b is near its maximum
    lr: learning rate
