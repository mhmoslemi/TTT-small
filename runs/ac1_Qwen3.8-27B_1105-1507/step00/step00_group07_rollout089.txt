"""
Hunt for low-peak nonnegative step functions by combining:
- FFT-based exact scoring
- several analytic/random initial constructions
- smooth log-mean-exp surrogate optimization in log-softmax and bounded-cos bases
- active-set linear programming side-information moves
- final stochastic refinement
"""

import time
import math
import numpy as np

from scipy.optimize import linprog, minimize


# -----------------------------
# Convolution / scoring helpers
# -----------------------------

def _next_pow2(m):
    if m <= 1:
        return 1
    return 1 << int((m - 1).bit_length())


def _conv_self(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    out_len = 2 * n - 1
    L = _next_pow2(out_len)
    fa = np.fft.rfft(a, L)
    out = np.fft.irfft(fa * fa, L)
    return out[:out_len]


def _conv_full(a, b):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    out_len = a.size + b.size - 1
    if out_len <= 0:
        return np.array([], dtype=np.float64)
    L = _next_pow2(out_len)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:out_len]


def _normalize_sum(a, target):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-14:
        return np.zeros_like(a)
    return a * (float(target) / s)


def _fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-12:
        return float("inf")
    c = _conv_self(a)
    if c.size == 0:
        return float("inf")
    m = float(c.max())
    if not math.isfinite(m) or m < 0.0:
        m = 0.0
    return float(2.0 * n * m / (s * s))


def _logmeanexp(c, lam):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return float("inf")
    z = float(lam) * c
    zmax = float(z.max())
    e = np.exp(z - zmax)
    return float((zmax + math.log(float(e.sum())) - math.log(float(c.size))) / float(lam))

        best_score = float(s)

    p = np.clip(np.asarray(p, dtype=np.float64).ravel(), 0.0, 1000.0)
    if p.size == 0:
        return None
    L = p.size
    target = math.sqrt(float(2 * n))
    if L >= n:
        out = np.empty(n, dtype=np.float64)
        for i in range(n):
            s = (i * L) // n
            e = ((i + 1) * L) // n
            if e <= s:
                out[i] = p[0]
            else:
                out[i] = float(p[s:e].mean())
    else:
        reps = (n + L - 1) // L
        out = np.tile(p, reps)[:n]
    out = np.maximum(out, 0.0)
    if out.sum() <= 1e-12:
        return None
    return _normalize_sum(out, target)


def _anti_random_r(n, rng):
    r = np.zeros(n, dtype=np.float64)
    m = n // 2
    if m > 0:
        left = rng.uniform(-1.0, 1.0, size=m)
        r[:m] = left
        r[n - 1 - np.arange(m)] = -left
    return r


def _make_batch(n, rng):
    n = int(n)
    target = math.sqrt(float(2 * n))
    arr = np.arange(n, dtype=np.float64)
    out = []

    # uniform
    out.append(np.full(n, target / n, dtype=np.float64))

    # random positive
    for _ in range(2):
        e = rng.exponential(1.0, size=n)
        out.append(_normalize_sum(e, target))
    for _ in range(1):
        e = rng.rayleigh(1.0, size=n)
        out.append(_normalize_sum(e, target))

    # power-law shapes
    for beta in (0.15, 0.45, 0.85):
        g = (arr + 1.0) ** (-beta)
        g = np.maximum(g, 0.0)
        if g.sum() > 1e-12:
            out.append(_normalize_sum(g, target))

    # anti-correlated bounded shapes
    r = _anti_random_r(n, rng)
    for a in (0.35, 0.75, 1.0):
        g = 1.0 + a * r
        g = np.maximum(g, 0.0)
        if g.sum() > 1e-12:
            out.append(_normalize_sum(g, target))

    # sinusoidal / polynomial phase shapes
    two_pi = 2.0 * math.pi
    for k in (1, 2, 3):
        phases = (0.0, math.pi * 0.25, math.pi * 0.5, math.pi)
        amps = (0.45, 0.85)
        for ph in phases:
            for a in amps:
                g = 1.0 + a * np.cos(two_pi * k * arr / n + ph)
                g = np.maximum(g, 0.0)
                if g.sum() > 1e-12:
                    out.append(_normalize_sum(g, target))

    # single sine/cosine anti-ish shapes
    for sgn in (-1.0, 1.0):
        g = 1.0 + 0.9 * sgn * np.sin(two_pi * arr / n)
        g = np.maximum(g, 0.0)
        if g.sum() > 1e-12:
            out.append(_normalize_sum(g, target))

    # alternating
    alt = np.where((np.arange(n) % 2) == 0, 1.0, -1.0)
    for a in (0.3, 0.7, 0.95):
        g = 1.0 + a * alt
        g = np.maximum(g, 0.0)
        if g.sum() > 1e-12:
            out.append(_normalize_sum(g, target))

    # parent, if present
    p = _parent_resample(n)
    if p is not None:
        out.append(p)

    return out

_CB_DEADLINE = 0.0
_LAST_P = None


def _minimize_cb(_xk):
    global _LAST_P
    try:
        if time.monotonic() > _CB_DEADLINE:
            _LAST_P = np.asarray(_xk, dtype=np.float64).copy()
            raise KeyboardInterrupt
        _LAST_P = np.asarray(_xk, dtype=np.float64).copy()
    except KeyboardInterrupt:
        raise
    except Exception:
        pass


# ----------------
# log-softmax base
# ----------------

_GS_N = 0
_GS_LAM = 1.0


def _softmax_f(x):
    x = np.asarray(x, dtype=np.float64)
    y = x - float(x.mean())
    y = y - float(y.max())
    p = np.exp(y)
    s = float(p.sum())
    if s <= 1e-14:
        return np.full_like(x, 1.0 / max(1, x.size))
    return p / s


def _fun_softmax(x):
    n = _GS_N
    lam = _GS_LAM
    f = _softmax_f(x)
    c = _conv_self(f)
    return _logmeanexp(c, lam)


def _jac_softmax(x):
    n = _GS_N
    lam = _GS_LAM
    f = _softmax_f(x)
    c = _conv_self(f)
    z = lam * c
    zmax = float(z.max())
    e = np.exp(z - zmax)
    H = e / float(e.sum())
    convH = _conv_full(f, H[::-1])
    grad_f = 2.0 * convH[n - 1:2 * n - 1][::-1]
    grad = grad_f - f * float(np.sum(grad_f))
    grad -= float(grad.mean())
    return grad


def _eval_softmax_p(x, n):
    f = _softmax_f(x)
    s = _fast_score(f)
    return f, s


def _optimize_softmax(n, deadline, rng):
    global _GS_N, _GS_LAM, _CB_DEADLINE, _LAST_P

    if n <= 0:
        return None, float("inf")

    n = int(n)
    target = math.sqrt(float(2 * n))

    starts = []
    z = np.zeros(n, dtype=np.float64)
    z += rng.normal(0.0, 1e-4, size=n)
    starts.append(z)
    starts.append(rng.normal(0.0, 0.7, size=n))
    starts.append(rng.normal(0.0, 1.5, size=n))
    r = _anti_random_r(n, rng)
    f0 = np.maximum(1.0 + 0.8 * r, 1e-6)
    f0 = f0 / float(f0.max())
    starts.append(np.log(f0 + 1e-8))

    if n >= 256:
        starts = starts[:2]

    best_f = None
    best_s = float("inf")
    schedule = (2.0, 5.0, 12.0, 30.0)

    now = time.monotonic()
    T = max(0.0, float(deadline) - now)
    if T <= 1.0 or len(starts) == 0:
        return best_f, best_s

    for ii, x0 in enumerate(starts):
        if time.monotonic() >= deadline:
            break
        child_deadline = now + T * (ii + 1) / float(len(starts))
        child_deadline = min(float(deadline), child_deadline - 0.05)
        if child_deadline <= time.monotonic() + 0.2:
            continue

        p = np.asarray(x0, dtype=np.float64).copy()
        f, s = _eval_softmax_p(p, n)
        if s < best_s:
            best_s = s
            best_f = f.copy()

        for lam in schedule:
            if time.monotonic() >= child_deadline:
                break
            _GS_N = n
            _GS_LAM = float(lam)
            _CB_DEADLINE = child_deadline - 0.02
            _LAST_P = p.copy()
            maxiter = max(15, int(35000 // max(1, n)))
            try:
                res = minimize(
                    _fun_softmax,
                    p,
                    jac=_jac_softmax,
                    method="L-BFGS-B",
                    callback=_minimize_cb,
                    options={
                        "maxiter": maxiter,
                        "maxfun": 3 * maxiter,
                        "ftol": 1e-10,
                        "gtol": 1e-7,
                        "disp": False,
                    },
                )
                if hasattr(res, "x") and res.x is not None:
                    p = np.asarray(res.x, dtype=np.float64).copy()
            except BaseException:
                pass
            if _LAST_P is not None:
                p = np.asarray(_LAST_P, dtype=np.float64).copy()
            if p.size != n:
                p = x0.copy()
            f, s = _eval_softmax_p(p, n)
            if s < best_s:
                best_s = s
                best_f = f.copy()
                print(f"softmax n={n} lam={lam} score={s:.6f}", flush=True)

    if best_f is not None and best_f.sum() > 1e-12:
        best_f = _normalize_sum(best_f, target)
        best_s = _fast_score(best_f)
    return best_f, best_s


# ----------------
# bounded cos base
# ----------------

_GC_N = 0
_GC_LAM = 1.0


def _cos_f_from_p(p, n):
    p = np.asarray(p, dtype=np.float64)
    if p.size != n + 1:
        return None
    alpha = float(np.clip(p[0], 0.0, 1.0))
    theta = p[1:]
    cth = np.cos(theta)
    g = 1.0 + alpha * cth
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-12:
        return None
    f = g / s
    return f


def _eval_cos_p(p, n):
    f = _cos_f_from_p(p, n)
    if f is None:
        return None, float("inf")
    s = _fast_score(f)
    return f, s


def _fun_cos(p):
    n = _GC_N
    lam = _GC_LAM
    f = _cos_f_from_p(p, n)
    if f is None:
        return 1e30
    c = _conv_self(f)
    return _logmeanexp(c, lam)


def _jac_cos(p):
    n = _GC_N
    lam = _GC_LAM
    p = np.asarray(p, dtype=np.float64)
    f = _cos_f_from_p(p, n)
    if f is None:
        return np.zeros(n + 1, dtype=np.float64)

    alpha = float(np.clip(p[0], 0.0, 1.0))
    theta = p[1:]
    cth = np.cos(theta)
    sth = np.sin(theta)
    g = 1.0 + alpha * cth
    g = np.maximum(g, 0.0)
    S = float(g.sum())
    if S <= 1e-12:
        return np.zeros(n + 1, dtype=np.float64)

    c = _conv_self(f)
    z = lam * c
    zmax = float(z.max())
    e = np.exp(z - zmax)
    H = e / float(e.sum())
    convH = _conv_full(f, H[::-1])
    grad_f = 2.0 * convH[n - 1:2 * n - 1][::-1]

    Gg = float(np.dot(grad_f, g))
    Gc = float(np.dot(grad_f, cth))

    dalpha = (Gc - Gg) / S
    dtheta = alpha * sth * (Gg / (S * S) - 1.0 / S)
    grad = np.empty(n + 1, dtype=np.float64)
    grad[0] = dalpha
    grad[1:] = dtheta
    return grad


def _make_theta_kind(n, kind, rng):
    arr = np.arange(n, dtype=np.float64)
    two_pi = 2.0 * math.pi
    if kind == 0:
        return rng.normal(0.0, 0.2, size=n)
    if kind == 1:
        return rng.uniform(0.0, two_pi, size=n)
    if kind == 2:
        return two_pi * arr / n
    if kind == 3:
        return (two_pi * arr * arr / n) % two_pi
    if kind == 4:
        return math.pi * (np.arange(n).astype(np.float64))
    r = _anti_random_r(n, rng)
    if kind == 5:
        return np.arccos(np.clip(r, -1.0, 1.0))
    if kind == 6:
        return np.arccos(np.clip(np.sin(two_pi * arr / n), -1.0, 1.0))
    # default
    return rng.normal(0.0, 1.0, size=n)


def _optimize_cos(n, deadline, rng):
    global _GC_N, _GC_LAM, _CB_DEADLINE, _LAST_P

    if n <= 0:
        return None, float("inf")

    n = int(n)
    target = math.sqrt(float(2 * n))

    if n >= 1024:
        kinds = (5, 1, 2)
    else:
        kinds = (5, 1, 2, 6, 0, 4)
    alphas = (0.85, 0.55)

    starts = []
    for k in kinds:
        th = _make_theta_kind(n, k, rng)
        for a in alphas:
            p = np.empty(n + 1, dtype=np.float64)
            p[0] = a
            p[1:] = th
            starts.append(p)
    if n >= 1024:
        starts = starts[:3]
    elif n >= 512:
        starts = starts[:4]
    else:
        starts = starts[:6]

    best_f = None
    best_s = float("inf")
    schedule = (3.0, 8.0, 20.0, 50.0)

    now = time.monotonic()
    T = max(0.0, float(deadline) - now)
    if T <= 1.0 or len(starts) == 0:
        return best_f, best_s

    bounds = [(0.0, 1.0)] + [(None, None)] * n

    for ii, p0 in enumerate(starts):
        if time.monotonic() >= deadline:
            break
        child_deadline = now + T * (ii + 1) / float(len(starts))
        child_deadline = min(float(deadline), child_deadline - 0.05)
        if child_deadline <= time.monotonic() + 0.2:
            continue

        p = np.asarray(p0, dtype=np.float64).copy()
        p[0] = float(np.clip(p[0], 0.0, 1.0))
        f, s = _eval_cos_p(p, n)
        if f is not None and s < best_s:
            best_s = s
            best_f = f.copy()

        for lam in schedule:
            if time.monotonic() >= child_deadline:
                break
            _GC_N = n
            _GC_LAM = float(lam)
            _CB_DEADLINE = child_deadline - 0.02
            _LAST_P = p.copy()
            maxiter = max(12, int(18000 // max(1, n)))
            try:
                res = minimize(
                    _fun_cos,
                    p,
                    jac=_jac_cos,
                    method="L-BFGS-B",
                    bounds=bounds,
                    callback=_minimize_cb,
                    options={
                        "maxiter": maxiter,
                        "maxfun": 3 * maxiter,
                        "ftol": 1e-10,
                        "gtol": 1e-6,
                        "disp": False,
                    },
                )
                if hasattr(res, "x") and res.x is not None:
                    p = np.asarray(res.x, dtype=np.float64).copy()
            except BaseException:
                pass
            if _LAST_P is not None:
                p = np.asarray(_LAST_P, dtype=np.float64).copy()
            if p.size != n + 1:
                p = p0.copy()
            p[0] = float(np.clip(p[0], 0.0, 1.0))
            f, s = _eval_cos_p(p, n)
            if f is not None and s < best_s:
                best_s = s
                best_f = f.copy()
                print(f"cos n={n} lam={lam} score={s:.6f}", flush=True)

    if best_f is not None and best_f.sum() > 1e-12:
        best_f = _normalize_sum(best_f, target)
        best_s = _fast_score(best_f)
    return best_f, best_s

# LP side-information improvement
# -----------------------------

def _build_A(f, idx):
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    idx = np.asarray(idx, dtype=np.int64)
    m = idx.size
    A = np.zeros((m, n), dtype=np.float64)
    for r, k in enumerate(idx):
        k = int(k)
        s = max(0, k - n + 1)
        e = min(n - 1, k) + 1
        if e > s:
            js = np.arange(s, e, dtype=np.int64)
            A[r, s:e] = f[k - js]
    return A


def _safe_linprog(c, A, b, n, max_time):
    if max_time <= 0.0:
        return None
    bounds = [(0.0, None)] * n
    try:
        return linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"time_limit": max(0.5, float(max_time))},
        )
    except Exception:
        pass
    try:
        return linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    except Exception:
        return None


def _lp_direction(f, M, deadline, rng, max_lp_time=6.0):
    n = f.size
    cff = _conv_self(f)
    m_total = 2 * n - 1
    if m_total <= 0:
        return None

    if n <= 512:
        max_active = m_total
    else:
        max_active = min(m_total, max(512, n))

    active = np.argsort(cff)[-max_active:]
    if active.size == 0:
        return None

    q = rng.normal(size=n)
    q -= float(q.mean())
    q_scale = 1e-5
    cobj = -np.ones(n, dtype=np.float64) + q_scale * q

    for _ in range(40):
        if time.monotonic() > deadline:
            return None
        A = _build_A(f, active)
        b = np.full(A.shape[0], float(M) * (1.0 + 1e-10) + 1e-12, dtype=np.float64)
        rem = max(0.0, float(deadline) - time.monotonic())
        max_time = min(max_lp_time, max(0.5, rem - 1.0))
        if max_time <= 0.5:
            return None

        res = _safe_linprog(cobj, A, b, n, max_time)
        if res is None or not getattr(res, "success", False) or res.x is None:
            res = _safe_linprog(-np.ones(n), A, b, n, max_time)
            if res is None or res.x is None:
                return None

        g = np.asarray(res.x, dtype=np.float64).copy()
        if not np.all(np.isfinite(g)):
            return None
        g = np.maximum(g, 0.0)
        if g.sum() <= 1e-14:
            return None

        cross = _conv_full(f, g)
        if cross.size != m_total:
            return None
        viol = cross > float(M) * (1.0 + 1e-7)
        if not bool(np.any(viol)):
            return g

        vi = np.where(viol)[0]
        union = np.union1d(active, vi)
        if union.size > max_active:
            vals = np.maximum(cff, cross)
            order = np.argsort(vals)[::-1]
            union = order[:max_active]
        active = union

    return g


def _lp_improve(f, deadline, rng):
    f = np.asarray(f, dtype=np.float64).copy()
    n = f.size
    if n == 0:
        return f, _fast_score(f)

    # The LP argument is cleanest when sum(f)^2 = 2n, i.e. peak = score.
    S = math.sqrt(float(2 * n))
    f = _normalize_sum(f, S)
    score = _fast_score(f)
    if not math.isfinite(score):
        return f, score

    if n > 1024:
        # Avoid very dense LPs.
        return f, score

    no_improve = 0
    max_iter = 200

    for _ in range(max_iter):
        if time.monotonic() >= deadline:
            break
        rem = max(0.0, float(deadline) - time.monotonic())
        if rem <= 3.0:
            break

        cff = _conv_self(f)
        M = float(cff.max())
        if not math.isfinite(M) or M <= 0.0:
            break

        g = _lp_direction(f, M, deadline, rng, max_lp_time=min(8.0, rem / 3.0))
        if g is None:
            break

        Sg = float(g.sum())
        if Sg <= 1e-12:
            break
        gN = _normalize_sum(g, S)

        cands = []
        ts = (0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0)
        for t in ts:
            cand = (1.0 - t) * f + t * gN
            cand = np.maximum(cand, 0.0)
            cands.append(cand)
            sym = 0.5 * (cand + cand[::-1])
            sym = np.maximum(sym, 0.0)
            cands.append(sym)

        best_local = f
        best_local_score = score
        for cand in cands:
            if cand.sum() <= 1e-12:
                continue
            cand = _normalize_sum(cand, S)
            ns = _fast_score(cand)
            if ns < best_local_score - 1e-12:
                best_local_score = ns
                best_local = cand

        if best_local_score < score - 1e-12:
            f = best_local
            score = float(best_local_score)
            no_improve = 0
            print(f"LP improved n={n} score={score:.6f}", flush=True)
        else:
            no_improve += 1
            if no_improve >= 4:
                # Small perturbation to make the next LP face a different face.
                cand = f.copy()
                bs = max(1, n // 64)
                st = int(rng.integers(0, max(1, n - bs + 1)))
                cand[st:st + bs] *= np.exp(rng.normal(0.0, 0.15, size=bs))
                cand = _normalize_sum(np.maximum(cand, 0.0), S)
                pc = _fast_score(cand)
                if pc < score * 1.08:
                    f = cand
                    score = float(pc)
                no_improve = 0

    return f, score


# -----------------------------
# Stochastic refinement
# -----------------------------

def _refine(f, deadline, rng):
    f = np.asarray(f, dtype=np.float64).copy()
    n = f.size
    if n == 0:
        return f, _fast_score(f)
    S0 = float(f.sum())
    if S0 <= 1e-12:
        f = np.ones(n, dtype=np.float64)
        S0 = float(f.sum())
    f = np.maximum(f, 0.0)
    score = _fast_score(f)
    if not math.isfinite(score):
        return f, score

    sigma = 0.12
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        cand = f.copy()
        r = rng.random()

        if r < 0.45:
            # block multiplicative perturbation
            choices = (1, 2, 3, 4, 6, 9, 14, 23)
            div = int(choices[int(rng.integers(0, len(choices)))])
            bs = max(1, n // div)
            st = int(rng.integers(0, max(1, n - bs + 1)))
            en = min(n, st + bs)
            cand[st:en] *= np.exp(rng.normal(0.0, sigma, size=en - st))
        elif r < 0.65:
            # peak-informed coordinate move
            c = _conv_self(f)
            peak = int(c.argmax())
            s = max(0, peak - n + 1)
            e = min(n, peak + 1)
            if e <= s:
                idx = int(rng.integers(0, n))
            else:
                idxs = np.arange(s, e, dtype=np.int64)
                idx = int(idxs[int(rng.integers(0, idxs.size))])
            if rng.random() < 0.55:
                cand[idx] *= np.exp(-sigma * 0.8)
            else:
                cand[idx] *= np.exp(sigma * 0.8)
        elif r < 0.8:
            t = rng.uniform(0.05, 0.35)
            cand = (1.0 - t) * f + t * f[::-1]
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            amount = f[i] * rng.uniform(0.05, 0.55)
            if i == j:
                continue
            if amount < f[i]:
                cand[i] -= amount
                cand[j] += amount

        cand = np.maximum(cand, 0.0)
        if cand.sum() <= 1e-12:
            continue
        cand = _normalize_sum(cand, S0)
        ns = _fast_score(cand)

        if ns < score - 1e-13:
            f = cand
            score = float(ns)
            last_improve = time.monotonic()
            sigma = min(0.35, sigma * 1.07)
        else:
            sigma = max(0.01, sigma * 0.985)
            if time.monotonic() - last_improve > 25.0:
                sigma = max(0.008, sigma * 0.85)

    return f, score

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a nonnegative coefficient list minimizing
    2*n*max(convolve(a,a)) / sum(a)^2.
    """
    budget_s = max(60.0, float(budget_s))
    rng = np.random.default_rng(int(seed))
    stop_time = time.monotonic() + budget_s - 25.0

    candidates = []
    best_f = None
    best_score = float("inf")

    def _consider(f):
        nonlocal best_f, best_score
        best_f, best_score = _add_candidate(f, candidates, best_f, best_score)

    # 1) quick broad initialization
    initial_budget = min(70.0, max(5.0, 0.05 * budget_s))
    for n in (256, 512, 1024, 2048):
        for cand in _make_batch(n, rng):
            _consider(cand)
    print(f"initial best={best_score:.6f}", flush=True)

    # 2) smooth optimization: smaller dimensions first
    smooth_budget = min(330.0, max(20.0, 0.27 * budget_s))
    smooth_deadline = time.monotonic() + smooth_budget

    if time.monotonic() < smooth_deadline:
        alloc = (0.35, 0.35, 0.30)
        t = time.monotonic()
        for idx, n in enumerate((128, 256, 512)):
            if time.monotonic() >= smooth_deadline:
                break
            rem = max(0.0, smooth_deadline - time.monotonic())
            child = min(smooth_deadline, time.monotonic() + rem * alloc[idx])
            if child <= time.monotonic() + 1.0:
                continue
            f, s = _optimize_softmax(n, child, rng)
            _consider(f)
            f, s = _optimize_cos(n, child, rng)
            _consider(f)
            t = time.monotonic()

    if time.monotonic() < smooth_deadline:
        # dedicate remaining smooth time to larger dimensions
        f, s = _optimize_cos(512, smooth_deadline, rng)
        _consider(f)
        f, s = _optimize_cos(1024, smooth_deadline, rng)
        _consider(f)

    print(f"after smooth best={best_score:.6f}", flush=True)

    # 3) LP improvement on the strongest candidates
    refine_reserve = min(180.0, max(20.0, 0.14 * budget_s))
    lp_deadline = time.monotonic() + max(0.0, (stop_time - time.monotonic()) - refine_reserve)

    candidates.sort(key=_candidate_key)
    selected = []
    seen_n = set()
    for f, s in candidates:
        if len(selected) >= 4:
            break
        n = int(f.size)
        if n in seen_n:
            continue
        if n > 1024:
            continue
        selected.append(f.copy())
        seen_n.add(n)

    if selected and time.monotonic() < lp_deadline:
        per = max(10.0, (lp_deadline - time.monotonic()) / float(len(selected)))
        for f0 in selected:
            child = min(lp_deadline, time.monotonic() + per)
            if child <= time.monotonic() + 2.0:
                break
            f, s = _lp_improve(f0, child, rng)
            _consider(f)
        print(f"after LP best={best_score:.6f}", flush=True)

    # 4) final stochastic refinement
    if best_f is None:
        best_f = np.ones(512, dtype=np.float64)
    final_deadline = stop_time
    best_f, best_score = _refine(best_f, final_deadline, rng)
    print(f"after refine best={best_score:.6f}", flush=True)

    # 5) sanity/verification and fallback
    seq = [float(x) for x in np.clip(np.asarray(best_f, dtype=np.float64), 0.0, 1000.0)]
    if (not seq) or any((not math.isfinite(x)) for x in seq):
        seq = [1.0] * 512

    try:
        fs = evaluate_sequence(seq)
        if not math.isfinite(fs):
            seq = [float(x) for x in np.ones(512)]
    except Exception:
        pass

    return seq
