def _autocorr(a):
    """Full linear autocorrelation b[k] = sum_j a[j] a[k-j]  (== np.convolve(a,a,'full'))."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    out = np.fft.irfft(fa * fa, fsize)
    return out[:size]


def _fscore(a):
    """Fast mirror of evaluate_sequence (assumes 0 <= a <= 1000, finite)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = _autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def _normalize(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 0.0:
        return a
    return a / s

def _smoothed_grad(a, beta):
    """Return (grad, maxb) where grad approximates d/da just_{k} exp(beta b[k])."""
    n = len(a)
    b = _autocorr(a)
    mb = b.max()
    E = np.exp(np.clip(beta * (b - mb), -60.0, 0.0))
    E = E / max(E.sum(), 1e-300)
    ar = a[::-1]
    grad = 2.0 * np.convolve(E, ar, mode="full")[n - 1:2 * n - 1]
    if grad.sum() <= 0:
        grad = np.ones(n)
    return grad, mb


def _descend_mix(a0, rng, mult_steps, block_steps, beta_lo, beta_hi, lr0,
                 end_time, quiet_every=2000):
    """Hybrid multiplicative smoothed-descent (exploit) + annealed SA block moves
    (explore).  Returns (best_a, best_score).  Uses only fast ops (no big matrix).
    """
    a = _normalize(a0.copy())
    n = len(a)
    best = a.copy()
    bs = _fscore(a)
    lr = lr0
    temp = 0.02
    stuck = 0
    it = 0

    while time.monotonic() < end_time:
        # ----- exploit: multiplicative smoothed-max descent (accept only if better)
        for _ in range(mult_steps):
            if time.monotonic() > end_time:
                break
            it += 1
            beta = beta_lo + (beta_hi - beta_lo) * min(1.0, it / 20000.0)
            grad, mb = _smoothed_grad(a, beta)
            step = lr * grad / max(mb, 1e-30)
            cand = a * np.exp(-np.clip(step, -5.0, 5.0))
            cand = _normalize(cand)
            cs = _fscore(cand)
            if np.isfinite(cs) and cs < bs:
                a, bs = cand, cs
                stuck = 0
                if bs < np.inf:
                    lr = min(lr * 1.02, 1.5)
            else:
                stuck += 1
                if stuck > 400:
                    lr = max(lr * 0.5, 1e-3)
        if not np.isfinite(bs):
            break

        # ----- explore: simulated-annealing random block rescales
        temp = max(1e-4, temp * 0.9)
        wmin = max(2, n // 512)
        wmax = max(wmin + 1, n // 64)
        for _ in range(block_steps):
            if time.monotonic() > end_time:
                break
            it += 1
            w = int(rng.integers(wmin, wmax + 1))
            st = int(rng.integers(0, max(1, n - w)))
            cand = a.copy()
            cand[st:st + w] *= float(np.exp(rng.normal(0.0, 0.25)))
            cand = _normalize(cand)
            cs = _fscore(cand)
            if not np.isfinite(cs):
                continue
            if cs < bs or rng.random() < np.exp(-(cs - bs) / max(temp, 1e-6)):
                a, bs = cand, cs

        if bs < _bs_global[0]:
            _bs_global[0], _bs_global[1] = bs, a.copy()
        if it % quiet_every < mult_steps:
            print("  descend n=%d score=%.6f (lr=%.4f temp=%.4f)" % (n, bs, lr, temp))

    if bs < _bs_global[0]:
        _bs_global[0], _bs_global[1] = bs, a.copy()
    return best, bs

# State-of-the-art LP direction blend

def _convmat_csc(a):
    """Build the (2n-1) x n convolution matrix M, M[k,j] = a[k-j] (0<=k-j<n), as CSC."""
    n = len(a)
    L = lil_matrix((2 * n - 1, n))
    for j in range(n):
        L[j:j + n, j] = a
    return L.tocsc()


def _lp_direction(a, M, time_limit=12.0):
    """Solve max sum(g) s.t. convolve(a,g) <= M, g>=0.  Return normalized g or None."""
    if not _HAVE_CPX:
        return None
    n = len(a)
    Mmat = _convmat_csc(a)
    g = cp.Variable(n, nonneg=True)
    cons = [Mmat @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), cons)
    try:
        prob.solve(solver=cp.SCS, eps=1e-7, max_iters=200000,
                   time_limit_epoch=time.monotonic() + time_limit, verbose=False)
    except Exception:
        return None
    if g.value is None:
        return None
    gv = np.asarray(g.value, dtype=np.float64)
    if not np.all(np.isfinite(gv)):
        return None
    gv = np.clip(gv, 0.0, None)
    if gv.sum() <= 1e-12:
        return None
    return gv / gv.sum()


def _lp_blend_line(a, g):
    """Pick t in [0,1] minimizing score of (1-t)*a + t*g.  Return (best_a, best_score, t)."""
    best_a, best_s, best_t = a, _fscore(a), 0.0
    cand_ts = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    for t in cand_ts:
        c = (1.0 - t) * a + t * g
        c = _normalize(c)
        s = _fscore(c)
        if np.isfinite(s) and s < best_s:
            best_a, best_s, best_t = c, s, t
    # refine around best_t
    if 0.0 < best_t < 1.0:
        lo, hi = max(0.0, best_t - 0.15), min(1.0, best_t + 0.15)
        for t in np.linspace(lo, hi, 9):
            c = _normalize((1.0 - t) * a + t * g)
            s = _fscore(c)
            if np.isfinite(s) and s < best_s:
                best_a, best_s, best_t = c, s, t
    return best_a, best_s, best_t


def _descend_lp(a0, dest_time, lp_budget):
    """Run SOTA LP-blend iteration (monotone) from a non-uniform seed a0 (sum=1)."""
    a = _normalize(a0.copy())
    n = len(a)
    bs = _fscore(a)
    no_improve = 0
    while time.monotonic() < dest_time:
        b = _autocorr(a)
        M = b.max()
        nt_start = time.monotonic()
        g = _lp_direction(a, M, time_limit=lp_budget)
        if g is None:
            # cheap fallback so we never fully stall
            grad, mb = _smoothed_grad(a, 30.0)
            g = _normalize(np.exp(-0.2 * grad / max(mb, 1e-30)))
        na, ns, _t = _lp_blend_line(a, g)
        if ns < bs:
            a, bs = na, ns
            no_improve = 0
        else:
            no_improve += 1
            a = _normalize(a + 1e-4 * (g - a))   # nudge to escape exact fixed point
            bs = _fscore(a)
        if bs < _bs_global[0]:
            _bs_global[0], _bs_global[1] = bs, a.copy()
        if (time.monotonic() - nt_start) < 0.5:
            print("  LP n=%d M=%.4f score=%.6f" % (n, M, bs))
        if no_improve > 40:
            break
    if bs < _bs_global[0]:
        _bs_global[0], _bs_global[1] = bs, a.copy()
    return a, bs

# Global bookkeeping (module-level, no closures)

_bs_global = [np.inf, np.ones(64)]


def _make_seeds(rng, n, kinds):
    seeds = []
    for k in range(len(kinds)):
        if kinds[k] == "randpos":
            v = np.abs(rng.normal(1.0, 0.9, n)) + 0.02
        elif kinds[k] == "rand01":
            v = (rng.random(n) < 0.5).astype(np.float64)
            if v.sum() == 0:
                v[0] = 1.0
        elif kinds[k] == "twoval":
            v = np.where(rng.random(n) < 0.5, 0.15, 1.0)
        elif kinds[k] == "smooth":
            t = np.arange(n)
            v = 1.0 + 0.6 * np.sin(2 * np.pi * t / n * (1 + k)) + \
                0.4 * np.cos(2 * np.pi * t / n * (2 + k))
            v = np.abs(v) + 0.05
        else:
            v = np.abs(rng.normal(1.0, 0.7, n)) + 0.02
        seeds.append(_normalize(v))
    return seeds


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Top-level search entry point.  Runs time-sliced across several lengths and
    seeds: SOTA LP-blend (guaranteed monotone) at moderate lengths, plus fast
    smoothed-descent/SA for large lengths and the provided height_sequence_1.
    Returns the best non-negative sequence found as a plain list of floats.
    """
    global _bs_global
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 8.0)

    _bs_global = [np.inf, np.ones(64)]

    # ---- jobs: (kind, n, seedvec).  kind in {'lp','mix'} ----
    jobs = []
    # LP-capable lengths (memory-safe, guaranteed monotone from non-uniform seeds)
    for n in (2048, 4096):
        for s in _make_seeds(rng, n, ["randpos", "rand01", "twoval"]):
            jobs.append(("lp", n, s))
    # fast large-length hybrids
    for n in (4096, 8192, 12288):
        for s in _make_seeds(rng, n, ["randpos", "smooth", "twoval"]):
            jobs.append(("mix", n, s))

    # provided construction (its own odd length) as a quick hybrid + LP attempt
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        base = np.clip(base, 0.0, 1000.0)
        if base.ndim == 1 and base.size >= 64 and np.all(np.isfinite(base)) and base.sum() > 0.01:
            jobs.append(("mix", len(base), _normalize(base.copy())))
            if len(base) <= 5000:
                jobs.append(("lp", len(base), _normalize(base.copy() + 0.01 * rng.normal(0, 1, len(base)))))
    except Exception:
        pass

    t0 = time.monotonic()
    print("Starting search: %d jobs, budget %.1fs" % (len(jobs), float(budget_s)))

    slices = 24.0          # seconds per job per round
    rounds = 0
    while time.monotonic() < deadline and rounds < 4000:
        rounds += 1
        for idx, (kind, n, s) in enumerate(jobs):
            if time.monotonic() >= deadline:
                break
            job_start = time.monotonic()
            if kind == "lp":
                dest = min(deadline, job_start + slices)
                _descend_lp(s, dest, lp_budget=10.0)
            else:
                dest = min(deadline, job_start + slices)
                _descend_mix(s, rng, mult_steps=400, block_steps=120,
                             beta_lo=8.0, beta_hi=60.0, lr0=0.12,
                             end_time=dest, quiet_every=10 ** 9)
            # refresh seeds occasionally to keep exploring new basins (cheap)
            if rounds % 6 == 0:
                if kind == "lp":
                    jobs[idx] = ("lp", n, _make_seeds(rng, n, ["randpos"])[0])
                else:
                    jobs[idx] = ("mix", n, _make_seeds(rng, n, ["randpos"])[0])

    # ---- optionally one final LP polish of the global best at its own length ----
    if _bs_global[0] < np.inf and time.monotonic() < deadline - 20 and _HAVE_CPX:
        gb = _normalize(_bs_global[1].copy())
        gn = len(gb)
        if gn <= 4096:
            print("Polishing global best (n=%d, score=%.6f) with LP..." % (gn, _bs_global[0]))
            _descend_lp(gb, deadline - 5, lp_budget=8.0)

    best = _normalize(np.asarray(_bs_global[1], dtype=np.float64).copy())
    # authoritative confirmation / repair
    blist = [float(max(0.0, min(1000.0, x))) for x in best]
    final = float(evaluate_sequence(blist))
    if not np.isfinite(final):
        # guaranteed-valid fallback (uniform always gives a finite score = 2)
        fb = np.ones(256, dtype=np.float64)
        blist = [float(x) for x in fb]
        final = float(evaluate_sequence(blist))
    print("Final best score (authoritative evaluate_sequence): %.6f  (n=%d)" % (final, len(blist)))
    _bs_global = [final, np.asarray(blist, dtype=np.float64)]
    return list(blist)


if __name__ == "__main__":  # pragma: no cover
    propose_candidate(42, 60)
