def build_constraint_matrix_sparse(n, a):
    """Build sparse constraint matrix for the LP."""
    # A[k, i] = a[i] if max(0, k-n+1) <= i <= min(n-1, k)
    row_indices = []
    col_indices =[]
    values = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            row_indices.append(k)
            col_indices.append(i)
            values.append(a[i])
    A = csr_matrix((values, (row_indices, col_indices)), shape=(2*n-1, n))
    return A

def solve_lp_step(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M elementwise, b >= 0."""
    n = len(a)
    c = -np.ones(n)  # minimize -sum(b)
    A = build_constraint_matrix_sparse(n, a)
    b_vec = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                     options={'presolve': True})
    if result.success:
        return result.x
    return None

    """Compute the score directly (for tracking, not for final evaluation)."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv) / (s ** 2)

        # Update

        M = M_new

        a_new[start:start + width] *= scale
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-6:
            continue
        a_new /= s
        val = np.max(np.convolve(a_new, a_new))
        if val < best_val:
            best_val = val
            a = a_new
    return a, best_val


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search function. Returns the best sequence found within budget."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_sequence = None
    best_score = np.inf

    def update_best(seq, label=""):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = np.array(seq, dtype=np.float64).copy()
            print(f"  NEW BEST [{label}]: score={score:.6f}, n={len(seq)}")
        return score

    # --- Try parent sequence ---
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000)
        s = np.sum(parent)
        if s > 0.01:
            p_norm = parent / s
            score = update_best(p_norm, "parent")
            print(f"  Parent: n={parent.size}, score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"  No parent available: {e}")

    # --- LP-based optimization for various n and initializations ---
    lp_configs = [
        (200, 60), (400, 50), (600, 45), (1000, 35), (1500, 25), (2000, 20),
    ]

    for n, max_it in lp_configs:
        if time.monotonic() >= deadline:
            break
        print(f"\n--- LP optimization: n={n}, max_iters={max_it} ---")

        # Init 1: flat convolution square root
        if time.monotonic() >= deadline:
            break
        a = flat_conv_init(n)
        result_a, _ = iter_lp_refine(a, max_iters=max_it, n=n, deadline=deadline, verbose=False)
        update_best(result_a, f"flat_n{n}")

        # Init 2: cosine window
        if time.monotonic() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        a2 = 1.0 + 0.5 * np.cos(2 * np.pi * i / max(1, n - 1))
        a2 = np.clip(a2, 0, None)
        s2 = np.sum(a2)
        if s2 > 0.01:
            a2 /= s2
            result_a, _ = iter_lp_refine(a2, max_iters=max_it, n=n, deadline=deadline)
            update_best(result_a, f"cos_n{n}")

        # Init 3: random
        if time.monotonic() >= deadline:
            break
        a3 = rng.random(n) + 0.1
        a3 /= np.sum(a3)
        result_a, _ = iter_lp_refine(a3, max_iters=max_it // 2, n=n, deadline=deadline)
        update_best(result_a, f"rand_n{n}")

    # --- Large-n block perturbation with flat init ---
    large_n_options = [3000, 5000, 8000, 12000]
    for n in large_n_options:
        if time.monotonic() >= deadline:
            break
        print(f"\n--- Block perturb: n={n} (flat init) ---")
        a = flat_conv_init(n)
        # Quick local refinement
        a, _ = local_perturb_refine(a, time.monotonic() + min(60, max(10, (deadline - time.monotonic()) // 4)), rng,
                                    n_perturb=5000)
        update_best(a, f"block_n{n}")

    # --- Final local refinement on best found ---
    if best_sequence is not None and time.monotonic() < deadline:
        print(f"\n--- Final refinement (n={len(best_sequence)}) ---")
        remaining = deadline - time.monotonic()
        refined, _ = local_perturb_refine(best_sequence, deadline, rng,
                                          n_perturb=max(1000, int(remaining * 500)))
        update_best(refined, "final_refine")

    # --- Fallback ---
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"Fallback uniform n=128: score={best_score:.6f}")

    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_sequence)} ===")
    return [float(x) for x in best_sequence]
