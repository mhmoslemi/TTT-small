import numpy as np
import time

"""
Minimizes 2n*max(convolve(a,a))/(sum(a))^2 using soft-gradient descent
with adaptive temperature and learning rate, multiple diverse initializations,
and iterative refinement. The gradient of the soft-max objective is computed
via cross-correlation (O(n log n) per iteration using FFT).
"""


def _compute_gradient(a, w, n):
    """Compute g[j] = 2 * sum_k w[k] * a[k-j] for j=0..n-1 using FFT."""
    # We need cross-correlation: r[j] = sum_k w[k] * a[k-j]
    # This equals: correlate(w, a) at lag j
    # Using FFT: ifft(conj(FFT(w)) * FFT(a)) gives the cross-correlation
    L = 1
    while L < 4 * n:
        L *= 2
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    r = np.fft.irfft(np.conj(W) * A, L)
    return 2.0 * r[:n]


def _get_convolution_max(a):
    """Compute max of a*a convolution."""
    c = np.convolve(a, a)
    return c.max()


def _soft_gradient_step(a, T, eta):
    """One step of soft-gradient descent.”
    Returns (updated_a, c_max)."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    # Softmax weights (numerically stable)
    c_shifted = c - c_max
    w = np.exp(c_shifted / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    # Gradient via cross-correlation
    g = _compute_gradient(a, w, n)
    # Update
    a = a - eta * g
    # Project onto feasible set
    np.maximum(a, 0.0, out=a)
    s = a.sum()
    if s < 1e-300:
        a[:] = 1.0 / n
    else:
        a /= s
    return a, c_max


def _optimize_sequence(a, rng, time_limit=60.0):
    """Run soft-gradient descent with adaptive parameters on a single sequence."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    a = a / a.sum()
    n = len(a)

    deadline = time.monotonic() + time_limit
    best_a = a.copy()
    best_cmax = _get_convolution_max(a)

    # Adaptive parameters
    T = 0.003  # temperature (small = focused on max peak)
    eta = 0.4  # learning rate
    no_improve = 0
    patience = 3000  # iterations without improvement before adapting

    iteration = 0
    while time.monotonic() < deadline:
        # Process in mini-batches for efficiency
        batch = min(100, max(1, int((deadline - time.monotonic()) * 1000 / 0.1)))
        for _ in range(batch):
            a, c_max = _soft_gradient_step(a, T, eta)
            iteration += 1
            if c_max < best_cmax - 1e-14:
                best_cmax = c_max
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        if time.monotonic() >= deadline:
            break

        # Adaptive strategy every ~5000 iterations
        if iteration % 5000 < batch:
            if no_improve > patience:
                # Escalate: increase temperature, reduce learning rate, perturb
                T = min(T * 2.5, 0.5)
                eta = max(eta * 0.6, 0.05)
                # Random perturbation to escape local minimum
                pertur = 0.03 * (1.0 + 0.1 * (T - 0.003))
                a += pertur * rng.standard_normal(n)
                np.maximum(a, 0.0, out=a)
                ss = a.sum()
                if ss > 1e-300:
                    a /= ss
                no_improve = 0
            elif no_improve > patience // 3:
                # Gentle adaptation
                T = min(T * 1.3, 0.3)
                eta = max(eta * 0.9, 0.1)

    return best_a


def _try_block_refinement(a, rng, time_limit=30.0):
    """Local block perturbation refinement (complements gradient method)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    a = a / a.sum()
    n = len(a)
    deadline = time.monotonic() + time_limit
    best_a = a.copy()
    best_cmax = _get_convolution_max(a)

    while time.monotonic() < deadline:
        # Random block perturbation
        width = max(1, int(rng.integers(n // 200, n // 8)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.15, stop - start))
        np.maximum(candidate, 0.0, out=candidate)
        cs = candidate.sum()
        if cs > 1e-300:
            candidate /= cs
            cmax = _get_convolution_max(candidate)
            if cmax < best_cmax:
                best_cmax = cmax
                best_a = candidate.copy()

    return best_a


def _make_u_shape(n, sharpness=15.0, rng=None):
    """Create a U-shaped (bathtub) initialization."""
    t = np.linspace(0.0, 1.0, n)
    base = 1.0 / (1.0 + sharpness * 4.0 * t * (1.0 - t))
    if rng is not None:
        base += 0.02 * rng.random(n)
    return np.maximum(base, 1e-6)


def _make_bimodal(n, gap_ratio=0.3, rng=None):
    """Create a bimodal initialization (two peaks)."""
    a = np.ones(n)
    center = n // 2
    half_gap = int(gap_ratio * center)
    a[center - half_gap:center + half_gap] = 0.1
    if rng is not None:
        a += 0.02 * rng.random(n)
    return np.maximum(a, 1e-6)
