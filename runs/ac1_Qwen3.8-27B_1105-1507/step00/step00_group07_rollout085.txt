def optimize_subgradient(a_init, n_steps, lr_start, lr_end, eval_every, seed):
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        return a, np.inf
    a = a / s
    
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_score = np.inf
    
    # Momentum
    v = np.zeros(n)
    
    for step in range(n_steps):
        frac = step / max(1, n_steps - 1)
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Compute autocorrelation
        r = full_conv(a)
        r_max = r.max()
        
        # Get weights: top-k lags with exponential weighting
        # Use a "soft" version: weight_k = exp(alpha * (r_k - r_max))
        alpha = 50.0 * (1.0 + 5.0 * frac)  # increase focus over time
        weights_r = np.exp(alpha * (r - r_max))
        weights_r = weights_r / weights_r.sum()
        
        # Gradient: grad[j] = 2 * sum_k weights_r[k] * a[k-j]
        # = 2 * (weights_r cross-correlate a)[j]
        grad = grad_weighted_autocorr(a, weights_r)
        
        # Momentum step
        v = 0.9 * v - lr * grad
        a = a + v
        
        # Project
        a = np.maximum(a, 0.0)
        s_new = a.sum()
        if s_new < 1e-15:
            a = best_a.copy()
            v = np.zeros(n)
            continue
        a = a / s_new
        
        # Noise
        if rng.random() < 0.01:
            a += 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
        
        # Evaluate
        if (step + 1) % eval_every == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score
