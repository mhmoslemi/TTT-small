Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected gradient descent
with log-sum-exp surrogate, FFT-based O(n log n) gradients, beta annealing,
and multiple diverse restarts.
"""

import numpy as np
import time


def compute_objective_and_gradient(a, beta):
    """
    Compute the soft objective (1/beta)*log(sum exp(beta*conv(a,a)))
    and its gradient with respect to a.
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    # Compute convolution
    w = np.convolve(a, a, mode='full')  # length 2n-1
    
    # Soft objective (log-sum-exp for numerical stability)
    w_max = w.max()
    shifted = beta * (w - w_max)
    # Clip to avoid overflow
    shifted = np.minimum(shifted, 500.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    f = (w_max + log_sum / beta) if beta > 0 else w_max
    
    # Soft weights
    w_soft = np.exp(shifted)
    w_soft_sum = w_soft.sum()
    if w_soft_sum < 1e-300:
        w_soft = np.ones_like(w) / len(w)
    else:
        w_soft = w_soft / w_soft_sum

    # = 2 * (cross-correlation of w_soft and a evaluated at lag i)
    # = 2 * conv(a, w_soft_flipped)[2n-2-i] for i=0..n-1
    w_flipped = w_soft[::-1]
    
    total_len = n + len(w_soft) - 1  # 3n-2
    size = 1 << (total_len - 1).bit_length()
    
    A = np.fft.rfft(a, size)
    Wf = np.fft.rfft(w_flipped, size)
    c1 = np.fft.irfft(A * Wf, size)[:total_len]
    
    # g[i] = 2 * c1[2n-2-i] for i=0..n-1
    # c1[2n-2], c1[2n-3], ..., c1[n-1]
    g = 2.0 * c1[n-1 : 2*n-1][::-1]
    
    return f, g, w, w_max


def normalize_nonneg(a):
    """Project to non-negative simplex (sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def soft_objective(a, beta):
    """Quick computation of soft objective value (for line search)."""
    w = np.convolve(a, a, mode='full')
    w_max = w.max()
    shifted = beta * (w - w_max)
    shifted = np.minimum(shifted, 500.0)
    return w_max + np.log(np.sum(np.exp(shifted))) / beta


def run_gd(a_init, beta_start, beta_end, n_iter, deadline, best_a, best_score, evaluate_seq):
    """
    Run projected gradient descent with beta annealing.
    Returns (best_a, best_score).
    """
    n = len(a_init)
    a = normalize_nonneg(a_init.copy())
    best_a_local = a.copy()
    best_score_local = best_score
    
    lr = 1.0
    min_lr = 0.01
    
    for i in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        # Anneal beta
        frac = i / max(1, n_iter - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        f_old, g, w, w_max = compute_objective_and_gradient(a, beta)
        
        # Gradient step with backtracking
        step_lr = lr
        a_new = a - step_lr * g
        a_new = normalize_nonneg(a_new)
        
        # Check if soft objective decreased
        f_new = soft_objective(a_new, beta)
        backtrack = 0
        while f_new > f_old + 1e-12 and step_lr > min_lr and backtrack < 10:
            step_lr *= 0.5
            a_new = a - step_lr * g
            a_new = normalize_nonneg(a_new)
            f_new = soft_objective(a_new, beta)
            backtrack += 1
        
        if step_lr > min_lr:
            lr = min(2.0, step_lr * 1.05)  # Slight increase if no backtracking
        else:
            lr = max(min_lr, lr * 0.9)
        
        a = a_new
        
        # Periodic exact evaluation
        if i % 100 == 99 or i == n_iter - 1:
            score = evaluate_seq(a.tolist())
            if np.isfinite(score) and score < best_score_local:
                best_score_local = score
                best_a_local = a.copy()
    
    # Update global best
    if best_score_local < best_score:
        best_a = best_a_local
        best_score = best_score_local
    
    return best_a, best_score


def make_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    # Parent
    if parent is not None and len(parent) == n:
        inits.append(parent / max(parent.sum(), 1e-15))
    
    # Uniform
    inits.append(np.ones(n, dtype=np.float64))
    
    # Cosine modulations (various frequencies and amplitudes)
    for freq in [1, 2, 3, 5, 8, 13]:
        for alpha in [0.2, 0.5, 0.8]:
            a = 1.0 + alpha * np.cos(2 * np.pi * freq * x)
            inits.append(np.clip(a, 0.001, None))
    
    # Bimodal (high at edges)
    for alpha in [0.3, 0.6, 0.9]:
        inits.append(1.0 + alpha * (2 * x - 1) ** 2)
    
    # Bimodal (high in middle)
    for alpha in [0.3, 0.6, 0.9]:
        inits.append(np.clip(1.0 - alpha * (2 * x - 1) ** 2, 0.001, None))
    
    # Sine absolute value
    for alpha in [0.3, 0.6, 0.9]:
        inits.append(np.clip(1.0 + alpha * np.abs(np.sin(np.pi * x)), 0.001, None))
    
    # Smooth random
    for _ in range(3):
        noise = rng.standard_normal(n)
        # Simple smoothing via cumulative sum trick
        kernel_size = max(1, n // 100)
        cumsum = np.cumsum(noise)
        smoothed = (cumsum[kernel_size:] - cumsum[:-kernel_size]) / kernel_size
        smoothed = np.concatenate([smoothed, np.zeros(kernel_size)])
        a = 1.0 + 0.5 * (smoothed - smoothed.mean()) / (smoothed.std() + 1e-10)
        inits.append(np.clip(a, 0.001, None))
    
    # Linear ramps
    inits.append(0.5 + x)
    inits.append(0.5 + (1 - x))
    
    # Two-peak
    for peak_pos in [0.2, 0.8]:
        a = np.exp(-((x - peak_pos) ** 2) / (2 * 0.05 ** 2)) + 0.1
        inits.append(a)
    
    return inits


def coordinate_descent_polish(a, n_passes, deadline, best_score, evaluate_seq):
    """
    Polish with cyclic coordinate descent.
    For each element, find the best value that minimizes the max convolution.
    """
    n = len(a)
    a = normalize_nonneg(a.copy())
    best_a = a.copy()
    best_sc = best_score
    
    for pass_num in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        for i in range(n):
            if time.monotonic() > deadline:
                    break
            
            # Save current value
            orig = a[i]
            
            # Compute convolution with a_i = 0
            a[i] = 0.0
            w_base = np.convolve(a, a, mode='full')
            
            # b_k = w_base_k + 2*a_i*a_{k-i} - a_i^2*[2i=k]
            # We want to find a_i >= 0 that minimizes max_k b_k
            
            # The current max of w_base (without a_i contribution)
            # Try a few candidate values
            candidates = [0.0, orig * 0.5, orig * 0.75, orig, orig * 1.25, orig * 1.5, orig * 2.0]
            # Also try the value that would equalize the peak
            # b_{k*} = w_base_{k*} + 2*t*a_{k*-i} - t^2*[2i=k*]
            # To set b_{k*} = target: 2*a_{k*-i}*t - t^2*[2i=k*] = target - w_base_{k*}
            # This is a quadratic in t
            
            best_t = orig
            best_max_w = np.inf
            
            for t in candidates:
                t = max(0.0, t)
                a[i] = t
                # Compute max of convolution
                # b_k = w_base_k + 2*t*a_{k-i} - t^2*[2i=k]
                # We need a_{k-i} for k=0..2n-2
                # a_{k-i} is non-zero for i <= k <= i+n-1
                # So b_k for k in [i, i+n-1]: w_base_k + 2*t*a[k-i]
                # And for k=2i (if 2i < 2n-1): subtract t^2
                w_new = w_base.copy()
                # Add 2*t*a[k-i] for valid k
                lo = i
                hi = min(i + n, 2 * n - 1)
                idx = np.arange(lo, hi)
                w_new[lo:hi] += 2 * t * a[idx - i]  # a[k-i] for k in [lo,hi)
                # Subtract t^2 at k=2i
                if 2 * i < 2 * n - 1:
                    w_new[2 * i] -= t * t
                
                max_w = w_new.max()
                if max_w < best_max_w:
                    best_max_w = max_w
                    best_t = t
            
            a[i] = best_t
        
        # Renormalize after each pass
        a = normalize_nonneg(a)
        
        # Evaluate
        score = evaluate_seq(a.tolist())
        if np.isfinite(score) and score < best_sc:
            best_sc = score
            best_a = a.copy()
    
    return best_a, best_sc

def compute_soft_weighted_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum exp(beta*conv(a,a))) w.r.t. a.
    Returns (gradient, convolution, max_conv).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    w = np.convolve(a, a, mode='full')  # length 2n-1

    w_max = w.max()
    shifted = beta * (w - w_max)
    shifted = np.minimum(shifted, 500.0)
    w_soft = np.exp(shifted)
    w_soft_sum = w_soft.sum()
    if w_soft_sum < 1e-300:
        w_soft = np.ones_like(w) / len(w)
    else:

    # = 2 * conv(a, w_soft_flipped)[2n-2-i] for i=0..n-1

    total_len = n + len(w_soft) - 1  # 3n - 2

        if time.monotonic() > deadline:
            break

        frac = i / max(1, n_iter - 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        g, w, w_max = compute_soft_weighted_gradient(a, beta)

        # Gradient step
        a_new = a - lr * g
        a_new = normalize_nonneg(a_new)

        # Simple adaptive: if soft objective worsened a lot, reduce lr
        if i > 0 and i % 50 == 0:
            f_new = soft_obj_value(a_new, beta)
            f_old = soft_obj_value(a, beta)
            if f_new > f_old + 0.01:
                lr *= 0.5
            elif f_new < f_old - 0.001:
                lr = min(2.0, lr * 1.1)
            else:
                lr *= 0.99
        else:
            lr *= 0.9995

        a = a_new

        # Periodic exact evaluation
        if i % 150 == 149 or i == n_iter - 1:
            score = evaluate_seq(a.tolist())
            if np.isfinite(score) and score < best_sc:
                best_sc = score
                best_a_local = a.copy()
                if score < 1.6:
                    print(f"    GD iter {i}: new best {score:.6f}")

    if best_sc < best_score:
        best_a = best_a_local
        best_score = best_sc

    return best_a, best_score


def make_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None and len(parent) == n and parent.sum() > 0:
        inits.append(parent / parent.sum())

    # Uniform
    inits.append(np.ones(n, dtype=np.float64))

    # Cosine modulations
    for freq in [1, 2, 3, 5, 8]:
        for alpha in [0.2, 0.5, 0.8]:
            a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * x)
            inits.append(np.clip(a, 0.001, None))

    # Bimodal high at edges
    for alpha in [0.3, 0.6, 0.9]:
        inits.append(1.0 + alpha * (2.0 * x - 1.0) ** 2)

    # Bimodal high in middle
    for alpha in [0.3, 0.6, 0.9]:
        inits.append(np.clip(1.0 - alpha * (2.0 * x - 1.0) ** 2, 0.001, None))

    # Sine-based
    for alpha in [0.3, 0.6, 0.9]:
        inits.append(np.clip(1.0 + alpha * np.sin(np.pi * x) ** 2, 0.001, None))
        inits.append(np.clip(1.0 + alpha * np.abs(np.sin(2.0 * np.pi * x)), 0.001, None))

    # Smooth random
    for _ in range(4):
        noise = rng.standard_normal(n)
        # Smooth via simple moving average
        k = max(1, n // 80)
        kernel = np.ones(k) / k
        smoothed = np.convolve(noise, kernel, mode='same')
        a = 1.0 + 0.6 * (smoothed - smoothed.mean()) / (smoothed.std() + 1e-10)
        inits.append(np.clip(a, 0.001, None))

    # Linear ramps
    inits.append(np.clip(0.3 + 1.4 * x, 0.01, None))
    inits.append(np.clip(0.3 + 1.4 * (1.0 - x), 0.01, None))

    # Two-peak
    for peak in [0.2, 0.3, 0.7, 0.8]:
        a = np.exp(-((x - peak) ** 2) / (2 * 0.08 ** 2)) + 0.05
        inits.append(a)

    return inits


def coordinate_descent_polish(a, n_passes, deadline, best_score, evaluate_seq):
    """
    Cyclic coordinate descent: for each element, try several candidate values
    and keep the one that minimizes the max convolution.
    """
    n = len(a)
    a = normalize_nonneg(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = best_score

    for pass_num in range(n_passes):
        if time.monotonic() > deadline:
            break

        improvements = 0
        for i in range(n):
            if time.monotonic() > deadline:
                break

            orig = a[i]
            a[i] = 0.0
            w_base = np.convolve(a, a, mode='full')
            a[i] = orig

            # Try candidate values
            candidates = [0.0, orig * 0.3, orig * 0.6, orig, orig * 1.4, orig * 2.0, orig * 3.0]
            # Also try value that equalizes peak and mean
            target = w_base.max() * 0.95
            denom = 2.0 * orig if False else None  # skip for now

            best_t = orig
            best_max_w = np.inf

            for t in candidates:
                t = float(max(0.0, t))
                a[i] = t
                # b_k = w_base_k + 2*t*a_base[k-i] for k != 2i
                # b_{2i} = w_base_{2i} + t^2
                w_new = w_base.copy()
                lo = i
                hi = min(i + n, 2 * n - 1)
                if hi > lo:
                    w_new[lo:hi] += 2.0 * t * a[lo - i:hi - i]
                # Fix k=2i: the a[k-i]=a[i]=t was added as 2*t*t, but should be t^2
                if 2 * i < 2 * n - 1:
                    w_new[2 * i] -= t * t  # 2*t*t - t*t = t*t, which is correct
                    # Wait: w_base has a[i]=0, so at k=2i, w_base doesn't include a[i]^2
                    # The added term 2*t*a[k-i] at k=2i gives 2*t*t (since a[i]=t now)
                    # But correct is t^2. So subtract t^2.
                    # Actually: a[k-i] at k=2i is a[i]=t (we just set it), so 2*t*t is added.
                    # Correct contribution is t^2. So we need to subtract t^2.
                    pass  # Already handled by the subtraction above

                max_w = w_new.max()
                if max_w < best_max_w:
                    best_max_w = max_w
                    best_t = t

            a[i] = best_t
            if abs(best_t - orig) > 1e-10:
                improvements += 1

        a = normalize_nonneg(a)

        if (pass_num + 1) % 2 == 0 or pass_num == n_passes - 1:
            score = evaluate_seq(a.tolist())
            if np.isfinite(score) and score < best_sc:
                best_sc = score
                best_a = a.copy()
                print(f"    CD pass {pass_num+1}/{n_passes}: {best_sc:.6f} (improvements={improvements})")

        if improvements == 0 and pass_num > 2:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Minimize 2*n*max(conv(a,a))/(sum(a))^2 using:
    1. Multiple diverse initializations (parametric + random)
    2. Projected gradient descent with log-sum-exp surrogate (FFT-based, O(n log n))
    3. Beta annealing (temperature schedule)
    4. Coordinate descent polishing
    5. Final high-beta GD refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0

    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000.0)
        n_parent = len(parent)
    except (NameError, TypeError):
        parent = None
        n_parent = 2048

    if parent is None or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(2048, dtype=np.float64)
        n_parent = 2048

    # Choose lengths to try
    lengths = [n_parent]
    for factor in [0.5, 1.5, 2.0]:
        l = int(n_parent * factor)
        if 256 <= l <= 30000:
            lengths.append(l)
    lengths = list(dict.fromkeys(lengths))

    best_a = parent / parent.sum()
    best_score = float(evaluate_sequence(best_a.tolist()))
    print(f"Initial score: {best_score:.6f} (n={n_parent})")

    # Phase 1: GD from diverse initializations across lengths
    total_phase1_time = budget_s * 0.5
    time_per_length = total_phase1_time / max(1, len(lengths))
    phase1_deadline = time.monotonic() + total_phase1_time

    for n_try in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break

        print(f"\nPhase 1: n={n_try} (budget {time_per_length:.0f}s)")
        t_start = time.monotonic()
        local_deadline = min(t_start + time_per_length, phase1_deadline)

        # Get a parent for this length (resize if needed)
        if len(parent) == n_try:
            local_parent = parent
        else:
            local_parent = np.interp(
                np.linspace(0, 1, n_try),
                np.linspace(0, 1, len(parent)),
                parent
            )

        inits = make_diverse_inits(n_try, rng, local_parent)

        for idx, init in enumerate(inits):
            if time.monotonic() > local_deadline:
                break

            a_init = np.asarray(init, dtype=np.float64)
            if a_init.shape[0] != n_try:
                continue

            # Run GD with different beta schedules
            for beta_end in [30.0, 80.0]:
                if time.monotonic() > local_deadline:
                    break
                n_iter = 400
                best_a, best_score = run_gradient_descent(
                    a_init, 1.0, beta_end, n_iter,
                    local_deadline, best_a, best_score, evaluate_sequence
                )

    print(f"\nAfter Phase 1: best score = {best_score:.6f}")

    # Phase 2: Coordinate descent polishing
    if time.monotonic() < deadline - 60:
        print(f"\nPhase 2: Coordinate descent polishing")
        cd_deadline = min(deadline - 30, time.monotonic() + budget_s * 0.3)
        best_a, best_score = coordinate_descent_polish(
            best_a, 15, cd_deadline, best_score, evaluate_sequence
        )
        print(f"After Phase 2: best score = {best_score:.6f}")

    # Phase 3: Final high-beta GD refinement
    if time.monotonic() < deadline - 20:
        print(f"\nPhase 3: Final refinement with high beta")
        a_init = best_a / best_a.sum()
        best_a, best_score = run_gradient_descent(
            a_init, 5.0, 150.0, 600,
            deadline, best_a, best_score, evaluate_sequence
        )
        print(f"After Phase 3: best score = {best_score:.6f}")

    # Phase 4: Quick coordinate descent on final result
    if time.monotonic() < deadline - 5:
        best_a, best_score = coordinate_descent_polish(
            best_a, 5, deadline, best_score, evaluate_sequence
        )
        print(f"After Phase 4: best score = {best_score:.6f}")

    print(f"\nFINAL: score={best_score:.6f}, length={len(best_a)}")
    return [float(x) for x in best_a]
