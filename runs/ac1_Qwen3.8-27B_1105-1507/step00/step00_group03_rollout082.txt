def mirror_update(a, c, m, beta, eta, rng):

    # To handle numerical zero, shift
    cshift = c - m
    # if beta large, exp may underflow; use log-sum-exp

       thresh = m - 0.01*m
       idx = c > thresh

       w = idx.astype(float)/idx.sum()

def _mirror_follow(a, c, m, power, eta, rng):
    if m is None or m <=0: return None
    ratio = np.maximum(c, 0.0) / m
    ratio = np.minimum(ratio, 1.0)
    w = ratio ** power

       idx = c >= thresh

       w = idx.astype(float) / float(cnt)

       w = w / sw
    g = 2.0 * correlate(w, a, mode='valid', method='fft')

import math
import time

import numpy as np

from scipy.signal import correlate
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _fft_len(n: int) -> int:
    """Next power-of-two FFT length for a linear convolution."""
    L = 1
    need = 2 * int(n) - 1
    while L < need:
        L <<= 1
    return L


def _conv_sqrt(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of a with itself, computed by zero-padded FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    return c[: 2 * n - 1]


def _conv_cross(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of a and b, computed by zero-padded FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = a.size
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    c = np.fft.irfft(fa * fb, L)
    return c[: 2 * n - 1]


def _score_fast_with_conv(a: np.ndarray, L: int):
    """Return (score, max_conv, convolution)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = float(np.sum(a))
    if s <= 1e-300:
        return np.inf, 0.0, None
    c = _conv_sqrt(a, L)
    m = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(m) or m <= 0.0:
        return np.inf, max(0.0, m), c
    return float(2.0 * n * m / (s * s)), m, c


def _normalize_array(a):
    """Clip to nonnegative finite values and normalize sum to one."""
    if a is None:
        return None
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(np.sum(arr))
    if s <= 1e-300:
        return None
    return arr / s


def _binomial_base(n: int) -> np.ndarray:
    """Unnormalized one-sided flat-square base: C(2k,k)/4^k."""
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    b = np.ones(n, dtype=np.float64)
    if n > 1:
        k = np.arange(1, n, dtype=np.float64)
        b[1:] = np.cumprod((2.0 * k - 1.0) / (2.0 * k))
    return b


def _record_sequence(top, a, score_hint, maxlen=10):
    """Insert a normalized sequence into a small top list."""
    if a is None:
        return
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(np.sum(arr))
    if s <= 1e-300:
        return
    arr = arr / s
    if not np.isfinite(score_hint):
        score_hint, _, _ = _score_fast_with_conv(arr, _fft_len(arr.size))
    if not np.isfinite(score_hint):
        return

    if maxlen > 0 and len(top) >= maxlen and score_hint >= top[-1][0]:
        return

    i = 0
    while i < len(top) and top[i][0] < score_hint:
        i += 1
    top.insert(i, (float(score_hint), arr.copy()))
    if maxlen > 0:
        while len(top) > maxlen:
            top.pop()

def _mirror_follow(a: np.ndarray, c: np.ndarray, m: float, power: float, eta: float, rng):
    """Mirror-descent-like update from a weighted max-convolution gradient."""
    if a is None or c is None:
        return None
    if not np.isfinite(m) or m <= 1e-300:
        return None

    ratio = np.maximum(c, 0.0) / m
    ratio = np.minimum(ratio, 1.0)
    w = ratio ** power
    sw = float(np.sum(w))

    if not np.isfinite(sw) or sw <= 1e-300:
        thresh = 0.99 * m
        idx = c >= thresh
        cnt = int(idx.sum())
        if cnt <= 0:
            idx[int(np.argmax(c))] = True
            cnt = 1
        w = idx.astype(np.float64) / float(cnt)
    else:
        w = w / sw

    try:
        g = 2.0 * correlate(w, a, mode="valid", method="fft")
    except TypeError:
        g = 2.0 * correlate(w, a, mode="valid")

    g = np.asarray(g, dtype=np.float64)
    if g.size != a.size:
        g = g[: a.size]
    if not np.all(np.isfinite(g)) or a.size == 0:
        return None

    gc = g - float(np.mean(g))
    scale = float(np.max(np.abs(gc)))
    if scale <= 1e-18:
        return None

    max_a = float(np.max(a))
    base = np.maximum(a, max_a * 1e-8 + 1e-300)
    a_new = base * np.exp(np.clip(-eta * gc / scale, -10.0, 10.0))
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a_new / s

                continue

            cand = a.copy()
            cand[i] = float(y)
            s = float(np.sum(cand))
            if s <= 1e-300:
                continue
            cand = cand / s
            sc2, _, _ = _score_fast_with_conv(cand, L)
            if np.isfinite(sc2) and sc2 < best_score - 1e-14:
                a = cand
                best = cand.copy()
                best_score = float(sc2)
                current_score = float(sc2)
                max_a = float(np.max(a))
                no_improve = 0
                break
        else:
            no_improve += 1

        if no_improve > 1000:
            if current_score > best_score * 1.02 + 1e-13:
                a = best.copy()
                current_score = best_score
            else:
                a = best.copy()
                sigma = 0.05
                a = a * np.exp(np.clip(rng.normal(0.0, sigma, n), -6.0, 6.0))
                a = _normalize_array(a)
                if a is None:
                    break
                current_score, _, _ = _score_fast_with_conv(a, L)
            no_improve = 0

    return best, best_score


def _lp_matrix_parts(n: int):
    """CSR structure for the linear map g -> f * g, rows = lags, cols = j."""
    rows = 2 * n - 1
    indptr = np.zeros(rows + 1, dtype=np.int64)
    starts = np.empty(rows, dtype=np.int64)
    ends = np.empty(rows, dtype=np.int64)
    total = n * n
    indices = np.empty(total, dtype=np.int32)

    off = 0
    for k in range(rows):
        if k < n:
            s = 0
            e = k
        else:
            s = k - n + 1
            e = n - 1
        starts[k] = s
        ends[k] = e
        length = e - s + 1
        indptr[k + 1] = indptr[k] + length
        indices[off:off + length] = np.arange(s, e + 1, dtype=np.int32)
        off += length

    return indptr, indices, starts, ends


def _lp_fill_data(f: np.ndarray, indptr, starts, ends) -> np.ndarray:
    """Fill data for matrix f*g from current f."""
    n = f.size
    rows = 2 * n - 1
    data = np.empty(n * n, dtype=np.float64)

    for k in range(rows):
        s = int(starts[k])
        e = int(ends[k])
        lo = k - e
        hi = k - s
        data[indptr[k]: indptr[k + 1]] = f[lo:hi + 1][::-1]

    return data


def _lp_improve(n: int, seconds: float, seed: int = 0):
    """LP-direction refinement in the normalized max-conv=1 representation."""
    if n <= 0 or seconds <= 20.0:
        return None, np.inf

    deadline = time.monotonic() + max(0.0, float(seconds))
    L = _fft_len(n)

    quick = min(5.0, max(1.0, 0.05 * float(seconds)), float(seconds) - 10.0)
    starts = _parameter_search(n, max(0.0, quick), seed)
    if not starts:
        a0 = _normalize_array(_binomial_base(n))
        if a0 is None:
            return None, np.inf
        sc0, _, _ = _score_fast_with_conv(a0, L)
        starts = [(sc0, a0)]

    best_seq = np.asarray(starts[0][1], dtype=np.float64).copy()
    best_sc = float(starts[0][0])
    if not np.isfinite(best_sc):
        return best_seq, best_sc

    # Precompute matrix structure once.
    indptr, indices, starts_row, ends_row = _lp_matrix_parts(n)
    b_ub = np.ones(2 * n - 1, dtype=np.float64) * (1.0 + 1e-7)
    c_obj = np.ones(n, dtype=np.float64)
    rng = np.random.default_rng(seed + 777)

    iters = 0
    while time.monotonic() < deadline - 5.0 and iters < 80:
        iters += 1

        sc, m, _ = _score_fast_with_conv(best_seq, L)
        if not np.isfinite(m) or m <= 0.0:
            break

        f = best_seq / np.sqrt(m)
        f = np.where(np.isfinite(f), f, 0.0)
        f = np.clip(f, 0.0, 1.0)
        s_f = float(np.sum(f))
        if s_f <= 1e-300:
            break

        data = _lp_fill_data(f, indptr, starts_row, ends_row)
        A = csr_matrix((data, indices, indptr), shape=(2 * n - 1, n))

        remaining = deadline - time.monotonic()
        if remaining < 12.0:
            break

        opts = {
            "presolve": True,
            "dual_feasibility_tolerance": 1e-9,
            "primal_feasibility_tolerance": 1e-9,
            "time_limit": max(5.0, remaining - 3.0),
        }
        try:
            res = linprog(
                c=c_obj,
                A_ub=A,
                b_ub=b_ub,
                bounds=(0.0, 1.0),
                method="highs",
                options=opts,
            )
        except Exception:
            try:
                opts2 = {
                    "presolve": True,
                    "dual_feasibility_tolerance": 1e-8,
                    "primal_feasibility_tolerance": 1e-8,
                }
                res = linprog(
                    c=c_obj,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=(0.0, 1.0),
                    method="highs",
                    options=opts2,
                )
            except Exception:
                break

        if res is None or not hasattr(res, "x") or res.x is None:
            break

        g = np.asarray(res.x, dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        G = float(np.sum(g))
        if not np.isfinite(G) or G <= s_f * 1.0001:
            break

        cross = _conv_cross(f, g, L)
        mx = float(np.max(cross)) if cross.size else 0.0
        if not np.isfinite(mx) or mx > 1.0 + 1e-6:
            g *= (1.0 / max(mx, 1e-300))
            G = float(np.sum(g))
            if not np.isfinite(G) or G <= s_f * 1.0001:
                break

        g = g * (s_f / G)

        improved = False
        for t in (0.005, 0.01, 0.02, 0.05, 0.10, 0.20, 0.30, 0.50, 0.80, 1.00):
            cand = f * (1.0 - t) + g * t
            cand = np.maximum(cand, 0.0)
            scand = float(np.sum(cand))
            if scand <= 1e-300:
                continue
            cand = cand / scand
            sc2, _, _ = _score_fast_with_conv(cand, L)
            if np.isfinite(sc2) and sc2 < best_sc - 1e-13:
                best_seq = cand.copy()
                best_sc = float(sc2)
                improved = True

        if not improved:
            if iters < 4:
                # Perturb and try another direction.
                best_seq = best_seq * np.exp(np.clip(rng.normal(0.0, 0.04, n), -5.0, 5.0))
                best_seq = _normalize_array(best_seq)
                if best_seq is None:
                    break
                best_sc, _, _ = _score_fast_with_conv(best_seq, L)
            else:
                break

    return best_seq, best_sc


def _sequence_from_array(a):
    """Convert an array to a valid list for evaluate_sequence."""
    if a is None:
        return None
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return None

    out = []
    for v in arr:
        x = float(v)
        if not math.isfinite(x):
            x = 0.0
        if x < 0.0:
            x = 0.0
        elif x > 1000.0:
            x = 1000.0
        out.append(x)

    s = float(sum(out))
    if s < 0.02:
        if s > 0.0:
            factor = 0.05 / s
            for i in range(len(out)):
                out[i] = min(1000.0, out[i] * factor)
        else:
            return None
    return out


def propose_candidate(seed: int = 42, budget_s: float = 1500.0, **kwargs):
    """
    Hybrid search: flat-square binomial initializers, lightweight tilts,
    mirror-descent local search, coordinate polish, and LP-direction refnement.
    """
    total_budget = max(0.0, float(budget_s) - 25.0)
    if total_budget <= 0.0:
        total_budget = max(0.0, float(budget_s) - 5.0)

    t0 = time.monotonic()
    deadline = t0 + total_budget
    top = []

    n_main = 4096
    n_lp = 2048

    # 1) Initial candidate family.
    param_budget = min(30.0, max(5.0, 0.04 * total_budget))
    starts = _parameter_search(n_main, param_budget, seed=seed)
    if not starts:
        a0 = _normalize_array(_binomial_base(n_main))
        if a0 is not None:
            sc0, _, _ = _score_fast_with_conv(a0, _fft_len(n_main))
            starts = [(sc0, a0)]

    for sc, a in starts:
        _record_sequence(top, a, sc, maxlen=12)

    # 2) Long local search on best initializer.
    if starts and time.monotonic() < deadline - 80.0:
        local_budget = min(0.42 * total_budget, max(60.0, 0.42 * total_budget))
        local_budget = min(local_budget, max(10.0, deadline - time.monotonic() - 120.0))
        if local_budget > 10.0:
            a_new, sc_new = _local_search(starts[0][1], local_budget, seed=seed + 10)
            _record_sequence(top, a_new, sc_new, maxlen=12)

    # 3) Secondary local search on a different initializer.
    if len(starts) > 1 and time.monotonic() < deadline - 80.0:
        local_budget2 = min(150.0, 0.10 * total_budget)
        local_budget2 = min(local_budget2, max(5.0, deadline - time.monotonic() - 100.0))
        if local_budget2 > 10.0:
            a_new2, sc_new2 = _local_search(starts[1][1], local_budget2, seed=seed + 11)
            _record_sequence(top, a_new2, sc_new2, maxlen=12)

    # 4) LP-direction refinement on a smaller, LP-tractable length.
    if time.monotonic() < deadline - 80.0:
        lp_budget = min(350.0, 0.35 * total_budget)
        lp_budget = min(lp_budget, max(5.0, deadline - time.monotonic() - 60.0))
        if lp_budget > 80.0:
            lp_a, lp_sc = _lp_improve(n_lp, lp_budget, seed=seed + 1000)
            _record_sequence(top, lp_a, lp_sc, maxlen=12)

    # 5) Final long polish of the best internal candidate.
    if top and time.monotonic() < deadline - 10.0:
        final_budget = min(600.0, max(10.0, deadline - time.monotonic() - 8.0))
        if final_budget > 20.0:
            a_final, sc_final = _local_search(top[0][1], final_budget, seed=seed + 2000)
            _record_sequence(top, a_final, sc_final, maxlen=12)

    # Exact-rank top candidates using the authoritative evaluator.
    best_seq = None
    best_exact = np.inf
    eval_count = 0
    for sc, a in top:
        seq = _sequence_from_array(a)
        if seq is None:
            continue
        exact = evaluate_sequence(seq)
        eval_count += 1
        if np.isfinite(float(exact)) and float(exact) < float(best_exact):
            best_exact = float(exact)
            best_seq = seq

    # Fallback to parent if available.
    if best_seq is None:
        parent = globals().get("height_sequence_1", None)
        seq = _sequence_from_array(parent)
        if seq is not None:
            exact = evaluate_sequence(seq)
            if np.isfinite(float(exact)):
                best_seq = seq
                best_exact = float(exact)

    # Safe fallback.
    if best_seq is None:
        base = _normalize_array(_binomial_base(512))
        if base is None:
            base = np.ones(512, dtype=np.float64)
        best_seq = _sequence_from_array(base)
        if best_seq is None:
            best_seq = [1.0] * 256

    return best_seq
