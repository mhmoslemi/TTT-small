"""
LP-iteration based optimizer for minimum peak autocorrelation non-negative sequences.
Uses cross-correlation LP to generate descent directions, with line search,
multiple restarts across different sequence lengths and initializations,
and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar

_DEADLINE = 0.0


def _fast_score(a):
    """Compute 2*n*max(convolve(a,a))/sum(a)^2 without overhead."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)


def _build_lp_and_solve(a):
    """Solve LP: min t s.t. (a*b)[k] <= t for all k, sum(b)=1, b>=0.
    Returns optimal b (normalized) or None."""
    n = len(a)
    c = np.zeros(n + 1)
    c[n] = 1.0

    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])

    A_ub = np.zeros((2 * n - 1, n + 1))
    # A_ub[k, j] = a[k-j] for valid (k,j), then A_ub[k, n] = -1
    # For each k, valid j range is [max(0,k-n+1), min(n-1,k)]
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        # k-j for j in [j_lo, j_hi] gives values [k-j_hi, k-j_lo]
        A_ub[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            b = np.clip(res.x[:n], 0, None)
            bs = b.sum()
            if bs > 1e-12:
                return b / bs
    except Exception:
        pass
    return None


def _lp_iterate(a, max_rounds=150, verbose=False):

    n = len(a)

        if b is None:

        # Check convergence

            break

        # Line search: try decreasing step sizes
        improved = False
        for step in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05]:
            a_new = (1.0 - step) * a + step * b
            a_new = np.maximum(a_new, 0.0)

            if s_sum < 1e-12:
                continue

    # 6. Edge-concentrated

    inits.append(ec / ec.sum())

    # 7. Random uniform
    ru = rng.uniform(0.01, 1.0, n)
    inits.append(ru / ru.sum())

    # 8. Alternating high/low
