"""
Minimizes J = 2n*max(conv(a,a))/sum(a)^2 using gradient descent with
log-sum-exp smoothing, softmax parameterization, Adam optimizer, tau annealing,
multiple sequence lengths, and random restarts. A final local polishing phase
applies direct subgradient steps on the best sequence found.
"""

import numpy as np
import time


def _softmax(x):
    x = x - np.max(x)
    e = np.exp(x)
    return e / e.sum()


def _compute_grad(a, tau, S):
    """Compute autocorrelation b, softmax weights p, and gradient h = dL/da."""
    n = len(a)
    fa = np.fft.rfft(a, S)
    b = np.fft.irfft(fa * fa, S)[:2 * n - 1]
    bm = b.max()
    p = np.exp(tau * (b - bm))
    p /= p.sum()
    fp = np.fft.rfft(p, S)
    corr = np.fft.irfft(fp * np.conj(fa), S)
    h = 2.0 * corr[:n]
    return b, h


def _init_sequence(n, rng, init_type, best_a=None):
    if init_type == 0:
        x = rng.normal(0, 1, n)
    elif init_type == 1:
        x = np.linspace(0, 2, n) + rng.normal(0, 0.1, n)
    elif init_type == 2:
        c = n / 2 + rng.normal(0, n / 4)
        w = n / 5 + rng.uniform(0, n / 5)
        x = -((np.arange(n, dtype=np.float64) - c) / w) ** 2
    elif init_type == 3:
        x = rng.exponential(1.0, n)
    else:
        if best_a is not None and len(best_a) == n:
            a0 = np.array(best_a, dtype=np.float64)
            a0 += rng.normal(0, 0.02, n)
            a0 = np.clip(a0, 1e-12, None)
            a0 /= a0.sum()
            x = np.log(a0)
        else:
            x = rng.normal(0, 1, n)
    return _softmax(x)


def _adam_update(x, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m[:] = beta1 * m + (1.0 - beta1) * grad
    v[:] = beta2 * v + (1.0 - beta2) * grad * grad
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    x -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return x, m, v


def _polish(a, n, S, deadline, rng):
    """Local refinement using subgradient steps on the actual max."""
    best = a.copy()
    best_J = evaluate_sequence(best.tolist())
    lr = 0.005
    for it in range(50000):
        if time.monotonic() > deadline:
            break
        fa = np.fft.rfft(best, S)
        b = np.fft.irfft(fa * fa, S)[:2 * n - 1]
        mx = b.max()
        idx = np.where(b >= mx - 1e-12)[0]
        # Subgradient: sg[i] = 2/|idx| * sum_{k in idx} a[k-i]
        sg = np.zeros(n)
        for k in idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            j = np.arange(lo, hi + 1)
            sg[j] += best[k - j]
        sg *= 2.0 / len(idx)
        a_new = best - lr * sg
        a_new = np.clip(a_new, 1e-12, None)
        a_new /= a_new.sum()
        J = evaluate_sequence(a_new.tolist())
        if J < best_J:
            best = a_new
            best_J = J
            lr *= 1.05
        else:
            lr *= 0.9
            best = a_new
    return best, best_J


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + budget_s - 8.0

    best_J = float('inf')
    best_a = None
    rng = np.random.default_rng(seed)

    lengths = [2048, 4096, 8192]
    time_fracs = [0.25, 0.40, 0.35]
    total_time = budget_s - 20.0
    time_alloc = [total_time * f for f in time_fracs]

    for li, n in enumerate(lengths):
        n_deadline = start_time + sum(time_alloc[:li + 1])
        if time.monotonic() > n_deadline:
            break
        print(f"Starting n={n}, deadline={n_deadline - start_time:.1f}s")

        S = 1
        while S < 3 * n:
            S <<= 1

        num_restarts = max(3, int(time_alloc[li] / 80.0))
        num_restarts = min(num_restarts, 12)
        print(f"  {num_restarts} restarts for n={n}")

        for r in range(num_restarts):
            r_deadline = start_time + time_alloc[li] * ((r + 1) / num_restarts)
            if time.monotonic() > r_deadline:
                break

            init_type = r % 5
            a = _init_sequence(n, rng, init_type, best_a)
            x = np.log(a + 1e-30)

            local_best_J = float('inf')
            local_best_a = a.copy()

            m_s = np.zeros(n)
            v_s = np.zeros(n)
            t_adam = 0
            lr = 0.3 / np.sqrt(n)
            beta1, beta2 = 0.9, 0.999

            tau_start, tau_end = 30.0, 1000.0
            num_phases = 20
            steps_per_phase = max(500, int((r_deadline - time.monotonic()) * 2500) // num_phases)

            for phase in range(num_phases):
                tau = tau_start * (tau_end / tau_start) ** (phase / max(1, num_phases - 1))
                phase_bias = 0.0

                for step in range(steps_per_phase):
                    if time.monotonic() > r_deadline:
                        break

                    b, h = _compute_grad(a, tau, S)
                    wmean = np.dot(h, a)
                    grad_x = a * (h - wmean)

                    t_adam += 1
                    x, m_s, v_s = _adam_update(x, grad_x, m_s, v_s, t_adam, lr, beta1, beta2, 1e-8)
                    a = _softmax(x)

                    if step % 2000 == 0:
                        J = evaluate_sequence(a.tolist())
                        if np.isfinite(J) and J < local_best_J:
                            local_best_J = J
                            local_best_a = a.copy()

                if phase % 5 == 4:
                    J = evaluate_sequence(local_best_a.tolist())
                    if np.isfinite(J) and J < local_best_J:
                        local_best_J = J
                    print(f"  n={n} r={r} phase={phase} tau={tau:.0f} local_J={local_best_J:.6f} global_J={best_J:.6f} t={time.monotonic()-start_time:.0f}s")

            J = evaluate_sequence(local_best_a.tolist())
            if np.isfinite(J) and J < local_best_J:
                local_best_J = J

            if local_best_J < best_J:
                best_J = local_best_J
                best_a = local_best_a.copy()
                print(f"  NEW BEST: J={best_J:.6f} n={n}")

    # Final polishing
    if best_a is not None and time.monotonic() < deadline:
        n = len(best_a)
        S = 1
        while S < 3 * n:
            S <<= 1
        print(f"Polishing n={n}, J={best_J:.6f}")
        best_a, best_J = _polish(best_a, n, S, deadline, rng)
        print(f"After polish: J={best_J:.6f}")

    if best_a is None:
        best_a = np.ones(256) / 256.0
        best_J = evaluate_sequence(best_a.tolist())

    print(f"FINAL: J={best_J:.6f}")
    return [float(v) for v in best_a]
