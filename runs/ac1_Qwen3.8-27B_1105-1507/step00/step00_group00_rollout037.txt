```python
import time

def compute_score(a):
    """Compute the score directly (for internal use, faster than evaluate_sequence)."""
    n = len(a)
    conv = np.convolve(a, a)
    M = conv.max()
    S = a.sum()
    return 2.0 * n * M / (S * S)

def subgradient_step(a, K=5):
    """Compute the subgradient of max(conv(a,a)) at point a."""
    n = len(a)
    conv = np.convolve(a, a)
    # Find top K positions
    top_idx = np.argpartition(conv, -K)[-K:]
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                grad[j] += 2.0 * a[m]
    grad /= K
    return grad, conv

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0
    
    best_score = np.inf
    best_seq = None
    
    # Try different lengths
    for n in [100, 200, 500, 1000, 2000]:
        if time.monotonic() > deadline:
            break
        
        # Try different initializations
        inits = []
        # Linear ramp (decreasing)
        t = np.linspace(1, 0.5, n)
        inits.append(t / t.sum())
        # Mild perturbation of uniform
        a = np.ones(n) + 0.1 * rng.normal(size=n)
        a = np.clip(a, 0.1, None)
        inits.append(a / a.sum())
        # Geometric (mild)
        r = 0.999
        a = np.power(r, np.arange(n))
        inits.append(a / a.sum())
        # Two-level
        a = np.ones(n)
        a[:n//2] = 1.5
        inits.append(a / a.sum())
        
        for a0 in inits:
            if time.monotonic() > deadline:
                break
            
            # Phase 1: Subgradient descent
            a = a0.copy()
            lr = 0.01
            for step in range(10000):
                if time.monotonic() > deadline:
                    break
                grad, conv = subgradient_step(a)
                norm = np.linalg.norm(grad)
                if norm > 1e-10:
                    a = a - lr * grad / norm
                    a = np.clip(a, 0, None)
                    if a.sum() > 1e-10:

    return [float(x) for x in best_seq]

def local_search(a, deadline, rng):
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    lr = 0.001
    
    while time.monotonic() < deadline:
        # Try different perturbation strategies
        strategy = rng.integers(0, 3)
        
        if strategy == 0:
            # Block multiplicative
            candidate = best_a.copy()
            width = max(1, n // 50)
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.05))
        elif strategy == 1:
            # Targeted: reduce the worst contributors
            conv = np.convolve(best_a, best_a)
            k_max = np.argmax(conv)
            # Reduce a[i] where i and k_max-i both have large values
            contributions = np.zeros(n)
            for j in range(n):
```

```python
def isotonic_regression(a, decreasing=True):
    """Project onto the monotone (decreasing if decreasing=True) cone."""
    if not decreasing:
        a = -a
    # Pool adjacent violators
    n = len(a)
    blocks = [(a[i], 1) for i in range(n)]  # (mean, count)
    stack = []
    for val, count in blocks:
        while stack and stack[-1][0] < val:
            prev_val, prev_count = stack.pop()
            val = (prev_val * prev_count + val * count) / (prev_count + count)
            count += prev_count
        stack.append((val, count))
    result = np.array([val * count for val, count in stack])
    # Expand back
    expanded = np.repeat(np.array([val for val, _ in stack]), np.array([c for _, c in stack]))
    if not decreasing:
        expanded = -expanded
    return expanded
```
