```python
import numpy as np
import time

def compute_subgradient(a, k_star, n):
    """Compute subgradient of (a*a)[k_star] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        k_minus_j = k_star - j
        if 0 <= k_minus_j < n:
            val += a[k_minus_j]
        j_minus_k = j - k_star
        if 0 <= j_minus_k < n:
            val += a[j_minus_k]
        grad[j] = val
    return grad
```

```python
def compute_subgradient_fast(a, k_star, n):

    # a[k_star - j] for j where 0 <= k_star - j < n
    # i.e., j in [k_star - n + 1, k_star] ∩ [0, n-1]
    j_start = max(0, k_star - n + 1)
    j_end = min(n - 1, k_star)
    if j_start <= j_end:
        indices = np.arange(j_start, j_end + 1)
        grad[indices] += a[k_star - indices]
    
    # a[j - k_star] for j where 0 <= j - k_star < n

    if j_start2 <= j_end2:
        indices2 = np.arange(j_start2, j_end2 + 1)
```
