```python
import numpy as np
from scipy.optimize import minimize
import time

def smooth_max(arr, p=50):
    """Smooth approximation of max."""
    arr = np.maximum(arr, 1e-30)  # avoid log(0)
    return np.sum(arr**p)**(1.0/p)

def objective_and_grad(x, p=50):
    """Compute objective and gradient for softmax parameterization."""
    # Softmax
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)
    n = len(a)
    
    # Convolution
    C = np.convolve(a, a)
    
    # Smooth max
    C_clipped = np.maximum(C, 1e-30)
    C_p = C_clipped**p
    sum_Cp = np.sum(C_p)
    obj = sum_Cp**(1.0/p)
    
    # Gradient with respect to a
    C_prime = (p * C_clipped**(p-1)) / sum_Cp**((p-1)/p)  # d smooth_max / d C
    # g[j] = 2 * sum_m C_prime[j+m] * a[m]
    corr = np.correlate(C_prime, a, mode='full')[:n]
    g_a = 2.0 * corr
    
    # Gradient with respect to x (through softmax)
    g_x = a * (g_a - np.sum(g_a * a))
    
    return obj, g_x
