def _peak_shave(a, beta, lr, n_iters, schedule='linear'):
    """Run peak-shaving iterations on sequence a. Returns best found."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    S = np.sum(a)
    if S <= 0:
        return a
    a = a / S  # normalize to sum 1
    best = a.copy()
    best_obj = _objective(best)
    
    for i in range(n_iters):
        # Adaptive learning rate
        if schedule == 'linear':
            lr_i = lr * (1 - i / max(1, n_iters)) + 0.001
        else:
            lr_i = lr
        
        c = np.convolve(a, a)
        M = c.max()

        p = np.correlate(w, a, mode='valid')

"""
Iterative peak-shaving optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multiplicative-weight updates driven by convolution peak pressure,
with parametric initializations, adaptive hyperparameters, and local search.
"""
import numpy as np
import time


def _compute_objective(a):
    """Compute 2*n*max_b/sum_a^2 for sequence a."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    M = c.max()
    return float(2.0 * n * M / (S * S))


def _peak_shave(a, beta, lr, n_iters, temp_anneal=False):
    """Run peak-shaving iterations. Returns best sequence found."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    S = np.sum(a)
    if S <= 0:
        return a
    a = a / S
    best = a.copy()
    best_obj = _compute_objective(best)

    for i in range(n_iters):
        frac = i / max(1, n_iters)
        if temp_anneal:
            beta_i = beta * (1.0 - 0.5 * frac)
            lr_i = lr * (1.0 - 0.7 * frac) + 0.005
        else:
            beta_i = beta
            lr_i = lr * (1.0 - 0.5 * frac) + 0.005

        c = np.convolve(a, a)
        M = c.max()
        if M <= 0:
            break
        w = (c / M) ** beta_i
        p = np.correlate(w, a, mode='valid')
        p_max = p.max()
        if p_max <= 1e-15:
            break
        a = a * np.exp(-lr_i * p / p_max)
        floor = 1e-6 / n
        a = np.maximum(a, floor)
        a = a / np.sum(a)

        obj = _compute_objective(a)
        if obj < best_obj:
            best = a.copy()
            best_obj = obj

    best = best * S
    return best


def _block_perturb(a, rng, width_frac, noise_std):
    """Apply random multiplicative perturbation to a random block."""
    a_new = a.copy()
    n = len(a_new)
    width = max(1, int(n * width_frac))

    a_new[start:start + width] *= np.exp(rng.normal(0.0, noise_std, size=width))
    return np.clip(a_new, 0.0, 1000.0)

    """Apply multiple small block perturbations."""
    a_new = a.copy()
    n = len(a_new)
    for _ in range(n_blocks):
        width_frac = rng.uniform(0.005, 0.05)
        noise_std = rng.uniform(0.02, 0.12)
        a_new = _block_perturb(a_new, rng, width_frac, noise_std)
    return a_new
