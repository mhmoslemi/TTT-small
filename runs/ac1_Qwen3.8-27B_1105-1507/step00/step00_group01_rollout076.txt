"""
Minimize F(a) = 2*n*max(conv(a,a)) / (sum a)^2 over non-negative real sequences.
The function is scale-invariant, so we fix sum(a)=1 and minimize max(conv(a,a)).

Key structural fact used: conv(a,a)[n-1] = <a, reverse(a)>. For a *palindromic*
sequence this equals sum a_i^2 >= (sum a)^2/n, forcing F >= 2 (the parent sits on
this symmetric floor). To go below 2 we MUST break the palindromic symmetry, so
every optimizer in this file starts from a deliberately asymmetric (ramped /
exponential / power-law / noisy) seed and drives down the convolution peak with a
beta-annealed smooth-max (softmax) projected gradient method.

Tracks run in parallel within the wall-clock budget:
  * small-n asymmetric sequences (fast direct O(n^2) conv + analytic gradient),
  * large-n (parent length) sequences (FFT convolution + analytic gradient),
with a self-test at startup that verifies the FFT gradient against a direct one.
The best sequence by the REAL evaluator is returned; parent is the floor fallback.
"""

import math
import time
import numpy as np


# ----------------------------- convolution + gradient -----------------------------

def _next_pow2(m):
    p = 2
    while p < m:
        p <<= 1
    return p


def conv_direct(a):
    """Exact linear convolution a*a (length 2n-1)."""
    return np.convolve(a, a)


def conv_fft(a):
    n = a.size
    L = _next_pow2(2 * n - 1)
    fa = np.rfft(a, L)
    return np.irfft(fa * fa, L)[: (2 * n - 1)]


def grad_direct(a, c, w):
    """d/d a_j of sum_k w_k c_k  where c_k = sum_i a_i a_{k-i}.
    d c_k / d a_j = 2 * a_{k-j}.  =>  g_j = 2 * sum_k w_k a_{k-j}.
    a_{k-j} valid for 0<=k-j<n  =>  k in [j, j+n-1].  Use zero-padded w."""
    n = a.size
    wpad = np.zeros(n + n)  # length 2n, w lives in first 2n-1
    wpad[: (2 * n - 1)] = w
    g = np.empty(n)
    for j in range(n):
        g[j] = 2.0 * np.dot(wpad[j:j + n], a)
    return g


def grad_fft(a, c, w):
    """Same as grad_direct but via FFT cross-correlation: g_j = 2*sum_m a_m w_{m+j}."""
    n = a.size
    L = _next_pow2(3 * n)
    fa = np.rfft(a, L)
    fw = np.rfft(w, L)
    corr = np.irfft(fw * np.conj(fa), L)  # corr[k] = sum_m w[m+k] a[m] (no wrap, k small)
    g = 2.0 * corr[:n]
    return g


def self_test_grad():
    rng = np.random.default_rng(0)
    ok_all = True
    for n in (5, 16, 33):
        a = rng.random(n) ** 2 + 0.05
        a /= a.sum()
        c = conv_direct(a)
        beta = 30.0
        e = np.exp(beta * (c - c.max()))
        w = e / e.sum()
        g1 = grad_direct(a, c, w)
        g2 = grad_fft(a, c, w)
        err = float(np.max(np.abs(g1 - g2)))
        rel = err / (float(np.max(np.abs(g1))) + 1e-12)
        ok = rel < 1e-6
        ok_all = ok_all and ok
        print("self-test n=%d grad rel err=%.3e ok=%s" % (n, rel, ok))
    return ok_all


# ----------------------------- optimizer -----------------------------

def project_normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-12:
        a = np.ones_like(a)
        s = a.sum()
    return a / s


def optimize_sequence(a0, budget_s, use_fft, lr=0.6, momentum=0.5,
                      beta_start=12.0, beta_end=400.0, stages=5,
                      step_fn="auto", seed=1):
    """Smooth-max projected gradient descent. Returns (best_a_sum1, best_maxc)."""
    rng = np.random.default_rng(seed)
    if use_fft:
        convf = conv_fft
        gradf = grad_fft
    else:
        convf = conv_direct
        gradf = grad_direct

    a = project_normalize(np.asarray(a0, dtype=np.float64).copy())
    c = convf(a)
    best_a = a.copy()
    best_mc = float(c.max())

    n = a.size
    total_budget = max(1, int(budget_s / max(1e-6, (beta_end - beta_start) / 100.0)))
    # per-stage step budget sized by n (direct is O(n^2), fft is O(n log n))
    if use_fft:
        steps_per_stage = max(40, int(20000.0 / max(1.0, math.log2(n) + 1)))
    else:
        steps_per_stage = max(30, int(1.2e6 / max(1.0, float(n))))

    deadline = time.monotonic() + budget_s
    betas = np.geomspace(beta_start, beta_end, stages) if beta_end > beta_start else [beta_start]
    for si, beta in enumerate(betas):
        v = np.zeros(n)
        for st in range(steps_per_stage):
            if time.monotonic() > deadline:
                break
            c = convf(a)
            m = float(c.max())
            if m < best_mc:
                best_mc = m
                best_a = a.copy()
            e = np.exp(beta * (c - m))
            w = e / e.sum()
            g = gradf(a, c, w)
            # adaptive step so the update magnitude is controlled
            gnorm = float(np.linalg.norm(g)) + 1e-12
            step = lr * (float(a.sum()) ** 2 / (float(n) * gnorm)) if gnorm > 1e-9 else lr
            step = min(step, 0.05)
            v = momentum * v - step * g
            a = a + v
            a = project_normalize(a)
    return best_a, best_mc


def make_tilts(n, seed):
    rng = np.random.default_rng(seed)
    tilts = []
    # exponential ramp (geometric) - the strongest asymmetry that we know works
    for ratio in (1.15, 1.25, 1.4, 1.7, 2.2):
        t = np.array([ratio ** i for i in range(n)])
        tilts.append(t / t.sum())
    # power-law ramp
    for p in (1.0, 1.5, 2.5, 4.0):
        t = np.array([(i + 1) ** p for i in range(n)])
        tilts.append(t / t.sum())
    # reversed power-law
    for p in (1.5, 3.0):
        t = np.array([(n - i) ** p for i in range(n)])
        tilts.append(t / t.sum())
    # asymmetric noise
    for _ in range(4):
        t = np.abs(rng.normal(1.0, 0.6, n)) + 0.02
        t = t * (0.6 + 0.8 * np.linspace(0.3, 1.4, n))
        tilts.append(t / t.sum())
    return tilts
