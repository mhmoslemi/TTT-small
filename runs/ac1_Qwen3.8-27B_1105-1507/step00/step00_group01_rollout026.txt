def smooth_max_objective(x, alpha):
    """Compute the smooth max of conv(p,p) where p = softmax(x)."""
    n = len(x)
    L = 1
    while L < 4*n:

    p /= p.sum()
    P = rfft(p, L)

    logsumexp = alpha * s.max() + np.log(np.sum(np.exp(alpha * (s - s.max()))))
    return logsumexp / alpha

        k_star = int(s.argmax())

        # Influence of each position j on the peak s[k_star]
        # s[k_star] = sum_j p[j] * p[k_star - j]
        influence = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= k_star - j < n:
                val += p[k_star - j]
            if 0 <= k_star - j < n:
                pass  # already counted
            influence[j] = val

        # Find position with highest "self" influence to reduce
        # (we want to reduce p[j] where p[k_star-j] is large, i.e., j's mirror is large)
        # And find a position to add mass where influence is low
        top_j = int(np.argmax(influence * p))
        # Find a target with low influence
        low_influence = np.argsort(influence)[:n // 4]
        target_j = low_influence[rng.integers(0, len(low_influence))]

        if top_j == target_j:
            continue

        # Try multiple step sizes
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            move_amt = p[top_j] * frac
            if move_amt <= 0:
                continue
            candidate = best.copy()
            candidate[top_j] -= move_amt
            candidate[target_j] += move_amt
            if candidate[top_j] < 0:
                continue
            score = _score_seq(candidate)
            if score < best_score:
                best = candidate
                best_score = score
                break

    return best, best_score


def _random_block_perturb(base, rng, n):
    """Apply random multiplicative perturbations to blocks."""
    candidate = base.copy()
    n_perturb = rng.integers(3, 15)
    for _ in range(n_perturb):
        width = rng.integers(max(2, n // 50), max(3, n // 5))
        start = rng.integers(0, max(1, n - width))
        candidate[start:start + width] *= np.exp(rng.normal(0, 0.3))
    candidate = np.clip(candidate, 0.0, 1000.0)
    if candidate.sum() < 1e-10:
        candidate = base.copy()
    return candidate

    Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
    1. Continuation method with log-sum-exp smoothed max (gradient-based L-BFGS)

        parent = np.ones(256)
