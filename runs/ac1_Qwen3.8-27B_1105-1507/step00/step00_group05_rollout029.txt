    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fc = np.fft.irfft(fa * fa, fsize)[:size]
