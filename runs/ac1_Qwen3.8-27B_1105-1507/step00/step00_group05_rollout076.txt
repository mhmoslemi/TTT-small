Minimizes 2n*max(conv(a,a))/(sum(a))^2 using gradient-based optimization
with softmax parameterization, temperature annealing, and exact local search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def full_convolution(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = fft(a, n=size)
    return np.real(ifft(fa * fa, n=size))[:2 * n - 1]

def soft_max_and_weights(conv, beta):
    """Compute soft-max value and softmax weights."""
    bc = beta * conv
    m = np.max(bc)
    exp_shifted = np.exp(bc - m)
    Z = np.sum(exp_shifted)
    w = exp_shifted / Z
    sm = m / beta + np.log(Z) / beta
    return sm, w

def smooth_obj_and_grad(x, beta):
    """Objective and gradient for softmax-parameterized a."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)
    n = len(a)
    
    conv = full_convolution(a)
    sm, w = soft_max_and_weights(conv, beta)
    obj = 2.0 * n * sm
    
    c1 = np.correlate(w, a, mode='valid')
    wc = np.dot(w, conv)
    grad = 4.0 * n * a * (c1 - wc)
    
    return obj, grad

    n = len(a)
