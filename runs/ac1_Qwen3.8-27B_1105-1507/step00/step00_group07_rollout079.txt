```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_constraint_matrix(a):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    total_nnz = n * n  # exact count
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    
    idx = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        count = j_end - j_start + 1
        if count > 0:
            rows[idx:idx+count] = k
            cols[idx:idx+count] = np.arange(j_start, j_end + 1)
            vals[idx:idx+count] = a[k - np.arange(j_start, j_end + 1)]
            idx += count
    
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_direction(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    A = build_conv_constraint_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success or result.x is None:
        return None
    b = result.x
    if b.sum() <= 1e-10:
        return None
    return b / b.sum()

def make_initial_candidates(rng, lengths):
    """Generate diverse initial sequences."""
    candidates = []
    for n in lengths:
        # Linear ramp
        a = np.arange(1, n+1, dtype=np.float64)
        a /= a.sum()
        candidates.append(a)
        
        # Reverse ramp
        a = np.arange(n, 0, -1, dtype=np.float64)
        a /= a.sum()
        candidates.append(a)
        
        # Exponential with various rates
        for alpha in [0.5, 1.0, 2.0, 4.0]:
            a = np.exp(alpha * np.arange(n, dtype=np.float64) / max(1, n-1))
            a /= a.sum()
            candidates.append(a)
            # Reverse
            a = a[::-1].copy()
            candidates.append(a)
        
        # Step
        for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
            a = np.ones(n)
            a[int(frac*n):] = 0.3
            a /= a.sum()
            candidates.append(a)
            a = a[::-1].copy()
            candidates.append(a)
        
        # Sine
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 + 0.05
        a /= a.sum()
        candidates.append(a)
        
        # Random
        for _ in range(3):
            a = rng.random(n) + 0.05
            a /= a.sum()
            candidates.append(a)
    
    return candidates

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0
    
    # Phase 1: Explore initial candidates
    lengths = [256, 512, 1024, 2048]
    candidates = make_initial_candidates(rng, lengths)
    
    best = None
    best_score = float('inf')
    
    for a in candidates:
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()
    
    print(f"Phase 1 best: score={best_score:.6f}, len={len(best)}")
    
    # Phase 2: LP-based refinement
    a = best.copy()
    n = len(a)
    t_start = time.monotonic()
    
    for iteration in range(500):
        if time.monotonic() > deadline:
            break
        
        # Compute peak
        conv = np.convolve(a, a)
        M = conv.max()
        
        # Solve LP
        b = lp_direction(a, M)
        if b is None:
            # Try smaller step perturbation
            break
        
        # Line search
        current_score = evaluate_sequence(a.tolist())
        best_t = 0
        best_trial = a.copy()
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.3, 0.15]:
            a_try = (1 - t) * a + t * b
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s  # renormalize
            score = evaluate_sequence(a_try.tolist())
            if np.isfinite(score) and score < current_score:
                current_score = score
                best_t = t
                best_trial = a_try.copy()
        
        if best_t == 0:
            # Try perturbation
            a_trial = a * np.exp(rng.normal(0, 0.05, n))
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 0:
                a_trial /= s
                score = evaluate_sequence(a_trial.tolist())
                if np.isfinite(score) and score < evaluate_sequence(a.tolist()):
                    a = a_trial
                else:
                    break
        else:
            a = best_trial
            best_score = current_score
        
        if iteration % 20 == 0:
            print(f"  Iter {iteration}: score={best_score:.6f}")
    
    print(f"Phase 2 final: score={best_score:.6f}, len={len(a)}")
    
    # Phase 3: Try other lengths with LP
    # ... (if time permits)
    
    return [float(x) for x in a]
```

```python
"""
Step function optimization: minimize 2n*max(conv(a,a))/sum(a)^2.
Uses diverse initial sequences (ramps, exponentials, power laws) of varying lengths,
then refines via LP-based direction finding and perturbation-based local search.
Key insight: non-palindromic sequences (especially ramps for large n) can achieve
objective well below 2.0, approaching 4/3 asymptotically for linear ramps.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_constraint_matrix(a):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    total_nnz = n * n
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)

    idx = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        count = j_end - j_start + 1
        if count > 0:
            rows[idx:idx + count] = k
            cols[idx:idx + count] = np.arange(j_start, j_end + 1)
            vals[idx:idx + count] = a[k - np.arange(j_start, j_end + 1)]
            idx += count

    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0. Returns normalized b or None."""
    n = len(a)
    A = build_conv_constraint_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success or result.x is None:
        return None
    b = np.maximum(result.x, 0.0)
    s = b.sum()
    if s <= 1e-10:
        return None
    return b / s


def generate_ramp(n):
    """Linear ramp normalized to sum=1, max value <= 1000."""
    a = np.arange(1, n + 1, dtype=np.float64)
    a /= a.sum()
    return a


def generate_exp_seq(n, alpha):
    """Exponential sequence normalized to sum=1."""
    a = np.exp(alpha * np.arange(n, dtype=np.float64) / max(1, n - 1))
    a /= a.sum()
    return a


def generate_power_seq(n, alpha):
    """Power law sequence normalized to sum=1."""
    a = np.power(np.arange(1, n + 1, dtype=np.float64), alpha)
    a /= a.sum()
    return a


def generate_skew_gaussian(n, center_frac, sigma_frac):
    """Skewed Gaussian-like sequence normalized to sum=1."""
    x = np.arange(n, dtype=np.float64) / n
    c = center_frac
    s = sigma_frac
    a = np.exp(-0.5 * ((x - c) / s) ** 2) + 0.01
    a /= a.sum()
    return a


def generate_candidates(rng, lengths):
    """Generate a diverse set of candidate sequences."""
    candidates = []
    for n in lengths:
        # Ramps (forward and reverse)
        candidates.append(generate_ramp(n))
        candidates.append(generate_ramp(n)[::-1].copy())

        # Exponentials
        for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
            candidates.append(generate_exp_seq(n, alpha))
            candidates.append(generate_exp_seq(n, -alpha))

        # Power laws
        for alpha in [0.25, 0.5, 1.0, 1.5, 2.0]:
            candidates.append(generate_power_seq(n, alpha))

        # Skewed Gaussians
        for cf in [0.3, 0.4, 0.6, 0.7]:
            for sf in [0.15, 0.25, 0.35]:
                candidates.append(generate_skew_gaussian(n, cf, sf))

        # Step functions
        for frac in [0.3, 0.5, 0.7]:
            a = np.ones(n)
            a[int(frac * n):] = 0.2
            a /= a.sum()
            candidates.append(a)

        # Random
        for _ in range(2):
            a = rng.random(n) + 0.05
            a /= a.sum()
            candidates.append(a)

    return candidates


def perturb_sequence(a, rng, scale=0.05):
    """Apply a random multiplicative perturbation."""
    n = len(a)
    # Block perturbation
    width = max(1, n // 50)
    n_blocks = rng.integers(1, 5)
    a_new = a.copy()
    for _ in range(n_blocks):
        start = rng.integers(0, max(1, n - width))
        end = min(n, start + width)
        a_new[start:end] *= np.exp(rng.normal(0, scale, end - start))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    return a_new


def lp_refine(a, deadline, max_iters=200):
    """Refine sequence using LP-based direction finding with line search."""
    n = len(a)
    current_score = evaluate_sequence(a.tolist())
    best_score = current_score
    best_a = a.copy()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = np.convolve(a, a)
        M = conv.max()

        b = lp_direction(a, M)
        if b is None:
            break

        # Line search
        found = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.4, 0.15]:
            a_try = (1 - t) * a + t * b
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            score = evaluate_sequence(a_try.tolist())
            if np.isfinite(score) and score < current_score - 1e-8:
                a = a_try
                current_score = score
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                found = True
                break

            if np.isfinite(score) and score < current_score - 1e-8:

                    best_score = score

            else:
                stall_count += 1

                    break
        else:
            stall_count = 0

        if it % 30 == 0:

def perturbation_refine(a, deadline, rng, max_iters=5000):

    n = len(a)
    current_score = evaluate_sequence(a.tolist())
    best_score = current_score
    best_a = a.copy()
    scale = 0.1
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        a_try = perturb_sequence(a, rng, scale)
        score = evaluate_sequence(a_try.tolist())

        if np.isfinite(score) and score < current_score:
            a = a_try
            current_score = score
            if score < best_score:
                best_score = score
                best_a = a.copy()
            stall = 0
            scale = min(scale * 1.05, 0.3)
        else:
            stall += 1
            if stall > 200:
                scale = max(scale * 0.7, 0.01)
                stall = 0
            if scale < 0.005:
                break

        if it % 500 == 0:
            print(f"    Pert iter {it}: score={best_score:.6f}, scale={scale:.4f}")

    return best_a, best_score
```
