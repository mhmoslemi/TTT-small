        outer_max = np.zeros(len(p_arr))
        for idx, p in enumerate(p_arr):
            left_max = cummax[p-1] if p > 0 else 0
            right_start = p + n
            right_max = rev_cummax[right_start] if right_start < 2*n-1 else 0
            outer_max[idx] = max(left_max, right_max)
        
        total_max = np.maximum(affected_max, outer_max)
        best_idx = np.argmin(total_max)
        best_p = int(p_arr[best_idx])
        
        conv[best_p:best_p+n] += 2*f
        conv[2*best_p] += 1
        f[best_p] = 1
    
    return f
```

def local_search_01(f, max_iters=100000):
    """Local search on a 0-1 sequence."""
    n = len(f)
    f = f.copy()
    conv = np.convolve(f, f)
    current_max = conv.max()
    
    improved = True
    iter_count = 0
    while improved and iter_count < max_iters:
        improved = False
        iter_count += 1
        
        # Try all flips
        best_gain = 0
        best_action = None
        
        for i in range(n):
            if f[i] == 1:
                # Try removing
                new_conv = conv.copy()
                new_conv[i:i+n] -= 2*f  # subtract the contribution
                new_conv[2*i] -= 1
                # But we need to be careful: f[i] is still 1 in the delta
                # Actually, removing f[i]: new_f = f - e_i
                # new_conv = (f-e_i)*(f-e_i) = f*f - 2*e_i*f + e_i*e_i = conv - 2*(e_i*f) + e_{2i}
                # (e_i*f)[j] = f[j-i]
                # So new_conv[j] = conv[j] - 2*f[j-i] + [j==2i]
                # Wait, that's conv[j] - 2*f[j-i] + [j==2i]
                # But f[j-i] includes f[0] when j=i, which is the position we're removing
                # Since f[i] = 1, f[j-i] at j=i is f[0]... no wait.
                # (e_i * f)[j] = f[j-i] (with f out of bounds = 0)
                # So new_conv = conv - 2*(e_i*f) + e_{2i}
                # new_conv[j] = conv[j] - 2*f[j-i] + [j==2i]
                
                # For j in [i, i+n-1]: new_conv[j] = conv[j] - 2*f[j-i] + [j==2i]
                # For j outside: new_conv[j] = conv[j]
                
                # But wait, f[j-i] here uses the CURRENT f (before removal).
                # Since we're only changing f[i], and f[i]=1, when j-i = i (j=2i), f[j-i] = f[i] = 1.
                # So at j=2i: new_conv[2i] = conv[2i] - 2*1 + 1 = conv[2i] - 1.
                # At other j in [i, i+n-1]: new_conv[j] = conv[j] - 2*f[j-i].
                
                delta = np.zeros(2*n-1)
                delta[i:i+n] = -2*f  # -2*f[j-i] for j in [i, i+n-1]
                delta[2*i] += 1
                new_max = np.max(conv + delta)
                gain = current_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_action = ('remove', i, delta)
            else:
                # Try adding
                delta = np.zeros(2*n-1)
                delta[i:i+n] = 2*f
                delta[2*i] += 1
                new_max = np.max(conv + delta)
                gain = current_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_action = ('add', i, delta)
        
        if best_action is not None and best_gain > 0:
            action, i, delta = best_action
            conv += delta
            f[i] = 1 - f[i]
            current_max = conv.max()
            improved = True
    
    return f, conv
```

**Continuous approach using subgradient:**

Start with the 0-1 sequence (or a continuous relaxation). Then:
1. Compute conv = f * f
2. Find k* = argmax(conv)
3. Compute gradient: grad[i] = 2*f[k*-i] (with f[j]=0 for j out of bounds)
4. Update: f = f - lr * grad
5. Project: f = max(f, 0)
6. Optionally renormalize

This is like gradient descent on the non-smooth function max_k Q_k(f).

The issue is that the gradient only "sees" the current argmax, so it might oscillate between different peaks. To handle this, we can use a "smoothed" version: replace max with a soft-max, or use a subgradient with averaging.

Let me try a simple version:

def subgradient_refine(f, num_iters=10000, lr_init=0.01):
    n = len(f)
    f = f.astype(float).copy()
    
    for t in range(num_iters):
        conv = np.convolve(f, f)
        k_star = np.argmax(conv)
        
        # Gradient of Q_{k*}
        grad = np.zeros(n)
        # grad[i] = 2*f[k_star - i] for valid k_star - i
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            grad[i] = 2 * f[k_star - i]
        
        lr = lr_init / (1 + t * 0.001)
        f -= lr * grad
        f = np.maximum(f, 0)
    
    return f

def greedy_construct(n, k, verbose=False):
    """Greedy: place k ones to minimize max convolution."""
    f = np.zeros(n)
    conv = np.zeros(2*n-1)
    
    for step in range(k):
        if verbose and step % 100 == 0:
            print(f"    Greedy: step {step}/{k}, current max={conv.max():.1f}")
        
        remaining = np.where(f == 0)[0]
        if len(remaining) == 0:
            break
        
        # Precompute prefix/suffix max of conv
        cummax = np.maximum.accumulate(conv)
        rev_cummax = np.maximum.accumulate(conv[::-1])[::-1]
        
        p_arr = remaining
        # Build affected matrix: row p = conv[p:p+n]
        cols = np.arange(n)
        indices = p_arr[:, None] + cols[None, :]
        affected = conv[indices]
        
        affected_plus = affected + 2*f[None, :]
        # Add 1 at column p for row p
        rows = np.arange(len(p_arr))
        affected_plus[rows, p_arr] += 1
        
        affected_max = affected_plus.max(axis=1)
        
        # Outer max
        p_vals = p_arr
        left_max = np.where(p_vals > 0, cummax[p_vals - 1], 0)
        right_start = p_vals + n
        right_max = np.where(right_start < 2*n-1, rev_cummax[right_start], 0)
        outer_max = np.maximum(left_max, right_max)
        
        total_max = np.maximum(affected_max, outer_max)
        best_idx = total_max.argmin()
        best_p = int(p_arr[best_idx])
        
        # Update
        conv[best_p:best_p+n] += 2*f
        conv[2*best_p] += 1
        f[best_p] = 1.0
    
    return f

    f = f.astype(float).copy()

    current_score_max = conv.max()
    S = f.sum()
    current_score = 2 * n * current_score_max / (S * S) if S > 0 else np.inf
    
    for it in range(max_iters):
        if verbose and it % 100 == 0:
            print(f"    LS iter {it}, max_conv={current_score_max:.2f}, score={current_score:.4f}")
        
        # Try all additions (0 -> 1)
        zeros = np.where(f == 0)[0]
        if len(zeros) > 0:
            cummax = np.maximum.accumulate(conv)
            rev_cummax = np.maximum.accumulate(conv[::-1])[::-1]
            
            cols = np.arange(n)
            indices = zeros[:, None] + cols[None, :]
            affected = conv[indices] + 2*f[None, :]
            affected[np.arange(len(zeros)), zeros] += 1
            affected_max = affected.max(axis=1)
            
            left_max = np.where(zeros > 0, cummax[zeros - 1], 0)
            right_start = zeros + n
            right_max = np.where(right_start < 2*n-1, rev_cummax[right_start], 0)
            outer_max = np.maximum(left_max, right_max)

            best_add_idx = total_max.argmin()
            best_add_max = total_max[best_add_idx]

            # But f[i] = 1, so at j=2i: conv[2i] - 2*1 + 1 = conv[2i] - 1
            # At other j in [i, i+n-1]: conv[j] - 2*f[j-i]
            affected = conv[indices] - 2*f[None, :]
            affected[np.arange(len(ones)), ones] += 1  # the [j==2i] term
            affected_max = affected.max(axis=1)
            
            left_max = np.where(ones > 0, cummax[ones - 1], 0)
            right_start = ones + n
            right_max = np.where(right_start < 2*n-1, rev_cummax[right_start], 0)
            outer_max = np.maximum(left_max, right_max)
            
            total_max = np.maximum(affected_max, outer_max)
            best_rmv_idx = total_max.argmin()
