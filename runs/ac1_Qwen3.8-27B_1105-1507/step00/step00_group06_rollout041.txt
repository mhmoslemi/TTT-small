def _next_pow2(n):
    """Next power of 2 >= n."""
    p = 1
    while p < n:
        p *= 2
    return p


def _conv_sq(p, size):
    """Compute linear convolution p*p using FFT of given size."""
    P = fft(p, n=size)
    c = ifft(P * P).real
    return c[:2 * len(p) - 1]


def _simplex_proj(v):
    """Project v onto the probability simplex {p: sum p = 1, p >= 0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, 1.0 / n)
    r = idx[-1]
    theta = (cssv[r] - 1.0) / (r + 1)
    p = np.maximum(v - theta, 0.0)
    s = p.sum()
    if s > 0:
        p /= s
    else:
        p = np.full(n, 1.0 / n)
    return p


def _compute_gradient(p, w, size_grad):
    """Compute gradient of smoothed max of conv(p,p) w.r.t. p.
    
    grad_j = 2 * sum_i p[i] * w[j+i] for j=0,...,n-1 (where j+i < 2n-1)
    This is the cross-correlation of p and w, evaluated at lags 0 to n-1.
    """
    n = len(p)
    P_ft = fft(p, n=size_grad)
    W_ft = fft(w, n=size_grad)
    corr = ifft(np.conj(P_ft) * W_ft).real
    return 2.0 * corr[:n]


def _smoothed_max_and_grad(p, c, beta, size_grad):
    """Compute the gradient of the log-sum-exp smoothing of max(c).
    
    c = conv(p, p) is already computed.
    Returns (grad, smoothed_value).

    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    Z = exp_bc.sum()
    w = exp_bc / Z
    grad = _compute_gradient(p, w, size_grad)
    smoothed_val = c_max + np.log(Z) / beta
    return grad, smoothed_val
