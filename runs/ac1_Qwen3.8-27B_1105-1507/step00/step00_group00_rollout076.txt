import numpy as np
import time
import math

"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using soft-max gradient descent
from the truncated infinite flat sequence (p_i = binom(2i,i)/4^i)
with multiple lengths, initializations, and parameter sweeps.
"""


def compute_convolution(a):
    """Compute linear self-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(fa * fa, fft_size)[:size]
    np.maximum(b, 0.0, out=b)
    return b


def compute_gradient(a, w):
    """
    Compute grad[j] = 2 * sum_k w[k] * a[k-j] for valid indices.
    Uses FFT-based convolution for efficiency.
    """
    n = len(a)
    w_len = len(w)
    conv_len = n + w_len - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    fw = np.fft.rfft(w[::-1], fft_size)
    conv_full = np.fft.irfft(fa * fw, fft_size)[:conv_len]
    return 2.0 * conv_full[n - 1:2 * n - 1][::-1]


def compute_p_coefficients(n):
    """Compute p_i = binom(2i, i) / 4^i for i = 0, ..., n-1."""
    p = np.empty(n)
    p[0] = 1.0
    for i in range(1, n):
        p[i] = p[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return p


def optimize_sequence(a_init, deadline, T_start, T_end, lr_start, lr_end, max_iters):
    """
    Optimize to minimize 2n*max(b) with sum(a)=1, a>=0 using soft-max GD.
    Returns (best_a_normalized, best_score).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = np.sum(a)
    if S <= 1e-15:
        return a_init.copy(), np.inf
    a = a / S

    best_a = a.copy()
    best_max_b = np.inf
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = compute_convolution(a)
        max_b = float(np.max(b))

        if max_b < best_max_b:
            best_max_b = max_b
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Adaptive: if stuck, boost temperature to escape
        frac = it / max(1, max_iters - 1)
        T = T_start * (T_end / T_start) ** frac
        if no_improve_count > 500:
            T = max(T, T_end)
            no_improve_count = 0

        lr = lr_start * (lr_end / lr_start) ** frac

        # Soft-max weights
        bT = b * T
        bT_max = np.max(bT)
        exp_bT = np.exp(bT - bT_max)
        w = exp_bT / np.sum(exp_bT)

        # Gradient via FFT
        grad = compute_gradient(a, w)

        # Update with non-negativity projection and renormalization
        a = a - lr * grad
        np.maximum(a, 0.0, out=a)
        S = np.sum(a)
        if S < 1e-15:
            break
        a = a / S

    score = 2.0 * n * best_max_b
    return best_a, score
