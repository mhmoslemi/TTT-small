def _next_pow2(m):
    n = 1

def _autocorr(f, N):
    """Linear autocorrelation (full convolution with self) via FFT."""
    F = np.fft.rfft(f, N)
    return np.fft.irfft(F * F, N)


def _proj_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _grad_soft(f, h, N2, n, gamma):
    """Soft-peak weighted gradient: g[i] = sum_k w[k]*f[k-i]."""
