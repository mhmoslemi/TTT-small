def softmax_subgradient_step(a, T, eta):
   n=len(a)
   L=2*n-1
   N=1 << (3*n-2).bit_length()
   A=np.fft.rfft(a, N)
   C=np.fft.irfft(A*A, N=N)[:L]
   P=float(np.max(C))
   # current score not needed
   m = T * (C - P)
   # To avoid overflow/under: m<=0; if m.max()>0 due round, shift
   if m.max() > 0: m -= m.max()
   w = np.exp(m)
   ws = w.sum()
   if ws <= 0 or not np.isfinite(ws):
       return a.copy(), fast_score(a)
   w /= ws
   W=np.fft.rfft(w, N)

"""
Searches for nonnegative step functions with small normalized convolution peak.

Strategy:
1. Use a long length near the available parent length.
2. Start from one-sided power-law / analytic seeds, especially around the
   classical continuum limit a_i ~ (i + c)^(-alpha).
3. Refine with a smooth softmax-subgradient projected onto the simplex.
4. Polish with exact one-dimensional coordinate descent while maintaining
   the convolution incrementally.
5. Return the best sequence by the authoritative evaluate_sequence score.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as scipy_minimize
except Exception:
    scipy_minimize = None


def next_power_two_ge(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()


def full_convolution_fft(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    out_len = 2 * n - 1
    if n <= 256:
        return np.convolve(a, a)
    nfft = next_power_two_ge(out_len)
    fa = np.fft.rfft(a, nfft)
    c = np.fft.irfft(fa * fa, n=nfft)
    return c[:out_len]


def normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return a
    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.0).copy()
    a = np.clip(a, 0.0, 1.0e6).copy()
    s = float(a.sum())
    if s <= 1.0e-300:
        return np.zeros_like(a)
    return a / s


def score_normalized(a):
    """
    Assumes a is nonnegative and normalized so that sum(a) = 1.
    Returns 2*n*max(a*a).
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.inf
    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1.0e6)
    conv = full_convolution_fft(a)
    if conv.size == 0:
        return np.inf
    p = float(np.max(conv))
    if not np.isfinite(p):
        return np.inf
    if p < 0.0:
        if p > -1.0e-14:
            p = 0.0
        else:
            return np.inf
    return float(2.0 * n * p)


def fast_score(a):
    a = normalize_sequence(a)
    if a.size == 0:
        return np.inf
    return score_normalized(a)


def make_params_sequence(params, n):
    n = int(n)
    if n <= 0:
        return None
    try:
        alpha = float(params[0])
        c = float(params[1])
        p1 = float(params[2])
        p2 = float(params[3])
        r = float(params[4])
    except Exception:
        return None

    if not np.isfinite(alpha) or not np.isfinite(c) or not np.isfinite(p1) \
            or not np.isfinite(p2) or not np.isfinite(r):
        return None

    alpha = max(0.05, min(0.95, alpha))
    c = max(1.0e-3, c)
    p1 = max(-0.9, min(0.9, p1))
    p2 = max(-0.9, min(0.9, p2))
    r = max(0.0, min(0.5, r))

    i = np.arange(n, dtype=np.float64)
    base = (i + c) ** (-alpha)

    if abs(p1) > 1.0e-12 or abs(p2) > 1.0e-12:
        t = i / float(n - 1) if n > 1 else np.zeros(1, dtype=np.float64)
        mult = 1.0 + p1 * t + p2 * (t * t)
        mult = np.clip(mult, 0.0, 4.0)
        base = base * mult

    if r > 1.0e-12:
        base = base + r * ((n - 1 - i) + c) ** (-alpha)

    base = np.clip(base, 0.0, 1.0e6)
    s = float(base.sum())
    if s <= 1.0e-300:
        return None
    return base / s


def score_params(params, n):
    seq = make_params_sequence(params, n)
    if seq is None:
        return 1.0e18
    sc = score_normalized(seq)
    if not np.isfinite(sc):
        return 1.0e18
    return float(sc)


def coarse_grid_n(n, deadline):
    n = int(n)
    best_seq = None
    best_score = np.inf
    best_alpha = 0.5
    best_c = 0.5

    i = np.arange(n, dtype=np.float64)
    alphas = np.arange(0.46, 0.6000001, 0.002)
    cs = np.array([0.25, 0.40, 0.50, 0.70, 1.00, 1.50], dtype=np.float64)

    for alpha in alphas:
        if time.monotonic() > deadline:
            break
        for c in cs:
            if time.monotonic() > deadline:
                break
            try:
                a = (i + c) ** (-alpha)
                s = float(a.sum())
                if s <= 0.0:
                    continue
                a = a / s
                sc = score_normalized(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = float(sc)
                    best_seq = a.copy()
                    best_alpha = float(alpha)
                    best_c = float(c)
            except Exception:
                continue

    return best_seq, best_score, best_alpha, best_c


def optimize_parametric(n, deadline):
    n = int(n)
    seq, sc, alpha, c = coarse_grid_n(n, deadline)

    if seq is None:
        seq = np.ones(n, dtype=np.float64) / float(n)
        sc = score_normalized(seq)
        alpha = 0.5
        c = 0.5

    if scipy_minimize is not None:
        x0_list = [
            [alpha, c, 0.0, 0.0, 0.0],
            [alpha * 0.995, c * 0.9, 0.0, 0.0, 0.0],
            [alpha * 1.005, c * 1.1, 0.0, 0.0, 0.0],
            [alpha, c, -0.10, 0.06, 0.0],
            [alpha, c,  0.10, 0.06, 0.0],
            [alpha, c,  0.0, 0.0, 0.08],
        ]

        opts = {
            "maxiter": 80,
            "maxfev": 260,
            "xtol": 1.0e-7,
            "ftol": 1.0e-9,
        }

        for x0 in x0_list:
            if time.monotonic() > deadline:
                break
            x0 = np.asarray(x0, dtype=np.float64)
            try:
                res = scipy_minimize(
                    score_params, x0, args=(n,),
                    method="Powell", options=opts
                )
                if res is not None and np.isfinite(res.fun):
                    seq2 = make_params_sequence(res.x, n)
                    if seq2 is not None:
                        sc2 = score_normalized(seq2)
                        if np.isfinite(sc2) and sc2 < sc:
                            seq = seq2
                            sc = float(sc2)
                            alpha = float(res.x[0])
                            c = float(res.x[1])
            except Exception:
                pass

    return seq, float(sc)


def softmax_subgradient_step(a, T, eta):
    a = normalize_sequence(a)
    n = a.size
    if n == 0:
        return a.copy(), np.inf

    L = 2 * n - 1
    nfft = next_power_two_ge(3 * n - 2)

    fa = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:L]

    if conv.size == 0:
        return a.copy(), np.inf

    p = float(np.max(conv))
    if not np.isfinite(p) or p < 0.0:
        p = 0.0

    m = T * (conv - p)
    mmax = float(np.max(m))
    if mmax > 0.0:
        m = m - mmax

    w = np.exp(m)
    ws = float(w.sum())
    if ws <= 1.0e-300 or not np.isfinite(ws):
        cur = score_normalized(a)
        return a.copy(), cur

    w = w / ws
    fw = np.fft.rfft(w, nfft)
    cross = np.fft.irfft(fa * np.conj(fw), n=nfft)
    grad = 2.0 * cross[:n]
    grad = np.where(np.isfinite(grad), grad, 0.0)

    if float(eta) <= 0.0:
        cur = score_normalized(a)
        return a.copy(), cur

    cand = a - float(eta) * grad
    cand = np.clip(cand, 0.0, None)
    s = float(cand.sum())

    if s <= 1.0e-300:
        cur = score_normalized(a)
        return a.copy(), cur

    cand = cand / s
    sc = score_normalized(cand)
    if not np.isfinite(sc):
        cur = score_normalized(a)
        return a.copy(), cur

    return cand, sc


def local_subgradient_optimize(start, deadline):
    a = normalize_sequence(start.copy())
    n = a.size
    if n == 0:
        return a, np.inf

    best = a.copy()
    best_score = score_normalized(a)

    if n <= 1:
        return best, float(best_score)

    t_factors = [1.0, 4.0, 16.0, 64.0]
    eta_candidates = [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 1.0]

    for tf in t_factors:
        if time.monotonic() > deadline:
            break

        T = tf * float(n)
        current = best.copy()
        current_score = best_score

        best_cand = None
        best_step_score = current_score
        best_eta = 0.01

        for eta in eta_candidates:
            if time.monotonic() > deadline:
                break
            try:
                cand, sc = softmax_subgradient_step(current, T, eta)
                if np.isfinite(sc) and sc < best_step_score:
                    best_step_score = float(sc)
                    best_cand = cand
                    best_eta = float(eta)
            except Exception:
                pass

        if best_cand is None:
            continue

        current = best_cand
        current_score = best_step_score
        eta = best_eta
        stalls = 0
        improvements = 0

        while time.monotonic() < deadline and stalls < 50:
            try:
                cand, sc = softmax_subgradient_step(current, T, eta)
            except Exception:
                break

            if np.isfinite(sc) and sc < current_score - 1.0e-12:
                current = cand
                current_score = float(sc)
                stalls = 0
                improvements += 1

                if current_score < best_score:
                    best = current.copy()
                    best_score = current_score
                    if improvements % 25 == 0 or best_score < 1.50:
                        print(f"  subgradient n={n} score={best_score:.9f}")

                eta *= 1.08
                if eta > 50.0:
                    eta = 50.0
            else:
                stalls += 1
                eta *= 0.5

    return best, float(best_score)


def coordinate_refine(start, deadline):
    a = normalize_sequence(start.copy())
    n = int(a.size)
    if n <= 0:
        return a, np.inf

    L = 2 * n - 1
    S = float(a.sum())
    if S <= 1.0e-300:
        return a, np.inf

    an = a / S
    conv_norm = full_convolution_fft(an)
    C = conv_norm * (S * S)
    current_score = float(2.0 * n * np.max(conv_norm))

    best = a.copy()
    best_score = current_score

    base = np.empty(n, dtype=np.float64)
    contrib = np.empty(n, dtype=np.float64)
    tmp = np.empty(n, dtype=np.float64)

    indices = np.arange(n, dtype=np.int64)
    rng = np.random.default_rng(20240517)

    updates = 0
    imps = 0

    for _pass in range(24):
        if time.monotonic() > deadline:
            break

        rng.shuffle(indices)
        pass_improved = False

        for pos in range(n):
            if time.monotonic() > deadline:
                break

            j = int(indices[pos])
            old = float(a[j])

            k_start = j
            k_stop = min(L, j + n)
            seg_len = k_stop - k_start
            if seg_len <= 0:
                continue

            np.multiply(a, 2.0 * old, out=contrib)
            if j < n:
                contrib[j] = old * old
            if seg_len < n:
                contrib[seg_len:] = 0.0

            np.subtract(C[k_start:k_stop], contrib[:seg_len], out=base[:seg_len])

            max_out = 0.0
            if k_start > 0:
                v = float(np.max(C[:k_start]))
                if v > max_out:
                    max_out = v
            if k_stop < L:
                v = float(np.max(C[k_stop:]))

                    max_out = v

            if old > 1.0e-300:
                x_vals = (

                    old * 0.75,
                    old,
                    old * 1.25,
                    old * 1.50,
                    old * 2.00,
                )
            else:
                pos_vals = a[a > 0.0]
                med = float(np.median(pos_vals)) if pos_vals.size > 0 else 1.0 / float(n)

                    0.0,

                    med * 0.50,

            best_x = old

                np.multiply(a, 2.0 * xv, out=contrib)
                if j < n:

                if seg_len < n:
                    contrib[seg_len:] = 0.0

                np.add(base[:seg_len], contrib[:seg_len], out=tmp[:seg_len])
                cmax_seg = float(np.max(tmp[:seg_len]))
                cmax = cmax_seg if cmax_seg > max_out else max_out
                if cmax < 0.0 and cmax > -1.0e-12:
                    cmax = 0.0

                S_new = S - old + xv
                if S_new <= 1.0e-300:
                    continue

                sc = 2.0 * float(n) * cmax / (S_new * S_new)
                if sc < best_sc:
                    best_sc = float(sc)
                    best_x = xv

            if best_x != old and best_sc < current_score:
                if old > 0.0 and best_x > 0.0:
                    refine_vals = (
                        best_x * 0.80,
                        best_x * 0.90,
                        best_x * 0.95,
                        best_x * 1.05,
                        best_x * 1.10,

                    )
                else:
                    refine_vals = (
                        best_x * 0.50,
                        best_x * 0.80,
                        best_x * 1.20,

                    )

                for xv in refine_vals:

                    if xv < 0.0:

                    np.multiply(a, 2.0 * xv, out=contrib)
                    if j < n:

                    if seg_len < n:

                    np.add(base[:seg_len], contrib[:seg_len], out=tmp[:seg_len])
                    cmax_seg = float(np.max(tmp[:seg_len]))
                    cmax = cmax_seg if cmax_seg > max_out else max_out
                    if cmax < 0.0 and cmax > -1.0e-12:
                        cmax = 0.0

                    S_new = S - old + xv
                    if S_new <= 1.0e-300:
                        continue

                    sc = 2.0 * float(n) * cmax / (S_new * S_new)
                    if sc < best_sc:
                        best_sc = float(sc)
                        best_x = xv

            if best_sc < current_score - 1.0e-12 and best_x >= 0.0:
                np.multiply(a, 2.0 * best_x, out=contrib)
                if j < n:
                    contrib[j] = best_x * best_x
                if seg_len < n:
                    contrib[seg_len:] = 0.0

                np.add(base[:seg_len], contrib[:seg_len], out=tmp[:seg_len])
                C[k_start:k_stop] = tmp[:seg_len]

                S = S - old + best_x
                a[j] = best_x
                current_score = best_sc

                if current_score < best_score:
                    best_score = current_score
                    best = a.copy()
                    pass_improved = True
                    imps += 1
                    if imps % 100 == 0 or best_score < 1.50:
                        print(f"  coordinate n={n} score={best_score:.9f}")

                updates += 1

                if S < 0.5 or S > 2.0:

                    a = a * scale
                    C = C * (scale * scale)
                    S = 1.0

                if updates % 512 == 0:
