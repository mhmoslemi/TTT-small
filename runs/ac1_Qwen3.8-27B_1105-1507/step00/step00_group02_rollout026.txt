from scipy.optimize import linprog

def lp_direction(f, M):
    """Find the LP direction: max sum(b) s.t. b>=0, |f*b|_inf <= M."""
    n = len(f)
    # Variables: b[0], ..., b[n-1]
    # Constraints: (f*b)[k] <= M for k = 0, ..., 2n-2
    # (f*b)[k] = sum_j f[j] * b[k-j] (with valid index range)
    
    # Build constraint matrix A (2n-1 rows, n cols)
    # A[k, j] = coefficient of b[j] in (f*b)[k]
    # (f*b)[k] = sum_j f[j] * b[k-j] where 0<=j<n and 0<=k-j<n
    # Let i = k-j, so j = k-i, and we need 0<=i<n: A[k, k-i] = f[i]
    # So A[k, j] = f[k-j] if 0<=k-j<n, else 0
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Bounds: b[j] >= 0
    bounds = [(0, None)] * n
    
    # Constraints: A*b <= M
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x
    return None

    max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns the optimal b or None.
    """
    # Build constraint matrix
    # (a*b)[k] = sum_j a[j] * b[k-j] for valid j
    # Coefficient of b[j] in (a*b)[k] is a[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency, build the matrix row by row
    # A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n and 0 <= j < n, else 0
    
    # Using the structure: for row k, the non-zero entries are at columns
    # j = max(0, k-n+1) to min(n-1, k), with values a[k-j] = a[max(0, k-n+1) ... k - max(0, k-n+1)]
    
    # Actually, let's just build it directly for clarity (n is at most 2048)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.status == 0:
        return res.x
    return None


def fast_objective(a):
    """Fast objective using FFT."""
    n = len(a)
    sum_a = a.sum()
    if sum_a < 1e-10:
        return np.inf
    # FFT-based convolution
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:size]
    return float(2.0 * n * conv.max() / (sum_a * sum_a))
