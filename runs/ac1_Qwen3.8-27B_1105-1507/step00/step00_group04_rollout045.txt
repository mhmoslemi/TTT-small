def evaluate_fast(a):

    """Small Gaussian perturbation of a sequence (keeping non-negative)."""

    new_a = a.copy()
    new_a[start:end] *= np.exp(rng.normal(0, 0.1, end - start))
    return np.maximum(new_a, 0.0)
