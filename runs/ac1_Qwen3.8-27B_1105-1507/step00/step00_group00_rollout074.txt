    j_arr = np.repeat(np.arange(n), n)
    t_arr = np.tile(np.arange(n), n)
    rows = j_arr + t_arr
    cols = j_arr

def lp_step(a, M, solver='CLARABEL'):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)<=M. Return normalized b."""
    n = len(a)
    A = build_conv_matrix(a)
    b = cp.Variable(n)
    constraints = [b >= 0, A @ b <= M * np.ones(2*n-1)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        prob.solve(solver=solver)

        try:

    # Parabolic
    para = 1.0 - (2 * t - 1) ** 2
    para = np.clip(para, 0.05, 1.0)
    inits.append(para / para.sum())
    # Sine window
    sin_w = np.sin(np.pi * np.arange(n) / n)
    sin_w = np.clip(sin_w, 0.01, 1.0)
    inits.append(sin_w / sin_w.sum())
    # Uniform
    inits.append(np.ones(n) / n)
    return inits


def lp_optimize(a, deadline, max_iters=50):
    """Run LP-based optimization on sequence a (should have sum ~1)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_M = np.convolve(a, a).max()
    alpha = 0.4
    it = 0

    while it < max_iters and time.monotonic() < deadline:
        C = np.convolve(a, a)
        M = C.max()
        if M < best_M:
            best_M = M
            best_a = a.copy()

        # Try LP step
        b = lp_step(a, M)
        if b is not None:
            a_sum = a.sum()
            b_scaled = b * a_sum
            a_new = (1 - alpha) * a + alpha * b_scaled
            a_new = np.clip(a_new, 0, None)
            sn = a_new.sum()
            if sn < 1e-10:
                break
            a_new = a_new / sn
        else:
            # Gradient descent fallback
            grad = grad_descent_step(a, top_k=8)
            if grad is None:
                break
            a_new = a - 0.03 * grad
            a_new = np.clip(a_new, 0, None)
            sn = a_new.sum()
            if sn < 1e-10:
                break
            a_new = a_new / sn

        a = a_new
        it += 1

        # Adapt step
        C_check = np.convolve(a, a)
        if C_check.max() < best_M * 1.005:
            alpha = max(0.1, alpha * 0.96)
        else:
            alpha = min(0.85, alpha * 1.04)

    return best_a


def grad_optimize(a, deadline, max_iters=5000):
    """Pure gradient descent optimization for larger n where LP is too expensive."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_M = np.convolve(a, a).max()
    eta = 0.05
    it = 0
    stall = 0

    while it < max_iters and time.monotonic() < deadline:
        # Batch of iterations to amortize overhead
        if it % 50 == 0:
            C = np.convolve(a, a)
            M = C.max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
            if M > best_M * 1.02:
                stall += 1
                if stall > 20:
                    # Perturb to escape
                    rng_local = np.random.default_rng(hash(tuple(a[:10])) & 0xFFFF)
                    a += rng_local.normal(0, 0.01, n)
                    a = np.clip(a, 0, None)
                    s = a.sum()
                    if s < 1e-10:
                        break
                    a = a / s
                    stall = 0

        grad = grad_descent_step(a, top_k=10)
        if grad is None:
            break
        a = a - eta * grad
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        eta = max(eta * 0.9995, 0.001)
        it += 1

    # Final check
    C = np.convolve(a, a)
    if C.max() < best_M:
        best_a = a
    return best_a


def local_search(a, deadline, rng):
    """Local search with block perturbations."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_M = np.convolve(a, a).max()
    scale = 0.15

    while time.monotonic() < deadline:
        width = int(rng.integers(1, max(2, n // 10)))
        start = int(rng.integers(0, max(1, n - width)))
        a_new = a.copy()
        a_new[start:start + width] *= np.exp(rng.normal(0, scale, width))
        a_new = np.clip(a_new, 0, None)
        sn = a_new.sum()
        if sn < 1e-10:
            continue
        a_new = a_new / sn
        M_new = np.convolve(a_new, a_new).max()

        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            scale = max(0.03, scale * 0.98)
        else:
            a = a_new
            if M_new > best_M * 1.5:
                a = best_a.copy()  # revert if too bad

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-length LP-based optimization with gradient descent
    fallback and local search refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = np.inf

    # Try previous best if available
    if 'height_sequence_1' in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0:
            prev_clip = np.clip(prev, 0, 1000)
            if prev_clip.sum() >= 0.01:
                sc = float(evaluate_sequence(prev_clip.tolist()))
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = prev_clip.copy()
                    print(f"Previous best: {sc:.6f}, len={prev.size}", flush=True)

    # Search configuration: (length, max_n_for_lp, num_inits, max_iters)
    search_plan = [
        (200, 200, 5, 40),
        (400, 400, 5, 35),
        (800, 800, 4, 25),
        (1500, 1500, 3, 20),
        (3000, 0, 3, 3000),  # 0 means use gradient descent only
        (5000, 0, 2, 5000),
    ]

    for n, lp_limit, n_inits, max_iters in search_plan:
        if time.monotonic() > deadline:
            break
        time_left = deadline - time.monotonic()
        if time_left < 40:
            break

        # Estimate time per init
        if lp_limit > 0:
            time_per_init = min(60, time_left * 0.15)
        else:
            time_per_init = min(40, time_left * 0.12)

        print(f"\n=== n={n}, LP={'yes' if lp_limit>0 else 'no'}, time_left={time_left:.0f}s ===", flush=True)

        inits = make_initializations(n)[:n_inits]

        for ii in range(len(inits)):
            if time.monotonic() > deadline:
                break
            if time.monotonic() + 10 > deadline:
                break

            a0 = inits[ii].astype(np.float64).copy()

            # Add some random variation for non-first inits
            if ii > 0:
                noise = rng.normal(0, 0.1, n)
                a0 = a0 + noise
                a0 = np.clip(a0, 0, None)
                s0 = a0.sum()
                if s0 < 1e-10:
                    continue
                a0 = a0 / s0

            sub_deadline = time.monotonic() + time_per_init

            if lp_limit > 0 and n <= lp_limit:
                a_opt = lp_optimize(a0, sub_deadline, max_iters=max_iters)
            else:
                a_opt = grad_optimize(a0, sub_deadline, max_iters=max_iters)

            # Evaluate
            a_eval = np.clip(a_opt, 0, 1000)
            sc = float(evaluate_sequence(a_eval.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_eval.copy()
                print(f"  NEW BEST: {sc:.6f} (n={n}, init={ii})", flush=True)

        print(f"  Best so far: {best_score:.6f}", flush=True)

    # Also try a random restart at the best length
    if best_seq is not None and time.monotonic() < deadline:
        time_left = deadline - time.monotonic()
        if time_left > 30:
            n_best = len(best_seq)
            # Try a few random perturbations of the best
            for attempt in range(5):
                if time.monotonic() > deadline:
                    break
                a_try = best_seq.astype(np.float64).copy()
                a_try *= np.exp(rng.normal(0, 0.1, n_best))
                a_try = np.clip(a_try, 0, None)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sub_dl = time.monotonic() + min(20, time_left * 0.3)
                a_opt = grad_optimize(a_try, sub_dl, max_iters=2000)
                a_eval = np.clip(a_opt, 0, 1000)
                sc = float(evaluate_sequence(a_eval.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_eval.copy()
                    print(f"  Random restart NEW BEST: {sc:.6f}", flush=True)

    # Final local search
    if best_seq is not None and time.monotonic() < deadline:
        time_left = deadline - time.monotonic()
        if time_left > 15:
            print(f"\n=== Final local search ({time_left:.0f}s) ===", flush=True)
            a_final = local_search(best_seq, deadline, rng)
            a_final = np.clip(a_final, 0, 1000)
            sc = float(evaluate_sequence(a_final.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_final.copy()
                print(f"  After local search: {sc:.6f}", flush=True)

    if best_seq is None:
        best_seq = np.ones(100)

    print(f"\nFinal: score={best_score:.6f}, length={len(best_seq)}", flush=True)
    return [float(x) for x in best_seq]
