    a = a.copy()

```python
def gradient_step(a, eta=0.01, n_peaks=5):
    """One gradient step to reduce the peak of conv(a,a)."""
    n = len(a)
    c = np.convolve(a, a)  # or use FFT
    # Find top n_peaks positions
    top_k = np.argsort(c)[::-1][:n_peaks]
    # Compute combined gradient
    g = np.zeros(n)
    for k in top_k:
        # ∂c[k]/∂a[j] = 2 * a[k-j] if valid
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        g[j_min:j_max+1] += 2.0 * a[k - j_max : k - j_min + 1][::-1]
    # Center to maintain sum
    g -= np.sum(g) / n
    # Update
    a_new = a - eta * g
    a_new = np.clip(a_new, 0, 1000)
    return a_new
```

This is O(n * n_peaks) per step (plus the O(n log n) for the convolution). Very fast!

import time
import math

def fft_convolve(a):
    """Full self-convolution using FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    Fa = np.fft.rfft(a, size)
    return np.fft.irfft(Fa * Fa, size)[:2 * n - 1]

def fast_J(a):
    """Fast J computation."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (s * s)
