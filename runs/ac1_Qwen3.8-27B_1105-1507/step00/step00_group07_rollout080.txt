def compute_grad(a, p, n):
    """Compute grad[j] = sum_d (p[j+d] + p[j-d]) * a[d] for j=0..2n-2,
    then return only the first n entries (since a has length n)."""
    # a has length n, p has length 2n-1
    a_padded = np.zeros(2*n-1)
    a_padded[:n] = a
    
    a_rev_padded = np.zeros(2*n-1)
    a_rev_padded[:n] = a[::-1]
    
    # Use FFT for convolution
    N = 2 * (2*n - 1)  # FFT size (or next power of 2)
    P = np.fft.rfft(p, N)
    
    A = np.fft.rfft(a_padded, N)

    term1_full = np.fft.irfft(P * A_rev, N)[n-1:n-1+(2*n-1)]
    
    grad = term1_full[:n] + term2_full[:n]
    return grad
