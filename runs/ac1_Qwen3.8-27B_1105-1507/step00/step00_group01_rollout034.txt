import numpy as np
import time
from scipy.fft import fft, ifft

def fast_convolve(a):
    """Compute full convolution of a with a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()  # next power of 2
    A = fft(a, fft_size)
    result = ifft(A * A).real
    return result[:size]

def fast_score(a):
    """Compute the score 2n * max(convolve(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fast_convolve(a)
    max_b = b.max()
    return 2.0 * n * max_b / (s * s)

def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if np.all(v <= 0):
        return np.full(n, target_sum / n)
    # Scale to get sum = target_sum using water-filling
    v = v.copy()
    # First, find the threshold
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho = np.where(mu * np.arange(1, n+1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    # Fix numerical issues
    result_sum = result.sum()
    if result_sum > 0:
        result *= target_sum / result_sum
    else:
        result = np.full(n, target_sum / n)
    return result

def compute_smoothed_gradient(a, tau, fft_size_grad):
    """Compute gradient of (1/tau)*log(sum(exp(tau * convolve(a,a)))) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    # Softmax
    shifted = tau * b - np.max(tau * b)

"""
Minimize 2n*max(convolve(a,a))/(sum(a))^2 using smoothed gradient descent
on the probability simplex with log-sum-exp or median smoothing,
multiple restarts, simulated annealing, and optional LP polishing.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    result = ifft(A * A).real
    return result[:size]


def fast_score(a):
    """Compute 2n * max(convolve(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fast_convolve(a)
    max_b = b.max()
    return 2.0 * n * max_b / (s * s)


def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = v.copy()
    if np.all(v <= 0):
        return np.full(n, target_sum / n)
    mu = np.sort(v)[::-1]
    cssv = np.cumsum(mu)
    rho_arr = np.where(mu * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_arr) == 0:
        return np.full(n, target_sum / n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    result_sum = result.sum()
    if result_sum > 0:
        result *= target_sum / result_sum
    else:
        result = np.full(n, target_sum / n)
    return result


def compute_gradient_logsumexp(a, tau, grad_fft_size):
    """Gradient of log-sum-exp(tau * convolve(a,a)) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    shifted = tau * b - np.max(tau * b)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w /= w_sum
    W = fft(w, grad_fft_size)
    A = fft(a, grad_fft_size)
    grad = np.real(ifft(np.conj(W) * A))[:n]
    return 2.0 * grad


def compute_gradient_median(a, k, grad_fft_size):
    """Gradient of k-th largest value of convolve(a,a) (subgradient)."""
    n = len(a)
    b = fast_convolve(a)
    idx = np.argpartition(b, -k)[-k:]
    w = np.zeros(2 * n - 1)
    w[idx] = 1.0 / k
    W = fft(w, grad_fft_size)
    A = fft(a, grad_fft_size)
    grad = np.real(ifft(np.conj(W) * A))[:n]
    return 2.0 * grad


def run_smoothed_descent(a0, n, conv_fft_size, grad_fft_size, budget_s, rng, tag=""):
    """Run smoothed gradient descent from initial point a0."""
    start = time.monotonic()
    deadline = start + budget_s - 5.0
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a)
    if not np.isfinite(best_score):
        return a0, np.inf

    tau = 20.0
    k_median = max(5, n // 50)
    n_iters = 0

    while time.monotonic() < deadline:
        b = fast_convolve(a)
        max_b = b.max()
        current_score = 2.0 * n * max_b  # since sum(a)=1

        # Annealing acceptance
        accept = False
        if current_score < best_score:
            accept = True
            best_a = a.copy()
            best_score = current_score
        else:
            # Occasionally accept worse with probability based on improvement potential
            if rng.random() < 0.02:
                accept = True

        # Compute gradient (alternate between logsumexp and median)
        if n_iters % 2 == 0:
            grad = compute_gradient_logsumexp(a, tau, grad_fft_size)
        else:
            grad = compute_gradient_median(a, k_median, grad_fft_size)

        # Adaptive learning rate
        lr = 0.01 / (1.0 + 0.01 * n_iters)
        a_new = a - lr * grad
        a_new = simplex_project(a_new, 1.0)

        if accept or rng.random() < 0.1:
            a = a_new

        # Cool down tau
        tau *= 0.97
        if tau < 1.0:
            tau = 1.0

        n_iters += 1
        if n_iters % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"  [{tag}] iter={n_iters}, best_score={best_score:.6f}, "
                  f"elapsed={elapsed:.1f}s", flush=True)

    return best_a, best_score


def run_lp_polish(a, n, budget_s, tag=""):
    """Try LP-based direction finding to improve a."""
    start = time.monotonic()
    deadline = start + budget_s - 5.0
    best_a = a.copy()
    best_score = fast_score(a)
    if not np.isfinite(best_score):
        return a, np.isfinite(best_score) and best_score

    try:
        import cvxpy as cp
    except ImportError:
        return a, best_score

    a = a.copy()
    for it in range(50):
        if time.monotonic() > deadline:
            break
        b = fast_convolve(a)
        M = b.max()
        # Use active set: top k constraints
        k_active = 50
        top_idx = np.argpartition(b, -k_active)[-k_active:]
        # Build constraint coefficients: for active m, cross_corr(a, g)[m] <= 0
        # cross_corr(a, g)[m] = sum_j a[m-j] * g[j] (valid j)
        # So constraint: sum_j a[m-j] * g[j] <= 0 for each active m
        # This is: c_m^T g <= 0 where c_m[j] = a[m-j] if valid else 0
        constraints = []
        for m in top_idx:
            c = np.zeros(n)
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            for j in range(lo, hi + 1):
                c[j] = a[m - j]
            # Normalize to avoid numerical issues
            if np.max(np.abs(c)) > 0:
                constraints.append(c @ cp.Variable(n) <= 0)

        g = cp.Variable(n)
        constraints.append(g >= 0)
        constraints.append(cp.sum(g) == 1)
        prob = cp.Problem(cp.Minimize(cp.sum_squares(g - a)), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iter=5000)
            if g.value is not None:
                g_val = np.clip(g.value, 0, 1000)
                gs = g_val.sum()
                if gs > 0.01:
                    g_val /= gs
                    # Try full step and partial steps
                    for alpha in [1.0, 0.5, 0.3, 0.1]:
                        candidate = alpha * g_val + (1 - alpha) * a
                        candidate = simplex_project(candidate, 1.0)
                        s = fast_score(candidate)
                        if np.isfinite(s) and s < best_score:
                            best_a = candidate
                            best_score = s
                if best_score < 2.0:
                    print(f"  [{tag}] LP iter {it}: score={best_score:.6f}", flush=True)
        except Exception:
            break

    # Final verification
    final_score = evaluate_sequence(best_a.tolist())
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    return best_a, best_score


def try_structured_patterns(n, rng):
    """Generate several structured candidate sequences."""
    candidates = []

    # Parent normalized
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    if len(parent) == n and parent.sum() > 0.01:
        candidates.append(parent / parent.sum())

    # Uniform
    candidates.append(np.ones(n) / n)

    # Raised cosine
    rc = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    rc /= rc.sum()
    candidates.append(rc)

    # Hann window
    hw = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    hw /= hw.sum()
    candidates.append(hw)

    # Triangle
    tri = np.minimum(np.arange(n), 2 * (n - 1) - np.arange(n))
    tri = tri.astype(float)
    tri = np.maximum(tri, 0)
    tri_sum = tri.sum()
    if tri_sum > 0:
        candidates.append(tri / tri_sum)

    # Random binary-like
    for _ in range(5):
        rb = rng.choice([0.0, 1.0], size=n, p=[0.3, 0.7])
        if rb.sum() > 0:
            candidates.append(rb / rb.sum())

    # Alternating with noise
    alt = np.ones(n)
    alt[1::2] = 0.3
    alt += rng.normal(0, 0.1, n)
    alt = np.maximum(alt, 0)
    if alt.sum() > 0:
        candidates.append(alt / alt.sum())

    # Few wide blocks
    for _ in range(3):
        nb = rng.integers(3, 15)
        block_val = rng.uniform(0.5, 2.0, nb)
        seg = np.array_split(np.arange(n), nb)
        blk = np.zeros(n)
        for i, s in enumerate(seg):
            blk[s] = block_val[i]
        if blk.sum() > 0:
            candidates.append(blk / blk.sum())

    return candidates
