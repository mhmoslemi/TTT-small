                best = [float(v) for v in xn]
                best_score = valn
                print(f"New best with n={n}: {best_score:.10f}")
            else:
                print(f"n={n}: best found = {valn:.6f} (not better than {best_score:.6f})")
        except Exception as e:
            print(f"n={n} check failed: {e}")

    # Final verification
    final_score = float(evaluate_sequence(best))
    print(f"\nFinal answer: {best}")
    print(f"Final score: {final_score:.10f}")
    print(f"Expected 32/27: {32.0/27.0:.10f}")

    # Ensure the returned sequence is valid
    best = [float(max(0.0, min(1000.0, x))) for x in best]
    if sum(best) < 0.01:
        best = [4.0, 2.0, 3.0]

    return best
