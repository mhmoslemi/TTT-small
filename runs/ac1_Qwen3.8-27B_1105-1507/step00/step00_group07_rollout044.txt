    """One step of mirror descent to reduce peak autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    M = np.max(c)
    k_star = np.argmax(c)
    
    # Subgradient
    s = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            s[j] = 2 * a[idx]
    
    if eta is None:
        eta = 1.0 / (np.max(s) + 1e-10)
    
    # Mirror descent update
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - eta * s
    a_new = np.exp(log_a_new)
    
    # Renormalize
    sum_a = np.sum(a)
    a_new = a_new * sum_a / np.sum(a_new)
    
    return np.clip(a_new, 0, 1000)
