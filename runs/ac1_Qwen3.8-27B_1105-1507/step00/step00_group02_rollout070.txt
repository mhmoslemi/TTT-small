def _next_pow2(x):
    n = 1
    while n < x:
        n *= 2
    return n


def _smooth_grad(a, tau):
    """Gradient of soft-max of autoconvolution w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)
    scaled = tau * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300 or not np.isfinite(ws):
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    # corr[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    w_rev = w[::-1]
    need = n + len(w_rev) - 1
    size = _next_pow2(need)
    A = np.fft.rfft(a, size)
    WR = np.fft.rfft(w_rev, size)
    cf = np.fft.irfft(A * WR, size)[:need]
    corr = cf[n - 1:2 * n - 1][::-1]
    return 2.0 * corr

with FFT-based gradients, flat-spectrum alternating projections, block local
search, and multi-restart exploration.

def _next_pow2(x):
    n = 1
    while n < x:
        n *= 2
    return n


def _smooth_grad(a, tau):
    n = len(a)
    conv = np.convolve(a, a)
    scaled = tau * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300 or not np.isfinite(ws):
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    w_rev = w[::-1]
    need = n + len(w_rev) - 1
    size = _next_pow2(need)
    A = np.fft.rfft(a, size)
    WR = np.fft.rfft(w_rev, size)
    cf = np.fft.irfft(A * WR, size)[:need]
    corr = cf[n - 1:2 * n - 1][::-1]
    return 2.0 * corr

def _flat_spectrum(a0, n_iter, deadline=None):
    a = np.array(a0, dtype=np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a[:] = 1.0 / n
    else:
        a /= s
    M = 2 * n
    for it in range(n_iter):
        if deadline is not None and time.monotonic() >= deadline:
            break
        Ap = np.fft.rfft(a, M)
        mag = np.abs(Ap)
        mag[mag < 1e-14] = 1e-14
        target = mag.mean()
        Ap_new = Ap / mag * target
        a_new = np.fft.irfft(Ap_new, M)[:n]
        a = np.maximum(a_new.real, 0.0)
        s = a.sum()
        if s < 0.01:
            a[:] = 1.0 / n
        else:
            a /= s
    return a


def _block_perturb(base, rng, n_widths=5):
    a = base.copy()
    n = len(a)
    candidates = []
    for _ in range(n_widths):
        c = base.copy()
        w = int(rng.integers(max(1, n // 200), max(2, n // 8)))
        st = int(rng.integers(0, max(1, n - w + 1)))
        EN = min(n, st + w)
        c[st:EN] *= float(np.exp(rng.normal(0.0, 0.2)))
        if rng.random() < 0.3:
            c = 0.5 * (c + c[::-1])
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.01:
            c /= s
        candidates.append(c)
    return candidates
