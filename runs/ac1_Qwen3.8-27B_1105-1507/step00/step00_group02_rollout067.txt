"""
Minimize 2*n*max(conv(p,p))/sum(p)^2 using:
1. Mirror descent (multiplicative weights) with FFT-based convolution for fast iterations
2. Alternating LP refinement for a final polish
Tries multiple n values and initializations.

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def fft_convolve(p):
    """Compute linear convolution of p with itself using FFT."""
    n = len(p)
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    F = fft(p, size)
    conv = np.real(ifft(F * F, size))[:m]
    return conv


def compute_gradient_fft(p, lambda_s):
    """
    Compute gradient of sum_s lambda[s] * conv(p,p)[s] with respect to p.
    grad[j] = 2 * sum_s lambda[s] * p[s-j] + 2 * p[j] * sum(lambda)
    where the first term is the cross-correlation of p and lambda.
    """
    n = len(p)
    m = len(lambda_s)
    size = 1 << (m + n - 2).bit_length()
    F_p = fft(p, size)
    F_lambda = fft(lambda_s, size)
    # Cross-correlation: c[j] = sum_u p[u] * lambda[j+u]
    c = np.real(ifft(np.conj(F_p) * F_lambda, size))[:n]
    lam_sum = lambda_s.sum()
    grad = 2.0 * c + 2.0 * p * lam_sum
    return grad

def build_conv_matrix(f, n):
    """Build sparse matrix A where A[k,j] = f[k-j] for valid k,j. Shape (2n-1, n)."""
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for d in range(n):
            k = j + d
            rows.append(k)
            cols.append(j)
            vals.append(f[d])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_given_other(f, n):
    """
    Solve: minimize t subject to conv(f, g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t) or (None, None) on failure.
    """
    A_conv = build_conv_matrix(f, n)
    m = 2 * n - 1

    # A_ub: [A_conv | -1] (m x (n+1)), b_ub = 0
    neg_col = sp.csc_matrix(-np.ones((m, 1)))
    A_ub = sp.hstack([A_conv, neg_col], format='csc')
    b_ub = np.zeros(m)

    # A_eq: [1 1 ... 1 0] (1 x (n+1)), b_eq = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c = np.zeros(n + 1)
    c[n] = 1.0

    bounds = [(0, None)] * n + [(0, None)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 120})
        if result.status == 0:
            g = np.maximum(result.x[:n], 0)
            g = g / g.sum()
            t_val = result.x[n]
            return g, float(t_val)
    except Exception:
        pass
    return None, None


def alternating_lp(f0, g0, n, budget_s):
    """Alternate solving LPs for f and g to minimize max(conv(f,g))."""
    deadline = time.monotonic() + budget_s
    f = f0.copy().astype(np.float64)
    g = g0.copy().astype(np.float64)
    f = f / f.sum()
    g = g / g.sum()

    best_f = f.copy()
    best_g = g.copy()
    best_score = np.inf

    for it in range(200):
        if time.monotonic() > deadline:
            break

        # Solve for g given f
        g_new, _ = solve_lp_given_other(f, n)
        if g_new is not None:
            g = g_new

        # Solve for f given g
        f_new, _ = solve_lp_given_other(g, n)
        if f_new is not None:
            f = f_new

        # Evaluate
        conv = fft_convolve(f)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_f = f.copy()
            best_g = g.copy()

        if time.monotonic() > deadline:
            break

    return best_f, best_score


def make_init(n, kind, rng):

        p = np.maximum(p, 1e-10)
        return p / p.sum()
    elif kind == 'gaussian':
        center = (n - 1) / 2.0
        sigma = n / rng.uniform(2.0, 6.0)
        x = np.arange(n)
        p = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        return p / p.sum()
    elif kind == 'tukey':
        r = rng.uniform(0.3, 0.7)
        p = np.ones(n)
        half = n // 2
        m = max(1, int(r * half))
        for i in range(m):
            frac = (i + 1) / (m + 1)
            val = 0.5 * (1 + np.cos(np.pi * (2 * frac - 1)))
            p[i] = val
            p[n - 1 - i] = val
        p = np.maximum(p, 1e-10)
        return p / p.sum()
    elif kind == 'random_smooth':
        p = np.abs(rng.normal(1.0, 0.5, n))
        # Smooth with Gaussian kernel

        kernel = kernel / kernel.sum()
        p = np.convolve(p, kernel, mode='same')
        p = np.maximum(p, 1e-10)
        return p / p.sum()
    else:
        return np.ones(n) / n
