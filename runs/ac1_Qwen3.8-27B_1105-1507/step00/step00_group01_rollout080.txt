    a = np.exp(-((j - center1) ** 2) / width) + np.exp(-((j - center2) ** 2) / width)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _init_random(n, rng):
    a = rng.random(n).astype(np.float64)
    return a / a.sum()


def _local_search_coord(a, c, n, rng, num_trials=3):
    """Try coordinate moves to reduce the peak autocorrelation."""
    k_star = int(np.argmax(c))
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n - 1, k_star)
    grad = np.zeros(n)
    if j_lo <= j_hi:
        idx = np.arange(j_lo, j_hi + 1)
        grad[idx] = 2.0 * a[k_star - idx]
        if k_star % 2 == 0:
            jd = k_star // 2
            if j_lo <= jd <= j_hi:
                grad[jd] = a[jd]

    positive_vals = grad[grad > 0]
    if len(positive_vals) == 0:
        return a
    med = np.median(positive_vals)
    high_idx = np.where(grad >= med * 1.5)[0]
    low_idx = np.where(grad <= med * 0.7)[0]

    if len(high_idx) < 5:
        high_idx = np.argsort(grad)[-10:]
    if len(low_idx) < 5:
        low_idx = np.argsort(grad)[:10]

    cur_obj = 2.0 * n * c.max()

    for _ in range(num_trials):
        s_idx = high_idx[rng.integers(len(high_idx))]
        d_idx = low_idx[rng.integers(len(low_idx))]
        if s_idx == d_idx or a[s_idx] < 1e-15:
            continue
        frac = float(rng.uniform(0.02, 0.08))
        step_amt = frac * a[s_idx]
        a_trial = a.copy()
        a_trial[s_idx] -= step_amt
        a_trial[d_idx] += step_amt
        a_trial = np.maximum(a_trial, 0.0)
        ts = a_trial.sum()
        if ts < 1e-30:
            continue
        a_trial = a_trial / ts
        c_trial = _fft_convolve_self(a_trial)
        obj_trial = 2.0 * n * c_trial.max()
        if obj_trial < cur_obj:
            return a_trial
    return a


def _block_perturb(a, n, rng, scale=0.03):
    """Apply a random block perturbation and normalize."""
    width = max(1, int(rng.integers(1, max(2, n // 80))))
    start = int(rng.integers(0, n - width))
    a_trial = a.copy()
    a_trial[start:start + width] *= np.exp(rng.normal(0, scale, width))
    a_trial = np.maximum(a_trial, 0.0)
    ts = a_trial.sum()
    if ts < 1e-30:
        return a
    return a_trial / ts


def propose_candidate(seed=42, budget_s=1500):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses multi-start mirror descent, coordinate descent, and local search.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 12.0

    # Determine sequence length from parent
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.size
    except Exception:
        n = 2048
        parent = np.ones(n)

    print(f"n = {n}")

    # Best tracking
    a_best = parent.copy()
    a_best = np.maximum(a_best, 0.0)
    s_best = a_best.sum()
    if s_best < 0.01:
        a_best = np.ones(n) / n
    else:
        a_best = a_best / s_best
    c_best = _fft_convolve_self(a_best)
    best_obj = 2.0 * n * c_best.max()
    print(f"Initial objective: {best_obj:.10f}")

    # === Phase 1: Multi-start mirror descent (50% of time) ===
    phase1_end = t0 + budget_s * 0.50

    a_p = _init_parent_seq(n)
    a_rc = _init_raised_cosine(n)
    a_hnn = _init_hann(n)
    a_bi1 = _init_bimodal(n, rng)
    a_bi2 = _init_bimodal(n, rng)
    a_r1 = _init_random(n, rng)
    a_r2 = _init_random(n, rng)

    inits_to_try = [
        ('parent', a_p),
        ('raised_cos', a_rc),
        ('hann', a_hnn),
        ('bimodal1', a_bi1),
        ('bimodal2', a_bi2),
        ('random1', a_r1),
        ('random2', a_r2),
    ]

    time_per_init = budget_s * 0.50 / len(inits_to_try)

    for init_name, a in inits_to_try:
        if time.monotonic() >= phase1_end:
            break
        init_deadline = min(phase1_end, time.monotonic() + time_per_init)

        eta = 2.0
        tau = 0.01
        num_active = 50
        it = 0

        while time.monotonic() < init_deadline:
            c = _fft_convolve_self(a)
            obj = 2.0 * n * c.max()
            if obj < best_obj:
                best_obj = obj
                a_best = a.copy()
                if it < 20 or it % 500 == 0:
                    print(f"  [P1:{init_name}] it={it} obj={obj:.10f}")

            a = _mirror_step(a, c, num_active, tau, eta)

            eta *= 0.9997
            if eta < 0.05:
                eta = 0.05
            tau *= 0.99995
            if tau < 0.001:
                tau = 0.001
            if it % 100 == 0:
                num_active = max(20, min(100, num_active + (1 if obj > 1.6 else -1)))

            it += 1

    print(f"\nAfter Phase 1: best_obj = {best_obj:.10f}")

    # === Phase 2: Refined mirror descent from best (28% of time) ===
    a = a_best.copy()
    phase2_end = t0 + budget_s * 0.78

    eta = 0.3
    tau = 0.003
    num_active = 40
    it = 0
    stagnation = 0

    while time.monotonic() < phase2_end:
        c = _fft_convolve_self(a)
        obj = 2.0 * n * c.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            a_best = a.copy()
            stagnation = 0
            if it % 500 == 0:
                print(f"  [P2] it={it} obj={obj:.10f}")
        else:
            stagnation += 1

        a = _mirror_step(a, c, num_active, tau, eta)

        if stagnation > 200:
            pert_scale = rng.uniform(0.05, 0.25)
            a = a_best * np.exp(rng.normal(0, pert_scale, n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            eta = float(rng.uniform(0.2, 0.8))
            tau = float(rng.uniform(0.001, 0.008))
            num_active = int(rng.integers(20, 80))
            stagnation = 0

        eta *= 0.99998
        if eta < 0.02:
            eta = 0.02
        tau *= 0.99999
        if tau < 0.0005:
            tau = 0.0005

        it += 1

    print(f"\nAfter Phase 2: best_obj = {best_obj:.10f}")

    # === Phase 3: Coordinate descent + local search (remaining time) ===
    a = a_best.copy()
    it = 0

    while time.monotonic() < deadline:
        phase = it % 4

        if phase == 0 or phase == 1:
            # Coordinate descent
            c = _fft_convolve_self(a)
            obj = 2.0 * n * c.max()
            if obj < best_obj:
                best_obj = obj
                a_best = a.copy()
                if it % 2000 == 0:
                    print(f"  [P3] it={it} obj={obj:.10f}")
            a = _local_search_coord(a, c, n, rng, num_trials=3)
        elif phase == 2:
            # Block perturbation
            a_trial = _block_perturb(a, n, rng, scale=0.03)
            c_trial = _fft_convolve_self(a_trial)
            obj_trial = 2.0 * n * c_trial.max()
            if obj_trial < best_obj:
                best_obj = obj_trial
                a_best = a_trial.copy()
                a = a_trial
                if it % 2000 == 0:
                    print(f"  [P3-block] it={it} obj={obj_trial:.10f}")
        else:
            # Mirror step from current
            c = _fft_convolve_self(a)
            obj = 2.0 * n * c.max()
            if obj < best_obj:
                best_obj = obj
                a_best = a.copy()
            a = _mirror_step(a, c, 30, 0.002, 0.1)

        it += 1

    # Final verification
    final_result = [float(x) for x in a_best]
    final_score = float(evaluate_sequence(final_result))
    print(f"\nFinal verified score: {final_score:.10f}")
    print(f"Final sequence sum: {a_best.sum():.6f}")
    print(f"Total iterations P3: {it}")

    return final_result
