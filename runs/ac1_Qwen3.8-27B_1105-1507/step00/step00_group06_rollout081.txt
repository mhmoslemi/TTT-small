    y[0] = P[0]
    if 2 * n - 1 > 1:
        y[1:] = P[1:] - P[:-1]
    tol = 1e-8 * max(M, 1e-10)
    active = np.where(y > tol)[0]
    # Dual constraint tightness
    slack = np.zeros(n)
    slack[0] = a_lp[0] * P[n - 1] - 1.0
    for i in range(1, n):
        slack[i] = a_lp[i] * (P[i + n - 1] - P[i - 1]) - 1.0
    s_max = np.abs(slack).max()
    if s_max < 1e-15:
        tight = np.arange(n)
    else:
        tight = np.where(np.abs(slack) < 1e-4 * s_max)[0]
    if len(active) == 0 or len(tight) == 0:
        return None
    if len(active) > 3000 or len(tight) > 3000:
        return None
    # Build and solve linear system
    A_idx = active
    S_idx = tight
    C = np.zeros((len(A_idx), len(S_idx)))
    rhs = M * np.ones(len(A_idx))
    for ki in range(len(A_idx)):
        k = A_idx[ki]
        for si in range(len(S_idx)):
            i = S_idx[si]
            j = k - i
            if 0 <= j < n:
                C[ki, si] = a_lp[j]
    try:
        sol, _, _, _ = np.linalg.lstsq(C, rhs, rcond=None)
        sol = np.maximum(sol, 0.0)
        s = sol.sum()
        if s < 1e-15:
            return None
        b_full = np.zeros(n)
        b_full[S_idx] = sol / s
        return b_full
    except Exception:
        return None


def run_gradient_descent(a_init, n, time_budget, rng):
    """Run gradient descent from a_init for time_budget seconds. Returns (best_a, best_score)."""
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = conv_self(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    if not np.isfinite(best_score):
        best_score = 2.0

    t_end = time.monotonic() + time_budget
    lr = 0.3
    lr_min = 1e-4
    lr_decay = 0.99995
    tau = max(conv.max(), 1e-10) * 0.08
    tau_floor = max(conv.max(), 1e-10) * 1e-5
    tau_decay = 0.9998
    it = 0
    no_improve = 0

    while time.monotonic() < t_end:
        M = conv.max()
        if M < 1e-15:
            break
        tau = max(tau * tau_decay, tau_floor)
        # Smooth max weights
        x = conv / tau
        x -= x.max()
        w = np.exp(x)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(2 * n - 1)
            am = np.where(conv == M)[0]
            w[am] = 1.0 / len(am)
        else:
            w /= ws
        # Gradient
        g = grad_smooth(a, w)
        gn = np.linalg.norm(g)
        if gn < 1e-20:
            break
        g /= gn
        # Step
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-10:
            lr = max(lr * 0.5, lr_min)
            continue
        a_new = a_new / sn
        # Score
        conv_new = conv_self(a_new)
        score_new = 2.0 * n * conv_new.max()
        if score_new < best_score:
            best_a = a_new.copy()
            best_score = score_new
            no_improve = 0
            lr = min(lr * 1.02, 1.5)
            a = a_new
            conv = conv_new
        else:
            no_improve += 1
            lr = max(lr * 0.85, lr_min)
            # Keep old a and conv
        it += 1
        if no_improve > 20000:
            break
        if it % 10000 == 0:
            el = time.monotonic()
            print("    GD iter={:d} score={:.6f} lr={:.5f}".format(it, best_score, lr))

    return best_a, best_score


def run_lp_refinement(a_init, n, time_budget, rng):
    """Run LP-based refinement for time_budget seconds. Returns (best_a, best_score)."""
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * conv_self(a).max()

    t_end = time.monotonic() + time_budget
    stall = 0

    while time.monotonic() < t_end:
        b = lp_direction(a, n)
        if b is None:
            stall += 1
            if stall > 100:
                break
            continue
        # Line search
        improved = False
        for t_try in (0.7, 0.5, 0.3, 0.15, 0.05, 0.02):
            a_try = (1.0 - t_try) * a + t_try * b
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-10:
                continue
            a_try = a_try / st
            score_try = 2.0 * n * conv_self(a_try).max()
            if score_try < best_score:
                best_a = a_try.copy()
                best_score = score_try
                a = a_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Minimizes 2n*max(a*a)/(sum(a))^2 using gradient descent on smoothed
    max-convolution (FFT-accelerated) with multiple restarts, followed by
    LP-based direction refinement via dual longest path.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0

    parent = np.asarray(height_sequence_1, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size
    ps = parent.sum()
    if ps < 0.01:
        parent = np.ones(n)
        ps = n
    a_parent = parent / ps

    best_a = a_parent.copy()
    best_score = 2.0 * n * conv_self(a_parent).max()
    print("Initial: score={:.6f} n={}".format(best_score, n))

    # Phase 1: Gradient descent with multiple restarts
    time_lp_reserved = 400.0
    time_per_gd = 70.0

    # Diverse initializations
    inits = []
    inits.append(a_parent.copy())
    inits.append(np.ones(n) / n)
    # Cosine bump
    c = np.cos(np.linspace(0.0, np.pi, n)) * 0.5 + 0.5
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    # Two bumps
    x = np.arange(n, dtype=np.float64)
    b2 = np.exp(-((x - 0.2 * n) ** 2) / (0.15 * n) ** 2) + np.exp(-((x - 0.8 * n) ** 2) / (0.15 * n) ** 2)
    b2 = np.maximum(b2, 0.01)
    inits.append(b2 / b2.sum())
    # Sine wave
    sw = 1.0 + 0.8 * np.sin(2.0 * np.pi * x / n)
    sw = np.maximum(sw, 0.01)
    inits.append(sw / sw.sum())
    # Edge-heavy
    eh = (x / n + (1.0 - x / n)) ** 4 + 0.01
    inits.append(eh / eh.sum())

    for idx, init in enumerate(inits):
        remaining = deadline - time.monotonic() - time_lp_reserved
        if remaining < 5.0:
            break
        tb = min(time_per_gd, remaining)
        ra, rs = run_gradient_descent(init, n, tb, rng)
        if rs < best_score:
            best_a = ra.copy()
            best_score = rs
            print("Init {} improved: {:.6f}".format(idx, best_score))

    # Random restarts
    while time.monotonic() < deadline - time_lp_reserved:
        alpha = float(rng.uniform(0.3, 1.5))
        a_rand = rng.dirichlet(np.ones(n) * alpha)
        remaining = deadline - time.monotonic() - time_lp_reserved
        if remaining < 5.0:
            break
        tb = min(time_per_gd * 0.8, remaining)
        ra, rs = run_gradient_descent(a_rand, n, tb, rng)
        if rs < best_score:
            best_a = ra.copy()
            best_score = rs
            print("Random restart improved: {:.6f}".format(best_score))

    # Phase 2: LP refinement
    remaining = deadline - time.monotonic()
    if remaining > 15.0:
        print("LP refinement: budget={:.1f}s score={:.6f}".format(remaining, best_score))
        la, ls = run_lp_refinement(best_a, n, remaining, rng)
        if ls < best_score:
            best_a = la.copy()
            best_score = ls
            print("LP improved: {:.6f}".format(best_score))

    # Phase 3: Short GD polish from LP result
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        ga, gs = run_gradient_descent(best_a, n, remaining, rng)
        if gs < best_score:
            best_a = ga.copy()
            best_score = gs
            print("Polish improved: {:.6f}".format(best_score))

    # Final validation
    result = [max(0.0, float(x)) for x in best_a]
    fs = float(evaluate_sequence(result))
    print("Final eval: {:.6f}".format(fs))
    if (not np.isfinite(fs)) or fs > 2.0:
        result = [max(0.0, float(x)) for x in a_parent]
        fs = float(evaluate_sequence(result))
        print("Fallback eval: {:.6f}".format(fs))

    return result
