"""
LP-based descent to minimize 2n*max(conv(a,a))/(sum(a))^2.
Each iteration solves an LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
Blends current sequence with LP solution to reduce the score.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(a):
    """Fast score computation mimicking evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_A_sparse(a, n):
    """Build sparse (2n-1) x n constraint matrix for LP.
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    total_nnz = 0
    for j in range(n):
        total_nnz += min(n, 2 * n - 1 - j)
    rows = np.zeros(total_nnz, dtype=np.int32)
    cols = np.zeros(total_nnz, dtype=np.int32)
    vals = np.zeros(total_nnz, dtype=np.float64)
    idx = 0
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        rows[idx:idx + length] = np.arange(j, j + length)
        cols[idx:idx + length] = j
        vals[idx:idx + length] = a[:length]
        idx += length
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, n, ub=100.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M elementwise, 0 <= b <= ub."""
    conv_aa = np.convolve(a, a)
    M = float(np.max(conv_aa))
    A = build_A_sparse(a, n)
    c_obj = -np.ones(n)
    b_ub_vec = np.full(2 * n - 1, M)
    bounds = [(0.0, ub)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub_vec, bounds=bounds, method='highs')
    if res.success:
        return res.x, M
    return None, M


def lp_descent(a0, n, deadline, verbose=True):
    """Run LP-based descent from initial sequence a0."""
    best = np.asarray(a0, dtype=np.float64).copy()
    best = np.clip(best, 0.0, 1000.0)
    best_score = compute_score(best)
    if verbose:
        print(f"  LP descent start: score={best_score:.6f}", flush=True)

    it = 0
    stall = 0
    for it in range(500):
        if time.monotonic() > deadline:
            break

        # Normalize to sum = n
        S = np.sum(best)
        if S < 1e-12:
            break
        a_norm = best * (n / S)

        t0 = time.monotonic()
        b, M = solve_lp(a_norm, n, ub=max(10.0, 5.0 * np.max(a_norm)))
        lp_time = time.monotonic() - t0

        if b is None:
            if verbose:
                print(f"  Iter {it}: LP failed", flush=True)
            break

        sum_b = float(np.sum(b))
        improvement_ratio = sum_b / n  # should be > 1 for improvement

        if improvement_ratio <= 1.0 + 1e-8:
            if verbose:
                print(f"  Iter {it}: No LP improvement (ratio={improvement_ratio:.6f})", flush=True)
            stall += 1
            if stall >= 3:
                break
            continue

        # Try blending with multiple t values
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            a_new = (1.0 - t) * a_norm + t * b
            a_new = np.clip(a_new, 0.0, 1000.0)
            sc = compute_score(a_new)
            if sc < best_score - 1e-10:
                best = a_new
                best_score = sc
                improved = True
                break

        if not improved:

            sc = compute_score(b)
            if sc < best_score - 1e-10:
                best = np.clip(b, 0.0, 1000.0)
                best_score = sc
                improved = True

        if improved:
            stall = 0
            if verbose:
                print(f"  Iter {it}: score={best_score:.6f} (t used, sum_b/n={improvement_ratio:.4f}, lp={lp_time:.2f}s)", flush=True)
        else:
            stall += 1
            if verbose:
                print(f"  Iter {it}: no blend improvement (lp={lp_time:.2f}s)", flush=True)
            if stall >= 5:
                break

    if verbose:
        print(f"  LP descent done: {it+1} iters, final score={best_score:.6f}", flush=True)
    return best, best_score


def block_perturb(a, rng, width_range=(3, 40)):
    """Apply a random block multiplicative perturbation."""
    c = a.copy()
    n = len(c)
    w = int(rng.integers(width_range[0], min(width_range[1], n) + 1))
    s = int(rng.integers(0, n - w + 1))
    c[s:s + w] *= float(np.exp(rng.normal(0.0, 0.15)))
    if rng.random() < 0.3:
        c = 0.5 * (c + c[::-1])
    return np.clip(c, 0.0, 1000.0)


def local_search_polish(a0, n, deadline, verbose=True):
    """Simple local search with block perturbations to polish."""
    rng = np.random.default_rng(12345)
    best = np.asarray(a0, dtype=np.float64).copy()
    best_score = compute_score(best)
    it = 0
    while time.monotonic() < deadline:
        cand = block_perturb(best, rng)
        sc = compute_score(cand)
        if sc < best_score - 1e-12:
            best = cand
            best_score = sc
            if verbose:
                print(f"  Polish it {it}: score={best_score:.6f}", flush=True)
        it += 1
    return best, best_score
