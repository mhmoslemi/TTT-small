import numpy as np

def fast_convolve(a):
    """Compute full convolution a*a using FFT (faster than np.convolve for large n)."""
    n = len(a)
    needed = 2 * n - 1
    fft_size = 1
    while fft_size < needed:
        fft_size <<= 1
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:needed]


def compute_score_normalized(a):
    """Compute score assuming sum(a)=1. Returns (score, c)."""
    n = len(a)
    c = fast_convolve(a)
    max_c = np.max(c)
    score = 2.0 * n * max_c
    return score, c


def gradient_for_lag(a, k):
    """Compute gradient of c_k w.r.t. a. Returns array of shape (n,)."""
    n = len(a)
    g = np.zeros(n)
    vs = max(0, k - n + 1)
    ve = min(n, k + 1)
    i_idx = np.arange(vs, ve)
    j_idx = k - i_idx
    g[i_idx] = 2.0 * a[j_idx]
    return g


def gd_step(a, g, lr):
    """One projected gradient step: subtract lr*g, clip to 0, renormalize."""
    a_new = a - lr * g
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-15:
        return a
    return a_new / s

                      top_k=5, restart_every=150, noise_level=0.005,
                      rng=None):

    Projected gradient descent to minimize max(convolve(a,a)) with sum(a)=1.
    Uses multi-lag gradient and periodic noise injection.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return a0, np.inf
    a = a / s

    n = len(a)
    best_a = a.copy()
    _, c0 = compute_score_normalized(a)

    lr = lr0
    start = time.monotonic()
    step = 0
    stagnation = 0

    while time.monotonic() - start < max_time:
        score, c = compute_score_normalized(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Find top-k lags
        if top_k >= len(c):
            top_lags = np.arange(len(c))
        else:
            top_lags = np.argpartition(c, -top_k)[-top_k:]

        # Compute averaged gradient over top-k lags
        g = np.zeros(n)
        for k in top_lags:
            g += gradient_for_lag(a, k)
        g /= len(top_lags)

        # Project: subtract mean to preserve sum constraint
        g -= np.mean(g)

        # Take step
        a = gd_step(a, g, lr)
        lr *= lr_decay
        step += 1

        # Periodic restart with noise
        if step % restart_every == 0:
            noise = noise_level * rng.standard_normal(n)
            a = a + noise
            a = np.maximum(a, 0.0)
            s = np.sum(a)
            a = a / s
            lr = lr0

            a = a + noise
            a = np.maximum(a, 0.0)
            a = a / np.sum(a)
            lr = lr0
