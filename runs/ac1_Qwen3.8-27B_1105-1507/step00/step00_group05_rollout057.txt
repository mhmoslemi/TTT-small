def solve_lp_direction(f, M):
    """Solve LP to maximize sum(g) subject to conv(f,g)[k] <= M and g >= 0."""
    n = len(f)
    c = -np.ones(n)  # minimize -sum(g)
    
    # Build sparse constraint matrix
    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    # For row k, non-zero columns are i where 0 <= k-i < n, i.e., k-n+1 <= i <= k
    # And 0 <= i < n
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:

    return None

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, lil_matrix
import time

def compute_convolution(f):
    """Compute full convolution using FFT for efficiency."""
    n = len(f)
    # Use FFT for O(n log n) convolution
    size = 2 * n
    F = np.fft.rfft(f, size=size)
    conv = np.fft.irfft(F * F, size=size)[:2*n-1]
    return conv

def evaluate(f):
    """Compute J = 2n * max(conv) / (sum)^2."""
    n = len(f)
    conv = compute_convolution(f)
    max_b = max(np.abs(conv))  # Actually, since f >= 0, conv >= 0
    sum_a = np.sum(f)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def solve_lp(f, M, verbose=False):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    
    # Build sparse matrix efficiently
    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    # Column i: non-zero in rows i to min(i+n-1, 2n-2), values f[0], f[1], ...
    
    rows_list = []
    cols_list = []
    vals_list = []
    
    for i in range(n):
        # Column i has non-zeros in rows k where 0 <= k-i < n
        # i.e., i <= k <= i+n-1, clamped to [0, 2n-2]
        k_start = i
        k_end = min(i + n - 1, 2*n - 2)
        for k in range(k_start, k_end + 1):
            rows_list.append(k)
            cols_list.append(i)
            vals_list.append(f[k - i])
    
    A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        if verbose:
            print(f"  LP solved: sum(g) = {-res.fun}, iterations = {res.iter}")
        return res.x
    else:
        if verbose:
            print(f"  LP failed: {res.message}")
        return None

def lp_iteration(f, t=0.1, verbose=False):
    """One LP-based iteration."""
    n = len(f)
    f = np.maximum(f, 0)  # Ensure non-negative
    sum_f = np.sum(f)
    if sum_f < 0.01:
        return f
    
    # Normalize
    f = f / sum_f * np.sqrt(2 * n)
    sum_f = np.sqrt(2 * n)
    
    # Compute current peak
    conv = compute_convolution(f)
    M = np.max(conv)
    
    # Solve LP
    g = solve_lp(f, M, verbose=verbose)
    if g is None:
        return f
    
    # Rescale g to match sum
    sum_g = np.sum(g)
    if sum_g < 0.01:
        return f
    g = g * sum_f / sum_g
    
    # Update
    f_new = (1 - t) * f + t * g
    f_new = np.maximum(f_new, 0)
    
    return f_new

"""
LP-based optimization for minimizing peak self-convolution.
Uses the linear programming approach to find directions that reduce
the maximum of the autocorrelation, combined with local search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def compute_conv_fft(f):
    """Compute full convolution of f with itself using FFT."""
    n = len(f)
    size = 2 * n
    F = np.fft.rfft(f, n=size)
    conv = np.fft.irfft(F * F, n=size)[:2 * n - 1]
    return conv


def build_constraint_matrix(f):
    """Build sparse constraint matrix for the LP.
    A[k, i] = f[k-i] if 0 <= k-i < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for i in range(n):
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        rows.extend(range(k_start, k_end + 1))
        cols.extend([i] * length)
        vals.extend(f[:length])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A = build_constraint_matrix(f)
    b = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None

    # Rescale and update
    g = g * sum_f / sum_g
    f_new = (1.0 - t) * f + t * g

"""
LP-based optimization to minimize the peak self-convolution of a
non-negative sequence. The evaluation function is
J = 2*n*max(conv(f,f)) / (sum(f))^2.
Uses linear programming to find descent directions that provably
reduce the peak, combined with local search for refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def compute_conv_fft(f):
    """Compute full self-convolution using FFT."""
    n = len(f)
    size = 2 * n
    F = np.fft.rfft(f, n=size)
    return np.fft.irfft(F * F, n=size)[:2 * n - 1]


def build_A(f):
    """Build sparse LP constraint matrix. A[k,i] = f[k-i] for valid (k,i)."""
    n = len(f)
    total = n * n
    rows = np.zeros(total, dtype=np.int64)
    cols = np.zeros(total, dtype=np.int64)
    vals = np.zeros(total, dtype=np.float64)
    for i in range(n):
        sl = slice(i * n, (i + 1) * n)
        rows[sl] = np.arange(i, i + n)
        cols[sl] = i
        vals[sl] = f
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M, time_limit=20):
    """Solve: max sum(g) s.t. A@f * g <= M, g >= 0."""
    n = len(f)
    A = build_A(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_step(f, t=0.1, time_limit=20):
    """Perform one LP-based descent step."""
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s < 1e-6:
        return f.copy()
    fn = f / s
    conv = compute_conv_fft(fn)
    M = float(np.max(conv))
    g = solve_lp(fn, M, time_limit)
    if g is None:
        return f.copy()
    sg = float(np.sum(g))
    if sg < 1e-10:
        return f.copy()
    g = g * s / sg
    return np.maximum((1.0 - t) * f + t * g, 0.0)
