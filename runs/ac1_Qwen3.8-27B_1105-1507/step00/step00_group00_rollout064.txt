import numpy as np
import time


def _fft_next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_max_b(a, N):
    """Return max of linear self-convolution of a (length n) using FFT of size N."""
    a = np.asarray(a, dtype=np.float64)
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)
    return float(b.max())


def _soft_step(a, N, tau, eta):
    """One multiplicative soft-max gradient step minimizing max(conv(a,a)). a has sum 1."""
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)
    n = a.size
    max_b = float(b.max())
    if max_b <= 1e-300:
        return a.copy(), max_b
    # softmax over b
    eb = np.exp(tau * (b - max_b))
    s = eb.sum()
    if s <= 1e-300:
        p = np.full_like(b, 1.0 / b.size)
    else:
        p = eb / s
    # grad_j = 2 * sum_i a_i p_{i+j}  (linear cross-correlation, j=0..n-1)
    P = np.fft.rfft(p, N)
    c = np.fft.irfft(np.conj(A) * P, N)[:n]
    grad = 2.0 * c
    step = eta * grad / max_b
    step = np.clip(step, -40.0, 40.0)
    a_new = a * np.exp(-step)
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return a.copy(), max_b
    a_new = a_new / ss
    return a_new, max_b


def _build_conv_matrix(f, n):
    """Sparse (2n-1) x n matrix F with (F g)_k = (f*g)_k."""
    from scipy.sparse import coo_matrix
    js = np.arange(n)
    rows = np.tile(np.arange(n), n) + np.repeat(js, n)
    cols = np.repeat(js, n)
    data = np.tile(f, n)
    return coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def _lp_direction(f, N, budget_s):
    """Solve max sum(g) s.t. (f*g)<=M, g>=0. Return normalized g (sum 1) or None."""
    from scipy.optimize import linprog
    n = f.size
    A_f = np.fft.rfft(f, N)
    b_f = np.fft.irfft(A_f * A_f, N)
    M = float(b_f.max())
    if M <= 1e-15:
        return None
    F = _build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=F, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method="highs", options={"time_limit": max(1.0, budget_s)})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0.0)
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs
