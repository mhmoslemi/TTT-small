```python
import time
import scipy.signal

def simplex_project(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    if n == 0:
        return x
    # Sort in descending order
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(x - theta, 0)

def compute_convolution(a):
    """Compute the full convolution of a with itself."""
    return np.convolve(a, a)

    # p_k = exp(c_k/T) / sum(exp(c_j/T))

    c_scaled -= c_scaled.max()  # for numerical stability
    p = np.exp(c_scaled)
```

```python
import numpy as np
import time
import scipy.signal

def simplex_project(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)

def run_gradient_descent(a_init, n_steps, T_init, T_min, lr_init, deadline):
    a = a_init.copy()
    a = a / a.sum()
    n = len(a)
    
    # Precompute FFT sizes
    L = 1
    while L < 3 * n:
        L *= 2
    
    best_score = np.inf
    best_a = a.copy()
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        T = T_init * (T_min / T_init) ** (step / max(n_steps, 1))
        
        # Compute convolution
        c = np.convolve(a, a)
        
        # Compute softmax weights
        c_scaled = c / T
        c_scaled -= c_scaled.max()
        p = np.exp(c_scaled)
        p_sum = p.sum()
        if p_sum == 0:
            p = np.ones(len(c)) / len(c)
        else:
            p /= p_sum
        
        # Compute gradient using FFT
        A = np.zeros(L)
        A[:n] = a
        P = np.zeros(L)
        P[:len(p)] = p
        
        FA = np.fft.rfft(A)
        FP = np.fft.rfft(P)
        
        g_full = np.fft.irfft(FP * np.conj(FA), n=L)
        g = 2.0 * g_full[:n]
        
        # Update
        a_new = a - lr_init * g
        a_new = simplex_project(a_new)
        
        # Evaluate
        score = 2.0 * n * np.max(np.convolve(a_new, a_new)) / (a_new.sum() ** 2)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score
```

import numpy as np
import time

def simplex_project(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    if n == 0:
        return x.copy()
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)

def softmax_gradient(a, c, T, L):
    """Compute the gradient of T*log(sum(exp(c/T))) w.r.t. a using FFT."""
    n = len(a)
    # Softmax weights
    c_scaled = c / T
    c_scaled -= c_scaled.max()
    p = np.exp(c_scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(len(c)) / len(c)
    else:
        p /= p_sum
    
    # FFT-based convolution to compute gradient
    A = np.zeros(L)
    A[:n] = a
    P = np.zeros(L)
    P[:len(p)] = p
    
    FA = np.fft.rfft(A)
    FP = np.fft.rfft(P)
    
    # cross-correlation: ifft(FFT(P) * conj(FFT(A)))
    corr = np.fft.irfft(FP * np.conj(FA), n=L)
    g = 2.0 * corr[:n]
    return g

def gradient_descent_phase(a_init, deadline, n_steps=500):
    """Run soft-max gradient descent."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    total = a.sum()
    if total <= 0:
        a = np.ones(n) / n
    else:
        a /= total
    
    # FFT size
    L = 1
    while L < 3 * n:
        L *= 2
    
    best_a = a.copy()
    best_score = np.inf
    
    T_init = 1.0 / n  # start with T such that c/T has O(1) range
    # Actually, c values are O(1/n) for normalized a, so c/T = O(1) when T = 1/n
    # Let's use T_init = max(1/n, 0.001) and T_min = 0.01/n
    
    c = np.convolve(a, a)
    c_range = c.max() - c.min()
    if c_range < 1e-15:
        c_range = 1.0 / n
    T_init = c_range * 0.5
    T_min = c_range * 0.01
    
    lr = 1.0
    momentum = np.zeros(n)
    beta = 0.9
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        progress = step / max(n_steps, 1)
        T = T_init * (T_min / T_init) ** progress
        lr_current = lr * (0.5 ** (step / max(n_steps // 3, 1)))
        
        c = np.convolve(a, a)
        g = softmax_gradient(a, c, T, L)
        
        momentum = beta * momentum + (1 - beta) * g
        a_new = a - lr_current * momentum
        a_new = simplex_project(a_new)
        
        # Evaluate
        c_new = np.convolve(a_new, a_new)
        score = 2.0 * n * c_new.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

def coordinate_descent_phase(a_init, deadline, max_passes=500):
    """Coordinate descent with incremental convolution updates."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    total = a.sum()
    if total <= 0:
        a = np.ones(n) / n
    else:
        a /= total
    
    c = np.convolve(a, a).copy()
    max_c = c.max()
    best_max_c = max_c
    best_a = a.copy()
    
    multipliers = np.array([0.1, 0.3, 0.5, 0.7, 1.3, 1.5, 2.0])
    
    passes = 0
    while time.monotonic() < deadline and passes < max_passes:
        passes += 1
        any_change = False
        
        # Process coordinates in a random or focused order
        # Focus on coordinates that contribute to the max
        idx = list(range(n))
        np.random.shuffle(idx)
        
        for i in idx:
            old = a[i]
            if old <= 1e-15:
                continue
            
            best_mult = 1.0
            best_local_max = max_c
            
            for mult in multipliers:
                delta = old * (mult - 1.0)
                if abs(delta) < 1e-15:
                    continue
                
                # Hypothetical new c in affected region
                # c[i:i+n] += 2*delta*a (using old a)
                # c[2*i] += delta^2
                affected = c[i:i+n] + 2.0 * delta * a
                if 2*i >= i and 2*i < i+n:
                    affected[2*i - i] += delta * delta
                
                local_max = max(max(c[:i].max() if i > 0 else 0, 
                                     c[i+n:].max() if i+n < len(c) else 0), 
                                 affected.max())
                
                if local_max < best_local_max:
                    best_local_max = local_max
                    best_mult = mult
            
            if best_mult != 1.0:
                delta = old * (best_mult - 1.0)
                c[i:i+n] += 2.0 * delta * a
                a[i] = old + delta
                if 2*i < len(c):
                    c[2*i] += delta * delta
                
                # Update max (conservative: take max of old max and new affected)
                max_c = max(max_c, c[i:i+n].max())
                # Actually, max_c might have decreased if the old max was in the affected region
                # and we decreased it. Let's just recompute periodically.
                
                any_change = True
        
        # Recompute exact max every 10 passes
        if passes % 10 == 0:
            c = np.convolve(a, a)
            max_c = c.max()
        
        if max_c < best_max_c:
            best_max_c = max_c
            best_a = a.copy()
        
        if not any_change:
            break
    
    return best_a, best_max_c
