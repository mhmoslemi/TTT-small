import cvxpy as cp

def solve_lp(a, M, solver=cp.SCS):
    n = len(a)
    b = cp.Variable(n)
    
    # Build constraint matrix
    # A[k, j] = a[k-j] if 0 <= k-j < n
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.where(valid, a[diff.clip(0, n-1)], 0.0)
    
    constraints = [A @ b <= M, b >= 0]
    objective = cp.Maximize(cp.sum(b))
    prob = cvxpy.Problem(objective, constraints)
    prob.solve(solver=solver)
    
    if b.value is None:
        return None
    return np.maximum(b.value, 0)

        else:
            no_improve += 1

        a = a_new
        lr *= 0.99999
        it += 1

        if verbose and it % 10000 == 0:
            print(f"  Subgrad iter {it}: score={best_score:.6f}, lr={lr:.6f}")

        if no_improve > patience:
            break
        if lr < lr_init * 1e-4:
            break

    return best_a, best_score


def coordinate_descent(a, n, deadline, n_sweeps_max=500, verbose=False):
    """Coordinate descent: optimize each a[i] exactly given others fixed."""
    a = a.copy()
    best_a = a.copy()
    best_score = compute_score_fast(a)
    no_improve = 0
    patience = 20000
    it = 0

    # Precompute initial convolution
    c = fast_convolve(a, a)

    while time.monotonic() < deadline and no_improve < patience:
        # Random order for diversity
        order = np.random.permutation(n)
        sweep_improved = False

        for i in order:
            # Current value
            old_val = a[i]
            if old_val < 1e-15:
                continue

            # Find max a[i] such that all c[k] stay <= current peak (or below)
            # c[k] = old_conv[k] + delta where delta involves a[i]
            # For k where k-i is valid (0 <= k-i < n):
            #   c[k] has term a[i]*a[k-i] (from j=i) and a[k-i]*a[i] (from j=k-i)
            #   If i != k-i: contribution is 2*a[i]*a[k-i]
            #   If i == k-i (k=2i): contribution is a[i]^2
            # Change: replace a[i] with new_val
            #   delta_c[k] = (new_val - old_val) * a[k-i] [for j=i term]
            #              + (new_val - old_val) * a[k-i] [for j=k-i term, if k-i != i]
            #   For k = 2i: delta = new_val^2 - old_val^2

            # The peak we want to not exceed is the current max
            target = c.max()

            # Constraints from linear terms (k != 2i):
            # c[k] + (new-old) * 2*a[k-i] <= target  (for k where 0<=k-i<n and k!=2i)
            # => new <= old + (target - c[k]) / (2*a[k-i])
            # Constraints from k = 2i: c[2i] + new^2 - old^2 <= target
            # => new <= sqrt(target - c[2i] + old^2)

            max_allowed = np.inf

            if 2 * i < 2 * n - 1:
                # Quadratic constraint at k = 2i
                rhs = target - c[2 * i] + old_val ** 2
                if rhs >= 0:
                    max_allowed = min(max_allowed, np.sqrt(rhs))

            # Linear constraints: for all k where 0 <= k-i < n and k != 2i
            # k ranges from i to i+n-1 (clipped to [0, 2n-2])
            lo_k = max(0, i)  # k-i >= 0 => k >= i... wait
            # k-i in [0, n-1] => k in [i, i+n-1], clipped to [0, 2n-2]
            lo_k = max(0, i - (n - 1))  # actually k-i >= 0 means k >= i, but also k-i < n means k < i+n
            # Let me redo: valid k are those with 0 <= k-i < n, i.e., i <= k <= i+n-1
            # Also k must be in [0, 2n-2]
            lo_k = max(0, i)
            hi_k = min(2 * n - 2, i + n - 1) + 1
            if lo_k < hi_k:
                k_range = np.arange(lo_k, hi_k)
                # a[k-i] for k in range
                a_vals = a[k_range - i]
                # Exclude k = 2i from linear constraints (handled above)
                mask = k_range != 2 * i
                if mask.any():
                    ks = k_range[mask]
                    av = a_vals[mask]
                    cv = c[ks]
                    pos = av > 1e-15
                    if pos.any():
                        bounds = old_val + (target - cv[pos]) / (2.0 * av[pos])
                        max_allowed = min(max_allowed, bounds.max() if len(bounds) > 0 else np.inf)
                        # Actually we need ALL constraints satisfied, so min of bounds
                        max_allowed = min(max_allowed, np.minimum.reduce(bounds) if len(bounds) > 0 else np.inf)

            # Also a[i] >= 0
            new_val = min(max(0.0, max_allowed), old_val * 1.5)  # don't increase too much
            # Actually, we want to potentially increase a[i] to help with the sum
            # But the constraint limits how much we can increase
            # Let's also try keeping it the same or reducing it
            new_val = max(0.0, min(new_val, old_val * 2.0))

            if abs(new_val - old_val) < 1e-15:
                continue

            # Apply update to c
            delta = new_val - old_val
            # Update c[k] for affected k
            lo_k = max(0, i)
            hi_k = min(2 * n - 2, i + n - 1)
            for k in range(lo_k, hi_k + 1):
                j = k - i
                if 0 <= j < n:
                    if k == 2 * i:
                        c[k] += new_val ** 2 - old_val ** 2
                    else:
                        c[k] += 2.0 * delta * a[j]
            a[i] = new_val
            sweep_improved = True

            # Update best
            new_peak = c.max()
            score_now = 2 * n * new_peak / (a.sum() ** 2)
            if score_now < best_score:
                best_score = score_now
                best_a = a.copy()

        no_improve = 0 if sweep_improved else no_improve + 1
        it += 1

        if verbose and it % 50 == 0:
            print(f"  CoordDescent sweep {it}: score={best_score:.6f}")

    return best_a, best_score


def lp_based_refinement(a, n, deadline, verbose=False):
    """LP-based peak reduction using projected subgradient on the LP dual."""
    a = a.copy()
    best_a = a.copy()
    best_score = compute_score_fast(a)
    it = 0

    while time.monotonic() < deadline:
        c = fast_convolve(a, a)
        M = c.max()
        if M < 1e-15:
            break

        # Solve LP approximately: find b maximizing sum(b) s.t. conv(a,b) <= M, b >= 0
        # Use a simplified approach: b[j] = max(0, min over k of (M - conv(a,b_partial)[k]) / a[k-j])
        # But this is greedy and might be slow. Instead, use a simple heuristic:
        # b[j] proportional to 1/sqrt(sum_k a[k-j]^2) (spreads weight where a is "weak")

        # Compute for each j: the "bottleneck" = min over k of M / a[k-j] (where a[k-j] > 0)
        # This is expensive. Use a simpler heuristic:
        # b[j] = 1 / (1 + norm of a's "shadow" at position j)

        # shadow[j] = sum_k (a[k-j])^2 over valid k = sum of a^2 (it's just sum(a^2) for all j)
        # That's not useful. Let me use:
        # b[j] = 1 / (1 + a[j])  -- simple inverse weighting

        b = 1.0 / (1.0 + a)
        b = b / b.sum()

        # Blend
        t = 0.1
        a_new = (1 - t) * a + t * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new /= s

        score = compute_score_fast(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            t = min(0.3, t * 1.05)
        else:
            t = max(0.01, t * 0.8)

        it += 1
        if verbose and it % 100 == 0:
            print(f"  LP-refine iter {it}: score={best_score:.6f}")

    return best_a, best_score


def full_lp_solve(a, M, n, max_time=30.0):
    """Attempt to solve the LP using cvxpy. Returns b or None on timeout."""
    start = time.monotonic()

    # For memory efficiency, work with a truncated version if n is too large
    work_n = min(n, 3000)
    if work_n < n:
        # Use center portion
        offset = (n - work_n) // 2
        a_work = a[offset:offset + work_n]
        scale_back = True
    else:
        a_work = a
        scale_back = False

    b_var = cp.Variable(work_n)

    # Build constraint: conv(a_work, b)[k] <= M for all k
    # Use the fact that this is a linear constraint in b
    # A[k, j] = a_work[k-j] if 0 <= k-j < work_n
    # To save memory, use a banded representation

    # Actually for work_n = 3000, the matrix is (5999, 3000) ~ 18M entries ~ 144MB
    # Let's try a smaller version
    work_n = min(n, 1500)
    if work_n < n:
        offset = (n - work_n) // 2
        a_work = a[offset:offset + work_n]
    else:
        a_work = a

    conv_len = 2 * work_n - 1

    # Build sparse constraint matrix
    # Row k has non-zeros at columns j where 0 <= k-j < work_n
    # i.e., j in [max(0, k-work_n+1), min(work_n-1, k)]
    rows = []
    cols = []
    vals = []
    for k in range(conv_len):
        lo = max(0, k - work_n + 1)
        hi = min(work_n, k + 1)
        if lo < hi:
            rows.extend([k] * (hi - lo))
            cols.extend(range(lo, hi))
            # a_work[k-j] for j in [lo, hi)
            idx = k - np.arange(lo, hi)
            vals.extend(a_work[idx].tolist())

    from scipy.sparse import csr_matrix
    A_sparse = csr_matrix((np.array(vals), (np.array(rows), np.array(cols))), shape=(conv_len, work_n))

    constraints = [A_sparse @ b_var <= M, b_var >= 0]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-5)
    except:
        return None

    if time.monotonic() - start > max_time:
        return None

    if b_var.value is None:
        return None

    b_val = np.maximum(b_var.value, 0.0)

    if not scale_back and work_n == n:
        return b_val
    else:
        # Scale back: place b in the center of a full-length vector
        b_full = np.zeros(n)
        b_full[offset:offset + work_n] = b_val
        return b_full


def multi_peak_subgradient(a, n, deadline, verbose=False):
    """Enhanced subgradient with multiple peaks and adaptive step size."""
    a = a.copy()
    best_a = a.copy()
    best_score = compute_score_fast(a)
    lr = 1.0 / n
    best_lr = lr
    no_improve = 0
    patience = 5000
    it = 0
    n_peaks = min(10, n // 100 + 1)

    while time.monotonic() < deadline:
        c = fast_convolve(a, a)
        peak_val = c.max()

        # Get top peaks
        top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        top_vals = c[top_idx]

        # Weight peaks by their value (emphasize the highest)
        weights = top_vals ** 2
        weights /= weights.sum()

        # Weighted subgradient
        g = np.zeros(n)
        for k, w in zip(top_idx, weights):
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            seg = a[k - hi + 1:k - lo + 1][::-1]
            g[lo:hi] += w * seg

        # Barzilai-Borwein style step: use recent change
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new /= s

        score = compute_score_fast(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            best_lr = lr
            no_improve = 0
            # Increase lr slightly
            lr = min(lr * 1.01, 1.0)
            a = a_new
        else:
            no_improve += 1
            lr *= 0.95
            if lr < best_lr * 0.1:
                lr = best_lr
                # Perturb to escape
                a = a + np.random.normal(0, 0.01 / n, n)
                a = np.maximum(a, 0)
                a /= a.sum()

        it += 1
        if verbose and it % 10000 == 0:
            print(f"  MultiPeak iter {it}: score={best_score:.6f}")

        if no_improve > patience:
            # Try a perturbation
            a = best_a + np.random.normal(0, 0.05 / n, n)
            a = np.maximum(a, 0)
            a /= a.sum()
            no_improve = 0
            lr = best_lr

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-strategy search for optimal step function sequence.
    
    Combines asymmetric initialization, subgradient descent, coordinate descent,
    and LP-based refinement to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    # Get parent sequence
    parent = np.array(height_sequence_1, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    n_parent = len(parent)

    # Choose working length
    # Keep the original length for the final answer
    n = n_parent

    # Generate diverse initializations
    initializations = []

    # 1. Normalized parent
    p = parent.copy()
    if p.sum() > 0:
        p /= p.sum()
    initializations.append(p)

    # 2. U-shaped asymmetric
    initializations.append(init_asymmetric_u(n, seed=101, curvature=4.0))

    # 3. Stronger U-shape
    initializations.append(init_asymmetric_u(n, seed=202, curvature=10.0))

    # 4. Inverse tapered
    initializations.append(init_asymmetric_inverse(n, seed=303, sharpness=1.5))

    # 5. Block asymmetric
    initializations.append(init_block_asymmetric(n, seed=404, n_blocks=6))

    # 6. Parent multiplied by U-shape
    u = np.arange(n, dtype=np.float64)
    t = 2.0 * u / (n - 1) - 1.0
    u_profile = 1.0 + 5.0 * t ** 2
    mixed = p * u_profile
    mixed = np.maximum(mixed, 0)
    if mixed.sum() > 0:
        mixed /= mixed.sum()
    initializations.append(mixed)

    # 7. Random with edge boost
    r = rng.random(n)
    edge_boost = 1.0 + 3.0 * t ** 2
    r2 = r * edge_boost
    r2 = np.maximum(r2, 0)
    if r2.sum() > 0:
        r2 /= r2.sum()
    initializations.append(r2)

    # 8. Parent with random perturbation
    p2 = p + rng.normal(0, 0.1, n) * p.sum() / n
    p2 = np.maximum(p2, 0)
    if p2.sum() > 0:
        p2 /= p2.sum()
    initializations.append(p2)

    best_a = None
    best_score = np.inf

    print(f"Starting search with n={n}, {len(initializations)} initializations")
    print(f"Budget: {budget_s}s, deadline set")

    # Phase 1: Quick evaluation of all initializations
    for idx, init in enumerate(initializations):
        score = compute_score_fast(init)
        if score < best_score:
            best_score = score
            best_a = init.copy()
        print(f"  Init {idx}: score = {score:.6f}")

    print(f"Best initial score: {best_score:.6f}")

    # Phase 2: Subgradient descent from best initializations (top 3)
    init_scores = [(compute_score_fast(init), idx) for idx, init in enumerate(initializations)]
    init_scores.sort()
    top_inits = [initializations[idx] for _, idx in init_scores[:3]]

    time_per_init = (deadline - time.monotonic()) / 3.0
    for i, init in enumerate(top_inits):
        phase_deadline = time.monotonic() + time_per_init
        if time.monotonic() > deadline:
            break
        print(f"\nPhase 2: Subgradient from init {i}, time budget={time_per_init:.0f}s")
        result_a, result_score = multi_peak_subgradient(init, n, phase_deadline, verbose=True)
        if result_score < best_score:
            best_score = result_score
            best_a = result_a.copy()
            print(f"  New best: {best_score:.6f}")

    # Phase 3: Further refinement with subgradient from the best
    if time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        print(f"\nPhase 3: Refined subgradient, time={remaining:.0f}s")
        result_a, result_score = multi_peak_subgradient(best_a, n, deadline, verbose=True)
        if result_score < best_score:
            best_score = result_score
            best_a = result_a.copy()
            print(f"  New best: {best_score:.6f}")

    # Phase 4: Coordinate descent refinement
    if time.monotonic() < deadline - 60:
        remaining = min(deadline - time.monotonic() - 30, 300.0)
        print(f"\nPhase 4: Coordinate descent, time={remaining:.0f}s")
        cd_deadline = time.monotonic() + remaining
        result_a, result_score = coordinate_descent(best_a, n, cd_deadline, verbose=True)
        if result_score < best_score:
            best_score = result_score
            best_a = result_a.copy()
            print(f"  New best: {best_score:.6f}")

    # Phase 5: Final verification and return
    # Verify with actual evaluate_sequence
    list_a = [float(x) for x in np.clip(best_a, 0, 1000)]
    actual_score = evaluate_sequence(list_a)
    print(f"\nFinal verified score: {actual_score:.6f}")

    if not np.isfinite(actual_score):
        # Fallback to parent
        print("WARNING: Fallback to parent sequence")
        parent_list = [float(x) for x in np.clip(parent, 0, 1000)]
        return parent_list

    return list_a
