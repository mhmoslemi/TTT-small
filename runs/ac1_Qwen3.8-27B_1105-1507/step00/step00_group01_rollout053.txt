def compute_smooth_grad(a, T):
    """Compute smooth approximation of max(conv(a,a)) and its gradient w.r.t. a."""
    n = len(a)
    b = np.convolve(a, a)  # length 2n-1, O(n^2) or use FFT
    # Softmax weights
    b_max = b.max()
    exp_b = np.exp((b - b_max) / T)
    w = exp_b / exp_b.sum()
    value = b_max + T * np.log(exp_b.sum())  # = T * logsumexp(b/T)
    
    # Gradient
    a_rev = a[::-1]
    # Need L >= (2n-1) + n - 1 = 3n-2 for both convolutions
    L = 1 << (3*n - 2).bit_length()
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    AR = np.fft.rfft(a_rev, n=L)
    
    # term1 = conv(w, a_rev)[n-1 : 2n-1]
    conv_w_ar = np.fft.irfft(W * AR)[:3*n-2]
    term1 = conv_w_ar[n-1:2*n-1]

    conv_a_w = np.fft.irfft(A * W)[:3*n-2]
    term2 = conv_a_w[:n]

    return value, grad

import numpy as np

def compute_objective_fast(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 using FFT."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    fa = np.zeros(L, dtype=np.complex128)
    fa[:n] = a
    fa = np.fft.rfft(fa)
    b = np.fft.irfft(fa * fa)[:size]
    max_b = b.max()
    return float(2.0 * n * max_b / (s * s))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = 1.0 - u / n * (cssv - 1.0) > 0
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n)
    theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)
