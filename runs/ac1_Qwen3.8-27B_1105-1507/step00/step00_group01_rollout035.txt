import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    # Start with parent sequence
    f = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    f = f / f.sum()  # normalize
    
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    
    deadline = time.monotonic() + budget_s - 5
    
    # LP-based iterations
    while time.monotonic() < deadline:
        c = np.convolve(f, f)
        M = c.max()
        
        # Build constraint matrix for LP
        # A @ g <= M, where A[k,j] = f[k-j] if 0 <= k-j < n else 0
        # g >= 0
        # Maximize sum(g)
        
        # Number of rows: 2n-1, columns: n
        # Build sparse matrix
        rows = []
        cols = []
        vals = []
        for k in range(2*n - 1):
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            for j in range(i_start, i_end + 1):
                # constraint k involves g[j] with coefficient f[k-j]
                rows.append(k)
                cols.append(j)
                vals.append(f[k-j])
        
        A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        b = np.full(2*n-1, M)
        
        # Maximize sum(g) = minimize -sum(g)
        c_obj = -np.ones(n)
        
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        
        if res.success:
            g = res.x
            g_sum = g.sum()
            if g_sum > 0:
                h = g / g_sum
                # Line search for step size
                for t in [0.5, 0.3, 0.1, 0.05, 0.01]:
                    f_new = (1-t)*f + t*h
                    if f_new.min() >= 0:
                        f_new = f_new / f_new.sum()
                        score = evaluate_sequence(f_new.tolist())
                        if score < best_score:
                            best_f = f_new.copy()
                            best_score = score
                            f = f_new
                            break
    
    return [float(x) for x in best_f]

**Faster approach: Projected subgradient / mirror descent**

Since the LP is expensive, let me use a simpler approach:

1. Compute c = f*f, find the set S of "active" indices where c[k] is close to max(c).
2. Compute a "direction" that reduces the active c[k] values.
3. Project onto the feasible set {g >= 0, sum(g) = 1}.

The subgradient of max_{k in S} c[k] with respect to f[i] is:
(1/|S|) * sum_{k in S} (dc[k]/df[i]) = (1/|S|) * sum_{k in S} 2*f[k-i] (with zero-padding)

Let me define the "autocorrelation gradient" at position i as:
grad[i] = (1/|S|) * sum_{k in S} 2*f[k-i] (where f is zero outside [0,n-1])

Then the descent direction is -grad. We update:
f_new[i] = max(0, f[i] - alpha * grad[i])
f_new = f_new / f_new.sum()

This is a projected subgradient step. The main cost is computing grad, which involves summing f[k-i] over k in S. If |S| is small, this is fast.

But which indices are in S? We can use a "soft" max: weight k by exp(c[k]/epsilon) for small epsilon, or just use the top-m values.

Let me think about the cost:
- Computing c = f*f: O(n^2) with direct convolution, or O(n log n) with FFT.
- Computing grad: O(|S| * n) if done naively, or O(n log n) with FFT.
- Projection: O(n).

With FFT, each iteration is O(n log n), which for n = 6121 is about 6121 * 13 ≈ 80,000 operations. Very fast!

import numpy as np
import time
from math import comb as math_comb

def _build_greedy(n):
    """Build the greedy sequence C(k) = binom(2k,k)/4^k, normalized."""
    f = np.zeros(n)
    for k in range(n):
        f[k] = math_comb(2*k, k) / (4.0 ** k)
    f /= f.sum()
    return f

def _convolve_full(f):
    """Full convolution using FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    c = np.fft.irfft(F * F, size)
    return c[:2 * n - 1]

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _subgradient_step(f, c, m, alpha):
    """One projected subgradient step."""
    n = len(f)
    top_idx = np.argsort(c)[-m:]
    grad = np.zeros(n)
    for k in top_idx:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            grad[i_start:i_end + 1] += f[k - i_end:k - i_start + 1][::-1]
    grad *= 2.0 / m
    f_new = f - alpha * grad
    f_new = _project_simplex(f_new)
    return f_new
