```python
import numpy as np
import time

def fast_score(a):
    """Fast computation of J = 2n * max(conv) / sum^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)

def normalize(a):
    """Normalize so sum = 1, clip to [0, 1000]."""
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s

def peak_directed_move(a, rng, delta):
    """Transfer mass from high-influence to low-influence positions."""
    n = len(a)
    a = a.copy()
    conv = np.convolve(a, a)
    k_star = np.argmax(conv)
    
    # Influence of position i on the peak at k_star:
    # (a*a)[k_star] includes a[i] * a[k_star - i]
    # So influence[i] = a[k_star - i] if 0 <= k_star - i < n, else 0
    influence = np.zeros(n)
    valid = (k_star >= np.arange(n)) & (k_star - np.arange(n) < n)
    ib = k_star - np.arange(n)
    ib[~valid] = 0
    influence[valid] = a[ib[valid]]
    
    # Donor: position with highest a[i] * influence[i] (contributes most to peak)
    contribution = a * influence
    # Recipient: position with lowest influence
    # But we also want to not create new peaks
    
    # Simple strategy: transfer from top contributor to random low-influence position
    donor = np.argmax(contribution)
    # Choose recipient from positions with low influence
    low_mask = influence < np.median(influence)
    low_positions = np.where(low_mask)[0]
    if len(low_positions) == 0:
        return a
    recipient = low_positions[rng.integers(len(low_positions))]
    
    # Transfer delta
    transfer = min(delta, a[donor] * 0.1)  # Don't transfer more than 10% of donor
    a[donor] -= transfer
    a[recipient] += transfer
    
    return np.clip(a, 0.0, None)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0
    
    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = None
    
    if parent is not None and parent.size > 0 and np.isfinite(parent).all():
        n = parent.size
        best = np.clip(parent, 0.0, None)
    else:
        n = 256
        best = np.ones(n)
    
    # Normalize
    best = normalize(best)
    best_score = fast_score(best)
    
    print(f"Initial: n={n}, score={best_score:.6f}")
    
    delta = 0.01  # Initial transfer amount
    
    while time.monotonic() < deadline:
        # Peak-directed move
        candidate = peak_directed_move(best, rng, delta)
        candidate = normalize(candidate)
        cand_score = fast_score(candidate)
        
        if cand_score < best_score:
            best = candidate
            best_score = cand_score
            delta = min(delta * 1.05, 0.1)  # Increase step slightly
        else:
            delta = max(delta * 0.95, 0.001)  # Decrease step
        
        # Occasionally try random block perturbation
        if rng.random() < 0.1:
            candidate = best.copy()
            width = rng.integers(max(1, n//100), max(2, n//10))
            start = rng.integers(0, n - width)
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.3))
            candidate = normalize(candidate)
            cand_score = fast_score(candidate)
            if cand_score < best_score:
                best = candidate
                best_score = cand_score
    
    print(f"Final: score={best_score:.6f}")
    return [float(x) for x in best]
```

```python
"""
Step function optimization: minimize J = 2n*max(conv(a,a))/sum(a)^2.
Uses peak-directed local search with diverse initializations.
"""
import numpy as np
import time

def fast_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))

def normalize(a):
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def make_initial_candidates(n, rng):
    """Generate diverse initial candidates."""
    candidates = []
    
    # Constant
    candidates.append(np.ones(n))
    
    # Linear ramp (increasing)
    candidates.append(np.linspace(0.1, 1.0, n))
    
    # Linear ramp (decreasing)
    candidates.append(np.linspace(1.0, 0.1, n))
    
    # Quadratic (bowl)
    x = np.linspace(-1, 1, n)
    candidates.append(1.0 - x**2)
    
    # Quadratic (hill)
    candidates.append(x**2 + 0.1)
    
    # Exponential decay left
    candidates.append(np.exp(-3.0 * np.arange(n) / n))
    
    # Exponential decay right
    candidates.append(np.exp(-3.0 * np.arange(n-1, -1, -1) / n))
    
    # Two blobs
    for ratio in [0.3, 0.4, 0.45]:
        L = int(n * ratio)
        blob = np.zeros(n)
        blob[:L] = 1.0
        blob[n-L:] = 1.0
        candidates.append(blob)
    
    # Sawtooth
    P = max(2, n // 10)
    saw = np.zeros(n)
    for i in range(n):
        saw[i] = (i % P) / P
    candidates.append(saw)
    
    # Random
    for _ in range(5):
        candidates.append(rng.random(n) + 0.1)
    
    # Parent-like with perturbation
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.size == n:
            candidates.append(parent)
            candidates.append(parent + 0.1 * rng.standard_normal(n))
    except (NameError, TypeError):
        pass
    
    # Normalize all
    normalized = []
    for c in candidates:
        c = normalize(c)
        normalized.append(c)
    
    return normalized

def local_search(a, rng, deadline, verbose=False):
    """
    Peak-directed local search.
    Moves: mass transfer from hot to cold positions, block scaling.
    """
    n = len(a)
    best = a.copy()
    best_score = fast_score(best)
    delta = 0.02
    iters = 0
    stall = 0
    
    while time.monotonic() < deadline:
        conv = np.convolve(best, best)
        k_star = int(np.argmax(conv))
        
        # Compute contribution of each position to the peak
        idx = np.arange(n)
        j_idx = k_star - idx
        valid = (j_idx >= 0) & (j_idx < n)
        contrib = np.zeros(n)
        contrib[valid] = best[idx[valid]] * best[j_idx[valid]]
        
        move_type = rng.random()
        
        if move_type < 0.6:
            # Peak-directed mass transfer
            candidate = best.copy()
            donor = int(np.argmax(contrib))
            
            # Choose recipient: low contribution, and also check it won't
            # create a new peak (approximate by checking its "global" influence)
            low_thresh = np.percentile(contrib, 20)
            low_positions = np.where(contrib <= low_thresh)[0]
            if len(low_positions) == 0:
                low_positions = np.array([int(np.argmin(contrib))])
            recipient = int(low_positions[rng.integers(len(low_positions))])
            
            transfer = delta * best[donor]
            transfer = min(transfer, 0.1 * best[donor])
            if transfer < 1e-12:
                continue
            candidate[donor] -= transfer
            candidate[recipient] += transfer
            
        elif move_type < 0.8:
            # Block scaling
            candidate = best.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 8)))
            start = int(rng.integers(0, n - width))
            factor = float(np.exp(rng.normal(0, 0.2)))
            candidate[start:start+width] *= factor
            
        else:
            # Single position perturbation
            candidate = best.copy()
            pos = int(rng.integers(n))
            candidate[pos] *= float(np.exp(rng.normal(0, 0.3)))
        
        candidate = np.clip(candidate, 0.0, None)
        candidate = normalize(candidate)
        
        cand_score = fast_score(candidate)
        
        if cand_score < best_score:
            best = candidate
            best_score = cand_score
            stall = 0
            delta = min(delta * 1.02, 0.2)
        else:
            stall += 1
            if stall > 100:
                delta = max(delta * 0.8, 0.001)
            # Simulated annealing: accept with small probability
            if rng.random() < 0.01:
                best = candidate
                best_score = cand_score
        
        iters += 1
        if verbose and iters % 50000 == 0:
            print(f"  Iter {iters}: score={best_score:.6f}")
    
    return best, best_score

    start_time = time.monotonic()
    total_budget = budget_s - 15.0  # Leave 15s for output

        n = 512

    phase1_time = min(60.0, total_budget * 0.1)
    phase1_deadline = time.monotonic() + phase1_time
    
    candidates = make_initial_candidates(n, rng)

    best_score = np.inf
    
    for c in candidates:
        sc = fast_score(c)
        if sc < best_score:
            best = c
            best_score = sc
            print(f"  New best (phase 1): {sc:.6f}")
    
    # Short refinement on top-5 candidates

        refined, refined_score = local_search(

        if refined_score < best_score:
            best = refined
            best_score = refined_score

    print(f"After Phase 1: best_score = {best_score:.6f}")
    
    # Phase 2: Long local search
    phase2_deadline = time.monotonic() + (total_budget - (time.monotonic() - start_time))
    best, best_score = local_search(best, rng, phase2_deadline, verbose=True)
    
    print(f"After Phase 2 (local search): best_score = {best_score:.6f}")
    
    # Final validation
    final_score = float(evaluate_sequence(best.tolist()))
    print(f"Final validated score: {final_score:.6f}")
    
    return [float(x) for x in best]
```
