def _active_gradient(f, D, K):
    n = f.size
    m = D.size
    if K <= 0 or n == 0:
        return np.zeros(n, dtype=np.float64)
    if K >= m:
        active = np.arange(m)
    else:
        active = np.argpartition(D, -K)[-K:]
    out = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            out[lo:hi+1] += f[k - idx]
    out /= float(active.size)
    return 2.0 * out

```python
"""
Searches for a non-negative sequence minimizing
    2 * n * max(self-convolution) / (sum(sequence))^2.

Algorithm:
- Normalize to the simplex sum = 1.
- Use FFT-based self-convolution and gradient curvature.
- Apply smooth multiplicative gradient descent with an adaptively scaled
  log-sum-exp / active-lag subgradient.
- Use backtracking line search, random perturbations, and periodic exact
  evaluation to avoid numerical trap accumulation.
"""

import numpy as np
import time


def _next_pow2(m):
    m = int(max(1, m))
    s = 1
    while s < m:
        s <<= 1
    return s


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _self_conv_fft(p, size):
    p = np.asarray(p, dtype=np.float64)
    m = 2 * p.size - 1
    F = np.fft.rfft(p, size)
    out = np.fft.irfft(F * F, size)
    out = out[:m]
    out = np.asarray(out, dtype=np.float64)
    np.maximum(out, 0.0, out=out)
    return out


def _weights_from_D(D, alpha):
    D = np.asarray(D, dtype=np.float64)
    if D.size == 0:
        return None, 0.0, 0.0

    dmax = float(np.max(D))
    if not np.isfinite(dmax):
        D = np.nan_to_num(D, nan=0.0, posinf=0.0, neginf=0.0)
        dmax = float(np.max(D))

    if dmax <= 1e-300:
        return None, max(0.0, dmax), max(0.0, dmax)

    dmin = float(np.min(D))
    if not np.isfinite(dmin):
        dmin = 0.0

    scale = dmax - dmin
    m = D.size
    a = float(alpha)
    if a <= 0.0:
        a = 1.0

    if scale <= 1e-300:
        w = np.full(m, 1.0 / float(m), dtype=np.float64)
        return w, dmax, dmax

    z = (a * (dmax - D)) / scale
    z = np.clip(z, 0.0, a)
    expvals = np.exp(-z)
    sum_exp = float(np.sum(expvals))

    if not np.isfinite(sum_exp) or sum_exp <= 1e-300:
        w = np.full(m, 1.0 / float(m), dtype=np.float64)
        return w, dmax, dmax

    w = expvals / sum_exp
    L = dmax + (scale / a) * float(np.log(sum_exp))
    return w, float(L), dmax


def _gradient_fft(w, f, size):
    n = f.size
    f_rev = f[n - 1::-1]
    Fw = np.fft.rfft(w, size)
    Fr = np.fft.rfft(f_rev, size)
    h = np.fft.irfft(Fw * Fr, size)
    c = h[n - 1:2 * n - 1].copy()
    np.maximum(c, 0.0, out=c)
    return 2.0 * c


def _active_gradient(f, D, K):
    n = f.size
    m = D.size
    if K <= 0 or n == 0:
        return np.zeros(n, dtype=np.float64)

    if K >= m:
        active = np.arange(m)
    else:
        active = np.argpartition(D, -K)[-K:]

    out = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            out[lo:hi + 1] += f[k - idx]

    out /= float(active.size)
    return 2.0 * out


def _perturb(f, rng, sigma):
    f = np.asarray(f, dtype=np.float64).copy()
    size = f.size
    if size == 0:
        return f

    if sigma > 0.0:
        f = f * np.exp(rng.normal(0.0, sigma, size))

    f = f + (1e-8 / float(size))
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(size, dtype=np.float64) / float(size)
    return f / s


def _safe_evaluate(a):
    try:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return np.inf
        vals = [float(x) for x in np.maximum(a, 0.0)]
        return float(evaluate_sequence(vals))
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=1500.0, **kwargs):
    start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1500.0
    if not np.isfinite(budget) or budget <= 30.0:
        budget = 1500.0

    deadline = start + max(10.0, budget - 25.0)

    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        base = np.ones(6121, dtype=np.float64)

    if base.size < 2:
        base = np.ones(6121, dtype=np.float64)

    # Safety cap for very large parents. The known parent is much smaller.
    if base.size > 12000:
        idx = np.linspace(0, base.size - 1, 12000, dtype=int)
        base = base[idx]

    base = np.where(np.isfinite(base), base, 0.0)
    base = np.maximum(base, 0.0)

    s0 = float(np.sum(base))
    if not np.isfinite(s0) or s0 < 0.01:
        f = np.ones(base.size, dtype=np.float64) / float(base.size)
    else:
        f = base / s0

    n = int(f.size)

    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    # Slightly break symmetry and ensure full support.
    f = 0.95 * f + 0.05 * (1.0 / float(n))
    f = f * np.exp(rng.normal(0.0, 0.03, n))
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 0.0)
    f = _normalize(f)

    size_self = _next_pow2(2 * n - 1)
    size_grad = _next_pow2(3 * n - 2)

    D = _self_conv_fft(f, size_self)
    dmax = float(np.max(D))
    best_score = 2.0 * float(n) * dmax
    best_f = f.copy()

    exact_best_score = np.inf
    exact_best_f = best_f.copy()

    print(f"n={n} initial approx score={best_score:.6f}", flush=True)

    alpha = 8.0
    eta = 0.35
    max_eta = 1.0
    min_eta = 1e-5

    no_improve = 0
    fails = 0
    successes = 0
    it = 0
    last_print = start
    next_exact = start + 45.0
    use_active_interval = 700

    while time.monotonic() < deadline:
        dmax = float(np.max(D))
        if not np.isfinite(dmax):
            f = _normalize(f)
            D = _self_conv_fft(f, size_self)
            continue

        current_score = 2.0 * float(n) * dmax
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400:
            alpha = min(25.0, alpha + 0.1)

        if no_improve > 4000:
            sigma = 0.35 + 0.45 * min(1.0, float(no_improve - 4000) / 8000.0)
            f = _perturb(best_f, rng, sigma)
            D = _self_conv_fft(f, size_self)
            no_improve = 0
            fails = 0
            successes = 0
            eta = 0.35
            alpha = 10.0
            continue

        w, L, _ = _weights_from_D(D, alpha)
        if w is None:
            f = _perturb(best_f, rng, 0.20)
            D = _self_conv_fft(f, size_self)
            continue

        use_active = (no_improve > 800 and (it % use_active_interval) == 0)
        if use_active:
            raw_grad = _active_gradient(f, D, 30)
        else:
            raw_grad = _gradient_fft(w, f, size_grad)

        raw_grad = np.where(np.isfinite(raw_grad), raw_grad, 0.0)
        raw_grad = np.maximum(raw_grad, 0.0)
        gsum = float(np.sum(raw_grad))

        if not np.isfinite(gsum) or gsum <= 1e-300:
            raw_grad = _active_gradient(f, D, 20)
            raw_grad = np.where(np.isfinite(raw_grad), raw_grad, 0.0)
            raw_grad = np.maximum(raw_grad, 0.0)
            gsum = float(np.sum(raw_grad))

        if not np.isfinite(gsum) or gsum <= 1e-300:
            f = _perturb(best_f, rng, 0.10)
            D = _self_conv_fft(f, size_self)
            fails += 1
            continue

        grad = raw_grad * (float(n) / gsum)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        grad = np.clip(grad, 0.0, 100.0)

        accepted = False
        accepted_f = None
        accepted_D = None
        accepted_max = dmax
        trial_eta = eta

        for _trial in range(4):
            factor = np.exp(-trial_eta * grad)
            f_new = f * factor
            f_new = np.where(np.isfinite(f_new), f_new, 0.0)
            f_new = np.maximum(f_new, 1e-300)

            snew = float(np.sum(f_new))
            if not np.isfinite(snew) or snew <= 1e-300:
                trial_eta *= 0.5
                continue

            f_new = f_new / snew
            D_new = _self_conv_fft(f_new, size_self)
            d_new = float(np.max(D_new))

            if not np.isfinite(d_new):
                trial_eta *= 0.5
                continue

            tol = max(1e-18, 1e-11 * dmax)

            if d_new <= dmax + tol:
                accepted = True
                accepted_f = f_new
                accepted_D = D_new
                accepted_max = d_new
                eta = trial_eta
                break

            if d_new <= dmax * (1.0 + 1e-5):
                _, L_new, _ = _weights_from_D(D_new, alpha)
                if np.isfinite(L_new) and L_new < L - 1e-15:
                    accepted = True
                    accepted_f = f_new
                    accepted_D = D_new
                    accepted_max = d_new
                    eta = trial_eta
                    break

            trial_eta *= 0.5
            if trial_eta < min_eta:
                break

        if accepted:
            f = accepted_f
            D = accepted_D
            dmax = accepted_max
            successes += 1
            fails = 0

            if 2.0 * float(n) * dmax < best_score - 1e-14:
                best_score = 2.0 * float(n) * dmax
                best_f = f.copy()
                no_improve = 0

            if successes % 60 == 0:
                eta = min(max_eta, eta * 1.05)

            alpha = max(5.0, alpha * 0.999)
        else:
            fails += 1
            eta = max(min_eta, trial_eta)

            if fails >= 6:
                f = _perturb(best_f, rng, 0.12)
                D = _self_conv_fft(f, size_self)
                fails = 0
                successes = 0
                eta = 0.35
                no_improve = min(no_improve, 1000)

        it += 1

        if it % 1000 == 0 or (time.monotonic() - last_print > 20.0):
            print(
                f"it={it} approx_best={best_score:.6f} "
                f"current={2.0 * n * dmax:.6f} alpha={alpha:.1f} "
                f"eta={eta:.3f} noimp={no_improve}",
                flush=True,
            )
            last_print = time.monotonic()

        if time.monotonic() >= next_exact:
            sc_f = _safe_evaluate(f)
            if np.isfinite(sc_f) and sc_f < exact_best_score:
                exact_best_score = sc_f
                exact_best_f = f.copy()

            sc_best = _safe_evaluate(best_f)
            if np.isfinite(sc_best) and sc_best < exact_best_score:
                exact_best_score = sc_best
                exact_best_f = best_f.copy()

            print(
                f"exact_best={exact_best_score:.6f} "
                f"approx_best={best_score:.6f}",
                flush=True,
            )
            next_exact = time.monotonic() + 60.0

    # Final exact selection among the best candidates.
    final_f = best_f.copy()
    final_score = np.inf

    candidates = [exact_best_f, best_f, f]
    for cand in candidates:
        if cand is None:
            continue
        cand = np.asarray(cand, dtype=np.float64).copy()
        if cand.size != n:
            continue
        cand = _normalize(cand)
        sc = _safe_evaluate(cand)
        if np.isfinite(sc) and sc < final_score:
            final_score = sc
            final_f = cand

    final_f = _normalize(final_f)
    last_sc = _safe_evaluate(final_f)

    if not np.isfinite(last_sc):
        final_f = np.ones(n, dtype=np.float64) / float(n)
        last_sc = _safe_evaluate(final_f)

    print(f"final exact score={last_sc:.6f}", flush=True)
    return [float(x) for x in final_f]
```
