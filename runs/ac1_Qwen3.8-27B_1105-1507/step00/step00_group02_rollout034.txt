"""
Minimizes 2n*max(convolve(a,a))/sum(a)^2 using smooth-max subgradient descent
with FFT-accelerated gradients, multi-start exploration, and coordinate descent refinement.
Key insight: non-palindromic sequences can achieve max_conv < 1/n, beating the uniform bound of 2.

import numpy as np

def compute_conv_fft(a):
    """Compute np.convolve(a, a) using FFT for O(n log n) efficiency."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def smooth_max_weights(conv, beta):
    """Compute softmax weights for smooth max approximation."""
    shifted = conv - np.max(conv)
    exp_vals = np.exp(beta * shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        # Fallback: uniform over the argmax
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return exp_vals / total


def gradient_fft(a, weights, beta):
    """Compute gradient of smooth_max(conv, beta) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    # g[j] = 2*beta * sum_k weights[k] * a[k-j] (for valid k-j)
    # This equals 2*beta * (cross-correlation of a with weights)[j]
    a_flipped = a[::-1].copy()
    # We need cross[j] = sum_m a[m] * weights[j+m] for j=0..n-1
    # = (a_flipped * weights)[j + n - 1]
    N2 = 1
    while N2 < n + 2 * n:
        N2 *= 2
    fa = np.fft.rfft(a_flipped, N2)
    fw = np.fft.rfft(weights, N2)
    cross = np.fft.irfft(fa * fw, N2)
    g = 2.0 * beta * cross[n - 1:2 * n - 1]
    return g


def simplex_project(v):
    """Project onto probability simplex (non-negative, sum=1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(len(v)) / len(v)
    return v / s


def make_linear_ramp(n):
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def make_cosine_ramp(n):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / (2.0 * n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_exponential(n, tau_frac):
    a = np.exp(-np.arange(n, dtype=np.float64) / (n * tau_frac))
    return a / a.sum()


def make_zadoff_chu_magnitude(n, m=1):
    """Create non-negative sequence from Zadoff-Chu based oscillation."""
    i = np.arange(n, dtype=np.float64)
    phase = np.pi * m * i * (i + 1) / n
    a = (np.cos(phase) + 1.0) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_random_positive(n, rng, smoothness=1.0):

    # Optional smoothing
    if smoothness > 0:
        kernel_size = max(2, int(smoothness * n / 100))
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def evaluate_normalized(a):
    """Compute score assuming sum(a)=1. Returns 2n*max_conv."""
    n = len(a)
    conv = compute_conv_fft(a)
    return 2.0 * n * np.max(conv)
