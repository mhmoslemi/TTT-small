def lp_solve(b, n):
    """Solve: min t s.t. sum_i a_i * b_{k-i} <= t, a_i >= 0, sum a_i = 1"""
    # Variables: a_0, ..., a_{n-1}, t
    # c = [0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1
    
    # Constraints: sum_i a_i * b_{k-i} - t <= 0 for each k
    # b[k-i] where 0 <= k-i < n, i.e., i in [k-n+1, k] intersect [0, n-1]

        A_ub[k, -1] = -1  # -t

    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1

"""
Minimize 2n*max(convolve(a,a))/sum(a)^2 using LP alternating minimization,
FISTA-based projected gradient refinement, and block local search.
Exploits scale invariance (normalize sum=1) and efficient gradient computation
via FFT-based convolution and numpy correlate.

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(a):
    """Compute the objective value directly (scale-invariant check)."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = np.sum(w)
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def lp_step(prev_a):
    """One LP alternating minimization step.
    Solves: min t s.t. (prev_a * x)[k] <= t, sum(x)=1, x>=0
    """
    n = len(prev_a)
    # Build convolution matrix M where (M @ x)[k] = (prev_a * x)[k]
    M = np.zeros((2 * n - 1, n))
    for i in range(n):
        M[i:i + n, i] = prev_a

    # LP formulation
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([M, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(-np.inf, np.inf)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual': False})

        x = np.clip(res.x[:n], 0, None)
        s = np.sum(x)
        if s < 1e-10:

        return x / s
    return None

    """Compute gradient of (1/beta)*log(sum_k exp(beta*(a*a)[k])).
    Uses: grad_j = 2 * correlate(weights, a, 'valid')[j]
    where weights are softmax of (a*a).
    """
    b = np.convolve(a, a)
    b_max = np.max(b)
    exp_b = np.exp(beta * (b - b_max))
    total = np.sum(exp_b)
    if total < 1e-300:
        return np.zeros(len(a))
    weights = exp_b / total
    grad = 2.0 * np.correlate(weights, a, mode='valid')

def fista_step(a, a_prev, t_old, beta, eta):
    """One FISTA iteration for min max(a*a) s.t. a in simplex."""
    n = len(a)
    t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
    y = a + ((t_old - 1.0) / t_new) * (a - a_prev)
    y = np.clip(y, 0, None)
    sy = np.sum(y)
    if sy < 1e-15:
        y = np.ones(n) / n
    else:
        y = y / sy

    grad = smoothed_gradient(y, beta)
    z = y - eta * grad
    x_new = project_simplex(z)
    return x_new, y, t_new


def block_perturb(a, rng, n):
    """Apply a random multiplicative block perturbation and re-project."""
    a_new = a.copy()
    n_blocks = int(rng.integers(1, 4))
    for _ in range(n_blocks):
        width = int(rng.integers(max(1, n // 200), max(2, n // 10)))
        start = int(rng.integers(0, n - width))
        a_new[start:start + width] *= np.exp(rng.normal(0.0, 0.08, width))
    a_new = project_simplex(a_new)
    return a_new


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search: LP alternating + FISTA refinement + local search."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 5.0
    n = 2535

    # --- Initialization ---
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(n)
    if parent.ndim != 1 or parent.size != n or not np.all(np.isfinite(parent)):
        parent = np.ones(n)
    parent = np.clip(parent, 0, 1000)
    if np.sum(parent) < 0.01:
        parent = np.ones(n)

    # Generate diverse initializations
    inits = []
    inits.append(parent / np.sum(parent))

    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    for sf in [0.15, 0.2, 0.25, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - center) / (sf * n)) ** 2)
        g = np.maximum(g, 1e-10)
        inits.append(g / np.sum(g))

    tri = np.minimum(x - 0.5, n - 1 - x - 0.5) + 0.5
    inits.append(tri / np.sum(tri))

    # Cosine-modulated
    for freq in [1, 2, 3]:
        mod = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * x / n)
        mod = np.maximum(mod, 0.1)
        inits.append(mod / np.sum(mod))

    # Random smooth
    for _ in range(5):
        r = np.abs(rng.normal(1.0, 0.3, n))
        r = np.convolve(r, np.ones(10) / 10, mode='same')
        r = np.maximum(r, 1e-10)
        inits.append(r / np.sum(r))

    best = None
    best_score = np.inf
    for init in inits:
        init = np.clip(init, 0, 1000)
        s = np.sum(init)
        if s < 0.01:
            continue
        init = init / s
        sc = compute_score(init)
        if sc < best_score:
            best_score = sc
            best = init.copy()

    print(f"Best init: score={best_score:.8f}")
    # Verify with actual evaluator
    actual = evaluate_sequence(best.tolist())
    if np.isfinite(actual) and actual < best_score:
        best_score = actual
    print(f"Verified: {best_score:.8f}")

    # --- Phase 1: LP Alternating Minimization ---
    print("Phase 1: LP alternating minimization")
    a = best.copy()
    lp_iter = 0
    stall = 0
    best_lp_score = best_score

    remaining = deadline - time.monotonic()
    lp_budget = remaining * 0.55
    lp_deadline = start_time + lp_budget

    while time.monotonic() < lp_deadline:
        result = lp_step(a)
        if result is None:
            break
        new_score = compute_score(result)
        if new_score < best_score:
            best_score = new_score
            best = result.copy()
            stall = 0
        else:
            stall += 1
        a = result
        lp_iter += 1

        if lp_iter % 50 == 0:
            print(f"  LP {lp_iter}: best={best_score:.8f}, cur={new_score:.8f}")

        if stall >= 40:
            # Perturb to escape
            a = block_perturb(a, rng, n)
            stall = 0

    print(f"Phase 1 done: {lp_iter} LP iters, best={best_score:.8f}")

    # --- Phase 2: FISTA Refinement ---
    print("Phase 2: FISTA refinement")
    a = best.copy()
    a_prev = best.copy()
    t_curr = 1.0
    beta = 30.0
    eta = 1.0 / (beta * 2.0 * n)  # initial step size (conservative)
    fista_iter = 0

    fista_deadline = min(deadline - 60, lp_deadline + (deadline - lp_deadline) * 0.6)

    while time.monotonic() < fista_deadline:
        a, _, t_curr = fista_step(a, a_prev, t_curr, beta, eta)
        a_prev = a
        fista_iter += 1

        if fista_iter % 100 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best = a.copy()

        if fista_iter % 2000 == 0:
            beta = min(beta * 1.4, 5000.0)
            eta = max(eta * 0.8, 1e-7)

    print(f"Phase 2 done: {fista_iter} FISTA iters, best={best_score:.8f}")

    # --- Phase 3: Second LP pass from refined point ---
    if time.monotonic() < deadline - 120:
        print("Phase 3: Second LP pass")
        a = best.copy()
        lp2_deadline = deadline - 120
        stall2 = 0
        lp2_iter = 0
        while time.monotonic() < lp2_deadline:
            result = lp_step(a)
            if result is None:
                break
            new_score = compute_score(result)
            if new_score < best_score:
                best_score = new_score
                best = result.copy()
                stall2 = 0
            else:
                stall2 += 1
            a = result
            lp2_iter += 1
            if stall2 > 50:
                a = block_perturb(a, rng, n)
                stall2 = 0
        print(f"  LP pass 2: {lp2_iter} iters, best={best_score:.8f}")

    # --- Phase 4: Local search ---
    print("Phase 4: Block local search")
    a = best.copy()
    ls_iter = 0
    while time.monotonic() < deadline - 10:
        a_new = block_perturb(a, rng, n)
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best = a_new
            a = a_new
        ls_iter += 1
        if ls_iter % 5000 == 0:
            print(f"  LS {ls_iter}: best={best_score:.8f}")

    # --- Final verification ---
    best = np.clip(best, 0, 1000)
    final_score = evaluate_sequence(best.tolist())
    print(f"Final score (evaluator): {final_score:.8f}")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    return [float(x) for x in best]
