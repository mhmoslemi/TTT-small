import numpy as np
from scipy.fft import fft, ifft
import time

def compute_objective(a):
    """Compute 2n * max(a*a) / (sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2 * n * c.max() / (s * s)

def peak_reducing_step(a, alpha=0.1, peak_fraction=0.95):
    """One step of peak-reducing gradient descent."""
    n = len(a)
    c = np.convolve(a, a)
    m = c.max()
    threshold = m * peak_fraction
    peak_indices = np.where(c >= threshold)[0]
    
    # Compute pressure
    pressure = np.zeros(n)
    for k in peak_indices:
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            idx = np.arange(i_min, i_max + 1)
            pressure[idx] += 2.0 * a[k - idx]
    
    # Multiplicative update
    max_p = pressure.max()
    if max_p < 1e-15:
        return a.copy()
    a_new = a * np.exp(-alpha * pressure / max_p)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s
