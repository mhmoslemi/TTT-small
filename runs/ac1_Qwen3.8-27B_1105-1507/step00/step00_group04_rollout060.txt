```python
import numpy as np
import time

def compute_score(a):
    """Fast computation of J without calling evaluate_sequence."""
    n = len(a)
    out = np.convolve(a, a)
    max_out = out.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_out / (sum_a * sum_a)

def gradient_active(a, k_star):
    """Compute subgradient of J w.r.t. a[i] using active lag k_star."""
    n = len(a)
    sum_a = a.sum()
    out = np.convolve(a, a)
    max_out = out[k_star]
    # dJ/da[i] = 2n/sum_a² * 2*a[k_star-i] - 4n*max_out/sum_a³
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    grad *= (2.0 * n) / (sum_a * sum_a)
    grad -= (4.0 * n * max_out) / (sum_a**3)
    return grad
```
