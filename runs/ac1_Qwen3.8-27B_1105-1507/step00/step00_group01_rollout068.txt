"""
Minimize score = 2 * n * max(convolve(a, a)) / (sum(a))**2 over non-negative
real sequences.  The score is scale-invariant, so we normalise every candidate
to sum = 1 and minimise the peak of the (linear) self-convolution.

Key idea (found by inspection of the n=2 case): the uniform sequence (score 2)
is NOT optimal.  An asymmetric / skewed shape lowers the convolution peak below
the 1/n uniform value.  We therefore search over a low-dimensional family of
smooth shapes (piecewise-linear interpolants of a few control points, plus a
few analytic priors) for many sequence lengths, using derivative-free
optimisation (COBYLA + Nelder-Mead polish).  During the search the peak is
evaluated cheaply via FFT; the exact evaluate_sequence is used to confirm the
best candidates.  The function always returns the best finite-score sequence
found (falling back to the supplied parent / uniform sequence).
"""

import time
import numpy as np
from scipy.optimize import minimize


# --------------------------------------------------------------------------
# small standalone helpers (no closures, no lambdas)
# --------------------------------------------------------------------------
def _softplus(z):
    """Numerically stable softplus -> element-wise positive."""
    z = np.asarray(z, dtype=np.float64)
    out = np.empty_like(z)
    m = z > 33.0
    out[m] = z[m]
    nz = ~m
    out[nz] = np.log1p(np.exp(z[nz]))
    return out


def _peak_fft(a):
    """Peak of the linear self-convolution of a, via FFT (fast surrogate)."""
    n = a.size
    nfft = 2 * n - 1
    fa = np.fft.rfft(a, nfft)
    c = np.fft.irfft(fa * fa, nfft)
    # clean tiny negative round-off
    return float(np.max(c))


def _peak_exact(a):
    c = np.convolve(a, a)
    return float(np.max(c))


def _score_seq(a):
    """Exact-ish score from a numpy array (non-negative, sum > 0)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return float(2.0 * a.size * _peak_exact(a) / (s * s))


def _a_from_ctrl(ctrl, n):
    """Length-n non-negative sequence interpolated from K positive control pts."""
    vals = _softplus(np.asarray(ctrl, dtype=np.float64))
    if vals.size == n:
        a = vals
    else:
        a = np.interp(np.arange(n), np.linspace(0.0, n - 1.0, vals.size),
                      np.linspace(0.0, n - 1.0, n) * 0 + 0 + vals)
    # the two linspace above are identical ranges; np.interp skill check:
    # (kept simple: interpolate from K points to n points)
    a = np.interp(np.arange(n), np.arange(vals.size), vals)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s <= 0.0:
        a = np.ones(n)
        s = a.size
    return a / s


def _objective_ctrl(ctrl, n):
    """Peak of self-conv of the normalised (sum=1) interpolated sequence."""
    a = _a_from_ctrl(ctrl, n)
    return _peak_fft(a)


def _initial_priors(n, K):
    """Return a list of K-vectors (in ctrl space) used as optimisation starts."""
    t = np.linspace(0.0, 1.0, K)
    starts = []
    # uniform
    starts.append(np.full(K, 0.0))
    # linear decreasing ramp
    starts.append((1.0 - t) * 3.0)
    # linear increasing ramp
    starts.append(t * 3.0)
    # bell (gaussian hump)
    g = np.exp(-4.0 * (t - 0.5) ** 2)
    starts.append(np.log(np.clip(g, 1e-3, None)))
    # skewed humps
    for p in (0.3, 0.6, 1.5, 2.5):
        s = (t + 1e-6) ** p
        starts.append(np.log(np.clip(s / s.max(), 1e-3, None)))
    for q in (0.3, 1.5, 3.0):
        s = (1.0 - t + 1e-6) ** q
        starts.append(np.log(np.clip(s / s.max(), 1e-3, None)))
    return starts


def _run_one_ctrl(n, K, z0, deadline, maxiter):
    """COBYLA (+ quick Nelder-Mead polish) on the K control points."""
    res = minimize(_objective_ctrl, z0, args=(n,), method="COBYLA",
                   options={"maxiter": maxiter, "maxfev": maxiter * 2,
                            "rhobeg": 1.0, "disp": False})
    if time.time() > deadline:
        # too slow: still accept if it improved, else abort flag handled outside
        pass
    best_z = np.asarray(res.x, dtype=np.float64)
    best_f = _objective_ctrl(best_z, n)
    # Nelder-Mead polish (cheap, small maxiter) to clean up
    if time.time() < deadline - 0.5 and K <= 80:
        try:
            nm = minimize(_objective_ctrl, best_z, args=(n,), method="Nelder-Mead",
                          options={"maxiter": 400, "xatol": 1e-5, "fatol": 1e-8})
            if nm.fun < best_f:
                best_z = np.asarray(nm.x, dtype=np.float64)
                best_f = float(nm.fun)
        except Exception:
            pass
    a = _a_from_ctrl(best_z, n)
    return a, _score_seq(a)


def _run_full(a_init, deadline, maxiter):
    """Optimise the full length-n vector directly (small n only)."""
    n = a_init.size
    z0 = np.log(np.clip(a_init, 1e-4, None) / a_init.max())

    def obj(z):
        a = _softplus(z)
        s = a.sum()
        if s <= 0:
            return np.inf
        return _peak_fft(a / s)

    res = minimize(obj, z0, method="Nelder-Mead",
                   options={"maxiter": maxiter, "xatol": 1e-4, "fatol": 1e-8})
    zbest = np.asarray(res.x, dtype=np.float64)
    a = _softplus(zbest)
    a = np.clip(a, 0.0, None) / max(a.sum(), 1e-12)
    return a, _score_seq(a)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Search for a non-negative sequence minimising evaluate_sequence."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + max(0.0, float(budget_s)) - 4.0

    # ---- parent / fallback -------------------------------------------------
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        parent = None
    if parent is None or parent.size == 0:
        parent = np.ones(256, dtype=np.float64)
    parent = parent / max(parent.sum(), 1e-12)

    best_a = parent.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    print("[init] parent n=%d score=%.6f" % (best_a.size, best_score))

    # ---- lengths to explore (log-spaced) ----------------------------------
    n_list = [2, 3, 4, 5, 6, 8, 10, 13, 16, 21, 27, 34, 45, 60, 80,
              110, 150, 200, 300, 450, 700, 1000, 1500, 2100, 3000, 4200]
    n_list = sorted(set([n for n in n_list if n <= max(64, parent.size * 2)]))

    def control_K(n):
        # resolution of the shape: a few dozen is plenty
        return int(min(max(12, n // 16), 96))

    for n in n_list:
        if time.time() > deadline:
            break
        K = control_K(n)
        starts = _initial_priors(n, K)
        # a couple of random smooth starts
        for _ in range(3):
            base = _initial_priors(n, K)[int(rng.integers(0, len(starts)))]
            starts.append(base + 0.6 * rng.standard_normal(K))

        starts = starts[:8]  # cap number of starts per length
        for z0 in starts:
            if time.time() > deadline:
                break
            # per-start iter budget scales down as we run low
            remaining = deadline - time.time()
            if remaining < 1.0:
                break
            maxiter = int(min(6000, max(300, 700000.0 / max(n, 1))))
            try:
                a, sc = _run_one_ctrl(n, K, np.asarray(z0, dtype=np.float64),
                                      deadline, maxiter)
            except Exception:
                continue
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                print("[n=%d] improved -> %.6f" % (n, best_score))

        # full-vector polish of the current best for this n (short n only)
        if n <= 96 and time.time() < deadline - 3.0:
            try:
                a, sc = _run_full(best_a if best_a.size == n else
                                  np.interp(np.arange(n), np.arange(parent.size),
                                           parent),
                                  deadline, 800)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print("[n=%d fullvec] improved -> %.6f" % (n, best_score))
            except Exception:
                pass

    # ---- final exact verification / light polish of the global best --------
    try:
        exact = float(evaluate_sequence([float(x) for x in best_a]))
        if np.isfinite(exact):
            best_score = exact
        else:
            best_a = parent.copy()
            best_score = float(evaluate_sequence(best_a.tolist()))
    except Exception:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # finale: a couple of extra random shape-starts on the winning length if
    # we still have time, to try to squeeze a little more out.
    if time.time() < deadline - 5.0:
        n = best_a.size
        K = control_K(n)
        for _ in range(4):
            if time.time() > deadline:
                break
            z0 = _initial_priors(n, K)[int(rng.integers(0, len(_initial_priors(n, K))))]
            z0 = z0 + 0.5 * rng.standard_normal(K)
            try:
                a, sc = _run_one_ctrl(n, K, z0, deadline, 4000)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print("[refine n=%d] improved -> %.6f" % (n, best_score))
            except Exception:
                continue

    # normalise output to sum=1 (score-invariant) and clip for safety
    out = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0)
    s = out.sum()
    if s <= 0.01:
        out = parent.copy()
        s = out.sum()
    out = out / max(s, 1e-12)

    try:
        final = float(evaluate_sequence([float(x) for x in out]))
    except Exception:
        final = np.inf
    if not np.isfinite(final):
        out = parent.copy()
        final = float(evaluate_sequence([float(x) for x in out]))
    print("[final] n=%d score=%.6f" % (out.size, final))
    print("[total time used] %.1fs" % (time.time() - (deadline - float(budget_s))))
    return [float(x) for x in out]
