"""
LP-based iterative optimization to minimize 2n*max(f*f)/(sum(f))^2.
Uses the approach of fixing f, solving an LP for the best g with f*g bounded,
then blending. Combined with local search for polishing.
"""

import numpy as np
from scipy.sparse import coo_matrix, csr_matrix
from scipy.optimize import linprog
import time


def build_convolution_matrix(f):
    """Build sparse matrix A where (A @ g)_k = (f * g)_k for full convolution."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A.tocsr()


def solve_lp_step(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M*1, g >= 0."""
    n = len(f)
    A = build_convolution_matrix(f)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x
    return None


def evaluate_score(f):
    """Compute the score without using evaluate_sequence (faster for internal use)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * np.max(conv) / (np.sum(f) ** 2)


def lp_iterate(f, max_iters=500, t_init=0.5, t_min=0.01, tol=1e-8, time_budget=600):
    """Iterative LP-based optimization."""
    f = f.copy().astype(np.float64)
    f = np.maximum(f, 0)
    norm = np.sum(f)
    if norm < 1e-10:
        return f
    f = f / norm  # Normalize to sum = 1
    
    best_f = f.copy()
    best_score = evaluate_score(best_f)
    
    start_time = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break
        
        conv = np.convolve(f, f)
        M = np.max(conv)
        
        # Solve LP
        g0 = solve_lp_step(f, M)
        if g0 is None:
            print(f"  LP failed at iteration {it}")
            break
        
        sum_g0 = np.sum(g0)
        if sum_g0 < 1e-10:
            break
        
        # Rescale g to have same sum as f (which is 1)
        g = g0 / sum_g0
        
        # Adaptive step size
        t = max(t_min, t_init * (0.95 ** it))
        
        # Blend
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        sum_fn = np.sum(f_new)
        if sum_fn < 1e-10:
            break
        f_new = f_new / sum_fn
        
        score_new = evaluate_score(f_new)
        
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
        
        f = f_new
        
        if it % 50 == 0:
            elapsed = time.monotonic() - start_time
            print(f"  Iter {it}: score={score_new:.6f}, best={best_score:.6f}, M={M:.2e}, time={elapsed:.1f}s")
    
    return best_f

LP-based iterative optimization to minimize 2n*max(f*f)/(sum(f))^2.
Uses the approach: fix f, solve LP for best g with f*g <= M, blend, repeat.
Combined with local search for final polishing.
"""

import numpy as np
from scipy.sparse import diags as sparse_diags
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute the objective score."""
    n = len(f)
    s = np.sum(f)
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * np.max(conv) / (s ** 2)


def build_A(f):
    """Build sparse (2n-1) x n matrix A where (A@g)_k = (f*g)_k."""
    n = len(f)
    diag_lists = [np.full(n, f[d]) for d in range(n)]
    offsets = list(range(0, -n, -1))  # [0, -1, -2, ..., -(n-1)]
    A = sparse_diags(diag_lists, offsets, shape=(2*n-1, n), format='csr')

def solve_lp(f):
    """Solve LP: max sum(g) s.t. A_f @ g <= M*1, g >= 0."""
    n = len(f)
    M = np.max(np.convolve(f, f))
    A = build_A(f)

    b = np.full(2*n-1, M)

def fft_convolve(a, b):
    """FFT-based convolution."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n]

LP-based iterative optimization to minimize 2n*max(f*f)/(sum(f))^2.
Uses the approach: fix f, solve LP for best g with f*g <= M, blend, repeat.
Combined with peak-directed local search for polishing.
"""

import numpy as np
from scipy.sparse import diags as sparse_diags
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute the objective score 2n*max(f*f)/(sum(f))^2."""
    n = len(f)
    s = np.sum(f)
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * np.max(conv) / (s ** 2)


def build_A(f):
    """Build sparse (2n-1) x n matrix A where (A@g)_k = (f*g)_k."""
    n = len(f)
    diag_lists = [np.full(n, f[d]) for d in range(n)]
    offsets = list(range(0, -n, -1))
    A = sparse_diags(diag_lists, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def solve_lp(f, time_limit=30):
    """Solve LP: max sum(g) s.t. A_f @ g <= M*1, g >= 0. Returns (g or None, M)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    A = build_A(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': time_limit})

    if res.status == 0:
        return res.x, M
    return None, M


def lp_iterate(f, step_size, time_limit=30):
    """One LP-based iteration. Returns (new_f, M, success)."""
    g0, M = solve_lp(f, time_limit=time_limit)
    if g0 is None:
        return f, M, False

    sum_g0 = float(np.sum(g0))
    if sum_g0 < 1e-15:
        return f, M, False

    g = g0 / sum_g0
    f_new = (1.0 - step_size) * f + step_size * g
    f_new = np.maximum(f_new, 0.0)
    s = np.sum(f_new)
    if s < 1e-15:
        return f, M, False
    f_new /= s
    return f_new, M, True
