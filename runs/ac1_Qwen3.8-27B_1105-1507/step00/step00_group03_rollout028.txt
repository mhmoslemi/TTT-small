import cvxpy as cp

def solve_lp_direction(f, M):
    """Find the LP direction g that maximizes sum(g) subject to f*g <= M and g >= 0."""
    n = len(f)
    g = cp.Variable(n)
    
    # Build the constraint matrix for f * g <= M
    # (f * g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
    # We need to express this as a matrix-vector product: (Fg)[k] <= M
    
    # For efficiency, we'll build the sparse matrix F
    # F[k, m] = f[m] if 0 <= k-m <= n-1, else 0
    # This is a "sliding window" matrix
    
    # Build using scipy sparse
    import scipy.sparse as sp
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        for m in range(max(0, k-n+1), min(n, k+1)):
            if 0 <= k - m and k - m < n:
                rows.append(k)
                cols.append(m)
                vals.append(f[m])
    
    F = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [F @ g <= M, g >= 0]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLOP)  # or cp.CBC
    
    return g.value

        b_shifted = b - b_max
        weights = np.exp(beta * b_shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.ones_like(weights) / len(weights)
        else:
            weights /= w_sum

        W = weights[w_idx_matrix]
        g = 2.0 * (W @ a_curr)

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            return a_curr, b_max, 0.0
        step = T / g_norm
        a_new = a_curr - step * g
        a_new = a_new + T * 0.005 * rng.standard_normal(n)
        a_new = simplex_project(a_new)
        return a_new, b_max, step

    def do_lp_step(a_curr):
        """One LP-based direction step."""
        b = fast_convolve(a_curr)
        M = b.max()
        g = solve_lp_direction(a_curr, M)
        if g is None or g.sum() < a_curr.sum() * 1.0005:
            return None
        # Normalize to same L1 as a_curr
        g_norm_l1 = g.sum()
        if g_norm_l1 < 1e-15:
            return None
        g = g / g_norm_l1 * a_curr.sum()
        # Line search
        best_t = 0.0
        best_s = compute_score(a_curr)
        best_a_try = a_curr.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = simplex_project((1 - t) * a_curr + t * g)
            s_try = compute_score(a_try)
            if s_try < best_s:
                best_s = s_try
                best_t = t
                best_a_try = a_try.copy()
        if best_t > 0:
            return best_a_try
        return None

    def random_restart():
        """Generate a random restart candidate."""
        choice = rng.integers(0, 6)
        if choice == 0:
            # Perturbed best
            a_r = best_a + 0.15 * rng.standard_normal(n)
        elif choice == 1:
            # Symmetric perturbation
            a_r = best_a.copy()
            a_r += 0.08 * rng.standard_normal(n)
            a_r = 0.5 * (a_r + a_r[::-1])
        elif choice == 2:
            # Gaussian bump profile
            center = np.arange(n)
            sigma = n / (4.0 + 4.0 * rng.random())
            a_r = np.exp(-0.5 * ((center - n / 2.0) / sigma) ** 2)
            a_r += 0.05 * rng.standard_normal(n)
        elif choice == 3:
            # Two-peak
            a_r = np.zeros(n)
            p1 = int(n * rng.uniform(0.15, 0.4))
            p2 = int(n * rng.uniform(0.6, 0.85))
            w = int(n * rng.uniform(0.05, 0.15))
            for center in [p1, p2]:
                lo = max(0, center - w)
                hi = min(n, center + w)
                a_r[lo:hi] = np.exp(-0.5 * ((np.arange(lo, hi) - center) / (w / 3.0)) ** 2)
            a_r += 0.02 * rng.standard_normal(n)
        elif choice == 4:
            # Cosinusoidal
            freq = rng.uniform(1.0, 5.0)
            a_r = 1.0 + 0.8 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
            a_r += 0.05 * rng.standard_normal(n)
        else:
            # Hybrid: mix best with random
            a_r = 0.5 * best_a + 0.5 * np.abs(rng.standard_normal(n))
        a_r = np.maximum(a_r, 0)
        return simplex_project(a_r)

    while time.monotonic() < deadline:
        it += 1

        # LP step periodically
        if it % lp_every == 1:
            a_lp = do_lp_step(a)
            if a_lp is not None:
                score = compute_score(a_lp)
                if score < best_score:
                    best_score = score
                    best_a = a_lp.copy()
                    no_improve = 0
                    a = a_lp.copy()
                    print(f"  LP it={it}: score={best_score:.6f}")
                else:
                    no_improve += lp_every
                T = max(T_min, T * T_cool)
                continue

        # Gradient step
        a_new, b_max, step = do_gradient_step(a)
        if a_new is not None and a_new is not a:
            new_score = compute_score(a_new)
            curr_score = compute_score(a)

            if new_score < curr_score:
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    no_improve = 0
                    if it % 50 == 0:
                        print(f"  Grad it={it}: score={best_score:.6f}")
                else:
                    no_improve += 1
            else:
                # Boltzmann acceptance
                delta = new_score - curr_score
                if delta < T * 100:
                    accept_prob = np.exp(-delta / max(T, 1e-12))
                else:
                    accept_prob = 0.0
                if rng.random() < accept_prob:
                    a = a_new
                no_improve += 1

        # Cool
        T = max(T_min, T * T_cool)

        # Restart if stuck
        if no_improve > max_no_improve:
            a = random_restart()
            s_r = compute_score(a)
            if s_r < best_score:
                best_score = s_r
                best_a = a.copy()
            no_improve = 0
            T = 0.02
            print(f"  Restart it={it}: T reset, best={best_score:.6f}")

    # Final: try a few more LP refinements from best
    remaining = deadline - time.monotonic()
    if remaining > 5:
        for _ in range(min(5, int(remaining) // 3)):
            a_try = do_lp_step(best_a)
            if a_try is not None:
                s = compute_score(a_try)
                if s < best_score:
                    best_score = s
                    best_a = a_try.copy()
                    print(f"  Final LP: score={best_score:.6f}")

    # Validate and return
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:
        best_a = np.ones(n) / n
    return [float(x) for x in best_a]
