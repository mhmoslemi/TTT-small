def solve_lp(f, M, tol=1e-6):
    """
    Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0
    
    conv(f,g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
    = sum_j A[k,j] * g[j]
    
    where A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    """
    n = len(f)
    # Build sparse constraint matrix A (2n-1) x n
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        # For row k, A[k,j] = f[k-j] if 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])

    if res.status == 0:
        return res.x
    else:
        return f  # fallback to current f

from scipy.optimize import linprog
import time

def solve_direction_lp(f, M):
    """Solve the LP to find the best direction g."""
    n = len(f)
    # Build sparse constraint matrix
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Shape: (2n-1) x n
    
    # Use a more efficient construction
    # For each j, the non-zero entries are in rows k = j, j+1, ..., j+n-1
    # A[k, j] = f[k-j] = f[0], f[1], ..., f[n-1] for k = j, j+1, ..., j+n-1
    
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        for i in range(n):
            k = j + i
            rows.append(k)
            cols.append(j)
            vals.append(f[i])

                  options={'presolve': True, 'time_limit': 60})
    
    return res.x if res.status == 0 else None

    """Compute np.convolve(a, b) using FFT."""
    from numpy.fft import fft, ifft
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = fft(a, size)
    fb = fft(b, size)
    return np.real(ifft(fa * fb))[:n]

def iterative_flattening(f, target_peak, n_iter=100):
    n = len(f)
    f = f.copy()
    S = f.sum()
    
    for it in range(n_iter):
        conv_f = np.convolve(f, f)
        peaks = conv_f > target_peak
        
        if not np.any(peaks):
            break
        
        # For each peak position, identify the contributing elements and reduce them
        peak_positions = np.where(peaks)[0]
        reduction = np.zeros(n)
        
        for k in peak_positions[:50]:  # limit to top 50 peaks
            # Contribution to peak k: f[j] * f[k-j] for valid j
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    # The "pressure" at position j from this peak
                    reduction[j] += f[idx] * (conv_f[k] - target_peak)
        
        # Reduce f proportionally to the reduction pressure
        if reduction.max() > 0:
            f -= target_peak * 0.01 * reduction / reduction.max()
            f = np.clip(f, 0, 1000)
            # Rescale to maintain sum
            if f.sum() > 0:
                f = f * (S / f.sum())
    
    return f

def fast_convolve_full(a, b):
    """Compute full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n]
