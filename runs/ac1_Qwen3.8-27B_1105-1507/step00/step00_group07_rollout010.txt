"""
Hybrid LP + local search optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses iterative LP (maximize sum(g) s.t. f*g <= M) for moderate-length sequences,
combined with targeted local search refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def convolve_fft(a):
    """Full linear convolution using FFT for efficiency."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    return conv


def project_onto_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def precompute_conv_structure(n):
    """Precompute sparse (row, col) indices for the convolution constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int64), np.array(cols, dtype=np.int64)


def solve_lp_step(f, M, rows, cols, n, time_limit=30.0):
    """Solve one LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Returns g or None."""
    f_idx = rows - cols
    vals = f[f_idx]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(
            c, A_ub=A, b_ub=b_ub, bounds=bounds,
            method='highs',
            options={'time_limit': time_limit, 'presolve': True}
        )
        if result.success and result.x is not None:
            return result.x
    except Exception:
        pass
    return None


def optimize_lp(n, time_budget, seed=42, max_iter=300):
    """
    LP-based iterative optimization for fixed sequence length n.
    Minimizes max(convolve(f,f)) subject to sum(f) = n, f >= 0.
    Returns (best_f, best_score).
    """
    t0 = time.time()
    deadline = t0 + time_budget

    rows, cols = precompute_conv_structure(n)
    f_idx = rows - cols

    # Initialize with uniform sequence
    f = np.ones(n, dtype=np.float64)
    conv = convolve_fft(f)
    M = float(conv.max())
    score = 2.0 * n * M / (f.sum() ** 2)
    best_f = f.copy()
    best_score = score
    print(f"    n={n}: initial score={score:.6f}, M={M:.6f}")

    step_size = 0.25
    no_improve = 0
    prev_M = M

    for it in range(max_iter):
        if time.time() > deadline - 5:
            break

        remaining = deadline - time.time()
        lp_time = min(45.0, remaining * 0.4)
        if lp_time < 3.0:
            break

        g = solve_lp_step(f, M, rows, cols, n, time_limit=lp_time)
        if g is None:
            break

        sum_g = g.sum()
        if sum_g < 1e-10:
            break

        # Rescale g to match f's sum
        target_sum = f.sum()
        g = g * (target_sum / sum_g)

        # Line search: find best step
        best_t = 0.0
        best_f_try = f.copy()
        best_M_try = M
        candidate_its = [step_size, step_size * 0.7, step_size * 0.5, step_size * 0.3, step_size * 0.1]

        for t in candidate_its:
            f_try = (1.0 - t) * f + t * g
            conv_try = convolve_fft(f_try)
            M_try = float(conv_try.max())
            if M_try < M:
                best_t = t
                best_f_try = f_try
                best_M_try = M_try
                break
            elif M_try <= M * 1.0001:
                best_t = t
                best_f_try = f_try
                best_M_try = M_try

        if best_t == 0.0:
            # Fallback: tiny step
            best_t = 0.005
            best_f_try = (1.0 - best_t) * f + best_t * g
            best_M_try = float(convolve_fft(best_f_try).max())

        f = best_f_try
        M = best_M_try
        score = 2.0 * n * M / (f.sum() ** 2)

        if score < best_score - 1e-12:
            best_score = score
            best_f = f.copy()
            no_improve = 0
            if it % 10 == 0 or score < best_score + 1e-9:
                print(f"    n={n} iter {it}: M={M:.8f}, score={score:.6f} *")
        else:
            no_improve += 1

        if it % 20 == 0:
            print(f"    n={n} iter {it}: M={M:.8f}, score={score:.6f}")

        # Adaptive step size
        if no_improve > 20:
            step_size *= 0.85
            no_improve = 0
            if step_size < 0.005:
                break

        # Convergence check
        if abs(prev_M - M) < 1e-9 * M and no_improve > 10:
            break
        prev_M = M

    elapsed = time.time() - t0
    print(f"    n={n}: done. best score={best_score:.6f}, elapsed={elapsed:.1f}s")
    return best_f, best_score


def subgradient_optimize(n, time_budget, seed=42, max_iter=50000):
    """
    Subgradient-based optimization as a faster alternative to LP.
    Minimizes max(conv(f,f)) subject to sum(f)=n, f>=0.
    """
    t0 = time.time()
    deadline = t0 + time_budget
    rng = np.random.default_rng(seed)

    f = np.ones(n, dtype=np.float64)
    conv = convolve_fft(f)
    M = float(conv.max())
    best_f = f.copy()
    best_score = 2.0 * n * M / (f.sum() ** 2)

    eta = 1.0 / n
    moment = 0.0

    for it in range(max_iter):
        if time.time() > deadline - 2:
            break
        if it % 10000 == 0 and it > 0:
            score = 2.0 * n * M / (f.sum() ** 2)
            print(f"    subgrad n={n} iter {it}: M={M:.8f}, score={score:.6f}")

        # Find subgradient at argmax
        conv = convolve_fft(f)
        M = float(conv.max())
        k_star = int(np.argmax(conv))

        # Subgradient: sg[i] = 2*f[k_star - i] if 0 <= k_star - i < n
        sg = np.zeros(n)
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        if i_max >= i_min:
            indices = np.arange(i_min, i_max + 1)
            sg[indices] = 2.0 * f[k_star - indices]

        # Nesterov-like averaging with decreasing step
        moment = it / (it + 1.0)
        eta_t = 1.0 / (np.sqrt(it + 1.0) * n * 0.1 + 1e-10)

        f_new = f - eta_t * sg
        f_new = project_onto_simplex(f_new) * n  # rescale to sum = n

        M_new = float(convolve_fft(f_new).max())
        if M_new < M:
            f = f_new
            M = M_new
            score = 2.0 * n * M / (f.sum() ** 2)
            if score < best_score:
                best_score = score
                best_f = f.copy()

    return best_f, best_score


def local_search(f, budget_s, seed=42):
    """Targeted local search using evaluate_sequence for final refinement."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s
    n = len(f)
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    if not np.isfinite(best_score):
        best = np.ones(n)
        best_score = float(evaluate_sequence(best.tolist()))

    iter_count = 0
    while time.time() < deadline - 3:
        candidate = best.copy()
        ptype = rng.random()

        if ptype < 0.4:
            # Block multiplicative perturbation
            width = int(rng.integers(max(2, n // 100), max(3, n // 10)))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.08, width))
        elif ptype < 0.6:
            # Pair swap (move mass from high to low)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and best[i] > 1e-10 and best[j] > 1e-10:
                amt = best[i] * rng.uniform(0.05, 0.3)
                candidate[i] -= amt
                candidate[j] += amt
        elif ptype < 0.8:
            # Symmetrize a portion
            half = n // 2
            width = int(rng.integers(half // 10, half // 2))
            start = int(rng.integers(0, max(1, half - width)))
            seg = candidate[start:start + width].copy()
            candidate[start:start + width] = 0.5 * (seg + seg[::-1])
        else:
            # Gaussian smoothing
            candidate = np.convolve(candidate, np.ones(5) / 5, mode='same')

        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        iter_count += 1

        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            if iter_count % 50 == 0:
                print(f"    local search: iter {iter_count}, score={best_score:.6f}")
