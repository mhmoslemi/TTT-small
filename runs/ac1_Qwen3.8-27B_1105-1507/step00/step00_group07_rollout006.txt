"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using power-ramp shapes (i/n)^p
with numpy-broadcast convolution and optional geometric random walk refinement.
Key insight: one-sided increasing sequences shift the convolution peak to pos n-1,
giving score ~ 2(p+1)^2*B(p+1,p+1) which decreases rapidly with p.
"""

import numpy as np
import time
import math


def fast_conv1d(x):
    """Full convolution of 1D array with itself using broadcasting (O(n^2) time, O(n) extra)."""
    n = x.shape[0]
    r = x.reshape(-1, 1)
    c = x.reshape(1, -1)
    res = np.zeros(2 * n - 1, dtype=np.float64)
    # Upper triangle indices: row+i, col+i where i=0..n-1, row+col = d
    for d in range(2 * n - 1):
        if d < n:
            i_start = 0
            i_end = d + 1
        else:
            i_start = d - n + 1
            i_end = n
        res[d] = np.dot(x[i_start:i_end], x[d - i_end + 1:d - i_start + 1])
    return res


def fast_conv1d_multi(x, y):
    """Cross-convolution of two 1D arrays using numpy broadcasting."""
    n1 = x.shape[0]
    n2 = y.shape[0]
    res = np.zeros(n1 + n2 - 1, dtype=np.float64)
    for d in range(n1 + n2 - 1):
        i_start = max(0, d - n2 + 1)
        i_end = min(n1, d + 1)
        if i_start < i_end:
            res[d] = np.dot(x[i_start:i_end], y[d - i_end + 1:d - i_start + 1])
    return res


def compute_score_np(a):
    """Compute the evaluation score using numpy (commit hook)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def make_power_ramp(n, p, normalize_to=100.0):
    """Create power ramp (i+1)^p normalized so max = normalize_to."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    if a[-1] > 0:
        a = a * (normalize_to / a[-1])
    return a


def try_small_n_brute(n, n_samples=30000, seed=0):
    """Random search for small n to find good configurations."""
    rng = np.random.default_rng(seed + n)
    best_score = float('inf')
    best_seq = None
    for _ in range(n_samples):
        # Mix of strategies
        r = rng.random()
        if r < 0.4:
            # Power ramp
            p = rng.uniform(0.5, min(8.0, 6.9 / max(0.5, math.log(n))))
            a = make_power_ramp(n, p, normalize_to=100.0)
        elif r < 0.7:
            # Random with exponential tilt
            a = np.exp(rng.uniform(-3, 3, size=n))
            a = np.maximum(a, 0.01)
        elif r < 0.9:
            # Two-segment: ramp then drop
            split = rng.integers(1, n)
            a1 = np.arange(1, split + 1, dtype=np.float64) ** rng.uniform(0.5, 3)
            a2 = np.zeros(n - split) if split < n else np.array([])
            a = np.concatenate([a1, a2]) if n > split else a1
        else:
            a = rng.random(n) ** rng.uniform(1, 5) + 0.01
        a = np.clip(a, 0, 100.0)
        if np.sum(a) < 0.01:
            continue
        s = compute_score_np(a)
        if s < best_score:
            best_score = s
            best_seq = a.copy()
    return best_score, best_seq


def geometric_refine(a, budget_s, seed=42):
    """Refine using geometric random walk in log-space with multi-step evaluation."""
    rng = np.random.default_rng(seed)
    n = a.shape[0]
    log_a = np.log(np.maximum(a, 1e-10))
    best_log = log_a.copy()
    best_score = compute_score_np(a)
    deadline = time.monotonic() + budget_s
    step_sizes = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
    step_idx = 0
    
    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 2.0:
            break
        # Decrease step size over time
        elapsed = budget_s - (deadline - time.monotonic())
        progress = min(1.0, elapsed / budget_s)
        step_idx = min(int(progress * len(step_sizes)), len(step_sizes) - 1)
        step = step_sizes[step_idx]
        
        # Block-wise perturbation
        block_size = max(1, int(n * rng.uniform(0.01, 0.1)))
        start = int(rng.integers(0, n - block_size + 1))
        end = min(n, start + block_size)
        
        candidate_log = best_log.copy()
        candidate_log[start:end] += rng.normal(0, step, size=end - start)
        candidate = np.clip(np.exp(candidate_log), 0, 100.0)
        candidate = np.maximum(candidate, 0)
        
        # Also try a global small perturbation occasionally
        if rng.random() < 0.2:
            candidate_log = best_log + rng.normal(0, step * 0.5, size=n)
            candidate = np.clip(np.exp(candidate_log), 0, 100.0)
        
        if np.sum(candidate) < 0.01:
            continue
        
        s = compute_score_np(candidate)
        if s < best_score:
            best_score = s
            best_log = candidate_log.copy()
            a = candidate
    
    return np.clip(np.exp(best_log), 0, 100.0), best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for optimal step function:
    1. Power ramp shapes (i/n)^p for various n and p
    2. Small-n brute force
    3. Geometric random walk refinement
    """
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(seed)
    
    best_score = float('inf')
    best_seq = np.ones(100, dtype=np.float64)
    
    # Phase 1: Power ramps (fast, O(n^2) per eval with numpy)
    print("Phase 1: Power ramp search")
    phase1_deadline = time.monotonic() + 120.0
    
    # Try various n values
    n_candidates = [2, 3, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 4000, 6733]
    
    for n in n_candidates:
        if time.monotonic() > phase1_deadline:
            break
        # Find optimal p for this n
        p_grid = np.linspace(0.3, min(15.0, 8.0 / max(0.5, math.log(max(2, n)))), 200)
        for p in p_grid:
            if time.monotonic() > phase1_deadline:
                break
            a = make_power_ramp(n, p, normalize_to=100.0)
            if np.sum(a) < 0.01:
                continue
            s = compute_score_np(a)
            if s < best_score:
                best_score = s
                best_seq = a.copy()
                if best_score < 1.0:
                    print(f"  n={n}, p={p:.3f}: score={s:.6f}")
    
    # Finer p search around best
    print(f"  Best after coarse: score={best_score:.6f}, n={len(best_seq)}")
    
    if best_score < float('inf'):
        # Refine p for the best n
        n_best = len(best_seq)
        p_center = None
        # Re-derive p from best_seq
        if n_best > 1 and best_seq[-1] > 0 and best_seq[0] > 0:
            ratio = best_seq[-1] / max(best_seq[0], 1e-10)
            p_center = math.log(ratio) / math.log(n_best)
        if p_center is not None:
            p_fine = np.linspace(max(0.1, p_center - 1), p_center + 1, 200)
            for p in p_fine:
                a = make_power_ramp(n_best, p, normalize_to=100.0)
                s = compute_score_np(a)
                if s < best_score:
                    best_score = s
                    best_seq = a.copy()
            print(f"  After fine p: score={best_score:.6f}")
    
    # Also try modified ramps: (i+1)^p with some smoothing
    if time.monotonic() < total_deadline:
        print("Phase 1b: Modified ramps")
        modified_deadline = min(total_deadline, time.monotonic() + 60.0)
        base_n = len(best_seq)
        for n in [max(2, base_n // 2), base_n, min(10000, base_n * 2)]:
            if time.monotonic() > modified_deadline:
                break
            for p in np.linspace(0.5, min(10, 7.0 / max(0.5, math.log(n))), 50):
                if time.monotonic() > modified_deadline:
                    break
                x = np.arange(1, n + 1, dtype=np.float64)
                # Try (i+1)^p * exp(-alpha*i/n) variants
                for alpha in [0, 0.5, 1.0]:
                    a = (x ** p) * np.exp(-alpha * np.arange(n, dtype=np.float64) / n)
                    if a[-1] > 0:
                        a = a * (100.0 / a[-1])
                    a = np.clip(a, 0, 100.0)
                    if np.sum(a) < 0.01:
                        continue
                    s = compute_score_np(a)
                    if s < best_score:
                        best_score = s
                        best_seq = a.copy()
    
    # Phase 2: Small-n random search
    if time.monotonic() < total_deadline:
        print("Phase 2: Small-n random search")
        phase2_deadline = min(total_deadline, time.monotonic() + 60.0)
        for n in [2, 3, 4, 5, 6, 7, 8, 10, 12, 15, 20]:
            if time.monotonic() > phase2_deadline:
                break
            s, seq = try_small_n_brute(n, n_samples=5000, seed=seed + n * 100)
            if s < best_score:
                best_score = s
                best_seq = seq.copy()
                print(f"  n={n}: score={s:.6f}")
    
    # Phase 3: Geometric random walk refinement
    if time.monotonic() < total_deadline:
        remaining = total_deadline - time.monotonic()
        print(f"Phase 3: Geometric refinement (remaining {remaining:.0f}s)")
        refine_budget = remaining * 0.7
        best_seq, best_score = geometric_refine(best_seq, refine_budget, seed=seed)
        print(f"  After refinement: score={best_score:.6f}")
        
        # Try refining with a different starting point
        if time.monotonic() < total_deadline:
            remaining2 = total_deadline - time.monotonic()
            if remaining2 > 10:
                # Start fresh from power ramp
                n_r = len(best_seq)
                p_r = 2.0
                a_r = make_power_ramp(n_r, p_r, normalize_to=100.0)
                s_r = compute_score_np(a_r)
                if s_r < best_score:
                    best_score = s_r
                    best_seq = a_r.copy()
                best_seq, best_score = geometric_refine(best_seq, remaining2 * 0.5, seed=seed + 999)
                print(f"  After 2nd refinement: score={best_score:.6f}")
    
    # Final verification with exact evaluate_sequence
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(final_list)
    print(f"Final score (exact): {final_score}")
    
    if not np.isfinite(final_score):
        # Fallback: simple power ramp
        n_fb = 100
        p_fb = 2.0
        final_list = [float(100.0 * ((i + 1) / n_fb) ** p_fb) for i in range(n_fb)]
        final_score = evaluate_sequence(final_list)
        print(f"Fallback score: {final_score}")
    
    return final_list
