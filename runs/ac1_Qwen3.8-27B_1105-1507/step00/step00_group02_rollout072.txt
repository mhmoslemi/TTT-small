import time

"""
Minimizes the evaluation function using FFT-based gradient descent on the
sum-of-squares proxy objective with momentum, multiple initializations,
and adaptive step sizes. Targets score < 1.5027.
"""

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if not np.any(v > 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def make_N(n):
    """Next power of 2 >= 2n-1."""
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    return N


def fft_sum_sq(p, N):
    """Compute sum of (p*p)[k]^2 using FFT."""
    n = len(p)
    p_pad = np.zeros(N)
    p_pad[:n] = p
    P = np.fft.rfft(p_pad)
    conv = np.fft.irfft(P * P, n=N)[:2 * n - 1]

def fft_grad(p, N):
    """Compute gradient of sum((p*p)[k]^2) w.r.t. p using FFT."""
    n = len(p)
    p_pad = np.zeros(N)
    p_pad[:n] = p
    P = np.fft.rfft(p_pad)
    mag2 = P.real * P.real + P.imag * P.imag
    grad = 4.0 * np.fft.irfft(P * mag2, n=N)[:n]

import numpy as np
import time

"""
Minimizes the evaluation function 2n*max(conv(p,p))/sum(p)^2 using FFT-based
gradient descent on the sum-of-squares proxy objective with momentum, multiple
structured initializations (U-shaped, random, given), and Polyak step sizes.
"""


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if not np.any(v > 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def make_N(n):
    """Next power of 2 >= 2n-1."""
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    return N


def fft_sum_sq(p, N):
    """Compute sum of (p*p)[k]^2 using FFT (circular with zero-padding)."""
    n = len(p)
    p_pad = np.zeros(N)
    p_pad[:n] = p
    P = np.fft.rfft(p_pad)
    conv = np.fft.irfft(P * P, n=N)[:2 * n - 1]
    return np.sum(conv * conv)


def fft_grad(p, N):
    """Gradient of sum((p*p)[k]^2) w.r.t. p, computed via FFT in O(n log n)."""
    n = len(p)
    p_pad = np.zeros(N)
    p_pad[:n] = p
    P = np.fft.rfft(p_pad)
    mag2 = P.real * P.real + P.imag * P.imag
    grad = 4.0 * np.fft.irfft(P * mag2, n=N)[:n]
    return grad


def run_descent(p_init, N, deadline, label=""):
    """
    Run FFT gradient descent with momentum and Polyak step sizes.
    Returns (best_p, best_score) where p is on the probability simplex.
    """
    n = len(p_init)
    p = project_simplex(np.asarray(p_init, dtype=np.float64).copy())
    if p.sum() < 0.01 or not np.all(np.isfinite(p)):
        p = np.ones(n) / n

    best_p = p.copy()
    best_score = float(evaluate_sequence(best_p.tolist()))
    J_lower = 1.0 / (2 * n - 1)

    v = np.zeros(n)
    momentum = 0.0
    eta = 1.0 / n
    it = 0
    stall = 0
    prev_J = fft_sum_sq(p, N)

    print(f"  [{label}] start: score={best_score:.6f}", flush=True)

    while time.monotonic() < deadline:
        # Compute gradient
        grad = fft_grad(p, N)
        gnorm2 = np.dot(grad, grad)

        if gnorm2 < 1e-20:
            break

        # Polyak step size
        J = fft_sum_sq(p, N)
        if J <= J_lower + 1e-12:
            break
        eta_polyak = (J - J_lower) / gnorm2
        eta = min(eta_polyak, 5.0 / n)

        # Animate momentum: start low, increase
        if it < 500:
            momentum = 0.3 * (it / 500.0)
        elif it < 5000:
            momentum = 0.3 + 0.4 * min(1.0, (it - 500) / 4500.0)
        else:
            momentum = 0.7

        # Momentum update + projection
        v = momentum * v - eta * grad
        p_new = p + v
        p_new = project_simplex(p_new)

        if not np.all(np.isfinite(p_new)) or p_new.sum() < 0.01:
            eta *= 0.1
            v *= 0.5
            continue

        p = p_new
        it += 1

        # Track convergence of proxy objective
        if it % 50 == 0:
            J_new = fft_sum_sq(p, N)
            if J_new > prev_J * 1.01:
                # Objective increasing; reduce step
                eta *= 0.5
                momentum *= 0.5
                v *= 0.5
            else:
                prev_J = J_new

        # Periodic evaluation of actual objective
        if it % 200 == 0:
            score = float(evaluate_sequence(p.tolist()))
            if np.isfinite(score) and score < best_score:
                best_p = p.copy()
                best_score = score
                stall = 0
                if it % 2000 == 0:
                    print(f"  [{label}] it={it} score={best_score:.6f} "
                          f"J={J_new:.8f} eta={eta:.2e}", flush=True)
            else:
                stall += 1

            # Restart if stalled
            if stall > 40:
                break

    # Final evaluation
    score = float(evaluate_sequence(p.tolist()))
    if np.isfinite(score) and score < best_score:
        best_p = p.copy()
        best_score = score

    print(f"  [{label}] done: score={best_score:.6f} iters={it}", flush=True)
    return best_p, best_score
