2. Use a projected subgradient method with careful step size selection
3. Periodically use an LP to get a better direction

- Update: a = max(0, a - lr*s): O(n)
- Normalize: O(n)

from scipy.optimize import linprog
from scipy import.sparse
import time

def make_gaussian(n, sigma_frac=0.25):
    """Create a Gaussian-shaped sequence of length n, normalized to sum=1."""
    center = (n-1)/2
    sigma = sigma_frac * n
    i = np.arange(n)
    a = np.exp(-(i-center)**2 / (2*sigma**2))
    a = a / np.sum(a)
    return a

def make_triangular(n):
    """Create a triangular (tent) sequence of length n, normalized to sum=1."""
    i = np.arange(n)
    center = (n-1)/2
    a = np.maximum(0, 1 - np.abs(i-center)/(center+1))
    if np.sum(a) < 1e-10:
        a = np.ones(n)
    a = a / np.sum(a)
    return a

def make_uniform(n):
    return np.ones(n) / n

def make_cosine(n):
    i = np.arange(n)
    a = np.cos(np.pi * i / n)**2 + 0.1
    a = a / np.sum(a)
    return a

def fast_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2*n - 1
    # Next power of 2 for efficient FFT
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)[:size]
    return c

def subgradient_loop(a, deadline, alpha=1.0, use_multiple_lags=3):
    """Run subgradient descent to minimize max(conv(a,a)) with sum(a)=1, a>=0."""
    n = len(a)
    a = a.copy()
    a = np.maximum(0, a)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_local = a.copy()
    best_local_score = 2 * n * np.max(fast_conv(a))
    no_improve_count = 0
    lr = 0.1
    
    step_count = 0
    while time.monotonic() < deadline:
        c = fast_conv(a)
        max_c = np.max(c)
        score = 2 * n * max_c
        
        if score < best_local_score:
            best_local_score = score
            best_local = a.copy()
            no_improve_count = 0
            # Increase LR slightly
            lr = min(lr * 1.05, 1.0)
        else:
            no_improve_count += 1
            if no_improve_count % 1000 == 0:
                lr *= 0.95
        
        # Find dominant lags
        # Use top-k lags for smoother gradient
        if use_multiple_lags > 1:
            # Get indices of top lags
            k_lags = np.argsort(c)[-use_multiple_lags:]
        else:
            k_lags = [np.argmax(c)]
        
        # Compute subgradient as weighted combination
        s_grad = np.zeros(n)
        for k in k_lags:
            # c[k] = sum_j a[j] * a[k-j] for valid j
            # d(c[k])/d(a[j]) = a[k-j] if 0 <= k-j < n
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                indices = np.arange(j_start, j_end + 1)
                s_grad[indices] += a[k - indices]
        
        # Update
        s_norm_sq = np.sum(s_grad * s_grad)
        if s_norm_sq < 1e-20:
            break
        step = lr * max_c / (s_norm_sq + 1e-10)
        a_new = a - step * s_grad
        a_new = np.maximum(0, a_new)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a = a.copy()  # Revert
            lr *= 0.5
            continue
        a = a_new / s_new
        step_count += 1
    
    return best_local, best_local_score
