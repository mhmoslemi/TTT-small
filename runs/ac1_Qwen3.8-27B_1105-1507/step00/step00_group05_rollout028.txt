```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix
import time

def build_conv_matrix(f, n):
    """Build sparse matrix A where (A @ g)[k] = sum_j f[k-j] * g[j] (convolution)."""
    # Column j has f[0], f[1], ..., f[n-1] at rows j, j+1, ..., j+n-1
    # Total: n columns, each with n entries
    # Shape: (2n-1, n)
    
    # Efficient construction using numpy
    data = np.tile(f, n)  # n*n elements
    rows = np.zeros(n*n, dtype=np.int32)
    cols = np.zeros(n*n, dtype=np.int32)
    
    for j in range(n):
        start = j * n
        rows[start:start+n] = j + np.arange(n)
        cols[start:start+n] = j
    
    return coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
```

        return f, M, 0, False
    
    t0 = time.monotonic()
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    t1 = time.monotonic()
    lp_time = t1 - t0
    
    if res.status != 0 or res.x is None:
        return f, M, lp_time, False
    
    g = res.x
    g_sum = g.sum()
    if g_sum < 1e-10:
        return f, M, lp_time, False
    
    f_sum = f.sum()
    g = g * (f_sum / g_sum)
    f_new = (1 - alpha) * f + alpha * g
    f_new = np.clip(f_new, 0, 1000)
    
    return f_new,
