def make_binomial_sequence(n):
    """Generate a[k] = C(2k,k)/4^k for k=0,...,n-1 using recursion."""
    a = np.ones(n, dtype=np.float64)
    for k in range(1, n):
        a[k] = a[k-1] * (2*k - 1) / (2*k)
    return a

def compute_conv(a):
    """Compute full linear convolution of a with itself."""
    return np.convolve(a, a)

def compute_score(a):
    """Compute the exact score: 2*n*max(conv)/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (s * s)

def smooth_objective(a, t):
    """Smooth approximation of the score using log-sum-exp."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return 1e10
    conv = compute_conv(a)
    tx = t * conv
    m = np.max(tx)
    lse = m + np.log(np.sum(np.exp(tx - m)))
    return lse / t - 2.0 * np.log(s)

def smooth_gradient(a, t):
    """Gradient of the smooth objective."""
    n = len(a)
    s = np.sum(a)
    conv = compute_conv(a)
    tx = t * conv
    m = np.max(tx)

    g = np.convolve(a[::-1], p, mode='full')[n-1:2*n-1]
    grad = 2.0 * g - 2.0 / s
    return grad

        # Row k, column j: a[k-j] if 0 <= k-j < n, else 0
        rows = []
        cols = []
        vals = []
        for k in range(2*n - 1):
            for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
                i = k - j
                if 0 <= i < n:
                    rows.append(k)
                    cols.append(j)
                    vals.append(a[i])

        a = np.maximum(a, 0)
