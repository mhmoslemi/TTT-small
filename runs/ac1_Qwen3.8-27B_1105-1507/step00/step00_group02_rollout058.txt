The objective (with sum(a) = n fixed) is: F(a) = max_k (a*a)[k]
The subgradient of F at a, evaluated at component i, is:

    top_k = np.argsort(r)[-K:][::-1]

        i_max = min(n-1, k)
        j = k - np.arange(i_min, i_max+1)
        pressure[i_min:i_max+1] += a[j]

    # a_new = a - alpha * pressure + lambda
    # sum(a_new) = n => lambda = (n - sum(a) + alpha * sum(pressure)) / n
    # Wait: sum(a_new) = sum(a) - alpha*sum(pressure) + n*lambda = n
    # => lambda = (n - sum(a) + alpha*sum(pressure)) / n
    # But sum(a) should already be n if we maintain it
    lambda_val = alpha * pressure.sum() / n
    a_new = a - alpha * pressure + lambda_val
    a_new = np.maximum(a_new, 0)
    # Rescale to fix sum (clipping may have changed it)
    s = a_new.sum()
    if s > 0:
        a_new *= n / s
    return a_new

def _fft_conv_square(a):
    """Compute a * a (full linear convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]

def _normalize_inplace(a, n):
    """Normalize a in-place to sum=n, non-negative."""
    np.maximum(a, 0, out=a)
    s = a.sum()
    if s < 1e-10:
        a[:] = 1.0
        return
    a *= (n / s)

def _sor_step_fft(a, alpha, K, r):
    """One SOR step, reusing the precomputed r."""
    n = len(a)
    if K <= 1:
        k_star = int(np.argmax(r))
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        j = k_star - np.arange(i_min, i_max + 1)
        # pressure at positions i_min..i_max is a[j]
        a[i_min:i_max+1] -= alpha * a[j]
    else:
        top_k = np.argpartition(r, -K)[-K:]
        for k in top_k:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            j = k - np.arange(i_min, i_max + 1)

    s = a.sum()

    np.maximum(a, 0, out=a)
