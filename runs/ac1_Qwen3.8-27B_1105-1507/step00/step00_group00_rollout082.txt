```python
from scipy.optimize import minimize

def score_n(n):
    # Minimize max(f*f) subject to sum(f)=1, f>=0
    # Variables: f[0..n-1], t
    # Objective: t
    # Constraints: (f*f)[k] <= t for k=0..2n-2, sum(f)=1, f>=0
    
    def objective(vars):
        f = vars[:n]
        t = vars[n]
        return t
    
    def convolve_k(f, k):
        # (f*f)[k] = sum_{i} f[i]*f[k-i] for valid i
        result = 0.0
        for i in range(max(0, k-n+1), min(n, k+1)):
            result += f[i] * f[k-i]
        return result
    
    constraints = []
    for k in range(2*n - 1):
        constraints.append({
            'type': 'ineq',
            'fun': lambda vars, k=k: vars[n] - convolve_k(vars[:n], k)
        })
    constraints.append({
        'type': 'eq',
        'fun': lambda vars: np.sum(vars[:n]) - 1
    })
    
    # Initial guess
    x0 = np.ones(n + 1)
    x0[:n] = 1.0 / n
    x0[n] = 1.0 / n  # initial t
    
    bounds = [(0, None)] * n + [(0, None)]
    
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints)
    
    if result.success:
        f_opt = result.x[:n]
        t_opt = result.x[n]
        score = 2 * n * t_opt
        return f_opt, t_opt, score
    else:
        return None, None, np.inf
```
