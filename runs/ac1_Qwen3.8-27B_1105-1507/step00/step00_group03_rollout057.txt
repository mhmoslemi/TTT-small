    # Use the fact that A is a banded Toeplitz matrix

    total_nnz = sum(min(n, 2*n-1-j) for j in range(n))  # = n^2

        rows[idx:idx+length] = np.arange(j, j+length)
        cols[idx:idx+length] = j
        vals[idx:idx+length] = a[:length]
        idx += length
