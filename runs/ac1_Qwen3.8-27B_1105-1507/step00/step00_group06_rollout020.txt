For the problem of minimizing `max_k b[k]` subject to `sum(a) = 1, a >= 0`:

Let me now put together a complete algorithm:

**Algorithm: Multiplicative Weight Descent with Softmax Peak Selection**

1. Initialize `a` (from parent or random)
2. Normalize: `a = a / sum(a)`
3. For each iteration:
   a. Compute `b = convolve(a, a)` (length 2n-1)
   b. Compute softmax weights: `w_k = exp(b_k / T) / sum(exp(b_k / T))`
   c. Compute gradient: `grad = 2 * correlate(w, a, mode='valid')`
   d. Update: `a *= exp(-eta * grad / max_b)` (normalize by max_b for scale invariance)
   e. Clip to positive, renormalize
4. Evaluate and keep best

The temperature T controls how focused the softmax is. Low T = focus on the maximum peak. High T = spread out.

The step size eta controls convergence speed.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _build_sparsity(n):
    rows, cols = [], []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(k - i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def _norm_pos(a, n):
    a = np.clip(a, 1e-15, 1000.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _score_norm(a, n):
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _mwd_step(a, eta, temp, n):
    b = np.convolve(a, a)
    mb = np.max(b)
    if mb < 1e-15:
        return a
    x = b / temp
    x -= np.max(x)
    w = np.exp(x)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(2 * n - 1)
        w[np.argmax(b)] = 1.0
    else:
        w /= ws
    grad = 2.0 * np.correlate(w, a, mode='valid')
    a_new = a * np.exp(-eta * grad / mb)
    return _norm_pos(a_new, n)


def _mwd_run(a0, n, deadline, save_margin):
    a = a0.copy()
    lb = a.copy()
    lbs = _score_norm(a, n)
    for phase in range(60):
        if time.monotonic() > deadline - save_margin:
            break
        eta = 1.5 / (1.0 + 0.08 * phase)
        temp = max(0.002, 0.04 * (0.82 ** phase))
        for it in range(800):
            if time.monotonic() > deadline - save_margin:
                break
            a = _mwd_step(a, eta, temp, n)
            s = _score_norm(a, n)
            if s < lbs:
                lbs = s
                lb = a.copy()
    return lb, lbs


def _lp_solve(f, rows_p, cols_p, n, time_limit=25):

    if M < 1e-15:
        return None
    vals = f[rows_p.astype(np.int64) - cols_p.astype(np.int64)]
    A = csr_matrix((vals, (rows_p, cols_p)), shape=(2 * n - 1, n))
    res = linprog(
        -np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
        bounds=[(0, None)] * n, method='highs',
        options={'time_limit': time_limit}
    )
    if res.success:
        return res.x
    return None
