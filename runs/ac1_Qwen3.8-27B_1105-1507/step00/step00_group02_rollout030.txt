def _fft_conv2(f, fft_size):
    """Linear convolution of f with itself via FFT."""
    F = np.fft.rfft(f, n=fft_size)
    return np.fft.irfft(F * F, n=fft_size)[:2*len(f)-1]

def _compute_grad_lse(c, f, beta, grad_fft_size):
    """Gradient of LSE(beta*c) w.r.t. f, where c = conv(f,f)."""
    n = len(f)
    c_shifted = beta * (c - c.max())
    A = np.exp(c_shifted)
    s = A.sum()
    if s < 1e-300:
        A = np.full(2*n - 1, 1.0 / (2*n - 1))
    else:
        A /= s
    f_rev = f[::-1]
    A_pad = np.zeros(grad_fft_size)
    A_pad[:2*n-1] = A
    FR = np.fft.rfft(f_rev, n=grad_fft_size)
    g_full = np.fft.irfft(np.fft.rfft(A_pad) * FR, n=grad_fft_size)
    return 2.0 * g_full[n-1 : n-1+n]

    """Gradient of sum(c^2) w.r.t. f, where c = conv(f,f)."""
