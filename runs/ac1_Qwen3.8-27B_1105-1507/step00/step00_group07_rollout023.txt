```python
def project_simplex(x):
    # Find τ such that sum(max(x_i - τ, 0)) = 1
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, len(x)+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)
```

```python
import numpy as np
import time

def fast_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2*n-1]
    return c

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_gradient(a, c, K=50):
    """Compute weighted subgradient using top-K lags."""
    n = len(a)
    top_k = np.argsort(c)[::-1][:K]
    w = c[top_k].copy()
    w = np.maximum(w, 0)
    w = w / w.sum()
    g = np.zeros(n)

        if wk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += 2.0 * wk * a[k-hi:k-lo+1][::-1]
    return g

def score(a):
    """Compute the exact score using the evaluation function."""
    return float(evaluate_sequence(a.tolist()))

def score_fft(a):

    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fast_convolve(a)
    max_b = c.max()
    return 2.0 * n * max_b / (S * S)

def optimize_sequence(a0, budget_s=300.0, seed=42):
    """Optimize a sequence using subgradient method with line search."""
    rng = np.random.default_rng(seed)
    n = len(a0)

    S = a.sum()

        S = n

    eta = 1.0
    K = 50

        c = fast_convolve(a)
        g = compute_gradient(a, c, K)

            a_try = project_simplex(a_try)
            s_try = score_fft(a_try)
            s_curr = score_fft(a)
            if s_try < s_curr * (1 - 1e-5):

            eta *= 0.5

        if True:  # check every iteration (score_fft is fast enough)
```

The max is NOT at the center but at k=1 (near the edge). This is because the first two elements are both large (1.5), so their product creates a large value at k=1.

So the "step" approach doesn't work directly. The issue is that having consecutive large values creates a large convolution value.

The problem is that the "matching" pairs (a[i] with a[n-1-i]) tend to be both large or both small, giving large products.
