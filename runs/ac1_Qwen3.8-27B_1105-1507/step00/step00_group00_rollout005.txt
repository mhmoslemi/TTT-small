import numpy as np
import time
from scipy.fft import rfft, irfft

"""
Algorithm: Smoothed subgradient optimization with temperature annealing for minimizing
the peak-to-average ratio of the autocorrelation of a non-negative sequence.
Uses multiple starting points, adaptive step sizes, and periodic re-optimization.
"""

def _fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    f = 1 << (n - 1).bit_length() if n > 1 else 2
    return irfft(rfft(a, f) * rfft(b, f), f)[:n]


def _compute_grad(a, b, T):
    """
    Gradient of (1/T)*log(sum(exp(T*b_k))) w.r.t. a.
    b = conv(a,a), so db_k/da_i = a[k-i] if 0<=k-i<n.
    grad[i] = sum_k weights[k] * a[k-i] * I(0<=k-i<n)
             = sum_j a[j] * weights[j+i]  (with j = k-i, so k = j+i)
    """
    n = len(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    s = exp_Tb.sum()
    if s < 1e-300:
        weights = np.zeros_like(b)
        weights[np.argmax(b)] = 1.0
    else:
        weights = exp_Tb / s

    f = 1 << (2 * n).bit_length()
    A = rfft(a, f)
    W = rfft(weights, f)
    cross = irfft(A * np.conj(W), f)
    # grad[i] = cross[(f - i) % f]
    idx = np.empty(n, dtype=np.intp)
    idx[0] = 0
    if n > 1:
        idx[1:] = f - np.arange(1, n)
    grad = np.real(cross[idx])
    return grad


def _optimize(a_init, deadline, T_start=50.0, T_end=1.0, lr_scale=0.5,
              reheat_every=500, reheat_factor=3.0, max_iters=500000):
    """
    Smoothed subgradient method with temperature annealing.
    Works with a normalized (sum=1) non-negative sequence.
    Returns the best (a, score) found.
    """
    a = np.array(a_init, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    b = _fft_convolve(a, a)
    avg_b = b.sum() / len(b)  # = 1/(2n-1) since sum(a)=1
    peak_b = b.max()
    score = 2.0 * n * peak_b  # since sum(a)=1
    best_a = a.copy()
    best_score = score

    T = T_start
    iter_count = 0
    stall_count = 0
    T_current = T_start

    while time.monotonic() < deadline and iter_count < max_iters:
        compute_b = (iter_count % 3 == 0)
        if compute_b:
            b = _fft_convolve(a, a)

        # Anneal temperature
        progress = min(1.0, iter_count / (max_iters * 0.8))
        T_current = T_start * (T_end / T_start) ** progress

        # Reheat if stalled
        if iter_count % reheat_every == 0 and iter_count > 0:
            if stall_count > reheat_every * 0.7:
                T_current = min(T_current * reheat_factor, T_start)
                stall_count = 0
            else:
                stall_count = 0

        # Compute gradient
        grad = _compute_grad(a, b, T_current)

        # Adaptive step size
        max_a = a.max()
        if max_a < 1e-15:
            break
        lr = lr_scale / max_a * (1.0 / (1.0 + 0.001 * iter_count))
        lr = min(lr, 0.5 / max_a)

        # Update
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-12:
            a_new = a  # keep old
        else:
            a_new = a_new / a_new_sum
            # Check improvement (using current b as proxy)
            b_new_approx = b  # cheap check
            score_new = 2.0 * n * b_new_approx.max()
            if score_new < score * (1.001 + 0.01 * T_current / T_start):
                a = a_new
            else:
                # Accept with some probability (simulated annealing style)
                if np.random.random() < 0.1:
                    a = a_new

        iter_count += 1

        # Periodic exact evaluation
        if iter_count % 200 == 0:
            b = _fft_convolve(a, a)
            peak_b = b.max()
            score = 2.0 * n * peak_b
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 200

    # Final evaluation
    b = _fft_convolve(a, a)
    score = 2.0 * n * b.max()
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def _pocs_init(n, n_iters=100, seed=0):
    """
    Initialize using Projection Onto Convex Sets:
    alternate between flat spectrum and non-negativity.
    """
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.5
    a = a / a.sum()
    f = 1 << (2 * n).bit_length()

    for _ in range(n_iters):
        # Project onto flat-spectrum constraint
        A = rfft(a, f)
        mags = np.abs(A)
        target_mag = mags.sum() / len(mags)
        phases = A / np.where(mags > 1e-15, mags, 1.0)
        A_new = target_mag * phases
        a_new = np.real(irfft(A_new, f))[:n]
        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        else:
            a = rng.random(n) + 0.1
            a = a / a.sum()

    return a


def _local_refine(a, deadline, n_eval_calls=200):
    """
    Fine-tune using small random perturbations and accept-if-better.
    """
    a = np.array(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        return a, np.inf
    a = a / s
    n = len(a)
    b = _fft_convolve(a, a)
    score = 2.0 * n * b.max()
    best_a = a.copy()
    best_score = score
    evals = 0

    rng = np.random.default_rng(42)

    while time.monotonic() < deadline and evals < n_eval_calls:
        # Random perturbation
        perturbation_scale = 0.05
        delta = rng.normal(0, perturbation_scale, n)
        a_new = a * (1.0 + delta)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-12:
            evals += 1
            continue
        a_new = a_new / s_new
        b_new = _fft_convolve(a_new, a_new)
        score_new = 2.0 * n * b_new.max()
        evals += 1
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_score

    Multi-phase optimization:
    1. Subgradient from height_sequence_1
    2. POCS init + subgradient for various lengths
    3. Random init + subgradient
    4. Local refinement of best solutions
    5. Verify with evaluate_sequence
    """

    print(f"Starting optimization at {time.strftime('%H:%M:%S')}, budget={budget_s}s")

    # Phase 1: Optimize from height_sequence_1
    t_phase1 = time.monotonic() + (deadline - time.monotonic()) * 0.30
    try:

        if a_init.ndim != 1 or a_init.size == 0:
            a_init = np.ones(1000)
        print(f"Phase 1: height_sequence_1, n={len(a_init)}")
        a_opt, score_opt = _optimize(a_init, t_phase1, T_start=30.0, T_end=0.5,
                                      lr_scale=0.3, reheat_every=300)
        # Verify with actual evaluator
        actual_score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(actual_score) and actual_score < best_score:
            best_score = actual_score

            print(f"  Phase 1 result: score={actual_score:.6f}")
    except Exception as e:

    return [float(x) for x in best_sequence]
