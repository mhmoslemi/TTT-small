"""
Minimizes 2n*max(convolve(a,a))/sum(a)^2 using FFT-based momentum gradient
descent on sum(c^p) surrogates (p=2,4), with multiple structured restarts,
adaptive step sizes, block/full perturbations to escape local minima, and
periodic evaluation against the true objective.
"""
import numpy as np
import time


def compute_gradient_fft(a, cw):
    """Compute gradient via FFT. Returns grad where grad[i]=2*(T1[i]+T2[i]),
    T1[i]=sum_j cw[j+i]*a[j], T2[i]=sum_j cw[i-j]*a[j]."""
    n = len(a)
    L = 1
    while L < 3 * n:
        L *= 2
    a_pad = np.zeros(L)
    a_pad[:n] = a
    cw_pad = np.zeros(L)
    cw_pad[:len(cw)] = cw
    A = np.fft.rfft(a_pad)
    CW = np.fft.rfft(cw_pad)
    T1 = np.fft.irfft(CW * np.conj(A), n=L)[:n]
    T2 = np.fft.irfft(CW * A, n=L)[:n]
    return 2.0 * (T1 + T2)


def grad_step(a, alpha, p, velocity, momentum):
    """One momentum gradient step minimizing sum(c^p). Normalized gradient."""
    n = len(a)
    c = np.convolve(a, a)
    if p == 2:
        cw = c
    elif p == 4:
        cw = c * c * c
    else:
        cw = c ** (p - 1)
    grad = float(p) * compute_gradient_fft(a, cw)
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad /= gnorm
    velocity = momentum * velocity - alpha * grad
    a_new = a + velocity
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a, velocity
    a_new /= s
    return a_new, velocity

    elif kind == 'quad_center':

    elif kind == 'quad_edge':
        a = (x / n)**2 + 0.1
    elif kind == 'lopsided':
        a = np.exp(3.0 * x / n)
    elif kind == 'random':
        a = np.abs(rng.standard_normal(n)) + 0.01
    elif kind == 'geometric':
        a = 1.3 ** (x / n * 5)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:

        s = float(n)
    return a / s


def block_perturb(a, rng, scale=0.1):
    """Perturb a random block of the sequence."""
    n = len(a)
    cand = a.copy()
    width = max(2, int(n * rng.uniform(0.01, 0.15)))
    start = int(rng.integers(0, max(1, n - width)))
    stop = min(n, start + width)
    w = stop - start
    if w > 0:
        cand[start:stop] *= np.exp(rng.normal(0, scale, size=w))
    cand = np.maximum(cand, 0.0)
    s = np.sum(cand)
    if s > 1e-10:
        cand /= s
    else:
        cand = a.copy()
    return cand


def full_perturb(a, rng, scale=0.05):
    """Perturb the entire sequence multiplicatively."""
    n = len(a)

    cand = np.maximum(cand, 0.0)
    s = np.sum(cand)
    if s < 1e-10:
        return a.copy()
    return cand / s
