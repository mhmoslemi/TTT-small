from scipy.optimize import linprog
import time

def build_conv_matrix(a):
    """Build the convolution matrix A where (A @ b)[k] = (a*b)[k]."""
    n = len(a)
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2*n-1, n))
    A[valid] = a[diff[valid]]
    return A

def gaussian_init(n, sigma_frac=0.3):
    """Create a Gaussian-shaped initial sequence."""
    center = (n-1) / 2.0
    sigma = sigma_frac * n
    x = np.arange(n)
    a = np.exp(-(x-center)**2 / (2*sigma**2))
    return a

def random_init(n, rng):
    """Create a random non-negative initial sequence."""
    return rng.random(n)

def sine_init(n, power=2.0):
    """Create a sine-shaped initial sequence."""
    x = np.linspace(0, np.pi, n)
    return np.sin(x)**power

def lp_step(a, M):
    """Solve LP to find best g: max sum(g) s.t. g>=0, a*g <= M."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                  options={'presolve': True})
    if res.success and np.all(res.x >= -1e-10):
        g = np.clip(res.x, 0, None)
        if np.sum(g) > 0:
            return g
    return None

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """LP-based local search with Gaussian initialization and random restarts."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0
    
    best_seq = None
    best_score = np.inf
    
    # Try different initializations and lengths
    configs = [
        (512, 'gaussian', 0.3),
        (512, 'gaussian', 0.2),
        (1024, 'gaussian', 0.3),
        (1024, 'sine', 2.0),
        (256, 'gaussian', 0.25),
        (2048, 'gaussian', 0.3),
        (512, 'random', None),
        (1024, 'random', None),
    ]
    
    for n, init_type, param in configs:
        if time.monotonic() > deadline:
            break
        
        if init_type == 'gaussian':
            a = gaussian_init(n, param)
        elif init_type == 'sine':
            a = sine_init(n, param)
        elif init_type == 'random':
            a = random_init(n, rng)
        
        a = np.clip(a, 0.0, 1000.0)
        if np.sum(a) < 0.01:
            a = np.ones(n) / n
        a = a / np.sum(a) * np.sqrt(2*n)  # normalize so sum = sqrt(2n)
        
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
        
        # LP optimization loop
        t = 0.1  # blending factor
        stuck_count = 0
        current_best = a.copy()
        current_score = score
        
        while time.monotonic() < deadline:
            M = float(np.max(np.convolve(current_best, current_best)))
            g = lp_step(current_best, M)
            
            if g is None:
                break
            
            # Rescale g to match sum of current_best
            g_scaled = g / np.sum(g) * np.sum(current_best)
            
            # Try different blending factors
            improved = False
            for t_try in [0.05, 0.1, 0.2, 0.3, 0.5]:
                candidate = (1-t_try) * current_best + t_try * g_scaled
                candidate = np.clip(candidate, 0.0, 1000.0)
                if np.sum(candidate) < 0.01:
                    continue
                s = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(s) and s < current_score:
                    current_best = candidate
                    current_score = s
                    improved = True
                    if s < best_score:
                        best_score = s
                        best_seq = candidate.copy()
                    break
            
            if not improved:
                stuck_count += 1
                if stuck_count > 50:
                    # Random perturbation to escape
                    pert = current_best * np.exp(rng.normal(0, 0.3, n))
                    pert = np.clip(pert, 0.0, 1000.0)
                    if np.sum(pert) > 0.01:
                        s = float(evaluate_sequence(pert.tolist()))
                        if np.isfinite(s):
                            current_best = pert
                            current_score = s
                    stuck_count = 0
                    if stuck_count > 200:
                        break
        
        print(f"n={n}, init={init_type}, best_score={best_score:.6f}")
    
    return [float(x) for x in best_seq]
