import time
import cvxpy as cp

def solve_lp_for_g(f, M):
    """Solve the LP: maximize sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0]
    
    # Build the convolution constraint matrix
    # (f*g)_k = sum_i f_i * g_{k-i} for valid i, k-i
    # This is a Toeplitz-like matrix
    for k in range(2*n - 1):
        # (f*g)_k = sum over i where 0 <= i < n and 0 <= k-i < n
        # i ranges from max(0, k-n+1) to min(n-1, k)
        idx_start = max(0, k - n + 1)
        idx_end = min(n - 1, k)
        # (f*g)_k = sum_{i=idx_start}^{idx_end} f_i * g_{k-i}
        # Let j = k-i, so j ranges from k-idx_end to k-idx_start
        # (f*g)_k = sum_{j=k-idx_end}^{k-idx_start} f_{k-j} * g_j
        coeffs = f[k - (k-idx_start):k - (k-idx_end)+1]  # f_{k-j} for j in range
        # Actually let me be more careful
        # j = k - i, so when i = idx_start, j = k - idx_start
        # when i = idx_end, j = k - idx_end
        # j ranges from k-idx_end to k-idx_start
        j_start = k - idx_end
        j_end = k - idx_start
        # (f*g)_k = sum_{j=j_start}^{j_end} f_{k-j} * g_j
        constraint = cp.sum(cp.multiply(f[k - np.arange(j_start, j_end+1)], g[j_start:j_end+1])) <= M

    prob = cp.Problem(objective, constraints)
    prob.solve()
    
    if g.value is None:
        return None
    return np.array(g.value)

def build_conv_matrix(f):
    """Build the matrix A such that A @ g = f * g (convolution)."""
    n = len(f)
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):

        i_end = min(n-1, k)

            top_idx = violated[np.argsort(viol_amounts)[-5:]]

"""
Iterative LP-based optimization for minimizing the auto-convolution peak ratio.
Uses separation oracle for the LP, gradient steps, and random perturbations.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _compute_score(a):
    """Compute the score directly (faster than evaluate_sequence for iteration)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = np.convolve(a, a)
    max_b = np.max(b)
    return float(2.0 * n * max_b / (S * S))


def _lp_update(f, M, max_separation=100):
    """
    Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Uses separation oracle (add violated constraints iteratively).
    """
    n = len(f)
    c = -np.ones(n)
    A_ub_list = []
    b_ub_list = []
    
    # Start with a reasonable initial constraint set
    # The most "central" constraints are usually the most violated
    mid = n - 1
    for k in [mid]:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        row = np.zeros(n)
        for i in range(i_start, i_end + 1):
            row[i] = f[k - i]
        A_ub_list.append(row)
        b_ub_list.append(M)
    
    g_val = None
    
    for sep_iter in range(max_separation):
        if A_ub_list:
            A_ub = np.array(A_ub_list)
            b_ub = np.array(b_ub_list)
        else:
            A_ub = None
            b_ub = None
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, 
                      bounds=[(0, None)] * n, 
                      method='highs',
                      options={'presolve': True})
        
        if not res.success:
            if g_val is not None:
                return g_val
            break
        
        g_val = res.x.copy()
        
        # Check all constraints via convolution
        conv = np.convolve(f, g_val)
        tolerance = 1e-6 * M
        violated = np.where(conv > M + tolerance)[0]
        
        if len(violated) == 0:
            return g_val
        
        # Add the most violated constraints (up to 20)
        viol_amount = conv[violated] - M
        if len(violated) > 20:
            top_idx = violated[np.argsort(viol_amount)[-20:]]
        else:
            top_idx = violated
        
        for k in top_idx:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            row = np.zeros(n)
            # Vectorized: row[i_start:i_end+1] = f[k-i for i in i_start..i_end]
            row[i_start:i_end+1] = f[k - np.arange(i_start, i_end + 1)]
            A_ub_list.append(row)
            b_ub_list.append(M)
        
        # Safety: if too many constraints, break
        if len(A_ub_list) > 2000:
            break
    
    return g_val


def _gradient_step(a, alpha=0.01):
    """
    Subgradient step to reduce the peak of the convolution.
    Reduces elements that contribute to the peak.
    """
    n = len(a)
    b = np.convolve(a, a)
    k_star = np.argmax(b)
    
    # Gradient of b_{k_star} with respect to a_i is a_{k_star - i}
    # So we reduce a_i by alpha * a_{k_star - i}
    grad = np.zeros(n)
    i_start = max(0, k_star - n + 1)
    i_end = min(n - 1, k_star)
    grad[i_start:i_end+1] = a[k_star - np.arange(i_start, i_end + 1)]
    
    a_new = a - alpha * grad
    a_new = np.clip(a_new, 0, 1000)
    
    # Re-normalize to maintain the same sum
    S = np.sum(a_new)
    if S > 0.01:
        a_new = a_new * (np.sum(a) / S)
    
    return a_new


def _block_perturbation(a, rng, num_blocks=5):
    """Apply random multiplicative perturbation to random blocks."""
    n = len(a)
    a_new = a.copy()
    block_size = max(1, n // (num_blocks * 2))
    for _ in range(num_blocks):
        start = rng.integers(0, n - block_size)
        end = start + block_size
        factor = np.exp(rng.normal(0, 0.1))
        a_new[start:end] *= factor
    a_new = np.clip(a_new, 0, 1000)
    return a_new


def _global_rescale(a, target_sum):
    """Rescale a to have the target sum."""
    S = np.sum(a)
    if S < 1e-10:
        return a
    return a * (target_sum / S)

Optimizes non-negative sequence to minimize 2n*max(convolve(a,a))/(sum(a))^2.
Uses KKT-inspired water-filling updates with random restarts and LP-based refinement.

def _fast_score(a):
    """Compute score: 2n * max(conv) / sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * b.max() / (S * S)


def _water_fill(a, alpha=0.03, n_iter=20, temp_scale=0.1):
    """
    KKT-inspired water-filling update.
    Iteratively adjusts a to flatten the auto-convolution.
    """
    n = len(a)
    a = a.copy()
    
    for it in range(n_iter):
        b = np.convolve(a, a)
        max_b = b.max()
        
        # Temperature for peak concentration
        T = max_b * temp_scale
        if T < 1e-10:
            T = 1e-10
        
        # Lambda: exponential weighting concentrated on peaks
        lam = np.exp((b - max_b) / T)
        lam_sum = lam.sum()
        if lam_sum < 1e-30:
            break
        lam = lam / lam_sum
        
        # c_j = sum_k lam_k * a_{k-j} (cross-correlation)
        # Using convolution: c = conv(a, lam[::-1])[n-1:2n-1][::-1]
        conv = np.convolve(a, lam[::-1])
        c = conv[n-1:2*n-1][::-1]
        
        # Target level: median of c on support
        threshold = a.max() * 1e-6
        support = a > threshold
        if support.sum() > 0:
            mu = np.median(c[support])
        else:
            mu = np.median(c)
        
        # Multiplicative update
        c_range = c.max() - c.min()
        if c_range < 1e-10:

        a = a * np.exp(alpha * (mu - c) / c_range)
        a = np.clip(a, 0.0, 1000.0)
        
        # Normalize to sum 1
        S = a.sum()
        if S > 0.01:
            a = a / S
        else:
            break
    
    return a
