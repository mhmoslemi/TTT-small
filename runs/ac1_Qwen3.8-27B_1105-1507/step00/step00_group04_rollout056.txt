def _compute_subgradient(a, k_star, n):
    """Compute subgradient of max_k C[k] w.r.t. a at the active constraint k_star."""
    grad = np.zeros(n)
    # dC[k*]/da[i] = 2*a[k*-i] if 0 <= k*-i < n
    # i ranges from max(0, k*-n+1) to min(n-1, k*)
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    if lo <= hi:
        # grad[i] = 2 * a[k_star - i] for i in [lo, hi]
