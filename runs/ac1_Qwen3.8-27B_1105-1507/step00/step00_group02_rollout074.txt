SOR (Successive Over-Relaxation) + local search hybrid for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key ideas:
- Normalize to sum=n (scale invariance), so score = 2*max_r/n
- Use FFT-based convolution for speed
- Anti-symmetric initial perturbations to break symmetry
- Multi-peak pressure for better convergence
- Local search with SOR refinement for fine-tuning
"""

import numpy as np
import time


def _fft_conv_sq(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]


def _normalize(a, n):
    """Return a new array normalized to sum=n, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n, dtype=np.float64)
    return a * (n / s)


def _score_from_r(r, n):
    """Compute score = 2*n*max_r/sum^2 given we normalized sum=n."""
    return 2.0 * r.max() / n


def _sor_single_peak(a, alpha):
    """One SOR step targeting the single highest peak."""
    n = len(a)
    r = _fft_conv_sq(a)
    k_star = int(np.argmax(r))
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)
    j = k_star - np.arange(i_min, i_max + 1)
    a[i_min:i_max + 1] -= alpha * a[j]
    np.maximum(a, 0.0, out=a)
    s = a.sum()
    if s < 1e-10:
        a[:] = 1.0
    else:
        a *= (n / s)
    return a


def _sor_multi_peak(a, alpha, K):
    """One SOR step targeting the top K peaks simultaneously."""
    n = len(a)
    r = _fft_conv_sq(a)
    top_k = np.argpartition(r, -K)[-K:]
    pressure = np.zeros(n, dtype=np.float64)
    for k in top_k:
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        j = k - np.arange(i_min, i_max + 1)
        pressure[i_min:i_max + 1] += a[j]
    pressure /= K
    a -= alpha * pressure
    np.maximum(a, 0.0, out=a)
    s = a.sum()
    if s < 1e-10:
        a[:] = 1.0
    else:
        a *= (n / s)
    return a


def _run_sor(a, n, alpha_start, alpha_min, n_iters, deadline, K=1, log_interval=5000):
    """Run SOR iterations until deadline or max iters. Returns (a, best_r_max)."""
    best_r_max = _fft_conv_sq(a).max()
    alpha = alpha_start
    it = 0
    while it < n_iters:
        if time.monotonic() > deadline:
            break
        if K <= 1:
            a = _sor_single_peak(a, alpha)
        else:
            a = _sor_multi_peak(a, alpha, K)
        it += 1
        if it % log_interval == 0:
            r = _fft_conv_sq(a)
            cur_max = r.max()
            if cur_max < best_r_max:
                best_r_max = cur_max
            alpha = max(alpha_min, alpha * 0.98)
    return a, best_r_max


def _local_search(best, n, deadline, rng, sors_per_trial=200):
    """Local search: random perturbations + SOR refinement. Returns (best, best_score)."""
    best_score = _score_from_r(_fft_conv_sq(best), n)
    improvements = 0
    stuck_count = 0
    noise_scale = 0.05

    while time.monotonic() < deadline:
        candidate = best.copy()
        n_blocks = int(rng.integers(1, 4))
        for _ in range(n_blocks):
            width = int(rng.integers(max(1, n // 500), max(2, n // 10)))
            start = int(rng.integers(0, n - width + 1))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, noise_scale, size=width))
        candidate = _normalize(candidate, n)

        for _ in range(sors_per_trial):
            if time.monotonic() > deadline:
                break
            candidate = _sor_single_peak(candidate, 0.03)

        r = _fft_conv_sq(candidate)
        score = _score_from_r(r, n)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = candidate.copy()
            improvements += 1
            stuck_count = 0
            if improvements <= 30 or improvements % 20 == 0:
                print(f"  LS improvement #{improvements}: score={best_score:.6f}")
        else:
            stuck_count += 1
            if stuck_count % 500 == 0:
                noise_scale = max(0.01, noise_scale * 0.9)
                if time.monotonic() > deadline:
                    break

    return best, best_score
