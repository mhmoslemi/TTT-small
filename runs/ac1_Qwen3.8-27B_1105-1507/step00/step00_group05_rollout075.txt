    # Phase 3: refine best found with sharper beta (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a3, v3 = _optimize_smooth_max(best, remaining, beta0=50.0, beta1=2000.0, lr0=0.1, seed=seed + 999)
        cand3 = a3 * best.sum()
        sc3 = float(evaluate_sequence(cand3.tolist()))
        print(f"Phase3 (refine): score={sc3:.6f}", flush=True)
        if np.isfinite(sc3) and sc3 < best_score:
            best = cand3
            best_score = sc3

    # Phase 4: final fine-tuning if time remains
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a4, v4 = _optimize_smooth_max(best, remaining, beta0=200.0, beta1=5000.0, lr0=0.05, seed=seed + 7777)
        cand4 = a4 * best.sum()
        sc4 = float(evaluate_sequence(cand4.tolist()))
        print(f"Phase4 (fine): score={sc4:.6f}", flush=True)
        if np.isfinite(sc4) and sc4 < best_score:
            best = cand4
            best_score = sc4

    # Final validation
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.ones(256)
    final_score = float(evaluate_sequence(best.tolist()))
    print(f"FINAL: n={len(best)}, score={final_score:.6f}", flush=True)
    return [float(x) for x in best]
