def normalize_p(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s <= 1e-300:
       return np.ones(p.size, dtype=np.float64) / max(1,p.size)
    return p / s

def next_pow2(x):
    n=1
    while n < x: n <<=1
    return n

def fft_convolve(a):
    n=a.size
    size = next_pow2(2*n-1)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa*fa, size)
    return c[:2*n-1]

def cross_conv_weights_p(w, p):
    n=p.size
    lp = len(w)
    size = next_pow2(lp + n - 1)
    fw = np.fft.rfft(w, size)
    fp = np.fft.rfft(p[::-1], size)
    h = np.fft.irfft(fw*fp, size)
    return h[n-1:n-1+n].copy()

```python
"""
Search for non-negative step-function heights minimizing
    score = 2 * n * max(convolve(a, a, 'full')) / (sum(a)**2).

The program works with the scale-invariant form: normalize a sequence to
sum 1, then minimize 2*n*max(convolve(p,p)).  It uses several smooth
random/Fourier/Gaussian starts, a soft-peak mirror-descent local optimizer,
and occasional exact pair-transfer refinements.  The best sequence found is
scaled so that every returned value is in [0, 1000].
"""

import time
import numpy as np


def normalize_p(p):
    """Clip to non-negative finite values and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64)
    if p.size == 0:
        return np.ones(1, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-300:
        return np.ones(p.size, dtype=np.float64) * (1.0 / max(1, p.size))
    return p / s


def next_power_of_two(x):
    """Smallest power of two >= x."""
    n = 1
    while n < x:
        n <<= 1
    return n


def fft_convolve_full(a):
    """Full linear convolution of a real 1-D sequence with itself via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = next_power_of_two(2 * n - 1)
    fa = np.fft.rfft(a, size)
    out = np.fft.irfft(fa * fa, size)
    return out[:2 * n - 1]


def cross_conv_weights_p(w, p):
    """
    Compute v[i] = sum_k w[k] * p[k-i] for i=0..n-1,
    where w has length 2*n-1 and p has length n.
    """
    n = p.size
    lw = len(w)
    size = next_power_of_two(lw + n - 1)
    fw = np.fft.rfft(np.asarray(w, dtype=np.float64), size)
    fp = np.fft.rfft(np.asarray(p, dtype=np.float64)[::-1], size)
    h = np.fft.irfft(fw * fp, size)
    return h[n - 1:n - 1 + n].copy()


def score_p(p):
    """
    Score for a normalized non-negative sequence p (sum approx 1).
    This equals the original evaluation function because the score is
    scale-invariant.
    """
    p = normalize_p(p)
    n = p.size
    if n == 0:
        return float("inf")
    c = fft_convolve_full(p)
    if c.size == 0:
        return float("inf")
    mx = float(c.max())
    if not np.isfinite(mx) or mx <= 0.0:
        return float("inf")
    return float(2.0 * n * mx)


def softmax_weights(c, temp):
    """Softmax over convolution lags with scale temp."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.empty(0, dtype=np.float64)
    if temp <= 1e-300:
        temp = 1e-300
    x = c / temp
    x = x - float(x.max())
    with np.errstate(over="ignore", under="ignore", invalid="ignore"):
        e = np.exp(x)
    e = np.where(np.isfinite(e), e, 0.0)
    s = float(e.sum())
    if s <= 1e-300:
        w = np.zeros(c.size, dtype=np.float64)
        int_idx = int(np.argmax(c))
        w[int_idx] = 1.0
        return w
    return e / s


def make_initial(n, kind, rng, parent=None):
    """
    Create a normalized non-negative initial sequence of length n.
    kind selects a family:
      0,1: log-normal noise
      2,3: Dirichlet noise
      4,5,6: sinusoidal/Fourier shapes
      7: Gaussian bumps
      8: downsampled parent sequence if available
    """
    n = int(max(1, n))
    kind = int(kind) % 9
    x = np.linspace(0.0, 1.0, n)

    if kind == 8 and parent is not None:
        parent = np.asarray(parent, dtype=np.float64)
        if parent.size >= n:
            idx = np.linspace(0.0, parent.size - 1.0, n)
            p = np.interp(idx, np.arange(parent.size, dtype=np.float64), parent)
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.maximum(p, 0.0)
            if float(p.sum()) > 1e-300:
                return normalize_p(p)

    if kind == 0:
        p = np.exp(rng.normal(0.0, 0.35, n))
    elif kind == 1:
        p = np.exp(rng.normal(0.0, 1.0, n))
    elif kind == 2:
        p = rng.dirichlet(np.full(n, 0.5, dtype=np.float64))
    elif kind == 3:
        p = rng.dirichlet(np.full(n, 2.0, dtype=np.float64))
    elif kind == 4:
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        eps = float(rng.uniform(0.3, 0.9))
        p = 1.0 + eps * np.sin(2.0 * np.pi * x + phase)
    elif kind == 5:
        a = float(rng.uniform(-0.8, 0.8))
        b = float(rng.uniform(-0.8, 0.8))
        ca = float(rng.uniform(-0.6, 0.6))
        cb = float(rng.uniform(-0.6, 0.6))
        p = (
            1.0
            + a * np.sin(2.0 * np.pi * x)
            + b * np.cos(2.0 * np.pi * x)
            + ca * np.sin(4.0 * np.pi * x)
            + cb * np.cos(4.0 * np.pi * x)
        )
    elif kind == 6:
        p = np.ones(n, dtype=np.float64)
        for m in range(1, 5):
            p += float(rng.normal(0.0, 0.25)) * np.sin(2.0 * np.pi * m * x)
            p += float(rng.normal(0.0, 0.25)) * np.cos(2.0 * np.pi * m * x)
        p += 0.25
    else:
        c1 = float(rng.uniform(0.1, 0.9))
        c2 = float(rng.uniform(0.1, 0.9))
        s1 = float(rng.uniform(0.02, 0.25))
        s2 = float(rng.uniform(0.02, 0.25))
        w1 = float(rng.uniform(0.3, 1.0))
        p = (
            w1 * np.exp(-(((x - c1) / s1) ** 2))
            + np.exp(-(((x - c2) / s2) ** 2))
            + 0.08
        )

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    if float(p.sum()) <= 1e-300:
        p = np.ones(n, dtype=np.float64)
    return normalize_p(p)


def pair_refine(p, n, sub_deadline, rng, w=None, c=None):
    """
    Exact local search by transferring mass from high-gradient positions to
    low-gradient positions.  Uses the current soft-peak weights v as a guide.
    """
    p = normalize_p(p)
    n = p.size
    if n == 0:
        return p, float("inf")

    if c is None:
        c = fft_convolve_full(p)
    if c.size == 0:
        return p, float("inf")
    C = float(c.max())
    if not np.isfinite(C) or C <= 0.0:
        return p, float("inf")

    if w is None:
        w = softmax_weights(c, max(0.05 * C, 1e-300))

    v = cross_conv_weights_p(w, p)
    best_p = p.copy()
    best_score = float(2.0 * n * C)

    if n <= 2:
        return best_p, best_score

    cnt = int(max(3, min(12, n // 32)))
    if n > cnt:
        hi = np.argpartition(v, -cnt)[-cnt:]
        lo = np.argpartition(v, cnt)[:cnt]
    else:
        hi = np.arange(n)
        lo = np.arange(n)

    pairs = []
    for i in hi:
        for j in lo:
            i = int(i)
            j = int(j)
            if i != j and p[i] > 0.0 and v[i] > v[j]:
                pairs.append((float(v[j] - v[i]), i, j))

    if not pairs:
        return best_p, best_score

    pairs.sort()

    max_eval = 70
    evaluated = 0
    for _, i, j in pairs:
        if evaluated >= max_eval or time.monotonic() > sub_deadline:
            break
        dmax = float(p[i])
        if dmax <= 1e-300:
            continue
        for frac in (0.2, 0.5, 0.8, 0.95):
            if evaluated >= max_eval or time.monotonic() > sub_deadline:
                break
            d = frac * dmax
            if d <= 0.0:
                continue
            q = p.copy()
            q[i] -= d
            q[j] += d
            s = float(q.sum())
            if s <= 1e-300:
                continue
            if abs(s - 1.0) > 1e-12:
                q /= s
            qs = score_p(q)
            evaluated += 1
            if np.isfinite(qs) and qs < best_score:
                best_score = qs
                best_p = q

    return best_p, best_score


def mirror_optimize(p0, n, sub_deadline, rng, max_no_improve=90, pair_every=8):
    """
    Soft-peak mirror-descent local optimizer.

    The current peak structure is softened into dual weights w over lags.
    v = cross-correlation(w, p) is a subgradient-like quantity for the
    peak objective.  Moving mass from high-v positions to low-v positions
    by multiplicative updates tends to flatten the convolution.
    """
    p = normalize_p(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    best_p = p.copy()
    best_score = score_p(p)
    if not np.isfinite(best_score):
        return best_p, best_score

    temp_frac = 0.40
    step_scale = 1.0
    no_improve = 0
    it = 0

    while time.monotonic() < sub_deadline and no_improve < max_no_improve:
        c = fft_convolve_full(p)
        C = float(c.max())
        if not np.isfinite(C) or C <= 1e-300:
            break

        current_score = float(2.0 * n * C)
        if current_score < best_score:
            best_score = current_score
            best_p = p.copy()

        temp = max(temp_frac * C, 1e-300)
        w = softmax_weights(c, temp)
        v = cross_conv_weights_p(w, p)
        dv = v - float(v.mean())
        sd = float(np.std(dv))

        if not np.isfinite(sd) or sd < 1e-15:
            p = normalize_p(np.exp(rng.normal(0.0, 0.30, n)))
            no_improve += 1
            it += 1
            continue

        if n <= 256:
            facs = (0.4 * step_scale, 1.2 * step_scale, 3.0 * step_scale)
            alphas = (0.5, 1.0)
            use_reset = True
        elif n <= 1024:
            facs = (0.5 * step_scale, 1.8 * step_scale)
            alphas = (0.5, 1.0)
            use_reset = True
        else:
            facs = (0.6 * step_scale, 2.0 * step_scale)
            alphas = (1.0,)
            use_reset = False

        best_cand = None
        best_cand_score = best_score

        for fac in facs:
            if fac <= 0.0:
                continue
            arg = -fac * dv / max(sd, 1e-300)
            arg = np.clip(arg, -40.0, 40.0)
            mult = np.exp(arg)
            mult = np.where(np.isfinite(mult), mult, 0.0)

            # Multiplicative update: p * mult.
            pm = p * mult
            s = float(pm.sum())
            if s > 1e-300:
                pm = pm / s
                for alpha in alphas:
                    if time.monotonic() > sub_deadline:
                        break
                    q = (1.0 - alpha) * p + alpha * pm
                    q = normalize_p(q)
                    qs = score_p(q)
                    if np.isfinite(qs) and qs < best_cand_score:
                        best_cand_score = qs
                        best_cand = q

            # Reset update: normalize(mult), mixed with old p.
            if use_reset:
                s = float(mult.sum())
                if s > 1e-300:
                    rm = mult / s
                    for alpha in alphas:
                        if time.monotonic() > sub_deadline:
                            break
                        q = (1.0 - alpha) * p + alpha * rm
                        q = normalize_p(q)
                        qs = score_p(q)
                        if np.isfinite(qs) and qs < best_cand_score:
                            best_cand_score = qs
                            best_cand = q

            if time.monotonic() > sub_deadline:
                break

        if best_cand is not None:
            best_p = best_cand
            best_score = best_cand_score
            p = best_cand.copy()
            no_improve = 0
            temp_frac = max(0.01, temp_frac * 0.80)

            if it % max(1, pair_every) == 0 and time.monotonic() < sub_deadline - 1.0:
                p2, s2 = pair_refine(p, n, sub_deadline, rng)
                if np.isfinite(s2) and s2 < best_score:
                    best_p = p2
                    best_score = s2
                    p = p2.copy()
        else:
            no_improve += 1
            if temp_frac > 0.08:
                temp_frac *= 0.70
            else:
                if step_scale > 0.1:
                    step_scale *= 0.6
                else:
                    sigma = float(0.25 + 0.05 * min(no_improve, 12))
                    p = normalize_p(np.exp(np.log(p + 1e-12) + rng.normal(0.0, sigma, n)))
                    temp_frac = 0.35
                    step_scale = 1.0

        it += 1

    if time.monotonic() < sub_deadline - 1.0:
        p2, s2 = pair_refine(p, n, sub_deadline, rng)
        if np.isfinite(s2) and s2 < best_score:
            best_p = p2
            best_score = s2

    return best_p, best_score


def scale_result(p):
    """Scale a normalized sequence so max value is exactly 1000 (clip safe)."""
    p = np.asarray(p, dtype=np.float64)
    if p.size == 0:
        return [1000.0]
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    mx = float(p.max())
    if mx <= 1e-300:
        return [1000.0] * p.size
    out = p * (1000.0 / mx)
    out = np.clip(out, 0.0, 1000.0)
    return [float(x) for x in out]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for the best valid coefficient sequence under the given time budget.
    Returns a list of floats in [0,1000].
    """
    start = time.monotonic()
    budget_s = max(0.0, float(budget_s))
    # Leave a small safety margin before the hard 1500s limit.
    deadline = start + max(0.0, budget_s - 18.0)
    rng = np.random.default_rng(seed)

    # Optional parent sequence from the previous construction.
    parent = None
    try:
        hs = height_sequence_1
        hs = np.asarray(hs, dtype=np.float64)
        if hs.ndim == 1 and hs.size > 0:
            hs = np.where(np.isfinite(hs), hs, 0.0)
            hs = np.maximum(hs, 0.0)
            if float(hs.sum()) > 1e-300:
                parent = normalize_p(hs)
    except Exception:
        parent = None

    # Reliable fallback baseline: uniform sequence has exact score 2.
    best_n = 512
    best_p = normalize_p(np.ones(best_n, dtype=np.float64))
    best_score = score_p(best_p)
    if not np.isfinite(best_score):
        best_score = 2.0

    if parent is not None:
        try:
            sp = score_p(parent)
            if np.isfinite(sp) and sp < best_score:
                best_p = parent.copy()
                best_score = sp
                best_n = int(parent.size)
                print("parent score %.6f n=%d" % (best_score, best_n))
        except Exception:
            pass

    # Quick starts of several lengths and smooth families.
    quick_lengths = [128, 256, 512]
    quick_kinds = [0, 4, 6, 7]
    for n in quick_lengths:
        if time.monotonic() > deadline - 30.0:
            break
        for kind in quick_kinds:
            try:
                p0 = make_initial(int(n), kind, rng, parent)
                s0 = score_p(p0)
                if np.isfinite(s0) and s0 < best_score:
                    best_p = p0.copy()
                    best_score = s0
                    best_n = int(n)
                    print("quick best score %.6f n=%d" % (best_score, best_n))
            except Exception:
                pass

    # If parent length is in a manageable range, try optimizing it directly.
    if parent is not None and 512 <= parent.size <= 8192:
        try:
            pn = int(parent.size)
            p0 = make_initial(pn, 8, rng, parent)
            s0 = score_p(p0)
            if np.isfinite(s0) and s0 < best_score:
                best_p = p0.copy()
                best_score = s0
                best_n = pn

            sub = min(deadline - 12.0, time.monotonic() + 45.0)
            if sub > time.monotonic() + 3.0:
                p_opt, s_opt = mirror_optimize(p0, pn, sub, rng)
                if np.isfinite(s_opt) and s_opt < best_score:
                    best_p = p_opt.copy()
                    best_score = s_opt
                    best_n = pn
                    print("parent-length best score %.6f" % best_score)
        except Exception:
            pass

    # Exploration schedule: (length, number of attempts, seconds per attempt)
    schedule = [
        (128, 4, 12),
        (256, 5, 22),
        (512, 4, 40),
        (64, 3, 10),
        (1024, 3, 60),
    ]

    explored = 0
    for n, count, per_attempt in schedule:
        if time.monotonic() > deadline - 45.0:
            break
        # Once we are comfortably below the target, spend less on broad search.
        if best_score < 1.42 and explored > 12:
            count = 1
            per_attempt = min(per_attempt, 12)

        for _ in range(count):
            if time.monotonic() > deadline - 30.0:
                break
            sub_deadline = min(deadline - 25.0, time.monotonic() + float(per_attempt))
            if sub_deadline <= time.monotonic() + 2.0:
                continue

            kind = int(rng.integers(0, 9))
            try:
                p0 = make_initial(int(n), kind, rng, parent)
            except Exception:
                continue

            try:
                s0 = score_p(p0)
            except Exception:
                s0 = float("inf")

            # Take a plain quick improvement from the initial point.
            if np.isfinite(s0) and s0 < best_score:
                best_p = p0.copy()
                best_score = s0
                best_n = int(n)
                print("init best score %.6f n=%d kind=%d" % (best_score, best_n, kind))

            # Optimize if the start is not hopelessly far from the current best.
            if np.isfinite(s0) and s0 < best_score + 2.0:
                try:
                    p_opt, s_opt = mirror_optimize(p0, int(n), sub_deadline, rng)
                    if np.isfinite(s_opt) and s_opt < best_score:
                        best_p = p_opt.copy()
                        best_score = s_opt
                        best_n = int(n)
                        print("sched best score %.6f n=%d" % (best_score, best_n))
                except Exception:
                    pass

            explored += 1
            if best_score <= 1.34 and time.monotonic() > start + 250.0:
                break

    # Final phase: use the remaining time to refine the incumbent and, if
    # still above the target, occasionally launch new smooth random starts.
    final_guard = 30.0
    while time.monotonic() < deadline - final_guard:
        chunk = min(45.0, deadline - time.monotonic() - final_guard)
        if chunk < 4.0:
            break

        if best_score > 1.5027:
            # Mix incumbent perturbation with fresh starts.
            if rng.random() < 0.55:
                choices = [128, 256, 512, 1024]
                n_choice = int(choices[int(rng.integers(0, len(choices)))])
                kind = int(rng.integers(0, 9))
                p0 = make_initial(n_choice, kind, rng, parent)
            else:
                n_choice = int(best_n)
                sig = float(rng.uniform(0.2, 0.9))
                p0 = normalize_p(
                    np.exp(np.log(best_p + 1e-12) + rng.normal(0.0, sig, best_n))
                )
        else:
            # Already below target; focus on improving incumbent.
            n_choice = int(best_n)
            if rng.random() < 0.45:
                sig = float(rng.uniform(0.15, 1.0))
                p0 = normalize_p(
                    np.exp(np.log(best_p + 1e-12) + rng.normal(0.0, sig, best_n))
                )
            else:
                p0 = best_p.copy()

        sub_deadline = min(deadline - final_guard + 5.0, time.monotonic() + chunk)
        try:
            p_opt, s_opt = mirror_optimize(p0, int(n_choice), sub_deadline, rng)
            if np.isfinite(s_opt) and s_opt < best_score:
                best_p = p_opt.copy()
                best_score = s_opt
                best_n = int(n_choice)
                print("final best score %.6f n=%d" % (best_score, best_n))
        except Exception:
            pass

    # One last exact refinement if there is time left.
    if time.monotonic() < deadline - 5.0:
        try:
            p_opt, s_opt = mirror_optimize(best_p, int(best_n), deadline - 5.0, rng)
            if np.isfinite(s_opt) and s_opt < best_score:
                best_p = p_opt.copy()
                best_score = s_opt
                best_n = int(best_n)
                print("last best score %.6f n=%d" % (best_score, best_n))
        except Exception:
            pass

    result = scale_result(best_p)

    # Validate once with the official evaluator if available.  This is only
    # for safety; the sequence should already be valid and unclipped.
    try:
        ev = evaluate_sequence(result)
        if not np.isfinite(ev):
            result = [1000.0] * 128
    except Exception:
        pass

    return result
```
