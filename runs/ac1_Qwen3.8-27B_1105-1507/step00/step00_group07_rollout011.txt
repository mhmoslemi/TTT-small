```python
import numpy as np
import time
from scipy.optimize import linprog

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. convolve(f, g) <= M, g >= 0"""
    n = len(f)
    # Construct A matrix: A[k, j] = f[k-j] if 0 <= k-j < n else 0
    k_idx = np.arange(2*n-1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.where(valid, np.where(valid, f[diff], 0), 0)
    
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n  # g >= 0
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

def lp_iteration(f, M, t):
    """One LP iteration: find g, blend with f"""
    g = solve_lp(f, M)
    if g is None:
        return None
    # Blend
    f_new = (1-t)*f + t*g
    # Ensure non-negative
    f_new = np.maximum(f_new, 0)
    # Normalize
    if np.sum(f_new) > 0:
        f_new = f_new / np.sum(f_new) * np.sum(f)
    return f_new
```

```python
import numpy as np
import time
from scipy.optimize import linprog

def build_A(n, f):
    """Build the constraint matrix A for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n else 0
    Shape: (2n-1, n)
    """
    k_idx = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    j_idx = np.arange(n)[None, :]       # (1, n)
    diff = k_idx - j_idx                # (2n-1, n)
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    A[valid] = f[diff[valid]]
    return A

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. A@ g <= M, g >= 0"""
    n = len(f)
    A = build_A(n, f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

def lp_optimize(f0, n, max_time, rng):
    """Run LP-based optimization starting from f0"""
    f = f0.copy()
    f = np.maximum(f, 0)
    f = f / np.sum(f) * n  # normalize to sum = n
    
    f_score = evaluate_sequence(f.tolist())
    best_f = f.copy()
    best_score = f_score
    
    t = 0.5
    stall = 0
    
    start = time.monotonic()
    while time.monotonic() - start < max_time:
        c = np.convolve(f, f)
        M = np.max(c)
        
        g = solve_lp(f, M)
        if g is None:
            t *= 0.5
            if t < 0.01:
                break
            continue
        
        # Try to improve
        improved = False
        for trial_t in [t, 0.5*t, 0.1*t]:
            f_new = (1 - trial_t) * f + trial_t * g
            f_new = np.maximum(f_new, 0)
            s = np.sum(f_new)
            if s < 1e-10:
                continue
            f_new = f_new / s * n
            
            score = evaluate_sequence(f_new.tolist())
            if score < f_score - 1e-10:
                f = f_new
                f_score = score
                improved = True
                if score < best_score:
                    best_score = score
                    best_f = f.copy()
                break
        
        if not improved:
            stall += 1
            t = max(0.01, t * 0.7)
            if stall > 30:
                break
        else:
            stall = 0
    
    return best_f, best_score

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5
    
    best_score = np.inf
    best_seq = None
    
    # Try different lengths and initializations
    configs = [
        (256, 'random'),
        (512, 'random'),
        (512, 'gaussian'),
        (1024, 'random'),
        (1024, 'gaussian'),
    ]
    
    for n, init_type in configs:
        if time.monotonic() > deadline:
            break
        
        remaining = deadline - time.monotonic()
        time_per_config = min(remaining / 2, 200)  # allocate time
        
        if init_type == 'random':
            f0 = rng.random(n) + 0.1
        elif init_type == 'gaussian':
            x = np.arange(n)
            f0 = np.exp(-0.5 * ((x - n/2) / (n/4))**2)
        else:
            f0 = np.ones(n)
        
        f0 = f0 / np.sum(f0) * n
        
        f, score = lp_optimize(f0, n, time_per_config, rng)
        
        if score < best_score:
            best_score = score
            best_seq = f
            print(f"New best: n={n}, init={init_type}, score={score:.6f}")
    
    if best_seq is None:
        best_seq = np.ones(256)
    
    return [float(x) for x in best_seq]
```
