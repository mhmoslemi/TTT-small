```
def sqrt_complex_spectrum(D):
    Pnew = np.empty_like(D, dtype=np.complex128)
    Pnew[0] = np.sqrt(max(float(D[0].real),0.0))
    if len(D)>1:
        Pnew[1:-1] = np.sqrt(D[1:-1] + 0j)

            ny = float(D[-1].real)
            Pnew[-1] = np.sqrt(ny) if ny > 0 else 0.0
    return Pnew
```

```
def make_projection(p, P, C, n, N, lam, lower_ratio, upper_ratio, method, mix):
    m=C.size
    if m <=0: return None
    mu=1.0/m
    lam = min(1.0, max(0.0, float(lam)))
    target = (1.0-lam)*C + lam*mu
    lower = mu*max(0.0, float(lower_ratio))
    target = np.maximum(target, lower)
    if upper_ratio is not None:
        upper = mu*float(upper_ratio)
        if upper > lower:
            target = np.minimum(target, upper)
    # ensure no NaN
    target = np.where(np.isfinite(target), target, mu)
    # If all zero, flat
    s = float(target.sum())
    if s <= 1e-12 or not np.isfinite(s):
        target = np.full(m, mu, dtype=np.float64)
    full = np.zeros(N, dtype=np.float64)
    full[:m] = target
    D = np.fft.rfft(full)
    if method == 0:
        Pnew = make_sqrt_spectrum(D)
    else:
        amp = np.sqrt(np.abs(D))
        pa = np.abs(P)
        phase = np.zeros(P.shape, dtype=np.float64)
        mask = pa > 1e-12
        phase[mask] = np.angle(P[mask])
        Pnew = amp * np.exp(1j*phase)
    x = np.fft.irfft(Pnew, N).real[:n]
    x = np.where(np.isfinite(x), x, 0.0)
    xp = np.maximum(x, 0.0)
    ax = np.abs(x)
    if xp.sum() <= 1e-12 or (ax.size > 0 and xp.sum() < 0.25*ax.sum()):
        xp = ax
    xp = np.clip(xp, 0.0, 1000.0)
    s = float(xp.sum())
    if s <= 1e-12:
        return None
    xp /= s
    if mix is not None and mix < 1.0:
        xp = (1.0 - mix)*p + mix*xp
        xp = np.maximum(xp, 0.0)
        s = float(xp.sum())
        if s <= 1e-12: return None
        xp /= s
    return xp
```

```
def make_gradient(p, P, C, n, N, beta, eta, mix):
    m=C.size
    if m<=0: return None
    maxC=float(np.max(C))
    if maxC <= 1e-300: return None
    beta=float(beta)
    if beta <= 0: beta=1.0
    temp=maxC/beta
    if temp <= 0: temp=1e-300
    vals=(C-maxC)/temp
    w=np.exp(np.clip(vals, -50.0, 0.0))
    full=np.zeros(N, dtype=np.float64)
    full[:m]=w
    W=np.fft.rfft(full)
    g=np.fft.irfft(np.conj(P)*W, N).real[:n]

    centered -= centered.mean()

    if pos <= 1e-300: return None

    # scale step to a fraction of mean

    q = p - step*centered

    q = np.maximum(q, 0.0)
    s=float(q.sum())
```

```
def resample_sequence(seq, n):
```

```python
"""
    Search for non-negative step-function heights minimizing
      J = 2*n*max(convolve(a,a)) / sum(a)^2
    using FFT-based convolution, spectral projection toward a flattened
    convolution, and soft-subgradient mirror/additive updates. Several
    lengths and random/deterministic initializations are tried.
"""

import numpy as np
import time


def next_pow2(x):
    if x <= 1:
        return 1
    n = 1
    while n < x:
        n <<= 1
    return n


def normalize_array(a):
    if a is None:
        return None
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def fast_score_norm(p, n, N):
    P = np.fft.rfft(p, N)
    C = np.fft.irfft(P * P, N).real[:2 * n - 1]
    if C.size == 0:
        return np.inf
    mc = float(np.max(C))
    if not np.isfinite(mc) or mc < 0.0:
        mc = 0.0
    return 2.0 * float(n) * mc


def fast_score_raw(a):
    p = normalize_array(a)
    if p is None:
        return np.inf
    n = p.size
    N = next_pow2(2 * n - 1)
    return fast_score_norm(p, n, N)


def convolution_data(p, n, N):
    P = np.fft.rfft(p, N)
    C = np.fft.irfft(P * P, N).real[:2 * n - 1]
    if C.size == 0:
        mc = 0.0
    else:
        mc = float(np.max(C))
        if not np.isfinite(mc) or mc < 0.0:
            mc = 0.0
    return P, C, mc


def make_sqrt_spectrum(D):
    P = np.empty(D.shape, dtype=np.complex128)
    nD = D.size
    if nD == 0:
        return P
    v0 = float(np.real(D[0]))
    P[0] = np.sqrt(v0) if v0 > 0.0 else 0.0
    if nD > 1:
        mid = D[1:-1]
        if mid.size > 0:
            P[1:-1] = np.sqrt(mid.astype(np.complex128))
        vy = float(np.real(D[-1]))
        P[-1] = np.sqrt(vy) if vy > 0.0 else 0.0
    return P


def make_projection(p, P, C, n, N, lam, lower_ratio, upper_ratio, method, mix):
    m = C.size
    if m <= 0:
        return None
    mu = 1.0 / float(m)
    lam = min(1.0, max(0.0, float(lam)))
    target = (1.0 - lam) * C + lam * mu
    target = np.where(np.isfinite(target), target, mu)

    lower = mu * max(0.0, float(lower_ratio))
    target = np.maximum(target, lower)
    if upper_ratio is not None:
        upper = mu * float(upper_ratio)
        if upper > lower:
            target = np.minimum(target, upper)

    s = float(target.sum())
    if not np.isfinite(s) or s <= 1e-12:
        target = np.full(m, mu, dtype=np.float64)

    full = np.zeros(N, dtype=np.float64)
    full[:m] = target
    D = np.fft.rfft(full)

    if method == 0:
        Pnew = make_sqrt_spectrum(D)
    else:
        amp = np.sqrt(np.abs(D))
        pa = np.abs(P)
        phase = np.zeros(P.shape, dtype=np.float64)
        mask = pa > 1e-12
        if mask.any():
            phase[mask] = np.angle(P[mask])
        Pnew = amp * np.exp(1j * phase)

    x = np.fft.irfft(Pnew, N).real[:n]
    x = np.where(np.isfinite(x), x, 0.0)

    # If N is exactly 2*n, recover useful tail mass by folding once.
    if N >= 2 * n:
        tail = x[n:min(2 * n, N)]
        if tail.size == n:
            head_pos = np.maximum(x, 0.0)
            tail_pos = np.maximum(tail, 0.0)
            if tail_pos.sum() > 0.05 * max(head_pos.sum(), 1e-300):
                x = head_pos + tail_pos

    xp = np.maximum(x, 0.0)
    ax = np.abs(x)
    if xp.sum() <= 1e-12 or xp.sum() < 0.25 * max(ax.sum(), 1e-300):
        xp = ax
    xp = np.clip(xp, 0.0, 1000.0)
    s = float(xp.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    xp /= s

    if mix is not None and mix < 1.0:
        xp = (1.0 - float(mix)) * p + float(mix) * xp
        xp = np.maximum(xp, 0.0)
        s = float(xp.sum())
        if not np.isfinite(s) or s <= 1e-12:
            return None
        xp /= s
    return xp


def make_gradient(p, P, C, n, N, beta, eta, mix):
    m = C.size
    if m <= 0:
        return None
    maxC = float(np.max(C))
    if not np.isfinite(maxC) or maxC <= 1e-300:
        return None

    beta = max(1e-12, float(beta))
    temp = maxC / beta
    if temp <= 1e-300:
        temp = 1e-300
    vals = (C - maxC) / temp
    w = np.exp(np.clip(vals, -50.0, 0.0))
    full = np.zeros(N, dtype=np.float64)
    full[:m] = w
    W = np.fft.rfft(full)
    g = np.fft.irfft(np.conj(P) * W, N).real[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    gm = float(np.max(g))
    if gm <= 1e-300:
        return None

    centered = g / gm
    centered -= centered.mean()
    pos = float(np.max(centered))
    if pos <= 1e-300:
        return None

    eta = max(0.0, float(eta))
    if eta == 0.0:
        return p

    step = eta * (float(np.mean(p)) / pos)
    if not np.isfinite(step) or step <= 0.0:
        step = eta * max(float(np.mean(p)), 1e-12)

    q = p - step * centered
    q = np.where(np.isfinite(q), q, 0.0)
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s <= 1e-12:
        q = p * np.exp(-np.clip(eta * centered, -50.0, 50.0))
        q = np.maximum(q, 0.0)
        s = float(q.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    q /= s

    if mix is not None and mix < 1.0:
        q = (1.0 - float(mix)) * p + float(mix) * q
        q = np.maximum(q, 0.0)
        s = float(q.sum())
        if s <= 1e-12:
            return None
        q /= s
    return q


def make_random(p, n, rng, sigma):
    sigma = float(sigma)
    if sigma <= 0.0:
        return p.copy()
    noise = rng.normal(0.0, sigma, n)
    q = p * np.exp(noise)
    q = np.where(np.isfinite(q), q, 0.0)
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s <= 1e-12:
        q = p + sigma * rng.normal(0.0, 1.0, n)
        q = np.where(np.isfinite(q), q, 0.0)
        q = np.maximum(q, 0.0)
        s = float(q.sum())
    if s <= 1e-12:
        return p.copy()
    return q / s


def make_sym(p):
    q = 0.5 * (p + p[::-1])
    q = np.maximum(q, 0.0)
    s = float(q.sum())
    if s <= 1e-12:
        return p.copy()
    return q / s


def make_block(p, n, rng, strength):
    if n <= 0:
        return p
    denom = int(rng.integers(8, 64)) + 1
    width = max(1, int(n // denom))
    max_start = max(1, n - width + 1)
    start = int(rng.integers(0, max_start))
    stop = min(n, start + width)
    mult = float(np.exp(rng.normal(0.0, float(strength))))
    q = p.copy()
    q[start:stop] *= mult
    return normalize_array(q)


def smooth_random_sequence(n, rng, sigma):
    i = np.arange(n, dtype=np.float64)
    K = min(16, max(2, n // 2))
    s = np.zeros(n, dtype=np.float64)
    for k in range(1, K + 1):
        amp = rng.normal(0.0, 1.0) / (1.0 + 0.1 * float(k))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        s += amp * np.cos(2.0 * np.pi * float(k) * i / float(n) + ph)
    p = 1.0 + float(sigma) * s
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.05)
    return p


def resample_sequence(seq, n):
    seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    if seq.size == 0:
        return np.ones(n, dtype=np.float64)
    if seq.size == n:
        return seq
    old = np.linspace(0.0, 1.0, seq.size, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    return np.interp(new, old, seq)


def base_initializations(n, rng, global_seq):
    out = []
    i = np.arange(n, dtype=np.float64)
    x = i / float(max(1, n - 1))

    raw = []
    raw.append(np.ones(n, dtype=np.float64))

    for amp in (0.3, 0.6, 0.9):
        raw.append(1.0 + amp * np.cos(2.0 * np.pi * i / float(n)))
        raw.append(1.0 - amp * np.cos(2.0 * np.pi * i / float(n)))
        raw.append(1.0 + amp * np.sin(2.0 * np.pi * i / float(n)))
        raw.append(1.0 - amp * np.sin(2.0 * np.pi * i / float(n)))

    for alpha in (0.3, 1.0, 3.0, 10.0):
        raw.append(np.exp(-alpha * x))
        raw.append(np.exp(-alpha * (1.0 - x)))
        raw.append(np.exp(-alpha * np.minimum(x, 1.0 - x)))

    eps = 1e-3
    for pw in (0.5, 1.0, 2.0, 4.0):
        raw.append(np.power(x + eps, -pw))
        raw.append(np.power(1.0 - x + eps, -pw))

    for f in (0.1, 0.2, 0.3, 0.5):
        kmax = max(1, int(f * n))
        for r in (0.2, 0.5, 2.0, 5.0):
            a = np.empty(n, dtype=np.float64)
            a[:kmax] = r
            a[kmax:] = 1.0
            raw.append(a)

    for sigma in (0.1, 0.3, 0.6, 1.0, 2.0):
        raw.append(smooth_random_sequence(n, rng, sigma))

    if global_seq is not None:
        gs = np.asarray(global_seq, dtype=np.float64).reshape(-1)
        gs = np.where(np.isfinite(gs), gs, 0.0)
        gs = np.clip(gs, 0.0, 1000.0)
        if gs.size > 0:
            raw.append(resample_sequence(gs, n))

    for a in raw:
        q = normalize_array(a)
        if q is not None:
            out.append(q)

    if not out:
        out.append(np.full(n, 1.0 / float(n), dtype=np.float64))
    return out


def optimize_sequence(p0, n, budget, seed, target_ratio=1.45, verbose=False):
    p0 = normalize_array(p0)
    if p0 is None:
        return None, np.inf

    start_t = time.monotonic()
    deadline = start_t + max(0.0, float(budget))
    N = next_pow2(2 * n - 1)
    rng = np.random.default_rng(int(seed))

    cur_p = p0.copy()
    cur_fast = fast_score_norm(cur_p, n, N)
    best_p = cur_p.copy()
    best_fast = cur_fast

    lam = 0.35
    beta = 8.0
    eta = 0.25
    no_imp = 0
    it = 0

    while time.monotonic() < deadline:
        # If the working point has drifted too far from the best, inject noise
        # around the best point to keep the search efficient.
        if cur_fast > best_fast + 0.05:
            q = make_random(best_p, n, rng, 0.05)
            if q is not None:
                cur_p = q
                cur_fast = fast_score_norm(cur_p, n, N)

        P, C, maxC = convolution_data(cur_p, n, N)
        cur_fast = 2.0 * float(n) * maxC

        if cur_fast < best_fast:
            best_fast = cur_fast
            best_p = cur_p.copy()
            no_imp = 0

        r = maxC * (2.0 * float(n) - 1.0)
        if not np.isfinite(r) or r < 1.0:
            r = 1.0

        target_ratio = max(1.05, float(target_ratio))
        upper = None
        if r > target_ratio + 0.05:
            upper = max(target_ratio, min(r - 0.10, 0.98 * r))

        cands = []

        lam1 = lam
        lam2 = min(1.0, lam + 0.25)

        q = make_projection(cur_p, P, C, n, N, lam1, 0.10, upper, 0, 0.70)
        if q is not None:
            cands.append(q)

        q = make_projection(cur_p, P, C, n, N, lam2, 0.10, upper, 1, 0.70)
        if q is not None:
            cands.append(q)

        q = make_projection(cur_p, P, C, n, N, 1.0, 0.15, upper, 0, 0.50)
        if q is not None:
            cands.append(q)

        q = make_gradient(cur_p, P, C, n, N, beta, eta, 0.80)
        if q is not None:
            cands.append(q)

        q = make_gradient(cur_p, P, C, n, N, min(300.0, beta * 3.0), eta, 0.80)
        if q is not None:
            cands.append(q)

        if it % 18 == 0:
            q = make_random(cur_p, n, rng, 0.15)
            if q is not None:
                cands.append(q)
            q = make_sym(cur_p)
            if q is not None:
                cands.append(q)
            q = make_block(cur_p, n, rng, 0.25)
            if q is not None:
                cands.append(q)

        accepted = False
        cand_best_score = cur_fast
        cand_best = None

        for q in cands:
            q = normalize_array(q)
            if q is None:
                continue
            qs = fast_score_norm(q, n, N)
            if np.isfinite(qs) and qs < cand_best_score:
                cand_best_score = qs
                cand_best = q
                if qs < best_fast:
                    best_fast = qs
                    best_p = q.copy()
                    no_imp = 0
                    if verbose and qs < 1.5027:
                        print(f"    n={n} new fast best {qs:.6f} at it {it}")

        if cand_best is not None:
            cur_p = cand_best
            cur_fast = cand_best_score
            accepted = True
            no_imp = 0
        else:
            no_imp += 1
            if no_imp == 10:
                lam = min(1.0, lam * 1.25 + 0.05)
                eta = 0.40 if eta < 0.40 else 0.10
                beta = beta * 1.5 if beta < 150.0 else 5.0
            if no_imp >= 25:
                sigma = 0.05 + 0.30 * min(1.0, float(no_imp) / 100.0)
                q = make_random(best_p, n, rng, sigma)
                if q is not None:
                    cur_p = q
                    cur_fast = fast_score_norm(cur_p, n, N)
                no_imp = 0
                lam = 0.50
                beta = 10.0
                eta = 0.30

        it += 1

    return best_p, best_fast


def clean_final_sequence(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        a = np.ones(256, dtype=np.float64)
        s = float(a.sum())
    return a / s


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(int(seed))
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 25.0)

    try:
        global_seq = height_sequence_1
    except NameError:
        global_seq = None

    lengths = []
    for L in (4096, 2048, 1024):
        if L not in lengths:
            lengths.append(L)

    if global_seq is not None:
        try:
            Lg = len(global_seq)
        except TypeError:
            Lg = 0
        if 512 <= Lg <= 8192 and Lg not in lengths:
            lengths.append(Lg)

    tasks = []
    for L in lengths:
        inits = base_initializations(L, rng, global_seq)
        scored = []
        for idx, a in enumerate(inits):
            sc = fast_score_raw(a)
            if np.isfinite(sc):
                scored.append((float(sc), idx, a))
        scored.sort()
        top = scored[:2]
        if not top:
            top = [(np.inf, -1, np.full(L, 1.0 / float(L), dtype=np.float64))]
        for sc, idx, a in top:
            tasks.append((L, a))

    if not tasks:
        tasks = [(2048, np.ones(2048, dtype=np.float64))]

    best_p = None
    best_exact = np.inf
    best_fast = np.inf

    ntasks = len(tasks)
    for ti, (L, a0) in enumerate(tasks):
        if time.monotonic() > deadline - 15.0:
            break

        remaining = deadline - time.monotonic()
        task_budget = remaining / float(max(1, ntasks - ti))
        if task_budget > 420.0:
            task_budget = 420.0
        if task_budget < 20.0:
            task_budget = 20.0

        tr = 1.45
        if np.isfinite(best_exact) and best_exact < 1.5027:
            tr = 1.40

        p, fs = optimize_sequence(a0, L, task_budget, int(seed) + 1000 * ti, tr, False)
        if p is None:
            continue

        if np.isfinite(fs) and fs < best_fast:
            best_fast = fs
            if not np.isfinite(best_exact):
                best_p = p.copy()

        check_exact = False
        if np.isfinite(fs):
            if not np.isfinite(best_exact):
                check_exact = True
            elif fs < best_exact + 0.05:
                check_exact = True
            elif fs < 1.60:
                check_exact = True
            elif ntasks - ti <= 1:
                check_exact = True

        if check_exact:
            try:
                ex = evaluate_sequence([float(x) for x in p])
            except Exception:
                ex = np.inf
            if np.isfinite(ex):
                best_exact = float(ex)
                best_p = p.copy()

        print(f"task={ti} n={L} fast={fs:.6f} best_exact={best_exact:.6f}")

    # If the initial round did not beat the target convincingly, spend remaining
    # time trying a flatter target on a power-of-two length.
    if time.monotonic() < deadline - 35.0 and (not np.isfinite(best_exact) or best_exact > 1.5027):
        L2 = 4096
        a0 = None
        if best_p is not None and len(best_p) == L2:
            a0 = best_p.copy()
        elif best_p is not None:
            a0 = resample_sequence(best_p, L2)

        extras = []
        if a0 is not None:
            extras.append(a0)
        extras.append(smooth_random_sequence(L2, rng, 0.5))
        extras.append(smooth_random_sequence(L2, rng, 1.2))
        extras.append(resample_sequence(np.ones(1, dtype=np.float64), L2))

        for ei, a0 in enumerate(extras):
            if time.monotonic() > deadline - 20.0:
                break
            rem = deadline - time.monotonic()
            budget = max(20.0, rem / float(max(1, len(extras) - ei)))
            budget = min(budget, 350.0)
            p, fs = optimize_sequence(a0, L2, budget, int(seed) + 9000 + ei, 1.40, False)
            if p is None:
                continue
            try:
                ex = evaluate_sequence([float(x) for x in p])
            except Exception:
                ex = np.inf
            if np.isfinite(ex):
                if ex < best_exact:
                    best_exact = float(ex)
                    best_p = p.copy()
                print(f"extra={ei} n={L2} fast={fs:.6f} exact={ex:.6f}")

    if best_p is None:
        best_p = np.ones(256, dtype=np.float64)

    best_p = clean_final_sequence(best_p)

    try:
        final_exact = evaluate_sequence([float(x) for x in best_p])
    except Exception:
        final_exact = np.inf

    if not np.isfinite(final_exact):
        best_p = clean_final_sequence(np.ones(256, dtype=np.float64))
        try:
            final_exact = evaluate_sequence([float(x) for x in best_p])
        except Exception:
            final_exact = np.inf

    print(f"final length={best_p.size} final exact={final_exact:.6f}")
    return [float(x) for x in best_p]
```
