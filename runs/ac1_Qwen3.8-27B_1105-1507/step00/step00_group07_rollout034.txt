"""
Minimizes the self-convolution peak score using FFT-accelerated mirror descent
with log-sum-exp smoothing. Explores multiple initializations including
the parent sequence, random, and windowed sequences.
"""

import numpy as np
import time


def fast_self_convolve(a):
    """Compute aperiodic self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)
    return conv[:size]


def compute_score_direct(a):
    """Compute score = 2n * max(conv) / sum(a)^2."""
    n = len(a)
    conv = fast_self_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-10:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))


def get_gradient(a, conv, epsilon):
    """Compute gradient of (1/epsilon)*log(sum exp(c[k]/epsilon)) w.r.t. a."""
    n = len(a)
    size = 2 * n - 1
    
    # Softmax weights (numerically stable)
    max_c = conv.max()
    exp_c = np.exp((conv - max_c) / epsilon)
    w = exp_c / exp_c.sum()
    
    # grad[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]
    # Using FFT: convolve(a, w[::-1]) and extract appropriate slice
    # Result length: n + size - 1 = 3n - 2
    # grad[i] = 2 * result[2n-2-i] for i=0,...,n-1
    
    conv_len = n + size - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    fw = np.fft.rfft(w[::-1], fft_size)
    result = np.fft.irfft(fa * fw, fft_size)[:conv_len]
    
    # Extract: we need result[2n-2], result[2n-3], ..., result[n-1]
    # That's result[n-1 : 2n-1] reversed
    grad = 2.0 * result[n-1:2*n-1][::-1]
    return grad


def optimize_sequence(a, time_budget, verbose=True, seed=0):
    """Run mirror descent optimization to minimize the convolution peak score."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a = a.astype(np.float64).copy()
    
    # Normalize to sum = 1
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        return a
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score_direct(best_a)
    
    eta = 0.5  # learning rate
    epsilon = 0.01  # log-sum-exp temperature
    eta_decay = 0.9995
    epsilon_decay = 0.999
    
    deadline = time.monotonic() + time_budget
    step = 0
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() < deadline:
        step += 1
        
        conv = fast_self_convolve(a)
        grad = get_gradient(a, conv, epsilon)
        
        # Normalize gradient for stability
        g_max = grad.max()
        if g_max <= 0:
            g_max = 1.0
        grad_norm = grad / g_max
        
        # Multiplicative update
        log_a = np.log(a + 1e-30) - eta * grad_norm
        log_a = log_a - log_a.max()  # for numerical stability
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-10:
            a_new = a.copy()
        
        score = compute_score_direct(a_new)
        
        if score < best_score:
            best_a = a_new.copy()
            best_score = score
            no_improve = 0
            if verbose and step % 1000 == 0:

            if no_improve >= max_no_improve:

                a = a + 0.05 * rng.standard_normal(n) * a.mean()
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

        a = a_new  # Always move to the new point

        epsilon = max(epsilon * epsilon_decay, 0.001)

            a = a + 0.01 * rng.standard_normal(n) * a.mean()
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a

    print(f"Final score (via evaluate_sequence): {final_score:.6f}")
    
    return [float(x) for x in best_overall]
