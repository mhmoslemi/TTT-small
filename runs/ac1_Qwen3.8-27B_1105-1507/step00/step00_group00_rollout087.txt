```
def project_simplex(x):
    x=np.asarray(x,float).copy()
    if x.size==0: return x
    if not np.all(np.isfinite(x)): x=np.where(np.isfinite(x),x,0.0)
    v = x
    u = np.sort(v)[::-1]

    return w/s
```

```
def param_p(params, basis, amp=0.99):
    z = basis @ params
    z = np.clip(z, -20.0, 20.0)
    th = np.tanh(z)
    f = 1.0 + amp * th
    f = np.maximum(f, 1e-12)
    s = f.sum()
    if s <= 1e-12: return np.full(basis.shape[0], 1.0/basis.shape[0])
    return f/s

def param_gradient(params, basis, g, p, amp=0.99):
    z = basis @ params
    th = np.tanh(z)
    f = 1.0 + amp*th
    f = np.maximum(f,1e-12)
    # fprime per dim: amp*(1-th^2)*basis
    fprime_scale = amp * (1.0 - th*th)  # n
    ratios = (basis * fprime_scale[:, None]) / f[:, None]
    pg = p * g
    first = ratios.T @ pg
    second = ratios.T @ p
    grad = first - float(pg.sum()) * second
    # numerical clean
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad
```

```python
import time
import numpy as np


def normalize_nonneg(a):
    """Clip to nonnegative finite values and normalize to sum one."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        a = np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def fast_convolution(p):
    """Linear self-convolution by FFT, length 2n-1."""
    n = int(p.size)
    if n <= 0:
        return np.array([0.0], dtype=np.float64)
    target = 2 * n - 1
    L = 1 << (target - 1).bit_length()
    f = np.fft.rfft(p, L)
    c = np.fft.irfft(f * f, L)[:target]
    return c


def fast_score(p):
    """Fast approximation of evaluate_sequence, used for local search."""
    c = fast_convolution(p)
    if c.size == 0:
        return 0.0
    mc = float(np.max(c))
    if not np.isfinite(mc) or mc <= 0.0:
        return np.inf
    return float(2.0 * p.size * mc)


def exact_score_of(seq):
    """Call the authoritative evaluator safely."""
    try:
        sc = evaluate_sequence(seq)
    except Exception:
        return np.inf
    try:
        sc = float(sc)
    except Exception:
        return np.inf
    if not np.isfinite(sc):
        return np.inf
    return sc


def select_active(c, K):
    """Select active lag indices near the largest convolution values."""
    c = np.asarray(c, dtype=np.float64)
    L = int(c.size)
    if L == 0:
        return np.array([0], dtype=int), np.array([1.0])
    K = int(min(max(1, K), L))
    if not np.all(np.isfinite(c)):
        return np.arange(L), np.full(L, 1.0 / L)
    maxc = float(np.max(c))
    if maxc <= 0.0:
        return np.array([0], dtype=int), np.array([1.0])
    tol = max(1e-12, 1e-7 * maxc)
    near = np.where(c >= maxc - tol)[0]
    if near.size == 0:
        near = np.array([int(np.argmax(c))], dtype=int)
    if near.size > K:
        vals = c[near]
        order = np.argpartition(vals, -K)[-K:]
        top = near[order]
    else:
        top = near
    vals = c[top]
    s = float(vals.sum())
    if not np.isfinite(s) or s <= 0.0:
        weights = np.full(top.size, 1.0 / top.size)
    else:
        weights = vals / s
    return top.astype(int), weights


def active_gradient(p, top, weights):
    """Gradient of a weighted sum of selected convolution lags."""
    n = int(p.size)
    g = np.zeros(n, dtype=np.float64)
    for idx in range(top.size):
        k = int(top[idx])
        w = float(weights[idx])
        if w == 0.0:
            continue
        hi = n - 1 if k > n - 1 else k
        lo = 0 if k - n + 1 < 0 else k - n + 1
        if hi >= lo:
            start = k - hi
            stop = k - lo
            g[lo:hi + 1] += w * p[start:stop + 1][::-1]
    return g


def project_simplex(x):
    """Euclidean projection onto {u >= 0, sum u = 1}."""
    x = np.asarray(x, dtype=np.float64).copy()
    if x.size == 0:
        return x
    x = np.where(np.isfinite(x), x, 0.0)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - 1.0
    ar = np.arange(1, x.size + 1, dtype=np.float64)
    ind = (u - cssv / ar) > 0
    rho = int(ind.sum())
    if rho == 0:
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    theta = cssv[rho - 1] / float(rho)
    w = np.maximum(x - theta, 0.0)
    s = float(w.sum())
    if s <= 1e-15:
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)
    return w / s


def multiplicative_update(p, g, rate, floor):
    """Multiplicative nonnegative update on the simplex."""
    gmax = float(np.max(g)) if g.size else 0.0
    if not np.isfinite(gmax) or gmax <= 0.0:
        return p.copy()
    gn = g / gmax
    q = p * np.exp(-rate * gn)
    q = np.maximum(q, floor)
    s = float(q.sum())
    if not np.isfinite(s) or s <= 0.0:
        return p.copy()
    return q / s


def smooth_random_factor(n, rng, max_log_amp):
    """Smooth positive random multiplier with controlled low-frequency content."""
    x = np.arange(n, dtype=np.float64)
    z = np.zeros(n, dtype=np.float64)
    freqs = [1, 2, 3, 5, 8, 13]
    for m in freqs:
        amp = float(max_log_amp) / (float(m) ** 0.5)
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        z += amp * np.cos(2.0 * np.pi * float(m) * x / float(n) + phase)
    zm = float(np.max(np.abs(z)))
    if zm > 1e-15:
        z *= float(max_log_amp) / zm
    z = np.clip(z, -8.0, 8.0)
    return np.exp(z)

    best = p.copy()
    best_score = fast_score(p)

        if time.monotonic() > end_time - 0.03:
            break

        K = k_cycle[it % len(k_cycle)]
        top, weights = select_active(c, K)

        improved = False

        if n > 8192:
            mul_rates = [0.12, 0.06, 0.03, 0.012]
        else:
            mul_rates = [0.18, 0.09, 0.045, 0.018, 0.008]

        for rate in mul_rates:
            q = multiplicative_update(p, g, rate, floor)
            qs = fast_score(q)
            if np.isfinite(qs) and qs < best_score - 1e-10:
                p = q
                best = q.copy()
                best_score = qs
                no_improve = 0
                improved = True
                break

        if not improved:
            gmean = float(np.mean(g)) if g.size else 0.0
            dirn = g - gmean
            if n > 8192:
                add_rates = [0.8 / n, 0.2 / n, 0.05 / n]
            else:
                add_rates = [2.0 / n, 0.5 / n, 0.1 / n, 0.02 / n]
            for rate in add_rates:
                q0 = p - rate * dirn
                q = project_simplex(q0)
                q = np.maximum(q, floor)
                q /= q.sum()
                qs = fast_score(q)
                if np.isfinite(qs) and qs < best_score - 1e-10:
                    p = q
                    best = q.copy()
                    best_score = qs
                    no_improve = 0
                    improved = True
                    break

        if not improved:
            amp = 0.25 + 0.45 * min(1.0, no_improve / 40.0)
            factor = smooth_random_factor(n, rng, amp)
            q = p * factor
            q = normalize_nonneg(q)
            q = np.maximum(q, floor)
            q /= q.sum()
            qs = fast_score(q)
            if np.isfinite(qs) and qs < best_score - 1e-10:
                p = q
                best = q.copy()
                best_score = qs
                no_improve = 0
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10 and no_improve % 5 == 0:
                mix = min(0.05, 0.005 * no_improve)
                q = (1.0 - mix) * best + (mix / float(n))
                q = normalize_nonneg(q)
                q = np.maximum(q, floor)
                q /= q.sum()
                qs = fast_score(q)
                if np.isfinite(qs) and qs < best_score - 1e-10:
                    p = q
                    best = q.copy()
                    best_score = qs
                    no_improve = 0
            if no_improve > 220:
                break

        it += 1

    return best, best_score


def make_basis(n, D):
    """Fourier basis: sin(2*pi*m*t), cos(2*pi*m*t) for t=(i+1/2)/n."""
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    basis = np.empty((n, int(D)), dtype=np.float64)
    idx = 0
    m = 1
    while idx < D:
        ang = 2.0 * np.pi * float(m) * t
        if idx < D:
            basis[:, idx] = np.sin(ang)
            idx += 1
        if idx < D:
            basis[:, idx] = np.cos(ang)
            idx += 1
        m += 1
    return basis


def param_p(params, basis, amp=0.99):
    """Convert Fourier parameters to a normalized nonnegative sequence."""
    z = basis @ params
    z = np.clip(z, -20.0, 20.0)
    th = np.tanh(z)
    f = 1.0 + amp * th
    f = np.maximum(f, 1e-12)
    s = float(f.sum())
    if s <= 1e-15:
        return np.full(basis.shape[0], 1.0 / float(basis.shape[0]), dtype=np.float64)
    return f / s


def param_gradient(params, basis, g, p, amp=0.99):
    """Gradient wrt Fourier parameters for an active-lag gradient g."""
    z = basis @ params
    z = np.clip(z, -20.0, 20.0)
    th = np.tanh(z)
    f = 1.0 + amp * th
    f = np.maximum(f, 1e-12)
    scale = amp * (1.0 - th * th)
    ratios = (basis * scale[:, None]) / f[:, None]
    pg = p * g
    first = ratios.T @ pg
    second = ratios.T @ p
    grad = first - float(pg.sum()) * second
    return np.where(np.isfinite(grad), grad, 0.0)


def optimize_params(params, basis, budget_s, rng, end_time=None):
    """Steepest descent with backtracking in a finite Fourier coefficient space."""
    params = np.asarray(params, dtype=np.float64).copy()
    p = param_p(params, basis)
    start = time.monotonic()
    if end_time is None:
        end_time = start + max(0.0, float(budget_s))
    best_params = params.copy()
    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0
    base_step = 0.25
    k_cycle = [1, 1, 3, 8, 16]
    it = 0

    while time.monotonic() < end_time:
        if time.monotonic() > end_time - 0.03:
            break
        c = fast_convolution(p)
        K = k_cycle[it % len(k_cycle)]
        top, weights = select_active(c, K)
        g = active_gradient(p, top, weights)
        grad = param_gradient(params, basis, g, p)
        nrm = float(np.linalg.norm(grad))
        if not np.isfinite(nrm) or nrm <= 1e-14:
            break
        dirn = grad / nrm
        improved = False
        steps = [base_step, 0.5 * base_step, 0.25 * base_step,
                 0.1 * base_step, 0.03 * base_step, 0.01 * base_step]

        for st in steps:
            new_params = params - st * dirn
            ln = float(np.linalg.norm(new_params))
            if not np.isfinite(ln) or ln > 25.0:
                if ln > 0.0:
                    new_params *= 25.0 / ln
                else:
                    new_params = params.copy()
            new_p = param_p(new_params, basis)
            sc = fast_score(new_p)
            if np.isfinite(sc) and sc < best_score - 1e-10:
                params = new_params
                p = new_p
                best_params = params.copy()
                best_p = p.copy()
                best_score = sc
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                jump = 0.05 + 0.02 * min(20, no_improve // 10)
                new_params = params + rng.normal(0.0, jump, size=params.size)
                ln = float(np.linalg.norm(new_params))
                if ln > 25.0 and ln > 0.0:
                    new_params *= 25.0 / ln
                new_p = param_p(new_params, basis)
                sc = fast_score(new_p)
                if np.isfinite(sc) and sc < best_score - 1e-10:
                    params = new_params
                    p = new_p
                    best_params = params.copy()
                    best_p = p.copy()
                    best_score = sc
                    no_improve = 0
            if no_improve % 80 == 0:
                base_step = max(0.02, min(3.0, base_step * 0.5))
            if no_improve > 240:
                break

        it += 1

    return best_params, best_p, best_score


def random_params(D, rng):
    """Random Fourier parameter seed with stronger low-frequency sine modes."""
    D = int(D)
    idx = np.arange(D, dtype=np.float64)
    base = 0.85 * np.exp(-idx * 0.17)
    base[1::2] *= 0.35
    params = rng.normal(0.0, base)
    if rng.random() < 0.35:
        j = int(rng.integers(0, D))
        params[j] += float(rng.choice([-1.0, 1.0])) * float(rng.uniform(0.5, 1.4))
    return np.clip(params, -5.0, 5.0)


def structured_params(D):
    """Deterministic seeds emphasizing the first antisymmetric sine mode."""
    D = int(D)
    out = []
    a_vals = [0.15, 0.30, 0.45, 0.60, 0.75, 0.90]
    b_vals = [0.0, -0.12, -0.25, -0.40]
    c_vals = [0.0, -0.10, 0.12]
    for a in a_vals:
        for b in b_vals:
            for c in c_vals:
                p = np.zeros(D, dtype=np.float64)
                p[0] = a
                if D > 2:
                    p[2] = b
                if D > 4:
                    p[4] = c
                out.append(p)
    return out


def run_param_plan(n, D, total_budget, rng):
    """Search a low-dimensional Fourier-sequence family and then refine in full space."""
    n = int(n)
    D = int(D)
    start = time.monotonic()
    total_budget = max(0.0, float(total_budget))
    end_time = start + total_budget
    if total_budget <= 1.0:
        p = np.full(n, 1.0 / float(n), dtype=np.float64)
        return p, fast_score(p)

    basis = make_basis(n, D)

    if total_budget <= 100.0:
        seed_deadline = start + min(8.0, 0.15 * total_budget)
        seed_count = 2
        param_f = 0.55
    else:
        seed_deadline = start + min(30.0, 0.15 * total_budget)
        seed_count = 4
        param_f = 0.62

    seeds = []
    for pp in structured_params(D):
        sp = param_p(pp, basis)
        sc = fast_score(sp)
        if np.isfinite(sc):
            seeds.append((sc, len(seeds), pp.copy(), sp.copy()))

    max_random_seeds = 2500 if n <= 4096 else 1200
    while len(seeds) < max_random_seeds and time.monotonic() < seed_deadline:
        pp = random_params(D, rng)
        sp = param_p(pp, basis)
        sc = fast_score(sp)
        if np.isfinite(sc):
            seeds.append((sc, len(seeds), pp.copy(), sp.copy()))

    if not seeds:
        p = np.full(n, 1.0 / float(n), dtype=np.float64)
        return p, fast_score(p)

    seeds.sort()
    top = seeds[:seed_count]

    param_opt_end = start + max(5.0, param_f * total_budget)
    best_p = None
    best_score = np.inf

    for i, (sc, sid, pp, sp) in enumerate(top):
        remaining = param_opt_end - time.monotonic()
        if remaining < 5.0:
            break
        sub = min(remaining / max(1, len(top) - i), 180.0)
        if sub < 2.0:
            continue
        _, op, oscore = optimize_params(pp, basis, sub, rng)
        if np.isfinite(oscore) and oscore < best_score:
            best_score = oscore
            best_p = op
        if time.monotonic() >= param_opt_end - 1.0:
            break

    if best_p is None:
        best_p = top[0][3]
        best_score = top[0][0]

    remaining_full = end_time - time.monotonic()
    if remaining_full > 10.0:
        best_p, best_score = optimize_full(best_p, min(remaining_full, 0.38 * total_budget), rng)

    return best_p, best_score


def sine_factor(n, a1=0.55, a2=-0.25, a3=0.05):
    """Explicit antisymmetric multiplicative factor."""
    t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    f = (1.0
         + a1 * np.sin(2.0 * np.pi * t)
         + a2 * np.sin(4.0 * np.pi * t)
         + a3 * np.sin(6.0 * np.pi * t))
    return np.maximum(f, 1e-6)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Fourier-seed active-lag descent plus full-vector nonnegative local search.

    The objective is 2n * max(self-convolution) / sum^2.  We normalize to sum 1,
    minimize the peak self-convolution using low-dimensional Fourier parameterization,
    and then refine with full-vector multiplicative/projected active-set updates.
    """
    start = time.monotonic()
    hard_deadline = start + max(60.0, float(budget_s) - 35.0)
    rng = np.random.default_rng(int(seed))

    best_seq = None
    best_score = np.inf

    def consider(seq, label):
        nonlocal best_seq, best_score
        if seq is None:
            return
        try:
            arr = np.asarray(seq, dtype=np.float64)
            if arr.size == 0:
                return
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.maximum(arr, 0.0)
            lst = [float(x) for x in arr]
        except Exception:
            return
        sc = exact_score_of(lst)
        fs = fast_score(arr)
        print(f"{label}: length={arr.size} fast={fs:.10f} exact={sc:.10f}")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = lst

    try:
        parent_raw = height_sequence_1
    except NameError:
        parent_raw = None

    parent_p = None
    if parent_raw is not None:
        try:
            parent_arr = np.asarray(parent_raw, dtype=np.float64)
            if parent_arr.ndim == 1 and parent_arr.size > 0:
                parent_p = normalize_nonneg(parent_arr)
                consider(parent_p, "parent")
        except Exception:
            parent_p = None

    # Quick low-dimensional search at moderate length.
    if time.monotonic() < hard_deadline - 40:
        b = min(260.0, max(20.0, hard_deadline - time.monotonic() - 60.0))
        seq, fs = run_param_plan(4096, 12, b, rng)
        consider(seq, "param 4096")

    # Larger smooth-parameter search.
    if time.monotonic() < hard_deadline - 60:
        b = min(380.0, max(20.0, hard_deadline - time.monotonic() - 60.0))
        seq, fs = run_param_plan(8192, 14, b, rng)
        consider(seq, "param 8192")

    # Very large length if time remains.
    if time.monotonic() < hard_deadline - 80:
        b = min(260.0, max(20.0, hard_deadline - time.monotonic() - 60.0))
        seq, fs = run_param_plan(16384, 16, b, rng)
        consider(seq, "param 16384")

    # If parent exists, perturb it before full optimization.
    if parent_p is not None and time.monotonic() < hard_deadline - 60:
        n = int(parent_p.size)
        if n < 64:
            n = 4096
            p0 = np.full(n, 1.0 / n)
        else:
            p0 = parent_p.copy()
        p0 = p0 * sine_factor(n, 0.55, -0.25, 0.05)
        p0 = normalize_nonneg(p0)
        b = min(200.0, max(20.0, hard_deadline - time.monotonic() - 40.0))
        seq, fs = optimize_full(p0, b, rng)
        consider(seq, f"parent full {seq.size}")

    # Small-length fine search can sometimes find discrete motifs.
    if time.monotonic() < hard_deadline - 70:
        b = min(160.0, max(20.0, hard_deadline - time.monotonic() - 40.0))
        seq, fs = run_param_plan(2048, 16, b, rng)
        consider(seq, "param 2048")

    # Final polishing on the current exact best if budget remains.
    if best_seq is not None and time.monotonic() < hard_deadline - 15:
        p = normalize_nonneg(best_seq)
        b = min(180.0, max(10.0, hard_deadline - time.monotonic() - 5.0))
        seq, fs = optimize_full(p, b, rng)
        consider(seq, "final polish")

    if best_seq is None:
        n = 256
        best_seq = [1.0 / float(n) for _ in range(n)]
        best_score = exact_score_of(best_seq)
        print(f"fallback uniform: exact={best_score:.10f}")

    # Guarantee a finite valid list.
    try:
        out = []
        for x in best_seq:
            y = float(x)
            if not np.isfinite(y) or y < 0.0:
                y = 0.0
            if y > 1000.0:
                y = 1000.0
            out.append(y)
        if not out:
            out = [1.0]
        s = float(np.sum(out))
        if s <= 0.01:
            out = [1.0 / 256.0] * 256
        return out
    except Exception:
        return [1.0]
```
