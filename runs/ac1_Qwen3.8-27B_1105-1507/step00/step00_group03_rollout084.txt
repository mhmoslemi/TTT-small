```python
"""
Search for non-negative step-function coefficients minimizing
    2*n*max(conv(a,a)) / (sum(a))**2.

Strategy:
  * Build diverse initial profiles, especially one-sided singular profiles
    near a_i ~ (i+1)^(-1/2), which already beats the uniform baseline.
  * Run several short stochastic projected multiplicative gradient refinements
    on the simplex sum(a)=1.  The smooth maximum of the quadratic convolution
    is minimized with FFT-accelerated convolution and cross-correlation.
  * Keep a population of best sequences, then verify the final candidates with
    the exact evaluator.
"""

import numpy as np
import time


def _pad_for(need: int) -> int:
    """Smallest power of two >= max(1, need)."""
    need = max(1, int(need))
    p = 1
    while p < need:
        p <<= 1
    return p


def _conv_fft(a: np.ndarray, pad: int) -> np.ndarray:
    """Linear convolution a*a for a real non-negative array."""
    n = int(a.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    if n <= 256:
        return np.convolve(a, a, mode="full").astype(np.float64, copy=False)
    fa = np.fft.rfft(a, pad)
    return np.fft.irfft(fa * fa, pad)[: 2 * n - 1].astype(np.float64, copy=False)


def _conv_fft_auto(a: np.ndarray) -> np.ndarray:
    """Choose padding automatically."""
    n = int(a.size)
    return _conv_fft(a, _pad_for(2 * n - 1))


def _cross_fft(w: np.ndarray, a: np.ndarray, pad: int) -> np.ndarray:
    """
    s_i = sum_k w_k * a_{k-i}, for i=0..n-1.
    This is np.correlate(w, a, 'valid') implemented with FFT.
    """
    n = int(a.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    if n <= 256:
        return np.correlate(w, a, mode="valid").astype(np.float64, copy=False)
    fw = np.fft.rfft(w, pad)
    fa = np.fft.rfft(a, pad)
    return np.fft.irfft(fw * np.conj(fa), pad)[:n].astype(np.float64, copy=False)


def _cross_fft_auto(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Cross-correlation with automatic padding."""
    n = int(a.size)
    return _cross_fft(w, a, _pad_for(3 * n - 2))


def _fast_score(a: np.ndarray) -> float:
    """Fast FFT-based score, closely matching evaluate_sequence for finite data."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float(np.inf)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 0.0:
        return float(np.inf)
    c = _conv_fft_auto(a)
    if c.size == 0:
        return float(np.inf)
    m = float(np.max(c))
    if not np.isfinite(m) or m < 0.0:
        m = 0.0
    val = float(2.0 * a.size * m / (s * s))
    return val if np.isfinite(val) else float(np.inf)


def _normalize(a: np.ndarray):
    """Normalize to sum 1, or return None if unusable."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 0.0:
        return None
    return a / s


def _positive_floor(a: np.ndarray, floor: float = 1e-15) -> np.ndarray:
    """Ensure strictly positive entries and sum 1."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    a = np.maximum(a, floor)
    s = float(a.sum())
    if s <= 0.0:
        raise ValueError("cannot make positive sequence")
    return a / s


def _make_power_tail(n: int, p: float, q: float):
    """Profile a_i ~ (i+1)^(-p) * (1 - i/(n-1))^q."""
    i = np.arange(n, dtype=np.float64)
    t = i / float(n - 1) if n > 1 else np.array([0.0], dtype=np.float64)
    a = np.power(i + 1.0, -p)
    if q != 0.0:
        tail = np.maximum(1.0 - t, 1e-8)
        a *= np.power(tail, q)
    return _normalize(a)


def _make_power_multiplier(n: int, p: float, multiplier: np.ndarray):
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + 1.0, -p)
    a *= np.maximum(multiplier, 1e-8)
    return _normalize(a)


def _make_linear_multiplier_profile(n: int, alpha: float):
    i = np.arange(n, dtype=np.float64)
    t = i / float(n - 1) if n > 1 else np.array([0.0], dtype=np.float64)
    b = np.power(i + 1.0, -0.5)
    bs = float(b.sum())
    if bs <= 0.0:
        return None
    mean_t = float(np.sum(b * t) / bs)
    m = 1.0 + alpha * (t - mean_t)
    return _make_power_multiplier(n, 0.5, m)


def _make_bump_profile(n: int, beta: float):
    i = np.arange(n, dtype=np.float64)
    t = i / float(n - 1) if n > 1 else np.array([0.0], dtype=np.float64)
    b = np.power(i + 1.0, -0.5)
    bs = float(b.sum())
    if bs <= 0.0:
        return None
    bump = t * (1.0 - t)
    mean_bump = float(np.sum(b * bump) / bs)
    m = 1.0 + beta * (bump - mean_bump)
    return _make_power_multiplier(n, 0.5, m)


def _make_step_profile(n: int, rho: float, delta: float):
    i = np.arange(n, dtype=np.float64)
    b = np.power(i + 1.0, -0.5)
    bs = float(b.sum())
    if bs <= 0.0:
        return None
    k = int(round(rho * n))
    if k <= 0 or k >= n:
        return None
    wl = float(b[:k].sum()) / bs
    wr = 1.0 - wl
    m = np.ones(n, dtype=np.float64)
    m[:k] = 1.0 - delta * wr
    m[k:] = 1.0 + delta * wl
    return _make_power_multiplier(n, 0.5, m)


def _make_exponential_profile(n: int, alpha: float):
    i = np.arange(n, dtype=np.float64)
    t = i / float(n - 1) if n > 1 else np.array([0.0], dtype=np.float64)
    a = np.exp(-alpha * t)
    return _normalize(a)


def _make_trend_profile(n: int, gamma: float):
    i = np.arange(n, dtype=np.float64)
    t = i / float(n - 1) if n > 1 else np.array([0.0], dtype=np.float64)
    b = np.power(i + 1.0, -0.5)
    a = b * np.exp(gamma * t)
    return _normalize(a)


def _make_random_profile(n: int, rng: np.random.Generator):
    kind = int(rng.integers(0, 6))
    if kind == 0:
        x = rng.normal(0.0, 0.8, size=n)
    elif kind == 1:
        t = np.linspace(0.0, 1.0, n)
        x = rng.normal(0.0, 0.6, size=n) + float(rng.uniform(-1.0, 1.0)) * t
    elif kind == 2:
        t = np.linspace(0.0, 1.0, n)
        x = rng.normal(0.0, 0.6, size=n) - float(rng.uniform(-1.0, 1.0)) * np.sqrt(t)
    elif kind == 3:
        t = np.linspace(0.0, 1.0, n)
        x = rng.normal(0.0, 0.6, size=n) + float(rng.uniform(-1.0, 1.0)) * (t * (1.0 - t))
    else:
        x = np.log1p(np.arange(n, dtype=np.float64) + 1.0)
        x += float(rng.normal(0.0, 0.5)) * np.linspace(0.0, 1.0, n)
        x += rng.normal(0.0, 0.3, size=n)
    x = np.clip(x, -5.0, 5.0)
    return _normalize(np.exp(x))


def _make_initial_candidates(n: int, seed: int, parent=None):
    rng = np.random.default_rng(seed)
    cands = []

    def add_seq(name, arr):
        if arr is None:
            return
        arr = _normalize(arr)
        if arr is None:
            return
        arr = _positive_floor(arr)
        sc = _fast_score(arr)
        if np.isfinite(sc):
            cands.append((float(sc), str(name), arr))

    if parent is not None:
        try:
            p = _normalize(parent)
            if p is not None:
                p = _positive_floor(p)
                sc = _fast_score(p)
                if np.isfinite(sc):
                    cands.append((float(sc), "parent", p))
        except Exception:
            pass

    i = np.arange(n, dtype=np.float64)
    t = i / float(n - 1) if n > 1 else np.array([0.0], dtype=np.float64)

    # Central one-sided singular profile.
    add_seq("power05", np.power(i + 1.0, -0.5))

    # Slight exponent changes.
    for p in (0.45, 0.48, 0.52, 0.55):
        add_seq(f"power{p}", np.power(i + 1.0, -p))

    # Tail / shape modifiers.
    for p in (0.45, 0.5, 0.55):
        for q in (-0.5, -0.25, -0.1, 0.0, 0.1, 0.25, 0.5, 1.0):
            add_seq(f"ptail_p{p}_q{q}", _make_power_tail(n, p, q))

    # Multiplicative smooth modifiers.
    for alpha in (-2.0, -1.0, -0.5, -0.25, 0.25, 0.5, 1.0, 2.0):
        add_seq(f"lin{alpha}", _make_linear_multiplier_profile(n, alpha))

    for beta in (-1.0, -0.5, 0.25, 0.5, 1.0, 2.0, 4.0):
        add_seq(f"bump{beta}", _make_bump_profile(n, beta))

    for rho in (0.15, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9):
        for delta in (-1.0, -0.5, -0.2, 0.2, 0.5, 1.0):
            add_seq(f"step_r{rho}_d{delta}", _make_step_profile(n, rho, delta))

    for alpha in (0.2, 0.5, 1.0, 2.0, 4.0):
        add_seq(f"exp{alpha}", _make_exponential_profile(n, alpha))

    for gamma in (-1.0, -0.5, -0.25, 0.25, 0.5, 1.0, 2.0):
        add_seq(f"trend{gamma}", _make_trend_profile(n, gamma))

    # Random smooth starts.
    for r in range(10):
        add_seq(f"rand{r}", _make_random_profile(n, rng))

    if not cands:
        # Very conservative fallback.
        a = _positive_floor(np.ones(n, dtype=np.float64))
        cands.append((_fast_score(a), "fallback_uniform", a))

    # Sort by score without lambda.
    scores = [c[0] for c in cands]
    order = sorted(range(len(cands)), key=scores.__getitem__)
    out = [cands[idx] for idx in order]
    return out


def _make_weights(c: np.ndarray, M: float) -> np.ndarray:
    """Soft weights emphasizing near-maximum convolution entries."""
    if c.size == 0:
        return np.array([], dtype=np.float64)
    if M <= 0.0:
        return np.ones(c.size, dtype=np.float64) / float(c.size)
    ratio = c / M
    ex = (ratio - 1.0) * 20.0
    ex = np.clip(ex, -40.0, 0.0)
    w = np.exp(ex)
    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.zeros_like(c, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= sw
    return w


def _local_descent(a0: np.ndarray, time_limit: float, rng: np.random.Generator,
                   initial_eta: float = 0.35, initial_noise: float = 0.02):
    """
    Multiplicative projected gradient / stochastic line search on the simplex.
    Returns (best_sequence, best_score), where best_score is exact if the
    exact evaluator can be called, otherwise the fast FFT score.
    """
    a = _positive_floor(np.asarray(a0, dtype=np.float64))
    n = int(a.size)
    deadline = time.monotonic() + max(0.0, time_limit)

    p_conv = _pad_for(2 * n - 1)
    p_cross = _pad_for(3 * n - 2)

    c = _conv_fft(a, p_conv)
    if c.size == 0:
        return a, float(np.inf)
    s = float(a.sum())
    M = float(np.max(c))
    score = float(2.0 * n * M / (s * s))
    if not np.isfinite(score):
        score = 1e100

    best_score = score
    best_a = a.copy()

    eval_func = globals().get("evaluate_sequence", None)
    true_best_score = float(np.inf)
    true_best_a = a.copy()

    if eval_func is not None:
        try:
            sc = float(eval_func(best_a.tolist()))
            if np.isfinite(sc):
                true_best_score = sc
                true_best_a = best_a.copy()
        except Exception:
            pass

    eta = float(initial_eta)
    noise = float(initial_noise)
    reject = 0
    restarts = 0
    max_restarts = 12
    no_progress = 0
    last_true_eval = time.monotonic()
    last_print = time.monotonic()

    history = []
    if len(history) == 0:
        history.append((best_score, best_a.copy()))

    while time.monotonic() < deadline - 0.25:
        c = _conv_fft(a, p_conv)
        if c.size == 0:
            break
        s = float(a.sum())
        M = float(np.max(c))
        if not np.isfinite(M) or M <= 0.0:
            break
        score = float(2.0 * n * M / (s * s))
        if not np.isfinite(score):
            score = 1e100

        old_best = best_score
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if len(history) == 0 or best_score < history[-1][0] - 1e-11:
                if len(history) < 60:
                    history.append((best_score, best_a.copy()))
                else:
                    # Keep only the best recent improvements.
                    if best_score < history[-1][0]:
                        history[-1] = (best_score, best_a.copy())
            no_progress = 0

        if best_score < true_best_score - 1e-12 and time.monotonic() - last_true_eval > 3.0:
            if eval_func is not None:
                try:
                    sc = float(eval_func(best_a.tolist()))
                    if np.isfinite(sc):
                        true_best_score = min(true_best_score, sc)
                        if sc < true_best_score:
                            true_best_a = best_a.copy()
                        last_true_eval = time.monotonic()
                except Exception:
                    pass

        if time.monotonic() - last_print > 30.0:
            print(f"  local: n={n} fast={best_score:.10f} exact={true_best_score:.10f} "
                  f"eta={eta:.4f} noise={noise:.5f} reject={reject} restarts={restarts}")
            last_print = time.monotonic()

        if best_score <= 1.0 + 1e-12 or best_score <= 1.30:
            break

        w = _make_weights(c, M)
        sx = _cross_fft(w, a, p_cross)
        smax = float(np.max(sx))
        if not np.isfinite(smax) or smax <= 0.0:
            break
        sx = sx / smax

        # Try several step sizes, sometimes adding small stochastic perturbation.
        cand_a = None
        cand_score = score
        cand_eta = eta
        for mult in (1.0, 0.5, 0.25):
            if time.monotonic() > deadline - 0.35:
                break
            e = eta * mult
            x = np.log(a) - e * sx
            if noise > 1e-9:
                scale = max(1e-9, noise * max(e, 0.05))
                x += rng.normal(0.0, scale, size=n)
            # Remove arbitrary additive scale.
            x -= float(np.max(x))
            x = np.clip(x, -60.0, 0.0)
            a2 = np.exp(x)
            s2 = float(a2.sum())
            if not np.isfinite(s2) or s2 <= 0.0:
                continue
            a2 /= s2
            c2 = _conv_fft(a2, p_conv)
            if c2.size == 0:
                continue
            m2 = float(np.max(c2))
            if not np.isfinite(m2) or m2 < 0.0:
                continue
            sc2 = float(2.0 * n * m2)  # a2 has sum 1
            if np.isfinite(sc2) and sc2 < cand_score:
                cand_score = sc2
                cand_a = a2
                cand_eta = e

        if cand_a is not None and cand_score < score * (1.0 - 1e-13):
            a = cand_a
            score = cand_score
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_progress = 0
                if len(history) < 60 and (len(history) == 0 or best_score < history[-1][0] - 1e-11):
                    history.append((best_score, best_a.copy()))
            eta = min(1.25, eta * 1.06)
            noise = max(0.0, noise * 0.998)
            reject = 0
        else:
            no_progress += 1
            eta = max(0.004, eta * 0.5)
            reject += 1

            if (reject >= 12 or no_progress >= 220) and restarts < max_restarts:
                # Restart from the best known sequence with a larger relative perturbation.
                sigma = 0.25 + 0.08 * float(min(5, restarts + 1))
                x = np.log(best_a) + rng.normal(0.0, sigma, size=n)
                x -= float(np.max(x))
                x = np.clip(x, -60.0, 0.0)
                a = np.exp(x)
                s = float(a.sum())
                if s > 0.0 and np.isfinite(s):
                    a /= s
                    a = _positive_floor(a)
                eta = 0.45
                noise = 0.05
                reject = 0
                no_progress = 0
                restarts += 1
            elif restarts >= max_restarts and no_progress > 500:
                break

    # Verify a few retained fast-score incumbents exactly.
    if eval_func is not None:
        # Evaluate the current best and the last few faster-score history points.
        to_check = []
        to_check.append(best_a)
        if a is not None:
            to_check.append(a)
        for hdr in history[-20:]:
            to_check.append(hdr[1])
        checked = 0
        for arr in to_check:
            if checked >= 24:
                break
            try:
                sc = float(eval_func(np.asarray(arr, dtype=np.float64).tolist()))
                checked += 1
                if np.isfinite(sc) and sc < true_best_score:
                    true_best_score = sc
                    true_best_a = np.asarray(arr, dtype=np.float64).copy()
            except Exception:
                pass

    if np.isfinite(true_best_score) and true_best_score < best_score:
        return true_best_a, true_best_score
    return best_a, best_score


def _select_final_best(final_list, eval_func):
    """Choose exact best among final_list, falling back to fast scores."""
    if not final_list:
        return None, float(np.inf)

    # If exact evaluator exists, verify the top fast-score candidates.
    if eval_func is not None:
        scores = [c[0] for c in final_list]
        order = sorted(range(len(final_list)), key=scores.__getitem__)
        best_score = float(np.inf)
        best_a = final_list[order[0]][1]
        checked = 0
        for idx in order:
            if checked >= 25:
                break
            arr = np.asarray(final_list[idx][1], dtype=np.float64)
            try:
                sc = float(eval_func(arr.tolist()))
                checked += 1
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = arr
            except Exception:
                pass
        if np.isfinite(best_score):
            return best_a, best_score

    # Fallback: best by fast score.
    scores = [c[0] for c in final_list]
    order = sorted(range(len(final_list)), key=scores.__getitem__)
    return final_list[order[0]][1], final_list[order[0]][0]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Construct and refine a non-negative coefficient sequence.

    Uses a multi-start stochastic projected gradient method on the simplex,
    initialized from one-sided singular/pronkore profiles and the provided
    parent sequence when available.
    """
    total_deadline = time.monotonic() + max(60.0, float(budget_s))
    reserve = 25.0
    rng = np.random.default_rng(int(seed) if seed is not None else 42)

    # Determine working length.  Prefer the supplied parent length, but avoid
    # extremely long sequences whose exact evaluation can become too costly.
    n = 2535
    parent = None
    hs = kwargs.get("height_sequence_1", None)
    if hs is None:
        hs = globals().get("height_sequence_1", None)
    if hs is not None:
        try:
            arr = np.asarray(hs, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 50 and np.all(np.isfinite(arr)):
                if arr.size <= 5000:
                    n = int(arr.size)
                    parent = np.clip(arr, 0.0, 1000.0)
                else:
                    # If the parent is too large, use a moderate length.
                    n = 4096
        except Exception:
            pass

    if n < 50:
        n = 2535
    if n > 5000:
        n = 5000

    try:
        cands = _make_initial_candidates(n, int(rng.integers(0, 2**31 - 1)), parent=parent)
        if not cands:
            a = _positive_floor(np.ones(n, dtype=np.float64))
            cands = [(_fast_score(a), "uniform", a)]

        # Ensure the parent, if present, is represented among stage-1 starts.
        starts = []
        has_parent = any(name == "parent" for _, name, _ in cands)
        if parent is not None and not has_parent:
            p = _normalize(parent)
            if p is not None:
                p = _positive_floor(p)
                starts.append((_fast_score(p), "parent_injected", p))

        # Take top diverse candidates by fast score.
        scores = [c[0] for c in cands]
        order = sorted(range(len(cands)), key=scores.__getitem__)
        for idx in order:
            sc, name, arr = cands[idx]
            starts.append((sc, name, arr))
            if len(starts) >= 6:
                break

        # Do a short refinement on several starts.
        final_list = []
        initial_best = cands[order[0]]
        final_list.append((initial_best[0], initial_best[2]))

        stage1_budget = max(90.0, min(180.0, (total_deadline - time.monotonic() - reserve - 300.0) / max(1.0, len(starts))))
        for sc, name, arr in starts:
            if time.monotonic() > total_deadline - reserve - 120.0:
                break
            stage_limit = min(stage1_budget, max(30.0, total_deadline - time.monotonic() - reserve - 60.0))
            if stage_limit < 25.0:
                break
            print(f"Stage 1 start {name} fast_score={sc:.10f} limit={stage_limit:.1f}s")
            a2, sc2 = _local_descent(arr, stage_limit, rng, initial_eta=0.35, initial_noise=0.025)
            if a2 is not None:
                final_list.append((float(sc2), a2))

        # Final deep refinement from the best candidate.
        scores_f = [c[0] for c in final_list]
        order_f = sorted(range(len(final_list)), key=scores_f.__getitem__)
        best_a, best_sc = final_list[order_f[0]]

        remaining = total_deadline - time.monotonic() - reserve
        if remaining > 45.0:
            print(f"Stage 2 deep refine fast_score={best_sc:.10f} limit={remaining:.1f}s")
            a3, sc3 = _local_descent(best_a, remaining, rng, initial_eta=0.25, initial_noise=0.012)
            if a3 is not None:
                final_list.append((float(sc3), a3))

        eval_func = globals().get("evaluate_sequence", None)
        best_a, best_score = _select_final_best(final_list, eval_func)

        if best_a is None or not np.all(np.isfinite(best_a)):
            best_a = _positive_floor(np.power(np.arange(n, dtype=np.float64) + 1.0, -0.5))
        if best_a.size == 0:
            best_a = _positive_floor(np.ones(2535, dtype=np.float64))

        # One last sanity check with the exact evaluator if available.
        if eval_func is not None:
            try:
                sc = float(eval_func(best_a.tolist()))
                if not np.isfinite(sc):
                    # Fallback to a known feasible profile.
                    best_a = _positive_floor(np.power(np.arange(n, dtype=np.float64) + 1.0, -0.5))
            except Exception:
                pass

        return [float(x) for x in np.asarray(best_a, dtype=np.float64)]

    except Exception:
        # Absolute fallback: one-sided singular profile, always feasible and
        # substantially better than a pure uniform sequence.
        try:
            a = _positive_floor(np.power(np.arange(n, dtype=np.float64) + 1.0, -0.5))
            return [float(x) for x in a]
        except Exception:
            return [1.0] * 2535
```
