def solve_lp(a):
    """Given current sequence a (sum=1, non-negative), solve LP to find b minimizing max(a*b)."""
    n = len(a)
    # Variables: b_0, ..., b_{n-1}, t
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t

    A_ub = np.zeros((2*n - 1, n + 1))

"""
Algorithm: Combined gradient descent + LP-based optimization to minimize
2n * max(convolve(a,a)) / (sum(a))^2 for non-negative sequences.

Uses FFT-based convolution for fast gradient computation, projected gradient
descent with Nesterov acceleration, and periodic LP-based "reboots" to
escape local minima.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Compute linear self-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    result = np.fft.irfft(fa * fa, fsize)[:size]
    return np.maximum(result, 0)  # clip tiny negatives from FFT

def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_subgradient(a, conv, peak_frac=0.999):
    """Compute subgradient of max(convolve(a,a)) at point a.
    
    Returns (gradient, max_value, num_peaks)
    """
    n = len(a)
    max_val = conv.max()
    # Find peaks (positions close to max)
    threshold = max_val * peak_frac
    peaks = np.where(conv >= threshold)[0]
    
    grad = np.zeros(n)
    for k in peaks:
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        idx = k - np.arange(j_start, j_end)
        grad[j_start:j_end] += 2.0 * a[idx]
    
    grad /= len(peaks)
    return grad, max_val, len(peaks)

def nesterov_gradient_descent(init, max_iters=200000, tol=1e-10, verbose=False):
    """Nesterov accelerated gradient descent on max(conv) s.t. sum(a)=1, a>=0."""
    a = project_simplex(init.copy().astype(np.float64))
    n = len(a)
    
    # Nesterov momentum
    a_prev = a.copy()
    momentum = 0.0  # will be t_k / t_{k+1}
    
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    
    t_k = 1.0
    L = 2.0  # Lipschitz constant estimate (upper bound)
    
    for it in range(max_iters):
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_next
        
        # Nesterov lookahead
        a_nest = a + beta * (a - a_prev)
        a_nest = np.maximum(a_nest, 0)
        if a_nest.sum() > 0:
            a_nest = a_nest / a_nest.sum()
        else:
            a_nest = a.copy()
        
        # Compute gradient at lookahead point
        conv = fast_convolve(a_nest)
        grad, max_val, _ = compute_subgradient(a_nest, conv)
        
        # Gradient step
        step = 1.0 / L
        a_new = a_nest - step * grad
        a_new = project_simplex(a_new)
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        val_new = conv_new.max()
        
        a_prev = a
        a = a_new
        t_k = t_next
        
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
        
        if verbose and it % 10000 == 0:
            print(f"  Iter {it}: max_conv = {best_val:.6f}, score = {2*n*best_val:.6f}")
        
        if it > 1000 and best_val - val_new < tol * best_val:
            # No improvement in non-accelerated value
            pass  # keep going
    
    return best_a, 2 * n * best_val

def lp_step(a, method='highs'):
    """Solve LP: minimize t s.t. (a*b)_k <= t, sum(b)=1, b>=0.
    
    Returns (b, t) where b is the solution.
    """
    n = len(a)
    # a should be normalized: sum(a) = 1
    a = a / a.sum()
    
    # Variables: [b_0, ..., b_{n-1}, t]
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Inequality: (a*b)_k - t <= 0 for k = 0, ..., 2n-2
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            val = a[k - j]
            if val > 0:
                rows.append(k)
                cols.append(j)
                vals.append(val)
        # -t term
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    bounds = [(0, None)] * n + [(0, None)]
    
    from scipy.optimize import linprog
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method=method)
    
    if result.success:
        b = result.x[:n]
        b = np.maximum(b, 0)
        b = b / b.sum()
        t = result.x[n]
        return b, t
    else:
        return a.copy(), None

Minimizes 2n*max(convolve(a,a))/(sum(a))^2 using LP-based alternating

def fast_convolve(a):
    """FFT-based linear self-convolution."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    result = np.fft.irfft(fa * fa, fsize)[:size]
    return np.maximum(result, 0.0)

def project_simplex(v):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def subgradient_max_conv(a, conv):
    """Subgradient of max(convolve(a,a))."""
    n = len(a)
    max_val = conv.max()
    peaks = np.where(conv >= max_val * (1 - 1e-9))[0]
    grad = np.zeros(n)
    for k in peaks:
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        grad[js:je] += 2.0 * a[k - np.arange(js, je)]
    grad /= len(peaks)
    return grad, max_val
