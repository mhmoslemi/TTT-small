        # j ranges from max(0, k-(n-1)) to min(n-1, k)

    if res.success:

    return None

"""
LP-based multiplicative weights method to minimize 2*n*max(f*f)/(sum(f))^2.
Uses iterative best-response via sparse linear programming with active constraint selection.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_objective_normalized(f):
    """Compute 2*n*max(f*f)/(sum(f))^2 for a normalized sequence f (sum(f)=1)."""
    n = len(f)
    c = np.convolve(f, f)
    return 2 * n * np.max(c)


def solve_best_response_sp(f, M, top_k=200):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for top-k constraints, g >= 0."""
    n = len(f)
    c = np.convolve(f, f)
    # Get top-k indices of the convolution
    active = np.argsort(c)[::-1][:min(top_k, len(c))]
    
    # Build sparse constraint matrix
    A_rows = []
    A_cols = []
    A_vals = []
    for idx, k in enumerate(active):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        ks = k - js  # indices into f
        A_rows.extend([idx] * len(js))
        A_cols.extend(js.tolist())
        A_vals.extend(f[ks].tolist())
    
    A = csr_matrix((A_vals, (A_rows, A_cols)), shape=(len(active), n))
    
    # Maximize sum(g) -> minimize -sum(g)
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(len(active), M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    
    if res.success:
        return res.x
    return None

        c = np.convolve(f, f)
        M = np.max(c)

        g = solve_best_response_sp(f, M, top_k=200)

                j_max = min(n - 1, k)

            grad /= len(active)
            eta = 0.5
            f = f * np.exp(-eta * grad)
            f = f / f.sum()
        else:
            # Normalize g
            g = g / g.sum()
            # Determine step size

def local_search(seq, rng, deadline):

"""

def lp_refine_step(f, M, top_k=200):
    """Solve LP for best response direction."""
    n = len(f)
    c = np.convolve(f, f)
    active = np.argsort(c)[::-1][:top_k]
    
    A_rows = []
    A_cols = []
    A_vals = []
    for idx, k in enumerate(active):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        A_rows.extend([idx] * len(js))
        A_cols.extend(js.tolist())
        A_vals.extend(f[k - js].tolist())
    
    A = csr_matrix((A_vals, (A_rows, A_cols)), shape=(len(active), n))
    
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(len(active), M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    
    if res.success:
        g = res.x
        if g.sum() > 0:
            return g / g.sum()
    return None

    f = f.copy().astype(np.float64)
    f = f / f.sum()
    
    best_f = f.copy()
    best_obj = 2 * n * np.max(np.convolve(f, f))

def local_search(seq, rng, deadline):
    """Local search with block perturbations."""
    seq = seq.copy().astype(np.float64)
    n = len(seq)
    best_score = float(evaluate_sequence(seq.tolist()))
    no_improve = 0
    
    while time.monotonic() < deadline:
        candidate = seq.copy()
        # Multi-scale perturbation
        n_blocks = rng.integers(1, 5)
        for _ in range(n_blocks):
            width = max(1, int(n * rng.uniform(0.001, 0.05)))
            start = rng.integers(0, max(1, n - width))
            scale = np.exp(rng.normal(0, 0.2, width))
            candidate[start:start+width] *= scale
        candidate = np.clip(candidate, 0, 1000)
        
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            seq = candidate
            best_score = score
            no_improve = 0
            if best_score < 2.0:
                print(f"    LS: {best_score:.6f}")
        else:
            no_improve += 1
    
    return seq
