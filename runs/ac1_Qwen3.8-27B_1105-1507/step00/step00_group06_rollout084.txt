```python
"""
Search for non-negative sequences minimizing
    2*n*max(convolve(a,a))/(sum(a)^2).

The optimizer works on variables x with sum x = n (so the evaluation score is
2*max(convolve(x,x))/n).  Candidates are produced from a rich pool of seeds
(plain uniform, edge blocks, staircases, chirps, anti-palindromic chirps,
parent resizing, etc.), then refined by a projected top-K subgradient method
with momentum, line search, adaptive step-size, and random multiplicative
restarting.  Several sequence lengths are searched within the time budget.
"""

import numpy as np
import time


def next_fft_size(m):
    """Smallest power of two >= m."""
    if m <= 1:
        return 1
    s = 1
    while s < m:
        s <<= 1
    return s


def fft_conv_full(x, N):
    """Full linear convolution x*x using FFT.  N must be at least len(x)*2-1."""
    n = x.size
    if n == 0:
        return np.zeros(0, dtype=float)
    F = np.fft.rfft(x, P=N)
    c = np.fft.irfft(F * F, N=N)
    c = c[:2 * n - 1]
    c[np.where(c < 0.0)[0]] = 0.0
    return c


def score_x_with_N(x, N, n):
    """Internal exact score for x with sum n: 2*max(conv(x,x))/n."""
    c = fft_conv_full(x, N)
    if c.size == 0:
        return np.inf
    return 2.0 * float(c.max()) / float(n)


def project_sum_n(x, n):
    """Cheap projection to {x >= 0, sum x = n, x_i <= 1000}."""
    x = np.asarray(x, dtype=float).ravel()
    if x.size != n:
        if x.size == 0:
            return np.ones(n, dtype=float)
        xp = np.linspace(0.0, 1.0, x.size)
        xq = np.linspace(0.0, 1.0, n)
        x = np.interp(xq, xp, x)
    x = x.copy()
    bad = ~np.isfinite(x)
    if bad.any():
        x[bad] = 1.0
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-12:
        return np.ones(n, dtype=float)
    x *= float(n) / s
    if float(x.max()) > 1000.0:
        return np.ones(n, dtype=float)
    return x


def top_gradient(x, c, beta, kmax):
    """
    Gradient of a smoothed log-sum-exp of convolution values c.
    Only the top kmax positions are used.  Returns the raw gradient w.r.t. x
    for d/dx logsumexp(beta*c), i.e. 2*sum_k p_k x_{k-i}.
    """
    n = x.size
    L = c.size
    if L == 0:
        return np.zeros(n, dtype=float)
    kmax = int(max(1, min(kmax, L)))
    cmax = float(c.max())
    if kmax >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, -kmax)[-kmax:]
    vals = c[idx]
    weights = np.exp(beta * (vals - cmax))
    sw = float(weights.sum())
    if not np.isfinite(sw) or sw <= 0.0:
        weights = np.ones_like(vals, dtype=float) / float(idx.size)
    else:
        weights /= sw

    grad = np.zeros(n, dtype=float)
    for ii in range(idx.size):
        w = float(weights[ii])
        if w <= 1e-15:
            continue
        k = int(idx[ii])
        lo = k - (n - 1)
        if lo < 0:
            lo = 0
        hi = k
        if hi >= n:
            hi = n - 1
        if hi < lo:
            continue
        # i in [lo, hi], j = k-i runs from k-hi down to k-lo.
        start_j = k - hi
        stop_j = k - lo + 1
        seg = x[start_j:stop_j][::-1]
        grad[lo:hi + 1] += w * seg
    return 2.0 * grad


def default_seed(n):
    """A simple two-level seed already better than all-ones."""
    x = np.ones(n, dtype=float)
    p = n // 2
    x[:p] = 2.0
    return project_sum_n(x, n)


def resize_sequence(seq, n):
    """Resize/normalize an existing sequence to length n, sum n."""
    if seq is None:
        return None
    arr = np.asarray(seq, dtype=float).ravel()
    if arr.size == 0:
        return None
    arr = np.clip(arr, 0.0, 1000.0)
    bad = ~np.isfinite(arr)
    if bad.any():
        arr[bad] = 0.0
    s = float(arr.sum())
    if s <= 1e-12:
        return None
    if arr.size == n:
        x = arr * (float(n) / s)
    elif arr.size == 1:
        x = np.full(n, float(arr[0]) * float(n) / s, dtype=float)
    else:
        xp = np.linspace(0.0, 1.0, arr.size)
        xq = np.linspace(0.0, 1.0, n)
        x = np.interp(xq, xp, arr)
        xs = float(x.sum())
        if xs <= 1e-12:
            return None
        x *= float(n) / xs
    return project_sum_n(x, n)


def make_levels_seed(n, counts, levels):
    """Piecewise-constant seed from counts and levels."""
    x = np.empty(n, dtype=float)
    pos = 0
    for cnt, lv in zip(counts, levels):
        cnt = int(max(0, cnt))
        if cnt <= 0:
            continue
        x[pos:pos + cnt] = float(lv)
        pos += cnt
    if pos < n:
        x[pos:] = float(levels[-1])
    return project_sum_n(x, n)


def make_search_seeds(n, rng, extra=None):
    """Generate a diverse pool of initial candidates for length n."""
    seeds = []

    # Uniform reference.
    seeds.append(np.ones(n, dtype=float))

    # Edge block seeds: x = 1 + a on a prefix.
    qs = [0.10, 0.15, 0.20, 0.25, 0.30, 0.40, 0.50, 0.60, 0.70, 0.80]
    amps = [0.25, 0.50, 0.75, 1.00, 1.50, 2.00, 3.00, 5.00, -0.20, -0.45, -0.75]
    for q in qs:
        p = int(round(q * float(n)))
        p = max(1, min(n - 1, p))
        for a in amps:
            x = np.ones(n, dtype=float)
            x[:p] += float(a)
            seeds.append(project_sum_n(x, n))

    # Deterministic staircases.
    for K in [3, 5, 8, 16]:
        w = np.ones(K, dtype=float) / float(K)
        counts = np.floor(n * w).astype(int)
        counts[-1] += n - int(counts.sum())
        for start, stop in [(3.0, 0.2), (2.0, 0.1), (1.2, 0.05)]:
            levels = np.linspace(start, stop, K)
            seeds.append(make_levels_seed(n, counts, levels))
            seeds.append(make_levels_seed(n, counts, levels[::-1]))

    # Random piecewise-constant seeds.
    for K in [4, 8, 16, 32]:
        for _ in range(8):
            w = rng.random(K)
            w /= w.sum()
            counts = np.floor(n * w).astype(int)
            counts[-1] += n - int(counts.sum())
            levels = np.exp(rng.normal(0.0, 0.8, K)) + 0.05
            if rng.random() < 0.5:
                levels = levels[::-1]
            seeds.append(make_levels_seed(n, counts, levels))

    # Sinusoidal / chirp seeds.
    for _ in range(30):
        i = np.arange(n, dtype=float) / float(n)
        f1 = rng.uniform(0.15, 6.0)
        f2 = rng.uniform(0.0, 3.0)
        phi = rng.uniform(0.0, 2.0 * np.pi)
        alpha = rng.uniform(0.2, 1.0)
        x = 1.0 + alpha * np.cos(2.0 * np.pi * (f1 * i + 0.5 * f2 * i * i) + phi)
        x += 0.05 * np.exp(rng.normal(0.0, 0.1, n))
        seeds.append(project_sum_n(x, n))

    # Anti-palindromic chirp seeds: x_i = 1 + alpha*s_i, s_{n-1-i}=-s_i.
    for _ in range(30):
        h = n // 2
        t = np.arange(h, dtype=float) / max(1, h)
        f1 = rng.uniform(0.15, 4.0)
        f2 = rng.uniform(0.0, 5.0)
        s = np.sin(2.0 * np.pi * (f1 * t + 0.5 * f2 * t * t))
        s += 0.5 * rng.normal(0.0, 1.0, h)
        amp = float(np.max(np.abs(s)))
        if amp > 1e-12:
            s /= amp
        alpha = rng.uniform(0.3, 1.0)
        x = np.empty(n, dtype=float)
        if h > 0:
            x[:h] = 1.0 + alpha * s
            if n % 2 == 0:
                x[h:] = 1.0 - alpha * s[::-1]
            else:
                x[h] = 1.0
                x[h + 1:] = 1.0 - alpha * s[::-1]
        else:
            x[:] = 1.0
        seeds.append(project_sum_n(x, n))

    # Use any external best sequence as a seed, blended with a block seed.
    if extra is not None:
        e = resize_sequence(extra, n)
        if e is not None:
            seeds.append(e)
            block = default_seed(n)
            for mix in [0.3, 0.5, 0.7]:
                m = float(mix)
                x = m * e + (1.0 - m) * block
                seeds.append(project_sum_n(x, n))

    return seeds


def optimize_x(x0, stop_time, rng, label=""):
    """
    Projected top-K subgradient optimizer on sum x = n, x >= 0.
    Returns the best x and its internal score found before stop_time.
    """
    n = int(x0.size)
    x = project_sum_n(np.asarray(x0, dtype=float).copy(), n)
    N = next_fft_size(2 * n - 1)

    c = fft_conv_full(x, N)
    score = 2.0 * float(c.max()) / float(n)
    best = x.copy()
    best_score = score
    last_print = best_score

    K0 = 24
    beta0 = 50.0
    lr0 = 0.10
    mom0 = 0.85
    target_avg = 0.10

    v = np.zeros(n, dtype=float)
    lr = lr0
    mom = mom0
    beta = beta0
    kmax = K0
    no_improve = 0
    fail = 0

    while time.monotonic() < stop_time:
        old_score = score

        if score < best_score - 1e-12:
            best_score = score
            best = x.copy()
            no_improve = 0
            fail = 0
            if label != "" and (last_print - best_score > 0.005 or best_score < last_print - 0.005):
                print(f"    {label} current={best_score:.6f}")
                last_print = best_score
        else:
            no_improve += 1

        # Periodic random multiplicative restart to escape plateaus.
        if no_improve > 3000:
            kick = np.exp(rng.normal(0.0, 0.25, n))
            x = project_sum_n(x * kick, n)
            c = fft_conv_full(x, N)
            score = 2.0 * float(c.max()) / float(n)
            v = np.zeros(n, dtype=float)
            lr = lr0
            beta = beta0
            kmax = K0
            no_improve = 0
            fail = 0
            continue

        # Sharpen the active set when stalled.
        if no_improve > 350:
            beta = min(beta * 1.03, 120.0)
        if no_improve > 900:
            kmax = max(1, kmax // 2)

        grad = top_gradient(x, c, beta, kmax)
        grad -= float(np.mean(grad))
        gnorm = float(np.linalg.norm(grad))
        if (not np.isfinite(gnorm)) or gnorm < 1e-12:
            x = project_sum_n(x + rng.normal(0.0, 0.05, n), n)
            c = fft_conv_full(x, N)
            score = 2.0 * float(c.max()) / float(n)
            v = np.zeros(n, dtype=float)
            lr = lr0
            beta = beta0
            kmax = K0
            no_improve = 0
            fail = 0
            continue

        # Momentum step with average-step-limiting.
        v = mom * v - lr * grad
        avg = float(np.linalg.norm(v) / np.sqrt(float(n)))
        if np.isfinite(avg) and avg > target_avg:
            v *= target_avg / avg

        trial = project_sum_n(x + v, n)
        c_t = fft_conv_full(trial, N)
        score_t = 2.0 * float(c_t.max()) / float(n)

        if np.isfinite(score_t) and score_t <= old_score + 1e-11:
            x = trial
            c = c_t
            score = score_t
            if score_t < old_score - 1e-14:
                lr = min(lr * 1.02, 0.60)
                beta = max(beta * 0.995, 10.0)
                fail = 0
            else:
                fail += 1
            continue

        # Rejected step: reduce lr and try a plain gradient step.
        fail += 1
        v = np.zeros(n, dtype=float)
        lr *= 0.5

        if lr < 1e-6:
            kick = np.exp(rng.normal(0.0, 0.12, n))
            x = project_sum_n(x * kick, n)
            c = fft_conv_full(x, N)
            score = 2.0 * float(c.max()) / float(n)
            lr = lr0
            beta = beta0
            kmax = K0
            fail = 0
            no_improve -= 500
            continue

        trial = project_sum_n(x - lr * grad, n)
        c_t = fft_conv_full(trial, N)
        score_t = 2.0 * float(c_t.max()) / float(n)
        if np.isfinite(score_t) and score_t < old_score:
            x = trial
            c = c_t
            score = score_t
            fail = 0
            continue

        # If even a small gradient step fails, this basin is done.
        break

    return best, float(best_score)


def search_length(n, stop_time, rng, extra=None):
    """Run seed scoring plus multiple optimization runs for one length n."""
    if n < 2:
        n = 2

    seeds = make_search_seeds(n, rng, extra)
    N = next_fft_size(2 * n - 1)

    scored = []
    for x in seeds:
        x = project_sum_n(x, n)
        s = score_x_with_N(x, N, n)
        if np.isfinite(s):
            scored.append((float(s), len(scored), x))
    scored.sort()

    if not scored:
        x = np.ones(n, dtype=float)
        return x, 2.0

    best_score, _, best_x = scored[0]
    print(f"length {n}: best seed score={best_score:.6f}")

    # Optimize the strongest seeds for modest chunks.
    top_count = min(5, len(scored))
    for rank in range(top_count):
        if time.monotonic() >= stop_time:
            break
        remaining = float(stop_time - time.monotonic())
        if remaining < 5.0:
            break
        if rank == 0:
            chunk = min(remaining, 90.0)
        elif rank == 1:
            chunk = min(remaining, 75.0)
        else:
            chunk = min(remaining, 45.0)
        x0 = np.asarray(scored[rank][2], dtype=float).copy()
        xo, so = optimize_x(x0, time.monotonic() + max(1.0, chunk), rng, label=f"n={n} rank={rank}")
        if so < best_score:
            best_score = so
            best_x = xo
            print(f"length {n}: improved to {best_score:.6f} from rank {rank}")

    # Continue polishing the current best in 30-second chunks.
    while time.monotonic() < stop_time:
        remaining = float(stop_time - time.monotonic())
        if remaining < 5.0:
            break
        chunk = min(remaining, 30.0)
        x0 = np.asarray(best_x, dtype=float).copy()
        sigma = 0.10 if best_score > 1.45 else 0.05
        x0 = project_sum_n(x0 * np.exp(rng.normal(0.0, sigma, n)), n)
        xo, so = optimize_x(x0, time.monotonic() + max(1.0, chunk), rng, label=f"n={n} polish")
        if so < best_score:
            best_score = so
            best_x = xo
            print(f"length {n}: polish improved to {best_score:.6f}")
        if best_score < 1.0001:
            break

    return np.asarray(best_x, dtype=float), float(best_score)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main entry point.  Searches several lengths and returns the best
    non-negative sequence found under the given time budget.
    """
    rng = np.random.default_rng(seed)
    budget_s = max(10.0, float(budget_s))
    deadline = time.monotonic() + budget_s - 5.0

    # Function global parent construction, if available.
    parent = None
    try:
        parent = globals().get("height_sequence_1", None)
    except Exception:
        parent = None

    # Fallback / initial best: explicit two-level block seed of length 5582.
    n_default = 5582
    best_x = default_seed(n_default)
    N_default = next_fft_size(2 * n_default - 1)
    best_score = score_x_with_N(best_x, N_default, n_default)
    print(f"initial block seed n={n_default}: score={best_score:.6f}")

    # Consider the parent itself at the default length.
    p_default = resize_sequence(parent, n_default)
    if p_default is not None:
        s = score_x_with_N(p_default, N_default, n_default)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_x = p_default
            print(f"parent resized to n={n_default}: score={best_score:.6f}")

    # Length schedule.  5582 is the parent length and is emphasized first.
    schedule = [
        (5582, 620),
        (4096, 300),
        (2048, 250),
        (1024, 160),
        (7168, 90),
    ]

    for n, alloc in schedule:
        now = time.monotonic()
        remaining = float(deadline - now)
        if remaining <= 10.0:
            break
        t = min(float(alloc), remaining)
        if t <= 10.0:
            continue

        extra = resize_sequence(best_x, n)
        try:
            local_x, local_score = search_length(n, now + t, rng, extra=extra)
            if local_x is not None and local_x.size == n:
                local_x = project_sum_n(local_x, n)
                N_local = next_fft_size(2 * n - 1)
                local_score = score_x_with_N(local_x, N_local, n)
                if np.isfinite(local_score) and local_score < best_score:
                    best_score = local_score
                    best_x = local_x
                    print(f"new global best: n={n}, score={best_score:.6f}")
        except Exception as exc:
            print(f"search_length({n}) failed: {exc}")

    # Final validation using the official evaluator, if accessible.
    final_list = [float(v) for v in np.clip(np.asarray(best_x, dtype=float), 0.0, 1000.0)]
    try:
        official = float(evaluate_sequence(final_list))
        if not np.isfinite(official):
            raise ValueError
        print(f"official final score: {official:.6f} (internal={best_score:.6f}, n={len(final_list)})")
    except Exception:
        # Fallback to a known valid block construction.
        final_list = [float(v) for v in np.clip(default_seed(n_default), 0.0, 1000.0)]
        try:
            official = float(evaluate_sequence(final_list))
            print(f"fallback official score: {official:.6f} (n={len(final_list)})")
        except Exception:
            print("fallback official evaluation unavailable; returning valid block seed.")

    return final_list
```
