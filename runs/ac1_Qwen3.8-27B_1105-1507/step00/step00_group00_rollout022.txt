def _next_pow2(x):
    """Return next power of 2 >= x."""
    if x <= 1:
        return 1
    r = 1
    while r < x:
        r *= 2
    return r

def _autoconvolve(a, fft_size):
    """Compute full linear autoconvolution using FFT."""
    n = len(a)
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)
    return C[:2*n-1]

def _spectral_compression(a, beta, fft_size):
    """Compress the magnitude of the spectrum."""
    n = len(a)
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_new = np.power(mag, beta)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)
    a_new = a_new[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return None
    return a_new / s
