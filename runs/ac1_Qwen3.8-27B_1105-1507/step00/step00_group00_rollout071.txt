def simplex_project(y):
    """Project y onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(y)
    if np.sum(y) <= 0:
        return np.ones(n) / n
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n+1) > (cssv[-1] - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(y - theta, 0)

"""
Optimizes a non-negative sequence to minimize 2n*max(convolve(a,a))/sum(a)^2.
Uses: flat-convolution initialization, gradient descent with smooth max (log-sum-exp),

def simplex_project(y):
    """Project y onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(y)
    s = np.sum(y)
    if s <= 1e-15:
        return np.ones(n) / n
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv[-1] - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(y - theta, 0.0)


def flat_convolve_init(n):
    """
    Initialize a sequence using the flat convolution recurrence.
    Sets b_k = t = 1/(2n-1) for as many k as possible,
    giving a sequence where the early convolution values are equal.
    """
    t = 1.0 / (2.0 * n - 1.0)
    a = np.zeros(n)
    a[0] = math.sqrt(t)
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            s += a[i] * a[k - i]
        val = (t - s) / (2.0 * a[0])
        if val < 0:
            break
        a[k] = val
    total = np.sum(a)
    if total > 1e-10:
        a = a / total
    else:
        a = np.ones(n) / n
    return a


def smooth_obj_grad(a, tau):
    """
    Compute smooth approximation of 2n*max(conv(a,a))/S^2 and its gradient.
    Returns (smooth_obj, grad_vector) where grad points in direction of steepest ascent.
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf, np.zeros(n)

    b = np.convolve(a, a)  # length 2n-1, all non-negative

    # Log-sum-exp for numerical stability
    b_max = float(np.max(b))
    shifted = (b - b_max) / tau
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(np.sum(exp_shifted)))
    B_smooth = b_max + tau * log_sum

    # Softmax weights
    p = exp_shifted / np.sum(exp_shifted)

    # Cross-correlation: R[i] = sum_j p[j+i] * a[j] for i=0..n-1
    # Use FFT: (p crosscorr a)[i] = sum_j p[j+i]*a[j]
    # FFT formula: IFFT(conj(FFT(a)) * FFT(p)) gives this
    N = 1
    while N < 3 * n:
        N *= 2
    P_ft = np.fft.fft(p, N)
    A_ft = np.fft.fft(a, N)
    R_full = np.fft.ifft(np.conj(A_ft) * P_ft).real
    R = R_full[:n]

    # Gradient of F_smooth w.r.t. a_i (direction of ascent):
    # proportional to S*(R[i] + a[i]) - 2*B_smooth
    grad = S * (R + a) - 2.0 * B_smooth

    obj = 2.0 * n * B_smooth / (S * S)
    return obj, grad

                    print(f"  [LS] n={n}, score={score_ls:.6f}", flush=True)

    if best_solution is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final polish, current best={best_score:.6f} ===", flush=True)
        a = best_solution.copy()
        polish_time = deadline - time.monotonic() - 5
        a_polished = local_block_search(a, rng, max_iters=10000, time_limit=polish_time)
        score_final = float(evaluate_sequence(a_polished.tolist()))
        if np.isfinite(score_final) and score_final < best_score:
            best_score = score_final
            best_solution = a_polished.copy()
            print(f"  [POLISH] score={score_final:.6f}", flush=True)

    # Fallback
    if best_solution is None:
        best_solution = np.ones(100) / 100.0
        best_score = float(evaluate_sequence(best_solution.tolist()))
        print(f"  [FALLBACK] score={best_score:.6f}", flush=True)

    print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_solution)} ===", flush=True)
    return [float(x) for x in best_solution]
