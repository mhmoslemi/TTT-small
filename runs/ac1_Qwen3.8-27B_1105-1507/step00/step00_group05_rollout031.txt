    a = a / total  # normalize

        top_k = np.argsort(c)[-n_peaks:]
        # Compute weighted gradient
        g = np.zeros(n)
        for k in top_k:
            # c[k] = sum_i a[i] * a[k-i]

            valid1 = (idx1 >= 0) & (idx1 < n)

        # Update
        a -= 0.01 * g
        a = np.maximum(0, a)
        total = a.sum()
        if total > 0:
            a /= total
    
    return a

    s = a.sum()
    if s < 0.01:

    return 2.0 * n * c.max() / (s * s)

def gradient_for_peak(a, k):
    """Compute the gradient of c[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)

    j = np.arange(n)
    idx = k - j
    valid = (idx >= 0) & (idx < n)
    g[valid] = 2.0 * a[idx[valid]]
    return g

    g = np.zeros(n)

        g[valid] += a[idx[valid]]

    # Update

    a_new = np.maximum(0, a_new)
    s = a_new.sum()
    if s > 0:

    return a_new

                    break
            eta_try *= 0.5

        if not accepted:
            eta = max(eta * 0.7, 1e-5)
        else:
            eta = min(eta * 1.02, 0.1)

        # Occasional random restart perturbation
        if rng.random() < 0.003:
            frac = 0.1
            idx = rng.choice(n, size=int(n * frac), replace=False)
            a[idx] *= np.exp(rng.normal(0, 0.3, len(idx)))
            a = np.maximum(0, a)
            s2 = a.sum()
            if s2 > 1e-10:
                a /= s2
                J = compute_J_fast(a)

        T = max(T * cooling, T_min)
        iter_count += 1

        if iter_count % 20000 == 0:
            print(f"    iter={iter_count}, J={J:.6f}, best={best_J:.6f}")

    return best_a, best_J


def make_chirp_init(n, rng):
    """Chirp-based initialization with random parameters."""
    i = np.arange(n, dtype=np.float64)
    f0 = rng.uniform(0.1, 5.0)
    f1 = rng.uniform(1.0, 20.0)
    amp = rng.uniform(0.1, 0.4)
    a = 1.0 + amp * np.cos(2.0 * np.pi * (f0 * i / n + f1 * (i / n) ** 2))
    a = np.maximum(0, a)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()


def make_multifreq_init(n, rng):
    """Multi-frequency cosine superposition."""
    i = np.arange(n, dtype=np.float64)
    n_freqs = rng.integers(3, 8)
    freqs = rng.uniform(0.5, 10.0, n_freqs)
    amps = np.abs(rng.normal(0, 0.08, n_freqs))
    phases = rng.uniform(0, 2 * np.pi, n_freqs)
    a = np.ones(n)
    for f, am, ph in zip(freqs, amps, phases):
        a += am * np.cos(2.0 * np.pi * f * i / n + ph)
    a = np.maximum(0.01, a)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for the sequence minimizing J = 2n*max(convolve(a,a))/(sum(a))^2.
    Uses gradient-based peak reduction with multiple initializations and annealing.
    """
    global height_sequence_1
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 10.0

    print(f"Starting optimization with budget {budget_s}s")

    # Build list of (length, initial_sequence) pairs
    init_list = []

    # 1. Provided sequence
    try:
        ps = np.asarray(height_sequence_1, dtype=np.float64)
        ps = np.clip(ps, 0, 1000)
        if ps.size > 10 and ps.sum() > 0:
            init_list.append((ps.size, ps / ps.sum()))
            print(f"Using provided sequence: n={ps.size}")
    except (NameError, TypeError):
        pass

    # 2. Uniform n=4000
    n2 = 4000
    init_list.append((n2, np.ones(n2) / n2))

    # 3. Chirp n=5000
    n3 = 5000
    init_list.append((n3, make_chirp_init(n3, rng)))

    # 4. Multi-frequency n=4000
    n4 = 4000
    init_list.append((n4, make_multifreq_init(n4, rng)))

    # 5. Random n=3000
    n5 = 3000
    a5 = rng.exponential(1.0, n5)
    init_list.append((n5, a5 / a5.sum()))

    # 6. Chirp n=8000
    n6 = 8000
    init_list.append((n6, make_chirp_init(n6, rng)))

    # 7. Multi-freq n=6000
    n7 = 6000
    init_list.append((n7, make_multifreq_init(n7, rng)))

    best_a = None
    best_J = np.inf

    # Time budget allocation: give more time to promising inits
    n_inits = len(init_list)
    # Allocate time proportionally, with minimum 30s each
    base_time = max(30.0, (budget_s - 60.0) / n_inits)

    for idx, (n, a0) in enumerate(init_list):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        alloc = min(base_time, remaining * 0.8)
        if alloc < 20.0:
            alloc = max(20.0, remaining * 0.3)

        print(f"Init {idx+1}/{n_inits}: n={n}, time_budget={alloc:.1f}s")
        t0 = time.monotonic()

        # Optimize with n_peaks that scales with size
        n_peaks = max(10, min(30, n // 200))
        a_opt, J_opt = optimize_sequence(a0.copy(), rng, t0 + alloc, n_peaks=n_peaks)

        actual_time = time.monotonic() - t0
        print(f"  Result: J={J_opt:.6f} (took {actual_time:.1f}s)")

        if J_opt < best_J:
            best_J = J_opt
            best_a = a_opt.copy()
            print(f"  NEW BEST: J={best_J:.6f} at n={n}")

    # Final polish on the best solution with remaining time
    if best_a is not None and time.monotonic() < deadline:
        print(f"Polishing best solution (J={best_J:.6f})...")
        n_peaks = max(15, min(40, len(best_a) // 150))
        best_a, best_J = optimize_sequence(best_a.copy(), rng, deadline, n_peaks=n_peaks)
        print(f"Final J={best_J:.6f}")

    if best_a is None:
        # Fallback
        best_a = np.ones(256, dtype=np.float64)
        best_J = 2.0

    # Scale to a safe range (avoid clipping at 0 or 1000)
    # Since J is scale-invariant, any scaling works as long as values are in [0, 1000]
    current_max = best_a.max()
    if current_max > 0:
        scale = min(500.0 / current_max, 1000.0 / current_max)
        result = best_a * scale
    else:
        result = best_a

    # Final verification
    final_J = evaluate_sequence(result.tolist())
    print(f"Verified final J = {final_J:.6f}")

    if not np.isfinite(final_J):
        # Fallback to uniform
        result = np.ones(256, dtype=np.float64) * 1.0
        final_J = evaluate_sequence(result.tolist())
        print(f"Fallback J = {final_J:.6f}")

    return [float(x) for x in result]
