```python
"""
LP-based optimization for minimizing 2n*max(a*a)/(sum(a))^2.
Uses iterative LP relaxation to find sequences with flat auto-convolution.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def _fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def _auto_convolve(a):
    """Full auto-convolution a * a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _score_fast(a):
    """Fast score computation (same as evaluate_sequence for valid inputs)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    conv = _auto_convolve(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _build_sparse_A(f):
    """Build sparse matrix A where A[k,j] = f[k-j] for 0<=k-j<n."""
    n = len(f)
    # Column j: rows j..j+n-1 have values f[0]..f[n-1]
    j_idx = np.arange(n)
    i_offset = np.arange(n)
    rows = (j_idx[:, None] + i_offset[None, :]).ravel()
    cols = np.repeat(j_idx, n)
    vals = np.tile(f, n)
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A.tocsc()


def _solve_lp(f, M, rng=None):
    """Solve LP: max 1^T g s.t. A(f)g <= M, g >= 0.
    Returns the optimal g or None if infeasible.
    """
    n = len(f)
    A = _build_sparse_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'presolve': True}
    if rng is not None:
        # Use a random objective to find a different extreme point
        c = -rng.standard_normal(n)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if res.success:
        return res.x
    return None


def _lp_iterate(f0, n_iters, rng, time_budget):
    """Run LP-based iterations starting from f0.
    Returns (best_f, best_score).
    """
    f = f0.copy().astype(np.float64)
    # Normalize to sum = 1
    s = np.sum(f)
    if s < 1e-10:
        f = np.ones_like(f) / len(f)
        s = 1.0
    else:
        f = f / s
    
    best_f = f.copy()
    best_score = _score_fast(f)
    
    t = 0.5  # step size
    stagnant = 0
    
    for it in range(n_iters):
        if time.time() > time_budget:
            break
        
        conv = _auto_convolve(f)
        M = np.max(conv)
        if M < 1e-15:
            break
        
        g = _solve_lp(f, M, rng)
        if g is None:
            break
        
        # Normalize g to have same sum as f (= 1)
        g_sum = np.sum(g)
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Update
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0.0, None)
        f_new_sum = np.sum(f_new)
        if f_new_sum < 1e-15:
```

"""
LP-based iterative optimization for minimizing 2n*max(a*a)/(sum(a))^2.
Uses iterative LP relaxation to find sequences with flat auto-convolution,
combined with coordinate descent and local search for refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def _fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def _auto_convolve(a):
    """Full auto-convolution a * a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _score_fast(a):
    """Fast score: 2n*max(a*a)/(sum(a))^2. Assumes non-negative a."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    conv = _auto_convolve(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _build_sparse_A(f):
    """Build sparse matrix A where A[k,j] = f[k-j] for 0<=k-j<n, else 0."""
    n = len(f)
    j_idx = np.arange(n)
    i_offset = np.arange(n)
    rows = (j_idx[:, None] + i_offset[None, :]).ravel()
    cols = np.repeat(j_idx, n)
    vals = np.tile(f, n)
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A.tocsc()


def _solve_lp(f, M, rng=None):
    """Solve LP: max c^T g s.t. A(f)g <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = _build_sparse_A(f)
    if rng is not None:
        c = -rng.standard_normal(n)
    else:
        c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _lp_iterate(f0, rng, time_budget):
    """Run LP-based iterations from initial f0 (normalized to sum=1)."""
    f = np.clip(f0.astype(np.float64).copy(), 0.0, None)
    s = np.sum(f)
    if s < 1e-12:
        f = np.ones_like(f) / len(f)
        s = 1.0
    else:
        f = f / s

    best_f = f.copy()
    best_score = _score_fast(f)
    t = 0.5
    stagnant = 0
    n = len(f)

    while time.time() < time_budget:
        conv = _auto_convolve(f)
        M = np.max(conv)
        if M < 1e-15:
            break

        # Solve LP with random objective for diversity
        g = _solve_lp(f, M, rng)
        if g is None:
            g = _solve_lp(f, M)  # fallback to standard objective
            if g is None:
                break

        g_sum = np.sum(g)
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Adaptive step size
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0.0, None)
        f_new_s = np.sum(f_new)
        if f_new_s < 1e-15:
            break
        f_new = f_new / f_new_s

        score_new = _score_fast(f_new)

            best_f = f_new.copy()

            stagnant += 1

    n = len(f)

        # Random subset of coordinates for speed
        n_try = min(n, max(50, n // 5))

            conv[k_valid] = old_vals + new_contribs

            new_max = max(np.max(conv[k_valid]), 
                         np.max(np.delete(conv, k_valid)) if len(np.delete(conv, k_valid)) > 0 else 0)
            # Actually this is expensive; just use the old max and check affected

                # The old max might be gone

            new_max = max(np.max(conv[k_valid]), unaffected_max) if len(conv[k_valid]) > 0 else unaffected_max
            # Simplify: just recompute max (O(n) which is fine)
            new_max = np.max(conv)

            new_score = 2.0 * n * new_max / 1.0  # sum is 1 (approximately)
            # Actually sum changed slightly; let's just use the fast scorer for accuracy
            # But that's O(n log n). For now, accept the approximation.
            old_score = 2.0
