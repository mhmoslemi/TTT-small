        p = np.maximum(p, 1e-3)
    elif init_type == 'quadratic':
        i = np.arange(n, dtype=np.float64)
        t = 2.0 * i / max(1, n - 1) - 1.0
        p = np.sqrt(np.maximum(1.0 - t * t, 0.0)) + 1e-3
    elif init_type == 'flat_edges':
        i = np.arange(n, dtype=np.float64)
        t = 2.0 * i / max(1, n - 1) - 1.0
        p = 1.0 - 0.4 * t * t
        p = np.maximum(p, 1e-3)
    else:
        p = np.ones(n, dtype=np.float64)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s > 1e-10:
        p = p / s
    else:
        p = np.ones(n) / n
    return p


def _convolve_fft(p):
    n = len(p)
    size = 2 * n - 1
    P = np.fft.rfft(p, size)
    c = np.fft.irfft(P * P, size)
    return c


def _build_A(p, n):
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    p2 = 2.0 * p
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = p2[lo:hi + 1][::-1]
    return A


def _lp_solve_step(p, c, n):
    num_c = 2 * n - 1
    A = _build_A(p, n)

    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[-1] = 1.0

    A_ub = np.empty((num_c, n + 1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -c.astype(np.float64)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    lower = -p
    bounds = [(lower[i], None) for i in range(n)]
    bounds.append((0.0, None))

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result is not None and result.success:
            return result.x[:n].copy()
    except Exception:
        pass
    return None


def _refine_lp(p, deadline, max_iters=200, verbose=False):
    n = len(p)
    p = p.copy().astype(np.float64)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        p = np.ones(n) / n
    else:
        p = p / s

    c = _convolve_fft(p)
    c = np.maximum(c, 0.0)
    c_max = c.max()
    best_p = p.copy()
    best_score = 2.0 * n * c_max

    stagnant = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        delta = _lp_solve_step(p, c, n)
        if delta is None:
            p = p * (1.0 + 0.005 * np.random.randn(n))
            p = np.maximum(p, 1e-15)
            p = p / p.sum()
            c = _convolve_fft(p)
            c = np.maximum(c, 0.0)
            c_max = c.max()
            stagnant += 1
            continue

        accepted = False
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            p_try = p + alpha * delta
            if (p_try < -1e-12).any():
                continue
            p_try = np.maximum(p_try, 0.0)
            ps = p_try.sum()
            if ps < 1e-10:
                continue
            p_try = p_try / ps

            c_try = _convolve_fft(p_try)
            c_try = np.maximum(c_try, 0.0)
            score_try = 2.0 * n * c_try.max()

            if score_try < best_score:
                best_score = score_try
                best_p = p_try.copy()

            if score_try <= 2.0 * n * c_max + 1e-12:
                p = p_try
                c = c_try
                c_max = c.max()
                accepted = True
                break

        if not accepted:
            stagnant += 1
            if stagnant > 15:
                p = p * (1.0 + 0.02 * np.random.randn(n))
                p = np.maximum(p, 1e-15)
                p = p / p.sum()
                c = _convolve_fft(p)
                c = np.maximum(c, 0.0)
                c_max = c.max()
                stagnant = 0
        else:
            stagnant = 0

        if verbose and (it % 20 == 0):
            print(f"    LP it={it}, best_score={best_score:.8f}")

    return best_p, best_score


def _refine_mul(p, deadline, max_iters=100000, beta=30.0, lr=0.15):
    n = len(p)
    p = p.copy().astype(np.float64)
    p = np.maximum(p, 1e-15)
    p = p / p.sum()

    size = 2 * n - 1
    best_p = p.copy()
    best_score = 2.0 * n * _convolve_fft(p).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _convolve_fft(p)
        c = np.maximum(c, 0.0)
        c_max = c.max()
        score = 2.0 * n * c_max

        if score < best_score:
            best_score = score
            best_p = p.copy()

        weights = np.exp(beta * (c - c_max))
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        w = weights / w_sum

        n_top = min(400, size)
        top_idx = np.argpartition(w, -n_top)[-n_top:]

        pressure = np.zeros(n, dtype=np.float64)
        for k in top_idx:
            wk = w[k]
            if wk < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            pressure[lo:hi + 1] += wk * p[lo:hi + 1][::-1]

        scale = lr / (n * 0.5)
        p = p * np.exp(-scale * pressure)
        p = np.maximum(p, 1e-15)
        p = p / p.sum()

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(convolve(a,a))/(sum(a))^2.
    Uses LP-based linearized optimization followed by multiplicative polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    total_deadline = start_time + budget_s - 10.0

    best_score = np.inf
    best_seq = None

    # Load given sequence
    given_p = None
    try:
        hs = globals().get('height_sequence_1')
        if hs is not None and len(hs) > 10:
            given_p = np.array(hs, dtype=np.float64)
            given_p = np.maximum(given_p, 0.0)
            if given_p.sum() > 0.01:
                given_p = given_p / given_p.sum()
                n_g = len(given_p)
                s_g = 2.0 * n_g * _convolve_fft(given_p).max()
                print(f"Given: n={n_g}, score={s_g:.6f}")
    except Exception as e:
        print(f"Could not load given: {e}")

    # Phase 1: LP search over configs (55% budget)
    phase1_dl = start_time + (budget_s - 10.0) * 0.55
    print(f"Phase 1: LP search ({phase1_dl - start_time:.0f}s budget)")

    configs = [
        (1024, 'u_shaped'),
        (1024, 'ramp'),
        (1024, 'bimodal'),
        (1500, 'u_shaped'),
        (1500, 'bimodal'),
        (2048, 'u_shaped'),
        (2048, 'ramp'),
        (2048, 'bimodal'),
    ]

    for n, init_type in configs:
        if time.monotonic() > phase1_dl:
            break
        p = _make_init(n, init_type)
        s0 = 2.0 * n * _convolve_fft(p).max()
        if np.isfinite(s0) and s0 < best_score:
            best_score = s0
            best_seq = p.copy()
            print(f"  Config n={n} init={init_type}: initial score={s0:.6f}")

        p_best, score = _refine_lp(p, phase1_dl, max_iters=80, verbose=False)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = p_best.copy()
            print(f"  Config n={n} init={init_type}: LP score={score:.8f} [NEW BEST]")

    # Try given sequence with LP if size allows
    if given_p is not None and len(given_p) <= 2500:
        p_best, score = _refine_lp(given_p, phase1_dl, max_iters=60, verbose=False)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = p_best.copy()
            print(f"  Given (LP): score={score:.8f} [NEW BEST]")

    if best_seq is None:
        best_seq = _make_init(1024, 'u_shaped')
        best_score = 2.0 * 1024 * _convolve_fft(best_seq).max()
        print(f"  Fallback init score={best_score:.6f}")

    print(f"After Phase 1: best_score={best_score:.8f}, n={len(best_seq)}")

    # Phase 2: Better LP refinement on the best (20% budget)
    phase2_dl = start_time + (budget_s - 10.0) * 0.75
    n_b = len(best_seq)
    if n_b <= 2500:
        print(f"Phase 2: LP refinement ({phase2_dl - start_time:.0f}s budget)")
        p_best, score = _refine_lp(best_seq, phase2_dl, max_iters=150, verbose=True)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = p_best.copy()
            print(f"  Phase 2 LP: score={score:.8f}")
    else:
        print(f"Phase 2: n={n_b} too large for LP, skipping")

    # Phase 3: Multiplicative polish (15% budget)
    phase3_dl = start_time + (budget_s - 10.0) * 0.92
    print(f"Phase 3: Multiplicative polish ({phase3_dl - start_time:.0f}s budget)")
    p_pol, score_pol = _refine_mul(best_seq, phase3_dl, max_iters=80000, beta=30.0, lr=0.12)
    if np.isfinite(score_pol) and score_pol < best_score:
        best_score = score_pol
        best_seq = p_pol.copy()
        print(f"  Phase 3 mul: score={score_pol:.8f}")

    # Phase 4: Final LP if time remains and n is small enough
    phase4_dl = total_deadline
    if len(best_seq) <= 2000 and time.monotonic() < phase4_dl:
        print(f"Phase 4: Final LP ({phase4_dl - start_time:.0f}s budget)")
        p_fin, score_fin = _refine_lp(best_seq, phase4_dl, max_iters=100, verbose=True)
        if np.isfinite(score_fin) and score_fin < best_score:
            best_score = score_fin
            best_seq = p_fin.copy()
            print(f"  Phase 4 LP: score={score_fin:.8f}")

    # Try multiplicative with different beta for diversity
    if time.monotonic() < total_deadline - 30.0 and len(best_seq) <= 5000:
        p_pol2, score_pol2 = _refine_mul(best_seq, total_deadline, max_iters=50000,
                                         beta=50.0, lr=0.08)
        if np.isfinite(score_pol2) and score_pol2 < best_score:
            best_score = score_pol2
            best_seq = p_pol2.copy()
            print(f"  Extra mul (beta=50): score={score_pol2:.8f}")

    # Prepare output
    print(f"\nFinal: local_score={best_score:.8f}, n={len(best_seq)}")

    p_out = best_seq.copy()
    p_out = np.maximum(p_out, 0.0)
    s_out = p_out.sum()
    if s_out < 1e-10:
        return [1.0] * 256
    p_out = p_out / s_out

    # Scale to keep values in [0, 1000]
    p_max = p_out.max()
    if p_max > 0.5:
        p_out = p_out * (0.5 / p_max)
    elif p_max < 0.001:
        p_out = p_out * (0.001 / p_max)

    final_seq = [float(x) for x in p_out]
    actual = evaluate_sequence(final_seq)
    print(f"Evaluated score: {actual:.8f}")

    if not np.isfinite(actual):
        return [1.0] * 256

    return final_seq
