        sequence_array = sequence_array * (np.sqrt(2 * target_len) / seq_sum)
    # Return as list

    return sequence_array_final

def calculate_convolution_directly(seq):
    """Directly calculate the convolution if numpy's convolve fails for some reason"""
    n = len(seq)
    conv_result = np.zeros(2*n-1)  # Initialize full convolution
    for i in range(n):
        for j in range(n):
            conv_result[i+j] += seq[i] * seq[j]
    return conv_result.tolist()

def scale_sequence_by_length(sequence, target_length):
    """Conservatively scale a sequence to a new length"""
    current_n = len(sequence)
    target_n = target_length
    # If the sequence is shorter than target_n, pad with zeros until target_n
    # But then ensure the sum becomes the desired one:
    if target_n > current_n:
        # We can only give + so this might not align to bounds
        sequence_padded = sequence + [0.0]*(target_n - current_n)
    elif target_n < current_n:
        sequence_padded = sequence[:target_n]
    else:
        sequence_padded = sequence
    # But now, the sum might be small. Scale to have sum = sqrt(2*target_n)
    sum_pad = max(0.001, np.sum(sequence_padded))  # Ensure at least 0.001 to avoid infinite score
    if sum_pad > 0:
        norm_factor = (np.sqrt(2*target_n) / sum_pad)
        sequence_out = [x*norm_factor for x in sequence_padded]
    else:
        sequence_out = sequence_padded  # Stay positive but unchanged
    return sequence_out
