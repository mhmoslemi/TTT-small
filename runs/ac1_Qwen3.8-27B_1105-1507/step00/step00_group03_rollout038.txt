import time

def project_simplex(v):
    """Project v onto the simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

    no_improve = 0
    max_no_improve = 3000
    momentum = np.zeros(n)
    mu = 0.9

    iteration = 0
    while time.monotonic() < deadline:
        # Compute convolution and find peak
        C = fft_convolve_full(a, a)
        k_star = int(np.argmax(C))

        # Compute gradient
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        grad = np.zeros(n)
        if j_min <= j_max:
            indices = k_star - np.arange(j_min, j_max + 1)
            grad[j_min:j_max + 1] = 2.0 * a[indices]

        # Momentum update
        momentum = mu * momentum + grad
        a_new = project_onto_simplex(a - lr * momentum)

        # Evaluate
        score = compute_score(a_new)

        if score < best_score - 1e-14:
            a = a_new
            best_a = a.copy()
            best_score = score
            no_improve = 0
            mu = min(0.99, mu + 0.001)
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                lr *= 0.8
                mu = max(0.5, mu - 0.05)
                if lr < min_lr:
                    # Random restart from best
                    a = best_a + 0.05 * rng.standard_normal(n)
                    a = project_onto_simplex(a)
                    lr = initial_lr
                    mu = 0.9
                    momentum = np.zeros(n)
                    no_improve = 0

        iteration += 1
        if iteration % 5000 == 0:
            print(f"  Local refine iter {iteration}: score={best_score:.8f}, lr={lr:.2e}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses multiple initial shapes and projected gradient descent.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    n = len(height_sequence_1)
    print(f"Sequence length: {n}")

    # Generate candidate initializations
    candidates = []

    # 1. Linear ramp (gives ~4/3)
    ramp = make_ramp(n)
    candidates.append(("ramp", ramp))

    # 2. Quadratic valley
    quad = make_quadratic_valley(n)
    candidates.append(("quadratic", quad))

    # 3. Sinusoidal
    sin_seq = make_sinusoidal(n)
    candidates.append(("sinusoidal", sin_seq))

    # 4. Two ramps (tent)
    tent = make_two_ramps(n)
    candidates.append(("tent", tent))

    # 5. Parent sequence
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.sum() > 0.01:
        parent = parent / parent.sum()
        candidates.append(("parent", parent))

    # 6. Random smooth sequences
    for ri in range(4):
        freqs = rng.integers(1, 5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        amps = rng.uniform(0.2, 0.8, size=3)
        i = np.arange(n, dtype=np.float64)
        a_rand = 1.0
        for f, p, amp in zip(freqs, phases, amps):
            a_rand += amp * np.cos(2 * np.pi * f * i / n + p)
        a_rand = np.clip(a_rand, 1e-8, None)
        a_rand /= a_rand.sum()
        candidates.append((f"random_{ri}", a_rand))

    # Evaluate all candidates
    best_score = np.inf
    best_a = None
    for name, cand in candidates:
        score = compute_score(cand)
        print(f"  {name}: score = {score:.8f}")
        if score < best_score:
            best_score = score
            best_a = cand.copy()

    print(f"Best initial: {best_score:.8f}")

    # Time allocation: spend time refining the best candidates
    # Sort candidates by score and refine top ones
    scored_candidates = []
    for name, cand in candidates:
        score = compute_score(cand)
        scored_candidates.append((score, name, cand))
    scored_candidates.sort()

    # Refine top 3 candidates
    time_per_candidate = (deadline - time.monotonic()) / 3.0
    for rank, (score, name, cand) in enumerate(scored_candidates[:3]):
        if time.monotonic() >= deadline:
            break
        print(f"Refining '{name}' (initial score={score:.8f}), time budget={time_per_candidate:.1f}s")
        cand_deadline = time.monotonic() + time_per_candidate
        refined, refined_score = local_refine(
            cand, n, rng, cand_deadline, initial_lr=1e-3
        )
        if refined_score < best_score:
            best_score = refined_score
            best_a = refined.copy()
            print(f"  New best from '{name}': {refined_score:.8f}")

    # Additional random restarts from best_a
    remaining_time = deadline - time.monotonic()
    if remaining_time > 30:
        print(f"Doing random restarts with {remaining_time:.1f}s remaining")
        n_restarts = max(2, int(remaining_time / 30))
        restart_score = best_score
        for r in range(n_restarts):
            if time.monotonic() >= deadline:
                break
            noise_scale = 0.1 / (r + 1)
            a_restart = best_a + noise_scale * rng.standard_normal(n)
            a_restart = project_onto_simplex(a_restart)
            cand_deadline = min(time.monotonic() + remaining_time / n_restarts, deadline)
            refined, refined_score = local_refine(
                a_restart, n, rng, cand_deadline, initial_lr=5e-4
            )
            if refined_score < best_score:
                best_score = refined_score
                best_a = refined.copy()
                print(f"  Restart {r}: new best = {refined_score:.8f}")

    # Final verification
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    final_score = float(evaluate_sequence(final_list))
    print(f"Final verified score: {final_score:.8f}")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    return final_list
