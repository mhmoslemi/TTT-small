"""
Gradient-based optimization using log-sum-exp and power-sum objectives
to minimize the peak of the linear autocorrelation of a non-negative sequence.
Uses FFT-accelerated gradient computation and multiplicative updates.
"""

import numpy as np
import time


def _fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    N = 1
    while N < n:
        N *= 2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]


def _compute_gradient_power(a, p):
    """Compute gradient of sum(b^p) w.r.t. a using FFT."""
    n = len(a)
    b = _fft_convolve(a, a)
    if p == 1:
        weight = np.ones_like(b)
        coeff = 1.0
    else:
        weight = np.maximum(b, 0.0) ** (p - 1)
        coeff = float(p)
    # grad[j] = coeff * sum_{d=0}^{n-1} a[d] * weight[j+d]
    # = coeff * correlate(weight, a, 'valid')[j]
    # Using FFT: correlate(w, a) = ifft(fft(w) * conj(fft(a)))
    N = 1
    while N < len(weight) + n - 1:
        N *= 2
    W = np.fft.rfft(weight, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    grad = coeff * corr[:n]
    return grad


def _compute_gradient_lse(a, T):
    """Compute gradient of log-sum-exp(T*b) w.r.t. a using FFT."""
    n = len(a)
    b = _fft_convolve(a, a)
    # Softmax weights (numerically stable)
    b_max = np.max(b)
    exp_b = np.exp(T * (b - b_max))
    weight = exp_b / np.sum(exp_b)
    # grad[j] = sum_{d=0}^{n-1} a[d] * weight[j+d]
    N = 1
    while N < len(weight) + n - 1:
        N *= 2
    W = np.fft.rfft(weight, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    grad = corr[:n]
    return grad


def _objective(a):
    """Compute J = 2n * max(b) / (sum a)^2 directly."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    b = _fft_convolve(a, a)
    max_b = np.max(b)
    return float(2.0 * n * max_b / (s * s))


def _multiplicative_update(a, grad, eta):
    """Multiplicative weights update with normalization."""
    grad_centered = grad - np.mean(grad)
    log_update = -eta * grad_centered
    # Clip to prevent underflow/overflow
    log_update = np.clip(log_update, -5.0, 5.0)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s * np.sum(a)
    return a_new


def _make_initializations(n, target_sum, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # 1. Parent (normalized)
    if parent is not None:
        p = np.maximum(parent[:n], 0.0).copy()
        s = np.sum(p)
        if s > 0:
            inits.append(p / s * target_sum)

    # 2. Random exponential
    for trial in range(3):
        a_r = np.exp(rng.normal(0, 1.5, n))
        inits.append(a_r / a_r.sum() * target_sum)

    # 3. Sinusoidal modulations
    for freq in [1, 2, 3, 5, 7, 12, 20]:
        amp = 0.3 + 0.4 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a_s = 1.0 + amp * np.cos(2 * np.pi * freq * x / n + phase)
        a_s = np.maximum(a_s, 0.0)
        s = a_s.sum()
        if s > 0:
            inits.append(a_s / s * target_sum)

    # 4. Multi-frequency
    a_mf = np.ones(n)
    for _ in range(4):
        f = rng.integers(1, 30)
        ph = rng.uniform(0, 2 * np.pi)
        a_mf += 0.3 * np.cos(2 * np.pi * f * x / n + ph)
    a_mf = np.maximum(a_mf, 0.0)
    s = a_mf.sum()
    if s > 0:
        inits.append(a_mf / s * target_sum)

    # 5. Bump / Gaussian
    for center_frac in [0.3, 0.5, 0.7]:
        sigma = n * rng.uniform(0.05, 0.2)
        a_b = np.exp(-0.5 * ((x - center_frac * n) / sigma) ** 2)
        a_b += 0.1  # avoid exact zeros
        inits.append(a_b / a_b.sum() * target_sum)

    # 6. Block random
    n_blocks = rng.integers(4, 15)
    a_blk = np.ones(n)
    for i in range(n_blocks):
        start = i * n // n_blocks
        end = (i + 1) * n // n_blocks
        a_blk[start:end] = rng.uniform(0.05, 2.0)
        # Add some smoothness within block
        half = (end - start) // 2
        if half > 2:
            a_blk[start:start+half] *= np.linspace(0.8, 1.2, half)
            a_blk[start+half:end] *= np.linspace(1.2, 0.8, end - start - half)
    inits.append(a_blk / a_blk.sum() * target_sum)

    # 7. Large perturbation of parent
    if parent is not None:
        p = np.maximum(parent[:n], 0.0).copy()
        a_pert = p * np.exp(rng.normal(0, 2.0, n))
        a_pert = np.maximum(a_pert, 1e-6)
        inits.append(a_pert / a_pert.sum() * target_sum)

    # 8. "Anti-constant" - emphasize edges
    envelope = np.abs(x - n / 2) / (n / 2)
    a_edge = 0.2 + envelope
    a_edge += 0.2 * np.cos(6 * 2 * np.pi * x / n)
    a_edge = np.maximum(a_edge, 0.0)
    inits.append(a_edge / a_edge.sum() * target_sum)

    return inits


def _optimize_sequence(a, target_sum, deadline, rng, p_values, T_values):
    """Run gradient descent on a single sequence. Returns best found."""
    n = len(a)
    best_local = a.copy()
    best_score = _objective(a)

    # Phase 1: Power-sum objectives (from low p to high p)
    for p in p_values:
        if time.monotonic() > deadline:
            break
        n_steps = 3000
        eta_start = 0.3
        eta_end = 0.005
        for step in range(n_steps):
            if time.monotonic() > deadline:
                break
            grad = _compute_gradient_power(a, p)
            t = step / n_steps
            eta = eta_start * (1 - t) ** 2 + eta_end * t
            a = _multiplicative_update(a, grad, eta)
            if step % 1000 == 0:
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_local = a.copy()

    # Phase 2: Log-sum-exp objectives (increasing T for sharper peak targeting)
    for T in T_values:
        if time.monotonic() > deadline:
            break
        n_steps = 2000
        eta_start = 0.2
        eta_end = 0.003
        for step in range(n_steps):
            if time.monotonic() > deadline:
                break
            grad = _compute_gradient_lse(a, T)
            t = step / n_steps
            eta = eta_start * (1 - t) ** 2 + eta_end * t
            a = _multiplicative_update(a, grad, eta)
            if step % 500 == 0:
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_local = a.copy()

    # Final check
    score = _objective(a)
    if score < best_score:
        best_score = score
        best_local = a.copy()

    return best_local, best_score


def _local_search(a, deadline, rng):
    """Fine-tune with block perturbations."""
    n = len(a)
    best = a.copy()
    best_score = _objective(a)
    target_sum = np.sum(best)

    width_min = max(1, n // 500)
    width_max = max(2, n // 10)

    while time.monotonic() < deadline:
        candidate = best.copy()
        width = int(rng.integers(width_min, width_max + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.08)))
        candidate = np.maximum(candidate, 0.0)
        s = np.sum(candidate)
        if s < 0.01:
            continue
        candidate = candidate / s * target_sum
        score = _objective(candidate)
        if score < best_score:
            best_score = score
            best = candidate
            if best_score < 1.51:
                print(f"  Local search: J={best_score:.6f}")

    return best, best_score
