```python
import numpy as np
import time
from scipy.sparse import csr_matrix, lil_matrix
from scipy.optimize import linprog

def build_conv_matrix(a):
    """Build sparse matrix for conv(a, b) where b is the variable vector."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        # conv(a,b)_k = sum_i a[i] * b[k-i]
        # For valid i: 0 <= i < n and 0 <= k-i < n
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i  # column in b
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def evaluate_custom(a):
    """Fast evaluation without the security checks."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a ** 2))

def binary_local_search(a, rng, n_iter=10000):
    """Local search on binary (0/1) sequence."""
    a = (a > 0.5).astype(np.float64)
    n = len(a)
    best_score = evaluate_custom(a)
    
    for _ in range(n_iter):
        # Try flipping a random bit
        i = rng.integers(n)
        a_new = a.copy()
        a_new[i] = 1 - a[i]
        score = evaluate_custom(a_new)
        if score < best_score:
            a = a_new
            best_score = score
        elif rng.random() < 0.05:
            # Occasionally accept worse solutions (simulated annealing-like)
            a = a_new
            best_score = score
    
    return a, best_score
```

    n = len(v)

    n = len(a)

    score = fast_score(a)
    best_a = a.copy()

    alpha_decay = 0.999

            break

        c = np.convolve(a, a)
        max_c = np.max(c)

        threshold = max_c - 1e-6 * max_c

            best_a = a.copy()
            best_score = score

            print(f"    Subgrad iter {it}: score={score:.6f}, best={best_score:.6f}, alpha={alpha:.6f}")

def fast_score(a):
    """Compute the score quickly without security overhead."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def simplex_project(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    
    # Check if already valid
    if np.all(v >= -1e-15) and abs(np.sum(v) - target_sum) < 1e-10:
        return np.maximum(v, 0)
    
    # Bisection for lambda: find lambda such that sum(max(v - lambda, 0)) = target_sum
    lo = np.min(v) - target_sum - 1.0
    hi = np.max(v) + 1.0
    
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    lambda_val = (lo + hi) / 2.0
    proj = np.maximum(v - lambda_val, 0.0)
    
    # Numerical correction
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-12:
        nz = proj > 0
        if np.sum(nz) > 0:
            proj[nz] += diff / np.sum(nz)
            proj = np.maximum(proj, 0.0)
    
    return proj


def subgradient_substep(a, c, n, active_set):
    """Compute subgradient direction for reducing max convolution."""
    g = np.zeros(n)
    for k in active_set:
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            g[j_start:j_end + 1] += 2.0 * a[k - j_end:k - j_start + 1][::-1]
    g /= max(1, len(active_set))
    return g

def structured_initialization(n, rng, idx):
    """Generate a structured initial binary pattern."""
    if idx == 0:
        return np.ones(n)
    elif idx == 1:
        return (np.arange(n) % 3 != 0).astype(float)
    elif idx == 2:
        return (np.arange(n) % 3 != 2).astype(float)
    elif idx == 3:
        # Alternating blocks
        a = np.zeros(n)
        block = max(2, n // 10)
        for i in range(0, n, block * 2):
            a[i:min(n, i + block)] = 1.0
        return a
    elif idx == 4:
        # Prime-based pattern
        p = 2
        while p * p <= n:
            if n % p == 0:
                break
            p += 1
        a = np.ones(n)
        a[(np.arange(n) % p) >= p // 2] = 0.0
        return a
    elif idx == 5:
        # Random with 70% ones
        return (rng.random(n) > 0.3).astype(float)
    elif idx == 6:
        # Random with 80% ones
        return (rng.random(n) > 0.2).astype(float)
    elif idx == 7:
        # Sinusoidal mask
        return (np.sin(np.arange(n) * 2 * np.pi * 3.0 / n) > 0).astype(float)
    elif idx == 8:
        # Random with 60% ones
        return (rng.random(n) > 0.4).astype(float)
    else:
        return (rng.random(n) > 0.25).astype(float)


def continuous_optimize(a, rng, time_budget):
    """Subgradient-based continuous optimization."""
    deadline = time.monotonic() + time_budget
    n = len(a)
    a = a.astype(np.float64).copy()
    target_sum = np.sum(a)
    if target_sum < 0.01:
        target_sum = 1.0
        a = a / np.sum(a) * target_sum
    
    score = fast_score(a)
    best_a = a.copy()
    best_score = score
    
    # Momentum
    velocity = np.zeros(n)
    mu = 0.9  # momentum coefficient
    alpha = 0.05  # initial step size
    
    for it in range(50000):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(a, a)
        max_c = np.max(c)
        
        # Active set: top positions
        n_active = max(5, min(50, n // 5))
        active_set = np.argsort(c)[-n_active:]
        
        # Compute subgradient
        g = subgradient_substep(a, c, n, active_set)
        
        # Momentum update
        velocity = mu * velocity - alpha * g
        a_new = a + velocity
        
        # Project
        a_new = simplex_project(a_new, target_sum)
        
        new_score = fast_score(a_new)
        if new_score < score:
            a = a_new
            score = new_score
            alpha = min(0.2, alpha * 1.005)
        else:
            alpha *= 0.98
            velocity *= 0.5
        
        if score < best_score:
            best_a = a.copy()
            best_score = score
        
        if it % 5000 == 0 and it > 0:
            elapsed = time.monotonic() - (deadline - time_budget)
            print(f"      Continuous iter {it}: score={score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_score

    candidate_lengths = sorted(set(candidate_lengths))
    
    # Allocate time: more time for promising lengths
    total_time = budget_s - 60
    # Give equal time to each length initially, adjust based on results
    time_per_length = total_time / len(candidate_lengths)
    
    for n_try in candidate_lengths:
        if time.monotonic() > deadline - 30:
            break
        
        # Dynamic time allocation: give more time to lengths that already show promise
        allocated = min(time_per_length * 1.5, deadline - time.monotonic() - 20)
        if allocated < 10:
            allocated = 10
        
        print(f"\n=== Trying length {n_try} (budget {allocated:.0f}s) ===")
        
        best_for_n = None
        best_for_n_score = np.inf
        
        # Phase A: Binary search with multiple initializations
        n_restarts = max(3, int(allocated / 30))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 40:
                break
            t_alloc = min(allocated / max(1, n_restarts - r), deadline - time.monotonic() - 30)
            if t_alloc < 5:
                t_alloc = 5
            
            init = structured_initialization(n_try, rng, r % 9)
            a, s = binary_local_search(n_try, rng, t_alloc, initial=init)
            
            if a is not None and s < best_for_n_score:
                best_for_n = a
                best_for_n_score = s
                print(f"    After binary restart {r}: {s:.6f}")
        
        # Phase B: Continuous refinement
        if best_for_n is not None:
            cont_time = min(allocated * 0.4, deadline - time.monotonic() - 30)
            if cont_time > 10:
                a_cont, s_cont = continuous_optimize(best_for_n, rng, cont_time)
                if s_cont < best_for_n_score:
                    best_for_n = a_cont
                    best_for_n_score = s_cont
                    print(f"    After continuous: {s_cont:.6f}")
        
        # Final check with real evaluator
        if best_for_n is not None:
            real_score = float(evaluate_sequence(best_for_n.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_seq = best_for_n.copy()
                best_score = real_score
                print(f"  *** New global best: {best_score:.6f} (length {n_try}) ***")
    
    # Final polish phase
    if time.monotonic() < deadline - 15:
        print(f"\n=== Final polish (score={best_score:.6f}) ===")
        polish_deadline = deadline - 5
        n_polish = len(best_seq)
        
        for iteration in range(100000):
            if time.monotonic() > polish_deadline:
                break
            
            # Small random perturbation
            sigma = 0.05 * (1.0 + 0.5 * np.cos(iteration * 0.01))
            candidate = best_seq + rng.normal(0, sigma, n_polish)
            candidate = np.clip(candidate, 0.0, 1000.0)
            
            s = fast_score(candidate)
            if s < best_score:
                real_s = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(real_s) and real_s < best_score:
                    best_seq = candidate.copy()
                    best_score = real_s
                    print(f"  Polish iter {iteration}: {best_score:.6f}")
    
    print(f"\nFinal result: score={best_score:.6f}, length={len(best_seq)}")
    return [float(x) for x in best_seq]
