def _project_simplex(v):
    n = len(v)
    s = np.sum(v)
    if s <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    cols = []

Uses LP-based iteration (with non-uniform starts to avoid the uniform fixed point),
subgradient descent, and block perturbation polishing across multiple lengths.
Key insight: uniform sequence is a fixed point of the LP iteration, so we must start
from non-uniform sequences (triangular, exponential, random) for the LP to make progress.
"""

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse


def _score_norm(a):
    """Compute score for a sequence assumed to have sum=1."""
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * float(np.max(c))


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.zeros(0)
    s = np.sum(v)
    if s <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_cands = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_cands) == 0:
        return np.ones(n) / n
    rho = rho_cands[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _build_sparse_conv_mat(a):
    """Build sparse (2n-1 x n) matrix where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    row_idx = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    col_idx = np.repeat(np.arange(n), n)
    data = np.tile(a, n)
    return scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))


def _solve_lp(a, M, timeout=30.0):
    """Solve: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0. Returns normalized b or None."""
    n = len(a)
    try:
        A = _build_sparse_conv_mat(a)
    except Exception:
        return None
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': timeout, 'presolve': True})
    except Exception:
        return None
    if res is None or not res.success or res.x is None:
        return None
    b = np.asarray(res.x, dtype=np.float64)
    s = float(np.sum(b))
    if s < 1e-12:
        return None
    return b / s


def _lp_optimize(a_init, time_limit, seed=0):
    """LP-based iterative optimization starting from non-uniform a_init."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = float(np.sum(a))
    if s < 1e-12:
        a = np.ones(n)
    a = a / np.sum(a)

    best = a.copy()
    best_score = _score_norm(a)
    stall = 0

    while time.monotonic() < time_limit:
        c = np.convolve(a, a)
        M = float(np.max(c))

        b = _solve_lp(a, M, timeout=20.0)
        if b is None:
            break

        improved = False
        for t in [0.1, 0.2, 0.5, 0.3, 0.05, 0.7, 0.15, 0.4, 0.8, 0.25, 0.6]:
            new_a = np.maximum((1.0 - t) * a + t * b, 0.0)
            ns = float(np.sum(new_a))
            if ns < 1e-12:
                continue
            new_a = new_a / ns
            sc = _score_norm(new_a)
            if sc < best_score - 1e-10:
                best = new_a.copy()
                best_score = sc
                a = new_a
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall > 80:
                # Perturb to escape fixed point
                a = a + rng.normal(0, 0.05, n)
                a = _project_simplex(a)
                stall = 0

    return best, best_score


def _subgrad_optimize(a_init, time_limit, seed=0):
    """Projected subgradient descent with momentum."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    a = a / np.sum(a)

    best = a.copy()
    best_score = _score_norm(a)

    momentum = 0.9
    v = np.zeros(n)

    for it in range(100000000):
        if time.monotonic() > time_limit:
            break

        c = np.convolve(a, a)
        k_star = int(np.argmax(c))

        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        g[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]

        v = momentum * v + (1.0 - momentum) * g
        lr = 4.0 / (np.sqrt(n) * (1.0 + 0.01 * float(it)))
        a_new = a - lr * v
        a_new = _project_simplex(a_new)

        sc = _score_norm(a_new)
        if sc < best_score:
            best = a_new.copy()
            best_score = sc

        a = a_new

    return best, best_score


def _block_perturb_optimize(a_init, time_limit, seed=0):
    """Local search with random block multiplicative perturbations."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    a = a / np.sum(a)

    best = a.copy()
    best_score = _score_norm(a)
    stall = 0

    while time.monotonic() < time_limit:
        width = int(rng.integers(max(1, n // 300), max(2, n // 5)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        candidate = a.copy()
        candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))

        if rng.random() < 0.25:
            candidate = 0.5 * (candidate + candidate[::-1])

        candidate = np.maximum(candidate, 0.0)
        st = float(np.sum(candidate))
        if st < 1e-12:
            stall += 1
            continue
        candidate = candidate / st
        sc = _score_norm(candidate)

        if sc < best_score - 1e-10:
            best = candidate.copy()
            best_score = sc
            a = candidate
            stall = 0
        else:
            stall += 1
            if stall > 1000:
                a = _project_simplex(a + rng.normal(0, 0.08, n))
                stall = 0

    return best, best_score


def _multiplicative_weights(a_init, time_limit, seed=0):
    """Multiplicative weights update penalizing peak contributors."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    a = a / np.sum(a)

    best = a.copy()
    best_score = _score_norm(a)
    eta = 0.05
    stall = 0

    for it in range(100000000):
        if time.monotonic() > time_limit:
            break

        c = np.convolve(a, a)
        k_star = int(np.argmax(c))

        mult = np.ones(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        mult[lo:hi + 1] = np.exp(-eta * a[k_star - hi:k_star - lo + 1][::-1])

        a_new = a * mult
        ns = float(np.sum(a_new))
        if ns < 1e-12:
            break
        a_new = a_new / ns

        sc = _score_norm(a_new)
        if sc < best_score:
            best = a_new.copy()
            best_score = sc
            stall = 0
        else:
            stall += 1

        a = a_new
        eta *= 0.9995

        if stall > 200:
            eta = 0.05
            a = _project_simplex(a + rng.normal(0, 0.03, n))
            stall = 0

    return best, best_score


def _make_initializations(n, seed):
    """Generate diverse non-uniform initializations of length n."""
    rng = np.random.default_rng(seed)
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Triangular (peaked in center)
    a_tri = np.maximum(1.0 - np.abs(x - mid) / (mid + 1.0), 0.0)
    if np.sum(a_tri) > 0:
        inits.append(a_tri / np.sum(a_tri))

    # Exponential (decaying)
    a_exp = np.exp(-2.0 * x / n)
    inits.append(a_exp / np.sum(a_exp))

    # Two-sided exponential
    a_bi = np.exp(-2.0 * np.abs(x - mid) / n)
    inits.append(a_bi / np.sum(a_bi))

    # Random
    a_rnd = rng.random(n)
    inits.append(a_rnd / np.sum(a_rnd))

    # Sine-modulated
    a_sin = np.maximum(0.5 + 0.4 * np.sin(2.0 * np.pi * x / n), 0.01)
    inits.append(a_sin / np.sum(a_sin))

    # Block pattern (alternating high/low)
    a_blk = np.where(np.arange(n) % 4 < 2, 1.0, 0.1)
    inits.append(a_blk / np.sum(a_blk))

    return inits

    """
    Search for non-negative sequence minimizing 2n*max(convolve(a,a))/(sum(a))^2.

    best_score = float('inf')

    print(f"Starting search. Budget: {budget_s}s. Deadline in {float(budget_s)-5:.0f}s.", flush=True)

    print("=== Phase 1: LP optimization ===", flush=True)

        if remaining < 20:
            break

        if n <= 128:
            alloc = min(30.0, remaining * 0.25)
        elif n <= 256:
            alloc = min(45.0, remaining * 0.25)
        elif n <= 512:
            alloc = min(60.0, remaining * 0.25)
        elif n <= 1024:
            alloc = min(100.0, remaining * 0.25)
        else:
            alloc = min(100.0, remaining * 0.25)

        inits = _make_initializations(n, seed + n)

        for idx, init in enumerate(inits[:3]):  # Try up to 3 inits per n
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a, sc = _lp_optimize(init, time.monotonic() + min(alloc / 2.0, remaining - 5), seed=seed + n * 10 + idx)
            if sc < best_score:
                best_score = sc
                best_seq = a
                print(f"  LP n={n} init={idx}: {sc:.8f} ({time.monotonic() + 0 - (deadline - remaining):.1f}s used)", flush=True)
            else:
                print(f"  LP n={n} init={idx}: {sc:.8f} (no improvement, best={best_score:.8f})", flush=True)

    # Phase 2: Multiplicative weights on best found so far
    print("=== Phase 2: Multiplicative weights ===", flush=True)
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            n = len(best_seq)
            alloc = min(90.0, remaining * 0.3)
            a, sc = _multiplicative_weights(best_seq, time.monotonic() + alloc, seed=seed + 100)
            if sc < best_score:
                best_score = sc
                best_seq = a
                print(f"  MW n={n}: {sc:.8f}", flush=True)
            else:
                print(f"  MW n={n}: {sc:.8f} (no improvement)", flush=True)

    # Phase 3: Subgradient for larger n
    print("=== Phase 3: Subgradient ===", flush=True)
    for n in [2048, 4096, 6733]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        alloc = min(120.0, remaining * 0.3)

        if best_seq is not None and len(best_seq) == n:
            init = best_seq
        else:
            # Use a non-uniform init
            inits = _make_initializations(n, seed + n * 7)
            init = inits[0]  # triangular

        a, sc = _subgrad_optimize(init, time.monotonic() + alloc, seed=seed + 200 + n)
        if sc < best_score:
            best_score = sc
            best_seq = a
            print(f"  Subgrad n={n}: {sc:.8f}", flush=True)
        else:
            print(f"  Subgrad n={n}: {sc:.8f} (no improvement)", flush=True)

    print("=== Phase 4: Block perturbation polishing ===", flush=True)
    if best_seq is not None:
        remaining = deadline - time.monotonic()

            n = len(best_seq)
            a, sc = _block_perturb_optimize(best_seq, deadline - 3, seed=seed + 300)
            if sc < best_score:
                best_score = sc
                best_seq = a
                print(f"  Block n={n}: {sc:.8f}", flush=True)

                print(f"  Block n={n}: {sc:.8f} (no improvement)", flush=True)
