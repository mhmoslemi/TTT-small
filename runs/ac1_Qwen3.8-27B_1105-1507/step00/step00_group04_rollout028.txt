    # For efficiency, we can use the fact that this is a convolution matrix.
    # A is (2n-1) x n, with A[k,j] = f[k-j] for max(0, k-n+1) <= j <= min(n-1, k)

    # Row k, col j is nonzero when 0 <= k-j < n, i.e., k-n+1 <= j <= k, and 0 <= j < n.
    # So j ranges from max(0, k-n+1) to min(n-1, k).
    
    rows = []
    cols = []
    vals = []

            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
