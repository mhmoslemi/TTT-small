def _clean_start():
    """Load the provided parent if it looks sane, else a gentle ramp+flat mix."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        if p.ndim == 1 and p.size >= 8 and np.isfinite(p).all():
            p = np.clip(p, 0.0, 1.0)
            if p.sum() > 0:
                return p / p.sum()
    except Exception:
        pass
    n = 511
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.sin(2 * np.pi * 7 * x) + 0.3 * np.cos(2 * np.pi * 13 * x)
    a = np.clip(a, 0.05, None)
    return a / a.sum()


def fast_score(a):
    """2*n*max(conv(a,a))/sum(a)^2 via rfft (exact same math as the evaluator)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    N = 2 * n - 1
    f = np.fft.rfft(a, N)
    c = np.fft.irfft(f * f, N)
    c = np.where(c > 0, c, 0.0)  # drop tiny negative round-off
    return float(2.0 * n * c.max() / (s * s))


def _active_lags(c, cap=128):
    m = c.max()
    idx = np.flatnonzero(c >= m * (1.0 - 1e-6))
    if idx.size > cap:
        order = np.argpartition(c, -cap)[-cap:]
        idx = order
    return idx


def _grad_for_lags(a, lags):
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    for k in lags:
        k = int(k)
        lo1, hi1 = max(0, k - n + 1), min(k + 1, n)
        if hi1 > lo1:
            g[lo1:hi1] += a[(k - lo1):(k - hi1 + 1):-1]
        lo2, hi2 = 0, min(n, n - k)
        if hi2 > lo2:
            g[lo2:hi2] += a[k:k + (hi2 - lo2)]
    return g


def _gaussian_blur(a, sigma):
    n = a.size
    if sigma <= 0:
        return a
    N = 2 * n
    x = np.fft.fftfreq(N)
    h = np.exp(-2 * (math.pi * sigma) ** 2 * x * x)
    b = np.fft.irfft(np.fft.rfft(a, N) * h, N)[:n]
    b = np.where(b > 0, b, 0.0)
    return b


def refine(a, rng, budget_s):
    """Multiplicative subgradient reduction of the convolution peak."""
    n = a.size
    a = a.copy()
    s = a.sum()
    if s <= 0:
        return a
    a = a / s
    score = fast_score(a)
    best = a.copy()
    best_score = score
    eta = 0.25
    sweep = 0
    t0 = time.monotonic()
    while time.monotonic() - t0 < budget_s:
        N = 2 * n - 1
        f = np.fft.rfft(a, N)
        c = np.fft.irfft(f * f, N)
        c = np.where(c > 0, c, 0.0)
        lags = _active_lags(c)
        g = _grad_for_lags(a, lags)
        gm = g.max()
        if gm <= 0:
            break
        h = g / gm
        a2 = a / (1.0 + eta * h)
        if a2.sum() <= 0:
            eta *= 0.5
            continue
        a2 = a2 / a2.sum()
        sc2 = fast_score(a2)
        if sc2 < score - 1e-12:
            a, score = a2, sc2
            if score < best_score:

            eta = min(0.9, eta * 1.03)
        else:

            if eta < 0.02:

                a = np.where(a > 0, a, 0.0)
                a = a / a.sum()
                score = fast_score(a)
                if score < best_score:
                    best_score, best = score, a.copy()
                eta = 0.2
        sweep += 1
        if sweep % 400 == 0:
            a = _gaussian_blur(a, max(1.0, n // 600))
            a = np.where(a > 0, a, 0.0)
            a = a / a.sum()
            score = fast_score(a)
            if score < best_score:
                best_score, best = score, a.copy()
    return best


def pulse_train(n, m, w, phase, rng):
    """n long, m equal top-hat pulses of width w, phase-decorrelated positions."""
    a = np.zeros(n, dtype=np.float64)
    if w < 1:
        w = 1
    n_per = n // float(m)
    for i in range(m):
        base = (i + 1) * n_per + phase
        pos = int(round(base % n))
        a[pos:(pos + w)] += 1.0
    s = a.sum()
    if s <= 0:
        return None
    return a / s


def concave_init(n, rng, alpha):
    """A smoothly decaying profile (large->small) which already beats the uniform T=2."""
    x = np.linspace(0.0, 1.0, n)
    a = (1.0 - alpha * x) ** 3 + 0.02
    a = np.clip(a, 0.0, None)
    if a.sum() <= 0:
        return None
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Docstring: pulse-train + concave inits polished by FFT multiplicative
    subgradient minimization of the self-convolution peak; final answer is the
    candidate with the lowest authoritative evaluate_sequence score."""
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 5.0)
    t_end = time.monotonic() + total_budget

    start = _clean_start()
    n = start.size
    if n < 64:
        n = 512
        start = concave_init(n, rng, 0.5)

    # Build a diverse pool of initializations (all sum-normalized, nonnegative).
    pool = []
    pool.append((start, fast_score(start)))
    for alpha in (0.3, 0.7, 1.0, 1.5):
        c = concave_init(n, rng, alpha)
        if c is not None:
            pool.append((c, fast_score(c)))
    # pulse trains: M chosen around sqrt(2n/T) with T in a sweep
    for target in (1.5, 1.7, 2.0, 2.5, 3.5):
        m = int(round(math.sqrt(2.0 * n / target)))
        m = max(8, min(m, n // 12))
        for w in (2, 3, 5, 8):
            if w * 2 >= m:
                continue
            for _ in range(2):
                phase = rng.uniform(0, n)
                p = pulse_train(n, m, w, phase, rng)
                if p is not None:
                    pool.append((p, fast_score(p)))
    for w in (4, 10, 20):
        m = max(8, n // (2 * w + 2))
        p = pulse_train(n, m, w, rng.uniform(0, n), rng)
        if p is not None:
            pool.append((p, fast_score(p)))

    # Keep the best few seeds and polish each within a sub-budget.
    pool.sort(key=lambda t: t[1])
    topk = pool[: min(8, len(pool))]

    # First pass: quick polish of every seed.
    quick = min(60.0, total_budget * 0.25)
    best_a, best_s = None, np.inf
    for a, sc in topk:
        if time.monotonic() > t_end:
            break
        a2 = refine(a, rng, quick)
        s2 = fast_score(a2)
        if s2 < best_s:
            best_s, best_a = s2, a2
    print("after quick polish: fast=%.6f" % best_s, flush=True)

    # Second pass: heavy polish on the current best (and a couple alternates).
    remaining = t_end - time.monotonic()
    if best_a is not None and remaining > 5:
        a2 = refine(best_a, rng, max(5.0, remaining * 0.7))
        s2 = fast_score(a2)
        if s2 < best_s:
            best_s, best_a = s2, a2
        print("after heavy polish: fast=%.6f" % best_s, flush=True)

    # Also try one more alternate seed with leftover time if it is close.
    remaining = t_end - time.monotonic()
    if remaining > 30 and len(topk) > 1:
        alt = topk[1][0]
        a3 = refine(alt, rng, remaining)
        s3 = fast_score(a3)
        if s3 < best_s:
            best_s, best_a = s3, a3
        print("after alt polish: fast=%.6f" % best_s, flush=True)

    if best_a is None:
        best_a = start

    # Validate a small set of finalists with the authoritative evaluator.
    finalists = [best_a]
    for a, _ in pool[:3]:
        finalists.append(a)
    finalists.append(start)
    out_best, out_score = None, np.inf
    for a in finalists:
        cand = [float(v) for v in np.clip(a, 0.0, 1000.0)]
        sc = float(evaluate_sequence(cand))
        print("finalist eval: %.6f" % sc, flush=True)
        if sc < out_score:
            out_score, out_best = sc, cand
    if out_best is None:
        out_best = [float(v) for v in np.clip(best_a, 0.0, 1000.0)]
    return out_best
