```python
"""
Search for low evaluation sequences for the auto-convolution objective

    score(a) = 2*n*max(convolve(a,a)) / (sum(a))^2

with a_i >= 0.

The code uses two complementary ideas:

1. An analytic strong starting family: the truncated coefficients of
   (1-z)^(-1/2), i.e.
       b_i = binom(2i, i) / 4^i.
   In raw scale this sequence has auto-convolution coefficients equal to 1
   on the first half of the full convolution, and its score approaches
   pi/2 from above as n grows.  This is far better than the uniform sequence
   (score 2) and is used as the safe fallback.

2. A randomized adaptive multiplicative search in log-space.  Starting from
   the base family (or small perturbed variants of it), it minimizes a smooth
   approximation to the maximum convolution using softmax-based weights, then
   accepts improvements with a line search.  A block multiplicative local
   search is also used for polishing.

All helper functions are top-level and there are no lambda functions or
closures.
"""

import numpy as np
import time

try:
    from scipy.signal import fftconvolve
except Exception:
    fftconvolve = None


FALLBACK_N = 8192

# (length, nominal duration in seconds, initial log-space perturbation scales)
SEARCH_CONFIGS = [
    (8192, 260, [0.0, 0.08, 0.16]),
    (4096, 220, [0.0, 0.10, 0.22]),
    (2048, 160, [0.0, 0.18, 0.35]),
]


def make_base_with_sum(n):
    """Return raw central-binomial base sequence b_i and its sum."""
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64), 0.0

    b = np.empty(n, dtype=np.float64)
    val = 1.0
    s = 1.0
    b[0] = 1.0

    for i in range(1, n):
        val *= (2 * i - 1) / (2 * i)
        b[i] = val
        s += val

    return b, s


def convolve_two(x, y):
    """Full linear convolution, using FFT for larger vectors when available."""
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)

    if x.size == 0 or y.size == 0:
        return np.array([], dtype=np.float64)

    if fftconvolve is not None and max(x.size, y.size) > 256:
        return fftconvolve(x, y)

    return np.convolve(x, y)


def score_normalized(a):
    """
    Score for a normalized sequence with sum(a) == 1.
    Returns 2*n*max(conv(a,a)).
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")

    c = convolve_two(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")

    return 2.0 * float(n) * m


def score_exact(a):
    """
    Exact non-FFT score using np.convolve.  Useful for final validation.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")

    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return float("inf")

    if abs(s) > 1.0e100:
        b = a / s
        c = np.convolve(b, b)
        m = float(np.max(c))
        if not np.isfinite(m):
            return float("inf")
        return 2.0 * float(n) * m

    c = np.convolve(a, a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")

    return 2.0 * float(n) * m / (s * s)


def normalize_exp(x):
    """Normalize exp(x) robustly."""
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return None

    if not np.all(np.isfinite(x)):
        x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=-1000.0)

    m = float(np.max(x))
    y = np.exp(x - m)
    s = float(np.sum(y))

    if s <= 0.0 or not np.isfinite(s):
        return np.full(x.size, 1.0 / x.size, dtype=np.float64)

    return y / s


def direction_h(a, factor):
    """
    Compute a centered multiplicative gradient direction for reducing the
    smooth maximum of the auto-convolution.

    factor controls softmax temperature: smaller factor gives cruder max.
    Returns h where log-update is x <- x - lr*h, or None on failure.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return None

    c = convolve_two(a, a)
    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax):
        return None

    temp = cmax / float(factor)
    z = (c - cmax) / temp
    z = np.clip(z, -60.0, 60.0)

    w = np.exp(z)
    ws = float(np.sum(w))
    if ws <= 0.0 or not np.isfinite(ws):
        w = np.zeros_like(c, dtype=np.float64)
        k = int(np.argmax(c))
        w[k] = 1.0
    else:
        w = w / ws

    # g_j = sum_k w_k * a_{k-j}   (with valid indices).
    # This is obtained by convolving w with reversed a and slicing.
    full = convolve_two(w, a[::-1])
    if full.size < (2 * n - 1):
        full = np.convolve(w, a[::-1])

    g = full[n - 1:2 * n - 1]

    if g.size != n or not np.all(np.isfinite(g)):
        return None

    h = (g - cmax) / cmax
    h -= float(np.mean(h))
    h = np.clip(h, -20.0, 20.0)

    return h


def block_search(a, n, seed, dur, best_sc=None):
    """
    Simple multiplicative block local search.  Keeps best incumbent and uses
    random blocks with random positive multipliers, followed by renormalization.
    """
    if dur <= 0.0:
        if best_sc is None:
            best_sc = score_normalized(a)
        return np.asarray(a, dtype=np.float64), best_sc

    rng = np.random.default_rng(seed)

    best_a = np.asarray(a, dtype=np.float64).copy()
    if best_a.size != n:
        best_a = normalize_exp(np.arange(n, dtype=np.float64))
        if best_a is None:
            best_a = np.full(n, 1.0 / n, dtype=np.float64)

    s = float(np.sum(best_a))
    if s <= 0.0 or not np.isfinite(s):
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        s = 1.0
    else:
        best_a = best_a / s

    if best_sc is None:
        best_sc = score_normalized(best_a)

    start = time.monotonic()
    no_improve = 0

    min_w = max(1, n // 1000)
    max_w = max(min_w + 1, n // 15)

    while time.monotonic() - start < dur:
        w = int(rng.integers(min_w, max_w + 1))
        if w > n:
            w = n
        st = int(rng.integers(0, n - w + 1))

        if rng.random() < 0.8:
            sigma = 0.05 + 1.2 * float(rng.random())
            fac = np.exp(rng.normal(0.0, sigma, w))
        else:
            fac = 0.25 + 3.0 * rng.random(w)

        cand = best_a.copy()
        cand[st:st + w] *= fac

        cs = float(np.sum(cand))
        if cs <= 0.0 or not np.isfinite(cs):
            continue

        cand = cand / cs
        sc = score_normalized(cand)

        if np.isfinite(sc) and sc < best_sc - 1.0e-12:
            best_a = cand
            best_sc = sc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                break

    return best_a, best_sc


def adam_search_from(a, n, seed, dur):
    """
    Adaptive multiplicative gradient search in log-space, followed by a
    block-local-search polishing phase.  Input a is approximately normalized.
    """
    if dur <= 0.0:
        a = np.asarray(a, dtype=np.float64)
        s = float(np.sum(a))
        if s <= 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
            s = 1.0
        else:
            a = a / s
        return a, score_normalized(a)

    rng = np.random.default_rng(seed)

    best_a = np.asarray(a, dtype=np.float64).copy()
    if best_a.size != n:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    s = float(np.sum(best_a))
    if s <= 0.0 or not np.isfinite(s):
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        s = 1.0
    else:
        best_a = best_a / s

    best_sc = score_normalized(best_a)

    # Log representation of the incumbent.
    x = np.log(np.maximum(best_a, 1.0e-300))
    x = x - float(np.max(x))

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1_power = 1.0
    b2_power = 1.0

    factors = [80, 40, 20, 10, 5, 2]
    lrs = [0.005, 0.02, 0.05, 0.1, 0.2]

    start = time.monotonic()
    adam_time = min(dur, 0.65 * dur)
    no_improve = 0
    last_print = start

    while time.monotonic() - start < adam_time:
        factor = factors[int(rng.integers(0, len(factors)))]
        h = direction_h(best_a, factor)
        if h is None:
            break

        t += 1
        b1_power *= 0.9
        b2_power *= 0.999

        m = 0.9 * m + 0.1 * h
        v = 0.999 * v + 0.001 * (h * h)

        mh = m / (1.0 - b1_power)
        vh = v / (1.0 - b2_power)
        step = mh / (np.sqrt(vh) + 1.0e-8)

        best_local = best_sc
        best_x = None
        best_cand = None

        for lr in lrs:
            xc = x - lr * step
            a2 = normalize_exp(xc)
            if a2 is None:
                continue

            sc = score_normalized(a2)
            if np.isfinite(sc) and sc < best_local - 1.0e-12:
                best_local = sc
                best_x = xc
                best_cand = a2

        if best_cand is not None:
            best_a = best_cand
            best_sc = best_local
            x = best_x - float(np.max(best_x))
            no_improve = 0

            if time.monotonic() - last_print > 5.0:
                print(f"adam n={n} score={best_sc:.6f}")
                last_print = time.monotonic()
        else:
            no_improve += 1
            if no_improve > 800:
                break

    remaining = dur - (time.monotonic() - start)
    if remaining > 5.0:
        ba, bs = block_search(best_a, n, seed + 12345, remaining, best_sc)
        best_a = ba
        best_sc = bs

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1500.0):
    """
    Returns a non-negative coefficient list.  The global variable
    height_sequence_1 may exist, but this constructor starts from the
    analytic central-binomial base and searches around it.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)) - 12.0

    # Safe analytic fallback.  Raw scale is fine; the score is scale-invariant.
    fallback_seq, fallback_sum = make_base_with_sum(FALLBACK_N)
    fallback_score = 2.0 * float(FALLBACK_N) / (fallback_sum * fallback_sum)

    best_seq = fallback_seq
    best_score = fallback_score
    using_fallback = True

    print(f"fallback n={FALLBACK_N} score={fallback_score:.6f}")

    # Precompute normalized base sequences for attempted lengths.
    base_by_n = {}
    for n, _dur, _perts in SEARCH_CONFIGS:
        b, s = make_base_with_sum(n)
        base_by_n[int(n)] = b / s

    # Scale nominal durations if the requested budget is small.
    total_nominal = float(sum(d for _n, d, _p in SEARCH_CONFIGS))
    remaining = deadline - time.monotonic()
    scale = 1.0
    if total_nominal > remaining:
        scale = max(0.0, (remaining - 20.0) / total_nominal)

    for n, nominal_dur, perts in SEARCH_CONFIGS:
        if time.monotonic() >= deadline:
            break

        dur = min(float(nominal_dur) * scale, max(0.0, deadline - time.monotonic() - 8.0))
        if dur < 20.0:
            continue

        base_a = base_by_n[int(n)]
        base_sc = score_normalized(base_a)

        if base_sc < best_score - 1.0e-9:
            # This can happen if fallback length is smaller than searched length
            # or due to finite effects.  Accept it.
            best_seq = base_a.copy()
            best_score = base_sc
            using_fallback = False
            print(f"base improvement n={n} score={best_sc:.6f}")

        for idx, pert in enumerate(perts):
            if time.monotonic() >= deadline:
                break

            rem = deadline - time.monotonic() - 8.0
            if rem <= 0:
                break

            # First attempt gets more time; perturbed starts are exploratory.
            if pert == 0.0:
                attempt_dur = min(rem, dur * 0.6)
            else:
                attempt_dur = min(rem, dur * 0.2)

            if attempt_dur < 8.0:
                continue

            a_start = base_a.copy()
            if pert > 0.0:
                rngp = np.random.default_rng(int(seed) * 1000003 + int(n) * 31 + idx)
                xp = np.log(a_start) + pert * rngp.normal(size=n)
                a_start = normalize_exp(xp)
                if a_start is None:
                    continue

            a_best, sc_best = adam_search_from(
                a_start,
                int(n),
                int(seed) * 7919 + int(n) * 257 + 11 * idx,
                attempt_dur,
            )

            if np.isfinite(sc_best) and sc_best < best_score - 1.0e-9:
                best_seq = a_best.copy()
                best_score = sc_best
                using_fallback = False
                print(f"search improvement n={n} pert={pert:.2f} score={best_score:.6f}")

    # If we are not using the analytic fallback, validate the returned candidate
    # with a non-FFT exact score when enough time remains.  If validation reveals
    # that the FFT-based incumbent was misleadingly worse than the fallback,
    # revert to the fallback.
    if not using_fallback and (deadline - time.monotonic()) > 25.0:
        try:
            exact_score = score_exact(best_seq)
        except Exception:
            exact_score = float("inf")

        if np.isfinite(exact_score):
            if exact_score <= best_score + 1.0e-7:
                best_score = exact_score
            elif exact_score >= fallback_score - 1.0e-9:
                # The "improvement" was not verified; fall back to the safe base.
                best_seq = fallback_seq
                best_score = fallback_score
                using_fallback = True
                print("reverted to analytic fallback after exact validation")

    # Ensure the returned object is a Python list of finite non-negative floats.
    out = np.asarray(best_seq, dtype=np.float64)
    out = np.nan_to_num(out, nan=0.0, posinf=1000.0, neginf=0.0)
    out = np.clip(out, 0.0, 1000.0)

    # If all values were clipped to zero for some unforeseen reason, return a
    # small valid uniform sequence.
    if out.size == 0 or float(np.sum(out)) <= 0.0:
        out = np.ones(64, dtype=np.float64)

    return [float(x) for x in out]
```
