    s = np.sum(a)

        return np.inf

    return 2.0 * n * np.max(b) / (s * s)

    grad[valid] = a[idx[valid]]

"""
Multi-method search for low-peak autocorrelation step function sequences.
Combines parametric initialization, LP-based directional moves, subgradient
descent with exponentiated gradients, and adaptive learning rates.
Explores multiple sequence lengths and starting points within time budget.
"""
import numpy as np
from scipy.optimize import linprog
import time


def compute_objective(a):
    """Fast local computation of the objective (matches evaluate_sequence)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * n * np.max(b) / (s * s))


def subgradient_step(a, lr):
    """One exponentiated-gradient subgradient step targeting the peak lag."""
    n = len(a)
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))
    # Subgradient of max(conv(a,a)) w.r.t. a: grad[j] = a[k-j] if valid
    idx = k - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    grad = np.zeros(n)
    grad[valid] = a[idx[valid]]
    gmax = np.max(grad)
    if gmax < 1e-15:
        return a, k
    a_new = a * np.exp(-lr * grad / gmax)
    s = np.sum(a_new)
    if s < 1e-15:
        return a, k
    a_new = a_new / s
    return a_new, k


def multi_subgradient(a, n_iter, lr_init, cooldown, time_limit):
    """Run subgradient iterations with adaptive learning rate."""
    a = a.copy()
    lr = lr_init
    best_a = a.copy()

    stall = 0
    t_start = time.monotonic()
    for it in range(n_iter):
        if time.monotonic() - t_start > time_limit:
            break
        a_new, k = subgradient_step(a, lr)
        obj = compute_objective(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            lr = min(lr * 1.05, 2.0)

            lr *= 0.9
            if stall > 80:
                break
    return best_a, best_obj


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def lp_iteration(a, n_iters, time_limit):
    """LP-based iterative improvement using cross-convolution constraints."""
    a = a.copy()
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_objective(a)
    t_start = time.monotonic()
    n = len(a)

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit:
            break
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        c = -np.ones(n)
        A = build_conv_matrix(a)
        b = np.full(2 * n - 1, M)
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                          method='highs', options={'presolve': True})
        except Exception:
            break
        if res.status != 0:
            break
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1]:
            a_try = t * g + (1.0 - t) * a
            a_try = np.maximum(a_try, 0.0)
            ss = np.sum(a_try)
            if ss < 1e-15:

            obj = compute_objective(a_try)

                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
        if not improved:
            if it > 10:
                break
    return best_a, best_obj

                best_score = obj_ref
                best_a = a_ref.copy()
                print(f"  After refinement: {best_score:.6f}")

            a_ref2, obj_ref2 = peak_shave(best_a, n_iter=500, time_limit=t_final * 0.3)
            if obj_ref2 < best_score:
                best_score = obj_ref2
                best_a = a_ref2.copy()
                print(f"  After peak shave: {best_score:.6f}")

    # Verify with actual evaluate_sequence
    if best_a is not None:
        final_check = evaluate_sequence([float(x) for x in best_a])
        print(f"\nFinal verified score: {final_check:.6f} (local: {best_score:.6f})")
        if final_check < best_score:
            best_score = final_check
        return [float(x) for x in best_a]

    # Fallback: uniform sequence
    print("WARNING: Using fallback uniform sequence")
    return [1.0] * 256
