import numpy as np

def fast_convolve(a, b):
    """FFT-based full convolution for speed."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score_internal(a):
    """Compute score without calling evaluate_sequence (faster for optimization)."""
    n = len(a)
    c = fast_convolve(a, a)
    max_c = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a * sum_a)


def gradient_step(a, eta, n_iter_topk=1):
    """
    One gradient descent step on max_k c_k.
    Uses the top excess positions to compute a smoothed gradient.
    Returns (new_a, c, max_c, sum_a).
    """
    n = len(a)
    c = fast_convolve(a, a)
    sum_a = np.sum(a)
    max_c = np.max(c)
    target = (sum_a * sum_a) / (2 * n - 1)
    excess = np.maximum(0.0, c - target)

    # Gradient of c_k w.r.t. a_j = 2*a_{k-j} for valid k-j
    # Full gradient: grad_j = 2 * sum_k excess_k * a_{k-j} * 1[0<=k-j<=n-1]
    # = 2 * (excess_reversed * a)_{n-1+j}  [verified via FFT index manipulation]

    size = 1
    while size < 3 * n:
        size <<= 1

    ext = np.zeros(size)
    ext[:2 * n - 1] = excess
    a_rev = np.zeros(size)
    a_rev[:n] = a[::-1]

    result = np.fft.irfft(
        np.fft.rfft(ext, size) * np.fft.rfft(a_rev, size), size
    )
    grad = result[n - 1:2 * n - 1] * 2.0

    # Normalize gradient by its norm for stable steps
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a.copy(), c, max_c, sum_a

    s = np.sum(a_new)
    if s > 1e-15:
        a_new = a_new / s

    return a_new, c, max_c, sum_a

def make_exponential_init(n, alpha, side='left', rng=None):
    """Create exponential decay initialization with decay parameter alpha."""
    if side == 'left':
        # Heavy at start, decays toward end
        a = np.exp(-alpha * np.arange(n) / n)
    elif side == 'right':
        # Heavy at end, decays toward start
        a = np.exp(-alpha * (n - 1 - np.arange(n)) / n)
    elif side == 'both':
        # Heavy at both ends
        center = np.arange(n) - (n - 1) / 2.0
        a = np.exp(-alpha * np.abs(center) / ((n - 1) / 2.0))
    else:
        a = np.ones(n)

    a = np.maximum(0.0, a)
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a


def make_bimodal_init(n, separation, width, rng):
    """Two Gaussian bumps at specified positions."""
    x = np.arange(n)
    c1 = separation * (n - 1)
    c2 = (1.0 - separation) * (n - 1)
    w = width * n
    a = np.exp(-0.5 * ((x - c1) / w) ** 2) + np.exp(-0.5 * ((x - c2) / w) ** 2)
    a = np.maximum(0.0, a)
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a


def make_polynomial_init(n, power, side='left'):
    """Polynomial-shaped initialization."""
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    if side == 'left':
        a = (1.0 - x) ** power
    else:
        a = x ** power
    a = np.maximum(0.0, a)
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a
