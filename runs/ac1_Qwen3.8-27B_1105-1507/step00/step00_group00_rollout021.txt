def peak_reduction_step(f, rng):
    """Reduce the peak of conv(f,f) by adjusting individual elements."""
    n = len(f)
    c = np.convolve(f, f)
    peak_val = np.max(c)
    peak_positions = np.where(c >= 0.95 * peak_val)[0]
    
    # For each peak position, find the top contributors
    # c[k] = sum_i f[i]*f[k-i]
    # The top contributors to c[k] are pairs (i, k-i) with large f[i]*f[k-i]
    
    # Strategy: reduce the largest elements that appear in multiple peak positions
    # Count how many peak positions each index appears in
    contribution = np.zeros(n)
    for k in peak_positions:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        i_vals = np.arange(j_start, j_end + 1)
        k_minus_i = k - i_vals
        # Contribution of f[i] to c[k] is f[i] * f[k-i]
        # Add f[k-i] to contribution[i] (since reducing f[i] by delta reduces c[k] by delta * f[k-i])
        contribution[i_vals] += f[k_minus_i]
    
    # Reduce elements with high contribution
    top_k = max(1, n // 100)
    top_indices = np.argsort(contribution)[-top_k:]
    
    f_new = f.copy()
    total_reduction = 0
    for idx in top_indices:
        delta = 0.01 * f[idx]
        f_new[idx] -= delta
        total_reduction += delta
    
    # Distribute the reduction to low-contribution positions
    low_indices = np.argsort(contribution)[:top_k]
    f_new[low_indices] += total_reduction / len(low_indices)
    
    f_new = np.clip(f_new, 0, 1000)
    return f_new
