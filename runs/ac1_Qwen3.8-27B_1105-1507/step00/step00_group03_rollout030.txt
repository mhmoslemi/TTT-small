"""
Hybrid optimization: subgradient descent + LP direction + structured perturbations
to minimize 2n * max(conv(a,a)) / sum(a)^2 for non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _norm(a):
    """Clip, normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones_like(a) / len(a)
    return a / s

def _score(a):
    """Quick score: 2n * max(conv) / sum^2."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv_max = np.convolve(a, a).max()
    return float(2.0 * len(a) * conv_max / (s * s))

def _peak_subgrad(a):
    """Subgradient of peak convolution at a. Returns (grad, conv, peak_idx, peak_val)."""
    conv = np.convolve(a, a)
    pk = int(np.argmax(conv))
    n = len(a)
    idx = pk - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    grad = np.zeros(n)
    grad[valid] = a[idx[valid]]
    return grad, conv, pk, conv[pk]

def _lp_step(a, t=0.1, n_constr=200):
    """LP to find direction reducing peak. Returns direction or None."""
    n = len(a)
    conv = np.convolve(a, a)
    M = conv.max()
    pk = int(np.argmax(conv))
    
    rhs = (M - conv) / (2.0 * t)
    # Use most binding constraints
    idx = np.argsort(rhs)[:n_constr]
    
    # Objective: minimize (a*d)[pk]
    c = np.zeros(n)
    c_idx = pk - np.arange(n)
    c_valid = (c_idx >= 0) & (c_idx < n)
    c[c_valid] = a[c_idx[c_valid]]
    
    # Build sparse constraint matrix
    rows, cols, vals = [], [], []
    for ci, k in enumerate(idx):
        i_lo = max(0, k - n + 1)
        i_hi = min(n, k + 1)
        ii = np.arange(i_lo, i_hi)
        rows.extend((np.full(len(ii), ci)).tolist())
        cols.extend(ii.tolist())
        vals.extend(a[k - ii].tolist())
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(len(idx), n))
    
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bounds = [(-a[i] / t, 1.0) for i in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=rhs[idx], A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True})
    
    if res.success:
        return res.x, conv, pk
    return None, conv, pk

    1. Multiple initializations (parent, sinusoidal, random)
    2. Alternating subgradient steps and LP direction steps
    3. Adaptive step size and periodic restarts
    4. Return best sequence found
    """

    start = time.monotonic()
    deadline = start + max(0, float(budget_s) - 5.0)
    
    # Get parent

    except:
        parent = None
    if parent is None or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(2535)
    parent = np.clip(parent, 0, 1000)
    n = len(parent)
    
    # Try initializations
    candidates = []
    candidates.append(_norm(parent.copy()))
    
    # Sinusoidal variations
    t_arr = np.arange(n)
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            seq = 1.0 + amp * np.cos(2 * np.pi * freq * t_arr / n)
            candidates.append(_norm(seq))
    
    # Random

    # Find best init
    best = candidates[0]
    best_score = _score(best)
    for c in candidates[1:]:
        s = _score(c)
