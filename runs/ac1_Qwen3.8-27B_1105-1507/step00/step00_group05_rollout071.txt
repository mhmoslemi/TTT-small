"""
Gradient-based optimization of non-negative step function to minimize J = 2*n*max_b/sum_a^2.
Uses multiple initializations, momentum-based gradient descent on the simplex (sum=1, s>=0),
with top-K lag gradient approximation and adaptive step sizing.
"""

import numpy as np
import time


def fast_obj(s):
    """Compute J = 2*n*max(conv(s,s))/sum(s)^2."""
    n = len(s)
    b = np.convolve(s, s)
    max_b = b.max()
    sa = s.sum()
    if sa < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sa * sa)


def compute_grad_topk(s, K):
    """Compute approximate gradient of max(conv(s,s)) using top-K lags. Returns gradient and max_b."""
    n = len(s)
    b = np.convolve(s, s)
    max_b = b.max()
    if K >= len(b):
        top = np.arange(len(b))
    else:
        top = np.argpartition(b, -K)[-K:]
    grad = np.zeros(n)
    for k in top:
        if k < n:
            grad[:k + 1] += s[k::-1]
        else:
            grad[k - n + 1:] += s[n - 1:k - n - 1:-1]
    grad /= float(K)
    return grad, max_b


def project_simplex(s):
    """Project onto {x >= 0, sum(x) = 1}."""
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss < 1e-14:
        return np.ones_like(s) / len(s)
    return s / ss


def optimize_single(s_init, time_budget, K=10, seed=0):
    """Run gradient descent on the simplex to minimize 2n*max_b (equiv to J when sum=1)."""
    rng = np.random.default_rng(seed)
    n = len(s_init)
    s = project_simplex(s_init.astype(np.float64).copy())
    b0 = np.convolve(s, s)
    best_J = 2.0 * n * b0.max()
    best_s = s.copy()

    momentum = np.zeros(n)
    beta = 0.85
    alpha = 0.1
    start_t = time.monotonic()
    iters = 0
    stall = 0

    while time.monotonic() - start_t < time_budget:
        grad, max_b = compute_grad_topk(s, K)
        grad_c = grad - grad.mean()

        momentum = beta * momentum + (1.0 - beta) * grad_c

        mnorm = np.linalg.norm(momentum)
        if mnorm < 1e-14:
            step = 0.0
        else:
            step = min(alpha, 0.05 / (mnorm + 1e-10))

        s_new = s - step * momentum
        s_new = project_simplex(s_new)

        b_new = np.convolve(s_new, s_new)
        J = 2.0 * n * b_new.max()

        if J < best_J:
            best_J = J
            best_s = s_new.copy()
            s = s_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Anneal and perturb
                alpha *= 0.8
                if stall > 200:
                    s = best_s + 0.003 * rng.standard_normal(n)
                    s = project_simplex(s)
                    momentum = np.zeros(n)
                    stall = 100
                elif stall > 100:
                    s = best_s + 0.001 * rng.standard_normal(n)
                    s = project_simplex(s)
                    stall = 50

        iters += 1
        if iters % 3000 == 0:
            print(f"    it={iters} J={best_J:.6f} step={step:.6f} stall={stall}")

    return best_s, best_J
