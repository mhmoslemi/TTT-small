import numpy as np
import time
import cvxpy as cp

def solve_lp(a, M, n):
    """Solve the LP: maximize sum(b) s.t. conv(a,b) <= M, b >= 0"""
    b = cp.Variable(n, nonneg=True)
    
    # Build the constraint matrix (sparse Toeplitz)
    # A[k, j] = a[k-j] if 0 <= k-j < n else 0
    # We need A @ b <= M (scalar)
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)  # or SCS, CLARABEL
    
    if b.value is None:
        return a.copy()  # fallback
    return np.maximum(b.value, 0)

    rows = []
    cols = []
    vals = []

        for offset in range(n):
            k = j + offset
            rows.append(k)
            cols.append(j)
            vals.append(a[offset])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

Let me focus on making the subgradient method work well, since it's simpler and more reliable:

**Subgradient method with restarts:**
1. Normalize a so sum(a) = 1.
2. Compute c = conv(a,a), find S = argmax(c).
3. Compute subgradient: for each i, sg[i] = (2/|S|) * sum_{k in S} a[k-i] * 1[0 <= k-i < n].
4. Update: a <- max(0, a - lr * sg), then renormalize to sum = 1.
5. Adapt lr: use a diminishing schedule or restart if stuck.

The key challenge with subgradient is the learning rate. Too large: oscillate. Too small: slow convergence.

A common trick is to use **proximal gradient** (like FISTA) for faster convergence, but the "prox" step for the constraint sum(a)=1, a>=0 is just: clip to non-negative, then subtract the median (or divide by the mean) to get sum = 1.

Actually, the projection onto {a: sum(a) = 1, a >= 0} is:
1. Start with a.
2. Subtract a constant to make sum = 1: a' = a - (sum(a) - 1)/n.
3. Clip to non-negative: a'' = max(0, a').
4. This doesn't preserve the sum, so we need to iterate (like the Euclidean projection onto the simplex).

Tries multiple sequence lengths and random restarts within a 1500s budget.

import numpy as np
import time

def simplex_project(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u + (1 - cssv) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(rho[cond])
    theta = (cssv[rho-1] - 1) / rho
    return np.maximum(v - theta, 0)

    """
