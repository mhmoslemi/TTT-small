def build_convolution_matrix(f, n):
    """Build the sparse matrix A where (A @ b)[k] = (f * b)[k]."""
    # A has shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # Build using the structure: column j has f[0..n-1] at rows j..j+n-1
    rows = []
    cols = []
    data = []
    
    for j in range(n):
        # Column j: A[k, j] = f[k-j] for k = j, j+1, ..., j+n-1
        # i.e., f[0] at row j, f[1] at row j+1, ..., f[n-1] at row j+n-1

        c = np.full(n, j)

        cols.append(c)

Actually, you know what, let me just try a different approach entirely. Instead of the LP, let me use a **coordinate descent / projected gradient** method on the objective directly.

The objective is: F(a) = 2n * max_k (convolve(a,a)[k]) / (sum(a))^2

Since the objective is scale-invariant (scaling a by c doesn't change F), I can fix sum(a) = 1 and minimize 2n * max_k c[k].

Or equivalently, fix ||a||_1 = 1 and minimize max_k c[k].

The constraint a >= 0 makes this a constrained optimization.

Approach: **Multiplicative Weights / Mirror Descent**

Initialize a > 0 with sum(a) = 1.
At each step:
1. Compute c = convolve(a, a)
2. Find k* = argmax c[k]
3. Update: a[j] *= exp(-eta * dc[k*]/da[j])
4. Renormalize: a = a / sum(a)

This is a form of mirror descent on the simplex. The update reduces the maximum autocorrelation.

The gradient dc[k]/da[j] = a[k-j] + a[j-k] (with appropriate boundary handling).

This is O(n) per update step (for a single k), and we do it for each coordinate or in batches.

Actually, a simpler approach: **Subgradient method**

minimize max_k c[k] subject to a >= 0, sum(a) = 1

Subgradient with respect to a[j]:
If k* = argmax c[k], then the subgradient of max c[k] at a[j] is dc[k*]/da[j].

The constrained subgradient update (projected onto the simplex a >= 0, sum = 1) is:
a_new = proj_simplex(a - eta * g)

where g[j] = dc[k*]/da[j] and proj_simplex projects onto {x >= 0, sum(x) = 1}.

This is O(n) per step! Very fast.

But subgradient methods converge slowly. We can use a more sophisticated approach:
- Use the average of the top few k values instead of just the argmax
- Use a decreasing step size
- Use Nesterov acceleration

import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_A_sparse(f, n):
    """Build sparse matrix A of shape (2n-1, n) where A @ b = convolve(f, b)."""
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

        rows[idx:idx+n] = np.arange(j, j+n)
        cols[idx:idx+n] = j
        data[idx:idx+n] = f
        idx += n
    
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def lp_iteration(f, n, M):
    """One LP iteration: find b that maximizes sum(b) subject to convolution constraints."""
    A = build_A_sparse(f, n)
    c_vec = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def run_lp_optimization(initial, n, max_iters, time_limit):
    """Run LP-based optimization iterations."""
    f = np.array(initial[:n], dtype=np.float64)
    f = np.clip(f, 0, 1000)
    if np.sum(f) < 1e-10:
        f = np.ones(n)
    
    sum_f = np.sum(f)
    c = np.convolve(f, f)
    M = np.max(c)
    best_f = f.copy()
    best_score = float(evaluate_sequence(f.tolist()))
    
    start_time = time.monotonic()
    t = 0.3  # mixing parameter
    
    for iter in range(max_iters):
        if time.monotonic() - start_time > time_limit:
            break
        
        b = lp_iteration(f, n, M)
        if b is None:
            print(f"  LP failed at iteration {iter}")
            break
        
        # Ensure b is non-negative
        b = np.clip(b, 0, 1000)
        
        if np.sum(b) < 1e-10:
            print(f"  LP solution is zero at iteration {iter}")
            break
        
        # Rescale b to match sum of f
        b = b * (sum_f / np.sum(b))
        
        # Update
        f_new = (1 - t) * f + t * b
        f_new = np.clip(f_new, 0, 1000)
        
        # Evaluate
        score = float(evaluate_sequence(f_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best_f = f_new.copy()
            best_score = score

        f = f_new
        sum_f = np.sum(f)
        c = np.convolve(f, f)
        M = np.max(c)
        
        # Decrease t slowly
        t = max(0.05, 0.3 * (0.95 ** iter))
    
    return best_f, best_score
