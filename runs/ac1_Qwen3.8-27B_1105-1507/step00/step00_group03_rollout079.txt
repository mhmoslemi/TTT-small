import numpy as np

def build_lp_matrices(a, n):
    """Build the LP constraint matrices for SQA."""
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            length = j_hi - j_lo + 1
            rows.extend([k] * length)
            cols.extend(range(j_lo, j_hi + 1))
            # a[k-j] for j in [j_lo, j_hi]
            # When j = j_lo: k - j_lo
            # When j = j_hi: k - j_hi
            # So a[k-j_lo], a[k-j_lo-1], ..., a[k-j_hi]
            idx = k - np.arange(j_lo, j_hi + 1)
            vals.extend(a[idx].tolist())
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_sqa_step(a, n):
    """One SQA step: solve the LP and return the solution."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix, eye as sp_eye, hstack as sp_hstack
    
    A = build_lp_matrices(a, n)
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # A_ub = [A | -I_{2n-1}]
    I = sp_eye(2*n-1, format='csr')
    neg_I = -I
    A_ub = sp_hstack([A, neg_I], format='csr')
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n+1)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs')
    return result

def subgradient_step(a, T, eta, momentum, beta):
    """One subgradient step with momentum."""
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    w = np.exp((b - b_max) / T)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(2*n-1)
        w[np.argmax(b)] = 1.0
    else:
        w = w / w_sum
    grad = 2.0 * np.correlate(w, a, 'valid')
    momentum = beta * momentum + (1.0 - beta) * grad
    a_new = a - eta * momentum
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-15:
        a_new = np.ones(n) / n
    else:
        a_new = a_new / s
    return a_new, momentum, np.max(b)

    """Precompute row/col indices for the constraint matrix."""
    n_rows = 2*n - 1
    total = n*n

def precompute_lp_structure(n):
    """Precompute row/col indices for the SQA constraint matrix."""
    n_rows = 2 * n - 1
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    pos = 0
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        count = j_hi - j_lo + 1
        rows[pos:pos + count] = k
        cols[pos:pos + count] = np.arange(j_lo, j_hi + 1)
        pos += count
    return rows[:pos].copy(), cols[:pos].copy()


def build_lp_matrix(a, n, precomputed):
    """Build sparse constraint matrix A where A[k,j] = a[k-j] if valid."""
    rows, cols = precomputed
    vals = a[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_sqa_step(a, n, precomputed):
    """Solve one SQA LP. Returns (success, solution)."""
    A = build_lp_matrix(a, n, precomputed)
    n_rows = 2 * n - 1

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    I = sp_eye(n_rows, format='csr')
    A_ub = sp_hstack([A, -I], format='csr')
    b_ub = np.zeros(n_rows)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * (n + 1)

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    if not result.success:

    return True, result.x[:n].copy()

        w = np.zeros(2 * n - 1)
        w[int(np.argmax(b))] = 1.0
    else:
        w = w / w_sum

    a_new = a - eta * momentum

    else:
