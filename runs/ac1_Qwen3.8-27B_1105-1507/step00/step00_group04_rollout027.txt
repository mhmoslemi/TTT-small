```python
import numpy as np
import time

def convolve_full_fast(a):
    """Compute full convolution using FFT for large n."""
    n = len(a)
    if n < 128:
        return np.convolve(a, a)
    # Use FFT
    size = 2 * n - 1
    # Next power of 2
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    B = np.fft.irfft(A * A, fft_size)[:size]
    return B

def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from Duchi et al. (2008)
    if v.sum() <= 1 and v.min() >= 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def frank_wolfe_optimize(n, max_iters=5000, use_fft=True):
    """Optimize max(a*a) subject to a >= 0, sum(a) = 1 using Frank-Wolfe."""
    a = np.ones(n) / n
    best_score = np.inf
    best_a = a.copy()

        # Compute convolution
        if use_fft and n >= 128:
            b = convolve_full_fast(a)
        else:
            b = np.convolve(a, a)
        
        max_b = b.max()

            best_a = a.copy()

        active = np.where(b >= max_b - tol)[0]

        # Where A-i = {k-i : k in A}
        
        # More efficiently: for each k in A, for each i in [0, n-1], add 2*a_{k-i} to g_i (if valid)

        g = np.zeros(n)
        for k in active:

            i_start = max(0, k - n + 1)
            i_end = min(n-1, k)
            j_start = k - i_end  # = 0 when i_end = k
            j_end = k - i_start  # = n-1 when i_start = k-n+1

            g[i_start:i_end+1] += a[j_start:j_end+1][::-1]
```

```python
import numpy as np
import time
import math

def convolve_fft(a):
    """Full convolution using FFT."""
    n = len(a)
    if n < 64:
        return np.convolve(a, a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(fa * fa, fft_size)[:size]
    # Clean up tiny numerical errors
    b[np.abs(b) < 1e-15] = 0
    return b

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def score_from_conv(b, n):
    """Compute score from convolution and length (assuming sum(a)=1)."""
    return 2.0 * n * np.max(b)

def opt_subgrad(n, time_budget, init='uniform', seed=42):
    """Optimize using projected subgradient method."""
    rng = np.random.default_rng(seed)
    
    if init == 'uniform':
        a = np.ones(n) / n
    elif init == 'ramp':
        a = np.arange(1, n + 1, dtype=np.float64)
        a /= a.sum()
    elif init == 'ramp_rev':
        a = np.arange(n, 0, -1, dtype=np.float64)
        a /= a.sum()
    elif init == 'bimodal':
        a = np.zeros(n)
        a[n//4] = 0.5
        a[3*n//4] = 0.5
        a /= a.sum()
    elif init == 'perturbed':
        a = np.ones(n) / n
        a *= np.exp(rng.normal(0, 0.1, n))
        a = project_simplex(a)
    else:
        a = np.ones(n) / n
    
    use_fft = n >= 128
    deadline = time.monotonic() + time_budget
    
    best_a = a.copy()
    best_score = float(score_from_conv(convolve_fft(a) if use_fft else np.convolve(a, a), n))
    
    eta = 1.0 / (10 * np.sqrt(n))
    
    it = 0
    while time.monotonic() < deadline:
        b = convolve_fft(a) if use_fft else np.convolve(a, a)
        max_b = np.max(b)
        score = float(score_from_conv(b, n))
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Active set (with tolerance)
        tol = max_b * 1e-4 + 1e-15
        active = np.where(b >= max_b - tol)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            j_lo = k - i_hi
            j_hi = k - i_lo
            g[i_lo:i_hi + 1] += a[j_lo:j_hi + 1][::-1]
        g *= 2.0 / len(active)
        
        # Step
        a_new = a - eta * g
        a = project_simplex(a_new)
        
        it += 1
        # Adapt step size
        if it % 1000 == 0:
            eta *= 0.95  # Slowly decrease
    
    return best_a, best_score
```
