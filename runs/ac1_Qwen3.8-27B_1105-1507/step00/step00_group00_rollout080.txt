def compute_conv_max(p):
    """Compute max of aperiodic convolution using FFT."""
    n = len(p)
    # Pad to next power of 2 for FFT efficiency
    size = 1
    while size < 2*n - 1:
        size *= 2
    f = np.fft.rfft(p, size)
    conv = np.fft.irfft(f * f, size)[:2*n-1]
    return np.max(conv)

def score_from_conv(p):
    """Compute score from sequence."""
    n = len(p)
