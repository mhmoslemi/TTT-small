"""
Minimizes the normalized max auto-convolution of a non-negative sequence.
Uses FISTA (accelerated projected gradient) with FFT-based gradients,
multiple initializations, and a final local search phase.
Key insight: asymmetric sequences (e.g., ramps) break the F >= 2 barrier
that constrains symmetric sequences.
"""

import numpy as np
import time


def _fft_size(n):
    """Smallest power of 2 >= n."""
    return 1 << (n - 1).bit_length()


def _full_convolution(a):
    """Full linear convolution of a with itself via real FFT."""
    n = len(a)
    nfft = _fft_size(2 * n - 1)
    A = np.fft.rfft(a, nfft)
    return np.fft.irfft(A * A, nfft)[:2 * n - 1]


def _cross_corr(w, a):
    """(w star a)[j] = sum_l w[j+l] * a[l] for j=0..n-1, via FFT."""
    n = len(a)
    nfft = _fft_size(2 * n - 1)
    W = np.fft.rfft(w, nfft)
    A = np.fft.rfft(a, nfft)
    return np.fft.irfft(W * np.conj(A), nfft)[:n]


def _project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u > (cssv - S) / np.arange(1, n + 1)
    rho = int(cond.sum())
    if rho == 0:
        return np.full(n, S / n, dtype=np.float64)
    theta = (cssv[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)
