import numpy as np
import time

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Subtract target_sum/n from each, then find threshold
    # Standard water-filling algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)

def subgradient_step(a, lr, top_k=3):
    """One step of subgradient descent on the simplex."""
    n = len(a)
    conv = np.convolve(a, a)
    
    # Find top-k positions
    top_k_positions = np.argsort(conv)[::-1][:top_k]
    
    # Compute subgradient (average of top-k gradients)
    grad = np.zeros(n)
    for k in top_k_positions:
        j_min = max(0, k - (n-1))
        j_max = min(k, n-1)
        indices = k - np.arange(j_min, j_max + 1)
        grad[j_min:j_max+1] += 2.0 * a[indices]
    grad /= top_k
    
    # Update with projection
    a_new = project_simplex(a - lr * grad, 1.0)
    return a_new

def project_simplex(v, target_sum=1.0):
    """Project v onto the probability simplex (scaled by target_sum)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # The projection onto {x: x >= 0, sum(x) = target_sum}
    # is the same as target_sum * projection onto standard simplex
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - target_sum)
    rho = np.max(np.where(cond)[0]) if np.any(cond) else n - 1
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_refine(f, M, time_budget=10):
    """Solve LP to find best g with f*g <= M."""
    from scipy.optimize import linprog
    n = len(f)
    
    # Variables: b_0, ..., b_{n-1} (n variables)
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: (f*b)[k] <= M for k = 0, ..., 2n-2
    # (f*b)[k] = sum_{i} f[i] * b[k-i] for valid i
    # For each k, the constraint is: sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[k-j] * b[j] <= M
    
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A_rows = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        j_min = max(0, k - (n-1))
        j_max = min(k, n-1)
        for j in range(j_min, j_max + 1):
            row[j] = f[k-j]
        A_rows.append(row)
    A = np.array(A_rows)
    b = np.full(2*n - 1, M)
    
    # Bounds: b_j >= 0
    bounds = [(0, None)] * n
    
    # Solve
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        return g
    else:
        return None

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_score(a):
    """Compute 2n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = fftconvolve(a, a)
    return 2.0 * n * conv.max() / (s * s)

def build_constraint_matrix(f):
    """Build the (2n-1) x n matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

    """Solve: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)

    if result.success:
        return result.x
    return None

            f = f_new

"""
Minimizes 2n*max(convolve(a,a))/(sum(a))^2 using LP-based feasible direction
method combined with differential evolution and subgradient polishing.
The LP approach finds a complementary direction g that maximizes mass while
keeping the cross-convolution bounded, then mixes toward it.
"""
import numpy as np
import time
from scipy.signal import fftconvolve
from scipy.optimize import linprog


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: x >= 0, sum(x) = target_sum} using water-filling."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    """Compute 2n*max(conv)/(sum)^2 using FFT."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = fftconvolve(a, a)
    return 2.0 * n * conv.max() / (s * s)


def build_constraint_matrix(f):
    """Build (2n-1)xn matrix where A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_complementary(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_constraint_matrix(f)
    c = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_optimize(f0, time_budget, seed=42, verbose=True):
    """LP-based feasible direction optimization."""
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = project_simplex(f0.astype(np.float64), 1.0)

    best_f = f.copy()
    best_score = compute_score(f)
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")

    start = time.monotonic()
    t = 0.15
    no_improve = 0
    max_no_improve = 30

    while time.monotonic() - start < time_budget:
        conv_f = fftconvolve(f, f)
        M = conv_f.max()

        g = solve_lp_complementary(f, M)
        if g is None or g.sum() < 1e-10:
            g = solve_lp_complementary(f, M * 1.02)
            if g is None or g.sum() < 1e-10:
                break

        sum_g = g.sum()
        g = g / sum_g

        f_new = (1.0 - t) * f + t * g
        f_new = project_simplex(f_new, 1.0)
        score_new = compute_score(f_new)

        if score_new < best_score - 1e-10:
            best_score = score_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            t = max(t * 0.97, 0.02)
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                # Perturb and restart
                f = project_simplex(f + rng.normal(0, 0.03, n), 1.0)
                if compute_score(f) > best_score * 1.02:
                    f = best_f.copy()
                t = 0.15
                no_improve = 0

        if verbose and (no_improve == 0 and int(time.monotonic() - start) % 30 < 1):
            print(f"    LP t={time.monotonic()-start:.0f}s: score={best_score:.6f}, mix_t={t:.3f}")

    return best_f, best_score


def subgradient_polish(a, time_budget=50, verbose=False):
    """Subgradient descent on the simplex for local improvement."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64), 1.0)
    best_a = a.copy()
    best_score = compute_score(a)
    lr = 1.0 / (4.0 * n)

    start = time.monotonic()
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = fftconvolve(a, a)
        current_max = conv.max()

        # Use top-5 active constraints for a smoother subgradient
        top_k = min(5, len(conv))
        active = np.argsort(conv)[::-1][:top_k]

        grad = np.zeros(n)
        for k in active:
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            if j_max >= j_min:
                indices = k - np.arange(j_min, j_max + 1)
                grad[j_min:j_max + 1] += 2.0 * a[indices]
        grad /= top_k

        # Backtracking line search
        improved = False
        for _ in range(10):
            a_new = project_simplex(a - lr * grad, 1.0)
            s = compute_score(a_new)
            if s < best_score:
                a = a_new
                best_a = a.copy()
                best_score = s
                no_improve = 0
                lr = min(lr * 1.02, 0.005)
                improved = True
                break
            lr *= 0.5

        if not improved:
            a = project_simplex(a - lr * grad, 1.0)
            no_improve += 1
            if no_improve > 500:
                break

    return best_a, best_score


def differential_evolution(pop, time_budget, seed=42, F=0.8, CR=0.9, verbose=False):
    """Differential evolution on the simplex."""
    rng = np.random.default_rng(seed)
    pop_size = len(pop)
    n = len(pop[0])

    scores = [compute_score(p) for p in pop]
    best_idx = np.argmin(scores)
    best_pop = pop[best_idx].copy()
    best_score = scores[best_idx]

    if verbose:
        print(f"    DE start: best_score={best_score:.6f}, pop_size={pop_size}")

    start = time.monotonic()
    gen = 0

    while time.monotonic() - start < time_budget:
        gen += 1
        for i in range(pop_size):
            if time.monotonic() - start > time_budget:
                break

            # Select three distinct parents
            r1, r2 = rng.integers(0, pop_size, 2)
            while r1 == r2:
                r2 = rng.integers(0, pop_size)
            while i == r1 or i == r2:
                r1, r2 = rng.integers(0, pop_size, 2)
                while r1 == r2:
                    r2 = rng.integers(0, pop_size)

            # Mutation: v = pop[i] + F*(pop[r1] - pop[r2])
            v = pop[i] + F * (pop[r1] - pop[r2])

            # Crossover
            trial = np.zeros(n)
            for j in range(n):
                if rng.random() < CR or j == rng.integers(0, n):
                    trial[j] = v[j]
                else:
                    trial[j] = pop[i][j]

            # Project to simplex
            trial = project_simplex(trial, 1.0)

            # Selection
            s = compute_score(trial)
            if s < scores[i]:
                pop[i] = trial
                scores[i] = s
                if s < best_score:
                    best_score = s
                    best_pop = trial.copy()
                    if verbose and gen % 500 == 0:
                        print(f"    DE gen={gen}: score={best_score:.6f}")

        # Occasionally add a random member
        if gen % 100 == 0 and pop_size > 5:
            worst_idx = np.argmax(scores)
            pop[worst_idx] = project_simplex(rng.uniform(0.1, 2.0, n), 1.0)
            scores[worst_idx] = compute_score(pop[worst_idx])

    return best_pop, best_score
