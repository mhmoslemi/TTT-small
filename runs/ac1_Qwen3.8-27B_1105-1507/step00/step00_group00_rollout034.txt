"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
- Direct SLSQP for small n (<=8)
- LP-based iterative refinement for larger n
- Multiple n values and random restarts
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def get_score(b):
    """Score for normalized b (sum=1): 2*n*max(conv(b,b))."""
    n = len(b)
    return 2.0 * n * np.max(np.convolve(b, b))

def solve_slsqp(n, n_trials=30, seed=42):
    """Solve small-n problem with SLSQP."""
    rng = np.random.default_rng(seed)
    best_b = np.ones(n) / n
    best_val = np.max(np.convolve(best_b, best_b))
    
    def obj(x):
        return np.max(np.convolve(x, x))
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, None)] * n
    
    for t in range(n_trials):
        if t == 0:
            x0 = np.ones(n) / n
        else:
            x0 = rng.dirichlet(np.ones(n))
        
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 500, 'ftol': 1e-12})
        if res.fun < best_val:
            best_val = res.fun
            best_b = res.x.copy()
    
    best_b = np.clip(best_b, 0, None)
    best_b /= best_b.sum()
    return best_b, 2.0 * n * best_val

def solve_lp_iter(b, n_iters=500, step=0.3, seed=42):
    """LP-based iterative method."""
    rng = np.random.default_rng(seed)
    n = len(b)
    current = b.copy()
    current_score = get_score(current)
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        for i in range(n):
            A[i+j, j] += current[i]
    
    for it in range(n_iters):
        conv = np.convolve(current, current)
        M = np.max(conv)
        
        # Rebuild A (since current changed)
        A[:] = 0
        for j in range(n):
            for i in range(n):
                A[i+j, j] += current[i]
        
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                     bounds=[(0.0, None)]*n, method='highs')
        if not res.success:
            break
        
        g = res.x
        sum_g = g.sum()
        if sum_g < 1e-15:
            break
        
        g_norm = g / sum_g
        
        # Try different step sizes
        best_new = current
        best_new_score = current_score
        for s in [step, step*0.5, step*0.1]:
            candidate = (1-s)*current + s*g_norm
            candidate = np.clip(candidate, 0, None)
            csum = candidate.sum()
            if csum < 1e-15:
                continue
            candidate /= csum
            sc = get_score(candidate)
            if sc < best_new_score:
                best_new_score = sc
                best_new = candidate
        
        if best_new_score >= current_score:
            # No improvement, try a random perturbation
            if rng.random() < 0.3:
                perturbation = rng.normal(0, 0.05, n)
                candidate = current + perturbation
                candidate = np.clip(candidate, 0, None)
                candidate /= candidate.sum()
                sc = get_score(candidate)
                if sc < current_score:
                    current = candidate
                    current_score = sc
            else:
                break
        else:
            current = best_new
            current_score = best_new_score
    
    return current, current_score

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 5.0
    
    best_sequence = None
    best_score = float('inf')
    
    def update_best(b, score):
        nonlocal best_sequence, best_score
        if score < best_score:
            best_score = score
            best_sequence = b.copy()
            print(f"  New best: score={score:.6f}, n={len(b)}, t={time.monotonic()-start_time:.1f}s")
    
    # Try small n with SLSQP
    for n in [3, 4, 5, 6, 7, 8]:
        if time.monotonic() > deadline - 60:
            break
        print(f"SLSQP with n={n}")
        b, score = solve_slsqp(n, n_trials=30, seed=seed)
        update_best(b, score)
    
    # Try LP-based method for various n
    n_values = [10, 15, 20, 30, 50, 80, 120, 200, 350, 500]
    for n in n_values:
        if time.monotonic() > deadline - 30:
            break
        print(f"LP-iter with n={n}")
        
        # Multiple starting points
        for init in range(3):
            if time.monotonic() > deadline - 10:
                break
            if init == 0:
                b0 = np.ones(n) / n
            elif init == 1:
                b0 = np.exp(-np.linspace(0, 2, n))
                b0 /= b0.sum()
            else:
                b0 = rng.dirichlet(np.ones(n))
            
            b, score = solve_lp_iter(b0, n_iters=200, step=0.3, seed=seed+init)
            update_best(b, score)
    
    # Use remaining time for additional LP iterations on the best
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("Final polish on best sequence")
        b, score = solve_lp_iter(best_sequence, n_iters=1000, step=0.1, seed=seed+100)
        update_best(b, score)
    
    if best_sequence is None:
        best_sequence = np.ones(10) / 10
    
    return [float(x) for x in best_sequence]
