def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    N = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    N_fft = 1
    while N_fft < 2 * N:
        N_fft *= 2
    A = rfft(a, N_fft)
    autocorr = irfft(A * A.conj(), N_fft)[:N]
    return autocorr


def soft_max_val(autocorr, beta):
    """Compute soft-max of autocorr."""
    max_val = np.max(autocorr)
    return max_val + np.log(np.sum(np.exp(beta * (autocorr - max_val)))) / beta

    """Compute gradient of soft_max(autocorr) w.r.t. a."""

Minimizes 2n*max(autocorr)/(sum)^2 for non-negative sequences using
projected gradient descent with adaptive soft-max, multi-length sweep,
and local search polishing. Exploits FFT-based autocorrelation and
cross-correlation for efficiency on longer sequences.

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_autocorr(a):
    """Compute full linear autocorrelation (same as np.convolve(a,a)) using FFT."""
    n = len(a)
    N = 2 * n - 1
    N_fft = 1 << (N - 1).bit_length()
    A = rfft(a, N_fft)
    autocorr = irfft(A * A.conj(), N_fft)[:N]
    return autocorr


def cross_corr_valid(w, a):
    """Compute sum_j w[i+j]*a[j] for i=0..len(a)-1 using FFT (fast for large n)."""
    n = len(a)
    m = len(w)
    N = m + n - 1
    N_fft = 1 << (N - 1).bit_length()
    W = rfft(w, N_fft)
    A = rfft(a, N_fft)
    # np.correlate(w, a, 'valid')[i] = sum_j w[j+i] * a[j]
    # = ifft(conj(FFT(w)) * FFT(a)) truncated
    corr = irfft(np.conj(W) * A, N_fft)[:n]
    return corr


def cross_corr_valid_direct(w, a):
    """Direct computation for small arrays (avoids FFT overhead)."""
    return np.correlate(w, a, mode='valid')


def compute_gradient(a, beta):
    """Compute gradient of soft_max(autocorr(a)) w.r.t. a (unconstrained)."""
    n = len(a)
    autocorr = fast_autocorr(a)
    max_val = np.max(autocorr)

def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(n, 0, -1) > (cssv - 1)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    autocorr = fast_autocorr(a)

    for n in [24, 28, 32, 40, 48, 56, 64, 80, 96, 112, 128, 160, 192, 224, 256,
              320, 384, 448, 512, 640, 768, 896, 1024, 1280, 1536, 1792, 2048,
              2560, 3072, 3584, 4096, 5120]:
        if n not in lengths:
            lengths.append(n)
    # Also add provided length
    try:
        if best_n > 0 and best_n not in lengths:
            lengths.append(best_n)
    except:
        pass
    lengths.sort()
    
    # Allocate time: more time for medium lengths (likely sweet spot)
    for n in lengths:
        if time.monotonic() > deadline:
            break
        
        remaining = deadline - time.monotonic()
        if remaining < 8:
            break
        
        # Time allocation based on length
        if n <= 50:
            time_alloc = 5.0
            max_iter = 500
        elif n <= 200:
            time_alloc = 8.0
            max_iter = 400
        elif n <= 1000:
            time_alloc = 12.0
            max_iter = 200
        else:
            time_alloc = 20.0
            max_iter = 100
        
        remaining = deadline - time.monotonic()
        if remaining < time_alloc + 3:
            break
        
        # Number of initializations
        n_inits = min(4, max(1, int(remaining / (time_alloc * 1.5))))
        
        for init_idx in range(n_inits):
            if time.monotonic() > deadline - 5:
                break
            
            # Different initializations
            if init_idx == 0:
                a0 = np.ones(n) / n
            elif init_idx == 1:
                # Asymmetric: more mass on one side
                a0 = np.linspace(0.5, 1.5, n)
                a0 = a0 / np.sum(a0)
            elif init_idx == 2:
                # Random
                a0 = rng.random(n) + 0.1
                a0 = a0 / np.sum(a0)
            elif init_idx == 3:
                # Triangular (peak in middle)
                mid = n // 2
                a0 = 1.0 - np.abs(np.arange(n) - mid) / (mid + 1)
                a0 = np.maximum(a0, 0.01)
                a0 = a0 / np.sum(a0)
            else:
                break
            
            t_start = time.monotonic()
            try:
                # Projected gradient optimization
                a_opt = optimize_projected_gradient(
                    a0,
                    beta_schedule=(20.0, min(200.0, 5.0 * n)),
                    step0=0.1,
                    max_iter=max_iter,
                    time_limit=time_alloc * 0.7
                )
                
                score = eval_score(a_opt)
                
                # Local search polish if time remains
                time_left = deadline - time.monotonic()
                if time_left > 10:
                    a_polished, score_polished = local_search_polish(
                        a_opt, max_iter=min(2000, int(time_left * 100)), rng=rng
                    )
                    if score_polished < score:
                        a_opt = a_polished
                        score = score_polished
                
                if score < best_score:
                    best_score = score
                    best_seq = a_opt.copy()
                    best_n = n
                    elapsed = time.monotonic() - t_start
                    print(f"n={n}, init={init_idx}: score={score:.6f} ({elapsed:.1f}s)")
                    
            except Exception as e:
                print(f"n={n}, init={init_idx}: error: {e}")
    
    # Final polishing on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\nPolishing best sequence (n={best_n}, score={best_score:.6f})...")
        a_polished, score_polished = local_search_polish(
            best_seq, max_iter=min(5000, int((deadline - time.monotonic()) * 100)), rng=rng
        )
        if score_polished < best_score:
            best_score = score_polished
            best_seq = a_polished
            print(f"After polish: score={best_score:.6f}")
        
        # Coordinate refinement
        if time.monotonic() < deadline - 10:
            a_coord, score_coord = coordinate_refine(
                best_seq, n_passes=10, rng=rng
            )
            if score_coord < best_score:
                best_score = score_coord
                best_seq = a_coord
                print(f"After coordinate refine: score={best_score:.6f}")
    
    # Final verification
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFinal score via evaluate_sequence: {final_score:.6f} (n={len(best_seq)})")
    
    return [float(x) for x in best_seq]
