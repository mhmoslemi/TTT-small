def solve_lp_for_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Constraint: for each k, sum_j f[k-j]*g[j] <= M
    # conv(f,g)[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[k-j]*g[j]

    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0

    constraints = [cp.sum(g) >= 0]  # just nonnegativity (already encoded)
    
    # For each lag k
    for k in range(2*n - 1):
        # g[j] appears with coefficient f[k-j] for j in [max(0,k-n+1), min(k,n-1)]
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        # Coefficients: f[k-j] for j in [j_start, j_end]
        coeffs = f[k-j_end:k-j_start+1][::-1]  # f[k-j_end], f[k-j_end+1], ..., f[k-j_start]

    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)  # or GLOP, CLARABEL

        return None

import cvxpy as cp

def eval_fast(a):
    """Fast evaluation without the security checks (we know our input is valid)."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    b = np.convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a ** 2))

def lp_direction(f, M, max_iters_lp=1):
    """Solve LP to find improvement direction."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    constraints = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        # conv(f,g)[k] = sum_{j=j_start}^{j_end} f[k-j] * g[j]
        coeffs = np.array([f[k - j] for j in range(j_start, j_end + 1)])
        constraints.append(coeffs @ g[j_start:j_end+1] <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    if g.value is None:
        return None
    g_val = np.clip(np.asarray(g.value).flatten(), 0, 1000)
    if not np.all(np.isfinite(g_val)) or np.sum(g_val) < 1e-10:
        return None
    return g_val

def build_conv_matrix(f):
