"""
Subgradient descent with momentum and adaptive stepping to minimize
max autocorrelation of a non-negative step function. Key insight:
symmetric sequences are bounded below by score 2, so the optimizer
naturally finds asymmetric solutions by attacking peaks with
shifted-reversed gradient directions.
"""

import numpy as np
import time


def fast_convolve(f):
    """Compute full linear convolution via FFT. Returns array of length 2n-1."""
    n = len(f)
    target = 2 * n - 1
    size = 1
    while size < target:
        size <<= 1
    F = np.fft.rfft(f, size)
    c = np.fft.irfft(F * F, size)
    return c[:target]


def compute_score(f):
    """Compute objective 2n*max(f*f)/sum(f)^2 assuming f is already non-negative."""
    n = len(f)
    S = f.sum()
    if S < 0.01:
        return float('inf')
    c = fast_convolve(f)
    M = c.max()
    return float(2.0 * n * M / (S * S))


def make_cosine_init(n, freq, phase, rng):
    i = np.arange(n, dtype=np.float64)
    f = 1.0 + rng.uniform(0.3, 1.0) * np.cos(2.0 * np.pi * freq * i / n + phase)
    return np.maximum(f, 0.01)


def make_random_init(n, rng, sharpness=1.0):
    f = rng.random(n) ** sharpness
    return np.maximum(f, 0.001)


def make_linear_init(n, rng, direction):
    i = np.arange(n, dtype=np.float64)
    if direction < 0:
        i = n - 1 - i
    slope = rng.uniform(0.5, 3.0)
    f = 1.0 + slope * i / n
    return f


def make_bimodal_init(n, rng):

    f = np.exp(-0.5 * ((i - c1) / s1) ** 2) + np.exp(-0.5 * ((i - c2) / s2) ** 2)
    return f


def subgradient_optimize(f_init, time_limit, seed=0):

    Subgradient descent with momentum to minimize max(f*f)

    """

    f = f / S
    n = len(f)

    best = f.copy()
    best_score = compute_score(best)

    momentum = np.zeros(n)
    beta = 0.7
    # Adaptive step

    min_eta = 0.001
    patience = 0
    max_patience = 300

    iter_count = 0

    while time.monotonic() < deadline:
        c = fast_convolve(f)
        k_max = int(c.argmax())
        peak_val = c[k_max]
        avg_val = 1.0 / (2 * n - 1)  # since sum(f)=1, sum(c)=1

def coordinate_polish(f_init, time_limit, seed=0):
    """
    Coordinate-wise line search to polish the solution.
    For each coordinate j, the objective max_k (f*f)_k is a convex
    piecewise-linear function of f_j (when others are fixed).
    We use ternary search on this 1D function.
    """
    rng = np.random.default_rng(seed)
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = np.maximum(f, 0.0)
    S = f.sum()
    if S < 1e-10:
        return f_init
    f = f / S
    n = len(f)

    best = f.copy()
    best_score = compute_score(best)
    deadline = time.monotonic() + time_limit

    # Cache convolution for speed
    def eval_1d(f_copy, j, val):
        f_copy[j] = val
        S_c = f_copy.sum()
        if S_c < 1e-10:
            return float('inf')
        f_normalized = f_copy / S_c
        c = fast_convolve(f_normalized)
        return 2.0 * n * c.max() / (n * n)  # simplified since sum=1

    # Random coordinate order, cycling
    order = rng.permutation(n)
    idx = 0

    while time.monotonic() < deadline:
        j = int(order[idx % n])
        idx += 1

        old_val = f[j]
        # Search range: [0, 2 * old_val + 0.1/n] to allow growth
        lo_bound = 0.0
        hi_bound = 2.0 * old_val + 0.05 / n

        # Ternary search for the minimum
        a = lo_bound
        b = hi_bound
        for _ in range(20):
            m1 = a + (b - a) / 3.0
            m2 = b - (b - a) / 3.0
            f_save = f.copy()
            s1 = eval_1d(f_save, j, m1)
            s2 = eval_1d(f, j, m2)
            f[:] = f_save
            if s1 < s2:
                b = m2
            else:
                a = m1
        optimal = (a + b) / 2.0
        f[j] = optimal

        # Renormalize
        S_now = f.sum()
        if S_now < 1e-10:
            f = best.copy()
            continue
        f = f / S_now

        cur_score = compute_score(f)
        if cur_score < best_score:
            best = f.copy()
            best_score = cur_score

    return best
