def simplex_project(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(f):
    """Build matrix A such that A @ g = convolve(f, g, mode='full')."""
    n = len(f)

def simplex_project(v):
    """Project v onto probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if v.sum() <= 0:

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]

                f_lp2 = lp_refine(f_for_lp2, max_iter=300, t_init=0.4, time_limit=t_lp2)
                score = evaluate_sequence(f_lp2.tolist())
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = f_lp2.copy()
                    print(f"  [LP2-{init_name}] n={n} score={score:.6f}")

    # Final local search polish on best sequence
    if best_seq is not None and time.monotonic() < deadline:
        t_rem = deadline - time.monotonic()
        best_seq, best_score = local_search(best_seq, best_score,
                                             n_steps=10000,
                                             time_limit=max(2.0, t_rem * 0.6),
                                             rng=rng)
        print(f"After local search: score={best_score:.6f}")

    # One more LP pass on the final best if time allows
    if best_seq is not None and time.monotonic() < deadline and len(best_seq) <= 2000:
        t_rem = deadline - time.monotonic()
        if t_rem > 10:
            f_final = lp_refine(best_seq, max_iter=200, t_init=0.3,
                                 time_limit=max(5.0, t_rem * 0.5))
            score = evaluate_sequence(f_final.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = f_final
                print(f"Final LP polish: score={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(20) / 20
        best_score = evaluate_sequence([float(x) for x in best_seq])

    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
