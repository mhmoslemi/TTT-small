def fft_convolve(a, b):
    """Compute full convolution using FFT for speed."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()  # next power of 2
    fa = np.fft.rfft(a, N)

```python
def project_simplex(v):
    """Project v onto the probability simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 1 and (v >= 0).all():
        # Already in simplex (up to scaling)
        return v / v.sum() if v.sum() > 0 else np.ones(n) / n
    
    # Find tau such that sum max(v_i - tau, 0) = 1
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    tau = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - tau, 0)
```
