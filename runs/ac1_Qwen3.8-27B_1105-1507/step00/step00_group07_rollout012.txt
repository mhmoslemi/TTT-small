def compute_score_fast(a):
    """Compute the score without calling evaluate_sequence (for speed)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def gradient_step(a, eta, top_k=1):
    """Perform one gradient step to reduce the score."""
    n = len(a)
    conv = np.convolve(a, a)
    sum_a = np.sum(a)
    
    # Find top-k positions
    idx = np.argsort(conv)[::-1][:top_k]
    
    # Compute gradient (sum over top-k positions)
    grad = np.zeros(n)
    for p in idx:
        # grad[j] += a[p-j] if 0 <= p-j <= n-1
        j_start = max(0, p - n + 1)
        j_end = min(n - 1, p)
        for j in range(j_start, j_end + 1):
            grad[j] += a[p - j]
    
    # Center the gradient to preserve the sum
    grad -= np.mean(grad)
    
    # Update
    a_new = a - eta * grad / sum_a
    a_new = np.clip(a_new, 0, 1000.0)
    
    # Renormalize
    s = np.sum(a_new)
    if s > 0:
        a_new = a_new * sum_a / s
    
    return a_new, conv
