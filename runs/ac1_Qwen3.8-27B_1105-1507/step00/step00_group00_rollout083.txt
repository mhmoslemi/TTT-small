"""
Soft-gradient descent with FFT-based convolution to minimize
2n*max(conv(a,a))/(sum(a))^2. Uses temperature annealing over the
convolution peak, multiple diverse starting points, and simulated
annealing perturbations to escape local minima.
"""

import numpy as np
import time


def _next_pow2(n):
    return 1 << (max(1, n) - 1).bit_length()


def _fft_full_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a) + len(b) - 1
    L = _next_pow2(n)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:n]


def _compute_score(a):
    """Compute objective: 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    b = np.fft.irfft(A * A, L)[:2 * n - 1]
    max_b = float(np.max(b))
    return 2.0 * n * max_b / (s * s)


def _soft_grad_step(a, T, lr, rng, do_anneal=True):
    """One step of soft-gradient descent. Returns (new_a, b)."""
    n = len(a)
    L = _next_pow2(max(2 * n - 1, 3 * n - 2))

    A = np.fft.rfft(a, L)

    # Full convolution b = a * a
    b = np.fft.irfft(A * A, L)[:2 * n - 1]

    # Soft weights from b
    b_pos = np.maximum(b, 1e-15)
    log_b = np.log(b_pos)
    max_log_b = np.max(log_b)
    exponents = T * (log_b - max_log_b)
    w = np.exp(exponents)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= w_sum

    W = np.fft.rfft(w, L)

    # p1[j] = sum_k w[k] * a[k-j]  (cross-correlation)
    p1 = np.fft.irfft(W * np.conj(A), L)[:n]

    # p2[j] = sum_k w[k] * a[j-k]  (convolution a*w)
    p2 = np.fft.irfft(A * W, L)[:n]

    grad = p1 + p2

    # Update
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = float(np.sum(a_new))
    target_s = float(np.sum(a))
    if s < 1e-10:
        a_new = a.copy()
    else:
        a_new *= target_s / s

    # Annealing perturbation
    if do_anneal and rng.random() < 0.12:
        noise = rng.normal(0.0, lr * 0.4, size=n)
        a_new = np.maximum(a_new + noise, 0.0)
        s = float(np.sum(a_new))
        if s > 1e-10:
            a_new *= target_s / s

    return a_new, b


def _optimize(a_init, n_iters, T_init, T_end, lr_init, lr_end, rng, label=""):
    """Run soft-gradient optimization loop."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    score = _compute_score(a)
    best_a = a.copy()
    best_score = score
    no_improve = 0
    stagnation_limit = 3000

    for it in range(n_iters):
        if time.monotonic() > _DEADLINE:
            break

        frac = it / max(1, n_iters - 1)
        T = T_init * (T_end / T_init) ** frac
        lr = lr_init * (lr_end / lr_init) ** frac

        a, b = _soft_grad_step(a, T, lr, rng)
        score = _compute_score(a)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Reseed if stuck too long
        if no_improve > stagnation_limit:
            n2 = len(a)
            a = np.maximum(a + rng.normal(0, 0.1, size=n2), 0.0)
            s = np.sum(a)
            if s > 1e-10:
                a *= float(np.sum(a_init)) / s
            no_improve = 0

        if it % 2000 == 0 and it > 0:
            print(f"    {label}: it={it}, T={T:.4f}, score={score:.6f}, best={best_score:.6f}", flush=True)

    return best_a, best_score


# Global deadline set by propose_candidate
_DEADLINE = 0.0


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses soft-gradient descent with FFT, temperature annealing, diverse
    initializations, and periodic re-seeding to escape local minima.
    """
    global _DEADLINE
    rng = np.random.default_rng(seed)
    _DEADLINE = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_score = np.inf
    best_seq = None

    print(f"Budget: {budget_s}s, deadline in {budget_s - 5:.0f}s", flush=True)

    # Build diverse starting points
    starts = []

    # 1. Try height_sequence_1
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, 1000)
        if len(hs) > 0 and np.sum(hs) > 0.01:
            starts.append((hs.copy(), "height_seq"))
    except NameError:
        pass

    # 2. Uniform sequences of various lengths
    for n in [1024, 2048, 4096, 8192]:
        starts.append((np.ones(n), f"u{n}"))

    # 3. Sine-modulated
    for n in [2048, 4096]:
        i = np.arange(n, dtype=np.float64)
        for f in [1, 2, 3, 5]:
            a = 1.0 + 0.4 * np.sin(2.0 * np.pi * f * i / n)
            starts.append((np.maximum(a, 0.05), f"sin{n}f{f}"))

    # 4. V-shape (anti-symmetric around center)
    for n in [2048, 4096]:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 + 1.0 * np.abs(2.0 * i / n - 1.0)
        starts.append((a, f"v{n}"))

    # 5. Bell-shape (peaked at center)
    for n in [2048, 4096]:
        i = np.arange(n, dtype=np.float64)
        a = 0.3 + 1.0 * np.cos(np.pi * i / n) ** 2
        starts.append((np.maximum(a, 0.05), f"bell{n}"))

    # 6. Random
    for n in [2048, 4096]:
        a = np.abs(rng.normal(0, 1, n)) + 0.1
        starts.append((a, f"rnd{n}"))

    # 7. Two-peak
    for n in [2048, 4096]:
        i = np.arange(n, dtype=np.float64)
        a = 0.3 + np.exp(-((i - 0.25 * n) / (0.1 * n)) ** 2) + np.exp(-((i - 0.75 * n) / (0.1 * n)) ** 2)
        starts.append((a, f"2pk{n}"))

    # Evaluate starting points
    scored = []
    for a, label in starts:
        sc = _compute_score(a)
        scored.append((sc, a, label))
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    scored.sort(key=lambda x: x[0])

    print(f"Best initial: {best_score:.6f}", flush=True)
    for sc, a, lb in scored[:5]:
        print(f"  {lb}: score={sc:.6f}, n={len(a)}", flush=True)

    # Run optimization from top starting points
    time_elapsed = time.monotonic() - (_DEADLINE - budget_s + 5.0)
    remaining = _DEADLINE - time.monotonic()
    n_to_try = min(6, len(scored))
    time_each = remaining / n_to_try

    for idx in range(n_to_try):
        if time.monotonic() > _DEADLINE:
            break
        init_sc, a_init, label = scored[idx]
        n = len(a_init)
        est_per_iter = n * np.log2(max(n, 2)) * 1.2e-8
        max_iters = int(min(20000, max(200, time_each / max(est_per_iter, 5e-7))))

        print(f"\nOptimizing {label} (n={n}, init={init_sc:.6f}, iters={max_iters}, ~{time_each:.0f}s)", flush=True)
        t0 = time.monotonic()

        local_best, local_sc = _optimize(
            a_init, max_iters,
            T_init=1.0, T_end=0.005,
            lr_init=0.02, lr_end=0.002,
            rng=rng, label=label
        )

        print(f"  Done in {time.monotonic()-t0:.1f}s: score={local_sc:.6f}", flush=True)

        if local_sc < best_score:
            best_score = local_sc
            best_seq = local_best.copy()
            print(f"  NEW BEST: {best_score:.6f}", flush=True)

    # Final polish from best found
    if time.monotonic() < _DEADLINE and best_seq is not None:
        rem = _DEADLINE - time.monotonic()
        n = len(best_seq)
        est_per_iter = n * np.log2(max(n, 2)) * 1.2e-8
        max_iters = int(min(30000, max(500, rem / max(est_per_iter, 5e-7))))
        print(f"\nPolish: n={n}, iters={max_iters}, time={rem:.0f}s", flush=True)

        lb, lsc = _optimize(
            best_seq, max_iters,
            T_init=0.1, T_end=0.001,
            lr_init=0.008, lr_end=0.001,
            rng=rng, label="polish"
        )
        if lsc < best_score:
            best_score = lsc
            best_seq = lb.copy()
            print(f"  Polish improved: {best_score:.6f}", flush=True)

        # Try mirror flip
        if time.monotonic() < _DEADLINE:
            flipped = best_seq[::-1].copy()
            fl_sc = _compute_score(flipped)
            if fl_sc < best_score:
                best_score = fl_sc
                best_seq = flipped
                print(f"  Flip improved: {best_score:.6f}", flush=True)

    # Final verification
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_sc = evaluate_sequence(result)
    print(f"\nFinal verified score: {final_sc:.6f}", flush=True)

    if not np.isfinite(final_sc):
        # Fallback: uniform sequence
        result = [1.0] * 1024
        final_sc = evaluate_sequence(result)
        print(f"Fallback score: {final_sc:.6f}", flush=True)

    return result
