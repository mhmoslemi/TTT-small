import time

def normalize_simplex(a):
    """Project onto the simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones_like(a) / len(a)
    return a / s

def compute_autocorr(a):
    """Compute full autocorrelation (convolution with itself)."""
    return np.convolve(a, a, mode='full')

def soft_max_grad(autocorr, a, T):
    """Compute the soft-max gradient of the autocorrelation peak."""
    # Soft weights
    shifted = autocorr - autocorr.max()
    weights = np.exp(T * shifted)
    weights /= weights.sum()
    # Gradient: sum_k weights[k] * shift(a, k)
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(autocorr)):
        if weights[k] < 1e-12:

        # shift(a, k)_i = a[k-i] if 0 <= k-i < n else 0

        hi = min(n - 1, k)

        idx = k - np.arange(lo, hi+1)  # = k-lo, k-lo-1, ..., k-hi
