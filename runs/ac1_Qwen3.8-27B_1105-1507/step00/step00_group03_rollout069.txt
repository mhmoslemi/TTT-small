2*n*max(convolve(a,a))/(sum(a))^2. Uses multiple initializations and
adaptive step sizes with periodic restarts.

import time
import math


def compute_subgradient_fft(a, active_mask, n):
    """Compute subgradient g[k] = sum_j e[j]*a[j-k] for k=0..n-1 using FFT."""
    L = 1
    while L < 3 * n:
        L *= 2
    E = np.fft.rfft(active_mask, L)
    A = np.fft.rfft(a, L)
    R = np.fft.irfft(E * np.conj(A), L)
    return R[:n]


def fast_score(a):
    """Compute score without full evaluate_sequence overhead."""
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    S = a.sum()
    if S < 1e-10:
        return float('inf')
    return 2.0 * n * M / (S * S)

    # Each constraint involves at most n terms.
    # Total non-zeros: O(n^2) which is ~6.4M for n=2535. This is fine.

    # Alternative: solve the dual which might be easier
    # Primal: max 1'b s.t. T(b) <= 1, b >= 0
    # where T is the (2n-1) x n convolution matrix
    # Dual: min sum(lambda) s.t. T'lambda >= 1, lambda >= 0

    # Let's use the dual for efficiency (n variables in dual vs 2n-1 in primal)
    # Actually both have similar sizes. Let's just solve the primal.

    # For n=2535, building the full constraint matrix is 5069 x 2535 ~ 12.8M entries
    # Most are zero (banded). Let's use sparse.
    import scipy.sparse as sp

    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            j = k - i  # index into b
            rows.append(k)
            cols.append(j)
            vals.append(a[i])

    T = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    constraints = [T @ b <= np.ones(2 * n - 1)]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000, eps=1e-5)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None

    if b.value is None:
        return None
    b_val = np.maximum(b.value, 0)
    s = b_val.sum()
    if s < 1e-10:
        return None
    return b_val / s


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function: subgradient descent with multiple initializations,
    FFT-accelerated correlation, and optional LP-based direction refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0

    # Get parent sequence
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = None

    if parent is None or parent.size == 0:
        parent = np.ones(2535)

    n = parent.size
    best = np.clip(parent, 0.0, 1000.0).copy()
    best_score = fast_score(best)
    print(f"Initial: n={n}, score={best_score:.6f}")

    # Phase 1: Optimize from parent (40% of budget)
    t1 = (deadline - time.monotonic()) * 0.35
    a0 = np.maximum(best, 0)
    if a0.sum() > 1e-10:
        a0 = a0 / a0.sum()
    a_opt, s_opt = subgradient_optimize(a0, n, t1, seed, top_k=50)
    if s_opt < best_score:
        best = a_opt * best.copy().max() * (best.sum() / max(a_opt.sum(), 1e-10))
        # Actually just scale back to original range
        best = (a_opt * parent.sum()).copy()
        best_score = s_opt
        print(f"Phase 1: score={best_score:.6f}")

    # Phase 2: Try edge-peaked initialization (20% of budget)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t2 = remaining * 0.35
        # Edge-peaked: large at edges, small in middle
        x = np.arange(n, dtype=np.float64)
        mid = (n - 1) / 2.0
        a_edge = np.exp(-1.5 * (x - mid) ** 2 / (n * n / 4.0))
        # Normalize to peak at edges: actually this peaks in middle. Reverse:
        a_edge = 1.0 / (1.0 + 5.0 * np.exp(-2.0 * (x - mid) ** 2 / (n * n / 4.0)))
        a_edge = np.maximum(a_edge, 0)
        a_edge = a_edge / a_edge.sum()
        a_opt, s_opt = subgradient_optimize(a_edge, n, t2, seed + 1, top_k=50)
        if s_opt < best_score:
            best = (a_opt * parent.sum()).copy()
            best_score = s_opt
            print(f"Phase 2 (edge): score={best_score:.6f}")

    # Phase 3: Try sinusoidal initializations (15% of budget each for a few)
    remaining = deadline - time.monotonic()
    freqs_to_try = [1, 2, 3, 5]
    for p in freqs_to_try:
        if remaining < 15:
            break
        t3 = min(remaining * 0.2, 80)
        x = np.arange(n, dtype=np.float64)
        a_sin = 1.0 + 0.7 * np.cos(2.0 * np.pi * p * x / n + rng.uniform(0, 2 * np.pi))
        a_sin = np.maximum(a_sin, 0)
        a_sin = a_sin / a_sin.sum()
        a_opt, s_opt = subgradient_optimize(a_sin, n, t3, seed + 10 + p, top_k=50)
        if s_opt < best_score:
            best = (a_opt * parent.sum()).copy()
            best_score = s_opt
            print(f"Phase 3 (sin p={p}): score={best_score:.6f}")
        remaining = deadline - time.monotonic()

    # Phase 4: Random restarts (10% of budget)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_restarts = min(5, int(remaining / 30))
        for r in range(n_restarts):
            if deadline - time.monotonic() < 10:
                break
            a_rand = rng.uniform(0, 1, n)
            a_rand = a_rand / a_rand.sum()
            t4 = min(deadline - time.monotonic() - 5, 40)
            if t4 < 5:
                break
            a_opt, s_opt = subgradient_optimize(a_rand, n, t4, seed + 100 + r, top_k=80)
            if s_opt < best_score:
                best = (a_opt * parent.sum()).copy()
                best_score = s_opt
                print(f"Phase 4 (rand {r}): score={best_score:.6f}")

    # Phase 5: Fine-tune from best (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_best = np.maximum(best, 0)
        if a_best.sum() > 1e-10:
            a_best = a_best / a_best.sum()
        a_opt, s_opt = subgradient_optimize(a_best, n, remaining - 2, seed + 999, top_k=30)
        if s_opt < best_score:
            best = (a_opt * parent.sum()).copy()
            best_score = s_opt
            print(f"Phase 5 (fintune): score={best_score:.6f}")

    # Final verification with actual evaluate_sequence
    final_score = evaluate_sequence([float(max(0.0, x)) for x in best])
    if not np.isfinite(final_score) or final_score > best_score:
        # Fallback: return parent
        print(f"Warning: final eval={final_score}, using parent (score={evaluate_sequence([float(max(0.0,x)) for x in parent])})")
        best = parent
        best_score = evaluate_sequence([float(max(0.0, x)) for x in parent])

    print(f"FINAL: score={best_score:.6f}")
    return [float(max(0.0, x)) for x in best]
