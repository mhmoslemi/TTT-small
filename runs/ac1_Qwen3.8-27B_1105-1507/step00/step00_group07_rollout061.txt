Uses convex LP "best response" iterations combined with line search and multi-length search.

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix

def build_conv_matrix(f, n):
    """Build sparse matrix A where A @ g gives full convolution of f and g."""
    all_rows = []
    all_cols = []
    all_vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            all_rows.append(np.full(len(js), k))
            all_cols.append(js)
            # A[k, j] = f[k - j]
            all_vals.append(f[k - js])
    row_arr = np.concatenate(all_rows)
    col_arr = np.concatenate(all_cols)
    val_arr = np.concatenate(all_vals)
    return csr_matrix((val_arr, (row_arr, col_arr)), shape=(2 * n - 1, n))

def solve_lp(f, M, n, solver=cp.CLARABEL):
    """Solve the LP: max sum(g) s.t. f*g <= M, g >= 0."""
    A = build_conv_matrix(f, n)
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= M, g >= 0]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None

        return None
    g_val = np.asarray(g.value, dtype=np.float64)

    return np.maximum(0, g_val)

"""
LP-based iterative optimization for minimizing peak autocorrelation ratio of step functions.
Uses scipy HiGHS LP solver with precomputed sparse matrix indices for efficient
'best response' direction finding, combined with line search, multi-length exploration,
and random perturbation polishing.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

_conv_indices_cache = {}


def get_conv_indices(n):
    """Get or compute row/col index arrays for the full convolution matrix of size n."""
    if n in _conv_indices_cache:
        return _conv_indices_cache[n]
    all_rows = []
    all_cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1, dtype=np.int64)
            all_rows.append(np.full(len(js), k, dtype=np.int64))
            all_cols.append(js)
    row_arr = np.concatenate(all_rows)
    col_arr = np.concatenate(all_cols)
    _conv_indices_cache[n] = (row_arr, col_arr)
    return row_arr, col_arr


def build_conv_matrix(f, n):
    """Build sparse matrix A (shape (2n-1, n)) where A @ g = full conv(f, g)."""
    row_arr, col_arr = get_conv_indices(n)
    vals = f.astype(np.float64)[row_arr - col_arr]
    return csr_matrix((vals, (row_arr, col_arr)), shape=(2 * n - 1, n))


def solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f,g)_k <= M for all k, g >= 0. Returns g or None."""
    try:
        A = build_conv_matrix(f, n)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 120})
        if res.status == 0 and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def normalize_seq(f, target_sum=1.0):
    """Normalize non-negative array to have the given sum."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = np.sum(f)
    if s > 1e-12:
        return f * (target_sum / s)
    return np.ones_like(f, dtype=np.float64) * (target_sum / len(f))


def do_line_search(f, g, n):
    """Find best blend coefficient t minimizing score along (1-t)*f + t*g."""
    best_score = float('inf')
    best_t = 0.0
    t_list = [0.02, 0.05, 0.1, 0.2, 0.4, 0.6, 0.8, 0.95]
    for t in t_list:
        candidate = (1.0 - t) * f + t * g
        candidate = normalize_seq(candidate, 1.0)
        try:
            score = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            score = float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_t = t
    return best_t, best_score


def lp_optimize(f_init, n, budget_s, verbose=True):
    """Run LP-based iterations to minimize peak autocorrelation. Returns (best_seq, best_score)."""
    deadline = time.monotonic() + budget_s
    f = normalize_seq(np.asarray(f_init, dtype=np.float64), 1.0).copy()

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    if verbose:
        print(f"    LP: n={n}, init={best_score:.6f}, budget={budget_s:.0f}s")

    iter_count = 0
    no_improve = 0
    rng_local = np.random.default_rng(12345 + n)

    while time.monotonic() < deadline - 3.0:
        t0 = time.monotonic()

        conv = np.convolve(f, f)
        M = float(np.max(conv))

        g = solve_lp(f, M, n)
        if g is None:
            if verbose:
                print(f"      LP solve failed at iter {iter_count}")
            break

        g_sum = np.sum(g)
        if g_sum < 1e-12:
            break
        g = normalize_seq(g, 1.0)

        best_t, best_new_score = do_line_search(f, g, n)

        if best_new_score < best_score - 1e-10:
            f = (1.0 - best_t) * f + best_t * g
            f = normalize_seq(f, 1.0)
            best = f.copy()
            best_score = best_new_score
            no_improve = 0
        else:
            no_improve += 1
            # Move slightly toward g to change LP landscape
            f = (1.0 - 0.02) * f + 0.02 * g
            f = normalize_seq(f, 1.0)
            if no_improve >= 8:

                f = normalize_seq(f, 1.0)

    if verbose:
        print(f"    LP done: {iter_count} iters, best={best_score:.6f}")
    return best, best_score
