Strategy:
1. Initialize with central binomial coefficient sequence (theoretical J → pi/2).
2. Apply subgradient descent + LP-based refinement.
3. Try multiple sequence lengths.

import numpy as np
import time
import math

def central_binom_sequence(n):
    """Generate a[k] = binom(2k,k)/4^k for k=0,...,n-1."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k-1) / (2*k)
    return a

def compute_conv_max_and_sum(a):
    """Compute max of convolve(a,a) and sum of a."""
    c = np.convolve(a, a)
    return c.max(), a.sum(), c

def subgradient_step(a, learning_rate):
    """One step of projected subgradient descent."""
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    S = a.sum()
    # Find all lags achieving the max
    max_lags = np.where(np.isclose(c, M, atol=1e-10))[0]
    # Subgradient of J = 2n*M/S^2 w.r.t. a[i]
    # (using S=1 normalization for simplicity, then renormalize)
    # ∂M/∂a[i] = (1/|K|) * sum_{k in K} c_k' where c_k = sum_j a[j]*a[k-j]
    # ∂c_k/∂a[i] = a[k-i] (if 0<=k-i<n)
    g = np.zeros(n)
    for k in max_lags:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += a[j]
    g /= len(max_lags)
    # Full subgradient of J = 2n*M/S^2:
    # ∂J/∂a[i] = 2n*(S*g[i] - 2*M)/S^3
    grad_J = 2*n*(S*g - 2*M) / S**3
    # Update
    a_new = a - learning_rate * grad_J
    # Project to non-negative
    a_new = np.maximum(a_new, 0)
    # Renormalize to keep sum constant (optional, since objective is scale-invariant)
    if a_new.sum() > 0:
        a_new = a_new / a_new.sum() * S
    return a_new

def central_binom_sequence(n):
    """Generate a[k] = binom(2k, k) / 4^k for k = 0, ..., n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2.0*k - 1.0) / (2.0*k)
    return a

def normalize_1(a):
    """Normalize a to have sum 1, maintaining non-negativity."""
    s = a.sum()
    if s > 1e-300:
        return a / s
    return a

def compute_conv(a):
    return np.convolve(a, a)

def subgradient_step(a, lr):
    """
    One step of projected subgradient descent on f(a) = max(convolve(a,a)).
    a is assumed to be normalized (sum=1).
    """
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    # Active set: lags within 1% of max (to handle near-ties)
    tol = 0.01 * M
    K = np.where(c >= M - tol)[0]
    m = len(K)
    
    # Compute subgradient g[i] = (1/m) * sum_{k in K} a[k-i]
    # Using: g = convolve(v, a[::-1])[n-1:2n-1] / m
    v = np.zeros(2*n-1)
    v[K] = 1.0
    a_rev = a[::-1]
    conv_result = np.convolve(v, a_rev)
    g = conv_result[n-1:2*n-1] / m
    
    # Update
    a_new = a - lr * g
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_1(a_new)
    return a_new

def lp_direction(a, M):
    """
    Solve LP to find direction b that maximizes sum(b) 
    subject to conv(a,b) <= M and b >= 0.
    Returns normalized b or None if LP fails.
    """
    from scipy.optimize import linprog
    n = len(a)
    
    # Variables: b[0], ..., b[n-1]
    # Objective: minimize -sum(b) (to maximize sum(b))
    c_obj = -np.ones(n)
    
    # Constraints: conv(a, b)[k] <= M for k = 0, ..., 2n-2
    # conv(a,b)[k] = sum_{i} a[i] * b[k-i] <= M
    # For each k, the constraint is: sum_i A[i,k] * b[i] <= M
    # where A[i,k] = a[i] if 0<=k-i<n else 0
    
    # Build constraint matrix (sparse would be better, but let's use dense for small n)
    # Number of constraints: 2n-1
    if n <= 3000:
        A_ub = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    A_ub[k, i] = a[i]
        b_ub = np.full(2*n-1, M)
    else:
        # Too large for dense, skip LP
        return None
    
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    b = res.x
    s = b.sum()
    if s < 1e-10:
        return None
    return b / s  # Normalize to sum=1

def block_perturb(a, rng, rel_width=0.02, scale=0.05):
    """Apply a random multiplicative block perturbation."""
    n = len(a)
    a_new = a.copy()
    width = max(1, int(n * rel_width))
    start = int(rng.integers(0, max(1, n - width)))
    a_new[start:start+width] *= np.exp(rng.normal(0, scale, size=width))
    a_new = np.maximum(a_new, 0.0)
    return a_new
