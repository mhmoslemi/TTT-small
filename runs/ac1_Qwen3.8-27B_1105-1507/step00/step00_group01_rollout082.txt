"""
Peak-reducing gradient descent with FFT-accelerated pressure computation
and multiple initializations for minimizing 2*n*max(a*a)/(sum(a))^2.
"""

import numpy as np
from scipy.fft import fft, ifft
import time


def fast_convolve(a, b):
    """Fast linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    return ifft(fft(a, size) * fft(b, size)).real[:n]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = fast_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def compute_pressure_fft(a, peak_indices):
    """Compute pressure using FFT: pressure[i] = 2 * sum_{k in peaks} a[k-i]."""
    n = len(a)
    c_len = 2 * n - 1
    size = 1 << (3 * n - 2).bit_length()
    I = np.zeros(c_len)
    I[peak_indices] = 1.0
    # Cross-correlation: R[i] = sum_m a[m] * I[m+i]
    R = ifft(fft(a, size) * conj(fft(I, size))).real[:n]
    return 2.0 * R

Hmm wait, I realize there might be an issue with the `conj` in the FFT pressure computation. Since I is real, `conj(fft(I, size))` is just the complex conjugate of the FFT of I, which is correct for the cross-correlation formula.

Let me double-check the formula:
R[i] = sum_m a[m] * I[m+i]

The circular cross-correlation theorem states:
ifft(FFT(x) * conj(FFT(y)))[k] = sum_m x[m] * y[(m-k) mod L]

Wait, that gives y[m-k], not y[m+k]. Let me be more careful.

The circular cross-correlation is defined as:
(x ⋆_c y)[k] = sum_m x[m] * conj(y[(m-k) mod L])

So ifft(FFT(x) * conj(FFT(y)))[k] = sum_m x[m] * conj(y[(m-k) mod L])

For real y: = sum_m x[m] * y[(m-k) mod L]

I want R[i] = sum_m a[m] * I[m+i]

So I need y[(m-k) mod L] = I[m+i], which means (m-k) mod L = m+i, so k = -i (mod L), i.e., k = L - i.

So R[i] = ifft(FFT(a) * conj(FFT(I)))[L - i]

For i = 0: R[0] = ifft(...)[L] = ifft(...)[0] (since L mod L = 0)
For i = 1: R[1] = ifft(...)[L-1]
For i = 2: R[2] = ifft(...)[L-2]

So R[i] = result[(L-i) % L] where result = ifft(FFT(a, L) * conj(FFT(I, L))).real

Equivalently: R[n-1:-1:-1] = result[L-1:0:-1] ... hmm this is getting confusing.
