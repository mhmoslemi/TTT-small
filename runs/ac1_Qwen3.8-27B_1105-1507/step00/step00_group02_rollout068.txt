```python
"""
Minimizes 2n*max(f*f)/(sum(f))^2 using L-BFGS-B with progressive top-K smoothing,
FFT-accelerated convolution, and multiple diverse restarts.
The objective is scale-invariant, so only shape matters.
Progressive K reduction transitions from a smooth approximation to the true max.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fft_convolve_square(f):
    """Compute linear convolution f*f using FFT. Returns array of length 2n-1."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    return np.fft.irfft(F * F, fft_size)[:size]


def topk_objective_and_gradient(f, K=5):
    """
    Compute the scaled top-K smoothed objective and its gradient.
    obj = 2n * mean(top-K of conv) / S^2  where S = sum(f).
    This is a smooth approximation to 2n*max(conv)/S^2.
    """
    n = len(f)
    conv = fft_convolve_square(f)
    S = np.sum(f)

    if S < 1e-15:
        return 1e10, np.zeros(n)

    # Get top K indices
    if K >= len(conv):
        top_k_idx = np.arange(len(conv))
    else:
        part_idx = np.argpartition(conv, -K)[-K:]
        top_k_idx = part_idx[np.argsort(conv[part_idx])[::-1]]

    top_vals = conv[top_k_idx]
    M = np.mean(top_vals)

    value = 2.0 * n * M / (S * S)

    # Gradient of M w.r.t. f_m:
    # dC_k/df_m = 2*f_{k-m} if 0 <= k-m < n
    # dM/df_m = (2/K) * sum_{k in topK, 0<=k-m<n} f_{k-m}
    g_M = np.zeros(n)
    for k in top_k_idx:
        k_int = int(k)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        if lo <= hi:
            sl_start = k_int - hi
            sl_end = k_int - lo + 1
            g_M[lo:hi + 1] += f[sl_start:sl_end][::-1]
    g_M *= 2.0 / K

    # d(obj)/df_m = (2n/S^3) * (dM/df_m * S - 2*M)
    grad = (2.0 * n / (S ** 3)) * (g_M * S - 2.0 * M)

    return value, grad


def make_u_shape(n):
    """U-shaped: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    return (x - 0.5) ** 2 * 4.0 + 0.05


def make_cosine_bump(n):
    """Cosine bump: high at edges (cos has max at 0)."""
    x = np.linspace(0, 1, n)
    return 1.0 + 0.8 * np.cos(2.0 * np.pi * x)


def make_two_peaks(n):
    """Two Gaussian peaks."""
    x = np.linspace(0, 1, n)
    return (np.exp(-30.0 * (x - 0.25) ** 2) +
            np.exp(-30.0 * (x - 0.75) ** 2) + 0.03)


def make_v_shape(n):
    """V-shape: high at edges, low at center."""
    x = np.linspace(0, 1, n)
    return np.abs(x - 0.5) * 2.0 + 0.02


def make_symmetric_random(n, rng):
    """Random smooth symmetric sequence."""
    half = n // 2
    h = np.abs(np.cumsum(rng.normal(0, 1, half))) / half + 0.1
    if n % 2 == 0:
        f = np.concatenate([h[::-1], h])
    else:
        f = np.concatenate([h[::-1], h[1:]])
    return f[:n]


def make_block_random(n, rng):
    """Random block-constant sequence."""
    nblocks = int(rng.integers(4, 12))
    f_init = np.empty(n)
    pos = 0
    for j in range(nblocks):
        val = float(rng.uniform(0.05, 2.0))
        end = min(n, pos + n // nblocks + int(rng.integers(-5, 6)))
        f_init[pos:end] = val
        pos = end
    if pos < n:
        f_init[pos:] = float(rng.uniform(0.05, 2.0))
    return f_init


def fast_score(f):
    """Fast score using FFT convolution."""
    conv = fft_convolve_square(f)
    S = np.sum(f)
    if S < 0.01:
        return np.inf
    return float(2 * len(f) * np.max(conv) / (S * S))


def _lbgss_obj(x, K, n):
    """Objective wrapper for L-BFGS-B (top-level, no closures)."""
    val, _ = topk_objective_and_gradient(x, K)
    return val


def _lbgss_grad(x, K, n):
    """Gradient wrapper for L-BFGS-B (top-level, no closures)."""
    _, g = topk_objective_and_gradient(x, K)
    return g


def run_lbfgs_phase(f0, K_schedule, deadline, label=""):
    """
    Run L-BFGS-B with progressive K reduction.
    K_schedule: list of (K, maxiter, maxls) tuples.
    Returns best found (array, score).
    """
    current = f0.copy()
    best = f0.copy()
    best_score = fast_score(f0)
    n = len(f0)

    for K, maxiter, maxls in K_schedule:
        if time.monotonic() > deadline:
            break

        t0 = time.monotonic()

        # Use the fitted=False, maxcor default
        result = minimize(
            _lbgss_obj, current, args=(K, n),
            jac=_lbgss_grad,
            method='L-BFGS-B',
            bounds=[(0.0, 1000.0)] * n,
            options={'maxiter': maxiter, 'maxls': maxls,
                     'ftol': 1e-14, 'gtol': 1e-8, 'maxfun': maxiter * maxls}
        )

        if result.fun < best_score - 1e-10:
            candidate = np.maximum(result.x, 0.0)
            sc = fast_score(candidate)
            if sc < best_score:
                best = candidate.copy()
                best_score = sc
                current = candidate.copy()
                print(f"  {label} K={K}: improved to {best_score:.8f}", flush=True)
            else:
                current = candidate.copy()
        else:
            current = np.maximum(result.x, 0.0)

        elapsed = time.monotonic() - t0
        print(f"  {label} K={K}: fun={result.fun:.8f}, best={best_score:.8f}, took {elapsed:.1f}s", flush=True)

    return best, best_score


def subgradient_polish(f, deadline, max_iters=1000):
    """
    Subgradient descent polishing for K=1 (exact max).
    Uses adaptive step size with backtracking.
    """
    n = len(f)
    best = f.copy()
    best_score = fast_score(f)
    alpha = 0.005

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fft_convolve_square(best)
        max_val = np.max(conv)
        # Find active lags (within 0.01% of max)
        threshold = max_val * 0.9999
        active_lags = np.where(conv >= threshold)[0]

        if len(active_lags) > 30:
            active_lags = np.argpartition(conv, -30)[-30:]

        # Compute subgradient (average over active lags)
        g = np.zeros(n)
        for k in active_lags:
            k_int = int(k)
            lo = max(0, k_int - n + 1)
            hi = min(n - 1, k_int)
            if lo <= hi:
                sl_s = k_int - hi
                sl_e = k_int - lo + 1
                g[lo:hi + 1] += 2.0 * best[sl_s:sl_e][::-1]
        g /= len(active_lags)

        # Center to preserve scale
        g_min = np.min(g)
        g = g - g_min

        f_new = best - alpha * g
        f_new = np.maximum(f_new, 0.0)

        sc = fast_score(f_new)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best = f_new
            best_score = sc
            alpha = min(alpha * 1.03, 0.1)
        else:
            alpha *= 0.92
            if alpha < 1e-6:
                break

    return best, best_score
```
