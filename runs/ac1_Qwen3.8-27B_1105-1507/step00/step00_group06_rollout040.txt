def gradient_at_argmax(v):
    s = v_to_s(v)
    F = compute_F(s)
    k_star = F.argmax()
    
    # grad_s[m] = 2*(1+s)[k*-m] if 0 <= k*-m < n else 0
    # Let's compute this:
    # For m = 0,...,n-1: j = k*-m, valid if 0 <= j < n, i.e., k*-n+1 <= m <= k*
    lo = max(0, k_star - (n-1))
    hi = min(n-1, k_star)
    
    grad_s = np.zeros(n)
    # For m in [lo, hi]: j = k_star - m ranges from k_star-lo to k_star-hi
    # grad_s[m] = 2*(1+s)[k_star - m]
    js = k_star - np.arange(lo, hi+1)
    grad_s[lo:hi+1] = 2.0 * (1.0 + s[js])
    
    # Project to free variables
    grad_v = grad_s[:m] - grad_s[n-1-np.arange(m)]
    return F.max(), grad_v
