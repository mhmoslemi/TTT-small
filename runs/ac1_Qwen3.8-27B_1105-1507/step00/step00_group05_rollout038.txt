= 2 * sum_{k=max(0,i)}^{min(2n-2, i+n-1)} s[k] * a[k-i]

This is like a "sliding dot product" of a with the shifted sequence s.

Alternative: compute it directly. For each i, g[i] = 2 * dot(a, s[i:i+n]). This is O(n^2) total if done naively. But with FFT, it's O(n log n).

The operation "g[i] = sum_j a[j] * s[i+j] for j=0..n-1" is a cross-correlation between a and s (of length 2n-1).

cross_correlation(a, s)[i] = sum_j a[j] * s[i+j]

This can be written as: cross_correlation(a, s) = convolution(a_reversed, s) (shifted appropriately).

Let a_rev[j] = a[n-1-j]. Then:
conv(a_rev, s)[i] = sum_j a_rev[j] * s[i-j] = sum_j a[n-1-j] * s[i-j]
Let k = n-1-j: = sum_k a[k] * s[i-n+1+k] = sum_k a[k] * s[(i-n+1)+k]

So cross_correlation(a, s)[i] = conv(a_rev, s)[i + n - 1] (shifted by n-1).

We can compute conv(a_rev, s) using FFT (it's the linear convolution of a_rev (length n) and s (length 2n-1), giving a result of length 3n-2).

So: g = 2 * ifft(fft(a_rev, 3n-1) * fft(s, 3n-1))[shifted]

This is doable in O(n log n).

def compute_gradient(a, s, n):
    """Compute g[i] = 2 * sum_j a[j] * s[i+j] for j=0..n-1."""
    a_rev = a[::-1]
    # conv(a_rev, s) has length n + (2n-1) - 1 = 3n-2
    N = 3*n  # next power of 2 or just a large enough size
    A = np.fft.rfft(a_rev, N)
    S = np.fft.rfft(s, N)
    conv_result = np.fft.irfft(A * S, N)
    # We want cross_corr[i] = conv_result[i + n - 1] for i = 0, ..., n-1
    g = 2 * conv_result[n-1 : 2*n-1]
    return g

def smoothed_objective(a, tau):
    """Compute the smoothed objective and its gradient."""
    n = len(a)
    # Compute convolution
    N = 2*n - 1
    c = np.fft.irfft(np.fft.rfft(a, N) ** 2, N)
    c = c[:2*n-1]  # ensure correct length
    # Compute softmax
    c_shifted = c - np.max(c)  # for numerical stability
    s = np.exp(c_shifted / tau) / np.sum(np.exp(c_shifted / tau))
    # Compute gradient
    a_rev = a[::-1]
    N2 = 3*n
    A = np.fft.rfft(a_rev, N2)
    S = np.fft.rfft(s, N2)
    conv_res = np.fft.irfft(A * S, N2)
    g = 2 * conv_res[n-1 : 2*n-1]
    return c, s, g

def optimize(a_init, tau, alpha_init, n_iter):
    a = a_init.copy().astype(float)
    a = np.maximum(0, a)
    S = np.sum(a)
    a = a * S / S  # normalize (no-op, but for clarity)
    best_a = a.copy()
    best_obj = 2*n*np.max(conv(a,a))/S**2
    for t in range(n_iter):
        c, s, g = smoothed_objective(a, tau)
        alpha = alpha_init / (1 + t * alpha_init / 100)  # diminishing step size
        a_new = a - alpha * g
        a_new = np.maximum(0, a_new)
        # Renormalize to maintain sum = S
        if np.sum(a_new) > 0:
            a_new = a_new * S / np.sum(a_new)
        a = a_new
        # Track best
        obj = 2*n*np.max(conv(a,a))/S**2
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    return best_a, best_obj

def fft_convolve(a, b):
    """Linear convolution using FFT."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()  # next power of 2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]

def compute_objective(a):
    """Compute J = 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    max_c = np.max(c)
    return 2.0 * n * max_c / (S * S), max_c, c
