import time

def propose_candidate():
    # Starting sequence from the provided global variable
    if not isinstance(height_sequence_1, (list, np.ndarray)):
        # Fallback if no initial sequence is provided

        n_val = len(height_sequence_1)
        if n_val != 6121:
            print(f"Warning: Initial sequence length {n_val} is not 6121. Using fallback.")
            candidate = np.ones(6121) * 0.01
        else:
            candidate = np.asarray(height_sequence_1, dtype=np.float64)
            # Normalize to meet the sum requirement
            sum_val = np.sum(candidate)

                candidate *= (0.01 / sum_val)
            # Clip values to ensure they are non-negative and not overscaling
            clipped = np.clip(candidate, 0.0, 1000.0)
            # If clipping was excessive, replace with fallback
            if np.any(clipped != candidate):
                print("Fallback: Initial sequence has zeros and must be reconstructed entirely.")
                candidate = np.ones(6121) * 0.01
    
    best = candidate.copy()
    best_score = np.inf
    n_elements = 6121  # Fixed target length

    time_budget = 3 * 60  # 3 minutes in seconds
    start_time = time.time()

    while time.time() - start_time < time_budget:

        try:
            current_score = float(2.0 * n_elements * max_conv / (sum_val**2))
            if np.isnan(current_score) or np.isinf(current_score):
                raise ValueError("Invalid score due to numerical issues")
                
            # If improvement is found, update best candidate
            if current_score < best_score:
                best_score = current_score
                best = best.copy()
                
            # Trigger diversity occasionally to break potential local minima
            if np.random.rand() < 0.2 or len(best) != n_elements:
                # This is unlikely, but use as indicator to vary density
                candidate_len = int(n_elements * (0.05 + 0.9 * np.random.rand()))
