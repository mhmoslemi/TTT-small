    M = max(conv)
    # Softmax weights
    logits = T * conv / M
    logits -= logits.max()  # for numerical stability

    w /= w.sum()

    grad = 2 * np.correlate(w, a, mode='full')[:n]
    # Update
    a_new = a - lr * grad
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() > 0:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new, M

import numpy as np
import time

def compute_score_fast(a):
    """Fast score computation (same as evaluate_sequence but without overhead)."""
    n = len(a)
    conv = np.convolve(a, a)
    M = conv.max()
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)

def soft_gradient_step(a, T, lr):
    """One step of gradient descent on smooth approximation of max(conv)."""
    conv = np.convolve(a, a)
    M = conv.max()
    if M <= 0:
        return a, M
    # Softmax weights concentrated on peaks
    shifted = T * conv / M - T  # range [-T, 0]
    w = np.exp(shifted)
    w /= w.sum()
    # Gradient
    grad = 2.0 * np.correlate(w, a, mode='full')[:len(a)]
    # Update with clipping and renormalization
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new * (a.sum() / s)
    return a_new, M

"""
Hybrid optimizer: soft-gradient descent on smoothed max-convolution,
with periodic LP refinement via cvxpy and local search polishing.
Targets score = 2n*max(conv(a))/sum(a)^2 for non-negative sequences.
"""

import numpy as np
import time

# ---------------------------------------------------------------------------
# Helper functions
# ---------------------------------------------------------------------------

def compute_score_fast(a):
    """Fast score: 2*max(conv(a))/ (sum(a)^2 / n)  ==  2n*max/S^2."""
    n = len(a)
    conv = np.convolve(a, a)
    M = conv.max()
    S = a.sum()
    if S < 1e-6:
        return float('inf')
    return 2.0 * n * M / (S * S)


def soft_gradient_step(a, T, lr):
    """
    One gradient step on f_T(a) = (1/T) log sum_p exp(T * conv_p)
    which approximates max(conv(a)) for large T.
    """
    conv = np.convolve(a, a)
    M = conv.max()
    if M <= 1e-15:
        return a.copy(), M
    # Softmax weights (numerically stable)
    logits = T * conv / M - T  # in [-T, 0]
    w = np.exp(logits)
    w /= w.sum()
    # grad_k = 2 * sum_p w[p] * a[p-k]  =  2 * correlate(w, a, 'full')[k]
    grad = 2.0 * np.correlate(w, a, mode='full')[:len(a)]
    # Update
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-12:
        a_new = a_new * (a.sum() / s)
    return a_new, M


def optimize_gradient(a, deadline, rng, verbose=False):
    """
    Soft-gradient descent with adaptive parameters and periodic resets.
    Returns (best_sequence, best_score).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 0.01:
        a = np.ones(n)
        S = n

    best = a.copy()
    best_score = compute_score_fast(a)
    # Also track an EMA of iterates
    a_ema = a.copy()
    ema_decay = 0.999

    T = 40.0       # temperature
    lr = 0.04      # learning rate
    step = 0
    stagnant = 0
    PATIENCE = 400

    while time.monotonic() < deadline:
        a, M = soft_gradient_step(a, T, lr)
        step += 1
        a_ema = ema_decay * a_ema + (1 - ema_decay) * a

        if step % 15 == 0:
            score = compute_score_fast(a)
            score_ema = compute_score_fast(a_ema)
            if score_ema < score:
                score = score_ema
                a = a_ema.copy()
            if score < best_score - 1e-12:
                best_score = score
                best = a.copy()
                stagnant = 0
            else:
                stagnant += 15

        # Adaptive schedule
        if step % 150 == 0:
            lr = max(0.005, lr * 0.92)
            T = min(800.0, T * 1.06)

        # Reset if stuck
        if stagnant >= PATIENCE:
            mode = rng.integers(3)
            if mode == 0:
                # multiplicative noise
                a = np.maximum(a * np.exp(rng.normal(0, 0.4, n)), 0)
            elif mode == 1:

            else:

            a = np.maximum(a, 1e-10)
            a *= S / a.sum()
            a_ema = a.copy()
            stagnant = 0
            T = 25.0
            lr = 0.05

    return best, best_score


def build_conv_matrix(f):
    """Build sparse matrix C such that C @ g = np.convolve(f, g)."""
    n = len(f)
    row = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    col = np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    from scipy.sparse import csr_matrix
    return csr_matrix((data, (row, col)), shape=(2 * n - 1, n))


def lp_step(f, M, solver=None):
    """
    Solve LP: max sum(g) s.t. C(f) @ g <= M, g >= 0.
    Returns optimal g (or None on failure).
    """
    import cvxpy as cp
    n = len(f)
    C = build_conv_matrix(f)
    g = cp.Variable(n, nonneg=True)
    rhs = M * np.ones(2 * n - 1)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [C @ g <= rhs])
    for sol in ([solver] if solver else [cp.SCS, cp.ECOS, cp.CLRABEL]) + [None]:
        try:
            if sol is not None:
                prob.solve(solver=sol, verbose=False)
            else:
                prob.solve(verbose=False)
            if g.value is not None and np.isfinite(g.value).all():
                return np.clip(g.value, 0, None)
        except Exception:
            continue
    return None


def optimize_lp(a_max_iter, deadline, rng, verbose=False):
    """
    LP-based iteration (SOTA-inspired):
    At each step, solve LP with current f to get g, then blend.
    """
    n = len(a_max_iter)
    a = a_max_iter.astype(np.float64).copy()
    S = a.sum()
    best = a.copy()
    best_score = compute_score_fast(a)
    t = 0.3  # blending parameter

    for it in range(a_max_iter):
        if time.monotonic() > deadline:
            break
        conv = np.convolve(a, a)
        M = conv.max()

        g = lp_step(a, M)
        if g is None:
            break

        # Rescale g to match sum
        gs = g.sum()
        if gs < 1e-12:
            break
        g = g * (S / gs)

        # Blend
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new *= S / a_new.sum()

        score = compute_score_fast(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new
            if verbose and it % 10 == 0:
                print(f"  LP iter {it}: score={score:.6f}, M={M:.6f}")
        else:
            a = a_new  # still update for exploration
            t = max(0.05, t * 0.9)

    return best, best_score


def local_search(a, deadline, rng):
    """Perturbation-based local search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best = a.copy()
    best_score = compute_score_fast(a)
    stall = 0
    max_stall = 200

    while time.monotonic() < deadline:
        strategy = rng.integers(4)
        if strategy == 0:
            # Single element
            a_try = a.copy()
            idx = rng.integers(n)
            a_try[idx] = max(0, a_try[idx] * np.exp(rng.normal(0, 0.5)))
        elif strategy == 1:
            # Block
            width = rng.integers(3, max(5, n // 30))
            start = rng.integers(0, n - width)
            a_try = a.copy()
            a_try[start:start+width] *= np.exp(rng.normal(0, 0.15, size=width))
        elif strategy == 2:
            # Mirror pair
            a_try = a.copy()
            idx = rng.integers(n // 2)
            f1 = np.exp(rng.normal(0, 0.3))
            a_try[idx] = max(0, a_try[idx] * f1)
            a_try[n - 1 - idx] = max(0, a_try[n - 1 - idx] / f1)
        else:
            # Smoothed random walk on log-scale
            a_try = a.copy()
            mask = rng.random(n) < 0.1
            a_try[mask] *= np.exp(rng.normal(0, 0.2, mask.sum()))
            a_try = np.maximum(a_try, 0)

        a_try *= S / max(a_try.sum(), 1e-12)
        score = compute_score_fast(a_try)
        if score < best_score - 1e-12:
            best_score = score
            best = a_try.copy()
            a = a_try
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                # Big jump
                a = np.maximum(a * np.exp(rng.normal(0, 0.8, n)), 0)
                a *= S / a.sum()
                stall = 0

    return best, best_score

                a_full_lp, sc_full = optimize_lp(
                    max(3, int((phase2_end - time.monotonic()) / 20.0)),
                    phase2_end, rng)
                sc_full = evaluate_sequence(a_full_lp.tolist())
                if np.isfinite(sc_full) and sc_full < best_score:
                    best_score = sc_full
                    best_overall = a_full_lp
                    print(f"[lp-full] (n={len(a_full_lp)}): {sc_full:.6f}")

    # ---- Phase 3: Local search polishing (~15% of time) ----
    phase3_end = t0 + (budget_s - 20.0) * 0.92
    if best_overall is not None and time.monotonic() < phase3_end - 10:
        a_ls, sc_ls = local_search(best_overall, phase3_end, rng)
        sc_ls = evaluate_sequence(a_ls.tolist())
        if np.isfinite(sc_ls) and sc_ls < best_score:
            best_score = sc_ls
            best_overall = a_ls
            print(f"[local] (n={len(a_ls)}): {sc_ls:.6f}")

    # ---- Phase 4: Final gradient sweep on best (~remaining time) ----
    if best_overall is not None and time.monotonic() < deadline - 5:
        a_final, sc_final = optimize_gradient(best_overall, deadline, rng)
        sc_final = evaluate_sequence(a_final.tolist())
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_overall = a_final
            print(f"[final-grad] (n={len(a_final)}): {sc_final:.6f}")

    # Fallback: ensure we always return something valid
    if best_overall is None or len(best_overall) == 0:
        best_overall = np.ones(100)
        best_score = evaluate_sequence(best_overall.tolist())

    print(f"\n=== FINAL: n={len(best_overall)}, score={best_score:.6f}, "
          f"elapsed={time.monotonic()-t0:.1f}s ===")

    return [float(x) for x in best_overall]
