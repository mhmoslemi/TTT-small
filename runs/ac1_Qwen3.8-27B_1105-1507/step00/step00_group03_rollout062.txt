```
def _score_fast(a):
    a = np.asarray(a, dtype=np.float64)
```

```python
"""
Search for a non-negative step-function height sequence minimizing
    2 * n * max(np.convolve(a, a)) / sum(a)^2.

Strategy:
- Use scale invariance and normalize candidate sequences to sum 1.
- Search a flexible one-sided power-exponential parameter family:
        a_i ~ (i+1)^(-beta) * exp(-delta * x_i - gamma * x_i^2)
  This family has empirically very flat self-convolutions when beta is
  moderately close to 0.5 and delta/gamma are tuned.
- Refine the best discrete sequence with an exact one-coordinate
  multiplicative coordinate-descent sweep.  The sweep updates the
  convolution in O(n) per coordinate/factor using the analytic effect
  of changing one coefficient.
"""

import numpy as np
import time


def _full_conv(a):
    """Full linear convolution for a 1D nonnegative float array."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([float(a[0] * a[0])], dtype=np.float64)

    # Direct convolution is faster and more accurate for small n.
    if n <= 256:
        b = np.convolve(a, a)
    else:
        m = 2 * n - 1
        N = 1
        while N < m:
            N <<= 1
        A = np.fft.rfft(a, N)
        b = np.fft.irfft(A * A, N)[:m]

    # The true convolution is nonnegative for nonnegative input.
    np.maximum(b, 0.0, out=b)
    return b


def _score_norm(a):
    """
    Score assuming sum(a) == 1 and no clipping is needed.
    This is used inside the controlled search where candidates are
    normalized and have tiny values.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 0:
        return np.inf
    b = _full_conv(a)
    mx = float(np.max(b))
    if mx < 0.0:
        mx = 0.0
    return float(2.0 * n * mx)


def _score_fast(a):
    """
    Robust score equivalent to evaluate_sequence, used for arbitrary
    starting sequences and final sanity checks.
    """
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    b = _full_conv(a)
    mx = float(np.max(b))
    if mx < 0.0:
        mx = 0.0
    return float(2.0 * a.size * mx / (s * s))


def _update_prefs(pref, suf, b):
    """
    pref[i] = max(b[0:i])
    suf[i]  = max(b[i:])
    """
    L = b.size
    pref[0] = 0.0
    np.maximum.accumulate(b, out=pref[1:])

    rev_acc = np.maximum.accumulate(b[::-1])
    suf[L] = 0.0
    suf[:L] = rev_acc[::-1]


def _make_param_seq(n, beta, delta, gamma):
    """
    Generate a normalized nonnegative sequence from a power-exponential template.
    """
    n = int(max(2, n))
    idx = np.arange(n, dtype=np.float64)
    x = idx / float(n - 1)
    base = (idx + 1.0) / float(n + 1.0)

    a = np.power(base, -float(beta))
    if delta != 0.0:
        a *= np.exp(-float(delta) * x)
    if gamma != 0.0:
        a *= np.exp(-float(gamma) * x * x)

    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 0.0:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a /= s
    return a


def _coordinate_sweep(a_arr, deadline, factors):
    """
    Exact one-coordinate multiplicative coordinate descent.

    For a candidate change a_i -> f*a_i, the change in every convolution
    coefficient is known analytically:
      - non-diagonal lag k=i+j: delta = 2*(f-1)*a_i*a_j
      - diagonal lag k=2*i:     delta = (f^2-1)*a_i^2

    This avoids recomputing the full convolution for every factor.
    """
    a = np.asarray(a_arr, dtype=np.float64).copy()
    n = a.size
    if n < 2:
        return a, np.inf

    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a = a / s

    b = _full_conv(a)
    L = 2 * n - 1

    pref = np.empty(L + 1, dtype=np.float64)
    suf = np.empty(L + 1, dtype=np.float64)
    _update_prefs(pref, suf, b)

    current_score = float(2.0 * n * np.max(b))
    start_score = current_score
    best_score = current_score
    best_a = a

    factors = tuple(float(f) for f in factors if abs(float(f) - 1.0) > 1e-12)
    if not factors:
        return a, current_score

    clean_count = 0
    progress = True
    min_pos = 1e-300

    while progress and time.monotonic() < deadline:
        progress = False

        for i in range(n):
            if time.monotonic() > deadline:
                break

            old = float(a[i])
            if old <= min_pos:
                continue

            base_slice = b[i:i + n]
            best_f = 1.0
            best_sc = current_score

            for f in factors:
                f = float(f)
                Snew = 1.0 + (f - 1.0) * old
                if Snew <= min_pos:
                    continue

                coef = 2.0 * (f - 1.0) * old

                # Modified slice b[i:i+n]
                cand = base_slice.copy()
                cand += coef * a
                cand[i] = b[2 * i] + (f * f - 1.0) * old * old

                inside = float(np.max(cand))
                if inside < 0.0:
                    inside = 0.0

                outside = pref[i]
                out2 = suf[i + n]
                if out2 > outside:
                    outside = out2

                M = inside if inside > outside else outside
                sc = (2.0 * n * M) / (Snew * Snew)

                if sc < best_sc:
                    best_sc = sc
                    best_f = f

            if best_f != 1.0:
                Snew = 1.0 + (best_f - 1.0) * old
                coef = 2.0 * (best_f - 1.0) * old

                b_cand = b.copy()
                b_cand[i:i + n] += coef * a
                b_cand[2 * i] = b[2 * i] + (best_f * best_f - 1.0) * old * old

                scale = 1.0 / (Snew * Snew)
                b_cand *= scale
                np.maximum(b_cand, 0.0, out=b_cand)

                a_cand = a.copy()
                a_cand[i] *= best_f
                a_cand *= (1.0 / Snew)

                # Remove rounding drift.
                scand = float(np.sum(a_cand))
                if scand > min_pos:
                    a_cand = a_cand / scand
                    b_cand = b_cand / (scand * scand)
                    np.maximum(b_cand, 0.0, out=b_cand)

                new_score = float(2.0 * n * np.max(b_cand))

                if np.isfinite(new_score) and new_score < current_score - 1e-12:
                    a = a_cand
                    b = b_cand
                    current_score = new_score
                    _update_prefs(pref, suf, b)
                    progress = True

                    if new_score < best_score:
                        best_score = new_score
                        best_a = a

                    clean_count += 1
                    if clean_count >= 50:
                        # Reclean with an FFT convolution to remove accumulation drift.
                        anorm = float(np.sum(a))
                        if anorm <= min_pos:
                            a = np.ones(n, dtype=np.float64) / float(n)
                        else:
                            a = a / anorm
                        b = _full_conv(a)
                        current_score = float(2.0 * n * np.max(b))
                        _update_prefs(pref, suf, b)
                        if current_score < best_score:
                            best_score = current_score
                            best_a = a
                        clean_count = 0

        if best_score < start_score:
            # At least one improvement occurred. Continue if time remains.
            pass

    # Final clean-up and protection against any tiny numerical regression.
    s = float(np.sum(a))
    if s > min_pos:
        a = a / s
    b = _full_conv(a)
    final_score = float(2.0 * n * np.max(b))

    if final_score < best_score:
        best_score = final_score
        best_a = a

    if best_score < final_score + 1e-12:
        return best_a, float(best_score)
    return a, final_score
```
