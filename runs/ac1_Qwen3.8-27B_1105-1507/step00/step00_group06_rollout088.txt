def resample_sequence(seq, n):
   if seq.size == n: return project_simplex(seq)
   x_old = np.linspace(0,1,seq.size)
   x_new = np.linspace(0,1,n)
   y = np.interp(x_new, x_old, seq)
   y = np.maximum(y,0)
   s=y.sum()
   if s<=0: return np.ones(n)/n
   return project_simplex(y/s)

import time
import numpy as np

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float).copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    den = np.arange(1, n + 1, dtype=float)
    conds = u > (cssv - 1.0) / den
    if not np.any(conds):
        return np.full(n, 1.0 / n)
    idx = np.flatnonzero(conds)
    rho = int(idx[-1]) + 1
    theta = cssv[rho - 1] / rho
    q = np.maximum(v - theta, 0.0)
    s = float(q.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / n)
    return q / s


def project_simplex_safe(v):
    """Fast projection when v is already very close to the simplex."""
    v = np.asarray(v, dtype=float)
    s = float(np.sum(v))
    if np.min(v) >= -1e-8 and abs(s - 1.0) < 1e-4:
        q = np.maximum(v, 0.0)
        qs = float(q.sum())
        if qs > 1e-300:
            return q / qs
    return project_simplex(v)


def normalize_nonneg(x):
    """Clip to nonnegative values and normalize to sum 1."""
    arr = np.asarray(x, dtype=float)
    if arr.ndim != 1 or arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 1e-300:
        return None
    return arr / s


def convolve_full(a, b):
    """Full nonnegative convolution, direct for small sizes and FFT otherwise."""
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    need = a.size + b.size - 1
    if need <= 0:
        return np.array([], dtype=float)
    if a.size * b.size <= 200000:
        c = np.convolve(a, b)
    else:
        size = 1 << int(max(1, need - 1).bit_length())
        fa = np.fft.rfft(a, size)
        fb = np.fft.rfft(b, size)
        c = np.fft.irfft(fa * fb, size)[:need]
    c = np.asarray(c, dtype=float)
    c = np.maximum(c, 0.0)
    return c


def score_normalized(seq):
    """Score for a sequence assumed or made to have sum 1."""
    p = normalize_nonneg(seq)
    if p is None:
        return np.inf
    c = convolve_full(p, p)
    return float(2.0 * p.size * np.max(c))


def smooth_max_weights(c, tau):
    """Weights proportional to exp(c/tau), the gradient of a smooth max."""
    c = np.asarray(c, dtype=float)
    if c.size == 0:
        return np.array([], dtype=float)
    if tau <= 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    shift = c - float(np.max(c))
    e = np.exp(shift / tau)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def smooth_max_value(c, tau):
    """Smooth maximum approximating max(c)."""
    c = np.asarray(c, dtype=float)
    if c.size == 0:
        return 0.0
    m = float(np.max(c))
    if tau <= 1e-300:
        return m
    e = np.exp((c - m) / tau)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return m
    return m + tau * float(np.log(s))


def conv_grad_A(p, w):
    """
    For c_k = sum_i p_i p_{k-i},
    sum_k w_k d c_k / d p_i = 2 * sum_k w_k p_{k-i}.
    This function returns the vector A_i = sum_k w_k p_{k-i}.
    """
    n = p.size
    if n == 0:
        return np.array([], dtype=float)
    return convolve_full(p[::-1], w)[n - 1:2 * n - 1]


def resample_sequence(seq, n):
    """Linearly resample a nonnegative sequence to a new length and normalize."""
    p = normalize_nonneg(seq)
    if p is None:
        return np.full(n, 1.0 / n)
    if p.size == n:
        return p.copy()
    x_old = np.linspace(0.0, 1.0, p.size)
    x_new = np.linspace(0.0, 1.0, n)
    y = np.interp(x_new, x_old, p)
    y = np.maximum(y, 0.0)
    s = float(y.sum())
    if s <= 1e-300:
        return np.full(n, 1.0 / n)
    return y / s


def smooth_noise(n, rng):
    """Smooth Gaussian noise of length n."""
    z = rng.normal(size=n)
    k = max(3, int(n // 128))
    kern = np.ones(k) / float(k)
    return np.convolve(z, kern, mode="same")


def block_perturb(p, rng, strength=0.25):
    """Multiplicative random block perturbation."""
    p = np.asarray(p, dtype=float)
    n = p.size
    q = p.copy()
    low = max(3, int(n // 300))
    high = max(low + 1, min(n, max(low + 1, int(n // 6))))
    for _ in range(3):
        width = int(rng.integers(low, high + 1))
        if width > n:
            width = n
        start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, strength)))
        q[start:start + width] *= factor
    return normalize_nonneg(q)


def make_initial_candidates(n, rng):
    """A diverse family of nonnegative normalized starting sequences."""
    cands = []
    x = np.linspace(0.0, 1.0, n)

    p = normalize_nonneg(np.ones(n))
    if p is not None:
        cands.append(p)

    # Monotone / midpoint-symmetric smooth shapes.
    for B in (0.2, 0.5, 0.8, 1.0, 1.2, 1.5, 1.8):
        p = normalize_nonneg(1.0 + B * (x - 0.5))
        if p is not None:
            cands.append(p)

    for lam in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        sig = 1.0 / (1.0 + np.exp(-lam * (x - 0.5)))
        p = normalize_nonneg(2.0 * sig)
        if p is not None:
            cands.append(p)

    for lam in (0.5, 1.0, 2.0, 4.0, 6.0):
        p1 = normalize_nonneg(np.exp(lam * x))
        if p1 is not None:
            cands.append(p1)
        p2 = normalize_nonneg(np.exp(-lam * x))
        if p2 is not None:
            cands.append(p2)

    for a in (0.25, 0.5, 1.0, 2.0):
        p1 = normalize_nonneg(np.power(x, a) + 0.01)
        if p1 is not None:
            cands.append(p1)
        p2 = normalize_nonneg(np.power(1.0 - x, a) + 0.01)
        if p2 is not None:
            cands.append(p2)

    # Odd-harmonic trigonometric candidates; these can depress the center peak.
    for a in (0.25, 0.45, 0.65, 0.80):
        for k in (1, 2, 3, 4):
            freq = (2 * k + 1) * np.pi
            for phi in (0.0, np.pi / 4.0, np.pi / 2.0):
                p = normalize_nonneg(1.0 + 0.05 + a * np.cos(freq * x + phi))
                if p is not None:
                    cands.append(p)

    # Random smooth Fourier candidates.
    for _ in range(5):
        z = np.zeros(n)
        for f in range(1, 13):
            amp = np.exp(-0.25 * f)
            ph1 = rng.uniform(0.0, 2.0 * np.pi)
            ph2 = rng.uniform(0.0, 2.0 * np.pi)
            z += amp * (
                rng.normal() * np.cos(2.0 * np.pi * f * x + ph1)
                + rng.normal() * np.sin(2.0 * np.pi * f * x + ph2)
            )
        z += 0.8 * x
        p = normalize_nonneg(np.exp(0.45 * z))
        if p is not None:
            cands.append(p)

    # Smooth Gaussian-noise candidates.
    for _ in range(4):
        z = rng.normal(size=n) * 0.6
        k = max(3, n // 80)
        z = np.convolve(z, np.ones(k) / float(k), mode="same")
        z += 1.2 * x
        p = normalize_nonneg(np.exp(z))
        if p is not None:
            cands.append(p)

    return cands


def item_score_key(item):
    return item[0]


def score_candidates(p_list):
    """Return list of (score, counter, sequence) sorted by score."""
    scored = []
    cnt = 0
    for p in p_list:
        q = normalize_nonneg(p)
        if q is None:
            continue
        s = score_normalized(q)
        if np.isfinite(s):
            scored.append((s, cnt, q))
            cnt += 1
    scored.sort(key=item_score_key)
    return scored


def get_parent_normalized():
    hs = globals().get("height_sequence_1", None)
    if hs is None:
        return None
    return normalize_nonneg(hs)


def lp_direction(p, M, ub=None):
    """
    Solve the SoA LP direction:
        maximize sum(r)
        subject to convolve(p, r) <= M, r >= 0, r <= ub.
    """
    if linprog is None:
        return None
    p = np.asarray(p, dtype=float)
    m = p.size
    if m == 0 or M <= 0.0:
        return None

    A = np.zeros((2 * m - 1, m), dtype=float)
    for j in range(m):
        A[j:j + m, j] = p

    b = np.full(2 * m - 1, float(M), dtype=float)
    c = -np.ones(m, dtype=float)

    if ub is None:
        ub = max(10.0 / float(m), 4.0 * float(np.max(p)), 10.0 * float(M))

    bounds = [(0.0, float(ub)) for _ in range(m)]

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    except Exception:
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="interior-point")
        except Exception:
            return None

    if res.success and res.x is not None:
        r = np.asarray(res.x, dtype=float)
        r = np.where(np.isfinite(r), r, 0.0)
        r = np.maximum(r, 0.0)
        s = float(r.sum())
        if s > 1e-12:
            return r
    return None


def lp_only_refine(p0, time_end, rng, max_iters=40):
    """A few LP-based SoA-style updates on a coarse sequence."""
    p = normalize_nonneg(p0)
    if p is None:
        p = np.ones(8) / 8.0

    m = p.size
    kernel = np.array([0.2, 0.3, 0.3, 0.2], dtype=float)

    best = p.copy()
    best_score = score_normalized(best)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= time_end:
            break

        s = score_normalized(p)
        if s < best_score - 1e-11:
            best = p.copy()
            best_score = s
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 12 and it >= 8:
            break

        c = convolve_full(p, p)
        M = float(np.max(c))
        r = lp_direction(p, M)
        if r is None:
            break

        r = normalize_nonneg(r)
        if r is None:
            break

        rs = normalize_nonneg(np.convolve(r, kernel, mode="same"))
        r_blend = normalize_nonneg(0.5 * r + 0.5 * (rs if rs is not None else r))

        directions = [r]
        if rs is not None:
            directions.append(rs)
        if r_blend is not None:
            directions.append(r_blend)

        improved = False
        for dr in directions:
            if dr is None:
                continue
            for t in (0.25, 0.5, 0.1, 0.75):
                q = normalize_nonneg((1.0 - t) * best + t * dr)
                if q is None:
                    continue
                qs = score_normalized(q)
                if qs < best_score - 1e-12:
                    best = q
                    best_score = qs
                    p = q.copy()
                    improved = True

        if not improved:
            p = normalize_nonneg(best * (1.0 + 0.12 * smooth_noise(m, rng)))
            if p is None:
                p = best.copy()

    return best


def projected_refine(p0, time_end, rng, tau_factor=3.0):
    """
    Projected subgradient/line-search refinement for the smooth peak objective.
    Returns the best sequence found.
    """
    p = normalize_nonneg(p0)
    if p is None:
        p = np.ones(256) / 256.0

    n = p.size
    best = p.copy()
    best_score = score_normalized(best)

    c0 = convolve_full(p, p)
    mean_c0 = float(np.mean(c0))
    tau = max(1e-12, tau_factor * mean_c0)
    tau_min = max(1e-12, 0.03 * mean_c0)

    alphas = (
        1.0,
        5.0,
        20.0,
        float(max(50, n // 10)),
        float(max(100, n // 2)),
        float(n),
        float(2 * n),
    )

    no_improve = 0
    max_no = 70

    while time.monotonic() < time_end:
        c = convolve_full(p, p)
        maxc = float(np.max(c))
        score = 2.0 * n * maxc

        if score < best_score - 1e-11:
            best = p.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no:
            p = best.copy()
            c = convolve_full(p, p)
            mean_c = float(np.mean(c))
            tau = max(tau_min, 3.0 * mean_c)
            no_improve = 0
            max_no = int(max_no * 1.3)
            continue

        w = smooth_max_weights(c, tau)
        A = conv_grad_A(p, w)
        d = -2.0 * A
        d = d - float(np.mean(d))
        dn = float(np.max(np.abs(d)))

        if not np.isfinite(dn) or dn <= 1e-300:
            p = block_perturb(best, rng, 0.25)
            tau = max(tau_min, tau * 0.75)
            continue

        d = d / dn * (1.0 / n)
        accepted = False

        for alpha in alphas:
            q = p + alpha * d
            q = 0.999 * q + 0.001 / n
            q = project_simplex_safe(q)

            if float(np.max(np.abs(q - p))) <= 1e-14:
                continue

            c2 = convolve_full(q, q)
            score2 = 2.0 * n * float(np.max(c2))

            if score2 < score - 1e-10:
                p = q
                accepted = True
                break

        if not accepted:
            if no_improve > 8:
                p = block_perturb(best, rng, 0.30)
            else:
                tmp = normalize_nonneg(p * (1.0 + 0.05 * smooth_noise(n, rng)))
                if tmp is None:
                    p = best.copy()
                else:
                    p = tmp

        tau = max(tau_min, tau * 0.99)
        if no_improve > 18:
            c = convolve_full(p, c)
            tau = max(tau, 0.7 * float(np.mean(c)))

    return best, best_score


def exact_score_sequence(seq):
    """Use the official evaluator when available."""
    try:
        vals = [float(x) for x in seq]
        return float(evaluate_sequence(vals))
    except Exception:
        return np.inf


def finalize_sequence(seq):
    arr = np.asarray(seq, dtype=float)
    if arr.ndim != 1 or arr.size == 0:
        arr = np.ones(2048)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        arr = np.ones(arr.size)
    else:
        arr = arr / s
    return arr


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hierarchical search for low 2*n*max(self-convolution)/sum^2.

    The program initializes many smooth nonnegative sequences, refines coarse
    patterns with LP-style SoA direction updates where available, then runs
    projected smooth-peak line search on the main working length and finally
    selects the best upsampling of the best pattern.
    """
    start = time.monotonic()
    deadline = start + max(10.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(seed)

    parent = get_parent_normalized()

    # Main work length.  This is long enough that finite-size endpoint effects
    # are small, but fast enough for many FFT-based line-search iterations.
    N = 2048

    # Initial direct candidates at the main length.
    init_cands = make_initial_candidates(N, rng)
    if parent is not None:
        init_cands.append(resample_sequence(parent, N))

    init_scored = score_candidates(init_cands)
    if init_scored:
        best = init_scored[0][2].copy()
        best_score = init_scored[0][0]
    else:
        best = np.ones(N) / N
        best_score = score_normalized(best)

    print(f"[start] N={N} best_score={best_score:.10f}", flush=True)

    # Collect a small pool of promising initial seeds.
    pool = [init_scored[i][2].copy() for i in range(min(3, len(init_scored)))]

    # Coarse density/pattern search.  This explores shapes that may be missed
    # by direct large-n search.
    if time.monotonic() < deadline - 260:
        m = 192
        dens_cands = make_initial_candidates(m, rng)
        if parent is not None:
            dens_cands.append(resample_sequence(parent, m))
        dens_scored = score_candidates(dens_cands)

        for i in range(min(3, len(dens_scored))):
            s0 = dens_scored[i][0]
            p0 = dens_scored[i][2].copy()
            if s0 >= best_score + 0.35:
                continue

            alloc = max(25.0, min(160.0, (deadline - time.monotonic() - 220.0) / 2.0))
            t_end_lp = start + 0.4 * (time.monotonic() - start) + alloc * 0.45
            t_end_lp = min(time.monotonic() + alloc * 0.45, deadline - 180.0)
            p_lp = p0
            if linprog is not None:
                p_lp = lp_only_refine(p0, t_end_lp, rng, max_iters=35)

            t_end_grad = min(time.monotonic() + alloc * 0.55, deadline - 200.0)
            p_gr, sg = projected_refine(p_lp, t_end_grad, rng)

            up = resample_sequence(p_gr, N)
            su = score_normalized(up)
            if su < best_score:
                best = up
                best_score = su
            print(f"[coarse] m={m} seed={i} score={su:.10f} best={best_score:.10f}", flush=True)

            if time.monotonic() > deadline - 260:
                break

    # Main direct refinement on the main length, using several seeds.
    if time.monotonic() < deadline - 150:
        top_pool = []
        for p in pool:
            q = normalize_nonneg(p)
            if q is not None:
                top_pool.append(q)
        top_pool.append(best.copy())
        top_pool.append(resample_sequence(best, N).copy())

        tscored = score_candidates(top_pool)
        if tscored:
            top_pool = [tscored[i][2].copy() for i in range(min(4, len(tscored)))]

        remain = deadline - time.monotonic()
        if remain > 160:
            # Give at least one seed a long slice, then shorter slices to others.
            # The best seed is always included first.
            order = [best.copy()]
            for p in top_pool:
                if p.size != N:
                    p = resample_sequence(p, N)
                if p.size == N:
                    order.append(p)

            # Keep order length small.
            order = order[:4]

            slices = []
            rem = deadline - time.monotonic()
            if rem > 300:
                slices.append((order[0], min(rem * 0.50, rem - 60.0)))
            if len(order) > 1 and rem > 220:
                slices.append((order[1], min(rem * 0.25, rem - 60.0)))
            if len(order) > 2 and rem > 180:
                slices.append((order[2], min(rem * 0.20, rem - 60.0)))
            if len(order) > 3 and rem > 150:
                slices.append((order[3], min(rem * 0.15, rem - 60.0)))

            for pseed, dt in slices:
                t_end = time.monotonic() + max(5.0, dt)
                t_end = min(t_end, deadline - 60.0)
                if time.monotonic() >= t_end:
                    break
                p_opt, s_opt = projected_refine(pseed, t_end, rng)
                if s_opt < best_score:
                    best = p_opt
                    best_score = s_opt
                print(f"[refine] N={N} score={s_opt:.10f} best={best_score:.10f}", flush=True)

    # Final upsampling / length selection.  The objective is nearly scale/length
    # invariant in the continuum limit; longer lengths can slightly reduce finite
    # endpoint artifacts.
    final_candidates = [best.copy()]
    if parent is not None:
        final_candidates.append(resample_sequence(parent, N))

    for nl in (N, 4096, 5582):
        try:
            cand = resample_sequence(best, nl)
            sc = score_normalized(cand)
            final_candidates.append(cand)
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc
                print(f"[upsample] n={nl} score={sc:.10f} best={best_score:.10f}", flush=True)
        except Exception:
            pass

    # Additional final polish if enough time remains.
    if time.monotonic() < deadline - 120:
        p_opt, s_opt = projected_refine(best, deadline - 45, rng)
        if s_opt < best_score:
            best = p_opt
            best_score = s_opt
        print(f"[final refine] n={len(best)} score={s_opt:.10f} best={best_score:.10f}", flush=True)

    # Final short validation with the official evaluator.  The evaluator is
    # authoritative and will be used to pick among the top final candidates.
    validation = []
    for cand in final_candidates:
        arr = finalize_sequence(cand)
        sc = exact_score_sequence(arr)
        if np.isfinite(sc):
            validation.append((sc, len(arr), arr))
    if validation:
        validation.sort(key=item_score_key)
        best = validation[0][2].copy()
        best_score = validation[0][0]

    # Absolute fallback to a valid sequence with score 2.
    if best is None or not np.all(np.isfinite(np.asarray(best, dtype=float))):
        best = np.ones(N, dtype=float) / N

    best = finalize_sequence(best)
    # One last exact score, but do not change the sequence if the evaluator is
    # temporarily unavailable.
    exact = exact_score_sequence(best)
    if np.isfinite(exact):
        best_score = exact

    print(f"[done] n={len(best)} best_score={best_score:.10f}", flush=True)
    return [float(x) for x in best]
