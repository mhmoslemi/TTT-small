```python
"""
Search for a low-scoring non-negative coefficient sequence.

Strategy:
- Always keep a guaranteed small construction: [0.5, 0, 0.5], which scores 1.5.
- Run a heavy small-n local search over lengths 2..12 to try to improve further.
- If time remains, run larger-sequence local search using FFT convolution,
  projected subgradient/contrast moves, random smooth perturbations, and
  low-frequency Fourier restarts.
- Return a verified candidate only if it beats the guaranteed baseline under
  evaluate_sequence(); otherwise return the baseline.
"""

import numpy as np
import time
import math


def normalize_sum1(a):
    """Normalize a non-negative 1D sequence to sum 1, clipping safely."""
    try:
        a = np.asarray(a, dtype=np.float64)
    except Exception:
        return None
    if a.ndim != 1 or a.size == 0:
        return None
    if not np.all(np.isfinite(a)):
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-18:
        return None
    a = a / s
    m = float(np.max(a))
    if m > 1000.0:
        a = a / m
        a = np.clip(a, 0.0, 1000.0)
        s2 = float(np.sum(a))
        if s2 <= 1e-18:
            return None
        a = a / s2
    if not np.all(np.isfinite(a)):
        return None
    return a


def score_dense_C(a):
    """Exact dense score for small arrays. Returns score, convolution, argmax."""
    a = normalize_sum1(a)
    if a is None:
        return np.inf, None, 0
    n = a.size
    if n < 1:
        return np.inf, None, 0
    c = np.convolve(a, a)
    c = np.maximum(c, 0.0)
    s = float(np.sum(a))
    if s <= 0.0:
        return np.inf, c, 0
    mx = float(np.max(c))
    k = int(np.argmax(c))
    sc = float(2.0 * n * mx / (s * s))
    return sc, c, k


def score_ft_C(a):
    """Fast FFT-based score and full convolution for larger arrays."""
    a = normalize_sum1(a)
    if a is None:
        return np.inf, None, 0
    n = a.size
    m = 2 * n - 1
    if m <= 512:
        c = np.convolve(a, a)
    else:
        N = 1 << (m - 1).bit_length()
        F = np.fft.rfft(a, N)
        c = np.fft.irfft(F * F, N)[:m]
        c = np.maximum(c, 0.0)
    c = np.maximum(c, 0.0)
    s = float(np.sum(a))
    if s <= 0.0:
        return np.inf, c, 0
    mx = float(np.max(c))
    k = int(np.argmax(c))
    sc = float(2.0 * n * mx / (s * s))
    if not np.isfinite(sc):
        return np.inf, c, k
    return sc, c, k


def active_top(C, R, rng=None, mode=0):
    """Choose top-R lags and soft weights over them."""
    L = int(C.size)
    R = int(R)
    if R <= 0 or L <= 0:
        return np.empty(0, dtype=int), np.empty(0, dtype=float)
    R = min(R, L)

    if rng is not None and mode == 1:
        sample = min(L, max(20, L // 8))
        idx = np.unique(rng.choice(L, size=sample, replace=False))
        if idx.size >= R:
            idx = idx[np.argsort(C[idx])][-R:]
        else:
            idx = np.argsort(C)[-R:]
        idx = idx.astype(int)
    else:
        if R == L:
            idx = np.arange(L, dtype=int)
        else:
            idx = np.argpartition(C, L - R)[L - R:].astype(int)

    vals = C[idx]
    vmin = float(np.min(vals))
    vmax = float(np.max(vals))
    if vmax - vmin <= 1e-18:
        w = np.ones(R, dtype=np.float64)
    else:
        t = (vals - vmin) / (vmax - vmin)
        w = (t * t * t + 0.02).astype(np.float64)
    ws = float(np.sum(w))
    if ws <= 0.0:
        w = np.ones(R, dtype=np.float64)
    else:
        w = w / ws
    return idx, w


def direction_weighted(a, C, S, idx, w):
    """
    Negative subgradient direction for a weighted sum of active lags.
    For one active lag k, decreasing direction is proportional to
    C[k]/S - a[k-i] on valid indices, plus constant elsewhere.
    """
    n = a.size
    d = np.zeros(n, dtype=np.float64)
    if idx.size == 0:
        return d
    if S <= 0.0:
        S = 1.0
    for t in range(idx.size):
        k = int(idx[t])
        wt = float(w[t])
        val = float(C[k]) / S
        d += val * wt
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            j = k - np.arange(lo, hi + 1, dtype=int)
            d[lo:hi + 1] -= wt * a[j]
    d -= float(np.mean(d))
    return d


def direction_contrast(a, idx, w):
    """
    Contrast-enhancement direction for active lags:
    for each pair (i, k-i), add a[i]-a[k-i] to i, reducing their product
    for fixed pair-sum in expectation.
    """
    n = a.size
    d = np.zeros(n, dtype=np.float64)
    for t in range(idx.size):
        k = int(idx[t])
        wt = float(w[t])
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[lo:hi + 1]
            j = k - np.arange(lo, hi + 1, dtype=int)
            d[lo:hi + 1] += wt * (seg - a[j])
    d -= float(np.mean(d))
    return d


def make_candidate_dir(a, d, alpha):
    """Apply a directional move, clip, and renormalize to sum 1."""
    c = a + alpha * d
    c = np.clip(c, 0.0, 1000.0)
    s = float(np.sum(c))
    if s <= 1e-18:
        return None
    c = c / s
    return c


def random_smooth_perturb(a, rng, strength):
    """Add a bounded low-frequency cosine perturbation."""
    n = a.size
    if n <= 1:
        c = a + rng.normal(0.0, strength)
        return normalize_sum1(c)
    num = int(rng.integers(3, 12))
    maxfreq = max(2, int(math.ceil(num * 2.0)))
    freqs = rng.integers(1, maxfreq, size=num)
    amps = rng.normal(0.0, strength, size=num)
    phases = rng.uniform(0.0, 2.0 * math.pi, size=num)
    i = np.arange(n, dtype=np.float64)
    noise = np.zeros(n, dtype=np.float64)
    for m in range(num):
        ang = (2.0 * math.pi * float(freqs[m]) / float(n)) * i + float(phases[m])
        noise += float(amps[m]) * np.cos(ang)
    c = a + noise
    return normalize_sum1(c)


def upsample_seq(a, n):
    """Linearly resample a normalized sequence to length n."""
    a = normalize_sum1(a)
    if a is None:
        return None
    if a.size == 0:
        return None
    if a.size == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size, endpoint=True)
    x_new = np.linspace(0.0, 1.0, n, endpoint=True)
    c = np.interp(x_new, x_old, a)
    return normalize_sum1(c)


def random_fourier_init(n, rng, M=6, amp_max=0.6):
    """Random low-frequency non-negative-ish Fourier initialization."""
    if n <= 0:
        return None
    if n == 1:
        return np.ones(1, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    scale = amp_max / math.sqrt(max(1, M))
    for m in range(1, M + 1):
        amp = scale * float(rng.uniform(0.25, 1.0))
        phase = float(rng.uniform(0.0, 2.0 * math.pi))
        a += amp * np.cos((2.0 * math.pi * float(m) / float(n)) * i + phase)
    return normalize_sum1(a)


def init_sine_seq(n, rng=None):
    """Smooth arc-shaped initialization."""
    if n <= 0:
        return None
    if n == 1:
        return np.ones(1, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    c = (1.0 - np.cos(2.0 * math.pi * i / float(max(2, n + 1)))) * 0.5
    if rng is not None:
        meanv = float(np.mean(c))
        c = c + rng.normal(0.0, max(1e-8, 0.1 * meanv), n)
    return normalize_sum1(c)


def init_ramp(n, rng=None):
    """Monotone ramp initialization."""
    if n <= 0:
        return None
    if n == 1:
        return np.ones(1, dtype=np.float64)
    a = np.linspace(0.1, 1.0, n, dtype=np.float64)
    if rng is not None:
        if rng.random() < 0.5:
            a = a[::-1]
        meanv = float(np.mean(a))
        a = a + rng.normal(0.0, max(1e-8, 0.05 * meanv), n)
    return normalize_sum1(a)


def init_small(n, rng, counter):
    """Diverse small-n initializers."""
    if n <= 0:
        return None
    if n == 3 and counter == 0:
        return np.array([0.5, 0.0, 0.5], dtype=np.float64)

    mode = int(counter) % 6
    if mode == 0:
        a = np.ones(n, dtype=np.float64)
        a += rng.normal(0.0, 0.2, n)
    elif mode == 1:
        a = rng.dirichlet(np.full(n, 0.5, dtype=np.float64))
    elif mode == 2:
        a = rng.dirichlet(np.full(n, 2.0, dtype=np.float64))
    elif mode == 3:
        a = np.zeros(n, dtype=np.float64)
        a[int(rng.integers(0, n))] += 0.5
        if n > 1:
            a[int(rng.integers(0, n))] += 0.5
    elif mode == 4:
        a = np.zeros(n, dtype=np.float64)
        for _ in range(3):
            a[int(rng.integers(0, n))] += 1.0 / 3.0
    else:
        a = np.linspace(0.1, 1.0, n, dtype=np.float64)
        if rng.random() < 0.5:
            a = a[::-1]

    return normalize_sum1(a)


def small_local_search(a, rng, max_iters):
    """Dense exact local search for small n."""
    a = normalize_sum1(a)
    if a is None:
        a = np.ones(3, dtype=np.float64) / 3.0

    st, C, k = score_dense_C(a)
    if not np.isfinite(st):
        return a, np.inf

    no_improve = 0
    for _ in range(int(max_iters)):
        idx = np.array([k], dtype=int)
        w = np.array([1.0], dtype=np.float64)
        improved = False

        d = direction_weighted(a, C, 1.0, idx, w)
        if float(np.max(np.abs(d))) > 1e-15:
            for alpha in (0.1, 0.3):
                c = make_candidate_dir(a, d, alpha)
                if c is None:
                    continue
                sc2, C2, k2 = score_dense_C(c)
                if sc2 < st:
                    a = c
                    st = sc2
                    C = C2
                    k = k2
                    improved = True
                    break

        if not improved:
            d = direction_contrast(a, idx, w)
            if float(np.max(np.abs(d))) > 1e-15:
                for alpha in (0.05, 0.2):
                    c = make_candidate_dir(a, d, alpha)
                    if c is None:
                        continue
                    sc2, C2, k2 = score_dense_C(c)
                    if sc2 < st:
                        a = c
                        st = sc2
                        C = C2
                        k = k2
                        improved = True
                        break

        if improved:
            no_improve = 0
        else:
            meanv = float(np.mean(a))
            c = random_smooth_perturb(a, rng, max(1e-8, 0.35 * meanv))
            if c is not None:
                sc2, C2, k2 = score_dense_C(c)
                if sc2 < st:
                    a = c
                    st = sc2
                    C = C2
                    k = k2
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
            if no_improve > 250:
                break

    return a, st


def small_search_phase(start, end_time, best_a, best_sc, rng):
    """Search small lengths with equal slice time per length."""
    phase_start = time.monotonic()
    if end_time <= phase_start:
        return best_a, best_sc

    ns = [2, 3, 4, 5, 6, 7, 8, 9, 10, 12]
    total_span = max(1e-9, end_time - phase_start)

    for idx, n in enumerate(ns):
        n_end = phase_start + total_span * (idx + 1) / float(len(ns))
        n_end = min(n_end, end_time)

        restart = 0
        while time.monotonic() < n_end and restart < 100:
            a = init_small(n, rng, restart)
            if a is None:
                restart += 1
                continue
            if n <= 5:
                max_iters = 1000
            elif n <= 8:
                max_iters = 800
            else:
                max_iters = 600

            a2, sc2 = small_local_search(a, rng, max_iters)
            if np.isfinite(sc2) and sc2 < best_sc:
                best_sc = float(sc2)
                best_a = np.array(a2, dtype=np.float64).copy()
            restart += 1

    return best_a, best_sc


def large_local_search(a, rng, deadline, current_best_sc):
    """FFT/local subgradient search for one fixed larger length."""
    a = normalize_sum1(a)
    if a is None:
        return None, np.inf

    st, C, k = score_ft_C(a)
    if not np.isfinite(st):
        return a, st

    local_best = a.copy()
    local_best_st = st
    base_alpha = 0.35
    no_improve = 0
    restarts = 0
    max_no_improve = 3000
    R = 8 if a.size <= 2048 else 14

    while time.monotonic() < deadline:
        if no_improve > max_no_improve:
            restarts += 1
            if restarts > 250:
                break
            n = a.size
            mode = int(rng.integers(0, 4))
            if mode == 0:
                meanv = max(1e-8, float(np.mean(a)))
                c = random_smooth_perturb(a, rng, meanv * rng.uniform(0.5, 1.5))
            elif mode == 1:
                c = random_fourier_init(n, rng, M=6, amp_max=0.6)
            elif mode == 2:
                c = init_sine_seq(n, rng)
            else:
                c = init_ramp(n, rng)

            if c is not None:
                sc2, C2, k2 = score_ft_C(c)
                if np.isfinite(sc2):
                    a = c
                    C = C2
                    k = k2
                    st = sc2
                    if st < local_best_st:
                        local_best = a.copy()
                        local_best_st = st
            no_improve = 0
            base_alpha = 0.35
            continue

        move = int(rng.integers(0, 10))
        idx, w = active_top(C, R, rng, mode=0)
        improved = False

        if move < 4:
            d = direction_weighted(a, C, 1.0, idx, w)
            if float(np.max(np.abs(d))) > 1e-15:
                for alpha in (0.08, base_alpha):
                    c = make_candidate_dir(a, d, alpha)
                    if c is None:
                        continue
                    sc2, C2, k2 = score_ft_C(c)
                    if sc2 < st:
                        a = c
                        st = sc2
                        C = C2
                        k = k2
                        improved = True
                        if st < local_best_st:
                            local_best = a.copy()
                            local_best_st = st
                        break

        elif move < 7:
            d = direction_contrast(a, idx, w)
            if float(np.max(np.abs(d))) > 1e-15:
                for alpha in (0.03, 0.12):
                    c = make_candidate_dir(a, d, alpha)
                    if c is None:
                        continue
                    sc2, C2, k2 = score_ft_C(c)
                    if sc2 < st:
                        a = c
                        st = sc2
                        C = C2
                        k = k2
                        improved = True
                        if st < local_best_st:
                            local_best = a.copy()
                            local_best_st = st
                        break

        else:
            meanv = max(1e-8, float(np.mean(a)))
            c = random_smooth_perturb(a, rng, meanv * rng.uniform(0.2, 1.2))
            if c is not None:
                sc2, C2, k2 = score_ft_C(c)
                if sc2 < st:
                    a = c
                    st = sc2
                    C = C2
                    k = k2
                    improved = True
                    if st < local_best_st:
                        local_best = a.copy()
                        local_best_st = st

        if improved:
            no_improve = 0
            base_alpha = min(1.0, base_alpha * 1.02)
        else:
            no_improve += 1
            base_alpha = max(0.03, base_alpha * 0.97)

    return local_best, local_best_st


def large_search_phase(global_deadline, remaining, best_a, best_sc, rng, parent_n):
    """Try several longer lengths, respecting the global deadline."""
    global_deadline = max(time.monotonic() + 1.0, float(global_deadline))
    now = time.monotonic()
    usable = max(0.0, min(float(remaining), global_deadline - now) - 40.0)
    if usable <= 0.0:
        return best_a, best_sc

    lengths = {128, 256, 512, 1024, 2048, 4096}
    if parent_n is not None:
        pn = int(parent_n)
        if 64 <= pn <= 8000:
            lengths.add(pn)

    lengths = sorted(lengths, reverse=True)
    if not lengths:
        return best_a, best_sc

    per_length = max(20.0, usable / float(len(lengths)))

    for n in lengths:
        now = time.monotonic()
        if now >= global_deadline - 15.0:
            break
        slice_end = min(global_deadline - 15.0, now + per_length)
        if slice_end <= now + 5.0:
            continue

        inits = []
        c = upsample_seq(best_a, n)
        if c is not None:
            inits.append(c)
        c = random_fourier_init(n, rng)
        if c is not None:
            inits.append(c)
        c = init_sine_seq(n, rng)
        if c is not None:
            inits.append(c)

        if not inits:
            continue

        for j, init in enumerate(inits):
            now = time.monotonic()
            if now >= global_deadline - 15.0 or now >= slice_end:
                break
            rem_slice = slice_end - now
            if rem_slice <= 5.0:
                break
            give = min(rem_slice, max(10.0, rem_slice / float(len(inits) - j)))
            init_deadline = time.monotonic() + give
            a2, sc2 = large_local_search(init, rng, init_deadline, best_sc)
            if a2 is not None and np.isfinite(sc2) and sc2 < best_sc:
                best_sc = float(sc2)
                best_a = np.array(a2, dtype=np.float64).copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1500.0, **kwargs):
    """
    Main entry point. Returns a list of non-negative floats.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # Guaranteed construction: 1.5.
    best_a = np.array([0.5, 0.0, 0.5], dtype=np.float64)
    best_sc = 1.5

    try:
        bsc = float(evaluate_sequence([0.5, 0.0, 0.5]))
        if np.isfinite(bsc):
            best_sc = bsc
    except Exception:
        pass

    # Small-n exact search can easily find or confirm the 1.5 construction,
    # and may find better short sequences.
    small_budget = min(240.0, max(20.0, float(budget_s) * 0.25))
    small_end = min(deadline - 120.0, start + small_budget)
    if small_end > start:
        try:
            best_a, best_sc = small_search_phase(start, small_end, best_a, best_sc, rng)
        except Exception:
            pass

    # Optional larger search if enough time remains.
    remaining = deadline - time.monotonic()
    if remaining > 180.0:
        parent_n = 0
        try:
            p = np.asarray(height_sequence_1, dtype=np.float64)
            if p.size:
                parent_n = int(p.size)
        except Exception:
            parent_n = 0
        try:
            best_a, best_sc = large_search_phase(deadline, remaining, best_a, best_sc, rng, parent_n)
        except Exception:
            pass

    # Final verification against the exact evaluator.
    try:
        cand = normalize_sum1(best_a)
        if cand is None:
            return [0.5, 0.0, 0.5]
        cand_list = [float(x) for x in cand]
        cand_sc = float(evaluate_sequence(cand_list))

        baseline = [0.5, 0.0, 0.5]
        base_sc = 1.5
        try:
            bsc = float(evaluate_sequence(baseline))
            if np.isfinite(bsc):
                base_sc = bsc
        except Exception:
            pass

        if np.isfinite(cand_sc) and cand_sc < base_sc:
            return cand_list
        return baseline
    except Exception:
        return [0.5, 0.0, 0.5]
```
