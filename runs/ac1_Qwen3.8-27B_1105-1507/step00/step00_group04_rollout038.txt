def solve_lp(a, M):

    # Build constraint matrix: (A_ub @ b) <= M
    # (a * b)_k = sum_i a_i * b_{k-i} for valid i, k-i
    # For each k (0 to 2n-2), the constraint involves b_0 through b_{n-1}
    
    # A_ub is (2n-1) x n
    # A_ub[k, j] = coefficient of b_j in (a*b)_k
    # (a*b)_k = sum_j a_{k-j} * b_j (where a_{k-j} is 0 if k-j out of range)

    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a[idx]
    
    c = -np.ones(n)  # minimize -sum b = max sum b
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')

So the derivative at $t=0$ for component $k$ is $2(a*b)_k - 2(a*a)_k$.

For indices $k^*$ where $(a*a)_{k^*} = M$: derivative is $2(a*b)_{k^*} - 2M \leq 2M - 2M = 0$.

So the directional derivative is $\leq 0$ at the peak points, meaning $f(t) \leq M$ for small $t$ (at least non-increasing).

    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        A[k, js:je] = a[k - je + 1:k - js + 1][::-1]
    return A

def score_of(a):
    """Compute the evaluation score without calling evaluate_sequence."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2 * n * np.max(c) / (S * S))

def lp_solve(a):
    """Solve LP: max sum(b) s.t. (a*b)_k <= max(a*a), b >= 0."""
    n = len(a)
    S = np.sum(a)
    a_norm = a / S
    c_conv = np.convolve(a_norm, a_norm)
    M = float(np.max(c_conv))

    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_vec, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None

def build_convolution_matrix(a):
    """Build (2n-1) x n matrix A where A @ b = np.convolve(a, b)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        A[k, js:je] = a[k - je + 1:k - js + 1][::-1]
    return A

def solve_lp(a):
    """Solve LP: max sum(b) s.t. (a*b)_k <= max(a*a), b >= 0.
    Returns the optimal b vector or None on failure."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return None
    a_norm = a / S
    c_conv = np.convolve(a_norm, a_norm)
    M = float(np.max(c_conv))
    
    A = build_convolution_matrix(a_norm)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

                  bounds=[(0, None)] * n, method='highs')
