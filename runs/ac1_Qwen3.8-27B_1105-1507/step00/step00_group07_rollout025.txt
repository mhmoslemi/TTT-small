import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import sys

def compute_score(a):
    """Compute the score for a sequence (assumes a is a numpy array)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def multiplicative_update(a, eta, n_iter=1000):
    """Apply multiplicative weight updates to minimize max autocorrelation."""
    a = a.copy().astype(np.float64)
    sum_a = np.sum(a)
    a = a / sum_a  # normalize to sum = 1
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iter):
        conv = np.convolve(a, a)
        k_star = np.argmax(conv)
        M = conv[k_star]
        
        # Gradient of b_{k*} with respect to a_i
        # grad[i] = 2 * a[k_star - i] (with a[j] = 0 for j outside [0, n-1])
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Multiplicative update
        a = a * np.exp(-eta * grad)
        
        # Project onto {a >= 0, sum(a) = 1}
        a = np.clip(a, 0, None)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

def lp_update(a, M):
    """Solve the LP to find b that maximizes sum(b) subject to conv(a,b) <= M, b >= 0."""
    n = len(a)
    # conv(a,b)_k = sum_j a[k-j] * b[j] for valid j
    # We want: for each k in [0, 2n-2], sum_j A[k,j] * b[j] <= M
    # where A[k,j] = a[k-j] if 0 <= k-j <= n-1, else 0
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Maximize sum(b) = -minimize -sum(b)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), 
                  bounds=[(0, None)] * n, method='highs', options={'time_limit': 60})
    
    if res.status == 0:
        return res.x
    return None

def lp_iteration(a, n_lp_iters=10, t=0.5):
    """Run LP-based iterations."""
    a = a.copy().astype(np.float64)
    sum_a = np.sum(a)
    a = a / sum_a
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_lp_iters):
        conv = np.convolve(a, a)
        M = np.max(conv)
        
        b = lp_update(a, M)
        if b is None:
            break
        
        sum_b = np.sum(b)
        if sum_b < 1e-10:
            break
        b = b / sum_b
        
        # Update
        a_new = (1 - t) * a + t * b
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

"""
Algorithm: Multi-strategy optimization for non-negative sequences minimizing
score = 2n * max(convolve(a,a)) / (sum(a))^2.

Uses multiplicative weight updates, LP-based best-response, and random restarts
across multiple sequence lengths to find low-scoring sequences.
"""

import numpy as np
import time
import math

def _compute_score(a):
    """Compute score using numpy (fast, for internal optimization)."""
    n = a.shape[0]
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))

def _normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / a.shape[0]
    return a / s

def _mul_update(a, eta, n_peak_lags=1):
    """One step of multiplicative weight update.
    
    Uses the top n_peak_lags lags to compute a combined gradient.
    Returns updated a (normalized).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    
    # Get top peak lags
    if n_peak_lags == 1:
        k_star = int(np.argmax(conv))
        lags = [k_star]
    else:
        top_idx = np.argsort(conv)[::-1][:n_peak_lags]
        lags = [int(k) for k in top_idx]
    
    # Combined gradient
    idx = np.arange(n)
    grad = np.zeros(n)
    for k in lags:
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
    
    # Multiplicative update
    a_new = a * np.exp(-eta * grad)
    return _normalize(a_new)

def _mul_optimize(a0, n_iters, eta_start, eta_end, n_peak_lags=1, seed=0, perturb_every=200, perturb_prob=0.2):
    """Run multiplicative weight optimization with optional random perturbations."""
    rng = np.random.default_rng(seed)
    a = _normalize(a0.copy().astype(np.float64))
    n = a.shape[0]
    best_a = a.copy()
    best_score = _compute_score(a)
    
    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        eta = eta_start * (eta_end / eta_start) ** t
        
        a = _mul_update(a, eta, n_peak_lags)
        
        # Random perturbation
        if it > 0 and it % perturb_every == perturb_every - 1:
            if rng.random() < perturb_prob:
                n_perturb = max(1, n // 10)
                idx = rng.choice(n, size=n_perturb, replace=False)
                a[idx] *= np.exp(rng.normal(0, 0.5, n_perturb))
                a = _normalize(a)
        
        score = _compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

def _lp_solve(a, M, timeout=30):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    
    n = a.shape[0]
    n_con = 2 * n - 1
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(n_con):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    
    A = coo_matrix((vals, (rows, cols)), shape=(n_con, n))
    c = -np.ones(n)
    b_ub = np.full(n_con, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'time_limit': timeout})
    
    if res.status == 0:
        return res.x
    return None

def _lp_optimize(a0, n_iters=20, t_blend=0.5, timeout_per_lp=30, seed=0):
    """Run LP-based best-response optimization."""
    rng = np.random.default_rng(seed)
    a = _normalize(a0.copy().astype(np.float64))
    n = a.shape[0]
    best_a = a.copy()
    best_score = _compute_score(a)
    
    for it in range(n_iters):
        conv = np.convolve(a, a)
        M = float(conv.max())
        
        b = _lp_solve(a, M, timeout=timeout_per_lp)
        if b is None:
            break
        
        b = _normalize(b)
        
        # Blend
        t = t_blend * (0.5 ** (it // 5))  # Decreasing blend
        a_new = (1 - t) * a + t * b
        a_new = _normalize(a_new)
        
        score = _compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

Actually, I just realized something. Let me re-examine the multiplicative update more carefully.

Starting from a random vector a (normalized to sum=1):
- k* = argmax of conv(a,a), typically near n-1.
- grad[i] = 2 * a[k*-i]
- Update: a_i ← a_i * exp(-η * 2 * a[k*-i])

For a random vector, a[k*-i] ≈ 1/n on average. So the update is approximately:
a_i ← a_i * exp(-2η/n)

This is the same for all i (approximately), so after normalization, a is approximately unchanged! The update is very weak for random vectors.

To make the update more effective, I should use a larger η or a different gradient normalization.

Alternative: Use the "relative" gradient: grad[i] / a[i] = 2 * a[k*-i] / a[i].
Then the update is: a_i ← a_i * exp(-η * 2 * a[k*-i] / a[i]).

This is more aggressive: if a[i] is small and a[k*-i] is large, the update will strongly decrease a[i]. If a[i] is large and a[k*-i] is small, the update will barely change a[i].
