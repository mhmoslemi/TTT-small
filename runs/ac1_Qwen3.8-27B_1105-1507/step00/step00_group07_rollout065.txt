def fft_self_conv(a, L):
    """Compute linear self-convolution of a using FFT (a zero-padded to length L)."""
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)


def smooth_max_grad(a, b, n, L, top_k):
    """Compute gradient of top-k weighted smooth max of b."""
    b_valid = b[:2 * n - 1]
    k = min(top_k, len(b_valid))
    top_indices = np.argpartition(b_valid, -k)[-k:]
    top_values = b_valid[top_indices]
    shifted = top_values - top_values.max()
    weights = np.exp(8.0 * shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    arange_n = np.arange(n)
    for i in range(k):
        if weights[i] < 1e-8:
            continue
        k_idx = int(top_indices[i])
        idx = (k_idx - arange_n) % L
        valid = idx < n
        grad[valid] += weights[i] * 2.0 * a[idx[valid]]
    return grad

def block_perturb(base, rng):
    """Return a nonnegative block perturbation of base."""
    n = base.size
    width = int(rng.integers(max(2, n // 50), max(3, n // 8)))
    start = int(rng.integers(0, max(1, n - width + 1)))

    candidate[start:stop] *= np.exp(rng.normal(0.0, 0.15))
    if rng.random() < 0.3:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)
