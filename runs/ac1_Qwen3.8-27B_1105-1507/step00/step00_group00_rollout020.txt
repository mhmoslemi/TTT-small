def _compute_score(p):

    s = p.sum()

    c = np.convolve(pn, pn, mode='full')

def _fft_convolve(p):
    """Compute convolution of p with itself using FFT."""
    n = len(p)
    size = 2 * n - 1
    P = np.fft.rfft(p, size)
    c = np.fft.irfft(P * P, size)
    return c  # length = size = 2n-1

def _refine(p, deadline, max_iters=50000):
    """Refine sequence p to minimize the score."""
    n = len(p)
    p = p.copy().astype(np.float64)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        p = np.ones(n) / n
    else:
        p = p / s
    
    size = 2 * n - 1
    P = np.fft.rfft(p, size)
    
    best_p = p.copy()
    best_score = 2 * n * np.max(np.fft.irfft(P * P, size))
    
    lr = 0.5
    beta = 20.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution
        c = np.fft.irfft(P * P, size)
        c = np.maximum(c, 0)  # numerical safety
        c_max = c.max()
        current_score = 2 * n * c_max
        
        if current_score < best_score:
            best_score = current_score
            best_p = p.copy()
        
        # Compute soft-max weights
        # Use shifted c to avoid overflow
        weights = np.exp(beta * (c - c_max))
        w_sum = weights.sum()
        w = weights / w_sum
        
        # Compute gradient (pressure)
        # grad[i] = sum_k w[k] * p[k-i] (cross-correlation of w and p)
        # We can compute this via FFT:
        # cross_corr = ifft(conj(fft(w)) * fft(p))
        # But we need to be careful about indexing.
        # Actually, let's use the direct formula for the top lags.
        
        # Get top lags (where w is significant)
        # For efficiency, use the top 100 lags or so
        n_top = min(200, size)
        top_indices = np.argpartition(w, -n_top)[-n_top:]
        
        pressure = np.zeros(n)
        for k in top_indices:
            wk = w[k]
            if wk < 1e-10:
                continue
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            pressure[lo:hi+1] += wk * p[lo:hi+1][::-1]
        
        # Multiplicative update
        p_new = p * np.exp(-lr * pressure / (n * w_sum))
        p_new = np.clip(p_new, 1e-15, None)
        p_new_sum = p_new.sum()
        if p_new_sum < 1e-10:
            break
        p_new = p_new / p_new_sum
        
        # Check if the update is too aggressive
        p = p_new
        P = np.fft.rfft(p, size)
        
        # Adapt learning rate
        if current_score > best_score * 1.01:
            lr *= 0.95
        elif it % 100 == 0 and current_score == best_score:
            lr *= 1.05
    
    return best_p, best_score

def _make_init(n, init_type):
    if init_type == 'uniform':
        p = np.ones(n)
    elif init_type == 'ramp':
        p = np.arange(1, n+1, dtype=float)
    elif init_type == 'u_shaped':
        i = np.arange(n, dtype=float)
        t = 2 * i / max(1, n-1) - 1
        p = 1.0 / (1.0 + 3.0 * t**2)
    elif init_type == 'bimodal':
        i = np.arange(n, dtype=float)
        p = np.exp(-((i - 0.15*n)**2) / (0.08*n)**2) + np.exp(-((i - 0.85*n)**2) / (0.08*n)**2) + 1e-3
    elif init_type == 'cosine_taper':
        i = np.arange(n, dtype=float)
        p = np.sqrt(0.5 * (1 + np.cos(np.pi * i / n))) + 1e-3  # half-cosine taper
    elif init_type == 'quadratic_taper':
        i = np.arange(n, dtype=float)
        p = np.sqrt(1.0 - (2*i/(n-1) - 1)**2) + 1e-3
    else:
        p = np.ones(n)
    p = np.maximum(p, 0)
    s = p.sum()
    if s > 0:
        p = p / s
    else:
        p = np.ones(n) / n
    return p

    """Score = 2n * c_max (when sum p = 1)."""
    return 2 * n * c_max

def _lp_step(p, c, n):
    """
    Solve the LP to find the best linearized step.
    Returns (delta, M_target) or (None, None) if LP fails.
    """
    num_c = 2 * n - 1
    
    # Build A matrix: A[k, i] = 2*p[k-i] (if valid)
    A = np.zeros((num_c, n))
    for k in range(num_c):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = 2.0 * p[lo:hi+1][::-1]
    
    # Objective: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints: A @ delta - M <= -c  (i.e., c + A@delta <= M)
    # In form: A_ub @ x <= b_ub, where x = [delta, M]
    A_ub = np.empty((num_c, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -c
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -p_i <= delta_i <= inf, M >= 0
    lower_bounds = -p
    bounds = list(zip(lower_bounds, [None] * n)) + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        delta = result.x[:n]
        M_target = result.x[n]
        return delta, M_target
    return None, None

def _lp_refine(p, deadline, max_iters=300):
    """Refine p using LP-based steps."""
    n = len(p)
    p = p.copy().astype(np.float64)
    n_c = 2 * n - 1
    size = n_c
    
    # Compute initial convolution
    P = np.fft.rfft(p, size)
    c = np.fft.irfft(P * P, size)
    c = np.maximum(c, 0)
    c_max = c.max()
    best_p = p.copy()
    best_score = 2 * n * c_max
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Solve LP
        delta, M_target = _lp_step(p, c, n)
        if delta is None:
            break
        
        # Take the step with line search
        accepted = False
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            p_try = p + alpha * delta
            if (p_try < -1e-10).any():
                continue
            p_try = np.maximum(p_try, 0)
            p_try = p_try / p_try.sum()
            
            # Compute score
            P_try = np.fft.rfft(p_try, size)
            c_try = np.fft.irfft(P_try * P_try, size)
            c_try = np.maximum(c_try, 0)
            score_try = 2 * n * c_try.max()
            
            if score_try < best_score:
                best_p = p_try.copy()
                best_score = score_try
            
            if score_try <= 2 * n * c_max * (1 + 1e-6):
                # Accept the step (score didn't increase too much)
                p = p_try
                c = c_try
                c_max = c.max()
                accepted = True
                break
        
        if not accepted:
            # Try a small random perturbation to escape
            p = p * (1 + 0.01 * np.random.randn(n))
            p = np.maximum(p, 1e-10)
            p = p / p.sum()
        
        # Update
        P = np.fft.rfft(p, size)
        c = np.fft.irfft(P * P, size)
        c = np.maximum(c, 0)
        c_max = c.max()
    
    return best_p, best_score
