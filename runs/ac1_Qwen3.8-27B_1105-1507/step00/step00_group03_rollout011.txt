def _fast_score(a):
    """Compute score without list conversion."""
    n = len(a)
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def _subgradient(a, top_frac=0.05):
    """Compute subgradient of max(convolve(a,a)) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    threshold = max_c * (1.0 - top_frac)
    active = c >= threshold
    n_active = max(1, active.sum())
    
    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * active[i+j] / n_active (where i+j < 2n-1)
    grad = np.zeros(n)
    for i in range(n):
        max_j = min(n, 2 * n - 1 - i)
        if max_j > 0:
            grad[i] = 2.0 * np.dot(a[:max_j], active[i:i+max_j])
    grad /= n_active
    return grad, c, max_c

def _projected_subgradient_step(a, lr, top_frac=0.05):
    """One step of projected subgradient descent (sum normalized)."""
    n = len(a)
    grad, c, max_c = _subgradient(a, top_frac)

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:

Wait, I have a problem with the coordinate descent phase. Computing np.convolve(a, a) for each coordinate i is O(n^2), and doing it for all n coordinates is O(n^3). For n=2535, that's about 16 billion operations per pass. Way too slow.

I need to be smarter. In coordinate descent, when I change a[i] from old_val to new_val, the change in the convolution is:
Δc[k] = (new_val - old_val) * a[k-i] for k in [i, i+n-1] (valid)
Δc[k] = 0 otherwise

So I can update the convolution in O(n) per coordinate change, and the total per pass is O(n^2). That's 6.4M per pass, which is fine.
