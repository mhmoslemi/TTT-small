"""
Step-function search minimizing  score = 2*n*max(convolve(a,a)) / (sum a)^2
over non‑negative a.  The objective is scale invariant so we optimize on the
simplex sum(a)=1 and minimise  max(convolve(a,a)).

Method (deliberately different from the old block‑perturbation search):
  * Several diverse initialisations (constant, ramps, concave/asymmetric,
    the supplied parent, plus random).
  * A smooth‑max (log‑sum‑exp / softmax) projected‑gradient descent in which
    the gradient of  max(conv)  is taken as a softmax‑weighted combination of
    the per‑bin gradients.  Gradients and convolutions are computed with FFTs
    so each iteration is O(n log n).  Steps are accepted only when the *hard*
    max(conv) decreases (a sub‑gradient / submodular‑style rule), with a
    decaying, line‑searched learning rate and momentum.
  * Softmax temperature is annealed from "flat" to "sharp" so we start by
    smoothing the whole convolution and finish by attacking the true peak.
  * A cheap local coordinate polish (exact FFT, hard acceptance).
  * The authoritative evaluate_sequence is called periodically / at the end;
    the best valid list found is always returned before the deadline.
"""

import numpy as np
import time


def _next_pow2(v):
    L = 1
    while L < v:
        L *= 2
    return L


def fft_convolve_square(a):
    """Linear a*a convolution via FFT, length 2n-1."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    C = np.fft.irfft(A * A, L)
    return C[:2 * n - 1]


def fft_cross_corr(p, a):
    """g[j] = sum_m p[j+m] * a[m]  for j = 0..n-1 (linear cross corr)."""
    n = a.size
    L = _next_pow2(p.size + n - 1)
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)
    cc = np.fft.irfft(P * np.conj(A), L)
    return cc[0:n]


def normalize_to_simplex(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.ones_like(a)
        s = a.sum()
    return a / s


def hard_score_internal(a):
    return 2.0 * a.size * fft_convolve_square(a).max()


def build_inits(n, parent):
    inits = []
    inits.append(np.ones(n))
    inits.append(np.arange(1, n + 1, dtype=np.float64))                 # rising ramp
    inits.append(np.arange(n, 0, -1, dtype=np.float64))                 # falling ramp
    # asymmetric concave / convex bump
    t = np.linspace(0.0, 1.0, n)
    inits.append(1.0 + 0.9 * (t - 0.5) ** 2)          # U (edges high)
    inits.append(1.0 + 0.9 * t * (1.0 - t))           # dome (center high)
    inits.append(np.exp(-((t - 0.25) ** 2) / 0.02))   # left bump
    inits.append(np.exp(-((t - 0.75) ** 2) / 0.02))   # right bump
    inits.append(1.0 + 0.6 * np.sin(2 * np.pi * t))
    inits.append(1.0 + 0.6 * np.cos(4 * np.pi * t))
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.size:
            p = np.clip(p, 0.0, 1000.0)
            if p.size == n:
                inits.append(p)
            else:
                inits.append(np.interp(np.linspace(0, 1, n),
                                       np.linspace(0, 1, p.size), p))
    return [normalize_to_simplex(x) for x in inits]

    a = normalize_to_simplex(a.copy())
    C = fft_convolve_square(a)
    best = a.copy()
    best_c = C.max()
    global_best_c = best_c
    rng = np.random.default_rng(12345)

    mom = np.zeros(n)
    lr = 1.0
    plateau = 0

    n = a.size
    two_n = 2.0 * n

        it += 1
        # anneal temperature: start flat -> sharp (approach hard max)
        beta = min(n, 8.0 + 0.25 * it)

        g = fft_cross_corr(p, a)

        a2 = a + mom

        C2 = fft_convolve_square(a2)

        if c2 < cur - 1e-15:

                plateau += 1

            plateau += 1

        if it % 25 == 0:

        # stop this init if clearly stuck

def local_polish(a, deadline, tag):
    """Cheap coordinate / small‑block line search with exact FFT acceptance."""
    n = a.size
    a = normalize_to_simplex(a.copy())
    best = a.copy()
    best_c = fft_convolve_square(a).max()
    rng = np.random.default_rng(777)
    it = 0
    while time.monotonic() < deadline:
        it += 1
        k = int(rng.integers(0, n))
        half = max(1, int(n // 400))
        lo = max(0, k - half)
        hi = min(n, k + half + 1)
        a2 = a
        for mult in (2.0, 4.0, 0.5, 0.25, 8.0):
            trial = a.copy()
            trial[lo:hi] *= mult
            trial = normalize_to_simplex(trial)
            c = fft_convolve_square(trial).max()
            if c < best_c - 1e-15:
                a = trial
                best = trial.copy()
                best_c = c
                a2 = trial
                break
        if it % 200 == 0:
            sc = evaluate_sequence(best.tolist())
            print(f"[polish {tag}] it={it} maxC={best_c:.6e} exact={sc:.6f}")

    return best, best_c

    except Exception:
        parent = None

    if parent is not None and parent.size > 0 and np.all(np.isfinite(parent)):
        n = parent.size
    else:
        n = 4096
        parent = np.ones(n)

    # cap absurd lengths for runtime safety (still large)
    if n > 20000:
        n = 20000

    inits = build_inits(n, parent)
    # add a couple of random initialisations
    inits.append(normalize_to_simplex(rng.random(n) + 0.05))
    inits.append(normalize_to_simplex(
        np.exp(-((np.linspace(0, 1, n) - rng.random()) ** 2) / 0.03)))

    global_best = inits[0].copy()
    global_best_score = evaluate_sequence(global_best.tolist())
    print(f"n={n} initial exact score={global_best_score:.6f}")

    # per‑attempt time budget so we explore several diverse starts
    per_attempt = max(40.0, (float(budget_s) - 30.0) / max(1.0, len(inits)))

    for idx, init in enumerate(inits):
        if time.monotonic() > total_deadline:
            break
        sub_deadline = min(total_deadline - 25.0, time.monotonic() + per_attempt)
        a, c, it, plat = softmax_refine(init, sub_deadline, tag=f"I{idx}")

        # polish the best found for this start
        a, c = local_polish(a,
                            min(total_deadline - 5.0, time.monotonic() + per_attempt * 0.4),
                            tag=f"I{idx}")

        sc = evaluate_sequence(a.tolist())
        print(f"I{idx} refined: internal_maxC={c:.6e} exact_score={sc:.6f}")
        if np.isfinite(sc) and sc < global_best_score:
            global_best_score = sc
            global_best = a.copy()
            print(f"  -> NEW BEST {sc:.6f}")

    # final safety net: always return the best valid sequence found
    a = np.clip(np.asarray(global_best, dtype=np.float64), 0.0, 1000.0)
    out = [float(x) for x in a]
    fin = evaluate_sequence(out)
    print(f"FINAL score={fin:.6f} (internal best {global_best_score:.6f})")
    return out
