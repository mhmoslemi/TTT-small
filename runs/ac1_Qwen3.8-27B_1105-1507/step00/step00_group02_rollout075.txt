def _compute_convolution(f):
    """Compute linear convolution f*f using FFT."""
    n = len(f)
    size = 2 * n - 1
    N = 1
    while N < size:
        N *= 2
    F = np.fft.rfft(f, N)
    conv = np.fft.irfft(F * F, N)[:size]
    return conv


def _compute_smoothed_gradient(f, conv, temperature):
    """
    Compute smoothed gradient of max(conv) w.r.t. f.
    Uses the softmax of conv/temperature as weights.
    """

The self-convolution peak is minimized by spreading convolution energy as flat as possible,
approaching the theoretical lower bound 2n/(2n-1).

import numpy as np

def _compute_convolution(f):
    """Compute linear convolution f*f using FFT."""
    n = len(f)
    size = 2 * n - 1
    N = 1
    while N < size:
        N *= 2
    F = np.fft.rfft(f, N)
    conv = np.fft.irfft(F * F, N)[:size]
    return conv


def _compute_smoothed_gradient(f, conv, temperature):
    """
    Compute smoothed gradient of max(conv) w.r.t. f using softmax weighting.
    Returns gradient array of same length as f.
    """
    n = len(f)
    conv_len = len(conv)  # 2n-1

    # Numerically stable softmax
    scaled = conv / temperature
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    weight = exp_vals / exp_vals.sum()

    # Compute cross-correlation via FFT
    # wr = reverse of weight (length 2n-1)
    wr = weight[::-1]

    # FFT size for linear convolution of wr (len 2n-1) and f (len n)
    conv_full_len = (2 * n - 1) + n - 1  # 3n - 2
    N = 1
    while N < conv_full_len:
        N *= 2

    F_wr = np.fft.rfft(wr, N)
    F_f = np.fft.rfft(f, N)
    conv_full = np.fft.irfft(F_wr * F_f, N)[:conv_full_len]

    # grad[i] = 2 * conv_full[2n-2-i] for i=0,...,n-1
    grad = 2.0 * conv_full[n - 1:2 * n - 1][::-1]

    return grad


def _optimize_one_start(f, rng, deadline, max_iters=200000):

    n = len(f)
    f = np.clip(f, 0.0, None).astype(np.float64)

    temp_init = 1.0

    temperature = temp_init

    stall_threshold = 5000

        conv = _compute_convolution(f)

        # Compute smoothed gradient

        else:
