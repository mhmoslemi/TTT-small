def normalize_nonnegative(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    np.maximum(a, 0.0, out=a)
    s = a.sum()
    if s <= 0.0 or not np.isfinite(s):
       n = a.size if a.size>0 else 1
       return np.ones(n, dtype=np.float64)/float(n)
    a /= s
    return a

"""
Search for non-negative step heights minimizing
2*n*max(convolve(a,a))/sum(a)^2.

The method uses a truncated minimum-phase square-root construction for a
strong initial point, then applies a smoothed-max (log-sum-exp) FFT gradient
flow in softmax/log-domain coordinates with Adam-like updates.  Several
deformed/log-noise starts are explored, and the best object is polished with
multiple annealing schedules.
"""

import numpy as np
import time

_BASE_CACHE = {}
_FFT_CTX_CACHE = {}


def next_power_of_two(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def get_base_sequence(n):
    """Truncated coefficients of (1-x)^(-1/2)."""
    n = int(n)
    if n in _BASE_CACHE:
        return _BASE_CACHE[n].copy()

    a = np.zeros(max(n, 1), dtype=np.float64)
    if n > 0:
        a[0] = 1.0
        for i in range(n - 1):
            a[i + 1] = a[i] * (2.0 * i + 1.0) / (2.0 * (i + 1.0))

    _BASE_CACHE[n] = a
    return a.copy()


def get_fft_context(n):
    """FFT length and phase factor for reversed-sequence FFT relation."""
    n = int(n)
    if n in _FFT_CTX_CACHE:
        return _FFT_CTX_CACHE[n]

    N = next_power_of_two(3 * n - 2)
    M = N // 2 + 1
    k = np.arange(M, dtype=np.float64)
    phase = np.exp((-2.0j * np.pi * float(n - 1) / float(N)) * k)
    ctx = (N, phase)
    _FFT_CTX_CACHE[n] = ctx
    return ctx


def normalize_nonnegative(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    np.maximum(a, 0.0, out=a)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        n = a.size if a.size > 0 else 1
        return np.ones(n, dtype=np.float64) / float(n)
    a /= s
    return a


def softmax_from_b(b, b_range=30.0):
    b = np.asarray(b, dtype=np.float64)
    b = np.where(np.isfinite(b), b, 0.0)
    b -= float(np.mean(b))
    np.clip(b, -float(b_range), float(b_range), out=b)

    shift = float(np.max(b))
    a = np.exp(b - shift)
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return None
    a /= s
    return a


def deformed_base(n, alpha=0.0, beta=0.0, amp=0.0):
    base = get_base_sequence(n)
    a = base.copy()
    x = (np.arange(n, dtype=np.float64) + 1.0) / float(n)

    if alpha != 0.0:
        a *= np.power(x, float(alpha))
    if beta != 0.0:
        a *= np.exp(float(beta) * (x - 0.5))
    if amp != 0.0:
        a *= (1.0 + float(amp) * np.sin(2.0 * np.pi * x))

    return normalize_nonnegative(a)


def mixed_reverse_base(n, eps):
    base = get_base_sequence(n)
    a = base + float(eps) * base[::-1]
    return normalize_nonnegative(a)


def noisy_base(n, sigma, seed):
    base = get_base_sequence(n)
    rng = np.random.default_rng(int(seed))
    a = base * np.exp(rng.normal(0.0, float(sigma), n))
    return normalize_nonnegative(a)


def fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf

    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.inf

    N = next_power_of_two(2 * n - 1)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    cm = float(np.max(c))
    if not np.isfinite(cm):
        return np.inf
    return float(2.0 * float(n) * cm / (s * s))


def prepare_starts(n, parent_arr=None):
    starts = []

    starts.append(deformed_base(n, 0.0, 0.0, 0.0))

    for alpha in (-0.75, -0.5, -0.25, 0.25, 0.5, 0.75):
        starts.append(deformed_base(n, alpha, 0.0, 0.0))

    for beta in (-1.5, -1.0, -0.5, 0.5, 1.0, 1.5):
        starts.append(deformed_base(n, 0.0, beta, 0.0))

    for amp in (-0.2, 0.2):
        starts.append(deformed_base(n, 0.0, 0.0, amp))

    for eps in (0.1, 0.3, 0.6):
        starts.append(mixed_reverse_base(n, eps))

    for sigma, seed in ((0.10, 11), (0.25, 22), (0.50, 33)):
        starts.append(noisy_base(n, sigma, seed))

    starts.append(normalize_nonnegative(np.ones(n, dtype=np.float64)))

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    cosw = 0.25 + 0.75 * (0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    starts.append(normalize_nonnegative(cosw))

    if parent_arr is not None:
        p = np.asarray(parent_arr, dtype=np.float64).ravel()
        if p.size == n:
            starts.append(normalize_nonnegative(p))

    return starts


def optimize_b(a0, deadline, N, phase, verbose=False, lr_start=0.06,
               t_start=5.0, t_end=100.0, b_range=30.0, initial_score=np.inf):
    """Log-domain Adam flow for log-sum-exp smoothed self-convolution max."""
    a0 = normalize_nonnegative(np.asarray(a0, dtype=np.float64))
    n = a0.size
    if n == 0:
        return np.array([], dtype=np.float64), np.inf

    best_a = a0.copy()
    best_score = float(initial_score)
    if not np.isfinite(best_score):
        best_score = fast_score(best_a)

    b = np.log(np.maximum(best_a, 1.0e-300))
    b -= float(np.mean(b))
    np.clip(b, -float(b_range), float(b_range), out=b)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    beta1 = 0.90
    beta2 = 0.999
    eps = 1.0e-12
    beta1_pow = 1.0
    beta2_pow = 1.0

    step = 0
    start_time = time.monotonic()
    total_budget = max(1.0e-6, float(deadline) - start_time)
    last_report = np.inf
    scale = float(2 * n - 1)

    while time.monotonic() < float(deadline):
        a = softmax_from_b(b, b_range)
        if a is None:
            break

        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)

        cmax = float(np.max(c))
        if not np.isfinite(cmax):
            break

        score = 2.0 * float(n) * cmax

        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose and best_score < last_report - 1.0e-6:
                print(f"    opt n={n} step={step} best={best_score:.9f}")
                last_report = best_score

        if step > 100 and (score > 3.0 or score > best_score + 0.75):
            b = np.log(np.maximum(best_a, 1.0e-300))
            b -= float(np.mean(b))
            np.clip(b, -float(b_range), float(b_range), out=b)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            step += 1
            continue

        elapsed = time.monotonic() - start_time
        frac = elapsed / total_budget
        if frac < 0.0:
            frac = 0.0
        elif frac > 1.0:
            frac = 1.0

        T = float(t_start) * (float(t_end) / float(t_start)) ** frac

        c_scaled = c * scale
        max_scaled = cmax * scale
        exps = T * (c_scaled - max_scaled)

        w = np.exp(exps)
        wsum = float(np.sum(w))
        if not np.isfinite(wsum) or wsum <= 0.0:
            w = np.full(2 * n - 1, 1.0 / float(2 * n - 1), dtype=np.float64)
        else:
            w /= wsum

        active = c_scaled >= (max_scaled - 1.0e-7)
        ac = int(np.sum(active))
        if ac > 0:
            w = w + 0.05 * (active.astype(np.float64) / float(ac))
            ws2 = float(np.sum(w))
            if ws2 > 0.0 and np.isfinite(ws2):
                w /= ws2

        W = np.fft.rfft(w, N)
        R = phase * np.conj(A)
        conv = np.fft.irfft(W * R, N)
        raw = 2.0 * conv[n - 1:2 * n - 1]
        g = scale * raw

        mean_g = float(np.dot(a, g))
        if not np.isfinite(mean_g):
            b = np.log(np.maximum(best_a, 1.0e-300))
            b -= float(np.mean(b))
            np.clip(b, -float(b_range), float(b_range), out=b)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            step += 1
            continue

        gb = a * (g - mean_g)
        if not np.isfinite(gb).all():
            break

        m = beta1 * m + (1.0 - beta1) * gb
        v = beta2 * v + (1.0 - beta2) * (gb * gb)

        beta1_pow *= beta1
        beta2_pow *= beta2

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = float(lr_start) / (1.0 + 3.0 * frac)
        b = b - lr * (m_hat / (np.sqrt(v_hat) + eps))

        np.clip(b, -float(b_range), float(b_range), out=b)
        b -= float(np.mean(b))

        step += 1

        if verbose and step % 50000 == 0:
            print(f"    opt n={n} step={step} T={T:.2f} current={score:.9f} best={best_score:.9f}")

    return best_a, best_score


def run_search_for_length(n, deadline, parent_arr=None, verbose=True):
    n = int(n)
    deadline = float(deadline)
    if n <= 0:
        return np.ones(1, dtype=np.float64), np.inf

    starts = prepare_starts(n, parent_arr)
    scored = []
    for idx, a in enumerate(starts):
        a = normalize_nonnegative(a)
        s = fast_score(a)
        if np.isfinite(s):
            scored.append((float(s), idx, a))

    if not scored:
        a = normalize_nonnegative(np.ones(n, dtype=np.float64))
        return a, fast_score(a)

    scored.sort()
    best_a = scored[0][2].copy()
    best_score = scored[0][0]

    if verbose:
        print(f"[len {n}] starts={len(scored)} best initial={best_score:.9f}")

    N, phase = get_fft_context(n)
    start_now = time.monotonic()
    total_budget = max(0.0, deadline - start_now)

    if total_budget < 1.0:
        return best_a, best_score

    weights = []
    for i in range(len(scored)):
        w = float(0.75 ** i)
        if i == 0:
            w *= 2.0
        weights.append(w)

    tw = float(sum(weights))
    if tw <= 0.0:
        tw = 1.0

    for i in range(len(scored)):
        if time.monotonic() >= deadline:
            break

        remaining = deadline - time.monotonic()
        if remaining < 1.0:
            break

        s_init, _idx, a = scored[i]
        alloc = total_budget * weights[i] / tw
        if i == len(scored) - 1:
            alloc = max(alloc, remaining - 0.5)

        if alloc < 1.0:
            alloc = 1.0
        if alloc > remaining - 0.5:
            alloc = remaining - 0.5
        if alloc < 0.5:
            break

        chunk_deadline = time.monotonic() + alloc
        a2, s2 = optimize_b(
            a,
            chunk_deadline,
            N,
            phase,
            verbose=(verbose and i < 3),
            initial_score=s_init
        )

        if np.isfinite(s2) and s2 < best_score:
            best_score = float(s2)
            best_a = a2.copy()
            if verbose:
                print(f"[len {n}] improved to {best_score:.9f} after start rank {i}")

    return best_a, best_score
