```python
def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Bisection on threshold
    lo = v.min() - 1
    hi = v.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > 1:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    return np.maximum(v - theta, 0)
```

```python
def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (sv - 1))[0][-1]
    theta = (sv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth gradient descent with adaptive temperature and backtracking,
combined with LP-based projection steps and multiple initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (sv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (sv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a, b):
    """Fast convolution using FFT."""
    n = len(a)
    m = len(b)
    N = 1
    while N < n + m - 1:
        N *= 2
    fa = np.fft.fft(a, N)
    fb = np.fft.fft(b, N)
    return np.real(np.fft.ifft(fa * fb))[:n + m - 1]

def compute_grad(a, tau):
    """Compute gradient of smooth max of self-convolution."""
    conv = fast_convolve(a, a)
    m = np.max(conv)
    weights = np.exp((conv - m) / tau)
    wsum = np.sum(weights)
    if wsum == 0 or not np.isfinite(wsum):
        # Fallback: uniform weights
        weights = np.ones(len(conv)) / len(conv)
    else:
        weights /= wsum
    grad = 2.0 * np.correlate(weights, a, mode='valid')
    return grad, np.max(conv)

    cols = []

    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                     method='highs', options={'presolve': True})
    
    if result.success:
        return result.x
    return None

def lp_refine(a, budget):
    """Refine sequence using LP iterations."""
    a = a.copy()
    n = len(a)
    deadline = time.monotonic() + budget
    
    best_a = a.copy()
    best_val = np.max(fast_convolve(a, a))
    
    t = 0.2  # blending parameter
    
    while time.monotonic() < deadline:
        conv = fast_convolve(a, a)
        cur_val = np.max(conv)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
        
        g = lp_iteration(a)
        if g is None:
            break
        
        sum_g = np.sum(g)
        if sum_g <= 0:
            break
        
        target_sum = np.sum(a)
        g = g * (target_sum / sum_g)
        
        a = (1 - t) * a + t * g
        a = np.maximum(a, 0)
        a = a / np.sum(a) * target_sum
        
        # Reduce step size over time
        t *= 0.95
    
    return best_a, best_val
