    A = np.fft.rfft(a, fft_size)

Minimizes score = 2*n*max(convolve(a,a))/(sum(a))^2 using the "flat convolution" 
initialization which achieves the theoretical lower bound 2n/(2n-1) for non-negative 
sequences. Optionally refines with gradient descent.

import numpy as np
import time


def flat_convolution_init(n):
    """
    Compute the non-negative sequence whose autocorrelation is as flat as possible.
    For a finite sequence of length n, c[k] = 1/(2n-1) is achieved for k = 0,...,n-1.
    """
    target = 1.0 / (2 * n - 1)
    a = np.zeros(n, dtype=np.float64)
    a[0] = np.sqrt(target)
    for k in range(1, n):
        # c[k] = 2*a[0]*a[k] + sum_{j=1}^{k-1} a[j]*a[k-j] = target
        if k >= 2:
            s = np.dot(a[1:k], a[k-1:0:-1])
        else:
            s = 0.0
        a[k] = (target - s) / (2.0 * a[0])
        # Due to floating point, a[k] might be slightly negative
        if a[k] < 0 and a[k] > -1e-15:
            a[k] = 0.0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total > 0:
        a = a / total
    return a


def fast_score(a):
    """Compute score = 2*n*max(conv)/(sum)^2 efficiently."""
    n = len(a)

    if total < 0.01:

    c = np.convolve(a, a, mode='full')
    return float(2.0 * n * c.max() / (total * total))


def gradient_refine(a, n, time_limit, rng):
    """
    Refine using gradient descent on log-sum-exp approximation of max(conv).
    """
    a = np.array(a, dtype=np.float64).copy()

    total = a.sum()
    if total < 1e-10:
        return a, fast_score(a)
    a = a / total  # normalize to sum=1
    
    best_a = a.copy()
    best_score = fast_score(a)  # = 2*n*max(c) since sum=1

        c = np.convolve(a, a, mode='full')
        cur = 2.0 * n * c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        cs = c - c.max()
        p = np.exp(cs / tau)
        p /= p.sum()

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
