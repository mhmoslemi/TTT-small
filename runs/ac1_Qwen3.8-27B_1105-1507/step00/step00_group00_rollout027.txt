"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Combines FFT-based subgradient descent, LP direction finding, spectral shaping,
and multi-start with different sequence lengths.
"""

import numpy as np
import time


def _fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    needed = len(a) + len(b) - 1
    size = 1 << (needed - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:needed]


def _compute_objective(a):
    """Compute objective value using FFT."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    max_c = np.max(c)
    return 2.0 * n * max_c / (S * S)


def _compute_subgradient(a):
    """Compute subgradient of the objective at a."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return None, np.inf

    c = _fft_convolve(a, a)
    max_c = np.max(c)
    obj = 2.0 * n * max_c / (S * S)

    # Find all indices at maximum (with tolerance)
    tol = 1e-9 * max_c + 1e-15
    max_mask = c >= max_c - tol
    K = np.where(max_mask)[0]

    if len(K) == 0:
        return None, np.inf

    # v_j = (1/|K|) * sum_{k in K} a[k-j] * 1[0 <= k-j < n]
    # = (a_rev * h)[n-1+j] where h[k] = 1/|K| if k in K
    a_rev = a[::-1]
    h = np.zeros(2 * n - 1)
    h[K] = 1.0 / len(K)

    conv = _fft_convolve(a_rev, h)
    v = conv[n - 1:2 * n - 1]

    # Full subgradient of F(a) = 2n * max(c) / S^2
    subgrad = (2.0 * n / (S * S)) * v - (4.0 * n * max_c / (S ** 3))

    return subgrad, obj

            if stall_count > stall_threshold * 3:
                # Random restart perturbation
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                stall_count = 0
                base_lr = 2.0 / (n * (np.max(a) + 1e-10))
            else:
                lr *= 2.0

        # Gradient step with projection
        a_new = a - lr * subgrad
        a_new = np.maximum(a_new, 0.0)

        if np.sum(a_new) < 0.01:
            # Scale back
            a_new = a
        else:
            a = a_new

        if t % 5000 == 0:
            print(f"    SG t={t}: obj={obj:.6f}, best={best_obj:.6f}")

    return best_a, best_obj


def _spectral_shaping(a_init, budget_s, seed=0):
    """GS-like spectral shaping with target based on flat convolution ideal."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_obj = _compute_objective(a)

    # FFT size (power of 2 >= 2n-1)
    size = 1 << (2 * n - 1).bit_length()

    # Target magnitude: for ideal flat c of length 2n-1,
    # |A[k]| would be S/sqrt(2n-1) * |Dirichlet|.
    # Use a softer target: geometric mean of current and flat.
    
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        S = np.sum(a)
        if S < 0.01:
            a = a_init.copy()
            continue

        A = np.fft.rfft(a, size)
        mags = np.abs(A)

        # Target: S/sqrt(size) gives a "white spectrum" target
        # Blend between current and target
        target = S / np.sqrt(size)

        # Soft blending: new_mag = mags^alpha * target^(1-alpha)
        alpha = min(1.0, 0.3 + 0.1 * np.log1p(iteration / 100.0))
        alpha = min(alpha, 0.85)

        # Avoid zero division
        safe_mags = np.maximum(mags, 1e-30)
        new_mags = (safe_mags ** alpha) * (target ** (1.0 - alpha))

        # Preserve DC
        new_mags[0] = S

        # Apply with existing phases
        phases = A / safe_mags
        A_new = phases * new_mags

        a_new = np.fft.irfft(A_new, size)[:n].copy()
        a_new = np.maximum(a_new, 0.0)

        if np.sum(a_new) < 0.01:
            a = a_init.copy()
        else:
            a = a_new

        obj = _compute_objective(a)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

        if iteration % 1000 == 0:
            print(f"    SPEC iter={iteration}: obj={obj:.6f}, best={best_obj:.6f}")

    return best_a, best_obj


def _lp_direction(f, M):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns the optimal g vector.
    """
    from scipy.optimize import linprog
    import scipy.sparse as sp

    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)

    # Convolution constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_{j} f_j * g_{k-j} where 0 <= k-j < n
    # = sum_{j: max(0,k-n+1) <= j <= min(n-1,k)} f_{k-j} * g_j

    # Build constraint matrix A_ub (2n-1 x n)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])

    A_ub = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)

    # Non-negativity: g >= 0 (handled by bounds)
    bounds = [(0, None)] * n

    # Solve
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return result.x
    else:
        return None


def _lp_iteration(f_init, budget_s, seed=0):
    """Iterative LP-based direction finding."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    f = np.asarray(f_init, dtype=np.float64).copy()
    n = len(f)
    best_f = f.copy()
    best_obj = _compute_objective(f)

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        S = np.sum(f)
        if S < 0.01:
            break

        c = _fft_convolve(f, f)
        M = np.max(c)

        # Solve LP for best direction
        g = _lp_direction(f, M)
        if g is None:
            break

        if np.sum(g) < 1e-10:
            break

        # Rescale g to match f's sum
        g = g * (S / np.sum(g))

        # Update: f <- (1-t)*f + t*g with small t
        t = 0.1
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)

        if np.sum(f_new) < 0.01:
            break

        f = f_new
        obj = _compute_objective(f)

        if obj < best_obj:
            best_f = f.copy()
            best_obj = obj

        if iteration % 5 == 0:
            print(f"    LP iter={iteration}: obj={obj:.6f}, best={best_obj:.6f}")

    return best_f, best_obj


def _try_size(n, budget_s, seed=0):
    """Try optimization with a specific sequence length."""
    rng = np.random.default_rng(seed)

    # Multiple initializations
    inits = []

    # Uniform
    inits.append(np.ones(n) / np.sqrt(2.0 * n))

    # Sine window
    j = np.arange(n)
    sine_win = np.sin(np.pi * (j + 0.5) / n)
    inits.append(sine_win / np.sqrt(2.0 * n))

    # Cosine-squared window
    cos_win = np.sin(np.pi * (j + 0.5) / n) ** 2
    inits.append(cos_win / np.sqrt(2.0 * n))

    # Random
    rand_init = rng.random(n)
    inits.append(rand_init / np.sqrt(2.0 * n))

    # "Edge-enhanced" (high at edges, low in middle)
    edge = 1.0 - np.sin(np.pi * (j + 0.5) / n)
    edge = np.maximum(edge, 0.01)
    inits.append(edge / np.sqrt(2.0 * n))

    # Split: high in first half, low in second (or vice versa)
    split = np.ones(n)
    split[n // 2:] *= 0.5
    inits.append(split / np.sqrt(2.0 * n))

    best_overall = None
    best_score = np.inf

    time_per_init = budget_s / len(inits)

    for i, init in enumerate(inits):
        if time.monotonic() + 2 > time.monotonic() + budget_s:
            break
        sub_budget = min(time_per_init, budget_s - 1.0)
        if sub_budget < 5:
            break

        # Subgradient descent
        result, score = _subgradient_descent(init, sub_budget * 0.6, seed=seed + i)
        if score < best_score:
            best_score = score
            best_overall = result.copy()

        # Spectral shaping
        result2, score2 = _spectral_shaping(result, sub_budget * 0.4, seed=seed + 100 + i)
        if score2 < best_score:
            best_score = score2
            best_overall = result2.copy()

        print(f"    Size {n}, init {i}: best={best_score:.6f}")

    return best_overall, best_score


def _polish(a, budget_s, seed=0):
    """Final polish with very fine subgradient steps."""
    deadline = time.monotonic() + budget_s
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_obj = _compute_objective(a)

    max_a = np.max(a) + 1e-10
    base_lr = 0.5 / (n * max_a)
    t = 0

    while time.monotonic() < deadline:
        t += 1
        subgrad, obj = _compute_subgradient(a)
        if subgrad is None:
            break

        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

        lr = base_lr / (1.0 + 0.01 * t)
        a_new = a - lr * subgrad
        a_new = np.maximum(a_new, 0.0)

        if np.sum(a_new) < 0.01:
            a_new = a
        else:
            a = a_new

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-strategy optimizer: combines FFT subgradient descent, spectral shaping,
    LP direction finding, and multi-start with different sizes.
    Returns the best sequence found within the time budget.
    """
    global deadline_global
    deadline_global = time.monotonic() + budget_s - 5.0

    print(f"Starting optimization with budget {budget_s}s")
    print(f"Deadline set to {budget_s - 5}s from now")

    # Initial sequence from the global variable
    init_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    init_n = len(init_seq)
    print(f"Initial sequence: n={init_n}")

    # Get initial score
    init_score = _compute_objective(init_seq)
    print(f"Initial FFT objective: {init_score:.6f}")

    best_a = init_seq.copy()
    best_score = init_score

    # === Phase 1: Subgradient descent from initial (300s) ===
    print("\n=== Phase 1: Subgradient descent from initial ===")
    t0 = time.monotonic()
    result, score = _subgradient_descent(init_seq, 300.0, seed=seed)
    if score < best_score:
        best_a = result.copy()
        best_score = score
        print(f"  Improved to {best_score:.6f}")
    print(f"Phase 1 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}")

    # === Phase 2: Spectral shaping from initial (200s) ===
    print("\n=== Phase 2: Spectral shaping from initial ===")
    t0 = time.monotonic()
    result, score = _spectral_shaping(best_a, 200.0, seed=seed + 1)
    if score < best_score:
        best_a = result.copy()
        best_score = score
        print(f"  Improved to {best_score:.6f}")
    print(f"Phase 2 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}")

    # === Phase 3: Subgradient from spectral result (200s) ===
    print("\n=== Phase 3: Subgradient from spectral result ===")
    t0 = time.monotonic()
    result, score = _subgradient_descent(best_a, 200.0, seed=seed + 2)
    if score < best_score:
        best_a = result.copy()
        best_score = score
        print(f"  Improved to {best_score:.6f}")
    print(f"Phase 3 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}")

    # === Phase 4: Try different sizes (200s) ===
    print("\n=== Phase 4: Different sizes ===")
    t0 = time.monotonic()
    sizes_to_try = [1024, 2048, 4096, 8192]
    for s in sizes_to_try:
        remaining = deadline_global - time.monotonic()
        if remaining < 30:
            break
        alloc = min(50.0, remaining - 5.0)
        if alloc < 20:
            break
        print(f"  Trying size {s} with {alloc:.0f}s budget...")
        result, score = _try_size(s, alloc, seed=seed + 1000 + s)
        if result is not None and score < best_score:
            best_a = result.copy()
            best_score = score
            print(f"  New best: {best_score:.6f} at size {s}")
    print(f"Phase 4 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}")

    # === Phase 5: LP iteration (150s) ===
    print("\n=== Phase 5: LP-based iteration ===")
    remaining = deadline_global - time.monotonic()
    if remaining > 30:
        lp_budget = min(150.0, remaining - 20.0)
        if lp_budget > 20:
            try:
                result, score = _lp_iteration(best_a, lp_budget, seed=seed + 5)
                if score < best_score:
                    best_a = result.copy()
                    best_score = score
                    print(f"  LP improved to {best_score:.6f}")
            except Exception as e:
                print(f"  LP failed: {e}")
    print(f"Phase 5 done, best={best_score:.6f}")

    # === Phase 6: Final polish (remaining time) ===
    print("\n=== Phase 6: Final polish ===")
    remaining = deadline_global - time.monotonic()
    if remaining > 10:
        polish_budget = min(remaining - 5.0, 200.0)
        if polish_budget > 10:
            result, score = _polish(best_a, polish_budget, seed=seed + 999)
            if score < best_score:
                best_a = result.copy()
                best_score = score
                print(f"  Polish improved to {best_score:.6f}")
    print(f"Phase 6 done, best={best_score:.6f}")

    # === Verify with evaluate_sequence ===
    print(f"\nFinal FFT score: {best_score:.6f}")
    print(f"Final sequence length: {len(best_a)}")

    # Use the exact evaluator to confirm
    exact_score = evaluate_sequence([float(x) for x in best_a])
    print(f"Exact evaluate_sequence score: {exact_score:.6f}")

    if not np.isfinite(exact_score):
        print("WARNING: Exact score is inf, falling back to uniform")
        best_a = np.ones(init_n) / np.sqrt(2.0 * init_n)
        exact_score = evaluate_sequence([float(x) for x in best_a])
        print(f"Fallback score: {exact_score:.6f}")

    return [float(x) for x in best_a]
