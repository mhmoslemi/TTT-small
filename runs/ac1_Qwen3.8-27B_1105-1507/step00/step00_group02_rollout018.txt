where `c0` is `(a*a)_{2j} - a_j^2` (the part not involving `a_j`), and the other terms are the various lag contributions.

This is a piecewise-linear (plus one quadratic) function of `x >= 0`. We can minimize it by:
1. Evaluating `f(x)` at various points
2. Or analytically: the minimum of `max(x^2 + c0, max_i (2*a_{k_i}*x + c_i), C)` where C is the peak of parts independent of x.

For a one-dimensional problem like this, we can use a simple golden section or binary search.

Actually, since `x >= 0` and the function is the max of a convex quadratic and linear functions (and a constant), the function is convex in x (max of convex functions is convex). So we can use binary search on the derivative (or subgradient).

The subgradient of `f(x)` contains:
- `2x` if `x^2 + c0` is the active max
- `2*a_{k_i}` if `2*a_{k_i}*x + c_i` is the active max
- Any value in the convex hull of active subgradients

Setting the subgradient to 0:
- If only the quadratic is active: `2x = 0`, so `x = 0`.
- If a linear term is active: the subgradient is a constant `2*a_{k_i} > 0`, so we can't set it to 0. The minimum would be at `x = 0` or at the boundary where the quadratic takes over.

```python
def coordinate_optimize_j(a, j):
    """Optimize a[j] given all other elements fixed."""
    n = len(a)
    # Compute the convolution without considering a[j]
    # (a*a)_k = sum_i a_i * a_{k-i}, and we remove terms involving a[j]
    # Terms involving a[j]:
    #   At lag k=2j: a[j]^2
    #   At lag k where k-j is valid and k-j != j: 2*a[j]*a[k-j]
    
    # Compute full convolution
    b = np.convolve(a, a)
    
    # Subtract the contributions of a[j]
    b_no_j = b.copy()
    # Lag 2j: remove a[j]^2
    if 0 <= 2*j < 2*n-1:
        b_no_j[2*j] -= a[j]**2
    # Other lags: remove 2*a[j]*a[k-j]
    for k in range(2*n-1):
        if k == 2*j:
            continue
        other = k - j
        if 0 <= other < n:
            b_no_j[k] -= 2 * a[j] * a[other]
    
    # Now find optimal x = a[j] to minimize max(b_no_j + contributions from x)
    # contributions: x^2 at lag 2j, 2*x*a[k-j] at other lags
    # We want to minimize max_lag (b_no_j[lag] + contribution[lag])
    
    # The peak without x is:
    base_peak = np.max(b_no_j)
    
    # Try x = 0:
    # peak at x=0: max(b_no_j) = base_peak (since contributions are 0)
    
    # Try x = a[j] (current):
    # peak = max(b) = current peak
    
    # Try intermediate values...
    # For convex 1D, we can binary search or use golden section.
    
    def peak_at_x(x):
        # Compute the peak of the convolution when a[j] = x
        # b_new[k] = b_no_j[k] + contribution
        # contribution at lag 2j: x^2
        # contribution at lag k: 2*x*a[k-j] (if k-j valid and k != 2j)
        # We only need the max, so:
        # max over k of (b_no_j[k] + contrib_k(x))
        # = max(base_peak_independent, x^2 + b_no_j[2j], max_k 2*x*a[k-j] + b_no_j[k])
        
        result = base_peak  # the parts not involving x at all
        # Add x^2 contribution at lag 2j
        if 0 <= 2*j < 2*n-1:
            result = max(result, x*x + b_no_j[2*j])
        # Add 2*x*a[k-j] contributions
        for k in range(2*n-1):
            if k == 2*j:
                continue
            other = k - j
            if 0 <= other < n:
                val = 2*x*a[other] + b_no_j[k]
                if val > result:
                    result = val
        return result
    
    # Minimize peak_at_x(x) over x >= 0
    # Use golden section search
    # First, find a bracket
    # peak_at_x(0) = base_peak
    # peak_at_x grows for large x (quadratic), so there's a finite minimum
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = max(a[j] * 3, 1.0)  # initial upper bound
    # Ensure hi is large enough that peak is increasing
    while peak_at_x(hi) < peak_at_x(hi/2):
        hi = hi * 2
        if hi > 1000:
            break
    
    # Golden section
    a_gs = hi - gr*(hi - lo)
    b_gs = lo + gr*(hi - lo)
    fa = peak_at_x(a_gs)
    fb = peak_at_x(b_gs)
    for _ in range(50):
        if fa < fb:
            hi = b_gs
            b_gs = a_gs
            fb = fa
            a_gs = hi - gr*(hi - lo)
            fa = peak_at_x(a_gs)
        else:
            lo = a_gs
            a_gs = b_gs
            fa = fb
            b_gs = lo + gr*(hi - lo)
            fb = peak_at_x(b_gs)
    
    optimal_x = (lo + hi) / 2
    return optimal_x
```

```
    return np.maximum(v - theta, 0)
```

```python
def compute_score(a):
    """Compute the score directly."""
    n = len(a)
    if np.sum(a) < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (np.sum(a)**2)

def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - s)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_step(a, step_size, rng):
    """Take one subgradient step."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = np.max(b)
    # Find active lags (within tolerance)
    tol = 1e-8 * max_b
    active = np.where(b >= max_b - tol)[0]
    
    # Subgradient: g[j] = (1/|S|) * sum_{k in S} 2*a[k-j] (if 0<=k-j<n)
    g = np.zeros(n)

        # i.e., j ranges from max(0, k-n+1) to min(n-1, k)
```
