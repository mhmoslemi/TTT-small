The LP is: maximize $\sum b_j$ subject to $b_j \geq 0$ and $(f*b)_k \leq M$ for all $k$.

The LP has:
- $n$ variables ($b_0, \ldots, b_{n-1}$).
- $2n-1$ inequality constraints ($(f*b)_k \leq M$).
- $n$ non-negativity constraints.

For $n = 4806$, that's 4806 variables and ~9611 constraints. A modern LP solver can handle this.

The solution $b^*$ gives a direction. We then set $g = \frac{\sum f}{\sum b^*} b^*$ (to match the normalization) and update $f \leftarrow (1-t)f + tg$.

Why does this work? The LP finds the "largest" $b$ (in sum) that doesn't make $f*b$ exceed $M$. This $b$ is "anti-correlated" with $f$ in the sense that it avoids the lags where $f$ has high values. When we mix $f$ and $b$, the new sequence has lower peak convolution.

```python
import numpy as np
import time
from scipy.optimize import linprog

def simplex_projection(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_step(a, eta, active tolerance=0.01):
    """Take one subgradient step for minimizing max(a*a)."""
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    # Active set: lags where conv[k] >= M - tol
    tol = active_t * M if M > 0 else active_t
    active = np.where(conv >= M - tol)[0]
    # Compute subgradient
    s = np.zeros(n)
    for k in active:
        # grad_k[j] = 2 * a[k-j] if 0 <= k-j < n
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            s[j] += 2 * a[k - j]
    s /= len(active)
    # Subgradient step
    a_new = a - eta * s
    # Project onto simplex
    a_proj = simplex_projection(a_new)
    return a_proj, M

def compute_score(a):
    """Compute the score 2n * max(conv) / sum(a)^2."""
    n = len(a)
    if np.sum(a) < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv) / (np.sum(a))**2

def lp_polish(f, M, max_iters=100, t_init=0.1):
    """LP-based polishing step."""
    n = len(f)
    f = f.copy()
    for it in range(max_iters):
        # Build constraint matrix
        # A[k, j] = f[k-j] if 0 <= k-j < n else 0
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            A[k, lo:hi+1] = f[k-hi:k-lo+1][::-1]
        
        # Solve LP: max sum(b) s.t. A b <= M, b >= 0
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                      bounds=[(0, None)]*n, method='highs',
                      options={'presolve': True})
        if res.status != 0:
            break
        b = res.x
        sum_b = np.sum(b)
        if sum_b < 1e-10:
            break
        # Rescale
        sum_f = np.sum(f)
        g = (sum_f / sum_b) * b
        # Update
        t = t_init / (1 + it * 0.1)
        f = (1 - t) * f + t * g
        f = np.clip(f, 0, None)
        f = f / np.sum(f)  # Renormalize
        # Update M
        conv = np.convolve(f, f)
        M = np.max(conv)
    return f
```

Minimizes the score 2n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Uses projected subgradient method, LP-based polishing, and multi-length search.

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
