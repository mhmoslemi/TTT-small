import time

def compute_conv(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fc = fa * fa
    c = np.fft.irfft(fc, size)[:2*n-1]
    return c

def subgradient_step(a, top_k_lags, lr):
    """Take a subgradient step to reduce the top-K lags of the autocorrelation."""
    n = len(a)
    g = np.zeros(n)
    for k in top_k_lags:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += a[j]
    return a - lr * g

def optimize_sequence(a, budget_s, top_k=5):

    S = np.sum(a)
    a = a / S  # normalize to sum = 1
    best = a.copy()
    best_max = np.max(compute_conv(a))
    
    lr = 0.01
    t = 0
    deadline = time.monotonic() + budget_s
    stuck_count = 0
    
    while time.monotonic() < deadline:
        C = compute_conv(a)
        # Find top-K lags
        top_k_lags = np.argsort(C)[-top_k:]
        current_max = C[top_k_lags[0]]
        
        if current_max < best_max:
            best = a.copy()
            best_max = current_max
            stuck_count = 0
        else:

            if stuck_count > 100:

                lr = max(0.001, lr * 0.5)
                stuck_count = 0

            # dC[k]/da[i] = a[k-i] (if valid)
            valid_i = np.arange(max(0, k-(n-1)), min(n-1, k)+1)
            valid_j = k - valid_i
            g[valid_i] += a[valid_j]
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 0:

    # Restore original scaling

import time

def fft_convolve(a):
    """Full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1

    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]

def score_from_conv(C, n, S):
    """Compute the score from the convolution."""
    return 2.0 * n * np.max(C) / (S * S)

def optimize(a, budget_s, top_k=10, verbose=True):

    n = len(a)

    a = a / S  # normalize to sum = 1

    best_maxC = np.max(fft_convolve(a))
    
    lr = 1.0 / n  # initial learning rate

    deadline = time.monotonic() + budget_s

        if current_max < best_maxC:

            best_maxC = current_max
            patience = 0

            if patience > max_patience:

def _fft_convolve_self(a):
    """Compute full autocorrelation (convolution with itself) using FFT."""
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)
    return result[:size]

def _fft_cross(a, b):
    """Compute full cross-correlation of a and b using FFT.
    result[i] = sum_j a[j] * b[i+j] for i = 0, ..., len(a)-1
    """
    n = a.size
    m = b.size
    # We need result[i] = sum_j a[j] * b[i+j]
    # This is the cross-correlation, which we can compute as:
    # rev_b = b[::-1]
    # conv = convolve(a, rev_b)  # full convolution
    # result[i] = conv[m-1-i]
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)

    # result[i] = conv[m-1-i] for i=0,...,m-1 (but we only need i=0,...,n-1)
    return conv[m-1 : m-n : -1]
