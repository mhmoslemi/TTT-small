Minimizes J = 2n * max(conv(a,a)) / (sum(a))^2 using FFT-based projected
subgradient descent with smooth-max surrogate, adaptive step size,

import numpy as np

def fast_convolve(a):
    """Compute full linear convolution a*a using FFT. O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, n=size)
    result = np.fft.irfft(fa * fa, n=size)
    return result[:2 * n - 1]


def conv_grad_at(a, k):
    """Gradient of (a*a)[k] w.r.t. a. Returns array of length n."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js <= je:
        # grad[j] = 2*a[k-j] for j in [js, je]
        # As j goes js..je, (k-j) goes (k-js)..(k-je) in reverse
        grad[js:je + 1] = 2.0 * a[k - je:k - js + 1][::-1]
    return grad


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 1e-15:
        return v / s
    return np.ones(len(v)) / len(v)


def smooth_max_grad(a, tau, num_peaks):

    Returns (gradient_vector, conv_array, max_val).

    return grad, conv, cmax


def run_descent(a_init, max_iters, lr_init, tau_init, tau_final, num_peaks, deadline):
    """Run smooth-max projected gradient descent. Returns (best_a, best_max)."""

    best_a = a.copy()

def block_perturb(a, rng, min_width, max_width, sigma):
    """Apply random multiplicative block perturbation. Input/output sum to 1."""
    n = len(a)
    candidate = a.copy()

    if w >= n:
        w = max(1, n // 4)
    start = int(rng.integers(0, n - w + 1))
    candidate[start:start + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    return project_simplex(candidate)
