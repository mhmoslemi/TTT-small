    if s < 1e-10:

    else:

    N = 1
    while N < 2*n - 1:
        N *= 2
    
    def get_conv(a):
        A = np.fft.rfft(a, N)
        C = np.fft.irfft(A * A, N)
        return C[:2*n-1]
    
    def get_obj(a):
        c = get_conv(a)
        return c.max()
    
    best_a = a.copy()
    best_val = get_obj(a)
    
    # Projected subgradient descent
    eta0 = 0.1
    for it in range(50000):
        if time.monotonic() > deadline - 100:  # leave time for local search
            break
        
        c = get_conv(a)
        max_c = c.max()
        
        # Subgradient: weight peaks
        tau = 0.01 * max_c  # soft max temperature
        weights = np.exp((c - max_c) / tau)
        weights /= weights.sum()

        # This is a correlation (circular, but since c is zero-padded to N, it's linear)
        W = np.fft.rfft(weights, N)
        A = np.fft.rfft(a, N)
        G = np.fft.irfft(A * W, N)  # This gives sum_k weights[k] * a[k-i] (circular conv)
        # Wait, I need to be careful. The linear convolution of a with weights is:
        # (a * w)[i] = sum_k a[k] * w[i-k]
        # But I want sum_k w[k] * a[k-i] = sum_k w[k] * a[k-i]
        # Let j = k-i: sum_j w[j+i] * a[j] = (w_flipped * a)[i] where w_flipped[k] = w[k]... 
        # Actually, sum_k w[k] * a[k-i] = sum_j w[j+i] * a[j] = (reverse_w * a)[i]
        # where reverse_w[k] = w[-k] (with appropriate handling).
        # In DFT: IFFT(FFT(a) * FFT(reverse_w))
        # FFT(reverse_w) = conj(FFT(w)) (for real w)
        # So G = IFFT(FFT(a) * conj(FFT(w)))
        # But we're using real FFT...

        # Find peaks (within 10% of max)
        threshold = 0.9 * max_c
        peak_indices = np.where(c >= threshold)[0]

        for k in peak_indices:

            # a[k-i] is nonzero when 0 <= k-i < n, i.e., k-n+1 <= i <= k
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                # a[k-i] for i in [lo, hi] is a[k-hi : k-lo+1] reversed
                g[lo:hi+1] += a[k-hi:k-lo+1][::-1]

        eta = eta0 / (1 + 0.001 * it)
        a = a - eta * g

        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.ones(n) / n
        
        val = get_obj(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    
    # Phase 2: Local search
    a = best_a.copy()
    best_val = get_obj(a)
    
    while time.monotonic() < deadline:
        # Random block perturbation
        width = rng.integers(max(1, n//50), max(2, n//5))
        start = rng.integers(0, n - width)
        perturbation = np.exp(rng.normal(0, 0.1, width))
        candidate = a.copy()
        candidate[start:start+width] *= perturbation
        candidate = np.maximum(candidate, 0)
        candidate_sum = candidate.sum()
        if candidate_sum > 0:
            candidate /= candidate_sum
        
        val = get_obj(candidate)
        if val < best_val:
            best_val = val
            best_a = candidate.copy()
    
    # Convert back to the scale that the evaluator expects
    # Since the objective is scale-invariant, any scaling works.
    # Use sum = sqrt(2n) to match the SOTA normalization.
    target_sum = np.sqrt(2 * n)
    result = best_a * target_sum
    result = np.clip(result, 0, 1000)
    
    return [float(x) for x in result]
