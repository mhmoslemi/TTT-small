"""
LP-based iterative improvement for minimizing 2*n*max(autoconv)/sum^2.
Key insight: non-uniform sequences (ramps) achieve score ~4/3 vs uniform's 2.
Uses LP block coordinate descent + targeted local search with multiple inits.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

_structures = {}

def _get_structure(n):
    if n in _structures:
        return _structures[n]
    ar = np.arange(n, dtype=np.int32)
    col_lengths = np.minimum(n, 2 * n - 1 - ar).astype(np.int32)
    cols = np.repeat(ar, col_lengths)
    cumstarts = np.concatenate([[0], np.cumsum(col_lengths[:-1])].astype(np.int32)) if n > 1 else np.array([0], dtype=np.int32)
    rows_base = np.repeat(ar, col_lengths)
    offsets = np.concatenate([np.arange(l, dtype=np.int32) for l in col_lengths]) if n > 1 else np.array([0], dtype=np.int32)
    rows = (rows_base + offsets).astype(np.int32)
    struct = (rows, cols, col_lengths)
    _structures[n] = struct
    return struct

def solve_lp(f, M, structure):
    n = len(f)
    rows, cols, col_lengths = structure
    if n == 1:
        return np.array([min(M / max(f[0], 1e-30), 1e30) if M > 0 else 0.0])
    vals = np.concatenate([f[:l] for l in col_lengths])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 50})
    if res.status == 0:
        return np.maximum(res.x, 0.0)
    return None

def make_ramp(n, alpha=1.0):
    i = np.arange(1, n + 1, dtype=np.float64)
    f = i ** alpha
    f /= f.sum()
    return f

def make_symmetric_bump(n, p=2.0):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    f = ((i - center) / center) ** 2
    f = 1.0 / (1.0 + f ** p)
    f /= f.sum()
    return f

def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = max(1.0, sigma_frac * n)
    f = np.exp(-(x - center) ** 2 / (2.0 * sigma ** 2))
    f /= f.sum()
    return f

def make_cosine_window(n):
    i = np.arange(n, dtype=np.float64)
    f = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / (n - 1)))
    f = np.clip(f, 0.01, None)
    f /= f.sum()
    return f

def make_two_bumps(n, pos1=0.3, pos2=0.7, sigma_frac=0.15):
    x = np.arange(n, dtype=np.float64)
    sigma = max(1.0, sigma_frac * n)
    f = np.exp(-(x - pos1 * n) ** 2 / (2 * sigma ** 2)) + \
        np.exp(-(x - pos2 * n) ** 2 / (2 * sigma ** 2))
    f /= f.sum()
    return f

def _evaluate(f):
    return float(evaluate_sequence(f.tolist()))

def lp_iterate(f, time_limit, rng, tag=""):
    n = len(f)
    structure = _get_structure(n)
    M = float(np.max(np.convolve(f, f)))
    best_f = f.copy()
    best_score = _evaluate(f)
    best_M = M
    stalled = 0
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_limit:
        it += 1
        g = solve_lp(f, M, structure)
        if g is None:
            break
        Sg = float(g.sum())
        if Sg < 1.0 + 1e-7:
            stalled += 1
            if stalled > 5:
                scale = 0.05 + 0.1 * rng.random()
                f = np.clip(f + scale * rng.normal(0, 1, n), 0, None)
                s = f.sum()
                if s < 1e-15:
                    break
                f /= s
                M = float(np.max(np.convolve(f, f)))
                stalled = 0
            continue
        stalled = 0
        g = g / Sg
        # Adaptive step
        excess = Sg - 1.0
        t = min(1.0, max(0.2, min(1.0, 5.0 * excess)))
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new /= s
        M_new = float(np.max(np.convolve(f_new, f_new)))
        if M_new < M:
            f = f_new
            M = M_new
            sc = _evaluate(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
                best_M = M
                if sc < 1.6:
                    print(f"[{tag}] LP it={it} sc={sc:.6f} M={M:.2e}")
        else:
            t2 = t * 0.3
            f_new2 = (1.0 - t2) * f + t2 * g
            f_new2 = np.clip(f_new2, 0, None)
            s2 = f_new2.sum()
            if s2 > 1e-15:
                f_new2 /= s2
                M_new2 = float(np.max(np.convolve(f_new2, f_new2)))
                if M_new2 < M:
                    f = f_new2
                    M = M_new2
    return best_f, best_score

def targeted_local_search(f, time_limit, rng, tag=""):
    n = len(f)
    best_f = f.copy()
    best_score = _evaluate(f)
    step = 0.005
    start = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_limit:
        it += 1
        conv = np.convolve(f, f)
        k_peak = int(np.argmax(conv))
        i_min = max(0, k_peak - (n - 1))
        i_max = min(n - 1, k_peak)
        contributing = np.arange(i_min, i_max + 1)
        if len(contributing) < 2:
            contributing = np.arange(n)
        # Weighted pick for decrease
        contrib_vals = f[contributing]
        other_vals = f[k_peak - contributing]
        weights = contrib_vals * other_vals
        tw = weights.sum()
        if tw > 1e-30:
            probs = weights / tw
            idx_dec = contributing[rng.choice(len(contributing), p=probs)]
        else:
            idx_dec = contributing[rng.integers(len(contributing))]
        # Non-contributing for increase
        mask = np.ones(n, dtype=bool)
        mask[i_min:i_max + 1] = False
        non_contrib = np.where(mask)[0]
        if len(non_contrib) == 0:
            non_contrib = np.array([1 - idx_dec if idx_dec < n - 1 else 0])
        idx_inc = non_contrib[rng.integers(len(non_contrib))]

        amount = step * f.mean()
        f_new = f.copy()
        f_new[idx_dec] = max(0.0, f_new[idx_dec] - amount)
        f_new[idx_inc] += amount
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new *= (f.sum() / s)
        sc = _evaluate(f_new)
        if sc < best_score - 1e-9:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if sc < 1.6:
                print(f"[{tag}] LS it={it} sc={sc:.6f}")
        else:
            no_improve += 1
            if no_improve > 200:
                step = min(0.02, step * 1.5)
                no_improve = 0
            elif rng.random() < 0.01:
                step = min(0.02, step * 1.2)
    return best_f, best_score
