def build_convolution_matrix(a, n):
    """Build the sparse convolution constraint matrix C where C[k,j] = a[k-j] if valid."""
    # C is (2n-1) x n
    # C[k, j] = a[k-j] if 0 <= k-j < n, else 0

def build_conv_matrix(a):
    """Build sparse (2n-1) x n matrix C where C[k,j] = a[k-j] if 0 <= k-j < n."""
    n = len(a)
    j = np.repeat(np.arange(n), n)
    d = np.tile(np.arange(n), n)

    C = csc_matrix((a[d], (k, j)), shape=(2 * n - 1, n))
    return C


def smooth_gradient_step(a, eta, temperature):
    """One step of smooth gradient descent on max convolution using softmax weights."""
    n = len(a)
    conv = np.convolve(a, a, mode='full')
    shifted = conv - conv.max()
    weights = np.exp(shifted / temperature)
    weights /= weights.sum()
    # grad[i] = 2 * sum_m a[m] * weights[m+i] = 2 * correlate(weights, a, 'valid')[i]
    grad = 2.0 * np.correlate(weights, a, mode='valid')

    a_new = np.maximum(a_new, 1e-30)

    return a_new


def solve_lp_for_g(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""

    return None
