import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_lp_direction(a, M):
    """Solve the LP: max sum(b) s.t. convolve(a,b) <= M, b >= 0."""
    n = len(a)
    # Build sparse constraint matrix
    # A[k, m] = a[k-m] if 0 <= k-m < n, else 0
    # Constraint: A @ b <= M (vector of M's, length 2n-1)
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        for m in range(m_min, m_max + 1):
            rows.append(k)
            cols.append(m)
            vals.append(a[k - m])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    # Objective: maximize sum(b), i.e., minimize -sum(b)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, 
                  method='highs', options={'presolve': True})
    
    if res.status == 0:
        return res.x
    return None
