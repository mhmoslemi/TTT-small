"""
Step function optimization: minimize J = 2n*max(conv(a,a))/sum(a)^2.
Uses peak-directed local search with diverse asymmetric initializations,
adaptive step sizes, and multi-phase search strategy.
"""
import numpy as np
import time


def fast_score(a):
    """Compute J = 2n * max(conv(a,a)) / sum(a)^2 directly."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s

def peak_directed_move(best, rng, delta):
    """
    Generate a candidate by transferring mass from a high-contribution
    position to a low-contribution position relative to the current peak.
    Returns list of candidate arrays (trying multiple recipients).
    """
    n = len(best)
    conv = np.convolve(best, best)
    k_star = int(np.argmax(conv))

    # Contribution of each position i to the peak at k_star:
    # (a*a)[k_star] includes a[i]*a[k_star-i] for valid k_star-i
    idx = np.arange(n)
    j_idx = k_star - idx
    valid = (j_idx >= 0) & (j_idx < n)
    contrib = np.zeros(n)
    contrib[valid] = best[idx[valid]] * best[j_idx[valid]]

    donor = int(np.argmax(contrib))

    # Find low-contribution positions
    low_thresh = np.percentile(contrib, 25)
    low_positions = np.where(contrib <= low_thresh)[0]
    if len(low_positions) == 0:
        low_positions = np.array([int(np.argmin(contrib))])

    # Try multiple recipients
    candidates_list = []
    n_tries = min(6, len(low_positions))
    chosen = rng.choice(low_positions, size=n_tries, replace=False)

    for recipient in chosen:
        candidate = best.copy()
        transfer = min(delta * best[donor], 0.08 * best[donor])
        if transfer < 1e-12:
            continue
        candidate[donor] -= transfer
        candidate[recipient] += transfer
        candidate = np.clip(candidate, 0.0, None)
        candidates_list.append(candidate)

    return candidates_list, conv, k_star


def block_scale_move(best, rng):
    """Generate a candidate by scaling a random block."""
    n = len(best)
    candidate = best.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 6)))
    start = int(rng.integers(0, n - width))
    factor = float(np.exp(rng.normal(0.0, 0.25)))
    candidate[start:start + width] *= factor
    return np.clip(candidate, 0.0, None)


def position_move(best, rng):
    """Perturb a single random position."""
    n = len(best)
    candidate = best.copy()
    pos = int(rng.integers(n))
    candidate[pos] *= float(np.exp(rng.normal(0.0, 0.35)))
    return np.clip(candidate, 0.0, None)


def local_search(a, rng, deadline, verbose=False):

    Peak-directed local search with adaptive step size and
    multiple move types. Returns (best_array, best_score).
    """
    n = len(a)
    best = a.copy()

    delta = 0.03
    iters = 0
    stall = 0
    best_seen = best_score

        if move_type < 0.55:
            # Peak-directed mass transfer
            cands, conv, k_star = peak_directed_move(best, rng, delta)
            best_cand_score = best_score
            best_cand = None

                sc = fast_score(cand)

                    best_cand_score = sc
                    best_cand = cand

                best = best_cand
                best_score = best_cand_score
                stall = 0
                best_seen = min(best_seen, best_score)
                delta = min(delta * 1.03, 0.25)
            else:
                stall += 1

        elif move_type < 0.75:
            # Block scaling

                best = candidate
                best_score = cand_score
                stall = 0
                best_seen = min(best_seen, best_score)
                delta = min(delta * 1.02, 0.25)

                stall += 1

            cand_score = fast_score(candidate)

    return best, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for optimal step function heights.
    Phase 1: Diverse initialization + short refinement.
    Phase 2: Extended local search on best candidate.
    Phase 3: Fine-tuning with smaller steps.
    Returns the best sequence found.

    start_time = time.monotonic()
    total_budget = max(10.0, float(budget_s) - 20.0)

    # Determine sequence length from parent
    n = 512
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.size > 0 and np.isfinite(parent).all():
            n = parent.size
    except (NameError, TypeError):
        pass

    print(f"Sequence length: {n}, budget: {budget_s}s")

    # Phase 1: Generate diverse candidates and do short refinement
    phase1_budget = min(120.0, total_budget * 0.15)
    phase1_deadline = time.monotonic() + phase1_budget

    candidates = make_initial_candidates(n, rng)
    print(f"Generated {len(candidates)} initial candidates")

    # Score all candidates
    scored = []
    for i, c in enumerate(candidates):
        sc = fast_score(c)
        scored.append((sc, i))
    scored.sort()

    best = candidates[scored[0][1]].copy()
    best_score = scored[0][0]
    print(f"Best initial candidate: {best_score:.6f}")

    # Short refinement on top candidates
    top_k = min(8, len(scored))
    for sc, idx in scored[:top_k]:
        if time.monotonic() > phase1_deadline:
            break
        refine_budget = min(8.0, phase1_deadline - time.monotonic())
        if refine_budget < 1.0:
            break
        refined, refined_score = local_search(
            candidates[idx].copy(), rng, time.monotonic() + refine_budget)
        if refined_score < best_score:
            best = refined
            best_score = refined_score
            print(f"  Phase 1 improvement: {best_score:.6f}")

    print(f"After Phase 1: {best_score:.6f} (elapsed {time.monotonic()-start_time:.1f}s)")

    # Phase 2: Extended local search
    phase2_deadline = start_time + total_budget * 0.85
    best, best_score = local_search(best, rng, phase2_deadline, verbose=True)
    print(f"After Phase 2: {best_score:.6f} (elapsed {time.monotonic()-start_time:.1f}s)")

    # Phase 3: Fine-tuning with smaller steps
    if time.monotonic() < start_time + total_budget:
        # Run local search again (will use smaller effective delta due to
        # the adaptive mechanism, but reset to encourage finer moves)
        best, best_score = local_search(best, rng, start_time + total_budget, verbose=False)
        print(f"After Phase 3: {best_score:.6f} (elapsed {time.monotonic()-start_time:.1f}s)")

    # Final validation with the official evaluator
    result = [float(x) for x in best]
    final_score = float(evaluate_sequence(result))
    print(f"Final validated score: {final_score:.6f}")
    print(f"Total time: {time.monotonic()-start_time:.1f}s")

    return result
