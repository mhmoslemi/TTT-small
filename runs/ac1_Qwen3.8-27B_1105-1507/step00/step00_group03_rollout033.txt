Minimize 2n*max(convolve(p,p))/sum(p)^2 using smooth projected gradient descent
with log-sum-exp approximation, multiple restarts, and local search refinement.

import numpy as np

def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u * ind > cssv
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0)

def fft_convolve(a, b):
    """FFT-based full convolution."""
    n = len(a) + len(b) - 1
    L = 1
    while L < n:
        L *= 2
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:n]

def compute_b(p):
    """Compute the full convolution of p with itself."""
    return fft_convolve(p, p)

def compute_gradient(p, t):
    """Compute gradient of soft_max(convolve(p,p)) with respect to p."""
    n = len(p)
    b = compute_b(p)

    exp_t = np.exp(t * (b - b_max))
    w = exp_t / exp_t.sum()

    # cross_correlation(w, p)[i] = sum_j w[i+j] * p[j] for j=0..n-1
