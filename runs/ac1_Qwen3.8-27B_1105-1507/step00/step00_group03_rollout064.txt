Subgradient descent with momentum, multiple initializations, and periodic
restarts to minimize the peak-to-average ratio of linear autocorrelation
for non-negative sequences. Uses direct score computation for speed.

import numpy as np
import time


def _normalize(x, n):
    """Normalize a non-negative array to unit sum."""
    s = x.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return x / s


def _subgradient_step(a, eta, K, momentum, beta, n):
    """
    Perform one subgradient step with momentum.
    Returns (new_a, new_momentum, score).
    """
    c = np.convolve(a, a)
    max_c = c.max()
    score = 2.0 * n * max_c  # since sum(a) = 1

    # Top-K peaks
    K_curr = min(K, 2 * n - 1)
    peak_idx = np.argpartition(c, -K_curr)[-K_curr:]

    # Gradient via cross-correlation
    pw = np.zeros(2 * n - 1)
    pw[peak_idx] = 1.0 / K_curr
    grad = (2.0 / K_curr) * np.correlate(pw, a, mode='valid')

    # Momentum update
    momentum = beta * momentum - eta * grad
    a_new = a + momentum
    a_new = np.clip(a_new, 0.0, None)
    a_new = _normalize(a_new, n)

    return a_new, momentum, score

def _lp_direction(a, n, M):
    """
    Solve LP: maximize sum(g) subject to g >= 0, (a*g)[k] <= M for all k.
    Returns the LP solution g normalized to unit sum, or None on failure.
    Uses scipy linprog.
    """
    from scipy.optimize import linprog

    # Variables: g[0], ..., g[n-1]
    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)

    # Constraints: (a * g)[k] <= M for k = 0, ..., 2n-2
    # (a * g)[k] = sum_j a[j] * g[k-j] for valid j
    # This is a matrix-vector product: A_mat[k, j] = a[k-j] if 0<=k-j<n else 0
    # A_mat has shape (2n-1, n)
    A_mat = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A_mat[k, j_start:j_end + 1] = a[j_start:j_end + 1]

    b_vec = np.full(2 * n - 1, M)

    # Bounds: g >= 0
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_vec, bounds=bounds, method='highs')
        if res.success and res.fun < -1e-10:
            g = res.x.copy()
            g = np.clip(g, 0, None)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass

    return None
