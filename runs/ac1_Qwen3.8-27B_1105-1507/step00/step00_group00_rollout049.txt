    """Solve LP: max sum(g) s.t. g >= 0, conv(f,g) <= M."""
    n = len(f)

def solve_lp(f, M):

    n = len(f)
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):

            rows.append(k)
            cols.append(j)
            vals.append(f[i])

    return None

"""
LP-based iterative improvement for minimizing autoconvolution peak-to-sum ratio.
Uses multiple initializations (Gaussian, ramp, flat-top, random, provided sequence)
and runs LP-based block coordinate descent with local search refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, element-wise conv(f,g) <= M."""
    n = len(f)
    if n == 0:
        return None
    
    # Build sparse constraint matrix A where A[k, j] = f[i] if j = k-i
    # (i.e., the contribution of g[j] to conv(f,g)[k])
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            vals.append(f[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    
    if res.status == 0:
        return res.x
    return None


def make_initial_sequence(n, init_type, rng):
    """Create an initial sequence of length n with the given type."""
    if init_type == 'gaussian':
        x = np.arange(n) - (n - 1) / 2.0
        sigma = max(1.0, n / 4.0)
        f = np.exp(-x ** 2 / (2 * sigma ** 2))
    elif init_type == 'ramp':
        f = np.linspace(0.01, 1.0, n)
    elif init_type == 'flat_top':
        f = np.ones(n)
        edge = max(1, n // 10)
        for i in range(edge):
            val = 0.5 * (1 + np.cos(np.pi * i / edge))
            f[i] = val
            f[n - 1 - i] = val
    elif init_type == 'bump':
        # Two bumps
        center1 = n // 4
        center2 = 3 * n // 4
        sigma = max(1.0, n / 8.0)
        x = np.arange(n).astype(float)
        f = np.exp(-(x - center1) ** 2 / (2 * sigma ** 2)) + \
            np.exp(-(x - center2) ** 2 / (2 * sigma ** 2))
    elif init_type == 'random':
        f = rng.random(n) + 0.01
    elif init_type == 'cosine':
        # Single cosine bump
        x = np.arange(n)
        f = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        f[0] = 0.01  # avoid exact zero
    else:
        f = np.ones(n)
    
    f = np.clip(f, 0.0, None)
    total = f.sum()
    if total > 1e-15:
        f = f / total
    else:
        f = np.ones(n) / n
    return f

    """Run LP-based iterations to improve the sequence. Returns best found."""
    n = len(f)
    M = np.max(np.convolve(f, f))

    best_M = M
    best_score = float(evaluate_sequence(f.tolist()))

    start_time = time.monotonic()
    
    while time.monotonic() - start_time < time_limit:

        g = solve_lp(f, M)
        if g is None:
            break

        if Sg <= 1.0 + 1e-8:
            # At local optimum for this f, try perturbation
            if verbose:

        if total < 1e-15:

        f_new = f_new / (total / f.sum())  # maintain same sum

        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            # Gradually decrease step size

        else:

            if rng.random() < 0.01:
