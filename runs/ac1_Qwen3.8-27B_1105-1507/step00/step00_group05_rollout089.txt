def _push_top(store, arr, score, maxlen=8):
    if not np.isfinite(score): return
    arr = np.asarray(arr, dtype=np.float64)
    if arr.size ==0 or not np.isfinite(arr).all(): return
    arr = np.clip(arr,0.0,1000.0)
    s=arr.sum()
    if s <=0:
       arr = np.ones(arr.size)/arr.size
    else:
       arr = arr/s
    arr = arr.copy()
    if len(store) < maxlen:
       store.append([score, arr])
    else:
       # if score worse than worst, ignore
       worst_idx=0; worst_val=store[0][0]
       for i in range(1,len(store)):
          if store[i][0] > worst_val:
             worst_val=store[i][0]; worst_idx=i
       if score < worst_val:
          store[worst_idx]=[score, arr]

"""
Searches for low evaluation sequences by optimizing positive sequences in
log-domain with a mirror-descent style subgradient method and random restarts.

The objective is scale-invariant.  Sequences are internally normalized to sum 1.
Fast FFT-based convolution and gradient correlation are used for many iterations,
and the final candidates are verified with evaluate_sequence.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    m = 1
    while m < x:
        m <<= 1
    return m


def _normalize_pos(a) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(a.size, dtype=np.float64)
    return a / s


def _fast_score(a) -> float:
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")

    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return float("inf")

    need = 2 * a.size - 1
    m = _next_pow2(need)

    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    c = c[:need]

    maxc = float(np.max(c))
    if not np.isfinite(maxc) or maxc < 0.0:
        maxc = 0.0

    return float(2.0 * a.size * maxc / (s * s))


def _make_v4_from_x(x: np.ndarray, c1: float, c2: float, c3: float, c4: float) -> np.ndarray:
    xm = x - 0.5
    xm2 = xm * xm
    V = c1 * x + c2 * xm2 + c3 * xm2 * xm + c4 * xm2 * xm2
    maxv = float(np.max(V))
    if not np.isfinite(maxv):
        return np.ones(x.size, dtype=np.float64) / float(x.size)
    V = V - maxv
    p = np.exp(V)
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(x.size, dtype=np.float64) / float(x.size)
    return p / s


def _make_log_from_x(x: np.ndarray, e0: float, e1: float, c1: float) -> np.ndarray:
    eps = 1e-9
    V = c1 * x + e0 * np.log(x + eps) + e1 * np.log(1.0 - x + eps)
    maxv = float(np.max(V))
    if not np.isfinite(maxv):
        return np.ones(x.size, dtype=np.float64) / float(x.size)
    V = V - maxv
    p = np.exp(V)
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(x.size, dtype=np.float64) / float(x.size)
    return p / s


def _push_top(store, arr, score: float, maxlen: int):
    """Keep up to maxlen sequences, ordered by implicitly tracked internal score."""
    if not np.isfinite(score):
        return

    arr = np.asarray(arr, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)

    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 0.0:
        arr = np.ones(arr.size, dtype=np.float64)
        s = float(arr.size)
    else:
        arr = arr / s

    arr = arr.copy()

    if len(store) < maxlen:
        store.append([score, arr])
    else:
        worst_idx = 0
        worst_val = store[0][0]
        for i in range(1, len(store)):
            if store[i][0] > worst_val:
                worst_val = store[i][0]
                worst_idx = i
        if score < worst_val:
            store[worst_idx] = [score, arr]


def _scale_output(p, max_val: float = 1000.0):
    p = np.asarray(p, dtype=np.float64)
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.clip(p, 0.0, max_val)

    if p.size == 0:
        return [1.0] * 16

    maxp = float(np.max(p))
    if not np.isfinite(maxp) or maxp <= 0.0:
        return [1.0] * 16

    out = p * (max_val / maxp)
    out = np.nan_to_num(out, nan=0.0, posinf=max_val, neginf=0.0)
    out = np.clip(out, 0.0, max_val)

    if not np.isfinite(out).all():
        return [1.0] * 16

    return [float(v) for v in out]


def _run_mirror(p, rng, until, store=None, lr: float = 0.35, K: int = 32):
    """
    Log-domain mirror descent / subgradient improvement.
    Returns (best_p, best_internal_score).
    """
    p = _normalize_pos(p)
    n = p.size

    if n < 8:
        return p, _fast_score(p)

    need = 2 * n - 1
    m = _next_pow2(need)

    pc = _fast_score(p)
    if store is not None:
        _push_top(store, p, pc, 8)

    if not np.isfinite(pc) or time.monotonic() >= until:
        return p, pc

    logp = np.log(np.clip(p, 1e-300, None))
    logp = logp - float(np.max(logp))
    P = np.fft.fft(p, m)

    fail = 0
    it = 0

    while time.monotonic() < until:
        c = np.fft.ifft(P * P).real[:need]
        cmax = float(np.max(c))
        if not np.isfinite(cmax) or cmax <= 0.0:
            break

        Kuse = int(min(K, need))
        if Kuse < need:
            part = np.argpartition(c, -Kuse)[-Kuse:]
        else:
            part = np.arange(need, dtype=int)

        c_top = c[part]
        wsum = float(np.sum(c_top))
        if not np.isfinite(wsum) or wsum <= 0.0:
            wsum = 1.0

        q = np.zeros(need, dtype=np.float64)
        q[part] = c_top / wsum

        Q = np.fft.fft(q, m)
        R = np.fft.ifft(P * np.conj(Q)).real

        grad = np.empty(n, dtype=np.float64)
        grad[0] = R[0]
        if n > 1:
            grad[1:] = R[m - 1:m - n:-1]

        g = (2.0 * float(Kuse) / cmax) * grad
        g = np.nan_to_num(g, nan=0.0, posinf=10.0, neginf=-10.0)
        g = np.clip(g, -10.0, 10.0)

        mu = float(np.dot(p, g))
        if not np.isfinite(mu):
            mu = 0.0
        g = g - mu

        newlog = logp - lr * g
        maxnl = float(np.max(newlog))
        minnl = float(np.min(newlog))

        if not np.isfinite(maxnl):
            lr = max(lr * 0.7, 0.01)
            fail += 1
            continue

        if maxnl - minnl > 50.0:
            newlog = np.clip(newlog, maxnl - 50.0, maxnl)
        maxnl = float(np.max(newlog))
        newlog = newlog - maxnl

        newp = np.exp(newlog)
        newp = np.nan_to_num(newp, nan=0.0, posinf=0.0, neginf=0.0)
        newp = newp + 1e-14

        ns = float(np.sum(newp))
        if ns <= 0.0 or not np.isfinite(ns):
            lr = max(lr * 0.7, 0.01)
            fail += 1
            continue

        newp = newp / ns

        Pnew = np.fft.fft(newp, m)
        cnew = np.fft.ifft(Pnew * Pnew).real[:need]
        newcmax = float(np.max(cnew))

        if not np.isfinite(newcmax):
            lr = max(lr * 0.7, 0.01)
            fail += 1
            continue

        if newcmax < 0.0:
            newcmax = 0.0

        newscore = float(2.0 * n * newcmax)

        if newscore < pc - 1e-12:
            p = newp
            pc = newscore
            logp = np.log(np.clip(p, 1e-300, None))
            logp = logp - float(np.max(logp))
            P = Pnew
            lr = min(lr * 1.03, 1.6)
            fail = 0

            if store is not None:
                _push_top(store, p, pc, 8)
        else:
            lr = max(lr * 0.95, 0.01)
            fail += 1

            if fail == 100 or fail == 200 or (fail > 200 and fail % 200 == 0):
                sigma = 0.12 + 0.03 * (fail // 200)
                if sigma > 1.0:
                    sigma = 1.0

                noise = rng.normal(0.0, sigma, n)
                pl = logp + noise

                maxpl = float(np.max(pl))
                minpl = float(np.min(pl))

                if not np.isfinite(maxpl):
                    fail += 1
                    continue

                if maxpl - minpl > 50.0:
                    pl = np.clip(pl, maxpl - 50.0, maxpl)

                maxpl = float(np.max(pl))
                pl = pl - maxpl

                pnl = np.exp(pl)
                pnl = np.nan_to_num(pnl, nan=0.0, posinf=0.0, neginf=0.0) + 1e-14
                sp = float(np.sum(pnl))

                if sp > 0.0 and np.isfinite(sp):
                    pnl = pnl / sp
                    Pnoise = np.fft.fft(pnl, m)
                    cnoise = np.fft.ifft(Pnoise * Pnoise).real[:need]
                    nmax = float(np.max(cnoise))

                    if np.isfinite(nmax) and nmax > 0.0:
                        nscore = float(2.0 * n * nmax)
                        if nscore < pc - 1e-12:
                            p = pnl
                            pc = nscore
                            logp = np.log(np.clip(p, 1e-300, None))
                            logp = logp - float(np.max(logp))
                            P = Pnoise
                            lr = 0.35
                            fail = 0

                            if store is not None:
                                _push_top(store, p, pc, 8)

        it += 1
        if it % 2000 == 0:
            print("mirror n=%d score=%.9f lr=%.4f fail=%d t=%.1f" % (n, pc, lr, fail, time.monotonic()))

    return p, pc
