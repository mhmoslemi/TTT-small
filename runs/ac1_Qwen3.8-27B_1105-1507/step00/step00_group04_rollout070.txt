Optimizes a nonnegative sequence to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses projected subgradient descent with FFT convolution, alternating projection
initialization, and LP-based improvement steps. Tries multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho_candidates = np.where(u * np.arange(1, n + 1) > cv)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fc = fa * fa
    result = np.fft.irfft(fc, N)[:size]
    # Clean up tiny numerical artifacts
    result[result < 0] = 0.0
    return result


def compute_subgradient(a, peak_indices):
    """Compute subgradient of max_k c[k] at point a, averaged over active peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= max(1, len(peak_indices))
    return grad

def lp_step(a, P):
    """
    LP step: maximize sum(b) subject to convolve(a,b) <= P, b >= 0.
    Returns rescaled solution or None.
    """
    n = len(a)
    if n > 3000:
        return None

    # Build convolution matrix A of shape (2n-1, x n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    # LP: min -1^T b s.t. A @ b <= P*ones, b >= 0
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, P)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': 8.0})
        if res.success:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-8:
                a_sum = a.sum()
                b = b * (a_sum / b_sum)
                return b
    except Exception:
        pass
    return None
