import numpy as np

def make_conv_template(n):
    if n <= 0: return None
    k = np.arange(2*n-1, dtype=np.int32)
    js = np.maximum(0, k - n + 1).astype(np.int32)
    je = np.minimum(n-1, k).astype(np.int32)
    counts = (je - js + 1).astype(np.int64)
    total = int(counts.sum())
    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts
    row = np.repeat(k, counts).astype(np.int32)
    col = (js[np.repeat(np.arange(2*n-1, dtype=np.int64), counts)] + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)
    A_off = csr_matrix((offsets, (row, col)), shape=(2*n-1, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    m = 2*n-1
    return A_template, edge_offsets, c_obj, bounds, m

"""
LP-based iterative search for a nonnegative step function minimizing
2*n*max(convolve(f,f))/(sum(f))^2.

At each iteration, for the current normalized f, we solve the LP

    max sum(g)
    s.t. (f * g)_k <= M   for all k
         g >= 0

where M = max_k (f*f)_k.  If the LP can make sum(g) larger than the
normalization sum, the rescaled g is a descent direction.  We take a
safe step toward g and keep the best sequence found.  The search is run
from several random starts and from up-sampled versions of the currently
best sequence.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def normalize_sequence(arr):
    """Return a nonnegative floating vector normalized to sum 1, or None."""
    try:
        x = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None
    if x.size == 0:
        return None
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return None
    x /= s
    return x


def score_normalized(x, n):
    """Score assuming x is nonnegative and sum(x) = 1."""
    conv = np.convolve(x, x)
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def mix_noisy(base, rng, eps):
    """Mix a base vector with a random Dirichlet vector for a random restart."""
    f = normalize_sequence(base)
    if f is None:
        return None
    n = f.size
    if eps <= 1e-12:
        return f
    try:
        mix = rng.dirichlet(np.ones(n, dtype=np.float64))
    except Exception:
        mix = np.ones(n, dtype=np.float64) / float(n)
    if eps >= 1.0:
        f = mix
    else:
        f = (1.0 - eps) * f + eps * mix
    return normalize_sequence(f)


def interpolate_sequence(x, n):
    """Resample a nonnegative sequence to length n and normalize it."""
    x = normalize_sequence(x)
    if x is None or n <= 0:
        return None
    m = x.size
    if m == n:
        return x
    if m == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(new, old, x)
    return normalize_sequence(out)


def add_unique_start(starts, f, tol=1e-6):
    """Append f if it is not numerically too similar to an existing start."""
    f = normalize_sequence(f)
    if f is None:
        return
    for g in starts:
        if float(np.linalg.norm(f - g)) < tol:
            return
    starts.append(f)


def choose_starts(n, budget, rng, best_f, given_f):
    """Choose a diverse set of initial sequences for a given length n."""
    starts = []

    if best_f is not None:
        up = interpolate_sequence(best_f, n)
        up = mix_noisy(up, rng, 0.05)
        add_unique_start(starts, up)

    uniform = mix_noisy(np.ones(n, dtype=np.float64), rng, 1.0)
    add_unique_start(starts, uniform)

    if budget > 25.0:
        d = mix_noisy(rng.dirichlet(np.ones(n, dtype=np.float64)), rng, 0.0)
        add_unique_start(starts, d)

    if budget > 80.0 and given_f is not None:
        g = interpolate_sequence(given_f, n)
        g = mix_noisy(g, rng, 0.05)
        add_unique_start(starts, g)

    if budget > 160.0 and n <= 256:
        d2 = mix_noisy(rng.dirichlet(np.full(n, 0.3, dtype=np.float64)), rng, 0.0)
        add_unique_start(starts, d2)

    out = []
    for f in starts:
        f = normalize_sequence(f)
        if f is None or f.size != n:
            continue
        out.append(f)
        if len(out) >= 4:
            break
    if not out:
        out.append(mix_noisy(np.ones(n, dtype=np.float64), rng, 1.0))
    return out


def make_conv_template(n):
    """
    Precompute the CSR matrix structure for convolve(f, g).

    Returned template has:
      A_template : sparse float template, data must be overwritten each call
      edge_offsets : for each matrix row entry, the index into f
      c_obj : LP objective vector
      bounds : LP bounds for g
      m : number of convolution coefficients
    """
    m = 2 * n - 1
    k64 = np.arange(m, dtype=np.int64)
    js = np.maximum(0, k64 - n + 1).astype(np.int32)
    je = np.minimum(n - 1, k64).astype(np.int32)
    counts = (je.astype(np.int64) - js.astype(np.int64) + 1).astype(np.int64)
    total = int(counts.sum())

    cum = np.cumsum(counts)
    starts = np.repeat(cum - counts, counts)
    local = np.arange(total, dtype=np.int64) - starts

    row = np.repeat(k64, counts).astype(np.int32)
    col_base = js[row.astype(np.int64)]
    col = (col_base + local).astype(np.int32)
    offsets = (row.astype(np.int64) - col.astype(np.int64)).astype(np.int32)

    A_off = csr_matrix((offsets, (row, col)), shape=(m, n), dtype=np.int32)
    edge_offsets = A_off.data.astype(np.int32, copy=True)
    A_template = A_off.astype(np.float64).copy()
    A_template.data.fill(0.0)

    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    return A_template, edge_offsets, c_obj, bounds, m


def solve_direction_lp(template, f, n, M):
    """Solve the LP direction problem for the given f and M."""
    A_template, edge_offsets, c_obj, bounds, m = template
    lp_scale = float(n)

    A = A_template.copy()
    A.data[:] = f[edge_offsets]
    A.data *= lp_scale

    b = np.full(m, M * lp_scale * lp_scale, dtype=np.float64)
    return linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")


def optimize_lp(n, f_init, time_budget, rng, template):
    """
    Run the LP-based iteration from one initial vector and return the best
    normalized vector and its normalized score.
    """
    f = normalize_sequence(f_init)
    if f is None:
        f = np.ones(n, dtype=np.float64) / float(n)

    if template is None:
        try:
            template = make_conv_template(n)
        except Exception:
            return f, score_normalized(f, n)

    best_f = f.copy()
    best_score = score_normalized(f, n)
    start_score = best_score

    deadline = time.monotonic() + max(0.0, float(time_budget))
    stall = 0
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        t0 = time.monotonic()

        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        if not np.isfinite(M) or M <= 0.0:
            break

        try:
            res = solve_direction_lp(template, f, n, M)
        except Exception as e:
            print(f"      n={n} linprog exception: {e}", flush=True)
            break

        x = getattr(res, "x", None)
        if x is None:
            stall += 1
            if stall > 12:
                break
            nf = mix_noisy(best_f, rng, 0.25)
            f = nf if nf is not None else best_f.copy()
            continue

        x = np.asarray(x, dtype=np.float64).ravel()
        if x.size != n:
            break
        x = np.where(np.isfinite(x), x, 0.0)
        x = np.maximum(x, 0.0)
        s_star = float(x.sum())
        if s_star <= 1e-300:
            stall += 1
            if stall > 12:
                break
            nf = mix_noisy(best_f, rng, 0.25)
            f = nf if nf is not None else best_f.copy()
            continue

        s_ratio = s_star / float(n)
        if s_ratio <= 1.0 + 1e-8:
            stall += 1
            if stall > 12:
                break
            nf = mix_noisy(best_f, rng, 0.25)
            f = nf if nf is not None else best_f.copy()
            continue

        g = x / s_star

        r = np.convolve(f, g)
        r_max = float(np.max(r))
        if (not np.isfinite(r_max)) or r_max <= 0.0:
            stall += 1
            if stall > 12:
                break
            nf = mix_noisy(best_f, rng, 0.25)
            f = nf if nf is not None else best_f.copy()
            continue

        # Need a strict bound r_max < M to guarantee descent.
        if r_max >= M * (1.0 - 1e-10):
            stall += 1
            if stall > 12:
                break
            nf = mix_noisy(best_f, rng, 0.25)
            f = nf if nf is not None else best_f.copy()
            continue

        S_eff = M / r_max
        if S_eff <= 1.0 + 1e-8:
            stall += 1
            if stall > 12:
                break
            nf = mix_noisy(best_f, rng, 0.25)
            f = nf if nf is not None else best_f.copy()
            continue

        G = float(np.max(np.convolve(g, g)))
        if not np.isfinite(G):
            stall += 1
            if stall > 12:
                break
            nf = mix_noisy(best_f, rng, 0.25)
            f = nf if nf is not None else best_f.copy()
            continue

        # Safe step from the bound:
        # H(t) = (1-t)^2 M + 2(1-t)t r_max + t^2 G <= M
        # H(t)-M = t*2*(r_max-M) + t^2*(G + M - 2*r_max)
        a_coef = 2.0 * (r_max - M)
        b_coef = G + M - 2.0 * r_max
        if b_coef > 1e-14:
            t_max = -a_coef / b_coef
        else:
            t_max = 1.0

        if (not np.isfinite(t_max)) or t_max <= 0.0:
            t_max = 1e-8
        base = min(1.0, 0.75 * t_max)
        if base < 1e-12:
            base = 1e-12

        raw_ts = [
            base,
            0.5 * base,
            0.1 * base,
            0.01 * base,
            min(base, 1e-3),
            min(base, 1e-5),
            min(base, 1e-7),
            1e-12,
        ]
        ts = []
        last = 2.0
        for tv in sorted((float(v) for v in raw_ts), reverse=True):
            if tv < 1e-14:
                continue
            if tv > 1.0:
                tv = 1.0
            if last - tv < 1e-13:
                continue
            last = tv
            ts.append(tv)
            if len(ts) >= 7:
                break

        best_cand_f = None
        best_cand_score = best_score
        best_t = None

        for tv in ts:
            f_new = (1.0 - tv) * f + tv * g
            f_new = np.maximum(f_new, 0.0)
            ssum = float(f_new.sum())
            if ssum <= 1e-300:
                continue
            f_new /= ssum
            sc = score_normalized(f_new, n)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_cand_f = f_new
                best_t = tv

        if best_cand_f is not None and best_cand_score < best_score:
            f = best_cand_f
            best_f = f.copy()
            best_score = best_cand_score
            stall = 0
            if iteration <= 5 or iteration % 10 == 0 or best_score < start_score - 0.05:
                print(
                    f"      n={n} iter={iteration} score={best_score:.6f} "
                    f"M={M:.6g} S_eff={S_eff:.4f} t={best_t:.6f} "
                    f"dt={time.monotonic()-t0:.2f}s",
                    flush=True,
                )
        else:
            stall += 1
            nf = mix_noisy(best_f, rng, 0.25)
            f = nf if nf is not None else best_f.copy()
            if iteration % 5 == 0:
                print(
                    f"      n={n} iter={iteration} no improvement, "
                    f"S_eff={S_eff:.4f}, best={best_score:.6f}",
                    flush=True,
                )
            if stall > 12:
                break

    return best_f, float(best_score)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a low-scoring nonnegative sequence.  The LP template is built
    for several lengths, and each length is started from random, given, and
    up-sampled best-sequence initializations.  The best sequence found before
    the time budget is returned.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    total_budget = max(1.0, float(budget_s))
    reserve = 18.0 if total_budget > 60.0 else 3.0
    start_time = time.monotonic()
    deadline = start_time + max(1.0, total_budget - reserve)

    best_f = None
    best_score = float("inf")
    given_f = None

    # If there is enough time, read and score the provided construction.
    if total_budget >= 30.0:
        try:
            raw = np.asarray(height_sequence_1, dtype=np.float64)
            given_f = normalize_sequence(raw)
            if given_f is not None:
                best_f = given_f.copy()
                try:
                    best_score = score_normalized(given_f, given_f.size)
                    print(
                        f"given sequence: n={given_f.size}, score={best_score:.6f}",
                        flush=True,
                    )
                except Exception:
                    best_score = float("inf")
        except Exception:
            given_f = None

    if best_f is None:
        best_f = np.ones(64, dtype=np.float64) / 64.0
        best_score = 2.0

    # Length/time plan.  Smaller lengths are tried first to find a good basin
    # quickly; larger lengths are used to improve further if time remains.
    plan = [
        (32, 10.0),
        (64, 25.0),
        (128, 50.0),
        (256, 110.0),
        (512, 300.0),
        (1024, 600.0),
        (1536, 350.0),
    ]

    for n, alloc in plan:
        rem = deadline - time.monotonic()
        if rem < 8.0:
            break
        alloc = min(alloc, rem - 3.0)
        if alloc <= 3.0:
            continue

        print(
            f"\n== n={n}, alloc={alloc:.1f}s, current_best={best_score:.6f}",
            flush=True,
        )

        try:
            template = make_conv_template(n)
        except MemoryError:
            print(f"  memory error building template for n={n}; skipping", flush=True)
            continue
        except Exception as e:
            print(f"  template build failed for n={n}: {e}", flush=True)
            continue

        starts = choose_starts(n, alloc, rng, best_f, given_f)
        if not starts:
            starts = [mix_noisy(np.ones(n, dtype=np.float64), rng, 1.0)]

        m = len(starts)
        if m == 1:
            weights = [1.0]
        elif m == 2:
            weights = [0.55, 0.45]
        elif m == 3:
            weights = [0.45, 0.35, 0.20]
        else:
            weights = [0.40, 0.30, 0.20, 0.10]

        for idx, init in enumerate(starts):
            rem = deadline - time.monotonic()
            if rem < 3.0:
                break
            tb = alloc * weights[idx]
            tb = min(tb, rem - 1.0)
            if tb < 2.0:
                continue

            print(f"  start {idx + 1}/{m}, budget={tb:.1f}s", flush=True)
            f_res, s_res = optimize_lp(n, init, tb, rng, template)

            if np.isfinite(s_res) and s_res < best_score:
                best_score = float(s_res)
                best_f = np.asarray(f_res, dtype=np.float64).copy()
                print(f"  new global best: n={best_f.size}, score={best_score:.6f}", flush=True)

        try:
            del template
        except Exception:
            pass

    # Final consolidation.
    if best_f is None:
        best_f = np.ones(64, dtype=np.float64) / 64.0

    seq = [float(x) for x in best_f]
    try:
        final_score = float(evaluate_sequence(seq))
    except Exception:
        final_score = float("inf")

    if not np.isfinite(final_score):
        try:
            if given_f is not None:
                seq = [float(x) for x in given_f]
                final_score = float(evaluate_sequence(seq))
        except Exception:
            pass

    if not np.isfinite(final_score):
        seq = [1.0 / 64.0] * 64

    print(
        f"\nFINAL: n={len(seq)}, local_best={best_score:.6f}, "
        f"final.evaluate={final_score:.6f}",
        flush=True,
    )
    return seq
