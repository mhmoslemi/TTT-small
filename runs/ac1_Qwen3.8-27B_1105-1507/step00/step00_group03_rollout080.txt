"""
Minimize  2*n*max(conv(a,a)) / (sum(a))**2  over non-negative step heights a.

Approach (scale-invariant, so we fix sum(a)=1 internally):
  1. Parametric scan of asymmetric "two-level step" and "ramp" families.
     Non-symmetry lowers the central convolution peak (b_{n-1}=sum a_i a_{n-1-i}),
     which is the key to beating the uniform solution (score 2.0).
  2. Constrained LP local search (state-of-the-art style): given f with M=max(f*f),
     solve  max sum(g)  s.t.  (f*g)_k <= M  for all k,  g>=0  (a real LP because
     f is fixed), rescale g, then line-search a convex update.
  3. Projected averaged sub-gradient descent on the max (averaged over near-active
     peaks to reduce oscillation) as a cheap refiner.
  4. Light random block-perturbation restarts to escape local minima.

All helpers are top-level; the authoritative score is obtained from the provided
global evaluate_sequence, while an identical FFT-based score drives the bulk search.
"""

import time
import numpy as np


# --------------------------------------------------------------------------- #
#  Core numerical helpers
# --------------------------------------------------------------------------- #
def fft_convolve(a, b):
    """Full linear convolution of two 1-D arrays via FFT."""
    size = len(a) + len(b) - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:size]


def internal_score(a):
    """Same quantity as evaluate_sequence for non-negative a with 0.01<=sum<=large.
    Scale invariant, so we use it for fast inner-loop feedback."""
    n = a.size
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    b = fft_convolve(a, a)
    return float(2.0 * n * np.max(b) / (s * s))


def clip_valid(a):
    """Project to the admissible box used by the evaluator."""
    return np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)


def normalize(a):
    s = np.sum(a)
    if s <= 0.0:
        return a.copy()
    return a / s


def build_conv_matrix(f):
    """Sparse matrix A ( (2n-1) x n ) with A[k, j] = f[k-j] if 0<=k-j<n else 0,
    which encodes the linear (in g) convolution constraints (f*g)_k = (A g)_k."""
    n = f.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        r0 = j
        r1 = min(2 * n - 1, j + n)
        m = r1 - r0
        if m > 0:
            A[r0:r1, j] = f[:m]
    return A


def lp_direction(f, M):
    """Solve  max sum(g) s.t. (f*g)_k <= M, g>=0.  Returns normalized g or None."""
    n = f.size
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"disp": False})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    g = np.clip(res.x, 0.0, None)
    gs = g.sum()
    if not np.isfinite(gs) or gs <= 1e-15:
        return None
    return g / gs


def line_search(f, g):
    """Best convex combination (1-t)f + t g by internal score."""
    base = internal_score(f)
    best_t = 0.0
    best_s = base
    t_grid = [0.1, 0.2, 0.3, 0.5, 0.6, 0.8, 0.9, 1.0]
    for t in t_grid:
        trial = (1.0 - t) * f + t * g
        trial = clip_valid(trial)
        s = internal_score(trial)
        if np.isfinite(s) and s < best_s:
            best_s = s
            best_t = t
    # refine around best_t
    if best_t > 0.0:
        lo = max(0.0, best_t - 0.15)
        hi = min(1.0, best_t + 0.15)
        for t in np.linspace(lo, hi, 25):
            trial = (1.0 - t) * f + t * g
            trial = clip_valid(trial)
            s = internal_score(trial)
            if np.isfinite(s) and s < best_s:
                best_s = s
                best_t = t
    return best_t, best_s


def grad_descent(f, budget_s):

    stall = 0

        else:
            f = f_new
            lr *= 0.9
            stall += 1
            if stall > 200 or lr < 1e-6:

def block_perturb(base, rng, strength):
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 400), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= np.exp(rng.normal(0.0, strength))
    r = int(rng.integers(0, max(1, n - width + 1)))
    w2 = int(rng.integers(max(1, n // 600), max(2, n // 60)))
    s2 = min(n, r + w2)
    cand[r:s2] *= np.exp(rng.normal(0.0, strength))
    return clip_valid(cand)

def two_level_sequence(n, k, t):
    """First k entries = x, rest = y = t*x, normalised to sum=1."""
    x = 1.0 / (k + (n - k) * t)
    a = np.empty(n)
    a[:k] = x
    a[k:] = t * x
    return a


def linear_ramp_sequence(n, c):
    """a_i proportional to (1 + c*(i/(n-1))) , clipped non-negative, normalised."""
    idx = np.linspace(0.0, 1.0, n)
    a = 1.0 + c * idx
    a = np.clip(a, 1e-9, None)
    return a / a.sum()


def geometric_sequence(n, r):
    a = r ** np.arange(n)
    a = np.clip(a, 1e-12, None)
    return a / a.sum()


def scan_parametric(n, budget_s, rng):
    """Coarse scan over the useful families; returns best sequence + score."""
    deadline = time.monotonic() + max(5.0, budget_s)
    best_a = np.ones(n) / n
    best_s = internal_score(best_a)

    # two-level step: scan k (fraction of length) and ratio t
    fracs = np.array([0.05, 0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40,
                      0.45, 0.50, 0.55, 0.60, 0.65, 0.70, 0.75, 0.80, 0.85, 0.90])
    ratios = np.array([0.2, 0.35, 0.5, 0.65, 0.8, 1.0, 1.25, 1.6, 2.2, 3.0, 4.5])
    evaluated = 0
    for fr in fracs:
        k = int(round(fr * n))
        k = min(max(1, k), n - 1)
        for t in ratios:
            a = two_level_sequence(n, k, t)
            s = internal_score(a)
            if np.isfinite(s) and s < best_s:
                best_s = s
                best_a = a.copy()
            evaluated += 1
            if time.monotonic() > deadline:
                break
        if time.monotonic() > deadline:
            break

    # linear ramps (both slopes)
    for c in np.array([-1.2, -0.9, -0.6, -0.4, -0.25, 0.25, 0.4, 0.6, 0.9, 1.2]):
        a = linear_ramp_sequence(n, c)
        s = internal_score(a)
        if np.isfinite(s) and s < best_s:
            best_s = s
            best_a = a.copy()
        if time.monotonic() > deadline:
            break

    # geometric (front and back)
    for r in np.array([0.97, 0.93, 0.88, 1.03, 1.08, 1.12]):
        a = geometric_sequence(n, r)
        s = internal_score(a)
        if np.isfinite(s) and s < best_s:
            best_s = s
            best_a = a.copy()
        if time.monotonic() > deadline:
            break

    # a few random-ish sparse-top candidates (unlikely to win, cheap)
    for _ in range(40):
        a = normalize(np.clip(rng.lognormal(0.0, 0.6, n), 0.0, None))
        s = internal_score(a)
        if np.isfinite(s) and s < best_s:
            best_s = s
            best_a = a.copy()

    print(f"[parametric] evaluated={evaluated} best={best_s:.6f}")
    return best_a, best_s
