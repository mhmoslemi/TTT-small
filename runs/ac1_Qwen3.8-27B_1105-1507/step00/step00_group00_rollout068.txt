def project_simplex(v):
    n = len(v)

def project_simplex(v):
    """Euclidean projection of v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 1:
        return np.ones(1, dtype=np.float64)
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (sv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.where(cond)[0][-1])
    theta = (sv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a, b):
    """Linear convolution via FFT."""
    n = a.size
    m = b.size
    need = n + m - 1
    N = 1
    while N < need:
        N <<= 1
    fa = np.fft.fft(a, N)
    fb = np.fft.fft(b, N)
    return np.real(np.fft.ifft(fa * fb))[:need]


def smooth_value_and_grad(a, tau):
    """Return (smoothed-peak-value, gradient) of the self-convolution peak."""
    conv = fast_convolve(a, a)
    m = np.max(conv)
    z = (conv - m) / tau
    ez = np.exp(z)
    ez = ez / np.sum(ez)
    val = m + tau * np.log(np.sum(np.exp(z)) * np.exp(m / tau)) if m / tau > 700 else m + tau * np.log(np.sum(ez) * np.exp(m / tau) / np.exp(m / tau))
    # simpler numerically-stable form
    val = m + tau * (np.log(np.sum(ez)) + (m - m))  # mean of shifted exp avg
    # correct stable evaluation:
    val = m + tau * np.log(np.sum(np.exp(z)) / np.exp(z).size) * np.exp(z).size / np.exp(z).size + 0.0
    val = m + tau * np.log(np.mean(np.exp(z))) + 0.0
    # gradient = 2 * correlate(softmax_weights, a)
    grad = 2.0 * np.correlate(ez, a, mode='valid')
    return val, grad


def smooth_gd(a0, budget, tau_start, tau_end, alpha0, seed=0):
    """Projected gradient descent on the smoothed peak convolution."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = project_simplex(a)
    n = a.size
    start = time.monotonic()
    deadline = start + float(budget)

    cur_val, cur_grad = smooth_value_and_grad(a, tau_start)
    best_a = a.copy()
    best_peak = float(np.max(fast_convolve(a, a)))
    alpha = float(alpha0)
    it = 0
    stuck = 0

    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-6, float(budget) * 0.85))
        tau = tau_start * (tau_end / tau_start) ** frac

        accept = False
        best_cand = None
        best_cval = cur_val
        step = alpha
        for _bt in range(6):
            cand = project_simplex(a - step * cur_grad)
            cval, _ = smooth_value_and_grad(cand, tau)
            if cval <= cur_val + 1e-12:
                accept = True
                best_cand = cand
                best_cval = cval
                break
            if cval < best_cval:
                best_cval = cval
                best_cand = cand
            step *= 0.5
        if not accept:
            # take the best tried step anyway if it is not clearly worse
            if best_cand is not None and best_cval <= cur_val + 1e-9:
                a = best_cand
                alpha = max(alpha * 0.9, 1e-4)
            else:
                alpha *= 0.6
                if alpha < 1e-5:
                    # diversification kick to escape a flat basin
                    w = max(2, n // 50)
                    s = int(rng.integers(0, n - w))
                    a[s:s + w] *= float(np.exp(rng.normal(0.0, 0.6)))
                    a = project_simplex(a)
                continue
        a = best_cand
        if accept and best_cval < cur_val - 1e-12:
            alpha = min(alpha * 1.05, 1.0)
            stuck = 0
        else:
            stuck += 1

        if it % 4 == 0:
            peak = float(np.max(fast_convolve(a, a)))
            if peak < best_peak - 1e-12:
                best_peak = peak
                best_a = a.copy()
        cur_val, cur_grad = smooth_value_and_grad(a, tau)

    peak = float(np.max(fast_convolve(a, a)))
    if peak < best_peak:
        best_peak = peak
        best_a = a.copy()
    return best_a, best_peak


def lp_best_response(a):
    """Solve max sum(g)  s.t.  g>=0, (a*g)[k] <= max(a*a) for all k."""
    n = a.size
    conv = fast_convolve(a, a)
    M = float(np.max(conv))
    if M <= 0.0:
        return a.copy()

    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(k - i)
            vals.append(a[i])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.asarray(res.x, dtype=np.float64)
    except Exception:
        pass
    return None


def lp_polish(a, budget, seed=0):
    """Blend LP best-response steps into the iterate, tracking best sequence."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    S = float(np.sum(a))
    deadline = time.monotonic() + float(budget)
    best_a = a.copy()
    best_peak = float(np.max(fast_convolve(a, a)))
    t = 0.25
    for _ in range(40):
        if time.monotonic() > deadline:
            break
        g = lp_best_response(a)
        if g is None:
            break
        sg = float(np.sum(g))
        if sg <= 0.0:
            break
        g = g * (S / sg)
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, 0.0)
        sa = float(np.sum(a))
        if sa <= 0.0:
            break
        a = a * (S / sa)
        peak = float(np.max(fast_convolve(a, a)))
        if peak < best_peak - 1e-12:
            best_peak = peak
            best_a = a.copy()
        t *= 0.9
    return best_a, best_peak


def build_inits(rng, parent):
    """Return a list of (n, a) candidates with a >= 0, sum(a) = 1."""
    cands = []
    if parent is not None and parent.size >= 20:
        p = np.maximum(parent, 0.0)
        sp_ = float(np.sum(p))
        if sp_ > 0.01:
            cands.append((p.size, p / sp_, 'parent'))

    ns = [1500, 2500, 4000]
    for n in ns:
        # uniform
        cands.append((n, np.ones(n) / n, 'uniform'))
        # linear decreasing / increasing (antisymmetric tilt)
        r = np.linspace(1.0, 0.0, n)
        cands.append((n, np.maximum(r, 1e-3), 'dec'))
        cands.append((n, np.maximum(1.0 - r, 1e-3), 'inc'))
        # sinusoidal (odd profile)
        i = np.arange(n)
        base = 1.0 + 0.85 * np.sin(np.pi * (i + 0.5) / n)
        base = np.maximum(base, 1e-3)
        cands.append((n, base, 'sino'))
        base2 = 1.0 + 0.6 * np.cos(2.0 * np.pi * (i + 0.5) / n * 1.0) + \
            0.3 * np.cos(4.0 * np.pi * (i + 0.5) / n * 1.0)
        base2 = np.maximum(base2, 1e-3)
        cands.append((n, base2, 'sino2'))
        # random dirichlet draws
        cands.append((n, rng.dirichlet(np.full(n, 0.5)), 'rand_lo'))
        cands.append((n, rng.dirichlet(np.full(n, 2.0)), 'rand_hi'))
    out = []
    for n, a, tag in cands:
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 1e-6)
        a = a / float(np.sum(a))
        out.append((n, a, tag))
    return out

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Search for the best coefficient sequence; always return a valid one."""
    total_start = time.monotonic()
    total_deadline = total_start + float(budget_s) - 8.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
        parent = np.maximum(parent, 0.0)
    except Exception:
        parent = None

    best_sequence = None
    best_score = np.inf

    def record(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = float(np.sum(a))
        if s < 0.01 or not np.all(np.isfinite(a)):
            return
        try:
            score = float(evaluate_sequence(a.tolist()))  # noqa: F821
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  NEW BEST  score=%.6f  n=%d  t=%.1fs" %
                  (best_score, a.size, time.monotonic() - total_start),
                  flush=True)

    cands = build_inits(rng, parent)
    # put parent first, then interleave to keep n balanced
    if cands and not cands[0][2].startswith('parent'):
        cands = sorted(cands, key=lambda x: (0 if x[0] < 3000 else 1))

    record(np.ones(500) / 500.0)  # guarantee a valid baseline (== 2.0)
    if parent is not None and parent.size >= 1:
        record(parent)

    n_cand = len(cands)
    phase1_frac = 0.62
    phase1_budget = (float(budget_s) - 8.0) * phase1_frac
    per_cand = max(4.0, phase1_budget / max(1, n_cand))

    print("Phase 1: %d candidates, ~%.1fs each" % (n_cand, per_cand), flush=True)
    scored = []
    for idx, (n, a0, tag) in enumerate(cands):
        if time.monotonic() > total_deadline - 30.0:
            break
        remaining = total_deadline - time.monotonic()
        alloc = min(per_cand, remaining * 0.4)
        if alloc < 3.0:
            break
        tau_start = 5.0 / n
        tau_end = 0.25 / n
        a_opt, peak = smooth_gd(a0, alloc, tau_start, tau_end, alpha0=0.4,
                                seed=seed + idx)
        a_opt = project_simplex(a_opt)
        try:
            sc = float(evaluate_sequence(a_opt.tolist()))  # noqa: F821
        except Exception:
            sc = np.inf
        scored.append((sc if np.isfinite(sc) else 1e9, n, a_opt, tag))
        record(a_opt)
        print("  cand %2d tag=%-7s n=%4d  eval=%.5f  maxconv=%.4g  t=%.1fs" %
              (idx, tag, n, sc, peak, time.monotonic() - total_start),
              flush=True)

    if scored:
        scored.sort(key=lambda x: x[0])

    # ---------------- Phase 2: extend + LP polish the top candidates --------
    print("Phase 2: deep optimization", flush=True)
    ranked = scored[:3] if scored else []
    # make sure the parent-like best found so far is considered
    for k, (sc, n, a0, tag) in enumerate(ranked):
        if time.monotonic() > total_deadline - 12.0:
            break
        remaining = total_deadline - time.monotonic() - 8.0
        if remaining < 8.0:
            break
        tau_start = 2.0 / n
        tau_end = 0.12 / n
        a_ext, peak = smooth_gd(a0, remaining * 0.55, tau_start, tau_end,
                                alpha0=0.4, seed=seed + 100 + k)
        record(project_simplex(a_ext))
        if time.monotonic() < total_deadline - 10.0:
            a_lp, peak_lp = lp_polish(project_simplex(a_ext),
                                      remaining * 0.40, seed=seed + 200 + k)
            record(project_simplex(a_lp))

    # ---- Phase 3: if still exactly ~2.0, try aggressive tilted restarts ----
    if best_score is None or best_score > 1.999 or time.monotonic() < total_deadline - 20.0:
        n_try = 3000
        for r2 in range(6):
            if time.monotonic() > total_deadline - 10.0:
                break
            prof = 1.0 - (1.9 - 0.3 * r2) * (np.arange(n_try) / (n_try - 1.0))
            a0 = np.maximum(prof, 1e-3)
            a0 = a0 / float(np.sum(a0))
            tau_start = 3.0 / n_try
            tau_end = 0.2 / n_try
            a_opt, _ = smooth_gd(a0, max(3.0, (total_deadline - time.monotonic() - 5.0) / 2.0),
                                 tau_start, tau_end, alpha0=0.4, seed=seed + 500 + r2)
            record(project_simplex(a_opt))

    # ------------------------- final fallback ------------------------------
    if best_sequence is None:
        if parent is not None and parent.size > 0:
            best_sequence = np.maximum(parent, 0.0)
        else:
            best_sequence = np.ones(500, dtype=np.float64)

    best_sequence = np.asarray(best_sequence, dtype=np.float64)
    best_sequence = np.maximum(best_sequence, 0.0)
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    print("FINAL  n=%d  score=%.6f  t=%.1fs" %
          (best_sequence.size, best_score, time.monotonic() - total_start),
          flush=True)
    return [float(x) for x in best_sequence]
