"""
Minimizes 2*n*max(convolution)/sum^2 using iterative multiplicative updates
with active-set identification and multiple diverse initializations.

import numpy as np
import time

def compute_objective(f):
    """Compute the objective value for a normalized sequence (sum=1)."""
    n = len(f)
    c = np.convolve(f, f)
    M = c.max()
    return 2.0 * n * M

def multiplicative_update(f, eta=0.5, top_k_frac=0.1):
    """One step of the multiplicative update."""
    n = len(f)
    c = np.convolve(f, f)
    M = c.max()
    
    # Find active set
    k = max(10, int(n * top_k_frac))
    threshold = np.partition(c, -k)[-k]  # k-th largest value
    I = (c >= threshold).astype(np.float64)
    
    # Compute pressure
    pressure = np.correlate(I, f, mode='valid')
    
    # Ensure pressure is positive
    pressure = np.maximum(pressure, 1e-15)
    
    # Update
    target = pressure.mean()
    f_new = f * np.power(target / pressure, eta)
    
    # Clip and renormalize
    f_new = np.clip(f_new, 0, None)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    f_new = f_new / s
    
    return f_new

    """Compute correlate(a, v, mode='valid') using FFT."""
    # a has length >= v
    # Result has length len(a) - len(v) + 1

def compute_obj(f):

def multiplicative_step(f, top_k_frac=0.08, eta=0.5):

def lp_step(f, M, top_k=200):
    """
    LP relaxation: find g>=0 maximizing sum(g) s.t. (f*g)[k] <= M for top-k k.
    Returns g normalized to sum=1, or None if LP fails.
    """
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    n = len(f)
    c = np.convolve(f, f)
    k_count = min(top_k, 2 * n - 1)
    active_idx = np.argsort(c)[-k_count:]

    rows = []
    cols = []
    vals = []
    for ki, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi < lo:
            continue
        rows.extend([ki] * (hi - lo + 1))
        cols.extend(range(lo, hi + 1))
        vals.extend(f[lo:hi + 1].tolist())

    if len(rows) == 0:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(k_count, n))
    c_obj = -np.ones(n)
    b_ub = np.full(k_count, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success and res.x is not None:
        g = np.asarray(res.x, dtype=np.float64)
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None

    stagnant = 0

        c = np.convolve(f, f)
        M = c.max()
        g = lp_step(f, M, top_k=min(300, n))
        if g is None:
            break
        # Try blending
        accepted = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            f_new = (1 - t) * f + t * g
            f_new = np.clip(f_new, 0, None)
            fs = f_new.sum()
            if fs < 1e-15:
                continue
            f_new = f_new / fs
            obj = compute_obj(f_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f_new.copy()
                f = f_new
                accepted = True
                stagnant = 0
                break
        if not accepted:
            stagnant += 1
            if stagnant > 5:
                # Perturb and retry
                f = best_f * np.exp(rng.normal(0, 0.1, n))
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs > 1e-15:
                    f = f / fs
                stagnant = 0

    return best_f, best_obj


def init_uniform(n):
    return np.ones(n) / n


def init_sinusoidal(n, freq, amp=0.9, seed=0):
    rng = np.random.default_rng(seed)
    i = np.arange(n)
    phase = rng.uniform(0, 2 * np.pi)
    f = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
    f = np.clip(f, 0, None)
    s = f.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return f / s


def init_dual_sinusoidal(n, freq1, freq2, seed=0):
    rng = np.random.default_rng(seed)
    i = np.arange(n)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    f = 1.0 + 0.6 * np.cos(2 * np.pi * freq1 * i / n + p1) + 0.5 * np.cos(2 * np.pi * freq2 * i / n + p2)
    f = np.clip(f, 0, None)
    s = f.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return f / s


def init_random(n, seed=0):
    rng = np.random.default_rng(seed)
    f = rng.random(n)
    s = f.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return f / s


def init_block(n, nblocks, seed=0):
    rng = np.random.default_rng(seed)
    f = np.zeros(n)
    bs = n // nblocks
    for b in range(nblocks):
        st = b * bs
        en = st + bs if b < nblocks - 1 else n
        f[st:en] = rng.uniform(0.3, 2.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return f / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Search for the best non-negative sequence minimizing 2*n*max(conv)/sum^2.
    Uses diverse initializations, multiplicative updates, and LP refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 5.0)

    # Determine n
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 4096
    n = max(256, min(n, 10000))
    print(f"Sequence length n={n}")

    # Build initializations
    inits = []
    inits.append(("uniform", init_uniform(n)))
    inits.append(("sin_3", init_sinusoidal(n, 3, 0.9, 1)))
    inits.append(("sin_7", init_sinusoidal(n, 7, 0.9, 2)))
    inits.append(("sin_13", init_sinusoidal(n, 13, 0.9, 3)))
    inits.append(("sin_21", init_sinusoidal(n, 21, 0.85, 4)))
    inits.append(("dual_3_11", init_dual_sinusoidal(n, 3, 11, 5)))
    inits.append(("dual_5_17", init_dual_sinusoidal(n, 5, 17, 6)))
    inits.append(("rand", init_random(n, seed)))
    inits.append(("block_4", init_block(n, 4, seed)))
    inits.append(("block_8", init_block(n, 8, seed + 1)))

    try:
        p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, None)
        ps = p.sum()
        if ps > 0.01:
            inits.append(("parent", p / ps))
    except Exception:
        pass

    best_f = None
    best_score = float('inf')

    # Phase 1: Short exploratory searches from multiple inits
    n_inits = len(inits)
    t_per_init = min(25.0, (budget_s - 120) / max(1, n_inits))
    print(f"Phase 1: {n_inits} initializations, ~{t_per_init:.1f}s each")

    for name, f0 in inits:
        rem = deadline - time.monotonic()
        if rem < 3.0:
            break
        t_alloc = min(t_per_init, rem)
        f_res, _ = local_search_mult(f0, t_alloc, seed=hash(name) % 50000)
        score = evaluate_sequence(f_res.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = f_res.copy()
            print(f"  [{name}] score={score:.6f}")

    print(f"After Phase 1: best={best_score:.6f}")

    # Phase 2: Extended multiplicative search from best
    rem = deadline - time.monotonic()
    if rem > 15 and best_f is not None:
        t_alloc = rem * 0.5
        print(f"Phase 2: extended mult search, {t_alloc:.1f}s")
        f_res, _ = local_search_mult(best_f, t_alloc, seed=seed + 111)
        score = evaluate_sequence(f_res.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = f_res.copy()
            print(f"  Phase 2 best: {score:.6f}")

    # Phase 3: LP-based refinement
    rem = deadline - time.monotonic()
    if rem > 10 and best_f is not None:
        print(f"Phase 3: LP refinement, {rem:.1f}s available")
        try:
            f_res, _ = lp_local_search(best_f, rem * 0.6, seed=seed + 222)
            score = evaluate_sequence(f_res.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_f = f_res.copy()
                print(f"  LP best: {score:.6f}")
        except Exception as e:
            print(f"  LP failed: {e}")

    # Phase 4: Final multiplicative polish
    rem = deadline - time.monotonic()
    if rem > 5 and best_f is not None:
        print(f"Phase 4: final polish, {rem:.1f}s")
        f_res, _ = local_search_mult(best_f, rem, seed=seed + 333)
        score = evaluate_sequence(f_res.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = f_res.copy()
            print(f"  Final: {score:.6f}")

    if best_f is None:
        best_f = init_uniform(n)

    print(f"FINAL SCORE: {best_score:.6f}")
    return [float(x) for x in best_f]
