"""
Minimize  score(a) = 2*n*max_k(a*a)_k / (sum a)^2
for a non-negative step function, using:
  * the closed-form "flat-aperiodic-autocorrelation" seed  b_k = binom(2k,k)/4^k
    (stably generated by the recurrence b_k = b_{k-1}*(2k-1)/(2k));
  * an exact LP direction-finding step ( scipy HiGHS ) that maximises sum(g)
     subject to the cross-correlation  (a*g)_k <= M  and  g >= 0,
     followed by a line-search over the convex blend  (1-t)a + t*gnorm ;
  * multiplicative log-normal perturbations on stalls for diversity;
  * a final projected log-sum-exp (soft-max) gradient polish.
The method is scale free so we always normalise to sum(a)=1, and every accepted
move is verified against the authoritative evaluate_sequence().
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


# ---------------------------------------------------------------------------
# Helper functions (all top level, no closures / lambdas)
# ---------------------------------------------------------------------------
def ideal_sequence(n):
    """b_k = binom(2k,k)/4^k, via stable recurrence; normalised to sum 1."""
    b = np.empty(n, dtype=np.float64)
    b[0] = 1.0
    for k in range(1, n):
        b[k] = b[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    b = np.clip(b, 0.0, None)
    s = b.sum()
    if s <= 0.0:                      # should never happen
        b = np.ones(n, dtype=np.float64)
        return b / n
    return b / s


def build_cross_matrix(n, a):
    """A with A[k, j] = a[k-j]  (0 otherwise), shape (2n-1, n)."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_direction(n, a, M):
    """max sum(g) s.t. (a*g)_k <= M, g >= 0.  Returns normalised g or None."""
    c = -np.ones(n, dtype=np.float64)
    A = build_cross_matrix(n, a)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"presolve": True})
    except Exception:
        return None
    if res is None or not res.success or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if not np.all(np.isfinite(g)):
        return None
    g = np.clip(g, 0.0, None)
    gs = g.sum()
    if gs <= 1e-12:
        return None
    return g / gs


def blend_best(a, gnorm, ts, cur_score):
    """Line search over t in ts.  Returns (new_a, new_score, improved_flag)."""
    best_cand = None
    best_sc = cur_score
    for t in ts:
        cand = (1.0 - t) * a + t * gnorm
        sc = evaluate_sequence(cand.tolist())
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_cand = cand
    if best_cand is None:
        return a, cur_score, False
    return best_cand, best_sc, True


def perturb(a, rng, sigma):
    """Multiplicative log-normal perturbation, renormalised to sum 1."""
    w = np.exp(rng.normal(0.0, sigma, size=a.size))
    cand = a * w
    cand = np.clip(cand, 0.0, None)
    s = cand.sum()
    if s <= 0.0:
        return a
    return cand / s


def proj_simplex(v):
    """Project vector onto {x >= 0, sum x = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_polish(a, deadline, rng, max_steps=4000):
    """Projected log-sum-exp gradient descent, line search on true score."""
    a = a.copy()
    n = a.size
    cur = evaluate_sequence(a.tolist())
    if not np.isfinite(cur):
        return a, cur, False
    tau = 6.0
    lrs = [4.0 / n, 2.0 / n, 1.0 / n, 0.5 / n, 0.25 / n, 0.1 / n]
    improved = False
    for _ in range(max_steps):
        if time.monotonic() >= deadline:
            break
        aa = np.convolve(a, a)
        e = np.exp(tau * (aa - aa.max()))
        w = e / e.sum()
        # grad_j = 2 * sum_m a_m * w_{j+m}
        grad = np.empty(n, dtype=np.float64)
        for j in range(n):
            grad[j] = 2.0 * np.dot(a, w[j:j + n])
        moved = False
        for lr in lrs:
            v = proj_simplex(a - lr * grad)
            sc = evaluate_sequence(v.tolist())
            if np.isfinite(sc) and sc < cur:
                a = v
                cur = sc
                improved = True
                moved = True
                break
        if not moved:
            # occasionally break symmetry to escape a plateau
            v = perturb(a, rng, 0.15)
            sc = evaluate_sequence(v.tolist())
            if np.isfinite(sc) and sc < cur:
                a = v
                cur = sc
                improved = True
            else:
                tau = max(2.0, tau * 0.7)
                if tau < 2.1 and rng.random() < 0.5:
                    break
    return a, cur, improved


def lp_phase(start_a, n, budget, global_best_holder):
    """Run the LP direction + line-search loop; returns (best_a, best_score)."""
    a = start_a.copy()
    cur = evaluate_sequence(a.tolist())
    best_a = a.copy()
    best_sc = cur
    ts = [1.0, 0.85, 0.7, 0.5, 0.35, 0.2]
    rng = global_best_holder["rng"]
    stall = 0
    iters = 0
    while time.monotonic() < budget:
        M = float(np.max(np.convolve(a, a)))
        if not np.isfinite(M) or M <= 0.0:
            break
        g = lp_direction(n, a, M)
        if g is None:
            stall += 1
        else:
            cand, sc, imp = blend_best(a, g, ts, cur)
            if imp:
                a = cand
                cur = sc
                stall = 0
                iters += 1
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    global_best_holder["score"] = best_sc
                    global_best_holder["seq"] = best_a.copy()
            else:
                stall += 1
        if stall >= 25:
            cand = perturb(a, rng, 0.20 + 0.10 * min(1.0, stall / 100.0))
            sc = evaluate_sequence(cand.tolist())
            if np.isfinite(sc) and sc < cur:
                a = cand
                cur = sc
            stall = 0
        if iters and iters % 40 == 0:
            print("   n=%d  it=%d  score=%.6f" % (n, iters, best_sc))
        if stall > 400:
            break
    return best_a, best_sc
