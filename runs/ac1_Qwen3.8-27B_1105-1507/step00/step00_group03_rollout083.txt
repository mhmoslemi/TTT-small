```
def _convolve(a):
```

```
def _valid_correlate(p,a):

    P=np.fft.rfft(p,L)
```

```
def _power_seq(n, alpha):
    i=np.arange(1,n+1,dtype=np.float64)
    u=i**(-alpha)

def _binomial_sqrt_seq(n):
    c=np.ones(n,dtype=np.float64)
    for i in range(1,n):
       c[i]=c[i-1]*((2*i-1.0)/(2*i))
    c/=c.sum()
    return c
def _blend_seq(a,b,w):
    s=w*a+(1.0-w)*b
    s=np.maximum(s,0.0)
    total=s.sum()
    if total<=0: return a/a.sum()
    return s/total
```

```
def _dilate_to(base, target):
    base=np.asarray(base,dtype=np.float64).ravel()
    m=base.size
    if target != m:
       widths=np.full(m, target//m, dtype=np.int64)
       rem=target % m
       if rem>0: widths[:rem]+=1
       out=np.empty(target,dtype=np.float64)
       pos=0
       for i in range(m):
          w=int(widths[i])
          out[pos:pos+w]=base[i]/float(w)
          pos+=w
       return out
    return base.copy()
```

```python
"""
Search for non-negative sequences minimizing
    2 * n * max(conv(s, s)) / (sum(s))**2.

The method is a multi-start multiplicative projected-gradient search in log
amplitude space.  It uses analytic "binomial square-root" seeds, optimizes
short base patterns, dilates them to a long candidate, and then refines promising
long candidates directly.  All sequences are normalized internally and returned
as non-negative floats.
"""

import numpy as np
import time


def _normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 0.0:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _uniform(n):
    return np.ones(n, dtype=np.float64) / float(n)


def _power_seq(n, alpha):
    x = np.arange(1, n + 1, dtype=np.float64)
    u = x ** (-float(alpha))
    s = float(u.sum())
    if s <= 0.0 or not np.isfinite(s):
        return _uniform(n)
    return u / s


def _binomial_sqrt_seq(n):
    # Coefficients of (1-z)^(-1/2): binom(2k,k)/4^k.
    c = np.ones(n, dtype=np.float64)
    for i in range(1, n):
        c[i] = c[i - 1] * float(2 * i - 1) / float(2 * i)
    s = float(c.sum())
    if s <= 0.0 or not np.isfinite(s):
        return _uniform(n)
    return c / s


def _blend_seq(a, b, w):
    w = float(w)
    s = w * np.asarray(a, dtype=np.float64) + (1.0 - w) * np.asarray(b, dtype=np.float64)
    s = np.clip(s, 0.0, 1000.0)
    s = np.where(np.isfinite(s), s, 0.0)
    total = float(s.sum())
    if total <= 0.0:
        a = np.asarray(a, dtype=np.float64)
        ta = float(a.sum())
        if ta <= 0.0:
            return _uniform(a.size)
        return a / ta
    return s / total


def _dilate_to(base, target):
    base = np.asarray(base, dtype=np.float64).ravel()
    m = base.size
    if target == m:
        return base.copy()
    if m == 0 or target <= 0:
        return _uniform(target)

    widths = np.full(m, target // m, dtype=np.int64)
    rem = target % m
    if rem > 0:
        widths[:rem] += 1

    out = np.empty(target, dtype=np.float64)
    pos = 0
    for i in range(m):
        w = int(widths[i])
        out[pos:pos + w] = base[i] / float(w)
        pos += w
    return out


def _convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)

    if n <= 256:
        c = np.convolve(a, a, mode="full")
    else:
        need = 2 * n - 1
        L = 1
        while L < need:
            L <<= 1
        f = np.fft.rfft(a, L)
        c = np.fft.irfft(f * f, L)[:need]

    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c


def _valid_correlate(p, a):
    p = np.asarray(p, dtype=np.float64).ravel()
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)

    if n <= 256:
        r = np.correlate(p, a, mode="valid")
        return np.where(np.isfinite(r), r, 0.0)

    need = 3 * n - 2
    L = 1
    while L < need:
        L <<= 1

    P = np.fft.rfft(p, L)
    Ar = np.fft.rfft(a[::-1], L)
    h = np.fft.irfft(P * Ar, L)[:need]
    r = h[n - 1:n - 1 + n]
    return np.where(np.isfinite(r), r, 0.0)


def _from_loga(loga):
    loga = np.asarray(loga, dtype=np.float64).ravel()
    x = np.clip(loga, -60.0, 60.0)
    a = np.exp(x)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        a = np.ones_like(a)
        s = float(a.size)
    return a / s


def _log_from_array(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.zeros(a.size, dtype=np.float64)
    a = a / s
    return np.log(np.maximum(a, 1e-300))


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.inf
    c = _convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * a.size * mx / (s * s))


def _optimize_loga(loga, start_time, end_time, rng, q0=6.0, rate0=0.18):
    n = int(np.asarray(loga).size)
    if n <= 0:
        return loga, np.inf, np.array([], dtype=np.float64)

    if start_time is None:
        start_time = time.monotonic()
    if end_time is None or end_time <= start_time:
        end_time = start_time + 1e-9
    if rng is None:
        rng = np.random.default_rng(0)

    loga = np.asarray(loga, dtype=np.float64).ravel().copy()
    a = _from_loga(loga)
    c = _convolve(a)
    score = float(2.0 * n * np.max(c))

    if not np.isfinite(score):
        a = _uniform(n)
        loga = _log_from_array(a)
        c = _convolve(a)
        score = float(2.0 * n * np.max(c))

    best_score = float(score)
    best_a = a.copy()
    best_loga = loga.copy()

    rate = float(rate0)
    q = float(q0)
    no_accept = 0
    last_best_time = time.monotonic()
    q_choices = (2.0, 4.0, 6.0, 10.0, 20.0, 50.0, 100.0)

    while time.monotonic() < end_time:
        if score > best_score * 1.05 and best_score < np.inf:
            a = best_a.copy()
            loga = best_loga.copy()
            c = _convolve(a)
            score = float(best_score)
            rate = 0.15
            q = max(2.0, min(20.0, q * 0.8))
            last_best_time = time.monotonic()

        if no_accept > 40 and time.monotonic() - last_best_time > 10.0:
            sigma = 0.1 + 0.25 * min(1.0, float(no_accept - 40) / 100.0)
            trial_loga = best_loga + rng.normal(0.0, sigma, n)
            trial_a = _from_loga(trial_loga)
            trial_c = _convolve(trial_a)
            trial_score = float(2.0 * n * np.max(trial_c))
            if np.isfinite(trial_score):
                if trial_score < score:
                    a = trial_a
                    loga = _log_from_array(trial_a)
                    c = trial_c
                    score = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_a = trial_a.copy()
                    best_loga = _log_from_array(trial_a)
                    last_best_time = time.monotonic()
            q = min(80.0, 2.0 + 0.8 * q)
            q = float(q_choices[int(rng.integers(0, len(q_choices)))])
            rate = float(rng.uniform(0.08, 0.25))
            no_accept = 0
            continue

        ref = float(np.max(c))
        if not np.isfinite(ref) or ref <= 0.0:
            break

        y = c / ref
        with np.errstate(under="ignore", over="ignore"):
            p = np.power(y, q)
        p = np.where(np.isfinite(p), p, 0.0)
        ps = float(p.sum())
        if ps <= 0.0 or not np.isfinite(ps):
            p = y.copy()
            ps = float(p.sum())
        if ps <= 0.0 or not np.isfinite(ps):
            p = np.zeros_like(c)
            p[0] = 1.0
            ps = 1.0
        p = p / ps

        g = 2.0 * _valid_correlate(p, a)
        g = np.where(np.isfinite(g), g, 0.0)
        g = g - float(g.mean())
        maxg = float(np.max(np.abs(g)))
        if not np.isfinite(maxg) or maxg == 0.0:
            break

        direction = -g / maxg
        accepted = False

        for trial_factor in (1.0, 0.5, 0.25, 0.1):
            r = rate * trial_factor
            if r <= 0.0 or not np.isfinite(r):
                continue
            delta = r * direction
            dmax = float(np.max(np.abs(delta)))
            if dmax > 0.35:
                delta *= 0.35 / dmax

            new_loga = loga + delta
            new_a = _from_loga(new_loga)
            new_c = _convolve(new_a)
            new_score = float(2.0 * n * np.max(new_c))
            if not np.isfinite(new_score):
                continue

            if new_score < score - 1e-14:
                a = new_a
                loga = _log_from_array(new_a)
                c = new_c
                score = new_score
                rate = min(0.45, r * 1.05)
                no_accept = 0
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                    best_loga = loga.copy()
                    last_best_time = time.monotonic()
                accepted = True
                break

        if not accepted:
            no_accept += 1
            rate *= 0.75
            if rate < 1e-4 or no_accept > 14:
                q = float(q_choices[int(rng.integers(0, len(q_choices)))])
                rate = float(rng.uniform(0.05, 0.25))
                no_accept = 0
        else:
            q = min(120.0, q + 0.03)

    return best_loga, best_score, best_a


def _full_candidate_pool(n, rng):
    pool = []

    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        hs = None
    if hs is not None and hs.size == n:
        pool.append(hs.copy())

    pool.append(_uniform(n))
    pool.append(_binomial_sqrt_seq(n))

    for alpha in (0.50, 0.49, 0.48, 0.47, 0.46, 0.45, 0.43, 0.40, 0.35):
        pool.append(_power_seq(n, alpha))

    base = _binomial_sqrt_seq(n)
    uni = _uniform(n)
    for w in (1.0, 0.97, 0.94, 0.90, 0.86, 0.82, 0.78, 0.72):
        if w >= 1.0:
            pool.append(base.copy())
        else:
            pool.append(_blend_seq(base, uni, w))

    # Mild right-side blends.  These are mostly seeds for the gradient search.
    right = _power_seq(n, 0.30)
    right = right[::-1].copy()
    for w in (0.96, 0.93, 0.90, 0.86, 0.82):
        pool.append(_blend_seq(base, right, w))

    right2 = _power_seq(n, 0.45)
    right2 = right2[::-1].copy()
    for w in (0.97, 0.95, 0.93):
        pool.append(_blend_seq(base, right2, w))

    # Random shape perturbations.
    for _ in range(10):
        alpha = float(rng.uniform(0.32, 0.56))
        sigma = float(rng.uniform(0.04, 0.35))
        a = _power_seq(n, alpha) * np.exp(rng.normal(0.0, sigma, n))
        a = np.maximum(a, 1e-12)
        a /= float(a.sum())
        pool.append(a)

    # A few symmetric-ish random walks, clipped to nonnegative.
    for _ in range(5):
        x = np.zeros(n, dtype=np.float64)
        x[0] = 1.0
        for i in range(1, n):
            x[i] = max(0.01, x[i - 1] * float(rng.uniform(0.75, 1.28)))
        x = np.where(np.isfinite(x), x, 1.0)
        x = np.clip(x, 0.01, 1000.0)
        x = x / float(x.sum())
        pool.append(x)

    return pool


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Return the best sequence found within the time budget."""
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 5.0)
    n = 2535
    rng = np.random.default_rng(int(seed))

    # Best sequence of any allowed length.
    best_seq = _uniform(n).copy()
    best_score = _score_normalized(best_seq)

    # Best full-length candidate, used for final races.
    best_full_seq = _uniform(n).copy()
    best_full_score = _score_normalized(best_full_seq)

    full_pool = []

    # Very small analytic seeds.  They can already improve over 2.
    for m in (4, 5, 6, 8, 10, 13, 16, 20):
        a = _binomial_sqrt_seq(m)
        sc = _score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
        full_pool.append(_dilate_to(a, n))

    # Optimized short base patterns, diluted to full length.
    base_time = min(230.0, max(0.0, (deadline - t0) * 0.22))
    if base_time > 15.0:
        base_sizes = (4, 5, 6, 8, 10, 13, 16, 20, 39, 65, 169)
        weights = (1, 1, 1, 1, 1, 1, 1, 1, 10, 24, 70)
        total_w = float(sum(weights))
        base_end = t0 + base_time

        for m, wgt in zip(base_sizes, weights):
            now = time.monotonic()
            if now >= base_end:
                break
            alloc = base_time * float(wgt) / total_w
            end = min(base_end, now + max(2.0, alloc))

            a0 = _binomial_sqrt_seq(m)
            a1 = _power_seq(m, 0.50)
            s0 = _score_normalized(a0)
            s1 = _score_normalized(a1)
            init = a1 if s1 < s0 else a0

            if end - now < 3.0:
                a = init
                sc = _score_normalized(a)
            else:
                loga = _log_from_array(init)
                loga, sc, a = _optimize_loga(loga, now, end, rng, q0=6.0, rate0=0.18)

            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a.copy()

            full = _dilate_to(a, n)
            sc_full = _score_normalized(full)
            if np.isfinite(sc_full):
                full_pool.append(full)
                if sc_full < best_full_score:
                    best_full_score = sc_full
                    best_full_seq = full.copy()
                if sc_full < best_score:
                    best_score = sc_full
                    best_seq = full.copy()

    # Construct the full-length search pool.
    pool_from_direct = _full_candidate_pool(n, rng)
    full_pool.extend(pool_from_direct)
    if best_full_seq is not None:
        full_pool.append(best_full_seq.copy())
    if best_seq.size == n:
        full_pool.append(best_seq.copy())

    full_scored = []
    idx = 0
    for cand in full_pool:
        arr = _normalize_sequence(cand)
        if arr.size != n:
            continue
        sc = _score_normalized(arr)
        if np.isfinite(sc):
            full_scored.append((float(sc), idx, arr))
            if sc < best_full_score:
                best_full_score = sc
                best_full_seq = arr.copy()
            if sc < best_score:
                best_score = sc
                best_seq = arr.copy()
        idx += 1

    full_scored.sort()

    # Multi-start refinement of the most promising full candidates.
    now = time.monotonic()
    rem = deadline - now
    if full_scored and rem > 30.0:
        full_multi_time = min(450.0, max(0.0, rem * 0.28))
        top_count = min(6, len(full_scored))
        if full_multi_time > 10.0:
            slice_time = full_multi_time / float(max(1, top_count))
            q_settings = (6.0, 10.0, 4.0, 20.0, 8.0, 12.0)
            for rank in range(top_count):
                now = time.monotonic()
                if now >= deadline:
                    break
                end = min(deadline, now + slice_time)
                if end - now < 3.0:
                    continue
                sc0, _, seq = full_scored[rank]
                q0 = q_settings[rank % len(q_settings)]
                loga = _log_from_array(seq)
                loga, sc, a = _optimize_loga(loga, now, end, rng, q0=q0, rate0=0.18)
                if np.isfinite(sc):
                    if sc < best_full_score:
                        best_full_score = sc
                        best_full_seq = a.copy()
                    if sc < best_score:
                        best_score = sc
                        best_seq = a.copy()

    # Final polishing.
    now = time.monotonic()
    if now < deadline:
        if best_seq.size == n:
            loga = _log_from_array(best_seq)
            loga, sc, a = _optimize_loga(loga, now, deadline, rng, q0=8.0, rate0=0.18)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a.copy()
                best_full_score = sc
                best_full_seq = a.copy()
        else:
            rem = deadline - now
            if rem > 2.0:
                small_end = now + 0.55 * rem
                loga = _log_from_array(best_seq)
                loga, sc, a = _optimize_loga(loga, now, small_end, rng, q0=8.0, rate0=0.18)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a.copy()

            if best_full_seq is not None and best_full_score < best_score + 0.35:
                now = time.monotonic()
                if now < deadline:
                    loga = _log_from_array(best_full_seq)
                    loga, sc, a = _optimize_loga(loga, now, deadline, rng, q0=12.0, rate0=0.15)
                    if np.isfinite(sc):
                        if sc < best_full_score:
                            best_full_score = sc
                            best_full_seq = a.copy()
                        if sc < best_score:
                            best_score = sc
                            best_seq = a.copy()

    best_seq = _normalize_sequence(best_seq)

    # Final sanity check against the official evaluator if available.
    try:
        official = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(official):
            print("final evaluator bound:", official)
        else:
            print("official evaluator returned inf; returning normalized best")
    except Exception:
        pass

    return [float(x) for x in best_seq]
```
