def precompute_conv_pattern(n):
    """Precompute sparsity pattern and value indices for convolution matrix."""
    # For CSR format
    # Row k has count(k) non-zeros
    counts = np.array([min(k+1, n, 2*n-1-k) for k in range(2*n-1)], dtype=np.int32)
    indptr = np.zeros(2*n, dtype=np.int32)
    indptr[1:] = np.cumsum(counts)
    total = n * n
    
    # Column indices
    indices = np.empty(total, dtype=np.int32)
    # Value indices (index into a)
    val_idx = np.empty(total, dtype=np.int32)
    
    pos = 0
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        count = j_end - j_start + 1
        indices[pos:pos+count] = np.arange(j_start, j_end+1)
        val_idx[pos:pos+count] = np.arange(k - j_start, k - j_end - 1, -1)
        pos += count

    b = fftconvolve(a, a)
    M = b.max()
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * M / (s * s)
