```python
import numpy as np
import time

def compute_convolution_max(a):
    """Compute max of autocorrelation."""
    conv = np.convolve(a, a)
    return np.max(conv), conv

def lp_step(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    # (f*g)[k] = sum_j f[k-j] * g[j] for valid j
    # For k from 0 to 2n-2:
    # (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[k-j]*g[j]
    #
    # A matrix: A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    # We want A @ g <= M (vector of M's, length 2n-1)
    # Maximize sum(g) = -minimize -sum(g)
    
    # Build A matrix (sparse would be better, but for n up to a few thousand, dense is OK)
    # Actually, for n=5000, A would be 9999 x 5000 = 50 million entries. That's too much memory.
    # Let me use a sparse representation or a simpler approach.
    
    # Actually, let's use scipy sparse.
    from scipy.sparse import csr_matrix
    import scipy.sparse.linalg
    
    # A is (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
```

```python
"""
Minimize the peak autocorrelation of a non-negative sequence using
subgradient descent and LP-based refinement.
The score is 2*n*max(a*a)/sum(a)^2, which is scale-invariant.
We normalize sum(a)=1 and minimize 2*n*max(a*a).
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def build_conv_matrix(f):
    """Build sparse matrix A where (A @ g) = f * g (linear convolution)."""
    n = len(f)
    rows = []
    cols = []
    data = []
    for d in range(n):
        js = np.arange(n)
        ks = js + d
        rows.extend(ks.tolist())
        cols.extend(js.tolist())
        data.extend(np.full(n, f[d]).tolist())
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def lp_update(f, M, max_iters=100000):
    """Solve LP: max sum(g) s.t. (f*g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs',
                  options={'max_iter': max_iters})
    if res.success:
        return res.x
    return None

def subgradient_descent(a, n_iterations, alpha=1.0):
    """
    Subgradient descent to minimize max(a*a) subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_val = np.max(np.convolve(a, a))
    
    for t in range(n_iterations):
        # Compute convolution and find max
        conv = np.convolve(a, a)
        v = conv.max()
        if v < best_val:
            best_val = v
            best_a = a.copy()
        
        # Find argmax
        k_star = int(np.argmax(conv))
        
        # Compute gradient: grad[i] = 2*a[k_star - i] for valid i
        i_lo = max(0, k_star - n + 1)
        i_hi = min(k_star, n - 1)
        grad = np.zeros(n)
        grad[i_lo:i_hi+1] = 2.0 * a[k_star - np.arange(i_lo, i_hi+1)]
        
        # Step size (normalize by gradient norm)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = alpha / (np.sqrt(t + 1) * grad_norm)
        
        # Update
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / S
        
        a = a_new
    
    return best_a, best_val

def lp_polish(a, n_iters=50):
    """LP-based refinement."""
    n = len(a)
    a = a.copy()
```

import numpy as np

from scipy.sparse import csr_matrix
import time


def build_conv_matrix(f):
    """Build sparse matrix A where A @ g = f * g (linear convolution)."""
    n = len(f)
    cols = np.tile(np.arange(n), n)
    rows = cols + np.repeat(np.arange(n), n)
    data = np.repeat(f, n)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(f, M):
    """LP: max 1^T g s.t. (f*g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return None


def subgradient_step(a, alpha=1.0, t=1):
    """One subgradient step to reduce max(a*a)."""
    n = len(a)
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))
    
    i_lo = max(0, k_star - n + 1)
    i_hi = min(k_star, n - 1)
    grad = np.zeros(n)
    indices = np.arange(i_lo, i_hi + 1)
    grad[indices] = 2.0 * a[k_star - indices]
    
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return a
    
    step = alpha / (np.sqrt(t + 1) * grad_norm)
    a_new = np.maximum(a - step * grad, 0.0)
    S = a_new.sum()
    if S < 1e-15:
        return np.ones(n) / n
    return a_new / S
