```python
"""
Search for a non-negative step function minimizing

    2 * n * max_i conv(f,f)_i / (sum f)^2

using a hybrid active-set / projected-gradient method.

The core idea is that, for a chosen set of active convolution lags, the
stationarity equations for the weighted average of those lags are linear in f.
We repeatedly:
  1. choose the currently worst lags,
  2. solve the associated linear stationarity system for a candidate f,
  3. blend/accept candidates with simulated annealing,
  4. polish with multiplicative projected gradient steps.

Multiple lengths and diverse smooth initializations are explored.
"""

import time
import numpy as np


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def _normalize_positive(f, floor=1e-15):
    f = np.asarray(f, dtype=np.float64).copy()
    f[~np.isfinite(f)] = 0.0
    f = np.clip(f, floor, 1000.0)
    s = f.sum()
    if not np.isfinite(s) or s <= 0.0:
        f = np.ones_like(f, dtype=np.float64)
    else:
        f /= s
    return f


def _score_and_conv(f, size):
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    s = f.sum()
    if not np.isfinite(s) or s <= 0.0:
        return np.inf, np.zeros(2 * n - 1, dtype=np.float64)

    ff = f / s
    F = np.fft.rfft(ff, size)
    conv = np.fft.irfft(F * F, size)[:2 * n - 1]
    conv = np.real(conv)
    conv[conv < 0.0] = 0.0
    m = float(np.max(conv)) if conv.size else 0.0
    return float(2.0 * n * m), conv


def _fft_convolve_ab(a, b, size=None):
    n = a.size + b.size - 1
    if size is None:
        size = _next_pow2(n)
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    out = np.fft.irfft(A * B, size)[:n]
    out = np.real(out)
    out[out < 0.0] = 0.0
    return out


def _build_windows(p, n):
    """Return matrix W[i, j] = p[i + j], i=0..n-1, j=0..n-1."""
    p = np.asarray(p, dtype=np.float64)
    try:
        view = np.lib.stride_tricks.sliding_window_view(p, n)
        return np.ascontiguousarray(view)
    except Exception:
        out = np.empty((n, n), dtype=np.float64)
        for i in range(n):
            out[i] = p[i:i + n]
        return out


def _select_active_lags(conv, n, rng, explore_frac=0.0):
    L = conv.size
    K = max(24, min(L - 1, n // 8))
    if K >= L:
        K = L
    if K <= 0:
        return np.arange(L)

    top = np.argpartition(conv, -K)[-K:]

    if explore_frac > 0.0:
        nr = int(explore_frac * K)
        if nr > 0 and L > 1:
            extra = rng.integers(0, L, size=nr)
            top = np.unique(np.concatenate((top, extra)))

    return top


def _active_weights(p_lags, conv, n):
    L = 2 * n - 1
    p = np.zeros(L, dtype=np.float64)
    if p_lags.size == 0:
        p[:] = 1.0
        p /= p.sum()
        return p

    vals = conv[p_lags]
    vmax = float(np.max(vals))
    vmin = float(np.min(vals))
    rng_span = vmax - vmin
    if rng_span <= 1e-300:
        w = np.ones(p_lags.size, dtype=np.float64)
    else:
        # Moderately emphasize the very worst lags, but keep neighbors.
        w = np.exp(-2.0 * (vmax - vals) / max(rng_span, 1e-300))
    w = np.maximum(w, 1e-8)
    w /= w.sum()

    p[p_lags] = w
    # Tiny background avoids singular degenerate active sets.
    p += 1e-6
    p /= p.sum()
    return p


def _solve_stationarity_from_weights(p, n):
    """
    Solve (T p)_i = lambda for all i, sum f = 1.

    Row i of the gradient operator is:
        f_i * sum_{k=i}^{i+n-1} p_k + sum_j p_{i+j} f_j
    The constant lambda is removed by subtracting row 0.
    """
    L = 2 * n - 1
    if p is None or p.size != L:
        return None
    psum = float(p.sum())
    if psum <= 0.0 or not np.isfinite(psum):
        return None

    segs = _build_windows(p, n)          # segs[i, j] = p[i+j]
    d = segs.sum(axis=1)                 # d[i] = sum_{k=i}^{i+n-1} p[k]

    M = np.empty((n, n), dtype=np.float64)
    b = np.zeros(n, dtype=np.float64)

    if n == 1:
        M[0, 0] = 1.0
        b[0] = 1.0
    else:
        M[:n - 1] = segs[1:] - segs[0]
        M[:n - 1, 0] -= d[0]
        rows = np.arange(1, n)
        M[:n - 1, rows] += d[1:]
        M[n - 1] = 1.0
        b[n - 1] = 1.0

    f = None
    try:
        f = np.linalg.solve(M, b)
        if not np.all(np.isfinite(f)):
            f = None
    except Exception:
        f = None

    if f is None:
        try:
            f = np.linalg.lstsq(M, b, rcond=None)[0]
        except Exception:
            return None

    if not np.all(np.isfinite(f)):
        return None

    # If solve was poor, fall back to least squares.
    try:
        res = float(np.linalg.norm(M @ f - b, ord=1))
        if res > 1e-5:
            f2 = np.linalg.lstsq(M, b, rcond=None)[0]
            if np.all(np.isfinite(f2)):
                res2 = float(np.linalg.norm(M @ f2 - b, ord=1))
                if res2 < res:
                    f = f2
    except Exception:
        pass

    if not np.all(np.isfinite(f)):
        return None

    s = float(f.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None

    # Accept slight numerical negatives; reject badly non-positive solves.
    if np.min(f) < -1e-3:
        fc = np.clip(f, 0.0, None)
        sc = float(fc.sum())
        if sc <= 0.05 * abs(s):
            return None
        f = fc
        s = sc

    f = np.clip(f, 0.0, None)
    s = float(f.sum())
    if s <= 1e-12:
        return None
    f /= s
    return f


def _gradient_from_weights_fft(p, f):
    """
    Gradient of sum_k p_k C_k with respect to f, where C = f*f (full).
    """
    n = f.size
    L = p.size
    if L != 2 * n - 1:
        p = np.pad(p, (0, (2 * n - 1) - p.size))
        L = 2 * n - 1

    # shift_i = sum_j p[i+j] f[j]
    q = p[::-1]
    size = _next_pow2(n + L - 1)
    conv = _fft_convolve_ab(f, q, size)
    idx = 2 * n - 2 - np.arange(n)
    shift = conv[idx]

    # count_i = sum_{k=i}^{i+n-1} p[k]
    cs = np.empty(L + 1, dtype=np.float64)
    cs[0] = 0.0
    cs[1:] = np.cumsum(p)
    count = cs[n:] - cs[:n]

    return shift + f * count


def _perturb_smooth(f, rng, amp):
    n = f.size
    x = np.arange(n, dtype=np.float64) / max(1, n)
    noise = np.zeros(n, dtype=np.float64)

    # Low-degree polynomial part.
    noise += rng.uniform(-amp, amp, size=1)[0] * (x - 0.5)
    noise += rng.uniform(-amp, amp, size=1)[0] * (x * x - x)

    # A few smooth Fourier modes.
    for m in range(1, 5):
        a = rng.uniform(-amp, amp) / m
        b = rng.uniform(-amp, amp) / m
        ph = rng.uniform(0.0, 2.0 * np.pi)
        noise += a * np.cos(2.0 * np.pi * m * x + ph) + b * np.sin(2.0 * np.pi * m * x + ph)

    cand = f * (1.0 + noise)
    return _normalize_positive(cand)


def _resize_array(arr, n):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.size == 0:
        return np.ones(n, dtype=np.float64)
    if arr.size == n:
        return arr.copy()
    if arr.size == 1:
        return np.full(n, arr[0], dtype=np.float64)
    old_x = np.linspace(0.0, 1.0, arr.size)
    new_x = np.linspace(0.0, 1.0, n)
    return np.interp(new_x, old_x, arr)


def _make_starts(n, rng, parent=None, count=8):
    starts = []

    # Parent, if available.
    if parent is not None and parent.size > 0:
        p = _resize_array(parent, n)
        starts.append(p)
        starts.append(p[::-1].copy())

    # Uniform baseline.
    starts.append(np.ones(n, dtype=np.float64))

    # Asymmetric smooth profiles.
    x = np.linspace(0.0, 1.0, n)
    xp = np.arange(n, dtype=np.float64) / max(1, n)

    for A in (0.4, 0.8, 1.3, 2.0):
        starts.append(np.exp(A * (x - 0.5)))
        starts.append(np.exp(-A * (x - 0.5)))
        starts.append(np.exp(A * x * (1.0 - x)))

    # Beta-like one-sided profiles.
    xx = np.linspace(1e-6, 1.0 - 1e-6, n)
    for a, b in ((0.1, 1.0), (0.3, 2.0), (1.0, 0.3), (2.0, 0.5), (0.5, 3.0), (3.0, 0.5)):
        starts.append(xx ** a * (1.0 - xx) ** b)
        starts.append(xx ** b * (1.0 - xx) ** a)

    # Random Fourier smooth profiles.
    for _ in range(6):
        f = np.ones(n, dtype=np.float64)
        for m in range(1, 5):
            amp = rng.uniform(-0.9, 0.9) / m
            ph = rng.uniform(0.0, 2.0 * np.pi)
            f += amp * np.sin(2.0 * np.pi * m * xp + ph)
        starts.append(f)

    # Random exponential polynomial.
    for _ in range(4):
        c1 = rng.uniform(-2.5, 2.5)
        c2 = rng.uniform(-3.0, 3.0)
        c3 = rng.uniform(-2.0, 2.0)
        f = np.exp(c1 * x + c2 * (x - 0.5) ** 2 + c3 * x * (1.0 - x))
        starts.append(f)

    normed = []
    for f in starts:
        g = _normalize_positive(f)
        normed.append(g)

    # Keep only count starts, in a stable order.
    return normed[:max(1, count)]


def _active_search(f0, n, size, budget, rng, tag=""):
    f = _normalize_positive(f0)
    cur_score, conv = _score_and_conv(f, size)

    if not np.isfinite(cur_score):
        f = np.ones(n, dtype=np.float64)
        cur_score, conv = _score_and_conv(f, size)

    best = f.copy()
    best_score = cur_score

    start_t = time.monotonic()
    end_t = start_t + max(0.0, float(budget))
    it = 0
    no_accept = 0
    temp = 0.02

    while time.monotonic() < end_t:
        active = _select_active_lags(conv, n, rng, 0.0 if no_accept < 6 else 0.25)
        p = _active_weights(active, conv, n)
        fsol = _solve_stationarity_from_weights(p, n)

        accepted = False
        if fsol is not None:
            base = f if no_accept == 0 else best
            base_score = cur_score if no_accept == 0 else best_score

            for mix in (0.45, 0.7, 0.2, 1.0, 0.35, 0.1, 0.05):
                cand = (1.0 - mix) * base + mix * fsol
                cand = _normalize_positive(cand)
                cs, cc = _score_and_conv(cand, size)

                acceptable = max(base_score, best_score) * (1.0 + temp)

                if np.isfinite(cs) and cs <= acceptable:
                    f = cand
                    conv = cc
                    cur_score = cs
                    accepted = True
                    if cs < best_score:
                        best_score = cs
                        best = f.copy()
                    break

        if accepted:
            no_accept = 0
            temp = max(0.01, temp * 0.97)
        else:
            no_accept += 1
            temp = min(0.12, temp * 1.05)

            # Periodically nudge the best point to escape a basin.
            if no_accept % 12 == 0:
                cand = _perturb_smooth(best, rng, 0.05)
                cs, cc = _score_and_conv(cand, size)
                if np.isfinite(cs) and cs < best_score:
                    best_score = cs
                    best = cand.copy()
                    f = best.copy()
                    conv = cc
                    cur_score = cs
                    no_accept = 0
                elif np.isfinite(cs) and cs < best_score * (1.0 + 0.08):
                    f = cand
                    conv = cc
                    cur_score = cs
                    no_accept = 0

        if cur_score < best_score:
            best_score = cur_score
            best = f.copy()

        it += 1
        if it % 200 == 0 and tag:
            print(f"{tag} active it={it} cur={cur_score:.6f} best={best_score:.6f}")

        if no_accept > 60:
            break

    return best, best_score


def _gradient_local(f0, n, size, budget, rng, tag=""):
    f = _normalize_positive(f0)
    cur_score, conv = _score_and_conv(f, size)

    if not np.isfinite(cur_score):
        f = np.ones(n, dtype=np.float64)
        cur_score, conv = _score_and_conv(f, size)

    best = f.copy()
    best_score = cur_score

    start_t = time.monotonic()
    end_t = start_t + max(0.0, float(budget))
    it = 0
    no_accept = 0
    alpha_max = 1.0

    while time.monotonic() < end_t:
        active = _select_active_lags(conv, n, rng, 0.0 if no_accept < 4 else 0.15)
        p = _active_weights(active, conv, n)
        g = _gradient_from_weights_fft(p, f)
        lam = float(np.dot(f, g))
        d = g - lam

        mad = float(np.max(np.abs(d)))
        if not np.isfinite(mad) or mad <= 1e-300:
            cand = _perturb_smooth(best, rng, 0.04)
            cs, cc = _score_and_conv(cand, size)
            if np.isfinite(cs) and cs < best_score:
                best_score = cs
                best = cand.copy()
                f = best.copy()
                conv = cc
                cur_score = cs
                no_accept = 0
            else:
                no_accept += 1
                if no_accept > 20:
                    break
            it += 1
            continue

        base = f if no_accept == 0 else best
        base_score = cur_score if no_accept == 0 else best_score
        improved = False

        for alpha in (0.10, 0.05, 0.025, 0.012, 0.006, 0.003):
            beta = alpha * alpha_max
            logq = -beta * d / mad
            logq = np.clip(logq, -60.0, 60.0)
            cand = base * np.exp(logq)
            cand = _normalize_positive(cand)
            cs, cc = _score_and_conv(cand, size)

            if np.isfinite(cs) and cs < base_score * (1.0 - 1e-12):
                f = cand
                conv = cc
                cur_score = cs
                improved = True
                alpha_max = min(2.0, alpha_max * 1.05)
                if cs < best_score:
                    best_score = cs
                    best = f.copy()
                break

        if improved:
            no_accept = 0
        else:
            no_accept += 1
            alpha_max = max(0.2, alpha_max * 0.85)

            if no_accept % 8 == 0:
                cand = _perturb_smooth(best, rng, 0.035)
                cs, cc = _score_and_conv(cand, size)
                if np.isfinite(cs) and cs < best_score:
                    best_score = cs
                    best = cand.copy()
                    f = best.copy()
                    conv = cc
                    cur_score = cs
                    no_accept = 0
                elif np.isfinite(cs) and cs < best_score * (1.0 + 0.03):
                    f = cand
                    conv = cc
                    cur_score = cs
                    no_accept = 0

        it += 1
        if it % 300 == 0 and tag:
            print(f"{tag} grad it={it} cur={cur_score:.6f} best={best_score:.6f}")

        if no_accept > 40:
            break

    if cur_score < best_score:
        best_score = cur_score
        best = f.copy()

    return best, best_score


def _random_active_candidate(n, rng):
    L = 2 * n - 1
    K = max(24, n // 8)
    centers = [
        n - 1, n, int(0.75 * n), int(1.25 * n),
        int(0.5 * n), int(1.5 * n), 0, L - 1
    ]
    centers = [max(0, min(L - 1, int(c))) for c in centers]

    active = []
    for _ in range(K):
        c = centers[rng.integers(0, len(centers))]
        v = int(c + rng.normal(0.0, max(2, n / 10.0)))
        v = max(0, min(L - 1, v))
        active.append(v)

    active = np.unique(np.asarray(active, dtype=int))
    while active.size < K:
        more = rng.integers(0, L, size=K - active.size)
        active = np.unique(np.concatenate((active, more)))

    p = np.zeros(L, dtype=np.float64)
    p[active] = 1.0
    p += 1e-5
    f = _solve_stationarity_from_weights(p, n)
    if f is None:
        return None
    return _normalize_positive(f)


def propose_candidate(seed=42, budget_s=1500, **kwargs):

    total = max(0.0, float(budget_s) - 20.0)
    if total <= 1.0:
        return [1.0 / 64.0] * 64

    deadline = time.monotonic() + total

    # Initial best: parent if available, otherwise uniform.
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.where(np.isfinite(parent), parent, 0.0)
        parent = np.clip(parent, 0.0, 1000.0)
        ps = float(parent.sum())
        if ps > 0.0:
            parent = parent / ps
        else:
            parent = None
    except Exception:
        parent = None

    if parent is not None and parent.size > 0:
        best_f = _normalize_positive(parent)
        size0 = _next_pow2(2 * best_f.size - 1)
        best_score, _ = _score_and_conv(best_f, size0)
    else:
        best_f = np.ones(384, dtype=np.float64)
        best_score, _ = _score_and_conv(best_f, _next_pow2(2 * 384 - 1))

    print(f"initial score={best_score:.6f} len={best_f.size}")

    # Main exploration over several lengths.
    # The fractions are intentionally biased to longer sequences for smoother behavior.
    phases = [
        (256, 0.22, 6),
        (384, 0.33, 7),
        (512, 0.45, 8),
    ]

    search_budget = total * 0.82
    phase_origin = time.monotonic()

    for n, frac, nstarts in phases:
        phase_time = search_budget * frac
        phase_end = min(deadline, phase_origin + phase_time)
        if time.monotonic() >= phase_end:
            continue

        starts = _make_starts(n, rng, parent, count=nstarts)
        if not starts:
            continue

        per_start = max(3.0, (phase_end - time.monotonic()) / max(1, len(starts)))
        size = _next_pow2(2 * n - 1)

        for si, s0 in enumerate(starts):
            if time.monotonic() >= phase_end:
                break
            rem_phase = max(0.0, phase_end - time.monotonic())
            if rem_phase < 5.0:
                break

            alloc = min(per_start, rem_phase)
            act_budget = max(2.0, alloc * 0.65)
            grad_budget = max(2.0, alloc * 0.30)

            tag = f"n{n}s{si}"
            f1, sc1 = _active_search(s0, n, size, act_budget, rng, tag=tag)

            if time.monotonic() < phase_end - 2.0 and grad_budget > 2.0:
                g_rem = min(grad_budget, phase_end - time.monotonic())
                f2, sc2 = _gradient_local(f1, n, size, g_rem, rng, tag=tag + ":grad")
                if np.isfinite(sc2) and sc2 < sc1:
                    f1, sc1 = f2, sc2

            if np.isfinite(sc1) and sc1 < best_score:
                best_score = sc1
                best_f = f1.copy()
                print(f"new phase best: len={n} score={best_score:.6f}")

    # If we still have a meaningful remainder, try random active-set solves
    # and then polish the best sequence, including a possible upsample.
    rem = deadline - time.monotonic()
    if rem > 45.0 and best_f is not None:
        n_best = best_f.size
        if n_best > 512:
            n_best = 512
            best_f = _resize_array(best_f, n_best)

        size_best = _next_pow2(2 * n_best - 1)

        # Random active-set attempts.
        rand_budget = min(120.0, rem * 0.35)
        if rand_budget > 15.0:
            t_end = time.monotonic() + rand_budget
            attempts = 0
            while time.monotonic() < t_end:
                rc = _random_active_candidate(n_best, rng)
                attempts += 1
                if rc is not None:
                    sc, cv = _score_and_conv(rc, size_best)
                    if np.isfinite(sc) and sc < best_score * 1.02:
                        # Quick polish only if promising.
                        polb = max(2.0, min(8.0, t_end - time.monotonic()))
                        rp, rsc = _gradient_local(rc, n_best, size_best, polb, rng, tag=f"rand{attempts}")
                        if np.isfinite(rsc) and rsc < best_score:
                            best_score = rsc
                            best_f = rp.copy()
                            print(f"random active improved: len={n_best} score={best_score:.6f}")
                attempts += 1
                if attempts % 200 == 0:
                    print(f"random active attempts={attempts} best={best_score:.6f}")

        # Final polish at the best length.
        rem = deadline - time.monotonic()
        if rem > 25.0:
            polish = min(rem * 0.7, 260.0)
            fpol, spol = _gradient_local(best_f, best_f.size, size_best, polish, rng, tag="finalpolish")
            if np.isfinite(spol) and spol < best_score:
                best_score = spol
                best_f = fpol.copy()

            # Dimension upgrading: if time remains, smooth-reshaped longer sequences often improve.
            rem = deadline - time.monotonic()
            if rem > 80.0:
                n_up = int(min(2048, max(1024, 2 * best_f.size)))
                fu = _resize_array(best_f, n_up)
                su, _ = _score_and_conv(fu, _next_pow2(2 * n_up - 1))
                size_up = _next_pow2(2 * n_up - 1)
                up_budget = min(rem * 0.6, 320.0)
                fu2, su2 = _gradient_local(fu, n_up, size_up, up_budget, rng, tag=f"up{n_up}")
                if np.isfinite(su2) and su2 < best_score:
                    best_score = su2
                    best_f = fu2.copy()
                    print(f"upsample improved: len={n_up} score={best_score:.6f}")

    # Final authoritative check.  The internal FFT score is very close, but the
    # official evaluator is authoritative.
    final_score = np.inf
    try:
        final_score = float(evaluate_sequence([float(x) for x in best_f]))
        print(f"final official score={final_score:.6f} len={best_f.size}")
    except Exception as e:
        print("final eval failed:", e)
        final_score = best_score

    if not np.isfinite(final_score):
        # Safe fallback: uniform sequence always has score 2.
        best_f = np.ones(384, dtype=np.float64)

    # Return plain floats, all non-negative.
    out = np.asarray(best_f, dtype=np.float64)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    oss = float(out.sum())
    if oss > 0.0:
        out = out / oss

    return [float(x) for x in out]
```
