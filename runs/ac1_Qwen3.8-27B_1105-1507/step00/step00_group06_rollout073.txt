def _full_convolution(a):

    """Generate an initial probability vector of length n."""
    x = np.linspace(0.0, 1.0, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'triangular':
        a = np.minimum(np.arange(n, dtype=float), (n - 1) - np.arange(n, dtype=float)) + 0.5
    elif kind == 'gaussian':
        a = np.exp(-0.5 * (np.linspace(-3.5, 3.5, n)) ** 2)
    elif kind == 'u_shaped':
        a = (x - 0.5) ** 2 + 0.05
    elif kind == 'edges_high':
        a = np.abs(x - 0.5) ** 0.4 + 0.05
    elif kind == 'cosine':
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * x)
    elif kind == 'sink':
        # Higher in middle, lower at edges
        a = 1.0 - (x - 0.5) ** 2
    elif kind == 'random_smooth':
        k = int(rng.integers(3, 25))
        cps = rng.random(k) + 0.01
        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, k), cps)
    elif kind == 'random_wiggle':
        freq = rng.uniform(1.0, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.sin(freq * 2 * np.pi * x + phase)
    else:
        a = np.ones(n)
    a = np.clip(a, 0.01, None)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses gradient-based optimization with soft-max peak approximation,
    multiple starting points, and random restarts.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s)) - 12.0

    # Track best
    best_score = np.inf
    best_seq = None

    def _try(a):
        nonlocal best_score, best_seq
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            el = time.monotonic() - t0
            print(f"  Best updated: T={score:.6f}  n={a.size}  t={el:.0f}s")

    # Get parent for potential use
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except NameError:
        parent = np.ones(256, dtype=np.float64)

    # --- Phase 1: Quick evaluations of various shapes ---
    print("Phase 1: Shape evaluations")
    for n in [64, 128, 256, 512, 1024, 2048, 4096, 8192]:
        if time.monotonic() > deadline - 300:
            break
        for kind in ['uniform', 'triangular', 'gaussian', 'u_shaped', 'edges_high', 'cosine', 'sink']:
            a = _make_init(n, kind, rng)
            _try(a * 100.0)

    # --- Phase 2: Gradient optimization at multiple scales ---
    print("Phase 2: Gradient optimization")
    n_list = [256, 512, 1024, 2048, 4096, 8192]
    kinds = ['uniform', 'gaussian', 'u_shaped', 'edges_high', 'sink']

    for n in n_list:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        per = remaining / max(1, len(n_list) - n_list.index(n))
        per_kind = per / max(1, len(kinds))

        for kind in kinds:
            if time.monotonic() > deadline - 5:
                break
            a = _make_init(n, kind, rng)
            sub_dl = time.monotonic() + min(per_kind, 90.0)
            r = _optimize_two_phase(a, rng, sub_dl)
            _try(r * 100.0)

    # --- Phase 3: Optimize from parent ---
    if parent.size > 0 and parent.size <= 12000 and time.monotonic() < deadline - 60:
        print(f"Phase 3: Optimize parent (n={parent.size})")
        a = parent / max(parent.sum(), 1e-10)
        r = _optimize_two_phase(a, rng, deadline - 20)
        _try(r * 100.0)

    # --- Phase 4: Random restarts ---
    if time.monotonic() < deadline - 60:
        print("Phase 4: Random restarts")
        ntrials = 0
        while time.monotonic() < deadline - 10 and ntrials < 30:
            ntrials += 1
            n = int(rng.integers(256, 6000))
            kind = rng.choice(['random_smooth', 'random_wiggle', 'gaussian',
                               'u_shaped', 'edges_high', 'sink', 'cosine'])
            a = _make_init(n, kind, rng)
            alloc = (deadline - time.monotonic() - 10) / max(1, 30 - ntrials)
            sub_dl = time.monotonic() + min(alloc, 80.0)
            r = _optimize(a, rng, sub_dl)
            _try(r * 100.0)

    # --- Phase 5: Refine the best found ---
    if best_seq is not None and time.monotonic() < deadline - 20:
        print("Phase 5: Final refinement")
        a = best_seq / max(best_seq.sum(), 1e-10)
        r = _optimize(a, rng, deadline - 5)
        _try(r * 100.0)

    # Fallback
    if best_seq is None:
        _try(np.ones(256) * 100.0)
    if best_seq is None:
        best_seq = parent if parent.size > 0 else np.ones(256)

    print(f"DONE: best_score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t0:.0f}s")
    return [float(x) for x in best_seq]
