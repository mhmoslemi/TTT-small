def local_optimize(p, rng, deadline):
    """Local optimization of probability distribution p to minimize max convolution."""
    p = p.copy()
    n = len(p)
    best_p = p.copy()
    best_score = fast_score(p)
    T = 1.0  # Temperature for simulated annealing
    T_min = 0.01
    cooling = 0.999
    
    step = 0
    while time.monotonic() < deadline:
        step += 1
        # Compute current convolution
        conv = np.convolve(p, p)
        max_val = conv.max()
        current_score = 2.0 * n * max_val  # Since S=1
        
        # Strategy: reduce the elements that contribute to the max
        # Find positions where conv is near max
        threshold = max_val * 0.95
        active_k = np.where(conv >= threshold)[0]

        pressure = np.zeros(n)

import numpy as np
import time

def _fast_score(p):
    """Compute 2*n*max(conv)/S^2 for a non-negative sequence p."""
    n = p.shape[0]
    S = p.sum()
    if S < 0.01:
        return np.inf
    conv = np.convolve(p, p, mode='full')
    return 2.0 * n * conv.max() / (S * S)

def _norm(p):
    """Normalize to unit sum."""
    S = p.sum()
    if S < 1e-15:
        return np.ones_like(p) / p.shape[0]
    return p / S

def _pressure(p, active):
    """Compute pressure vector for active convolution positions."""
    n = p.shape[0]
    # active is a boolean array of length 2n-1
    # pressure[i] = sum_k active[k] * p[k-i]
    # = np.correlate(active.astype(float), p, 'full')[:n]
    return np.correlate(active.astype(np.float64), p, 'full')[:n]

def _local_search(p, rng, deadline, verbose=False):
    """Local search to minimize max convolution of p."""
    n = p.shape[0]
    p = p.astype(np.float64).copy()
    p = np.clip(p, 0, None)
    p = _norm(p)
    
    best_p = p.copy()
    best_score = _norm_score(p)
    T = 0.1  # temperature
    T_min = 0.001
    no_improve = 0
    max_no_improve = 200
    
    alpha = 0.05  # step size
    
    while time.monotonic() < deadline:
        conv = np.convolve(p, p, mode='full')
        max_val = conv.max()
        cur_score = 2.0 * n * max_val  # since sum(p) = 1
        
        # Find active positions (near max)
        threshold = max_val * 0.9
        active = conv >= threshold
        
        if active.sum() == 0:
            active[np.argmax(conv)] = True
        
        # Compute pressure
        pres = _pressure(p, active)
        max_pres = pres.max()
        if max_pres < 1e-15:
            break
        
        # Update: reduce p where pressure is high
        # p_new[i] = p[i] * (1 - alpha * pres[i] / max_pres * (1 - p[i]))
        # The (1-p[i]) factor avoids over-reducing small elements
        pres_norm = pres / max_pres
        p_new = p * (1.0 - alpha * pres_norm)
        p_new = np.clip(p_new, 0, None)
        
        # Renormalize
        S = p_new.sum()
        if S < 0.01:
            # Too much reduction, reduce step
            alpha *= 0.5
            continue
        p_new = p_new / S
        
        new_score = 2.0 * n * np.convolve(p_new, p_new, mode='full').max()

    """Reduce the convolution peak at position k_star."""
    n = len(p)
    # Find contributing pairs
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)

        j = k_star - i

            p[i] *= (1 - alpha)
        else:
            p[j] *= (1 - alpha)

            alpha = min(0.25, alpha * 1.015)
        elif no_improve > 40:
            alpha = max(0.015, alpha * 0.96)

        # Diversification restart
        if no_improve > 600:
            p = np.clip(p * (1.0 + 0.08 * rng.standard_normal(n)), 0, None)
            p = _norm(p)
            no_improve = 0
            T = 0.04
            alpha = 0.06

    return best_p, best_score


def _make_inits(n, rng):
    """Generate diverse initial probability distributions of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xm = (x - (n - 1) / 2.0) / max(1.0, (n - 1) / 2.0)  # normalized to [-1, 1]

    # Uniform
    inits.append(np.ones(n))

    # Half-sine
    inits.append(np.sin(np.pi * (x + 0.5) / n))

    # Squared half-sine (raised cosine)
    inits.append(np.sin(np.pi * (x + 0.5) / n) ** 2)

    # Cosine modulated (various frequencies and amplitudes)
    for f in [1, 2, 3, 5]:
        for amp in [0.3, 0.5, 0.7]:
            p = 1.0 + amp * np.cos(2 * np.pi * f * x / n)
            inits.append(np.clip(p, 0.01, None))

    # Geometric decay (left and right)
    for r in [0.95, 0.99, 0.995]:
        inits.append(r ** x)
        inits.append(r ** (n - 1 - x))

    # Bimodal (Gaussian mixture)
    for sf in [0.05, 0.1, 0.15, 0.25, 0.35]:
        sigma = max(1.0, sf * n)
        c1, c2 = n // 4, 3 * n // 4
        p = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
        inits.append(p)

    # Triangular (peaked center)
    inits.append(1.0 - np.abs(xm))

    # U-shaped (peaked edges)
    inits.append(1.0 + 0.5 * xm * xm)

    # W-shaped
    inits.append(1.0 + 0.3 * np.cos(4 * np.pi * x / n))

    # Quartic center peak
    inits.append(1.0 + 0.5 * (1.0 - xm ** 2) ** 2)

    # Random smooth
    for _ in range(3):
        p = np.abs(rng.standard_normal(n))
        k = max(1, n // 80)
        p = np.convolve(p, np.ones(k) / k, mode='same')
        inits.append(np.clip(p, 0.001, None))

    # Random low-frequency
    for _ in range(3):
        p = np.ones(n)
        for _ in range(4):
            f = int(rng.integers(1, max(2, n // 15)))
            a = rng.uniform(-0.6, 0.6)
            b = rng.uniform(0, 2 * np.pi)
            p += a * np.cos(2 * np.pi * f * x / n + b)
        inits.append(np.clip(p, 0.001, None))

    return inits


def _interpolate_to_length(p, n_new):
    """Interpolate a sequence to a new length."""
    from scipy.interpolate import interp1d
    old_x = np.linspace(0, 1, len(p))
    new_x = np.linspace(0, 1, n_new)
    p_clipped = np.clip(p, 0, None)
    f = interp1d(old_x, p_clipped, kind='cubic', fill_value=0, bounds_error=False)
    return np.clip(f(new_x), 0, None)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for step function coefficients minimizing 2*n*max(autoconv)/sum^2.
    Strategy: multi-size exploration with diverse initializations,
    pressure-based local search with simulated annealing, and a final
    intensive refinement phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0

    best_p = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if p_raw.ndim == 1 and p_raw.size >= 20:
            p_raw = np.clip(p_raw, 0, None)
            if p_raw.sum() > 0.01:
                parent = p_raw
    except Exception:
        pass

    # Define sizes to explore (coarse to fine)
    sizes = [80, 150, 300, 600, 1200, 2500, 5000, 7500]
    if parent is not None:
        ps = parent.size
        if ps not in sizes and 50 <= ps <= 15000:
            sizes.append(ps)
            sizes.sort()

    print(f"Sizes to explore: {sizes}")
    print(f"Parent available: {parent is not None}, time budget: {deadline - start_time:.0f}s")

    # Phase 1: Broad exploration across sizes
    for idx, n in enumerate(sizes):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        sizes_left = len(sizes) - idx
        time_per_size = min(400.0, remaining / max(1, sizes_left))
        if time_per_size < 10:
            continue
        size_deadline = time.monotonic() + time_per_size

        # Benchmark
        t0 = time.monotonic()
        _ = np.convolve(np.ones(n), np.ones(n))
        t_conv = time.monotonic() - t0
        # Estimate iterations possible
        est_its = int(time_per_size / max(t_conv * 3, 0.001))
        print(f"\n  n={n} (conv={t_conv*1e3:.1f}ms, ~{est_its} its available, budget={time_per_size:.0f}s)")

        # Generate initializations
        inits = _make_inits(n, rng)

        # Add interpolated parent if available
        if parent is not None and parent.size != n:
            p_parent_interp = _interpolate_to_length(parent, n)
            inits.append(p_parent_interp)
        elif parent is not None and parent.size == n:
            inits.append(parent.copy())

        # Limit inits based on time
        max_inits = min(len(inits), max(4, int(time_per_size * 0.8 / max(t_conv * 80, 0.02))))
        max_inits = min(max_inits, len(inits))

        local_best_p = None
        local_best_score = np.inf

        for i in range(max_inits):
            if time.monotonic() > size_deadline:
                break
            p_init = np.clip(inits[i].astype(np.float64), 0, None)
            if p_init.sum() < 0.01:
                continue
            p_init = _norm(p_init)

            # Time for this init
            init_time = max(5.0, (size_deadline - time.monotonic()) / max(1, max_inits - i))
            init_deadline = time.monotonic() + min(init_time, size_deadline - time.monotonic())

            p_opt, score = _local_search(p_init, rng, init_deadline, verbose=False, label=f"n={n},i={i}")

            if score < local_best_score:
                local_best_score = score
                local_best_p = p_opt
            if score < best_score:
                best_score = score
                best_p = p_opt.copy()
                print(f"    NEW BEST: n={n}, init={i}, score={score:.6f} (t={time.monotonic()-start_time:.0f}s)")

        # Continue refining the best for this size
        if local_best_p is not None and time.monotonic() < size_deadline:
            p_opt, score = _local_search(local_best_p, rng, size_deadline, verbose=False, label=f"n={n},refine")
            if score < best_score:
                best_score = score
                best_p = p_opt.copy()
                print(f"    REFINED: n={n}, score={score:.6f}")

    # Phase 2: Intensive refinement of global best
    if best_p is not None:
        remaining = deadline - time.monotonic()
        print(f"\nPhase 2: intensive refinement, {remaining:.0f}s left, current best={best_score:.6f}")
        if remaining > 15:
            # Multiple refinement passes with fresh RNG
            for pass_i in range(3):
                if time.monotonic() > deadline:
                    break
                p_opt, score = _local_search(best_p, rng, deadline, verbose=True, label=f"pass{pass_i}")
                if score < best_score:
                    best_score = score
                    best_p = p_opt.copy()
                    print(f"  Pass {pass_i}: {score:.6f}")

    # Final check with a broader perturbation (escape potential plateau)
    if best_p is not None and time.monotonic() < deadline - 2:
        n = len(best_p)
        # Try a few different perturbation magnitudes
        for mag in [0.05, 0.1, 0.2]:
            if time.monotonic() > deadline - 2:
                break
            p_try = np.clip(best_p * (1.0 + mag * rng.standard_normal(n)), 0, None)
            p_try = _norm(p_try)
            p_try, s_try = _local_search(p_try, rng, min(deadline - 2, time.monotonic() + 30), verbose=False)
            if s_try < best_score:
                best_score = s_try
                best_p = p_try
                print(f"  Perturb(mag={mag}): {best_score:.6f}")

    # Verify and return
    if best_p is not None:
        result = [float(x) for x in np.clip(best_p, 0, 1000)]
        final = evaluate_sequence(result)
        print(f"\nFinal sequence: n={len(result)}, verified_score={final:.6f}")
        return result

    # Fallback: uniform
    print("FALLBACK: using uniform sequence")
    return [1.0]
