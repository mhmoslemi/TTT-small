def lp_alternation_step(a, K=10):
    """One step of the LP-based alternating method."""
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    
    # Find top K active constraints
    top_k = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix: K x n
    # Row k: (a * g)_k = sum_j a_j * g_{k-j} for valid j
    # = sum_m a_{k-m} * g_m for valid m
    # So A[k, m] = a[k-m] if 0 <= k-m <= n-1, else 0
    A = np.zeros((K, n))
    for i, k in enumerate(top_k):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = a[k-hi:k-lo+1][::-1]
    
    # LP: max sum(g) s.t. A @ g <= M, g >= 0
    # In standard form: min -sum(g) s.t. A @ g <= M, g >= 0
    from scipy.optimize import linprog
    c_obj = -np.ones(n)

    g = res.x
    s_g = g.sum()
    if s_g <= 0:
        return a
    
    # Normalize to match sum(a)
    s_a = a.sum()
    g = g * s_a / s_g
    
    return g

import numpy as np
import time
import sys

def project_simplex(v):
    """Project v onto {x: x_i >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n+1) - cv
    rho = np.where(cond > 0)[0]
    if len(rho) == 0:
        return np.full(n, 1.0/n)
    rho = rho[-1]
    theta = cv[rho] / (rho + 1)
    return np.maximum(v - theta, 0)

def compute_conv(a):
    """Fast convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A)[:2*n-1]
    return c

def get_score(c, n):
    """Compute score from convolution (assuming sum(a) = 1)."""
    return 2.0 * n * c.max()

"""
Minimizes 2n*max(convolve(a,a))/sum(a)^2 using accelerated subgradient descent
on the probability simplex with diverse initializations, momentum, and restarts.
Also attempts LP-based alternating refinement for final polish.
"""

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto the probability simplex {x: x_i >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 1:
        return np.maximum(v, 0)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - cv
    rho = np.where(cond > 0)[0]
    if rho.size == 0:
        return np.full(n, 1.0 / n)
    rho = int(rho[-1])
    theta = cv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fft_convolve(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def compute_subgrad(a, c, n_active_frac=0.03):
    """Compute averaged subgradient of max_k (convolve(a,a))_k at point a."""
    n = len(a)
    max_c = c.max()
    n_active = max(1, int(n_active_frac * len(c)))
    top_idx = np.argpartition(c, -n_active)[-n_active:]
    grad = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= len(top_idx)
    return grad


def subgrad_optimize(a0, max_time, seed=42, base_lr=None):
    """Run accelerated subgradient descent from a0 for at most max_time seconds."""
    n = len(a0)
    a = a0.copy()
    c = fft_convolve(a)
    cur_score = 2.0 * n * c.max()
    local_best = a.copy()
    local_best_score = cur_score

    if base_lr is None:
        base_lr = 3.0 / np.sqrt(n)

    lr = base_lr
    mu = 0.85
    mom = np.zeros(n)
    patience = 300
    no_improve = 0
    t_start = time.monotonic()
    iteration = 0

    while time.monotonic() - t_start < max_time:
        if time.monotonic() - t_start > max_time - 1.0:
            break

        c = fft_convolve(a)
        max_c = c.max()

        # Increase active set size as we converge (for more precise gradient)
        frac = 0.05 if no_improve < 50 else (0.03 if no_improve < 200 else 0.01)
        grad = compute_subgrad(a, c, n_active_frac=frac)

        # Momentum update
        mom = mu * mom - lr * grad
        a_new = project_simplex(a + mom)

        c_new = fft_convolve(a_new)
        score_new = 2.0 * n * c_new.max()

        if score_new < local_best_score - 1e-10:
            local_best_score = score_new
            local_best = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        iteration += 1

        # Decay learning rate slowly
        lr *= 0.9997

        # Restart if stuck
        if no_improve > patience:
            # Blend best with original and reset
            a = 0.6 * local_best + 0.4 * a0
            a = project_simplex(a)
            mom *= 0.3
            no_improve = 0
            lr = base_lr * (0.8 + 0.4 * np.random.default_rng(seed + iteration).random())

    return local_best, local_best_score


def lp_refine(a0, max_time, seed=42):
    """LP-based alternating method for final refinement."""
    from scipy.optimize import linprog
    n = len(a0)
    a = a0.copy()
    c = fft_convolve(a)
    cur_score = 2.0 * n * c.max()
    local_best = a.copy()
    local_best_score = cur_score

    t_start = time.monotonic()
    step = 0

    while time.monotonic() - t_start < max_time - 2.0:
        c = fft_convolve(a)
        M = c.max()

        # Build constraints: center + top 4
        ks = [n - 1]
        top4 = np.argsort(c)[::-1][:4]

            if k != n - 1 and k not in ks:

        K = len(ks)
        A_ub = np.zeros((K, n))
        for i, k in enumerate(ks):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_ub[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

        try:
            res = linprog(-np.ones(n), A_ub=A_ub, b_ub=np.full(K, M),
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': 5})
            if not res.success:
                break
            g = res.x
            s_g = g.sum()
            if s_g < 1e-10:
                break

            # Try different blend factors
            best_blend = None
            best_blend_score = np.inf
            for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
                a_new = (1.0 - t) * a + t * g * (a.sum() / s_g)
                a_new = project_simplex(a_new)
                c_new = fft_convolve(a_new)
                s_new = 2.0 * n * c_new.max()
                if s_new < best_blend_score:
                    best_blend_score = s_new
                    best_blend = a_new.copy()

            if best_blend is not None and best_blend_score < local_best_score:
                local_best_score = best_blend_score
                local_best = best_blend.copy()
                a = best_blend.copy()
                step += 1
            elif best_blend is not None:
                a = best_blend.copy()
                step += 1
            else:
                break
        except Exception:
            break

    return local_best, local_best_score


def make_starts(n, seed):
    """Generate diverse starting points of length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    starts = []

    # Uniform
    starts.append(np.ones(n, dtype=np.float64) / n)

    # Cosine window (Hann-like)
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    starts.append(w / w.sum())

    # Shifted cosine
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x + 0.5) / n))
    starts.append(w / w.sum())

    # Double-sided exponential
    w = np.exp(-1.5 * x / n) + np.exp(-1.5 * (n - 1 - x) / n)
    starts.append(w / w.sum())

    # Two Gaussian peaks
    w = np.exp(-((x - n // 3) ** 2) / (n / 5.0)) + np.exp(-((x - 2 * n // 3) ** 2) / (n / 5.0))
    starts.append(w / w.sum())

    # Three Gaussian peaks
    w = (np.exp(-((x - n // 4) ** 2) / (n / 8.0)) +
         np.exp(-((x - n // 2) ** 2) / (n / 8.0)) +
         np.exp(-((x - 3 * n // 4) ** 2) / (n / 8.0)))
    starts.append(w / w.sum())

    # Random
    w = rng.random(n)
    starts.append(w / w.sum())

    # Chirp magnitude
    w = np.abs(np.cos(np.pi * 3.0 * (x / n) ** 2))
    if w.sum() > 1e-10:
        starts.append(w / w.sum())

    # Random cosine sum
    phases = rng.uniform(0, 2 * np.pi, size=4)
    freqs = rng.integers(1, 5, size=4).astype(float)
    w = np.zeros(n)
    for f, p in zip(freqs, phases):
        w += np.abs(np.cos(f * np.pi * x / n + p))
    if w.sum() > 1e-10:
        starts.append(w / w.sum())

    return starts
