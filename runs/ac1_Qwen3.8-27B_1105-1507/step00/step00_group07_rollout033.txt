from scipy.optimize import linprog
from scipy.sparse import lil_matrix

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    
    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # A[k, j] = f[k-j] if 0 <= k-j < n
    A = lil_matrix((2*n-1, n))
    for j in range(n):
        for k in range(j, j+n):  # k-j ranges from 0 to n-1
            A[k, j] = f[k-j]
    A = A.tocsr()
    b_rhs = M * np.ones(2*n-1)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_rhs, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

def _build_matrix(f, n):
    """Build the sparse constraint matrix for the LP."""
    # For each column j (0 to n-1):
    #   rows: j, j+1, ..., min(j+n-1, 2n-2)
    #   values: f[0], f[1], ..., f[min(n-1, 2n-2-j)]
    
    all_rows = []
    all_cols = []
    all_vals = []
    
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        if length <= 0:
            continue
        rows = np.arange(j, j + length)
        cols = np.full(length, j)
        vals = f[:length]
        mask = vals > 1e-15
        if np.any(mask):
            all_rows.append(rows[mask])
            all_cols.append(cols[mask])
            all_vals.append(vals[mask])
    
    if not all_rows:
        return None
    
    return csr_matrix(
        np.concatenate(all_vals),
        (np.concatenate(all_rows), np.concatenate(all_cols)),
        shape=(2 * n - 1, n)

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _build_conv_matrix(n):
    """Build sparse matrix A where A[k,j] = 1 if 0<=k-j<n (structure only)."""
    # For LP: A[k,j] = f[k-j], but structure is fixed per n
    rows_list = []
    cols_list = []
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        if length <= 0:
            continue
        rows_list.append(np.arange(j, j + length))
        cols_list.append(np.full(length, j))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    return rows, cols


def _solve_lp(f, M, n, rows, cols):
    """Solve: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k=0..2n-2."""
    # A[k,j] = f[k-j] if 0<=k-j<n
    vals = f[rows - cols]
    mask = vals > 1e-15
    if not np.any(mask):
        return None
    A = csr_matrix((vals[mask], (rows[mask], cols[mask])), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)

    res = linprog(
        -np.ones(n), A_ub=A, b_ub=b,
        bounds=[(0, None)] * n,
        method='highs', options={'presolve': True}
    )
    if res.success and res.x is not None:
        return res.x
    return None


def _optimize_lp(f, n, rng, time_limit):
    """LP-based iteration to minimize 2n*max(conv) for sum(f)=1."""
    deadline = time.monotonic() + time_limit
    f = np.maximum(f, 0)
    f = f / f.sum()

    rows, cols = _build_conv_matrix(n)

    M = float(np.max(np.convolve(f, f)))
    best_M = M
    best_f = f.copy()

    t_blend = 0.05
    fail_count = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        g = _solve_lp(f, M, n, rows, cols)

        if g is None:
            fail_count += 1
            if fail_count > 15:
                f = best_f + rng.normal(0, 0.005, n)
                f = np.maximum(f, 0)
                s = f.sum()
                if s < 1e-10:
                    break
                f = f / s
                M = float(np.max(np.convolve(f, f)))
                fail_count = 0
                t_blend = 0.05
            continue

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        # If g is nearly identical to f, perturb
        if np.max(np.abs(g - f)) < 1e-10:
            f = best_f + rng.normal(0, 0.01, n)
            f = np.maximum(f, 0)
            s = f.sum()
            if s < 1e-10:
                break
            f = f / s
            M = float(np.max(np.convolve(f, f)))
            t_blend = 0.1
            continue

        # Blend
        f_new = (1 - t_blend) * f + t_blend * g
        c_new = np.convolve(f_new, f_new)
        M_new = float(np.max(c_new))

        if M_new <= M * (1 + 1e-9):
            f = f_new
            M = M_new
            fail_count = 0
            if M < best_M:
                best_M = M
                best_f = f.copy()
        else:
            t_blend *= 0.5
            fail_count += 1
            if fail_count > 40:
                # Escape local optimum
                f = best_f + rng.normal(0, 0.01, n)
                f = np.maximum(f, 0)
                s = f.sum()
                if s < 1e-10:
                    break
                f = f / s
                M = float(np.max(np.convolve(f, f)))
                t_blend = 0.1
                fail_count = 0

        # Gradually increase t_blend
        if it % 40 == 0:
            t_blend = min(0.25, t_blend * 1.4)

        if it % 200 == 0:
            print(f"    it={it}, best_M={best_M:.8f}, score≈{2.0*n*best_M:.6f}")

    score = 2.0 * n * best_M
    return best_f, score


def _random_perturb_search(f_best, score, n, rng, time_limit):
    """Try random perturbations to improve upon a local optimum."""
    deadline = time.monotonic() + time_limit
    best = f_best.copy()
    best_score = score

    while time.monotonic() < deadline:
        # Block perturbation
        candidate = best.copy()
        width = max(2, n // 50)
        start = int(rng.integers(0, max(1, n - width)))
        candidate[start:start + width] *= np.exp(rng.normal(0, 0.15))
        candidate = np.maximum(candidate, 0)
        s = candidate.sum()
        if s < 1e-10:
            continue
        candidate = candidate / s

        c = np.convolve(candidate, candidate)
        new_score = 2.0 * n * float(np.max(c))

        if new_score < best_score:
            best = candidate
            best_score = new_score

    return best, best_score

    """
    Minimizes 2n*max(convolve(a,a))/sum(a)^2 for nonnegative sequences.
    Uses iterated LP (max sum(g) s.t. g>=0, (f*g)<=M) with blending,
    combined with random perturbations. Tries multiple lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    best_score = np.inf
    best_seq = None

    # Get parent sequence
    parent = None
    try:

        if p.size > 0 and np.all(np.isfinite(p)) and float(p.sum()) > 0.01:
            parent = p
    except Exception:
        pass

    # Lengths to try (smaller first for quick wins)
    lengths = [128, 256, 512, 1024, 2048]

    for n in lengths:
        if time.monotonic() > deadline - 30:
            break

        remaining = deadline - time.monotonic()
        # Allocate time proportional to potential (smaller n = cheaper iterations)
        time_for_n = min(350.0, remaining * 0.35)

        # Random Dirichlet (spread out)
