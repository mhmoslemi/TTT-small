Step function optimization using LP-guided local search with FFT-based

import numpy as np

def _fft_convolve(a, b):
    """Full convolution via FFT."""
    n_out = len(a) + len(b) - 1
    fsize = 1 << (n_out - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    result = np.fft.irfft(fa * fb, fsize)[:n_out]
    return np.maximum(result, 0.0)  # clip tiny negatives from FFT


def _fast_score(a):
    """Compute score = 2n*max(conv)/sum^2 quickly."""
    n = len(a)

    if s < 0.01:
        return np.inf
