def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.fft(a, size)
    fb = np.fft.fft(b, size)
    return np.real(np.fft.ifft(fa * fb))[:n+m-1]

    b_shifted = b - b.max()
    p = np.exp(tau * b_shifted)

    v_hat = v / (1 - beta2**t)
