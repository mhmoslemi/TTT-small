        if jump_period > 0 and step_count % jump_period == 0:
            # Random perturbation to escape local minima
            mask = np.random.random(n) < 0.3
            perturbation = np.random.normal(0, 0.1, n) * mask
            a_pert = a * (1.0 + perturbation)
            a_pert = np.maximum(0.0, a_pert)
            s_pert = np.sum(a_pert)
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert
                c_pert = fast_conv(a_pert)
                score_pert = 2.0 * n * np.max(c_pert)
                if score_pert < best_score:
                    a = a_pert
                    best_score = score_pert
                    best_a = a_pert.copy()
                else:
                    a = a.copy()  # keep original
            v = np.zeros(n)

    return best_a, best_score


def lp_refine(a, deadline, max_outer=30, max_constraints=300):
    n = len(a)
    a = np.maximum(0.0, a).copy()
    s = np.sum(a)
    if s < 1e-20:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    c = fast_conv(a)
    best_score = 2.0 * n * np.max(c)

    c_a = fast_conv(a)
    M = np.max(c_a)
    n_init = min(100, 2 * n - 1)
    active_list = list(np.argpartition(c_a, -n_init)[-n_init:])

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        c_a = fast_conv(a)
        M = np.max(c_a)

        rows = []
        cols = []
        vals = []
        b = []

        for k in active_list:
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start > j_end:
                continue
            js = np.arange(j_start, j_end + 1)
            row_idx = len(b)
            rows.extend([row_idx] * len(js))
            cols.extend(js.tolist())
            vals.extend(a[ki - js].tolist())
            b.append(M)

        if len(b) == 0:
            break

        A = sparse.csr_matrix((vals, (rows, cols)), shape=(len(b), n))

        try:
            res = linprog(-np.ones(n), A_ub=A, b_ub=np.array(b),
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True})
        except Exception:
            break

        if not res.success:
            break

        g = np.maximum(0.0, res.x)
        g_sum = np.sum(g)
        if g_sum < 1e-20:
            break
        g = g / g_sum

        c_g = fast_conv(g)
        score_g = 2.0 * n * np.max(c_g)

        if score_g < best_score:
            best_score = score_g
            best_a = g.copy()

        # Find most violated constraint not in active set
        active_set = set(int(k) for k in active_list)
        max_viol = 0.0
        worst_k = -1
        for k in range(len(c_g)):
            if int(k) not in active_set:
                viol = c_g[k] - M
                if viol > max_viol:
                    max_viol = viol
                    worst_k = int(k)

        if worst_k >= 0 and max_viol > M * 0.005:
            active_list.append(worst_k)
            if len(active_list) > max_constraints:
                c_full = fast_conv(a)
                top = np.argpartition(c_full, -max_constraints)[-max_constraints:]
                active_list = list(int(k) for k in top)
                if worst_k not in active_list:
                    active_list[-1] = worst_k

        t_mix = 0.5
        a = (1.0 - t_mix) * a + t_mix * g
        a = np.maximum(0.0, a)
        s_a = np.sum(a)
        if s_a < 1e-20:
            a = best_a.copy()
            continue
        a = a / s_a

    return best_a, best_score


def try_init_pair(n, init, deadline, start_time):
    """Run subgradient then LP on one initialization. Return (seq, score)."""
    a = np.maximum(0.0, init).copy()
    s = np.sum(a)
    if s < 1e-10:
        return None, np.inf
    a = a / s

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return a, 2.0 * n * np.max(fast_conv(a))

    sub_deadline = time.monotonic() + remaining * 0.6
    a_sg, score_sg = subgradient_search(a, sub_deadline, K=20, momentum=0.3)

    if time.monotonic() > deadline - 30:
        return a_sg, score_sg

    lp_deadline = min(time.monotonic() + remaining * 0.35, deadline - 15)
    if score_sg < 2.5 and lp_deadline > time.monotonic() + 10:
        a_lp, score_lp = lp_refine(a_sg, lp_deadline)
        if score_lp < score_sg:
            return a_lp, score_lp

    return a_sg, score_sg


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Searches for non-negative sequences minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FFT subgradient descent with momentum and LP refinement (cutting planes).
    Explores multiple initializations and sequence lengths within time budget.
    """
    deadline = time.monotonic() + budget_s - 10.0
    start_time = time.monotonic()
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Sequence lengths to try
    n_list = [4806, 2048, 1024, 512]

    # Build initialization pool per length
    for n in n_list:
        if time.monotonic() > deadline - 90:
            break

        inits = []
        inits.append(("g35", make_gaussian_init(n, 0.35)))
        inits.append(("g15", make_gaussian_init(n, 0.15)))
        inits.append(("tri", make_triangular_init(n)))
        inits.append(("cos", make_cosine_init(n)))
        inits.append(("rip4", make_ripple_init(n, 4.0, 0.4)))
        inits.append(("rip8", make_ripple_init(n, 8.0, 0.3)))
        inits.append(("blk4", make_block_init(n, 4, 3.0)))
        inits.append(("blk8", make_block_init(n, 8, 2.0)))

        # Include given sequence
        try:
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            hs = np.maximum(0.0, hs)
            if len(hs) >= n:
                inits.append(("given", hs[:n].copy()))
            elif len(hs) > 0:
                interp = np.interp(
                    np.linspace(0, n - 1, n),
                    np.linspace(0, len(hs) - 1, len(hs)),
                    hs)
                inits.append(("gint", interp))
        except (NameError, Exception):
            pass

        # Shuffle for diversity
        order = list(range(len(inits)))
        rng.shuffle(order)
        inits = [inits[i] for i in order]

        # Time allocation per init
        time_remaining = deadline - time.monotonic()
        time_per_init = max(25, min(200, time_remaining / max(1, len(inits)) - 5))

        for name, init in inits:
            if time.monotonic() > deadline - 25:
                break

            seq, score = try_init_pair(n, init, deadline, start_time)

            if seq is not None and np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:.0f}s] n={n} init={name} score={score:.6f}", flush=True)

        # After exploring all inits for this n, do a focused LP on the best
        if best_seq is not None and time.monotonic() < deadline - 60:
            n_best = len(best_seq)
            if n_best == n:
                lp_deadline = min(time.monotonic() + 60, deadline - 20)
                a_final, score_final = lp_refine(best_seq, lp_deadline, max_outer=40, max_constraints=500)
                if score_final < best_score:
                    best_score = score_final
                    best_seq = a_final.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:.0f}s] n={n} final LP score={score_final:.6f}", flush=True)

    # Final smoothing attempt on best
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        a_sm = gaussian_filter1d(best_seq, sigma=max(2, n_best // 200))
        a_sm = np.maximum(0.0, a_sm)
        s_sm = np.sum(a_sm)
        if s_sm > 1e-10:
            a_sm = a_sm / s_sm
            score_sm = 2.0 * n_best * np.max(fast_conv(a_sm))
            if score_sm < best_score:
                best_score = score_sm
                best_seq = a_sm.copy()

    # Verify with actual evaluator
    if best_seq is not None:
        final_check = evaluate_sequence([float(x) for x in best_seq])
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:.0f}s] FINAL n={len(best_seq)} eval={final_check:.6f}", flush=True)
        if not np.isfinite(final_check):
            # Fallback
            best_seq = make_gaussian_init(1024, 0.3)
            print("[FALLBACK] using gaussian init", flush=True)

    if best_seq is None:
        best_seq = make_gaussian_init(1024, 0.3)
        print("[FALLBACK] no valid sequence found, using gaussian", flush=True)

    return [float(x) for x in best_seq]
