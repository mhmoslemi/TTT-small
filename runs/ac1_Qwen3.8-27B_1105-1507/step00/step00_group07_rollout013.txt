import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp

def build_conv_matrix(a):
    """Build sparse matrix A such that A @ b = full_convolve(a, b)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for j in range(n):
            rows.append(i + j)
            cols.append(i)
            vals.append(a[j])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_step(a, M):
    """Solve the LP: max sum(b) s.t. A@b <= M, b >= 0. Returns b."""
    n = len(a)
    A = build_conv_matrix(a)
    b = cp.Variable(n)
    constraints = [A @ b <= M, b >= 0]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, verbose=False, eps=1e-7)
    if b.value is None:
        return None
    return np.maximum(0, b.value)

def score_of(a):
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * M / (s * s)

"""
LP-based iterative optimization for minimizing the peak autocorrelation of non-
negative sequences. Uses the "solve LP, update" scheme from the state of the art.
"""
import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp


def _build_conv_matrix(a):
    """Build sparse matrix A such that A @ b = np.convolve(a, b, mode='full')."""
    n = len(a)
    row_indices = []
    col_indices = []
    data = []
    for i in range(n):
        for j in range(n):
            row_indices.append(i + j)
            col_indices.append(i)
            data.append(a[j])
    return sp.csr_matrix((data, (row_indices, col_indices)), shape=(2*n-1, n))


def _score(a):
    """Compute the objective value."""
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    return 2.0 * n * M / (s * s)


def _lp_optimize(n, a_init, time_limit=60.0, max_iters=100, seed=0):
    """Run LP-based optimization starting from a_init."""
    rng = np.random.default_rng(seed)
    a = np.array(a_init, dtype=np.float64).copy()
    s0 = np.sum(a)
    if s0 < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s0  # normalize to sum 1
    
    best_a = a.copy()
    best_score = _score(a)
    deadline = time.monotonic() + time_limit
    
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        
        t0 = time.monotonic()
        conv = np.convolve(a, a)
        M = np.max(conv)
        cur_score = 2.0 * n * M  # since sum(a) = 1
        
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        
        # Build constraint matrix and solve LP
        A = _build_conv_matrix(a)
        b = cp.Variable(n)
        constraints = [A @ b <= M, b >= 0]
        objective = cp.Maximize(cp.sum(b))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS, verbose=False, eps=1e-7)
        
        if b.value is None:
            break
        b_val = np.maximum(0, b.value)
        sum_b = np.sum(b_val)
        if sum_b < 1e-12:
            break
        
        # Line search over step sizes
        best_t = 0.0
        best_t_score = cur_score
        best_a_new = a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * b_val / sum_b
            a_try = np.maximum(0, a_try)
            s_try = _score(a_try)
            if s_try < best_t_score:
                best_t_score = s_try
                best_t = t
                best_a_new = a_try
        
        if best_t == 0.0:

            for t in [0.01, 0.05, 0.1]:

                a_try = np.maximum(0, a_try)
                s_try = _score(a_try)
                if s_try < best_t_score:
                    best_t_score = s_try
                    best_t = t
                    best_a_new = a_try
        
        a = best_a_new
        
        # Print progress occasionally
        if it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter {it}: score={cur_score:.6f}, LP time={elapsed:.2f}s")
