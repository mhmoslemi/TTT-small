    for k in k_stars:
        idx = k - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        grad[valid] += 2.0 * a[idx[valid]]

import numpy as np
import time
import math

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
- Multiple diverse initializations (parent, random, structured)
- Top-K subgradient descent with adaptive learning rate
- Noise-based escape mechanism for flat regions
- LP-based jump steps for sequences with n <= 2048
- Final fine-tuning phase
All helper functions are top-level; no closures or lambdas.
"""


def _conv(a):
    return np.convolve(a, a)


def _subgrad_full(a, k_stars):
    n = len(a)
    grad = np.zeros(n)
    for k in k_stars:
        idx = k - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        grad[valid] += 2.0 * a[idx[valid]]
    return grad / max(1, len(k_stars))


def _project_nn_sum(a, S):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.full_like(a, S / max(1, len(a)))
    return a * (S / s)


def _topk_indices(conv, k):
    if k >= conv.size:
        return np.arange(conv.size)
    part = np.argpartition(conv, -k)[-k:]
    return part[np.argsort(conv[part])[::-1]]


def _gd_run(a_init, S, time_limit, lr_init, top_k, seed, noise_thresh, noise_frac, noise_mag):
    a = a_init.astype(np.float64).copy()
    n = len(a)
    rng = np.random.default_rng(seed)
    start_t = time.monotonic()
    best_h = float('inf')
    best_a = a.copy()
    stagnant = 0
    global_stagnant = 0
    step = 0

    while time.monotonic() - start_t < time_limit:
        conv = _conv(a)
        h = float(conv.max())

        if h < best_h * (1.0 - 1e-15):
            best_h = h
            best_a = a.copy()
            stagnant = 0
            global_stagnant = 0
        else:
            stagnant += 1
            global_stagnant += 1

        k_stars = _topk_indices(conv, top_k)
        grad = _subgrad_full(a, k_stars)

        lr = lr_init / (1.0 + 0.00005 * step)
        a_new = a - lr * grad

        if stagnant >= noise_thresh:
            frac = min(noise_frac, 0.15)
            n_noise = max(1, int(n * frac))
            idx = rng.choice(n, size=n_noise, replace=False)
            scale = noise_mag * (1.0 + 0.2 * (global_stagnant // noise_thresh))
            a_new[idx] *= np.exp(rng.normal(0.0, scale, n_noise))
            a_new = _project_nn_sum(a_new, S)
            stagnant = 0

        a = _project_nn_sum(a_new, S)
        step += 1

    return best_a, best_h


def _try_lp(f, n, time_budget):
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix

        conv = _conv(f)
        M = float(conv.max())

        rows_list = []
        cols_list = []
        vals_list = []

        for k in range(2 * n - 1):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            js = np.arange(j_lo, j_hi + 1)
            rows_list.extend([k] * len(js))
            cols_list.extend(js.tolist())
            vals_list.extend(f[k - js].tolist())

        if not rows_list:
            return None

        A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max(10, time_budget)})

        if res.success and res.x is not None:
            b = res.x
            s_b = float(b.sum())
            if s_b > 1e-10:
                S = math.sqrt(2.0 * n)
                g = b * (S / s_b)
                return g
    except Exception:
        pass
    return None
