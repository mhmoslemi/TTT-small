"""
Minimize J(a)=2n*max(conv(a,a))/sum(a)^2 (scale invariant -> work at sum=1).

Core method: smooth the non-differentiable max with a log-sum-exp surrogate
(f(a)=cmax + logsumexp(tau*(c-cmax))/tau), compute its exact gradient using the
identity grad[j]=sum_k w_k a[k-j] + a[j] (w=softmax(tau*c)) evaluated via a single
FFT-less O(n^2)-free convolve, push the iterate onto the simplex {a>=0,sum=1} with a
water-filling projection, and adapt the step with Adam.  tau is annealed so the
surrogate sharpens toward the true peak.  Multiple lengths and several diverse
random/structured starts are tried; a banded sparse LP (max sum(g) s.t. g>=0,
(a*g)_k<=M) provides a principled alternative descent direction used to polish and
escape plateaus.  The exact evaluator is used as the authoritative scorer.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse


# ---------------------------------------------------------------------------
# Helpers (top-level, no closures, no lambdas)
# ---------------------------------------------------------------------------

def _stable_conv(a):
    """Full linear convolution of a real vector with itself."""
    return np.convolve(a, a)


def _smoothed_max_and_weights(c, tau):
    """Return (smoothed_max, softmax_weights) of c at temperature 1/tau."""
    cmax = float(np.max(c))
    shifted = tau * (c - cmax)
    m = float(np.max(shifted))
    e = np.exp(shifted - m)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        # degenerate fallback: uniform over argmax
        idx = np.where(c >= cmax - 1e-12)[0]
        w = np.zeros_like(c)
        w[idx] = 1.0 / max(1, len(idx))
        return cmax, w
    w = e / s
    val = cmax + (m - (-m)) / tau  # m - (-m) == 0 placeholder replaced below
    # correct logsumexp: log(sum exp(shifted)) = m + log(sum e)
    val = cmax + (m + np.log(s)) / tau
    return val, w


def _gradient_of_smoothmax(a, w):
    """grad_j = sum_k w_k a[k-j]  +  a_j*sum(w)  ==  cross[j] + a_j (sum w =1)."""
    n = a.size
    # cross[j] = sum_m a_m w[m+j] = convolve(a[::-1], w, 'full')[n-1+j]
    cc = np.convolve(a[::-1], w, 'full')
    cross = cc[n - 1: n - 1 + n]
    return cross + a * float(np.sum(w))


def _simplex_project(v):
    """Project v onto {x>=0, sum(x)=1} (Euclidean)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if not np.all(np.isfinite(v)) or float(np.sum(v)) <= 0.0:
        return np.ones(n, dtype=np.float64) / n
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rng = np.arange(1, n + 1)
    cond = (u * rng) > (css - 1.0)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n, dtype=np.float64) / n
    rho = int(rho_idx[0])
    theta = (css[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = float(np.sum(out))
    if s <= 0.0:
        return np.ones(n, dtype=np.float64) / n
    return out / s


def _normalize_sum1(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, None)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s


def _build_banded_matrix(n):
    """Constraint structure for (a*g)_k <= M: returns (rows, cols) index lists per k."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        for j in range(jlo, jhi + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int64), np.array(cols, dtype=np.int64)


def _lp_direction(a, M, allow_fail=True):
    """Solve max sum(g) s.t. g>=0, (a*g)_k <= M. Returns g (or None)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    rows, cols = _build_banded_matrix(n)
    vals = np.empty_like(rows, dtype=np.float64)
    # value A[k,j] = a[k-j]
    for t in range(rows.size):
        k = rows[t]
        j = cols[t]
        idx = k - j
        vals[t] = a[idx] if 0 <= idx < n else 0.0
    if not np.any(vals > 0) or M <= 0:
        return None
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 8.0})
    except Exception:
        return None
    if res is None or not getattr(res, 'success', False):
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.where(np.isfinite(g) & (g > 0), g, 0.0)
    if float(np.sum(g)) <= 0.0:
        return None
    return g


def _score_and_record(seq, state):
    ls = [float(x) for x in seq]
    # guard the evaluator's lower bound on the sum
    s = float(np.sum(ls))
    if s < 0.01:

        ls = [float(x) * scale for x in ls]
    sc = float(evaluate_sequence(ls))
    if np.isfinite(sc) and (state['best_score'] is None or sc < state['best_score']):
        state['best_score'] = sc
        state['best_seq'] = [float(x) for x in np.clip(np.asarray(ls), 0.0, 1000.0)]
        print("  new best score = %.6f  (n=%d)" % (sc, len(ls)))
    return sc

def _init_uniform(n, rng):
    return np.ones(n, dtype=np.float64)


def _init_random_smooth(n, rng, smooth=0.45):
    base = rng.exponential(1.0, size=n) + 0.05
    # simple moving-average smoothing for low frequency structure
    k = max(3, int(0.05 * n))
    kernel = np.ones(k) / k
    base = np.convolve(base, kernel, mode='same')
    # blend with raw random for some high-freq content
    base = (1.0 - smooth) * base + smooth * (1.0 / (k + 1)) * (1.0 + 0.7 * rng.standard_normal(n))
    base = np.clip(base, 0.0, None)
    return base


def _init_raised_cosine(n, rng):
    idx = np.arange(n, dtype=np.float64)
    x = (idx - (n - 1) / 2.0) / max(1.5, (n - 1) / 2.0)
    txt = np.clip((x - 1.0) / -2.0, 0.0, 1.0)
    base = 0.5 * (1.0 + np.cos(np.pi * txt))
    base = np.clip(base, 1e-4, None)
    base += 0.02 * rng.exponential(1.0, size=n)
    return base


def _init_from_parent(n, rng):

            p = np.interp(xt, xf, p)
        p = p + 0.02 * rng.exponential(1.0, size=n)
        p = np.clip(p, 0.0, None)
        return p
    except Exception:
        return _init_random_smooth(n, rng)


_INIT_FUNCS = [

# ---------------------------------------------------------------------------
# Single-sequence optimizer: Adam + log-sum-exp smoothing + simplex projection
# ---------------------------------------------------------------------------

def _optimize_one(a0, tau0, tau1, deadline, do_lp, lp_every=40):
    a = _normalize_sum1(a0)
    n = a.size
    M = float(np.max(_stable_conv(a)))

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5
    best_local = a.copy()
    best_local_M = M

    while time.monotonic() < deadline:
        t += 1
        # anneal tau
        tau = tau0 * (tau1 / tau0) ** (t / max(1, 400))
        tau = min(tau, tau1)

        c = _stable_conv(a)
        M = float(np.max(c))
        _, w = _smoothed_max_and_weights(c, tau)
        grad = _gradient_of_smoothmax(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        mhat = m / (1 - beta1 ** t)
        vhat = v / (1 - beta2 ** t)
        lr = lr0 / (np.sqrt(t) + 1.0)
        step = -lr * mhat / (np.sqrt(vhat) + eps)

        a = _simplex_project(a + step)

        # occasional principled LP polish direction
        if do_lp and (t % lp_every == 0):
            g = _lp_direction(a, M)
            if g is not None:
                gs = float(np.sum(g))
                if gs > 1.0 + 1e-4:
                    gn = g / gs
                    for tt in (1.0, 0.7, 0.5, 0.3, 0.15):
                        cand = _simplex_project((1.0 - tt) * a + tt * gn)
                        cm = float(np.max(_stable_conv(cand)))
                        if cm < M - 1e-9:
                            a = cand
                            M = cm
                            break

        if M < best_local_M - 1e-9:
            best_local_M = M
            best_local = a.copy()

        if M < 1e-12:
            break

    return best_local, best_local_M

def propose_candidate(budget_s=1500):
    """Search for the lowest-scoring non-negative sequence under the budget."""
    rng = np.random.default_rng(12345)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s) - 20.0)

    state = {'best_score': None, 'best_seq': None}

    # establish a safe fallback with the parent / uniform
    try:
        fb = np.asarray(height_sequence_1, dtype=np.float64)
        fb = np.clip(fb, 0.0, None)
        fb = np.where(np.isfinite(fb), fb, 0.0)
        if fb.ndim != 1 or fb.size == 0 or float(np.sum(fb)) <= 0:
            fb = None
    except Exception:
        fb = None
    if fb is None:
        fb = np.ones(256, dtype=np.float64)
    _score_and_record(fb, state)
    if state['best_seq'] is None:
        state['best_seq'] = [float(x) for x in fb]

    lengths = [256, 384, 512, 640, 768, 1024, 1280]
    # give more of the budget to mid/large lengths which have more attack surface
    weight = {256: 0.6, 384: 0.8, 512: 1.0, 640: 1.1, 768: 1.2, 1024: 1.4, 1280: 1.6}
    total_w = sum(weight[n] for n in lengths)
    avail = max(60.0, deadline - time.monotonic())

    for n in lengths:
        n_budget = avail * (weight[n] / total_w)
        n_deadline = time.monotonic() + n_budget
        if time.monotonic() >= n_deadline:
            continue
        print("[len=%d] budget=%.0fs" % (n, n_budget), flush=True)

        # trial a handful of diverse starts, splitting the length budget
        tries = 4
        per = max(5.0, (min(n_deadline, deadline) - time.monotonic()) / tries)
        for it in range(tries):
            sub_deadline = min(n_deadline, deadline, time.monotonic() + per)
            if time.monotonic() >= sub_deadline:
                break
            tag, fn = _INIT_FUNCS[(it + i_name_index(it, n)) % len(_INIT_FUNCS)]
            a0 = _normalize_sum1(fn(n, rng))
            if time.monotonic() >= sub_deadline:
                break
            res, _ = _optimize_one(a0, tau0=2.0, tau1=800.0,
                                   deadline=sub_deadline,
                                   do_lp=True, lp_every=30)
            _score_and_record(_normalize_sum1(res), state)

        # final LP polish on the best sequence of this length (re-extract best)
        if state['best_seq'] is not None and len(state['best_seq']) == n \
                and time.monotonic() < min(n_deadline, deadline):
            polish_deadline = min(n_deadline, deadline, time.monotonic() + max(8.0, 0.25 * n_budget))
            best_n = _normalize_sum1(state['best_seq'])
            res, _ = _optimize_one(best_n, tau0=50.0, tau1=2000.0,
                                   deadline=polish_deadline, do_lp=True, lp_every=10)
            _score_and_record(_normalize_sum1(res), state)

    # global final polish on overall best
    if state['best_seq'] is not None and time.monotonic() < deadline:
        bd = deadline - 2.0
        bb = _normalize_sum1(state['best_seq'])
        res, _ = _optimize_one(bb, tau0=100.0, tau1=5000.0, deadline=bd,
                               do_lp=True, lp_every=8)
        _score_and_record(_normalize_sum1(res), state)

    best = state['best_seq']
    if best is None:
        best = [float(x) for x in fb]
    best = [float(x) for x in np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)]
    print("FINAL best_score = %s  n=%d  time=%.0fs" % (
        state['best_score'], len(best), time.monotonic() - t_start), flush=True)
    return best


def i_name_index(it, n):
    """Deterministic small index to vary initialisation across trials/lengths."""
    return (it + int(n) % len(_INIT_FUNCS)) % len(_INIT_FUNCS)
