def build_conv_matrix(f):
    """Build sparse matrix A such that A @ b = convolve(f, b)."""
    n = len(f)

    cols = []
    data = []
    for k in range(2*n - 1):

            if f_new.sum() <= 1e-10:
                continue
            f_new = f_new / f_new.sum()
            score_new = compute_score(f_new)
            if score_new < best_score * (1.0 - 1e-10):
                f = f_new
                best_score = score_new
                best_f = f.copy()
                improved = True
                if verbose and (it % 5 == 0 or it == max_iters - 1):
                    print(f"    LP iter {it}: score={best_score:.6f} (t={t}, lp={lp_time:.1f}s)")
                break
        if not improved:
            break
    return best_f, best_score


def local_search(f, deadline, rng, verbose=True):
    """Adaptive local search with multiple perturbation strategies."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = f / f.sum()
    score = compute_score(f)
    best_f = f.copy()
    best_score = score
    base_sigma = 0.08
    it = 0
    stale = 0

    while time.monotonic() < deadline:
        it += 1
        f_new = f.copy()
        strategy = rng.integers(4)

        if strategy == 0:
            width = max(1, int(rng.integers(max(1, n // 500), max(2, n // 15))))
            start = int(rng.integers(0, max(1, n - width)))
            f_new[start:start + width] *= np.exp(rng.normal(0.0, base_sigma, size=width))
        elif strategy == 1:
            width = max(1, int(rng.integers(max(1, n // 100), max(2, n // 10))))
            start = int(rng.integers(0, max(1, n - width)))
            f_new[start:start + width] = f_new[start:start + width] * np.exp(rng.normal(0.0, base_sigma * 1.5, size=width))
            f2 = f.copy()
            w2 = max(1, int(rng.integers(max(1, n // 200), max(2, n // 40))))
            s2 = int(rng.integers(0, max(1, n - w2)))
            f2[s2:s2 + w2] *= np.exp(rng.normal(0.0, base_sigma, size=w2))
            f_new = 0.5 * (f_new + f2)
        elif strategy == 2:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                frac = float(rng.uniform(0.05, 0.4))
                amt = f_new[i] * frac
                f_new[i] -= amt
                f_new[j] += amt
            width = max(1, int(rng.integers(max(1, n // 2000), max(2, n // 500))))
            start = int(rng.integers(0, max(1, n - width)))
            f_new[start:start + width] *= np.exp(rng.normal(0.0, base_sigma * 0.5, size=width))
        else:
            # Symmetrize bias
            f_new = (1.0 - 0.3) * f_new + 0.3 * f_new[::-1]
            width = max(1, int(rng.integers(max(1, n // 100), max(2, n // 30))))
            start = int(rng.integers(0, max(1, n - width)))
            f_new[start:start + width] *= np.exp(rng.normal(0.0, base_sigma, size=width))

        f_new = np.clip(f_new, 0.0, None)
        if f_new.sum() > 1e-10:
            score_new = compute_score(f_new)
            if score_new < score:
                f = f_new
                score = score_new
                stale = 0
                if score < best_score:
                    best_score = score
                    best_f = f.copy()
                    if verbose and it % 5000 == 0:
                        print(f"    LS iter {it}: score={best_score:.6f}")
            else:
                stale += 1
                if stale % 5000 == 0:
                    base_sigma = max(0.02, base_sigma * 0.9)
        else:
            stale += 1

    return best_f, best_score


def guided_local_search(f, deadline, rng):
    """Local search guided by peak lag information."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = f / f.sum()
    best_f = f.copy()
    best_score = compute_score(f)
    it = 0

    while time.monotonic() < deadline:
        it += 1
        conv = np.convolve(f, f)
        k_star = int(np.argmax(conv))
        # Subgradient: d(conv(c,c)[k*])/d(c[j]) = c[(k*-j) mod ... ]
        # (c*c)[k] = sum_i c[i]*c[k-i], so d/dc[j] = c[k-j] if 0<=k-j<n
        sg = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                sg[j] = c[idx] if False else f[idx]
        # Reduce coordinates where sg is large, increase where sg is small
        top_k = max(1, n // 100)
        top_indices = np.argsort(sg)[-top_k:]
        bot_indices = np.argsort(sg)[:top_k]

        f_new = f.copy()
        for idx in top_indices:
            if rng.random() < 0.5:
                f_new[idx] *= float(np.exp(-rng.uniform(0.05, 0.3)))
        for idx in bot_indices:
            if rng.random() < 0.5:
                f_new[idx] *= float(np.exp(rng.uniform(0.05, 0.3)))

        # Small random perturbation
        width = max(1, int(rng.integers(max(1, n // 500), max(2, n // 50))))
        start = int(rng.integers(0, max(1, n - width)))
        f_new[start:start + width] *= np.exp(rng.normal(0.0, 0.05, size=width))

        f_new = np.clip(f_new, 0.0, None)
        if f_new.sum() > 1e-10:
            f_new = f_new / f_new.sum()
            score_new = compute_score(f_new)
            if score_new < best_score:
                f = f_new
                best_score = score_new
                best_f = f.copy()
                if it % 1000 == 0:
                    print(f"    GLS iter {it}: score={best_score:.6f}")

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Uses LP-based optimization on multiple starting
    points of various shapes and lengths, then refines with local search.
    """
    rng = np.random.default_rng(seed)
    total_time = float(budget_s)
    lp_fraction = 0.35
    lp_deadline = time.monotonic() + total_time * lp_fraction - 5
    search_deadline = time.monotonic() + total_time - 10

    print(f"Budget: {total_time}s, LP phase: {lp_fraction*100:.0f}%")

    best_seq = None
    best_score = np.inf

    # Collect starting points
    starting_points = []

    # Try the provided sequence
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0.0, None)
        if len(hs) > 0 and np.isfinite(hs).all() and hs.sum() > 0:
            hs_norm = hs / hs.sum()
            s0 = compute_score(hs_norm)
            print(f"Provided seq (n={len(hs)}): score={s0:.6f}")
            if s0 < best_score:
                best_score = s0
                best_seq = hs_norm.copy()
            starting_points.append(('provided', hs_norm))
    except Exception:
        pass

    # Diverse starting points
    configs = [
        (512, 'gauss', 0.25), (512, 'gauss', 0.4),
        (1024, 'gauss', 0.25), (1024, 'gauss', 0.35), (1024, 'gauss', 0.5),
        (2048, 'gauss', 0.3), (2048, 'gauss', 0.45),
        (1024, 'expon', 1.5), (1024, 'expon', 3.0),
        (2048, 'expon', 2.0), (2048, 'expon', 4.0),
        (1024, 'raised', 0.0), (2048, 'raised', 0.0),
    ]

    for n, shape, param in configs:
        if shape == 'gauss':
            x = np.arange(n, dtype=np.float64)
            center = (n - 1) / 2.0
            sigma = param * n
            c = np.exp(-(x - center) ** 2 / (2.0 * sigma ** 2))
        elif shape == 'expon':
            x = np.arange(n, dtype=np.float64)
            c = np.exp(-param * x / n)
        elif shape == 'raised':
            x = np.arange(n, dtype=np.float64) / max(1, n - 1)
            c = 0.5 * (1.0 + np.cos(np.pi * (2.0 * x - 1.0)))
        else:
            continue
        c = np.clip(c, 0.0, None)
        if c.sum() > 0:
            c = c / c.sum()
            name = f"{shape}_n{n}_p{param}"
            starting_points.append((name, c))

    # Run LP optimization on candidates (skip very large ones)
    for name, f in starting_points:
        if time.monotonic() > lp_deadline:
            print("LP phase time expired")
            break
        n = len(f)
        if n > 4000:
            score_val = compute_score(f)
            if score_val < best_score:
                best_score = score_val
                best_seq = f.copy()
            continue
        print(f"LP on {name} (n={n}), current best={best_score:.6f}")
        f_opt, score_val = lp_optimize(f, max_iters=25, deadline=lp_deadline)
        print(f"  {name} LP result: {score_val:.6f}")
        if score_val < best_score:
            best_score = score_val
            best_seq = f_opt.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # Local search phase
    remaining = search_deadline - time.monotonic()
    print(f"\nLocal search phase: {remaining:.0f}s available, best={best_score:.6f}")

    if best_seq is not None:
        # Split time: 60% on best, 40% on provided (if different)
        t_best = remaining * 0.6
        t_prov = remaining * 0.4

        # Local search on best sequence
        deadline_ls1 = time.monotonic() + t_best
        if len(best_seq) <= 5000:
            f_opt, score_val = local_search(best_seq, deadline_ls1, rng)
        else:
            f_opt, score_val = guided_local_search(best_seq, deadline_ls1, rng)
        if score_val < best_score:
            best_score = score_val
            best_seq = f_opt.copy()
        print(f"After LS on best (n={len(best_seq)}): {best_score:.6f}")

        # Local search on provided sequence if competitive and different length
        try:
            hs = np.array(height_sequence_1, dtype=np.float64)
            hs = np.clip(hs, 0.0, None)
            if (len(hs) > 0 and hs.sum() > 0 and len(hs) != len(best_seq)):
                hs_norm = hs / hs.sum()
                score_hs = compute_score(hs_norm)
                if score_hs < best_score + 0.3:
                    deadline_ls2 = time.monotonic() + t_prov
                    print(f"LS on provided (n={len(hs)}, init={score_hs:.6f}):")
                    f_opt2, score_val2 = local_search(hs_norm, deadline_ls2, rng)
                    if score_val2 < best_score:
                        best_score = score_val2
                        best_seq = f_opt2.copy()
                        print(f"  New best from provided: {best_score:.6f}")
        except Exception:
            pass

    # Final polish with smaller perturbations
    if best_seq is not None and time.monotonic() < search_deadline - 5:
        deadline_p = time.monotonic() + min(120.0, search_deadline - time.monotonic() - 5)
        print(f"Final polish ({deadline_p - time.monotonic():.0f}s):")
        f = best_seq.copy()
        score = best_score
        n = len(f)
        base_sigma = 0.03
        while time.monotonic() < deadline_p:
            f_new = f.copy()
            width = max(1, int(rng.integers(max(1, n // 1000), max(2, n // 200))))
            start = int(rng.integers(0, max(1, n - width)))
            f_new[start:start + width] *= np.exp(rng.normal(0.0, base_sigma, size=width))
            f_new = np.clip(f_new, 0.0, None)
            if f_new.sum() > 1e-10:
                f_new = f_new / f_new.sum()
                s_new = compute_score(f_new)
                if s_new < score:
                    f = f_new
                    score = s_new
                    if score < best_score:
                        best_score = score
                        best_seq = f.copy()

    # Scale result to [0, 1000]
    if best_seq is None:
        print("FALLBACK: returning uniform sequence")
        return [1.0] * 128

    max_val = best_seq.max()
    if max_val > 0:
        scale = 50.0 / max_val
    else:
        scale = 1.0
    result = best_seq * scale
    result = np.clip(result, 0.0, 1000.0)

    # Final verification
    final_score = compute_score(result)
    print(f"\n=== FINAL: n={len(result)}, score={final_score:.6f} ===")

    return [float(x) for x in result]
