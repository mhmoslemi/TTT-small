```python
from scipy.optimize import linprog
import time

def lp_descent_step(f, M, maxiter=100):
    """One step of the LP-based descent."""
    n = len(f)
    # A is (2n-1) x n, A[k,j] = f[k-j] if valid
    # We need A @ g <= M
    # Maximize sum(g)
    
    # Build A efficiently
    # A[k, j] = f[k-j] if 0 <= k-j < n and 0 <= j < n, else 0
    
    # Use scipy sparse
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            rows.append(k)
            cols.append(j)
            data.append(f[k-j])
    
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    
    # Solve LP: max 1^T g s.t. A g <= M, g >= 0
    c = -np.ones(n)  # minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```
