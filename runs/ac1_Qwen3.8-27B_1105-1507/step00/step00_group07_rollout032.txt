def build_conv_matrix(a):
    """Build the convolution constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    # A is (2n-1) x n
    # A[k,j] = a[k-j] if 0 <= k-j < n
    # For column j, non-zero entries are at rows j, j+1, ..., j+n-1
    # with values a[0], a[1], ..., a[n-1]
    rows = []
    cols = []
    data = []
    for j in range(n):
        for m in range(n):
            k = j + m
            rows.append(k)
            cols.append(j)
            data.append(a[m])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

The project_simplex function projects onto {x >= 0, sum(x) = 1}:

```python
def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""

    n = len(a)
    # Build constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0
    # A is (2n-1) x n, banded with bandwidth n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    b_ub = np.full(2*n-1, M)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0,None)]*n, method='highs')
    if not res.success:
        return None
    b = res.x
    s = np.sum(b)
    if s < 1e-12:
        return None
    return b / s

def optimize_lp(a_init, time_limit):
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    a = a / np.sum(a)
    best = a.copy()
    best_score = score_norm(a)
    patience = 0
    
    while time.monotonic() < time_limit:
        c = np.convolve(a, a)
        M = np.max(c)
        b = lp_maximize_sum(a, M)
        if b is None:
            break
        
        # Try step sizes
        best_t = None
        for t in [0.1, 0.2, 0.5, 0.3, 0.05, 0.7]:
            new_a = (1-t)*a + t*b
            new_a = np.maximum(new_a, 0)
            ns = np.sum(new_a)
            if ns < 1e-10:
                continue
            new_a /= ns
            sc = score_norm(new_a)
            if sc < best_score:
                best_t = t
                best_score = sc
                best = new_a
                break
            elif sc < score_norm(a):
                # Accept even if not new best (to keep moving)
                if best_t is None or sc < score_norm((1-best_t)*a + best_t*b):
                    best_t = t
                    best = ((1-best_t)*a + best_t*b)
                    best = np.maximum(best, 0)
                    best /= np.sum(best)
                    best_score = min(best_score, score_norm(best))
        
        if best_t is None:
            patience += 1
            if patience > 20:
                break
        else:
            a = best.copy()
            patience = 0
    
    return best, best_score

def optimize_subgrad(a_init, time_limit):
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    a = a / np.sum(a)
    best = a.copy()
    best_score = score_norm(a)
    
    lr0 = 2.0 / np.sqrt(n)
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(1000000):
        if time.monotonic() > time_limit:
            break
        c = np.convolve(a, a)
        k_star = int(np.argmax(c))
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        g[lo:hi+1] = 2.0 * np.flip(a[k_star-hi:k_star-lo+1])
        
        v = momentum * v + (1-momentum) * g
        lr = lr0 / (1 + it * 0.01)
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        sc = score_norm(a_new)
        if sc < best_score:
            best = a_new.copy()
            best_score = sc
        
        a = a_new
    
    return best, best_score

Let me think about what's realistic. The LP approach is essentially a "game theory" approach: a and b are players, and the LP finds the best response of b to a. The iteration converges to a "Nash equilibrium" of this game, which might or might not be the optimum.

Let me think about whether the LP approach is guaranteed to decrease the score. At each step:
- We have current a with peak M = max(a*a)
- We find b that maximizes sum(b) subject to (a*b)[k] <= M
- We set a_new = (1-t)*a + t*b

The score of a_new is 2n * max(a_new * a_new). We want this to be less than 2n * M.

Since (a*b)[k] <= M for all k, and a_new = (1-t)*a + t*b:
(a_new * a_new)[k] = (1-t)^2 (a*a)[k] + 2t(1-t)(a*b)[k] + t^2 (b*b)[k]
<= (1-t)^2 M + 2t(1-t) M + t^2 (b*b)[k]
= M((1-t)^2 + 2t(1-t)) + t^2 (b*b)[k]
= M(1-t) + t^2 (b*b)[k]

So the peak of a_new * a_new is at most M(1-t) + t^2 * max(b*b). If max(b*b) <= M/t (which is not guaranteed), then the peak decreases.

In general, the peak might increase if b has a very high autocorrelation peak. The step size t controls the trade-off: small t means we move slowly towards b, reducing the risk of increasing the peak.

So the LP approach is not guaranteed to monotonically decrease the score. That's why I use line search (trying different t values) and keep track of the best.

For simulated annealing: start from a random sequence, make random perturbations, and accept worse solutions with some probability to escape local minima.

These approaches are very general and don't exploit the structure of the problem, but they might work.
