def lp_step(a, M):
    """Solve LP to find best direction b that keeps convolution <= M."""
    n = len(a)
    # Build constraint matrix A: A[k, j] = a[k-j] if 0 <= k-j < n
    # This is a banded matrix (bandwidth n)
    # A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For row k: valid j are max(0, k-n+1) <= j <= min(n-1, k)
    
    # Using sparse matrix for efficiency
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    # Objective: maximize sum(b) = minimize -sum(b)
    c = -np.ones(n)

    bounds = [(0, None)] * n

    f = conv[m]
    grad = np.zeros(n)
    j_min = max(0, m - n + 1)
    j_max = min(n - 1, m)
    if j_max >= j_min:
        l = m - np.arange(j_min, j_max + 1)
        grad[j_min:j_max + 1] = 4.0 * n * (a[l] * S - f) / (S * S * S)
    return grad

def compute_J_smooth(a):
    """Compute smoothed objective using log-sum-exp."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e10
    conv = fast_convolution(a)
    max_c = np.max(conv)
    scaled = (conv - max_c) / _global_tau
    exp_vals = np.exp(scaled)
    sum_exp = np.sum(exp_vals)
    lse = max_c + _global_tau * np.log(sum_exp)
    return 2.0 * n * lse / (S * S)

def compute_grad_smooth(a):
    """Compute gradient of smoothed objective."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.zeros(n)
    conv = fast_convolution(a)
    max_c = np.max(conv)
    scaled = (conv - max_c) / _global_tau
    exp_vals = np.exp(scaled)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    lse = max_c + _global_tau * np.log(sum_exp)
    # Gradient: 2n/S^2 * (1/tau) * corr(w, a) - 2*J/S
    corr = np.correlate(w, a, mode='valid')
    grad = (2.0 * n * corr) / (_global_tau * S * S) - (2.0 * compute_J_smooth(a)) / S
    return grad

def fun_factory_smooth(x):
    """Wrapper for smooth objective (top-level to avoid closures)."""
    return compute_J_smooth(x)

def jac_factory_smooth(x):
    """Wrapper for smooth gradient."""
    return compute_grad_smooth(x)

def fun_factory_exact(x):
    """Wrapper for exact objective."""
    return compute_J_exact(x)

def jac_factory_exact(x):
    """Wrapper for exact gradient."""
    return compute_grad_exact(x)

def generate_initial(n, init_i, rng, parent):
    """Generate diverse initial sequences."""
    if init_i == 0:
        a = rng.random(n) * 0.08 + 0.02
    elif init_i == 1:
        p = np.asarray(parent, dtype=np.float64)
        if len(p) >= n:
            a = p[:n].copy()
        else:
            a = np.zeros(n)
            a[:len(p)] = p
            if len(p) > 0:
                a[len(p):] = p[-1]
            else:
                a[:] = 0.05
    elif init_i == 2:
        # Pseudo-random binary-like
        a = rng.integers(0, 2, n).astype(np.float64) * 0.04 + 0.02
    elif init_i == 3:
        # Low-frequency smooth with noise
        t = np.linspace(0, 4 * np.pi, n)
        a = (1.0 + 0.3 * np.sin(t) + 0.2 * np.cos(2.7 * t)) * 0.04 + 0.02
        a += rng.normal(0, 0.005, n)
    elif init_i == 4:
        # Random with structure (blocks)
        a = rng.random(n) * 0.06 + 0.02
        block_size = max(2, n // 32)
        for start in range(0, n, block_size):
            scale = np.exp(rng.normal(0, 0.3))
            a[start:start + block_size] *= scale
    elif init_i == 5:
        # Cosine basis combination
        coeffs = rng.normal(0, 1, min(20, n))
        t = np.linspace(0, 2 * np.pi, n)
        a = np.zeros(n)
        for i, c in enumerate(coeffs):
            a += c * np.cos((i + 1) * t)
        a = np.abs(a) * 0.03 + 0.02
    else:
        a = rng.random(n) * 0.08 + 0.02
    return np.clip(a, 0.0, 1000.0)

def run_optimization(a0, use_smooth, tau, time_budget):
    """Run L-BFGS-B optimization for a given time budget."""
    global _global_tau
    n = len(a0)
    
    if n > 8000:
        # Too large for efficient L-BFGS-B, truncate
        a0 = a0[:8000]
        n = len(a0)
    
    if use_smooth:
        _global_tau = tau
        fun = fun_factory_smooth
        jac = jac_factory_smooth
    else:
        fun = fun_factory_exact
        jac = jac_factory_exact
    
    bounds = [(0.0, 1000.0)] * n
    max_iter = max(100, int(time_budget / 0.02))
    max_fun = max_iter * 4
    
    start_time = time.monotonic()
    
    try:
        result = minimize(
            fun, a0.copy(), jac=jac,
            method='L-BFGS-B',
            bounds=bounds,
            options={
                'maxiter': max_iter,
                'maxfun': max_fun,
                'ftol': 1e-15,
                'gtol': 1e-8,
                'maxls': 40,
                'iprint': -1
            }
        )
        elapsed = time.monotonic() - start_time
        if time.monotonic() > _global_deadline:
            return a0
        return np.clip(result.x, 0.0, 1000.0)
    except Exception as e:
        elapsed = time.monotonic() - start_time
        return a0

def run_lp_refinement(a, max_iters=5):
    """LP-based refinement: find direction b that keeps a*b bounded."""
    n = len(a)
    if n > 2000:
        return a  # Too large for LP approach
    
    current_a = a.copy()
    for it in range(max_iters):
        if time.monotonic() > _global_deadline - 10:
            break
        M = np.max(fast_convolution(current_a))
        S = np.sum(current_a)
        
        # Build constraint matrix efficiently
        # A[k, j] = a[k-j] if 0 <= k-j < n
        k_idx = np.arange(2 * n - 1)[:, None]
        j_idx = np.arange(n)[None, :]
        diff = k_idx - j_idx
        valid = (diff >= 0) & (diff < n)
        A = np.where(valid, current_a[diff.clip(0, n - 1)], 0.0)
        
        # LP: max sum(b) s.t. A@b <= M, b >= 0
        c = -np.ones(n)
        try:
            from scipy.optimize import linprog
            res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                         bounds=[(0, None)] * n, method='highs')
            if res.success and res.x is not None:
                b = res.x
                b_sum = np.sum(b)
                if b_sum > 1e-10:
                    b = b * S / b_sum  # Normalize to match sum
                    # Update
                    t = 0.3
                    new_a = (1 - t) * current_a + t * b
                    new_a = np.clip(new_a, 0.0, 1000.0)
                    new_score = compute_J_exact(new_a)
                    if new_score < compute_J_exact(current_a):
                        current_a = new_a
        except Exception:
            pass
    return current_a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main optimization: two-phase (smooth + exact) L-BFGS-B with multiple
    initializations and sequence lengths. Includes LP refinement for small n.
    """
    global _global_deadline
    _global_deadline = time.monotonic() + budget_s - 8.0
    print(f"Total budget: {budget_s}s, deadline at t={_global_deadline - time.monotonic():.1f}s from now")
    
    best_seq = None
    best_score = np.inf
    
    rng = np.random.default_rng(seed)
    
    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(256) * 0.05
    
    parent = np.clip(parent, 0.0, 1000.0)
    
    # Configuration: (length, num_inits, smooth_frac)
    # smooth_frac: fraction of time for smooth phase
    configs = [
        (1024, 4, 0.5),
        (2048, 3, 0.4),
        (512, 3, 0.5),
        (4096, 2, 0.4),
        (768, 2, 0.5),
    ]
    
    total_configs = sum(c[1] for c in configs)
    per_config_time = (budget_s - 30.0) / total_configs  # Reserve 30s for final polish
    
    for n, num_inits, smooth_frac in configs:
        for init_i in range(num_inits):
            elapsed_total = time.monotonic() - (_global_deadline - budget_s + 8.0)
            remaining = _global_deadline - time.monotonic()
            if remaining < 15:
                break
            
            alloc = min(per_config_time, remaining - 10)
            if alloc < 10:
                break
            
            a0 = generate_initial(n, init_i, rng, parent)
            
            # Phase 1: Smooth optimization
            t1 = alloc * smooth_frac
            a1 = run_optimization(a0, True, 0.05, t1)
            
            if time.monotonic() > _global_deadline - 10:
                break
            
            # Phase 2: Exact optimization
            t2 = alloc * (1.0 - smooth_frac)
            a2 = run_optimization(a1, False, 0.0, t2)
            
            # Optional LP refinement (only for smaller n)
            if n <= 1024 and time.monotonic() < _global_deadline - 15:
                a2 = run_lp_refinement(a2, max_iters=3)
            
            score = evaluate_sequence(a2.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a2.copy()
                print(f"  n={n}, init={init_i}: J={score:.8f} (elapsed={elapsed_total:.0f}s)")
    
    # Final polishing phase with remaining time
    if best_seq is not None:
        remaining = _global_deadline - time.monotonic()
        if remaining > 20:
            print(f"Final polishing with {remaining:.0f}s remaining...")
            a_final = run_optimization(best_seq.copy(), False, 0.0, remaining - 10)
            score = evaluate_sequence(a_final.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_final
                print(f"  After polish: J={score:.8f}")
        
        # One more smooth pass if time allows
        remaining = _global_deadline - time.monotonic()
        if remaining > 30:
            a_polished = run_optimization(best_seq.copy(), True, 0.02, remaining - 15)
            a_polished = run_optimization(a_polished, False, 0.0, 10)
            score = evaluate_sequence(a_polished.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_polished
                print(f"  After 2nd polish: J={score:.8f}")
    
    if best_seq is None:
        # Fallback
        best_seq = np.ones(512) * 0.05
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"Fallback: J={best_score:.6f}")
    
    final = np.clip(best_seq, 0.0, 1000.0)
    print(f"Best score: {best_score:.8f}, length={len(final)}")
    return [float(x) for x in final]
