Let me formalize:
- Current sequence: `a` with `sum(a) = S`, `peak = max(convolve(a,a)) = P`
- Score = `2n * P / S^2`
- LP: maximize `sum(b)` subject to `b >= 0` and `convolve(a, b) <= P` (element-wise, so all `2n-1` constraints)
- Optimal solution: `b*`
- Rescale: `b_rescaled = b* * S / sum(b*)`
- Update: `a_new = (1-t)*a + t*b_rescaled`

            best_peak = peak

        # FFT

    """
    LP step: maximize sum(b) subject to convolve(a,b) <= P, b >= 0
    Returns the LP solution rescaled to have the same sum as a.
    """
    n = len(a)
    # Build constraint matrix: A[k, j] = a[k-j] if 0 <= k-j < n
    # Shape: (2n-1) x n
    # For efficiency, only do this for n <= 2000
    if n > 2000:
        return None
    
    # Build A (convolution matrix)
    # A is (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = a[k-hi:k-lo+1][::-1]
    
    # LP: max 1^T b s.t. A @ b <= P*ones, b >= 0
    # linprog minimizes, so: min -1^T b s.t. A @ b <= P*ones, b >= 0
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(2*n-1, P)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        b = res.x
        b_sum = b.sum()
        if b_sum > 0:
            # Rescale to match sum of a
            a_sum = a.sum()
            b = b * (a_sum / b_sum)
            return b
    return None

def optimize_sequence(a, deadline, rng, use_lp=False):
    """Projected gradient descent with optional LP steps"""
    n = len(a)
    a = project_simplex(a.copy())
    
    lr = 1.0 / (2.0 * np.max(np.abs(fast_convolve(a))))  # Initial learning rate
    # Actually, let's use a simpler initialization
    lr = 0.1 / n
    best_a = a.copy()
    best_peak = np.max(fast_convolve(a))
    
    stall_count = 0
    max_stall = 500
    
    for it in range(100000):
        if time.time() > deadline:
            break
        
        c = fast_convolve(a)
        peak = np.max(c)
        
        if peak < best_peak * (1 - 1e-8):
            best_peak = peak
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > max_stall:
            # Reduce learning rate or break
            lr *= 0.5
            stall_count = 0
            if lr < 1e-10:
                break
        
        # Get peak indices
        peak_indices = np.where(c >= peak * (1 - 1e-6))[0]
        # Limit to top few peaks for efficiency
        if len(peak_indices) > 50:
            top_idx = np.argsort(c[peak_indices])[::-1][:20]
            peak_indices = peak_indices[top_idx]
        
        # Compute subgradient
        grad = compute_subgradient(a, peak_indices)
        
        # Gradient step
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        a = a_new
        
        # Occasional LP step
        if use_lp and n <= 1500 and it % 50 == 0 and time.time() < deadline - 30:
            b = lp_step(a, best_peak, max_time=5)
            if b is not None:
                # Convex combination
                t = 0.5
                a_new = (1-t) * a + t * b
                a = project_simplex(a_new)
    
    return best_a
