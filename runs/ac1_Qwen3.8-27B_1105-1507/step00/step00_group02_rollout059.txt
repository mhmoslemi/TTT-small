import numpy as np

def fast_self_convolve(a):
    """Compute linear self-convolution using FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = int(2 ** np.ceil(np.log2(2 * n - 1)))
    A = fft(a, size)
    return ifft(A * A).real[:2 * n - 1]

    """Compute objective assuming sum(a) = sqrt(2n). Returns max_b."""

def full_evaluate(a):
    """Compute the actual objective 2n*max_b/(sum_a)^2."""
    n = len(a)
    S = float(np.sum(a))

        return np.inf

    max_b = float(np.max(b))
    return float(2.0 * n * max_b / (S * S))


def compute_gradient(a, b, m=5, use_soft=False, soft_alpha=1.0):
    """
    Compute gradient of weighted peak objective.
    
    grad[i] = 2 * sum_k w[k] * a[k-i]  (for valid k-i in [0,n-1])
    = 2 * cross_correlation(w, a)[i]
    """
    n = len(a)
    b_len = len(b)
    w = np.zeros(b_len)

    if m > 0:
        top_idx = np.argpartition(b, -m)[-m:]
        w[top_idx] = 1.0 / m

    if use_soft:
        mean_b = float(np.mean(b))
        if max_b := float(np.max(b)) > mean_b:
            excess = np.maximum(b - mean_b, 0)
            sw = float(np.sum(excess))
            if sw > 1e-15:
                w_soft = excess / sw
                w = soft_alpha * w_soft + (1.0 - soft_alpha) * w

    s = float(np.sum(w))
    if s < 1e-15:
        w = np.zeros(b_len)
        w[np.argmax(b)] = 1.0

    pad_size = int(2 ** np.ceil(np.log2(b_len + n - 1)))
    W = fft(w, pad_size)
    A = fft(a, pad_size)
    c = ifft(W * np.conj(A)).real[:n]

    return 2.0 * c

            if alloc < 2:

            a2, sc2 = flatten_sequence(a, alloc, seed=seed + 100 + attempt)
            full_sc = full_evaluate(a2)
            if full_sc < best_score:
                best_score = full_sc
                best_a = a2.copy()
                print(f"  Restart best: score={best_score:.6f}")

    # Final verification
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    final_score = evaluate_sequence(result)
    print(f"Final verified score: {final_score:.6f}, n={len(result)}")

    return result
