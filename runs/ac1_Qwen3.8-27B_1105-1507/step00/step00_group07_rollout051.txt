        k_star = np.argmax(c)
        max_val = c[k_star]
        # Gradient of c[k_star] w.r.t. a
        # c[k] = sum_i a[i] * a[k-i]
        # d(c[k])/d(a[j]) = 2*a[k-j] if 0<=k-j<n else 0

"""
Step function optimization to minimize 2n*max(convolve(a,a))/(sum(a))^2.
Uses subgradient descent with top-K lag weighting, combined with
parameterized initializations and block perturbations for diversity.
"""
import numpy as np
import time


def normalize_nonneg(a):
    """Clamp to non-negative and normalize to unit sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def compute_score(a):
    """Compute objective: 2n * max(conv) / (sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def subgrad_step(a, K=30, lr=0.08, eta_reg=0.0):
    """
    One subgradient step to flatten the autocorrelation.
    Reduces top-K lags and optionally adds regularization.
    Returns (new_a, current_max_conv).
    """
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    
    # Get top K lags
    if K >= c.size:
        top_idx = np.arange(c.size)
        top_val = c.copy()
    else:
        top_idx = np.argpartition(c, -K)[-K:]
        top_val = c[top_idx]
    
    # Weights: proportional to value (emphasize peaks)
    w = top_val ** 2
    w_sum = w.sum()
    if w_sum < 1e-15:
        return a, max_c
    w = w / w_sum
    
    # Compute gradient (sum of weighted gradients from top lags)
    grad = np.zeros(n, dtype=np.float64)
    for wi, ki in zip(w, top_idx):
        ki = int(ki)
        # d(c[ki])/d(a[j]) = 2*a[ki-j] for j where 0 <= ki-j < n
        j_lo = max(0, ki - n + 1)
        j_hi = min(n - 1, ki)
        if j_lo <= j_hi:
            # a[ki - j] for j = j_lo, ..., j_hi
            # ki - j ranges from ki - j_lo down to ki - j_hi
            idx = ki - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi+1] += wi * 2.0 * a[idx]
    
    # Step size
    gnorm_sq = (grad * grad).sum()
    if gnorm_sq < 1e-15:
        return a, max_c
    step = lr * max_c / gnorm_sq
    a_new = a - step * grad
    
    # Regularization: pull toward uniform
    if eta_reg > 0:
        a_new = (1.0 - eta_reg) * a_new + eta_reg / n
    
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_nonneg(a_new)
    return a_new, max_c


def subgrad_step_flatten(a, K_top=20, K_bot=20, lr=0.05):
    """
    Subgradient step that both reduces peaks and raises troughs.
    """
    n = len(a)
    c = np.convolve(a, a)
    max_c = c.max()
    min_c = c.min()
    range_c = max_c - min_c
    if range_c < 1e-15:
        return a, max_c
    
    # Top K lags (to reduce)
    if K_top >= c.size:
        top_idx = np.arange(c.size)
        top_val = c.copy()
    else:
        top_idx = np.argpartition(c, -K_top)[-K_top:]
        top_val = c[top_idx]
    
    # Bottom K lags (to raise)
    if K_bot >= c.size:
        bot_idx = np.arange(c.size)
        bot_val = c.copy()
    else:
        bot_idx = np.argpartition(c, K_bot)[:-K_bot] if K_bot < c.size else np.arange(c.size)
        # Actually, argpartition(c, K_bot) gives the smallest K_bot+1 elements in the first K_bot+1 positions
        # Let's use a different approach

    w_top = w_top / w_top.sum()
    
    # Bottom weights (inverted: more weight on the lowest)
    w_bot = (max_c - bot_val + 1e-10) ** 2

def make_parabolic(n, alpha):
    """U-shaped parabolic sequence."""
    i = np.arange(n, dtype=np.float64)
    if n > 1:
        t = 2.0 * i / (n - 1) - 1.0
    else:
        t = np.zeros(1)
    a = 1.0 + alpha * t ** 2
    return normalize_nonneg(a)


def make_cosine_u(n, alpha):
    """U-shaped cosine sequence: high at edges, low at center."""
    i = np.arange(n, dtype=np.float64)
    if n > 1:
        t = np.pi * i / (n - 1)
    else:
        t = np.zeros(1)
    a = 1.0 + alpha * np.cos(t)
    # cos goes from 1 to -1, so 1 + alpha*cos goes from 1+alpha to 1-alpha
    # For U-shape, we want high at edges: cos(0)=1, cos(pi)=-1
    # So 1 + alpha*cos(t) is high at edges for alpha > 0
    return normalize_nonneg(a)


def make_power_u(n, alpha, p):
    """U-shaped power sequence."""
    i = np.arange(n, dtype=np.float64)
    if n > 1:
        t = 2.0 * i / (n - 1) - 1.0
    else:
        t = np.zeros(1)
    a = (1.0 + alpha * t ** 2) ** p
    return normalize_nonneg(a)


def block_perturb(a, rng, min_width=None, max_width=None, sigma=0.1):
    """Apply a random multiplicative block perturbation."""
    n = len(a)
    if min_width is None:
        min_width = max(1, n // 5000)
    if max_width is None:
        max_width = max(min_width + 1, n // 50)
    width = int(rng.integers(min_width, max_width + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    end = min(n, start + width)
    
    a_new = a.copy()
    a_new[start:end] *= np.exp(rng.normal(0.0, sigma, end - start))
    a_new = np.maximum(a_new, 0.0)
    return normalize_nonneg(a_new)


def gs_iterations(a, n_iter=50):
    """Gerchberg-Saxton iterations for flat spectrum."""
    n = len(a)
    N = 2 * n  # FFT size
    for _ in range(n_iter):
        A = np.fft.rfft(a, N)
        # Target: flat magnitude
        target_mag = np.abs(A).mean()
        target_mag = np.full(A.shape, target_mag)
        # Keep phase
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        # Inverse
        a_new = np.fft.irfft(A_new, N)[:n]
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_nonneg(a_new)
        a = a_new
    return a
