Minimize J = 2*n*max(autocorrelation)/sum^2 using mirror descent on a
smooth-max (log-sum-exp) surrogate with annealed temperature, multiple
diverse initializations, and local refinement.

Key insight: the autocorrelation of the uniform sequence is triangular
with peak-to-average ratio ~2.  We use gradient-based optimization of a
smoothed maximum of the full autocorrelation to flatten it, reducing J
below 2 toward the theoretical bound 2n/(2n-1) ~ 1.
"""

import numpy as np
import time


def _compute_grad(a, w):
    """Compute grad_i = 2 * sum_k w_k * a_{k-i} via FFT."""
    n = len(a)
    L = 1
    while L < 3 * n:
        L <<= 1
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a[::-1], L)
    result = np.fft.irfft(W * A, L)
    return 2.0 * result[n - 1:2 * n - 1]


def _smooth_max_grad(a, T):
    """Return (smooth_value, gradient, actual_max) of autocorrelation."""
    c = np.convolve(a, a)
    c_max = float(c.max())
    z = (c - c_max) / T
    z = np.maximum(z, -50.0)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w = e / s
    grad = _compute_grad(a, w)
    smooth = c_max + T * np.log(s)
    return smooth, grad, c_max


def _normalize(a, S=None):
    """Clip to non-negative and rescale to sum S."""
    a = np.maximum(a, 0.0)
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = float(a.sum())
    if s < 1e-14:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def _mirror_optimize(a0, budget, lr=0.5, T0=5.0, T1=0.02, rng=None):
    """Mirror descent with annealed temperature on smooth max autocorrelation."""
    a = _normalize(np.asarray(a0, dtype=np.float64).copy())
    n = len(a)
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_mx = float(np.max(np.convolve(best_a, best_a)))
    t_start = time.monotonic()

    while time.monotonic() - t_start < budget:
        el = time.monotonic() - t_start
        frac = min(el / max(budget, 1e-6), 1.0)
        T = max(T1, T0 * (T1 / T0) ** frac)

        _, grad, mx = _smooth_max_grad(a, T)
        if mx < best_mx:
            best_mx = mx
            best_a = a.copy()

        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            break
        g = grad / gn

        # Adaptive learning rate: shrink if stuck
        a_new = a * np.exp(-lr * g)
        a_new = _normalize(a_new, S)
        mx_new = float(np.max(np.convolve(a_new, a_new)))

        if mx_new < mx:
            a = a_new
            lr = min(lr * 1.02, 2.0)
        else:
            lr = max(lr * 0.85, 0.05)
            if lr <= 0.051 and rng is not None:
                # Random restart nudge
                a = a * np.exp(rng.normal(0, 0.3, n))
                a = _normalize(a, S)
                lr = 0.5

    J = 2.0 * n * best_mx / (S * S)
    return best_a, J
