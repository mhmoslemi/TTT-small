"""
Optimizer for non-negative step-function heights minimizing
    2 * n * max(convolve(a, a)) / (sum(a))**2
via (i) cheap multiplicative "blow-down" steps on the current maximum
convolution lag (scale-invariant, so we normalize to sum=1) and (ii)
occasional sparse-LP kicks that find a direction keeping the cross
convolution (a*g) bounded while pushing mass into under-used lags, used
to escape plateaus. Crucially NO palindromic symmetry is enforced, since
palindromic sequences are provably bounded below by score=2. Multiple
random restarts explore different basins. The authoritative evaluator is
used to track the global best.
"""

import numpy as np
import time
import scipy.sparse as sp
import scipy.optimize as opt


def _normalize(a):
    """Rescale a (non-negative) to sum 1, guarding tiny sums."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / a.size
    return a / s


def _score_quick(a):
    """Fast internal score identical in form to the evaluator (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _blowdown(a, steps, eta0=0.35, eta1=0.05, floor=0.005):
    """
    Multiplicative blow-down: repeatedly reduce the currently maximal
    convolution lag by decaying the coordinates that feed it, then
    re-normalize. eta anneals eta0->eta1 over the steps; a tiny uniform
    floor prevents degeneration to a spike.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    a = _normalize(a)
    n = a.size
    j = np.arange(n)
    for it in range(steps):
        frac = it / max(1, steps - 1)
        eta = eta0 + (eta1 - eta0) * frac
        c = np.convolve(a, a)
        k = int(np.argmax(c))
        kj = k - j
        mask = (kj >= 0) & (kj < n)
        grad = np.zeros(n)
        grad[mask] = a[kj[mask]] + a[j[mask]]
        gm = grad.max()
        if gm > 1e-30:
            a = a * np.exp(-eta * grad / gm)
        if floor > 0.0:
            a = a * (1.0 - floor) + floor * (1.0 / n)
        a = _normalize(a)
    return a


def _lp_kick(a, top_k, timeout):
    """
    Sparse-LP kick. With a fixed current a (sum=1) and M = max(a*a),
    solve  max sum(g)  s.t.  (a*g)_k <= (a*a)_k  for the top-k lags k,
    g in [0, cap].  The returned g is a direction that does not (under the
    relaxed constraints) blow the cross-convolution, biased toward lags
    where 'a' leaves room, hence often a flattening direction.
    Returns the blended, re-normalized candidate, or None.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    c = np.convolve(a, a)
    if top_k is None or top_k >= (2 * n - 1):
        ks = np.arange(2 * n - 1)
    else:
        ks = np.argsort(c)[::-1][:top_k]
    ks = np.asarray(ks, dtype=np.int64)

    row_lists = []
    col_lists = []
    val_lists = []
    for k in ks:
        lo = max(0, int(k) - (n - 1))
        hi = min(n - 1, int(k))
        cols = np.arange(lo, hi + 1)
        vals = a[k - cols]
        r = np.full(cols.size, len(row_lists))
        row_lists.append(r)
        col_lists.append(cols)
        val_lists.append(vals)

    rows = np.concatenate(row_lists)
    cols = np.concatenate(col_lists)
    vals = np.concatenate(val_lists)
    A = sp.csr_matrix((vals, (rows, cols)), shape=(ks.size, n))
    b = c[ks]

    cap = max(1.0, 2.0 * a.sum())
    try:
        res = opt.linprog(
            -np.ones(n),
            A_ub=A,
            b_ub=b,
            bounds=[(0.0, cap)] * n,
            method="highs",
            options={"time_limit": max(1.0, timeout)},
        )
        if not res.success:
            return None
        g = np.asarray(res.x, dtype=np.float64)
        if g.sum() < 1e-12:
            return None
        g = _normalize(g)
        return g
    except Exception:
        return None


def _random_start(n, rng):
    """Diverse, strictly non-uniform non-negative start (front/back bias)."""
    base = np.exp(rng.normal(0.0, 1.0, size=n))
    pat = rng.random()
    if pat < 0.34:
        w = np.linspace(1.0, 0.1, n)          # front heavy
    elif pat < 0.67:
        w = np.linspace(0.1, 1.0, n)          # back heavy
    else:
        w = np.ones(n)
    a = base * w
    return _normalize(a)
