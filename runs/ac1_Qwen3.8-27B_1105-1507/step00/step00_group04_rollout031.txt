import numpy as np
import time

def fast_convolve(a):
    """FFT-based full linear convolution."""
    n = len(a)
    size = 1 << (2*n - 2).bit_length()  # next power of 2 >= 2n-1
    fa = np.fft.rfft(a, n=size)
    result = np.fft.irfft(fa * fa, n=size)
    return result[:2*n-1]

def gradient_at_peak(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    # grad[j] = 2*a[k-j] for max(0,k-n+1) <= j <= min(n-1,k)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js <= je:
        # a[k-js], a[k-js-1], ..., a[k-je] (reversed)
        grad[js:je+1] = 2.0 * a[k-je:k-js+1][::-1]
    return grad

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s > 1e-15:
        return v / s
    return np.ones(len(v)) / len(v)

def subgradient_step(a, lr, num_peaks=5):
    """One step of projected subgradient descent."""
    conv = fast_convolve(a)
    grad = np.zeros_like(a)
    
    # Find top peaks
    top_k = np.argsort(conv)[-num_peaks:]
    for k in top_k:
        grad += gradient_at_peak(a, int(k))
    grad /= num_peaks
    
    # Step
    a_new = a - lr * grad
    a_new = project_simplex(a_new)
    return a_new, conv

def run_sg(a, max_iters, lr_init, num_peaks=5):
    """Run projected subgradient descent."""
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    lr = lr_init
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    
    for it in range(max_iters):
        a_new, conv = subgradient_step(a, lr, num_peaks)
        current_max = conv.max()
        
        if current_max < best_max:
            best_max = current_max
            best_a = a_new.copy()
            a = a_new
            # Maybe increase lr slightly
            lr = min(lr * 1.01, lr_init * 4)
        else:
            a = a_new  # still move (for exploration)
            lr = lr * 0.9  # decrease lr
            # Re-project
            a = project_simplex(a)
    
    return best_a

Minimizes J = 2n * max(conv(a,a)) / (sum(a))^2.

import numpy as np
import time


def fast_convolve(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, n=size)
    return np.fft.irfft(fa * fa, n=size)[:2 * n - 1]


def conv_grad_at(a, k):
    """Gradient of (a*a)[k] w.r.t. a. Returns array of same length as a."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js <= je:
        grad[js:je + 1] = 2.0 * a[k - je:k - js + 1][::-1]
    return grad


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s > 1e-15:
        return v / s
    return np.ones(len(v)) / len(v)


def smooth_max_grad(a, tau, num_peaks):
