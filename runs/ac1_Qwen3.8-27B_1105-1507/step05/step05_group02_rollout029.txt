"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral (frequency domain) initialization
2. Multiplicative Weights Update (MWU) / exponential descent
3. LP mirror descent with adaptive steps

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M(a):
    """Compute max autocorrelation of non-negative sequence a."""
    a = np.maximum(a, 0)
    b = fast_conv(a)
    return float(b.max())


def compute_J(a):
    """Compute the objective J(a) = 2n * max(conv(a,a)) / (sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    M = compute_M(a)
    return float(2 * len(a) * M / (s * s))


def normalize(a):
    """Normalize so sum = 1, clip to non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smooth_max_and_grad(a, tau):
    """
    Compute smooth approximation to max of autocorrelation and its gradient.
    Returns (smooth_val, gradient, autocorr, max_val).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Log-sum-exp for numerical stability
    shifted = tau * (b - b_max)
    log_sum = b_max + np.log(np.exp(shifted).sum()) / tau
    # Actually: LSE = b_max + (1/tau) * log(sum(exp(tau*(b - b_max))))
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    # Gradient of (1/tau) * log(sum(exp(tau * b_k))) w.r.t. a_i
    # = sum_k p_k * (db_k/da_i)
    # db_k/da_i = 2 * a_{k-i} (for valid indices)
    grad = np.zeros(n)
    for k in range(len(b)):
        if p[k] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # b_k = sum_{j=lo}^{hi} a_j * a_{k-j}
        # db_k/da_i = 2 * a_{k-i} for i in [lo, hi]
        grad[lo:hi+1] += 2.0 * p[k] * a[k-hi:k-lo+1][::-1]
    return log_sum, grad, b, b_max


def run_mwu(a, deadline, max_iters=3000, tau_start=5.0, tau_end=500.0,
            eta_start=0.5, eta_end=0.01):
    """
    Multiplicative Weights Update on the probability simplex.
    a_i^{(t+1)} ∝ a_i^{(t)} * exp(-eta * grad_i / grad_norm_scale)
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        eta = eta_start * (eta_end / eta_start) ** frac

        _, grad, b, b_max = smooth_max_and_grad(a, tau)
        if b_max < best_M * (1 - 1e-9):
            best_M = b_max
            best_a = a.copy()

        # Scale gradient for numerical stability
        g_scale = np.abs(grad).max()
        if g_scale < 1e-15:
            break
        grad_s = grad / g_scale

        # Multiplicative update
        update = np.exp(-eta * grad_s)
        a = a * update
        a = np.maximum(a, 1e-10)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
        else:
            a = a / s

    return best_a

            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac

        _, grad, b, b_max = smooth_max_and_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix for LP: row k, col j has value a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=20):
    """Solve LP: max sum(b) s.t. a*b <= M, b >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror_v2(a, deadline, max_iters=300, lp_time=20):
    """
    LP mirror descent with adaptive step sizes.
    Tries a wider range of steps including very small ones for fine tuning.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    # Adaptive step sizes: start large, get finer over time
    base_steps = [0.8, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        # Adapt which steps to try based on iteration
        if it < 50:
            steps = base_steps[:6]
        elif it < 150:
            steps = base_steps[3:]
        else:
            steps = base_steps[6:]

        g = lp_best_response(a, M, time_limit=lp_time)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        improved = False
        for t in steps:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break

    return best_a


def block_lp_optimize(a, deadline, n_blocks=8, iterations_per_block=5, lp_time=10):
    """
    Block coordinate LP optimization.
    Split a into blocks, optimize each block with others fixed.
    Uses linearized constraint (dropping quadratic self-convolution term).
    """
    a = np.maximum(a.astype(np.float64), 1e-8).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()

    block_size = n // n_blocks
    if block_size < 4:
        return best_a

    for outer in range(iterations_per_block):
        if time.monotonic() > deadline:
            break
        for b_idx in range(n_blocks):
            if time.monotonic() > deadline:
                break
            lo = b_idx * block_size
            hi = min(n, (b_idx + 1) * block_size)
            m = hi - lo

            # Compute the "background" convolution (with block zeroed out)
            a_bg = a.copy()
            a_bg[lo:hi] = 0
            bg_conv = fast_conv(a_bg)

            # The constraint (a*a)_k <= M becomes approximately:
            # (a_block * a_full)_k + (a_full \ a_block) * (a_block)_k <= M - (a_bg * a_bg)_k
            # Since a = a_bg + a_block:
            # (a*a)_k = (a_bg*a_bg)_k + (a_bg*a_block)_k + (a_block*a_bg)_k + (a_block*a_block)_k
            # We drop (a_block*a_block)_k for linearization (it's small if block is small)
            # So: (a_bg*a_block)_k + (a_block*a_bg)_k <= M - (a_bg*a_bg)_k
            # Note: (a_bg*a_block)_k = (a_block*a_bg)_{n-1-k+...} -- not exactly symmetric
            # Actually for linear convolution: (a*b)_k = (b*a)_k, so this IS symmetric.
            # (a_bg * a_block)_k = sum_j a_bg[j] * a_block[k-j]
            # This is linear in a_block.

            # Build the constraint matrix for a_block variables
            # (a_bg * a_block)_k = sum_{j in block} a_block[j] * a_bg[k-j]
            # Plus (a_block * a_bg)_k = sum_{j in block} a_block[j] * a_bg[k-j] (same by symmetry of conv)
            # Wait, (a_b * a_bg)_k = sum_j a_b[j] * a_bg[k-j]
            # And (a_bg * a_b)_k = sum_j a_bg[j] * a_b[k-j]
            # These are different! (a_b * a_bg)_k != (a_bg * a_b)_k in general
            # Actually they ARE the same: (a*b)_k = sum_j a[j]b[k-j] = sum_j b[j]a[k-j] = (b*a)_k
            # So (a_bg * a_block)_k = (a_block * a_bg)_k

            # Total constraint: 2 * (a_bg * a_block)_k <= M - (a_bg * a_bg)_k
            # Let rhs_k = M - (a_bg * a_bg)_k
            # And the constraint is: sum_j a_block[j] * 2 * a_bg[k-j] <= rhs_k

            rhs = best_M - bg_conv
            # Only keep constraints where rhs > 0
            valid = rhs > 0

            if valid.sum() < 1:
                continue

            # Build sparse matrix: for valid constraint k, coefficient of a_block[j] is 2*a_bg[k-(lo+j)]
            # But k-(lo+j) must be in [0, n-1] for a_bg to be defined
            rows_idx = []
            cols_idx = []
            vals_idx = []

            for k in range(2 * n - 1):
                if not valid[k]:
                    continue
                for j in range(m):
                    global_j = lo + j
                    bg_idx = k - global_j
                    if 0 <= bg_idx < n:
                        rows_idx.append(len(rows_idx))
                        cols_idx.append(j)
                        vals_idx.append(2.0 * a_bg[bg_idx])

            if len(rows_idx) == 0:
                continue

            A_block = csr_matrix((vals_idx, (rows_idx, cols_idx)),
                                  shape=(len(rows_idx), m))
            c_block = -np.ones(m)
            b_ub_block = rhs[valid]
            bounds_block = [(0, None)] * m

            try:
                r = linprog(c_block, A_ub=A_block, b_ub=b_ub_block,
                           bounds=bounds_block, method='highs',
                           options={'time_limit': lp_time, 'presolve': True})
                if r.success and r.x is not None:
                    a_new_block = np.maximum(r.x, 0)
                    # The LP maximizes sum, so we want to scale back to maintain total sum
                    old_sum = a[lo:hi].sum()
                    new_sum = a_new_block.sum()
                    if new_sum > 1e-10:
                        # Scale to maintain total sum
                        scale = old_sum / new_sum if old_sum > 1e-10 else 1.0
                        a[lo:hi] = a_new_block * scale
                        M_new = fast_conv(a).max()
                        if M_new < best_M - 1e-12:
                            best_M = M_new
                            best_a = a.copy()
            except Exception:
                pass

            # Re-normalize
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a = a / s

    return best_a


def spectral_init(n, rng, mode='random_phase'):
    """
    Generate a sequence via frequency domain synthesis.
    Start with a flat power spectrum, apply random phase, inverse FFT, project to non-negative.
    """
    if mode == 'random_phase':
        # Random phase, constant amplitude
        N = 2 * n  # zero-pad to avoid circular conv artifacts
        amp = np.ones(N // 2 + 1)
        phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        spectrum = amp * np.exp(1j * phase)
        # Make it Hermitian-symmetric for real output
        full_spectrum = np.zeros(N, dtype=complex)
        full_spectrum[:N//2+1] = spectrum
        for k in range(1, N//2):
            full_spectrum[N-k] = np.conj(full_spectrum[k])
        a = np.fft.ifft(full_spectrum).real[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n
        return a

    elif mode == 'tapered':
        # Tapered amplitude spectrum (lower frequencies have more energy)
        N = 2 * n
        freqs = np.arange(N // 2 + 1)
        amp = 1.0 / (1.0 + 0.1 * freqs)
        phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        spectrum = amp * np.exp(1j * phase)
        full_spectrum = np.zeros(N, dtype=complex)
        full_spectrum[:N//2+1] = spectrum
        for k in range(1, N//2):
            full_spectrum[N-k] = np.conj(full_spectrum[k])
        a = np.fft.ifft(full_spectrum).real[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n
        return a

    else:
        return np.ones(n) / n


def cosine_init(n, p, rng, center=0.5, width=1.0):
    """Generate cosine-power initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = np.abs(np.cos(np.pi * (t - center) / width)) ** p
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a


def gaussian_init(n, sigma, rng, center=0.5):
    """Generate Gaussian-shaped initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a = a / a.sum()
    return a


def resample_seq(src, n):
    """Interpolate a sequence to a new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def freq_perturb(a, rng, strength=0.1):
    """Perturb in frequency domain."""
    n = len(a)
    N = 2 * n
    fa = np.fft.rfft(a, N)
    # Perturb the phase
    phase_pert = rng.normal(0, strength, len(fa))
    new_fa = fa * np.exp(1j * phase_pert)
    new_a = np.fft.irfft(new_fa, N)[:n]
    return np.maximum(new_a, 0)


def run_subgradient(a, deadline, max_iters=2000):
    """Subgradient descent with Polyak averaging."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

        tol = 1e-5 * M
        active = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad /= gn

        lr = 0.3 / np.sqrt(it + 1)
        a_new = np.maximum(a - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        running += a_new
        count += 1
        a = a_new

    if count > 100:
        avg = running / count
        avg = np.maximum(avg, 0.0)
        s = avg.sum()
        if s > 1e-10:
            avg /= s
            M_avg = fast_conv(avg).max()
            if M_avg < best_M:
                best_a = avg
    return best_a


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0
    best = BestTracker()

    # Load parent
    parent = None
    try:

        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    # === PHASE 1: Diverse Initialization ===
    p1_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Diverse Initialization", flush=True)
    print(f"{'='*50}", flush=True)

    base_n = parent.size if parent.size >= 128 else 256
    # Focus on a range of lengths
    lengths = sorted(set([
        max(128, base_n - 64), max(128, base_n - 32),
        base_n,
        base_n + 32, base_n + 64, base_n + 128,
        max(128, base_n // 2), base_n * 2 if base_n * 2 <= 2048 else base_n,
        # Smooth numbers (good for FFT and arithmetic)
        192, 256, 320, 384, 448, 512, 576, 640, 768, 896, 1024,
        1152, 1280, 1536, 2048,
    ] & {x for x in range(128, 2049) if x % 16 == 0}))
    lengths = [x for x in lengths if 128 <= x <= 2048]

    print(f"Lengths to try: {lengths}", flush=True)

    # Generate initial candidates for each length
    candidates = {}  # length -> list of initial sequences
    for n in lengths:
        cands = []
        # From parent (resampled)
        if parent is not None:
            cands.append(resample_seq(parent, n))
            cands.append(np.maximum(resample_seq(parent, n) + rng.normal(0, 0.02, n), 0.01))
        # Spectral
        cands.append(spectral_init(n, rng, 'random_phase'))
        cands.append(spectral_init(n, rng, 'tapered'))
        # Cosine
        for p_val in [1.0, 1.5, 2.0, 3.0]:
            cands.append(cosine_init(n, p_val, rng))
        # Gaussian
        cands.append(gaussian_init(n, 0.2, rng))
        cands.append(gaussian_init(n, 0.35, rng))
        # Uniform
        cands.append(np.ones(n) / n)
        # Random smoothed
        k = max(3, n // 32)
        kernel = np.ones(k) / k
        cands.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kernel, 'same') + 0.5, 0.01))
        candidates[n] = cands

    # Quick evaluation of all candidates
    for n, cands in candidates.items():
        for c in cands:
            c = normalize(c)
            M = compute_M(c)
            J = compute_J(c)
            if np.isfinite(J):
                best.update(c)

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: MWU + Nesterov on promising candidates ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: MWU + Nesterov Optimization", flush=True)
    print(f"{'='*50}", flush=True)

    # Focus optimization on the top few lengths
    # Score each length by best J found so far
    length_scores = {}
    for n in lengths:
        for cand in candidates[n]:

            if np.isfinite(J):
                if n not in length_scores or J < length_scores[n]:
                    length_scores[n] = J

    # Sort lengths by best score
    if length_scores:
        sorted_lengths = sorted(length_scores.keys(), key=lambda x: length_scores[x])
    else:
        sorted_lengths = lengths

    # Allocate time: more time to shorter sequences (faster per iteration)
    p2_budget = min(350.0, (deadline - p2_start))

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M(a):
    """Compute max autocorrelation of non-negative sequence a."""
    a = np.maximum(a, 0)
    b = fast_conv(a)
    return float(b.max())


def compute_J(a):
    """Compute the objective J(a) = 2n * max(conv(a,a)) / (sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    M = compute_M(a)
    return float(2 * len(a) * M / (s * s))


def normalize_seq(a):
    """Normalize so sum = 1, clip to non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smooth_max_and_grad(a, tau):
    """
    Compute smooth approximation to max of autocorrelation and its gradient.
    Returns (logsumexp_val, gradient, autocorr, max_val).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    log_sum = b_max + np.log(exp_shifted.sum()) / tau
    grad = np.zeros(n)
    for k in range(len(b)):
        if p[k] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return log_sum, grad, b, b_max


def run_mwu(a, deadline, max_iters=3000, tau_start=5.0, tau_end=500.0,
            eta_start=0.5, eta_end=0.01):
    """
    Multiplicative Weights Update on the probability simplex.
    a_i^{(t+1)} proportional to a_i^{(t)} * exp(-eta * grad_i / scale)
    Naturally stays in the simplex without projection.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        eta = eta_start * (eta_end / eta_start) ** frac

        _, grad, b, b_max = smooth_max_and_grad(a, tau)
        if b_max < best_M * (1 - 1e-9):
            best_M = b_max
            best_a = a.copy()

        g_scale = np.abs(grad).max()
        if g_scale < 1e-15:
            break
        grad_s = grad / g_scale

        update = np.exp(-eta * grad_s)
        a = a * update
        a = np.maximum(a, 1e-12)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
        else:
            a = a / s

    return best_a


def run_smooth_nesterov(a, deadline, max_iters=5000, tau_start=3.0, tau_end=500.0,
                        lr_init=0.3, lr_end=0.002):
    """
    Nesterov accelerated smooth gradient descent with Polyak-style normalization.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac

        _, grad, b, b_max = smooth_max_and_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid indices."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=15):
    """Solve LP: max sum(b) s.t. a*b <= M, b >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=300, lp_time=15):
    """
    LP mirror descent with adaptive step sizes.
    Uses coarser steps early, finer steps late.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    base_steps = [0.8, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        if it < 30:
            steps = base_steps[:6]
        elif it < 100:
            steps = base_steps[3:]
        else:
            steps = base_steps[5:]

        g = lp_best_response(a, M, time_limit=lp_time)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        improved = False
        for t in steps:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break

    return best_a


def block_lp_optimize(a, deadline, n_blocks=8, iterations_per_block=3, lp_time=8):
    """
    Block coordinate LP optimization (linearized).
    Split a into blocks; for each block, solve LP with linearized constraints.
    The quadratic self-convolution term of the block is dropped (valid for small blocks).
    """
    a = np.maximum(a.astype(np.float64), 1e-8).copy()
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()

    block_size = n // n_blocks
    if block_size < 4:
        return best_a

    for outer in range(iterations_per_block):
        if time.monotonic() > deadline:
            break
        for b_idx in range(n_blocks):
            if time.monotonic() > deadline:
                break
            lo = b_idx * block_size
            hi = min(n, (b_idx + 1) * block_size)
            m = hi - lo

            a_bg = a.copy()
            a_bg[lo:hi] = 0
            bg_conv = fast_conv(a_bg)

            rhs = best_M - bg_conv
            valid = rhs > 0
            if valid.sum() < 1:
                continue

            rows_idx = []
            cols_idx = []
            vals_idx = []

            for k in range(2 * n - 1):
                if not valid[k]:
                    continue
                for j in range(m):
                    global_j = lo + j
                    bg_idx = k - global_j
                    if 0 <= bg_idx < n:
                        rows_idx.append(len(rows_idx))
                        cols_idx.append(j)
                        vals_idx.append(2.0 * a_bg[bg_idx])

            if len(rows_idx) == 0:
                continue

            A_block = csr_matrix((vals_idx, (rows_idx, cols_idx)),
                                 shape=(len(rows_idx), m))
            c_block = -np.ones(m)
            b_ub_block = rhs[valid]
            bounds_block = [(0, None)] * m

            try:
                r = linprog(c_block, A_ub=A_block, b_ub=b_ub_block,
                            bounds=bounds_block, method='highs',
                            options={'time_limit': lp_time, 'presolve': True})
                if r.success and r.x is not None:
                    a_new_block = np.maximum(r.x, 0)
                    old_sum = a[lo:hi].sum()
                    new_sum = a_new_block.sum()
                    if new_sum > 1e-10:
                        scale = old_sum / new_sum if old_sum > 1e-10 else 1.0
                        a[lo:hi] = a_new_block * scale
                        M_new = fast_conv(a).max()
                        if M_new < best_M - 1e-12:
                            best_M = M_new
                            best_a = a.copy()
            except Exception:
                pass

            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
            if time.monotonic() > deadline:
                break

    return best_a


def spectral_init(n, rng, mode='random_phase'):
    """Generate sequence via frequency domain synthesis."""
    N = 2 * n
    if mode == 'random_phase':
        amp = np.ones(N // 2 + 1)
        phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        spectrum = amp * np.exp(1j * phase)
    elif mode == 'tapered':
        freqs = np.arange(N // 2 + 1)
        amp = 1.0 / (1.0 + 0.15 * freqs)
        phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        spectrum = amp * np.exp(1j * phase)
    elif mode == 'bandpass':
        freqs = np.arange(N // 2 + 1)
        center_f = N // 4
        width = N // 8
        amp = np.exp(-0.5 * ((freqs - center_f) / width) ** 2)
        phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        spectrum = amp * np.exp(1j * phase)
    else:
        return np.ones(n) / n

    full_spectrum = np.zeros(N, dtype=complex)
    full_spectrum[:N // 2 + 1] = spectrum
    for k in range(1, N // 2):
        full_spectrum[N - k] = np.conj(full_spectrum[k])
    a = np.fft.ifft(full_spectrum).real[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def cosine_init(n, p, center=0.5):
    """Generate cosine-power initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = np.abs(np.cos(np.pi * (t - center))) ** p
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a


def gaussian_init(n, sigma, center=0.5):
    """Generate Gaussian-shaped initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a = a / a.sum()
    return a


def resample_seq(src, n):
    """Interpolate a sequence to a new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
