"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.signal import fftconvolve
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse matrix A where (A@g)[k] = (a*g)[k] for g of length n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max < j_min:
            continue
        rows.extend(range(k, k + (j_max - j_min + 1)))
        cols.extend(range(j_min, j_max + 1))
        vals.extend(a[k - j_min:k - j_max - 1:-1] if j_max > j_min else [a[k - j_min]])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def build_conv_matrix_fast(n, a):
    """Faster construction using numpy vectorization."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    nnz_per_row = np.minimum(n, np.minimum(np.arange(1, 2*n), np.arange(2*n-1, 0, -1)))
    total_nnz = nnz_per_row.sum()
    
    rows = np.repeat(np.arange(2*n-1), nnz_per_row)
    cols = np.zeros(total_nnz, dtype=np.int32)
    vals = np.zeros(total_nnz)
    
    pos = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        cnt = j_max - j_min + 1
        cols[pos:pos+cnt] = range(j_min, j_max + 1)
        vals[pos:pos+cnt] = a[k - j_max : k - j_min + 1][::-1]
        pos += cnt
    
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_conv_matrix_fast(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 2, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None

def smooth_max_grad(a, tau):
    """Gradient of soft-max approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi : k - lo + 1][::-1]
        grad[lo : hi + 1] += 2.0 * pk * seg
    return grad, b

def resample_seq(src, dest_n):

def fast_conv(a):

    size = 1

def build_conv_matrix_fast(n, a):
    """Build sparse matrix A where (A@g)[k] = sum_j a[j]*g[k-j]."""
    a = np.asarray(a, dtype=np.float64)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)

        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_unit(a):

            a = a / a.sum()
    
    if verbose:
        print(f"  CoordDesc: J={best_J:.8f} iters={it} t={time.monotonic()-deadline+deadline:.0f}s", flush=True)
    
    return best_a, best_J


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_palindrome(a):
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def generate_inits(n, rng, parent=None):
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.2, n), 0.01))
    
    inits.append(np.ones(n))
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t_arr - 1)), 0.01))
    inits.append(np.maximum(0.3 + 0.7 * np.exp(-3 * (t_arr - 0.5) ** 2), 0.01))
    inits.append(np.maximum(0.3 + 0.7 * np.exp(-4 * (t_arr - 0.35) ** 2), 0.01))
    inits.append(np.maximum(np.exp(-2 * t_arr) + np.exp(-2 * (t_arr - 1)) + 0.2, 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4 * np.pi * t_arr), 0.01))
    
    w = np.ones(n)
    wc = n // 4
    w[wc : n - wc] = 1.4
    w[:wc] = 0.6
    w[n - wc:] = 0.6
    inits.append(w)
    
    tri = np.concatenate([np.linspace(0.2, 1.0, n // 2), np.linspace(1.0, 0.2, n - n // 2)])
    inits.append(tri)
    
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0.5, 1.5, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01))
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a, label=""):
        if a is None or a.size == 0:
            return
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = compute_J(a)
        if score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            msg = f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s"
            if label:
                msg += f" [{label}]"
            print(msg, flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Diverse init + quick gradient (80s)
    Phase 2: LP mirror descent on best candidates (850s)
    Phase 3: Coordinate descent polishing (250s)
    Phase 4: Multi-length + final polish (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "parent")
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "default")
    
    # Phase 1
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Diverse Init + Quick Gradient", flush=True)
    print(f"{'='*60}", flush=True)
    
    p1_dl = time.monotonic() + 80.0
    n_main = 512
    
    candidates = []
    inits = generate_inits(n_main, rng, parent)
    
    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_dl:
            break
        a0 = np.maximum(a0, 1e-6).astype(np.float64)
        a0 = a0 / a0.sum()
        a_opt = run_smooth_gradient(a0, 50, min(time.monotonic() + 8, p1_dl), tau=15, lr=0.12)
        J_val = compute_J(a_opt)
        candidates.append((J_val, a_opt, idx))
        print(f"  init {idx:2d}: J={J_val:.8f}", flush=True)
    
    candidates.sort(key=lambda x: x[0])
    for J_val, a_opt, idx in candidates[:3]:
        tracker.update(a_opt, f"p1_{idx}")
    print(f"Phase 1 best: J={candidates[0][0]:.8f}", flush=True)
    
    # Phase 2
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)
    
    p2_dl = time.monotonic() + 850.0
    n_cands = min(4, len(candidates))
    t2 = p2_dl - time.monotonic()
    allocs = [max(50, int(t2 * f)) for f in [0.4, 0.25, 0.2, 0.15][:n_cands]]
    
    for ci in range(n_cands):
        if time.monotonic() > p2_dl:
            break
        J_i, a_i, _ = candidates[ci]
        ph_dl = min(time.monotonic() + allocs[ci], p2_dl)
        print(f"\n  Cand {ci} (J={J_i:.8f}), budget={allocs[ci]}s", flush=True)
        a_best, J_best = lp_mirror_descent(a_i, ph_dl, verbose=True, max_no_improve=150)
        tracker.update(a_best, f"lp_{ci}")
    
    # Phase 3
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Coordinate Descent Polishing", flush=True)
    print(f"{'='*60}", flush=True)
    
    p3_dl = min(time.monotonic() + 250.0, deadline - 80.0)
    
    if tracker.best is not None and time.monotonic() < p3_dl:
        a_pol, _ = coordinate_descent_polish(tracker.best, p3_dl, verbose=True)
        tracker.update(a_pol, "cd")
        
        if time.monotonic() < p3_dl - 20:
            a_lp2, _ = lp_mirror_descent(tracker.best, min(time.monotonic() + 80, p3_dl - 10),
                                          verbose=False, max_no_improve=100)
            tracker.update(a_lp2, "lp2")
    
    # Phase 4
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Multi-length + Final", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"  Remaining: {deadline - time.monotonic():.0f}s", flush=True)
    
    if tracker.best is not None:
        base_n = tracker.best.size
        try_lengths = [base_n + 32, base_n - 32, base_n * 2, base_n // 2]
        try_lengths = [t for t in try_lengths if 128 <= t <= 2048 and t != base_n]
        
        for n_try in try_lengths[:3]:
            if time.monotonic() > deadline - 40:
                break
            a_ext = resample_seq(tracker.best, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(time.monotonic() + 35, deadline - 40)
            a_opt, _ = lp_mirror_descent(a_ext, sub_dl, verbose=False, max_no_improve=80)
            tracker.update(a_opt, f"n={n_try}")
            if time.monotonic() < deadline - 40:
                a_cd, _ = coordinate_descent_polish(a_opt, min(time.monotonic() + 12, deadline - 40), verbose=False)
                tracker.update(a_cd, f"n={n_try}_cd")
        
        if time.monotonic() < deadline - 30:
            a_pal = make_palindrome(tracker.best)
            a_pal = np.maximum(a_pal, 1e-6) / a_pal.sum()
            a_opt, _ = lp_mirror_descent(a_pal, min(time.monotonic() + 25, deadline - 30),
                                          verbose=False, max_no_improve=60)
            tracker.update(a_opt, "pal")
        
        if time.monotonic() < deadline - 10:
            a_fin, _ = coordinate_descent_polish(tracker.best, deadline - 5, verbose=False)
            tracker.update(a_fin, "final")
    
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
