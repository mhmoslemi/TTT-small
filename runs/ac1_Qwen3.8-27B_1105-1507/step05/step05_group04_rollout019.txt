            a_opt = a_opt / s
        else:
            a_opt = a0.copy()
        return a_opt
    except Exception:
        return a0.copy()


def lbfgs_anneal(a0, tau_schedule, max_iters_each, deadline):
    """Run L-BFGS with progressive temperature increase."""
    a = a0.copy()
    for tau, max_iter in zip(tau_schedule, max_iters_each):
        if time.monotonic() > deadline:
            break
        a = lbfgs_smooth_max(a, tau, max_iter=max_iter)
    return a


def build_lp_matrix(n, a, k_indices=None):
    """Build sparse matrix A where (A@g)[k] = (a*g)[k] for selected k."""
    if k_indices is None:
        k_indices = range(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    if not rows:
        return csr_matrix((len(k_indices), n))
    return csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))


def lp_mirror_descent(a, deadline, verbose=True, top_frac=0.5,
                      max_no_improve=100, base_step=0.3):
    """LP-based mirror descent for final refinement."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = float(b.max())

        # Select top constraints
        if top_frac < 1.0:
            threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
            active_k = np.where(b >= threshold)[0]
        else:
            active_k = np.arange(2 * n - 1)

        if len(active_k) < 5:
            active_k = np.argpartition(b, -10)[-10:]

        A = build_lp_matrix(n, a, active_k)
        c = -np.ones(n)
        b_ub = np.full(len(active_k), M)
        bounds = [(0, np.inf)] * n

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                        method='highs',
                        options={'time_limit': 5, 'presolve': True})
            if not r.success or r.x is None:
                no_improve += 1
                if no_improve > max_no_improve:
                    break
                continue
            g = np.maximum(r.x, 0.0)
        except Exception:
            no_improve += 1
            if no_improve > max_no_improve:
                break
            continue

        if g.sum() < 1e-10:
            break
        g = g / g.sum()

        # Try multiple step sizes
        current_J = compute_J(a)
        improved = False
        for t in [base_step, base_step * 0.5, base_step * 0.25, 0.1, 0.05]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
            elif J_trial < current_J - 1e-12:
                a = a_trial
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > max_no_improve:
                # Perturb and try again
                rng = np.random.default_rng(it * 31 + 7)
                a = a * (1.0 + 0.1 * rng.normal(0, 1, n))
                a = np.maximum(a, 1e-8)
                a = a / a.sum()
                no_improve = 0
                if no_improve > max_no_improve * 3:
                    break

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - (deadline - (deadline - time.monotonic()))
            print(f"    LP iter={it} J={best_J:.8f} M={M:.6f} "
                  f"no_improve={no_improve}", flush=True)

    return best_a, best_J


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences (non-palindromic preferred)."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n))

    # If parent available
    if parent is not None:
        p = parent.astype(np.float64)
        if p.size != n:
            t_old = np.linspace(0, 1, p.size)
            t_new = np.linspace(0, 1, n)
            p = np.maximum(np.interp(t_new, t_old, p), 0.0)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        # Shifted version (breaks palindrome)
        shift = n // 4
        inits.append(np.roll(p, shift))

    # Flat-top window (DFT design, good starting point)
    a0, a1, a2, a3, a4 = 0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368
    ft = a0 - a1 * np.cos(2 * np.pi * t) + a2 * np.cos(4 * np.pi * t) \
         - a3 * np.cos(6 * np.pi * t) + a4 * np.cos(8 * np.pi * t)
    inits.append(np.maximum(ft, 0.01))

    # Asymmetric cosine (non-palindromic)
    inits.append(np.maximum(0.3 + 0.4 * np.cos(np.pi * (2 * t - 0.3)) + 0.2 * np.sin(2 * np.pi * t), 0.01))

    # Exponential taper (asymmetric)
    inits.append(np.maximum(np.exp(-2 * t) + 0.3, 0.01))
    inits.append(np.maximum(np.exp(-2 * (1 - t)) + 0.3, 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.exp(-3 * (t - 0.3) ** 2) + 0.2 * np.exp(-3 * (t - 0.7) ** 2), 0.01))

    # Linear ramp with modulation
    inits.append(np.maximum(0.2 + t + 0.2 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.maximum(0.2 + (1 - t) + 0.2 * np.cos(4 * np.pi * t), 0.01))

    # Multi-frequency non-palindromic
    for f1, f2 in [(1, 3), (2, 5), (1, 4)]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f1 * t) + 0.2 * np.sin(2 * np.pi * f2 * t + 0.5), 0.01))

    # Gaussian bumps at different positions
    inits.append(np.maximum(0.2 + 0.8 * np.exp(-8 * (t - 0.3) ** 2) + 0.4 * np.exp(-8 * (t - 0.8) ** 2), 0.01))
    inits.append(np.maximum(0.2 + 0.6 * np.exp(-5 * (t - 0.2) ** 2) + 0.6 * np.exp(-5 * (t - 0.6) ** 2) + 0.2 * np.exp(-5 * (t - 0.9) ** 2), 0.01))

    # Random smooth
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0.2, 1.0, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01))

    # Sawtooth-like
    inits.append(np.maximum(0.3 + 0.4 * (t % 0.5) * 2.0, 0.01))

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: L-BFGS with smooth max + temperature annealing (600s)
    Phase 2: LP mirror descent refinement (600s)
    Phase 3: Multi-length exploration + final polish (250s)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_a = None
    best_score = float('inf')

    def update_best(a, label=""):
        nonlocal best_a, best_score
        if a is None or a.size == 0:
            return
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = compute_J(a)
        if score < best_score:
            best_a = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST J={score:.8f} n={a.size} t={elapsed:.0f}s [{label}]", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent, "parent")
            print(f"Parent: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        update_best(parent, "default_parent")

    # ===== PHASE 1: L-BFGS with temperature annealing =====
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: L-BFGS Smooth Max Optimization", flush=True)
    print(f"{'='*60}", flush=True)

    p1_deadline = t_start + 600.0
    n_main = 512

    # Try multiple lengths
    lengths = [256, 384, 512, 640, 768]
    # If parent has a good length, include it
    if parent.size not in lengths and 64 <= parent.size <= 1024:
        lengths.append(parent.size)
    lengths = sorted(set(lengths))

    candidates = []  # (J, a, n)

    for n in lengths:
        if time.monotonic() > p1_deadline - 30:
            break
        inits = generate_inits(n, rng, parent)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > p1_deadline - 10:
                break
            a0 = np.maximum(a0, 1e-6).astype(np.float64)
            a0 = a0 / a0.sum()

            # Temperature annealing schedule
            tau_schedule = [5.0, 10.0, 20.0, 40.0, 80.0, 150.0]
            max_iters_each = [300, 300, 300, 300, 300, 300]

            a_opt = lbfgs_anneal(a0, tau_schedule, max_iters_each,
                                 min(time.monotonic() + 60, p1_deadline - 5))
            J_val = compute_J(a_opt)
            candidates.append((J_val, a_opt.copy(), n))
            if (idx + 1) % 5 == 0 or J_val < 1.55:
                print(f"  n={n} init={idx:2d}: J={J_val:.8f}", flush=True)
            update_best(a_opt, f"lbfgs n={n} i={idx}")

    # Sort candidates
    candidates.sort(key=lambda x: x[0])
    print(f"\nPhase 1 best: J={candidates[0][0]:.8f} (n={candidates[0][2]})", flush=True)

    # ===== PHASE 2: LP Mirror Descent Refinement =====
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: LP Mirror Descent Refinement", flush=True)
    print(f"{'='*60}", flush=True)

    p2_deadline = min(t_start + 1200.0, deadline - 100.0)

    # Refine top candidates
    top_n = min(3, len(candidates))
    time_per_cand = (p2_deadline - time.monotonic()) / max(top_n, 1)

    for ci in range(top_n):
        if time.monotonic() > p2_deadline:
            break
        J_init, a_init, n = candidates[ci]
        phase_dl = min(time.monotonic() + time_per_cand, p2_deadline)

        print(f"\n  Refining candidate {ci} (J={J_init:.8f}, n={n}), "
              f"budget={time_per_cand:.0f}s", flush=True)

        # Run LP mirror descent with progressively tighter constraints
        a_lp, J_lp = lp_mirror_descent(
            a_init, phase_dl, verbose=True,
            top_frac=0.6, max_no_improve=80, base_step=0.3
        )
        update_best(a_lp, f"lp_ci{ci}")

        # If still have time, run with tighter constraints
        if time.monotonic() < p2_deadline - 10 and J_lp < best_score + 0.01:
            phase_dl2 = min(time.monotonic() + 60, p2_deadline)
            a_lp2, J_lp2 = lp_mirror_descent(
                a_lp, phase_dl2, verbose=False,
                top_frac=0.4, max_no_improve=50, base_step=0.2
            )
            update_best(a_lp2, f"lp_tight_ci{ci}")

    # ===== PHASE 3: Multi-length + Final Exploration =====
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Multi-length + Final Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None and time.monotonic() < deadline - 20:
        base_n = best_a.size

        # Try nearby lengths
        for n_try in [base_n + 64, base_n - 64, base_n * 2, base_n // 2]:
            if time.monotonic() > deadline - 20:
                break
            if n_try < 64 or n_try > 2048:
                continue
            if n_try == base_n:
                continue

            a_ext = resample_seq(best_a, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            # Quick L-BFGS
            a_ext = lbfgs_anneal(a_ext, [20.0, 50.0, 100.0], [200, 200, 200],
                                 min(time.monotonic() + 30, deadline - 15))
            update_best(a_ext, f"n={n_try}_lbfgs")

            # Short LP
            if time.monotonic() < deadline - 15:
                a_ext2, _ = lp_mirror_descent(
                    a_ext, min(time.monotonic() + 25, deadline - 10),
                    verbose=False, top_frac=0.5, max_no_improve=50
                )
                update_best(a_ext2, f"n={n_try}_lp")

        # Random perturbations of best
        for perturb_round in range(5):
            if time.monotonic() > deadline - 30:
                break
            rng_p = np.random.default_rng(seed + 1000 + perturb_round)
            scale = 0.05 + 0.03 * perturb_round
            a_pert = best_a * (1.0 + scale * rng_p.normal(0, 1, best_a.size))
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert = a_pert / a_pert.sum()

            # Quick optimize
            a_pert = lbfgs_anneal(a_pert, [30.0, 80.0], [150, 150],
                                  min(time.monotonic() + 20, deadline - 10))
            update_best(a_pert, f"perturb_{perturb_round}")

            if time.monotonic() < deadline - 15:
                a_pert2, _ = lp_mirror_descent(
                    a_pert, min(time.monotonic() + 15, deadline - 10),
                    verbose=False, top_frac=0.5, max_no_improve=30
                )
                update_best(a_pert2, f"perturb_{perturb_round}_lp")

    # ===== PHASE 4: Final Full LP Polish =====
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)
    remaining = deadline - time.monotonic()
    print(f"  Remaining time: {remaining:.0f}s", flush=True)

    if best_a is not None and remaining > 30:
        # Full LP polish (all constraints)
        a_final, J_final = lp_mirror_descent(
            best_a, deadline - 5, verbose=True,
            top_frac=1.0, max_no_improve=200, base_step=0.2
        )
        update_best(a_final, "final_full_lp")

        # One more with reduced constraints but more iterations
        if time.monotonic() < deadline - 10:
            a_final2, _ = lp_mirror_descent(
                best_a, deadline - 5, verbose=False,
                top_frac=0.3, max_no_improve=500, base_step=0.15
            )
            update_best(a_final2, "final_red_lp")

    # Fallback
    if best_a is None:
        best_a = np.ones(256)
        best_score = compute_J(best_a)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_a]
