import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    M = np.max(conv)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * M / (s*s)

Strategy: Use LP-based directional search as the main optimizer,

from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):

    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)

    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * M / (s*s)

def build_sparsish_toeplitz(a):

    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        for k in range(max(0, j), min(2*n-1, j+n)):
            A[k, j] = a[k-j]
    return A

def lp_direction(a, M, n_top_constraints=0):

    Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0.
    Optionally only use top n_top_constraints constraints.
    Returns normalized g (sum=1) or None.
    """
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        a_j = a  # each column is a shifted version
        # A[k, j] = a[k-j] if 0 <= k-j < n
        k_min = max(0, j)
        k_max = min(2*n-1, j+n)
        A[k_min:k_max, j] = a[:k_max-k_min]
    
    if n_top_constraints > 0:
        conv_aa = compute_conv(a)
        sorted_idx = np.argsort(conv_aa)[::-1]
        top_k = sorted_idx[:n_top_constraints]
        A = A[top_k]
        M_vec = np.full(len(top_k), M)
    else:
        M_vec = np.full(2*n-1, M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=M_vec, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g/s
    except:
        pass
    return None

    best_a = a.copy()

    for step in range(n_steps):

                a /= np.sum(a)

            i = rng.integers(n)
            j = rng.integers(n)

        else:

                M = new_M

                    best_E = E
                    best_a = a.copy()
            else:
                a = old_a

def normalize(a):

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_min = max(0, 0)
        k_max = min(2 * n - 1, j + n)
        A[k_min:k_max, j] = a[:k_max - k_min]
    return A


def lp_direction_fast(a, M, top_k=0, time_limit=5.0):

        A[:k_max, j] = a[:k_max]

    if top_k > 0 and top_k < 2 * n - 1:
        conv_aa = compute_conv(a)
        sorted_idx = np.argsort(conv_aa)[::-1]
        top_idx = sorted_idx[:top_k]
        A_use = A[top_idx]
        b_use = np.full(top_k, M)
    else:
        A_use = A
        b_use = np.full(2 * n - 1, M)

    c = -np.ones(n)
    bounds = [(0.0, None)] * n

def subgradient_step(a, step_size, momentum_v=None, mu=0.8):
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]

    grad -= np.mean(grad)

    if momentum_v is None:
        momentum_v = np.zeros(n)
    momentum_v = mu * momentum_v - step_size * grad
    a_new = a + momentum_v
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a, momentum_v
    a_new = a_new / s
    return a_new, momentum_v

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def make_polynomial_init(n, rng):
    coeffs = rng.normal(0, 1, min(n, 20))
    x = np.linspace(-1, 1, n)
    poly = np.zeros(n)
    for i, c in enumerate(coeffs):
        poly += c * x**i
    a = np.abs(poly)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

    if best_seq is None:
        best_seq = np.ones(700)
        best_score = compute_obj(best_seq)
        print(f"No parent, using uniform: obj={best_score:.8f}")

    n_parent = len(best_seq)

    # Phase 1: LP polish the parent (200s)
    print(f"\n=== Phase 1: LP Polish (n={n_parent}) ===")
    phase1_end = min(start_time + 200.0, deadline - 300.0)
    a_work = normalize(best_seq.copy())
    a_work, obj_work = lp_polish(a_work, max_iters=80, time_budget=phase1_end - time.monotonic())
    if obj_work < best_score:
        best_score = obj_work
        best_seq = a_work.copy()
        print(f"  After LP polish: {best_score:.8f}")

    # Phase 2: SA + LP cycling (700s)
    print(f"\n=== Phase 2: SA + LP Cycling ===")
    phase2_end = min(start_time + 200.0 + 700.0, deadline - 200.0)
    a_work = normalize(best_seq.copy())

    cycle = 0
    while time.monotonic() < phase2_end - 30:
        remaining = phase2_end - time.monotonic()
        if remaining < 20:
            break

        # SA phase
        sa_budget = min(remaining * 0.6, 120.0)
        n_sa_steps = int(sa_budget * 6000)  # ~6000 steps/sec target
        T_start = 0.04 + 0.01 * (cycle % 3)
        T_end = 0.0005 + 0.0001 * (cycle % 5)
        rng_sa = np.random.default_rng(seed + cycle * 1000)
        a_sa, obj_sa = sa_optimize(a_work, n_sa_steps, T_start, T_end, rng_sa)
        obj_sa_full = compute_obj(a_sa)
        if obj_sa_full < best_score:
            best_score = obj_sa_full
            best_seq = a_sa.copy()
            a_work = a_sa.copy()
            print(f"  Cycle {cycle} SA: {best_score:.8f}")

        # LP phase
        remaining = phase2_end - time.monotonic()
        if remaining < 10:
            break
        lp_budget = min(remaining * 0.4, 40.0)
        a_lp, obj_lp = lp_polish(a_work, max_iters=20, time_budget=lp_budget)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            a_work = a_lp.copy()
            print(f"  Cycle {cycle} LP: {best_score:.8f}")

        # Subgradient polish
        remaining = phase2_end - time.monotonic()
        if remaining > 5:
            a_sg = a_work.copy()
            v = np.zeros(len(a_sg))
            step = 0.03
            for _ in range(200):
                if time.monotonic() > phase2_end - 5:
                    break
                a_sg, v = subgradient_step(a_sg, step, v)
                step *= 0.995
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                a_work = a_sg.copy()
                print(f"  Cycle {cycle} SG: {best_score:.8f}")

        cycle += 1

    # Phase 3: Try different n values (250s)
    print(f"\n=== Phase 3: Explore n values ===")
    print(f"  Current best: {best_score:.8f} (n={len(best_seq)})")
    phase3_end = min(start_time + 200.0 + 700.0 + 250.0, deadline - 80.0)

    n_options = sorted(set([
        n_parent - 50, n_parent - 25, n_parent,
        n_parent + 25, n_parent + 50, n_parent + 100,
        500, 600, 800, 900, 1000
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1200]

    for n_try in n_options:
        if time.monotonic() > phase3_end - 20:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 15:
            break

        if n_try == len(best_seq):
            a_try = normalize(best_seq.copy())
        else:
            a_try = resample(best_seq, n_try)

        # Quick SA
        rng_try = np.random.default_rng(seed + n_try * 7)
        n_steps = int(min(remaining * 0.4, 30000))
        a_try, _ = sa_optimize(a_try, n_steps, 0.03, 0.001, rng_try)

        # Quick LP
        remaining = phase3_end - time.monotonic()
        if remaining > 10:
            a_try, obj_try = lp_polish_topk(a_try, max_iters=15,
                                            time_budget=min(remaining * 0.5, 20.0), top_k=25)
        else:
            obj_try = compute_obj(a_try)

        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: {best_score:.8f} (UPDATE) **")
        else:
            print(f"  n={n_try}: {obj_try:.8f}")

    # Phase 4: Alternative initializations with SA (120s)
    print(f"\n=== Phase 4: Alternative starts ===")
    phase4_end = min(start_time + 1500.0 - 40.0, deadline - 30.0)
    n_use = len(best_seq)

    inits = [
        ("poly1", make_polynomial_init(n_use, np.random.default_rng(seed + 1))),
        ("poly2", make_polynomial_init(n_use, np.random.default_rng(seed + 2))),
        ("alt1", make_alternating_init(n_use, np.random.default_rng(seed + 3))),
        ("alt2", make_alternating_init(n_use, np.random.default_rng(seed + 4))),
        ("rand1", np.maximum(np.random.default_rng(seed + 5).normal(1, 0.5, n_use), 0.01)),
    ]

    for name, a_init in inits:
        if time.monotonic() > phase4_end - 15:
            break
        remaining = phase4_end - time.monotonic()
        if remaining < 10:
            break

        a_init = normalize(a_init)
        # Short SA
        rng_alt = np.random.default_rng(seed + hash(name) % 10000)
        n_steps = int(min(remaining * 0.5, 30000))
        a_res, _ = sa_optimize(a_init, n_steps, 0.05, 0.001, rng_alt)

        # Quick LP
        remaining = phase4_end - time.monotonic()
        if remaining > 10:
            a_res, obj_res = lp_polish_topk(a_res, max_iters=10,
                                            time_budget=min(remaining * 0.4, 15.0), top_k=20)
        else:
            obj_res = compute_obj(a_res)

        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  {name}: {best_score:.8f} (UPDATE)")
        else:
            print(f"  {name}: {obj_res:.8f}")

    # Phase 5: Final intensive polish (remaining time)
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"  Before final: {best_score:.8f} (n={len(best_seq)})")

    if time.monotonic() < deadline - 10:
        a_final = normalize(best_seq.copy())
        remaining = deadline - time.monotonic() - 5.0

        # Multiple LP rounds
        for round_idx in range(5):
            if time.monotonic() > deadline - 10:
                break
            rem = deadline - time.monotonic() - 5.0
            if rem < 5:
                break
            a_final, obj_f = lp_polish(a_final, max_iters=30, time_budget=min(rem * 0.3, 30.0))
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_final.copy()
                print(f"  Final LP round {round_idx}: {best_score:.8f}")

            # SA perturbation
            if time.monotonic() < deadline - 20:
                rng_fin = np.random.default_rng(seed + 100 + round_idx)
                a_pert = a_final * np.exp(rng_fin.normal(0, 0.015, len(a_final)))
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = normalize(a_pert)
                rem = deadline - time.monotonic() - 10.0
                if rem > 5:
                    a_pert, _ = sa_optimize(a_pert, int(rem * 3000), 0.02, 0.0005,
                                            np.random.default_rng(seed + 200 + round_idx))
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_final = a_pert.copy()
                        print(f"  Final SA pert {round_idx}: {best_score:.8f}")

        # Very fine subgradient
        if time.monotonic() < deadline - 10:
            a_sg = normalize(best_seq.copy())
            v = np.zeros(len(a_sg))
            step = 0.01
            rem = deadline - time.monotonic() - 5.0
            max_iters = int(rem * 2000)
            for _ in range(max_iters):
                if time.monotonic() > deadline - 5:
                    break
                a_sg, v = subgradient_step(a_sg, step, v)
                step *= 0.998
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  Final SG: {best_score:.8f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
