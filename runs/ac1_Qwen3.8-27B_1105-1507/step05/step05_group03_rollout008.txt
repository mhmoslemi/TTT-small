import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def solve_lp_tight(a, M, row_idx, col_idx, tight_fraction=0.8):
    """Solve LP with slightly tighter constraints to encourage more aggressive updates."""
    n = len(a)
    conv_a = fast_conv(a)
    b_ub = np.full(2 * n - 1, M)
    # For the top constraints, use a tighter bound
    top_k = max(1, int(tight_fraction * (2 * n - 1)))
    sorted_idx = np.argsort(conv_a)
    for idx in sorted_idx[-top_k:]:
        b_ub[idx] = M * 0.95
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2_coeffs."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def line_search_multi(a, g):
    """Try multiple step sizes and return the best."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    # Ternary search first
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t_best = (lo + hi) / 2.0
    f_best = float(np.max(a_coef + b_coef * t_best + c_coef * t_best * t_best))

    # Also try some fixed step sizes
    for t_try in [0.1, 0.25, 0.5, 0.75, 1.0]:
        f_try = float(np.max(a_coef + b_coef * t_try + c_coef * t_try * t_try))
        if f_try < f_best:
            t_best = t_try
            f_best = f_try

    return t_best, f_best


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False,
                          use_tight=False, tight_frac=0.8):
    """LP mirror descent with analytic line search and adaptive M."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        if use_tight and it > 10:
            g = solve_lp_tight(a, M, row_idx, col_idx, tight_frac)
            if g is None:
                g = solve_lp(a, M, row_idx, col_idx)
        else:
            g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_multi(a, g)

        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Try a smaller step
            if t > 0.05:
                t_small = t * 0.3
                a_new = (1.0 - t_small) * a + t_small * g
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    actual_M = compute_M(a_new)
                    if actual_M < best_M:
                        a = a_new
                        best_M = actual_M
                        best_a = a.copy()
                        no_improve = 0

        if no_improve > 80:
            break

        if verbose and it % 100 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def variance_reduction_step(a, n_iters, deadline):
    """
    Reduce the variance of the autocorrelation using smooth gradient descent.
    Objective: sum_k ((a*a)[k] - mean)^2, which encourages flat autocorrelation.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = float('inf')

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        r = fast_conv(a)
        r_mean = r.mean()
        residuals = r - r_mean

        # Gradient of sum_k residuals[k]^2 w.r.t. a
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            res_k = residuals[k]
            if abs(res_k) < 1e-12:
                continue
            # d(a*a)[k]/da[i] = 2*a[k-i] (if valid), approximately
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # grad[i] += 2 * res_k * a[k-i] for i in [lo, hi]
            for i in range(lo, hi + 1):
                idx = k - i
                if 0 <= idx < n:
                    grad[i] += 2.0 * res_k * a[idx]

        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        a_new = a - 0.1 * grad / grad_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

        # Track objective
        r_new = fast_conv(a)
        obj = np.sum((r_new - r_new.mean()) ** 2)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def spectral_flatten(a, n_iters, deadline):
    """
    Iterative spectral flattening: modify the amplitude spectrum to flatten autocorrelation.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        # Compute autocorrelation
        r = fast_conv(a)
        r_max = r.max()
        r_mean = r.mean()

        # Compute FFT
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Modify magnitude: reduce peaks in autocorrelation domain
        # r = IFFT(|A|^2), so to flatten r, we want |A|^2 to be flatter
        power = mag ** 2
        # Apply a "soft cap" to the power spectrum
        # This encourages the autocorrelation to be flatter
        power_smooth = power + r_mean * (1.0 + 0.5 * np.cos(np.pi * np.arange(len(power)) / (len(power) - 1)))
        mag_new = np.sqrt(power_smooth)

        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

        M = compute_M(a)
        if M < best_M:
            best_M = M
            best_a = a.copy()

    return best_a


def smooth_gradient_descent(a, max_iters, deadline, initial_tau=5.0, final_tau=100.0):
    """Smooth gradient descent on log-sum-exp approximation of max."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = 0.2 * (0.02 / 0.2) ** frac

        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                idx = k - i
                if 0 <= idx < n:
                    grad[i] += 2.0 * pk * a[idx]

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    j = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n))

    # Cosine-modified: 1 + alpha*cos(2*pi*j/n)
    for alpha in [0.3, 0.5, 0.7, 0.9]:
        seq = 1.0 + alpha * np.cos(2 * np.pi * j / n)
        seq = np.maximum(seq, 0.0)
        if seq.sum() > 0:
            inits.append(seq)

    # Cosine-modified: 1 + alpha*cos(4*pi*j/n)
    for alpha in [0.3, 0.5, 0.7]:
        seq = 1.0 + alpha * np.cos(4 * np.pi * j / n)
        seq = np.maximum(seq, 0.0)
        if seq.sum() > 0:
            inits.append(seq)

    # Sinusoidal: sin^2(pi*j/n)
    inits.append(np.sin(np.pi * j / n) ** 2)

    # Sinusoidal: cos^2(pi*j/(2*n))
    inits.append(np.cos(np.pi * j / (2 * n)) ** 2)

    # Gaussian centered
    center = n / 2.0
    inits.append(np.exp(-2.0 * (j - center) ** 2 / n))

    # Gaussian edge-weighted
    inits.append(np.exp(-2.0 * (j - center) ** 2 / (n * 0.3)))

    # Triangular (peak in middle)
    tri = 1.0 - np.abs(2.0 * j / (n - 1) - 1.0)
    inits.append(np.maximum(tri, 0.01))

    # Triangular (peak at edges)
    tri_edge = np.abs(2.0 * j / (n - 1) - 1.0) + 0.1
    inits.append(tri_edge)

    # Multi-peak
    inits.append(1.0 + 0.5 * np.cos(6 * np.pi * j / n))

    # Random (a few)
    for _ in range(5):
        seq = rng.random(n) + 0.1
        inits.append(seq)

    # Random with structure
    for _ in range(3):
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        seq = 1.0 + 0.5 * np.cos(2 * np.pi * freq * j / n + phase)
        seq = np.maximum(seq, 0.0)
        if seq.sum() > 0:
            inits.append(seq)

    return inits


def structured_perturb(a, rng, n_perturbations=10):
    """Apply structured perturbations (cosine modes) to a sequence."""
    n = len(a)
    j = np.arange(n, dtype=np.float64)
    results = [a.copy()]

    for _ in range(n_perturbations):
        mode = rng.integers(1, max(2, n // 4))
        amplitude = rng.uniform(0.02, 0.1)
        sign = rng.choice([1, -1])
        perturbation = sign * amplitude * np.cos(2 * np.pi * mode * j / n + rng.uniform(0, 2 * np.pi))
        a_new = a + perturbation
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() > 1e-10:
            a_new = a_new / a_new.sum()
            results.append(a_new)

    return results

        if tracker.best is not None:
            a_hp = tracker.best.copy() / tracker.best.sum()
            a_hp = run_lp_mirror_descent(a_hp, 5000, min(time.monotonic() + 20, phase3_deadline),
                                         r_idx_b, c_idx_b, verbose=False)
            tracker.update(a_hp, "(P3 spectral+LP)")

        # Structured perturbations
        if time.monotonic() < phase3_deadline:
            perturbations = structured_perturb(tracker.best, rng, n_perturbations=8)
            n_pert = len(perturbations)
            time_per_pert = max(5, (phase3_deadline - time.monotonic()) / max(1, n_pert))

            for pi, a_pert in enumerate(perturbations):
                if time.monotonic() > phase3_deadline:
                    break
                sub_dl = time.monotonic() + time_per_pert
                # Quick gradient then LP
                a_q = smooth_gradient_descent(a_pert, 100, min(time.monotonic() + 5, sub_dl),
                                              initial_tau=10, final_tau=60)
                a_opt = run_lp_mirror_descent(a_q, 3000, sub_dl, r_idx_b, c_idx_b, verbose=False)
                tracker.update(a_opt, f"(P3 pert-{pi})")

    # Phase 4: Multi-length exploration
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    phase4_deadline = t_start + budget_s * 0.85

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.5, 0.625, 0.75, 1.25, 1.5, 1.75, 2.0, 2.5]:
            n_trial = int(base_n * mult)
            if 64 <= n_trial <= 8192 and n_trial != base_n:
                trial_lengths.append(n_trial)
        trial_lengths = sorted(set(trial_lengths))

        n_trials = len(trial_lengths)
        time_per_trial = max(10, (phase4_deadline - time.monotonic()) / max(1, n_trials))

        for n_trial in trial_lengths:
            if time.monotonic() > phase4_deadline:
                break

            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()

            r_idx_t, c_idx_t = build_lp_matrix(n_trial)
            sub_dl = time.monotonic() + time_per_trial

            # Quick gradient first
            a_q = smooth_gradient_descent(a_init, 200, min(time.monotonic() + 8, sub_dl),
                                          initial_tau=5, final_tau=50)
            # Then LP
            a_opt = run_lp_mirror_descent(a_q, 5000, sub_dl, r_idx_t, c_idx_t, verbose=False)
            tracker.update(a_opt, f"(P4 n={n_trial})")

    # Phase 5: Final extended LP polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)
        a = tracker.best.copy()
        a = a / a.sum()

        # Variance reduction first
        a = variance_reduction_step(a, 100, min(time.monotonic() + 15, deadline))

        # Then extended LP
        a_opt = run_lp_mirror_descent(a, 50000, deadline, r_f, c_f, verbose=True,
                                      use_tight=True, tight_frac=0.75)
        tracker.update(a_opt, "(P5 final)")

        # One more round: perturb + LP
        if tracker.best is not None and time.monotonic() < deadline - 30:
            r_idx_f2, c_idx_f2 = build_lp_matrix(tracker.best.size)
            a_final = tracker.best.copy() / tracker.best.sum()
            j = np.arange(tracker.best.size, dtype=np.float64)
            for mode in [1, 2, 3]:
                if time.monotonic() > deadline - 10:
                    break
                for amp in [0.01, 0.03]:
                    a_tr = a_final + amp * np.cos(2 * np.pi * mode * j / tracker.best.size)
                    a_tr = np.maximum(a_tr, 0.0)
                    if a_tr.sum() < 1e-10:
                        continue
                    a_tr = a_tr / a_tr.sum()
                    sub_dl = min(time.monotonic() + 15, deadline)
                    a_opt = run_lp_mirror_descent(a_tr, 3000, sub_dl, r_idx_f2, c_idx_f2, verbose=False)
                    tracker.update(a_opt, f"(P5 m={mode} a={amp})")

    # Ultimate fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    tracker.report("FINAL")
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
