                A[i, j] = 1.0
    
    # System: A * a_T = lambda * ones, ones^T * a_T = 1
    # Combined: [A, -ones] * [a_T; lambda] = 0
    #           [ones^T, 0] * [a_T; lambda] = 1
    M_mat = np.zeros((m+1, m+1))
    M_mat[:m, :m] = A
    M_mat[:m, m] = -1.0
    M_mat[m, :m] = 1.0
    b = np.zeros(m+1)
    b[m] = 1.0
    
    try:
        sol = np.linalg.solve(M_mat, b)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(M_mat, b, rcond=None)[0]
    
    a_sol = np.zeros(n)
    for i, t in enumerate(T):
        a_sol[t] = sol[i]
    return a_sol

Wait, but this system might be singular or ill-conditioned. Let me think about when it's well-conditioned.

The matrix A[i,j] = 1 if T[i] + T[j] in S is a kind of "sum matrix" restricted to T. If S is the full range {0, ..., 2n-2}, then A is all-ones (since T[i] + T[j] is always in {0, ..., 2n-2} for T[i], T[j] in {0, ..., n-1}). In that case, the system is:
m * a[j] = λ for all j (all rows of A are the same)
sum a[j] = 1

This gives a[j] = 1/m for all j, λ = m/m = 1. But this is the uniform distribution, which we know is not optimal.

The issue is that when S is too large, the constraints are redundant and the system is underdetermined or degenerate.

For a good solution, S should be a proper subset of {0, ..., 2n-2}. The active set at the optimum is the set of k where (a*a)[k] achieves its maximum. For a "good" sequence, this should be a small set (maybe O(n) but not all 2n-1).

Let me think about what the active set looks like for a good sequence.

For the uniform sequence, (a*a)[k] = (n - |k - (n-1)|) / n^2. The maximum is at k = n-1, and the active set (within ε of max) would be a small interval around n-1.

For a sequence optimized to reduce the peak, the active set might be spread out more.

**Iterative Active Set Method:**

1. Start with a good initial a (from LP mirror descent)
2. Compute M = max(a*a), identify S = {k : (a*a)[k] >= M - ε}
3. Identify T = {i : a[i] > δ}
4. Solve the linear system to get a_new
5. If a_new is valid (all components in T are positive, all (a*a)[k] <= M_new for k not in S_new where S_new is the new active set), accept and refine
6. Otherwise, adjust S or T:
   - If some a_new[t] < 0: remove t from T
   - Recompute M_new = max(a_new * a_new)
   - If new max locations not in S: add them to S
   - If some old S locations are no longer at the max: remove them
7. Repeat

The dual problem is:

At the optimum, y is supported on the active set S, and for j in T (support of a):
sum_{k in S} a[k-j] * y[k] = 1

So if I parametrize y (which is supported on S), the dual constraint becomes:

And the objective is M * sum(y).

"""
Optimized solver for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:

import numpy as np

from scipy.sparse import csr_matrix, csc_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def evaluate_and_track(a, tracker):

def lp_best_response(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    # Build constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception as e:
        pass
    return None

def lp_mirror_descent(a, max_iters, deadline, verbose=True):
    """LP mirror descent for Nash equilibrium."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b_conv = fast_conv(a)
        M = float(b_conv.max())
        
        t0 = time.monotonic()
        g = lp_best_response(a, M, time_limit=5)
        lp_time = time.monotonic() - t0
        
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Geometric line search
        improved = False
        base_t = max(0.02, 0.4 * (1.0 - 0.3 * it / max(1, max_iters)))

            a_trial = np.maximum((1-t)*a + t*g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:

            print(f"    LP iter {it}: M={M:.10f}, best_M={best_M:.10f}, lp_time={lp_time*1000:.0f}ms", flush=True)
    
    return best_a

def active_set_solve(a_init, n, S, T, max_iter=100):
    """
    Active set method: solve KKT conditions for given active set S and support T.
    Iterate: solve linear system, check validity, adjust S and T.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    S = list(set(S))
    T = list(set(T))
    
    for outer in range(max_iter):
        m = len(T)
        if m == 0:
            return a
        
        # Build matrix A[i,j] = 1 if T[i] + T[j] in S
        S_set = set(S)
        A = np.zeros((m, m))
        for i in range(m):
            ti = T[i]
            for j in range(m):
                if ti + T[j] in S_set:
                    A[i, j] = 1.0
        
        # System: [A, -1; 1^T, 0] * [a_T; lambda] = [0; 1]
        M_mat = np.zeros((m + 1, m + 1))
        M_mat[:m, :m] = A
        M_mat[:m, m] = -1.0
        M_mat[m, :m] = 1.0
        rhs = np.zeros(m + 1)
        rhs[m] = 1.0
        
        try:
            sol = np.linalg.solve(M_mat, rhs)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(M_mat, rhs, rcond=1e-12)[0]
            except:
                break
        
        # Extract solution
        a_new = np.zeros(n)
        a_T = sol[:m]
        lam = sol[m]
        
        # Check positivity
        if np.any(a_T < -1e-10):
            # Remove negative coordinates from T
            for idx in np.where(a_T < -1e-10)[0]:
                if T[idx] in T:
                    T.remove(T[idx])
            continue
        
        # Clip small negatives to 0
        a_T = np.maximum(a_T, 0.0)
        a_T_sum = a_T.sum()
        if a_T_sum < 1e-15:
            break
        a_T /= a_T_sum
        
        for i, t in enumerate(T):
            a_new[t] = a_T[i]
        
        # Check if (a*a)[k] = M for k in S and (a*a)[k] <= M for k not in S
        conv_new = fast_conv(a_new)
        M_new = conv_new.max()
        
        # Check active set validity
        S_valid = all(conv_new[k] >= M_new * (1 - 1e-6) for k in S)
        S_no_excess = all(conv_new[k] <= M_new * (1 + 1e-6) + 1e-15 for k in range(2*n-1) if k not in S_set)
        
        if S_valid and S_no_excess:
            # Check KKT: for i in T, sum_{k in S} a[k-i] should be constant
            kkt_vals = []
            for t in T:
                val = sum(a_new[k-t] for k in S if 0 <= k-t < n)
                kkt_vals.append(val)
            kkt_vals = np.array(kkt_vals)
            if np.std(kkt_vals) < 1e-8 * max(1, np.max(np.abs(kkt_vals))):
                a = a_new
                # Converged!
                if verbose:
                    print(f"    Active set converged at outer iter {outer}, M={M_new:.10f}", flush=True)
                return a_new
        
        # Adjust active set
        # Add new active constraints
        new_active = np.where(conv_new >= M_new * 0.999)[0]
        S = list(set(list(S) + list(new_active)))
        
        # Check if any S elements are no longer active
        S = [k for k in S if conv_new[k] >= M_new * 0.99]
        
        # Update support
        a = a_new
        
    return a

def coord_descent_fine(a, deadline, n_steps_per_coord=5, seed=42):
    """Fine coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return a
    
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    
    no_improve = 0
    max_no_improve = 50
    
    while time.monotonic() < deadline:
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            old = a[i]
            if old < 1e-12:
                continue
            
            # Try a fine grid around the current value
            ratios = np.array([0.5, 0.75, 0.9, 0.95, 1.05, 1.1, 1.25, 1.5, 0.0])
            candidates = np.unique(np.concatenate([[0.0], ratios * old]))
            
            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)
            
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2
            
            for x in candidates:
                delta = x - old
                if abs(delta) < 1e-14:
                    continue
                
                # Apply
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                
                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                
                # Rollback
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
            
            if best_val != old:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True
        
        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
    
    a = best_a
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a

def resample_seq(src, dest_n):

    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def solve_kkt_system(n, S, T):
    """
    Solve the KKT linear system for given active set S and support T.
    
    For i in T: sum_{k in S} a[k-i] = lambda
    sum_{i in T} a[i] = 1
    
    Returns (a, lambda, success)
    """
    m = len(T)
    if m == 0:
        return None, None, False
    
    S_set = set(S)
    
    # Build the system matrix
    # Unknowns: a[t_0], a[t_1], ..., a[t_{m-1}], lambda
    # Equations: for each t_p in T: sum_{t_q in T, t_p+t_q in S} a[t_q] - lambda = 0
    #            sum a[t_q] = 1
    
    A = np.zeros((m + 1, m + 1))
    for p in range(m):
        tp = T[p]
        for q in range(m):
            if (tp + T[q]) in S_set:
                A[p, q] = 1.0
        A[p, m] = -1.0  # coefficient of lambda
    
    A[m, :m] = 1.0  # sum equation
    A[m, m] = 0.0
    
    rhs = np.zeros(m + 1)
    rhs[m] = 1.0
    
    # Solve
    try:
        # Check condition number
        cond = np.linalg.cond(A)
        if cond > 1e15:
            # Use lstsq
            sol, residuals, rank, sv = np.linalg.lstsq(A, rhs, rcond=1e-12)
        else:
            sol = np.linalg.solve(A, rhs)
    except np.linalg.LinAlgError:
        sol, residuals, rank, sv = np.linalg.lstsq(A, rhs, rcond=1e-12)
    
    a_T = sol[:m]
    lam = sol[m]
    
    # Check validity
    if np.any(a_T < -1e-10):
        return None, None, False
    
    a = np.zeros(n)
    for i, t in enumerate(T):
        a[t] = max(0.0, a_T[i])
    
    # Renormalize
    s = a.sum()
    if s < 1e-15:
        return None, None, False
    a /= s
    
    return a, lam, True

"""
Optimized solver for J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Key innovations:
1. LP mirror descent with adaptive line search
2. Active set KKT solver (exploits the linear structure of optimality conditions)
3. Fine coordinate descent with incremental convolution
4. Multi-length exploration with resampling

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[: 2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=5):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, verbose=False):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    step_decay = 0.97
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b_conv = fast_conv(a)
        M = float(b_conv.max())
        
        g = lp_best_response(a, M, time_limit=min(5, max(1, (deadline - time.monotonic()) / 10)))
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        improved = False
        base_t = 0.3 * step_decay ** it
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02, 0.01, 0.005]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 60:
                break
        
        if verbose and it % 5 == 0:
            print(f"    LP it={it}: M={M:.12f} best_M={best_M:.12f}", flush=True)
    
    return best_a


def active_set_iteration(a_init, n, max_outer=100, verbose=False):
    """
    Active set KKT iteration:
    1. Identify S (active constraints) and T (support)
    2. Solve linear KKT system
    3. Check validity, adjust S/T if needed
    4. Iterate until convergence
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for outer in range(max_outer):
        conv = fast_conv(a)
        M = conv.max()
        
        # Identify active set: constraints within 0.05% of max
        tol = 1e-4 * M
        S_mask = conv >= (M - tol)
        S = np.where(S_mask)[0].tolist()
        
        # Identify support
        T = np.where(a > 1e-8 * a.max())[0].tolist()
        
        if len(S) == 0 or len(T) == 0:
            break
        
        m = len(T)
        S_set = set(S)
        
        # Build KKT matrix: B[i,j] = 1 if T[i] + T[j] in S
        B = np.zeros((m, m), dtype=np.float64)
        for i in range(m):
            ti = T[i]
            # T[j] ranges over T, check if ti + T[j] in S
            sums = ti + np.array(T)
            mask = np.array([s_val in S_set for s_val in sums])
            B[i, mask] = 1.0
        
        # System: [B, -1; 1^T, 0] [a_T; lambda] = [0; 1]
        sys_mat = np.zeros((m + 1, m + 1))
        sys_mat[:m, :m] = B
        sys_mat[:m, m] = -1.0
        sys_mat[m, :m] = 1.0
        rhs = np.ones(m + 1)
        rhs[:m] = 0.0
        
        # Solve
        try:
            cond = np.linalg.cond(sys_mat)
            if cond > 1e14:
                sol = np.linalg.lstsq(sys_mat, rhs, rcond=1e-12)[0]
            else:
                sol = np.linalg.solve(sys_mat, rhs)
        except np.linalg.LinAlgError:
