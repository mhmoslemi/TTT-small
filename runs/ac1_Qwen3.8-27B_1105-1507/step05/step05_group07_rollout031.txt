def fast_conv(a):

    s = a.sum()

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def subgradient_mirror_descent(a, max_iters, deadline, lr_scale=1.0):
    """
    Mirror descent with subgradients of max(conv(a,a)).
    Multiplicative updates: a[i] *= exp(-eta * g[i]), then normalize.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    eta = lr_scale / np.sqrt(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 0.01 * M + 1e-10
        active = np.where(b >= M - tol)[0]

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # g[j] += 2*a[k-j] for j in [lo, hi]
            partner_idx = k - np.arange(lo, hi + 1)
            g[lo:hi + 1] += 2.0 * a[partner_idx]

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Adaptive learning rate
        step = eta * M / (gnorm * a.sum())
        step = min(step, 0.5)

        # Mirror update
        a_new = a * np.exp(-step * g / M)
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()

        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            eta *= 0.9

        if no_improve > 50:
            break

    return best_a


def simulated_annealing(a, deadline, seed=0, T0=0.02, cooldown=0.995):
    """
    Simulated annealing with diverse moves:
    - Single coordinate perturbation
    - Block perturbation
    - Frequency-domain sinusoidal perturbation
    - Asymmetric shift (break symmetry)
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    a = a / a.sum()
    rng = np.random.default_rng(seed)
    T = T0
    best_a = a.copy()
    best_obj = compute_J(a)
    current_obj = best_obj

    b = fast_conv(a)
    no_improve = 0

    while time.monotonic() < deadline:
        a_trial = a.copy()
        move_type = rng.integers(0, 4)

        if move_type == 0:
            # Single coordinate
            i = rng.integers(n)
            scale = a[i] if a[i] > 1e-10 else 1.0 / n
            a_trial[i] = max(0.0, a[i] + rng.normal(0, 0.1 * scale))
        elif move_type == 1:
            # Block
            block_len = rng.integers(5, min(50, n // 4))
            start = rng.integers(0, n - block_len)
            perturbation = rng.normal(0, 0.05 / n, block_len)
            a_trial[start:start + block_len] = np.maximum(a_trial[start:start + block_len] + perturbation, 0)
        elif move_type == 2:
            # Frequency perturbation
            freq = rng.uniform(0.5, n / 4)
            phase = rng.uniform(0, 2 * np.pi)
            idx = np.arange(n)
            perturbation = 0.05 * np.sin(2 * np.pi * freq * idx / n + phase) / n
            a_trial = np.maximum(a + perturbation, 0)
        else:
            # Asymmetric shift: move mass from one side to the other
            shift_amt = rng.uniform(0.01, 0.05)
            half = n // 2
            a_trial[:half] *= (1 + shift_amt)
            a_trial[half:] *= (1 - shift_amt * half / (n - half))
            a_trial = np.maximum(a_trial, 0)

        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s

        obj_trial = compute_J(a_trial)

        if obj_trial < current_obj:
            a = a_trial
            current_obj = obj_trial
            if obj_trial < best_obj - 1e-10:
                best_obj = obj_trial
                best_a = a.copy()
                no_improve = 0
                print(f"    SA: J={best_obj:.8f} T={T:.5f}", flush=True)
            else:
                no_improve += 1
        else:
            # Accept with probability exp((current - trial) / T)
            if rng.random() < np.exp(-(obj_trial - current_obj) / max(T, 1e-10)):
                a = a_trial
                current_obj = obj_trial
            no_improve += 1

        T *= cooldown
        if T < 1e-6:
            break
        if no_improve > 200:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            asym[:n//2] *= (1 + shift)
            asym[n//2:] *= (1 - shift * (n//2) / (n - n//2))
            asym = np.maximum(asym, 1e-4)
            asym = asym / asym.sum()
            inits.append(asym)

    # Linear ramp (asymmetric)
    ramp = np.linspace(0.1, 2.0, n)
    inits.append(ramp / ramp.sum())
    ramp2 = np.linspace(2.0, 0.1, n)
    inits.append(ramp2 / ramp2.sum())

    # Exponential (asymmetric)
    for alpha in [0.5, 1.0, 2.0]:
        exp_seq = np.exp(alpha * t)
        inits.append(exp_seq / exp_seq.sum())
        exp_seq2 = np.exp(-alpha * t)
        inits.append(exp_seq2 / exp_seq2.sum())

    # Two-bump asymmetric
    for c1, c2 in [(0.2, 0.7), (0.3, 0.6), (0.15, 0.85)]:
        for w in [0.08, 0.15]:
            bump = np.exp(-((t - c1) / w) ** 2) + 0.7 * np.exp(-((t - c2) / w) ** 2)
            inits.append(bump / bump.sum())

    # Sine variants (asymmetric phase)
    for freq in [1, 2, 3, 5]:
        for amp in [0.3, 0.5, 0.7]:
            for phase in [0, np.pi/3, np.pi/2, 2*np.pi/3]:
                seq = np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t + phase), 0.01)
                inits.append(seq / seq.sum())

    # Random smooth
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        raw = rng.uniform(0, 1, n)
        smooth = np.convolve(raw, kernel, mode='same')
        smooth = np.maximum(smooth, 0.01)
        # Add asymmetry
        asym_factor = 1.0 + 0.3 * rng.uniform(-1, 1) * t
        smooth = np.maximum(smooth * asym_factor, 0.01)
        inits.append(smooth / smooth.sum())

    # Sawtooth
    saw = (t % 0.5) * 2
    inits.append(saw / saw.sum())
    saw2 = 1.0 - (t % 0.5) * 2
    inits.append(saw2 / saw2.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining analytical CD, subgradient mirror descent,
    and simulated annealing with asymmetric initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        tracker.update(parent)

    base_n = len(parent)

    # === PHASE 1: Analytical CD on parent length ===
    p1_deadline = min(time.monotonic() + 400, deadline - 600)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Analytical CD, n={base_n}, budget={p1_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    inits = generate_asymmetric_inits(base_n, rng, parent)
    # Shuffle and take a good subset
    rng.shuffle(inits)
    n_inits = min(12, len(inits))

    for idx in range(n_inits):
        if time.monotonic() > p1_deadline - 30:
            break

        a0 = np.maximum(inits[idx], 0)
        s0 = a0.sum()
        if s0 < 1e-10:
            continue
        a0 = a0 / s0

        t0 = time.monotonic()
        # Split remaining time among remaining inits
        remaining_inits = n_inits - idx
        per_init_budget = (p1_deadline - time.monotonic()) / max(1, remaining_inits) - 5
        a_opt = analytical_cd(a0, max_sweeps=20000,
                              deadline=time.monotonic() + max(30, per_init_budget),
                              seed=seed + idx, top_k=30)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_opt)
        print(f"  Init {idx}: J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
        tracker.update(a_opt)

    tracker.report("Phase 1")

    # === PHASE 2: Subgradient mirror descent on best ===
    p2_deadline = min(time.monotonic() + 250, deadline - 350)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Subgradient Mirror Descent, budget={p2_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a_sg = subgradient_mirror_descent(tracker.best, max_iters=5000,
                                          deadline=p2_deadline, lr_scale=0.5)
        J_val = compute_J(a_sg)
        print(f"  SG-MD: J={J_val:.8f}", flush=True)
        tracker.update(a_sg)

        # Second pass with different lr
        if time.monotonic() < p2_deadline - 60:
            a_sg2 = subgradient_mirror_descent(tracker.best, max_iters=3000,
                                               deadline=p2_deadline, lr_scale=1.5)
            J_val2 = compute_J(a_sg2)
            print(f"  SG-MD2: J={J_val2:.8f}", flush=True)
            tracker.update(a_sg2)

    tracker.report("Phase 2")

    # === PHASE 3: Simulated Annealing ===
    p3_deadline = min(time.monotonic() + 200, deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Simulated Annealing, budget={p3_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a_sa = simulated_annealing(tracker.best, deadline=p3_deadline,
                                   seed=seed + 500, T0=0.01, cooldown=0.997)
        J_val = compute_J(a_sa)
        print(f"  SA: J={J_val:.8f}", flush=True)
        tracker.update(a_sa)

    tracker.report("Phase 3")

    # === PHASE 4: CD on SA result + longer lengths ===
    p4_deadline = min(time.monotonic() + 350, deadline - 100)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: CD refinement + longer lengths, budget={p4_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    # CD on current best
    if tracker.best is not None and time.monotonic() < p4_deadline - 60:
        a_cd = analytical_cd(tracker.best, max_sweeps=20000,
                             deadline=min(time.monotonic() + 150, p4_deadline - 100),
                             seed=seed + 900, top_k=40)
        J_val = compute_J(a_cd)
        print(f"  CD refine: J={J_val:.8f}", flush=True)
        tracker.update(a_cd)

    # Try longer lengths
    best_n = tracker.best.size if tracker.best is not None else base_n
    for new_n in [int(best_n * 1.5), int(best_n * 2)]:
        if new_n > 3000 or time.monotonic() > p4_deadline - 30:
            break
        if new_n <= best_n:
            continue

        a_ext = resample_seq(tracker.best, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        a_opt = analytical_cd(a_ext, max_sweeps=10000,
                              deadline=time.monotonic() + min(100, p4_deadline - time.monotonic() - 20),
                              seed=seed + 1000 + new_n, top_k=40)
        J_val = compute_J(a_opt)
        print(f"  n={new_n}: J={J_val:.8f}", flush=True)
        tracker.update(a_opt)

    tracker.report("Phase 4")

    # === PHASE 5: Final polish ===
    p5_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish, budget={p5_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p5_deadline - 10:
        # Final CD
        a_final = analytical_cd(tracker.best, max_sweeps=20000,
                                deadline=p5_deadline - 10,
                                seed=seed + 2000, top_k=50)
        J_val = compute_J(a_final)
        print(f"  Final CD: J={J_val:.8f}", flush=True)
        tracker.update(a_final)

        # Quick SA polish
        if time.monotonic() < p5_deadline - 30:
            a_sa2 = simulated_annealing(tracker.best,
                                        deadline=min(time.monotonic() + 60, p5_deadline - 15),
                                        seed=seed + 3000, T0=0.005, cooldown=0.995)
            J_val2 = compute_J(a_sa2)
            print(f"  Final SA: J={J_val2:.8f}", flush=True)
            tracker.update(a_sa2)

            # CD after SA
            if time.monotonic() < p5_deadline - 15:
                a_cd2 = analytical_cd(tracker.best, max_sweeps=10000,
                                      deadline=p5_deadline - 5,
                                      seed=seed + 4000, top_k=50)
                J_val3 = compute_J(a_cd2)
                print(f"  Final CD2: J={J_val3:.8f}", flush=True)
                tracker.update(a_cd2)

    tracker.report("Phase 5")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
