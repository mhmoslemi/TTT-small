"""
Combined LP-alternating + analytical subgradient + peak equalization
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous:
1. Analytical subgradient for principled non-LP updates
2. Peak equalization when multiple peaks are close
3. Better line search (backtracking)
4. Dual-guided perturbations
5. Multi-length exploration with adaptive time allocation
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

def solve_lp(a, M, relax=1.0):
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    A = build_toeplitz(a)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            duals = res.ineqlin.marginals if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals') else None
            return g, s, duals
    except Exception:
        pass
    return None, 0.0, None

def subgradient_step(a, alpha, num_nearest=3):
    """Analytical subgradient step targeting the highest peaks."""
    n = len(a)
    conv = compute_conv(a)
    max_val = np.max(conv)
    # Get positions of peaks (sorted by value, take top few)
    threshold = max_val - 1e-6 * max_val
    peak_pos = np.where(conv >= threshold)[0]
    if len(peak_pos) == 0:
        peak_pos = np.argmax(conv)
        peak_pos = np.array([peak_pos])
    # Take only the top num_nearest peaks to stay focused
    if len(peak_pos) > num_nearest:
        top_vals = conv[peak_pos]
        order = np.argsort(top_vals)[::-1][:num_nearest]
        peak_pos = peak_pos[order]
    
    grad = np.zeros(n)
    for k in peak_pos:
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        # grad[i] += 2 * a[k-i] for i in [i_min, i_max]
        # a[k-i] for i in [i_min, i_max] is a[k-i_max : k-i_min+1] reversed
        seg = a[k-i_min:k-i_max+1][::-1]  # a[k-i_min], a[k-i_min-1], ..., a[k-i_max]
        # Wait, let me redo this.
        # For i in [i_min, i_max], we want a[k-i].
        # When i = i_min: a[k-i_min]
        # When i = i_max: a[k-i_max]
        # So the values are a[k-i_min], a[k-i_min-1], ..., a[k-i_max]
        # which is a[k-i_min : k-i_max-1 : -1] if k-i_min > k-i_max
        # i.e., a[k-i_min : k-i_max-1 : -1]
        # Or equivalently: reversed(a[k-i_max : k-i_min+1])
        vals = a[k-i_max : k-i_min+1][::-1]
        grad[i_min:i_max+1] += 2.0 * vals
    
    a_new = a - alpha * grad
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    a_new = a_new / s
    return a_new

def peak_equalization(a, num_peaks=5, strength=0.1):
    """Try to equalize the top peaks by reducing contributions to the highest peak."""
    n = len(a)
    conv = compute_conv(a)
    sorted_idx = np.argsort(conv)[::-1]
    top = sorted_idx[:num_peaks]
    max_val = conv[top[0]]
    
    # For the highest peak at position k, reduce a at positions that contribute most
    k = top[0]
    i_min = max(0, k - n + 1)
    i_max = min(k, n - 1)
    
    a_new = a.copy()
    # Reduce a_i for i in [i_min, i_max] proportionally to a_i * a_{k-i}
    contributions = np.zeros(n)
    for i in range(i_min, i_max + 1):
        contributions[i] = a[i] * a[k-i]
    
    total = np.sum(contributions)
    if total > 1e-15:
        # Reduce proportionally
        reduction = strength * contributions / total
        a_new[i_min:i_max+1] -= reduction[i_min:i_max+1] * max_val
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / a.sum()  # Keep same sum
    return a_new

def lp_with_subgradient(a, time_budget=100.0, verbose=False):
    """Combined LP alternating and subgradient optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 50
    alpha = 0.01  # Subgradient step size
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        conv = compute_conv(best_a)
        M = float(np.max(conv))
        
        # Try LP first
        # Adaptive relaxation
        relax = 1.0 + 0.02 * min(stall // 20, 5)
        g, S, duals = solve_lp(best_a, M, relax=relax)
        
        improved = False
        if g is not None and S > 1.0 + 1e-6:
            # Line search with backtracking
            g_norm = g / S
            for t in [0.3, 0.5, 0.7, 0.85, 1.0, 0.15, 0.2]:
                a_try = (1-t) * best_a + t * g_norm
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break
        
        if not improved:
            # Try subgradient step
            a_sg = subgradient_step(best_a, alpha)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj - 1e-14:
                best_obj = obj_sg
                best_a = a_sg.copy()
                improved = True
                alpha = min(alpha * 1.2, 0.1)
            else:
                # Decay alpha
                alpha = max(alpha * 0.9, 0.001)
        
        if not improved:
            stall += 1
            if stall > patience:
                # Perturb to escape
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                scale = rng.uniform(0.02, 0.06)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s_pert = np.sum(a_pert)
                if s_pert > 1e-10:
                    a_pert = a_pert / s_pert
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
                stall = 0
                patience = max(20, patience // 2)
        else:
            stall = 0
        
        if verbose and int(time.monotonic() - start) % 10 == 0:
            print(f"    obj={best_obj:.6f}, stall={stall}, relax={relax:.2f}")
    
    return best_a, best_obj

def random_init(n, seed):

    elif choice == 2:

    elif choice == 3:
        # Block structure
        nblocks = rng.integers(3, 8)
        a = np.ones(n)
        for b in range(nblocks):
            s = b*n//nblocks
            e = (b+1)*n//nblocks
            a[s:e] = rng.uniform(0.3, 1.5)
    elif choice == 4:
        # Gaussian window
        center = n*rng.uniform(0.3, 0.7)
        width = n*rng.uniform(0.1, 0.4)
        a = 1.0 + 0.5*np.exp(-0.5*((k-center)/width)**2)
    else:
        a = rng.random(n) + 0.1

    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)/n
    else:
        a = a / s
    return a

def spectral_flatten(a, iters=20, time_budget=10.0):

        if time.monotonic() - start > time_budget:
            break

        choice = rng.integers(0, 3)
        if choice == 0:

            thresh = np.percentile(mag[mag > 0], 90)
