def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0 : sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # Standard simplex projection
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    rho = np.nonzero(v * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        contributions = np.zeros(n)
        for k in peak_bins:
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                contributions[j] += a[j] * a[k - j]

        # Sort coordinates by contribution (most contributing first)
        order = np.argsort(-contributions)

        improved_this_round = False
        for i in order[:20]:  # Only try top 20 contributing coordinates
            if time.monotonic() > deadline:
                break
            if a[i] < 1e-8:
                continue

            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            # Try reducing i and see if we can compensate by increasing others
            best_val = old
            best_obj_local = 2.0 * n * conv.max()

            # Try several reduction levels
            for frac in [0.3, 0.5, 0.7, 0.9]:
                x = old * (1.0 - frac)
                delta = x - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                obj_candidate = 2.0 * n * conv.max()
                if obj_candidate < best_obj_local:
                    best_obj_local = obj_candidate
                    best_val = x
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if best_val < old - 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                # Redistribute the freed mass to low-contribution coordinates
                freed = old - best_val
                low_contrib = np.argsort(contributions)[:10]
                low_contrib = [j for j in low_contrib if j != i and a[j] < 0.05]
                if low_contrib:
                    per = freed / len(low_contrib)
                    for j in low_contrib:
                        old_j = a[j]
                        new_j = old_j + per
                        lo_j = max(0, j)
                        hi_j = min(2 * n - 2, j + n - 1)
                        a_slice_j = a[lo_j - j:hi_j - j + 1].copy()
                        a_slice_j[j - (lo_j - j)] = new_j  # Update the self-reference
                        kq_j = 2 * j
                        valid_kq_j = kq_j <= 2 * n - 2
                        d = new_j - old_j
                        conv[lo_j:hi_j + 1] += 2.0 * d * a_slice_j
                        if valid_kq_j:
                            conv[kq_j] += d * d
                        a[j] = new_j
                improved_this_round = True

        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

    a = np.maximum(best_a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def block_coord_descent(a, deadline, seed=0, block_size=3):
    """
    Block coordinate descent: update groups of adjacent coordinates together.
    Uses a small grid to search the block jointly.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() < deadline:
        improved = False
        # Process blocks
        for start in range(0, n - block_size + 1, block_size):
            if time.monotonic() > deadline:
                break
            block = a[start:start + block_size].copy()
            if block.sum() < 1e-10:
                continue

            # Try redistributing mass within the block
            # Use a few random redistribution patterns
            for _ in range(5):
                new_block = block * rng.uniform(0.5, 1.5, block_size)
                # Preserve the sum of the block
                if new_block.sum() > 1e-10:
                    new_block = new_block * (block.sum() / new_block.sum())
                else:
                    continue

                delta = new_block - block
                if np.max(np.abs(delta)) < 1e-12:
                    continue

                # Apply
                for idx in range(block_size):
                    i = start + idx
                    d = delta[idx]
                    if abs(d) < 1e-12:
                        continue
                    lo = max(0, i)
                    hi = min(2 * n - 2, i + n - 1)
                    a_slice = a[lo - i:hi - i + 1].copy()
                    # Account for other block changes already applied
                    a_slice[idx] = a[i] + d  # tentative value at self-position
                    kq = 2 * i
                    conv[lo:hi + 1] += 2.0 * d * a_slice
                    if kq <= 2 * n - 2:
                        conv[kq] += d * d
                    a[i] = a[i] + d

                new_obj = 2.0 * n * conv.max()
                if new_obj < best_obj - 1e-10:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                else:
                    # Rollback
                    for idx in range(block_size):
                        i = start + idx
                        d = -delta[idx]
                        if abs(d) < 1e-12:
                            continue
                        lo = max(0, i)
                        hi = min(2 * n - 2, i + n - 1)
                        a_slice = a[lo - i:hi - i + 1].copy()
                        a_slice[idx] = a[i] + d
                        kq = 2 * i
                        conv[lo:hi + 1] += 2.0 * d * a_slice
                        if kq <= 2 * n - 2:
                            conv[kq] += d * d
                        a[i] = a[i] + d

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

    a = np.maximum(best_a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a, max_iters, deadline):
    """LP mirror descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = fast_conv(a).max()
        t0 = time.monotonic()
        g = lp_best_response(a, M)
        lp_time = time.monotonic() - t0

        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        improved = False
        base_t = max(0.03, 0.3 * (1.0 - 0.3 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        remaining = deadline - time.monotonic()
        if remaining < lp_time * 3:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic-like interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smoothed_gradient_step(a, t_smooth=50.0):
    """
    Compute gradient of log-sum-exp smoothed max of convolution.
    Returns (gradient, smoothed_value).
    """
    n = len(a)
    conv = fast_conv(a)
    # Numerical stability
    conv_max = conv.max()
    exp_conv = np.exp(t_smooth * (conv - conv_max))
    weights = exp_conv / exp_conv.sum()
    smoothed_val = conv_max + np.log(exp_conv.sum()) / t_smooth

    # Gradient: grad[i] = sum_k weights[k] * 2 * a[k-i] * 1_{0<=k-i<n}
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * weights[k] * a[k - hi:k - lo + 1][::-1]

    return grad, smoothed_val


def smoothed_gradient_descent(a, deadline, t_init=20.0, t_final=200.0, max_iters=5000):
    """
    Gradient descent on the smoothed objective with adaptive temperature.
    Uses backtracking line search and simplex projection.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()
    best_M = fast_conv(a).max()
    lr = 0.05
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive temperature
        progress = min(1.0, it / max(1, max_iters * 0.8))
        t_cur = t_init * (t_final / t_init) ** progress

        grad, sm_val = smoothed_gradient_step(a, t_cur)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        # Normalize gradient
        grad_normed = grad / gnorm

        # Backtracking line search
        improved = False
        for step_mult in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            a_trial = a - step_mult * gnorm * grad_normed
            a_trial = project_simplex(a_trial, 1.0)
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
            lr = min(lr * 1.05, 0.2)
        else:
            no_improve += 1
            lr *= 0.7
            if no_improve > 50:
                break

    return best_a


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 0)
        if p.sum() > 0:
            p = p / p.sum()
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.001))
        inits.append(np.maximum(p * rng.uniform(0.5, 1.5, n), 0.001))

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-concentrated (should help reduce center peak)
    edge = np.maximum(np.exp(-5 * t) + np.exp(-5 * (1 - t)), 0.01)
    inits.append(edge / edge.sum())

    # Bimodal
    bimodal = np.maximum(np.exp(-((t - 0.25) / 0.15) ** 2) + np.exp(-((t - 0.75) / 0.15) ** 2), 0.01)
    inits.append(bimodal / bimodal.sum())

    # Smooth random
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.01, 0.001))

    # Cosine
    for freq in [2, 4, 8]:
        c = np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.05)
        inits.append(c / c.sum())

    # Normalize all
    result = []
    for a in inits:
        a = np.maximum(a, 0)
        if a.sum() > 1e-10:
            a = a / a.sum()
            result.append(a)
    return result


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining refined coordinate descent, peak targeting,
    smoothed gradient descent, block moves, LP refinement, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512
        tracker.update(parent)
        print(f"No parent, using uniform n=512, J={tracker.best_score:.8f}", flush=True)

    current_best = tracker.best.copy()

    # === PHASE 1: Refined coordinate descent on parent length ===
    n_base = len(current_best)
    p1_deadline = min(time.monotonic() + 350, deadline - 600)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Refined CD n={n_base} (budget {p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    inits = generate_diverse_inits(n_base, current_best, rng)
    for idx, a0 in enumerate(inits[:6]):
        if time.monotonic() > p1_deadline:
            break
        t0 = time.monotonic()
        a_opt = coord_descent_refined(a0, deadline=p1_deadline, seed=seed + idx,
                                       n_candidates=13, refine=True)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_opt)
        print(f"  Init {idx}: J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
        tracker.update(a_opt)

    tracker.report("Phase 1")

    # === PHASE 2: Peak targeting ===
    if time.monotonic() < deadline - 500:
        p2_deadline = min(time.monotonic() + 150, deadline - 450)
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 2: Peak Targeting (budget {p2_deadline - time.monotonic():.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        a_pt = peak_targeting_search(tracker.best, deadline=p2_deadline, seed=seed + 100)
        J_val = compute_J(a_pt)
        print(f"  Peak targeting: J={J_val:.8f}", flush=True)
        tracker.update(a_pt)

        # Follow up with CD
        if time.monotonic() < p2_deadline:
            a_cd = coord_descent_refined(tracker.best, deadline=p2_deadline, seed=seed + 200,
                                         n_candidates=13, refine=True)
            J_val = compute_J(a_cd)
            print(f"  Post-PT CD: J={J_val:.8f}", flush=True)
            tracker.update(a_cd)

        tracker.report("Phase 2")

    # === PHASE 3: Smoothed gradient descent ===
    if time.monotonic() < deadline - 400:
        p3_deadline = min(time.monotonic() + 150, deadline - 350)
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 3: Smoothed Gradient Descent (budget {p3_deadline - time.monotonic():.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        a_sg = smoothed_gradient_descent(tracker.best, deadline=p3_deadline,
                                          t_init=10.0, t_final=100.0, max_iters=3000)
        J_val = compute_J(a_sg)
        print(f"  Smoothed GD: J={J_val:.8f}", flush=True)
        tracker.update(a_sg)

        # Polish with CD
        if time.monotonic() < p3_deadline - 10:
            a_cd = coord_descent_refined(tracker.best, deadline=p3_deadline, seed=seed + 300,
                                         n_candidates=15, refine=True)
            J_val = compute_J(a_cd)
            print(f"  Post-SGD CD: J={J_val:.8f}", flush=True)
            tracker.update(a_cd)

        tracker.report("Phase 3")

    # === PHASE 4: Block coordinate descent ===
    if time.monotonic() < deadline - 350:
        p4_deadline = min(time.monotonic() + 100, deadline - 300)
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 4: Block CD (budget {p4_deadline - time.monotonic():.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        a_blk = block_coord_descent(tracker.best, deadline=p4_deadline, seed=seed + 400, block_size=3)
        J_val = compute_J(a_blk)
        print(f"  Block CD: J={J_val:.8f}", flush=True)
        tracker.update(a_blk)

        # Quick polish
        if time.monotonic() < p4_deadline:
            a_cd = coord_descent_refined(tracker.best, deadline=p4_deadline, seed=seed + 500,
                                         n_candidates=10, refine=False)
            J_val = compute_J(a_cd)
            print(f"  Post-block CD: J={J_val:.8f}", flush=True)
            tracker.update(a_cd)

        tracker.report("Phase 4")

    # === PHASE 5: Multi-n exploration ===
    if time.monotonic() < deadline - 200:
        p5_deadline = deadline - 150
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 5: Multi-n Exploration (budget {p5_deadline - time.monotonic():.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        base_n = tracker.best.size
        target_ns = []
        for target in [480, 512, 544, 576, 640, 768, 1024]:
            if target != base_n and target <= 1024:
                target_ns.append(target)

        # Sort by distance from base_n (try closest first)
        target_ns.sort(key=lambda x: abs(x - base_n))

        for new_n in target_ns[:5]:  # Try up to 5 different lengths
            if time.monotonic() > p5_deadline - 30:
                break

            t0 = time.monotonic()
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            # Quick CD
            remaining = p5_deadline - time.monotonic() - 10
            alloc = min(remaining / max(1, len(target_ns) - target_ns.index(new_n)), 120)
            a_opt = coord_descent_refined(a_ext, deadline=time.monotonic() + alloc,
                                           seed=seed + 600 + new_n, n_candidates=10, refine=True)
            elapsed = time.monotonic() - t0
            J_val = compute_J(a_opt)
            print(f"  n={new_n}: J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
            tracker.update(a_opt)

        tracker.report("Phase 5")

    # === PHASE 6: LP refinement ===
    if time.monotonic() < deadline - 30:
        p6_deadline = deadline - 10
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 6: LP Refinement (budget {p6_deadline - time.monotonic():.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        n_best = tracker.best.size
        # Estimate LP time
        est_lp_time = 0.005 * (n_best / 128) ** 2
        max_iters = max(5, min(50, int((p6_deadline - time.monotonic()) / (2 * max(0.01, est_lp_time)))))
        print(f"  n={n_best}, est LP time={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)

        a_lp = run_lp_mirror(tracker.best, max_iters, p6_deadline)
        J_val = compute_J(a_lp)
        print(f"  LP: J={J_val:.8f}", flush=True)
        tracker.update(a_lp)

        # Final quick CD polish
        if time.monotonic() < p6_deadline - 5:
            a_final = coord_descent_refined(tracker.best, deadline=p6_deadline, seed=seed + 999,
                                            n_candidates=10, refine=True)
            J_val = compute_J(a_final)
            print(f"  Final CD: J={J_val:.8f}", flush=True)
            tracker.update(a_final)

        tracker.report("Phase 6")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
