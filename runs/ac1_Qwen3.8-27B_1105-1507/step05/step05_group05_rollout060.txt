"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
Phase 1: KL Mirror Descent with FFT-accelerated gradient on log-sum-exp smoothed objective
Phase 2: Fast coordinate descent with vectorized 1D subproblem solutions
Phase 3: LP mirror descent for final polish
Key innovations: FFT-based gradient (O(n log n) per iteration), vectorized coordinate updates,
adaptive tau scheduling, and multi-length exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_conv(conv, tau):
    """Compute softmax of conv with temperature tau."""
    c_max = conv.max()
    exp_vals = np.exp(tau * (conv - c_max))
    return exp_vals / exp_vals.sum()


def compute_gradient_fft(a, p, tau):
    """Compute gradient of smooth max via FFT cross-correlation."""
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    h = np.fft.irfft(P * np.conj(A), size)[:n]
    return (2.0 / tau) * h


def project_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def mirror_descent_smooth(a, tau, max_iters, eta0, deadline):
    """KL mirror descent on log-sum-exp smoothed objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv = fast_conv(a)
        p = softmax_conv(conv, tau)
        g = compute_gradient_fft(a, p, tau)
        
        eta = eta0 / np.sqrt(it + 1)
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
    
    return best_a

            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                if k_lo > 0 and k_hi < 2 * n - 2:
                    base_max = max(conv[:k_lo].max(), conv[k_hi+1:].max())
                elif k_lo > 0:
                    base_max = conv[:k_lo].max()
                else:
                    base_max = conv[k_hi+1:].max()
            else:
                base_max = 0.0
            
            # Affected positions
            is_quad_k = 2 * i
            affected_range = np.arange(k_lo, k_hi + 1)
            lin_mask = (affected_range != is_quad_k)
            
            if lin_mask.any():
                lin_k = affected_range[lin_mask]
                lin_slope = 2.0 * a[lin_k - i]
                lin_base = conv[lin_k]
            else:
                lin_k = np.array([], dtype=np.int64)
                lin_slope = np.array([])
                lin_base = np.array([])
            
            has_quad = (is_quad_k >= k_lo) and (is_quad_k <= k_hi)
            q_base_val = conv[is_quad_k] if has_quad else None
            
            S = sum_a
            
            # Golden section search
            lo_b = 0.0
            hi_b = max(3.0 * x + 0.01, 2.0 * S / n)
            
            gr = (np.sqrt(5) - 1) / 2
            ga = hi_b - gr * (hi_b - lo_b)
            gb = lo_b + gr * (hi_b - lo_b)
            
            def evaluate_y(y_val):
                ns = S - x + y_val
                if ns < 1e-10:
                    return 1e10
                m = base_max
                if len(lin_k) > 0:
                    lv = np.max(lin_base + lin_slope * y_val)
                    if lv > m:
                        m = lv
                if has_quad:
                    qv = q_base_val + y_val * y_val - x * x
                    if qv > m:
                        m = qv
                return m / (ns * ns)
