import numpy as np
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def perfect_truncated_seq(n):
    """Generate the 'perfect truncated' sequence."""
    k = np.arange(n)
    # binom(2k, k) / 4^k computed stably
    # Use log-gamma for stability
    from scipy.special import gammaln
    log_a = gammaln(2*k+1) - 2*gammaln(k+1) - k*np.log(4)
    a = np.exp(log_a)
    a /= a.sum()
    return a

def compute_J(a):
    """Compute J(a)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa*fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2*len(a)*b.max()/(s*s))

def perfect_truncated_seq(n):
    k = np.arange(n, dtype=np.float64)
    log_a = gammaln(2*k+1) - 2*gammaln(k+1) - k*np.log(4.0)
    a = np.exp(log_a)
    a /= a.sum()
    return a

def subgradient_of_max_conv(a, tol_frac=1e-6):
    """Compute subgradient of max_k conv(a,a)[k]."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = tol_frac * M
    active = b >= M - tol

def proj_simplex(v):
    """Project v onto the simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cssv - 1) / rho > 0
    rho_hat = np.max(rho[cond])
    theta = (cssv[rho_hat-1] - 1) / rho_hat
    return np.maximum(v - theta, 0)

    """FISTA-style accelerated subgradient descent."""
    n = len(a0)
    x = a0.copy()

    x = x / x.sum()
    x_prev = x.copy()
    t = 1.0
    best_x = x.copy()
    best_M = fast_conv(x).max()
    
    # Running average for better convergence
    avg = np.zeros(n)
    count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # FISTA extrapolation
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        y = x + ((t-1)/t_new) * (x - x_prev)
        y = np.maximum(y, 0)
        
        # Compute subgradient at y
        b = fast_conv(y)
        M = b.max()
        tol = 1e-6 * M

            hi = min(n-1, k)

        grad_norm = np.linalg.norm(grad)

        # Update

        # Actually compute at x_new for accuracy
        M_xnew = fast_conv(x_new).max()
        if M_xnew < best_M:
            best_M = M_xnew
            best_x = x_new.copy()
        
        avg += x_new

        x_prev = x
        x = x_new
