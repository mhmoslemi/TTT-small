So for symmetric a, the peak is at k = n-1 and equals sum a[j]^2. To minimize this, we want a to be as uniform as possible (which gives sum a[j]^2 = 1/n for normalized a). But then the edge values (a*a)[0] = a[0]^2 must also be small, which conflicts with uniformity at the edges.

So symmetry might not be optimal, but it's a good starting constraint.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length exploration with diverse initializations
2. Smooth gradient descent with adaptive tau and proper simplex projection
3. LP mirror descent with analytic line search
4. Chunk-flip mutations to escape local minima
5. Final extended LP polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v, target_sum=1.0):
    """Project v onto {x : sum(x) = target_sum, x >= 0}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # We want to find tau such that sum(max(v - tau, 0)) = target_sum
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - target_sum) / rho > 0
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho_max = np.max(rho[cond])
    tau = (cssv[rho_max - 1] - target_sum) / rho_max
    return np.maximum(v - tau, 0)


def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 15, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = project_simplex(a_new)
            a = a_new
            actual_M = compute_M(a)
            if actual_M < best_M:
                best_M = actual_M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 150:
            break

        if verbose and it % 200 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def smooth_gradient_step(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Clip to avoid overflow
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad


def run_smooth_gradient(a, max_iters, deadline, initial_tau=3.0, final_tau=200.0, 
                        lr_start=0.5, lr_end=0.01, momentum=0.9):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * frac
        lr = lr_start * (lr_end / lr_start) ** frac

        grad = smooth_gradient_step(a, tau)
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = project_simplex(a_new)
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a


def chunk_flip(a, rng, flip_count=3):
    """Reverse random subarrays of a. Preserves sum and non-negativity."""
    a = a.copy()
    n = len(a)
    for _ in range(flip_count):
        i = rng.integers(0, n - 1)
        j = rng.integers(i + 1, n)
        a[i:j+1] = a[i:j+1][::-1]
    return a


def apply_noise(a, rng, scale):
    """Add Gaussian noise and project back to simplex."""
    a = a + rng.normal(0, scale, len(a))
    return project_simplex(a)


def make_init(n, kind, rng):
    """Create initial sequence of length n."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'raised_cosine':
        j = np.arange(n)
        x = 2 * j / (n - 1) - 1 if n > 1 else 0
        a = (1 - x * x) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'hann':
        j = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1))) if n > 1 else np.array([1.0])
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'gaussian':
        j = np.arange(n)
        center = (n - 1) / 2
        sigma = n / 4
        a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'random':
        a = rng.random(n) + 0.1
        return a / a.sum()
    elif kind == 'flat_center':
        # Flat in the middle, tapered at edges
        j = np.arange(n)
        x = np.abs(2 * j / (n - 1) - 1) if n > 1 else 0
        a = np.where(x < 0.5, 1.0, 1.0 - (x - 0.5) * 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Proper simplex projection (Euclidean, not clip+normalize)
2. Nesterov-momentum smooth gradient with adaptive tau
3. LP mirror descent with analytic line search
4. Multi-length exploration with diverse parametric initializations
5. Chunk-flip and swap mutations to escape local minima
6. Direct subgradient steps for coarse adjustment
7. Final extended LP polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x: sum(x)=target_sum, x>=0}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - target_sum) / rho > 0
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho_max = np.max(rho[cond])
    tau = (cssv[rho_max - 1] - target_sum) / rho_max
    return np.maximum(v - tau, 0.0)


def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(a, M, row_idx, col_idx):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 15, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        t, M_new = line_search_analytic(a, g)
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = project_simplex(a_new)
            a = a_new
            actual_M = compute_M(a)
            if actual_M < best_M:
                best_M = actual_M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 200:
            break
        if verbose and it % 200 == 0:
            print(f"      iter {it}: M={M:.12f} best_M={best_M:.12f} t={t:.6f}", flush=True)
    return best_a


def smooth_grad(a, tau):
    """Compute gradient of soft-max approximation of max(conv(a,a))."""
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad


def subgrad_step(a, scale=0.01):
    """Direct subgradient step: move away from the peak position."""
    b = fast_conv(a)
    k_star = int(np.argmax(b))
    n = len(a)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    contrib = a[lo:hi + 1] * a[k_star - hi:k_star - lo + 1][::-1]
    total = contrib.sum()
    if total < 1e-15:
        return a
    # Reduce elements proportional to their contribution to the peak
    reduction = contrib / total * scale
    a_new = a.copy()
    a_new[lo:hi + 1] -= reduction * 0.5
    # The paired elements also contribute
    pair_idx = k_star - np.arange(lo, hi + 1)
    valid = (pair_idx >= 0) & (pair_idx < n)
    a_new[pair_idx[valid]] -= reduction[valid] * 0.5
    return project_simplex(a_new)


def run_smooth_opt(a, max_iters, deadline, tau_start=5.0, tau_end=300.0,
                   lr_start=0.15, lr_end=0.005, momentum=0.92):
    """Momentum smooth gradient descent with adaptive tau and learning rate."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = project_simplex(a)
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.6)
        lr = lr_start * (lr_end / lr_start) ** frac
        grad = smooth_grad(a, tau)
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = project_simplex(a_new)
        a = a_new
        M = compute_M(a)
        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()
    return best_a


def run_hybrid_opt(a, deadline, row_idx, col_idx, lp_time_frac=0.5):
    """Hybrid: smooth gradient then LP. Returns optimized sequence."""
    elapsed_total = deadline - time.monotonic()
    if elapsed_total < 5:
        return a
    smooth_dl = time.monotonic() + elapsed_total * (1.0 - lp_time_frac)
    a_sg = run_smooth_opt(a, 10000, smooth_dl)
    lp_dl = deadline
    a_lp = run_lp_mirror_descent(a_sg, 20000, lp_dl, row_idx, col_idx, verbose=False)
    return a_lp


def make_init(n, kind, rng):
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'raised_cosine':
        j = np.arange(n, dtype=np.float64)
        x = 2.0 * j / (n - 1) - 1.0 if n > 1 else np.array([0.0])
        a = (1.0 - x * x) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'hann':
        j = np.arange(n, dtype=np.float64)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * j / max(1, n - 1)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'gaussian':
        j = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        sigma = max(1.0, n / 4.0)
        a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'kaiser':
        j = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (j - center) / center if n > 1 else np.array([0.0])
        beta = 5.0
        if n > 1:
            from scipy.special import i0
            a = np.where(np.abs(x) <= 1, i0(beta * np.sqrt(1 - x**2)) / i0(beta), 0.0)
        else:
            a = np.array([1.0])
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'flat_taper':
        j = np.arange(n, dtype=np.float64)
        x = np.abs(2.0 * j / max(1, n - 1) - 1.0)
        taper = 0.3
        a = np.where(x < 1.0 - taper, 1.0,
                     0.5 * (1.0 + np.cos(np.pi * (x - (1.0 - taper)) / taper)))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == 'random':
        a = rng.random(n) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def chunk_flip(a, rng, n_flips):
    a = a.copy()
    n = len(a)
    for _ in range(n_flips):
        i = int(rng.integers(0, max(1, n - 1)))
        j = int(rng.integers(i + 1, n))
        a[i:j + 1] = a[i:j + 1][::-1]
    return a


def swap_pairs(a, rng, n_swaps):
    a = a.copy()
    n = len(a)
    for _ in range(n_swaps):
        i, j = rng.integers(0, n, size=2)
        if i != j:
            a[i], a[j] = a[j], a[i]
    return a


class Tracker:
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining smooth gradient, LP mirror descent,
    multi-length exploration, and mutation-based restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    # Phase 1: Optimize parent (first 25% of time)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Optimize parent n={parent.size}", flush=True)
    print(f"{'='*60}", flush=True)

    n0 = parent.size
    ri0, ci0 = build_lp_matrix(n0)
    p1_dl = min(t_start + budget_s * 0.25, deadline)

    # Smooth gradient
    a1 = run_smooth_opt(parent, 10000, min(t_start + budget_s * 0.12, p1_dl),
                        tau_start=3.0, tau_end=150.0)
    tracker.update(a1, "(P1 smooth)")

    # LP from smooth result
    a1 = run_lp_mirror_descent(a1, 30000, p1_dl, ri0, ci0, verbose=True)
    tracker.update(a1, "(P1 LP)")

    # Phase 2: Multi-length & multi-init exploration (next 35%)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    p2_start = max(time.monotonic(), t_start + budget_s * 0.25)
    p2_dl = min(t_start + budget_s * 0.60, deadline)

    if tracker.best is not None:
        base_n = tracker.best.size
        # Generate candidate lengths
        n_candidates = []
        seen = set()
        for target_n in [128, 160, 192, 224, 256, 288, 320, 384, 448, 512,
                         576, 640, 768, 896, 1024, 1280]:
            if 64 <= target_n <= 4096 and target_n not in seen:
                seen.add(target_n)
                n_candidates.append(target_n)
        # Also add rescaled versions of base_n
        for mult in [0.5, 0.75, 1.25, 1.5, 2.0]:
            nn = int(base_n * mult)
            nn = max(64, min(4096, (nn // 16) * 16))
            if nn not in seen:
                seen.add(nn)
                n_candidates.append(nn)
        n_candidates = sorted(n_candidates)

        total_p2_time = max(10.0, p2_dl - p2_start)
        per_n_time = total_p2_time / max(1, len(n_candidates))

        for n_trial in n_candidates:
            if time.monotonic() > p2_dl:
                break
            if n_trial == base_n:
                # Already optimized in Phase 1, skip or do quick check
                continue

            inits_to_try = ['raised_cosine', 'kaiser', 'flat_taper']
            per_init_time = per_n_time / len(inits_to_try)

            for ik in inits_to_try:
                if time.monotonic() > p2_dl:
                    break
                a_init = make_init(n_trial, ik, rng)
                ri, ci = build_lp_matrix(n_trial)
                sub_dl = min(time.monotonic() + per_init_time, p2_dl)
                if sub_dl - time.monotonic() < 3:
                    continue
                a_opt = run_hybrid_opt(a_init, sub_dl, ri, ci, lp_time_frac=0.55)
                tracker.update(a_opt, f"(P2 n={n_trial} {ik})")

            # Also try resampling the current best
            if tracker.best is not None and tracker.best.size != n_trial:
                a_init = resample_seq(tracker.best, n_trial)
                a_init = project_simplex(a_init)
                ri, ci = build_lp_matrix(n_trial)
                sub_dl = min(time.monotonic() + per_init_time * 0.6, p2_dl)
                if sub_dl - time.monotonic() > 3:
                    a_opt = run_lp_mirror_descent(a_init, 10000, sub_dl, ri, ci, verbose=False)
                    tracker.update(a_opt, f"(P2 n={n_trial} resamp)")

    # Phase 3: Mutation-based exploration (next 20%)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Mutation exploration", flush=True)
    print(f"{'='*60}", flush=True)

    p3_start = max(time.monotonic(), t_start + budget_s * 0.60)
    p3_dl = min(t_start + budget_s * 0.80, deadline)

    if tracker.best is not None:
        n_best = tracker.best.size
        ri_b, ci_b = build_lp_matrix(n_best)
        n_muts = 50
        per_mut_time = max(5.0, (p3_dl - p3_start) / n_muts)

        for r in range(n_muts):
            if time.monotonic() > p3_dl:
                break

            a = tracker.best.copy().astype(np.float64)
            a = project_simplex(a)

            mut_type = r % 4
            if mut_type == 0:
                a = chunk_flip(a, rng, int(rng.integers(1, 5)))
            elif mut_type == 1:
                a = swap_pairs(a, rng, int(rng.integers(5, 20)))
            elif mut_type == 2:
                noise = rng.normal(0, 0.015, n_best)
                a = project_simplex(a + noise)
            else:
                a = chunk_flip(a, rng, 2)
                noise = rng.normal(0, 0.008, n_best)
                a = project_simplex(a + noise)

            sub_dl = min(time.monotonic() + per_mut_time, p3_dl)
            if sub_dl - time.monotonic() < 3:
                continue

            # Quick smooth + LP
            a_opt = run_hybrid_opt(a, sub_dl, ri_b, ci_b, lp_time_frac=0.5)
            tracker.update(a_opt, f"(P3 r={r} t={mut_type})")

    # Phase 4: Try different n for the best shape (next 10%)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Best shape at other lengths", flush=True)
    print(f"{'='*60}", flush=True)

    p4_start = max(time.monotonic(), t_start + budget_s * 0.80)
    p4_dl = min(t_start + budget_s * 0.90, deadline)

    if tracker.best is not None:
        best_n = tracker.best.size
        # Try nearby lengths
        for delta in [-64, -32, 32, 64, 96, 128, -128]:
            n_try = best_n + delta
            if n_try < 64 or n_try > 4096:
                continue
            if time.monotonic() > p4_dl:
                break
            a_init = resample_seq(tracker.best, n_try)
            a_init = project_simplex(a_init)
            ri, ci = build_lp_matrix(n_try)
            sub_dl = min(time.monotonic() + 30, p4_dl)
            if sub_dl - time.monotonic() < 3:
                continue
            a_opt = run_lp_mirror_descent(a_init, 10000, sub_dl, ri, ci, verbose=False)
            tracker.update(a_opt, f"(P4 n={n_try})")

    # Phase 5: Final extended polish (last 10%)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        ri_f, ci_f = build_lp_matrix(n_final)
        a = tracker.best.copy()
        a = project_simplex(a)

        # Extended LP
        a_opt = run_lp_mirror_descent(a, 100000, deadline, ri_f, ci_f, verbose=True)
        tracker.update(a_opt, "(P5 LP)")

        # Smooth polish
        if time.monotonic() < deadline - 10:
            a_sg = run_smooth_opt(a_opt, 5000, deadline,
                                  tau_start=50.0, tau_end=500.0,
                                  lr_start=0.05, lr_end=0.002)
            tracker.update(a_sg, "(P5 smooth)")

            # One more LP
            if time.monotonic() < deadline - 5:
                a_final = run_lp_mirror_descent(a_sg, 20000, deadline, ri_f, ci_f)
                tracker.update(a_final, "(P5 final)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
