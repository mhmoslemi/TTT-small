"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase optimization:
1. Structured initialization with diverse profiles (linear, quadratic, etc.)
2. Active-set coordinate descent with golden section
3. Nesterov-accelerated smooth gradient descent (adaptive log-sum-exp)
4. LP-based best response mirror descent
5. Final coordinate polish

Key innovations over previous approach:
- Active set tracking reduces gradient computation cost
- Nesterov acceleration for faster convergence of smooth method
- Structured initializations based on theoretical insights (monotone profiles)
- Adaptive LP step size with backtracking
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J_normalized(a):
    """Compute J(a) assuming sum(a) = 1. J = 2*n*max(conv)."""
    n = len(a)
    if a.sum() < 1e-10:
        return float('inf')
    a = a / a.sum()
    b = fast_conv(a)
    return float(2 * n * b.max())


def compute_grad_smooth(a, tau):
    """
    Compute gradient of smooth max (log-sum-exp) of self-convolution w.r.t. a.
    Uses FFT for efficiency.
    grad_i = 2 * sum_j a[j] * p[i+j] where p_k = softmax(tau * conv_k)
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    log_p = tau * (b - b_max)
    # Numerical stability
    log_p -= log_p.max()
    p = np.exp(log_p)
    p /= p.sum()
    
    # grad_i = 2 * sum_j a[j] * p[i+j]
    # = 2 * (a * p_rev)[2n-2-i] where p_rev = p[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p[::-1], size)
    conv = np.fft.irfft(fa * fp, size)
    grad = 2.0 * conv[2 * n - 2 - np.arange(n)]
    return grad


def normalize_nonnegative(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), 1.0 / len(a))
    return a / s

def coordinate_1d(a, i, n_iter_golden=40):
    """
    Golden section search for optimal a[i] given all other coordinates fixed.
    Assumes sum(a) = 1 before and after.
    """
    n = len(a)
    x_old = a[i]
    S = a.sum()  # should be ~1
    
    # Precompute unaffected max
    b = fast_conv(a)
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[k_lo:k_hi + 1] = False
    base_max = b[mask].max() if mask.any() else 0.0
    
    # Precompute coefficients for affected positions
    affected = list(range(k_lo, k_hi + 1))
    c = np.zeros(len(affected))
    s_coef = np.zeros(len(affected))
    quad_idx = -1
    
    for idx, k in enumerate(affected):
        j = k - i
        if 0 <= j < n:
            if j == i:
                c[idx] = b[k] - x_old * x_old
                s_coef[idx] = 0.0
                quad_idx = idx
            else:
                c[idx] = b[k] - 2 * x_old * a[j]
                s_coef[idx] = 2 * a[j]
        else:
            c[idx] = b[k]
            s_coef[idx] = 0.0
    
    def obj(x):
        new_sum = S - x_old + x
        if new_sum <= 1e-15:
            return 1e10
        scale = 1.0 / new_sum
        m = base_max
        for idx in range(len(affected)):
            v = c[idx] + s_coef[idx] * x
            if v > m:
                m = v
        if quad_idx >= 0:
            v = c[quad_idx] + x * x
            if v > m:
                m = v
        return m * m * (scale * scale)  # Wait, this is wrong
    
    # Actually, let me reconsider. If a_new = a with a[i] = x, and we normalize to sum=1:
    # a_norm = a_new / sum(a_new)
    # conv(a_norm) = conv(a_new) / sum(a_new)^2
    # max(conv(a_norm)) = max(conv(a_new)) / sum(a_new)^2
    # So the objective to minimize is max(conv(a_new)) / sum(a_new)^2
    
    def obj2(x):
        new_sum = S - x_old + x
        if new_sum <= 1e-15:
            return 1e10
        # conv values for affected positions
        m = base_max
        for idx in range(len(affected)):
            v = c[idx] + s_coef[idx] * x
            if v > m:
                m = v
        if quad_idx >= 0:
            v = c[quad_idx] + x * x
            if v > m:
                m = v
        return m / (new_sum * new_sum)
    
    # Golden section
    lo = 0.0
    hi = max(x_old * 3.0, 0.5, np.sqrt(max(b.max(), 1e-10)) * 3)
    hi = min(hi, 2.0)  # Cap since we're normalized
    
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj2(a_pt)
    fb = obj2(b_pt)
    
    for _ in range(n_iter_golden):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj2(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj2(b_pt)
    
    x_opt = (lo + hi) / 2.0
    # Check boundary
    if obj2(0.0) < obj2(x_opt):
        x_opt = 0.0
    return x_opt


def run_coord_descent(a, max_sweeps, deadline, seed=0, shuffle=True):
    """Coordinate descent with shuffling."""
    rng = np.random.default_rng(seed)
    a = normalize_nonnegative(a.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        if shuffle:
            rng.shuffle(order)
        
        for i in order:
            x_opt = coordinate_1d(a, i, 35)
            if abs(x_opt - a[i]) > 1e-12:
                a[i] = x_opt
        
        a = normalize_nonnegative(a)
        M = fast_conv(a).max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20:
            break
    
    return best_a


def nesterov_smooth_descent(a, max_iters, deadline, tau_start=10.0, tau_end=200.0, lr=0.1):
    """
    Nesterov-accelerated gradient descent on smooth max of self-convolution.
    """
    a = normalize_nonnegative(a.copy())
    n = len(a)
    y = a.copy()  # Nesterov momentum variable
    v = np.zeros(n)  # velocity
    beta = 0.9  # momentum
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        lr_t = lr / (1 + 0.05 * it)
        
        # Nesterov: evaluate gradient at y = a + beta * v
        y = a + beta * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = a.copy()
        else:
            y = y / s_y
        
        grad = compute_grad_smooth(y, tau)
        
        # Project gradient onto tangent space (sum = 0)
        grad -= grad.mean()
        
        # Update
        v = beta * v - lr_t * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s
        a = a_new
        beta = (1 + beta + np.sqrt(1 + 4 * beta * beta)) / 2
        
        M = fast_conv(a).max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
    
    return best_a


def build_lp_matrix(n):
    """Build the constraint matrix for the LP: (a*g)[k] <= M."""
    # For the LP max sum(g) s.t. (a*g)[k] <= M, g >= 0
    # (a*g)[k] = sum_j a[j] * g[k-j] for valid j
    # This is a sparse matrix A of size (2n-1) x n
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a_global[k - j])  # Will be set properly
    return rows, cols, vals


def lp_best_response(a, M, time_limit=10):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns g or None.
    """
    n = len(a)
    A_rows = []
    A_cols = []
    A_vals = []
    
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A_rows.append(k)
            A_cols.append(j)
            A_vals.append(a[k - j])
    
    A = csr_matrix((A_vals, (A_rows, A_cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'max_iter': 100000})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_iterations(a, max_iters, deadline, time_per_lp=8):
    """Iterative LP-based mirror descent."""
    a = normalize_nonnegative(a.copy())
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M, time_limit=time_per_lp)
        if g is None:
            no_improve += 1
            if no_improve > 3:
                break
            continue
        
        g = normalize_nonnegative(g)
        
        # Backtracking line search
        improved = False
        base_t = max(0.1, 0.5 * (1 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02, 0.01]:
            a_trial = normalize_nonnegative((1 - t) * a + t * g)
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
    
    return best_a


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    t = np.linspace(0, 1, n)
    t_e = np.linspace(0, 1, n, endpoint=False)
    
    # Linear increasing
    inits.append(0.5 + t_e)
    # Linear decreasing
    inits.append(1.5 - t_e)
    # Quadratic increasing
    inits.append(0.5 + 2 * t_e ** 2)
    # Exponential increasing
    inits.append(np.exp(2 * t_e) / np.exp(2 * t_e).sum())
    # Cosine profile
    inits.append(np.maximum(1.0 + 0.5 * np.cos(np.pi * t_e), 0.01))
    # Sinusoidal
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * t_e), 0.01))
    # Two-peaked
    inits.append(np.exp(-10 * (t_e - 0.25) ** 2) + np.exp(-10 * (t_e - 0.75) ** 2) + 0.05)
    # Uniform
    inits.append(np.ones(n))
    # Random smooth
    ksize = max(3, n // 50)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    # Power law
    inits.append((t_e + 0.1) ** 3)
    # Inverse (decreasing fast then flat)
    inits.append(1.0 / (t_e + 0.5))
    
    return [normalize_nonnegative(np.asarray(x, dtype=np.float64)) for x in inits]


def resample(a, new_n):
    """Resample 1D array to new length."""
    a = np.asarray(a, dtype=np.float64)
    if len(a) == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(t_new, t_old, a), 0.0)


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            print(f"  ** J={score:.8f} n={a.size} t={time.monotonic()-self.t0:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)
