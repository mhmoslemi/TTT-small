def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)

    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute the objective J(a)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_grad(a, tau):
    """Compute smooth max gradient using log-sum-exp."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Numerical stability
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        segment = a[k - hi : k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * segment
    return grad, p, b, b_max

def run_smooth_descent(a, deadline, max_iters=5000, tau_start=1.0, tau_end=200.0,
                       lr_init=0.4, lr_end=0.003):

    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.85  # momentum
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = (it - 1) / max(1, max_iters - 1)
        # Smoothed tau schedule
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        # Nesterov step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        momentum = (t_k - 1.0) / (t_k + 1.0)
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:

        else:

        y = a_new + momentum * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    
    return best_a

6. Repeat

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length exploration with bias toward larger n (better theoretical bound)
2. Edge-aware structured initialization exploiting autocorrelation flatness theory
3. Enhanced smooth gradient descent with adaptive tau and Nesterov acceleration
4. Frequency-domain spectral peak reduction (novel)
5. LP mirror descent for Remez-style equalization
6. Local coordinate refinement for final polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi : k - lo + 1][::-1]
        grad[lo : hi + 1] += 2.0 * p[k] * seg
    return grad, p, b, b_max


def run_smooth_descent(a, deadline, max_iters=8000, tau_start=1.0, tau_end=500.0,
                        lr_init=0.5, lr_end=0.002, momentum=0.85):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    stall = 0
    stall_threshold = max(200, max_iters // 20)

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = (it - 1) / max(1, max_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.45)
        lr = lr_init * (lr_end / lr_init) ** (frac ** 0.8)

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-11):
            best_M = b_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_threshold:
            break

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        mom = (t_k - 1.0) / (t_k + 1.0)
        a_new = np.maximum(y - lr * (grad + momentum * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + mom * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y /= sy
        a = a_new

    return best_a


def spectral_peak_reduction(a, deadline, max_iters=3000, n_reduce_frac=0.1):
    """
    Frequency-domain spectral peak reduction.
    Iteratively reduces the top power bins in the spectrum to flatten
    the autocorrelation, then projects back to non-negative time domain.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    best_a = a.copy()
    best_M = fast_conv(a).max()
    stall = 0
    stall_limit = 500

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2
        conv = np.fft.irfft(power, N)[:2 * n - 1]
        M = conv.max()
        mean_c = conv.mean()

        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_limit:
            break

        # Adaptive: reduce more aggressively when peak >> mean
        ratio = M / max(mean_c, 1e-15)
        n_bins = len(power)
        n_reduce = max(2, int(n_bins * n_reduce_frac))

        # Sort non-DC bins by power descending
        idx = np.argsort(power[1:])[::-1][:n_reduce] + 1
        # Adaptive damping: stronger when peak is much higher than mean
        damping = np.sqrt(max(0.7, 1.0 - 0.3 * (ratio - 2.0) / max(ratio - 1.0, 1.0)))

        A_new = A.copy()
        for i in idx:
            A_new[i] *= damping

        a_new = np.fft.irfft(A_new * np.ones_like(A_new, dtype=complex).real if False else A_new, N)[:n]
        # Fix: just use irfft of modified A
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new

    return best_a


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=20):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=400):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=25)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        for t in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-13:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break
    return best_a


def coordinate_refine(a, deadline, n_trials=5000):
    """Local coordinate descent refinement."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        trial_best = a.copy()
        current_M = best_M
        # Try different scales for a[j]
        for factor in [0.5, 0.7, 0.85, 1.0, 1.1, 1.2, 1.4, 1.7, 2.0]:
            trial_best[j] = a[j] * factor
            if trial_best[j] < 0:
                continue
            s = trial_best.sum()
            if s < 1e-10:
                continue
            trial_best_norm = trial_best / s
            M_t = fast_conv(trial_best_norm).max()
            if M_t < current_M - 1e-12:
                current_M = M_t
                trial_best = trial_best_norm.copy()
        if current_M < best_M - 1e-12:
            best_M = current_M
            a = trial_best.copy()

    return a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        for sigma in [0.01, 0.03, 0.06]:
            inits.append(np.maximum(p + rng.normal(0, sigma, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.005))

    # Uniform
    inits.append(np.ones(n))

    # Cosine (evided center, elevated edges) - a_j = 1 + eps*cos(2*pi*j/n)
    for eps in [0.1, 0.3, 0.5, 0.7]:
        inits.append(np.maximum(1.0 + eps * np.cos(2 * np.pi * t), 0.01))

    # Inverted cosine (elevated center)
    for eps in [0.1, 0.3, 0.5]:
        inits.append(np.maximum(1.0 - eps * np.cos(2 * np.pi * t), 0.01))

    # Power-law shapes
    for alpha in [0.3, 0.5, 0.7, 1.5, 2.0, 3.0]:
        shape = (t ** alpha + (1 - t) ** alpha)
        shape = shape / shape.max()
        inits.append(np.maximum(0.2 + 0.8 * shape, 0.01))

    # Trapezoidal windows
    for ramp in [0.1, 0.2, 0.3, 0.5]:
        trapezoid = np.clip(np.minimum(t / ramp, (1 - t) / ramp), 0, 1)
        inits.append(np.maximum(0.1 + 0.9 * trapezoid, 0.01))

    # Inverted trapezoid (U-shape)
    for ramp in [0.1, 0.2, 0.3]:
        trapezoid = np.clip(np.minimum(t / ramp, (1 - t) / ramp), 0, 1)
        inits.append(np.maximum(1.0 - 0.5 * trapezoid, 0.01))

    # Multi-frequency cosine
    for f in [1, 2, 3, 4]:
        for amp in [0.1, 0.3]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * f * t), 0.01))

    # Gaussian bump (center)
    for sig in [0.15, 0.25, 0.4]:
        inits.append(np.maximum(np.exp(-((t - 0.5) / sig) ** 2) + 0.1, 0.01))

    # Double bump (edges)
    for sig in [0.1, 0.15, 0.25]:
        inits.append(np.maximum(np.exp(-((t - 0) / sig) ** 2) +
                                np.exp(-((t - 1) / sig) ** 2) + 0.1, 0.01))

    # Random smooth
    k = max(3, n // 20)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n + k - 2), kern, 'same')[:n] + 0.1, 0.01))

    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)

    return inits
