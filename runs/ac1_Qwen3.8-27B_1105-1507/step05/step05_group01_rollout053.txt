"""
Hybrid LP-Subgradient optimization with Nesterov momentum and adaptive 
perturbation for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    m = 2*n - 1
    # A[i, j] = a[i-j] if i >= j, else a[j-i]... 
    # Actually for convolution: (a*g)[k] = sum_j a[j] * g[k-j]
    # So the constraint is sum_j A[k,j] * g[j] <= M
    # where A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        # a[k-j] for k = j, j+1, ..., j+n-1
        # So A[k,j] = a[k-j] for k in [j, j+n-1]
        end = min(j + n, m)
        A[j:end, j] = a[:end-j]
    return A

def solve_lp(a, M, n_constraints=None):
    """Solve LP: max sum(g) s.t. A@b_ub, g >= 0. Optionally use only top constraints."""
    n = len(a)
    m = 2*n - 1
    c = -np.ones(n)
    
    if n_constraints is not None and n_constraints < m:
        # Use only the tightest constraints
        conv_ag = compute_conv(a)  # approximate: use a*a as proxy for which constraints are tight
        # Actually we need conv(a, g) but we don't know g yet.
        # Use the constraint indices where conv(a,a) is largest as a heuristic
        idx = np.argsort(conv_ag)[::-1][:n_constraints]
        idx = np.sort(idx)
        A_full = build_toeplitz(a)
        A_sub = A_full[idx, :]
        b_sub = np.full(len(idx), M)
    else:
        A_full = build_toeplitz(a)
        A_sub = A_full
        b_sub = np.full(m, M)
    
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_sub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0

def subgradient_step(a, num_peaks=5, eta=0.1):
    """Compute and apply a subgradient step for the max convolution peak."""
    n = len(a)
    conv = compute_conv(a)
    # Find top peaks
    top_k = np.argsort(conv)[::-1][:num_peaks]
    
    # Compute subgradient: average of gradients at top peaks
    s = np.zeros(n)
    for k in top_k:
        k = int(k)
        # grad_k(j) = 2*a(k-j) if 0 <= k-j < n
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            s[j] += 2.0 * a[k - j]
    s /= len(top_k)
    
    # Apply subgradient step with projection
    a_new = a - eta * s
    a_new = np.maximum(a_new, 0)
    sum_new = np.sum(a_new)
    if sum_new < 1e-10:
        return a.copy()
    a_new /= sum_new
    return a_new

def hybrid_optimize(a, time_budget=60.0, num_peaks=3, use_momentum=True):
    """Main optimization loop combining LP and subgradient with momentum."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n

    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""

Exponential-weight gradient flow with Nesterov momentum and periodic LP polishing
for minimizing 2*n*max(conv(a,a))/(sum(a))^2. Uses FFT-accelerated gradient
computation, adaptive beta annealing, and multi-length exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj_fast(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_grad_exponent(a, weights):
    """
    Compute grad(j) = 2 * sum_k weights[k] * a(k-j) for valid indices.
    Uses FFT cross-correlation for O(n log n) speed.
    weights has length 2n-1.
    """
    n = len(a)
    m = len(weights)  # 2n-1
    # grad(j) = 2 * sum_{m=0}^{n-1} a(m) * weights(j+m)
    # = 2 * (a * w_rev)(2n-2-j) where w_rev = weights[::-1]
    w_rev = weights[::-1]
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w_rev, size)
    conv = irfft(fa * fw, size)[:n + m - 1]
    # grad[j] = 2 * conv[2n-2-j] for j=0,...,n-1
    # conv indices: for j=0 -> conv[2n-2], for j=n-1 -> conv[n-1]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = np.sum(w)
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

    """Build constraint matrix for LP: (a*g)[k] = sum_j A[k,j]*g[j]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

    """Solve LP: max sum(g) s.t. conv(a,g)(k) <= M for all k, g >= 0."""

    bounds = [(0.0, None)] * n
    
    A = build_toeplitz(a)

        conv_aa = compute_conv_fft(a)
        idx = np.argsort(conv_aa)[::-1][:max_constraints]
        idx = np.sort(idx)
        A_sub = A[idx, :]
        b_sub = np.full(len(idx), M)
    else:
        A_sub = A
        b_sub = np.full(m, M)
    
    try:

                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0


def gradient_flow_optimize(a, time_budget=300.0, beta_start=5.0, beta_end=50.0,
                           use_nesterov=True, lp_every=500):
    """
    Main gradient flow optimizer with exponential-weights and adaptive beta.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    
    # Nesterov momentum
    y = a.copy()
    momentum = 0.0
    momentum_decay = 0.9
    
    it = 0
    stall = 0
    patience = 2000
    
    while time.monotonic() - start < time_budget - 2:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 2:
            break
        
        # Adaptive beta
        progress = min(elapsed / time_budget, 1.0)
        beta = beta_start + (beta_end - beta_start) * (progress ** 0.5)
        
        # Nesterov lookahead
        if use_nesterov:
            momentum = momentum_decay * momentum
            y = a + momentum
            y = np.maximum(y, 0)
            s_y = np.sum(y)
            if s_y < 1e-10:
                y = a.copy()
            else:
                y = y / s_y
            
            conv = compute_conv_fft(y)
        else:
            y = a
            conv = compute_conv_fft(a)
        
        # Exponential weights
        conv_shifted = conv - conv.max()
        exp_vals = np.exp(beta * conv_shifted)
        w_sum = np.sum(exp_vals)
        if w_sum < 1e-300:
            weights = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            weights = exp_vals / w_sum
        
        # Compute gradient using FFT
        grad = compute_grad_exponent(y, weights)
        
        # Adaptive step size
        grad_norm = np.max(np.abs(grad))
        if grad_norm < 1e-15:
            eta = 0.01
        else:
            eta = 0.5 / (beta * grad_norm + 1e-10)
        
        # Gradient step
        a_new = a - eta * grad
        a_new = proj_simplex(a_new)
        
        # Update momentum
        if use_nesterov:
            momentum = a_new - a
        
        obj_new = compute_obj_fast(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        it += 1
        
        # Periodic LP polish
        if lp_every > 0 and it % lp_every == 0 and remaining > 30:
            conv_cur = compute_conv_fft(best_a)
            M_cur = float(np.max(conv_cur))
            g, S = solve_lp_direction(best_a, M_cur)
            if g is not None and S > 1.0 + 1e-6:
                g_norm = g / S
                # Blend
                for t in [0.3, 0.5, 0.7, 1.0]:
                    a_blend = (1.0 - t) * a + t * g_norm
                    a_blend = proj_simplex(a_blend)
                    obj_blend = compute_obj_fast(a_blend)
                    if obj_blend < best_obj - 1e-12:
                        best_obj = obj_blend
                        best_a = a_blend.copy()
                        a = a_blend
                        stall = 0
                        momentum = np.zeros(n)
                        break
        
        # Stall handling: perturbation
        if stall > patience:
            rng = np.random.default_rng(it * 31 + 7)
            scale = rng.uniform(0.01, 0.05)
            a = a * np.exp(rng.normal(0, scale, n))
            a = proj_simplex(a)
            momentum = np.zeros(n)
            stall = 0
            patience = max(500, patience // 2)
            beta = max(beta_start, beta * 0.5)
    
    return best_a, best_obj


def lp_polish(a, time_budget=100.0):
    """Pure LP-based polishing (best response iteration)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget - 3:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        conv = compute_conv_fft(a)
        M = float(np.max(conv))
        
        g, S = solve_lp_direction(a, M)
        if g is None or S <= 1.0 + 1e-8:
            stall += 1
            if stall > 30:
                rng = np.random.default_rng(int(time.monotonic() * 10000))
                a = a * np.exp(rng.normal(0, 0.03, n))
                a = proj_simplex(a)
                stall = 0
            continue
        
        improved = False
        for t in [0.3, 0.5, 0.7, 0.85, 1.0]:
            a_try = (1.0 - t) * a + t * g / S
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj_fast(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            stall += 1
            if stall > 50:
                # Try with relaxed constraints
                g2, S2 = solve_lp_direction(a, M * 1.1)
                if g2 is not None and S2 > 1.0 + 1e-6:
                    a_try = 0.5 * a + 0.5 * g2 / S2
                    a_try = proj_simplex(a_try)
                    obj_try = compute_obj_fast(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        stall = 0
                        continue
                # Perturb
                rng = np.random.default_rng(stall * 997)
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = proj_simplex(a)
                stall = 0
    
    return best_a, best_obj


def peak_targeted_refine(a, time_budget=20.0, num_peaks=5):
    """Target reduction of top peaks in the convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for _ in range(100):
        if time.monotonic() - start > time_budget - 1:
            break
        
        conv = compute_conv_fft(a)
        top_idx = np.argsort(conv)[::-1][:num_peaks]
        
        # For each top peak, find which a indices contribute most
        k = int(top_idx[rng.integers(0, len(top_idx))])
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        affected = list(range(lo, hi + 1))
        
        if len(affected) == 0:
            continue
        
        # Reduce affected indices
        a_new = a.copy()
        for pos in affected:
            a_new[pos] *= rng.uniform(0.75, 0.95)
        
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj_fast(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_obj


def spectral_refine(a, time_budget=20.0):
    """Spectral shaping to flatten autocorrelation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    for _ in range(50):
        if time.monotonic() - start > time_budget - 1:
            break
        
        fa = rfft(a, n)
        mag = np.abs(fa)
        phase = np.angle(fa)
        
        choice = rng.integers(0, 3)
        if choice == 0:
            # Soft-clip high magnitude
            threshold = np.percentile(mag, 90)
            mag_new = np.where(mag > threshold, threshold + (mag - threshold) * 0.5, mag)
            fa_new = mag_new * np.exp(1j * phase)
        elif choice == 1:
            # Spectral flattening
            target = np.mean(mag)
            alpha = rng.uniform(0.1, 0.4)
            mag_new = (1 - alpha) * mag + alpha * target
            fa_new = mag_new * np.exp(1j * phase)
        else:
            # Random spectral perturbation
            mag_new = mag * np.exp(rng.normal(0, 0.05, len(fa)))
            fa_new = mag_new * np.exp(1j * phase)
        
        a_new = irfft(fa_new, n)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj_fast(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)
    
    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        # Chirp-like (quadratic phase)
        f0 = rng.uniform(0.5, 2.0)
        f1 = rng.uniform(1.0, 5.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + f1 * (k / n) ** 2))
    else:
        a = rng.random(n) + 0.05
    
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def try_parent():

def propose_candidate(seed=42, budget_s=1500):

    Main search: Exponential-weight gradient flow with Nesterov momentum,

    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    print(f"Initial best: {best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")
    
    # Phase 1: Quick gradient flow on parent
    if best_seq is not None:
        n_parent = len(best_seq)
        remaining = deadline - time.monotonic()
        gf_budget = min(remaining * 0.25, 300.0)
        print(f"\n=== Phase 1: Gradient Flow (n={n_parent}, budget={gf_budget:.0f}s) ===")
        
        a_gf, obj_gf = gradient_flow_optimize(
            best_seq, time_budget=gf_budget,
            beta_start=5.0, beta_end=80.0,
            use_nesterov=True, lp_every=800
        )
        if obj_gf < best_score:
            best_score = obj_gf
            best_seq = a_gf.copy()
            print(f"  After GF: {best_score:.6f}")
        else:
            print(f"  After GF: {obj_gf:.6f} (no improvement)")
    
    # Phase 2: Multi-length exploration
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    
    if best_seq is not None:

    for n in n_values:

        if remaining < 40:

        alloc = min(remaining * 0.15, 100.0)

        if best_seq is not None and len(best_seq) != n:
            a_start = resample_sequence(best_seq, n)
        elif best_seq is not None:
            a_start = best_seq.copy()

        # Gradient flow

            a_start, time_budget=alloc * 0.5,

        # LP polish
        if time.monotonic() < deadline - 30:
            a_opt, obj_opt = lp_polish(a_opt, time_budget=min(alloc * 0.3, 30.0))
        
        # Peak targeting
        if time.monotonic() < deadline - 20:
            a_opt, obj_opt = peak_targeted_refine(
                a_opt, time_budget=min(5.0, alloc * 0.1), num_peaks=5)
        
        # Spectral
        if time.monotonic() < deadline - 20:
            a_opt, obj_opt = spectral_refine(a_opt, time_budget=min(5.0, alloc * 0.1))
        
        # Final LP
        if time.monotonic() < deadline - 15:
            a_opt, obj_opt = lp_polish(a_opt, time_budget=min(alloc * 0.1, 15.0))
        
        print(f"  N={n}: obj={obj_opt:.6f}")
        
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")
    
    # Phase 3: Intensive final optimization on global best

        remaining = deadline - time.monotonic()

            if obj4b < best_score:

                best_seq = a_cur.copy()

        # Multiple short gradient flow runs with different parameters
        if time.monotonic() < deadline - 15:

            for beta_s in [10.0, 30.0, 100.0]:
