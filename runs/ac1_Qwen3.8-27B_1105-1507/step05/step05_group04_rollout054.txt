"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
- Efficient FFT-based gradient computation
- Adam optimizer with adaptive smooth-max parameter
- Palindrome symmetry exploitation
- Multi-length exploration
- LP-based active set identification
- Coordinate descent polishing

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time
import math

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_corr(p, a):
    """Compute h_i = sum_k p_k * a_{k-i} for i=0,...,n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size *= 2
    Fp = np.fft.rfft(p, size)
    Fa = np.fft.rfft(a, size)
    # Cross-correlation: DFT(p xor a) = P * conj(A)
    h = np.fft.irfft(Fp * np.conj(Fa), size)
    # We need h_i = sum_k p_k * a_{k-i}
    # The cross-correlation r(i) = sum_k p(k+i) * conj(a(k)) 
    # In our case (real): r(i) = sum_k p(k+i) * a(k)
    # We want h(i) = sum_k p(k) * a(k-i) = sum_j p(j+i) * a(j) = r(i)
    # So h = r, which is what we computed
    return h[:n]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_grad(a, tau):
    """Compute gradient of J using smooth max approximation.
    Returns (grad, J_approx, kkt_residual, phi)."""
    n = len(a)
    S = a.sum()
    b = fast_conv(a)
    
    # Numerically stable log-sum-exp
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    log_sum = np.log(exp_b.sum())
    phi = b_max + log_sum / tau
    
    # Weights p_k
    p = exp_b / exp_b.sum()
    
    # Cross-correlation h_i = sum_k p_k * a_{k-i}
    h = cross_corr(p, a)
    
    # J approximation
    J_approx = 2 * n * phi / (S * S)
    
    # Gradient direction (to minimize J): (phi/S - h_i) * (4n/S)
    # We'll use (phi/S - h_i) as the direction (ignoring the constant factor)
    grad = phi / S - h
    
    # KKT residual
    kkt = np.max(np.abs(h - phi / S))
    if phi / S > 1e-15:
        kkt /= (phi / S)
    
    return grad, J_approx, kkt, phi

def adam_step(a, grad, m, v, t, lr, beta1, beta2, epsilon=1e-8):
    """One Adam step. Returns updated (a, m, v)."""
    m_new = beta1 * m + (1 - beta1) * grad
    v_new = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m_new / (1 - beta1**t)
    v_hat = v_new / (1 - beta2**t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
    return a_new, m_new, v_new

def gradient_descent_multi(a0, deadline, tau_start=5, tau_end=200,
                           lr_start=0.1, lr_end=0.001, use_palindrome=False,
                           verbose=False):
    """Run gradient descent with Adam and adaptive tau."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    if use_palindrome:
        a = make_palindrome(a)
    a = a / a.sum()
    n = len(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    
    max_iters = 50000
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Adaptive tau and learning rate
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        grad, J_approx, kkt, phi = smooth_grad(a, tau)
        
        a_new, m, v = adam_step(a, grad, m, v, t, lr, 0.9, 0.999)
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        if use_palindrome:
            a_new = make_palindrome(a_new)
        
        # Normalize
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        a = a_new
        
        # Check improvement (every 10 iterations to save time)
        if it % 10 == 0:
            J_actual = compute_J(a)
            if J_actual < best_J - 1e-12:
                best_J = J_actual
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 1000 == 0:
                    print(f"    GD iter={it} J={J_actual:.8f} tau={tau:.1f} kkt={kkt:.4f}", flush=True)
            else:
                no_improve += 10
        
        if no_improve > 500:
            # Restart with perturbation
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.05 + 0.05 * (no_improve / 500)
            a = a * (1 + scale * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-6)
            if use_palindrome:
                a = make_palindrome(a)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
    
    return best_a, best_J

def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def coordinate_descent(a, deadline, n_passes=50, verbose=False):
    """Coordinate descent for fine-tuning."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for path in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            # Try several values for a[i]
            current = a[i]
            s_others = a.sum() - current
            
            candidates = [0.0, current * 0.5, current * 0.8, current,
                         current * 1.2, current * 1.5, max(s_others/n, 1e-8)]
            # Add values based on neighbors
            if i > 0:
                candidates.append(a[i-1])
            if i < n-1:
                candidates.append(a[i+1])
            # Add a few random values
            rng = np.random.default_rng(i + path * 1000)
            candidates.extend([current * rng.uniform(0.5, 2.0) for _ in range(3)])
            
            best_val = current
            best_local_J = best_J  # We'll compute actual J
            
            for val in candidates:
                val = max(0.0, min(val, 10.0))
                a[i] = val
                s = a.sum()
                if s < 0.01:
                    continue
                J_val = compute_J(a)
                if J_val < best_local_J:
                    best_local_J = J_val
                    best_val = val
            
            a[i] = best_val
            if best_val != current:
                improved = True
        
        a = a / a.sum()
        J_actual = compute_J(a)
        if J_actual < best_J - 1e-14:
            best_J = J_actual
            best_a = a.copy()
            if verbose and path % 10 == 0:
                print(f"    CD pass={path} J={J_actual:.10f}", flush=True)
    
    return best_a, best_J

def lp_refine(a, deadline, verbose=False, n_iterations=200):
    """LP-based refinement using the active set approach."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(n_iterations):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Identify active set (top constraints)
        threshold = M * 0.999
        active_k = np.where(b >= threshold)[0]
        
        if len(active_k) == 0:
            active_k = np.where(b >= M * 0.99)[0]
        if len(active_k) == 0:
            active_k = np.where(b >= M * 0.9)[0]
        
        # Solve LP: max sum(g) s.t. (a*g)_k <= M for k in active_k, g >= 0
        # (a*g)_k = sum_j a_j * g_{k-j}
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active_k):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(a[k - j])
        
        A = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
        c = -np.ones(n)
        b_ub = np.full(len(active_k), M)
        bounds = [(0, np.inf)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 5, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0.0)
                if g.sum() > 1e-10:
                    g = g / g.sum()
                    
                    # Try different step sizes
                    for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
                        a_trial = (1 - t) * a + t * g
                        a_trial = np.maximum(a_trial, 0.0)
                        s = a_trial.sum()
                        if s < 1e-10:
                            continue
                        a_trial = a_trial / s
                        J_trial = compute_J(a_trial)
                        if J_trial < best_J - 1e-14:
                            a = a_trial
                            best_J = J_trial
                            best_a = a.copy()
                            if verbose and it % 20 == 0:
                                print(f"    LP iter={it} J={J_trial:.10f} active={len(active_k)}", flush=True)
                            break
        except Exception:
            pass
    
    return best_a, best_J

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
- Efficient FFT-based gradient computation with smooth max
- Adam optimizer with adaptive tau
- Palindrome symmetry exploitation
- Multi-length, multi-start exploration
- LP-based active set refinement
- Coordinate descent polishing
- Hybrid local search combining all methods
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_corr_p_a(p, a):
    """Compute h_i = sum_k p_k * a_{k-i} for i=0,...,n-1 using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size *= 2
    Fp = np.fft.rfft(p, size)
    Fa = np.fft.rfft(a, size)
    result = np.fft.irfft(Fp * np.conj(Fa), size)
    return result[:n]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def make_palindrome(a):
    """Enforce palindrome symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    a_new = np.zeros(n)
    half = (n + 1) // 2
    for i in range(half):
        val = 0.5 * (a[i] + a[n - 1 - i])
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_gradient(a, tau):
    """Compute gradient of J using smooth max. Returns (grad_dir, J_approx, kkt_res)."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.zeros(n), float('inf'), float('inf')

    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_b = np.exp(shifted)
    log_sum = np.log(exp_b.sum())
    phi = b_max + log_sum / tau
    p = exp_b / exp_b.sum()

    h = cross_corr_p_a(p, a)
    J_approx = 2.0 * n * phi / (S * S)
    target = phi / S
    grad_dir = target - h
    kkt_res = float(np.max(np.abs(h - target))) / (target + 1e-15)
    return grad_dir, J_approx, kkt_res


def gd_adam(a0, deadline, tau_start=5.0, tau_end=150.0,
            lr_start=0.05, lr_end=0.0005, use_palindrome=False,
            max_iters=50000, verbose=False):
    """Gradient descent with Adam and adaptive tau."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    if use_palindrome:
        a = make_palindrome(a)
    a = a / a.sum()
    n = len(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    check_interval = 15

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t += 1

        frac = min(1.0, it / (max_iters * 0.8))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** min(1.0, it / max_iters)

        grad_dir, J_approx, kkt = smooth_gradient(a, tau)

        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1 - b1) * grad_dir
        v = b2 * v + (1 - b2) * grad_dir * grad_dir
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = np.maximum(a_new, 0.0)
        if use_palindrome:
            a_new = make_palindrome(a_new)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s

        if it % check_interval == 0:
            J_act = compute_J(a)
            if J_act < best_J - 1e-13:
                best_J = J_act
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 500 == 0:
                    print(f"    GD it={it} J={J_act:.10f} tau={tau:.1f} kkt={kkt:.5f}", flush=True)
            else:
                no_improve += check_interval

        if no_improve > 600:
            rng = np.random.default_rng(t * 104729 + 3)
            scale = min(0.3, 0.05 + 0.02 * (no_improve // 600))
            a = a * (1.0 + scale * rng.standard_normal(n))
            a = np.maximum(a, 1e-6)
            if use_palindrome:
                a = make_palindrome(a)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0

    return best_a, best_J


def lp_refine(a0, deadline, verbose=False, max_iters=500, frac_active=0.98):
    """LP-based refinement solving max sum(g) s.t. a*g <= M."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        threshold = M * frac_active
        active_k = np.where(b >= threshold)[0]

        if len(active_k) < 5:
            idx = np.argsort(b)[-max(5, n // 4):]
            active_k = idx
        if len(active_k) == 0:
            break

        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active_k):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(a[k - j])

        A = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
        c = -np.ones(n)
        b_ub = np.full(len(active_k), M)
        bounds = [(0, np.inf)] * n

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 3, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0.0)
                if g.sum() > 1e-10:
                    g = g / g.sum()
                    improved = False
                    for t_step in [0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03]:
                        a_trial = (1.0 - t_step) * a + t_step * g
                        a_trial = np.maximum(a_trial, 0.0)
                        s = a_trial.sum()
                        if s < 1e-10:
                            continue
                        a_trial = a_trial / s
                        J_trial = compute_J(a_trial)
                        if J_trial < best_J - 1e-14:
                            a = a_trial
                            best_J = J_trial
                            best_a = a.copy()
                            improved = True
                            if verbose and it % 30 == 0:
                                print(f"    LP it={it} J={best_J:.10f} n_act={len(active_k)}", flush=True)
                            break
                    if not improved:
                        # Try smaller steps
                        for t_step in [0.02, 0.01, 0.005]:
                            a_trial = (1.0 - t_step) * a + t_step * g
                            a_trial = np.maximum(a_trial, 0.0)
                            s = a_trial.sum()
                            if s < 1e-10:
                                continue
                            a_trial = a_trial / s
                            J_trial = compute_J(a_trial)
                            if J_trial < best_J - 1e-14:
                                a = a_trial
                                best_J = J_trial
                                best_a = a.copy()
                                break
        except Exception:
            pass

    return best_a, best_J


def coordinate_descent(a0, deadline, n_passes=30, verbose=False):
    """Coordinate descent for fine-tuning each element."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for path in range(n_passes):
        if time.monotonic() > deadline:
            break

        for i in range(n):
            if time.monotonic() > deadline:
                break

            current = a[i]
            if current < 1e-15:
                continue

            s_others = a.sum() - current
            if s_others < 1e-10:
                continue

            # Generate candidate values
            cands = [current]
            cands.append(current * 0.7)
            cands.append(current * 1.3)
            cands.append(current * 0.5)
            cands.append(current * 1.5)
            cands.append(0.0)
            cands.append(s_others / n)
            if i > 0:
                cands.append(a[i - 1])
            if i < n - 1:
                cands.append(a[i + 1])
            # Mean of neighbors
            nb = []
            if i > 0:
                nb.append(a[i - 1])
            if i < n - 1:
                nb.append(a[i + 1])
            if nb:
                cands.append(np.mean(nb))

            best_val = current
            # Use current best_J as threshold
            local_best = compute_J(a)

            for val in cands:
                val = max(0.0, min(float(val), 10.0 * s_others / n))
                if abs(val - current) < 1e-12:
                    continue
                a[i] = val
                s = a.sum()
                if s < 0.01:
                    a[i] = current
                    continue
                J_val = compute_J(a)
                if J_val < local_best:
                    local_best = J_val
                    best_val = val

            a[i] = best_val

        # Normalize
        s = a.sum()
        if s > 1e-10:
            a = a / s

        J_now = compute_J(a)
        if J_now < best_J - 1e-14:
            best_J = J_now
            best_a = a.copy()
            if verbose and path % 5 == 0:
                print(f"    CD path={path} J={best_J:.10f}", flush=True)

    return best_a, best_J


def generate_inits(parent, rng, n_list):
    """Generate diverse initializations."""
    inits = []
    for n in n_list:
        # Parent resampled
        p = resample_seq(parent, n)
        inits.append(('parent', np.maximum(p, 1e-6)))

        # Palindrome parent
        inits.append(('pal', make_palindrome(np.maximum(p, 1e-6))))

        # Perturbed parent
        for sigma in [0.1, 0.3]:
            pert = np.maximum(p * (1 + sigma * rng.standard_normal(n)), 1e-6)
            inits.append((f'pert{sigma}', pert))

        # Structured
        t_arr = np.linspace(0, 1, n, endpoint=False)
        inits.append(('cos1', np.maximum(1 + 0.3 * np.cos(2 * np.pi * t_arr), 0.1)))
        inits.append(('cos2', np.maximum(1 + 0.3 * np.cos(4 * np.pi * t_arr), 0.1)))
        inits.append(('gauss', np.maximum(np.exp(-6 * (t_arr - 0.5) ** 2), 0.01)))
        inits.append(('bilateral', np.maximum(np.exp(-4 * t_arr**2) + np.exp(-4 * (t_arr-1)**2), 0.01)))
        inits.append(('uniform', np.ones(n) * 0.1))

    return inits


def class_score_tracker_init():
    return {'best': None, 'best_json': float('inf'), 't_start': time.monotonic()}


def class_score_tracker_update(tracker, a, label=""):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 0.01:
        return
    J = compute_J(a)
    if J < tracker['best_json']:
        tracker['best_json'] = J
        tracker['best'] = a.copy()
        elapsed = time.monotonic() - tracker['t_start']
        print(f"  ** NEW BEST: J={J:.10f} n={len(a)} t={elapsed:.0f}s [{label}]", flush=True)
