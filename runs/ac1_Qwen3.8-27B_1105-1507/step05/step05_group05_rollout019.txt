import numpy as np
from scipy.optimize import linprog

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    # Bound candidates
    max_delta = 10.0  # cap to avoid numerical issues
    candidates = [d for d in candidates if d > -x_old - 1e-10 and d < max_delta]
    candidates = [max(d, -x_old + 1e-12) for d in candidates]
    
    # Evaluate F at all candidates
    best_d = 0.0
    best_F = F_of_delta(0.0)
    for d in candidates:
        f_val = F_of_delta(max(d, -x_old + 1e-12))
        if f_val < best_F:
            best_F = f_val
            best_d = d
    
    # Refine with golden section in a small window around best_d
    window = max(0.01, abs(best_d) * 0.1 + 0.001)
    lo = max(best_d - window, -x_old + 1e-12)
    hi = best_d + window
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = F_of_delta(max(a_pt, -x_old + 1e-12))
    fb = F_of_delta(max(b_pt, -x_old + 1e-12))
    
    for _ in range(25):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = F_of_delta(max(a_pt, -x_old + 1e-12))
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = F_of_delta(max(b_pt, -x_old + 1e-12))
    
    d_opt = (lo + hi) / 2.0
    # Check boundary
    f_boundary = F_of_delta(-x_old + 1e-12)
    if f_boundary < F_of_delta(d_opt):
        d_opt = -x_old + 1e-12
    
    return max(d_opt, -x_old + 1e-12)


def run_fast_coordinate_descent(a, max_sweeps, deadline, seed=0):
    """Fast coordinate descent with incremental convolution updates."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    S = 1.0
    n = len(a)
    conv = fast_conv(a)
    
    best_a = a.copy()
    best_M = conv.max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = np.arange(n)
        rng_cd.shuffle(order)
        
        sweep_improved = False
        for i in order:
            x_old = a[i]
            if x_old < 1e-15 and conv[k_lo_check := i] < best_M * 0.5:
                # If this coordinate is near zero and its peak contribution is small, skip
                continue
            delta = solve_1d_coordinate(i, a, conv, S, x_old)
            new_x = x_old + delta
            if abs(delta) > 1e-14:
                # Update convolution incrementally
                k_start = i
                k_end = min(2 * n - 2, i + n - 1)
                j_range = np.arange(k_start, k_end + 1) - i
                conv[k_start:k_end + 1] += 2.0 * delta * a[j_range]
                if 2 * i <= 2 * n - 2:
                    conv[2 * i] += delta * delta
                a[i] = new_x
                S += delta
                sweep_improved = True
        
        # Renormalize
        if S < 1e-10:
            break
        a = a / S
        conv = conv / (S * S)
        S = 1.0
        
        M = conv.max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 3
        if no_improve > 25:
            break
    
    return best_a


def spectral_init(n, iters=200):
    """
    Initialize via alternating projections: project onto non-negative simplex
    and onto flat-spectrum constraint in Fourier domain.
    """
    a = np.ones(n) / n
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for it in range(iters):
        # Current spectrum
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        # Target: flat magnitude (proportional to 1/sqrt(2n-1) in convolution domain)
        # In Fourier: |A[k]|^2 should be constant => |A[k]| = 1/sqrt(n) (after normalization)
        target_mag = np.ones(len(A)) / np.sqrt(n)
        # Scale current to match target
        scale = target_mag / (mag + 1e-15)
        A_new = A * scale
        # Inverse FFT
        a_new = np.fft.irfft(A_new, size)[:n]
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        # Normalize
        s = a_new.sum()
        if s < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new
    
    return a


def smooth_max_subgradient(a, tau):
    """Compute subgradient of smooth max (log-sum-exp) of convolution using FFT."""
    n = len(a)
    conv = fast_conv(a)
    b_max = conv.max()
    exp_c = np.exp(tau * (conv - b_max))
    p = exp_c / exp_c.sum()
    
    # Gradient of conv[k] w.r.t. a[i] is 2*a[k-i] (if valid)
    # So grad[i] = sum_k p[k] * 2*a[k-i] (for valid k-i)
    # This is 2 * (p_reversed * a) -- a cross-correlation
    # Actually: grad[i] = 2 * sum_k p[k] * a[k-i] (for 0<=k-i<n)
    # = 2 * (p full convolve with a, but only p is short... no, p has length 2n-1)
    
    # Use FFT for efficiency: grad = 2 * ifft(fft(p) * conj(fft(a)))
    # Actually, grad[i] = 2 * sum_k p[k] a[k-i]
    # Let's define: (p cross-correlate a)[i] = sum_k p[k] a[k-i]
    # This is the same as convolve(p, flip(a))[i + n - 1]... let me just do it directly with FFT
    
    # grad[i] = 2 * sum_{k=i}^{i+n-1} p[k] * a[k-i]  (valid k-i means 0<=k-i<n, so k in [i, i+n-1])
    # This is a cross-correlation. Let's use FFT.
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    # We want: grad[i] = 2 * sum_k p[k] a[k-i]
    # = 2 * (p * flip(a))[i] where * is cross-correlation
    # = 2 * ifft(P * conj(A)) ... let me verify
    # ifft(fft(p) * conj(fft(a)))[i] = sum_k p[k] a[(k-i) mod size]
    # For our case with zero-padding, this should give the cross-correlation
    cross = np.fft.irfft(P * np.conj(A), size)[:2 * n - 1]
    # cross[i] = sum_k p[k] a[(k-i) mod size] -- but we need a[k-i] not a[i-k]
    # Actually let me just compute it properly
    # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]  (for valid i+j < 2n-1)
    
    # Let me use a simpler approach: grad = 2 * (flip(p) convolve a) appropriately indexed
    # p has length 2n-1. flip(p) has length 2n-1.
    # (flip(p) * a)[m] = sum_j flip(p)[j] * a[m-j] = sum_j p[2n-2-j] * a[m-j]
    # Let k = 2n-2-j: = sum_k p[k] * a[m-(2n-2-k)] = sum_k p[k] * a[m-2n+2+k]
    # We want grad[i] = 2 * sum_k p[k] * a[k-i]
    # Set m - 2n + 2 + k = k - i => m = 2n - 2 - i
    # So grad[i] = 2 * (flip(p) * a)[2n-2-i]
    
    p_flip = p[::-1]
    cross2 = np.fft.irfft(np.fft.rfft(p_flip, size) * A, size)
    grad = 2.0 * cross2[2 * n - 2 - np.arange(n)][:n]
    
    # Numerical check (for small n)
    # Actually the FFT approach should be correct. Let me just use it.
    return grad


def run_subgradient_descent(a, max_iters, deadline, tau=20.0, lr=0.01):
    """Subgradient descent on smooth max with Adam."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Adaptive tau
        tau_t = tau * (1 + 0.5 * it / max(1, max_iters))
        
        conv = fast_conv(a)
        b_max = conv.max()
        exp_c = np.exp(tau_t * (conv - b_max))
        p = exp_c / exp_c.sum()
        
        # Gradient via FFT cross-correlation
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        P = np.fft.rfft(p, size)
        A = np.fft.rfft(a, size)
        p_flip = p[::-1]
        cross = np.fft.irfft(np.fft.rfft(p_flip, size) * A, size)
        grad = 2.0 * cross[2 * n - 2 - np.arange(n)][:n]
        
        # Project to tangent space (sum-zero)
        grad = grad - grad.mean()
        
        # Adam
        lr_t = lr / (1 + 0.05 * it / max(1, max_iters))
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
    
    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix for LP: (a*g)[k] = sum_j a[k-j]*g[j] <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=15):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a, max_iters, deadline):
    """LP mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv = fast_conv(a)
        M = float(conv.max())
        
        # Find active lags (top few)
        n_active = min(10, 2 * n_best := len(a))
        active_threshold = M * (1 - 0.01)
        g = lp_best_response(a, M, time_limit=10)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        improved = False
        base_t = max(0.05, 0.5 * (1 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01, 0.005]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
    
    return best_a


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        if parent.size != n:
            t_old = np.linspace(0, 1, parent.size)
            t_new = np.linspace(0, 1, n)
            p = np.interp(t_new, t_old, parent)
        else:
            p = parent.copy()
        inits.append(np.maximum(p, 1e-6))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
    
    inits.append(np.ones(n))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * t), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.exp(-3 * (t - 0.5) ** 2) + 0.05)
    inits.append(0.3 + 0.7 * np.sin(np.pi * t))
    
    # Spectral init
    spec = spectral_init(n, iters=100)
    inits.append(spec)
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Fast coordinate descent (primary engine) across multiple lengths
    Phase 2: Spectral + subgradient refinement
    Phase 3: LP mirror descent polish
    Phase 4: Final fine coordinate descent
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
    
    # === PHASE 1: Fast Coordinate Descent ===
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Fast Coordinate Descent", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Try multiple lengths, weighted toward parent length
    parent_n = tracker.best.size if tracker.best is not None else 512
    lengths = sorted(set([256, 384, parent_n, max(1, parent_n + 128), max(1, parent_n - 128), 768, 1024]))
    lengths = [l for l in lengths if 128 <= l <= 2048]
    
    # Allocate time: more time for lengths near parent
    total_cd_budget = 700
    weights = {}
    for l in lengths:
        dist = abs(l - parent_n)
        weights[l] = 1.0 / (1.0 + dist / 256.0)
    total_w = sum(weights.values())
    time_alloc = {l: total_cd_budget * weights[l] / total_w for l in lengths}
    
    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        t_alloc = time_alloc.get(n, 60)
        phase_deadline = min(time.monotonic() + t_alloc, deadline - 350)
        
        if time.monotonic() >= phase_deadline:
            continue
        
        t0 = time.monotonic()
        inits = diverse_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_for_n = None
        best_M_for_n = float('inf')
        
        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_deadline - 10:
                break
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()
            
            max_sw = max(30, int(5000 / max(1, n // 64)))
            a_opt = run_fast_coordinate_descent(a0, max_sw, phase_deadline - 5, seed=seed + idx * 13)
            
            M_val = fast_conv(a_opt).max()
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a_opt.copy()
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: M={best_M_for_n:.8f} J={J_val:.6f} ({elapsed:.1f}s)", flush=True)
            tracker.update(best_for_n)
    
    tracker.report("Phase 1 complete")
    print(f"Phase 1 total: {time.monotonic()-p1_start:.1f}s", flush=True)
    
    # === PHASE 2: Subgradient + Coordinate Descent Refinement ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Subgradient + CD Refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 200:
        a_sub = run_subgradient_descent(tracker.best, 500, min(time.monotonic() + 100, deadline - 150),
                                        tau=20.0, lr=0.02)
        tracker.update(a_sub)
        
        # Follow up with coordinate descent
        a_cd = run_fast_coordinate_descent(a_sub, 100, min(time.monotonic() + 80, deadline - 100), seed=777)
        tracker.update(a_cd)
    
    tracker.report("Phase 2 complete")
    print(f"Phase 2 total: {time.monotonic()-p2_start:.1f}s", flush=True)
    
    # === PHASE 3: LP Mirror Descent ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 60:
        n_best = tracker.best.size
        est_lp_time = 0.02 * (n_best / 128) ** 2
        max_iters = max(10, min(100, int(120 / max(0.05, est_lp_time))))
        print(f"  n={n_best}, est LP iter time={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)
        
        a_lp = run_lp_mirror(tracker.best, max_iters, min(time.monotonic() + 130, deadline - 50))
        tracker.update(a_lp)
        
        # CD after LP
        a_cd2 = run_fast_coordinate_descent(a_lp, 50, min(time.monotonic() + 40, deadline - 30), seed=888)
        tracker.update(a_cd2)
    
    tracker.report("Phase 3 complete")
    print(f"Phase 3 total: {time.monotonic()-p3_start:.1f}s", flush=True)
    
    # === PHASE 4: Final Polish ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 5:
        # Fine coordinate descent
        a_final = run_fast_coordinate_descent(tracker.best, 200, deadline - 3, seed=12345)
        tracker.update(a_final)
        
        # Try slight perturbation + CD
        a_pert = tracker.best + rng.normal(0, 0.01, tracker.best.size)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert = a_pert / a_pert.sum()
        a_final2 = run_fast_coordinate_descent(a_pert, 100, deadline - 2, seed=99999)
        tracker.update(a_final2)
    
    tracker.report("Phase 4 complete")
    print(f"Phase 4 total: {time.monotonic()-p4_start:.1f}s", flush=True)
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
