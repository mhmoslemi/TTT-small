"""
KKT Active Set Optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses the KKT conditions of the equivalent problem min max(a*a) s.t. sum(a)=1, a>=0.
At the optimum, the "blurred" sequence (a convolved with the active set kernel) must be constant on the support of a.
This leads to a fixed-point iteration solving a Toeplitz system at each step.
Combined with coordinate descent polishing and multi-length search.
"""

import numpy as np
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def compute_gradient_active_set(a, w):
    """
    Compute grad_i = 2 * sum_j a_j * w[i+j]
    where w is defined on [0, 2n-2].
    Uses O(n^2) direct computation (fast enough for n <= 2048).
    """
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min > i_max:
            continue
        j_vals = k - np.arange(i_min, i_max + 1)
        grad[i_min:i_max + 1] += 2.0 * w[k] * a[j_vals]
    return grad

def kkt_descent(a, max_iters=500, deadline=None, p_start=5, p_end=50, exp_avg=False, seed=0):
    """
    KKT active set descent for min max(a*a) s.t. sum(a)=1, a>=0.
    Uses smooth approximation with geometrically increasing power p.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Adaptive learning rate
    lr = 0.3
    patience = 0
    max_patience = 50
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute convolution
        b = fast_conv(a)
        M = b.max()
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        # Adaptive power p
        p = p_start + (p_end - p_start) * it / max(1, max_iters)
        
        # Compute soft-max weights (normalized)
        if M > 0:
            b_norm = b / M
            w = b_norm ** p
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.ones(2*n-1) / (2*n-1)
            else:
                w = w / w_sum
        else:
            w = np.ones(2*n-1) / (2*n-1)
        
        # Compute gradient
        grad = compute_gradient_active_set(a, w)
        
        # Gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        
        # Simple line search: check if improvement
        b_new = fast_conv(a_new)
        M_new = b_new.max()
        if M_new > M * 1.01:  # got worse
            lr *= 0.7
            if lr < 1e-4:
                break
        else:
            #Accept (use exponential averaging for stability)
            if exp_avg:
                a = 0.9 * a_new + 0.1 * a
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                a = a_new
                lr = min(lr * 1.02, 0.5)  # slowly increase
    
    return best_a

def kkt_toeplitz_solve(a, w, regularize=1e-6):
    """
    Solve the KKT linear system: T @ a = 1, where T[i,j] = 2*w[i+j].
    Then normalize so sum(a) = 1 and project onto a >= 0.
    """
    n = len(a)
    # Build Toeplitz matrix T[i,j] = 2*w[i+j]
    # T is symmetric since T[i,j] = 2*w[i+j] = 2*w[j+i] = T[j,i]
    
    # For efficiency, build T as a sum of shifted diagonal matrices
    # But for n <= 1024, just build it directly
    idx = np.arange(n)
    T = 2.0 * w[np.ix_(idx[:, None] + idx[None, :], np.ones(1))[0]]
    
    # Hmm, that's not right. Let me just use broadcasting.
    i_idx = np.arange(n)[:, None]
    j_idx = np.arange(n)[None, :]
    T = 2.0 * w[i_idx + j_idx]
    
    # Solve (T + reg*I) @ x = 1
    try:
        x = np.linalg.solve(T + regularize * np.eye(n), np.ones(n))
    except np.linalg.LinAlgError:
        x = np.linalg.lstsq(T + regularize * np.eye(n), np.ones(n), rcond=None)[0]
    
    # Project and normalize
    x = np.maximum(x, 0)
    s = x.sum()
    if s < 1e-10:
        return a
    x = x / s
    return x

def kkt_fixed_point(a, max_iters=100, deadline=None, p=100, use_toeplitz=True, regularize=1e-4):
    """
    Fixed-point iteration for KKT conditions:
    a_new = solve(T(a) @ x = 1) where T is built from active set of a.
    """
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
        
        # Build weights
        if M > 0:
            b_norm = b / M
            w = b_norm ** p
            w_sum = w.sum()
            if w_sum < 1e-15:
                break
            w = w / w_sum
        else:
            break
        
        if use_toeplitz:
            a_new = kkt_toeplitz_solve(a, w, regularize)
        else:
            # Use gradient step instead
            grad = compute_gradient_active_set(a, w)
            a_new = a - 0.5 * grad
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
        
        # Mixing for stability
        a = 0.5 * a + 0.5 * a_new
        a = np.maximum(a, 0)
        a = a / a.sum()
    
    return best_a

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a.astype(np.float64), size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_kkt_gradient(a, w):
    """
    Compute gradient vector: grad_i = 2 * sum_{j=0}^{n-1} a_j * w[i+j]
    where w is defined on [0, 2n-2].
    O(n^2) direct computation.
    """
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min > i_max:
            continue
        j_vals = k - np.arange(i_min, i_max + 1)
        grad[i_min:i_max + 1] += 2.0 * w[k] * a[j_vals]
    return grad

def coord_descent_fast(a, max_sweeps=500, deadline=None, seed=0, n_candidates=5):

    Each coordinate update is O(n) via direct array operations.

    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    s = 1.0

    best_conv_max = conv.max()
    best_a = a.copy()

    no_improve = 0
    max_no_improve = 50

        if deadline is not None and time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            old = a[i]
            if old < 1e-10:
                candidates = [0.0, 1e-4, 1e-3, 5e-3, 1e-2]
            else:

                              1.15 * old, 1.5 * old]

            hi = min(2 * n - 2, i + n - 1)

            cur_M = conv.max()
            best_local_M = cur_M
            best_val = old

            for x in candidates:
                x = float(max(0, x))
                if abs(x - old) < 1e-12:
                    continue
                delta = x - old

                # Apply update to conv
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                # s_new = 1 - old + x

                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (s_new * s_new)

                    # Simplified: just track max
                    pass

                # Just track the best by actual objective
                # Rollback
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            # Redo: properly track best
            best_obj_local = 2.0 * n * conv.max()  # current (s=1)
            best_val = old

            for x in candidates:
                x = float(max(0, x))
                if abs(x - old) < 1e-12:
                    continue
                delta = x - old

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                s_new = 1.0 - old + x
                if s_new < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (s_new * s_new)

                if new_obj < best_obj_local:
                    best_obj_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            # Apply best
            if best_val != old and abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = 1.0 - old + best_val
                sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via interpolation."""
    src = np.asarray(src, dtype=np.float64)

    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, parent, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 0:

            inits.append(np.maximum(p + rng.normal(0, 0.15, n), 1e-4) / n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine modulation
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            c = np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01)
            inits.append(c / c.sum())
            s = np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t), 0.01)
            inits.append(s / s.sum())

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        g = np.exp(-((t - center) / 0.2) ** 2) + 0.1
        inits.append(g / g.sum())

    # Two-peak
    two_p = np.exp(-((t - 0.3) / 0.15) ** 2) + np.exp(-((t - 0.7) / 0.15) ** 2) + 0.05
    inits.append(two_p / two_p.sum())

    # Linear ramp
    for direction in [1, -1]:
        r = np.maximum(np.linspace(0.1, 1.0, n)[::direction] + 0.1, 0.01)
        inits.append(r / r.sum())

    # Smooth random

    kernel = np.ones(ksize) / ksize
    for _ in range(2):
        smooth = np.convolve(rng.uniform(0.1, 1, n), kernel, mode='same')
        inits.append(smooth / smooth.sum())

    return inits

                score_and_track(a_new_cd, tracker)
                print(f"  CD  n={new_n}: J={compute_J(a_new_cd):.8f}", flush=True)

    print(f"  Phase 3 best: J={tracker[0]:.8f} n={tracker[1].size if tracker[1] is not None else 0}", flush=True)

    # === PHASE 4: Random restarts and final polish ===
    p4_end = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Restart + Final Polish (budget {p4_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker[1] is not None and time.monotonic() < p4_end - 30:
        n_best = tracker[1].size

        # Try several random perturbations
        for trial in range(8):
            if time.monotonic() > p4_end - 60:
                break

            noise_scale = 0.05 * (1 - trial / 10.0)
            a_pert = np.maximum(tracker[1] + rng.normal(0, noise_scale, n_best), 0)
            a_pert = a_pert / a_pert.sum()

            # Quick KKT
            t0 = time.monotonic()
            a_res = kkt_descent(a_pert, max_iters=100,
                                deadline=min(p4_end - 60, t0 + 40),
                                p_start=5.0, p_end=50.0, lr_init=0.15,
                                seed=seed + 3000 + trial)
            score_and_track(a_res, tracker)

            # Quick CD
            if time.monotonic() < p4_end - 60:
                a_res_cd = coord_descent_fast(a_res, max_sweeps=500,
                                              deadline=min(p4_end - 60, time.monotonic() + 30),
                                              seed=seed + 4000 + trial)
                score_and_track(a_res_cd, tracker)

        # Final coordinate descent on overall best
        if time.monotonic() < p4_end - 20:
            a_final_cd = coord_descent_fast(tracker[1], max_sweeps=3000,
                                            deadline=p4_end - 5,
                                            seed=seed + 9999)
            score_and_track(a_final_cd, tracker)
            print(f"  Final CD: J={compute_J(a_final_cd):.8f}", flush=True)

    # Safety: ensure we have a valid result
    if tracker[1] is None:
        tracker[1] = np.ones(512) / 512.0
        tracker[0] = float(evaluate_sequence(tracker[1].tolist()))

    total_time = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker[0]:.8f} n={tracker[1].size} time={total_time:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker[1]]
