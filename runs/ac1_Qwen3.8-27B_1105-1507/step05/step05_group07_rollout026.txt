```python
def optimize_coordinate_1d(i, a, conv, s, n):
    """Find optimal value for a[i] given all other coordinates fixed."""
    old = a[i]
    lo = max(0, i)
    hi = min(2*n-2, i+n-1)
    
    # Precompute: for each k in [lo, hi], the linear/quadratic coefficients
    # conv_k(x) = A_k * x + B_k (for k != 2i)
    # conv_k(x) = x^2 + C_k (for k == 2i)
    
    # M_unchanged
    if lo > 0:
        M_unch = conv[:lo].max()
    elif hi < 2*n-2:
        M_unch = conv[hi+1:].max()
    else:
        M_unch = 0.0
    if lo > 0 and hi < 2*n-2:
        M_unch = max(conv[:lo].max(), conv[hi+1:].max())
    
    # Build the affected functions
    # For each k in [lo, hi]:
    #   if k == 2*i: quadratic x^2 + (conv[k] - old^2)
    #   else: linear (2*a[k-i])*x + (conv[k] - 2*old*a[k-i])
    
    # Now minimize f(x) = 2n * max(M_unch, max_affected(x)) / (s - old + x)^2
    # for x >= 0
    
    # Strategy: golden section search on [0, x_max]
    # where x_max is chosen so that f(x_max) > f(0)
    
    d = s - old  # so denominator = (d + x)^2
    
    def f(x):
        if x < 0:
            return float('inf')
        denom = d + x
        if denom < 1e-30:
            return float('inf')
        # Compute max of affected conv coeffs
        max_val = M_unch
        for k in range(lo, hi+1):
            if k == 2*i:
                val = x*x + (conv[k] - old*old)
            else:
                val = 2*a[k-i]*x + (conv[k] - 2*old*a[k-i])
            if val > max_val:
                max_val = val
        return 2*n*max_val / (denom*denom)
    
    # Find a reasonable upper bound
    x_max = max(2*old + 1, 1.0)
    # Expand if needed
    while f(x_max) < f(0) and x_max < 100:
        x_max *= 2
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    a_g = 0
    b_g = x_max
    c_g = b_g - gr*(b_g - a_g)
    d_g = a_g + gr*(b_g - a_g)
    fc = f(c_g)
    fd = f(d_g)
    
    for _ in range(60):  # 60 iterations gives precision ~ (0.618)^60 ≈ 1e-13
        if fc < fd:
            b_g = d_g
            d_g = c_g
            fd = fc
            c_g = b_g - gr*(b_g - a_g)
            fc = f(c_g)
        else:
            a_g = c_g
            c_g = d_g
            fc = fd
            d_g = a_g + gr*(b_g - a_g)
            fd = f(d_g)
    
    x_opt = (a_g + b_g) / 2
    # Also check x = 0
    if f(0) < f(x_opt):
        x_opt = 0
    
    return x_opt
```

```python
def optimize_coord_analytical(i, a, conv, s, n):
    """Find optimal a[i] using analytical 1D optimization."""
    old = a[i]
    lo = max(0, i)
    hi = min(2*n-2, i+n-1)
    
    # M_unchanged (max of unaffected conv coefficients)
    M_unch = 0.0
    if lo > 0:
        M_unch = max(M_unch, conv[:lo].max())
    if hi < 2*n-2:
        M_unch = max(M_unch, conv[hi+1:].max())
    
    # Affected coefficients: for k in [lo, hi]
    affected_ks = np.arange(lo, hi+1)
    other_indices = affected_ks - i  # indices into a for the "other" term
    
    # Current values
    current_vals = conv[lo:hi+1].copy()
    
    # For each affected k, the function is:
    # k != 2i: f_k(x) = 2*a[k-i]*x + (conv_k - 2*old*a[k-i])  [linear: A_k*x + B_k]
    # k == 2i: f_k(x) = x^2 + (conv_k - old^2)  [quadratic: x^2 + C_k]
    
    A = 2 * a[other_indices]  # slopes for linear terms
    B = current_vals - 2 * old * a[other_indices]  # intercepts for linear terms
    # For k = 2i, A and B don't apply; we handle separately
    
    is_quad = (affected_ks == 2*i)
    C = np.where(is_quad, current_vals - old*old, 0.0)  # C_k for quadratic term
    
    # d = s - old (denominator is (d + x)^2)
    d = s - old
    
    # Strategy: find the top 5 affected coefficients at x=old
    # and also consider M_unch
    # For each, try to solve the 1D problem with that as the active constraint
    
    candidates = [0.0, old]  # always consider x=0 and x=old
    
    # Also try the "optimal" x for the current max coefficient
    max_idx = np.argmax(current_vals)
    if not is_quad[max_idx]:
        # Linear: A*x + B, optimal x = d - 2B/A (if A > 0)
        A_k = A[max_idx]
        B_k = B[max_idx]
        if A_k > 1e-15:
            x_opt = d - 2*B_k/A_k
            if x_opt > 0:
                candidates.append(x_opt)
    else:
        # Quadratic: x^2 + C, optimal x = C/d (if d > 0)
        C_k = C[max_idx]
        if d > 1e-15:
            x_opt = C_k / d
            if x_opt > 0:
                candidates.append(x_opt)
    
    # Try top 3 coefficients
    top_indices = np.argsort(current_vals)[-5:]
    for idx in top_indices:
        if is_quad[idx]:
            C_k = C[idx]
            if d > 1e-15:
                x_opt = C_k / d
                if x_opt > 0:
                    candidates.append(x_opt)
        else:
            A_k = A[idx]
            B_k = B[idx]
            if A_k > 1e-15:
                x_opt = d - 2*B_k/A_k
                if x_opt > 0:
                    candidates.append(x_opt)
    
    # Also try x values that equalize the top 2 coefficients
    if len(top_indices) >= 2:
        for p in range(len(top_indices)):
            for q in range(p+1, len(top_indices)):
                idx1, idx2 = top_indices[p], top_indices[q]
                # Solve f_k1(x) = f_k2(x)
                if not is_quad[idx1] and not is_quad[idx2]:
                    # A1*x + B1 = A2*x + B2
                    denom = A[idx2] - A[idx1]
                    if abs(denom) > 1e-15:
                        x_eq = (B[idx1] - B[idx2]) / denom
                        if x_eq > 0:
                            candidates.append(x_eq)
                elif is_quad[idx1] and not is_quad[idx2]:
                    # x^2 + C1 = A2*x + B2
                    # x^2 - A2*x + (C1-B2) = 0
                    A2, B2, C1 = A[idx2], B[idx2], C[idx1]
                    disc = A2*A2 - 4*(C1-B2)
                    if disc >= 0:
                        sqrt_disc = np.sqrt(disc)
                        x1 = (A2 + sqrt_disc) / 2
                        x2 = (A2 - sqrt_disc) / 2
                        if x1 > 0:
                            candidates.append(x1)
                        if x2 > 0:
                            candidates.append(x2)
                elif not is_quad[idx1] and is_quad[idx2]:
                    # A1*x + B1 = x^2 + C2
                    # x^2 - A1*x + (C2-B1) = 0
                    A1, B1, C2 = A[idx1], B[idx1], C[idx2]
                    disc = A1*A1 - 4*(C2-B1)
                    if disc >= 0:
                        sqrt_disc = np.sqrt(disc)
                        x1 = (A1 + sqrt_disc) / 2
                        x2 = (A1 - sqrt_disc) / 2
                        if x1 > 0:
                            candidates.append(x1)
                        if x2 > 0:
                            candidates.append(x2)
    
    # Also try to equalize with M_unch (if it's above the affected max at x=old)
    # max(A*x + B, M_unch) → optimal is where A*x + B = M_unch (if that gives lower f)
    for idx in range(len(A)):
        if not is_quad[idx] and A[idx] > 1e-15:
            # A*x + B = M_unch → x = (M_unch - B)/A
            x_eq = (M_unch - B[idx]) / A[idx]
            if x_eq > 0:
                candidates.append(x_eq)
    
    # Evaluate f at all candidates
    best_x = old
    best_f = 2*n*max(M_unch, current_vals.max()) / (d*d)
    
    for x_c in candidates:
        x_c = max(0.0, float(x_c))
        if x_c < 1e-15:
            x_c = 0.0
        # Compute f(x_c)
        denom = d + x_c
        if denom < 1e-30:
            continue
        # Max of affected
        vals = A * x_c + B
        if np.any(is_quad):
            quad_idx = np.where(is_quad)[0]
            vals[quad_idx] = x_c*x_c + C[quad_idx]
        max_val = max(M_unch, vals.max())
        f_val = 2*n*max_val / (denom*denom)
        if f_val < best_f:
            best_f = f_val
            best_x = x_c
    
    return best_x
```

```python
import numpy as np
```

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):

def optimize_coord_1d(i, a, conv, s, n, top_k=10):
    """
    Find optimal value for a[i] using analytical 1D optimization.
    Returns (optimal_x, new_conv_max, new_s).
    """
    old = a[i]
    if old == 0 and s == 0:
        return 0, conv, 0
    
    lo = max(0, i)
    hi = min(2 * n - 2, i + n - 1)
    
    # Unaffected max
    M_unch = 0.0
    if lo > 0:
        M_unch = max(M_unch, float(conv[:lo].max()))
    if hi < 2 * n - 2:
        M_unch = max(M_unch, float(conv[hi+1:].max()))
    
    # Affected range
    affected_len = hi - lo + 1
    other_idx = np.arange(lo, hi + 1) - i  # indices into a for the "other" term
    
    # Verify other_idx is valid (should be, by construction of lo, hi)
    # current values in affected range
    curr_vals = conv[lo:hi+1].copy()
    
    # Linear coefficients: for k != 2i, conv_k(x) = A_k * x + B_k
    A = 2.0 * a[other_idx]
    B = curr_vals - 2.0 * old * a[other_idx]
    
    # Quadratic: for k = 2i, conv_k(x) = x^2 + C
    quad_mask = (np.arange(lo, hi+1) == 2 * i)
    C = np.where(quad_mask, curr_vals - old * old, 0.0)
    
    d = s - old  # denominator base: (d + x)^2
    
    # Generate candidate x values
    candidates = [0.0]
    if old > 1e-15:

    else:
        top_idx = np.argsort(curr_vals)[-top_k:]

        if quad_mask[idx]:
            # Quadratic: x^2 + C, optimal x = C/d
            if d > 1e-15 and C[idx] > 0:
                x_opt = C[idx] / d
                if x_opt > 1e-12:
                    candidates.append(float(x_opt))
        else:
            # Linear: A*x + B, optimal x = d - 2B/A (if A > 0)
            if A[idx] > 1e-15:

                if x_opt > 1e-12:
                    candidates.append(float(x_opt))

                candidates.append(old * 2.0)
                candidates.append(old * 3.0)

    for p in range(len(top_idx)):

                # x^2 + C1 = A2*x + B2 → x^2 - A2*x + (C1-B2) = 0
                A2, B2, C1 = A[idx2], B[idx2], C[idx1]
                disc = A2*A2 - 4.0*(C1 - B2)
                if disc >= 0:
                    sq = np.sqrt(disc)
                    for x_eq in [(A2 + sq) / 2.0, (A2 - sq) / 2.0]:

                            candidates.append(float(x_eq))
            elif not quad_mask[idx1] and quad_mask[idx2]:
                A1, B1, C2 = A[idx1], B[idx1], C[idx2]

                        if x_eq > 1e-12:
                            candidates.append(float(x_eq))
            elif not quad_mask[idx1] and not quad_mask[idx2]:
                diffA = A[idx2] - A[idx1]
                if abs(diffA) > 1e-15:
                    x_eq = (B[idx1] - B[idx2]) / diffA
                    if x_eq > 1e-12:

                    candidates.append(float(x_eq))

                # x^2 + C = M_unch → x = sqrt(M_unch - C)

                if val > 0:
                    candidates.append(float(np.sqrt(val)))

            continue

        vals = A * x_c + B

def coord_descent_precise(a, max_sweeps, deadline, random_order=False, seed=0, verbose=True):

    best_a = a.copy()
    best_obj = cur_obj
    
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 50
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        sweep_changed = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            new_val = optimize_coord_1d(i, a, conv, s, n)
            
            if abs(new_val - a[i]) > 1e-12:
                old = a[i]
                delta = new_val - old
                
                # Update convolution
                lo = max(0, i)
                hi = min(2 * n - 2, i + n - 1)
                other_idx = np.arange(lo, hi + 1) - i
                conv[lo:hi+1] += 2.0 * delta * a[other_idx]

                kq = 2 * i
                if kq <= 2 * n - 2:
                    conv[kq] += delta * (old + new_val)  # (new-old)*(new+old) = new² - old²
                
                a[i] = new_val
                s = s - old + new_val  # should stay ~1
                sweep_changed = True
        
        # Renormalize periodically to prevent drift
        if sweep % 10 == 0:
            s = a.sum()
            if abs(s - 1.0) > 1e-6:
                a = a / s
                s = 1.0
                conv = fast_conv(a)
        
        cur_obj = 2.0 * n * conv.max() / (s * s)
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0

                print(f"    CD sweep {sweep}: J={cur_obj:.10f}", flush=True)

            no_improve += 1
        
        if no_improve > max_no_improve:
            if verbose:
                print(f"    CD converged at sweep {sweep}: J={best_obj:.10f}", flush=True)
            break
    
    best_a = np.maximum(best_a, 0)
    bs = best_a.sum()
    if bs > 0:
        best_a /= bs
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    src = np.maximum(src, 0)
    s = src.sum()
    if s > 0:
        src = src / s
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    out = np.interp(t_new, t_old, src)
    return np.maximum(out, 0)


def generate_cosine_init(n, freq=1, power=1.0, rng=None):
    """Generate a cosine-based initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    base = np.abs(np.cos(np.pi * freq * t)) ** power
    if rng is not None:
        noise = rng.normal(0, 0.05, n)
        base = np.maximum(base + noise, 0)
    s = base.sum()
    if s > 0:
        base /= s
    return base

def proposed_anneal(a, n, deadline, tracker, seed=0):
    """
    Simulated annealing wrapper around coordinate descent.
    Periodically perturb and re-optimize.
    """
    rng = np.random.default_rng(seed)
    current = a.copy()
    current = normalize(current)
    best = current.copy()
    best_obj = compute_J(best)
    tracker.update(best)
    
    temperature = 0.1
    cool_rate = 0.98
    iter_count = 0
    
    while time.monotonic() < deadline:
        iter_count += 1
        
        # Perturb
        if rng.random() < 0.7:
            # Small perturbation (Gaussian noise)
            noise_scale = temperature
            perturbed = current + rng.normal(0, noise_scale, n)
        else:
            # Larger perturbation (mix with random)
            other = rng.uniform(0, 1, n)
            mix = rng.uniform(0.1, 0.4)
            perturbed = (1 - mix) * current + mix * other
        
        perturbed = normalize(perturbed)
        
        # Quick coordinate descent on perturbed (limited)
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        opt = coord_descent_precise(perturbed, max_sweeps=50,
                                     deadline=time.monotonic() + min(remaining * 0.3, 30),
                                     random_order=True, seed=seed + iter_count,
                                     verbose=False)
        
        opt_obj = compute_J(opt)
        
        # Accept if better or with probability exp(-delta/T)
        delta = opt_obj - best_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(temperature, 1e-10)):
            current = opt
            if delta < 0:
                best = opt.copy()
                best_obj = opt_obj
                tracker.update(best)
        
        temperature *= cool_rate
    
    return best
