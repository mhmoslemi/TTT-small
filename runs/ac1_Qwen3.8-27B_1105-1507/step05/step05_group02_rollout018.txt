```python
def dual_ascent(a, deadline, max_iters=10000):
    """Solve min max_k (a*a)_k s.t. sum(a)=1, a>=0 using dual ascent"""
    a = np.maximum(a, 0).astype(np.float64)
    a /= a.sum()
    n = len(a)
    
    best_M = fast_conv(a).max()
    best_a = a.copy()
    
    # Initialize p (dual variable - distribution over lags)
    b = fast_conv(a)
    M = b.max()
    # p concentrated on active lags
    active = b >= M * 0.99
    p = np.zeros(2*n-1)
    p[active] = 1.0 / active.sum()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Primal step: solve QP min a^T Q_p a s.t. sum(a)=1, a>=0
        a_new = solve_qp_fista(n, p, a, max_iter=200)
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        
        # Dual step: update p based on new a
        b = fast_conv(a_new)
        M_new = b.max()
        
        # Soft update: p_k proportional to exp(b_k / temperature)
        # or hard: p_k = 1/|active| for active k
        tol = 1e-8 * M_new
        active_new = b >= M_new - tol
        p_new = np.zeros(2*n-1)
        p_new[active_new] = 1.0 / active_new.sum()
        
        # Blend p for stability
        alpha = 0.5  # mixing parameter
        p = alpha * p_new + (1 - alpha) * p
        
        a = a_new
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
    
    return best_a

def qp_matvec(a, p, n):
```

```python
def solve_qp_fista(n, p, a_init, max_iter=200):
    """Solve min a^T Q_p a s.t. sum(a)=1, a>=0 using FISTA
    where (Q_p a)_j = sum_m a_m * p_{m+j}
    """
    # Precompute FFT of p
    size = 1
    while size < 3 * n:
        size <<= 1
    P = np.fft.rfft(p, size)
    
    # Lipschitz constant of the gradient (2*Q_p)
    # ||Q_p||_2 <= max row sum of |Q_p| (since Q_p is positive)
    # Actually, since all entries are non-negative, ||Q_p||_2 <= max row sum = max col sum
    # Row j of Q_p has entries (Q_p)_{j,m} = p_{j+m} for valid m
    # Row sum for row j = sum_m p_{j+m} = sum of p from index j to j+n-1
    # Max row sum <= sum(p) = 1 (since p is a probability distribution)
    # So L = 2 * 1 = 2
    # Actually more precisely: L = 2 * max_j (sum_m p_{j+m})
    # But for safety, let's use L = 2 (since p is normalized)
    L = 2.0
    
    # Projected FISTA
    x = np.maximum(a_init, 0).copy()
    s = x.sum()
    if s > 0:
        x /= s
    else:
        x = np.ones(n) / n
    y = x.copy()
    t = 1.0
    
    for it in range(max_iter):
        # Gradient: 2 * Q_p @ y
        grad = 2.0 * _qp_matvec_fft(y, P, size, n)
        
        # FISTA step
        x_new = y - grad / L
        # Project onto {z: sum(z)=1, z>=0}
        x_new = _project_simplex(x_new)
        
        # Update y
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0)
        t = t_new
        x = x_new
        
        # Check convergence
        if it % 20 == 19:
            diff = np.max(np.abs(x_new - x))
            if diff < 1e-10:
                break
    
    return x
```

```python
def _project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Shape-based initialization with quick refinement
2. Smooth Nesterov gradient descent for warm start
3. Intensive LP mirror descent (main optimization)
4. Dual ascent via QP (alternative local method)
5. Multi-length exploration and final polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A(n, a):
    """Build sparse matrix A where A @ g = conv(a, g) (linear convolution)"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        cnt = j_max - j_min + 1
        rows.extend([k] * cnt)
        cols.extend(range(j_min, j_max + 1))
        # a[k-j] for j in range(j_min, j_max+1)
        # = a[k-j_max : k-j_min+1][::-1]
        idx = np.arange(k - j_max, k - j_min + 1)
        vals.extend(a[idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def project_simplex(v):
    """Project v onto {x: sum(x)=1, x>=0}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g)_k <= M, g >= 0"""
    n = len(a)
    A = build_A(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def smooth_max_grad(a, tau):
    """Compute gradient of smooth approximation to 2*n*max(conv(a,a))/(sum(a))^2
    (with sum(a) fixed to 1, this is just 2*n*smooth_max(conv(a,a)))"""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    # grad_j = 2 * sum_k p_k * a_{k-j} (with bounds)
    # = 2 * (Q_p @ a)_j
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad


def qp_matvec_fft(a, P_fft, size, n):
    """Compute Q_p @ a using precomputed FFT of p"""
    A = np.fft.rfft(a, size)
    result = np.fft.irfft(A.conj() * P_fft, size)[:n]
    return result


def solve_qp_fista(n, P_fft, size, a_init, max_iter=300, L=None):
    """Solve min a^T Q_p a s.t. sum(a)=1, a>=0 using FISTA"""
    if L is None:
        L = 2.0
    
    x = np.maximum(a_init, 0).copy()
    s = x.sum()
    if s > 0:
        x /= s
    else:
        x = np.ones(n) / n
    y = x.copy()
    t = 1.0
    
    for it in range(max_iter):
        grad = 2.0 * qp_matvec_fft(y, P_fft, size, n)
        x_new = y - grad / L
        x_new = project_simplex(x_new)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0)
        t = t_new
        x = x_new
    
    return x


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=2.0, tau_end=100.0,
                        lr_init=0.3, lr_end=0.01):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    v = np.zeros(n)
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad = smooth_max_grad(y, tau)
        b_max = fast_conv(a).max()
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def run_lp_mirror(a, deadline, max_iters=5000):
    """LP mirror descent - the main optimization engine"""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    last_imp_M = best_M

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        # Try multiple step sizes
        for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.02, 0.01]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-15:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break
    
    return best_a


def run_dual_ascent(a, deadline, max_iters=2000):
    """Dual ascent: alternate between QP (primal) and p-update (dual)"""
    a = np.maximum(a.astype(np.float64), 0)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    size = 1
    while size < 3 * n:
        size <<= 1
    
    # Initialize p
    b = fast_conv(a)
    M = b.max()
    p = np.zeros(2 * n - 1)
    active = b >= M - 1e-6 * M
    p[active] = 1.0 / active.sum()
    P_fft = np.fft.rfft(p, size)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Primal: solve QP min a^T Q_p a s.t. sum(a)=1, a>=0
        a_new = solve_qp_fista(n, P_fft, size, a, max_iter=150)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        
        b_new = fast_conv(a_new)
        M_new = b_new.max()
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
        
        # Dual: update p
        # Soft max with annealing
        tau = max(0.1, 5.0 / (np.sqrt(it + 1)))
        exp_b = np.exp(b_new / tau * n / 10)
        p_new = exp_b / exp_b.sum()
        # Blend
        alpha = 0.3
        p = alpha * p_new + (1 - alpha) * p
        P_fft = np.fft.rfft(p, size)
    
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)
