LP-based optimization with subgradient perturbations and spectral shaping
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

When the LP alternation gets stuck (S ≈ 1), it means that `a` is a fixed point of the map `a -> g/S` where `g` is the LP optimum. This is a local optimum of the peak minimization problem.

- For each pair (i, k*-i) that contributes to the peak, reduce the larger element

Multi-strategy LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternation, subgradient perturbations, spectral shaping, and multi-start.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize a to have sum 1, ensuring non-negativity."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=build_toeplitz(a), b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = float(np.sum(g))
            return g, S
    except Exception:
        pass
    return None, 0.0


def subgradient_step(a, top_k=1, alphas=None):
    """Subgradient step to reduce top-K convolution peaks with line search."""
    n = len(a)
    if alphas is None:
        alphas = [0.01, 0.02, 0.03, 0.05, 0.1]
    
    conv = compute_conv(a)
    top_indices = np.argsort(conv)[::-1][:top_k]
    
    # Compute subgradient (average of top-k peak gradients)
    grad = np.zeros(n)
    for k in top_indices:
        k = int(k)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(top_indices)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for alpha in alphas:
        a_try = a - alpha * grad
        a_try = np.maximum(a_try, 0)
        s = np.sum(a_try)
        if s < 1e-15:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_try.copy()
    
    return best_a, best_obj


def spectral_perturb(a, seed):
    """Apply random spectral perturbation."""
    n = len(a)
    rng = np.random.default_rng(seed)
    fa = rfft(a)
    mag = np.abs(fa)
    phase = np.angle(fa)
    
    # Random multiplicative perturbation
    pert = np.exp(rng.normal(0, 0.1, len(fa)))
    # Ensure symmetry for real signal
    pert = np.concatenate([[pert[0]], pert[1:]])
    
    mag_new = mag * pert
    fa_new = mag_new * np.exp(1j * phase)
    a_new = irfft(fa_new, n)
    a_new = np.maximum(a_new, 0)
    return normalize(a_new)


def peak_deflate(a, deflate_factor=0.03, num_peaks=3, top_pairs=20):
    """Deflate top contributing pairs for the top peaks."""
    n = len(a)
    conv = compute_conv(a)
    top_indices = np.argsort(conv)[::-1][:num_peaks]
    
    a_new = a.copy()
    for k in top_indices:
        k = int(k)
        # Compute all pairs contributing to this peak
        contributions = []
        for i in range(n):
            j = k - i
            if 0 <= j < n and i < j:
                contributions.append((a_new[i] * a_new[j], i, j))
        contributions.sort(key=lambda x: x[0], reverse=True)
        
        for idx in range(min(top_pairs, len(contributions))):
            val, i, j = contributions[idx]
            a_new[i] *= (1 - deflate_factor)
            a_new[j] *= (1 - deflate_factor)
    
    a_new = np.maximum(a_new, 0)

                    if obj_tmp < best_obj:

                        best_a = a_tmp.copy()
                        a = a_tmp.copy()
                    stall = 0

        if verbose and iter_count % 100 == 0:
            print(f"    LP it={iter_count}, obj={best_obj:.6f}, S={S:.5f}, stall={stall}, rem={remaining:.0f}s")

    # Final LP pass at relax=1.0
    conv = compute_conv(a)
    M = float(np.max(conv))
    g, S = solve_lp(a, M, relax=1.0)
    if g is not None and S > 1.0 + 1e-6:
        for t in [1.0, 0.7, 0.5]:
            a_try = normalize((1.0 - t) * a + t * g / S)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a, best_obj


def subgradient_long(a, time_budget):
    """Repeated subgradient steps."""
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        for top_k in [1, 2, 3, 5]:
            a_try, obj_try = subgradient_step(a, top_k=top_k,
                                              alphas=[0.003, 0.005, 0.01, 0.02, 0.05])
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
    return best_a, best_obj


def peak_deflate_long(a, time_budget):
    """Repeated peak deflation with varying parameters."""
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    while time.monotonic() - start < time_budget:
        improved = False
        for df in [0.01, 0.02, 0.03, 0.05]:
            for npk in [1, 2, 3]:
                for tp in [8, 12, 20]:
                    a_try = peak_deflate(a, df, npk, tp)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
        if not improved:
            break
    return best_a, best_obj


def phase_long(a, time_budget):
    """Repeated phase perturbation."""
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    seed = 0
    while time.monotonic() - start < time_budget:
        seed += 1
        for npk in [2, 3, 5]:
            for damp in [0.1, 0.2, 0.3]:
                a_try = phase_perturb(a, npk, damp)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
    return best_a, best_obj


def random_init(n, seed):
    """Diverse random initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    choice = rng.integers(0, 8)
    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a = a + amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.3)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 5:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            s = b * n // nblocks
            e = (b + 1) * n // nblocks
            a[s:e] = rng.uniform(0.3, 1.5)
    elif choice == 6:
        a = np.ones(n)
        a[:n // 2] *= rng.uniform(0.5, 1.2)
        a[n // 2:] *= rng.uniform(0.5, 1.2)
    else:
        a = rng.random(n) + 0.05
    return normalize(a)


def resample_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternation with multi-strategy perturbations and
    multi-start exploration over sequence lengths and initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent = normalize(parent)
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            best_score = parent_obj
            best_seq = parent.copy()
    except Exception:
        pass

    if best_seq is None:
        best_seq = normalize(np.ones(700, dtype=np.float64))
        best_score = compute_obj(best_seq)
        print(f"Default init (n=700 uniform): obj={best_score:.6f}")

    print(f"Initial best: {best_score:.6f}")

    # ---- Phase 1: Multi-start exploration ----
    print("\n=== Phase 1: Multi-start Exploration ===")
    p1_budget = min(250.0, (deadline - time.monotonic()) * 0.18)

    if best_seq is not None:
        np_ = len(best_seq)
        n_values = sorted(set([
            max(100, np_ - 200), max(100, np_ - 100), np_,
            np_ + 50, np_ + 100, np_ + 200, np_ + 300, np_ + 500
        ]))
    else:
        n_values = [300, 500, 700, 900, 1100, 1300]

    n_values = [nv for nv in n_values if 100 <= nv <= 1500]
    # Sort by proximity to parent length first
    if best_seq is not None:
        np_ = len(best_seq)
        n_values.sort(key=lambda x: abs(x - np_))
        n_values = n_values[:8]
    print(f"  n values: {n_values}")

    total_starts = 0
    for n in n_values:
        if time.monotonic() > deadline - p1_budget:
            break
        remaining = p1_budget - (time.monotonic() - start_time + (1500.0 - float(budget_s) + 30.0))
        # Simpler: just check deadline
        if time.monotonic() > deadline - 200:
            break

        num_starts = 3
        alloc_each = min(25.0, max(8.0, p1_budget / (len(n_values) * num_starts)))

        for s in range(num_starts):
            if time.monotonic() > deadline - 180:
                break
            total_starts += 1

            if s == 0 and best_seq is not None:
                a_start = resample_sequence(best_seq, n)
            else:
                a_start = random_init(n, seed + total_starts * 104729)

            time_budget_s = alloc_each
            if time.monotonic() > deadline - 100:
                time_budget_s = min(time_budget_s, (deadline - time.monotonic()) - 60)
            if time_budget_s < 3:
                break

            a_opt, obj_opt = lp_alternation(a_start, time_budget=time_budget_s)

            # Quick perturbation pass
            if time.monotonic() < deadline - 100:
                for perturb_func in [subgradient_step, None]:
                    if time.monotonic() > deadline - 100:
                        break
                    if perturb_func is not None:
                        a_opt2, obj2 = subgradient_step(a_opt, top_k=3,
                                                        alphas=[0.005, 0.01, 0.02])
                        if obj2 < obj_opt:
                            a_opt, obj_opt = a_opt2, obj2

            print(f"  n={n}, start={s}: obj={obj_opt:.6f}", end="")
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print("  ** UPDATE **")
            else:
                print()

    # ---- Phase 2: Intensive LP polishing ----
    print(f"\n=== Phase 2: Intensive LP Polishing ===")
    print(f"  Best before: {best_score:.6f}, n={len(best_seq)}")

    p2_budget = min(700.0, (deadline - time.monotonic()) * 0.45)
    a_cur, obj_cur = lp_alternation(best_seq, time_budget=p2_budget, verbose=True)
    if obj_cur < best_score:
        best_score = obj_cur
        best_seq = a_cur.copy()
        print(f"  After Phase 2 LP: {best_score:.6f}")
    else:
        print(f"  After Phase 2 LP: no improvement (best={best_score:.6f})")

    # ---- Phase 3: Multi-strategy perturbation + LP ----
    print(f"\n=== Phase 3: Multi-Strategy Perturbation ===")

    remaining = deadline - time.monotonic()
    if remaining > 60:
        strategies = [
            ("Subgrad", subgradient_long),
            ("PeakDeflate", peak_deflate_long),
            ("Phase", phase_long),
            ("LP", lp_alternation),
        ]

        for name, func in strategies:
            if time.monotonic() > deadline - 30:
                break
            alloc = min(remaining * 0.15, 80.0)
            alloc = max(10.0, alloc)
            if alloc < 5:
                break

            a_try, obj_try = func(best_seq, alloc)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  {name}: {best_score:.6f} ** UPDATE **")
            else:
                print(f"  {name}: {obj_try:.6f} (no improvement)")

            # Follow up with LP
            if time.monotonic() < deadline - 30:
                alloc_lp = min(40.0, (deadline - time.monotonic()) * 0.1)
                if alloc_lp > 5:
                    a_lp, obj_lp = lp_alternation(best_seq, time_budget=alloc_lp)
                    if obj_lp < best_score:
                        best_score = obj_lp
                        best_seq = a_lp.copy()
                        print(f"  {name}+LP: {best_score:.6f} ** UPDATE **")

    # ---- Phase 4: Try different n values from best ----
    print(f"\n=== Phase 4: n-Refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 60 and best_seq is not None:
        n_best = len(best_seq)
        n_candidates = sorted(set([max(100, n_best - 50), n_best,
                                    n_best + 50, n_best + 100]))
        for n_try in n_candidates:
            if time.monotonic() > deadline - 30:
                break
            if n_try == n_best:
                continue
            alloc = min(40.0, (deadline - time.monotonic()) - 30)
            if alloc < 10:
                break
            a_res = resample_sequence(best_seq, n_try)
            a_opt, obj_opt = lp_alternation(a_res, time_budget=alloc)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.6f} ** UPDATE **")
            else:
                print(f"  n={n_try}: {obj_opt:.6f}")

    # ---- Phase 5: Final intensive LP ----
    print(f"\n=== Phase 5: Final LP Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 15 and best_seq is not None:
        final_budget = min(remaining - 10, 200.0)
        if final_budget > 5:
            a_final, obj_final = lp_alternation(best_seq, time_budget=final_budget)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
            print(f"  Final: {best_score:.6f}")

    # ---- Validation ----
    print(f"\n=== VALIDATION ===")
    if best_seq is None:
        best_seq = np.ones(200) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        if val < best_score:
            best_score = val
        print(f"  evaluate_sequence: {val:.6f}")
    else:
        print("  WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0 / 100.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    return result
