"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP mirror descent
with multi-M exploration, coordinate descent, and adaptive perturbations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a, k_range=None):
    """Build sparse matrix A such that A @ g = conv(a, g)"""
    rows = []
    cols = []
    vals = []
    if k_range is None:
        k_range = range(2 * n - 1)
    for k in k_range:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(k_range), n))


def lp_best_response(a, M, time_limit=10.0):
    """Find g >= 0 maximizing sum(g) s.t. conv(a, g) <= M"""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_best_response_multi(a, M, time_limit_total=30.0):
    """Try multiple M values and return the best g (by resulting J after blend)."""
    n = len(a)
    best_g = None
    best_score = float('inf')
    
    M_values = [M * 0.99, M, M * 1.01, M * 1.05, M * 1.1]
    # Remove values that are too close
    unique_Ms = []
    for m in M_values:
        if m <= 0:
            continue
        if not unique_Ms or abs(m - unique_Ms[-1]) > 1e-10 * M:
            unique_Ms.append(m)
    
    t0 = time.monotonic()
    for i, M_i in enumerate(unique_Ms):
        remaining = time_limit_total - (time.monotonic() - t0)
        if remaining < 2:
            break
        g = lp_best_response(a, M_i, time_limit=min(remaining - 1, 8.0))
        if g is None:
            continue
        gs = g.sum()
        if gs < 1e-10:
            continue
        g = g / gs
        # Quick score: try a few blend values
        for t in [0.3, 0.5, 0.7]:
            a_t = (1 - t) * a + t * g
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = fast_conv(a_t).max()
            if M_t < best_score:
                best_score = M_t
                best_g = g.copy()
    
    return best_g, best_score


def smooth_max_grad(a, tau):
    """Compute smooth max gradient of ||conv(a,a)||_inf"""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Numerical stability
    shifted -= shifted.max()
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max


def run_smooth_descent(a, deadline, max_iters=3000, tau_start=5.0, tau_end=200.0,
                       lr_init=0.2, lr_end=0.005):
    """Smooth gradient descent with Nesterov momentum"""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.85
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M - 1e-12:
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    
    return best_a


def coordinate_descent_lp(a, deadline, n_coords=30, n_rounds=100):
    """Optimize subsets of coordinates using LP, fixing others."""
    n = len(a)
    a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(12345)
    
    for round_i in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        # Pick a random subset of coordinates to optimize
        idx = rng.choice(n, size=min(n_coords, n), replace=False)
        idx.sort()
        
        # Build LP: maximize sum(a[idx]) subject to:
        #   conv(a, a) <= best_M * 1.01 (allow slight increase)
        #   a[j] >= 0 for all j
        #   a[j] fixed for j not in idx
        # This is non-convex (quadratic constraints), so use a linear relaxation:
        # Instead, use the dual: for fixed a_fixed, find a_free >= 0 such that
        # conv(a, a) is minimized at the max point.
        
        # Simpler approach: use the smooth gradient to get direction, then
        # do a line search on just those coordinates
        b = fast_conv(a)
        b_max = b.max()
        active_k = np.argmax(b)
        
        # Compute gradient restricted to active coordinates
        k = active_k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad_full = np.zeros(n)
        grad_full[lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        
        # Line search on active coordinates
        best_t = 0
        best_val = b_max
        for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_try = a.copy()
            a_try[idx] = np.maximum(a_try[idx] - t * grad_full[idx], 0)
            a_try /= a_try.sum()
            M_try = fast_conv(a_try).max()
            if M_try < best_val:
                best_val = M_try
                best_t = t
        
        if best_t > 0:
            a[idx] = np.maximum(a[idx] - best_t * grad_full[idx], 0)
            a /= a.sum()
            M_new = fast_conv(a).max()
            if M_new < best_M:
                best_M = M_new
    
    return a, best_M

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Enhanced LP mirror descent with multi-M exploration and adaptive time
2. Smooth gradient descent (Nesterov) for basin escaping

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(a, M, time_limit=10.0):
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_best_response(a, M, time_limit=10.0):
    g = lp_solve(a, M, time_limit)
    if g is not None:
        gs = g.sum()
        if gs > 1e-10:
            return g / gs
    return None


def lp_with_multi_M(a, M, time_per_lp=6.0, multipliers=None):
    """Try multiple M scales, return best (g, blend_t, resulting_M)."""
    if multipliers is None:
        multipliers = [1.0, 1.02, 1.05, 1.08, 1.12]
    
    best_g = None
    best_t = 0.5
    best_score = float('inf')
    
    for mult in multipliers:
        M_try = M * mult
        if M_try <= 0:
            continue
        g = lp_solve(a, M_try, time_limit=time_per_lp)
        if g is None:

            M_t = fast_conv(a_t).max()
            if M_t < best_score:
                best_score = M_t
                best_g = g.copy()
                best_t = t
    
    return best_g, best_t, best_score


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted -= shifted.max()
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=5.0, tau_end=200.0,
                        lr_init=0.25, lr_end=0.005, beta=0.85):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M - 1e-12:
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:

    return best_a

def coordinate_descent_a(a, deadline, n_sub=40, n_rounds=300):
    """Optimize random subsets of coordinates via line search on the max convolution entry."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(777)
    
    for rd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_max = int(np.argmax(b))
        
        # Determine which coordinates affect position k_max
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        active_range = np.arange(lo, hi + 1)
        
        # Pick a subset from active range (or all if small)
        if len(active_range) <= n_sub:
            idx = active_range
        else:
            idx = rng.choice(active_range, size=n_sub, replace=False)
        
        # Gradient at the active position
        grad = np.zeros(n)
        grad[lo:hi+1] = 2.0 * a[k_max - hi:k_max - lo + 1][::-1]
        
        # Line search on idx
        best_t = 0
        best_val = b[k_max]
        for t in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
            a_try = a.copy()
            a_try[idx] = np.maximum(a_try[idx] - t * grad[idx], 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            M_try = fast_conv(a_try).max()
            if M_try < best_val:
                best_val = M_try
                best_t = t
        
        if best_t > 0:
            a[idx] = np.maximum(a[idx] - best_t * grad[idx], 0)
            a /= a.sum()
            M_new = fast_conv(a).max()
            if M_new < best_M:
                best_M = M_new
    
    return a, best_M


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                                          lr_init=0.15, lr_end=0.01)
                best.update(a_p, f"pert_{scale}")
                
                if time.monotonic() < p5_dl - 25:
                    lp_dl = min(time.monotonic() + 35, p5_dl)
                    a_lp = run_lp_mirror(a_p, lp_dl, max_iters=lp_it,
                                         use_multi_M=True,
                                         multipliers=[1.0, 1.05, 1.1])
                    best.update(a_lp, f"pert_{scale}_LP")
    
    # === PHASE 6: Length Exploration ===
    if best.seq is not None and time.monotonic() < deadline - 50:
        print(f"\n{'='*50}", flush=True)
        print(f"PHASE 6: Length Exploration", flush=True)
        print(f"{'='*50}", flush=True)
        n_best = best.seq.size
        for delta in [16, -16, 32, -32, 48, -48, 64, -64]:
            if time.monotonic() > deadline - 30:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 2048:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            a_ext = run_lp_mirror(a_ext, min(time.monotonic() + 30, deadline - 25),
                                  max_iters=25, use_multi_M=False)
            best.update(a_ext, f"len_{n_try}")
    
    # === FINAL: LP Polish ===
    if best.seq is not None and time.monotonic() < deadline - 3:
        print(f"\nFinal LP polish...", flush=True)
        a_fin = run_lp_mirror(best.seq, deadline, max_iters=40,
                              use_multi_M=True,
                              multipliers=[1.0, 1.02, 1.05])
        best.update(a_fin, "final")
    
    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)
    
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} t={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
