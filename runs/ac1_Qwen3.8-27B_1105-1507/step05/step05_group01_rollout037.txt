```python
    # Random phase in frequency domain
    fa = np.exp(1j * rng.uniform(0, 2*np.pi, n//2+1))
    # Set DC to real positive
    fa[0] = 1.0
    a = irfft(fa * np.abs(fa), n)  # Same as irfft(fa, n) since |fa|=1
    a = np.abs(a)  # Take absolute value to ensure non-negative
    a = np.maximum(a, 0.01)
    return a / np.sum(a)
```

```python
"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast gradient descent with FFT, diverse spectral initializations,
peak-targeted subgradient methods, and LP polishing. Key insight: the
objective equals 2n * max_k (a*a)_k / S^2, and the theoretical minimum
is ~1 (flat convolution). We aim to flatten the autocorrelation peak.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (S * S)


def normalize(a):
    """Normalize to sum=1, clip non-negative."""
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / S


def gradient_of_peak(a, k):
    """Compute gradient of (a*a)_k w.r.t. a. Result: 2*a[k-j] for valid j."""
    n = len(a)
    grad = np.zeros(n)
    kj = k - np.arange(n)
    valid = (kj >= 0) & (kj < n)
    grad[valid] = 2.0 * a[kj[valid]]
    return grad


def fast_gradient_descent(a, time_budget=60.0, momentum=0.85,
                          eta_start=0.05, eta_min=0.0001, verbose=False):
    """
    Fast gradient descent on max_k (a*a)_k with S(a)=1 constraint.
    Uses FFT for convolution, subgradient for max, momentum for convergence.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)  # momentum
    eta = eta_start
    stall = 0
    max_stall = 200

    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        conv = compute_conv(a)
        M = float(np.max(conv))
        obj = 2.0 * n * M  # since S=1
        
        if obj < best_obj * (1 - 1e-12):
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            # Random perturbation to escape
            rng = np.random.default_rng(it * 7919 + 42)
            scale = rng.uniform(0.05, 0.15)
            a = a * np.exp(rng.normal(0, scale, n))
            a = normalize(a)
            v = np.zeros(n)
            eta = eta_start
            stall = 0
            max_stall = max(50, max_stall // 2)
            continue

        # Find peaks (within 99.5% of max for subgradient)
        threshold = M * 0.995
        peak_mask = conv >= threshold
        peak_indices = np.where(peak_mask)[0]

        if len(peak_indices) == 0:
            peak_indices = np.array([int(np.argmax(conv))])

        # Compute subgradient
        grad = np.zeros(n)
        for k in peak_indices:
            grad += gradient_of_peak(a, int(k))
        grad /= len(peak_indices)

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad

        # Adaptive step size with backtracking
        improved = False
        for _ in range(5):
            a_new = a - eta * v
            a_new = np.maximum(a_new, 0.0)

            if obj_new < obj * (1 - 1e-10):

def spectral_init(n, seed, method='random_phase'):
    """Generate initial sequences with flat-ish spectral magnitude."""
    rng = np.random.default_rng(seed)
    nfft = n // 2 + 1

    if method == 'random_phase':
        # Random phase, flat magnitude
        phase = rng.uniform(0, 2 * np.pi, nfft)
        fa = np.exp(1j * phase)
        fa[0] = 1.0
        a = irfft(fa, n)
        # Smooth rectification
        a = np.maximum(a, 0)
        if np.sum(a) < 1e-10:
            a = np.abs(a)
        return normalize(a)

    elif method == 'chirp':
        # Quadratic phase chirp
        k = np.arange(n)
        alpha = rng.uniform(0.5, 5.0)
        a = np.cos(2 * np.pi * alpha * k * k / n + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0)
        if np.sum(a) < 1e-10:
            a = np.abs(a) + 0.01
        return normalize(a)

    elif method == 'multi_chirp':
        # Sum of multiple chirps
        nchirps = rng.integers(2, 6)
        a = np.zeros(n)
        k = np.arange(n)
        for _ in range(nchirps):
            alpha = rng.uniform(0.3, 8.0)
            phase = rng.uniform(0, 2 * np.pi)
            a += np.cos(2 * np.pi * alpha * k * k / n + phase)
        a = np.maximum(a, 0)
        if np.sum(a) < 1e-10:
            a = np.abs(a) + 0.01
        return normalize(a)

    elif method == 'exp_noise':
        # exp of smoothed noise
        noise = rng.normal(0, 1, n)
        # Smooth it
        win = max(3, n // 30)
        kernel = np.hanning(win)
        kernel /= kernel.sum()
        noise = np.convolve(noise, kernel, mode='same')
        a = np.exp(0.5 * noise)
        return normalize(a)

    elif method == 'bipartite':
        # Two-block structure with random ratio
        ratio = rng.uniform(0.3, 0.7)
        split = int(n * ratio)
        a = np.ones(n)
        a[:split] = rng.uniform(0.8, 1.5)
        a[split:] = rng.uniform(0.3, 1.0)
        # Add small noise
        a += rng.normal(0, 0.05, n)
        a = np.maximum(a, 0.01)
        return normalize(a)

    elif method == 'sinusoidal':
        # Sum of sinusoids with random frequencies
        k = np.arange(n)
        a = np.ones(n)
        nsin = rng.integers(2, 8)
        for _ in range(nsin):
            f = rng.uniform(0.5, 10.0)
            amp = rng.uniform(0.1, 0.5)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.sin(2 * np.pi * f * k / n + phase)
        a = np.maximum(a, 0.01)
        return normalize(a)

    else:
        a = rng.random(n) + 0.05
        return normalize(a)


def peak_targeted_reduction(a, num_tries=50, time_budget=10.0):
    """Reduce the top peaks by locally adjusting contributing elements."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for _ in range(num_tries):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Find top peaks
        sorted_idx = np.argsort(conv)[::-1]
        num_peaks_to_fix = min(5, len(sorted_idx))
        
        # Pick a peak to reduce
        k = int(sorted_idx[rng.integers(0, num_peaks_to_fix)])
        
        # Find elements contributing to this peak
        kj = k - np.arange(n)
        valid = (kj >= 0) & (kj < n)
        
        if not np.any(valid):
            continue
        
        # Randomly reduce some contributing elements
        valid_indices = np.where(valid)[0]
        num_to_change = rng.integers(1, max(2, len(valid_indices) // 4))
        chosen = rng.choice(valid_indices, size=min(num_to_change, len(valid_indices)), replace=False)
        
        a_new = a.copy()
        for j in chosen:
            a_new[j] *= rng.uniform(0.5, 0.9)
        
        # Redistribute: boost non-contributing elements
        non_contrib = np.zeros(n, dtype=bool)
        non_contrib[:] = True
        non_contrib[valid_indices] = False
        if np.any(non_contrib):
            boost_indices = np.where(non_contrib)[0]
            num_boost = rng.integers(1, max(2, len(boost_indices) // 4))
            boost_chosen = rng.choice(boost_indices, size=min(num_boost, len(boost_indices)), replace=False)
            for j in boost_chosen:
                a_new[j] *= rng.uniform(1.1, 1.5)
        
        a_new = normalize(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing: find direction g that maximizes sum(g) subject to (a*g)_k <= M."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget - 2:
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            A = build_toeplitz(a)
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': remaining * 0.3})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                S_g = float(np.sum(g))
                if S_g > 1.0 + 1e-8:
                    # Line search
                    improved = False
                    for t in [0.5, 0.7, 0.85, 1.0, 0.3]:
                        a_try = (1.0 - t) * a + t * g / S_g
                        a_try = np.maximum(a_try, 0)
                        S_try = np.sum(a_try)
                        if S_try < 1e-10:
                            continue
                        a_try = a_try / S_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj * (1 - 1e-10):
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break
                    if improved:
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
        except Exception:
            stall += 1

        if stall > 20:
            # Perturb
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            a = a * np.exp(rng.normal(0, 0.05, n))
            a = normalize(a)
            stall = 0

        if verbose:
            elapsed = time.monotonic() - start
            print(f"      LP it, obj={best_obj:.6f}, elapsed={elapsed:.1f}s")

    return best_a


def resample_to_n(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

            update_best(a_opt, f"init-{method}-n{n}")

    # ===== Phase 3: Peak-targeted reduction on best =====
    print(f"\n=== Phase 3: Peak Targeted Reduction ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        pt_budget = min(30.0, (deadline - time.monotonic()) * 0.15)
        a_pt = peak_targeted_reduction(best_seq, num_tries=100, time_budget=pt_budget)
        update_best_a = compute_obj(a_pt)
        if update_best_a < best_score:
            best_score = update_best_a
            best_seq = a_pt.copy()
            print(f"  ** Peak reduction: {best_score:.6f} **")

    # ===== Phase 4: LP polishing =====
    print(f"\n=== Phase 4: LP Polishing ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        lp_budget = min(120.0, (deadline - time.monotonic()) * 0.4)
        a_lp = lp_polish(best_seq, time_budget=lp_budget)
        update_best(a_lp, "LP")

    # ===== Phase 5: Final gradient polish (multiple rounds with perturbation) =====
    print(f"\n=== Phase 5: Final GD Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_rounds = 3
        per_round = remaining / (n_rounds + 1)

        for r in range(n_rounds):
            if time.monotonic() > deadline - 15:
                break
            # Small perturbation
            rng = np.random.default_rng(seed + 1000 + r * 777)
            a_pert = best_seq * np.exp(rng.normal(0, 0.02, len(best_seq)))
            a_pert = normalize(a_pert)

            gd_b = min(per_round * 0.7, 60.0)
            a_final = fast_gradient_descent(a_pert, time_budget=gd_b)
            update_best(a_final, f"final-GD-{r}")

            # Also try from current best without perturbation
            if time.monotonic() < deadline - 20:
                gd_b2 = min(per_round * 0.3, 30.0)
                a_final2 = fast_gradient_descent(best_seq, time_budget=gd_b2)
                update_best(a_final2, f"final-GD2-{r}")

    # ===== Phase 6: Try slightly different n values with best structure =====
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Phase 6: Multi-n Final ===")
        n_best = len(best_seq)
        for delta in [-50, -25, 25, 50, 100, -100, 150]:
            n_try = n_best + delta
            if n_try < 50 or n_try > 1500:
                continue
            if time.monotonic() > deadline - 20:
                break
            a_res = resample_to_n(best_seq, n_try)
            gd_b = min(30.0, (deadline - time.monotonic()) * 0.2)
            if gd_b < 10:
                break
            a_opt = fast_gradient_descent(a_res, time_budget=gd_b)
            update_best(a_opt, f"multin-{n_try}")

    # ===== Final validation =====
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
```
