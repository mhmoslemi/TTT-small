import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
        if a.size == 0 or a.sum() < 0.01:
            return False
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            print(f"  ** J={score:.8f} n={a.size} t={time.monotonic()-self.t0:.0f}s", flush=True)
            return True
        return False

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def build_conv_matrix(n, a):
    """Build the convolution matrix: (M @ g)[k] = (a*g)[k]"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def coord_descent_adaptive(a, max_sweeps, deadline, seed=0):
    """Coordinate descent with adaptive candidate grid and ternary refinement."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0
    
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    # Adaptive grid: 15 candidates
    coarse_grid = np.array([0, 0.05, 0.1, 0.2, 0.35, 0.5, 0.65, 0.8, 0.9, 1.0, 1.1, 1.3, 1.6, 2.0, 3.0])
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = rng.permutation(n) if sweep % 3 == 0 else range(n)
        sweep_improved = False
        
        for i in order:
            old = a[i]
            lo_k = max(0, i)
            hi_k = min(2 * n - 2, i + n - 1)
            a_slice = a[lo_k - i:hi_k - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2
            
            current_max = conv.max()
            current_obj = 2.0 * n * current_max / (s * s)
            
            if old < 1e-10:
                candidates = np.array([0.0, 1e-4, 5e-4, 1e-3, 5e-3, 0.01, 0.05])
            else:
                candidates = coarse_grid * old
                # Filter: skip values too close to old
                mask = ~np.isclose(candidates, old, rtol=1e-4)
                candidates = candidates[mask]
            
            best_val = old
            best_local = current_obj
            
            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                
                conv[lo_k:hi_k + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                
                new_s = s + delta
                if new_s < 1e-10:
                    obj = float('inf')
                else:
                    obj = 2.0 * n * conv.max() / (new_s * new_s)
                
                if obj < best_local:
                    best_local = obj
                    best_val = x
                
                conv[lo_k:hi_k + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
            
            # Ternary refinement around best_val
            if best_val != old:
                lo_b = max(0, best_val * 0.85)
                hi_b = best_val * 1.15
                
                for _ in range(6):
                    m1 = lo_b + (hi_b - lo_b) / 3
                    m2 = hi_b - (hi_b - lo_b) / 3
                    
                    objs = []
                    for xm in [m1, m2]:
                        delta = xm - old
                        conv[lo_k:hi_k + 1] += 2.0 * delta * a_slice
                        if valid_kq:
                            conv[kq] += delta * delta
                        new_s = s + delta
                        objs.append(2.0 * n * conv.max() / (new_s * new_s) if new_s > 1e-10 else float('inf'))
                        conv[lo_k:hi_k + 1] -= 2.0 * delta * a_slice
                        if valid_kq:
                            conv[kq] -= delta * delta
                    
                    if objs[0] < objs[1]:
                        hi_b = m2
                    else:
                        lo_b = m1
                
                refined = (lo_b + hi_b) / 2
                delta = refined - old
                conv[lo_k:hi_k + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                new_s = s + delta
                refined_obj = 2.0 * n * conv.max() / (new_s * new_s) if new_s > 1e-10 else float('inf')
                
                if refined_obj <= best_local:
                    a[i] = refined
                    s = new_s
                else:
                    # Rollback to best_val
                    conv[lo_k:hi_k + 1] -= 2.0 * delta * a_slice
                    if valid_kq:
                        conv[kq] -= delta * delta
                    delta = best_val - old
                    conv[lo_k:hi_k + 1] += 2.0 * delta * a_slice
                    if valid_kq:
                        conv[kq] += delta * delta
                    a[i] = best_val
                    s = s + delta
                sweep_improved = True
        
        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 80:
            break
    
    a = np.maximum(best_a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a

def lp_mirror_descent(a, deadline, max_iters=200):
    """LP mirror descent with backtracking line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = compute_J(a)
    no_improve = 0
    
    # Pre-build the matrix (it changes with a, so we need to rebuild each iter)
    # Actually, the matrix A depends on a, so we can't pre-build it.
    # But we can cache it if a doesn't change much.
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t0 = time.monotonic()
        
        # Compute current convolution and M
        b_conv = fast_conv(a)
        M = float(b_conv.max())
        M = max(M, 1e-10)
        
        # Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0
        A = build_conv_matrix(n, a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 60, 'presolve': True})
            if not r.success or r.x is None:
                break
            g = np.maximum(r.x, 0)
        except Exception:
            break
        
        if g.sum() < 1e-10:
            break
        
        g = g / g.sum()
        
        # Backtracking line search
        improved = False
        t = 0.5
        while t >= 0.001:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            if a_trial.sum() < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / a_trial.sum()
            obj_trial = compute_J(a_trial)
            if obj_trial < best_obj - 1e-14:
                a = a_trial
                best_obj = obj_trial
                best_a = a.copy()
                improved = True
                break
            t *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
        
        lp_time = time.monotonic() - t0
        remaining = deadline - time.monotonic()
        if remaining < lp_time * 2.5:
            break
    
    return best_a

def resample(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0)

def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample(parent, n)
        p = p / p.sum() if p.sum() > 0 else np.ones(n) / n
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1]:
            inits.append(np.maximum(p + rng.normal(0, sigma, n), 1e-4))
    
    inits.append(np.ones(n) / n)
    
    for freq in [1, 2, 3, 4, 7]:
        for amp in [0.1, 0.3, 0.5]:
            inits.append(np.maximum(1 + amp * np.cos(2 * np.pi * freq * t), 0.01) / n)
    
    for center in [0.3, 0.5, 0.7]:
        for width in [0.15, 0.25, 0.4]:
            inits.append(np.maximum(np.exp(-((t - center) / width) ** 2) + 0.1, 0.01) / n)
    
    # Smooth random
    ksize = max(5, n // 25)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.05, 0.01) / n)
    
    return inits

def simulated_annealing_refine(a, deadline, seed=0, n_trials=15):
    """Perturb and re-optimize to escape local minima."""
    best_a = a.copy()
    best_obj = compute_J(a)
    rng = np.random.default_rng(seed)
    n = len(a)
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        # Geometrically decreasing perturbation
        eps = 0.3 * (0.7 ** trial)
        a_pert = np.maximum(a + eps * rng.standard_normal(n), 0)
        a_pert = a_pert / a_pert.sum()
        
        # Quick coordinate descent
        a_opt = coord_descent_adaptive(a_pert, max_sweeps=500,
                                        deadline=min(time.monotonic() + 30, deadline),
                                        seed=seed + trial * 100)
        obj = compute_J(a_opt)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            a = a_opt  # Continue from new best
            print(f"    SA trial {trial}: J={obj:.8f} (eps={eps:.3f})", flush=True)
    
    return best_a
