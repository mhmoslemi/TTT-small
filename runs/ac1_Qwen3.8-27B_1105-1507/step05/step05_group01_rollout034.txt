def project_simplex(v, s=1.0):
    """Project v onto {x : x >= 0, sum(x) = s}."""
    n = len(v)
    # We want the closest point to v on the simplex scaled by s
    # First project onto {x >= 0, sum(x) = 1}, then scale by s
    u = v / s if s > 0 else v
    # Project u onto standard simplex
    theta = 0.0
    sorted_u = np.sort(u)[::-1]
    cssv = np.cumsum(sorted_u)
    rho = np.nonzero(sorted_u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    proj = np.maximum(u - theta, 0)
    return proj * s

    n = len(a)

    s = np.sum(a)
    a = a / s  # Normalize to sum = 1
    
    best_a = a.copy()

    momentum = np.zeros(n)

        if time.monotonic() - start > time_budget:

        T = T_schedule(it)
        lr = lr_schedule(it)

        fw = rfft(w, size2)
        conv_arw = irfft(far * fw, size2)
        grad_g = 2.0 * conv_arw[n-1 : 2*n-1]
        
        # Nesterov momentum
        momentum = 0.9 * momentum + 0.1 * grad_g
        a_new = a - lr * momentum
        
        # Project onto simplex {x >= 0, sum(x) = 1}

        obj_new, _ = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

"""
Smooth gradient descent with log-sum-exp, Nesterov momentum, adaptive annealing,
coordinate descent polish, and multi-n exploration for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses FFT-accelerated gradients and simulated
annealing restarts to escape local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def smooth_max_and_weights(conv, T):
    """Compute smooth max value and softmax weights."""
    c = conv - np.max(conv)
    exp_Tc = np.exp(T * c)
    w = exp_Tc / np.sum(exp_Tc)
    g_val = np.log(np.sum(exp_Tc)) / T + np.max(conv)
    return g_val, w


def compute_grad_smooth(a, T):
    """Compute gradient of smooth max of conv(a,a) with respect to a."""
    n = len(a)
    conv = compute_conv(a)
    g_val, w = smooth_max_grad_inner(conv, T)
    # grad[j] = 2 * sum_{l=0}^{n-1} a[l] * w[j+l]
    ar = a[::-1]
    size2 = 1
    while size2 < 3 * n - 1:
        size2 <<= 1
    far = rfft(ar, size2)
    fw = rfft(w, size2)
    conv_arw = irfft(far * fw, size2)
    grad = 2.0 * conv_arw[n - 1:2 * n - 1]
    return g_val, grad


def smooth_max_grad_inner(conv, T):
    """Helper: compute smooth max and weights from conv."""
    c = conv - np.max(conv)
    exp_Tc = np.exp(T * c)
    w = exp_Tc / np.sum(exp_Tc)
    g_val = np.log(np.sum(exp_Tc)) / T + np.max(conv)
    return g_val, w


def project_simplex(v, target_sum=1.0):
    """Project v onto {x : x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    u = v / target_sum
    sorted_u = np.sort(u)[::-1]
    cssv = np.cumsum(sorted_u)
    rho_mask = sorted_u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(rho_mask):
        return np.full(n, target_sum / n)
    rho = np.nonzero(rho_mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(u - theta, 0.0)
    return proj * target_sum


def nesterov_smooth_descent(a, time_budget, T_start=0.5, T_end=50.0,
                            n_iters_max=500000, verbose=False):
    """Nesterov-accelerated projected gradient descent on smooth max."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    momentum = np.zeros(n)
    beta = 0.9

    for it in range(n_iters_max):
        if time.monotonic() - start > time_budget:
            break

        # Annealing schedule for T
        frac = min(1.0, it / max(1, n_iters_max * 0.8))
        T = T_start * (T_end / T_start) ** (frac ** 0.7)

        # Learning rate: proportional to 1/(T * n)
        lr = 2.0 / (T * n * (1.0 + 0.1 * T))
        lr = min(lr, 0.05)

        # Nesterov lookahead
        y = a + beta * (a - a + momentum)  # y = a + beta * velocity
        # Actually standard Nesterov: y = a + beta_t * (a - a_prev)
        # Let's use a simpler momentum formulation
        # grad at y
        conv_y = compute_conv(y)
        _, grad_y = smooth_max_grad_inner(conv_y, T)
        # Full gradient: grad_g = 2 * cross_corr(a, w)
        # Actually we need gradient at y, not a
        ar_y = y[::-1]
        size2 = 1
        while size2 < 3 * n - 1:
            size2 <<= 1
        far_y = rfft(ar_y, size2)
        fw_y = rfft(grad_y[0] if isinstance(grad_y, tuple) else grad_y, size2)

        # Let me redo this more carefully
        # Gradient at y:
        c_y = conv_y - np.max(conv_y)
        exp_Tc_y = np.exp(T * c_y)
        w_y = exp_Tc_y / np.sum(exp_Tc_y)

        ar_y = y[::-1]
        size2 = 1
        while size2 < 3 * n - 1:
            size2 <<= 1
        far_y = rfft(ar_y, size2)
        fw_y = rfft(w_y, size2)
        conv_arw_y = irfft(far_y * fw_y, size2)
        grad_y = 2.0 * conv_arw_y[n - 1:2 * n - 1]

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad_y
        a_new = a - lr * momentum
        a_new = project_simplex(a_new, 1.0)

        # Nesterov: update y for next iteration
        # y_next = a_new + (beta_{t+1}/beta_t) * (a_new - a)
        a_prev = a
        a = a_new

        obj_new = compute_obj(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"      it={it}, T={T:.2f}, obj={best_obj:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_descent_polish(a, time_budget, max_rounds=1000, verbose=False):
    """Coordinate descent: try changing individual coordinates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for round_idx in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        # Try a random subset of coordinates
        num_tries = min(n, 50)
        indices = rng.choice(n, size=num_tries, replace=False)
        improved = False

        for j in indices:
            if time.monotonic() - start > time_budget:
                break

            # Try several perturbation sizes
            for delta_frac in [0.01, 0.05, 0.1, -0.01, -0.05, -0.1, 0.2, -0.2]:
                delta = delta_frac * a[j]
                if a[j] + delta < 0:
                    continue

                a_trial = a.copy()
                a_trial[j] += delta
                s_trial = np.sum(a_trial)
                if s_trial < 1e-10:
                    continue
                a_trial = a_trial / s_trial
                obj_trial = compute_obj(a_trial)

                if obj_trial < best_obj - 1e-12:
                    best_obj = obj_trial
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                    break

        if not improved and round_idx % 50 == 0:
            # Random perturbation to escape
            a_pert = a * np.exp(rng.normal(0, 0.05, n))
            a_pert = np.maximum(a_pert, 0)
            s_p = np.sum(a_pert)
            if s_p > 1e-10:
                a_pert = a_pert / s_p
                obj_p = compute_obj(a_pert)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_pert.copy()
                    a = a_pert

        if verbose and round_idx % 100 == 0:
            print(f"      CD round={round_idx}, obj={best_obj:.6f}")

    return best_a, best_obj


def chirp_init(n, seed=42):
    """Initialize with a chirp-like pattern."""
    rng = np.random.default_rng(seed)

    phase = rng.uniform(0, 2 * np.pi)
    # Quadratic chirp, take real part, add offset for positivity
    a = 1.0 + 0.5 * np.cos(2 * np.pi * alpha * k * k / (n * n) + phase)
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def decreasing_init(n, seed=42):

def cos_init(n, seed=42):
    """Initialize with cosine pattern."""
    rng = np.random.default_rng(seed)

    a = np.maximum(a, 0.01)

def random_smooth_init(n, seed=42):
    """Random smooth initialization."""
    rng = np.random.default_rng(seed)

    a = np.ones(n)

def lp_polish_step(a, M, relax=1.0):
    """Single LP-based direction step (from previous approach, simplified)."""
    from scipy.optimize import linprog
    n = len(a)
    m = 2 * n - 1

    # Build constraint matrix: (Toep(a) @ g)[k] = conv(a, g)[k] <= M * relax
    # conv(a, g)[k] = sum_i a[i] * g[k-i]
    # For fixed a, this is linear in g.
    # (A @ g)[k] = sum_i A[k,i] * g[i] where A[k,i] = a[k-i] if 0 <= k-i < n else 0
    A = np.zeros((m, n))
    for i in range(n):
        # g[i] contributes to conv[k] for k = i, i+1, ..., i+n-1
        # conv[k] gets contribution a[k-i] * g[i]
        for k in range(i, min(i + n, m)):
            A[k, i] = a[k - i]

    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = float(np.sum(g))
            if S > 1.0 + 1e-6:
                return g, S
    except Exception:
        pass
    return None, 0.0


def combined_polish(a, time_budget):

def annealing_search(a, time_budget, n_restarts=5):
    """Simulated annealing style: grad descent + perturb + repeat."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    restart = 0
    while time.monotonic() - start < time_budget - 10:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 30:
            break

        budget_this = min(remaining * 0.25, 60.0)
        T_start = 0.5 + 0.5 * restart
        T_end = 20.0 + 5.0 * restart

        a_cur, obj_cur = nesterov_smooth_descent(a, budget_this,
                                                 T_start=T_start, T_end=T_end,
                                                 n_iters_max=200000)
        if obj_cur < best_obj:
            best_obj = obj_cur
            best_a = a_cur.copy()
            a = a_cur.copy()
        else:
            # Perturb and try again
            a = a * np.exp(rng.normal(0, 0.08, n))
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()

        restart += 1

    return best_a, best_obj
