Smoothed gradient descent with adaptive annealing for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses log-sum-exp to smooth the max,
projected gradient descent on the simplex, and multiple diverse
initializations to avoid local minima.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute exact objective."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    cond = u - cssv / np.arange(1, n + 1) > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[0] + 1
    theta = cssv[rho - 1] / rho
    return np.maximum(v - theta, 0.0)


def smoothed_obj_grad(a, tau):
    """
    Compute smoothed objective and its gradient.
    a is assumed to be on the simplex (a >= 0, sum(a) = 1).
    Returns (f, grad) where f = 2n * LSUM(conv(a,a), tau)
    and grad is the gradient projected to the tangent space of the simplex.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = conv.max()
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    log_sum = cmax + tau * np.log(sum_exp)
    f = 2.0 * n * log_sum
    
    # Softmax weights
    p = exp_shifted / sum_exp
    
    # Gradient wrt a: grad_a_i = 2 * sum_k p_k * a_{k-i} (for valid k-i)
    # This is the cross-correlation (p, a)
    # q_i = sum_j p[i+j] * a_j for j=0..n-1, with p indexed 0..2n-2
    # Using FFT: q = ifft(conj(FFT(a)) * FFT(p))[:n]
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = rfft(a, N)
    fp = rfft(p, N)
    q = irfft(np.conj(fa) * fp, N)[:n]
    grad_a = 2.0 * q
    
    # Project gradient onto tangent space of simplex (sum of a is fixed)
    grad_projected = grad_a - np.mean(grad_a)
    
    return f, grad_projected


def projected_gradient_descent(a, max_iters=5000, tau_start=1.0, tau_end=0.01,
                                 time_budget=60.0, verbose=False):
    """
    Projected gradient descent on the simplex with adaptive annealing of tau.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Backtracking line search parameters
    step = 1.0
    c1 = 1e-4  # Armijo condition
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive tau: anneal from tau_start to tau_end
        progress = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        f, grad = smoothed_obj_grad(a, tau)
        
        # Gradient step with backtracking
        # Direction: -grad (already projected)
        direction = -grad
        
        # Try step size with backtracking
        accepted = False
        for ls_iter in range(20):
            a_new = project_simplex(a + step * direction)
            f_new, _ = smoothed_obj_grad(a_new, tau)
            if f_new <= f + c1 * step * np.dot(grad, direction):
                accepted = True
                break
            step *= 0.5
        
        if accepted:
            a = a_new
            # Check exact objective periodically
            if it % 50 == 0:
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose and it % 500 == 0:
                        print(f"    PGD it{it}: obj={best_obj:.8f}, tau={tau:.4f}, step={step:.4f}")
        
        step = min(step * 1.5, 2.0)  # Increase step if backtracking succeeded quickly
    
    # Final check
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a, best_obj

                if verbose and it % 1000 == 0:
                    print(f"    PGD it{it}: obj={best_obj:.8f}, tau={tau:.5f}, step={step:.3f}")

    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def chase_peak_smooth(a, n_steps=200, time_budget=20.0, verbose=False):
    """
    Chase-the-peak using smoothed softmax weights for gradient direction.
    More targeted than PGD: focuses on reducing the actual peak.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.03

    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = conv.max()
        # Softmax with small tau to focus on peak
        tau = 0.01 + 0.5 * step
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500.0, 500.0)
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            idx = int(np.argmax(conv))
            p = np.zeros(len(conv))
            p[idx] = 1.0
        else:
            p /= p_sum

        # Gradient direction
        N = 1
        while N < 3 * n:
            N <<= 1
        fa = rfft(a, N)
        fp = rfft(p, N)
        q = irfft(np.conj(fa) * fp, N)[:n]
        grad = 4.0 * n * q
        grad = grad - np.mean(grad)

        # Step
        a_new = project_simplex(a - step * grad)
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.05, 0.1)
        else:
            step *= 0.7
            # Try a smaller step
            a_new2 = project_simplex(a - step * grad)
            obj2 = compute_obj(a_new2)
            if obj2 < best_obj:
                best_obj = obj2
                best_a = a_new2.copy()
                a = a_new2

        if verbose and it % 50 == 0:
            print(f"    ChasePeak it{it}: obj={best_obj:.8f}, step={step:.4f}")

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def short_lp_polish(a, time_limit=15):
    """
    Quick LP-based polish: minimize max(conv(a,g)) s.t. sum(g)=1, g>=0.
    Uses the min-peak LP formulation.
    """
    n = len(a)
    if n > 800:
        return a.copy(), compute_obj(a)

    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-8,
                               'primal_feasibility_tolerance': 1e-8})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                # Blend with current
                best_a = a.copy()
                best_obj = compute_obj(a)
                for t in [0.3, 0.5, 0.7, 1.0]:
                    a_try = (1 - t) * a + t * g
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                return best_a, best_obj
    except Exception:
        pass
    return a.copy(), compute_obj(a)


def generate_gaussian_init(n, sigma_frac=0.3, center_frac=0.5):
    """Generate a Gaussian-shaped initial sequence."""
    x = np.arange(n)
    center = center_frac * (n - 1)
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def generate_ramp_init(n, direction=1):
    """Generate a ramp-shaped initial sequence (asymmetric)."""
    if direction == 1:
        a = np.linspace(0.1, 1.0, n)
    else:
        a = np.linspace(1.0, 0.1, n)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def generate_bimodal_init(n):
    """Generate a bimodal initial sequence (mass at edges)."""
    a = np.zeros(n)
    a[0:n//4] = 1.0
    a[3*n//4:n] = 1.0
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        parent = parent[parent > 0]  # remove zeros for cleaner optimization
        if parent.size > 0 and np.isfinite(parent).all():
            s = parent.sum()
            if s > 1e-10:
                parent = parent / s
                obj = compute_obj(parent)
                print(f"Parent (n={parent.size}): obj={obj:.6f}")
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_single(a, time_budget=60.0, verbose=False):
    """
    Full optimization pipeline for a single sequence:
    PGR anneal -> chase peak -> PGR again -> LP polish.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    # Stage 1: PGR with wide tau (explore)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        alloc = min(remaining * 0.3, 25.0)
        a1, obj1 = pgd_anneal(a, max_iters=5000, tau_start=2.0, tau_end=0.05,
                              time_budget=alloc, verbose=verbose)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            if verbose:
                print(f"    PGR1: {best_obj:.8f}")

    # Stage 2: Chase peak (targeted)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        alloc = min(remaining * 0.25, 20.0)
        a2, obj2 = chase_peak_smooth(best_a, n_steps=500,
                                     time_budget=alloc, verbose=verbose)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            if verbose:
                print(f"    Chase: {best_obj:.8f}")

    # Stage 3: PGR with narrow tau (exploit)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        alloc = min(remaining * 0.3, 25.0)
        a3, obj3 = pgd_anneal(best_a, max_iters=5000, tau_start=0.1, tau_end=0.001,
                              time_budget=alloc, verbose=verbose)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            if verbose:
                print(f"    PGR2: {best_obj:.8f}")

    # Stage 4: Short LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8 and n <= 700:
        a4, obj4 = short_lp_polish(best_a, time_limit=min(remaining * 0.4, 15))
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
            if verbose:
                print(f"    LP: {best_obj:.8f}")

    # Stage 5: Final chase peak
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        a5, obj5 = chase_peak_smooth(best_a, n_steps=200,
                                     time_budget=min(remaining * 0.5, 10.0))
        if obj5 < best_obj:
            best_obj = obj5
            best_a = a5.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses smoothed projected gradient descent with
    adaptive annealing, diverse initializations, chase-the-peak, and
    LP polishing. Explores multiple sequence lengths.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}")

    # Phase 1: Optimize parent (or default) thoroughly
    print(f"\n=== Phase 1: Optimize n={n_parent} ===")
    phase1_deadline = start_time + min(500.0, (deadline - start_time) * 0.35)

    if best_seq is not None:
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        # Multiple PGR passes with different hyperparameters
        for pass_idx in range(4):
            if time.monotonic() > phase1_deadline:
                break
            remaining = phase1_deadline - time.monotonic()
            if remaining < 10:
                break

            alloc = min(remaining * 0.35, 80.0)
            tau_s = [2.0, 0.5, 0.1, 0.05][pass_idx % 4]
            tau_e = [0.1, 0.02, 0.005, 0.001][pass_idx % 4]

            a_new, obj_new = pgd_anneal(a, max_iters=8000, tau_start=tau_s,
                                         tau_end=tau_e, time_budget=alloc,
                                         verbose=True)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  PGR pass {pass_idx}: {best_score:.8f}")

            # Chase peak
            remaining = phase1_deadline - time.monotonic()
            if remaining > 10:
                a_cp, obj_cp = chase_peak_smooth(a, n_steps=500,
                                                 time_budget=min(remaining * 0.2, 20.0))
                if obj_cp < best_score:
                    best_score = obj_cp
                    best_seq = a_cp.copy()
                    a = a_cp.copy()
                    print(f"  Chase pass {pass_idx}: {best_score:.8f}")

    # Phase 2: Diverse initializations at parent length
    print(f"\n=== Phase 2: Diverse inits at n={n_parent} ===")
    phase2_deadline = start_time + min(900.0, (deadline - start_time) * 0.6)

    rng = np.random.default_rng(seed)
    inits = []

    if best_seq is not None:
        inits.append(("parent", best_seq.copy()))

    inits.append(("gauss_narrow", generate_gaussian_init(n_parent, sigma_frac=0.15)))
    inits.append(("gauss_wide", generate_gaussian_init(n_parent, sigma_frac=0.4)))
    inits.append(("gauss_off", generate_gaussian_init(n_parent, sigma_frac=0.25, center_frac=0.3)))
    inits.append(("ramp_up", generate_ramp_init(n_parent, direction=1)))
    inits.append(("ramp_down", generate_ramp_init(n_parent, direction=-1)))
    inits.append(("bimodal", generate_bimodal_init(n_parent)))
    inits.append(("random", project_simplex(rng.normal(1, 0.5, n_parent))))
    inits.append(("random_sparse", project_simplex(rng.exponential(1, n_parent))))

    # Drifted parent
    if best_seq is not None:
        drift = best_seq * np.exp(rng.normal(0, 0.1, n_parent))
        inits.append(("drifted", project_simplex(drift)))

    for name, a_init in inits:
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 20:
            break

        print(f"  Init '{name}': {remaining:.0f}s left", end="")

        a_opt, obj_opt = optimize_single(a_init, time_budget=min(remaining * 0.3, 50.0),
                                         verbose=False)
        print(f" -> {obj_opt:.8f}", end="")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(" ** NEW BEST **")
        else:
            print()

    # Phase 3: Length exploration
    print(f"\n=== Phase 3: Length exploration ===")
    phase3_deadline = start_time + min(1200.0, (deadline - start_time) * 0.8)

    n_options = []
    for frac in [0.5, 0.6, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4]:
        n_try = int(n_parent * frac)
        if 100 <= n_try <= 1200 and n_try not in n_options:
            n_options.append(n_try)

    for n_try in sorted(n_options):
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        if remaining < 25:
            break

        if best_seq is not None:
            a_resampled = resample_sequence(best_seq, n_try)
        else:
            a_resampled = np.ones(n_try) / n_try

        # Also try Gaussian at this length
        a_gauss = generate_gaussian_init(n_try, sigma_frac=0.25)

        print(f"  n={n_try}: {remaining:.0f}s left", end="")

        a_opt1, obj1 = optimize_single(a_resampled, time_budget=min(remaining * 0.25, 40.0))
        a_opt2, obj2 = optimize_single(a_gauss, time_budget=min(remaining * 0.15, 25.0))

        obj_best = min(obj1, obj2)
        a_best_local = a_opt1 if obj1 <= obj2 else a_opt2
        print(f" -> {obj_best:.8f}", end="")

        if obj_best < best_score:
            best_score = obj_best
            best_seq = a_best_local.copy()
            print(" ** NEW BEST **")
        else:
            print()

    # Phase 4: Final intensive polish on the best
    print(f"\n=== Phase 4: Final polish ===")
    print(f"Best before final: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)
        a_final = best_seq.copy()
        s = np.sum(a_final)
        if s > 1e-10:
            a_final = a_final / s

        # Multiple polishing rounds
        for round_idx in range(8):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            a_pol, obj_pol = optimize_single(a_final, time_budget=min(remaining * 0.2, 40.0),
                                             verbose=False)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol.copy()
                print(f"  Round {round_idx}: {best_score:.8f}")
            else:
                # Try perturbation + re-optimize
                rng2 = np.random.default_rng(round_idx * 7919 + seed)
                a_pert = a_final * np.exp(rng2.normal(0, 0.03 + 0.01 * round_idx, n_best))
                a_pert = project_simplex(a_pert)
                a_pert_opt, obj_pert = optimize_single(a_pert, time_budget=min(remaining * 0.15, 25.0))
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert_opt.copy()
                    a_final = a_pert_opt.copy()
                    print(f"  Perturb {round_idx}: {best_score:.8f}")

        # Very final: narrow PGR + chase peak
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            a_narrow, obj_narrow = pgd_anneal(a_final, max_iters=10000,
                                              tau_start=0.01, tau_end=0.0005,
                                              time_budget=min(remaining * 0.3, 20.0))
            if obj_narrow < best_score:
                best_score = obj_narrow
                best_seq = a_narrow.copy()
                print(f"  Final narrow PGR: {best_score:.8f}")

            remaining = deadline - time.monotonic()
            if remaining > 5:
                a_chase, obj_chase = chase_peak_smooth(best_seq, n_steps=500,
                                                       time_budget=min(remaining * 0.5, 15.0))
                if obj_chase < best_score:
                    best_score = obj_chase
                    best_seq = a_chase.copy()
                    print(f"  Final chase: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
