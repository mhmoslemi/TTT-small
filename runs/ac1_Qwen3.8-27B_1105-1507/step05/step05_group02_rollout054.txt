```python
import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse matrix C such that C @ x = conv(x, a) (full convolution)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max 1^T b s.t. conv(b,a) <= M, b >= 0"""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 8, 'presolve': True})
    if r.success and r.x is not None and r.x.sum() > 1e-15:
        return np.maximum(r.x, 0)
    return None


def run_lp_minimax(a, deadline, max_iters=2000, verbose=True, start_time=None):

    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    last_report = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:

            print(f"      LP iter {it}: best_M={best_M:.8f}, no_imp={no_imp}, elapsed={elapsed:.0f}s", flush=True)
            last_report = it
        
        g = lp_best_response(a, M)
        if g is None:
            break
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Line search
        improved = False
        for t in [0.4, 0.25, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_t = (1.0 - t) * a + t * g

                best_M = M_t

                improved = True
                break

            # Try larger steps or direct replacement
            for t in [0.6, 0.7, 0.8, 0.9, 1.0]:
                a_t = (1.0 - t) * a + t * g
                a_t = np.maximum(a_t, 0.0)
                s = a_t.sum()
                if s < 1e-10:
                    continue
                a_t = a_t / s

                if M_t < best_M - 1e-14:
                    a = a_t
                    best_M = M_t
                    best_a = a.copy()
                    improved = True
                    break

        print(f"      LP done: best_M={best_M:.8f}, iterations={it}, elapsed={elapsed:.0f}s", flush=True)
    
    return best_a, best_M


def cvxpy_qp_active_set(a_init, M_init, deadline, n_constraints_max=200):
    """
    Solve the QP: min M s.t. sum(a)=1, a>=0, (a*a)[k]<=M for active k.
    Uses active-set: start with most violated constraints, add more as needed.
    """
    n = len(a_init)
    a = a_init.copy()
    M = M_init
    
    # Identify active constraints
    b = fast_conv(a)
    # Sort by value, take top n_constraints_max
    active_idx = np.argsort(b)[::-1][:n_constraints_max]
    
    for outer in range(10):
        if time.monotonic() > deadline:
            break
        
        # Set up QP
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        
        for k in active_idx:
            # (a*a)[k] = sum_j a[j] * a[k-j]
            terms = []
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                terms.append(a_var[j] * a_var[k - j])
            constraints.append(cp.sum(terms) <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, max_iter=500)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=5000)
            except Exception:
                break
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        
        a_new = a_var.value
        M_new = M_var.value
        
        if a_new is None or M_new is None:
            break
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        M_new = float(M_new)
        
        # Check improvement
        b_new = fast_conv(a_new)
        M_actual = float(b_new.max())
        
        if M_actual < M - 1e-8:
            a = a_new
            M = M_actual
            # Update active set
            active_idx = np.argsort(b_new)[::-1][:n_constraints_max]
        else:
            # No improvement, try adding more constraints
            violated = np.where(b_new > M * 1.001)[0]
            if len(violated) > 0:
                active_idx = np.concatenate([active_idx, violated])
                active_idx = np.unique(active_idx)
                active_idx = active_idx[:n_constraints_max]
            else:
                break
    
    return a, M


def freq_domain_init(n, rng):
    """Generate a sequence with approximately flat power spectrum."""
    # In frequency domain, create a flat-ish spectrum with random phases
    # Then take IFFT and project to non-negative
    freqs = n // 2 + 1
    # Create flat magnitude with some random variation
    mag = np.ones(freqs)
    # Add some structure to help asymmetry
    bias = np.linspace(1.0, 0.7, freqs)
    mag = bias * (1 + 0.2 * rng.normal(0, 1, freqs))
    mag = np.maximum(mag, 0.3)
    
    phases = rng.uniform(0, 2 * np.pi, freqs)
    phases[0] = 0  # DC
    phases[-1] = 0  # Nyquist (for even n)
    
    spectrum = mag * np.exp(1j * phases)
    a = np.fft.irfft(spectrum, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. High-speed active-set LP minimax (von Neumann alternating) with adaptive line search
2. Momentum-accelerated LP iterations
3. CVXPY QP with active-set constraint management
4. Multi-length exploration with diversity injection
Key innovations: active-set LP (faster), momentum acceleration, QP polish, freq-domain init
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time
import cvxpy as cp


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix_rows(n, k_list):
    """Build sparse matrix for selected constraint rows k_list."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_list):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(k - j)
    # Return index arrays so we can multiply by a[k-j]
    row_idx = np.array(rows)
    col_idx = np.array(cols)
    val_idx = np.array(vals)  # these are the positions in 'a' to pick
    return row_idx, col_idx, val_idx


def build_conv_matrix_from_a(n, a, k_list=None):
    """Build sparse matrix C such that C @ x = [conv(x,a)[k] for k in k_list]."""
    if k_list is None:
        k_list = list(range(2 * n - 1))
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_list):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(k_list), n))


def lp_best_response_active(a, M, n_active=150):
    """Fast LP: max sum(b) s.t. conv(b,a)[k] <= M for top n_active k, b >= 0."""
    n = len(a)
    b = fast_conv(a)
    # Select active constraints (top n_active by value)
    if n_active < len(b):
        threshold = np.sort(b)[::-1][n_active - 1]
        active = list(np.where(b >= threshold)[0])
    else:
        active = list(range(2 * n - 1))

    A = build_conv_matrix_from_a(n, a, active)
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 5, 'presolve': True})
    if r.success and r.x is not None and r.x.sum() > 1e-15:
        return np.maximum(r.x, 0)
    return None


def lp_best_response_full(a, M):
    """Full LP with all constraints."""
    n = len(a)
    A = build_conv_matrix_from_a(n, a, list(range(2 * n - 1)))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 8, 'presolve': True})
    if r.success and r.x is not None and r.x.sum() > 1e-15:
        return np.maximum(r.x, 0)
    return None


def run_lp_minimax(a, deadline, max_iters=5000, verbose=True, start_time=None,
                   use_momentum=True, momentum_beta=0.7):
    """
    LP-based minimax with momentum acceleration and adaptive line search.
    """
    if start_time is None:
        start_time = time.monotonic()
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    last_report = 0
    v = np.zeros(n)  # momentum velocity
    iters_done = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        iters_done = it + 1

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 300:
            break

        if verbose and (it - last_report) >= 200:
            elapsed = time.monotonic() - start_time
            print(f"      LP iter {it}: best_M={best_M:.9f}, no_imp={no_imp}, elapsed={elapsed:.0f}s", flush=True)
            last_report = it

        # Solve LP (use active set for speed)
        g = lp_best_response_active(a, M, n_active=120)
        if g is None:
            g = lp_best_response_active(a, M, n_active=250)
        if g is None:
            g = lp_best_response_full(a, M)
        if g is None:
            break

        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Compute direction (with or without momentum)
        if use_momentum:
            direction = g - a
            v = momentum_beta * v + direction
            # Try a few step sizes on the momentum direction
            step_dir = v
        else:
            step_dir = g - a

        # Line search
        improved = False
        if use_momentum:
            # For momentum, the "new point" is a + t * step_dir
            for t in [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
                a_t = a + t * step_dir
                a_t = np.maximum(a_t, 0.0)
                s = a_t.sum()
                if s < 1e-10:
                    continue
                a_t = a_t / s
                M_t = fast_conv(a_t).max()
                if M_t < best_M - 1e-14:
                    a = a_t
                    best_M = M_t
                    best_a = a.copy()
                    improved = True
                    break
        else:
            for t in [0.4, 0.25, 0.15, 0.1, 0.06, 0.03, 0.01]:
                a_t = (1.0 - t) * a + t * g
                a_t = np.maximum(a_t, 0.0)
                s = a_t.sum()
                if s < 1e-10:
                    continue
                a_t = a_t / s
                M_t = fast_conv(a_t).max()
                if M_t < best_M - 1e-14:
                    a = a_t
                    best_M = M_t
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Try direct convex combo without momentum
            for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.7, 0.8]:
                a_t = (1.0 - t) * a + t * g
                a_t = np.maximum(a_t, 0.0)
                s = a_t.sum()
                if s < 1e-10:
                    continue
                a_t = a_t / s
                M_t = fast_conv(a_t).max()
                if M_t < best_M - 1e-14:
                    a = a_t
                    best_M = M_t
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Periodic perturbation to escape plateaus
            if no_imp % 200 == 100:
                rng_local = np.random.default_rng(hash((n, it)) & 0xFFFFFFFF)
                scale = min(0.02, 0.005 * (no_imp / 100.0))
                a_pert = a + rng_local.normal(0, scale, n)
                a_pert = np.maximum(a_pert, 1e-6)
                a_pert /= a_pert.sum()
                M_pert = fast_conv(a_pert).max()
                if M_pert < best_M:
                    a = a_pert
                    best_M = M_pert
                    best_a = a.copy()
                    v = np.zeros(n)
                else:
                    decay = 0.5 ** (no_imp / 50.0)
                    best_a_local = best_a.copy()
                    a = best_a_local + 0.01 * rng_local.normal(0, 1, n)
                    a = np.maximum(a, 1e-6)
                    a /= a.sum()

    if verbose:
        elapsed = time.monotonic() - start_time
        print(f"      LP done: best_M={best_M:.9f}, iters={iters_done}, elapsed={elapsed:.0f}s", flush=True)

    return best_a, best_M


def cvxpy_qp_polish(a_init, M_init, deadline, max_constraints=80, max_outer=8):
    """
    QP: min M s.t. sum(a)=1, a>=0, (a*a)[k]<=M for selected k.
    Active-set method: start with top constraints, add violated ones.
    """
    n = len(a_init)
    a = a_init.copy()
    M = M_init

    b = fast_conv(a)
    active_idx = np.argsort(b)[::-1][:max_constraints].tolist()

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1, M_var >= 0]

        for k in active_idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            quad_terms = [a_var[j] * a_var[k - j] for j in range(j_min, j_max + 1)]
            constraints.append(cp.sum(quad_terms) <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)

        solved = False
        for solver in [cp.CLARABEL, cp.ECOS, cp.CVXOPT]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue

        if not solved or prob.status not in ['optimal', 'optimal_inaccurate']:
            break

        a_new = np.asarray(a_var.value).flatten()
        M_new = float(M_var.value)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        M_actual = float(fast_conv(a_new).max())

        if M_actual < M - 1e-7:
            a = a_new
            M = M_actual
            b_new = fast_conv(a)
            active_idx = np.argsort(b_new)[::-1][:max_constraints].tolist()
            print(f"      QP outer {outer}: M={M:.9f} (improved)", flush=True)
        else:
            violated = np.where(fast_conv(a_new) > M * 1.0001)[0].tolist()
            if violated:
                for v_k in violated[:10]:
                    if v_k not in active_idx and len(active_idx) < max_constraints:
                        active_idx.append(v_k)
            else:
                break

    return a, M


def freq_domain_init(n, rng):
    """Generate a sequence with approximately flat power spectrum."""
    freqs = n // 2 + 1
    mag = np.linspace(1.0, 0.6, freqs) * (1 + 0.15 * rng.normal(0, 1, freqs))
    mag = np.maximum(mag, 0.3)
    phases = rng.uniform(0, 2 * np.pi, freqs)
    phases[0] = 0
    if n % 2 == 0:
        phases[-1] = 0
    spectrum = mag * np.exp(1j * phases)
    a = np.fft.irfft(spectrum, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            old_score = self.score
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} (prev {old_score:.6f}) t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)
```
