```python
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return np.maximum(conv[:2 * n - 1], 0)  # numerical safety

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def softmax_gradient_step(a, beta, lr):
    """One step of soft-max gradient descent.
    a should be normalized so sum(a) = 1.
    Returns new a (also normalized).
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Numerically stable softmax
    conv_shifted = conv - np.max(conv)
    exp_vals = np.exp(beta * conv_shifted)
    p = exp_vals / np.sum(exp_vals)
    
    # Gradient: grad[m] = 2 * sum_j a[j] * p[m+j]
    # This is a cross-correlation of a with p
    grad = np.zeros(n)
    for m in range(n):
        # sum_{j=0}^{n-1} a[j] * p[m+j] where m+j < 2n-1
        max_j = min(n, 2*n-1-m)
        if max_j > 0:
            grad[m] = 2.0 * np.dot(a[:max_j], p[m:m+max_j])

    lambda_est = np.mean(grad)
    grad_centered = grad - lambda_est

    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:

    a_new = a_new / s
    return a_new
```

```python
    # So hankel([p0,p1,...,p_{n-1}], [p_{n-1}, p_n, ..., p_{2n-2}])

    # So hankel(p[:n], p[n-1:2n-1]) gives H[m,j] = p[m+j]
```

```python
def softmax_gradient_descent(a, n_max_steps=50000, beta_start=5, beta_end=50, 
                              lr_start=0.5, lr_end=0.01, time_budget=60.0):
    """Soft-max gradient descent to minimize max(conv(a,a)) subject to sum(a)=1, a>=0."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for step in range(n_max_steps):
        if time.monotonic() - start > time_budget:
            break
        
        # Progress parameters
        frac = step / max(1, n_max_steps - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start + (lr_end - lr_start) * frac
        
        # Compute soft-max weights
        conv = compute_conv(a)
        conv_shifted = conv - np.max(conv)
        exp_vals = np.exp(beta * conv_shifted)
        p = exp_vals / np.sum(exp_vals)
        
        # Compute gradient
        H = hankel(p[:n], p[n-1:2*n-1])
        grad = 2.0 * (H @ a)
        
        # Center the gradient (to maintain sum constraint)
        # At optimum, grad should be constant on the active set
        # Approximate: subtract the weighted mean
        active = a > 1e-10
        if np.sum(active) > 0:
            lam = np.sum(grad[active] * a[active]) / np.sum(a[active])
        else:
            lam = np.mean(grad)
        
        grad_centered = grad - lam
        
        # Update
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj
```

Hmm, but I'm worried about the learning rate. Let me think about what scale the gradient is.

With sum(a) = 1, each a[m] ≈ 1/n. The convolution values are approximately 1/(2n) (flat case). The soft-max p[k] is approximately 1/(2n-1) for flat case, or concentrated on peaks.

grad[m] = 2 * sum_j a[j] * p[m+j]

In the flat case: a[j] ≈ 1/n, p[k] ≈ 1/(2n). Then grad[m] ≈ 2 * sum_j (1/n) * (1/(2n)) = 2 * n * (1/n) * (1/(2n)) = 1/n.

So the gradient is O(1/n) when a is flat. The step size lr * grad is O(lr/n). For meaningful movement, we want lr/n ~ 1/n, so lr ~ 1. Let me set lr around 1-10.
