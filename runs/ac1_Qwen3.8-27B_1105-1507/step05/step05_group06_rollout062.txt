    Adam-optimized smoothed subgradient descent with beta annealing.
    """
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    anneal_start = int(max_iters * anneal_frac)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_J = compute_J(a)
    patience = 0
    patience_limit = 500

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal beta
        if it < anneal_start:
            beta = beta_start
        else:
            frac = (it - anneal_start) / max(1, max_iters - anneal_start)
            beta = beta_start * (beta_end / beta_start) ** frac

        # Compute smoothed gradient
        grad, smooth_val = smoothed_gradient_step(a, beta)

        # Adam update
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        # Adaptive learning rate based on beta
        step_lr = lr / (1.0 + 0.1 * beta)
        a_new = a - step_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s

        # Check improvement every 20 steps
        if it % 20 == 0:
            J = compute_J(a)
            if J < best_J - 1e-10:
                best_J = J
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

        # If stuck, perturb
        if patience > patience_limit:
            # Try random perturbation
            n_perturb = max(1, n // 10)
            idx = np.random.choice(n, n_perturb, replace=False)
            a[idx] *= np.random.uniform(0.5, 1.5, n_perturb)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 0.01:
                a = a / s
            # Reset Adam state partially
            m *= 0.5
            v *= 0.5
            patience = 0

    J_final = compute_J(a)
    if J_final < best_J:
        best_J = J_final
        best_a = a.copy()

    return best_a, best_J


def coordinate_descent_polish(a, deadline, max_passes=30):
    """
    Coordinate descent with exact incremental convolution updates.
    For each coordinate i, find best value by trying candidates.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    b = fast_conv(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        S = a.sum()
        b = fast_conv(a)
        improved = False
        order = np.random.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Outside max (unchanged)
            outside_parts = []
            if k_lo > 0:
                outside_parts.append(b[:k_lo].max())
            if k_hi < 2 * n - 2:
                outside_parts.append(b[k_hi + 1:].max())
            outside_max = max(outside_parts) if outside_parts else 0.0

            # Try candidates
            if ai < 1e-12:
                candidates = [0.01 / n, 0.05 / n, 0.1 / n]
            else:
                candidates = [0.0, ai * 0.5, ai * 0.7, ai * 1.3, ai * 1.6, ai * 2.0]

            best_cand = ai
            best_score = 2.0 * n * max(b.max(), outside_max) / (S * S)

            for cand in candidates:
                delta = cand - ai
                if abs(delta) < 1e-15:
                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue

                # Compute max over affected range
                inside_max = 0.0
                two_i = 2 * i
                for k in range(k_lo, k_hi + 1):
                    if k == two_i:
                        new_val = b[k] + 2.0 * ai * delta + delta * delta
                    else:
                        j = k - i
                        if 0 <= j < n:
                            new_val = b[k] + 2.0 * delta * a[j]
                        else:
                            new_val = b[k]
                    if new_val > inside_max:
                        inside_max = new_val

                new_max = max(outside_max, inside_max)
                score = 2.0 * n * new_max / (S_new * S_new)

                if score < best_score - 1e-14:
                    best_score = score
                    best_cand = cand

            if abs(best_cand - ai) > 1e-14:
                delta = best_cand - ai
                a[i] = best_cand
                S = S + delta
                # Update b incrementally
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += 2.0 * ai * delta + delta * delta
                    else:
                        j = k - i
                        if 0 <= j < n and j != i:
                            b[k] += 2.0 * delta * a[j]
                        elif j == i:
                            b[k] += 2.0 * delta * ai
                improved = True

        if not improved:
            break

        # Renormalize
        a = np.maximum(a, 0)
        S = a.sum()
        if S > 0.01:
            a = a / S
            J = compute_J(a)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    from scipy.interpolate import interp1d
    f = interp1d(t_old, src, kind='cubic', fill_value='extrapolate')
    result = f(t_new)
    return np.maximum(result, 0.0)


def random_perturb(a, sigma=0.05):
    """Apply random multiplicative perturbation."""
    a_new = a * np.random.uniform(1 - sigma, 1 + sigma, len(a))
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0.01:
        a_new = a_new / s
    return a_new


def lp_mirror_descent_fast(a, deadline, max_iters=1000):
    """
    Fast LP-free mirror descent: use subgradient of max (active set based)
    with multiplicative weight updates.
    """
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    lr = 0.3
    momentum = np.zeros(n)
    mom_rate = 0.7

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = conv.max()

        # Active set: lags within 1% of max
        threshold = M * 0.99
        active = conv >= threshold
        n_active = active.sum()
        if n_active == 0:
            active = conv == M
            n_active = active.sum()

        # Gradient: (2/n_active) * sum_{k in active} a[k-j] * 1[valid]
        # = (2/n_active) * (active_weights * a_reversed) via FFT
        c = compute_gradient_fft(active.astype(np.float64), a)
        grad = (2.0 / n_active) * c

        # Momentum
        momentum = mom_rate * momentum + (1 - mom_rate) * grad

        # Multiplicative update (mirror descent with entropy)
        a_new = a * np.exp(-lr * momentum)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J = compute_J(a_new)
        if J < best_J - 1e-12:
            best_J = J
            best_a = a_new.copy()
            a = a_new
        else:
            # Accept anyway with some probability (simulated annealing)
            if np.random.random() < 0.1:
                a = a_new

        # Decay learning rate
        lr *= 0.999

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses smoothed subgradient descent, LP-free mirror descent, and coordinate descent.
    """
    np.random.seed(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        if a is None or len(a) == 0:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256.0
        update_best(parent)

    # ===== PHASE 1: Smoothed subgradient descent with Adam =====
    p1_deadline = min(t_start + 300, deadline - 400)
    print(f"\nPhase 1: Adam Smoothed Subgradient (budget={p1_deadline - t_start:.0f}s)", flush=True)

    a = parent / parent.sum()
    a1, J1 = adam_smoothed_descent(a, p1_deadline, max_iters=10000,
                                    beta_start=5.0, beta_end=500.0, lr=0.3,
                                    anneal_frac=0.2)
    update_best(a1)

    # Polish with CD
    a1_cd = coordinate_descent_polish(a1, min(p1_deadline + 30, deadline - 350), max_passes=15)
    update_best(a1_cd)
    print(f"  Phase 1 done: J={best_score:.8f}", flush=True)

    # ===== PHASE 2: LP-free mirror descent (multiplicative weights) =====
    p2_deadline = min(t_start + 500, deadline - 400)
    print(f"\nPhase 2: Multiplicative Mirror Descent (budget={p2_deadline - t_start:.0f}s)", flush=True)

    base = best_seq if best_seq is not None else a1
    a2, J2 = lp_mirror_descent_fast(base, p2_deadline, max_iters=5000)
    update_best(a2)
    a2_cd = coordinate_descent_polish(a2, min(p2_deadline + 30, deadline - 350), max_passes=15)
    update_best(a2_cd)
    print(f"  Phase 2 done: J={best_score:.8f}", flush=True)

    # ===== PHASE 3: Multi-length exploration =====
    p3_deadline = min(t_start + 850, deadline - 300)
    print(f"\nPhase 3: Multi-length Exploration (budget={p3_deadline - t_start:.0f}s)", flush=True)

    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = sorted(set([
            int(n_base * 0.75), int(n_base * 1.25), int(n_base * 1.5),
            max(128, n_base // 2), min(2048, n_base * 2),
            256, 384, 512, 640, 768, 1024, 1280
        ]))
        target_lengths = [L for L in target_lengths if 128 <= L <= 2048 and L != n_base]

        for tgt_n in target_lengths[:6]:
            if time.monotonic() > p3_deadline:
                break
            phase_dl = min(time.monotonic() + 40, p3_deadline)

            a_res = resample_seq(best_seq, tgt_n)
            a_res = np.maximum(a_res, 1e-8)
            a_res = a_res / a_res.sum()

            # Quick optimization: subgradient + CD
            a_r, _ = adam_smoothed_descent(a_res, phase_dl, max_iters=2000,
                                            beta_start=5.0, beta_end=200.0, lr=0.3,
                                            anneal_frac=0.3)
            update_best(a_r)
            a_cd = coordinate_descent_polish(a_r, min(phase_dl + 15, p3_deadline), max_passes=10)
            update_best(a_cd)

        print(f"  Phase 3 done: J={best_score:.8f}", flush=True)

    # ===== PHASE 4: Diverse restarts with perturbation =====
    p4_deadline = min(t_start + 1150, deadline - 200)
    print(f"\nPhase 4: Diverse Restarts (budget={p4_deadline - t_start:.0f}s)", flush=True)

    if best_seq is not None:
        n_b = best_seq.size
        attempts = 5
        per_attempt_time = (p4_deadline - time.monotonic()) / max(1, attempts)

        for att in range(attempts):
            if time.monotonic() > p4_deadline:
                break
            phase_dl = min(time.monotonic() + per_attempt_time, p4_deadline)

            # Different perturbation strategies
            if att == 0:
                a_pert = random_perturb(best_seq, sigma=0.1)
            elif att == 1:
                a_pert = random_perturb(best_seq, sigma=0.05)
            elif att == 2:
                # Flip a random chunk
                a_pert = best_seq.copy()
                chunk_size = max(1, n_b // 8)
                start = np.random.randint(0, n_b - chunk_size)
                a_pert[start:start + chunk_size] *= np.random.uniform(0.3, 2.0)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
            elif att == 3:
                # Blend with uniform
                a_uni = np.ones(n_b) / n_b
                a_pert = 0.7 * best_seq / best_seq.sum() + 0.3 * a_uni
                a_pert = a_pert / a_pert.sum()
            else:
                # Add noise
                a_pert = best_seq + np.random.normal(0, 0.02, n_b)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()

            a_r, _ = adam_smoothed_descent(a_pert, phase_dl, max_iters=3000,
                                            beta_start=5.0, beta_end=300.0, lr=0.3,
                                            anneal_frac=0.25)
            update_best(a_r)
            a_cd = coordinate_descent_polish(a_r, min(phase_dl + 10, p4_deadline), max_passes=8)
            update_best(a_cd)

        print(f"  Phase 4 done: J={best_score:.8f}", flush=True)

    # ===== PHASE 5: Aggressive coordinate descent on best =====
    p5_deadline = deadline
    print(f"\nPhase 5: Final CD Polishing (budget={p5_deadline - t_start:.0f}s)", flush=True)

    if best_seq is not None:
        # Multiple CD passes with different random orderings
        for cd_round in range(3):
            if time.monotonic() > p5_deadline - 5:
                break
            remaining = p5_deadline - time.monotonic()
            a_cd = coordinate_descent_polish(best_seq, p5_deadline, max_passes=50)
            update_best(a_cd)
            if best_score >= best_seq is not None and best_seq is not None:
                # Do a subgradient restart from the CD result
                a_sg, _ = adam_smoothed_descent(a_cd, min(time.monotonic() + 60, p5_deadline - 10),
                                                 max_iters=3000, beta_start=10.0, beta_end=500.0,
                                                 lr=0.2, anneal_frac=0.3)
                update_best(a_sg)
                a_cd2 = coordinate_descent_polish(a_sg, min(time.monotonic() + 30, p5_deadline - 5),
                                                   max_passes=20)
                update_best(a_cd2)

    print(f"  Phase 5 done: J={best_score:.8f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128.0
        update_best(best_seq)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
