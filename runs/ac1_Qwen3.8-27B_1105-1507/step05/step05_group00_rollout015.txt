"""
Multi-scale smooth optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses softmax parameterization, log-sum-exp smoothing with adaptive temperature,
L-BFGS-B optimization, multi-restart strategy, and LP-based polishing.
Explores multiple sequence lengths and diverse initializations.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def fft_conv_self(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj_normalized(a):
    """Compute objective for normalized sequence (sum=1). Returns 2n*max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = fft_conv_self(a)
    max_c = float(np.max(c))
    return 2.0 * n * max_c / (s * s)


def smooth_obj_grad_z(z, tau):
    """
    Smooth objective and gradient w.r.t. z (log-amplitude parameters).
    a = softmax(z), objective = 2n * soft_max(conv(a,a)).
    """
    n = len(z)
    z_shift = z - np.max(z)
    exp_z = np.exp(z_shift)
    a = exp_z / np.sum(exp_z)

    c = fft_conv_self(a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    w = np.exp(shifted)
    w_sum = np.sum(w)
    obj_val = 2.0 * n * (c_max + tau * np.log(w_sum))
    w /= w_sum

    # Gradient of soft_max w.r.t. a: grad_a[j] = 2*(w*flip_a)[n-1+j] - w[2j]*a[j]
    flip_a = a[::-1]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    Fw = rfft(w, size)
    Ffl = rfft(flip_a, size)
    corr = irfft(Fw * Ffl, size)
    grad_a = 2.0 * corr[n - 1:n - 1 + n]

    # Diagonal correction
    k_diag = np.arange(0, 2 * n - 1, 2)
    i_diag = k_diag // 2
    valid = i_diag < n
    grad_a[i_diag[valid]] -= w[k_diag[valid]] * a[i_diag[valid]]

    grad_a *= 2.0 * n

    # Chain rule for softmax: grad_z[i] = a[i] * (grad_a[i] - dot(grad_a, a))
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a)

    return obj_val, grad_z


def exact_obj_from_z(z):
    """Compute exact objective from z parameterization."""
    z_shift = z - np.max(z)
    a = np.exp(z_shift)
    a /= np.sum(a)
    return compute_obj_normalized(a), a


def run_lbfgsb(z_init, tau_sequence, max_iter=5000, time_budget=30.0):
    """Run L-BFGS-B with decreasing temperature schedule."""
    z = z_init.copy()
    best_obj = np.inf
    best_z = z.copy()
    start = time.monotonic()

    for tau in tau_sequence:
        if time.monotonic() - start > time_budget:
            break

        func = lambda z_, t=tau: smooth_obj_grad_z(z_, t)
        try:
            res = minimize(func, z, method='L-BFGS-B', jac=True,
                           options={'maxiter': max_iter, 'gtol': 1e-8,
                                    'ftol': 1e-10, 'maxfun': max_iter * 2})
        except Exception:
            break

        z = res.x
        exact_obj, a = exact_obj_from_z(z)
        if exact_obj < best_obj:
            best_obj = exact_obj
            best_z = z.copy()

    return best_z, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_for_direction(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    if n > 1200:
        return None
    A = build_toeplitz(a)
    c_neg = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_neg, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

    n = len(a)
    if n > 1200:
        return a, compute_obj_normalized(a)

    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget - 2:
        conv = fft_conv_self(a)
        M = float(np.max(conv))

        g = solve_lp_for_direction(a, M, time_limit=min(10, time_budget * 0.3))
        if g is None:
            break

        # Try several blend parameters
        improved = False
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:

            a_try = a_try / s_try
            obj_try = compute_obj_normalized(a_try)

                a = a_try.copy()

                s = np.sum(a)

    start = time.monotonic()

    # Phase 1: L-BFGS-B with smooth objective
    z_init = np.log(a + 1e-30)
    conv0 = fft_conv_self(a)
    max_c0 = float(np.max(conv0))

    # Temperature schedule
    tau_0 = max_c0 * 0.3
    tau_seq = [tau_0 * (0.4 ** i) for i in range(8)]
    tau_seq = [max(t, max_c0 * 1e-5) for t in tau_seq]

    remaining = time_budget - (time.monotonic() - start)
    alloc1 = min(remaining * 0.5, 30.0)
    z_best, obj_smooth = run_lbfgsb(z_init, tau_seq, max_iter=3000, time_budget=alloc1)

    _, a_lbfgs = exact_obj_from_z(z_best)
    obj_lbfgs = compute_obj_normalized(a_lbfgs)
    if obj_lbfgs < best_obj:
        best_obj = obj_lbfgs
        best_a = a_lbfgs.copy()
        if verbose:
            print(f"    L-BFGS-B: obj={best_obj:.8f}")

    z_current = z_best

    # Phase 2: More L-BFGS-B with smaller tau
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        conv_now = fft_conv_self(best_a)
        max_c_now = float(np.max(conv_now))
        tau_seq2 = [max_c_now * 0.1 * (0.3 ** i) for i in range(6)]
        tau_seq2 = [max(t, max_c_now * 1e-6) for t in tau_seq2]

        alloc2 = min(remaining * 0.4, 25.0)
        z_best2, _ = run_lbfgsb(z_current, tau_seq2, max_iter=3000, time_budget=alloc2)
        _, a2 = exact_obj_from_z(z_best2)
        obj2 = compute_obj_normalized(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            z_current = z_best2.copy()
            if verbose:
                print(f"    L-BFGS-B2: obj={best_obj:.8f}")

    # Phase 3: LP polishing
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10 and n <= 1200:
        a_lp, obj_lp = lp_polish(best_a, time_budget=min(remaining * 0.3, 20.0))
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"    LP: obj={best_obj:.8f}")

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        a_cd, obj_cd = coordinate_descent_polish(best_a, time_budget=min(remaining * 0.3, 15.0))
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
            if verbose:
                print(f"    CD: obj={best_obj:.8f}")

    # Phase 5: Final L-BFGS-B polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8:
        z_final = np.log(best_a + 1e-30)
        conv_f = fft_conv_self(best_a)
        mc_f = float(np.max(conv_f))
        tau_final = [mc_f * 0.05 * (0.3 ** i) for i in range(4)]
        tau_final = [max(t, mc_f * 1e-7) for t in tau_final]
        alloc5 = min(remaining * 0.5, 15.0)
        z_pol, _ = run_lbfgsb(z_final, tau_final, max_iter=2000, time_budget=alloc5)
        _, a_pol = exact_obj_from_z(z_pol)
        obj_pol = compute_obj_normalized(a_pol)
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()
            if verbose:
                print(f"    Final L-BFGS-B: obj={best_obj:.8f}")

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = np.sum(a_new)
    if s > 0:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses multi-restart L-BFGS-B with smooth log-sum-exp
    objective, LP polishing, coordinate descent, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try to load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent = parent / np.sum(parent)
            pobj = compute_obj_normalized(parent)
            print(f"Parent (n={parent.size}): obj={pobj:.8f}")
            best_score = pobj
            best_seq = parent.copy()
    except Exception:
        parent = None
        print("No parent available.")

    n_parent = len(best_seq) if best_seq is not None else 700

    # Define sequence of n values to explore
    n_values = [n_parent]
    for offset in [-100, -50, 50, 100, 200, 300, 500]:
        nn = n_parent + offset
        if 100 <= nn <= 3000:
            n_values.append(nn)
    # Sort by preference: parent first, then nearby
    n_values = sorted(set(n_values), key=lambda x: abs(x - n_parent))

    init_types = ['random_log', 'bipolar_smooth', 'chirp', 'sine',
                  'cosine', 'random', 'gaussian_edges', 'triangle']

    # Phase 1: Optimize at parent length with multiple initializations
    print(f"\n=== Phase 1: Multi-restart at n={n_parent} ===")
    phase1_budget = (deadline - start_time) * 0.45

    for init_i, init_type in enumerate(init_types):
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if init_i == 0 and best_seq is not None:
            a_init = best_seq.copy()
        else:
            a_init = make_initialization(n_parent, init_type, rng)

        if init_i == 0:
            tag = "parent"
        else:
            tag = init_type

        # Add random perturbation for diversity
        if init_i > 0:
            a_init = a_init * np.exp(rng.normal(0, 0.1, n_parent))
            a_init = np.maximum(a_init, 0)
            s = np.sum(a_init)
            if s > 0:
                a_init = a_init / s

        alloc = min(phase1_budget / max(1, len(init_types) - init_i), 80.0)
        print(f"  Init '{tag}' ({alloc:.0f}s budget)...")

        a_opt, obj = optimize_single(n_parent, a_init, time_budget=alloc, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: obj={best_score:.8f} (init={tag}) **")

    # Phase 2: Explore other lengths
    print(f"\n=== Phase 2: Length exploration ===")
    phase2_budget = (deadline - start_time) * 0.35

    for n_try in n_values:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_parent:
            continue  # Already done

        # Resample best sequence to new length
        if best_seq is not None:
            a_init = resample_sequence(best_seq, n_try)
        else:
            a_init = make_initialization(n_try, 'random_log', rng)

        alloc = min(phase2_budget / max(1, len(n_values)), 60.0)
        print(f"  n={n_try} ({alloc:.0f}s budget)...")

        a_opt, obj = optimize_single(n_try, a_init, time_budget=alloc, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: n={n_try}, obj={best_score:.8f} **")

    # Phase 3: Intensive final polish on the best sequence
    print(f"\n=== Phase 3: Final intensive polish ===")
    print(f"Best before: {best_score:.8f}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()

        # Multiple rounds of polishing with perturbations
        a_current = best_seq.copy()
        for round_idx in range(5):
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            # Perturb
            scale = 0.02 + 0.01 * round_idx
            a_pert = a_current * np.exp(rng.normal(0, scale, len(a_current)))
            a_pert = np.maximum(a_pert, 0)
            s = np.sum(a_pert)
            if s > 0:
                a_pert = a_pert / s

            alloc = min(remaining * 0.2, 30.0)
            a_opt, obj = optimize_single(len(a_current), a_pert, time_budget=alloc,
                                         verbose=False)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                a_current = a_opt.copy()
                print(f"  Round {round_idx}: obj={best_score:.8f}")
            else:
                # Try LP direct
                if len(a_current) <= 1000:
                    a_lp, obj_lp = lp_polish(a_current, time_budget=min(alloc * 0.5, 15.0))
                    if obj_lp < best_score:
                        best_score = obj_lp
                        best_seq = a_lp.copy()
                        a_current = a_lp.copy()
                        print(f"  Round {round_idx} LP: obj={best_score:.8f}")

        # Final coordinate descent
        if time.monotonic() < deadline - 15:
            a_cd, obj_cd = coordinate_descent_polish(best_seq, time_budget=10.0)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: obj={best_score:.8f}")

    # Validate and return
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
