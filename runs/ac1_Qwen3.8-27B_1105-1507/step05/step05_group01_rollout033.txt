def build_toeplitz(a):
    """Build matrix A where A @ g gives conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_direction(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=build_toeplitz(a), b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = float(np.sum(g))
            return g, S
    except Exception:
        pass
    return None, 0.0

def random_init(n, seed):

        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        a = 1.0 + rng.uniform(0.2, 0.6) * np.sin(2 * np.pi * rng.uniform(1, 5) * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 3:

        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        # Two-frequency cosine
        f1, f2 = rng.uniform(0.5, 2), rng.uniform(2, 6)
        a = 1.0 + 0.3 * np.cos(2*np.pi*f1*k/n) + 0.3 * np.cos(2*np.pi*f2*k/n + rng.uniform(0, 2*np.pi))
    else:

def softmax_gradient_descent(a, time_budget, beta_start=5.0, beta_end=50.0,
                              lr_start=0.01, lr_end=0.001, seed=42):
    """Annealed soft-max gradient descent with normalization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Learning rate schedule
    total_iters = 500
    stall = 0
    patience = 30
    
    for it in range(total_iters):
        if time.monotonic() - start > time_budget - 1:
            break
        
        # Anneal beta
        frac = it / total_iters
        beta = beta_start * (beta_end / beta_start) ** frac
        # Learning rate
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Compute smooth objective and gradient
        f_smooth, grad = softmax_obj_and_grad(a, beta)
        
        # Gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Evaluate
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                a_try = a - lr * factor * grad
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    stall = 0
                    break
            
            if stall > 0:
                stall += 1
            
            if stall > patience:
                # Perturb to escape
                scale = rng.uniform(0.02, 0.05)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 0:
                    a = a / s2
                stall = 0
                patience = max(15, patience // 2)
    
    return best_a, compute_obj(best_a)


def lp_polish(a, time_budget=100.0, seed=42):

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.01)

    A = rfft(a, N)
    W = rfft(w_active, N)
    R = irfft(np.conj(A) * W, N)[:n]
    
    # Water-filling: find lambda such that sum_j max(0, R_j - lambda) = sum_a
    # Sort R in decreasing order
    R_sorted = np.sort(R)[::-1]
    cumsum = np.cumsum(R_sorted)
    # Find largest k such that R_sorted[k] > (cumsum[k] + sum_a) / (k+1)
    # i.e., R_sorted[k] * (k+1) > cumsum[k] + sum_a
    k_range = np.arange(1, n+1)
    condition = R_sorted[:n] * k_range > cumsum[:n] + sum_a
    if np.any(condition):
        k_star = np.max(np.where(condition)[0]) + 1
    else:
        k_star = 1  # At least one positive
    
    lambda_val = (cumsum[k_star-1] + sum_a) / k_star
    a_new = np.maximum(R - lambda_val, 0)
    s_new = np.sum(a_new)
    if s_new > 0:
        a_new = a_new / s_new
    return a_new

"""
Soft-max annealing with water-filling fixed-point iteration and LP corrections.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Key innovations: FFT-based gradient computation, water-filling update rule derived
from KKT conditions, and adaptive switching between gradient/LP methods.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def cross_corr_
fft(a, w):
    """
    Compute R[j] = sum_{i=0}^{n-1} a[i] * w[i+j] for j = 0..n-1
    using FFT. w has length 2n-1.
    """
    n = len(a)
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    A = rfft(a, N)
    W = rfft(w, N)
    C = irfft(np.conj(A) * W, N)
    return C[:n]


def softmax_grad(a, beta):
    """
    Compute smoothed objective and gradient.
    f = 2n * (1/beta) * log(sum exp(beta * conv_k)) / (sum a)^2
    Returns (f, grad).
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    # Numerically stable log-sum-exp
    beta_conv = beta * conv
    beta_max = np.max(beta_conv)
    exp_vals = np.exp(beta_conv - beta_max)
    log_sum = beta_max / beta + np.log(np.sum(exp_vals))
    weights = exp_vals / np.sum(exp_vals)

    g = log_sum  # soft-max value = (1/beta)*log(sum exp(beta*conv_k))
    f = 2.0 * n * g / (sum_a * sum_a)

    # Gradient: df/da_j = (4n/S^2) * [R_j - g/S]
    R = cross_corr_
fft(a, weights)
    grad = (4.0 * n / (sum_a * sum_a)) * (R - g / sum_a)

    return f, grad


def waterfilling_update(a, num_peaks=8, top_frac=0.1):
    """
    Water-filling fixed-point update based on KKT conditions.
    At optimum: for a_j > 0, R_j = const; for a_j = 0, R_j <= const.
    One step: a_new_j = max(0, R_j - lambda) with lambda chosen for normalization.
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))

    # Find active peaks
    n_active = max(3, min(num_peaks, int(top_frac * (2 * n - 1))))
    sorted_idx = np.argsort(conv)[::-1][:n_active]

    # Weights proportional to peak values (or uniform for robustness)
    peak_vals = conv[sorted_idx]
    weights = peak_vals / np.sum(peak_vals)

    # Build weight vector
    w_vec = np.zeros(2 * n - 1)
    w_vec[sorted_idx] = weights

    # Compute R_j = sum_i a[i] * w[i+j]
    R = cross_corr_
fft(a, w_vec)

    # Water-filling: find lambda such that sum max(0, R_j - lambda) = sum_a
    R_sorted = np.sort(R)[::-1]
    cumsum = np.cumsum(R_sorted)
    # Condition: R_sorted[k-1] > (cumsum[k-1] + sum_a) / k
    k_range = np.arange(1, n + 1, dtype=np.float64)
    condition = R_sorted * k_range > cumsum + sum_a
    if np.any(condition):
        k_star = int(np.max(np.where(condition)[0])) + 1
    else:
        k_star = 1

    lambda_val = (cumsum[k_star - 1] + sum_a) / k_star
    a_new = np.maximum(R - lambda_val, 0)
    s_new = np.sum(a_new)
    if s_new > 1e-10:
        a_new = a_new / s_new
    else:
        a_new = a.copy() / sum_a

    return a_new


def build_toeplitz(a):
    """Build matrix A where A @ g gives conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_direction(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=build_toeplitz(a), b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = float(np.sum(g))
            return g, S
    except Exception:
        pass
    return None, 0.0

def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 7)

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 2:
        f1 = rng.uniform(1, 3)
        f2 = rng.uniform(3, 7)
        f3 = rng.uniform(7, 15)
        a = 1.0 + 0.25 * np.cos(2*np.pi*f1*k/n) + 0.2 * np.cos(2*np.pi*f2*k/n + rng.uniform(0,2*np.pi)) + 0.15 * np.cos(2*np.pi*f3*k/n + rng.uniform(0,2*np.pi))
    elif choice == 3:
        # Smooth random
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        # Block structure
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 5:

        a = 1.0 + 0.3 * np.sin(np.pi * k / n) + 0.2 * np.cos(2 * np.pi * rng.uniform(1, 4) * k / n)
    else:

        base = 0.5 + 0.5 * np.cos(np.pi * k / (n - 1))
        f = rng.uniform(2, 6)
        a = base * (1.0 + 0.5 * np.cos(2*np.pi*f*k/n + rng.uniform(0, 2*np.pi)))

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a


def gradient_phase(a, time_budget, beta_start=5.0, beta_end=200.0,
                   mode='combined', seed=42):
    """
    Main gradient-based optimization phase.
    mode: 'grad' = soft-max gradient only, 'wf' = water-filling only, 'combined' = alternate
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    patience = 25
    total_iters = 800

    for it in range(total_iters):
        if time.monotonic() - start > time_budget - 0.5:
            break

        frac = min(1.0, it / total_iters)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = 0.02 * (0.01 / 0.02) ** frac

        if mode == 'wf' or (mode == 'combined' and it % 3 != 2):
            # Water-filling update
            a_wf = waterfilling_update(a, num_peaks=max(5, (2*n-1)//50), top_frac=0.05)
            # Blend with current
            for t in [1.0, 0.7, 0.5, 0.3]:
                a_try = (1 - t) * a + t * a_wf
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    stall = 0
                    break
        else:
            # Gradient step
            f_smooth, grad = softmax_grad(a, beta)
            for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
                a_try = a - lr_try * grad
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    stall = 0
                    break

        if stall > 0:
            stall += 1

        if stall > patience:
            # Escape: perturb
            scale = rng.uniform(0.02, 0.06)
            a_pert = a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            s_p = np.sum(a_pert)
            if s_p > 0:
                a_pert = a_pert / s_p
                obj_p = compute_obj(a_pert)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_pert.copy()
                    a = a_pert
            stall = 0
            patience = max(10, patience // 2)

        # Periodically try peak-targeted reduction
        if it % 40 == 39 and time.monotonic() - start < time_budget - 2:
            conv = compute_conv(a)
            top_idx = np.argmax(conv)
            a_pert = a.copy()
            for j in range(n):
                j2 = top_idx - j
                if 0 <= j2 < n:
                    a_pert[j] *= 0.95
            a_pert = np.maximum(a_pert, 0)
            s_p = np.sum(a_pert)
            if s_p > 0:
                a_pert = a_pert / s_p
                obj_p = compute_obj(a_pert)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_pert.copy()
                    a = a_pert

    return best_a, compute_obj(best_a)


def lp_phase(a, time_budget, seed=42):
    """LP-based polishing phase."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s > 0:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget - 1:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(best_a)
        M = float(np.max(conv))

        relax = 1.0 + 0.02 * min(stall, 30)
        relax = min(relax, 1.25)

        g, S = solve_lp_direction(best_a, M, relax=relax)

        if g is not None and S > 1.0 + 1e-8:
            improved = False
            for t in [0.5, 0.7, 0.85, 1.0, 0.3, 0.15, 0.6]:
                a_try = (1.0 - t) * best_a + t * g / S
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break

            if improved:
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 8:
            # Peak-targeted escape
            conv = compute_conv(best_a)
            sorted_idx = np.argsort(conv)[::-1]
            for _ in range(3):
                top_k = int(sorted_idx[rng.integers(0, min(5, len(sorted_idx)))])
                a_pert = best_a.copy()
                for j in range(n):
                    j2 = top_k - j
                    if 0 <= j2 < n:
                        a_pert[j] *= rng.uniform(0.85, 0.98)
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 0:
                    a_pert = a_pert / s_p
                    obj_p = compute_obj(a_pert)
                    if obj_p < best_obj:
                        best_obj = obj_p
                        best_a = a_pert.copy()
                        stall = 0
                        break
            if stall > 0:
                stall = 0

def remaining_or_full(total, frac):

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.01)

            print(f"    Using random init (obj={obj_random:.6f} vs resample {obj_resample:.6f})")

        # Combined optimization
        a_opt, obj_opt = combined_optimize(a_start, alloc, seed=seed)
        obj_opt = compute_obj(a_opt)

        print(f"  N={n}: obj={obj_opt:.6f}")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")
            seed += 9999

    # Phase 3: Final intensive polish
    print(f"\n=== Phase 3: Final Polish ===")
    print(f"  Best before: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  Polishing n={n_best}, {remaining:.0f}s remaining")

        a_cur = best_seq.copy()

        for round_num in range(8):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if round_num % 3 == 0:
                # Water-filling focused
                alloc = min(remaining * 0.2, 40.0)
                a_cur, obj_r = gradient_phase(a_cur, alloc, beta_start=10.0,
                                               beta_end=300.0, mode='wf',
                                               seed=seed + round_num * 50)
            elif round_num % 3 == 1:
                # LP focused
                alloc = min(remaining * 0.2, 40.0)
                a_cur, obj_r = lp_phase(a_cur, alloc, seed=seed + round_num * 50)
            else:
                # Gradient + blend
                alloc = min(remaining * 0.2, 40.0)
                a_cur, obj_r = gradient_phase(a_cur, alloc, beta_start=5.0,
                                               beta_end=100.0, mode='combined',
                                               seed=seed + round_num * 50)

            obj_r = compute_obj(a_cur)
            if obj_r < best_score:
                best_score = obj_r
                best_seq = a_cur.copy()
                print(f"  Round {round_num + 1}: {best_score:.6f}")

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
