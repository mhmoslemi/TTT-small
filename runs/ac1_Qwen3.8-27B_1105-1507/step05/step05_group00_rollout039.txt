def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp(a, M, time_limit=10):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

    g_at_low = C + D * x_low + E * x_low * x_low
    active = np.abs(g_at_low - np.max(g_at_low)) < 1e-12
    subgrad_max = np.max(D[active] + 2 * E[active] * x_low)
    if subgrad_max <= 0:

    # (expand x_high if needed)

  (the term a[i]*a[k-i] = a[i]*a[i] = a[i]^2 becomes x^2)
  
- If k != 2i:
  conv_new[k] = conv[k] - 2*old_val*a[k-i] + 2*x*a[k-i]
  (the two terms a[i]*a[k-i] and a[k-i]*a[i] together give 2*a[i]*a[k-i], which becomes 2*x*a[k-i])

So:
conv_new[k] = conv[k] + 2*a[k-i]*(x - old_val) for k != 2i
conv_new[k] = conv[k] + x^2 - old_val^2 for k = 2i

In the form conv_new[k] = C[k] + D[k]*x + E[k]*x^2:
- For k != 2i: C[k] = conv[k] - 2*a[k-i]*old_val, D[k] = 2*a[k-i], E[k] = 0
- For k = 2i: C[k] = conv[k] - old_val^2, D[k] = 0, E[k] = 1
- For k outside [i, i+n-1]: C[k] = conv[k], D[k] = 0, E[k] = 0

Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact 1D coordinate descent (bisection on convex sub-problem)
2. LP-based alternating direction
3. Subgradient method with soft-max and Nesterov momentum
4. Multi-length exploration and structured restarts

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    # Vectorized construction
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M, time_limit=5):
    """Solve max sum(g) s.t. Aepp(g) <= M, g >= 0 where Aepp*g = conv(a,g)."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None
