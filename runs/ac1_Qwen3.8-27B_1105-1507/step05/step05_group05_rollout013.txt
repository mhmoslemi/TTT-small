def lp_best_response(a, M, timeout=5.0):
    """
    Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': timeout, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def get_subgradient(a, b, rel_tol=1e-8):
    """
    Compute an averaged subgradient of max_k (a*a)[k] at a.
    Returns gradient vector of same shape as a.
    """
    n = len(a)
    b_max = b.max()
    threshold = b_max * (1 - rel_tol)
    active = np.where(b >= threshold)[0]
    
    if len(active) == 0:
        return np.zeros(n)
    
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            kj = k - j
            if kj == j:
                grad[j] += 2.0 * a[j]
            else:
                if 0 <= kj < n:
                    grad[j] += a[kj]
                if 0 <= kj < n:
                    grad[j] += a[j]
    
    grad /= len(active)
    return grad


def get_subgradient_fast(a, b, rel_tol=1e-8):
    """
    Faster subgradient using the structure:
    grad[j] = (1/|A|) * sum_{k in A} [a[k-j] + a[k-j]] (with special case k=2j)
    
    This can be computed as a convolution with the active mask.
    """
    n = len(a)
    b_max = b.max()
    threshold = b_max * (1 - rel_tol)
    active_mask = (b >= threshold).astype(np.float64)
    
    # The "full" gradient (without averaging) is:

    #

    # (when both indices are valid)

    # This is essentially 2 * (active_mask * a) convolved appropriately
    # Let me use a different approach: 
    # (a*a)[k] = sum_j a[j]*a[k-j]
    # d/d a[j] (a*a)[k] = a[k-j] (if k-j valid and k-j != j) + a[j] (if k-j valid and j is the "other" index)
    # Actually the cleanest way:
    # grad[j] = sum_k active_mask[k] * (a[k-j] if 0<=k-j<n) [from the a[j]*a[k-j] term]
    #          + sum_k active_mask[k] * (a[j] if 0<=k-j<n and k-j != j) [from the a[k-j]*a[j] term]
    # Hmm this is getting confusing. Let me just use the loop version but limit active set.
    
    active_k = np.where(active_mask > 0)[0]

        return np.zeros(n)

    for k in active_k:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)

        contributions = np.where(kj_range == j_range, 2.0 * a[j_range], a[kj_range] + a[j_range])
        grad[j_range] += contributions
    
    grad /= n_active
    return grad


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def coordinate_sweep(a, b, order, gold_iters=35):
    """
    One sweep of coordinate descent.
    a: current sequence (modified in place)
    b: current convolution (recomputed at start)
    order: permutation of indices
    Returns updated a and b.
    """
    n = len(a)
    S = a.sum()
    gr = (np.sqrt(5) - 1) / 2
    
    for i in order:
        old = a[i]
        
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        
        # Unaffected max (from current b, which reflects previous updates this sweep)
        B = 0.0
        if k_lo > 0:
            B = max(B, b[:k_lo].max())
        if k_hi < 2 * n - 2:
            B = max(B, b[k_hi + 1:].max())
        
        # Precompute linear and quadratic parts
        n_affected = k_hi - k_lo + 1
        c = np.zeros(n_affected)
        s = np.zeros(n_affected)
        quad_idx = -1
        
        for idx in range(n_affected):
            k = k_lo + idx
            if k == 2 * i:
                c[idx] = b[k] - old * old
                s[idx] = 0.0
                quad_idx = idx
            else:
                kj = k - i
                if 0 <= kj < n:
                    c[idx] = b[k] - 2 * old * a[kj]
                    s[idx] = 2 * a[kj]
                else:
                    c[idx] = b[k]
                    s[idx] = 0.0
        
        c_quad = c[quad_idx] if quad_idx >= 0 else 0.0
        
        # Objective function
        def objective(x_t):
            ns = S - old + x_t
            if ns <= 1e-15:
                return 1e15
            v = c + s * x_t
            m = B
            vm = v.max()
            if vm > m:
                m = vm
            if quad_idx >= 0:
                qv = c_quad + x_t * x_t
                if qv > m:
                    m = qv
            return m / (ns * ns)
        
        # Golden section search
        lo = 0.0
        hi = max(old * 2.0 + 0.05, np.sqrt(max(b[k_lo:k_hi+1].max(), 0.0)) + 0.05, 0.02)
        
        # Don't let hi be too large
        hi = min(hi, S * 2.0)
        
        a_pt = hi - gr * (hi - lo)
        b_pt = lo + gr * (hi - lo)
        fa = objective(a_pt)
        fb = objective(b_pt)
        
        for _ in range(gold_iters):
            if fa < fb:
                hi = b_pt
                b_pt = a_pt
                fb = fa
                a_pt = hi - gr * (hi - lo)
                fa = objective(a_pt)
            else:
                lo = a_pt
                a_pt = b_pt
                fa = fb
                b_pt = lo + gr * (hi - lo)
                fb = objective(b_pt)
        
        x_opt = (lo + hi) / 2.0
        
        # Check boundary
        f0 = objective(0.0)
        f_opt = objective(x_opt)
        if f0 < f_opt:
            x_opt = 0.0
        
        # Update
        new_val = x_opt
        a[i] = new_val
        delta = new_val - old
        
        # Update b incrementally
        if abs(delta) > 1e-15:
            for idx in range(n_affected):
                k = k_lo + idx
                if k == 2 * i:
                    b[k] += delta * (new_val + old)
                else:
                    kj = k - i
                    if 0 <= kj < n:
                        b[k] += 2 * delta * a[kj]
    
    # Renormalize
    S_new = a.sum()
    if S_new < 1e-10:
        a[:] = 1.0 / n
        b = fast_conv(a)
    else:
        a /= S_new
        # After normalization, b scales as b / S_new^2
        # But we also need to account for the actual values
        b = fast_conv(a)  # Recompute for correctness after normalization
    
    return a, b


def run_cd(a, max_sweeps, deadline, seed=0, gold_iters=35):
    """Run coordinate descent for up to max_sweeps or until deadline."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = np.arange(n)
        rng.shuffle(order)
        
        # Periodically re-sort to put "active" coordinates first
        if sweep % 10 == 0:
            b_cur = fast_conv(a)
            # Sort by local gradient magnitude (proxy for importance)
            b_max = b_cur.max()
            active_k = np.where(b_cur >= b_max * 0.999)[0]
            # Coordinates that appear in active k values
            coord_importance = np.zeros(n)
            for k in active_k:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                coord_importance[lo:hi+1] += 1
            order = np.argsort(-coord_importance)
        
        a, b = coordinate_sweep(a, fast_conv(a), order, gold_iters)
        
        M = b.max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            break
    
    return best_a


def run_lp_cd_alternation(a, max_iters, deadline, cd_sweeps=5, lp_timeout=3.0):
    """
    Alternate between LP best response and coordinate descent.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # LP best response
        g = lp_best_response(a, M, timeout=lp_timeout)
        if g is None:
            break
        
        # Try different step sizes
        improved = False
        best_trial_M = best_M
        best_trial_a = best_a
        
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_trial_a = a_trial.copy()
                improved = True
        
        if improved:
            a = best_trial_a
            best_a = a.copy()
            best_M = best_trial_M
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
        
        # Coordinate descent polish
        if time.monotonic() < deadline - 5:
            a = run_cd(a, cd_sweeps, deadline - 2, seed=it * 31 + 7)
            M_cd = fast_conv(a).max()
            if M_cd < best_M:
                best_a = a.copy()
                best_M = M_cd
        
    return best_a


def subgradient_step(a, step_size=0.02):
    """Take a step in the negative subgradient direction (projected to simplex tangent)."""
    n = len(a)
    b = fast_conv(a)
    grad = get_subgradient_fast(a, b, rel_tol=1e-6)
    
    # Project to tangent space of simplex (sum = 0)
    grad = grad - grad.mean()
    
    # Normalize
    gnorm = np.sqrt((grad ** 2).sum())
    if gnorm < 1e-15:
        return a.copy()
    grad = grad / gnorm
    
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def run_subgradient_cd(a, max_iters, deadline, step_size=0.02, cd_sweeps=3):
    """Alternate subgradient steps with coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Subgradient step with decreasing step size
        s_t = step_size / (1 + 0.01 * it)
        a_sg = subgradient_step(a, s_t)
        
        # CD polish
        if time.monotonic() < deadline - 2:
            a_cd = run_cd(a_sg, cd_sweeps, deadline - 1, seed=it * 13 + 1)
        else:
            a_cd = a_sg
        
        M = fast_conv(a_cd).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a_cd.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_cd
        
        if no_improve > 30:
            break
    
    return best_a
