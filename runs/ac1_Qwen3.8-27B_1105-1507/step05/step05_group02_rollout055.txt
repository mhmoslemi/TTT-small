"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smooth gradient descent with Nesterov acceleration
2. LP mirror descent with FFT-accelerated line search
3. Multi-restart with structured perturbations

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def fast_cross_conv(a, g):
    """Compute cross-correlation a*g efficiently."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fg = np.fft.rfft(g, size)
    return np.fft.irfft(fa * fg, size)[:2 * n - 1]

def compute_M_from_fft_fa(fa, t, fg):
    """Compute max of conv((1-t)*a + t*g, (1-t)*a + t*g) using precomputed FFTs."""
    fmix = (1 - t) * fa + t * fg
    conv = np.fft.irfft(fmix * fmix, len(fa))
    return conv.max()

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    """Build sparse Toeplitz matrix for the LP constraint (a@g) <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        if r.success and r.x is not None and r.x.sum() > 1e-15:

def smooth_max_grad(a, tau):
    """Compute smooth max gradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-11)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[j] += 2 * sum_k p[k] * a[k-j] for valid k
        # This is 2 * p-convolved-with-a, but only at valid positions
        # Actually: d/d a_j [sum_k p[k] * conv(a,a)[k]] = 2 * sum_k p[k] * a[k-j]
        # = 2 * (p conv a)[j] ... hmm let me think more carefully
        # conv(a,a)[k] = sum_j a[j]*a[k-j]
        # d/d a_j [conv(a,a)[k]] = a[k-j] + a[k-j] = 2*a[k-j] (when both indices valid)
        # So grad[j] = 2 * sum_k p[k] * a[k-j] where k-j is valid
        idx = k - np.arange(lo, hi + 1)  # k - j for j in [lo, hi]
        grad[lo:hi + 1] += 2.0 * p[k] * a[idx]
    return grad, b, b_max

def run_smooth_nesterov(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,
                        lr_init=0.4, lr_end=0.005):
    """Nesterov-accelerated smooth max descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y /= sy
        a = a_new

    return best_a

def run_lp_mirror_descent(a, deadline, max_iters=10000, line_search_points=7):
    """LP mirror descent with FFT-accelerated line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    iter_count = 0
    
    # Precompute FFT size
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        iter_count += 1
        
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break
        
        # Solve LP for best response
        g = lp_best_response(a, M, time_limit=5)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10 or gs <= 1.0 + 1e-10:
            # No improvement possible
            no_imp += 5  # accelerate convergence detection
            if no_imp > 80:
                break
            continue
        
        g = g / gs  # normalize to sum=1
        
        # FFT-accelerated line search
        size = fft_size
        fa = np.fft.rfft(a, size)
        fg = np.fft.rfft(g, size)
        
        # Try several t values
        t_candidates = np.array([0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8])
        best_t = 0.0
        best_M_t = M
        
        for t in t_candidates:
            fmix = (1 - t) * fa + t * fg
            conv_mix = np.fft.irfft(fmix * fmix, size)
            M_t = conv_mix[:2*n-1].max()
            if M_t < best_M_t:
                best_M_t = M_t
                best_t = t
        
        if best_t == 0.0:
            no_imp += 1
            if no_imp > 50:
                break
            continue
        
        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def make_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.005))
        inits.append(np.maximum(p * (1 + 0.15 * rng.normal(0, 1, n)), 0.005))
    inits.append(np.ones(n))
    # Raised cosine variants
    for width in [0.3, 0.5, 0.7, 1.0]:
        xc = (t - 0.5) / width
        inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * xc), 0.01))
    # Gaussian variants
    for sigma in [0.15, 0.25, 0.35, 0.5]:
        inits.append(np.exp(-(t - 0.5)**2 / (2 * sigma**2)) + 0.05)
    # Double bump
    inits.append(np.exp(-5*(t-0.3)**2) + np.exp(-5*(t-0.7)**2) + 0.05)
    # Power law edges
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        inits.append(np.maximum(t**alpha * (1-t)**alpha, 0.005))
    # Random smooth
    k = max(3, n // 50)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    return inits

    """Precompute the structure of the sparse matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)

def build_A_from_structure(n, a, rows, cols):

    # For each (k, j) pair, the value is a[k-j]

    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

c[k] = sum_j a[j] * a[j+k] (for valid j)

This is the "infinite" version. The DFT relationship is different.

For the linear convolution: (a * a)[k] = sum_j a[j] * a[k-j]
The DFT of the linear convolution is the product of DFTs (with appropriate zero-padding).

So if we zero-pad a to length 2n, take its DFT, square it, and inverse DFT, we get the linear autocorrelation.

To minimize the peak of the linear autocorrelation, we want |A_pad(f)|^2 to be as "spread out" as possible in the frequency domain, which would make the time-domain autocorrelation more concentrated... no wait, that's the wrong direction.

Actually:
- If |A(f)|^2 is concentrated at f=0 → autocorrelation is spread out (like a triangle)
- If |A(f)|^2 is spread out → autocorrelation is concentrated at k=0 (like a delta)

We want to minimize the PEAK of the autocorrelation. The DC component (f=0) gives A(0) = sum(a) = 1 (after normalization). So |A(0)|^2 = 1. This corresponds to the "mean" of the autocorrelation.

The peak of the autocorrelation is at k=0 (always, for non-negative a, since c[0] = sum a[j]^2 is the largest by Cauchy-Schwarz... wait, is that right?

c[0] = sum a[j]^2 >= (sum a[j])^2 / n = 1/n (by Cauchy-Schwarz)

For other k: c[k] = sum a[j] * a[k-j] <= sqrt(sum a[j]^2 * sum a[k-j]^2) = sqrt(c[0] * c[0]) = c[0] (by Cauchy-Schwarz)

Wait, that's not right either. c[k] = sum_j a[j] * a[k-j]. By Cauchy-Schwarz:
c[k] <= sqrt(sum_j a[j]^2 * sum_j a[k-j]^2) = sqrt(c[0] * c[0]) = c[0]

So c[0] is always the maximum! The peak is always at k=0.

Wait, that can't be right for the uniform sequence. For a = [1/n, ..., 1/n]:
c[0] = n * (1/n)^2 = 1/n
c[n-1] = n * (1/n)^2 = 1/n (for the center of the triangular shape)

So c[0] = c[n-1] for the uniform case. They're equal!

Let me recheck: for a = [1, 1, 1, 1] (not normalized):
c[0] = 1+1+1+1 = 4
c[1] = 1+1+1 = 3
c[2] = 1+1 = 2
c[3] = 1

Ah I see, I was computing the convolution, not the autocorrelation. For non-negative sequences, the convolution (a*a)[k] = sum_j a[j] * a[k-j] is the same as the "autocorrelation" in the sense that it's symmetric.

For a = [1,1,1,1]: (a*a) = [1, 2, 3, 4, 3, 2, 1]. The maximum is at k=3 = n-1, not at k=0!

So my Cauchy-Schwarz argument was wrong. Let me redo it.

(a*a)[0] = a[0]^2 (only one term)
(a*a)[n-1] = sum_{j=0}^{n-1} a[j] * a[n-1-j] (n terms)

So (a*a)[n-1] can be much larger than (a*a)[0]!

For non-negative a, (a*a)[k] = sum_j a[j] * a[k-j]. The maximum is not necessarily at k=0.

OK so the peak can be anywhere. For the uniform sequence, it's at k=n-1 (the center). For a sequence that's concentrated at the edges, it might be at other positions.

This means the problem is more complex than I initially thought. The peak could be at any lag.

Going back to the optimization: the LP approach handles this correctly because it constrains ALL lags, not just one.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LP mirror descent with precomputed matrix structure
2. Smooth gradient descent with Nesterov acceleration for initialization
3. Multi-restart with adaptive perturbations
4. Length exploration and final polish
Key improvements: FFT-based line search eliminates redundant convolutions,
precomputed sparse structure speeds up LP builds, and adaptive time allocation
focuses compute on the LP phase which gives the best local results.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_fft_size(n):
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    return size


def precompute_lp_structure(n):
    """Precompute row/col indices and value index array for the LP sparse matrix."""
    rows_list = []
    cols_list = []
    val_idx_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
            val_idx_list.append(k - j)
    return (np.array(rows_list, dtype=np.int32),
            np.array(cols_list, dtype=np.int32),
            np.array(val_idx_list, dtype=np.int32))


def solve_lp_best_response(a, M, structure, time_limit=5):
    """Solve: max sum(g) s.t. (a@g)[k] <= M for all k, g >= 0.
    Uses precomputed sparse structure for speed."""
    n = len(a)
    rows, cols, val_idx = structure
    vals = a[val_idx]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def smooth_max_grad(a, tau):
    """Compute gradient of smooth max (log-sum-exp) of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = k - np.arange(lo, hi + 1)
        grad[lo:hi + 1] += 2.0 * p[k] * a[idx]
    return grad, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=1.0, tau_end=150.0,
                        lr_init=0.35, lr_end=0.005):
    """Nesterov-accelerated smooth max descent on simplex."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y /= sy
        a = a_new

    return best_a


def run_lp_mirror_descent(a, deadline, max_iters=50000, structure=None,
                          t_candidates=None):
    """LP mirror descent with FFT-accelerated line search.
    
    At each iteration:
    1. Compute M = max(conv(a,a))
    2. Solve LP: max sum(g) s.t. (a@g) <= M, g >= 0
    3. If sum(g) > 1, normalize g and do line search over t in [0,1]
       to minimize max(conv((1-t)*a + t*g, (1-t)*a + t*g))
    4. Update a to the best point found
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    fft_size = get_fft_size(n)
    
    if structure is None:
        structure = precompute_lp_structure(n)
    
    if t_candidates is None:
        t_candidates = np.array([0.08, 0.12, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.65, 0.8])
    
    iter_count = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        iter_count += 1
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 100:
                break
        
        # Solve LP
        g = solve_lp_best_response(a, M, structure, time_limit=3)
        if g is None:
            break
        gs = g.sum()
        
        if gs <= 1.0 + 1e-8:
            # LP says no improvement possible (within tolerance)
            no_imp += 10
            if no_imp > 120:
                break
            continue
        
        g = g / gs  # Normalize to sum = 1
        
        # FFT-accelerated line search
        fa = np.fft.rfft(a, fft_size)
        fg = np.fft.rfft(g, fft_size)
        
        best_t = 0.0
        best_M_t = M
        
        for t in t_candidates:
            fmix = (1.0 - t) * fa + t * fg
            conv_mix = np.fft.irfft(fmix * fmix, fft_size)
            M_t = float(conv_mix[:2 * n - 1].max())
            if M_t < best_M_t - 1e-15:
                best_M_t = M_t
                best_t = t
        
        if best_t == 0.0:
            no_imp += 1
            if no_imp > 100:
                break
            continue
        
        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
        else:
            a /= s
    
    if iter_count > 0:
        pass  # Could add debug output
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initializations for a given length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.025, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.06 * rng.normal(0, 1, n)), 0.005))
        inits.append(np.maximum(p * (1 + 0.12 * rng.normal(0, 1, n)), 0.005))
    inits.append(np.ones(n, dtype=np.float64))
    # Raised cosine / window variants
    for width in [0.3, 0.4, 0.5, 0.6, 0.8, 1.0]:
        xc = (t - 0.5) / max(width, 0.01)
        inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * np.clip(xc, -1, 1)), 0.01))
    # Gaussian center
    for sigma in [0.2, 0.3, 0.4, 0.5]:
        inits.append(np.exp(-(t - 0.5) ** 2 / (2 * sigma ** 2)) + 0.05)
    # Double bump
    inits.append(np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2) + 0.03)
    # Power law
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        inits.append(np.maximum((t * (1 - t)) ** alpha, 0.001))
    # Smooth random
    k = max(3, n // 60)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    # Symmetric
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.2, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    # Edge-enhanced
    edge = np.maximum(1.0 - 2.0 * np.abs(t - 0.5), 0.0)
    inits.append(0.3 + 0.7 * edge)
    return inits
