Actually, let me simplify my approach. Since the problem is scale-invariant, I'll normalize to sum(a)=1 at each step. Then the objective is just 2n * max(conv), and the gradient (of the soft-max version) is simpler.

With sum(a) = 1 (maintained by projection):
- Objective: f(a) = 2n * m_T(a) where m_T = T * log(sum exp(conv/T))
- Gradient: d m_T / d a[i] = sum_k w_k * (a[k-i]*I(0<=k-i<n) + a[i-k]*I(0<=i-k<n))

But we also need to project back to sum(a) = 1, which means we should use the gradient projected onto the hyperplane sum(a) = 1:
- grad_proj[i] = grad[i] - mean(grad)

Then take a step: a_new = a - alpha * grad_proj, project to a >= 0 and sum = 1.

Let me now write the complete solution:

```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa*fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0*n*max_b/(sum_a*sum_a)

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n+m-1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa*fb, size)[:n+m-1]

def soft_max_gradient(a, T):
    """Compute gradient of soft-max of conv(a,a) w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    w = np.exp((conv - cmax) / T)
    w_sum = np.sum(w)
    w /= w_sum
    m_T = cmax + T * np.log(w_sum)  # soft-max value
    
    # part1[i] = sum_j a[j]*w[i+j] for j=0..n-1 = np.correlate(w, a, 'valid')
    part1 = np.correlate(w, a, mode='valid')  # length n
    
    # part2[i] = sum_j a[j]*w[i-j] for valid = convolve(a, w)[i] for i=0..n-1
    full_conv = fft_convolve(a, w)  # length n + (2n-1) - 1 = 3n-2
    part2 = full_conv[:n]
    
    grad = part1 + part2
    return grad, m_T, conv

def project_normalize(a):
    """Project to a >= 0 and sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lse_optimize(a, max_iters=1000, time_budget=60.0, T_init=0.1, T_final=0.01):
    """Optimize using soft-max smoothing with adaptive T and backtracking."""
    n = len(a)
    a = project_normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    alpha = 0.05  # step size
    v = np.zeros(n)  # momentum
    mu = 0.7  # momentum
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive T: start large, decrease
        T = T_init * (T_final/T_init) ** (it/max_iters)
        
        grad, m_T, conv = soft_max_gradient(a, T)
        
        # True objective
        obj = 2.0 * n * float(np.max(conv))
        
        # Project gradient
        grad_proj = grad - np.mean(grad)
        
        # Momentum
        v = mu * v + grad_proj
        v_norm = np.linalg.norm(v)
        if v_norm > 1e-10:
            v = v / v_norm  # normalize direction
        
        # Backtracking line search
        best_alpha = alpha
        best_obj_try = obj
        for _ in range(10):
            a_try = a - alpha * v
            a_try = project_normalize(a_try)
            conv_try = compute_conv(a_try)
            obj_try = 2.0 * n * float(np.max(conv_try))
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_alpha = alpha
            alpha *= 0.5
        
        alpha = best_alpha
        a = a - alpha * v
        a = project_normalize(a)
        
        obj = 2.0 * n * float(np.max(compute_conv(a)))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Occasionally increase step size if we made progress
        if obj < best_obj * 0.999:
            alpha = min(alpha * 1.2, 0.3)
    
    return best_a
```

```python
Optimization of step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max smoothed gradient descent, fast swap moves, LP refinement,
and multiple structured initializations (decreasing, asymmetric, random).

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0 or np.sum(a) < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (np.sum(a)**2)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def soft_max_grad(a, T):
    """
    Returns (gradient, soft_max_value, conv) of the soft-max of conv(a,a).
    Gradient is w.r.t. a, for the case sum(a)=1 (so obj = 2n * soft_max).
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    shifted = (conv - cmax) / T
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w /= w_sum
    m_T = cmax + T * np.log(w_sum)
    
    # Gradient: grad[i] = sum_k w[k] * (a[k-i]*I(0<=k-i<n) + a[i-k]*I(0<=i-k<n))
    # Part 1: sum_k w[k]*a[k-i]*I(0<=k-i<n) = sum_j a[j]*w[i+j] = correlate(w, a, 'valid')[i]

    size = 1

    part2_full = irfft(fa * fw, size)
    part2 = part2_full[:n]
    
    grad = part1 + part2
    return grad, m_T, conv
```
