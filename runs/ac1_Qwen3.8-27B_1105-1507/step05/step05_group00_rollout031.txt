def fista_softmax(a, max_iters=500, tau_init=0.1, tau_decay=0.995, time_budget=60.0):
    """FISTA (accelerated gradient) with soft-max approximation of max(conv)."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Lipschitz constant estimate
    # The gradient of soft-max(conv) has norm bounded by ~ 2 * max(a) * (2n-1) / tau * tau = 2*max(a)*(2n-1)
    # More carefully: ||grad|| <= 2 * max_k w_k * sum_i a[k-i]... 
    # Let's just use L = 2*n*max(a) as a rough bound
    L = 2.0 * n * np.max(a) + 1e-10
    tau = tau_init
    
    y = a.copy()
    t_k = 1.0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute gradient at y
        grad_y = smooth_grad_fft(y, tau)
        
        # Project gradient onto sum=1 hyperplane
        grad_proj = grad_y - np.mean(grad_y)
        
        # FISTA update
        step = 1.0 / L
        x_new = y - step * grad_proj
        x_new = np.maximum(x_new, 0)
        x_new = normalize(x_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0)
        y = normalize(y)
        
        a = x_new
        t_k = t_new
        
        # Evaluate
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Decay tau and update L
        tau *= tau_decay
        L = 2.0 * n * np.max(a) + 1e-10
    
    return best_a, best_obj

def multiplicative_update(a, max_iters=500, time_budget=60.0, init_eta=0.01):
    """Multiplicative (exponential tilting) updates to reduce max convolution."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = init_eta
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute soft-max gradient
        tau = max(0.01, 0.1 * np.exp(-0.01 * it))
        grad = smooth_grad_fft(a, tau)
        grad_centered = grad - np.mean(grad)
        
        # Multiplicative update: a_new = a * exp(-eta * grad_centered)
        log_update = -eta * grad_centered
        log_update = np.clip(log_update, -5, 5)  # prevent overflow
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce eta
            eta *= 0.95
            if eta < 1e-6:
                break
    
    return best_a, best_obj

def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A

def lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None)."""
    n = len(a)
    if n > 1500:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None

def lp_max_sum_blended(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g>=0. Normalize result."""
    n = len(a)
    if n > 1500:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def golden_section_search(f, a_lo, a_hi, tol=1e-5, max_iter=50):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if a_hi - a_lo < tol:
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0

def dual_active_set_refine(a, time_budget=30.0):
    """
    Focus optimization on the "active set" - positions that contribute most to the peak.
    Use the LP dual variables to identify which convolution positions are tight.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    # Find active set: positions where conv is close to M
    threshold = M * 0.95
    active = np.where(conv >= threshold)[0]
    
    # For each active position k, the constraint is sum_i a[i]*a[k-i] <= M
    # To reduce these, we should decrease a[i] for i that appear in many active constraints
    # and increase a[i] for i that appear in few
    
    contribution = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                contribution[i] += a[i] * a[j]
    
    # Direction: decrease high-contribution, increase low-contribution
    # Normalize to be a direction on the sum=1 hyperplane
    direction = -contribution
    direction -= np.mean(direction)
    
    # Line search
    for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
        if time.monotonic() - start > time_budget:
            break
        a_try = a + t * direction / (np.max(np.abs(direction)) + 1e-10)
        a_try = np.maximum(a_try, 0)
        a_try = normalize(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
    
    return best_a, best_obj

def stochastic_restart(a, rng, scale=0.1):
    """Add multiplicative noise to restart from current point."""
    n = len(a)
    noise = rng.normal(0, scale, n)
    a_new = a * np.exp(noise)
    a_new = np.maximum(a_new, 0)
    a_new = normalize(a_new)
    return a_new

def structured_start(n, seed=0):
    """Generate structured starting sequences."""
    rng = np.random.default_rng(seed)
    structures = []
    
    # Uniform
    structures.append(np.ones(n) / n)
    
    # Triangular
    t = np.arange(n)
    tri = np.maximum(0, 1.0 - 2.0 * t / n)
    structures.append(normalize(tri))
    
    # Gaussian
    x = np.linspace(-3, 3, n)
    gauss = np.exp(-0.5 * x**2)
    structures.append(normalize(gauss))
    
    # Chirp (quadratic phase) - as magnitude
    phase = 2 * np.pi * t * t / (2 * n)
    chirp_mag = np.abs(np.exp(1j * phase))  # all 1s, not useful
    # Instead use a modulated sequence
    mod = 0.5 + 0.5 * np.cos(2 * np.pi * t / n)
    structures.append(normalize(mod))
    
    # Random
    structures.append(normalize(rng.uniform(0, 1, n)))
    
    # Bimodal
    bimo = np.exp(-((t - n//4)**2) / (n//10)**2) + np.exp(-((t - 3*n//4)**2) / (n//10)**2)
    structures.append(normalize(bimo))
    
    return structures

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using FISTA with soft-max,
multiplicative reweighting, LP-guided direction, dual active-set focusing,
and stochastic restarts with structured initializations.
Key innovations: adaptive FISTA, golden-section line search for LP blends,
pairwise local search for fine improvements, and multi-scale length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize a to sum to 1, ensuring non-negativity."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smooth_grad_fft(a, tau):
    """
    Compute gradient of soft-max(conv(a,a)) w.r.t. a using FFT.
    grad_i = 2 * sum_{j} a[j] * w[i+j] where w are soft-max weights.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -50, 50)
    e = np.exp(shifted)
    w = e / np.sum(e)

    # grad_i = 2 * sum_m a[m] * w[i+m]
    # = 2 * (a_rev * w)[n-1+i] where a_rev[m] = a[n-1-m]
    a_rev = a[::-1]
    size = 1
    while size < n + 2 * n - 1:
        size <<= 1
    fa_rev = rfft(a_rev, size)
    fw = rfft(w, size)
    corr = irfft(fa_rev * fw, size)
    grad = 2.0 * corr[n - 1:2 * n - 1]

    return grad


def fista_softmax(a, max_iters=500, tau_init=0.05, tau_end=0.01, time_budget=60.0):
    """
    FISTA (accelerated proximal gradient) with soft-max approximation.
    Minimizes max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Estimate Lipschitz constant
    L = 2.0 * n * np.max(a) + 1e-10
    tau = tau_init

    y = a.copy()
    t_k = 1.0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Animate tau from tau_init to tau_end
        progress = it / max(1, max_iters)
        tau = tau_init * (tau_end / tau_init) ** progress

        # Compute gradient at y
        grad_y = smooth_grad_fft(y, tau)

        # Project gradient onto sum=1 hyperplane
        grad_proj = grad_y - np.mean(grad_y)

        # Update Lipschitz estimate
        L = 2.0 * n * np.max(a) + 1e-10
        step = 1.0 / L

        # FISTA update
        x_new = y - step * grad_proj
        x_new = np.maximum(x_new, 0)
        x_new = normalize(x_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0)
        y = normalize(y)

        a = x_new
        t_k = t_new

        # Evaluate every few iterations for speed
        if it % 5 == 0 or it == max_iters - 1:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    # Final evaluation
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def multiplicative_update(a, max_iters=500, time_budget=60.0, init_eta=0.05):
    """
    Multiplicative (exponential tilting) updates.
    a_new = normalize(a * exp(-eta * centered_grad))
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = init_eta
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        tau = max(0.01, 0.1 * np.exp(-0.005 * it))
        grad = smooth_grad_fft(a, tau)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.max(np.abs(grad_centered))
        if grad_norm < 1e-15:
            break

        # Normalized direction
        direction = grad_centered / grad_norm

        # Try a few eta values
        best_eta = eta
        best_obj_local = np.inf
        best_a_local = a.copy()

        for eta_try in [eta * 0.5, eta, eta * 2.0]:
            log_update = -eta_try * direction
            log_update = np.clip(log_update, -5, 5)
            a_new = a * np.exp(log_update)
            a_new = normalize(a_new)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_local:
                best_obj_local = obj_new
                best_eta = eta_try
                best_a_local = a_new

        if best_obj_local < best_obj - 1e-14:
            best_obj = best_obj_local
            best_a = best_a_local
            a = best_a_local
            eta = best_eta
            stall = 0
        else:
            stall += 1
            eta *= 0.8
            if stall > 30:
                break

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def lp_minpeak(a, time_limit=15):
    """
    Solve LP: min M s.t. (a*g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    if n > 1200:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def lp_max_sum(a, M, time_limit=15):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g>=0.
    Returns normalized g or None.
    """
    n = len(a)
    if n > 1200:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_section_min(f, lo, hi, tol=0.005, max_iter=40):
    """Golden section search for minimum of unimodal f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def lp_alternate(a, max_iters=60, time_budget=120.0):
    """
    LP alternating with golden section line search.
    At each step: solve LP for best direction, then line search for optimal blend.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Solve LP for direction
        remaining_lp = max(3, (time_budget - (time.monotonic() - start)) * 0.4)
        g = lp_max_sum(a, M, time_limit=min(10, remaining_lp))
        if g is None:
            # Try min-peak LP instead
            if n <= 1000:
                g, M_opt = lp_minpeak(a, time_limit=min(10, remaining_lp))
                if g is None:
                    break
            else:
                break

        # Golden section line search
        def blend_obj(t):
            a_try = (1.0 - t) * a + t * g
            a_try = normalize(a_try)
            return compute_obj(a_try)

        best_t = golden_section_min(blend_obj, 0.0, 1.2, tol=0.01, max_iter=25)
        a_new = (1.0 - best_t) * a + best_t * g
        a_new = normalize(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            break

    return best_a, best_obj


def dual_active_refine(a, time_budget=20.0):
    """
    Focus on active constraints (peaks near max) and reduce their contributions.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for _ in range(20):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Active set: positions where conv >= 0.97*M
        active = np.where(conv >= 0.97 * M)[0]

        # Compute contribution of each a[i] to active peaks
        contribution = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contribution[i] += a[j]

        # Direction: reduce high-contribution, boost low-contribution
        direction = -contribution
        direction -= np.mean(direction)
        direction_norm = np.linalg.norm(direction)
        if direction_norm < 1e-12:
            break
        direction = direction / direction_norm

        # Try several step sizes
        improved = False
        for scale in [0.01, 0.02, 0.05, 0.1, 0.2]:
            a_try = a + scale * direction
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
        if not improved:
            break

    return best_a, best_obj


def pairwise_local_search(a, time_budget=30.0, max_sweeps=5):
    """
    Fine-grained pairwise local search: for each pair (i,j), try moving
    a small fraction of mass from one to the other.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(a)
    k_star = int(np.argmax(conv))

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        # Focus on positions that contribute to the peak
        # Position i contributes a[i]*a[k_star-i] to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
            else:
                contrib[i] = a[i]  # indirect contribution

        # Sort by contribution (high first) - we want to reduce high-contrib positions
        order = np.argsort(-contrib)

        for i_idx in order:
            if time.monotonic() - start > time_budget:
                break
            i = int(i_idx)
            if a[i] < 1e-10:
                continue

            # Try moving mass from i to several target positions
            for j_offset in range(1, min(n, 50)):
                j = (i + j_offset) % n
                if j == i or a[j] < 1e-12:
                    continue

                # Move small fraction from i to j
                for frac in [0.01, 0.03, 0.05]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[j] += delta
                    new_conv = compute_conv(a)
                    new_max = float(np.max(new_conv))
                    new_sum = float(np.sum(a))
                    new_obj = 2.0 * n * new_max / (new_sum * new_sum)

                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        k_star = int(np.argmax(conv))
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break

        if not improved:
            break

    return best_a, best_obj


def stochastic_restart(a, rng, scale=0.05):
    """Add multiplicative log-normal noise."""
    n = len(a)
    noise = rng.normal(0, scale, n)
    a_new = a * np.exp(noise)
    return normalize(a_new)


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize(a_new)


def structured_starts(n, seed=0):
    """Generate diverse structured starting sequences."""
    rng = np.random.default_rng(seed)
    t = np.arange(n)
    results = []

    # Uniform
    results.append(np.ones(n) / n)

    # Triangular
    tri = np.maximum(0, 1.0 - 2.0 * t / n)
    results.append(normalize(tri))

    # Gaussian
    x = np.linspace(-3, 3, n)
    results.append(normalize(np.exp(-0.5 * x ** 2)))

    # Cosine window
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * t / (n - 1)) if n > 1 else np.ones(1)
    results.append(normalize(cos_w))

    # Bimodal
    bimo = (np.exp(-((t - n // 4) ** 2) / max(1, n // 8)) +
            np.exp(-((t - 3 * n // 4) ** 2) / max(1, n // 8)))
    results.append(normalize(bimo))

    # Random
    results.append(normalize(rng.uniform(0.1, 1.0, n)))

    # Power law
    results.append(normalize((t + 1) ** 0.5))

    return results


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def full_polish(a, time_budget=60.0, n_max_lp=1000):
    """
    Full polishing pipeline: FISTA -> LP -> multiplicative -> dual -> pairwise -> FISTA2.
    Returns best (a, obj) found.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # 1. FISTA with moderate tau
    rem = time_budget - (time.monotonic() - start)
    if rem > 10:
        alloc = min(15, rem * 0.25)
        a_f, obj_f = fista_softmax(best_a, max_iters=400, tau_init=0.05,
                                   tau_end=0.02, time_budget=alloc)
        if obj_f < best_obj:
            best_obj = obj_f
            best_a = a_f
            print(f"      FISTA: {best_obj:.8f}")

    # 2. LP alternating
    rem = time_budget - (time.monotonic() - start)
    if n <= n_max_lp and rem > 15:
        alloc = min(30, rem * 0.35)
        a_lp, obj_lp = lp_alternate(best_a, max_iters=30, time_budget=alloc)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp
            print(f"      LP: {best_obj:.8f}")

    # 3. Multiplicative update
    rem = time_budget - (time.monotonic() - start)
    if rem > 10:
        alloc = min(15, rem * 0.2)
        a_mu, obj_mu = multiplicative_update(best_a, max_iters=400, time_budget=alloc)
        if obj_mu < best_obj:
            best_obj = obj_mu
            best_a = a_mu
            print(f"      Mult: {best_obj:.8f}")

    # 4. Dual active set
    rem = time_budget - (time.monotonic() - start)
    if rem > 8:
        alloc = min(10, rem * 0.2)
        a_da, obj_da = dual_active_refine(best_a, time_budget=alloc)
        if obj_da < best_obj:
            best_obj = obj_da
            best_a = a_da
            print(f"      Dual: {best_obj:.8f}")

    # 5. Pairwise local search (fine improvements)
    rem = time_budget - (time.monotonic() - start)
    if rem > 10:
        alloc = min(15, rem * 0.25)
        a_pw, obj_pw = pairwise_local_search(best_a, time_budget=alloc, max_sweeps=4)
        if obj_pw < best_obj:
            best_obj = obj_pw
            best_a = a_pw
            print(f"      Pair: {best_obj:.8f}")

    # 6. FISTA with small tau (fine convergence)
    rem = time_budget - (time.monotonic() - start)
    if rem > 8:
        alloc = min(12, rem * 0.3)
        a_f2, obj_f2 = fista_softmax(best_a, max_iters=300, tau_init=0.02,
                                     tau_end=0.005, time_budget=alloc)
        if obj_f2 < best_obj:
            best_obj = obj_f2
            best_a = a_f2
            print(f"      FISTA2: {best_obj:.8f}")

    # 7. Final LP if time allows
    rem = time_budget - (time.monotonic() - start)
    if n <= n_max_lp and rem > 10:
        alloc = min(15, rem * 0.4)
        a_lp2, obj_lp2 = lp_alternate(best_a, max_iters=15, time_budget=alloc)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2
            print(f"      LP2: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines FISTA, LP alternating, multiplicative updates,
    dual active-set refinement, pairwise local search, and stochastic restarts.
    Explores multiple sequence lengths and structured starts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = normalize(parent.copy())
        print(f"Parent: n={len(best_seq)}, obj={best_score:.8f}")

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}, budget={budget_s}s")

    # === Phase 1: Intensive optimization at parent length ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Intensive optimization (n={n_parent})")
    print(f"{'='*60}")

    if best_seq is not None:
        a_work = best_seq.copy()

        for round_idx in range(10):
            if time.monotonic() > deadline - 300:
                break
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            alloc = min(80.0, remaining * 0.2)
            print(f"  Round {round_idx}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

            a_pol, obj_pol = full_polish(a_work, time_budget=alloc)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_work = a_pol
                print(f"  ** IMPROVED: {best_score:.8f} **")
            else:
                # Stochastic restart
                scale = 0.03 + 0.02 * round_idx
                a_restart = stochastic_restart(best_seq, rng, scale=scale)
                a_work = a_restart

                # Quick polish the restart
                if time.monotonic() < deadline - 50:
                    alloc2 = min(30, remaining * 0.15)
                    a_pol2, obj_pol2 = full_polish(a_restart, time_budget=alloc2)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_work = a_pol2
                        print(f"  ** IMPROVED (restart): {best_score:.8f} **")
                    else:
                        a_work = best_seq.copy()

    # === Phase 2: Length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Length exploration")
    print(f"{'='*60}")

    n_factors = [0.6, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.3, 1.5, 1.8]
    n_options = []
    for f in n_factors:
        n_try = int(n_parent * f)
        if 80 <= n_try <= 2000:
            n_options.append(n_try)
    n_options = sorted(set(n_options))

    for n_try in n_options:
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        a_try = resample_sequence(best_seq, n_try)
        print(f"  n={n_try}: {remaining:.0f}s left")

        alloc = min(35.0, remaining * 0.18)
        a_opt, obj_opt = full_polish(a_try, time_budget=alloc)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** LENGTH WIN: n={n_try}, obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.8f}")

    # === Phase 3: Structured starts at best length ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Structured starts (n={len(best_seq)})")
    print(f"{'='*60}")

    n_best = len(best_seq)
    structures = structured_starts(n_best, seed=seed + 1)

    for idx, s in enumerate(structures):
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        alloc = min(20.0, remaining * 0.12)
        a_opt, obj_opt = full_polish(s, time_budget=alloc)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Structure {idx}: ** obj={best_score:.8f} **")
        else:
            print(f"  Structure {idx}: obj={obj_opt:.8f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Final polish")
    print(f"{'='*60}")
    print(f"Best before final: {best_score:.8f}")

    a_final = best_seq.copy()
    for fr in range(8):
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Alternating: polish directly vs perturb then polish
        if fr % 2 == 0:
            alloc = min(30, remaining * 0.2)
            a_pol, obj_pol = full_polish(a_final, time_budget=alloc)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Final {fr}: {best_score:.8f}")

        else:
            # Perturb and polish
            a_pert = stochastic_restart(a_final, rng, scale=0.02 + 0.01 * fr)
            alloc = min(25, remaining * 0.2)
            a_pol, obj_pol = full_polish(a_pert, time_budget=alloc)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol
                print(f"  Final perturb {fr}: {best_score:.8f}")

    # === Phase 5: Very fine pairwise search on the best ===
    if time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print(f"\n  Phase 5: Final pairwise search ({remaining:.0f}s)")
            a_pw, obj_pw = pairwise_local_search(best_seq, time_budget=min(remaining * 0.5, 40),
                                                 max_sweeps=10)
            if obj_pw < best_score:
                best_score = obj_pw
                best_seq = a_pw.copy()
                print(f"  Pairwise final: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
