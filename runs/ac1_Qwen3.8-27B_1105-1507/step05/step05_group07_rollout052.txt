def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def quadratic_residue_sequence(n):
    """Generate sequence based on quadratic residues modulo p (p prime, p ≡ 1 or 3 mod 4)."""

            break
        p += 1
    # Legendre symbol sequence
    a = np.zeros(p)
    for i in range(p):
        a[i] = 1.0 if (pow(i, (p-1)//2, p) != p-1) or i == 0 else 0.0
    # Resample to desired length
    return resample_seq(a, n)

def is_prime(n):
    if n < 2:
        return False
    if n < 4:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:

def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n, else 0."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, time_limit=60):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 2048:
        # Too large for sparse LP in reasonable time
        return None
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception as e:
        print(f"  LP error: {e}")
    return None

def lp_mirror_descent(a, max_iters, deadline, lp_time_limit=30):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_conv = fast_conv(a)
        M = float(b_conv.max())
        t0 = time.monotonic()
        g = lp_best_response(a, M, time_limit=lp_time_limit)
        lp_time = time.monotonic() - t0

        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search
        improved = False
        base_t = max(0.1, 0.8 * (1.0 - 0.3 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        remaining = deadline - time.monotonic()
        if remaining < lp_time * 1.5:
            break

    return best_a

def simulated_annealing(a, max_iters, deadline, t_init=0.1, t_min=0.001, cooling=0.999):
    """Simulated annealing to escape local minima."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    a = a / a.sum()
    current_J = compute_J(a)
    best_a = a.copy()

    t = t_init
    rng = np.random.default_rng(42)

        if time.monotonic() > deadline:
            break

        # Random perturbation
        a_pert = a.copy()
        # Perturb a random subset of elements
        n_perturb = max(1, n // 10)
        indices = rng.choice(n, size=n_perturb, replace=False)
        for idx in indices:
            a_pert[idx] *= (1.0 + rng.normal(0, t))
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert /= s

        J_pert = compute_J(a_pert)

        # Acceptance criterion
        if J_pert < current_J or rng.random() < np.exp(-(J_pert - current_J) / max(t, 1e-8)):
            a = a_pert
            current_J = J_pert
            if J_pert < best_J:
                best_J = J_pert
                best_a = a.copy()
                print(f"    SA: J={J_pert:.8f} t={t:.5f} iter={it}")

        t = max(t_min, t * cooling)

def block_coordinate_descent(a, block_size, max_sweeps, deadline, seed=0):
    """
    Block coordinate descent: update blocks of 'block_size' consecutive elements.
    For each block, try a small set of patterns.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    a = a / a.sum()
    conv = fast_conv(a)
    s = a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * conv.max() / (s * s)
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        improved = False
        for start in range(0, n - block_size + 1, block_size):
            old_block = a[start:start+block_size].copy()
            if old_block.max() < 1e-10:
                continue

            # Try several patterns for the block
            total = old_block.sum()
            patterns = [
                old_block,  # no change
                np.ones(block_size) * (total / block_size),  # uniform
                np.maximum(old_block + rng.normal(0, 0.1 * total / block_size, block_size), 0),  # noise
                np.maximum(old_block * (1 + rng.normal(0, 0.3, block_size)), 0),  # scale noise
            ]

            best_pattern = old_block
            best_local = 2.0 * n * conv.max() / (s * s)

            for pattern in patterns:
                pattern = np.maximum(pattern, 0)
                p_sum = pattern.sum()
                if p_sum < 1e-10:
                    continue
                # Scale to maintain total sum
                pattern_scaled = pattern * (total / p_sum) if total > 0 else pattern

                # Compute new conv (approximate: only affected region)
                delta = pattern_scaled - old_block
                # This is approximate - for exact we'd need full recompute
                # But for a heuristic, let's just compute J directly
                a_trial = a.copy()
                a_trial[start:start+block_size] = pattern_scaled
                J_trial = compute_J(a_trial)
                if J_trial < best_local:
                    best_local = J_trial
                    best_pattern = pattern_scaled

            if not np.allclose(best_pattern, old_block):
                a[start:start+block_size] = best_pattern
                s = a.sum()
                conv = fast_conv(a)
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()

        if not improved:
            break

    return best_a

"""
Hybrid optimizer for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Uses: combinatorial initializations, adaptive coordinate descent with incremental
convolution, LP mirror descent, simulated annealing, active-set refinement,
and block coordinate descent. Multi-length exploration with resampling.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def is_prime_test(n):
    if n < 2:
        return False
    if n < 4:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True


def quadratic_residue_seq(n):
    """Generate sequence based on quadratic residues modulo p."""
    p = n
    while p < n + 20:
        if is_prime_test(p):
            break
        p += 1
    a = np.zeros(p)
    for i in range(1, p):
        a[i] = 1.0 if pow(i, (p - 1) // 2, p) == 1 else 0.0
    a[0] = 0.0
    return resample_seq(a, n)


def m_seq_init(n, seed=12345):
    """Generate m-sequence-like smooth initialization."""
    rng = np.random.default_rng(seed)
    binary = (rng.random(n) > 0.5).astype(float)
    k = max(3, n // 64)
    kernel = np.ones(k) / k
    a = np.convolve(binary, kernel, mode='same')
    return np.maximum(a, 0.01)


def adaptive_coord_descent(a, max_sweeps, deadline, seed=0, n_cand=8):
    """
    Adaptive coordinate descent with incremental convolution updates.
    O(n) per coordinate step via direct array operations.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n

    conv = fast_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * conv.max() / (s * s)

    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 250

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            if old < 1e-8:
                candidates = [0.0, 1e-5, 1e-4, 1e-3, 5e-3, 0.01, 0.05, 0.1, 0.3]
            else:
                ratios = np.concatenate([
                    np.array([0.0, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85,
                              1.15, 1.3, 1.5, 1.8, 2.2, 3.0])
                ])
                candidates = [float(r) * old for r in ratios if abs(r * old - old) > old * 0.02]

            best_val = old
            best_local_obj = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = max(0.0, x)
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local_obj:
                    best_local_obj = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if abs(best_val - old) > 1e-12:

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                s = s - old + best_val

        current_obj = 2.0 * n * conv.max() / (s * s)

            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:

    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix for LP constraints: (a*g)[k] <= M for all k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=30):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 2048:
        return None
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, lp_time_limit=30):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    no_improve = 0
    ema_lp_time = 0.1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if deadline - time.monotonic() < ema_lp_time * 2:
            break

        b_conv = fast_conv(a)
        M = float(b_conv.max())
        t0 = time.monotonic()
        g = lp_best_response(a, M, time_limit=lp_time_limit)
        lp_time = time.monotonic() - t0
        ema_lp_time = 0.7 * ema_lp_time + 0.3 * lp_time

        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        improved = False
        base_t = max(0.08, 0.6 * (1.0 - 0.4 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:

            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 25:
                break
        else:
            no_improve = 0

    return best_a

            break

        a_pert = a.copy()
        indices = rng.choice(n, size=n_perturb, replace=False)
        for idx in indices:
            a_pert[idx] *= max(0.0, 1.0 + rng.normal(0, t))
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert /= s

        J_pert = compute_J(a_pert)

        if J_pert < current_J or (t > t_min and rng.random() < np.exp(min(0, -(J_pert - current_J) / t))):
            a = a_pert
            current_J = J_pert
            if J_pert < best_J:
                best_J = J_pert
                best_a = a.copy()
                iterations_since_best = 0
                if it % 100 == 0:
                    print(f"    SA iter={it}: J={best_J:.8f} t={t:.6f}", flush=True)
            else:
                iterations_since_best += 1
        else:
            iterations_since_best += 1

        t = max(t_min, t * cooling)

        if iterations_since_best > 2000 and t < t_min * 10:
            break

    return best_a


def active_set_cd(a, max_iters, deadline, threshold=0.9):
    """
    Coordinate descent focused on coordinates affecting the peak convolution values.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    a = a / a.sum()
    conv = fast_conv(a)
    s = a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = conv.max()
        active_k = np.where(conv >= threshold * M)[0]

        active_coords = set()
        for k in active_k:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for i in range(j_min, j_max + 1):
                active_coords.add(i)

        if len(active_coords) == 0:
            active_coords = set(range(n))
        elif len(active_coords) > n // 2:
            active_coords = set(range(n))

        for i in sorted(active_coords):
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            if old < 1e-8:
                candidates = [0.0, 1e-3, 1e-2, 0.05]
            else:
                candidates = [0.0, old * 0.4, old * 0.7, old * 1.0, old * 1.3, old * 1.8]

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                new_s = s - old + x
                if new_s > 1e-10:
                    obj = 2.0 * n * conv.max() / (new_s * new_s)
                else:
                    obj = float('inf')
                if obj < best_local:
                    best_local = obj
                    best_val = x
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val

        conv = fast_conv(a)
        current_J = 2.0 * n * conv.max() / (s * s)
        if current_J < best_J - 1e-12:
            best_J = current_J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    return best_a


def block_cd(a, block_size, max_sweeps, deadline, seed=0):
    """Block coordinate descent: optimize groups of consecutive elements."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        improved_sweep = False
        for start in range(0, n - block_size + 1, block_size):
            old_block = a[start:start + block_size].copy()
            block_sum = old_block.sum()
            if block_sum < 1e-10:
                continue

            # Generate candidate patterns
            patterns = [old_block.copy()]
            # Uniform
            patterns.append(np.full(block_size, block_sum / block_size))
            # Random perturbations
            for _ in range(4):
                p = np.maximum(old_block * (1.0 + rng.normal(0, 0.3, block_size)), 0)
                if p.sum() > 0:
                    p = p * (block_sum / p.sum())
                    patterns.append(p)
            # Ramp up/down
            ramp = np.linspace(0.5, 1.5, block_size)
            patterns.append(ramp * (block_sum / ramp.sum()))
            ramp2 = np.linspace(1.5, 0.5, block_size)
            patterns.append(ramp2 * (block_sum / ramp2.sum()))

            best_pattern = old_block
            best_local_J = best_J

            for pattern in patterns:
                if np.allclose(pattern, old_block, rtol=0.01):
                    continue
                a_trial = a.copy()
                a_trial[start:start + block_size] = pattern
                J_trial = compute_J(a_trial)
                if J_trial < best_local_J:
                    best_local_J = J_trial
                    best_pattern = pattern

            if not np.allclose(best_pattern, old_block, rtol=0.01):
                a[start:start + block_size] = best_pattern
                improved_sweep = True

        current_J = compute_J(a)
        if current_J < best_J - 1e-12:
            best_J = current_J
            best_a = a.copy()
            no_improve = 0
            improved_sweep = True
        else:
            no_improve += 1

        if no_improve > 20 or (not improved_sweep and sweep > 10):
            break

    return best_a


def generate_initializations(n, parent, rng, seed):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Parent-based
    if parent is not None:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            p = p / p.sum()
            inits.append(("parent", p))
            for nl in [0.03, 0.08, 0.15, 0.25, 0.4]:
                pn = np.maximum(p + rng.normal(0, nl / n, n), 0)
                if pn.sum() > 0:
                    inits.append((f"p+n{nl}", pn / pn.sum()))

    # Uniform
    inits.append(("uniform", np.ones(n) / n))

    # Quadratic residue
    try:
        qr = quadratic_residue_seq(n)
        if qr.sum() > 0:
            inits.append(("QR", qr / qr.sum()))
    except Exception:
        pass

    # M-sequence
    m = m_seq_init(n, seed=seed)
    inits.append(("mseq", m / m.sum()))

    # Cosine/sine waves
    for freq in [1, 2, 3, 5, 8]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            c = np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01)
            inits.append((f"cos{freq}_{amp:.1f}", c / c.sum()))
            s = np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t), 0.01)
            inits.append((f"sin{freq}_{amp:.1f}", s / s.sum()))

    # Gaussian bumps
    for center in [0.15, 0.3, 0.5, 0.7, 0.85]:
        for width in [0.03, 0.08, 0.15, 0.3]:
            g = np.exp(-((t - center) / width) ** 2) + 0.01
            inits.append((f"g{center}_{width}", g / g.sum()))

    # V and inverted V
    v = 0.5 + 0.5 * np.abs(2 * t - 1) + 0.01
    inits.append(("V", v / v.sum()))
    iv = 1.0 - 0.5 * np.abs(2 * t - 1) + 0.01
    inits.append(("invV", iv / iv.sum()))

    # Smooth random
    ksize = max(5, n // 25)
    kernel = np.ones(ksize) / ksize
    for i in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        sr = smooth + 0.05
        inits.append((f"rand{i}", sr / sr.sum()))

    # Power-law
    for alpha in [0.3, 0.7, 1.5, 2.0]:
        pl = np.maximum(t ** alpha + (1 - t) ** alpha + 0.01, 0.01)
        inits.append((f"pl{alpha}", pl / pl.sum()))

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.5
    inits.append(("alt", alt / alt.sum()))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase hybrid optimizer:
    Phase 1: Adaptive CD on diverse initializations (n=parent_len)
    Phase 2: Multi-length exploration
    Phase 3: LP mirror descent
    Phase 4: Simulated annealing + active-set refinement
    Phase 5: Final polish
    """
    deadline = time.monotonic() + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    best_a = None
    best_score = float('inf')
    t_global_start = time.monotonic()

    def try_update(a, label):
        nonlocal best_a, best_score
        if a is None or a.size == 0:
            return
        a = np.clip(a.astype(np.float64), 0, 1000)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_a = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_global_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} [{label}] t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            try_update(parent, "parent")
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        try_update(parent, "default")

    n_base = parent.size

    # === PHASE 1: Diverse inits + Adaptive CD ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Adaptive CD on diverse inits (n={n_base})", flush=True)
    print(f"{'='*60}", flush=True)

    p1_deadline = min(time.monotonic() + 380, deadline - 700)

    inits = generate_initializations(n_base, parent, rng, seed)
    n_inits = len(inits)
    time_per_init = max(15, (p1_deadline - time.monotonic()) / max(1, n_inits))

    for idx, (name, a0) in enumerate(inits):
        if time.monotonic() > p1_deadline:
            break
        if a0.sum() < 1e-10:
            continue

        t0 = time.monotonic()
        a_opt = adaptive_coord_descent(a0, max_sweeps=2000,
                                       deadline=min(t0 + time_per_init, p1_deadline),
                                       seed=seed + idx * 7)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_opt)
        if J_val < 2.0:  # Only report if meaningful
            print(f"  [{name:12s}] J={J_val:.8f} ({elapsed:.0f}s)", flush=True)
        try_update(a_opt, name)

    # Quick pass on current best with more sweeps
    if best_a is not None and time.monotonic() < p1_deadline - 30:
        t0 = time.monotonic()
        a_ref = adaptive_coord_descent(best_a, max_sweeps=5000,
                                       deadline=p1_deadline, seed=seed + 1000)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_ref)
        print(f"  [refine-best ] J={J_val:.8f} ({elapsed:.0f}s)", flush=True)
        try_update(a_ref, "refine")

    print(f"  Phase 1 done: best J={best_score:.8f}", flush=True)

    # === PHASE 2: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    p2_deadline = min(time.monotonic() + 250, deadline - 500)

    if best_a is not None:
        base_n = best_a.size
    else:
        base_n = n_base

    lengths = []
    for dn in [-256, -128, 0, 128, 256, 384, 512]:
        ln = base_n + dn
        if 128 <= ln <= 4096:
            lengths.append(ln)
    # Add some standard lengths
    for ln in [256, 384, 512, 640, 768, 1024, 1280, 1536]:
        if 128 <= ln <= 4096 and ln not in lengths:
            lengths.append(ln)

    lengths = sorted(set(lengths))
    time_per_len = max(20, (p2_deadline - time.monotonic()) / max(1, len(lengths)))

    for new_n in lengths:
        if time.monotonic() > p2_deadline:
            break

        src = best_a if best_a is not None else parent
        a_ext = resample_seq(src, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        t0 = time.monotonic()
        a_opt = adaptive_coord_descent(a_ext, max_sweeps=2000,
                                       deadline=min(t0 + time_per_len, p2_deadline),
                                       seed=seed + 2000 + new_n)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_opt)
        print(f"  n={new_n:5d}: J={J_val:.8f} ({elapsed:.0f}s)", flush=True)
        try_update(a_opt, f"len{new_n}")

    print(f"  Phase 2 done: best J={best_score:.8f}", flush=True)

    # === PHASE 3: LP Mirror Descent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    p3_deadline = min(time.monotonic() + 300, deadline - 150)

    if best_a is not None and best_a.size <= 2048:
        n_best = best_a.size
        est_lp = 0.003 * (n_best / 128) ** 1.5
        max_iters = max(15, min(300, int(250 / max(0.02, est_lp))))
        lp_limit = max(10, min(60, int(200 // max(1, max_iters))))
        print(f"  n={n_best}, est_lp={est_lp*1000:.0f}ms, max_iters={max_iters}, lp_limit={lp_limit}s", flush=True)

        a_lp = lp_mirror_descent(best_a, max_iters, p3_deadline, lp_time_limit=lp_limit)
        J_val = compute_J(a_lp)
        print(f"  LP-MD pass 1: J={J_val:.8f}", flush=True)
        try_update(a_lp, "LP1")

        if time.monotonic() < p3_deadline - 3 * est_lp:
            a_lp2 = lp_mirror_descent(a_lp, max(10, max_iters // 3), p3_deadline, lp_time_limit=lp_limit)
            J_val2 = compute_J(a_lp2)
            print(f"  LP-MD pass 2: J={J_val2:.8f}", flush=True)
            try_update(a_lp2, "LP2")

    print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)

    # === PHASE 4: Simulated Annealing + Active Set ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: SA + Active Set Refinement", flush=True)
    print(f"{'='*60}", flush=True)

    p4_deadline = min(time.monotonic() + 150, deadline - 60)

    if best_a is not None and time.monotonic() < p4_deadline:
        # Simulated annealing
        t0 = time.monotonic()
        sa_deadline = min(t0 + 80, p4_deadline - 30)
        a_sa = simulated_annealing(best_a, max_iters=10000, deadline=sa_deadline,
                                   t_init=0.05, t_min=0.0005, cooling=0.9995, seed=seed)
        J_val = compute_J(a_sa)
        print(f"  SA: J={J_val:.8f} ({time.monotonic()-t0:.0f}s)", flush=True)
        try_update(a_sa, "SA")

        # Active set CD
        if time.monotonic() < p4_deadline:
            t0 = time.monotonic()
            a_as = active_set_cd(best_a, max_iters=200, deadline=p4_deadline, threshold=0.85)
            J_val = compute_J(a_as)
            print(f"  Active-set: J={J_val:.8f} ({time.monotonic()-t0:.0f}s)", flush=True)
            try_update(a_as, "AS")

    print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Final Polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None and time.monotonic() < deadline - 10:
        # Final adaptive CD with many sweeps
        t0 = time.monotonic()
        a_final = adaptive_coord_descent(best_a, max_sweeps=10000,
                                         deadline=deadline - 5, seed=seed + 9999)
        J_val = compute_J(a_final)
        print(f"  Final CD: J={J_val:.8f} ({time.monotonic()-t0:.0f}s)", flush=True)
        try_update(a_final, "final-CD")

        # Block CD if time remains
        if time.monotonic() < deadline - 15:
            t0 = time.monotonic()
            a_bc = block_cd(best_a, block_size=4, max_sweeps=300,
                            deadline=min(t0 + 40, deadline - 5), seed=seed + 7777)
            J_val = compute_J(a_bc)
            print(f"  Block CD: J={J_val:.8f} ({time.monotonic()-t0:.0f}s)", flush=True)
            try_update(a_bc, "block-CD")

        # One more CD on the very best
        if time.monotonic() < deadline - 5 and best_a is not None:
            a_last = adaptive_coord_descent(best_a, max_sweeps=3000,
                                            deadline=deadline - 2, seed=seed + 12345)
            J_val = compute_J(a_last)
            print(f"  Last CD: J={J_val:.8f}", flush=True)
            try_update(a_last, "last-CD")

    # Fallback
    if best_a is None:
        best_a = np.ones(128)
        best_score = float(evaluate_sequence(best_a.tolist()))
        print(f"  FALLBACK: J={best_score:.8f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size} total_time={time.monotonic()-t_global_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_a]
