The LP mirror descent is essentially doing:
g = argmax { sum(g) : a*g <= M, g >= 0 }
Then updating a = (1-t)*a + t*g

This is a form of "best response" in a game-theoretic sense. The LP finds the direction that maximizes the sum while keeping the convolution bounded.

An alternative: instead of the LP, use a **multiplicative update**:
g[i] = a[i] * exp(-eta * subgrad[i])

This is the mirror map for the entropy regularization. It's simpler than solving an LP and might work just as well.

minimize max_k (a*a)[k] subject to a >= 0, sum(a) = 1

This is a convex optimization problem (the max of convex functions is convex, and the constraint set is convex). So we can use standard convex optimization methods.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast FFT-based subgradient descent with momentum and adaptive step size
2. Exact coordinate descent with golden-section 1D optimization
3. Multiplicative (exponential) updates as alternative to LP
4. Multi-length exploration with diverse initializations
5. Final polishing with combined methods
"""

import numpy as np
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def subgradient_descent(a, deadline, max_iters=100000, top_k_ratio=0.1, 
                        momentum=0.9, step_scale=0.5, rerun_interval=500):
    """
    Subgradient descent with momentum and adaptive step size.
    Uses FFT to compute the subgradient efficiently.
    """
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 0.01:
        return a
    a = a / S
    n = len(a)
    
    best_a = a.copy()
    best_J = 2 * n * fast_conv(a).max()
    
    v = np.zeros(n)  # momentum
    step = step_scale / np.sqrt(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        J = 2 * n * M  # since sum(a) = 1
        
        if J < best_J - 1e-12:
            best_J = J
            best_a = a.copy()
        
        # Find active set: positions where b[k] is close to max
        threshold = M * (1 - top_k_ratio * 0.1)  # within top_k_ratio*10% of max
        # Actually let's use: positions where b[k] > M - (M - b.min()) * top_k_ratio
        spread = M - b.min()
        if spread < 1e-15:
            break
        threshold = M - spread * top_k_ratio
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute subgradient via cross-correlation
        # subgrad[i] = 2 * sum_{k in active} a[k-i] (a[x]=0 if x not in [0,n-1])
        # = 2 * cross_corr(indicator_active, a)[i]
        # where cross_corr(c, a)[i] = sum_j c[i+j] * a[j]
        
        size = 1
        while size < 3 * n:
            size <<= 1
        
        ind = np.zeros(size)
        ind[active] = 1.0
        fa = np.fft.rfft(a, size)
        fi = np.fft.rfft(ind, size)
        corr = np.fft.irfft(fi * np.conj(fa), size)
        subgrad = 2.0 * corr[:n]
        
        # Center the subgradient (to maintain sum(a) = 1)
        subgrad_centered = subgrad - subgrad.mean()
        
        # Momentum update
        v = momentum * v + subgrad_centered
        a = a - step * v
        a = np.maximum(a, 0)
        S = a.sum()
        if S < 0.01:
            a = best_a.copy()
            v = np.zeros(n)
            continue
        a = a / S
        
        # Adaptive step size (decrease over time)
        if it % rerun_interval == rerun_interval - 1:
            step *= 0.95
        
    return best_a

def coordinate_descent_exact(a, deadline, max_passes=100):
    """
    Exact coordinate descent: for each coordinate, find the optimal value
    via golden-section search on a 1D objective.
    """
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 0.01:
        return a
    a = a / S
    n = len(a)
    b = fast_conv(a)  # b = conv(a, a), sum(a) = 1
    
    best_a = a.copy()
    best_M = b.max()
    
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        
        for i in order:
            old_ai = a[i]
            if old_ai < 1e-15:
                # Try a few small values
                candidates = [0.001, 0.005, 0.01, 0.02, 0.05]
                best_new = old_ai
                best_obj = 2 * n * b.max()
                for cand in candidates:
                    delta = cand
                    S_new = 1.0 + delta
                    # Compute new max
                    k_lo = max(0, i)
                    k_hi = min(2*n-2, i+n-1)
                    # Outside
                    if k_lo > 0:
                        m_out_left = b[:k_lo].max()
                    else:
                        m_out_left = 0
                    if k_hi < 2*n-2:
                        m_out_right = b[k_hi+1:].max()
                    else:
                        m_out_right = 0
                    m_out = max(m_out_left, m_out_right)
                    # Inside (excluding 2i)
                    m_in = 0
                    for k in range(k_lo, k_hi+1):
                        if k == 2*i:
                            continue
                        j = k - i
                        if 0 <= j < n:
                            val = b[k] + 2*delta*a[j]
                            if val > m_in:
                                m_in = val
                    # 2i
                    if 2*i <= 2*n-2:
                        val_2i = b[2*i] + delta*delta  # 2*delta*old_ai = 0 since old_ai ~ 0
                        if val_2i > m_in:
                            m_in = val_2i
                    m_new = max(m_out, m_in)
                    obj = 2 * n * m_new / (S_new * S_new)
                    if obj < best_obj:
                        best_obj = obj
                        best_new = cand
                
                if best_new != old_ai:
                    delta = best_new - old_ai
                    a[i] = best_new
                    S_total = 1.0 + delta
                    # Update b
                    k_lo = max(0, i)
                    k_hi = min(2*n-2, i+n-1)
                    for k in range(k_lo, k_hi+1):
                        if k == 2*i:
                            b[k] += 2*delta*old_ai + delta*delta
                        else:
                            j = k - i
                            if 0 <= j < n:
                                b[k] += 2*delta*a[j]
                    a = a / S_total
                    b = b / (S_total * S_total)
                    improved = True
                    M = b.max()
                    if M < best_M:
                        best_M = M
                        best_a = a.copy()
                continue
            
            # Golden section search for optimal delta
            # Objective: max(m_out, max_{k in A} (b[k] + 2*delta*a[k-i]), b[2i] + 2*delta*a[i] + delta^2) / (1+delta)^2
            
            k_lo = max(0, i)
            k_hi = min(2*n-2, i+n-1)
            k_2i = 2*i if 2*i <= 2*n-2 else -1
            
            # Outside max
            m_out_parts = []
            if k_lo > 0:
                m_out_parts.append(b[:k_lo].max())
            if k_hi < 2*n-2:
                m_out_parts.append(b[k_hi+1:].max())
            m_out = max(m_out_parts) if m_out_parts else 0.0
            
            # Affected: k in [k_lo, k_hi], k != 2i
            # b[k] + 2*delta*a[k-i]
            # We need the max of these linear functions at the optimal delta
            
            # For golden section, we just evaluate the objective
            lo = -old_ai * 0.99
            hi = old_ai * 3 + 0.05
            
            def obj(delta):
                S_new = 1.0 + delta
                if S_new < 0.01:
                    return 1e10
                # Max of affected linear terms
                m_aff = m_out
                for k in range(k_lo, k_hi+1):
                    if k == k_2i:
                        continue
                    j = k - i
                    val = b[k] + 2*delta*a[j]
                    if val > m_aff:
                        m_aff = val
                if k_2i >= 0:
                    val_2i = b[k_2i] + 2*delta*old_ai + delta*delta
                    if val_2i > m_aff:
                        m_aff = val_2i
                return m_aff / (S_new * S_new)
            
            # Golden section
            a1 = hi - gr * (hi - lo)
            b1 = lo + gr * (hi - lo)
            fa1 = obj(a1)
            fb1 = obj(b1)
            
            for _ in range(50):
                if fa1 < fb1:
                    hi = b1
                    b1 = a1
                    fb1 = fa1
                    a1 = hi - gr * (hi - lo)
                    fa1 = obj(a1)
                else:
                    lo = a1
                    a1 = b1
                    fa1 = fb1
                    b1 = lo + gr * (hi - lo)
                    fb1 = obj(b1)
            
            delta_opt = (a1 + b1) / 2
            new_ai = old_ai + delta_opt
            if new_ai < 0:
                new_ai = 0.0
                delta_opt = -old_ai
            
            if abs(new_ai - old_ai) < 1e-10:
                continue
            
            # Apply update
            delta = new_ai - old_ai
            S_total = 1.0 + delta
            a[i] = new_ai
            for k in range(k_lo, k_hi+1):
                if k == k_2i:
                    b[k] += 2*delta*old_ai + delta*delta
                else:
                    j = k - i
                    if 0 <= j < n:
                        b[k] += 2*delta*a[j]
            a = a / S_total
            b = b / (S_total * S_total)
            improved = True
            M = b.max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a

def multiplicative_update(a, deadline, max_iters=5000, eta=0.1):
    """
    Multiplicative (exponential) update: g[i] = a[i] * exp(-eta * normalized_subgrad[i])
    Then a = (1-t)*a + t*g, projected.
    """
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 0.01:
        return a
    a = a / S
    n = len(a)
    
    best_a = a.copy()
    best_J = 2 * n * fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Find active set
        spread = M - b.min()
        if spread < 1e-15:
            break
        threshold = M - spread * 0.1
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient
        size = 1
        while size < 3 * n:
            size <<= 1
        ind = np.zeros(size)
        ind[active] = 1.0
        fa = np.fft.rfft(a, size)
        fi = np.fft.rfft(ind, size)
        corr = np.fft.irfft(fi * np.conj(fa), size)
        subgrad = 2.0 * corr[:n]
        subgrad = subgrad - subgrad.mean()
        
        # Multiplicative update
        scale = np.max(np.abs(subgrad))
        if scale < 1e-15:
            break
        subgrad_norm = subgrad / scale
        g = a * np.exp(-eta * subgrad_norm)
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 0.01:
            break
        g = g / g_sum
        
        # Interpolate
        t = 0.1
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        J_new = 2 * n * fast_conv(a_new).max()
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Decrease eta
            eta *= 0.9
            if eta < 1e-6:
                break
    
    return best_a

def diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random
    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())
    
    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())
    
    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())
    
    # Bell
    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())
    
    # Double bump
    dbl = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())
    
    # Triple bump
    tri = np.exp(-10 * (t - 0.2) ** 2) + np.exp(-10 * (t - 0.5) ** 2) + np.exp(-10 * (t - 0.8) ** 2) + 0.02
    inits.append(tri / tri.sum())
    
    # Frequency modulated
    for freq in [1, 2, 3, 5, 7]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())
    
    # Power law
    for alpha in [0.5, 1.0, 2.0]:
        pl = (t + 0.01) ** alpha
        inits.append(pl / pl.sum())
    
    # Zigzag
    zz = 0.5 + 0.3 * np.abs(2 * (t * 4 % 1) - 1)
    inits.append(zz / zz.sum())
    
    # Sinc-like
    sinc = np.sinc(2 * (t - 0.5)) + 0.1
    inits.append(np.maximum(sinc, 0.01) / np.maximum(sinc, 0.01).sum())
    
    return inits

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast FFT-based subgradient descent with momentum and adaptive step size
2. Exact coordinate descent with golden-section 1D optimization per coordinate
3. Multiplicative (exponential) updates for diverse exploration
4. Multi-length exploration with diverse initializations
5. Iterative peak-reduction targeting
Key differences from previous: replaces expensive LP with fast FFT subgradient,
uses exact 1D optimization in CD instead of candidate sampling, and adds
targeted peak-reduction as an additional local search mode.

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_subgradient_fft(a, active):
    """Compute subgradient of max_k (a*a)[k] using FFT cross-correlation.
    subgrad[i] = 2 * sum_{k in active} a[k-i] (a[x]=0 if x not in [0,n-1])
    """
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    ind = np.zeros(size)
    ind[active] = 1.0
    fa = np.fft.rfft(a, size)
    fi = np.fft.rfft(ind, size)
    corr = np.fft.irfft(fi * np.conj(fa), size)
    return 2.0 * corr[:n]


def subgradient_descent(a, deadline, max_iters=100000, top_frac=0.1,
                        momentum=0.9, step_scale=0.5, decay=0.98):
    """
    Subgradient descent with momentum and decaying step size.
    Uses FFT to compute the subgradient of the max of self-convolution.
    """
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 0.01:
        return a
    a = a / S
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = 2.0 * n * best_M

    v = np.zeros(n)
    step = step_scale / np.sqrt(n)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        J = 2.0 * n * M

        if J < best_J - 1e-13:
            best_J = J
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            # Perturb to escape local optimum
            a = a + np.random.normal(0, 0.01, n)
            a = np.maximum(a, 1e-5)
            a = a / a.sum()
            v = np.zeros(n)
            step = step_scale / np.sqrt(n)
            no_improve = 0
            continue

        spread = M - b.min()
        if spread < 1e-14:
            break

        threshold = M - spread * top_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        subgrad = compute_subgradient_fft(a, active)
        subgrad_centered = subgrad - subgrad.mean()

        v = momentum * v + subgrad_centered
        a = a - step * v
        a = np.maximum(a, 0)
        S = a.sum()
        if S < 0.01:
            a = best_a.copy()
            v = np.zeros(n)
            continue
        a = a / S

        step = step * decay
        if step < 1e-8 / n:
            step = 1e-8 / n

    return best_a


def cd_obj_func(delta, b, a, i, k_lo, k_hi, k_2i, m_out, n, old_ai):
    """Evaluate the 1D objective for coordinate descent at a given delta."""
    S_new = 1.0 + delta
    if S_new < 0.01:
        return 1e10
    m = m_out
    for k in range(k_lo, k_hi + 1):
        if k == k_2i:
            continue
        j = k - i
        val = b[k] + 2.0 * delta * a[j]
        if val > m:
            m = val
    if k_2i >= 0:
        val_2i = b[k_2i] + 2.0 * delta * old_ai + delta * delta
        if val_2i > m:
            m = val_2i
    return m / (S_new * S_new)


def coordinate_descent_exact(a, deadline, max_passes=100):
    """
    Exact coordinate descent: for each coordinate, find optimal value
    via golden-section search on the 1D objective.
    """
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 0.01:
        return a
    a = a / S
    n = len(a)
    b = fast_conv(a)

    best_a = a.copy()
    best_M = b.max()
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        order = np.random.permutation(n)

        for i in order:
            old_ai = a[i]

            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            k_2i = 2 * i if 2 * i <= 2 * n - 2 else -1

            # Outside max
            m_out_parts = []
            if k_lo > 0:
                m_out_parts.append(b[:k_lo].max())
            if k_hi < 2 * n - 2:
                m_out_parts.append(b[k_hi + 1:].max())
            m_out = max(m_out_parts) if m_out_parts else 0.0

            if old_ai < 1e-12:
                # Try a few small values
                best_new = old_ai
                best_obj = 2.0 * n * b.max()
                for cand in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    obj_val = cd_obj_func(cand, b, a, i, k_lo, k_hi, k_2i, m_out, n, 0.0)
                    if obj_val < best_obj:
                        best_obj = obj_val
                        best_new = cand

                if best_new != old_ai:
                    delta = best_new
                    a[i] = best_new
                    S_total = 1.0 + delta
                    for k in range(k_lo, k_hi + 1):
                        if k == k_2i:
                            b[k] += delta * delta
                        else:
                            j = k - i
                            b[k] += 2.0 * delta * a[j]
                    a = a / S_total
                    b = b / (S_total * S_total)
                    improved = True
                    M = b.max()
                    if M < best_M:
                        best_M = M
                        best_a = a.copy()
                continue

            # Golden section search
            lo = -old_ai * 0.999
            hi = old_ai * 2.0 + 0.01

            a1 = hi - gr * (hi - lo)
            b1 = lo + gr * (hi - lo)
            fa1 = cd_obj_func(a1, b, a, i, k_lo, k_hi, k_2i, m_out, n, old_ai)
            fb1 = cd_obj_func(b1, b, a, i, k_lo, k_hi, k_2i, m_out, n, old_ai)

            for _ in range(45):
                if fa1 < fb1:
                    hi = b1
                    b1 = a1
                    fb1 = fa1
                    a1 = hi - gr * (hi - lo)
                    fa1 = cd_obj_func(a1, b, a, i, k_lo, k_hi, k_2i, m_out, n, old_ai)
                else:
                    lo = a1
                    a1 = b1
                    fa1 = fb1
                    b1 = lo + gr * (hi - lo)
                    fb1 = cd_obj_func(b1, b, a, i, k_lo, k_hi, k_2i, m_out, n, old_ai)

            delta_opt = (a1 + b1) / 2.0
            new_ai = old_ai + delta_opt
            if new_ai < 0.0:
                new_ai = 0.0
                delta_opt = -old_ai

            if abs(new_ai - old_ai) < 1e-12:
                continue

            # Apply update
            delta = new_ai - old_ai
            S_total = 1.0 + delta
            a[i] = new_ai
            for k in range(k_lo, k_hi + 1):
                if k == k_2i:
                    b[k] += 2.0 * delta * old_ai + delta * delta
                else:
                    j = k - i
                    b[k] += 2.0 * delta * a[j]
            a = a / S_total
            b = b / (S_total * S_total)
            improved = True
            M = b.max()
            if M < best_M:
                best_M = M
                best_a = a.copy()

        if not improved:

def peak_reduction(a, deadline, max_rounds=50):
    """
    Targeted peak reduction: identify the top peaks in the convolution
    and directly reduce the contributing products.
    """
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 0.01:
        return a
    a = a / S
    n = len(a)
    b = fast_conv(a)
    best_a = a.copy()
    best_M = b.max()

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Find top peaks
        max_b = b.max()
        threshold = max_b * 0.95
        peaks = np.where(b >= threshold)[0]

        if len(peaks) == 0:
            break

        # For each peak, find the dominant contributing pair and reduce it
        changed = False
        for k in peaks[:10]:
            if time.monotonic() > deadline:
                break
            # Find j that maximizes a[j]*a[k-j]
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            if j_min > j_max:
                continue
            js = np.arange(j_min, j_max + 1)
            products = a[js] * a[k - js]
            j_star = js[np.argmax(products)]
            prod_val = products.max()
            if prod_val < 1e-10:
                continue

            # Reduce the larger of a[j_star] and a[k-j_star]
            a1 = a[j_star]
            a2 = a[k - j_star]
            if a1 >= a2:
                idx_reduce = j_star
                idx_incr = k - j_star
                frac = 0.15
            else:
                idx_reduce = k - j_star
                idx_incr = j_star
                frac = 0.15

            delta = a[idx_reduce] * frac
            if delta < 1e-8:
                continue

            # Move mass: reduce idx_reduce, increase idx_incr
            # But check if increasing idx_incr helps other peaks
            old_b_max = b.max()
            a[idx_reduce] -= delta
            a[idx_incr] += delta
            a = np.maximum(a, 0)
            S_new = a.sum()
            a = a / S_new
            b_new = fast_conv(a)
            M_new = b_new.max()

            if M_new < old_b_max - 1e-10:
                b = b_new
                changed = True
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
            else:
                # Revert
                a[idx_reduce] += delta
                a[idx_incr] -= delta
                a = np.maximum(a, 0)
                a = a / a.sum()
                b = b_new = fast_conv(a)

        if not changed:
            break

    return best_a


def multiplicative_update(a, deadline, max_iters=3000, eta_start=0.15):
    """
    Multiplicative (exponential) update: g[i] = a[i] * exp(-eta * subgrad[i])
    Then a = (1-t)*a + t*g.
    """
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 0.01:
        return a
    a = a / S
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    eta = eta_start

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()

        spread = M - b.min()
        if spread < 1e-14:
            break

        threshold = M - spread * 0.15
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        subgrad = compute_subgradient_fft(a, active)
        subgrad_centered = subgrad - subgrad.mean()

        scale = np.max(np.abs(subgrad_centered))
        if scale < 1e-15:
            break
        subgrad_norm = subgrad_centered / scale

        g = a * np.exp(-eta * subgrad_norm)
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 0.01:
            eta *= 0.5
            continue
        g = g / g_sum

        t = 0.15
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        S_new = a_new.sum()
        if S_new < 0.01:
            break
        a_new = a_new / S_new

        J_new_M = fast_conv(a_new).max()
        if J_new_M < M - 1e-13:
            a = a_new
            eta = max(eta * 0.995, 0.01)
        else:
            eta *= 0.7
            if eta < 1e-5:
                break

    return best_a


def diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n) / n)

    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())

    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())

    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())

    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())

    dbl = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())

    tri = np.exp(-10 * (t - 0.2) ** 2) + np.exp(-10 * (t - 0.5) ** 2) + np.exp(-10 * (t - 0.8) ** 2) + 0.02
    inits.append(tri / tri.sum())

    for freq in [1, 2, 3, 5, 7]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    for alpha in [0.3, 0.7, 1.5]:
        pl = (t + 0.01) ** alpha
        inits.append(pl / pl.sum())

    zz = 0.5 + 0.3 * np.abs(2 * (t * 4 % 1) - 1)
    inits.append(zz / zz.sum())

    # Edge-concentrated
    edge = np.exp(-3 * (t - 0.5) ** 2) * 0.1 + np.exp(-20 * t**2) + np.exp(-20 * (t-1)**2)
    inits.append(edge / edge.sum())

    # Anti-bell (U-shaped)
    ushape = (t - 0.5) ** 2 + 0.05
    inits.append(ushape / ushape.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Fast subgradient descent (main workhorse, replaces LP)
    Phase 2: Exact coordinate descent polishing
    Phase 3: Peak reduction targeting
    Phase 4: Subgradient + CD cycles
    Phase 5: Diverse initializations
    Phase 6: Multi-length exploration
    Phase 7: Final polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() >= 0.01:
                sc = float(evaluate_sequence(parent.tolist()))
                if np.isfinite(sc):
                    best_score = sc
                    best_seq = parent.copy()
                print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256.0
        best_seq = parent.copy()
        best_score = compute_J(parent)
        print(f"No parent; using uniform n=256, J={best_score:.6f}", flush=True)

    # === PHASE 1: Subgradient descent from parent ===
    n_p = parent.size
    p1_deadline = min(t_start + 280, deadline - 400)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Subgradient Descent (n={n_p}, budget~{p1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    a_sg = subgradient_descent(parent / parent.sum(), p1_deadline,
                               max_iters=200000, top_frac=0.1,
