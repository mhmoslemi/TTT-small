    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def golden_section_minimize(f, a, b, tol=1e-10, max_iter=50):
    """Golden section search for 1D minimization."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2

"""
Optimized minimizer for J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous version:
1. Constraint-reduced LP for faster mirror descent iterations
2. Adaptive line search with backtracking
3. Multi-length exploration with smart resampling
4. Faster coordinate descent with parabolic refinement
5. Better time management with phase deadlines

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_active_indices(conv, M, frac=0.1, max_count=100):
    """Get indices where conv is close to M (the top frac*M range)."""
    threshold = M * (1 - frac)
    active = np.where(conv >= threshold)[0]
    if len(active) > max_count:
        # Keep only the top max_count
        top_idx = np.argsort(conv[active])[-max_count:]
        active = active[top_idx]
    return active


def build_A_reduced(n, a, active_indices):
    """Build sparse constraint matrix for reduced LP."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(float(a[k - j]))
    m = len(active_indices)
    return csr_matrix((vals, (rows, cols)), shape=(m, n))


def lp_best_response_reduced(a, M, n_active=80, time_limit=10):
    """LP with reduced constraints: max sum(g) s.t. (a*g)[k] <= M for active k, g >= 0."""
    n = len(a)
    conv_a = fast_conv(a)
    active = get_active_indices(conv_a, M, frac=0.15, max_count=n_active)
    
    if len(active) == 0:
        return np.ones(n) / n
    
    A = build_A_reduced(n, a, active)
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters=200, deadline=None, n_active=80, lp_time=10):
    """LP mirror descent with reduced constraints and backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    best_obj = 2.0 * n * best_M  # since sum = 1
    no_improve = 0
    max_no_improve = 40
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        t0 = time.monotonic()
        b_conv = fast_conv(a)
        M = float(b_conv.max())
        
        g = lp_best_response_reduced(a, M, n_active=n_active, time_limit=lp_time)
        lp_elapsed = time.monotonic() - t0
        
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Backtracking line search
        improved = False
        step_sizes = [0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]
        for t in step_sizes:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                best_obj = 2.0 * n * best_M
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        else:
            no_improve = 0
        
        # Adapt LP time based on remaining time
        if deadline is not None:
            remaining = deadline - time.monotonic()
            if remaining < lp_elapsed * 3:
                break
    
    return best_a


def coord_descent_v2(a, max_sweeps=3000, deadline=None, random_order=True, seed=0):
    """Coordinate descent with parabolic refinement."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 80

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        if random_order and sweep % 4 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        sweep_improved = False

        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            # Evaluate at key points
            if old < 1e-10:
                candidates = [0.0, 1e-3, 1e-2, 0.05, 0.1]
            else:
                candidates = [0.0, 0.1, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 4.0]
                candidates = [x * old for x in candidates]
                candidates = [x for x in candidates if not np.isclose(x, old, rtol=1e-3)]

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)
            best_pairs = []  # (x, obj) for parabolic interpolation

            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s + delta
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                best_pairs.append((x, new_obj))

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            # Parabolic refinement around best_val
            if best_val != old and len(best_pairs) >= 3:
                # Sort by x value for interpolation
                best_pairs.sort()
                # Find 3 points around the minimum
                min_idx = min(range(len(best_pairs)), key=lambda i: best_pairs[i][1])
                
                if 0 < min_idx < len(best_pairs) - 1:
                    (x0, y0) = best_pairs[min_idx - 1]
                    (x1, y1) = best_pairs[min_idx]
                    (x2, y2) = best_pairs[min_idx + 1]
                    
                    denom = (x0 - x1) * (x0 - x2) * (x1 - x2)
                    if abs(denom) > 1e-20:
                        A = (x2 * (y1 - y0) + x1 * (y0 - y2) + x0 * (y2 - y1)) / denom
                        B = (x2*x2 * (y0 - y1) + x1*x1 * (y2 - y0) + x0*x0 * (y1 - y2)) / denom
                        if abs(A) > 1e-20:
                            xt = -B / (2 * A)
                            xt = max(0.0, min(xt, max(x0, x2)))
                            
                            # Evaluate at xt
                            delta_t = xt - old
                            conv[lo:hi + 1] += 2.0 * delta_t * a_slice
                            if valid_kq:
                                conv[kq] += delta_t * delta_t
                            new_s_t = s + delta_t
                            yt = 2.0 * n * conv.max() / (new_s_t * new_s_t) if new_s_t > 1e-10 else float('inf')
                            conv[lo:hi + 1] -= 2.0 * delta_t * a_slice
                            if valid_kq:
                                conv[kq] -= delta_t * delta_t
                            
                            if yt < best_local:
                                best_local = yt
                                best_val = xt

            # Apply best
            if best_val != old and abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s + delta
                sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            p = p / p.sum()
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.3, n), 0.001))
        # Scaled versions
        inits.append(np.maximum(p ** 1.5, 0.001))
        inits.append(np.maximum(p ** 0.7, 0.001))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine variants
    for freq in [1, 2, 3, 5]:
        for amp in [0.15, 0.3, 0.5]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01) / n)

    # Gaussian
    for center in [0.3, 0.5, 0.7]:
        for width in [0.15, 0.25]:
            inits.append(np.maximum(np.exp(-((t - center) / width) ** 2) + 0.05, 0.01) / n)

    # Smooth random
    ksize = max(7, n // 15)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.02, 0.01) / n)

    # Power spectrum based: try to match a target spectrum
    # (This is more exotic)
    for power in [0.5, 1.0, 1.5, 2.0]:
        freqs = np.fft.rfftfreq(n)
        spectrum = np.maximum(1.0 - freqs / (n / 2), 0.01) ** power
        sig = np.fft.irfft(spectrum * np.sqrt(n))
        if sig.size >= n:
            sig = sig[:n]
        inits.append(np.maximum(sig, 0.001) / max(1e-10, np.maximum(sig, 0).sum()))

    #Normalize all
    result = []
    for x in inits:
        x = np.maximum(x, 0)
        if x.sum() > 1e-10:
            result.append(x / x.sum())
    return result


def subgradient_polish(a, max_iters=3000, deadline=None, lr=0.05):
    """Subgradient descent with averaging."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-6 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]

        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad /= gn

        a_new = a - lr / (1 + 0.01 * it) * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        running_sum += a_new
        count += 1
        a = a_new

        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    if count > 20:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0)
        if a_avg.sum() > 1e-10:
            a_avg /= a_avg.sum()
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg.copy()
                best_M = M_avg

    return best_a


class Tracker:

"""
Optimized minimizer for J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements:
1. Constraint-reduced LP for fast mirror descent iterations
2. Backtracking line search in LP mirror descent

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_active_indices(conv, M, frac=0.15, max_count=100):
    """Get indices where conv is close to M."""
    threshold = M * (1.0 - frac)
    active = np.where(conv >= threshold)[0]
    if len(active) > max_count:
        sort_idx = np.argsort(conv[active])
        top = sort_idx[-max_count:]
        active = active[top]
    return active


def build_A_reduced(n, a, active_indices):
    """Build sparse constraint matrix using only active constraints."""
    rows = []
    cols = []
    vals = []
    for idx in range(len(active_indices)):
        k = int(active_indices[idx])
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(float(a[k - j]))
    m = len(active_indices)
    return csr_matrix((vals, (rows, cols)), shape=(m, n))


def lp_best_response(a, M, n_active=100, time_limit=12):
    """LP: max sum(g) s.t. (a*g)[k] <= M for active k, g >= 0."""
    n = len(a)
    conv_a = fast_conv(a)
    active = get_active_indices(conv_a, M, frac=0.15, max_count=n_active)

    if len(active) == 0:
        return np.ones(n) / n

    A = build_A_reduced(n, a, active)
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters=200, deadline=None, n_active=100, lp_time=12):
    """LP mirror descent with reduced constraints and backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    no_improve = 0
    max_no_improve = 50

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        t0 = time.monotonic()
        b_conv = fast_conv(a)
        M = float(b_conv.max())

        g = lp_best_response(a, M, n_active=n_active, time_limit=lp_time)
        lp_elapsed = time.monotonic() - t0

        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        improved = False
        step_sizes = [0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
        for t in step_sizes:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        else:
            no_improve = 0

        if deadline is not None:
            remaining = deadline - time.monotonic()
            if remaining < lp_elapsed * 3:
                break

    return best_a


def coord_descent_v2(a, max_sweeps=3000, deadline=None, random_order=True, seed=0):
    """Coordinate descent with parabolic refinement."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 80

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        if random_order and sweep % 4 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            if old < 1e-10:
                candidates = [0.0, 1e-3, 1e-2, 0.05, 0.1]
            else:
                candidates = [0.0, 0.1 * old, 0.25 * old, 0.5 * old,
                              0.75 * old, 1.0 * old, 1.25 * old, 1.5 * old,
                              2.0 * old, 4.0 * old]
                candidates = [x for x in candidates if not np.isclose(x, old, rtol=1e-3)]

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)
            evals = []

            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s + delta
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                evals.append((x, new_obj))

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            # Parabolic refinement
            if best_val != old and len(evals) >= 3:
                evals.sort()
                ys = np.array([e[1] for e in evals])
                min_idx = int(np.argmin(ys))

                if 0 < min_idx < len(evals) - 1:
                    x0, y0 = evals[min_idx - 1]
                    x1, y1 = evals[min_idx]
                    x2, y2 = evals[min_idx + 1]

                    denom = (x0 - x1) * (x0 - x2) * (x1 - x2)
                    if abs(denom) > 1e-30:
                        A_coef = (x2 * (y1 - y0) + x1 * (y0 - y2) + x0 * (y2 - y1)) / denom
                        B_coef = (x2 * x2 * (y0 - y1) + x1 * x1 * (y2 - y0) + x0 * x0 * (y1 - y2)) / denom
                        if abs(A_coef) > 1e-30:
                            xt = -B_coef / (2.0 * A_coef)
                            xt = max(0.0, min(xt, max(x0, x2) + abs(x1 - x0)))

                            delta_t = xt - old
                            conv[lo:hi + 1] += 2.0 * delta_t * a_slice
                            if valid_kq:
                                conv[kq] += delta_t * delta_t
                            new_s_t = s + delta_t
                            yt = 2.0 * n * conv.max() / (new_s_t * new_s_t) if new_s_t > 1e-10 else float('inf')
                            conv[lo:hi + 1] -= 2.0 * delta_t * a_slice
                            if valid_kq:
                                conv[kq] -= delta_t * delta_t

                            if yt < best_local:
                                best_local = yt
                                best_val = xt

            if best_val != old and abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s + delta

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            p = p / p.sum()
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.2, n), 0.001))
        inits.append(np.maximum(p ** 1.5, 0.0001))
        inits.append(np.maximum(p ** 0.7, 0.0001))
        # Rotation
        shift = rng.integers(1, max(2, n // 10))
        inits.append(np.roll(p, shift))
        inits.append(np.roll(p, -shift))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine variants
    for freq in [1, 2, 3, 5, 7]:
        for amp in [0.1, 0.2, 0.4]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01))

    # Sin variants
    for freq in [1, 2, 3]:
        for amp in [0.2, 0.4]:
            inits.append(np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t), 0.01))

    # Gaussian bumps
    for center in [0.25, 0.5, 0.75]:
        for width in [0.1, 0.2, 0.35]:
            inits.append(np.maximum(np.exp(-((t - center) / width) ** 2) + 0.02, 0.01))

    # Smooth random
    ksize = max(7, n // 12)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.01, 0.01))

    # Power spectrum based
    for power in [0.5, 1.0, 1.5, 2.0, 3.0]:
        freqs = np.fft.rfftfreq(n)
        max_f = n / 2
        spectrum = np.maximum(1.0 - freqs / max_f, 0.01) ** power
        sig = np.fft.irfft(spectrum * np.sqrt(n))
        if sig.size >= n:
            sig = sig[:n]
        sig = np.maximum(sig, 0.001)
        if sig.sum() > 1e-10:
            inits.append(sig)

    # V-shape and inverted V
    inits.append(np.maximum(0.3 + 0.7 * np.abs(2 * t - 1), 0.01))
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2 * t - 1), 0.01))

    # Plateau
    for width in [0.3, 0.5, 0.7]:
        plateau = np.where(np.abs(t - 0.5) < width / 2, 1.0, 0.2)
        inits.append(plateau)

    # Normalize all
    result = []
    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 0)
        s = x.sum()
        if s > 1e-10:
            result.append(x / s)
    return result


def subgradient_polish(a, max_iters=3000, deadline=None, lr_init=0.04):
    """Subgradient descent with averaging."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        lr = lr_init / (1.0 + 0.02 * it)
        b = fast_conv(a)
        M = b.max()
        tol = 1e-5 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]

        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad /= gn

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        running_sum += a_new
        count += 1
        a = a_new

        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()

    if count > 30:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0)
        if a_avg.sum() > 1e-10:
            a_avg /= a_avg.sum()
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M * (1.0 - 1e-12):
                best_a = a_avg.copy()

    return best_a


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return False
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
            return True
        return False

    def report(self, label=""):
        if self.best is not None:
            print(f"  [{label}] best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: Coordinate descent from diverse initializations (~350s)
    Phase 2: LP mirror descent - the main workhorse (~550s)
    Phase 3: Multi-length exploration (~250s)
    Phase 4: Final polishing with subgradient + CD + LP (~300s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = Tracker()

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0:
                parent = parent / parent.sum()
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        tracker.update(parent)
        print(f"Using default parent: n={parent.size}", flush=True)

    n_base = parent.size

    # === PHASE 1: Coordinate Descent ===
    p1_end = min(time.monotonic() + 350, deadline - 700)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Coordinate Descent (n={n_base}, budget={p1_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_inits(n_base, rng, parent)
    # Sort by initial quality
    init_list = []
    for a0 in inits:
        s = a0.sum()
        if s > 1e-10:
            M = fast_conv(a0).max()
            J = 2.0 * n_base * M / (s * s)
            init_list.append((J, a0))
    init_list.sort(key=lambda x: x[0])

    # Run CD on top initializations
    for idx in range(min(10, len(init_list))):
        if time.monotonic() > p1_end:
            break
        J_init, a0 = init_list[idx]
        t0 = time.monotonic()
        a_opt = coord_descent_v2(a0, max_sweeps=5000, deadline=p1_end,
                                  random_order=True, seed=seed + idx)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_opt)
        print(f"  Init {idx:2d} (J0={J_init:.4f}): J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
        tracker.update(a_opt)

    tracker.report("Phase 1")

    # === PHASE 2: LP Mirror Descent ===
    p2_end = min(time.monotonic() + 550, deadline - 250)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (budget={p2_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        a_start = tracker.best.copy()
        n = a_start.size

        # Adapt parameters to sequence length
        if n <= 384:
            n_active = 80
            lp_time = 10
            max_iters = 400
        elif n <= 640:
            n_active = 100
            lp_time = 15
            max_iters = 300
        elif n <= 1024:
            n_active = 120
            lp_time = 20
            max_iters = 200
        else:
            n_active = 150
            lp_time = 30
            max_iters = 100

        t0 = time.monotonic()
        a_lp = lp_mirror_descent(a_start, max_iters=max_iters, deadline=p2_end,
                                  n_active=n_active, lp_time=lp_time)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_lp)
        print(f"  LP MD pass 1: J={J_val:.8f} n={n} ({elapsed:.1f}s)", flush=True)
        tracker.update(a_lp)

        # Second LP pass with wider active set
        if time.monotonic() < p2_end - 30:
            a_lp2 = lp_mirror_descent(a_lp, max_iters=100, deadline=p2_end,
                                       n_active=min(n_active + 60, 250), lp_time=10)
            J_val2 = compute_J(a_lp2)
            print(f"  LP MD pass 2: J={J_val2:.8f}", flush=True)
            tracker.update(a_lp2)

        # Third pass if time remains
        if time.monotonic() < p2_end - 20:
            a_lp3 = lp_mirror_descent(tracker.best, max_iters=50, deadline=p2_end,
                                       n_active=n_active, lp_time=8)
            J_val3 = compute_J(a_lp3)
            print(f"  LP MD pass 3: J={J_val3:.8f}", flush=True)
            tracker.update(a_lp3)

    tracker.report("Phase 2")

    # === PHASE 3: Multi-length exploration ===
    p3_end = min(time.monotonic() + 250, deadline - 100)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration (budget={p3_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        candidate_lengths = set()
        for delta in [-96, -64, -32, 32, 64, 96, 128, 192]:
            new_n = base_n + delta
            if 128 <= new_n <= 2048:
                candidate_lengths.add(new_n)
        for nice_n in [256, 320, 384, 448, 480, 512, 544, 576, 640, 704, 768, 832, 896, 960, 1024, 1280]:
            if 128 <= nice_n <= 2048:
                candidate_lengths.add(nice_n)
        candidate_lengths.discard(base_n)
        candidate_lengths = sorted(candidate_lengths)

        tried = 0
        for new_n in candidate_lengths:
            if time.monotonic() > p3_end - 20 or tried >= 5:
                break
            tried += 1
            t0 = time.monotonic()
            a_ext = resample_seq(tracker.best, new_n)
            if a_ext.sum() < 1e-10:
                continue
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            a_opt = coord_descent_v2(a_ext, max_sweeps=2000, deadline=p3_end - 20,
                                      random_order=True, seed=seed + 2000 + new_n)
            elapsed = time.monotonic() - t0
            J_val = compute_J(a_opt)
            print(f"  n={new_n:4d}: J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
            tracker.update(a_opt)

    tracker.report("Phase 3")

    # === PHASE 4: Final Polish ===
    p4_end = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Final Polish (budget={p4_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None and time.monotonic() < p4_end - 20:
        # Subgradient polishing
        sg_end = min(time.monotonic() + 80, p4_end - 100)
        if time.monotonic() < sg_end:
            t0 = time.monotonic()
            a_sg = subgradient_polish(tracker.best, max_iters=3000, deadline=sg_end, lr_init=0.03)
            elapsed = time.monotonic() - t0
            J_val = compute_J(a_sg)
            print(f"  Subgradient: J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
            tracker.update(a_sg)

        # Coordinate descent polish
        cd_end = min(time.monotonic() + 80, p4_end - 50)
        if time.monotonic() < cd_end:
            a_cd = coord_descent_v2(tracker.best, max_sweeps=3000, deadline=cd_end,
                                     random_order=True, seed=seed + 9000)
            J_val = compute_J(a_cd)
            print(f"  CD polish: J={J_val:.8f}", flush=True)
            tracker.update(a_cd)

        # Final LP pass
        if time.monotonic() < p4_end - 10:
            n_best = tracker.best.size
            n_act = min(100, max(40, n_best // 5))
            a_final = lp_mirror_descent(tracker.best, max_iters=40,
                                         deadline=p4_end, n_active=n_act, lp_time=8)
            J_val = compute_J(a_final)
            print(f"  Final LP: J={J_val:.8f}", flush=True)
            tracker.update(a_final)

        # One more CD after LP
        if time.monotonic() < p4_end - 5:
            a_final2 = coord_descent_v2(tracker.best, max_sweeps=1000, deadline=p4_end,
                                        random_order=True, seed=seed + 9999)
            J_val = compute_J(a_final2)
            print(f"  Final CD: J={J_val:.8f}", flush=True)
            tracker.update(a_final2)

    tracker.report("Phase 4")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
