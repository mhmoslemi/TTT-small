"""
Optimized search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Key improvements:
1. Precise per-coordinate 1D optimization via ternary search (not grid)
2. LP mirror descent with adaptive constraint relaxation

4. Active-set balancing refinement
5. Better time management and diverse initializations

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ScoreTracker:

def precise_coord_descent(a, max_sweeps, deadline, seed=0):
    """
    Coordinate descent with precise 1D per-coordinate optimization via ternary search.
    Much more effective than grid-based approaches.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0

    conv = fast_conv(a)
    best_a = a.copy()
    best_obj = conv.max() / (s * s)

    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = min(30, max_sweeps // 3)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if sweep % 2 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        sweep_improved = False

        for i in order:
            old = a[i]
            if old < 1e-12 and conv.max() / (s * s) < 1e-15:
                continue

            # Precompute w_k = a[k-i] for valid k (length 2n-1)
            w = np.zeros(2 * n - 1)
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            w[lo:hi + 1] = a[0:n]
            kq = 2 * i
            has_quad = kq <= 2 * n - 2

            def eval_obj(t):
                """Evaluate max_k(conv_k + 2t*w_k + t^2*delta) / (s+t)^2"""
                new_s = s + t
                if new_s < 1e-10:
                    return float('inf')
                # Vectorized: conv + 2*t*w
                val = conv + 2.0 * t * w
                if has_quad:
                    # At k=kq: conv[kq] + 2*t*a_i + t^2 (the w[kq] = a[i] = old)
                    val[kq] = conv[kq] + 2.0 * t * old + t * t
                return val.max() / (new_s * new_s)

            # Coarse search to find good bracket
            t_candidates = [-old, -0.5 * old, 0.0, 0.5 * old, old, 1.5 * old, 2.0 * old,
                            3.0 * old, s / n]
            t_candidates = [t for t in t_candidates if t >= -old + 1e-12]
            # Add some random candidates
            for _ in range(3):
                t_c = rng.uniform(-old, 3 * max(old, s / n))
                t_candidates.append(t_c)

            best_t = 0.0
            best_local_obj = eval_obj(0.0)
            for t_c in t_candidates:
                obj_c = eval_obj(t_c)
                if obj_c < best_local_obj:
                    best_local_obj = obj_c
                    best_t = t_c

            # Fine ternary search
            spread = max(0.3 * abs(best_t) + 0.3 * old + 0.01 * s / n, 1e-6)
            tlo = max(-old + 1e-12, best_t - spread)
            thi = best_t + spread * 2

            for _ in range(35):
                m1 = tlo + (thi - tlo) / 3.0
                m2 = thi - (thi - tlo) / 3.0
                f1 = eval_obj(m1)
                f2 = eval_obj(m2)
                if f1 < f2:
                    thi = m2
                else:
                    tlo = m1

            t_opt = (tlo + thi) / 2.0
            obj_opt = eval_obj(t_opt)

            current_obj = conv.max() / (s * s)
            if obj_opt < current_obj * (1 - 1e-12) and abs(t_opt) > 1e-12:
                new_val = old + t_opt
                if new_val < 0:
                    new_val = 0.0
                    t_opt = -old
                    obj_opt = eval_obj(t_opt)

                # Apply update in-place
                conv[lo:hi + 1] += 2.0 * t_opt * a[0:n]
                if has_quad:
                    conv[kq] += t_opt * t_opt  # additional quadratic term (2*t*old already done)
                a[i] = new_val
                s += t_opt
                sweep_improved = True

        if sweep_improved:
            current_obj = conv.max() / (s * s)
            if current_obj < best_obj * (1 - 1e-12):
                best_obj = current_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    a = np.maximum(best_a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    return a


def build_A_sparse(n, a):
    """Build sparse constraint matrix A[k,j] = a[k-j] + a[j-k] for valid indices."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            # (a*g)_k = sum_j a_j * g_{k-j}
            # For LP: we want A[k,j] such that (A @ g)[k] = sum_j A[k,j]*g_j = (a*g)_k
            # (a*g)_k = sum_j a_j * g_{k-j}, so coefficient of g_j is a_{k-j}
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_mirror_descent(a, max_iters, deadline, relax=1.0):
    """
    LP mirror descent: iteratively solve LP max sum(g) s.t. (a*g)_k <= M*relax, g>=0
    and update a = (1-t)*a + t*g.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_conv = fast_conv(a)
        M = float(b_conv.max())
        M_eff = M * relax

        t0 = time.monotonic()
        A = build_A_sparse(n, a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_eff)
        bounds = [(0, None)] * n
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': min(60, max(5, 0.5 * (deadline - time.monotonic()))),
                                 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
            else:
                break
        except Exception:
            break

        lp_time = time.monotonic() - t0
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search
        improved = False
        base_t = max(0.05, 0.4 * (1.0 - 0.7 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.7, base_t * 0.4, base_t * 0.2, 0.05, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-12):
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                break

        # Check if we have time for another LP
        remaining = deadline - time.monotonic()
        if remaining < lp_time * 2.5:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_refine(a, max_iters, deadline, lr_init=0.1):
    """Subgradient descent with averaging for polishing."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lr = lr_init / (it + 1) ** 0.5
        b = fast_conv(a)
        M = b.max()
        tol = 1e-6 * M if M > 0 else 1e-10
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]

        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        a_new = a - lr * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        running_sum += a_new
        count += 1
        a = a_new

    if count > 20:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0.0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return best_a


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initial sequences for multi-start."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-8)
        p = p / p.sum()
        inits.append(p.copy())
        for noise_level in [0.02, 0.05, 0.1, 0.2]:
            noisy = np.maximum(p + rng.normal(0, noise_level, n), 1e-6)
            inits.append(noisy / noisy.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine/sine waves
    for freq in [1, 2, 3, 5]:
        for amp in [0.1, 0.3, 0.5]:
            sig = np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t), 0.01)
            inits.append(sig / sig.sum())
            cos = np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01)
            inits.append(cos / cos.sum())

    # Gaussian bumps
    for center in [0.25, 0.5, 0.75]:
        for width in [0.08, 0.15, 0.25]:
            g = np.exp(-((t - center) / width) ** 2) + 0.05
            inits.append(g / g.sum())

    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0.1, 1.0, n), kernel, mode='same')
        inits.append(smooth / smooth.sum())

    # Power-law
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        pl = np.maximum(t ** alpha + (1 - t) ** alpha + 0.01, 0.01)
        inits.append(pl / pl.sum())

    return inits
