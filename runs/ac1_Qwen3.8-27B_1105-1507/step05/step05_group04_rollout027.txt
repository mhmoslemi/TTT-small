            best_M = M
            best_a = a.copy()
        exp_b = np.exp(tau * (b - M))
        p = exp_b / exp_b.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            seg = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * pk * seg
        v = 0.85 * v + lr * grad
        a_new = np.maximum(a - v, 0.0)
        sn = a_new.sum()
        if sn > 1e-10:
            a = a_new / sn

    return best_a


def solve_lp_direction(a, M):
    n = len(a)
    b = fast_conv(a)
    threshold = np.percentile(b, 40.0)
    active_k = np.where(b >= threshold)[0]

    if len(active_k) < 10:
        active_k = np.arange(2 * n - 1)

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])

    A = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, np.inf)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                ag = fftconvolve(a, g)[:2*n-1]
                if ag.max() <= M * (1.0 + 1e-4):
                    return g / g.sum()
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_palindrome(a):
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def lp_perturb_and_cd(a, deadline, tracker):
    """LP direction + coordinate descent hybrid."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10).copy()
    a = a / a.sum()
    b = fast_conv(a)
    M = b.max()

    g = solve_lp_direction(a, M)
    if g is None:
        return a, compute_J(a)

    # Try multiple mixing ratios
    best_a = a.copy()
    best_J = compute_J(a)

    for t in [0.3, 0.5, 0.7, 1.0]:
        if time.monotonic() > deadline - 30:
            break
        a_trial = np.maximum((1.0 - t) * a + t * g, 1e-10)
        a_trial = a_trial / a_trial.sum()
        # Quick CD
        a_opt, J_opt = coord_descent(a_trial, min(60.0, deadline - time.monotonic() - 5), verbose=False)
        if J_opt < best_J:
            best_J = J_opt
            best_a = a_opt.copy()

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_a = None
    best_J = float('inf')

    def update_tracker(a, label=""):
        nonlocal best_a, best_J
        if a is None or len(a) == 0:
            return
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={J:.10f} n={a.size} t={elapsed:.0f}s [{label}]", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_tracker(parent, "parent")
            print(f"Parent loaded: n={parent.size}, J={best_J:.10f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        update_tracker(parent, "default")

    n = parent.size

    # === PHASE 1: Coordinate descent on parent (50% of time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Coordinate Descent on parent (n={n})", flush=True)
    print(f"{'='*60}", flush=True)

    p1_time = (deadline - time.monotonic()) * 0.50
    p1_deadline = time.monotonic() + p1_time

    a_cd, J_cd = coord_descent(parent, p1_time, verbose=True)
    update_tracker(a_cd, "cd_parent")

    # === PHASE 2: Smooth gradient + CD from perturbed starts (20%) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: Diversified restarts", flush=True)
    print(f"{'='*60}", flush=True)

    p2_time = (deadline - time.monotonic()) * 0.25
    p2_deadline = time.monotonic() + p2_time

    # Generate diverse perturbations
    n_perms = 0
    t_arr = np.linspace(0, 1, n, endpoint=False)

    perturbations = []
    # Smooth frequency perturbations
    for f in [2, 3, 5, 7, 11]:
        for ph in [0, np.pi / 2, np.pi]:
            per = np.cos(2 * np.pi * f * t_arr + ph)
            perturbations.append(per)
    # Palindrome
    perturbations.append(make_palindrome(a_cd) - a_cd)
    # Edge concentration
    perturbations.append(np.exp(-5 * t_arr**2) + np.exp(-5 * (t_arr-1)**2) - 1.0)

    for pidx, pert in enumerate(perturbations):
        if time.monotonic() > p2_deadline - 20:
            break
        for scale in [0.05, 0.1, 0.2]:
            if time.monotonic() > p2_deadline - 10:
                break
            a_pert = np.maximum(a_cd + scale * pert, 1e-8)
            a_pert = a_pert / a_pert.sum()
            remaining = p2_deadline - time.monotonic() - 5
            if remaining < 10:
                break
            a_opt, J_opt = coord_descent(a_pert, min(remaining, 40.0), verbose=False)
            update_tracker(a_opt, f"pert{pidx}_s{scale}")
            n_perms += 1

    print(f"  Phase 2 done: {n_perms} perturbations tried, best J={best_J:.10f}", flush=True)

    # === PHASE 3: LP perturbation + CD (15%) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: LP Perturbation + CD", flush=True)
    print(f"{'='*60}", flush=True)

    p3_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.15

    if time.monotonic() < p3_deadline - 30:
        a_lp, J_lp = lp_perturb_and_cd(best_a, p3_deadline, None)
        update_tracker(a_lp, "lp_perturb")

    # === PHASE 4: Multi-length exploration (10%) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    p4_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.10

    if time.monotonic() < p4_deadline - 10:
        for n_try in [n * 2, n + 128, n - 128, int(n * 1.5)]:
            if time.monotonic() > p4_deadline - 5:
                break
            if n_try < 64 or n_try > 4096:
                continue
            a_ext = resample_seq(best_a, n_try)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()
            remaining = p4_deadline - time.monotonic() - 5
            if remaining < 15:
                break
            a_opt, J_opt = coord_descent(a_ext, min(remaining, 30.0), verbose=False)
            update_tracker(a_opt, f"n={n_try}")

    # === PHASE 5: Final polishing with extended CD ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 5: Final Polishing", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"  Remaining: {deadline - time.monotonic():.0f}s", flush=True)

    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 5
        a_final, J_final = coord_descent(best_a, remaining, verbose=True)
        update_tracker(a_final, "final_polish")

    # Fallback
    if best_a is None:
        best_a = np.ones(128)
        best_J = compute_J(best_a)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_J:.10f} n={best_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_a]
