Multi-stage optimization combining subgradient descent, LP polishing, and spectral
shaping to minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

    if n == 0:
        return np.inf

"""
Multi-stage optimizer combining multiplicative subgradient descent, LP polishing,
spectral equalization, and coordinate descent to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses top-p peak averaging, adaptive learning rates, and Nesterov-inspired momentum.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=build_toeplitz(a), b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g
    except Exception:
        pass
    return None


def multiplicative_sgd(a, time_budget, verbose=False):
    """
    Multiplicative subgradient descent with top-p peak averaging.
    Uses exp(-lr * sg) update which preserves positivity naturally.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lr = 0.05
    patience = 300
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        progress = 1.0 - remaining / max(time_budget, 1e-10)

        # Adaptive number of peaks to average
        p = int(3 + 17 * progress)

        conv = compute_conv(a)
        if p >= len(conv):
            top_p = np.arange(len(conv))
        else:
            top_p = np.argpartition(conv, -p)[-p:]

        # Compute averaged subgradient: sg[j] = (1/p) * sum_{k in top_p} 2*a[k-j]
        sg = np.zeros(n)
        for k in top_p:
            k = int(k)
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end + 1] += 2.0 * a[k - np.arange(j_start, j_end + 1)]
        sg /= len(top_p)

        # Multiplicative update: a[j] *= exp(-lr * sg[j])
        # This preserves positivity and is like exponential weights
        a_new = a * np.exp(-lr * sg)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a = best_a.copy()
            lr *= 0.5
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slowly decrease lr
            lr = max(0.0005, lr * 0.9998)
        else:
            stall += 1
            if stall > patience:
                # Perturb to escape

                a = a / np.sum(a)
                lr = min(0.1, max(lr, 0.02))

        if verbose and it % 5000 == 0:

def spectral_equalize(a, num_iters=30, time_budget=10.0):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    for _ in range(num_iters):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a)
        mag = np.abs(fa)
        phase = np.angle(fa)

        # Try to equalize the magnitude spectrum
        choice = rng.integers(0, 3)
        if choice == 0:
            # Soft equalization toward mean
            target = np.median(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha = rng.uniform(0.1, 0.4)
            mag_new = (1 - alpha) * mag + alpha * target
        elif choice == 1:
            # Reduce peaks in spectrum
            threshold = np.percentile(mag, 80)
            mag_new = np.where(mag > threshold, threshold + (mag - threshold) * 0.3, mag)
        else:
            # Random spectral perturbation
            mag_new = mag * np.exp(rng.normal(0, 0.05, len(mag)))

        fa_new = mag_new * np.exp(1j * phase)
        a_new = irfft(fa_new, n)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def coordinate_descent(a, time_budget=30.0, verbose=False):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        j = int(rng.integers(0, n))
        old_val = a[j]

        # Optimize a[j] by trying several values
        best_val = old_val
        best_local_obj = best_obj

        # Try geometric progression around current value
        for multiplier in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]:
            new_val = old_val * multiplier
            if new_val < 0:
                new_val = 0.0
            a[j] = new_val
            s_new = np.sum(a)
            if s_new < 1e-10:
                a[j] = old_val
                continue
            obj_try = compute_obj(a)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_val = new_val

        a[j] = best_val
        if best_local_obj < best_obj - 1e-12:
            best_obj = best_local_obj
            best_a = a.copy()

    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    choice = rng.integers(0, 8)

    if choice == 0:
        # Monotone increasing (motivated by rearrangement inequality)
        a = rng.uniform(0.5, 2.0, n)
        a = np.cumsum(a) / np.sum(a)
    elif choice == 1:
        # Cosine window
        a = np.cos(np.pi * k / (n - 1)) ** 2 + 0.1
    elif choice == 2:
        # Gaussian
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.4)
        a = np.exp(-0.5 * ((k - center) / width) ** 2) + 0.05
    elif choice == 3:
        # Cosine with frequency
        f = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        # Sine squared
        f = rng.uniform(1.0, 4.0)
        a = np.sin(np.pi * f * k / n) ** 2 + 0.1
    elif choice == 5:
        # Monotone decreasing
        a = rng.uniform(0.5, 2.0, n)
        a = np.cumsum(a[::-1])[::-1] / np.sum(a)
    elif choice == 6:
        # V-shape
        center = n // 2
        a = np.abs(k - center) / center + 0.3
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines multiplicative subgradient descent, LP polishing,
    spectral equalization, and coordinate descent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.6f}")

    # Phase 1: Multiplicative SGD on parent
    if best_seq is not None:
        n_parent = len(best_seq)
        print(f"\n=== Phase 1: Multiplicative SGD (n={n_parent}) ===")
        alloc = min(400.0, (deadline - time.monotonic()) * 0.35)
        a_sgd, obj_sgd = multiplicative_sgd(best_seq, time_budget=alloc, verbose=True)
        if obj_sgd < best_score:
            best_score = obj_sgd
            best_seq = a_sgd.copy()
            print(f"  After MultSGD: {best_score:.6f}")
        else:
            print(f"  After MultSGD: {obj_sgd:.6f} (no improvement)")

    # Phase 2: LP Polish
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Phase 2: LP Polish ===")
        remaining = deadline - time.monotonic()
        alloc = min(200.0, remaining * 0.25)
        a_lp, obj_lp = lp_polish(best_seq, time_budget=alloc, verbose=True)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  After LP: {best_score:.6f}")
        else:
            print(f"  After LP: {obj_lp:.6f} (no improvement)")

    # Phase 3: Spectral + Coordinate
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Phase 3: Spectral + Coordinate ===")
        remaining = deadline - time.monotonic()

        # Spectral equalization
        alloc_spec = min(30.0, remaining * 0.15)
        a_spec, obj_spec = spectral_equalize(best_seq, num_iters=50, time_budget=alloc_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  After spectral: {best_score:.6f}")

        # Coordinate descent
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            alloc_cd = min(60.0, remaining * 0.3)
            a_cd, obj_cd = coordinate_descent(best_seq, time_budget=alloc_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  After coordinate: {best_score:.6f}")

        # Another round of MultSGD
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            alloc2 = min(100.0, remaining * 0.4)
            a_sgd2, obj_sgd2 = multiplicative_sgd(best_seq, time_budget=alloc2)
            if obj_sgd2 < best_score:
                best_score = obj_sgd2
                best_seq = a_sgd2.copy()
                print(f"  After MultSGD round 2: {best_score:.6f}")

    # Phase 4: Multi-length exploration
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    if best_seq is not None:
        n_best = len(best_seq)
        n_values = sorted(set([
            max(100, n_best - 100),
            max(100, n_best - 50),
            n_best,
            n_best + 50,
            n_best + 100,
            n_best + 200,
        ]))
    else:
        n_values = [400, 500, 600, 700, 800]

    n_values = [n for n in n_values if 100 <= n <= 2000]

    for n in n_values:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        alloc = min(remaining * 0.2, 120.0)
        alloc = max(15.0, alloc)

        print(f"\n  N={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        if best_seq is not None:
            a_start = resample_sequence(best_seq, n)
        else:
            a_start = random_init(n, seed + n * 7919)

        # MultSGD
        a_opt, obj_opt = multiplicative_sgd(a_start, time_budget=alloc * 0.5)

        # LP polish
        if time.monotonic() < deadline - 30:
            a_opt, obj_opt = lp_polish(a_opt, time_budget=min(alloc * 0.3, 40.0))

        # Spectral
        if time.monotonic() < deadline - 30:
            a_opt, obj_opt = spectral_equalize(a_opt, num_iters=20, time_budget=min(alloc * 0.1, 10.0))

        print(f"  N={n}: obj={obj_opt:.6f}")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Phase 5: Final intensive polish
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"  Best before: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  Polishing n={n_best}, {remaining:.0f}s remaining")

        # Round: MultSGD + LP
        alloc1 = min(remaining * 0.3, 150.0)
        a_cur, obj1 = multiplicative_sgd(best_seq, time_budget=alloc1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a_cur.copy()
            print(f"  Final MultSGD: {best_score:.6f}")

        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            a_cur, obj2 = lp_polish(best_seq, time_budget=min(remaining * 0.3, 80.0))
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_cur.copy()
                print(f"  Final LP: {best_score:.6f}")

        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            a_cur, obj3 = coordinate_descent(best_seq, time_budget=min(remaining * 0.3, 40.0))
            if obj3 < best_score:
                best_score = obj3
                best_seq = a_cur.copy()
                print(f"  Final CD: {best_score:.6f}")

        # One more MultSGD with smaller lr
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            a_cur, obj4 = multiplicative_sgd(best_seq, time_budget=min(remaining * 0.5, 30.0))
            if obj4 < best_score:
                best_score = obj4
                best_seq = a_cur.copy()
                print(f"  Final MultSGD2: {best_score:.6f}")

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
