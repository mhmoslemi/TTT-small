    x = a_init.copy()  # current iterate
    z = a_init.copy()  # momentum variable
    t = 1.0

    L = 2.0 * np.sum(w) + 1e-8
    lr = 1.0 / L
    
    for _ in range(n_steps):
        # Gradient of f(x) = x^T Q x at z
        grad = 2.0 * toeplitz_matvec(w, z)

        x_new = project_simplex(x_new)

        z = x_new + (t - 1.0) / t_new * (x_new - x)
        
        x = x_new
        t = t_new

    T_min = 0.001
    cooling = 0.999
    
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        if T < T_min:
            T = T_min
        
        # Random block move
        block_size = rng.integers(1, max(2, n // 10))
        start_idx = rng.integers(0, n - block_size)
        
        a_new = a.copy()
        delta = rng.normal(0, T, block_size)
        a_new[start_idx:start_idx+block_size] += delta
        a_new = np.maximum(a_new, 0)
        
        # Project to simplex
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
        else:
            a_new = a.copy()
        
        obj_new = compute_obj(a_new)
        obj_cur = compute_obj(a)
        
        # Metropolis acceptance
        if obj_new < obj_cur or rng.random() < np.exp(-(obj_new - obj_cur) / T):
            a = a_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
        
        T *= cooling
    
    return best_a

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative weights QP (FISTA with FFT matvec), LP-based directions,
simulated annealing, and frequency-domain shaping. Explores multiple n values.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v):
    """Project v onto probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_last = rho[-1]
    theta = (cssv[rho_last] - 1) / (rho_last + 1)
    return np.maximum(v - theta, 0)


def toeplitz_matvec(w, a):
    """Compute Q @ a where Q is Toeplitz with Q[i,j] = w[i+j].
    Uses FFT-based cross-correlation."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    W = np.zeros(size, dtype=np.complex128)
    W[:len(w)] = w
    A = np.zeros(size, dtype=np.complex128)
    A[:n] = a
    # cross-correlation: sum_j w[j+i]*a[j] = ifft(fft(w) * conj(fft(a)))
    result = np.real(np.fft.ifft(np.fft.fft(W) * np.conj(np.fft.fft(A))))
    return result[:n]


def fista_qp(w, a_init, n_steps=300, verbose=False):
    """Solve min a^T Q a s.t. a>=0, sum(a)=1 using FISTA.
    Q is Toeplitz with first row (upper) w."""
    n = len(a_init)
    x = a_init.copy()
    z = a_init.copy()
    t_mom = 1.0

    # Estimate Lipschitz constant via circulant embedding
    size = 1
    while size < 2 * n:
        size <<= 1
    W_ext = np.zeros(size)
    W_ext[:len(w)] = w
    # Eigenvalues of circulant embed
    eig_vals = np.abs(np.fft.fft(W_ext))
    L = 2.0 * float(np.max(eig_vals)) + 1e-10
    lr = 1.0 / L

    obj_init = float(a_init @ toeplitz_matvec(w, a_init))

    for step in range(n_steps):
        # Gradient at z
        grad = 2.0 * toeplitz_matvec(w, z)

        # Gradient step
        x_new = z - lr * grad

        # Project onto simplex
        x_new = project_simplex(x_new)

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        z = x_new + (t_mom - 1.0) / t_new * (x_new - x)

        x = x_new.copy()
        t_mom = t_new

        if verbose and step % 50 == 0:
            obj_cur = float(x @ toeplitz_matvec(w, x))
            print(f"      FISTA step {step}: obj={obj_cur:.8f} (init={obj_init:.8f})")

    return x

            # Or perturb a
            rng = np.random.default_rng(outer * 7919)
            a_pert = a * np.exp(rng.normal(0, 0.03, n))
            a_pert = np.maximum(a_pert, 0)
            sp = np.sum(a_pert)
            if sp > 1e-10:
                a_pert = a_pert / sp
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    a = a_pert

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix for LP: (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_direction(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c_obj = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=build_toeplitz(a), b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0


def backtracking_line_search(a, direction, obj_cur, max_iter=50):
    """Backtracking line search along a direction (a + t*(direction - a))."""
    n = len(a)
    best_t = 0.0
    best_obj = obj_cur
    t = 1.0
    for _ in range(max_iter):
        a_try = a + t * (direction - a)
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            t *= 0.5
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
        t *= 0.5
        if t < 1e-6:
            break
    if best_t > 0:
        a_result = a + best_t * (direction - a)
        a_result = np.maximum(a_result, 0)
        s = np.sum(a_result)
        if s > 1e-10:
            a_result = a_result / s
            return a_result, best_obj
    return a.copy(), best_obj


def lp_alternation_enhanced(a, time_budget=60.0):
    """Enhanced LP alternation with backtracking line search and adaptive relaxation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    relax = 1.0
    rng = np.random.default_rng(54321)

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, S = solve_lp_direction(a, M, relax=relax)
        if g is None or S <= 1.0 + 1e-8:
            stall += 1
            if stall > 20:
                relax = min(relax + 0.1, 2.0)
                if relax > 1.5:
                    a = a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    s2 = np.sum(a)
                    if s2 > 1e-10:
                        a = a / s2
                    else:
                        a = best_a.copy()
                    relax = 1.0
                    stall = 0
            continue

        # Backtracking line search
        g_normalized = g / S
        a_new, obj_new = backtracking_line_search(a, g_normalized, best_obj)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            relax = max(1.0, relax - 0.05)
        else:
            stall += 1
            if stall > 15:
                # Peak-targeted perturbation
                sorted_idx = np.argsort(conv)[::-1]
                top_k = int(sorted_idx[rng.integers(0, 5)])
                affected = []
                for i in range(n):
                    j = top_k - i
                    if 0 <= j < n:
                        affected.append(i)
                if affected:
                    a_pert = a.copy()
                    for pos in affected:
                        a_pert[pos] *= rng.uniform(0.7, 0.95)
                    a_pert = np.maximum(a_pert, 0)
                    sp = np.sum(a_pert)
                    if sp > 1e-10:
                        a_pert = a_pert / sp
                        obj_pert = compute_obj(a_pert)
                        if obj_pert < best_obj:
                            best_obj = obj_pert
                            best_a = a_pert.copy()
                            a = a_pert
                            stall = 0
                            relax = 1.0
                            continue
                # Random restart of relaxation
                relax = rng.uniform(1.0, 1.3)
                stall = 0

    return best_a, best_obj


def simulated_annealing_block(a_init, time_budget=60.0):
    """Simulated annealing with block moves in time domain."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(77777)

    T = 0.005
    T_min = 0.0001
    cooling_rate = 0.9995

    while time.monotonic() - start < time_budget:
        if T < T_min:
            T = T_min

        # Random block perturbation
        block_size = rng.integers(2, max(3, n // 20))
        start_idx = rng.integers(0, n - block_size)

        a_new = a.copy()
        a_new[start_idx:start_idx + block_size] += rng.normal(0, T * 2, block_size)
        a_new = np.maximum(a_new, 0)

        sp = np.sum(a_new)
        if sp < 1e-10:
            T *= cooling_rate
            continue
        a_new = a_new / sp

        obj_new = compute_obj(a_new)

        # Metropolis acceptance
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        if obj_new <= best_obj or rng.random() < np.exp(-(obj_new - best_obj) / T):
            a = a_new

        T *= cooling_rate

    return best_a, best_obj


def freq_domain_shape(a_init, time_budget=30.0):
    """Frequency domain shaping: modify magnitude spectrum to flatten autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(31415)

    while time.monotonic() - start < time_budget:
        fa = rfft(a)
        mag = np.abs(fa)
        phase = np.angle(fa)

        choice = rng.integers(0, 4)
        if choice == 0:
            # Soft-clip high magnitude bins
            thresh = np.percentile(mag, 90)
            mag_new = np.where(mag > thresh, thresh + 0.3 * (mag - thresh), mag)
        elif choice == 1:
            # Spectral flattening
            mean_mag = np.mean(mag)
            alpha = rng.uniform(0.1, 0.5)
            mag_new = (1 - alpha) * mag + alpha * mean_mag
        elif choice == 2:
            # Low-pass tilt (emphasize low freqs for smoother time signal)
            freqs = np.linspace(0, 1, len(mag))
            tilt = rng.uniform(0.3, 1.5)
            mag_new = mag / (1.0 + tilt * freqs)
            mag_new = mag_new / (mag_new[0] + 1e-10) * mag[0]
        else:
            # Random spectral perturbation
            mag_new = mag * np.exp(rng.normal(0, 0.15, len(mag)))

        fa_new = mag_new * np.exp(1j * phase)
        a_new = irfft(fa_new, n)
        a_new = np.maximum(a_new, 0)
        sn = np.sum(a_new)
        if sn < 1e-10:
            continue
        a_new = a_new / sn

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence from global."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.3)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 5:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            st = b * n // nblocks
            en = (b + 1) * n // nblocks
            a[st:en] = rng.uniform(0.3, 1.5)
    elif choice == 6:
        a = np.ones(n)
        a[:n // 2] *= rng.uniform(0.5, 1.2)
        a[n // 2:] *= rng.uniform(0.5, 1.2)
    else:
        a = rng.random(n) + 0.05

    a = np.maximum(a, 0.001)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines multiplicative weights QP, LP alternation,
    simulated annealing, and frequency-domain shaping across multiple n values.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 35.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < np.inf:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Starting with parent: n={len(best_seq)}, score={best_score:.6f}")

    # Determine n values to explore
    if best_seq is not None:
        n_parent = len(best_seq)
        n_values = sorted(set([
            max(200, n_parent - 100),
            max(200, n_parent - 50),
            n_parent,
            n_parent + 50,
            n_parent + 100,
            n_parent + 200,
            n_parent + 300,
        ]))
    else:
        n_values = [400, 500, 600, 700, 800, 1000]

    n_values = [max(200, min(1500, n)) for n in n_values]
    print(f"n values to explore: {n_values}")

    # Track per-n best
    n_best = {n: (None, np.inf) for n in n_values}

    # If we have a parent, set its n as best
    if best_seq is not None:
        np_ = len(best_seq)
        if np_ in n_best:
            n_best[np_] = (best_seq.copy(), best_score)

    # Strategy cycling
    strategies = ['mw_qp', 'lp', 'sa', 'freq']
    strat_cycle = 0

    while time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        strat = strategies[strat_cycle % len(strategies)]
        strat_cycle += 1

        # Pick n: prioritize n's that haven't been explored much or have potential
        # Cycle through n values
        n = n_values[strat_cycle % len(n_values)]

        # Get starting point
        if n in n_best and n_best[n][0] is not None:
            a_start = n_best[n][0].copy()
            obj_start = n_best[n][1]
        elif best_seq is not None:
            a_start = resample_sequence(best_seq, n)
            obj_start = compute_obj(a_start)
        else:
            a_start = random_init(n, seed + n * 104729)
            obj_start = compute_obj(a_start)

        # Allocate time based on remaining and number of n values
        alloc = min(remaining * 0.15, 80.0)
        alloc = max(10.0, alloc)

        print(f"\n[Strat={strat}, n={n}] t={time.monotonic()-start_time:.0f}s, "
              f"remaining={remaining:.0f}s, alloc={alloc:.0f}s, init_obj={obj_start:.6f}")

        a_cur = a_start.copy()
        s = np.sum(a_cur)
        if s < 1e-10:
            a_cur = np.ones(n) / n
        else:
            a_cur = a_cur / s

        if strat == 'mw_qp':
            # Multiplicative weights QP with varying alpha
            alphas = [1.0, 2.0, 3.0, 5.0]
            alpha = alphas[strat_cycle % len(alphas)]
            a_new, obj_new = multiplicative_weights_qp(
                a_cur, time_budget=alloc, alpha=alpha, num_inner=150)

        elif strat == 'lp':
            # LP alternation with backtracking
            a_new, obj_new = lp_alternation_enhanced(a_cur, time_budget=alloc)

        elif strat == 'sa':
            # Simulated annealing
            a_new, obj_new = simulated_annealing_block(a_cur, time_budget=alloc)

        else:
            # Frequency domain shaping
            a_new, obj_new = freq_domain_shape(a_cur, time_budget=alloc)

        # Track best for this n
        if obj_new < n_best[n][1]:
            n_best[n] = (a_new.copy(), obj_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  ** NEW GLOBAL BEST: n={n}, score={best_score:.6f} **")
            else:
                print(f"  n-best updated: n={n}, score={obj_new:.6f}")
        else:
            print(f"  No improvement: obj={obj_new:.6f} vs n_best={n_best[n][1]:.6f}")

    # Final polishing phase
    print(f"\n=== Final Polishing ===")
    print(f"Best before polish: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_final = len(best_seq)
        print(f"Polishing n={n_final}, {remaining:.0f}s remaining")

        a_polish = best_seq.copy()

        # Round 1: MW-QP polish
        if remaining > 60:
            alloc1 = min(remaining * 0.35, 120.0)
            a_p, obj_p = multiplicative_weights_qp(
                a_polish, time_budget=alloc1, alpha=3.0, num_inner=200)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_polish = a_p.copy()
                print(f"  MW-QP polish: {best_score:.6f}")

        # Round 2: LP polish
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.3, 80.0)
            a_p, obj_p = lp_alternation_enhanced(a_polish, time_budget=alloc2)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_polish = a_p.copy()
                print(f"  LP polish: {best_score:.6f}")

        # Round 3: SA + short LP
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            a_p, obj_p = simulated_annealing_block(a_polish, time_budget=min(20.0, remaining * 0.2))
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                a_polish = a_p.copy()
                print(f"  SA polish: {best_score:.6f}")

            if time.monotonic() < deadline - 20:
                remaining = deadline - time.monotonic()
                a_p, obj_p = lp_alternation_enhanced(a_polish, time_budget=min(30.0, remaining * 0.5))
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Final LP: {best_score:.6f}")

        # Try a few different n values for final best
        for n_try in [len(best_seq) - 20, len(best_seq) + 20, len(best_seq) + 50]:
            if time.monotonic() > deadline - 30:
                break
            if n_try < 200:
                continue
            a_resamp = resample_sequence(best_seq, n_try)
            remaining = deadline - time.monotonic()
            a_p, obj_p = multiplicative_weights_qp(
                a_resamp, time_budget=min(30.0, remaining * 0.3), alpha=2.0, num_inner=100)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  n_try={n_try}: {best_score:.6f}")

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(500) / 500
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 500
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
