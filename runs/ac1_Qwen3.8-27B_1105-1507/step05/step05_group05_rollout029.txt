"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Global search via differential evolution with Fourier parametrization
2. Full-space Adam optimization with FFT-based smooth max gradient
3. Block coordinate descent with L-BFGS-B

import numpy as np
from scipy.optimize import differential_evolution, minimize

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_value(b, tau):
    b_max = b.max()
    return b_max + np.log(np.exp(tau * (b - b_max)).sum()) / tau

def smooth_max_grad_fft(a, tau):
    """Compute gradient of smooth max of conv(a,a) using FFT."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    e = np.exp(tau * (b - b_max))
    p = e / e.sum()
    
    # grad[i] = 2 * sum_j p[i+j] * a[j] for valid j
    # This is 2 * cross_correlation(p, a)[i]
    N = 1
    while N < 2 * n - 1 + n - 1:
        N <<= 1
    fp = np.fft.rfft(p, N)
    fa = np.fft.rfft(a, N)
    # cross-correlation: ifft(conj(FFT(p)) * FFT(a))
    corr = np.fft.irfft(np.conj(fp) * fa, N)
    # corr[k] = sum_m p[m] * a[m-k] (circular)
    # We want v[i] = sum_j p[i+j] * a[j] = sum_m p[m] * a[m-i] = corr[i]
    grad = 2.0 * corr[:n]
    return grad, b

def fourier_to_seq(params, n):
    """Convert Fourier coefficients to a non-negative sequence."""
    K = len(params) // 2
    alpha = params[:K]
    beta = params[K:2*K]
    t = np.arange(n)
    a = np.ones(n)
    for k in range(1, K+1):
        a += alpha[k-1] * np.cos(2*np.pi*k*t/n) + beta[k-1] * np.sin(2*np.pi*k*t/n)
    return np.maximum(a, 0.0)

def de_objective(params, n):
    a = fourier_to_seq(params, n)
    s = a.sum()
    if s < 0.01:
        return 1e10
    a_norm = a / s
    b = fast_conv(a_norm)
    return 2 * n * b.max()

def run_adam_smooth(a, max_iters, deadline, tau_start=10.0, tau_end=200.0, lr=0.05):
    """Adam optimizer on smooth max objective with FFT gradient."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.8)
        lr_t = lr / (1 + 0.02 * it)
        
        grad, b = smooth_max_grad_fft(a, tau)
        
        # Project gradient onto simplex tangent space
        grad = grad - grad.mean()
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
    
    return best_a

def block_optimize(a, block_size=8, max_sweeps=20, deadline=None):
    """Block coordinate descent: optimize blocks of consecutive coordinates."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    n_blocks = (n + block_size - 1) // block_size
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        for b_idx in range(n_blocks):
            if deadline is not None and time.monotonic() > deadline:
                break
            
            lo = b_idx * block_size
            hi = min(n, lo + block_size)
            m = hi - lo
            
            # Define objective for this block
            def block_obj(x):
                a_trial = a.copy()
                a_trial[lo:hi] = np.maximum(x, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    return 1e10
                a_trial = a_trial / s
                b = fast_conv(a_trial)
                return b.max()
            
            x0 = a[lo:hi].copy()
            res = minimize(block_obj, x0, method='L-BFGS-B', 
                          bounds=[(0, None)]*m,
                          options={'maxiter': 50, 'gtol': 1e-8})
            
            if res.fun < block_obj(x0):
                a[lo:hi] = np.maximum(res.x, 0.0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
        
        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
    
    return best_a

def coordinate_descent(a, max_sweeps=50, deadline=None):
    """Standard coordinate descent with golden section search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    gr = (np.sqrt(5) - 1) / 2
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        for i in range(n):
            x_cur = a[i]
            S = a.sum()
            b = fast_conv(a)
            
            # Find optimal a[i] given all others
            def obj(x):
                a_new = a.copy()
                a_new[i] = x
                s_new = a_new.sum()
                if s_new <= 0:
                    return 1e10
                b_new = fast_conv(a_new)
                return b_new.max() / (s_new * s_new)
            
            # Golden section search
            lo = 0.0
            hi = max(x_cur * 2.0, 0.1)
            
            # Expand hi if needed
            if obj(hi) < obj(hi * 0.5) and hi < 1.0:
                hi = min(hi * 2.0, 1.0)
            
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj(a_pt)
            fb = obj(b_pt)
            
            for _ in range(40):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj(b_pt)
            
            x_opt = (lo + hi) / 2.0
            if obj(0.0) < obj(x_opt):
                x_opt = 0.0
            
            a[i] = max(0.0, x_opt)
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
    
    return best_a

    return np.maximum(np.interp(t_new, t_old, src), 0.0)

```python
"""
Multi-phase optimizer for J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Phases: Fourier DE -> Adam smooth grad -> Block CD -> Coordinate CD -> Random restarts.
"""

import numpy as np
from scipy.optimize import differential_evolution, minimize
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad_fft(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    e = np.exp(tau * (b - b_max))
    p = e / e.sum()
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    fp = np.fft.rfft(p, N)
    fa = np.fft.rfft(a, N)
    corr = np.fft.irfft(np.conj(fp) * fa, N)
    grad = 2.0 * corr[:n]
    return grad, b


def fourier_to_seq(params, n):
    K = len(params) // 2
    alpha = params[:K]
    beta = params[K:2 * K]
    t = np.arange(n)
    a = np.ones(n)
    for k in range(1, K + 1):
        a = a + alpha[k - 1] * np.cos(2 * np.pi * k * t / n)
        a = a + beta[k - 1] * np.sin(2 * np.pi * k * t / n)
    return np.maximum(a, 0.0)


def de_objective(params, n):
    a = fourier_to_seq(params, n)
    s = a.sum()
    if s < 0.01:
        return 1e10
    a = a / s
    b = fast_conv(a)
    return 2.0 * n * b.max()


def run_adam_smooth(a, max_iters, deadline, tau_start=10.0, tau_end=300.0, lr=0.05):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.7)
        lr_t = lr / (1 + 0.01 * it)
        
        grad, b = smooth_max_grad_fft(a, tau)
        grad = grad - grad.mean()
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
    
    return best_a

    a_trial[lo:hi] = np.maximum(x, 0.0)
    s = a_trial.sum()
