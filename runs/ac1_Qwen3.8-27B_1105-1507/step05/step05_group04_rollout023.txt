    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_max_obj_grad(a, tau):
    """Compute soft-max approximation of J and its gradient.
    J_soft = 2n * soft_max(conv(a,a)) / (sum(a))^2
    """
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    F = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient via FFT: grad_F[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * conv(p, a_rev)[n-1+i]
    a_rev = a[::-1]
    size = 1
    while size < (2 * n - 1) + n:
        size <<= 1
    P = np.fft.rfft(p, size)
    AR = np.fft.rfft(a_rev, size)
    conv_pa = np.fft.irfft(P * AR, size)
    grad_F = 2.0 * conv_pa[n - 1:2 * n - 1]
    
    J = 2.0 * n * F / (s * s)
    grad_J = 2.0 * n * (grad_F * s - 2.0 * F) / (s ** 3)
    
    return J, grad_J


def run_lbfgs(a0, deadline, tau_schedule=None, max_iter=500):
    """Run L-BFGS-B with soft-max objective and adaptive tau."""
    a0 = np.maximum(a0, 0).astype(np.float64)
    s = a0.sum()
    if s < 1e-10:
        return a0
    a0 = a0 / s
    n = len(a0)
    
    if tau_schedule is None:
        tau_schedule = [5, 10, 20, 40, 80, 160]
    
    best_a = a0.copy()
    best_J = compute_J(a0)
    
    for tau in tau_schedule:
        if time.monotonic() > deadline - 10:
            break
        
        def obj_grad(x):
            x = np.maximum(x, 0)
            val, grad = soft_max_obj_grad(x, tau)
            return val, grad
        
        try:
            res = minimize(obj_grad, a0, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                                  'ftol': 1e-15, 'gtol': 1e-12,
                                  'maxls': 50})
            a0 = np.maximum(res.x, 0)
            s = a0.sum()
            if s > 1e-10:
                a0 = a0 / s
        except Exception:
            break
        
        J_val = compute_J(a0)
        if J_val < best_J:
            best_J = J_val
            best_a = a0.copy()
    
    return best_a


def build_lp_matrix_reduced(n, a, active_k):
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))


def solve_lp(a, M, top_frac=0.7, time_limit=3):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    
    if top_frac < 1.0:
        threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
        active_k = np.where(b >= threshold)[0]
        if len(active_k) < 10:
            active_k = np.arange(2 * n - 1)
    else:
        active_k = np.arange(2 * n - 1)
    
    A = build_lp_matrix_reduced(n, a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, np.inf)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                # Verify
                ag = fast_cross_conv(a, g)
                if ag.max() <= M * (1.0 + 1e-4):
                    return g
                # Relax: scale down
                scale = M / ag.max()
                g = g * scale
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, verbose=True, top_frac=0.7, max_no_improve=150):
    """LP mirror descent with adaptive step sizing."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline - 5:
        it += 1
        b = fast_conv(a)
        M = float(b.max())
        
        # Try with slightly tighter M
        M_tight = M * 0.999
        g = solve_lp(a, M_tight, top_frac=top_frac, time_limit=3)
        if g is None:
            g = solve_lp(a, M, top_frac=top_frac, time_limit=3)
        
        if g is None or g.sum() < 1e-10:
            if no_improve > max_no_improve:
                break
            no_improve += 1
            continue
        
        g = g / g.sum()
        
        # Line search
        improved = False
        diff = np.max(np.abs(g - a))
        base_t = min(0.7, max(0.05, diff * 1.5))
        t_candidates = [base_t, base_t * 0.5, base_t * 0.25, base_t * 0.1, 0.05, 0.02, 0.01]
        
        current_J = best_J
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            
            if J_trial < best_J - 1e-14:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
            elif J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
        
        # Periodic perturbation
        if no_improve > max_no_improve:
            rng = np.random.default_rng(hash((it, no_improve)) % (2**32))
            scale = 0.1 / (1.0 + 0.1 * (no_improve - max_no_improve))
            freqs = rng.integers(1, 8, size=3)
            phases = rng.uniform(0, 2 * np.pi, size=3)
            amplitudes = rng.uniform(0.2, 0.8, size=3)
            t_arr = np.linspace(0, 1, n, endpoint=False)
            pert = np.zeros(n)
            for f, ph, amp in zip(freqs, phases, amplitudes):
                pert += amp * np.cos(2 * np.pi * f * t_arr + ph)
            a = a + scale * pert
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            no_improve = 0
        
        if verbose and it % 1000 == 0:
            print(f"    LP iter={it} J={best_J:.8f} M={M:.6f} no_imp={no_improve}", flush=True)
        
        if it > 100000:
            break
    
    return best_a, best_J


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
    
    inits.append(np.ones(n))
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t - 1)), 0.01))
    inits.append(np.maximum(np.exp(-8 * (t - 0.5) ** 2), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 2 * t), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t), 0.01))
    inits.append(np.maximum(1.0 + 0.2 * np.cos(2 * np.pi * 5 * t) + 0.1 * np.cos(2 * np.pi * 7 * t), 0.01))
    
    # Random smooth
    ksize = max(3, n // 64)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Block structure
    n_blocks = max(4, n // 128)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    inits.append(np.repeat(block_vals, n // n_blocks + 1)[:n])
    
    return inits

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. L-BFGS-B with FFT-based soft-max gradients for fast convergence
2. LP mirror descent with dual LP formulation (min-peak) for fine-tuning

4. Final full-constraint LP polishing
Key improvements: min-peak LP formulation, FFT gradients, adaptive tau,

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_max_obj_grad(a, tau):
    """Compute soft-max approximation of J and its gradient via FFT."""
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    F = b_max + np.log(exp_b.sum()) / tau

    # Gradient via FFT: grad_F[i] = 2 * sum_k p[k] * a[k-i]
    a_rev = a[::-1]
    size = 1
    while size < (2 * n - 1) + n:
        size <<= 1
    P = np.fft.rfft(p, size)
    AR = np.fft.rfft(a_rev, size)
    conv_pa = np.fft.irfft(P * AR, size)
    grad_F = 2.0 * conv_pa[n - 1:2 * n - 1]

    grad_J = 2.0 * n * (grad_F * s - 2.0 * F) / (s ** 3)
    return J, grad_J

            x = np.maximum(x, 0)
            val, grad = soft_max_obj_grad(x, tau)
            return val

            x = np.maximum(x, 0)
            val, grad = soft_max_obj_grad(x, tau)
            return val, grad

    return best_a


def make_conv_matrix(n, a, active_k=None):
    """Build sparse convolution matrix. (A @ g)[idx] = (a*g)[k]."""
    if active_k is None:
        active_k = np.arange(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
