import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

def solve_lp(a, M, relax=1.0):
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    A_ub = build_toeplitz(a)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = float(np.sum(g))
            return g, S
    except:
        pass
    return None, 0.0

def gradient_step(a, lr):
    """Take a projected gradient step to reduce the autocorrelation peak."""
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    # Subgradient: grad[j] = a[k_star - j] if 0 <= k_star - j < n, else 0
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = a[idx]
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 1e-10)  # Small floor to avoid zeros
    a_new = a_new / np.sum(a_new)
    return a_new, conv

def multi_peak_gradient_step(a, lr, num_peaks=3):
    """Take a gradient step considering multiple peaks."""
    n = len(a)
    conv = compute_conv(a)
    # Find top peaks
    sorted_idx = np.argsort(conv)[::-1]
    top_peaks = sorted_idx[:num_peaks]
    
    # Sum gradients from all top peaks (weighted by how close to max)
    max_val = conv[0]
    grad = np.zeros(n)
    for k in top_peaks:
        weight = conv[k] / max_val  # Weight by relative magnitude
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / np.sum(a_new)
    return a_new, conv

"""
Hybrid gradient + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact subgradients of the autocorrelation peak, multi-peak awareness,
log-barrier regularization, LP-based escape, and multi-start with diverse
initializations and adaptive n selection.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_direction(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    A_ub = build_toeplitz(a)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = float(np.sum(g))
            return g, S
    except Exception:
        pass
    return None, 0.0


def subgradient_peak(a, k_star):
    """Compute subgradient of q_{k_star}(a) = sum_i a[i]*a[k_star-i].
    Returns grad[j] = a[k_star - j] if 0 <= k_star-j < n, else 0."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(k_star, n - 1)
    for j in range(lo, hi + 1):
        grad[j] = a[k_star - j]
    return grad


def gradient_step_multi_peak(a, lr, num_peaks=3, barrier_mu=0.0):
    """Projected gradient step considering multiple peaks + optional log barrier."""
    n = len(a)
    conv = compute_conv(a)
    sorted_idx = np.argsort(conv)[::-1]
    top_peaks = sorted_idx[:num_peaks]
    max_val = conv[top_peaks[0]]
    
    grad = np.zeros(n)
    for k in top_peaks:
        weight = (conv[k] / max_val) ** 2  # Squared weight emphasizes top peaks
        sg = subgradient_peak(a, int(k))
        grad += weight * sg
    
    # Add barrier term to keep a away from zero
    if barrier_mu > 0:
        grad -= barrier_mu / (a + 1e-10)
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 1e-8)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new, conv


def gradient_descent_phase(a, time_budget=30.0, lr_init=0.005, lr_min=1e-6,
                           num_peaks=3, barrier_mu=0.0, verbose=False):
    """Rapid gradient descent with adaptive learning rate."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    stall = 0
    
    while time.monotonic() - start < time_budget:
        a_new, conv = gradient_step_multi_peak(a, lr, num_peaks, barrier_mu)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Adaptive learning rate
        if stall > 10:
            lr *= 0.95
            if lr < lr_min:
                break
            if stall > 100:
                break
        elif stall == 0 and lr < lr_init:
            lr = min(lr * 1.1, lr_init)
    
    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.01)

def peak_burst_reduce(a, num_tries=30, time_budget=10.0):
    """Targeted perturbation to reduce top peaks."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    for _ in range(num_tries):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        sorted_idx = np.argsort(conv)[::-1]
        top_peaks = sorted_idx[:5]
        k = int(top_peaks[rng.integers(0, min(5, len(top_peaks)))])
        
        # Find affected positions
        affected = []
        for j in range(n):
            idx = k - j
            if 0 <= idx < n and a[j] > 0 and a[idx] > 0:
                affected.append(j)
        
        if len(affected) == 0:
            continue
        
        # Reduce at affected positions, redistribute
        mask = np.zeros(n)
        for pos in affected:
            mask[pos] = 1.0
        
        factor = rng.uniform(0.6, 0.9)
        a_new = a.copy()
        a_new[affected] *= factor
        
        # Redistribute to non-affected positions
        reduction = np.sum(a[affected]) - np.sum(a_new[affected])
        free_mask = 1.0 - mask
        free_sum = np.sum(a[free_mask > 0])
        if free_sum > 0:
            a_new[free_mask > 0] += reduction * a_new[free_mask > 0] / free_sum
        
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
    
    return best_a

            obj_r2 = compute_obj(a_r2)
            if obj_r2 < best_score:
                best_score = obj_r2
                best_seq = a_r2.copy()
                print(f"  Round 2a (peaks): {best_score:.6f}")
            
            if time.monotonic() < deadline - 20:
                remaining = deadline - time.monotonic()
                alloc2 = min(remaining * 0.3, 80.0)
                a_r2b, obj_r2b = aggressive_polish(best_seq, time_budget=alloc2)
                if obj_r2b < best_score:
                    best_score = obj_r2b
                    best_seq = a_r2b.copy()
                    print(f"  Round 2b: {best_score:.6f}")
        
        # Round 3: Try slightly different n (remap)
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            n_alt = n_best + 10
            if 200 <= n_alt <= 1200:
                a_alt = resample_sequence(best_seq, n_alt)
                a_alt, obj_alt = aggressive_polish(a_alt, time_budget=min(remaining * 0.2, 40.0))
                if obj_alt < best_score:
                    best_score = obj_alt
                    best_seq = a_alt.copy()
                    print(f"  Round 3 (n={n_alt}): {best_score:.6f}")
            
            n_alt2 = n_best - 10
            if 200 <= n_alt2 <= 1200 and n_alt2 != n_alt:
                remaining = deadline - time.monotonic()
                if remaining > 30:
                    a_alt2 = resample_sequence(best_seq, n_alt2)
                    a_alt2, obj_alt2 = aggressive_polish(a_alt2, time_budget=min(remaining * 0.2, 30.0))
                    if obj_alt2 < best_score:
                        best_score = obj_alt2
                        best_seq = a_alt2.copy()
                        print(f"  Round 3 (n={n_alt2}): {best_score:.6f}")
        
        # Round 4: Final fine gradient
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            a_r4, obj_r4 = gradient_descent_phase(best_seq, time_budget=min(remaining * 0.5, 30.0),
                                                  lr_init=0.0005, num_peaks=5, barrier_mu=0.001)
            if obj_r4 < best_score:
                best_score = obj_r4
                best_seq = a_r4.copy()
                print(f"  Round 4: {best_score:.6f}")
    
    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    
    return result
