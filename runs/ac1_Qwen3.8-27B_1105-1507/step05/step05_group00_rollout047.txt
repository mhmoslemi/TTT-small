"""
Optimizes step function heights using smoothed log-sum-exp objective with L-BFGS-B,
combined with LP alternation, multi-n exploration, and random restarts.
Key innovation: reparameterizes via softmax to handle non-negativity and sum constraints,
then uses τ-annealing on the log-sum-exp smoothing for effective gradient-based search.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(z), 1.0 / len(z))
    return e / s


def smooth_obj_and_grad(z, tau):
    """
    Compute smoothed objective and gradient w.r.t. z (softmax parameters).
    Objective: 2n * (1/tau) * log(sum_k exp(tau * (a*a)_k)) where a = softmax(z).
    """
    n = len(z)
    a = softmax(z)
    conv = compute_conv(a)
    max_c = np.max(conv)
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_sum = np.sum(np.exp(shifted))
    if exp_sum < 1e-300:
        obj = 2.0 * n * max_c
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        log_sum = max_c + np.log(exp_sum)
        obj = 2.0 * n * log_sum / tau
        weights = np.exp(shifted) / exp_sum

    # Gradient of obj w.r.t. a: grad_a[i] = 2 * sum_k weights[k] * a[k-i] (if 0<=k-i<n)
    # = 2 * correlate via FFT
    w_rev = weights[::-1]
    conv_aw = fft_convolve(a, w_rev)
    # grad_a[i] = 2 * conv_aw[2n-2-i] for i=0..n-1
    grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]

    # Gradient w.r.t. z via reparameterization:
    # df/dz_k = a_k * (grad_a_k - dot(a, grad_a))
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)

    return float(obj), grad_z


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns normalized g."""
    n = len(a)
    if n > 2000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def compute_obj(a):
    """Compute the true objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def optimize_smoothed(a_init, n, time_budget=120.0, seed=42):
    """
    Optimize using smoothed log-sum-exp with L-BFGS-B and tau annealing.
    a_init: initial sequence (will be normalized).
    Returns optimized sequence and objective.
    """
    start = time.monotonic()
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    # Initialize z from a: z = log(a) (since softmax(log(a)) = a)
    z = np.log(a + 1e-30)

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Tau schedule: start smooth, go sharp
    tau_schedule = [1.0, 5.0, 10.0, 50.0, 100.0, 300.0, 500.0, 1000.0, 2000.0]

    for tau in tau_schedule:
        if time.monotonic() - start > time_budget - 10:
            break

        # Adjust max iterations based on remaining time
        remaining = time_budget - (time.monotonic() - start)
        max_iter = max(50, int(remaining * 8))

        try:
            res = minimize(
                smooth_obj_and_grad,
                z,
                args=(tau,),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8,
                         'maxfun': max_iter * 3}
            )
            if res.x is not None:
                a_new = softmax(res.x)
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    z = res.x.copy()
        except Exception:
            pass

    # Final: if we improved, use the best; otherwise keep original
    return best_a, best_obj


def lp_alternate(a, max_iters=50, time_budget=60.0):
    """LP alternating optimization with adaptive step sizes."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

        M = float(np.max(conv))

        g = solve_lp_max_sum(a, M, time_limit=15)
        if g is None:

        # Try multiple step sizes
        best_local_obj = best_obj
        best_local_a = best_a

        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_a
        else:
            # Perturb to escape
            rng = np.random.default_rng(it * 104729 + 7)

            a_pert = np.maximum(a_pert, 0)

def subgradient_steps(a, max_iters=300, time_budget=30.0):
    """Subgradient optimization with momentum on smoothed max."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.85
    step = 0.03

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        tau = 1.0 + 3.0 * it / max_iters
        shifted = np.clip((conv - cmax) / tau, -300.0, 300.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
        grad_centered = grad_a - np.mean(grad_a)

        v = mu * v - step * grad_centered
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        step *= 0.998

    return best_a


def random_structured_init(n, rng):
    """Generate a random structured initialization."""
    pattern = rng.integers(0, 6)
    if pattern == 0:
        # Pure random
        a = rng.random(n)
    elif pattern == 1:
        # Block structure
        nblocks = rng.integers(3, 15)
        block_sizes = rng.integers(1, max(2, n // nblocks), nblocks)
        a = np.zeros(n)
        pos = 0
        for bs in block_sizes:
            if pos >= n:
                break
            a[pos:pos + bs] = rng.uniform(0.1, 1.0)
            pos += bs
    elif pattern == 2:
        # Periodic with noise
        period = rng.integers(3, min(20, n // 2))
        base = rng.uniform(0.2, 1.0, period)
        a = np.tile(base, n // period + 1)[:n]
        a *= np.exp(rng.normal(0, 0.3, n))
    elif pattern == 3:
        # Spike pattern
        a = np.full(n, 0.1)
        n_spikes = rng.integers(n // 10, n // 3)
        spike_pos = rng.choice(n, size=n_spikes, replace=False)
        a[spike_pos] = rng.uniform(0.5, 2.0, n_spikes)
    elif pattern == 4:
        # Symmetric (a_i = a_{n-1-i})
        half = n // 2
        a_half = rng.uniform(0.1, 1.0, half)
        a = np.concatenate([a_half, a_half[::-1]])[:n]
    else:
        # Low-pass filtered random
        a = rng.random(n)
        kernel_size = rng.integers(3, min(11, n))
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

            print(f"  Round {round_idx}: {alloc:.0f}s")

            # Smoothed L-BFGS-B with different seeds
            rng = np.random.default_rng(seed + round_idx * 12345)
            a_pert = best_seq * np.exp(rng.normal(0, 0.03, n_best))
            a_pert = np.maximum(a_pert, 0)
            s = np.sum(a_pert)
            if s > 1e-10:
                a_pert = a_pert / s

            a_opt, obj_opt = optimize_smoothed(a_pert, n_best,
                                               time_budget=alloc * 0.5,
                                               seed=seed + round_idx)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"    Updated: {best_score:.8f}")

            # LP polish
            remaining = deadline - time.monotonic()
            if remaining > 20 and n_best <= 1200:
                a_lp = lp_alternate(best_seq, max_iters=30,
                                    time_budget=min(remaining * 0.2, 40.0))
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"    LP updated: {best_score:.8f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_sg = subgradient_steps(best_seq, max_iters=300,
                                         time_budget=min(remaining * 0.15, 20.0))
                obj_sg = compute_obj(a_sg)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    print(f"    SG updated: {best_score:.8f}")

    # Phase 3: Try nearby n values around the best
    print(f"\n=== Phase 3: Neighboring n exploration ===")
    if best_seq is not None:
        n_best = len(best_seq)
        neighbors = [n_best - 25, n_best - 10, n_best + 10, n_best + 25,
                     n_best + 50, n_best - 50]
        neighbors = [n for n in neighbors if 100 <= n <= 2000]
        neighbors = sorted(set(neighbors))

        for n_nb in neighbors:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_nb = resample_sequence(best_seq, n_nb)
            s = np.sum(a_nb)
            if s > 1e-10:
                a_nb = a_nb / s

            # Quick optimization
            a_opt, obj_opt = optimize_smoothed(a_nb, n_nb,
                                               time_budget=min(remaining * 0.2, 40.0),
                                               seed=seed + n_nb)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** Neighbor n={n_nb}: {best_score:.8f} **")

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_final, obj_final = final_polish(best_seq,
                                              time_budget=min(remaining * 0.7, 90.0))
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final polish: {best_score:.8f}")

    # Validation and return
    if best_seq is None:
        best_seq = np.ones(200, dtype=np.float64) / 200.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
