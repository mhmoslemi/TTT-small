import numpy as np

def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_constraint_matrix(a, peak_indices):
    """Build constraint matrix rows for given peak indices.
    Row k: (a*g)[k] = sum_i a[k-i] * g[i] where 0<=k-i<n.
    So row k has entry a[k-i] at column i for valid i.
    """
    n = len(a)
    idx = np.array(peak_indices, dtype=int)
    # For each peak index k, the constraint row is:
    # (a*g)[k] = sum_{i=max(0,k-(n-1))}^{min(k,n-1)} a[k-i] * g[i]
    # = sum_i g[i] * a[k-i] (zero-padded)
    # We can construct this efficiently using the structure.
    A = np.zeros((len(idx), n))
    for row, k in enumerate(idx):
        k = int(k)
        # g[i] appears with coefficient a[k-i] for i such that 0<=k-i<n
        # i.e., k-(n-1) <= i <= k, and 0 <= i <= n-1
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start <= i_end:
            # coefficients are a[k - i] for i in [i_start, i_end]
            # = a[k - i_start : k - i_end + 1][::-1]
            coeffs = a[k - i_end : k - i_start + 1][::-1]
            A[row, i_start:i_end + 1] = coeffs
    return A


def solve_lp_fast(a, M, top_k=10, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax for top_k peaks, g >= 0."""
    n = len(a)
    conv = compute_conv_fft(a)
    # Get top_k peak indices
    sorted_idx = np.argsort(conv)[::-1][:top_k]
    sorted_idx = sorted_idx.astype(int)
    
    A = build_constraint_matrix(a, sorted_idx)
    c = -np.ones(n)
    b_ub = np.full(len(sorted_idx), M * relax)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 5})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, 0.0

def project_simplex(v, s=1.0):
    """Project v onto {x: x>=0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho_max = np.max(rho[cond])
    theta = (cssv[rho_max - 1] - s) / rho_max
    return np.maximum(v - theta, 0)


def subgradient_step(a, step_size=0.01):
    """Take a subgradient step to reduce the peak autocorrelation."""
    n = len(a)
    conv = compute_conv_fft(a)
    k_star = int(np.argmax(conv))
    
    # Gradient of (a*a)[k_star] w.r.t. a is 2 * a[k_star - i] (zero-padded)
    # i.e., grad[i] = 2 * a[k_star - i] if 0 <= k_star - i < n, else 0
    grad = np.zeros(n)
    i_start = max(0, k_star - (n - 1))
    i_end = min(k_star, n - 1)
    if i_start <= i_end:
        grad[i_start:i_end + 1] = 2.0 * a[k_star - i_end : k_star - i_start + 1][::-1]
    
    a_new = a - step_size * grad
    a_new = project_simplex(a_new, 1.0)
    return a_new

        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def joint_optimize(a, time_budget=50.0):
    """Combined LP + subgradient + spectral optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(2024)
    
    phase = 0
    stall = 0
    
    while time.monotonic() - start < time_budget - 2:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break
        
        # Phase cycling: LP -> subgrad -> spectral
        if phase == 0:
            # LP phase
            a_new, obj_new = lp_optimize(best_a, time_budget=min(remaining * 0.4, 30.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
            phase = 1
            
        elif phase == 1:
            # Subgradient phase
            for _ in range(min(50, int(remaining * 2))):
                if time.monotonic() - start > time_budget - 2:
                    break
                step = 0.003 * (1.0 + 0.1 * rng.random())
                a_sg = subgradient_step(best_a, step_size=step)
                obj_sg = compute_obj(a_sg)
                if obj_sg < best_obj:
                    best_obj = obj_sg
                    best_a = a_sg.copy()
                    stall = 0
            phase = 2
            
        elif phase == 2:
            # Spectral escape phase
            a_spec = spectral_shaping_escape(best_a, num_iters=20,
                                             time_budget=min(8.0, remaining * 0.2))
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_obj:
                best_obj = obj_spec
                best_a = a_spec.copy()
            stall += 1
            phase = 0
        
        # If very stalled, try a bigger perturbation
        if stall > 4:
            scale = rng.uniform(0.05, 0.15)
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            s_p = np.sum(a_pert)
            if s_p > 1e-10:
                a_pert = a_pert / s_p
                # Quick LP polish
                a_pert, obj_p = lp_optimize(a_pert, time_budget=min(5.0, remaining * 0.1))
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_pert.copy()
                    stall = 0
                else:
                    stall += 1
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Hybrid LP + Subgradient + Spectral optimization with
    multi-length exploration and adaptive time allocation.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            pobj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={pobj:.6f}")
            parent_obj = pobj
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    # Phase 1: Intensive optimization on parent
    if best_seq is not None:
        n_parent = len(best_seq)
        print(f"\n=== Phase 1: Optimize parent (n={n_parent}) ===")
        
        # Normalize
        s = np.sum(best_seq)
        a_work = best_seq / s if s > 1e-10 else np.ones(n_parent) / n_parent
        
        # Joint optimization
        alloc1 = min(400.0, (deadline - time.monotonic()) * 0.35)
        a_opt, obj_opt = joint_optimize(a_work, time_budget=alloc1)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  After joint opt: {best_score:.6f}")
        else:
            print(f"  After joint opt: {obj_opt:.6f} (no improvement)")
    
    # Phase 2: Multi-length exploration
    print(f"\n=== Phase 2: Multi-length Exploration ===")
    
    n_candidates = []
    if best_seq is not None:
        np_ = len(best_seq)
        n_candidates = [np_ - 50, np_, np_ + 50, np_ + 100, np_ + 150, np_ + 200]
    else:
        n_candidates = [400, 500, 600, 700, 800, 900]
    
    n_candidates = sorted(set(max(100, min(1500, n)) for n in n_candidates))
    print(f"  Lengths to try: {n_candidates}")
    
    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        # Skip if same as current best length (already optimized)
        if best_seq is not None and len(best_seq) == n_try:
            continue
        
        alloc = min(remaining * 0.18, 100.0)
        alloc = max(15.0, alloc)
        
        print(f"\n  N={n_try}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")
        
        # Initialize
        if best_seq is not None:
            a_start = resample_sequence(best_seq, n_try)
        else:
            a_start = random_init(n_try, seed + n_try * 31)
        
        # Optimize
        a_opt, obj_opt = joint_optimize(a_start, time_budget=alloc)
        
        # Extra peak reduction
        if time.monotonic() < deadline - 30:
            a_opt = peak_targeted_reduction(a_opt, num_tries=20,
                                            time_budget=min(5.0, (deadline - time.monotonic()) * 0.05))
            obj_opt = compute_obj(a_opt)
        
        # Final LP polish
        if time.monotonic() < deadline - 30:
            a_opt, obj_opt = lp_optimize(a_opt, time_budget=min(20.0, (deadline - time.monotonic()) * 0.1))
        
        print(f"  N={n_try}: obj={obj_opt:.6f}")
        
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: n={n_try}, obj={best_score:.6f} **")
    
    # Phase 3: Final intensive polish on global best
    print(f"\n=== Phase 3: Final Polish ===")
    print(f"  Best before: {best_score:.6f}")
    
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        
        # Multiple rounds of different strategies
        a_cur = best_seq.copy()
        
        # Round 1: Heavy LP
        alloc_r1 = min(remaining * 0.25, 150.0)
        a_cur, obj_r1 = lp_optimize(a_cur, time_budget=alloc_r1, verbose=True)
        if obj_r1 < best_score:
            best_score = obj_r1
            best_seq = a_cur.copy()
            print(f"  Round 1 (LP): {best_score:.6f}")
        
        # Round 2: Spectral + LP
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            a_cur = spectral_shaping_escape(best_seq, num_iters=30,
                                            time_budget=min(10.0, remaining * 0.1))
            obj_r2 = compute_obj(a_cur)
            if obj_r2 < best_score:
                best_score = obj_r2
                best_seq = a_cur.copy()
                print(f"  Round 2 (spec): {best_score:.6f}")
            
            a_cur, obj_r2b = lp_optimize(a_cur, time_budget=min(40.0, remaining * 0.15))
            if obj_r2b < best_score:
                best_score = obj_r2b
                best_seq = a_cur.copy()
                print(f"  Round 2b (LP): {best_score:.6f}")
        
        # Round 3: Peak reduction + subgradient + LP
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            a_cur = peak_targeted_reduction(best_seq, num_tries=40,
                                            time_budget=min(8.0, remaining * 0.1))
            obj_r3 = compute_obj(a_cur)
            if obj_r3 < best_score:
                best_score = obj_r3
                best_seq = a_cur.copy()
            
            # Subgradient burst
            a_sg = best_seq.copy()
            for _ in range(200):
                if time.monotonic() > deadline - 20:
                    break
                step = 0.002 * (1.0 + 0.05 * np.random.random())
                a_sg = subgradient_step(a_sg, step_size=step)
                obj_sg = compute_obj(a_sg)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
            
            a_cur, obj_r3b = lp_optimize(best_seq, time_budget=min(30.0, remaining * 0.15))
            if obj_r3b < best_score:
                best_score = obj_r3b
                best_seq = a_cur.copy()
                print(f"  Round 3 (final LP): {best_score:.6f}")
        
        # Round 4: Try a few random perturbations + quick LP
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            rng = np.random.default_rng(seed + 9999)
            for ptry in range(8):
                if time.monotonic() > deadline - 15:
                    break
                scale = rng.uniform(0.02, 0.1)
                a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    a_pert, obj_p = lp_optimize(a_pert, time_budget=min(5.0, remaining * 0.1))
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_pert.copy()
                        print(f"  Round 4 perturb {ptry}: {best_score:.6f}")
    
    # Phase 4: Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(500) / 500
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    
    return result
