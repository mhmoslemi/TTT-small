import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_qp_matrix(a, M):
    """Build QP: min 0.5*g^T Q g + c^T g s.t. A*g <= M, g >= 0
    where Q = 2I (so objective is ||g - a||^2) and c = -2a
    A is the convolution matrix with rows from a.
    """
    n = len(a)
    # A[k, j] = a[k-j] for valid j
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_qp(a, M, A):
    """Solve QP: min ||g - a||^2 s.t. A@g <= M, g >= 0 using scipy minimize (SLSQP)."""
    n = len(a)
    
    def objective(g):
        d = g - a
        return d @ d
    
    def jac(g):
        return 2 * (g - a)
    
    constraints = [{'type': 'ineq', 'fun': lambda g: M - A @ g, 'jac': lambda g: -A.T}]
    bounds = [(0, None)] * n
    
    try:
        result = minimize(objective, a, jac=jac, bounds=bounds, method='SLSQP',
                          options={'maxiter': 200, 'ftol': 1e-10})
        if result.success:
            return np.maximum(result.x, 0)
    except Exception:
        pass
    return None

def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    # First check if going in this direction helps at all
    f0 = float(np.max(a_coef))
    f1 = float(np.max(a_coef + b_coef + c_coef))
    
    if f1 >= f0:
        # Direction doesn't help even at t=1, try small t
        # Use quadratic approximation at t=0
        # f'(0) = max_k b_coef[k] (approximately, if the max is unique)
        # But we need to be more careful. Let's just do a fine search in [0, 0.1]
        lo, hi = 0.0, 0.1
    else:
        lo, hi = 0.0, 1.0
    
    # Also try t > 1
    f2 = float(np.max(a_coef + 2*b_coef + 4*c_coef))
    f3 = float(np.max(a_coef + 4*b_coef + 16*c_coef))
    if f2 < f1:
        hi = 2.0
        if f3 < f2:
            hi = 4.0
    
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(a_coef + b_coef * m1 + c_coef * m1 * m1))
        f2 = float(np.max(a_coef + b_coef * m2 + c_coef * m2 * m2))
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, t)
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-10:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            break

    return best_a


def run_subgradient_method(a, max_iters, deadline, tau0=5.0, tau1=50.0, lr0=0.01, lr1=0.001):
    """Subgradient method with momentum on max of quadratics."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)
    mu = 0.9  # momentum

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau0 + (tau1 - tau0) * frac
        lr = lr0 + (lr1 - lr0) * frac

        grad = subgradient_step_fast(a, tau)
        v = mu * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()

    return best_a


def coordinate_descent(a, max_iters, deadline):
    """
    Coordinate descent: optimize each coordinate in turn.
    For each coordinate j, given all other coordinates fixed,
    optimize a_j to minimize max_k (a*a)[k].
    
    This is a 1D problem: for fixed a_{-j}, the convolution values are
    quadratic in a_j: b[k](a_j) = b0[k] + a_j * b1[k] + a_j^2 * (1 if k=2j else 0)
    Wait, that's not quite right. Let me think...
    
    (a*a)[k] = sum_{i} a_i * a_{k-i}
    When we vary a_j:
    - Terms involving a_j: a_j * a_{k-j} (from i=j) + a_{k-j} * a_j (from i=k-j=j, same term)
      Actually: (a*a)[k] = ... + a_j * a_{k-j} + ... + a_{k-j} * a_j + ...
      If j != k-j: two terms, each a_j * a_{k-j}, total 2*a_j*a_{k-j}
      If j == k-j (i.e., k=2j): one term a_j^2
    
    So b[k] = c[k] + a_j * d[k] + a_j^2 * e[k]
    where c[k] = b[k] with a_j set to 0 (i.e., the part not involving a_j)
    d[k] = 2*a_{k-j} (if k != 2j and 0 <= k-j < n) or a_{k-j} = a_j... 
    
    Let me be more careful. Set a_j = 0 initially, compute b0[k] = (a_{-j} * a_{-j})[k]
    Then b[k](t) = b0[k] + t * 2*a_{k-j} + t^2 * delta(k, 2j)
    where delta(k, 2j) = 1 if k = 2j (and 0 <= 2j < 2n-1), else 0.
    Wait, we also need 0 <= k-j < n for the cross term.
    
    So b[k](t) = b0[k] + 2t * a_{k-j} * I(0 <= k-j < n) + t^2 * I(k = 2j)
    
    We want to find t >= 0 minimizing max_k b[k](t) subject to keeping sum(a) = 1.
    But changing a_j changes the sum, so we need to renormalize. This makes it more complex.
    
    Alternative: work with the unnormalized objective max_k b[k](t) / (sum(a))^2.
    sum(a) = S (constant) - a_j_old + t, where S = sum of all a.
    Actually sum(a) = S0 + t where S0 = sum(a) - a_j_old.
    
    So we minimize: max_k [b0[k] + 2t*a_{k-j}*I(0<=k-j<n) + t^2*I(k=2j)] / (S0+t)^2
    
    This is a 1D minimization problem. We can solve it with golden section or ternary search.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    coord_order = np.random.permutation(n)  # Random order to avoid bias
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        j = coord_order[it % n]
        aj = a[j]
        
        # Compute b0 (convolution with a_j = 0)
        a[j] = 0.0
        b0 = fast_conv(a)
        a[j] = aj
        
        # d[k] = 2 * a[k-j] for valid k
        d = np.zeros(2 * n - 1)
        k_min = max(0, j)  # k-j >= 0 => k >= j
        k_max = min(2 * n - 2, j + n - 1)  # k-j <= n-1 => k <= j+n-1
        for k in range(k_min, k_max + 1):
            d[k] = 2.0 * a[k - j]
        
        # e[k] = 1 if k = 2j, else 0
        e = np.zeros(2 * n - 1)
        if 0 <= 2 * j < 2 * n - 1:
            e[2 * j] = 1.0
        
        S0 = a.sum() - aj  # sum of all other coordinates
        
        # Minimize max_k [b0[k] + t*d[k] + t^2*e[k]] / (S0+t)^2 for t >= 0
        # Let f(t) = max_k (b0[k] + t*d[k] + t^2*e[k]) / (S0+t)^2
        # This is a ratio of convex functions... not necessarily convex.
        # Use ternary search on a bounded interval.
        
        def f_1d(t):
            if t < 0:
                t = 0
            num = b0 + t * d + t * t * e
            den = (S0 + t) ** 2
            if den < 1e-30:
                return float('inf')
            return float(np.max(num / den))
        
        # Find bounds
        t_lo = 0.0
        t_hi = 2.0 * aj + 0.1  # Try at most double the current value + a bit
        # Also try t = 0 and t = aj
        f_aj = f_1d(aj)
        f_0 = f_1d(0.0)
        f_2aj = f_1d(2 * aj + 0.01)
        
        if f_2aj < f_aj:
            t_hi = 4 * aj + 0.01
            f_4aj = f_1d(t_hi)
            if f_4aj < f_2aj:
                t_hi = 8 * aj + 0.01
        
        # Ternary search
        lo, hi = t_lo, t_hi
        for _ in range(40):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = f_1d(m1)
            f2 = f_1d(m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        
        t_opt = (lo + hi) / 2.0
        t_opt = max(0.0, t_opt)
        
        a[j] = t_opt
        s = a.sum()
        if s > 1e-10:
            a = a / s
        
        M = compute_M(a)
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_diverse_initializations(n, rng, count=10):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    # 1. Uniform
    inits.append(np.ones(n) / n)
    
    # 2. Gaussian bump
    x = np.arange(n)
    sigma = n / 4
    g = np.exp(-0.5 * ((x - n/2) / sigma) ** 2)
    inits.append(g / g.sum())
    
    # 3. Two peaks
    g1 = np.exp(-0.5 * ((x - n*0.25) / (n*0.1)) ** 2)
    g2 = np.exp(-0.5 * ((x - n*0.75) / (n*0.1)) ** 2)
    g = g1 + g2
    inits.append(g / g.sum())
    
    # 4. Ramp
    r = np.linspace(0.1, 1, n)
    inits.append(r / r.sum())
    
    # 5. Triangle
    tri = np.minimum(x, n - 1 - x) + 0.1
    inits.append(tri / tri.sum())
    
    # 6. Sparse random
    for _ in range(count // 2):
        sparsity = rng.integers(n // 4, n // 2)
        idx = rng.choice(n, size=sparsity, replace=False)
        s = np.zeros(n)
        s[idx] = rng.dirichlet(np.ones(sparsity))
        inits.append(s + 1e-6)
        inits[-1] = inits[-1] / inits[-1].sum()
    
    # 7. Oscillatory
    for freq in [3, 7, 13]:
        osc = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n)
        inits.append(osc / osc.sum())
    
    # 8. Power law
    pl = (x + 1) ** (-0.5)
    inits.append(pl / pl.sum())
    pl2 = (n - x) ** (-0.5)
    inits.append(pl2 / pl2.sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining:
    1. LP mirror descent on parent (proven effective)
    2. Subgradient method with momentum (faster per iteration)
    3. Coordinate descent (cheap local refinement)
    4. Diverse multi-start with structured initializations
    5. Adaptive perturbation restarts
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    n0 = parent.size
    row_idx0, col_idx0 = build_lp_matrix(n0)

    # === PHASE 1: LP mirror descent on parent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent on parent (n={n0})", flush=True)
    print(f"{'='*60}", flush=True)

    phase1_deadline = min(t_start + budget_s * 0.30, deadline)
    a_p1 = parent.copy() / parent.sum()
    a_p1 = run_lp_mirror_descent(a_p1, 50000, phase1_deadline, row_idx0, col_idx0, verbose=True)
    tracker.update(a_p1, "(P1 LP-MD parent)")

    # === PHASE 2: Subgradient polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Subgradient Method Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        phase2_dl = min(t_start + budget_s * 0.40, deadline)
        a_p2 = tracker.best.copy() / tracker.best.sum()
        a_p2 = run_subgradient_method(a_p2, 5000, phase2_dl, tau0=10.0, tau1=80.0, lr0=0.005, lr1=0.0005)
        tracker.update(a_p2, "(P2 subgrad)")
        
        # Short LP polish
        n_p2 = tracker.best.size
        if n_p2 != n0:
            ri2, ci2 = build_lp_matrix(n_p2)
        else:
            ri2, ci2 = row_idx0, col_idx0
        phase2b_dl = min(t_start + budget_s * 0.45, deadline)
        a_p2b = tracker.best.copy() / tracker.best.sum()
        a_p2b = run_lp_mirror_descent(a_p2b, 5000, phase2b_dl, ri2, ci2)
        tracker.update(a_p2b, "(P2b LP polish)")

    # === PHASE 3: Coordinate Descent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Coordinate Descent", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        phase3_dl = min(t_start + budget_s * 0.60, deadline)
        a_p3 = tracker.best.copy()
        a_p3 = coordinate_descent(a_p3, 200 * a_p3.size, phase3_dl)
