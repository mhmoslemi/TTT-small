        a = a_new

        if f < best_f:
            best_f = f
            best_a = a.copy()

            print(f"    iter {it}: f={f:.10f} best={best_f:.10f} beta={beta:.1f}", flush=True)

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_h_and_grad_lse(a, beta):
    """Compute self-convolution and LSE gradient in one pass."""
    n = len(a)
    size1 = 1

    h = np.fft.irfft(fa1 * fa1, size1)[:2 * n - 1]

    h_max = h.max()
    logit = np.clip(beta * (h - h_max), -700, 700)
    exp_logit = np.exp(logit)
    p = exp_logit / exp_logit.sum()

    size2 = 1

        size2 <<= 1

    fa2 = np.fft.rfft(a, size2)

def compute_h_and_grad_powersum(a, p_power):
    """Compute self-convolution and power-sum gradient."""
    n = len(a)
    size1 = 1
    while size1 < 2 * n - 1:
        size1 <<= 1
    fa1 = np.fft.rfft(a, size1)
    h = np.fft.irfft(fa1 * fa1, size1)[:2 * n - 1]

    h_clipped = np.maximum(h, 1e-30)
    w = h_clipped ** (p_power - 1)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = w / np.power(np.sum(h_clipped ** p_power), 1.0 - 1.0 / p_power)

    size2 = 1
    while size2 < 3 * n:
        size2 <<= 1
    Fw = np.fft.rfft(w, size2)
    fa2 = np.fft.rfft(a, size2)
    cross = np.fft.irfft(Fw * np.conj(fa2), size2)
    grad = 2.0 * cross[:n]

    return h, grad


def proj_simplex(x):
    """Project x onto the probability simplex {a >= 0, sum a = 1}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (cssv - 1)
    valid = np.where(cond)[0]
    if len(valid) == 0:
        return np.full(n, 1.0 / n)
    rho = valid[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(x - theta, 0)


def run_lse_adam(a, max_iters, deadline, beta_init=3.0, beta_final=800.0,

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        beta = beta_init + (beta_final - beta_init) * (frac ** 0.7)
        lr = lr_init + (lr_final - lr_init) * frac

        h, grad = compute_h_and_grad_lse(a, beta)

def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2 via ternary search."""
    size = 1
    while size < 2 * len(a) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fg = np.fft.rfft(g, size)
    A = np.fft.irfft(fa * fa, size)[:2 * len(a) - 1]
    B = np.fft.irfft(fa * fg, size)[:2 * len(a) - 1]
    C = np.fft.irfft(fg * fg, size)[:2 * len(g) - 1]

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    print(f"{'=' * 60}", flush=True)

    n2 = tr['best'].size
    ri, ci = build_lp_matrix(n2)
    a2 = tr['best'].copy() / tr['best'].sum()
    dl2 = t_start + budget_s * 0.45
    a2_opt = run_lp_mirror_descent(a2, 20000, min(dl2, deadline), ri, ci, verbose=True)
    tracker_update(tr, a2_opt, "(P2 LP)")

    # Brief LSE after LP
    dl2b = t_start + budget_s * 0.50
    a2_lse = run_lse_adam(a2_opt, 10000, min(dl2b, deadline),
                          beta_init=50.0, beta_final=1000.0,
                          lr_init=0.01, lr_final=0.0005, verbose=False)
    tracker_update(tr, a2_lse, "(P2 LSE post-LP)")

    # ===================== PHASE 3: Diverse initializations =====================
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Multi-Init Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = []
    n_base = tr['best'].size

    # Various structured inits at the same length
    inits.append(('gauss_wide', gen_gaussian(n_base, 0.5, 0.4)))
    inits.append(('gauss_narrow', gen_gaussian(n_base, 0.5, 0.15)))
    inits.append(('raised_cos', gen_raised_cosine(n_base)))
    inits.append(('bimodal', gen_bimodal(n_base, 0.1)))
    inits.append(('hamming', gen_windowed(n_base, 'hamming')))
    inits.append(('blackman', gen_windowed(n_base, 'blackman')))

    # Random inits with different smoothness
    for smooth in [0.01, 0.05, 0.2]:
        r = rng.random(n_base)
        # Smooth it
        k = max(2, int(n_base * smooth))
        kernel = np.ones(k) / k
        r = np.convolve(r, kernel, mode='same')
        r = np.maximum(r, 0)
        r = r / r.sum()
        inits.append((f'random_s{smooth}', r))

    # Perturbed parent
    pert = tr['best'].copy()
    pert += rng.normal(0, 0.02 * tr['best'].max(), n_base)
    pert = np.maximum(pert, 1e-6)
    pert = pert / pert.sum()
    inits.append(('perturbed_best', pert))

    dl3 = t_start + budget_s * 0.70
    per_init_time = max(5, (dl3 - time.monotonic()) / len(inits))

    for name, a_init in inits:
        if time.monotonic() > dl3:
            break
        a_try = a_init.copy()
        sub_dl = min(time.monotonic() + per_init_time, dl3)
        a_opt = run_lse_adam(a_try, 5000, sub_dl,
                             beta_init=5.0, beta_final=300.0,
                             lr_init=0.04, lr_final=0.002, verbose=False)
        tracker_update(tr, a_opt, f"(P3 {name})")

    # ===================== PHASE 4: Multi-length exploration =====================
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    dl4 = t_start + budget_s * 0.82
    n_best = tr['best'].size
    trial_lengths = []
    for mult in [0.5, 0.75, 1.25, 1.5, 2.0]:
        n_trial = int(n_best * mult)
        if 128 <= n_trial <= 4096 and n_trial != n_best:
            trial_lengths.append(n_trial)
    trial_lengths = sorted(set(trial_lengths))

    per_len_time = max(3, (dl4 - time.monotonic()) / max(1, len(trial_lengths)))
    for n_trial in trial_lengths:
        if time.monotonic() > dl4:
            break
        a_init = resample_seq(tr['best'], n_trial)
        a_init = np.maximum(a_init, 1e-6)
        a_init = a_init / a_init.sum()
        sub_dl = min(time.monotonic() + per_len_time, dl4)
        a_opt = run_lse_adam(a_init, 5000, sub_dl,
                             beta_init=3.0, beta_final=400.0,
                             lr_init=0.04, lr_final=0.002, verbose=False)
        tracker_update(tr, a_opt, f"(P4 n={n_trial})")

    # ===================== PHASE 5: Perturbation restarts on best =====================
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Perturbation Restarts", flush=True)
    print(f"{'=' * 60}", flush=True)

    dl5 = t_start + budget_s * 0.90
    n_best = tr['best'].size
    temperature = 0.03

    for r in range(20):
        if time.monotonic() > dl5:
            break
        a = tr['best'].copy().astype(np.float64)
        noise_scale = temperature * 0.01
        a = a + rng.normal(0, noise_scale, n_best)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()

        sub_dl = min(time.monotonic() + 5, dl5)
        a = run_lse_adam(a, 2000, sub_dl,
                         beta_init=20.0, beta_final=500.0,
                         lr_init=0.02, lr_final=0.001, verbose=False)
        tracker_update(tr, a, f"(P5 r={r})")
        temperature *= 0.85

    # ===================== PHASE 6: Final LP + LSE polish =====================
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 6: Final Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    n_final = tr['best'].size
    rf, cf = build_lp_matrix(n_final)
    a_final = tr['best'].copy() / tr['best'].sum()

    # LP polish
    dl6 = min(t_start + budget_s * 0.95, deadline)
    a_lp = run_lp_mirror_descent(a_final, 20000, dl6, rf, cf, verbose=True)
    tracker_update(tr, a_lp, "(P6 LP)")

    # Final LSE with very high beta
    if time.monotonic() < deadline:
        a_lse_final = run_lse_adam(a_lp, 50000, deadline,
                                   beta_init=100.0, beta_final=2000.0,
                                   lr_init=0.005, lr_final=0.0001, verbose=True)
        tracker_update(tr, a_lse_final, "(P6 LSE final)")

    # Fallback
    if tr['best'] is None:
        tr['best'] = parent.copy()
        tr['best_score'] = compute_J(tr['best'])

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tr['best_score']:.8f} n={tr['best'].size} time={elapsed:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tr['best']]
