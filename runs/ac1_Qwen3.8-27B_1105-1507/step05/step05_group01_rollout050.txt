Key insights:
- The objective is `2n * max_b / sum_a^2`. Since we can scale the sequence arbitrarily (the ratio is scale-invariant), we can normalize `sum_a = 1` and minimize `2n * max_b`.
- The convolution `conv(a,a)` has `2n-1` elements. The maximum is at least `sum(a)^2 / (2n-1)` by averaging, so `max_b >= sum_a^2/(2n-1)`, giving `2n * max_b / sum_a^2 >= 2n/(2n-1)` which approaches 1 as n grows.
- For a constant sequence `a = [c, c, ..., c]`, `conv(a,a) = [c^2, 2c^2, ..., nc^2, (n-1)c^2, ..., c^2]`, so `max_b = n*c^2`, `sum_a = n*c`, and the objective = `2n * n*c^2 / (n*c)^2 = 2n/n = 2`.
- So constant sequence gives 2. We need to do much better.
- For a "random" sequence, the convolution values are roughly `sum_a^2 / (2n)` on average with some variance, so the peak would be around `sum_a^2 * (1 + O(sqrt(log n / n))) / (2n)`, giving objective around `1 + O(sqrt(log n / n))`.

The key challenge is making the autocorrelation as flat as possible while keeping all values non-negative.

5. **New idea - "Peak shaving" via direct computation**:
   - The convolution at position k is `sum_i a[i] * a[k-i]`
   - If position k is a peak, we want to reduce `sum_i a[i] * a[k-i]`
   - We can do this by reducing some `a[i]` values where both `a[i]` and `a[k-i]` are large
   - But we also need to maintain the sum (or we can rescale)
   - A gradient-based approach: the gradient of `max_k sum_i a[i]*a[k-i]` with respect to `a[j]` is `sum_{k: k is argmax} a[k-j]` (for the active peaks)

6. **Subgradient method**: Since we're minimizing a convex function (the max of linear functions) subject to non-negativity, we can use subgradient descent:
   - At each step, find which convolution positions are at the maximum
   - Compute the subgradient with respect to each `a[j]`
   - Move in the negative subgradient direction
   - Project back to the non-negative orthant
   - Rescale to maintain sum = 1

7. **Even better - use the LP more cleverly**:

8. **Another idea - "proportional reduction"**:

   - The contribution of `a[j]` to this peak is `a[j]*a[k-j]` (if `k-j` is in range)
   - We can reduce `a[j]` proportionally to its contribution to the peak
   - But we need to do this carefully to not create new peaks elsewhere

Let me think about a more systematic approach:

**Subgradient/projection approach:**

- Compute `conv = a*a`
- Find all positions where `conv[k] >= M - epsilon` (near-maximum positions)

"""
Hybrid LP-subgradient optimization for minimizing autocorrelation peak ratio.
Uses LP directions, subgradient refinement, peak-pair targeting, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

def solve_lp(a, M, relax=1.0):
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    A = build_toeplitz(a)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            duals = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
            return g, s, duals
    except:
        pass
    return None, 0.0, None

def subgradient_step(a, step_size, n_active=10):
    """Subgradient descent on the non-negative simplex."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    # Find active (near-max) positions
    active = np.where(conv >= M - 0.01 * M)[0]
    if len(active) > n_active:
        active = active[np.argsort(conv[active])[::-1][:n_active]]
    
    # Compute subgradient
    sg = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                sg[i] += a[j]
    
    # Normalize subgradient
    sg_norm = np.linalg.norm(sg)
    if sg_norm < 1e-10:
        return a.copy()
    sg = sg / sg_norm
    
    # Apply step
    a_new = a - step_size * sg
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s

def peak_pair_target(a, num_rounds=100, time_budget=20.0):

            if not contributions:

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def dual_guided_perturb(a, duals, strength=0.1):
    """Use LP dual variables to guide perturbation direction."""
    n = len(a)
    if duals is None:
        return a.copy()
    
    # Dual variables correspond to convolution positions
    # High dual value = binding constraint = peak position
    # We want to perturb in a way that helps at binding constraints
    
    # For each a[j], its "importance" at binding constraint k is a[k-j]
    # So we want to reduce a[j] where it contributes to binding constraints
    
    M_dual = float(np.max(np.abs(duals))) if len(duals) > 0 else 0
    if M_dual < 1e-10:
        return a.copy()
    
    # Compute how much each a[j] contributes to binding constraints
    contribution = np.zeros(n)
    for k in range(len(duals)):
        if abs(duals[k]) > 0.1 * M_dual:
            weight = abs(duals[k]) / M_dual
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contribution[i] += weight * a[j]
    
    # Perturb: reduce where contribution is high
    max_contrib = np.max(contribution)
    if max_contrib < 1e-10:
        return a.copy()
    
    a_new = a * (1.0 - strength * contribution / max_contrib)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s

"""
Optimized LP-based search with dual-guided perturbations and peak-pair targeting
for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

def solve_lp_fast(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. Toeplitz(a) @ g <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    A = build_toeplitz(a)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            duals = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
            return g, s, duals
    except Exception:
        pass
    return None, 0.0, None

"""
LP-based optimization with adaptive multi-strategy escape, peak-targeted
perturbations, spectral shaping, and multi-length exploration for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. Toeplitz(a)@g <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    A = build_toeplitz(a)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            duals = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
            return g, s, duals
    except Exception:
        pass
    return None, 0.0, None

    return np.maximum(np.interp(x_new, x_old, a), 0)
