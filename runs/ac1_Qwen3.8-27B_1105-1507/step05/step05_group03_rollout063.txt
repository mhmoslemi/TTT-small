"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix, lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def logsumexp_conv(a, tau):
    """Compute (1/tau)*log(sum exp(tau*(a*a))) and its gradient."""
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    s = exp_b.sum()
    val = (1.0 / tau) * (tau * b_max + np.log(s))
    p = exp_b / s  # softmax weights
    
    # Gradient: sum_k p_k * grad_k(a)
    # grad_k(a)[m] = 2*a[k-m] if 0 <= k-m < n else 0
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        # grad_k[a] = 2 * flip(a) shifted by k
        # (a*a)[k] = sum_j a[j]*a[k-j]
        # d/d(a[m]) = 2*a[k-m] for valid k-m
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[m] += 2 * p[k] * a[k-m] for m in [lo, hi]
        # a[k-m] for m in [lo, hi] is a[k-hi : k-lo+1] reversed
        grad[lo:hi+1] += 2.0 * p[k] * a[k-hi:k-lo+1][::-1]
    
    return val, grad


def logsumexp_conv_fast(a, tau):
    """Faster version using matrix-free gradient."""
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    s = exp_b.sum()
    if s == 0 or not np.isfinite(s):
        return (1.0/tau) * (tau * b_max), np.zeros(len(a))
    val = (1.0 / tau) * (tau * b_max + np.log(s))
    p = exp_b / s
    
    n = len(a)
    # Use FFT-based approach for gradient
    # grad[m] = sum_k p[k] * 2*a[k-m] for valid k-m
    # This is a cross-correlation of p with a (reversed)
    # grad = 2 * convolve(p, a_reversed) ... let me think
    
    # Actually: grad[m] = 2 * sum_k p[k] * a[k-m] (where a[k-m] is valid)
    # = 2 * (p * a_reversed)[m - (n-1)] ... 
    
    # Let's just use the loop for now, it's fast enough for n=512
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += 2.0 * pk * a[k-hi:k-lo+1][::-1]
    
    return val, grad


def lsmd_objective(a, tau):
    """Objective and gradient for L-BFGS."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(a)

    #

    # where a_norm = a/s.

    return val, grad


def run_lbfgs_smooth(a, max_iters, deadline, tau_start=5.0, tau_end=200.0, verbose=False):
    """
    Run L-BFGS on the log-sum-exp approximation with adaptive temperature.
    Optimizes on the probability simplex (a >= 0, sum(a) = 1).
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    best_val = compute_J(a)
    
    # We'll do multiple rounds with increasing tau
    num_rounds = 5
    round_time = (deadline - time.monotonic()) / num_rounds
    
    for round_idx in range(num_rounds):
        if time.monotonic() > deadline:
            break
        
        frac = round_idx / max(1, num_rounds - 1)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.7)
        
        def obj_and_grad(x):
            x = np.maximum(x, 0)
            s = x.sum()
            if s < 1e-10:
                return 1e10, np.zeros_like(x)
            xn = x / s
            val, grad = logsumexp_conv(xn, tau)
            # Chain rule: xn = x/s
            # dxn/dx is the projection onto the tangent space of the simplex
            # Grad of val w.r.t. x: 
            # val depends on xn = x/s
            # d(val)/dx[m] = sum_j (d(val)/dxn[j]) * (dxn[j]/dx[m])
            # dxn[j]/dx[m] = delta[j,m]/s - x[j]/s^2
            # So d(val)/dx = (1/s)*grad - (1/s^2)*sum(grad)*x
            # = (1/s) * (grad - mean(grad) * ... ) 
            # Actually: d(val)/dx[m] = (1/s)*grad[m] - (1/s^2)*sum_j(grad[j]*x[j])
            # = (1/s) * (grad[m] - (sum_j grad[j]*x[j])/s)
            # Since xn = x/s, sum_j grad[j]*x[j] = s * sum_j grad[j]*xn[j]
            # So d(val)/dx[m] = (1/s) * grad[m] - (1/s) * sum_j grad[j]*xn[j]
            # = (1/s) * (grad[m] - <grad, xn>)
            g_x = (grad - np.dot(grad, xn)) / s
            return val, g_x
        
        # Use L-BFGS-B with bounds [0, inf)
        options = {
            'maxiter': max_iters // num_rounds,
            'ftol': 1e-15,
            'gtol': 1e-12,
            'maxls': 50,
            'disp': False
        }
        
        res = minimize(obj_and_grad, a, method='L-BFGS-B', 
                      bounds=[(0, None)] * n, options=options)
        
        if res.x is not None and res.fun < float('inf'):
            a_new = np.maximum(res.x, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                M = compute_M(a_new)
                J = compute_J(a_new)
                if J < best_val:
                    best_val = J
                    best_M = M
                    best_a = a_new.copy()
                a = a_new
        
        if verbose:
            elapsed = time.monotonic()
            print(f"    L-BFGS round {round_idx+1}: tau={tau:.1f} J={best_val:.8f}", flush=True)
    
    return best_a


def build_lp_matrix_sparse(n, K=None):
    """Build sparse structure for LP constraint matrix.
    If K is given, only include top-K constraint structure (all constraints are included
    in the structure, but we'll mask later)."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp_reduced(a, M, row_idx_all, col_idx_all, K=50):
    """Solve LP with only top-K constraints.
    max 1^T g s.t. (a@g)[k] <= M for top-K peak locations, g >= 0."""
    n = len(a)
    
    # Find top-K peak locations in a*a
    b = fast_conv(a)
    top_k = np.argsort(b)[::-1][:K]
    
    # Filter to only top-K constraints
    mask = np.isin(row_idx_all, top_k)
    row_idx = row_idx_all[mask]
    col_idx = col_idx_all[mask]
    
    if len(row_idx) == 0:
        return None
    
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(K, n))
    
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def solve_lp_full(a, M, row_idx, col_idx):
    """Solve LP with all constraints."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g, n_iter=60):
    """
    Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2.
    Both a and g normalized to sum 1.
    """
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    lo = 0.0
    hi = 1.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent_v2(a, max_iters, deadline, row_idx, col_idx, 
                              use_reduced=True, K=80, verbose=False):
    """
    Improved LP mirror descent with constraint reduction and adaptive K.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    it = 0
    
    while it < max_iters:
        if time.monotonic() > deadline:
            break
        
        M = compute_M(a)
        
        if use_reduced:
            g = solve_lp_reduced(a, M, row_idx, col_idx, K=K)
        else:
            g = solve_lp_full(a, M, row_idx, col_idx)
        
        if g is None:
            # Try full LP as fallback
            g = solve_lp_full(a, M, row_idx, col_idx)
            if g is None:
                break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        
        t, M_new = line_search_analytic(a, g)
        
        improved = False
        if t > 1e-8 and M_new < best_M * (1 - 1e-12):
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Adaptive K: increase K when stuck
        if no_improve > 50:
            K = min(K + 20, 200)
            no_improve = 0
        
        if no_improve > 200:
            break
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} K={K}", flush=True)
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self):
        if self.best is not None:
            print(f"  best: J={self.best_score:.8f} n={self.best.size}", flush=True)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on adaptive log-sum-exp approximation (fast convergence)
2. LP mirror descent with constraint reduction (fine-tuning)
3. Multi-length exploration with structured resampling
4. Subgradient perturbation restarts
5. Final extended LP polish with symmetry projection
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lse_val_grad(a, tau):
    """Compute log-sum-exp of (a*a) values and gradient w.r.t. a."""
    b = fast_conv(a)
    b_max = b.max()
    scaled = tau * (b - b_max)
    # Numerical stability
    if scaled.max() > 500:
        scaled = scaled / 10.0
        tau_eff = tau / 10.0
    else:
        tau_eff = tau
    exp_b = np.exp(scaled)
    s_exp = exp_b.sum()
    if s_exp < 1e-300:
        return (1.0 / tau) * b_max, np.zeros(len(a))
    val = (1.0 / tau_eff) * (tau_eff * b_max + np.log(s_exp))
    p = exp_b / s_exp

    n = len(a)
    grad = np.zeros(n)
    active = p > 1e-12
    for k in np.where(active)[0]:
        pk = p[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return val, grad


def run_lbfgs_smooth(a, max_total_iters, deadline, tau_start=5.0, tau_end=200.0,
                     num_rounds=6, verbose=False):
    """
    L-BFGS on log-sum-exp approximation with adaptive temperature.
    Optimizes on the probability simplex (a>=0, sum(a)=1).
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    best_val = compute_J(a)

    for round_idx in range(num_rounds):
        if time.monotonic() > deadline:
            break

        frac = round_idx / max(1, num_rounds - 1)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.6)
        iters_this_round = max(50, max_total_iters // num_rounds)

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s < 1e-10:
                return 1e10, np.zeros_like(x)
            xn = x / s
            val, grad = lse_val_grad(xn, tau)
            if not np.isfinite(val):
                return 1e10, np.zeros_like(x)
            # Chain rule for normalization: xn = x/s
            # d(val)/dx[m] = (1/s)*grad[m] - (1/s^2)*sum_j(grad[j]*x[j])
            dot_grad_xn = np.dot(grad, xn)
            g_x = (grad - dot_grad_xn) / s
            return val, g_x

        bounds = [(0.0, None)] * n
        options = {
            'maxiter': iters_this_round,
            'ftol': 1e-16,
            'gtol': 1e-14,
            'maxls': 40,
            'disp': False
        }

        try:
            res = minimize(obj_and_grad, a, method='L-BFGS-B',
                          bounds=bounds, options=options)
            if res.x is not None and np.isfinite(res.fun):
                a_new = np.maximum(res.x, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    J = compute_J(a_new)
                    if J < best_val:
                        best_val = J
                        best_M = compute_M(a_new)
                        best_a = a_new.copy()
                    a = a_new
        except Exception:
            pass

        if verbose:
            print(f"    L-BFGS round {round_idx+1}/{num_rounds}: "
                  f"tau={tau:.1f} J={best_val:.8f}", flush=True)

    return best_a


def build_lp_structure(n):
    """Precompute row/col index structure for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_reduced(a, M, row_idx, col_idx, K=100):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M for top-K peaks, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    top_k = np.argsort(b)[::-1][:K]

    mask = np.isin(row_idx, top_k)
    r = row_idx[mask]
    c = col_idx[mask]

    if len(r) == 0:
        return None

    vals = a[r - c].astype(np.float64)
    A = csr_matrix((vals, (r, c)), shape=(K, n))

    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'time_limit': 5, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def solve_lp_full(a, M, row_idx, col_idx):
    """Solve LP with all 2n-1 constraints."""
    n = len(a)
    vals = a[row_idx - col_idx].astype(np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'time_limit': 8, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g, n_iter=65):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_c = A
    b_c = 2.0 * (B - A)
    c_c = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_c + b_c * m1 + c_c * m1 * m1)
        f2 = np.max(a_c + b_c * m2 + c_c * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_c + b_c * t + c_c * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx,
                          use_reduced=True, K=100, verbose=False):
    """LP mirror descent with adaptive constraint reduction."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    it = 0

    while it < max_iters:
        if time.monotonic() > deadline:
            break

        M = compute_M(a)

        if use_reduced:
            g = solve_lp_reduced(a, M, row_idx, col_idx, K=K)
            if g is None:
                g = solve_lp_full(a, M, row_idx, col_idx)
        else:
            g = solve_lp_full(a, M, row_idx, col_idx)

        if g is None:
            break

        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t > 1e-8 and M_new < best_M * (1 - 1e-13):
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adaptive K: expand when stuck
        if no_improve > 60 and use_reduced:
            K = min(K + 25, 250)
            no_improve = 0

        if no_improve > 250:
            break

        it += 1
        if verbose and it % 200 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} "
                  f"K={K if use_reduced else 'full'} t={t:.6f}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def try_symmetry(a):
    """Try palindromic projection and return the better of a and its symmetrized version."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s < 1e-10:
        return a
    a_sym = a_sym / s
    J_orig = compute_J(a)
    J_sym = compute_J(a_sym)
    if J_sym < J_orig:
        return a_sym
    return a
