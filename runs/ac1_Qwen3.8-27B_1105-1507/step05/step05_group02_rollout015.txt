    for i in range(1, n_pop):
        scale = 0.005 + 0.02 * (i / n_pop)
        noise = rng.normal(0, scale, n)
        cand = np.maximum(base + noise, 0)
        s = cand.sum()
        if s > 1e-10:
            cand /= s
        pop.append(cand)

    for gen in range(n_gen):
        if deadline is not None and time.monotonic() > deadline:
            break
        new_pop = []
        for i in range(n_pop):
            if i == 0:
                new_pop.append(pop[0].copy())
                continue
            idxs = [j for j in range(n_pop) if j != i]
            r1, r2 = rng.choice(len(idxs), 2, replace=False)
            r1, r2 = idxs[r1], idxs[r2]
            mutant = pop[r1] + F * (pop[r2] - pop[i])
            mutant = np.maximum(mutant, 0)
            s = mutant.sum()
            if s < 1e-10:
                new_pop.append(pop[i].copy())
                continue
            mutant /= s
            M_mut = fast_conv(mutant).max()
            M_cur = fast_conv(pop[i]).max()
            if M_mut <= M_cur:
                new_pop.append(mutant)
                if M_mut < best_M:
                    best_M = M_mut
                    best_out = mutant.copy()
            else:
                new_pop.append(pop[i].copy())
        pop = new_pop

    return best_out


def resample_seq(src, n):
    """Resample a sequence to a new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.015, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.04, n), 0.001))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.001))

    inits.append(np.ones(n))

    for center, width in [(0.5, 0.15), (0.5, 0.25), (0.3, 0.2), (0.7, 0.2)]:
        bump = np.exp(-((t - center) / width) ** 2)
        inits.append(np.maximum(bump + 0.05, 0.01))

    for f, amp in [(1, 0.3), (2, 0.2), (1, 0.5), (3, 0.15), (2, 0.4)]:
        inits.append(np.maximum(0.5 + amp * np.cos(2 * np.pi * f * t), 0.01))

    for alpha in [0.5, 1.0, 1.5, 2.0, 4.0]:
        inits.append(np.maximum(t**alpha + (1 - t)**alpha + 0.05, 0.01))

    k = max(3, n // 20)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))

    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)

    # Normalize all
    result = []
    for a in inits:
        a = np.maximum(a.astype(np.float64), 0)
        s = a.sum()
        if s > 1e-10:
            result.append(a / s)
    return result


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            s = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(budget_s=1500):
    """Main search function. Returns best sequence found within budget."""
    rng = np.random.default_rng(12345)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    # Try to get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(512)
        best.update(parent)
        print(f"Using default parent: n=512", flush=True)

    base_n = parent.size
    if best.seq is not None:
        base_n = best.seq.size

    # Determine candidate lengths
    candidate_lengths = set()
    for n_try in range(256, 2049, 32):
        candidate_lengths.add(n_try)
    for mult in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
        n_try = int(base_n * mult)
        n_try = max(128, min(3072, (n_try // 32) * 32))
        candidate_lengths.add(n_try)
    lengths = sorted(candidate_lengths)
    print(f"Lengths to explore: {lengths[:5]}...{lengths[-5:]} (total {len(lengths)})", flush=True)

    # === PHASE 1: Smooth Nesterov across lengths (250s) ===
    p1_deadline = time.monotonic() + 250
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smooth Nesterov (250s budget)", flush=True)
    print(f"{'='*60}", flush=True)

    # Prioritize lengths near the parent
    length_priority = []
    for n in lengths:
        dist = abs(n - base_n) / base_n
        length_priority.append((dist, n))
    length_priority.sort()

    best_per_length = {}
    for dist, n in length_priority:
        if time.monotonic() > p1_deadline - 20:
            break
        if dist > 1.5:
            break
        # Time allocation: more time for lengths closer to parent
        allot = max(8, 60 * (1 - dist))
        dl = min(time.monotonic() + allot, p1_deadline - 10)
        if time.monotonic() >= dl:
            continue

        t0 = time.monotonic()
        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n_arr = None
        best_n_M = float('inf')

        n_iters = max(100, int(3000 * (512.0 / n)))
        for a0 in inits:
            if time.monotonic() > dl:
                break
            a = nesterov_smooth(a0, dl, max_iters=n_iters,
                                tau_start=1.0, tau_end=200.0,
                                lr_init=0.4, lr_end=0.003)
            M = fast_conv(a).max()
            if M < best_n_M:
                best_n_M = M
                best_n_arr = a.copy()

        if best_n_arr is not None:
            J_n = compute_J(best_n_arr)
            print(f"  n={n:4d}: M={best_n_M:.7f} J={J_n:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n_arr)
            best_per_length[n] = best_n_arr.copy()

    best.report("Phase 1")

    # === PHASE 2: LP Mirror Descent (600s) ===
    p2_deadline = time.monotonic() + 600
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (600s budget)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        # Run LP on the best sequence
        a = lp_mirror(best.seq, p2_deadline, max_iters=1000)
        best.update(a)
        best.report("Phase 2 main")

    # Also try LP on the second best if time remains
    if len(best_per_length) > 1 and time.monotonic() < p2_deadline - 30:
        sorted_lengths = sorted(best_per_length.keys(),
                                key=lambda n: compute_J(best_per_length[n]))
        if len(sorted_lengths) > 1:
            n2 = sorted_lengths[1]
            if n2 != best.seq.size:
                print(f"  Trying LP on second-best length n={n2}", flush=True)
                a2 = lp_mirror(best_per_length[n2], min(time.monotonic() + 60, p2_deadline - 10),
                               max_iters=100)
                best.update(a2)

    best.report("Phase 2")

    # === PHASE 3: DE Kicks + LP (250s) ===
    p3_deadline = time.monotonic() + 250
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: DE Kicks + LP (250s budget)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        kicks = 0
        while time.monotonic() < p3_deadline - 20:
            kicks += 1
            print(f"  DE kick {kicks}...", flush=True)
            a_kick = differential_evolution_kick(
                best.seq, rng, n_gen=25, n_pop=12, F=0.35,
                deadline=time.monotonic() + 40)
            best.update(a_kick)

            # Brief LP refinement after kick
            if time.monotonic() < p3_deadline - 15:
                a_lp = lp_mirror(a_kick, time.monotonic() + 30, max_iters=50)
                best.update(a_lp)

            if best.score < 1.5030:
                print(f"  Approaching target! J={best.score:.6f}", flush=True)

    best.report("Phase 3")

    # === PHASE 4: Final LP Polish (remaining time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        # Multiple LP passes with fresh starts
        for pass_num in range(3):
            if time.monotonic() > deadline - 5:
                break
            a = lp_mirror(best.seq, deadline, max_iters=500)
            best.update(a)
            if time.monotonic() > deadline - 5:
                break

        # Try a slightly different starting point for the last LP
        if time.monotonic() < deadline - 20:
            a_pert = best.seq + rng.normal(0, 0.002, best.seq.size)
            a_pert = np.maximum(a_pert, 0)
            a_pert /= a_pert.sum()
            a = lp_mirror(a_pert, deadline, max_iters=200)
            best.update(a)

    best.report("Phase 4")

    # === Final verification ===
    if best.seq is None:
        best.seq = np.ones(256)
        best.score = compute_J(best.seq)

    # One last check with evaluate_sequence
    final_score = evaluate_sequence(best.seq.tolist())
    if np.isfinite(final_score) and final_score < best.score:
        best.score = final_score

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best.seq]
