"""
Optimized peak-reduction search using alternating LP, subgradient methods,
and multi-initialization strategy for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0 or np.sum(a) < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def normalize(a):
    s = np.sum(a)
    if s < 1e-10:
        return a
    return a / s


def solve_minpeak_lp(a, time_limit=10):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, g>=0, sum(g)=1."""
    n = len(a)
    # Build constraint matrix
    # (a*g)[k] = sum_i a[i] * g[k-i] for valid indices
    # Constraint: sum_i a[k-i] * g[i] <= M for each k
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        # a[i] multiplies g[k] for k such that k-i is in [0, n-1], i.e., k in [i, i+n-1]
        # So A[k, i] = a[k-i] for k in [i, min(i+n-1, 2n-2)]
        # Equivalently, A[k, i] = a[k-i] if 0 <= k-i < n
        start = i  # k-i = 0 when k = i
        end = min(i + n, 2 * n - 1)  # k-i = n-1 when k = i+n-1
        A[start:end, i] = a[:end - start]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # -M on the left side
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(-np.inf, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception as e:
        print(f"LP error: {e}")
    return None, None


def alternating_lp_optimize(a, max_iters=50, time_budget=60.0, verbose=False):
    """Alternating LP optimization with adaptive step size."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break
        
        g, M_opt = solve_minpeak_lp(a, time_limit=min(15, time_budget - (time.monotonic() - start) - 2))
        if g is None:
            break
        
        # Try different step sizes
        best_t_obj = best_obj
        best_t = 0
        best_a_local = best_a.copy()
        
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
                best_a_local = a_try.copy()
        
        if best_t > 0:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local.copy()
            if verbose:
                print(f"  LP alt it{it}: obj={best_obj:.8f}, t={best_t:.2f}, M_opt={M_opt:.6f}")
        else:
            # No improvement, try smaller steps

"""

Uses alternating LP with LP-dual-guided subgradient, multi-initialization,

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def normalize(a):
    s = float(np.sum(a))
    if s < 1e-10:
        return a.copy()
    return a / s


def build_constraint_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        start = i
        end = min(i + n, 2 * n - 1)
        A[start:end, i] = a[:end - start]
    return A


def solve_minpeak_lp(a, time_limit=15):
    n = len(a)
    A = build_constraint_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(-np.inf, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, float(res.x[n]), res
    except Exception:
        pass
    return None, None, None


def subgradient_momentum(a, max_iters=300, time_budget=20.0, top_k=5):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    mu = 0.85
    step = 0.03

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Use top-k peaks for subgradient
        if top_k < len(conv):
            top_indices = np.argsort(conv)[-top_k:]
            weights = np.zeros(len(conv))
            for idx in top_indices:
                weights[idx] = 1.0 / top_k
        else:
            weights = np.full(len(conv), 1.0 / len(conv))

        # Gradient: grad[i] = sum_k weights[k] * (a[k-i] + a[i-k]) with bounds
        w_for_conv = weights.copy()
        # First term: sum_k w[k] * a[k-i] for valid k-i -> use convolution
        grad1 = fft_convolve(a, w_for_conv)[n - 1:2 * n - 1]
        # Second term: sum_k w[k] * a[i-k] for valid i-k

        grad2 = fft_convolve(a, w_rev)[:n][::-1]
        grad = grad1 + grad2

        # Center gradient
        grad = grad - np.mean(grad)

        v = mu * v - step * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        step = max(0.001, step * 0.998)

    return best_a, best_obj

def make_tapered_init(n, mode=0):
    if mode == 0:
        # Raised cosine
        return 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))
    elif mode == 1:
        # Linear ramp
        return np.linspace(0.1, 1.0, n)
    elif mode == 2:
        # U-shape
        x = np.linspace(-1, 1, n)
        return x ** 2 + 0.3
    elif mode == 3:
        # Two bump
        centers = np.array([n * 0.25, n * 0.75])
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-((np.arange(n) - c) ** 2) / (2.0 * (n / 8.0) ** 2))
        return a + 0.05
    elif mode == 4:
        # Anti-symmetric-ish: high at edges, low in middle
        x = np.linspace(0, 1, n)
        return (x - 0.5) ** 2 + 0.2
    else:
        return np.ones(n)

def resample_sequence(a, new_n):

                if remaining > 5:
                    a_cd, obj_cd = coordinate_refine(a_final,
                                                      time_budget=min(remaining * 0.3, 10.0))
                    if obj_cd < best_score:
                        best_score = obj_cd
                        best_seq = a_cd.copy()
                        a_final = a_cd.copy()
                        print(f"  Final CD {round_idx}: {best_score:.8f}")

            # Perturbation attempt
            remaining = deadline - time.monotonic()
            if remaining > 10 and round_idx % 2 == 1:
                rng_p = np.random.default_rng(round_idx * 31 + 7)
                for pert in range(2):
                    if time.monotonic() > deadline - 8:
                        break
                    scale = 0.02 + 0.01 * round_idx
                    a_pert = best_seq * np.exp(rng_p.normal(0, scale, len(best_seq)))
                    a_pert = np.maximum(a_pert, 0.0)
                    a_pert = normalize(a_pert)
                    a_pert, obj_pert = subgradient_momentum(a_pert, max_iters=100,
                                                             time_budget=5.0, top_k=3)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_final = a_pert.copy()
                        print(f"  Perturb {round_idx}.{pert}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
