```python
import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_and_grad(b, tau):
    """Compute smooth max and its gradient w.r.t. a."""
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    smooth_max = b_max + np.log(exp_b.sum()) / tau
    return smooth_max, p

def conv_gradient_from_p(p, a):
    """Given soft weights p over convolution indices, compute gradient of smooth max w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if p[k] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n-1, k)
        # For i in [lo, hi]: contribution is 2 * p[k] * a[k-i]
        # a[k-hi], a[k-hi+1], ..., a[k-lo]
        idx = np.arange(k-hi, k-lo+1)
        grad[lo:hi+1] += 2.0 * p[k] * a[idx]
    return grad

def full_J_gradient(a, tau):
    """Compute gradient of J(a) = 2n * smooth_max(conv(a,a), tau) / S^2."""
    n = len(a)
    b = fast_conv(a)
    S = a.sum()
    smooth_max, p = smooth_max_and_grad(b, tau)
    g = conv_gradient_from_p(p, a)
    # dJ/da_i = (2n/S^2) * g_i - (4n * smooth_max / S^3)
    J_grad = (2.0 * n / (S * S)) * g - (4.0 * n * smooth_max / (S * S * S))
    return J_grad, smooth_max

def adam_step(a, grad, m, v, t, lr, beta1, beta2, eps, constraint_sum=True):
    """Adam step with projection to a >= 0."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    if constraint_sum:
        s = a_new.sum()
        if s > 1e-15:

def nesterov_subgradient(a, deadline, max_iters=5000, lr_init=0.01):
    """Nesterov accelerated projected subgradient method on the exact (non-smooth) objective."""
    n = len(a)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    
    y = a.copy()
    momentum = 0.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead: y = a + momentum * (a - prev_a) -- simplified
        # Compute subgradient at y
        b = fast_conv(y)
        k_star = np.argmax(b)
        S = a.sum()
        
        # Subgradient of max(conv(a,a)) at y, evaluated as direction from y
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n-1, k_star)
        idx = np.arange(k_star - hi, k_star - lo + 1)
        g[lo:hi+1] = 2.0 * y[idx]
        
        # Gradient of J
        M = b.max()
        J_grad = (2.0 * n / (S * S)) * g - (4.0 * n * M / (S * S * S))
        
        # Update
        t_now = it + 1
        momentum = (1 + momentum) / (1 + momentum + 0.1)  # ~ 2/3 eventually
        # Actually let's use standard Nesterov
        # a_new = y - lr * J_grad
        # y_new = a_new + ((t)/(t+3)) * (a_new - a)
        
        lr = lr_init / np.sqrt(t_now)
        a_new = y - lr * J_grad
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        
        # Nesterov update
        theta = it / (it + 3.0)
        y = a_new + theta * (a_new - a)
        y = np.maximum(y, 0.0)
        y = y / y.sum()
        a = a_new
    
    return best_a, best_J

def spectral_shaping(a, alpha=2.0):
    """Sharpen the spectrum of a to make autocorrelation flatter."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    power = np.abs(A)**2
    # Sharpen: raise to alpha power
    power_sharp = power ** alpha
    # Need to find A' such that |A'|^2 = power_sharp (up to scale)
    # A' = A * (power_sharp / power)^0.5 = A * |A|^(alpha-1)
    # This preserves the phase of A
    A_new = A * (power / power.max() + 1e-20) ** ((alpha - 1) / 2)
    a_new = np.fft.irfft(A_new, size)[:n]
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() > 1e-10:
        a_new = a_new / a_new.sum()
    return a_new
```

```python
        if no_improve == 200:
            base_lr *= 0.7
            no_improve = 0
        elif no_improve > 2000:
            break

    return best_a, best_J


def run_nesterov_subgrad(a, deadline, lr_init=0.05, max_iters=100000):
    """Nesterov accelerated projected subgradient on exact J."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)

    y = a.copy()
    it = 0

    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(y)
        S = y.sum()
        M = b.max()
        k_star = np.argmax(b)

        # Subgradient of the objective at y
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        idx = np.arange(k_star - hi, k_star - lo + 1)
        g[lo:hi + 1] = 2.0 * y[idx]
        J_grad = (2.0 * n / (S * S)) * g - (4.0 * n * M) / (S * S * S)

        lr = lr_init / (1.0 + 0.01 * it)
        a_new = y - lr * J_grad
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / S_new

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()

        # Nesterov momentum update
        theta = it / (it + 3.0)
        y = a_new + theta * (a_new - a)
        y = np.maximum(y, 0.0)
        Sy = y.sum()
        if Sy < 1e-15:
            y = a_new.copy()
        else:
            y = y / Sy
        a = a_new

    return best_a, best_J


def spectral_shaping(a, alpha=2.0):
    """Sharpen the power spectrum to make autocorrelation flatter."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    power = np.abs(A) ** 2
    power_max = power.max()
    if power_max < 1e-20:
        return a.copy()
    # A_new = A * normalized_power^((alpha-1)/2)
    norm_power = power / power_max
    A_new = A * (norm_power + 1e-20) ** ((alpha - 1.0) / 2.0)
    a_new = np.fft.irfft(A_new, size)[:n]
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S < 1e-10:
        return a.copy()
    return a_new / S


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse non-symmetric initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Shifted Gaussians (non-symmetric centers)
    for center in [0.25, 0.35, 0.45, 0.55, 0.65, 0.75]:
        for width in [0.08, 0.15, 0.25, 0.35]:
            g = np.exp(-0.5 * ((t - center) / width) ** 2)
            g = np.maximum(g, 1e-6)
            inits.append(g / g.sum())

    # Ramps and power laws
    inits.append((np.arange(n, dtype=float) + 1) / (np.arange(n, dtype=float) + 1).sum())
    inits.append((n - np.arange(n, dtype=float)) / (n - np.arange(n, dtype=float)).sum())
    for alpha in [0.5, 1.0, 2.0, 4.0]:
        w = (t + 0.01) ** alpha
        inits.append(w / w.sum())
        w2 = (1.0 - t + 0.01) ** alpha
        inits.append(w2 / w2.sum())

    # Cosine with phase (breaks symmetry)
    for f in [1, 2, 3, 4]:
        for phase in [0.3, 0.8, 1.2, 1.7, 2.5]:
            sig = 1.0 + 0.4 * np.cos(2.0 * np.pi * f * t + phase)
            sig = np.maximum(sig, 0.01)
            inits.append(sig / sig.sum())

    # Asymmetric windows
    inits.append((np.sin(np.pi * t) ** 2 * np.exp(-t * 2)) / (np.sin(np.pi * t) ** 2 * np.exp(-t * 2)).sum())
    inits.append((np.sin(np.pi * t) ** 2 * np.exp(-(1-t) * 2)) / (np.sin(np.pi * t) ** 2 * np.exp(-(1-t) * 2)).sum())

    # Two-peak asymmetric
    for c1, c2, r1, r2 in [(0.3, 0.7, 0.1, 0.2), (0.2, 0.6, 0.08, 0.15), (0.4, 0.8, 0.1, 0.1)]:
        g = r1 * np.exp(-0.5 * ((t - c1) / 0.1) ** 2) + r2 * np.exp(-0.5 * ((t - c2) / 0.15) ** 2)
        g = np.maximum(g, 1e-6)
        inits.append(g / g.sum())

    # Random smooth
    for _ in range(8):
        n_freqs = rng.integers(2, 6)
        freqs = rng.integers(1, 10, size=n_freqs)
        phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
        amps = rng.uniform(0.1, 0.8, size=n_freqs)
        sig = np.ones(n)
        for f, ph, amp in zip(freqs, phases, amps):
            sig += amp * np.cos(2 * np.pi * f * t + ph)
        sig = np.maximum(sig, 0.01)
        inits.append(sig / sig.sum())

    # Parent-based
    if parent is not None and parent.size != n:
        t_old = np.linspace(0, 1, parent.size)
        p_res = np.maximum(np.interp(t, t_old, parent), 0.01)
        inits.append(p_res / p_res.sum())
        # Perturbed parent
        p_pert = np.maximum(p_res + 0.1 * rng.normal(0, 1, n), 0.01)
        inits.append(p_pert / p_pert.sum())
        p_pert2 = np.maximum(p_res + 0.3 * rng.normal(0, 1, n), 0.01)
        inits.append(p_pert2 / p_pert2.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a, label=""):
        if a is None or a.size == 0:
            return
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = compute_J(a)
        if score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s [{label}]", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_multi_tau_optimization(a, deadline, taus=[15, 30, 50, 80], lr=0.01):
    """Run Adam optimization with multiple temperatures sequentially."""
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    n_taus = len(taus)
    per_tau_time = (deadline - time.monotonic()) / n_taus

    for tau in taus:
        t_dead = time.monotonic() + per_tau_time
        if time.monotonic() >= t_dead:
            break
        a_out, J_out = run_adam_direct(best_a, t_dead, tau=tau, lr=lr)
        if J_out < best_J:
            best_a = a_out
            best_J = J_out

    return best_a, best_J


def run_exhaustive_polish(a, deadline):
    """Aggressive polishing with alternating Adam and Nesterov."""
    best_a = a.copy()
    best_J = compute_J(a)
    n = len(a)

    # Alternate between methods
    chunk_time = 20.0
    while time.monotonic() < deadline - 5:
        # Adam round
        t_dl = min(time.monotonic() + chunk_time, deadline - 5)
        a1, J1 = run_adam_direct(best_a, t_dl, tau=40, lr=0.003)
        if J1 < best_J:
            best_a, best_J = a1, J1

        # Nesterov round
        if time.monotonic() < deadline - 10:
            t_dl2 = min(time.monotonic() + chunk_time // 2, deadline - 5)
            a2, J2 = run_nesterov_subgrad(best_a, t_dl2, lr_init=0.02)
            if J2 < best_J:
                best_a, best_J = a2, J2

        # Spectral perturbation
        if time.monotonic() < deadline - 15:
            for alpha in [1.5, 2.5]:
                a_s = spectral_shaping(best_a, alpha)
                if a_s.sum() > 0.01:
                    J_s = compute_J(a_s)
                    if J_s < best_J:
                        best_A = a_s
                        best_J = J_s
                    # Quick Adam from shaped
                    t_dl3 = min(time.monotonic() + 5, deadline - 5)
                    a3, J3 = run_adam_direct(a_s, t_dl3, tau=30, lr=0.002)
                    if J3 < best_J:
                        best_a, best_J = a3, J3

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Diverse initialization screening with Adam (50% time)
    Phase 2: Multi-temperature deep optimization (25% time)
    Phase 3: Spectral shaping + multi-length exploration (10% time)
    Phase 4: Exhaustive final polishing (15% time)
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    t_start = time.monotonic()
    tracker = ScoreTracker()

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "parent")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "default")

    remaining = total_budget - 60.0

    # === PHASE 1: Multi-length initialization + screening ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: Multi-length Screening", flush=True)
    print(f"{'=' * 60}", flush=True)

    p1_deadline = t_start + remaining * 0.45
    lengths = list(dict.fromkeys([512, 384, 640, 320]))
    if parent is not None:
        lengths.insert(0, parent.size)

    all_candidates = []

    for n in lengths:
        if time.monotonic() > p1_deadline:
            break

        print(f"\n  Processing n={n}...", flush=True)
        inits = generate_diverse_inits(n, rng, parent)

        # Quick screen with short Adam runs
        time_per_init = max(3.0, min(15.0, (p1_deadline - time.monotonic()) / max(1, len(inits))))

        for idx, a0 in enumerate(inits):
            if time.monotonic() > p1_deadline - 10:
                break
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            # Very quick Adam
            t_dl = min(time.monotonic() + time_per_init, p1_deadline - 10)
            a_opt, J_opt = run_adam_direct(a0, t_dl, tau=30.0, lr=0.02)
            all_candidates.append((J_opt, a_opt, n))
            tracker.update(a_opt, f"screen_n{n}_{idx}")

            if (idx + 1) % 20 == 0:
                print(f"    {idx+1}/{len(inits)} done, best for n={n}: "
                      f"{min(c[0] for c in all_candidates if c[2]==n):.6f}", flush=True)

    all_candidates.sort(key=lambda x: x[0])
    print(f"\n  Top 5 candidates after screening:", flush=True)
    for i, (J, a, n) in enumerate(all_candidates[:5]):
        print(f"    #{i+1}: J={J:.8f} n={n}", flush=True)

    # === PHASE 2: Deep multi-temperature optimization on top candidates ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 2: Deep Multi-Temperature Optimization", flush=True)
    print(f"{'=' * 60}", flush=True)

    p2_deadline = min(t_start + remaining * 0.75, time.monotonic() + (p1_deadline - t_start) * 0.8)

    top_k = min(4, len(all_candidates))
    time_per_cand = (p2_deadline - time.monotonic()) / top_k if top_k > 0 else 0

    for ci in range(top_k):
        if time.monotonic() > p2_deadline:
            break
        J_init, a_init, n_init = all_candidates[ci]
        t_dl = min(time.monotonic() + time_per_cand, p2_deadline)

        print(f"\n  Deep-opt candidate #{ci+1} (J={J_init:.6f}, n={n_init}), budget={time_per_cand:.0f}s", flush=True)

        # Multi-tau Adam
        a_multi, J_multi = run_multi_tau_optimization(
            a_init, t_dl, taus=[15, 30, 50, 80], lr=0.008
        )
        tracker.update(a_multi, f"deep_tau_{ci}")

        # Nesterov from best
        if time.monotonic() < t_dl - 5:
            a_nest, J_nest = run_nesterov_subgrad(
                a_multi, min(time.monotonic() + 20, t_dl - 5), lr_init=0.03
            )
            tracker.update(a_nest, f"deep_nest_{ci}")

    print(f"\n  After Phase 2: best J={tracker.best_score:.8f} n={tracker.best.size}", flush=True)

    # === PHASE 3: Spectral shaping + multi-length ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 3: Spectral Shaping + Multi-length", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_deadline = min(t_start + remaining * 0.88, time.monotonic() + 60.0)

    if tracker.best is not None:
        base_n = tracker.best.size

        # Spectral shaping with various alphas
        for alpha in [1.5, 2.0, 3.0, 4.0, 6.0]:
            if time.monotonic() > p3_deadline:
                break
            a_shaped = spectral_shaping(tracker.best, alpha=alpha)
            if a_shaped.sum() > 0.01:
                J_shaped = compute_J(a_shaped)
                if J_shaped < tracker.best_score:
                    tracker.update(a_shaped, f"spectral_a{alpha}")
                # Re-optimize from shaped
                t_dl = min(time.monotonic() + 10, p3_deadline)
                a_opt, J_opt = run_adam_direct(a_shaped, t_dl, tau=30, lr=0.005)
                tracker.update(a_opt, f"spectral_a{alpha}_opt")

        # Try different lengths
        for n_try in [base_n - 64, base_n + 64, base_n + 128, base_n * 2]:
            if time.monotonic() > p3_deadline:
                break
            if n_try < 64 or n_try > 2048:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = np.maximum(a_res, 0.01)
            a_res = a_res / a_res.sum()
            t_dl = min(time.monotonic() + 15, p3_deadline)
            a_opt, J_opt = run_adam_direct(a_res, t_dl, tau=30, lr=0.01)
            tracker.update(a_opt, f"len_{n_try}")

    print(f"\n  After Phase 3: best J={tracker.best_score:.8f} n={tracker.best.size}", flush=True)

    # === PHASE 4: Final polishing ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 4: Final Polishing", flush=True)
    print(f"{'=' * 60}", flush=True)

    p4_deadline = t_start + total_budget - 40.0

    if tracker.best is not None and time.monotonic() < p4_deadline - 10:
        a_polish, J_polish = run_exhaustive_polish(tracker.best, p4_deadline)
        tracker.update(a_polish, "final_polish")

        # One more round with very low LR Adam
        if time.monotonic() < p4_deadline - 10:
            a_final, J_final = run_adam_direct(
                tracker.best, p4_deadline, tau=50, lr=0.0005
            )
            tracker.update(a_final, "final_adam_low_lr")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
```
