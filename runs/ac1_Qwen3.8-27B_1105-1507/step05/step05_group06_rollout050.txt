```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact coordinate descent with analytical 1D optimization
2. Softmax-smoothed subgradient descent with adaptive temperature
3. Nesterov-accelerated gradient steps
4. Multi-start with diverse initializations and perturbations
5. Multi-length exploration
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def exact_coordinate_descent(a, deadline, max_passes=50):
    """
    Exact coordinate descent: for each coordinate, find the optimal value
    using analytical 1D optimization.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        order = np.random.permutation(n)
        b = fast_conv(a)
        S = a.sum()
        improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            ai = a[i]
            S = a.sum()
            b = fast_conv(a)
            
            # Generate candidate delta values
            # For each k where a[k-i] > 0: delta_k = b[k]/a[k-i] - S
            # For k=2i: delta_2i = b[2i]/S
            # Boundary: delta = -ai
            # Current: delta = 0
            
            candidates = [0.0, -ai]
            
            # Valid range for k when changing a[i]:
            # k ranges from i to i+n-1 (since k-i must be in [0,n-1])
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            
            for k in range(k_lo, k_hi + 1):
                k_minus_i = k - i
                if k == 2 * i:
                    # Quadratic case: delta = b[2i] / S
                    if S > 1e-15:
                        candidates.append(b[k] / S)
                else:
                    # Linear case: delta = b[k] / a[k-i] - S (if a[k-i] > 0)
                    if a[k_minus_i] > 1e-15:
                        candidates.append(b[k] / a[k_minus_i] - S)
            
            # Also add some heuristic candidates
            candidates.append(ai * 0.5)
            candidates.append(ai * 2.0)
            candidates.append(0.01)
            candidates.append(0.1)
            
            # Filter valid candidates (delta > -ai, i.e., new value > 0)
            valid_candidates = [d for d in candidates if d > -ai - 1e-15]
            
            best_delta = 0.0
            best_score = compute_J(a)
            
            for delta in valid_candidates:
                new_val = ai + delta
                if new_val < -1e-15:
                    continue
                new_val = max(new_val, 0.0)
                delta_actual = new_val - ai
                S_new = S + delta_actual
                if S_new < 0.01:
                    continue
                
                # Compute new b values (only affected range changes)
                # b_new[k] = b[k] + 2*delta*a[k-i] + delta^2*I(k=2i)
                new_b_val = 0.0
                for k in range(k_lo, k_hi + 1):
                    k_minus_i = k - i
                    if k == 2 * i:
                        bv = b[k] + 2 * delta_actual * a[k_minus_i] + delta_actual ** 2
                    else:
                        bv = b[k] + 2 * delta_actual * a[k_minus_i]
                    if bv > new_b_val:
                        new_b_val = bv
                
                # Check outside range
                if k_lo > 0:
                    out_max = b[:k_lo].max()
                else:
                    out_max = 0.0
                if k_hi < 2 * n - 2:
                    out_max = max(out_max, b[k_hi + 1:].max())
                else:
                    out_max = max(out_max, 0.0)
                
                max_b_val = max(new_b_val, out_max)
                score = 2 * n * max_b_val / (S_new * S_new)
                
                if score < best_score:
                    best_score = score
                    best_delta = delta_actual
            
            if best_delta != 0.0:
                a[i] = max(ai + best_delta, 0.0)
                improved = True
        
        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a

def softmax_subgradient_descent(a, deadline, max_iters=10000, tau_start=50, tau_end=5000):
    """
    Subgradient descent with softmax smoothing of the max.
    Uses Nesterov acceleration and adaptive step size.
    """
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    y = a.copy()  # Nesterov position
    v = np.zeros(n)  # Momentum
    mu = 1.0  # Momentum parameter
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Compute convolution at y
        b = fast_conv(y)
        S = y.sum()
        max_b = b.max()
        J = 2 * n * max_b / (S * S)
        
        if J < best_J:
            best_J = J
            best_a = y.copy()

            if k_hi < 2 * n - 2:
                out_max = max(out_max, b[k_hi+1:].max())
            
            max_total = np.maximum(max_aff, out_max)
            scores = 2.0 * n * max_total / (S_new ** 2)

            if scores[best_idx] < 2.0 * n * b.max() / (S * S) - 1e-14:
                a[i] = max(ai + cands[best_idx], 0.0)
                improved = True
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J - 1e-15:
                best_J = J
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a

def lp_mirror_descent_fast(a, deadline, max_iters=500):
    """LP mirror descent with fast line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Build LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0
        # A[k,j] = a[k-j] if 0<=k-j<n
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 30, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                g_sum = g.sum()
                if g_sum > 1e-15:
                    g = g / g_sum
                    
                    # Line search: minimize J((1-t)*a + t*g)
                    # J(t) = 2n * max_k ((1-t)^2*b_aa[k] + 2t(1-t)*b_ag[k] + t^2*b_gg[k]) / ((1-t)*S_a + t*S_g)^2
                    # Since a and g are both normalized (sum=1), denominator is always 1
                    # So J(t) = 2n * max_k (quadratic in t)
                    b_ag = cross_conv(a, g)
                    b_gg = fast_conv(g)
                    
                    # Find t* that minimizes max_k of the quadratic
                    # Use golden section or similar
                    from scipy.optimize import minimize_scalar
                    def obj(t):
                        b = (1-t)**2 * b + 2*t*(1-t) * b_ag + t**2 * b_gg
                        return b.max()
                    
                    result = minimize_scalar(obj, bounds=(0, 1), method='bounded', options={'xatol': 1e-10})
                    t_opt = result.x
                    
                    a_new = (1 - t_opt) * a + t_opt * g
                    a_new = np.maximum(a_new, 0)
                    s = a_new.sum()
                    if s > 0.01:
                        a_new = a_new / s
                        J_new = compute_J(a_new)
                        if J_new < best_J - 1e-14:
                            best_J = J_new
                            best_a = a_new.copy()
                            a = a_new
        except Exception:
            pass
    
    return best_a

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Vectorized exact coordinate descent (primary, very fast per pass)
2. Perturbation-and-restart strategy for escaping local minima
3. LP mirror descent for diverse basins
4. Multi-length and multi-init exploration
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def vectorized_cd(a, deadline, max_passes=50000, stagnation_limit=200):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    stagnant = 0
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        S = a.sum()
        cur_M = b.max()
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            ai = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            c = a[:k_hi - i + 1]
            b_aff = b[k_lo:k_hi + 1]
            has_2i = (2 * i >= k_lo) and (2 * i <= k_hi)
            idx_2i = 2 * i - k_lo if has_2i else -1
            # Build candidates
            cands = np.array([0.0, max(-ai + 1e-8, -ai * 0.5)])
            nz = c > 1e-14
            if np.any(nz):
                cands = np.concatenate([cands, b_aff[nz] / c[nz] - S])
            if has_2i:
                cands = np.concatenate([cands, np.array([b_aff[idx_2i] / S])])
            cands = np.concatenate([cands, np.array([ai * 0.5 - ai, ai * 2.0 - ai])])
            cands = cands[cands > -ai - 1e-12]
            if len(cands) == 0:
                continue
            S_new = S + cands
            valid = S_new > 0.01
            if not np.all(valid):
                cands = cands[valid]
                S_new = S[valid] + cands[valid]
            if len(cands) == 0:
                continue
            b_new = b_aff[None, :] + 2.0 * cands[:, None] * c[None, :]
            if has_2i:
                b_new[:, idx_2i] += cands ** 2
            max_aff = b_new.max(axis=1)
            out_max = 0.0
            if k_lo > 0:
                out_max = b[:k_lo].max()
            if k_hi < 2 * n - 2:
                out_max = max(out_max, b[k_hi + 1:].max())
            max_total = np.maximum(max_aff, out_max)
            scores = 2.0 * n * max_total / (S_new ** 2)
            best_idx = np.argmin(scores)
            cur_score = 2.0 * n * cur_M / (S * S)
            if scores[best_idx] < cur_score * (1 - 1e-12):
                a[i] = max(ai + cands[best_idx], 0.0)
                improved = True
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J - 1e-14:
                best_J = J
                best_a = a.copy()
                stagnant = 0
            else:
                if not improved:
                    stagnant += 1
        if stagnant >= stagnation_limit:
            break
    return best_a


def cd_with_perturb(a, deadline, max_passes=50000, perturb_every=500, noise_scale=0.05):
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    passes_done = 0
    while time.monotonic() < deadline:
        a_cd = vectorized_cd(a, deadline, max_passes=max_passes, stagnation_limit=100)
        passes_done += 1
        J = compute_J(a_cd)
        if J < best_J - 1e-14:
            best_J = J
            best_a = a_cd.copy()
        a = a_cd
        # Check if we should perturb
        if passes_done % perturb_every == 0 or compute_J(a) == best_J:
            a_pert = a + np.random.normal(0, noise_scale, n)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            a = a_pert
        if time.monotonic() > deadline:
            break
    return best_a
