"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smooth gradient descent with Nesterov acceleration
2. LP mirror descent for precise local refinement
3. Diverse asymmetric initializations (key insight: asymmetric sequences minimize max conv)
4. Multi-length exploration (256-4096)
5. Genetic crossover between top solutions
6. Active-set LP acceleration for larger sequences

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad_fft(a, tau):
    """Compute gradient of smooth max using FFT-based cross-correlation."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = np.clip(tau * (b - b_max), -60, 60)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()

    # grad[j] = sum_{m=0}^{n-1} p[j+m] * a[m]
    # Use FFT cross-correlation
    size = 1
    while size < 3 * n:
        size <<= 1
    Fa = np.fft.rfft(a, size)
    Fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(Fa * np.conj(Fp), size)
    # grad[j] = cross[(size - j) % size]
    indices = (size - np.arange(n)) % size
    grad = cross[indices]
    return grad, p, b, b_max


def smooth_max_grad_loop(a, tau):
    """Compute gradient of smooth max using direct loop (for small n)."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = np.clip(tau * (b - b_max), -60, 60)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_arr = np.arange(lo, hi + 1)
        grad[j_arr] += pk * a[k - j_arr]
    return grad, p, b, b_max


def smooth_max_grad(a, tau):
    n = len(a)
    if n <= 1024:
        return smooth_max_grad_loop(a, tau)
    else:
        return smooth_max_grad_fft(a, tau)


def run_smooth_nesterov(a, deadline, max_iters=5000, tau_start=0.5, tau_end=300.0,
                        lr_init=0.5, lr_end=0.002, momentum=0.9):
    """Nesterov-accelerated smooth gradient descent with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    y = a.copy()
    t_k = 1.0

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        # Exponential tau schedule with faster ramp
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        # Learning rate decay
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1.0 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        # Nesterov step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        # Gradient step on y (which is ahead of a)
        a_new = np.maximum(y - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s

        # Update y with momentum
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y /= sy
        a = a_new

    # Final check
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_a = a.copy()
        best_M = M_final
    return best_a


def build_A_sparse(n, a, active_only=False, M=None, frac=0.15):
    """Build the constraint matrix for the LP. If active_only, only include
    constraints where conv(a,a) is close to M."""
    rows = []
    cols = []
    vals = []
    if active_only and M is not None:
        b = fast_conv(a)
        threshold = M * (1.0 - frac)
        active_ks = np.where(b >= threshold)[0]
    else:
        active_ks = np.arange(2 * n - 1)

    for k in active_ks:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k if not active_only else list(active_ks).index(k))
            cols.append(j)
            vals.append(a[k - j])

    if active_only:
        shape = (len(active_ks), n)
    else:
        shape = (2 * n - 1, n)
    return csr_matrix((vals, (rows, cols)), shape=shape), active_ks


def build_A_sparse_fast(n, a):
    """Fast construction of A matrix using vectorized approach."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        vals.extend(a[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=8, active_frac=0.0):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(a)
    try:
        if active_frac > 0:
            b_conv = fast_conv(a)
            threshold = M * (1.0 - active_frac)
            active_ks = np.where(b_conv >= threshold)[0]
            if len(active_ks) < 5:
                active_ks = np.arange(2 * n - 1)
            # Build reduced matrix
            rows = []
            cols = []
            vals = []
            idx_map = {}
            for idx, k in enumerate(active_ks):
                j_min = max(0, k - n + 1)
                j_max = min(n - 1, k)
                for j in range(j_min, j_max + 1):
                    rows.append(idx)
                    cols.append(j)
                    vals.append(a[k - j])
            A = csr_matrix((vals, (rows, cols)), shape=(len(active_ks), n))
            b_ub = np.full(len(active_ks), M)
        else:
            A = build_A_sparse_fast(n, a)
            b_ub = np.full(2 * n - 1, M)

        c = -np.ones(n)
        bounds = [(0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=300, time_per_lp=5.0, active_frac=0.0):
    """LP mirror descent: iteratively compute best response and blend."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        # Adapt time per LP based on remaining time
        remaining = deadline - time.monotonic()
        t_lp = min(time_per_lp, remaining / 3.0)
        if t_lp < 1.0:
            break

        g = lp_best_response(a, M, time_limit=t_lp, active_frac=active_frac)
        if g is None:
            # Try with active constraints
            g = lp_best_response(a, M, time_limit=t_lp, active_frac=0.3)
            if g is None:
                break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        # Try a range of step sizes
        for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.06, 0.03, 0.01]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = float(fast_conv(a_t).max())
            if M_t < best_M * (1.0 - 1e-14):
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 15:
                break

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_asymmetric_inits(n, rng, parent=None):
    """Generate diverse asymmetric initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Linear ramp (strongly asymmetric)
    inits.append(np.maximum(t + 0.1, 0.01))
    # Quadratic ramp
    inits.append(np.maximum(t**2 + 0.05, 0.01))
    # Exponential ramp
    inits.append(np.maximum(np.exp(2 * t) / np.exp(2.0) + 0.05, 0.01))
    # Reverse exponential
    inits.append(np.maximum(np.exp(2 * (1 - t)) / np.exp(2.0) + 0.05, 0.01))
    # Skewed triangle (peak at 1/3)
    tri = np.minimum(t / (1/3), (1 - t) / (2/3))
    inits.append(np.maximum(tri + 0.05, 0.01))
    # Skewed triangle (peak at 2/3)
    tri2 = np.minimum(t / (2/3), (1 - t) / (1/3))
    inits.append(np.maximum(tri2 + 0.05, 0.01))
    # Power law
    inits.append(np.maximum(t**0.5 + 0.1, 0.01))
    # Very skewed: most mass at one end
    inits.append(np.maximum(0.1 + 0.9 * t**4, 0.01))
    inits.append(np.maximum(0.1 + 0.9 * (1 - t)**4, 0.01))

    # Random Fourier (random phases)
    for _ in range(3):
        freqs = np.arange(n // 2 + 1)
        phases = rng.uniform(0, 2 * np.pi, n // 2 + 1)
        amps = np.ones(n // 2 + 1) / np.sqrt(freqs + 1)
        amps[0] = 1.0
        spectrum = amps * np.exp(1j * phases)
        sig = np.fft.irfft(spectrum * np.sqrt(n), n)
        sig = np.maximum(sig, 0)
        s = sig.sum()
        if s > 1e-6:
            sig /= s
            inits.append(sig + 0.01)

    # Smooth random
    k = max(3, n // 32)
    kern = np.ones(k) / k
    sig = np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same'), 0.01)
    inits.append(sig)

    # Cosine-based asymmetric
    inits.append(np.maximum(0.3 + 0.5 * np.cos(np.pi * t - np.pi / 4), 0.01))
    inits.append(np.maximum(0.2 + 0.6 * np.cos(np.pi * (t - 0.3)), 0.01))

    # Parent-based
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.005))
        # Flip parent (exploits asymmetry)
        inits.append(p[::-1].copy())
        inits.append(np.maximum(p[::-1] + rng.normal(0, 0.02, n), 0.005))

    return inits


def run_subgradient(a, deadline, max_iters=2000, lr_const=0.3):
    """Subgradient descent with Polyak averaging."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    running = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()

        # Active set: positions within 1% of max
        tol = 0.01 * M
        active = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            j_arr = np.arange(lo, hi + 1)
            grad[j_arr] += a[k - j_arr]

        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad /= gn
            lr = lr_const / np.sqrt(it + 1)
            a_new = np.maximum(a - lr * grad, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new /= s
                a = a_new

        running += a
        count += 1

    if count > 50:
        avg = running / count
        avg = np.maximum(avg, 0.0)
        s = avg.sum()
        if s > 1e-10:
            avg /= s
            M_avg = float(fast_conv(avg).max())
            if M_avg < best_M:
                best_a = avg
    return best_a


def crossover_mutate(a1, a2, n, rng, mu_scale=0.02):
    """Crossover two sequences (resampled to length n) with mutation."""
    a1r = resample_seq(a1, n)
    a2r = resample_seq(a2, n)
    # Uniform crossover
    mask = rng.random(n) < 0.5
    child = np.where(mask, a1r, a2r)
    child = np.maximum(child, 0.001)
    # Mutation
    child += rng.normal(0, mu_scale, n)
    child = np.maximum(child, 0.001)
    child /= child.sum()
    return child

    print(f"PHASE 4: Genetic Crossover", flush=True)
    print(f"{'='*50}", flush=True)

    # Collect diverse solutions
    pool = []
    if best.seq is not None:
        pool.append(best.seq)
    for sol in lp_solutions:
        if sol is not None and sol.size > 0:
            pool.append(sol)
    # Add parent
    if parent is not None:
        pool.append(parent)
    # Add some from best_per_length
    for n, a in list(best_per_length.items())[:4]:
        pool.append(a)

    # Remove duplicates (approximately)
    unique_pool = [pool[0]] if pool else []
    for a in pool[1:]:
        if not any(np.allclose(a, u, atol=0.01) for u in unique_pool):
            unique_pool.append(a)

    print(f"  Pool size: {len(unique_pool)}", flush=True)

    n_crossover_attempts = 8
    for i in range(n_crossover_attempts):
        if time.monotonic() > deadline - 50:
            break
        if len(unique_pool) < 2:
            break

        # Pick two diverse parents
        idx1 = min(i, len(unique_pool) - 1)
        idx2 = (i + 1) % len(unique_pool)
        if idx1 == idx2 and len(unique_pool) > 2:
            idx2 = (idx2 + 1) % len(unique_pool)

        a1 = unique_pool[idx1]
        a2 = unique_pool[idx2]

        # Try crossover at several lengths
        for n_try in [512, 1024, 1536, 2048]:
            if time.monotonic() > deadline - 40:
                break
            child = crossover_mutate(a1, a2, n_try, rng, mu_scale=0.01)
            # Quick polish
            child = run_smooth_nesterov(child, min(time.monotonic() + 25, deadline - 30),
                                        max_iters=500, tau_start=2.0, tau_end=100.0,
                                        lr_init=0.2, lr_end=0.01)
            best.update(child, f"xover {i} n={n_try}")
            # Quick LP if n small enough
            if n_try <= 1024 and time.monotonic() < deadline - 20:
                a_lp = run_lp_mirror(child, min(time.monotonic() + 15, deadline - 20),
                                     max_iters=30, time_per_lp=3.0)
                best.update(a_lp, f"xover_lp {i} n={n_try}")

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p3_start + time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Length refinement around best ===
    if best.seq is not None and time.monotonic() < deadline - 30:
        p5_start = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"PHASE 5: Length Refinement", flush=True)
        print(f"{'='*50}", flush=True)

        n_best = best.seq.size
        for delta in [-64, -32, 32, 64, 128, 256]:
            if time.monotonic() > deadline - 20:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 4096:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            a_ext = run_smooth_nesterov(a_ext, min(time.monotonic() + 15, deadline - 15),
                                        max_iters=400, tau_start=3.0, tau_end=100.0,
                                        lr_init=0.2, lr_end=0.01)
            best.update(a_ext, f"lenref n={n_try}")
        best.report("Phase 5")
        print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === FINAL: LP polish on absolute best ===
    if best.seq is not None and time.monotonic() < deadline - 5:
        n_final = best.seq.size
        if n_final <= 2048:
            est_lp_time = max(2.0, 0.000005 * n_final ** 1.8)
            max_iters = max(5, int((deadline - time.monotonic()) / est_lp_time))
            print(f"\nFinal LP: n={n_final}, max_iters={max_iters}", flush=True)
            a_final = run_lp_mirror(best.seq, deadline, max_iters=max_iters,
                                    time_per_lp=est_lp_time)
            best.update(a_final, "FINAL_LP")
        else:
            a_final = run_smooth_nesterov(best.seq, deadline, max_iters=1000,
                                          tau_start=5.0, tau_end=500.0,
                                          lr_init=0.1, lr_end=0.001)
            best.update(a_final, "FINAL_SMOOTH")

    # Fallback
    if best.seq is None:
        best.seq = np.linspace(0.1, 1.0, 512)
        best.update(best.seq, "fallback")

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
