def lp_best_response_cvxpy(a, M):
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    # Constraints: (a*g)[k] <= M for all k
    # (a*g)[k] = sum_j a[k-j] * g[j] for valid j
    # This is a matrix-vector product with a Toeplitz-like matrix

    # A[k, j] = a[k-j] if 0 <= k-j < n
    # Using scipy sparse
    from scipy.sparse import csr_matrix
    rows, cols, vals = [], [], []
    for k in range(2*n-1):
        j_min = max(0, k-n+1)
        j_max = min(n-1, k)
        for j in range(j_min, j_max+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [A @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.PDLP)  # or ECOS, or MOSEK
    if g.value is not None:
        return np.maximum(g.value, 0)
    return None

A[k,j] = a[k-j] if 0 <= k-j < n, else 0.

For a fixed column j:
- k jellyfish from j to j+n-1 (so that k-j ranges from 0 to n-1)
- A[k,j] = a[k-j]

So for column j, the non-zero entries are at rows j, j+1, ..., j+n-1, with values a[0], a[1], ..., a[n-1].

But we only have rows 0 to 2n-2. So:
- For j = 0: rows 0 to n-1, values a[0] to a[n-1]
- For j = 1: rows 1 to n, values a[0] to a[n-1]
- ...
- For j = n-1: rows n-1 to 2n-2, values a[0] to a[n-1]

So: A[j:j+n, j] = a for all j. But we need to clip to the row range [0, 2n-2].

For j=0: A[0:n, 0] = a ✓ (rows 0 to n-1)
For j=n-1: A[n-1:2n-1, n-1] = a ✓ (rows n-1 to 2n-2)

Since j ranges from 0 to n-1, and j+n-1 ranges from n-1 to 2n-2, all rows are within [0, 2n-2]. ✓

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_constraint_matrix(a):
    """Build dense constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    j = np.arange(n)
    k = j[:, None] + np.arange(n)[None, :]
    A = np.zeros((2 * n - 1, n))
    A[k.flatten(), np.repeat(np.arange(n), n)] = np.tile(a, n)
    return A

def lp_best_response(a, M, time_limit=5.0):
    """LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    A = build_constraint_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    # Try HiGHS first
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    
    # Fallback to CVXPY
    try:
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        for solver in [cp.HIGHS, cp.PDLP, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None:
                    return np.maximum(g.value, 0)
            except Exception:
                continue
    except Exception:
        pass
    
    return None

def lp_mirror_descent(a, deadline, verbose=True):

    return float(2 * len(a) * b.max() / (s * s))


def build_constraint_matrix(a):
    """Build dense constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n, else 0."""
    n = len(a)
    j = np.arange(n)
    k = j[:, None] + np.arange(n)[None, :]
    A = np.zeros((2 * n - 1, n))
    A[k.flatten(), np.repeat(np.arange(n), n)] = np.tile(a, n)
    return A


def lp_best_response(a, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_constraint_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    try:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        A_sp = A
        constraints = [A_sp @ g <= M]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        for solver in [cp.PDLP, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if g.value is not None:
                    return np.maximum(np.asarray(g.value), 0)
            except Exception:
                continue
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, verbose=True, name=""):
    """LP mirror descent with adaptive step sizes and convergence detection."""
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_conv(a)
    M = float(conv.max())
    best_a = a.copy()
    best_M = M
    no_improve = 0
    max_no_improve = 80
    t_start = time.monotonic()
    last_lp_time = 0.1
    it = 0
    prev_relative_improve = 1.0

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < last_lp_time * 2.0:
            break

        conv = fast_conv(a)
        M = float(conv.max())

        t0 = time.monotonic()
        g = lp_best_response(a, M, time_limit=max(1.0, remaining * 0.25))
        lp_time = time.monotonic() - t0
        last_lp_time = 0.6 * last_lp_time + 0.4 * lp_time

        if g is None or g.sum() < 1e-12:
            break

        g = np.maximum(g, 0.0)
        g = g / g.sum()
        it += 1

        # Adaptive step size: start larger, shrink over time
        base_t = min(0.6, max(0.03, 0.8 / (1.0 + 0.05 * it)))
        candidates_t = [base_t, base_t * 0.7, base_t * 0.5, base_t * 0.3, 0.05, 0.02, 0.01]

        improved = False
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial
            M_trial = float(fast_conv(a_trial).max())

            if M_trial < M - 1e-13:
                relative_improve = (M - M_trial) / M
                a = a_trial
                M = M_trial
                if M < best_M - 1e-15:
                    best_M = M
                    best_a = a.copy()
                no_improve = 0
                improved = True
                if verbose and it % 20 == 0:
                    elapsed = time.monotonic() - t_start
                    print(f"    {name} LP-MD it={it} M={M:.10f} best={best_M:.10f} t={t:.3f} ({elapsed:.0f}s)", flush=True)
                break

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
            # If relative improvement is tiny, we've likely converged
            if it > 20:
                if relative_improve < 1e-7 and no_improve > 20:
                    break

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    {name} LP-MD it={it} M={M:.10f} best={best_M:.10f} ({elapsed:.0f}s) no_imp={no_improve}", flush=True)

    return best_a


def coord_descent_fast(a, deadline, max_sweeps=5000):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        sweep_improved = False

        for i in range(n):
            old = a[i]
            if old < 1e-12:
                candidates = [0.0, 1e-4, 1e-3, 1e-2, 0.05, 0.1, 0.2]
            else:
                ratios = np.array([0.0, 0.1, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0])
                candidates = ratios * old
                candidates = candidates[np.abs(candidates - old) > 1e-4 * max(old, 1e-3)]

            if len(candidates) == 0:
                continue

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-13:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a = a / a.sum()
    return a


def subgradient_polish(a, deadline, max_iters=20000):
    """Subgradient descent with averaging and Nesterov momentum."""
    a = np.maximum(a.astype(np.float64), 0).copy()
    if a.sum() < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    y = a.copy()
    running_sum = np.zeros(n)
    count = 0
    t_start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t_it = (it + 1) / max_iters
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t_it * t_it)) / 2.0

        conv_y = fast_conv(y)
        M_y = float(conv_y.max())
        k_star = int(np.argmax(conv_y))

        # Subgradient: grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        j = np.arange(lo, hi + 1)
        grad[j] = 2.0 * a[k_star - j]

        # Nesterov update
        x_new = y + (beta - 1.0) / (it + 2.0) * (y - a)
        lr = 1.0 / np.sqrt(it + 1.0)
        x_new = x_new - lr * grad
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-10:
            x_new = a.copy()
        else:
            x_new = x_new / s

        running_sum += x_new
        count += 1
        y = x_new
        a = x_new

        M_new = float(fast_conv(a).max())
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

        if count % 5000 == 0:
            a_avg = running_sum / count
            a_avg = np.maximum(a_avg, 0)
            if a_avg.sum() > 1e-10:
                a_avg = a_avg / a_avg.sum()
                M_avg = float(fast_conv(a_avg).max())
                if M_avg < best_M:
                    best_M = M_avg
                    best_a = a_avg.copy()

    # Final averaging
    if count > 50:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0)
        if a_avg.sum() > 1e-10:
            a_avg = a_avg / a_avg.sum()
            M_avg = float(fast_conv(a_avg).max())
            if M_avg < best_M:
                best_a = a_avg

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def eval_and_track(a, best_state, label=""):
    """Evaluate sequence and update best state. Returns (best_seq, best_score)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0 or a.sum() < 0.01:
        return best_state
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state[1]:
        best_state = (a.copy(), score)
        print(f"  ** {label} J={score:.10f} n={a.size}", flush=True)
    return best_state


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: Coordinate descent initialization (120s)
    Phase 2: LP Mirror Descent main optimization (700s)
    Phase 3: Try longer sequences with LP-MD (350s)
    Phase 4: Final polishing (remaining time)
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** {label}: J={score:.10f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            try_update(parent, "parent")
            print(f"Parent loaded: n={parent.size}, J={best_score:.10f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512
        try_update(parent, "default")

    # === PHASE 1: Coordinate descent (120s) ===
    p1_deadline = min(t_start + 120, deadline - 1100)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Coord Descent ({p1_deadline - time.monotonic():.0f}s budget)", flush=True)
    print(f"{'='*60}", flush=True)

    n_base = len(parent)
    a_cd = coord_descent_fast(parent, p1_deadline, max_sweeps=5000)
    try_update(a_cd, "CD")

    # Perturbed initialization
    a_pert = np.maximum(parent + rng.normal(0, 0.03, n_base), 0)
    a_pert = a_pert / a_pert.sum()
    if time.monotonic() < p1_deadline - 20:
        a_cd2 = coord_descent_fast(a_pert, p1_deadline - 5, max_sweeps=2000)
        try_update(a_cd2, "CD-pert")

    print(f"Phase 1 done: best J={best_score:.10f}", flush=True)

    # === PHASE 2: LP Mirror Descent (main, ~700s) ===
    p2_deadline = min(t_start + 820, deadline - 350)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent ({p2_deadline - time.monotonic():.0f}s budget)", flush=True)
    print(f"{'='*60}", flush=True)

    a_current = best_seq.copy()
    a_current = np.maximum(a_current, 0)
    a_current = a_current / a_current.sum()

    a_lp = lp_mirror_descent(a_current, p2_deadline, verbose=True, name="P2")
    try_update(a_lp, "LP-MD")

    # Check if we should do another round
    if time.monotonic() < deadline - 350:
        print("  Running second LP-MD round...", flush=True)
        a_lp2 = lp_mirror_descent(a_lp, deadline - 350, verbose=True, name="P2b")
        try_update(a_lp2, "LP-MD-2")

    print(f"Phase 2 done: best J={best_score:.10f}", flush=True)

    # === PHASE 3: Longer sequences (350s) ===
    p3_deadline = deadline - 120
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Longer Sequences ({p3_deadline - time.monotonic():.0f}s budget)", flush=True)
    print(f"{'='*60}", flush=True)

    base_n = len(best_seq)
    target_lengths = [l for l in [640, 768, 1024, 1280, 1536] if l > base_n and l <= 2048]

    for new_n in target_lengths:
        if time.monotonic() > p3_deadline - 80:
            break

        print(f"  Trying n={new_n}...", flush=True)
        a_ext = resample_seq(best_seq, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        # Quick CD
        cd_deadline = min(time.monotonic() + 40, p3_deadline - 40)
        a_cd_ext = coord_descent_fast(a_ext, cd_deadline, max_sweeps=1000)
        try_update(a_cd_ext, f"CD n={new_n}")

        # LP-MD on the result
        if time.monotonic() < p3_deadline - 30:
            a_lp_ext = lp_mirror_descent(a_cd_ext, p3_deadline - 30, verbose=False, name=f"n={new_n}")
            try_update(a_lp_ext, f"LP n={new_n}")

    print(f"Phase 3 done: best J={best_score:.10f}", flush=True)

    # === PHASE 4: Final Polish (120s) ===
    p4_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final Polish ({p4_deadline - time.monotonic():.0f}s budget)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None and time.monotonic() < p4_deadline - 10:
        # Subgradient polish
        sg_deadline = min(time.monotonic() + 50, p4_deadline - 40)
        a_sg = subgradient_polish(best_seq, sg_deadline)
        try_update(a_sg, "SG")

        # Final coordinate descent
        if time.monotonic() < p4_deadline - 10:
            a_fin = coord_descent_fast(best_seq, min(time.monotonic() + 40, p4_deadline - 5), max_sweeps=2000)
            try_update(a_fin, "Final-CD")

        # One more LP-MD if time
        if time.monotonic() < p4_deadline - 5:
            a_fin2 = lp_mirror_descent(best_seq, p4_deadline, verbose=False, name="Final")
            try_update(a_fin2, "Final-LP")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_seq.tolist()))

    total_time = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.10f} n={len(best_seq)} time={total_time:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
