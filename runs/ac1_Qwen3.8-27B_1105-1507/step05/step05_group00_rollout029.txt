"""
Optimized peak auto-correlation minimization using soft-max L-BFGS and LP alternating.
Uses temperature-annealing on the soft-max smoothing, combined with block-coordinate
LP solves for the min-peak cross-correlation problem.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def cross_corr(a, w):
    """Compute r[m] = sum_j a[j] * w[j+m] for m=0..n-1 using FFT."""
    n = len(a)
    # r[m] = conv_full[2n-2-m] where conv_full = a * w[::-1]
    b_rev = w[::-1]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    A = rfft(a, size)
    B = rfft(b_rev, size)
    conv_full = irfft(A * B, size)[:3 * n - 2]
    # r[m] = conv_full[2n-2-m] for m=0..n-1
    # So r = conv_full[n-1 : 2n-1][::-1]
    r = conv_full[n-1:2*n-1][::-1]
    return r


def build_toeplitz(a):
    n = len(a)
    T = np.zeros((2*n-1, n))
    for j in range(n):
        k_end = min(j + n, 2*n-1)
        T[j:k_end, j] = a[:k_end-j]
    return T


def solve_minpeak_lp(a, time_limit=10.0):
    """Solve: min M s.t. (a*g)[k] <= M for k=0..2n-2, sum(g)=1, g>=0."""
    n = len(a)
    T = build_toeplitz(a)
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize M
    
    # Inequality: T@g - M <= 0
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None


def lbfqs_smoothed(a, tau, max_iter=200, time_budget=30.0):
    """
    Run L-BFGS-B on the smoothed objective with given temperature tau.
    a is normalized (sum=1). Returns optimized a.
    """
    n = len(a)
    z0 = np.log(a + 1e-30)  # inverse softmax (approximate)
    
    start = time.monotonic()
    
    def fun_and_grad(z):
        a_local = softmax_fn(z)
        conv = compute_conv(a_local)
        
        # Soft-max
        cmax = np.max(conv)
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500, 500)
        e = np.exp(shifted)
        w = e / np.sum(e)
        smooth_max = cmax + tau * np.log(np.sum(e))
        f = 2.0 * n * smooth_max
        
        # Gradient
        grad_a = 2.0 * cross_corr(a_local, w)
        grad_z = a_local * (grad_a - np.dot(grad_a, a_local))
        
        return f, grad_z
    
    try:
        res = minimize(fun_and_grad, z0, method='L-BFGS-B',
                      jac=True, options={'maxiter': max_iter, 'maxfun': 1000,
                                         'ftol': 1e-15, 'gtol': 1e-10})
        a_new = softmax_fn(res.x)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
        return a_new
    except Exception:
        return a.copy()


def lp_alternate(a, max_iters=50, time_budget=100.0, verbose=False):
    """
    LP alternating: alternate between optimizing g (given a) and a (given g).
    Each iteration solves two LPs.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Step 1: Given a, find best g
        g, M_g = solve_minpeak_lp(a, time_limit=5.0)
        if g is None:
            break
        
        # Step 2: Given g, find best a_new
        a_new, M_a = solve_minpeak_lp(g, time_limit=5.0)
        if a_new is None:
            # Just use g
            a_new = g
        
        # Evaluate
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 5 == 0:
                print(f"    LP alt it{it}: obj={best_obj:.8f}, M_a={M_a:.6f}")
        else:
            # Try blending
            for t in [0.3, 0.5, 0.7]:
                a_blend = (1-t) * best_a + t * a_new
                a_blend = np.maximum(a_blend, 0)
                s_b = np.sum(a_blend)
                if s_b < 1e-10:
                    continue
                a_blend = a_blend / s_b
                obj_blend = compute_obj(a_blend)
                if obj_blend < best_obj - 1e-12:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
                    a = a_blend
                    break
            else:
                # No improvement, try small perturbation
                if it > 10:
                    # Check if we're stuck
                    if it > 30:
                        break
    
    return best_a

temperature annealing, combined with LP block-coordinate alternating and
projected subgradient steps. Explores multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time

    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))

    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None


def lbfgs_smoothed(a, tau, max_iter=300):
    """
    Run L-BFGS-B on smoothed objective 2n*softmax(conv(a,a), tau)
    where a = softmax(z). Returns optimized a (sum=1).
    """
    n = len(a)
    z0 = np.log(a + 1e-30)

    def fun_and_grad(z):
        a_loc = softmax_fn(z)
        conv = compute_conv(a_loc)
        cmax = np.max(conv)
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500, 500)
        e = np.exp(shifted)
        w = e / np.sum(e)
        smooth_max = cmax + tau * np.log(np.sum(e))
        f = 2.0 * n * smooth_max
        grad_a = 2.0 * cross_corr_fft(a_loc, w)
        grad_z = a_loc * (grad_a - np.dot(grad_a, a_loc))
        return f, grad_z

    try:
        res = minimize(fun_and_grad, z0, method='L-BFGS-B',
                       jac=True, options={'maxiter': max_iter, 'maxfun': 2000,
                                          'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50})
        a_new = softmax_fn(res.x)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
        return a_new
    except Exception:
        return a.copy()


def anneal_lbfgs(a, tau0=None, tau1=None, n_levels=8, max_iter_per_level=200,
                 time_budget=200.0, verbose=False):
    """
    Temperature-annealed L-BFGS on smoothed objective.
    Starts with large tau (smooth), ends with small tau (sharp).
    """
    n = len(a)
    start = time.monotonic()
    a_local = a.copy()
    s = np.sum(a_local)
    if s < 1e-10:
        return a
    a_local = a_local / s

    best_a = a_local.copy()
    best_obj = compute_obj(a_local)

    if tau0 is None:
        conv = compute_conv(a_local)
        tau0 = max(float(np.max(conv)) / 3.0, 1e-6)
    if tau1 is None:
        tau1 = tau0 / (10.0 ** (n_levels - 1))

    taus = np.geomspace(tau0, tau1, n_levels)

    for li, tau in enumerate(taus):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        level_budget = min(remaining * 0.6, 30.0)

        a_local = lbfgs_smoothed(a_local, tau, max_iter=max_iter_per_level)
        obj = compute_obj(a_local)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_local.copy()
            if verbose:
                print(f"    Anneal level {li}: tau={tau:.6f}, obj={best_obj:.8f}")
        else:
            if verbose:
                print(f"    Anneal level {li}: tau={tau:.6f}, obj={obj:.8f} (no improvement)")

    return best_a


def lp_alternate_pure(a, max_iters=60, time_budget=100.0, verbose=False):
    """
    Pure LP block-coordinate alternating:
    - Given a, solve min M s.t. T(a)*g <= M, sum(g)=1, g>=0 -> g
    - Given g, solve min M s.t. T(g)*a <= M, sum(a)=1, a>=0 -> a_new
    Track best auto-correlation.
    """
    n = len(a)
    a_local = a.copy().astype(np.float64)
    s = np.sum(a_local)
    if s < 1e-10:
        return a
    a_local = a_local / s

    best_a = a_local.copy()
    best_obj = compute_obj(a_local)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break

        # Solve for g given a
        g, _ = solve_minpeak_lp(a_local, time_limit=5.0)
        if g is None:
            break

        # Solve for a_new given g
        a_new, _ = solve_minpeak_lp(g, time_limit=5.0)
        if a_new is None:
            a_new = g

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a_local = a_new
            stall = 0
            if verbose and it % 10 == 0:
                print(f"    LP pure it{it}: obj={best_obj:.8f}")
        else:
            stall += 1
            # Try blending with best
            improved_blend = False
            for t in [0.2, 0.4, 0.6]:
                a_blend = (1 - t) * best_a + t * a_new
                a_blend = np.maximum(a_blend, 0)
                s_b = np.sum(a_blend)
                if s_b < 1e-10:
                    continue
                a_blend = a_blend / s_b
                obj_b = compute_obj(a_blend)
                if obj_b < best_obj - 1e-13:
                    best_obj = obj_b
                    best_a = a_blend.copy()
                    a_local = a_blend
                    improved_blend = True
                    break
            if stall > 20 and not improved_blend:
                # Perturb to escape
                rng = np.random.default_rng(it * 31 + 7)
                a_pert = a_local * np.exp(rng.normal(0, 0.05, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    a_local = a_pert
                stall = 0
            elif stall > 40:
                break

    return best_a


def subgrad_project(a, max_iters=100, time_budget=20.0, verbose=False):
    """
    Projected subgradient with restart. Uses the smoothed subgradient
    (soft-max weights) and projects onto the simplex.
    """
    n = len(a)
    a_local = a.copy().astype(np.float64)
    s = np.sum(a_local)
    if s < 1e-10:
        return a
    a_local = a_local / s

    best_a = a_local.copy()
    best_obj = compute_obj(a_local)
    start = time.monotonic()

    # Simplex projection: project x onto {y: sum(y)=1, y>=0}
    def project_simplex(x, target_sum=1.0):
        n_x = len(x)
        if np.all(x <= 0):
            return np.ones(n_x) / n_x * target_sum
        v = np.sort(x)[::-1]
        cssv = np.cumsum(v)
        rho = np.nonzero(v * np.arange(1, n_x + 1) > (cssv - target_sum))[0][-1]
        theta = (cssv[rho] - target_sum) / (rho + 1.0)
        w = np.maximum(x - theta, 0)
        sw = np.sum(w)
        if sw < 1e-15:
            return np.ones(n_x) / n_x * target_sum
        return w / sw * target_sum

    tau = 0.01
    step = 0.5
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a_local)
        cmax = np.max(conv)
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500, 500)
        e = np.exp(shifted)
        w = e / np.sum(e)

        grad_a = 2.0 * cross_corr_fft(a_local, w)
        # Center gradient (project off the sum constraint)
        grad_centered = grad_a - np.sum(grad_a * a_local)

        a_local = a_local - step * grad_centered
        a_local = project_simplex(a_local, 1.0)

        obj = compute_obj(a_local)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_local.copy()

        # Adaptive step
        step = max(0.01, step * 0.99)
        # Anneal tau
        tau = max(0.001, tau * 0.97)

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    # Ensure non-negative
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_single(a, time_budget=100.0, verbose=False):
    """
    Full optimization pipeline for a single sequence:
    1. LP alternating
    2. L-BFGS annealing
    3. More LP alternating
    4. Subgradient polish
    """
    n = len(a)
    start = time.monotonic()
    a_local = a.copy().astype(np.float64)
    s = np.sum(a_local)
    if s < 1e-10:
        return a
    a_local = a_local / s

    best_a = a_local.copy()
    best_obj = compute_obj(a_local)

    # Round 1: LP alternating
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        alloc = min(remaining * 0.3, 60.0)
        a_new = lp_alternate_pure(a_local, max_iters=40,
                                  time_budget=alloc, verbose=verbose)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a_local = a_new
            if verbose:
                print(f"      R1 LP: {best_obj:.8f}")

    # Round 2: L-BFGS annealing
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        alloc = min(remaining * 0.3, 60.0)
        a_new = anneal_lbfgs(a_local, n_levels=6, max_iter_per_level=150,
                             time_budget=alloc, verbose=verbose)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a_local = a_new
            if verbose:
                print(f"      R2 LBF: {best_obj:.8f}")

    # Round 3: More LP alternating
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        alloc = min(remaining * 0.25, 40.0)
        a_new = lp_alternate_pure(a_local, max_iters=30,
                                  time_budget=alloc, verbose=verbose)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a_local = a_new
            if verbose:
                print(f"      R3 LP: {best_obj:.8f}")

    # Round 4: Subgradient polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.2, 20.0)
        a_new = subgrad_project(a_local, max_iters=200,
                                time_budget=alloc, verbose=verbose)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"      R4 SG: {best_obj:.8f}")

    # Round 5: Final LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        alloc = min(remaining * 0.3, 30.0)
        a_new = lp_alternate_pure(best_a, max_iters=20,
                                  time_budget=alloc, verbose=verbose)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"      R5 LP: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start optimization combining LP alternating,
    L-BFGS on smoothed max objective with temperature annealing,
    and projected subgradient methods.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n={n_parent}")

    if best_seq is None:
        best_seq = np.ones(500)
        best_score = compute_obj(best_seq)
        n_parent = 500

    a_main = best_seq.copy()
    s = np.sum(a_main)
    if s > 1e-10:
        a_main = a_main / s

    # Phase 1: Intensive optimization on parent length
    print(f"\n=== Phase 1: Optimize n={n_parent} (budget ~600s) ===")
    phase1_budget = min(600.0, (deadline - start_time) * 0.4)

    # 1a: LP alternating (long)
    remaining = deadline - start_time
    if remaining > 30:
        alloc = min(phase1_budget * 0.3, remaining * 0.4)
        print(f"  LP alternating ({alloc:.0f}s)...")
        a_new = lp_alternate_pure(a_main, max_iters=50,
                                  time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_main = a_new
        print(f"  After LP: {best_score:.8f}")

    # 1b: L-BFGS annealing
    remaining = deadline - start_time
    if remaining > 20:
        alloc = min(phase1_budget * 0.25, remaining * 0.3)
        print(f"  L-BFGS annealing ({alloc:.0f}s)...")
        a_new = anneal_lbfgs(a_main, n_levels=8, max_iter_per_level=250,
                             time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_main = a_new
        print(f"  After LBF: {best_score:.8f}")

    # 1c: More LP
    remaining = deadline - start_time
    if remaining > 20:
        alloc = min(phase1_budget * 0.2, remaining * 0.3)
        print(f"  LP round 2 ({alloc:.0f}s)...")
        a_new = lp_alternate_pure(a_main, max_iters=40,
                                  time_budget=alloc, verbose=True)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_main = a_new
        print(f"  After LP2: {best_score:.8f}")

    # 1d: Subgradient
    remaining = deadline - start_time
    if remaining > 15:
        alloc = min(30.0, remaining * 0.2)
        a_new = subgrad_project(a_main, max_iters=300, time_budget=alloc)
        obj = compute_obj(a_new)
        if obj < best_score:
            best_score = obj
            best_seq = a_new.copy()
            a_main = a_new
        print(f"  After SG: {best_score:.8f}")

    # 1e: Final LP + LBF
    remaining = deadline - start_time
    if remaining > 20:
        alloc = min(remaining * 0.3, 60.0)
        a_new, obj_new = optimize_single(a_main, time_budget=alloc, verbose=True)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a_main = a_new
        print(f"  Phase 1 final: {best_score:.8f}")

    # Phase 2: Try different n values
    print(f"\n=== Phase 2: Explore n values ===")
    n_candidates = [n_parent]
    for delta in [-100, -50, 50, 100, 150, 200]:
        n_try = n_parent + delta
        if 200 <= n_try <= 1200 and n_try not in n_candidates:
            n_candidates.append(n_try)
    # Also try some absolute values
    for n_try in [400, 500, 600, 800, 900, 1000]:
        if n_try not in n_candidates and 200 <= n_try <= 1200:
            n_candidates.append(n_try)

    phase2_budget = min(500.0, (deadline - start_time) * 0.35)
    per_n_budget = phase2_budget / max(len(n_candidates), 1)

    for n_try in sorted(n_candidates):
        remaining = deadline - start_time
        if remaining < 30:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(per_n_budget * 0.8, remaining * 0.4)
        if alloc < 10:
            continue

        print(f"  n={n_try} ({alloc:.0f}s)...")
        a_new, obj_new = optimize_single(a_try, time_budget=alloc, verbose=False)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  ** UPDATE: n={n_try}, obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_new:.8f}")

    # Phase 3: Final intensive polish on best
    print(f"\n=== Phase 3: Final Polish ===")
    remaining = deadline - start_time
    print(f"  Budget: {remaining:.0f}s, current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)
        a_final = best_seq.copy()
        s_f = np.sum(a_final)
        if s_f > 1e-10:
            a_final = a_final / s_f

        # Multiple rounds of optimization
        for rnd in range(5):
            remaining = deadline - start_time
            if remaining < 15:
                break
            alloc = min(remaining * 0.2, 40.0)

            # Alternate between LP and LBF
            if rnd % 2 == 0:
                a_new = lp_alternate_pure(a_final, max_iters=25,
                                          time_budget=alloc)
            else:
                a_new = anneal_lbfgs(a_final, n_levels=5,
                                     max_iter_per_level=100,
                                     time_budget=alloc)
            obj = compute_obj(a_new)
            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                a_final = a_new
                print(f"  Final {rnd}: {best_score:.8f}")

            # Subgradient micro-steps
            remaining = deadline - start_time
            if remaining > 10:
                a_sg = subgrad_project(a_final, max_iters=100,
                                       time_budget=min(10.0, remaining * 0.2))
                obj_sg = compute_obj(a_sg)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    a_final = a_sg
                    print(f"  Final {rnd} SG: {best_score:.8f}")

    # Phase 4: Try perturbations and re-optimize
    remaining = deadline - start_time
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Phase 4: Perturbation search ===")
        n_pb = len(best_seq)
        rng = np.random.default_rng(seed)

        for pert_idx in range(8):
            remaining = deadline - start_time
            if remaining < 20:
                break
            alloc = min(remaining * 0.15, 30.0)

            scale = 0.03 + 0.02 * pert_idx
            a_pert = best_seq * np.exp(rng.normal(0, scale, n_pb))
            a_pert = np.maximum(a_pert, 0)
            s_p = np.sum(a_pert)
            if s_p < 1e-10:
                continue
            a_pert = a_pert / s_p

            # Quick optimize
            a_new = lp_alternate_pure(a_pert, max_iters=15,
                                      time_budget=alloc * 0.6)
            obj = compute_obj(a_new)
            if obj > best_score:
                a_new = anneal_lbfgs(a_pert, n_levels=4,
                                     max_iter_per_level=80,
                                     time_budget=alloc * 0.4)
                obj = compute_obj(a_new)

            if obj < best_score:
                best_score = obj
                best_seq = a_new.copy()
                print(f"  Pert {pert_idx}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
