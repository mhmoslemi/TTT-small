```python
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS optimization
of a smooth max surrogate with softmax parameterization, combined with
LP-based refinement and multi-scale exploration.

Key improvements over parent:

2. Softmax parameterization naturally handles simplex constraint

4. Multi-start with diverse initializations

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
from scipy.signal import fftconvolve
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def smooth_max_and_grad_from_b(b, tau):
    """Compute smooth max value and weights from a vector b."""
    b_max = b.max()
    shifted = tau * (b - b_max)
    # For numerical stability, clip
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    log_sum = b_max + np.log(exp_shifted.sum()) / tau
    p = exp_shifted / exp_shifted.sum()
    return float(log_sum), p


def smooth_max_value_and_grad_a(a, tau):
    """Compute smooth max value and gradient w.r.t. a of the convolution."""
    n = len(a)
    b = fast_conv(a)
    val, p = smooth_max_and_grad_from_b(b, tau)
    
    # Gradient: grad_i = 2 * sum_k p_k * a_{k-i} (where k-i in [0,n-1])
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # For i in [lo, hi]: contribution is 2 * p[k] * a[k-i]
        # k-i ranges from k-lo to k-hi (decreasing)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    return val, grad


def objective_and_grad_b(b, tau):
    """Objective and gradient in softmax parameterization."""
    a = softmax(b)
    val, grad_a = smooth_max_value_and_grad_a(a, tau)
    # Chain rule through softmax:
    # d a_i / d b_j = a_i * (delta_{ij} - a_j)
    # grad_b_j = sum_i grad_a_i * a_i * (delta_{ij} - a_j)
    #           = a_j * (grad_a_j - sum_i grad_a_i * a_i)
    dot = np.dot(grad_a, a)
    grad_b = a * (grad_a - dot)
    return val, grad_b


def lbfgs_smooth_optimize(a0, deadline, tau_start=5.0, tau_end=50.0, 
                           verbose=True, max_total_iters=5000):
    """
    Optimize using L-BFGS with adaptive tau.
    a0 is the initial sequence (will be converted to softmax parameterization).
    """
    n = len(a0)
    a0 = np.maximum(a0, 1e-10).astype(np.float64)
    a0 = a0 / a0.sum()
    
    # Initialize b from a0: b_i = log(a_i)
    b = np.log(a0)
    b = b - b.mean()  # center for numerical stability
    
    best_a = a0.copy()
    best_J = compute_J(a0)
    best_smooth_val = smooth_max_value_and_grad_a(a0, tau_start)[0]
    
    total_iters = 0
    tau = tau_start
    
    # Progressive tau schedule
    n_phases = 3
    taus = [tau_start, (tau_start * tau_end) ** 0.5, tau_end]
    
    for phase in range(n_phases):
        if time.monotonic() > deadline:
            break
        tau = taus[phase]
        phase_deadline = deadline  # use full remaining time for each phase
        # Actually, let's allocate time proportionally
        # Phase 0: 40%, Phase 1: 35%, Phase 2: 25%
        
        if verbose:
            print(f"  L-BFGS phase {phase}: tau={tau:.1f}, n={n}", flush=True)
        
        # Define objective with fixed tau
        def obj_grad(b_vec):
            nonlocal total_iters
            total_iters += 1
            val, grad = objective_and_grad_b(b_vec, tau)
            # Add a small regularization to prevent b from going to infinity
            reg = 0.01 * np.dot(b_vec, b_vec)
            grad = grad + 0.02 * b_vec
            return val + reg, grad
        
        try:
            result = minimize(obj_grad, b, method='L-BFGS-B', 
                             jac=True,  # objective_and_grad_b returns (f, g)
                             options={
                                 'maxiter': 2000,
                                 'ftol': 1e-14,
                                 'gtol': 1e-10,
                                 'maxfun': 10000,
                                 'iprint': 0
                             })
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error: {e}", flush=True)
            break
        
        a_new = softmax(result.x)
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)
        
        if J_new < best_J:
            best_a = a_new.copy()
            best_J = J_new
            if verbose:
                print(f"    Phase {phase} result: J={J_new:.10f} (iters={total_iters})", flush=True)
        
        # Update b for next phase
        b = result.x - result.x.mean()
    
    return best_a, best_J


def build_lp_matrix(n, a, active_k=None):
    """Build sparse convolution matrix."""
    if active_k is None:
        active_k = range(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(list(active_k)) if not isinstance(active_k, range) else len(active_k), n))


def lp_mirror_step(a, M, top_frac=0.7, time_limit=5):
    """One step of LP mirror descent."""
    n = len(a)
    b = fast_conv(a)
    
    if top_frac < 1.0:
        threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
        active_k = np.where(b >= threshold)[0]
    else:
        active_k = np.arange(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, np.inf)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                # Verify
                ag = fast_cross_conv(a, g)
                if ag.max() <= M * (1.0 + 1e-4):
                    return g
    except Exception:
        pass
    return None


def lp_refine(a, deadline, verbose=True, max_no_improve=100):
    """LP mirror descent refinement."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_mirror_step(a, M, top_frac=0.7, time_limit=3)
        if g is None:
            g = lp_mirror_step(a, M, top_frac=1.0, time_limit=5)
        if g is None:
            break
        
        # Line search
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = (1.0 - t) * a + t * (g / g.sum())
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            
            if J_trial < best_J - 1e-12:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
            elif J_trial < compute_J(a) - 1e-12:
                a = a_trial
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
            if no_improve > max_no_improve:
                # Perturbation
                rng = np.random.default_rng(it * 12345)
                a = a * (1.0 + 0.1 * rng.normal(0, 1, n))
                a = np.maximum(a, 1e-6)
                a = a / a.sum()
                no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    LP iter {it}: J={best_J:.10f}", flush=True)
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        p = p / p.sum()
        inits.append(('parent', p))
        inits.append(('parent_noisy', np.maximum(p + rng.normal(0, 0.05, n), 1e-6)))
        inits.append(('parent_noisy2', np.maximum(p + rng.normal(0, 0.1, n), 1e-6)))
    
    inits.append(('uniform', np.ones(n) / n))
    inits.append(('sine', (0.5 + 0.5 * np.cos(2 * np.pi * t)).clip(1e-6, None)))
    inits.append(('gaussian', np.exp(-4 * (t - 0.5) ** 2).clip(1e-6, None)))
    inits.append(('edge', (np.exp(-3 * t**2) + np.exp(-3 * (t-1)**2) + 0.1).clip(1e-6, None)))
    inits.append(('triang', (1 - np.abs(2*t - 1)).clip(1e-6, None)))
    
    # Random smooth
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(('smooth_rand', smooth.clip(1e-6, None)))
    
    # Multi-frequency
    for f in [2, 3, 5]:
        inits.append((f'sine_{f}', (1.0 + 0.3 * np.cos(2 * np.pi * f * t)).clip(1e-6, None)))
    
    # Normalize all
    normalized = []
    for name, a in inits:
        a = np.maximum(a, 1e-8).astype(np.float64)
        a = a / a.sum()
        normalized.append((name, a))
    
    return normalized
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. L-BFGS optimization of a smooth max (log-sum-exp) surrogate with softmax parameterization
2. Adaptive tau schedule for progressive sharpening
3. Multi-start diversification
4. LP mirror descent refinement
5. Multi-length exploration

Key improvements over parent:
- Second-order (L-BFGS) optimization exploits curvature information
- Softmax parameterization naturally enforces the simplex constraint
- Adaptive tau annealing navigates from smooth to sharp landscape
- Combines L-BFGS directions with LP directions for robustness
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a.astype(np.float64), size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a.astype(np.float64), size)
    fb = np.fft.rfft(b.astype(np.float64), size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_fn(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def smooth_max_val_p(b_conv, tau):
    """Compute smooth max value and probability weights."""
    b_max = b_conv.max()
    shifted = tau * (b_conv - b_max)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    val = float(b_max + np.log(total) / tau)
    p = exp_shifted / total
    return val, p


def smooth_max_grad_a(a, tau):
    """Compute smooth max value and gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    val, p = smooth_max_val_p(b, tau)

    grad = np.zeros(n, dtype=np.float64)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return val, grad


def smooth_obj_b(b_param, tau):
    """Objective and gradient in softmax parameterization for scipy.minimize."""
    a = softmax_fn(b_param)
    val, grad_a = smooth_max_grad_a(a, tau)
    dot = np.dot(grad_a, a)
    grad_b = a * (grad_a - dot)
    return val, grad_b


def lbfgs_optimize(a0, deadline, tau_start=5.0, tau_end=50.0, verbose=True,
                   max_lbfgs_iters=3000):
    """
    Optimize using L-BFGS with adaptive tau (progressive sharpening).
    Returns (best_a, best_J).
    """
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 1e-10)
    a0 = a0 / a0.sum()

    b = np.log(a0)
    b = b - b.mean()

    best_a = a0.copy()
    best_J = compute_J(a0)

    n_phases = 4
    taus = np.geomspace(tau_start, tau_end, n_phases)

    for phase in range(n_phases):
        if time.monotonic() > deadline:
            break
        tau = taus[phase]

        if verbose:
            print(f"    L-BFGS phase {phase}: tau={tau:.2f}, current J={best_J:.10f}", flush=True)

        # Use L-BFGS-B (no bounds needed since b is unconstrained)
        try:
            result = minimize(
                smooth_obj_b, b, args=(tau,),
                method='L-BFGS-B', jac=True,
                options={
                    'maxiter': max_lbfgs_iters,
                    'ftol': 1e-15,
                    'gtol': 1e-10,
                    'maxfun': max_lbfgs_iters * 3,
                    'iprint': 0
                }
            )
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error in phase {phase}: {e}", flush=True)
            break

        if not result.success and result.status not in [0, 1, 2]:
            if verbose:
                print(f"    L-BFGS phase {phase} status: {result.status} - {result.message}", flush=True)

        a_new = softmax_fn(result.x)
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)

        if J_new < best_J:
            best_a = a_new.copy()
            best_J = J_new
            if verbose:
                print(f"    Phase {phase} improved: J={J_new:.10f} (smooth_obj={result.fun:.6f})", flush=True)
        else:
            if verbose:
                print(f"    Phase {phase}: J={J_new:.10f} (no improvement, best={best_J:.10f})", flush=True)

        # Update b for next phase (use the result even if no improvement, as it's a better approximation)
        b = result.x - result.x.mean()

    return best_a, best_J


def lbfgs_with_restarts(a0, deadline, tau_start=5.0, tau_end=80.0,
                        n_restarts=3, verbose=True):
    """L-BFGS with periodic restarts and perturbations."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()

    best_a = a.copy()
    best_J = compute_J(a)

    rng = np.random.default_rng(42)
    time_per_restart = max(10.0, (deadline - time.monotonic()) / n_restarts)

    for restart in range(n_restarts):
        if time.monotonic() > deadline:
            break

        phase_dl = min(time.monotonic() + time_per_restart, deadline)

        # For restarts > 0, perturb the current best
        if restart > 0:
            pert_scale = 0.05 * (restart + 1)
            a = best_a * (1.0 + pert_scale * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-8)
            a = a / a.sum()

        tau_s = tau_start * (1.0 + 0.5 * restart)
        tau_e = tau_end * (1.0 + 0.3 * restart)

        a_opt, J_opt = lbfgs_optimize(a, phase_dl, tau_start=tau_s, tau_end=tau_e,
                                      verbose=verbose)
        if J_opt < best_J:
            best_a = a_opt.copy()
            best_J = J_opt

    return best_a, best_J


def lp_mirror_step(a, M, top_frac=0.7, time_limit=4):
    """One step of LP mirror descent: find g maximizing sum(g) s.t. conv(a,g) <= M."""
    n = len(a)
    b = fast_conv(a)

    if top_frac < 1.0:
        threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
        active_k = np.where(b >= threshold)[0]
    else:
        active_k = np.arange(2 * n - 1)

    n_active = len(active_k)
    rows = []
    cols = []
    vals = []
    for idx in range(n_active):
        k = active_k[idx]
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])

    if not rows:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(n_active, n))
    c = -np.ones(n)
    b_ub = np.full(n_active, M)
    bounds = [(0.0, np.inf)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                ag = fast_cross_conv(a, g)
                if ag.max() <= M * (1.0 + 1e-4):
                    return g
    except Exception:
        pass
    return None


def lp_refine(a, deadline, verbose=True, max_no_improve=200):
    """LP mirror descent refinement with line search and restarts."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = float(b.max())

        g = lp_mirror_step(a, M, top_frac=0.65, time_limit=3)
        if g is None:
            g = lp_mirror_step(a, M, top_frac=0.85, time_limit=5)
        if g is None:
            break

        g = g / g.sum()
        current_J = compute_J(a)

        improved = False
        for t in [0.4, 0.25, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)

            if J_trial < best_J - 1e-13:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
            elif J_trial < current_J - 1e-13:
                a = a_trial
                current_J = J_trial
                no_improve = 0
                break

        if not improved:
            no_improve += 1

        if no_improve > max_no_improve:
            # Perturbation to escape
            scale = 0.08 + 0.02 * (no_improve // max_no_improve)
            a = a * (1.0 + scale * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            no_improve = 0

        if verbose and it % 300 == 0:
            print(f"    LP iter {it}: J={best_J:.10f}, no_improve={no_improve}", flush=True)

        if it > 200000:
            break

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-8)
        p = p / p.sum()
        inits.append(('parent', p))
        inits.append(('parent_n1', np.maximum(p + rng.normal(0, 0.03, n), 1e-8)))
        inits.append(('parent_n2', np.maximum(p + rng.normal(0, 0.08, n), 1e-8)))
        inits.append(('parent_n3', np.maximum(p + rng.normal(0, 0.15, n), 1e-8)))

    inits.append(('uniform', np.ones(n).copy()))
    inits.append(('sine1', 1.0 + 0.4 * np.cos(2.0 * np.pi * t)))
    inits.append(('sine2', 1.0 + 0.3 * np.cos(4.0 * np.pi * t)))
    inits.append(('sine3', 1.0 + 0.3 * np.cos(6.0 * np.pi * t)))
    inits.append(('gauss_center', np.exp(-6.0 * (t - 0.5) ** 2) + 0.05))
    inits.append(('edge_conc', np.exp(-8.0 * t ** 2) + np.exp(-8.0 * (t - 1) ** 2) + 0.05))
    inits.append(('triang', 1.0 - np.abs(2.0 * t - 1.0)))
    inits.append(('ramp', t + 0.1))

    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(('smooth_rand', smooth + 0.1))

    # Block structure
    n_blocks = max(4, n // 64)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(('block', block_seq))

    # Normalize all
    normalized = []
    for name, a in inits:
        a = np.maximum(a, 1e-8).astype(np.float64)
        a = a / a.sum()
        normalized.append((name, a))

    return normalized


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining L-BFGS and LP mirror descent.
    Phase 1: L-BFGS at n=512 with multiple diverse starts
    Phase 2: L-BFGS at other lengths
    Phase 3: LP mirror descent refinement
    Phase 4: Final combined polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    best_a = None
    best_J = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                best_a = parent.copy()
                best_J = compute_J(parent)
                print(f"Parent loaded: n={parent.size}, J={best_J:.10f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        best_a = parent.copy()
        best_J = compute_J(parent)
        print(f"Default parent: J={best_J:.10f}", flush=True)

    # === PHASE 1: L-BFGS at main length n=512 ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: L-BFGS Optimization (n=512)", flush=True)
    print(f"{'=' * 60}", flush=True)

    n_main = 512
    p1_deadline = min(t_start + 350.0, deadline - 500.0)

    inits = generate_inits(n_main, rng, parent)

    # Sort by initial J
    inits_sorted = []
    for name, a in inits:
        J_val = compute_J(a)
        inits_sorted.append((J_val, name, a))
    inits_sorted.sort(key=lambda x: x[0])

    # Allocate time to top candidates
    n_cands = min(len(inits_sorted), 8)
    remaining_time = p1_deadline - time.monotonic()
    if remaining_time < 30:
        p1_deadline = time.monotonic() + 10

    # Top candidates get more time
    time_weights = [3.0, 2.5, 2.0, 1.5, 1.0, 1.0, 0.7, 0.7]
    total_weight = sum(time_weights[:n_cands])

    for idx in range(n_cands):
        if time.monotonic() > p1_deadline:
            break

        J_init, name, a0 = inits_sorted[idx]
        alloc = remaining_time * time_weights[idx] / total_weight
        alloc = max(15.0, min(alloc, 120.0))
        phase_dl = min(time.monotonic() + alloc, p1_deadline)

        print(f"  [{idx}] '{name}' init_J={J_init:.6f}, budget={alloc:.0f}s", flush=True)

        a_opt, J_opt = lbfgs_optimize(a0, phase_dl, tau_start=4.0, tau_end=60.0,
                                      max_lbfgs_iters=2000, verbose=True)

        if J_opt < best_J:
            best_J = J_opt
            best_a = a_opt.copy()
            print(f"  ** NEW BEST: J={J_opt:.10f} from '{name}'", flush=True)
        else:
            print(f"  Result: J={J_opt:.10f} (best={best_J:.10f})", flush=True)

    # === PHASE 2: L-BFGS at other lengths ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 2: Multi-length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    p2_deadline = min(t_start + 700.0, deadline - 300.0)

    if best_a is not None:
        lengths_to_try = [384, 640, 768, 1024]
        for n_try in lengths_to_try:
            if time.monotonic() > p2_deadline:
                break

            a_res = resample_seq(best_a, n_try)
            a_res = np.maximum(a_res, 1e-8)
            a_res = a_res / a_res.sum()
            J_res = compute_J(a_res)

            alloc = min(100.0, max(20.0, p2_deadline - time.monotonic() - 30.0))
            phase_dl = min(time.monotonic() + alloc, p2_deadline)

            print(f"  n={n_try}: init_J={J_res:.6f}, budget={alloc:.0f}s", flush=True)

            a_opt, J_opt = lbfgs_optimize(a_res, phase_dl, tau_start=4.0, tau_end=50.0,
                                          max_lbfgs_iters=1500, verbose=False)
            if J_opt < best_J:
                best_J = J_opt
                best_a = a_opt.copy()
                print(f"  ** NEW BEST: J={J_opt:.10f} at n={n_try}", flush=True)
            else:
                print(f"  n={n_try}: J={J_opt:.10f} (best={best_J:.10f})", flush=True)

    # === PHASE 3: LP Mirror Descent Refinement ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 3: LP Mirror Descent Refinement", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_deadline = min(t_start + 1200.0, deadline - 60.0)

    if best_a is not None and time.monotonic() < p3_deadline:
        print(f"  Starting J={best_J:.10f}, n={best_a.size}", flush=True)
        a_lp, J_lp = lp_refine(best_a, p3_deadline, verbose=True, max_no_improve=150)
        if J_lp < best_J:
            best_J = J_lp
            best_a = a_lp.copy()
            print(f"  ** LP improved to J={J_lp:.10f}", flush=True)

    # === PHASE 4: Final L-BFGS Polish at high tau ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 4: Final Polish (High-Tau L-BFGS + LP)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_a is not None and time.monotonic() < deadline - 10:
        # High tau L-BFGS for final precision
        remaining = deadline - time.monotonic()
        half = remaining * 0.5

        if half > 20:
            a_pol, J_pol = lbfgs_optimize(best_a, time.monotonic() + half,
                                          tau_start=30.0, tau_end=120.0,
                                          max_lbfgs_iters=2000, verbose=True)
            if J_pol < best_J:
                best_J = J_pol
                best_a = a_pol.copy()
                print(f"  ** Polish improved to J={J_pol:.10f}", flush=True)

        # Short LP refinement
        if time.monotonic() < deadline - 5:
            a_lp2, J_lp2 = lp_refine(best_a, deadline - 5, verbose=False, max_no_improve=80)
            if J_lp2 < best_J:
                best_J = J_lp2
                best_a = a_lp2.copy()
                print(f"  ** Final LP improved to J={J_lp2:.10f}", flush=True)

    # Fallback
    if best_a is None:
        best_a = np.ones(128)
        best_J = compute_J(best_a)

    # Verify with evaluate_sequence
    final_list = [float(x) for x in best_a]
    final_score = evaluate_sequence(final_list)
    if final_score < best_J:
        best_J = final_score

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_J:.10f} n={best_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return final_list
```
