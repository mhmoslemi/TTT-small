"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Symmetric optimization (a_j = a_{n-1-j})

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp_full(a, M, row_idx, col_idx, time_limit=5):
    """Solve LP: max 1^T g s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_quad(a, g):
    """
    Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2.
    Uses the exact quadratic structure.
    """
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    # f(t) = max_k [(1-t)^2 A_k + 2t(1-t) B_k + t^2 C_k]
    # = max_k [A_k + 2t(B_k - A_k) + t^2(C_k - 2B_k + A_k)]
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    lo = 0.0
    hi = 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_accelerated_lp(a, max_iters, deadline, row_idx, col_idx, 
                       momentum=0.5, verbose=False, verbose_every=100):
    """
    LP mirror descent with Nesterov-style acceleration.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    x_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        if it > 0 and momentum > 0:
            y = a + momentum * (a - x_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-10:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a.copy()
        
        M = compute_M(y)
        g = solve_lp_full(y, M, row_idx, col_idx, time_limit=5)
        if g is None:
            # Fallback: no momentum
            M = compute_M(a)
            g = solve_lp_full(a, M, row_idx, col_idx, time_limit=5)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_symmetric(a):
    """Symmetrize a sequence: a_j = (a_j + a_{n-1-j})/2."""
    n = len(a)
    sym = (a + a[::-1]) / 2.0
    return np.maximum(sym, 0)

def solve_lp_flat(a, row_idx, col_idx, time_limit=5):
    """
    Solve LP: min t s.t. (a*g)_k <= t for all k, sum(g) = 1, g >= 0.
    Returns the optimal g and t.
    """
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    
    # Variables: g_0, ..., g_{n-1}, t
    # Objective: minimize t → c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: (a*g)_k - t <= 0 for all k
    # And: sum(g) = 1 (equality)
    # And: g >= 0, t is free (but t >= 0 implicitly)
    
    # Inequality constraints: A*g - t*e <= 0
    # A_ub is (2n-1) x (n+1)
    A_ub = np.hstack([np.array(A.todense()), -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # g >= 0, t >= 0
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                    bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent with "flatten" LP variant and Nesterov acceleration
2. Multi-direction exploration (standard LP vs flatten LP)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def build_lp_matrix(n):
    """Precompute the structure (row/col indices) of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp_standard(a, M, row_idx, col_idx, time_limit=4):
    """Solve LP: max 1^T g s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def solve_lp_flatten(a, row_idx, col_idx, time_limit=4):
    """
    Solve LP: min T s.t. (a*g)_k <= T for all k, sum(g)=1, g>=0.
    This finds the direction that flattens the cross-correlation peak.
    """
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    
    # Build constraint: (a*g)_k - T <= 0
    # Variables: [g_0,...,g_{n-1}, T]
    # A_ub has shape (2n-1) x (n+1)
    # Each row k: vals for row k of A, and -1 in the T column
    
    rows_ext = []
    cols_ext = []
    vals_ext = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows_ext.append(k)
            cols_ext.append(j)
            vals_ext.append(a[k - j])
        rows_ext.append(k)
        cols_ext.append(n)
        vals_ext.append(-1.0)
    
    A_ub = csr_matrix((vals_ext, (rows_ext, cols_ext)), shape=(2 * n - 1, n + 1))
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize T
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def line_search_quad(a, g):
    """
    Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2_k.
    Uses ternary search on the convex(ish) function.
    """
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    lo = 0.0
    hi = 1.0
    for _ in range(75):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx,
                          use_flatten=False, momentum=0.0,
                          verbose=False, verbose_every=50, lp_time=4):
    """
    LP mirror descent with optional Nesterov momentum and flatten LP.
    Input a should be non-negative with sum = 1.
    Returns optimized non-negative sequence (normalized to sum 1).
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    x_prev = a.copy()
    mom = momentum
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        if it > 0 and mom > 0.05:
            y = a + mom * (a - x_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-10:
                y = y / sy
            else:
                y = a.copy()
        else:
            y = a.copy()
        
        if use_flatten:
            g = solve_lp_flatten(y, row_idx, col_idx, time_limit=lp_time)
        else:
            M = compute_M(y)
            g = solve_lp_standard(y, M, row_idx, col_idx, time_limit=lp_time)
        
        if g is None:
            # Fallback without momentum
            if use_flatten:
                g = solve_lp_flatten(a, row_idx, col_idx, time_limit=lp_time)
            else:
                M = compute_M(a)
                g = solve_lp_standard(a, M, row_idx, col_idx, time_limit=lp_time)
            if g is None:
                break
            y = a.copy()
        
        t, M_new = line_search_quad(y, g)
        x_prev = a.copy()
        
        if t < 1e-9:
            no_improve += 1
        elif M_new < best_M:
            a_new = (1.0 - t) * y + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Adaptive momentum: reduce when stuck
        if no_improve > 30:
            mom *= 0.7
            no_improve = 0
            if mom < 0.02:
                mom = 0.0
        
        if no_improvement_count := no_improve > 200:
            break
        
        if verbose and it % verbose_every == 0:
            print(f"      iter {it}: M={compute_M(a):.10f} best={best_M:.10f} "
                  f"t={t:.4f} mom={mom:.3f}", flush=True)
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_symmetric(a):
    """Symmetrize: a_j = (a_j + a_{n-1-j})/2."""
    return np.maximum((a + a[::-1]) / 2.0, 0)

    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
