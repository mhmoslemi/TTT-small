Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    """Precompute row/col indices for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(a, M, row_idx, col_idx, time_limit=5):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def mwu_solve(a, M, n_inner=10, eta=1.0):
    """
    Approximate LP solution using Multiplicative Weight Updates.
    Solves: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns approximate solution g (normalized to sum 1).
    """
    n = len(a)
    g = np.ones(n) / n
    for _ in range(n_inner):
        h = cross_conv(a, g)
        # g_new = g * exp(-eta * h / M), normalized
        log_g = np.log(np.maximum(g, 1e-30))
        log_g -= eta * h / (M + 1e-30)
        # Normalize
        log_g -= log_g.max()
        g = np.exp(log_g)
        s = g.sum()
        if s < 1e-30:
            g = np.ones(n) / n
        else:
            g = g / s
    return g


def line_search_analytic(a, g):
    """
    Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)[k]^2.
    Both a and g should be normalized to sum 1.
    """
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx,
                          verbose=False, lp_time_limit=5, use_mwu=False, mwu_inner=8):
    """
    LP mirror descent with analytic line search.
    Input a should be non-negative, normalized to sum 1.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        if M < 1e-15:
            break

        if use_mwu:
            g = mwu_solve(a, M, n_inner=mwu_inner, eta=1.0)
        else:
            g = solve_lp(a, M, row_idx, col_idx, time_limit=lp_time_limit)

        if g is None:
            break

        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-10:
            no_improve += 1
        elif M_new < best_M - 1e-12:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 200:
            break

        if verbose and it % 500 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f} "
                  f"no_improve={no_improve}", flush=True)

    return best_a


def smooth_gradient_step(a, tau):
    """Gradient of (1/tau)*log(sum exp(tau*(a*a)))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[j] += 2 * pk * a[k-j] for j in [lo, hi]
        indices = np.arange(lo, hi + 1)
        grad[indices] += 2.0 * pk * a[k - indices]

    return grad


def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=100.0):
    """Smooth gradient descent with adaptive tau and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.7)
        lr = 0.5 * (0.02 / 0.5) ** frac

        grad = smooth_gradient_step(a, tau)
        v = 0.9 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 1e-10)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: LP mirror descent on multiple lengths with
    MWU-accelerated restarts and aggressive final polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    # Phase 1: LP mirror descent on parent
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent n={parent.size}", flush=True)
    print(f"{'='*60}", flush=True)

    n0 = parent.size
    row_idx0, col_idx0 = build_lp_matrix(n0)
    a = parent.copy()
    a = a / a.sum()

    phase1_deadline = min(t_start + budget_s * 0.20, deadline)
    a_opt = run_lp_mirror_descent(a, 50000, phase1_deadline, row_idx0, col_idx0,
                                  verbose=True, lp_time_limit=5)
    tracker.update(a_opt, "(P1 LP-MD)")

    # Phase 2: Multi-length exploration
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    base = tracker.best if tracker.best is not None else parent
    base_n = base.size

    # Try multiple lengths around the base
    trial_lengths = []
    for target_n in [384, 512, 640, 768, 1024, 1280, 1536, 2048]:
        if target_n != base_n and 128 <= target_n <= 2048:
            trial_lengths.append(target_n)
    trial_lengths = sorted(set(trial_lengths))

    # Allocate time proportional to inverse of n (smaller n gets more iterations)
    phase2_deadline = min(t_start + budget_s * 0.55, deadline)
    total_weight = sum(1.0 / np.sqrt(n) for n in trial_lengths)

    for n_trial in trial_lengths:
        if time.monotonic() > phase2_deadline:
            break

        remaining = phase2_deadline - time.monotonic()
        weight = (1.0 / np.sqrt(n_trial)) / total_weight
        n_trial_budget = max(20, remaining * weight)

        print(f"\n  --- n={n_trial}, budget={n_trial_budget:.0f}s ---", flush=True)

        a_init = resample_seq(base, n_trial)
        a_init = np.maximum(a_init, 1e-6)
        a_init = a_init / a_init.sum()

        r_idx, c_idx = build_lp_matrix(n_trial)
        sub_deadline = time.monotonic() + n_trial_budget

        # LP mirror descent
        a_opt = run_lp_mirror_descent(a_init, 20000, sub_deadline, r_idx, c_idx,
                                      verbose=False, lp_time_limit=3)
        tracker.update(a_opt, f"(P2 n={n_trial} LP)")

        # Also try MWU-based (faster, can do more iterations)
        if time.monotonic() < sub_deadline - 10:
            a_init2 = resample_seq(base, n_trial)
            a_init2 = np.maximum(a_init2, 1e-6)
            a_init2 = a_init2 / a_init2.sum()
            mwu_deadline = min(time.monotonic() + n_trial_budget * 0.3, sub_deadline)
            a_mwu = run_lp_mirror_descent(a_init2, 50000, mwu_deadline, r_idx, c_idx,
                                          verbose=False, use_mwu=True, mwu_inner=12)
            tracker.update(a_mwu, f"(P2 n={n_trial} MWU)")

    # Phase 3: Perturbation restarts on best
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_idx_b, c_idx_b = build_lp_matrix(n_best)
        phase3_deadline = min(t_start + budget_s * 0.75, deadline)

        # Different perturbation strategies
        strategies = [
            ("gauss_small", 0.005),
            ("gauss_med", 0.015),
            ("gauss_large", 0.04),
            ("sparse", 0.03),
            ("gauss_small", 0.003),
            ("gauss_med", 0.01),
        ]

        restart_idx = 0
        for strategy_name, scale in strategies:
            n_restarts = 8
            for r in range(n_restarts):
                if time.monotonic() > phase3_deadline:
                    break
                restart_idx += 1

                a = tracker.best.copy().astype(np.float64)
                if strategy_name == "sparse":
                    # Perturb only a random subset
                    n_perturb = max(4, n_best // 8)
                    idx = rng.choice(n_best, size=n_perturb, replace=False)
                    a[idx] += rng.normal(0, scale, n_perturb)
                else:
                    a = a + rng.normal(0, scale, n_best)
                a = np.maximum(a, 1e-6)
                a = a / a.sum()

                # Quick smooth then LP
                sub_dl1 = min(time.monotonic() + 10, phase3_deadline)
                a = run_smooth_gradient(a, 100, sub_dl1, initial_tau=10, final_tau=60)

                sub_dl2 = min(time.monotonic() + 25, phase3_deadline)
                a_opt = run_lp_mirror_descent(a, 3000, sub_dl2, r_idx_b, c_idx_b,
                                              verbose=False, lp_time_limit=3)
                tracker.update(a_opt, f"(P3 {strategy_name} r={r})")

        # Also try perturbing to nearby lengths
        for delta_n in [-64, 64]:
            n_try = n_best + delta_n
            if n_try < 128 or n_try > 2048:
                continue
            if time.monotonic() > phase3_deadline:
                break
            r_idx_t, c_idx_t = build_lp_matrix(n_try)
            for r in range(4):
                if time.monotonic() > phase3_deadline:
                    break
                a = resample_seq(tracker.best, n_try)
                a = np.maximum(a, 1e-6)
                a = a + rng.normal(0, 0.01, n_try)
                a = np.maximum(a, 1e-6)
                a = a / a.sum()
                sub_dl = min(time.monotonic() + 20, phase3_deadline)
                a_opt = run_lp_mirror_descent(a, 2000, sub_dl, r_idx_t, c_idx_t,
                                              verbose=False, lp_time_limit=3)
                tracker.update(a_opt, f"(P3 n={n_try} r={r})")

    # Phase 4: Final extended LP polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)
        a = tracker.best.copy()
        a = a / a.sum()

        # Split final phase: LP then smooth
        mid_deadline = min(t_start + budget_s * 0.95, deadline - 20)
        a_opt = run_lp_mirror_descent(a, 100000, mid_deadline, r_f, c_f,
                                      verbose=True, lp_time_limit=5)
        tracker.update(a_opt, "(P4 LP final)")

        # Smooth polish on the LP result
        if tracker.best is not None and time.monotonic() < deadline - 5:
            a_sg = tracker.best.copy()
            a_sg = a_sg / a_sg.sum()
            a_sg = run_smooth_gradient(a_sg, 500, deadline - 2,
                                       initial_tau=50, final_tau=300)
            tracker.update(a_sg, "(P4 smooth final)")

            # One more LP after smooth
            if time.monotonic() < deadline - 3:
                r_f2, c_f2 = build_lp_matrix(tracker.best.size)
                a_lp2 = tracker.best.copy() / tracker.best.sum()
                a_lp2 = run_lp_mirror_descent(a_lp2, 5000, deadline - 1, r_f2, c_f2,
                                              verbose=False, lp_time_limit=3)
                tracker.update(a_lp2, "(P4 LP after smooth)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
