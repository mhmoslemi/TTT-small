def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.where(cond)[0][-1]
    rho = (cssv[i] + 1) / (i + 1)
    return np.maximum(v - rho, 0)

Let me think about what the current LP mirror descent does well:

The main issue is that the LP is expensive. What if we use a cheaper approximation of the LP direction?

**Idea: Greedy "LP approximation"**

Instead of solving the full LP, we can approximate the direction g by:
1. Start with g = a (current point)
2. For each active constraint k (where (a*a)_k is close to M), try to decrease the contribution to that lag
3. The most effective way to decrease (a*a)_k is to decrease the product a_i * a_{k-i} for the pairs with the largest product

This is like a coordinate descent on the "peaks" of the autocorrelation.

Actually, here's a cleaner idea:

**Idea: Iterative "peak shaving"**

1. Compute b = a*a
2. Find the peak lag k* where b_k* is maximum
3. To reduce b_k*, we want to reduce the products a_i * a_{k*-i}
4. The most effective single change is to reduce the maximum a_i * a_{k*-i}
5. This suggests: find the pair (i, k*-i) with the largest product, and reduce one of them
6. To maintain the sum constraint, increase some other coordinate (one that doesn't contribute much to any peak)

This is essentially a coordinate descent / steepest descent in a combinatorial sense.

  But it also affects other lags: b_{j} for all j that involve a_i
- The "cost" of decreasing a_i is the increase in other peaks

Let me also think about the "weighted subgradient" idea more carefully:

When we solve the LP:
max 1^T g s.t. A g <= M, g >= 0

The dual is:
min M * 1^T y s.t. A^T y >= 1, y >= 0

If we have the dual solution y, then the weighted subgradient of max(a*a) at the current point is:
grad_weighted = sum_k y_k * grad_k / sum_k y_k

where grad_k is the gradient of (a*a)_k.

This gives a "smoother" subgradient that focuses on the most important constraints. We can use this in the smooth gradient descent between LP solves.

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. FFT-accelerated smooth gradient descent with adaptive temperature
2. LP mirror descent with analytic line search for powerful corrections
3. Weighted subgradient using LP dual for informed descent between LP solves
4. Multi-length exploration with resampling
5. Peak-shaving coordinate descent for final polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def build_lp_matrix(n):
    """Precompute (row, col) indices for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx, time_limit=5):
    """Solve LP: max 1^T g s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g, r
    except Exception:
        pass
    return None, None


def solve_lp_dual(a, M, row_idx, col_idx, time_limit=5):
    """Solve LP and return dual variables."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                # Get dual variables (shadow prices)
                # In scipy linprog, r.ineqlin.marginals gives duals for A_ub x <= b_ub
                duals = r.ineqlin.marginals if hasattr(r, 'ineqlin') and r.ineqlin is not None else None
                if duals is not None:
                    # marginals are non-positive for <= constraints
                    duals = -np.abs(duals)  # make positive
                return g, duals
    except Exception:
        pass
    return None, None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def smooth_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*(a*a))) using FFT.
    Returns the gradient (not divided by tau, so it's the gradient of the log-sum-exp).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    logits = np.clip(tau * (b - b_max), -500, 0)
    exp_b = np.exp(logits)
    p = exp_b / exp_b.sum()
    
    # grad[i] = 2 * sum_k p[k] * a[k-i] for valid indices
    # = 2 * cross_correlation(a, p)[i]
    # cross_correlation(a, p)[i] = sum_m a[m] * p[m+i]
    
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    corr = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    return 2.0 * corr, p


def run_smooth(a_init, max_iters, deadline, tau_schedule='exp'):

        frac = min(1.0, it / max(1, max_iters))
        if tau_schedule == 'exp':

        elif tau_schedule == 'lin':
            tau = 5.0 + 195.0 * frac
        else:
            tau = 50.0
        
        # Compute gradient
        grad, p = smooth_gradient(a, tau)
        grad = grad / tau  # This is the gradient of (1/tau)*log(sum exp(tau*b))
        
        # Learning rate: decrease over time
        # The Lipschitz constant of the smooth function is approximately O(n/tau)
        lr = 1.0 / (n * tau * 0.1)
        lr = min(lr, 0.5 / (1 + 0.05 * it))

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()

def run_lp_descent(a_init, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = compute_M(a)
        g, _ = solve_lp(a, M, row_idx, col_idx, time_limit=5)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        
        t, M_new = line_search_analytic(a, g)
        
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:

            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:

                a = a_new

                if actual_M < best_M:

                    best_a = a.copy()
                    no_improve = 0
                else:

            else:

        if verbose and it % 100 == 0:

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coordinate_descent_peak_shave(a, max_iters, deadline):
    """
    Coordinate descent that focuses on reducing the peak of the autocorrelation.
    At each step, identify the peak lag, then find the coordinate pair that
    contributes most to that peak and reduce one of them (compensating elsewhere).
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_peak = int(np.argmax(b))
        M = b[k_peak]
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
        
        # Find the pair (i, k_peak - i) with the largest product
        j_min = max(0, k_peak - n + 1)
        j_max = min(n - 1, k_peak)
        
        best_reduction = 0
        best_i = -1
        best_partner = -1
        
        for i in range(j_min, j_max + 1):
            partner = k_peak - i
            if partner < 0 or partner >= n:
                continue
            if i == partner:
                # Reduce a_i^2: reduction = 2 * a_i * da
                # We'll handle this separately
                continue
            prod = a[i] * a[partner]
            # If we reduce a[i] by delta, the reduction in b_k_peak is approximately partner_val * delta
            # If we reduce a[partner] by delta, the reduction is a[i] * delta
            # We want the largest marginal reduction per unit decrease
            # Reducing a[i]: marginal effect = a[partner] (and also affects other lags)
            # Reducing a[partner]: marginal effect = a[i]
            # Choose the one with larger marginal effect (larger partner value)
            # But we also need to consider the cost (effect on other lags)
            # Simple heuristic: reduce the smaller of the two (less "expensive")
            target = i if a[i] < a[partner] else partner
            other = partner if target == i else i
            marginal = a[other]
            # Cost: how much does reducing a[target] affect other peaks?
            # Approximate cost: sum of a[j] for j where target and j form a peak
            # For simplicity, just use the marginal
            if marginal > best_reduction:
                best_reduction = marginal
                best_i = target
                best_partner = other
        
        if best_i < 0:
            # All pairs are i==partner (k_peak is even and only one term)
            i = k_peak // 2
            if 0 <= i < n:
                best_i = i
                best_reduction = a[i]
        
        if best_i < 0 or best_reduction < 1e-15:
            break
        
        # Reduce a[best_i] slightly, increase the smallest element to maintain sum
        delta = 0.01 * a[best_i]  # reduce by 1%
        a[best_i] -= delta
        
        # Find the smallest element to distribute the mass to
        # (to minimize impact on other peaks)
        min_idx = int(np.argmin(a))
        a[min_idx] += delta
        
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        # Check improvement
        M_new = compute_M(a)
        if M_new > best_M * (1 + 0.1):
            # Got worse, revert
            a = best_a.copy()
            break
    
    return best_a


class Tracker:
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self):
        if self.best is not None:
            print(f"  best: J={self.best_score:.8f} n={self.best.size}", flush=True)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def build_lp_structure(n):
    """Precompute (row, col) index arrays for the full LP constraint matrix."""
    rows = []
    cols = []
    row_starts = [0]
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
        row_starts.append(len(rows))
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), row_starts


def solve_lp_active(a, M, row_idx, col_idx, row_starts, top_k=60, time_limit=3):
    """
    Solve LP: max 1^T g s.t. (a*g)[k] <= M for active k, g >= 0.
    Uses only the top_k largest (a*a)_k constraints for speed.
    """
    n = len(a)
    b = fast_conv(a)
    # Select top lags by value
    if len(b) <= top_k:
        active_k = np.arange(len(b))
    else:
        active_k = np.argsort(b)[::-1][:top_k]
    active_k = np.sort(active_k)

    # Extract rows corresponding to active_k using row_starts
    row_mask = np.zeros(len(row_idx), dtype=bool)

    n_constraints = len(active_k)
    A = csr_matrix((r_vals, (r_rows, r_cols)), shape=(2 * n - 1, n))

def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def smooth_gradient_fft(a, tau, top_lags=None):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*(a*a))) using FFT.
    If top_lags is given, only weight those lags (hard threshold).
    Returns (gradient, max_conv_value).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    logits = np.clip(tau * (b - b_max), -500, 0)
    exp_b = np.exp(logits)
    p = exp_b / exp_b.sum()

    if top_lags is not None:
        mask = np.zeros(len(b), dtype=bool)
        mask[top_lags] = True
        p = np.where(mask, p, 0)
        s = p.sum()
        if s > 0:
            p = p / s
        else:
            p = p.copy()  # fallback

    # grad[i] = 2 * sum_k p[k] * a[k-i] = 2 * cross_correlation(a, p)[i]
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    corr = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    return (2.0 / tau) * corr, b_max


def run_smooth(a_init, max_iters, deadline, tau_start=5.0, tau_end=300.0, momentum=0.9, verbose=False):
    """Smooth gradient descent with momentum and adaptive tau."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.4)

        grad, b_max = smooth_gradient_fft(a, tau)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Adaptive learning rate
        lr = 1.0 / (n * 0.5 * (1 + 0.1 * it))

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            break

        if verbose and it % 500 == 0:
            print(f"        smooth iter {it}: M={M:.10f} tau={tau:.1f}", flush=True)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 1e-6)


def peak_shave(a_init, max_iters, deadline):
    """
    Coordinate descent: at each step, reduce the largest contributor to the peak lag.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    stuck = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_peak = int(np.argmax(b))
        M = b[k_peak]

        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 50:
                break

        # Find pair (i, k_peak - i) with largest product contribution
        j_min = max(0, k_peak - n + 1)
        j_max = min(n - 1, k_peak)

        best_target = -1
        best_marginal = 0
        for i in range(j_min, j_max + 1):
            partner = k_peak - i
            if partner < 0 or partner >= n:
                continue
            if i == partner:
                marginal = 2.0 * a[i]
                if marginal > best_marginal:
                    best_marginal = marginal
                    best_target = i
            else:
                marginal = a[partner]
                if marginal > best_marginal:
                    best_marginal = marginal
                    best_target = i

        if best_target < 0 or best_marginal < 1e-15:
            break

        # Reduce a[best_target] and add to a random small element
        delta = 0.02 * a[best_target]
        a[best_target] -= delta

        # Distribute to elements not involved in the peak
        candidates = np.where(a < 0.05 * a.max())[0]
        if len(candidates) > 0:
            add_to = candidates[np.random.randint(len(candidates))]
        else:
            add_to = int(np.argmin(a))
        a[add_to] += delta

        a = np.maximum(a, 0)
        a = a / a.sum()

    return best_a

        cur = run_lp_mirror(cur / cur.sum(), 5000, sub_dl2, r_idx, c_idx, r_starts,
                            use_active=True, top_k=60, verbose=False)
        tracker.update(cur, f"(P3 r{round_idx} LP)")

    # === PHASE 4: Multi-length exploration (15% budget) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    dl4 = t_start + budget_s * 0.80
    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = set()
        for mult in [0.5, 0.75, 1.25, 1.5, 2.0, 0.625, 1.333]:
            n_t = int(round(base_n * mult))
            if 64 <= n_t <= 4096:
                trial_lengths.add(n_t)

        for n_t in sorted(trial_lengths):
            if time.monotonic() > dl4:
                break
            a_init = resample_seq(tracker.best, n_t)
            a_init = a_init / a_init.sum()
            r_t, c_t, rs_t = build_lp_structure(n_t)

            sub_dl = min(time.monotonic() + 20, dl4)
            a_sm = run_smooth(a_init, 10000, sub_dl, tau_start=5.0, tau_end=150.0)
            tracker.update(a_sm, f"(P4 n={n_t} smooth)")

            sub_dl2 = min(time.monotonic() + 40, dl4)
            a_lp = run_lp_mirror(a_sm / a_sm.sum(), 5000, sub_dl2, r_t, c_t, rs_t,
                                 use_active=True, top_k=60)
            tracker.update(a_lp, f"(P4 n={n_t} LP)")

    # === PHASE 5: Perturbation restarts (10% budget) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    dl5 = t_start + budget_s * 0.90
    if tracker.best is not None:
        n_best = tracker.best.size
        r_b, c_b, rs_b = build_lp_structure(n_best)
        temperature = 0.1

        for r in range(30):
            if time.monotonic() > dl5:
                break

            a = tracker.best.copy().astype(np.float64)
            noise = rng.normal(0, temperature * 0.01, n_best)
            a = a + noise
            a = np.maximum(a, 1e-8)
            a = a / a.sum()

            sub_dl = min(time.monotonic() + 10, dl5)
            a = run_smooth(a, 5000, sub_dl, tau_start=10.0, tau_end=100.0, momentum=0.85)

            sub_dl2 = min(time.monotonic() + 25, dl5)
            a = run_lp_mirror(a / a.sum(), 3000, sub_dl2, r_b, c_b, rs_b,
                              use_active=True, top_k=60)
            tracker.update(a, f"(P5 r={r} T={temperature:.3f})")
            temperature *= 0.85

    # === PHASE 6: Peak shaving + final LP polish (5% budget) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Peak Shaving + Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_fin = tracker.best.size
        r_f, c_f, rs_f = build_lp_structure(n_fin)

        # Peak shave
        a_ps = peak_shave(tracker.best.copy(), 5000, min(time.monotonic() + 20, deadline))
        tracker.update(a_ps, "(P6 peak_shave)")

        # Final LP
        a_fin = run_lp_mirror(tracker.best.copy() / tracker.best.sum(),
                              50000, deadline, r_f, c_f, rs_f,
                              use_active=True, top_k=50, verbose=True)
        tracker.update(a_fin, "(P6 final LP)")

        # One more peak shave on the result
        a_ps2 = peak_shave(a_fin, 3000, deadline)
        tracker.update(a_ps2, "(P6 final shave)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
