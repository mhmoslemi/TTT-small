import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, x0=None, time_limit=20):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None. x0 is optional warm start.
    """
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        options = {'presolve': True, 'time_limit': time_limit,
                   'dual_feasibility_tolerance': 1e-10,
                   'primal_feasibility_tolerance': 1e-10}
        if x0 is not None:
            options['presolve'] = False
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      x0=x0, options=options)
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def smoothed_gradient(a, tau=0.05):
    """
    Compute gradient of smoothed max of conv(a,a) w.r.t. a.
    Uses soft-max weighting with temperature tau.
    Returns gradient vector of length n.
    """
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -300.0, 300.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    # grad_i = 2 * sum_k weights[k] * a[k-i] for valid k-i
    # This is 2 * (weights_reversed * a)[valid range]
    w_rev = weights[::-1]
    conv_fw = fft_convolve(a, w_rev)
    # The gradient component for a[i] is 2 * sum_k w[k] * a[k-i]
    # = 2 * (a * w_rev)[2(n-1)-i] ... let me think again
    # Actually: grad_i = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
    # Let j = k-i, then k = i+j, j ranges over [0, n-1]
    # grad_i = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is the cross-correlation of w with a, evaluated at lag i
    # = (w * a_reversed) reversed... 
    # More simply: it's the convolution (w @ a) at position (n-1+i)
    # where (w @ a)[m] = sum_j w[m-j] * a[j]
    # We want sum_j w[i+j] * a[j] = (w_reversed @ a)[(len(w)-1-i)]
    # Wait, let me just use the direct formula:
    # grad_i = 2 * sum_j a[j] * w[i+j] for j=0..n-1, where w is defined on [0, 2n-2]
    # This is the cross-correlation. Let's use FFT:
    # If we compute C = fft_convolve(w, a[::-1]), then C[m] = sum_j w[m-j]*a[n-1-j]
    # That's not quite right either. Let me just do it with a direct FFT approach.
    
    # Actually the cleanest way: grad_i = 2 * sum_j a[j] * w[i+j]
    # = 2 * (cross_correlation of a and w) at lag i
    # cross_correlation(a, w)[i] = sum_j a[j] * w[j+i]
    # This equals (fft(a) * conj(fft(w))) inverse, at the right offset
    
    # Simpler: just use the fact that this is a Toeplitz multiplication
    # grad = 2 * T(w) @ a where T(w) is Toeplitz with w[i+j] at (i,j)
    # But that's O(n^2). For n=700, that's 490K ops which is fine.
    
    # For efficiency, use the convolution formulation:
    # sum_j a[j] * w[i+j] for j=0..n-1
    # = sum_j a[j] * w[i+j]
    # Let's define w_ext as w padded, and compute (a * w_ext_reversed) appropriately
    
    # Actually, let's just use the FFT approach from the previous program:
    # w_rev = weights[::-1] has length 2n-1
    # conv_fw = fft_convolve(a, w_rev) has length n + (2n-1) - 1 = 3n-2
    # conv_fw[m] = sum_j a[j] * w_rev[m-j] = sum_j a[j] * w[(2n-2)-(m-j)] = sum_j a[j] * w[2n-2-m+j]
    # We want sum_j a[j] * w[i+j]. Set 2n-2-m = i, so m = 2n-2-i.
    # So grad_i = 2 * conv_fw[2n-2-i]
    
    grad = np.zeros(n)
    for i in range(n):
        idx = 2 * n - 2 - i

    """
    Nesterov-accelerated optimization of smoothed max objective.

    """

    v = np.zeros(n)
    mu = 0.9
    step = 0.03
    tau = tau_init

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Nesterov lookahead
        a_lookahead = a + mu * v

        # Compute gradient at lookahead point
        grad = smoothed_gradient(a_lookahead, tau=tau)

        # Project onto sum=const (remove mean component)
        grad_centered = grad - np.mean(grad)

        # Update momentum
        v = mu * v - step * grad_centered

        obj = compute_obj(a)

            best_obj = obj
            best_a = a.copy()

        # Decay
        step *= 0.998
        tau *= tau_decay

    return best_a

def multi_peak_lp(a, M, top_k=5, time_limit=15):
    """
    LP with weighted peaks: emphasize the top-k peaks more strongly.

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    # Tighten constraints for non-top peaks (they should be below M anyway)
    # Actually, loosen them slightly to allow the solver to focus on top peaks
    for k in range(2 * n - 1):

                      options={'presolve': True, 'time_limit': time_limit,

def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.ones(new_n) * a[0] if n == 1 else np.ones(new_n)
    
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    
    from scipy.interpolate import PchipInterpolator
    try:
        interp = PchipInterpolator(x_old, a)
        a_new = interp(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

                                            time_budget=min(remaining * 0.15, 12.0),
                                            tau_init=0.02, tau_decay=0.995)
            obj_nag = compute_obj(a_new2)
            if obj_nag < obj_lp:
                a_new = a_new2
                obj_lp = obj_nag

        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_new.copy()
            best_n = n_try
            print(f"  ** n={n_try}: {best_score:.8f} **")
        else:
            print(f"  n={n_try}: {obj_lp:.8f}")

    n_best = len(best_seq)
    print(f"  Phase 2 done: best={best_score:.8f}, n={n_best}")

    # === Phase 3: Final intensive polish ===
    print(f"\n=== Phase 3: Final polish (n={n_best}) ===")
    a = best_seq.copy()

    for round_idx in range(10):
        if time.monotonic() > deadline - 20:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        a_new, obj_new = polish_sequence(a, time_budget=min(remaining * 0.25, 50.0))
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  Polish R{round_idx}: {best_score:.8f}")
        else:
            # Try perturbation + re-optimize
            rng = np.random.default_rng(round_idx * 31337 + seed)
            for pert in range(3):
                if time.monotonic() > deadline - 15:
                    break
                remaining = deadline - time.monotonic()
                if remaining < 10:
                    break

                # Different perturbation strategies
                strategy = (pert + round_idx) % 3
                if strategy == 0:
                    # Gaussian noise
                    noise = rng.normal(0, 0.03, n_best)
                    a_pert = a * np.exp(noise)
                elif strategy == 1:
                    # Sinusoidal
                    freq = rng.uniform(0.5, n_best / 3)
                    phase = rng.uniform(0, 2 * np.pi)
                    a_pert = a * (1 + 0.05 * np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase))
                else:
                    # Block perturbation
                    block = max(1, n_best // 20)
                    a_pert = a.copy()
                    for _ in range(5):
                        start_idx = rng.integers(0, n_best - block)
                        factor = rng.uniform(0.7, 1.3)
                        a_pert[start_idx:start_idx + block] *= factor

                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p < 1e-10:
                    continue
                a_pert = a_pert / s_p

                # Quick optimize
                a_opt = lp_alternate_enhanced(a_pert, max_iters=20, time_budget=8.0)
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    a = a_opt.copy()
                    print(f"  Pert {pert} (R{round_idx}): {best_score:.8f}")

    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: Ultra-fine final steps ===
    if time.monotonic() < deadline - 15:
        print(f"\n=== Phase 4: Ultra-fine polish ===")
        remaining = deadline - time.monotonic()

        # Very low temperature Nesterov
        a_fine = nesterov_smoothed_opt(best_seq, max_iters=500,
                                        time_budget=min(remaining * 0.4, 30.0),
                                        tau_init=0.005, tau_decay=0.999)
        obj_fine = compute_obj(a_fine)
        if obj_fine < best_score:
            best_score = obj_fine
            best_seq = a_fine.copy()
            print(f"  Ultra-fine NAG: {best_score:.8f}")

        # One more LP pass
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_final = lp_alternate_enhanced(best_seq, max_iters=30,
                                             time_budget=min(remaining * 0.5, 25.0))
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Ultra-fine LP: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
