import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v, S=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 1:

        weights_raw = np.exp((peak_vals - M) / temp)
        weights = weights_raw / np.sum(weights_raw)

    # Compute C[i] = sum_k w_k * a[k-i] for valid k-i
    # Build sparse weight vector
    w_vec = np.zeros(2 * n - 1)
    for idx, w in zip(sorted_idx[:top_k], weights if top_k < len(conv) else weights[:len(sorted_idx)]):
        w_vec[idx] = w
    # Normalize weights
    w_sum = np.sum(w_vec)
    if w_sum < 1e-30:
        w_vec[int(np.argmax(conv))] = 1.0
        w_sum = 1.0
    else:
        w_vec /= w_sum

    # C[i] = sum_k w_vec[k] * a[k-i] (for valid 0<=k-i<n)
    # = conv(w_vec, a_rev)[n-1+i] where a_rev = a[::-1]
    a_rev = a[::-1]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    cw = rfft(w_vec, size)
    ca = rfft(a_rev, size)
    conv_wa = irfft(cw * ca, size)
    C = conv_wa[n - 1:n + n - 1]

    # Gradient: 2n/S^3 * (S * 2*C - 2*M*a) = 4n/S^3 * (S*C - M*a)
    # But we use the soft max value instead of M for smoothness
    soft_M = float(np.dot(w_vec, conv))
    grad = (4.0 * n / (S ** 3)) * (S * C - soft_M * a)

    return grad, M, conv


def lse_gradient(a, beta):
    """
    Compute gradient using log-sum-exp smoothing of the max.
    More stable for very flat or very peaked convolutions.
    """
    n = len(a)
    conv = compute_conv(a)
    S = float(np.sum(a))
    M = float(np.max(conv))

    # Softmax weights: w_k = exp(beta * conv_k) / sum exp(beta * conv_k)
    shifted = beta * (conv - M)
    shifted = np.clip(shifted, -500, 500)
    w_vec = np.exp(shifted)
    w_sum = np.sum(w_vec)
    if w_sum < 1e-300:
        w_vec = np.zeros(2 * n - 1)
        w_vec[int(np.argmax(conv))] = 1.0
    else:
        w_vec /= w_sum

    # LSE value
    LSE = M + np.log(w_sum) / beta

    # C[i] = sum_k w_k * a[k-i]
    a_rev = a[::-1]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    cw = rfft(w_vec, size)
    ca = rfft(a_rev, size)
    conv_wa = irfft(cw * ca, size)
    C = conv_wa[n - 1:n + n - 1]

    # grad = 2n/S^2 * d(LSE)/da - 2n*LSE/S^3 * da
    # d(LSE)/da[i] = 2*C[i]
    grad = (2.0 * n / S ** 2) * 2.0 * C - (2.0 * n * LSE / S ** 3) * a

    return grad, M, conv


def chop_peak(a, n_iters=200, base_step=0.08, verbose=False):
    """
    Aggressively reduce the single dominant peak using direct subgradient
    with diminishing steps. Very effective when one peak dominates.
    """
    n = len(a)
    a = a.copy()
    S = np.sum(a)

    for it in range(n_iters):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        M = float(conv[k_star])

        # Subgradient: sg[i] = 2*a[k_star - i] if valid
        sg = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                sg[i] = 2.0 * a[j]

        # Center (project onto sum=const)
        sg = sg - np.mean(sg)

        # Diminishing step
        step = base_step * (1.0 / (1.0 + 0.05 * it))

        a_new = a - step * sg
        a_new = project_simplex(a_new, S)
        a = a_new

    return a


def solve_lp_direction(a, M, time_limit=8):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    if n > 1200:
        return None
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = a[:end - start]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def line_search_blend(a, g, n_trials=8):
    """Find best t in [0,1] for blend (1-t)*a + t*g."""
    best_obj = compute_obj(a)
    best_t = 0.0
    best_a = a.copy()
    for t in np.linspace(0.05, 1.0, n_trials):
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s = np.sum(a_try)
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj = compute_obj(a_try)
        if obj < best_obj:
            best_obj = obj
            best_t = t
            best_a = a_try.copy()
    return best_a, best_obj, best_t


def optimize_single(a, time_budget=60.0, verbose=False):
    """
    Full optimization pipeline for a single sequence:
    1. LSE gradient descent with adaptive beta and Nesterov momentum
    2. Peak chasing
    3. LP-based direction search
    4. Final gradient polish
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if verbose:
        print(f"    Start: n={n}, obj={best_obj:.8f}")

    # === Stage 1: LSE gradient with adaptive parameters ===
    beta = 3.0
    mu = 0.85
    v = np.zeros(n)
    eta = 0.15
    stall = 0
    max_stall = 40

    for it in range(800):
        elapsed = time.monotonic() - start
        if elapsed > time_budget * 0.55:
            break

        # Adaptive beta: start broad, narrow down
        beta = 3.0 * (1.0 + it / 100.0)
        beta = min(beta, 30.0)

        grad, M, conv = lse_gradient(a, beta)

        # Check if we should use top-k instead for more focused gradient
        if it > 200:
            grad, M, conv = soft_gradient(a, beta, top_k=3)

        # Center gradient
        grad_c = grad - np.mean(grad)

        # Nesterov momentum
        v = mu * v - eta * grad_c
        a_new = a - v
        a_new = project_simplex(a_new, 1.0)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        # Adaptive step size
        if stall > max_stall:
            eta *= 0.7
            mu = min(mu + 0.02, 0.93)
            stall = 0
            # Occasionally perturb
            if stall // max_stall > 3:
                rng = np.random.default_rng(it)
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = project_simplex(a, 1.0)
                best_a = a.copy()
                best_obj = compute_obj(a)
                v = np.zeros(n)

        eta *= 0.9995
        mu = min(mu * 1.0005, 0.92)

    if verbose:
        print(f"    After grad descent: obj={best_obj:.8f}")

    # === Stage 2: Peak chasing ===
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_chase = chop_peak(best_a, n_iters=min(200, int(remaining * 10)), base_step=0.06)
        obj_chase = compute_obj(a_chase)
        if obj_chase < best_obj:
            best_obj = obj_chase
            best_a = a_chase.copy()
            if verbose:
                print(f"    After chase: obj={best_obj:.8f}")

    # === Stage 3: LP direction ===
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10 and n <= 900:
        conv = compute_conv(best_a)
        M = float(np.max(conv))
        g = solve_lp_direction(best_a, M, time_limit=min(remaining * 0.4, 10))
        if g is not None:
            a_blend, obj_blend, t_used = line_search_blend(best_a, g)
            if obj_blend < best_obj:
                best_obj = obj_blend
                best_a = a_blend.copy()
                if verbose:
                    print(f"    After LP blend (t={t_used:.2f}): obj={best_obj:.8f}")

    # === Stage 4: Final short gradient pass ===
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        v = np.zeros(n)
        eta = 0.05
        mu = 0.8
        for it in range(int(remaining * 20)):
            if time.monotonic() - start > time_budget - 1:
                break
            grad, M, conv = lse_gradient(best_a, 15.0)
            grad_c = grad - np.mean(grad)
            v = mu * v - eta * grad_c
            a_new = project_simplex(best_a - v, 1.0)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
            else:
                best_a = a_new  # Still track for momentum
            eta *= 0.998

    return best_a, best_obj


def resample_to_length(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Loaded parent: n={parent.size}, obj={obj:.8f}")
            return parent.copy(), obj
    except NameError:
        pass
    except Exception as e:
        print(f"Error loading parent: {e}")
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization:
    1. Load and optimize parent sequence
    2. Perturbation-based restarts
    3. Length exploration
    4. Final intensive polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    else:
        best_seq = np.ones(500)
        best_score = compute_obj(best_seq)
        print(f"Default init: n=500, obj={best_score:.8f}")

    n_parent = len(best_seq)
    print(f"\nInitial best: n={n_parent}, score={best_score:.8f}")

    # === Phase 1: Optimize at parent length ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Optimize n={n_parent} (budget: 450s)")
    print(f"{'='*60}")

    a = best_seq.copy()
    S = np.sum(a)
    a = a / S

    alloc1 = min(450.0, (deadline - start_time) * 0.35)
    a_opt, obj_opt = optimize_single(a, time_budget=alloc1, verbose=True)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"Phase 1 result: {best_score:.8f}")

    # === Phase 2: Second optimization pass (with different params) ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Second pass on best (budget: 200s)")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        a2 = best_seq.copy()
        S2 = np.sum(a2)
        a2 = a2 / S2
        alloc2 = min(200.0, remaining * 0.15)
        a_opt2, obj_opt2 = optimize_single(a2, time_budget=alloc2, verbose=True)
        if obj_opt2 < best_score:
            best_score = obj_opt2
            best_seq = a_opt2.copy()
            print(f"Phase 2 result: {best_score:.8f}")

    # === Phase 3: Perturbation restarts ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Perturbation restarts")
    print(f"{'='*60}")

    rng_main = np.random.default_rng(seed)
    n_restarts = 8
    for r in range(n_restarts):
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Different perturbation strategies
        strategy = r % 4
        if strategy == 0:
            # Multiplicative noise
            scale = 0.04 + 0.02 * (r // 4)
            a_pert = best_seq * np.exp(rng_main.normal(0, scale, len(best_seq)))
        elif strategy == 1:
            # Additive noise with floor
            a_pert = best_seq + rng_main.normal(0, 0.01 * np.mean(best_seq), len(best_seq))
            a_pert = np.maximum(a_pert, 0)
        elif strategy == 2:
            # Swap-like perturbation: randomly redistribute some mass
            a_pert = best_seq.copy()
            n_swaps = max(3, len(a_pert) // 50)
            for _ in range(n_swaps):
                i = rng_main.integers(0, len(a_pert))
                j = rng_main.integers(0, len(a_pert))
                if i != j:
                    transfer = 0.05 * a_pert[i]
                    a_pert[i] -= transfer
                    a_pert[j] += transfer
            a_pert = np.maximum(a_pert, 0)
        else:
            # Smoothing + noise (simulates different frequency content)
            from scipy.ndimage import gaussian_filter1d
            a_smooth = gaussian_filter1d(best_seq, sigma=3.0)
            a_pert = a_smooth + 0.02 * np.mean(best_seq) * rng_main.standard_normal(len(best_seq))
            a_pert = np.maximum(a_pert, 0)

        S_p = np.sum(a_pert)
        if S_p < 1e-10:
            continue
        a_pert = a_pert / S_p

        obj_init = compute_obj(a_pert)
        alloc = min(remaining * 0.2, 70.0)
        a_opt_r, obj_opt_r = optimize_single(a_pert, time_budget=alloc, verbose=False)

        if obj_opt_r < best_score:
            best_score = obj_opt_r
            best_seq = a_opt_r.copy()
            print(f"  Restart {r} (strat={strategy}): {best_score:.8f} ** IMPROVED **")
        else:
            print(f"  Restart {r} (strat={strategy}): {obj_opt_r:.8f} (init={obj_init:.8f})")

    # === Phase 4: Length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Length exploration")
    print(f"{'='*60}")

    n_options = sorted(set([
        max(200, n_parent - 200),
        max(200, n_parent - 100),
        n_parent,
        n_parent + 100,
        n_parent + 200,
        min(2000, n_parent + 400),
    ]))

    for n_try in n_options:
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_to_length(best_seq, n_try)

        S_t = np.sum(a_try)
        if S_t > 1e-10:
            a_try = a_try / S_t

        obj_try_init = compute_obj(a_try)
        alloc = min(remaining * 0.25, 50.0)
        a_opt_n, obj_opt_n = optimize_single(a_try, time_budget=alloc, verbose=False)

        if obj_opt_n < best_score:
            best_score = obj_opt_n
            best_seq = a_opt_n.copy()
            n_parent = n_try
            print(f"  n={n_try}: {best_score:.8f} ** IMPROVED **")
        else:
            print(f"  n={n_try}: {obj_opt_n:.8f} (init={obj_try_init:.8f})")

    # === Phase 5: Final intensive polish ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Final intensive polish")
    print(f"{'='*60}")
    print(f"  Best before final: {best_score:.8f}")

    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_final = best_seq.copy()
        S_f = np.sum(a_final)
        a_final = a_final / S_f

        # Multiple short optimization rounds
        for round_idx in range(5):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            alloc = min(remaining * 0.3, 80.0)
            a_pol, obj_pol = optimize_single(a_final, time_budget=alloc, verbose=False)

            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol.copy()
                print(f"  Round {round_idx}: {best_score:.8f}")
            else:
                # Try a slightly different perturbation
                rng_r = np.random.default_rng(seed + 500 + round_idx)
                a_pert = a_final * np.exp(rng_r.normal(0, 0.02, len(a_final)))
                a_pert = project_simplex(a_pert, 1.0)
                a_pol2, obj_pol2 = optimize_single(a_pert, time_budget=min(alloc * 0.5, 30.0))
                if obj_pol2 < best_score:
                    best_score = obj_pol2
                    best_seq = a_pol2.copy()
                    a_final = a_pol2.copy()
                    print(f"  Round {round_idx} (pert): {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
