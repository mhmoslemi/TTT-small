import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_obj_and_grad(a, tau):
    """Log-sum-exp smoothed objective and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # For numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    logsumexp = b_max + np.log(exp_shifted.sum()) / tau
    
    # Gradient
    p = exp_shifted / exp_shifted.sum()
    grad = np.zeros(n)
    for i in range(n):
        k_lo = i
        k_hi = min(2*n-2, i + n - 1)
        if k_lo <= k_hi:
            grad[i] = 2.0 * np.dot(p[k_lo:k_hi+1], a[k_lo-i:k_hi-i+1])
    return logsumexp, grad

        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0
    
    affected = np.arange(k_lo, k_hi + 1)
    c_vals = np.zeros(len(affected))
    slopes = np.zeros(len(affected))
    quad_idx = -1
    
    for idx, k in enumerate(affected):
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            slopes[idx] = 0.0
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2 * x_cur * a[j]
                slopes[idx] = 2 * a[j]
            else:
                c_vals[idx] = b[k]
                slopes[idx] = 0.0
    
    def obj(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 0:
            return 1e10
        vals = c_vals + slopes * x_test
        m = base_max
        max_v = np.max(vals) if len(vals) > 0 else 0
        if max_v > m:
            m = max_v
        if quad_idx >= 0:
            qv = c_vals[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)
    
    # Golden section
    lo = 0.0
    hi = max(x_cur * 3.0, np.sqrt(max(b.max(), 0)) + 0.5, 0.01)
    
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(60):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)

def build_conv_matrix(n, a):
    """Build the convolution matrix A where (A @ g)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception as e:
        pass
    return None

def run_lp_descent(a, max_iters, deadline, time_limit=10):
    """LP-guided mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit)
        if g is None:
            break
        
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-12):
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
    
    return best_a

def resample(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage approach combining:
1. Projected smooth gradient descent (log-sum-exp) with adaptive temperature
2. Fine coordinate descent with exact golden-section subproblem solving
3. LP-guided mirror descent for large structural changes
4. Perturbation + re-optimization to escape local optima
5. Multi-length exploration
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_obj_grad(a, tau):
    """Compute log-sum-exp smoothed max of convolution and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -700, 700)
    exp_s = np.exp(shifted)
    logsumexp = b_max + np.log(exp_s.sum()) / tau
    p = exp_s / exp_s.sum()
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    grad = np.zeros(n)
    for i in range(n):
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        if k_lo <= k_hi:
            grad[i] = 2.0 * np.dot(p[k_lo:k_hi + 1], a[0:k_hi - k_lo + 1])
    return logsumexp, grad


def run_smooth_gradient(a, max_iters, deadline, tau_start=5.0, tau_end=300.0,
                        lr=0.1, momentum=0.9, verbose=False):
    """Projected gradient descent with momentum on smooth max objective."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)  # momentum

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        lr_t = lr / (1 + 0.05 * frac * max_iters ** 0.5)

        _, grad = smooth_obj_grad(a, tau)
        # Project to tangent space (sum-zero)
        grad = grad - grad.mean()

        # Momentum update
        v = momentum * v + lr_t * grad
        a_new = a - v

        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()

    return best_a


def coord_update(a, i, b_cache):
    """Exact 1D coordinate update via golden-section search."""
    n = len(a)
    x_cur = a[i]
    S = a.sum()

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # Unaffected max
    base_max = 0.0
    if k_lo > 0:
        base_max = max(base_max, b_cache[:k_lo].max())
    if k_hi < 2 * n - 2:
        base_max = max(base_max, b_cache[k_hi + 1:].max())

    # Affected part
    affected = np.arange(k_lo, k_hi + 1)
    c_vals = np.empty(len(affected))
    slopes = np.empty(len(affected))
    quad_idx = -1

    for idx, k in enumerate(affected):
        if k == 2 * i:
            c_vals[idx] = b_cache[k] - x_cur * x_cur
            slopes[idx] = 0.0
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b_cache[k] - 2 * x_cur * a[j]
                slopes[idx] = 2.0 * a[j]
            else:
                c_vals[idx] = b_cache[k]
                slopes[idx] = 0.0

    def obj(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 1e-15:
            return 1e20
        m = base_max
        if len(c_vals) > 0:
            m = max(m, np.max(c_vals + slopes * x_test))
        if quad_idx >= 0:
            m = max(m, c_vals[quad_idx] + x_test * x_test)
        return m / (new_sum * new_sum)

    # Golden section search
    lo = 0.0
    hi = max(x_cur * 3.0, np.sqrt(max(b_cache.max(), 0)) + 0.5, 0.01)
    if hi > 1.0:
        hi = 1.0  # since we work with normalized sequences

    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)

    for _ in range(60):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coord_descent(a, max_sweeps, deadline, seed=0):
    """Coordinate descent with periodic fresh convolution."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        b_cache = fast_conv(a)
        order = np.arange(n)
        rng.shuffle(order)

        improved = False
        for i in order:
            x_opt = coord_update(a, i, b_cache)
            if abs(x_opt - a[i]) > 1e-13:
                a[i] = x_opt
                improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not improved:
            no_improve += 5
        if no_improve > 30:
            break

    return best_a


def build_conv_matrix(n, a):
    """Build sparse matrix A where (A @ g) = (a * g)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def run_lp_descent(a, max_iters, deadline, time_limit=10):
    """LP-guided mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit)
        if g is None:
            break

        improved = False
        for t in [0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.02]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-13):
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 8:
                break
        else:
            no_improve = 0

    return best_a


def resample_seq(src, n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={time.monotonic()-self.t0:.0f}s {label}", flush=True)

    def report(self):
        if self.best is not None:
            print(f"  Best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t0:.0f}s", flush=True)


def try_smoothing_refine(a, deadline, seed=0):
    """Try multiple tau values for smooth gradient refinement."""
    rng = np.random.default_rng(seed)
    best = a.copy()
    best_M = fast_conv(a).max()

    for tau in [8, 15, 30, 60, 120, 250]:
        if time.monotonic() > deadline:
            break
        a_curr = run_smooth_gradient(best, 100, deadline, tau_start=tau * 0.3, tau_end=tau * 2, lr=0.05)
        M = fast_conv(a_curr).max()
        if M < best_M:
            best_M = M
            best = a_curr.copy()

    return best


def try_block_perturbation(a, deadline, seed=0, n_trials=10):
    """Try random block perturbations and re-optimize."""
    rng = np.random.default_rng(seed)
    n = len(a)
    best = a.copy()
    best_M = fast_conv(a).max()

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        # Create a perturbed version
        a_pert = a.copy()
        n_blocks = rng.integers(1, 4)
        for _ in range(n_blocks):
            block_size = rng.integers(4, max(5, n // 8))
            start = rng.integers(0, max(1, n - block_size))
            scale = rng.uniform(0.5, 1.5)
            a_pert[start:start + block_size] *= scale

        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        # Quick coordinate descent on perturbed
        budget = min(40, int((deadline - time.monotonic()) / max(1, n_trials - trial)))
        a_opt = run_coord_descent(a_pert, budget, deadline, seed=seed + trial * 13)
        M = fast_conv(a_opt).max()
        if M < best_M:
            best_M = M
            best = a_opt.copy()

    return best


def try_pair_exchange(a, deadline, seed=0):
    """Try exchanging blocks between positions and re-optimizing."""
    rng = np.random.default_rng(seed)
    n = len(a)
    best = a.copy()
    best_M = fast_conv(a).max()
    n_trials = 15

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        a_pert = a.copy()
        block_len = rng.integers(4, max(5, n // 10))
        # Pick two non-overlapping positions
        i = rng.integers(0, max(1, n - 2 * block_len))
        j = rng.integers(i + block_len, max(i + block_len + 1, n - block_len))
        if j + block_len > n:
            continue
        # Swap
        a_pert[i:i + block_len], a_pert[j:j + block_len] = a_pert[j:j + block_len].copy(), a_pert[i:i + block_len].copy()
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        budget = 20
        a_opt = run_coord_descent(a_pert, budget, deadline, seed=seed + 500 + trial)
        M = fast_conv(a_opt).max()
        if M < best_M:
            best_M = M
            best = a_opt.copy()

    return best


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-stage optimization:
    Phase 1: Smooth gradient descent with adaptive temperature
    Phase 2: Fine coordinate descent (main optimization)
    Phase 3: LP-guided mirror descent
    Phase 4: Perturbation + re-optimization (multiple strategies)
    Phase 5: Multi-length exploration
    Phase 6: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
    tracker.update(parent, "parent")
    print(f"Starting: n={parent.size}, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size

    # === PHASE 1: Smooth gradient (150s) ===
    print("\n=== PHASE 1: Smooth Gradient Descent ===", flush=True)
    p1_deadline = min(time.monotonic() + 120, deadline - 800)

    a_s1 = try_smoothing_refine(parent, p1_deadline, seed=seed)
    tracker.update(a_s1, "smooth_grad")

    # === PHASE 2: Coordinate Descent (350s) ===
    print("\n=== PHASE 2: Coordinate Descent ===", flush=True)
    p2_deadline = min(time.monotonic() + 350, deadline - 350)

    a_cd = run_coord_descent(tracker.best, 80, p2_deadline, seed=1)
    tracker.update(a_cd, "coord_descent_main")

    # Additional CD passes with different orderings
    a_cd2 = run_coord_descent(tracker.best, 60, p2_deadline, seed=42)
    tracker.update(a_cd2, "coord_descent_pass2")

    # === PHASE 3: LP Descent (300s) ===
    print("\n=== PHASE 3: LP Mirror Descent ===", flush=True)
    p3_deadline = min(time.monotonic() + 300, deadline - 150)

    if tracker.best is not None and time.monotonic() < p3_deadline:
        n = tracker.best.size
        time_limit = min(15, max(2, int(20 * (128.0 / n) ** 1.5)))
        print(f"  n={n}, lp_time_limit={time_limit}s", flush=True)

        a_lp = run_lp_descent(tracker.best, 25, p3_deadline, time_limit=time_limit)
        tracker.update(a_lp, "lp_descent")

        # Follow up with CD after LP
        if tracker.best is not None and time.monotonic() < p3_deadline:
            a_cd_post = run_coord_descent(tracker.best, 40, p3_deadline, seed=7)
            tracker.update(a_cd_post, "post_lp_cd")

    # === PHASE 4: Perturbation strategies (200s) ===
    print("\n=== PHASE 4: Perturbation + Reoptimize ===", flush=True)
    p4_deadline = min(time.monotonic() + 200, deadline - 80)

    if tracker.best is not None and time.monotonic() < p4_deadline:
        # Random perturbations
        a_pert = try_block_perturbation(tracker.best, p4_deadline, seed=1000, n_trials=8)
        tracker.update(a_pert, "block_perturb")

        # Pair exchanges
        a_pair = try_pair_exchange(tracker.best, p4_deadline, seed=2000)
        tracker.update(a_pair, "pair_exchange")

        # Small noise + re-optimize
        for noise_scale in [0.005, 0.01, 0.02]:
            if time.monotonic() > p4_deadline:
                break
            a_noisy = tracker.best + rng.normal(0, noise_scale, tracker.best.size)
            a_noisy = np.maximum(a_noisy, 0)
            a_noisy = a_noisy / a_noisy.sum()
            a_opt = run_coord_descent(a_noisy, 30, min(time.monotonic() + 30, p4_deadline), seed=3000)
            tracker.update(a_opt, f"noise_{noise_scale}")

    # === PHASE 5: Multi-length exploration (100s) ===
    print("\n=== PHASE 5: Multi-Length ===", flush=True)
    p5_deadline = min(time.monotonic() + 100, deadline - 30)

    if tracker.best is not None and time.monotonic() < p5_deadline:
        n_best = tracker.best.size
        candidates_n = []
        for mult in [0.75, 1.0, 1.25, 1.5, 2.0]:
            new_n = int(round(n_best * mult))
            if new_n >= 128 and new_n <= 2048 and new_n != n_best:
                candidates_n.append(new_n)

        for new_n in candidates_n[:3]:  # Limit to 3 to save time
            if time.monotonic() > p5_deadline:
                break
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 1e-6)
            a_res = a_res / a_res.sum()
            a_opt = run_coord_descent(a_res, 30, min(time.monotonic() + 25, p5_deadline), seed=5000)
            tracker.update(a_opt, f"n={new_n}")

    # === PHASE 6: Final polish (remaining time) ===
    print("\n=== PHASE 6: Final Polish ===", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 5:
        # Final CD
        a_final = run_coord_descent(tracker.best, 80, deadline - 15, seed=9999)
        tracker.update(a_final, "final_cd")

        # Final smooth gradient
        if time.monotonic() < deadline - 10:
            a_smooth = run_smooth_gradient(tracker.best, 50, deadline - 10,
                                           tau_start=50, tau_end=200, lr=0.02)
            tracker.update(a_smooth, "final_smooth")

        # Final LP if time allows
        if time.monotonic() < deadline - 8:
            n = tracker.best.size
            tl = min(5, max(1, int(10 * (128.0 / n) ** 1.5)))
            a_lp_final = run_lp_descent(tracker.best, 5, deadline, time_limit=tl)
            tracker.update(a_lp_final, "final_lp")

    # One more CD if best changed
    if tracker.best is not None and time.monotonic() < deadline - 3:
        a_last = run_coord_descent(tracker.best, 30, deadline, seed=12345)
        tracker.update(a_last, "very_final_cd")

    tracker.report()

    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n=== FINAL: J={tracker.best_score:.8f} n={tracker.best.size} t={time.monotonic()-tracker.t0:.0f}s ===", flush=True)
    return [float(x) for x in tracker.best]
