import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix_sparse(n, a):
    """Build sparse constraint matrix A for (a*g)(k) <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((np.array(vals), (np.array(rows), np.array(cols))),
                   shape=(2 * n - 1, n))
    return A

    n = len(a)
    if active_lags is None:
        active_lags = range(2 * n - 1)
    active_lags = list(active_lags)
    if not active_lags:
        return None
    
    rows = []
    cols = []
    vals = []
    b_ub = []
    
    for k in active_lags:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        b_ub.append(M)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((np.array(vals, dtype=np.float64),
                    (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))),
                   shape=(2 * n - 1, n))
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=np.array(b_ub, dtype=np.float64),
                    bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def get_active_lags(b, frac=0.7):
    """Get lags where b[k] > frac * max(b)."""
    M = b.max()
    threshold = M * frac
    return [k for k in range(len(b)) if b[k] > threshold]


def lp_mirror_descent_v2(a, deadline, use_active_set=True, active_frac=0.7,
                         momentum_beta=0.5, max_iters=3000):
    """
    LP mirror descent with active-set LP and momentum.
    """
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    v = np.zeros(n)  # momentum
    no_improve = 0
    iter_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        iter_count += 1

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        # Active set: only constrain lags near the maximum
        if use_active_set and n > 64:
            active = get_active_lags(b_aa, active_frac)
            if len(active) < 10:
                active = list(range(len(b_aa)))
        else:
            active = list(range(2 * n - 1))

        g = lp_best_response(a, M, active_lags=active)
        if g is None:
            # Fallback: full LP
            g = lp_best_response(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Line search
        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)

        def obj_line(t):
            b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
            return b.max()

        result = minimize_scalar(obj_line, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-8})
        t_opt = float(result.x)

        # Nesterov momentum
        a_candidate = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_candidate.sum()
        if s < 1e-10:
            a_candidate = a
            t_opt = 0
        else:
            a_candidate = a_candidate / s

        J_new = compute_J(a_candidate)

        if J_new < best_J - 1e-14:
            v = momentum_beta * v + (a_candidate - a)
            best_a = a_candidate.copy()
            best_J = J_new
            a = a_candidate
            no_improve = 0
        else:
            no_improve += 1
            # Try with momentum
            a_mom = a + 0.1 * v
            a_mom = np.maximum(a_mom, 0)
            sm = a_mom.sum()
            if sm > 0.01:
                a_mom = a_mom / sm
                J_mom = compute_J(a_mom)
                if J_mom < best_J - 1e-14:
                    best_a = a_mom.copy()
                    best_J = J_mom
                    a = a_mom
                    no_improve = 0
            v = momentum_beta * v

        if no_improve > 40:
            # Perturb
            a = a + np.random.normal(0, 0.03, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_J


def coordinate_descent_exact(a, deadline, max_passes=30):
    """
    Coordinate descent with exact 1D minimization per coordinate.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        S = a.sum()
        improved = False

        # Random order for diversity
        order = np.random.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            if ai < 1e-14:
                # Try small positive values
                candidates = [0.005, 0.01, 0.02, 0.05, 0.1]
                best_cand = 0.0
                best_score = compute_J(a)
                for c in candidates:
                    a[i] = c
                    sc = compute_J(a)
                    if sc < best_score:
                        best_score = sc
                        best_cand = c
                a[i] = best_cand
                if best_cand > 0:
                    improved = True
                continue

            # Find optimal value for a[i] using minimize_scalar
            k_lo = max(0, i)
            k_hi = min(i + n - 1, 2 * n - 2)

            # Precompute outside max
            if k_lo > 0:
                outside_left = b[:k_lo].max() if k_lo < len(b) else 0
            else:
                outside_left = 0
            if k_hi < 2 * n - 2:
                outside_right = b[k_hi + 1:].max() if k_hi + 1 < len(b) else 0
            else:
                outside_right = 0
            outside_max = max(outside_left, outside_right)

            def obj_delta(delta):
                S_new = S + delta
                if S_new < 0.01:
                    return float('inf')
                max_val = outside_max
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j < 0 or j >= n:
                        continue
                    if k == 2 * i:
                        new_val = b[k] + 2 * delta * a[i] + delta * delta
                    else:
                        new_val = b[k] + 2 * delta * a[j]
                    if new_val > max_val:
                        max_val = new_val
                return 2 * n * max_val / (S_new * S_new)

            # Bracket: delta in [-ai, some_upper]
            lo_bound = -ai + 1e-12
            hi_bound = max(0.5, ai * 3)

            result = minimize_scalar(obj_delta, bounds=(lo_bound, hi_bound),
                                     method='bounded', options={'xatol': 1e-8})
            best_delta = float(result.x)
            new_ai = max(0.0, ai + best_delta)

            if abs(new_ai - ai) > 1e-12:
                a[i] = new_ai
                improved = True
                # Update b incrementally
                delta = new_ai - ai
                S = S + delta
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j < 0 or j >= n:
                        continue
                    if k == 2 * i:
                        b[k] += 2 * delta * new_ai + delta * delta - 2 * delta * ai - delta * delta
                        # Actually: old contribution was 2*ai*ai=2ai^2... 
                        # Let me just recompute properly
                        # b[k] was computed with old a[i]. New a[i] = new_ai.
                        # Change: b[k] += delta * (2*a[i]_old + delta) for k=2i
                        # Actually the change in b[k] when a[i] changes by delta:
                        # For k=2i: b_new = b_old + 2*delta*a_old[i] + delta^2
                        # For k!=2i: b_new = b_old + 2*delta*a[k-i]
                        pass

        # Recompute b to avoid drift
        b = fast_conv(a)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J - 1e-14:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def cd_fast_vectorized(a, deadline, max_passes=20):
    """
    Faster CD: recompute conv each pass, use vectorized objective.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        S = a.sum()
        improved = False
        order = np.random.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]

            # Determine range of affected lags
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Precompute the "partner" values a[k-i] for k in [k_lo, k_hi]
            j_vals = np.arange(k_lo, k_hi + 1) - i
            j_vals = j_vals[(j_vals >= 0) & (j_vals < n)]
            k_valid = j_vals + i
            a_partners = a[j_vals]

            def obj_delta(delta):
                S_new = S + delta
                if S_new <= 0.01:
                    return 1e10
                # Compute new b values for affected lags
                new_vals = b[k_valid] + 2.0 * delta * a_partners
                # Handle k = 2i case
                if k_lo <= 2 * i <= k_hi:
                    idx = 2 * i - k_lo
                    new_vals[idx] = b[2 * i] + 2.0 * delta * ai + delta * delta
                # Outside unaffected
                max_val = new_vals.max()
                if k_lo > 0:
                    max_val = max(max_val, b[:k_lo].max())
                if k_hi < 2 * n - 2:
                    max_val = max(max_val, b[k_hi + 1:].max())
                return 2.0 * n * max_val / (S_new * S_new)

            lo_b = -ai + 1e-10
            hi_b = max(1.0, ai * 5.0)

            res = minimize_scalar(obj_delta, bounds=(lo_b, hi_b),
                                  method='bounded', options={'xatol': 1e-7})
            new_ai = max(0.0, ai + float(res.x))

            if abs(new_ai - ai) > 1e-11:
                a[i] = new_ai
                improved = True

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def parametric_search(n, deadline):
    """Try various parametric families to find good initializations."""
    results = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Cosine families: a_i = (1 + alpha*cos(2*pi*i/n + phi))^beta
    for alpha in [0.3, 0.5, 0.7, 0.9, 1.0]:
        for phi in [0, np.pi/4, np.pi/2, 3*np.pi/4, np.pi]:
            for beta in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
                if time.monotonic() > deadline:
                    return results
                base = 1.0 + alpha * np.cos(2 * np.pi * t + phi)
                base = np.maximum(base, 1e-6)
                a = base ** beta
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 0.01:
                    a = a / s
                    J = compute_J(a)
                    results.append((J, a.copy()))

    # Gaussian families
    for center in [0.25, 0.4, 0.5, 0.6, 0.75]:
        for width in [0.1, 0.15, 0.2, 0.3, 0.4]:
            if time.monotonic() > deadline:
                return results
            a = np.exp(-(t - center) ** 2 / (2 * width ** 2))
            a = a + 0.01
            a = a / a.sum()
            J = compute_J(a)
            results.append((J, a.copy()))

    # Power law from edges
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        if time.monotonic() > deadline:
            return results
        dist = np.minimum(t, 1 - t)
        a = dist ** p + 0.01
        a = a / a.sum()
        J = compute_J(a)
        results.append((J, a.copy()))

    # Inverted power law (bowl)
    for p in [0.5, 1.0, 1.5, 2.0]:
        if time.monotonic() > deadline:
            return results
        dist = np.abs(t - 0.5)
        a = dist ** p + 0.05
        a = a / a.sum()
        J = compute_J(a)
        results.append((J, a.copy()))

    # Frequency-modulated
    for freq in [1, 2, 3, 4, 6, 8]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            if time.monotonic() > deadline:
                return results
            a = 1.0 + amp * np.cos(2 * np.pi * freq * t)
            a = np.maximum(a, 0.01)
            a = a / a.sum()
            J = compute_J(a)
            results.append((J, a.copy()))

    # W-shaped
    for freq in [2, 3, 4]:
        if time.monotonic() > deadline:
            return results
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        a = a ** 2 + 0.01
        a = a / a.sum()
        J = compute_J(a)
        results.append((J, a.copy()))

    # Sorted
    results.sort(key=lambda x: x[0])
    return results


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def peak_shaving(a, deadline, n_iters=200):
    """
    Directly target the top autocorrelation peaks:
    For each peak lag k*, try to reduce conv(a,k*) by adjusting contributing a's.
    """
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        S = a.sum()
        M = b.max()

        # Find the top lags
        top_lags = np.argsort(b)[-min(5, n):][::-1]

        improved = False
        for k in top_lags:
            if time.monotonic() > deadline:
                break

            # conv(k) = sum_{i+j=k} a[i]*a[j]
            # To reduce it, decrease the a's that contribute most
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)

            # For each i in [i_min, i_max], the partner is j = k - i
            # The product a[i]*a[k-i] contributes to b[k]
            # Try decreasing a[i] and increasing a[i] at a "quiet" position

            # Simple approach: try reducing the largest a[i] among contributing indices
            contributing = [(a[i], i) for i in range(i_min, i_max + 1)
                           if 0 <= k - i < n]
            if not contributing:
                continue

            contributing.sort(reverse=True)
            ai_val, ai_idx = contributing[0]

            # Try a few reduction factors
            for factor in [0.7, 0.5, 0.3]:
                new_val = ai_val * factor
                delta = new_val - ai_val
                S_new = S + delta
                if S_new < 0.01:
                    continue

                a[ai_idx] = new_val
                a_new = np.maximum(a, 0)
                s_check = a_new.sum()
                if s_check < 0.01:
                    a[ai_idx] = ai_val
                    continue
                J_new = compute_J(a_new / s_check)
                if J_new < best_J - 1e-12:
                    best_J = J_new
                    best_a = (a_new / s_check).copy()
                    a = (a_new / s_check).copy()
                    S = a.sum()
                    improved = True
                    break
                else:
                    a[ai_idx] = ai_val

        if not improved:
            # Perturb
            a = a + np.random.normal(0, 0.01, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()

    return best_a


def propose_candidate(seed=42, budget_s=1500):

    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.

    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent)

    # === PHASE 2: LP mirror descent from best ===

        a_f = lp_mirror_descent_v2(
            best_seq, min(t_start + budget_s - 10, deadline),
            use_active_set=True, active_frac=0.4,
            momentum_beta=0.2, max_iters=500)
        update_best(a_f)

        # Final CD
        sub_dl = min(time.monotonic() + 20, deadline)
        a_cd = cd_fast_vectorized(best_seq, sub_dl, max_passes=10)
        update_best(a_cd)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        best_score = compute_J(best_seq)
        update_best(best_seq)

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
