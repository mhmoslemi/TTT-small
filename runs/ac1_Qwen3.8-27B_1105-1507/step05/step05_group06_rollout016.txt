"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based mirror descent with two formulations (max-mass and min-peak)
2. Exact 1D coordinate descent using scipy.minimize_scalar
3. Subgradient method with Polyak step size
4. Simulated annealing for global exploration
5. Multiple initializations including "flat autocorrelation" sequence

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x) = target_sum, x >= 0}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    # Ensure sum is exactly target_sum (handle numerical issues)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference to positive entries
        pos = result > 0
        if pos.sum() > 0:
            result[pos] += diff / pos.sum()
            result = np.maximum(result, 0)
    return result

def lp_max_mass(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    # Build constraint matrix: A[k,j] = a[k-j] if valid
    # (a*g)[k] = sum_j a[k-j] * g[j]
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def lp_min_peak(a, b):
    """LP: min t s.t. 2*(g*a)[k] - b[k] <= t, sum(g)=1, g>=0.
    This is the linearization of the autocorrelation at point a."""
    n = len(a)
    # Variables: g_0,...,g_{n-1}, t (n+1 variables)
    # Constraints: 2*(g*a)[k] - t <= b[k] for k=0,...,2n-2
    #              sum(g) = 1
    #              g >= 0, t free
    
    # The constraint matrix for the inequalities:
    # For each k: sum_j 2*a[k-j]*g_j - t <= b[k]
    # In standard form: A_ub @ x <= b_ub where x = [g, t]
    
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)  # coefficient of g_j
            vals.append(2 * a[k - j])
        # coefficient of t is -1
        rows.append(k)
        cols.append(n)  # t is the (n+1)-th variable
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub_vec = b.copy()  # RHS is b[k]
    
    # Objective: minimize t → c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0, t free
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub_vec, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            if g.sum() > 1e-15:
                return g / g.sum()  # normalize
    except Exception:
        pass
    return None

def line_search_bilinear(b_aa, b_ag, b_gg):
    """Find t in [0,1] minimizing max((1-t)^2*b_aa + 2t(1-t)*b_ag + t^2*b_gg)."""
    def obj(t):
        b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
        return b.max()
    
    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                             options={'xatol': 1e-10})
    return float(result.x)

def subgradient_method(a, deadline, max_iters=10000):
    """
    Subgradient method with Polyak step size for minimizing max(conv(a,a)) s.t. sum(a)=1, a>=0.
    """
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)
    
    # Best known upper bound on max(b)
    best_M = fast_conv(a).max()
    current_a = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(current_a)
        M = b.max()
        
        if M < best_M:
            best_M = M
        
        # Polyak step size: alpha = (M - M_lower) / ||g||^2
        # M_lower is a lower bound on the optimal value of max(b)
        # A simple lower bound: max(b) >= S^2/(2n-1) where S = sum(b) = (sum(a))^2 = 1
        # So M_lower = 1/(2n-1)
        M_lower = 1.0 / (2 * n - 1)
        
        # Subgradient: average of gradients at active indices
        active = np.where(b >= M - 1e-12)[0]
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2 * current_a[j]
        grad /= len(active)
        
        grad_norm_sq = np.sum(grad**2)
        if grad_norm_sq < 1e-20:
            break
        
        alpha = (M - M_lower) / grad_norm_sq
        alpha = min(alpha, 0.1)  # Cap the step size
        
        # Gradient step
        a_new = current_a - alpha * grad
        
        # Project onto simplex
        a_new = project_simplex(a_new, 1.0)
        
        current_a = a_new
    
    return current_a

def generate_flat_autocorr_init(n):
    """
    Generate initialization based on the 'flat autocorrelation' sequence.
    a[k] = binom(2k, k) / 4^k (truncated to length n)
    This minimizes the autocorrelation peak for the infinite-length case.
    """
    a = np.zeros(n)
    for k in range(n):
        # binom(2k, k) / 4^k
        if k == 0:
            a[k] = 1.0
        else:
            # Use log-space to avoid overflow
            log_val = 0.0
            for i in range(1, k + 1):
                log_val += np.log(2*k - i + 1) - np.log(i)
            log_val -= k * np.log(4)
            a[k] = np.exp(log_val)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a
