import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    """
    Compute gradient of smooth max of conv(a,a) w.r.t. a.
    Returns (grad, R, p) where grad = d/d a [1/tau * log(sum(exp(tau*R_k)))]
    and p_k = softmax(tau*R).
    """
    n = len(a)
    R = fast_conv(a)
    R_max = R.max()
    scaled = tau * (R - R_max)
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(R)
        p[np.argmax(R)] = 1.0
    else:
        p /= p_sum
    
    # grad[i] = 2 * sum_{l=0}^{n-1} a[l] * p[i+l]
    # = 2 * correlate(a, p, 'full')[2n-2 + i]
    # Use FFT for speed
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    # correlate(a, p) = ifft(conj(fft(a)) * fft(p))
    corr = np.fft.irfft(np.conj(fa) * fp, N)
    # Offset: correlate(a,p,'full') has length len(a)+len(p)-1 = n+(2n-1)-1 = 3n-2
    # We want indices [2n-2] to [2n-2+n-1] = [2n-2] to [3n-3]
    start = 2 * n - 2
    grad = 2.0 * corr[start:start + n]
    
    return grad, R, p


def nesterov_smooth_descent(a, deadline, max_iters=5000, tau_init=15.0, tau_end=80.0,
                            beta=0.9, lr_init=0.15, verbose=False):

    Nesterov-accelerated smooth max gradient descent with adaptive temperature.
    """

    a = a / a.sum()

    best_a = a.copy()
    best_J = compute_J(a)
    
    v = np.zeros(n)
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if time.monotonic() > deadline - 0.5:
            break
        
        # Adaptive temperature
        progress = min(1.0, it / float(max_iters))
        tau = tau_init + (tau_end - tau_init) * progress ** 0.7
        
        # Compute gradient at Nesterov lookahead point
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 1e-10)
        grad, R, _ = smooth_max_grad(y, tau)
        
        # Nesterov update
        v = beta * v + grad
        lr = lr_init / (1.0 + 0.05 * progress)

            a_new = a
        else:
            a_new = a_new / s
        
        a_prev = a
        a = a_new

            best_a = a.copy()

def build_lp_matrix_reduced(n, a, active_k):
    """Build sparse matrix for LP with only active constraints."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))


def solve_lp(a, M, top_frac=0.5, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for active k, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    
    # Select top-fraction active constraints
    num_active = max(10, int(len(b) * top_frac))
    active_idx = np.argsort(b)[-num_active:]
    
    A = build_lp_matrix_reduced(n, a, active_idx)
    c = -np.ones(n)
    b_ub = np.full(len(active_idx), M)
    bounds = [(0, np.inf)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                # Verify against ALL constraints
                ag = fast_conv(a)  # just for checking max
                # Actually need cross-conv
                N = 1
                while N < 2 * n - 1:
                    N <<= 1
                fa = np.fft.rfft(a, N)
                fg = np.fft.rfft(g, N)
                cross = np.fft.irfft(fa * fg, N)[:2 * n - 1]
                if cross.max() <= M * (1.0 + 1e-4):
                    return g
    except Exception:
        pass
    return None

        else:
            no_improve = 0

        improved = False

def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    # Truncated exponential (concentrated in center)

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 best: J={candidates[0][0]:.8f} n={candidates[0][2]}", flush=True)
    
    # === PHASE 2: LP Mirror Descent ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: LP Mirror Descent Refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    p2_deadline = t_start + 1050.0
    
    # Take top unique-length candidates
    seen_lengths = set()
    top_cands = []
    for J_val, a_opt, n_try in candidates:
        if n_try not in seen_lengths:
            top_cands.append((J_val, a_opt, n_try))
            seen_lengths.add(n_try)
        if len(top_cands) >= 4:
            break
    
    # Also include the overall best regardless of length
    if candidates and candidates[0][1] is not best_sequence:
        top_cands.append((candidates[0][0], candidates[0][1], candidates[0][2]))
    
    total_p2_time = p2_deadline - time.monotonic()
    num_cands = len(top_cands)
    time_per_cand = total_p2_time / max(1, num_cands)
    
    for ci, (J_init, a_init, n_c) in enumerate(top_cands):
        if time.monotonic() > p2_deadline - 10:
            break
        
        cand_dl = min(time.monotonic() + time_per_cand * 1.5, p2_deadline - 10)
        
        print(f"\n  LP cand {ci}: J_init={J_init:.8f} n={n_c} budget={time_per_cand:.0f}s", flush=True)
        
        a_best, J_best = lp_mirror_descent(
            a_init, cand_dl, top_frac=0.5,
            max_no_improve=120, verbose=(ci == 0)
        )
        update_best(a_best, f"LP n={n_c} cand{ci}")
    
    # === PHASE 3: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    p3_deadline = min(t_start + 1350.0, deadline - 80.0)
    
    if best_sequence is not None and time.monotonic() < p3_deadline:
        base_n = best_sequence.size
        base_J = best_score
        
        # Try nearby and scaled lengths
        lengths = set()
        for factor in [0.75, 1.0, 1.25, 1.5, 2.0]:
            ln = int(base_n * factor)
            if 128 <= ln <= 2048:
                lengths.add(ln)
        # Add some fixed good lengths
        for ln in [256, 512, 768, 1024]:
            lengths.add(ln)
        lengths.discard(base_n)
        
        lengths = sorted(lengths)
        
        for n_try in lengths:
            if time.monotonic() > p3_deadline:
                break
            
            a_ext = resample_seq(best_sequence, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            
            # Quick gradient
            quick_dl = min(time.monotonic() + 15.0, p3_deadline - 30.0)
            a_ext, J_quick = nesterov_smooth_descent(
                a_ext, quick_dl, max_iters=500,
                tau_init=15.0, tau_end=50.0,
                beta=0.85, lr_init=0.10, verbose=False
            )
            update_best(a_ext, f"explor n={n_try}")
            
            # Short LP
            if time.monotonic() < p3_deadline - 10:
                lp_dl = min(time.monotonic() + 30.0, p3_deadline - 5.0)
                a_ext2, J_lp = lp_mirror_descent(
                    a_ext, lp_dl, top_frac=0.6,
                    max_no_improve=80, verbose=False
                )
                update_best(a_ext2, f"explor n={n_try} LP")
    
    # === PHASE 4: Final polishing ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Final Polishing", flush=True)
    print(f"{'='*60}", flush=True)
    
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s", flush=True)
    
    if best_sequence is not None and remaining > 30:
        # Full polynomial-time LP (larger top_frac for more constraints)
        a_polish, J_polish = lp_mirror_descent(
            best_sequence, deadline - 5, top_frac=0.8,
            max_no_improve=400, verbose=True
        )
        update_best(a_polish, "polish_full")
        
        # One more pass with even more constraints
        if time.monotonic() < deadline - 10:
            a_polish2, J_polish2 = lp_mirror_descent(
                best_sequence, deadline - 5, top_frac=1.0,
                max_no_improve=200, verbose=False
            )
            update_best(a_polish2, "polish_all")
        
        # Try palindrome version
        if time.monotonic() < deadline - 20:
            a_pal = make_palindrome(best_sequence)
            a_pal = np.maximum(a_pal, 1e-6)
            a_pal = a_pal / a_pal.sum()
            a_pal, J_pal = lp_mirror_descent(
                a_pal, min(time.monotonic() + 30, deadline - 5),
                top_frac=0.7, max_no_improve=100, verbose=False
            )
            update_best(a_pal, "palindrome")
    
    # Fallback
    if best_sequence is None:
        best_sequence = parent
        best_score = compute_J(best_sequence)
    
    # Final Nesterov polish on the best
    if time.monotonic() < deadline - 5:
        a_final, J_final = nesterov_smooth_descent(
            best_sequence, deadline - 2, max_iters=1000,
            tau_init=30.0, tau_end=100.0,
            beta=0.9, lr_init=0.05, verbose=False
        )
        update_best(a_final, "final_NSD")
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={len(best_sequence)} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_sequence]
