        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def hankel_matvec(w, x):
    """Compute Q_w @ x where Q_w[j,i] = w[j+i]. Uses FFT."""
    n = len(x)
    # (Q_w @ x)[j] = sum_i w[j+i] * x[i]
    # This is the cross-correlation of w and x at lag j
    # corr[j] = sum_i w[j+i]*x[i] for j=0..n-1
    # = conv(flip(w), x)[len(w)-1-j]
    w_flip = w[::-1]
    size = 1
    while size < len(w_flip) + n - 1:
        size <<= 1
    W = np.fft.rfft(w_flip, size)
    X = np.fft.rfft(x, size)
    result = np.fft.irfft(W * X, size)[:len(w_flip) + n - 1]
    # result[k] = conv(w_flip, x)[k]
    # We want corr[j] = result[len(w)-1-j] = result[2n-2-j]
    corr = result[2*n-2 : n-1 : -1]
    return corr

def solve_qp_hankel(w, warm_start=None, max_iters=500):
    """
    Solve min a^T Q_w a s.t. sum(a)=1, a>=0
    where Q_w[j,i] = w[j+i] (Hankel matrix)
    Uses active-set + conjugate gradient.
    """
    n = len(w) // 2 + 1  # n such that w has length 2n-1
    # Actually, w has length 2n-1, so n = (len(w)+1)//2
    n = (len(w) + 1) // 2
    
    if warm_start is not None:
        a = warm_start.copy()
        active = a > 1e-10
    else:
        a = np.ones(n) / n
        active = np.ones(n, dtype=bool)
    
    for outer in range(50):
        S = np.where(active)[0]
        nS = len(S)
        if nS == 0:
            a = np.ones(n) / n
            return a
        
        # Solve the equality-constrained problem on the active set:
        # min a_S^T Q_SS a_S s.t. 1^T a_S = 1
        # KKT: Q_SS a_S + lambda * 1 = 0, 1^T a_S = 1
        # => a_S = -lambda * Q_SS^{-1} * 1
        # => 1^T a_S = -lambda * 1^T Q_SS^{-1} 1 = 1
        # => lambda = -1 / (1^T Q_SS^{-1} 1)
        # => a_S = Q_SS^{-1} 1 / (1^T Q_SS^{-1} 1)
        
        # Solve Q_SS * y = 1 using CG
        # Define the matrix-vector product on the active set
        def matvec_SA(y):
            # Compute Q[a_S] @ y where a_S is the subsequence of w
            # Q_SS[j,i] = w[S[j]+S[i]]
            # (Q_SS @ y)[j] = sum_i w[S[j]+S[i]] * y[i]
            # This is NOT a simple Hankel product unless S is contiguous
            # For general S, we need a different approach
            
            # For now, if S is all indices (contiguous), use FFT
            if nS == n:
                return hankel_matvec(w, y)
            else:
                # Dense computation (slow but correct)
                return np.array([sum(w[S[j]+S[i]] * y[i] for i in range(nS)) for j in range(nS)])
        
        # Use CG
        b_vec = np.ones(nS)
        y = np.zeros(nS)
        r = b_vec.copy()
        p_vec = r.copy()
        rs = r @ r
        
        for cg_iter in range(min(nS, max_iters)):
            Ap = matvec_SA(p_vec)
            pAp = p_vec @ Ap
            if pAp < 1e-20:
                break
            alpha = rs / pAp
            y += alpha * p_vec
            r -= alpha * Ap
            rs_new = r @ r
            if rs_new < 1e-14 * rs:
                break
            p_vec = r + (rs_new / rs) * p_vec
            rs = rs_new
        
        # Normalize: a_S = y / sum(y)
        y_sum = y.sum()
        if y_sum <= 0 or not np.isfinite(y_sum):
            # Fallback
            a_S = np.ones(nS) / nS
        else:
            a_S = y / y_sum
        
        # Check non-negativity
        a_new = np.zeros(n)
        a_new[S] = a_S
        
        if np.all(a_S >= -1e-10):
            # Success! Check KKT for inactive variables
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
            # Check if any inactive variable should be added
            # (Q @ a)_i for i not in S should be >= (Q @ a)_j for j in S
            Qa = hankel_matvec(w, a_new)
            active_val = Qa[S].mean()  # Should be constant at optimum
            inactive_min = Qa[~active].min() if (~active).any() else np.inf
            
            if inactive_min > active_val + 1e-8:
                # KKT satisfied (within tolerance)
                return a_new
            else:
                # Add the most violating inactive variable
                violating = np.where(~active)[0]
                if len(violating) > 0:
                    worst = violating[np.argmin(Qa[violating])]
                    active[worst] = True
                continue
        else:
            # Remove the most negative variable
            worst_in_S = S[np.argmin(a_S)]
            active[worst_in_S] = False
            continue
    
    # Return best found
    a_new = np.zeros(n)
    a_new[np.where(active)[0]] = np.ones(active.sum()) / active.sum()
    return a_new

This is interesting! The dual is a covering problem: we want to "cover" the vector 1 with non-negative combinations of shifted copies of a.

The optimal dual solution y* gives us: M * sum(y*) = optimal primal objective = max sum(g).

And the primal-dual relationship: at optimality, sum(g*) = M * sum(y*).

Now, the key insight for the algorithm:
- Given current a, the LP finds the best g (primal) or equivalently the best y (dual)
- The constraint (y ⋈ a) >= 1 means that y "covers" the all-ones vector when correlated with a
- This is like a "harmonic analysis" problem: find the minimal-weight non-negative measure y whose correlation with a dominates 1

For the next iteration, we update a towards g (the primal solution).

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Alternating QP method with log-sum-exp weight annealing (main engine)
2. LP mirror descent with improved active set management
3. Coordinate descent polishing
4. Multi-length and multi-initialization exploration
The QP method solves min a^T H_w a s.t. sum(a)=1, a>=0 where H_w is a Hankel
matrix, using FFT-accelerated matrix-vector products and active-set CG.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def hankel_matvec(w, x):
    """Compute H_w @ x where H[j,i] = w[j+i], using FFT."""
    n = len(x)
    w_flip = w[::-1]
    size = 1
    while size < len(w_flip) + n - 1:
        size <<= 1
    W = np.fft.rfft(w_flip, size)
    X = np.fft.rfft(x, size)
    result = np.fft.irfft(W * X, size)[:len(w_flip) + n - 1]
    return result[2*n-2 : n-1 : -1]

def solve_qp_hankel(w, n, warm_start=None, tolerance=1e-8):
    """
    Solve min a^T H_w a s.t. sum(a)=1, a>=0
    H_w[j,i] = w[j+i], w has length 2n-1.
    Uses active set + conjugate gradient with FFT matvec.
    """
    if warm_start is not None:
        a = np.maximum(warm_start, 0).copy()
        a = a / a.sum()
        active = a > 1e-12
    else:
        a = np.ones(n) / n
        active = np.ones(n, dtype=bool)
    
    best_a = a.copy()
    best_obj = a @ hankel_matvec(w, a)
    
    for outer_iter in range(100):
        S = np.where(active)[0]
        nS = len(S)
        if nS == 0:
            return np.ones(n) / n
        
        # Solve equality-constrained QP on active set:
        # min a_S^T H_SS a_S s.t. 1^T a_S = 1
        # Solution: a_S = H_SS^{-1} 1 / (1^T H_SS^{-1} 1)
        
        # Use CG to solve H_SS * y = 1
        def matvec_active(y):
            # H_SS[j,i] = w[S[j] + S[i]]
            if nS == n:
                return hankel_matvec(w, y)
            else:
                # For non-contiguous active set, use direct computation
                # This is O(nS^2) but nS should be small in practice
                result = np.zeros(nS)
                for idx_j in range(nS):
                    result[idx_j] = sum(w[S[idx_j] + S[idx_i]] * y[idx_i] for idx_i in range(nS))
                return result
        
        # CG
        b_vec = np.ones(nS)
        y = np.zeros(nS)
        r = b_vec.copy()
        p = r.copy()
        rs = r @ r
        rs_initial = rs
        
        for cg_iter in range(min(nS * 2, 2000)):
            Ap = matvec_active(p)
            pAp = p @ Ap
            if pAp < 1e-20:
                break
            alpha = rs / pAp
            y = y + alpha * p
            r = r - alpha * Ap
            rs_new = r @ r
            if rs_new < 1e-16 * rs_initial:
                break
            p = r + (rs_new / rs) * p
            rs = rs_new
        
        y_sum = y.sum()
        if y_sum <= 1e-15 or not np.isfinite(y_sum):
            # Fallback: uniform on active set
            a_new = np.zeros(n)
            a_new[S] = 1.0 / nS
        else:
            a_new = np.zeros(n)
            a_new[S] = y / y_sum
        
        # Check non-negativity of active variables
        min_active = a_new[S].min()
        
        if min_active >= -1e-10:
            # Clip tiny negatives
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            obj = a_new @ hankel_matvec(w, a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
            
            # Check KKT for inactive variables
            H_a = hankel_matvec(w, a_new)
            if nS == n:
                # All active, check if solution is valid
                if min_active > 1e-12:
                    return a_new
                else:
                    # Remove the smallest active variable
                    idx_min = S[np.argmin(a_new[S])]
                    active[idx_min] = False
                    continue
            else:
                active_vals = H_a[S]
                # At optimum, H_a should be constant on S
                # Check: for inactive variables, H_a[i] >= active_vals.mean()
                inactive = np.where(~active)[0]
                if len(inactive) > 0:
                    min_inactive = H_a[inactive].min()
                    mean_active = active_vals.mean()
                    if min_inactive >= mean_active - 1e-10:
                        # KKT satisfied
                        return a_new
                    else:
                        # Add the most violating inactive variable
                        worst = inactive[np.argmin(H_a[inactive])]
                        active[worst] = True
                        continue
        else:
            # Remove the most negative variable
            idx_worst = S[np.argmin(a_new[S])]
            active[idx_worst] = False
            continue
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent with constraint reduction and warm starts
2. FFT-accelerated smooth gradient (Nesterov) with log-sum-exp max approximation
3. Exact 1D coordinate descent with smart candidate generation
4. Multi-length exploration and diverse initialization
Key novelty: Combines LP (exact best-response) with smooth gradient (fast local search)
in an alternating fashion, plus Nesterov acceleration for the gradient phase.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*conv(a,a)))) using FFT.
    Returns (gradient, softmax_weights, conv_values)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    p = exp_b / exp_b.sum()
    # grad[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation)
    # = 2 * conv(flip(p), a)[2n-2-i]
    p_flip = p[::-1]
    size = 1
    while size < len(p_flip) + n - 1:
        size <<= 1
    P = np.fft.rfft(p_flip, size)
    A = np.fft.rfft(a, size)
    conv_result = np.fft.irfft(P * A, size)[:len(p_flip) + n - 1]
    corr = conv_result[2 * n - 2:n - 1:-1]
    grad = 2.0 * corr
    return grad, p, b


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def nesterov_smooth_descent(a, deadline, beta_start=5.0, beta_end=200.0,
                            lr_start=0.02, max_iters=50000):
    """Nesterov-accelerated projected gradient on smoothed objective."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    current_J = best_J

    y = a.copy()
    a_prev = a.copy()

    for t in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(t / max(1, max_iters), 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start / (1.0 + 0.05 * t)

        # Nesterov momentum
        momentum = (t + 1) / (t + 3)
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        y = y / y.sum() if y.sum() > 0.01 else np.ones(n) / n

        # Compute gradient
        grad, p, b = softmax_grad(y, beta)

        # Gradient step
        y_new = y - lr * grad
        y_new = project_simplex(y_new)

        a_prev = a
        a = y_new

        # Evaluate periodically
        if t % 20 == 0 or t == max_iters - 1:
            J = compute_J(a)
            if J < best_J - 1e-10:
                best_J = J
                best_a = a.copy()

    return best_a, best_J


def build_lp_matrix_reduced(n, a, b, frac_keep=0.7):
    """Build reduced sparse constraint matrix keeping only top frac_keep constraints."""
    n_within = int((2 * n - 1) * frac_keep)
    # Find the top n_within values in b
    sorted_idx = np.argsort(b)[::-1]
    keep = sorted_idx[:n_within]
    keep = np.sort(keep)

    rows_list = []
    cols_list = []
    vals_list = []
    for k in keep:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(a[k - j])

    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    vals = np.array(vals_list, dtype=np.float64)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A, keep


def lp_best_response(a, M, frac_keep=0.7, time_limit=5.0):
    """LP: max sum(g) s.t. (a*g)[k] <= M for top constraints, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    A, keep = build_lp_matrix_reduced(n, a, b, frac_keep)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent_fast(a, deadline, max_iters=2000, frac_keep=0.7):
    """Fast LP mirror descent with constraint reduction."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        g = lp_best_response(a, M, frac_keep=frac_keep, time_limit=3.0)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Adaptive step size: try multiple t values
        best_t = 0.1
        best_val = compute_J(np.maximum(a + 0.1 * (g - a), 0))
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            a_try = np.maximum((1 - t) * a + t * g, 0)
            s_try = a_try.sum()
            if s_try < 0.01:
                continue
            a_try = a_try / s_try
            val = compute_J(a_try)
            if val < best_val:
                best_val = val
                best_t = t

        a_new = np.maximum((1 - best_t) * a + best_t * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)

        if J_new < best_J - 1e-12:
            best_a = a_new.copy()
            best_J = J_new
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                a = a + np.random.normal(0, 0.03, n)
                a = np.maximum(a, 0.001)
                a = a / a.sum()
                no_improve = 0

    return best_a, best_J


def coordinate_descent_exact(a, deadline, max_passes=20):
    """Coordinate descent with exact 1D optimization per coordinate."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = np.random.permutation(n)
        b = fast_conv(a)
        S = a.sum()
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            # Determine affected range
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            if k_lo > 0 and k_hi < 2 * n - 2:
                unaffected_max = max(b[:k_lo].max(), b[k_hi + 1:].max())
            elif k_lo == 0:
                unaffected_max = b[k_hi + 1:].max() if k_hi < 2 * n - 2 else 0.0
            else:
                unaffected_max = b[:k_lo].max()

            # Generate candidate deltas
            candidates = [0.0]
            if ai > 1e-12:
                candidates.extend([
                    -ai,  # set to 0
                    -0.5 * ai,
                    0.5 * ai,
                    ai,  # double
                    2.0 * ai,
                ])
            else:
                candidates.extend([0.01, 0.05, 0.1, 0.2, 0.5])

            # Also try equalization: find d that makes the top two affected values equal
            if ai > 0:
                # The current peak in affected region
                affected_vals = b[k_lo:k_hi + 1]
                if len(affected_vals) >= 2:
                    top_idx = np.argsort(affected_vals)[::-1][:3]
                    for ti in range(len(top_idx)):
                        for tj in range(ti + 1, len(top_idx)):
                            k1 = k_lo + top_idx[ti]
                            k2 = k_lo + top_idx[tj]
                            d1 = a[k1 - i] if k1 - i < n else 0
                            d2 = a[k2 - i] if k2 - i < n else 0
                            if abs(d1 - d2) > 1e-12:
                                d_eq = (b[k2] - b[k1]) / (2.0 * (d1 - d2))
                                if -ai < d_eq < 5 * ai + 0.5:
                                    candidates.append(d_eq)

            best_val = 2 * n * max(b.max(), unaffected_max) / (S * S)
            best_delta = 0.0

            for d in candidates:
                if abs(d) < 1e-14:
                    continue
                S_new = S + d
                if S_new < 0.05:
                    continue
                if d + ai < -1e-12:
                    continue

                # Compute new max
                new_max = unaffected_max
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j == i:
                        new_val = b[k] + 2 * d * a[i] + d * d
                    elif j >= 0 and j < n:
                        new_val = b[k] + 2 * d * a[j]
                    else:
                        new_val = b[k]
                    if new_val > new_max:
                        new_max = new_val

                val = 2 * n * new_max / (S_new * S_new)
                if val < best_val - 1e-13:
                    best_val = val
                    best_delta = d

            if best_delta != 0.0:
                a[i] += best_delta
                if a[i] < 0:
                    a[i] = 0
                improved = True

        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            b = fast_conv(a)
            M = b.max()
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()

        if not improved:
            break

    return best_a


def generate_inits(n, rng, best_seq=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random
    r = rng.uniform(0.3, 1.7, n)
    inits.append(r / r.sum())

    # Cosine
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())

    # Bell
    bell = np.exp(-3 * (t - 0.5) ** 2) + 0.1
    inits.append(bell / bell.sum())

    # Double bump
    dbl = np.exp(-6 * (t - 0.3) ** 2) + np.exp(-6 * (t - 0.7) ** 2) + 0.05
    inits.append(dbl / dbl.sum())

    # Linear ramp
    ramp = np.linspace(0.1, 2.0, n)
    inits.append(ramp / ramp.sum())

    # Zigzag
    zz = 0.5 + 0.5 * np.abs(np.sin(np.pi * t * 4))
    inits.append(zz / zz.sum())

    # Frequency modulated
    for freq in [2, 3, 5]:
        fm = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
        fm = np.maximum(fm, 0.05)
        inits.append(fm / fm.sum())

    # From best_seq if available
    if best_seq is not None:
        bs = best_seq / best_seq.sum()
        # Perturbed versions
        for sigma in [0.05, 0.1, 0.2]:
            pert = np.maximum(bs + rng.normal(0, sigma, n), 0.01)
            inits.append(pert / pert.sum())
        # Smoothed
        ksize = max(3, n // 16)
        kernel = np.ones(ksize) / ksize
        sm = np.convolve(bs, kernel, mode='same')
        sm = np.maximum(sm, 0.01)
        inits.append(sm / sm.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        ]))
        target_lengths = [L for L in target_lengths if 128 <= L <= 1536 and L != n_base]
        # Limit to 5 to stay within budget
        target_lengths = target_lengths[:5]

        for tgt_n in target_lengths:
            if time.monotonic() > p4_dl:
                break
            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            # Quick LP + smooth + CD
            phase_budget = max(10.0, (p4_dl - time.monotonic()) / max(1, len(target_lengths)))
            dl_lp = time.monotonic() + phase_budget * 0.4
            a_r, J_r = lp_mirror_descent_fast(a_ext, dl_lp, max_iters=500)
            dl_smooth = time.monotonic() + phase_budget * 0.3
            a_r, J_r2 = nesterov_smooth_descent(a_r, dl_smooth, beta_start=10, beta_end=100,
                                                 lr_start=0.02, max_iters=20000)
            dl_cd = min(time.monotonic() + phase_budget * 0.3, p4_dl)
            a_r = coordinate_descent_exact(a_r, dl_cd, max_passes=5)
            update_best(a_r)

        print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Diverse initializations + optimization ===
    p5_dl = min(t_start + 1200, deadline - 250)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Diverse Inits (budget={p5_dl - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_best = best_seq.size
        inits = generate_inits(n_best, rng, best_seq)
        init_scores = [(compute_J(init), init) for init in inits]
        init_scores.sort(key=lambda x: x[0])

        # Try top 5 initializations with combined optimization
        for score_init, init in init_scores[:5]:
            if time.monotonic() > p5_dl:
                break
            remaining = max(5.0, p5_dl - time.monotonic())
            per_init_budget = remaining / max(1, 5)

            # LP
            dl1 = time.monotonic() + per_init_budget * 0.4
            a_r, _ = lp_mirror_descent_fast(init, dl1, max_iters=300)
            # Smooth
            dl2 = time.monotonic() + per_init_budget * 0.3
            a_r, _ = nesterov_smooth_descent(a_r, dl2, beta_start=10, beta_end=200,
                                              lr_start=0.02, max_iters=10000)
            # CD
            dl3 = min(time.monotonic() + per_init_budget * 0.3, p5_dl)
            a_r = coordinate_descent_exact(a_r, dl3, max_passes=5)
            update_best(a_r)

        print(f"  Phase 5 done: best J={best_score:.8f}", flush=True)

    # === PHASE 6: Final polishing ===
    p6_dl = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final Polishing (budget={p6_dl - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        remaining = max(5.0, p6_dl - time.monotonic())
        a_f, _ = lp_mirror_descent_fast(best_seq, time.monotonic() + remaining * 0.35, max_iters=500)
        update_best(a_f)
        a_f, _ = nesterov_smooth_descent(a_f, time.monotonic() + remaining * 0.3,
                                          beta_start=50, beta_end=500,
                                          lr_start=0.01, max_iters=30000)
        update_best(a_f)
        a_f = coordinate_descent_exact(a_f, p6_dl, max_passes=15)
        update_best(a_f)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
