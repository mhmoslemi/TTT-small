Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-sum-exp smoothed objective with tau annealing
2. LP mirror descent (best-response iteration)

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_fft(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_obj(x, tau):
    """Smoothed objective: (1/tau)*log(sum(exp(tau*(a*a)))) where a=softmax(x)."""
    a = np.exp(x - x.max())
    a = a / a.sum()
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    log_sum_exp = max_b + np.log(np.sum(np.exp(tau * (b - max_b))))
    return (1.0 / tau) * log_sum_exp


def softmax_grad(x, tau):
    """Gradient of smoothed objective w.r.t. x."""
    a = np.exp(x - x.max())
    a = a / a.sum()
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    p = np.exp(tau * (b - max_b))
    p = p / p.sum()
    # grad_a[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # Using cross-correlation: c[i] = sum_j a[j] * p[i+j]
    # c[i] = (a * p_rev)[2n-2-i] where p_rev[k] = p[2n-2-k]
    p_rev = p[::-1].copy()
    conv = cross_fft(a, p_rev)
    # grad_a[i] = 2 * conv[2n-2-i] for i=0,...,n-1
    # Indices: 2n-2, 2n-3, ..., n-1
    grad_a = 2.0 * conv[n-1:2*n-1][::-1]
    # Convert to gradient w.r.t. x (softmax parameterization)
    mean_grad = np.sum(grad_a * a)
    grad_x = a * (grad_a - mean_grad)
    return grad_x

            result = minimize(
                softmax_obj, x, jac=softmax_grad,
                method='L-BFGS-B', args=(tau,),
                options={'maxiter': max_iter_per_tau, 'ftol': 1e-14, 'gtol': 1e-12}
            )
            x = result.x
            a = np.exp(x - x.max())
            a = a / a.sum()
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
                print(f"    tau={tau}: J={J:.8f} (iter={result.nit})", flush=True)
        except Exception as e:
            print(f"    tau={tau}: L-BFGS failed: {e}", flush=True)
    
    return best_a, best_J


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_mirror_descent(a, max_iters, deadline, use_cd_every=40):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b_aa = fast_conv(a)
        M = float(b_aa.max())
        
        # LP best response
        A = build_lp_matrix(n, a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 30, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                g_sum = g.sum()
                if g_sum < 1e-15:
                    break
                g = g / g_sum
                
                # Line search
                b_ag = cross_fft(a, g)
                b_gg = fast_conv(g)
                
                def obj(t):
                    b = (1-t)**2 * b_aa + 2*t*(1-t) * b_ag + t*t * b_gg
                    return b.max()
                
                from scipy.optimize import minimize_scalar
                res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                       options={'xatol': 1e-8})
                t_opt = float(res.x)
                
                a_new = np.maximum((1-t_opt) * a + t_opt * g, 0)
                s = a_new.sum()
                if s < 1e-10:
                    break
                a_new = a_new / s
                J_new = compute_J(a_new)
                
                if J_new < best_J - 1e-13:
                    best_a = a_new.copy()
                    best_J = J_new
                    a = a_new
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                break
        except Exception:
            break
        
        if no_improve > 50:
            a = a + np.random.normal(0, 0.02, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0
    
    return best_a, best_J


def coordinate_descent(a, deadline, max_passes=20, n_candidates=8):
    """Coordinate descent with multiple candidates per coordinate."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        order = np.random.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            ai = a[i]
            
            # Generate candidates
            if ai < 1e-10:
                candidates = np.linspace(0, 0.3, n_candidates)
            else:
                candidates = np.concatenate([
                    [0.0],
                    ai * np.linspace(0.1, 0.9, n_candidates // 2),
                    ai * np.linspace(1.1, 2.5, n_candidates - n_candidates // 2 - 1)
                ])
            
            best_i = ai
            best_score = best_J
            
            for cand in candidates:
                if abs(cand - ai) < 1e-14:
                    continue
                a[i] = max(cand, 0)
                s_check = a.sum()
                if s_check < 0.01:
                    a[i] = ai
                    continue
                score = compute_J(a / s_check)
                if score < best_score - 1e-12:
                    best_score = score
                    best_i = cand
            
            if best_i != ai:
                a[i] = best_i
                improved = True
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a


def local_search(a, deadline, n_trials=5000, max_perturb=0.1):
    """Local search: small random perturbations with acceptance."""
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    T = 0.01
    for it in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        # Perturb a random subset of coordinates
        n_pert = np.random.choice([1, 2, 3, 5], p=[0.5, 0.25, 0.15, 0.1])
        indices = np.random.choice(n, min(n_pert, n), replace=False)
        old_vals = a[indices].copy()
        
        scale = max_pert * (0.5 + 0.5 * np.random.random())
        a[indices] += np.random.normal(0, scale, len(indices))
        a[indices] = np.maximum(a[indices], 0)
        
        s = a.sum()
        if s < 0.01:
            a[indices] = old_vals
            continue
        a_new = a / s
        J_new = compute_J(a_new)
        
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
        elif J_new < best_J and np.random.random() < np.exp(-(J_new - best_J) / T):
            pass  # Accept (already in a)
        else:
            a[indices] = old_vals
        
        T = max(0.001, T * 0.999)
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-sum-exp smoothed objective with tau annealing (fast local optimizer)
2. LP mirror descent with best-response iteration (global exploration)
3. Fine-grained coordinate descent (precise polishing)
4. Local search with random perturbations (escape local minima)
Key innovation: L-BFGS with softmax parameterization provides superlinear convergence
near the optimum, complementing the sublinear LP mirror descent.
"""

import numpy as np
from scipy.optimize import minimize, linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_fft(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_obj(x, tau):
    """Smoothed objective: (1/tau)*log(sum(exp(tau*(a*a)))) where a=softmax(x)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    log_sum_exp = max_b + np.log(np.sum(np.exp(tau * (b - max_b))))
    return (1.0 / tau) * log_sum_exp


def softmax_grad(x, tau):
    """Gradient of smoothed objective w.r.t. x (softmax parameterization)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    p = np.exp(tau * (b - max_b))
    p = p / p.sum()
    # grad_a[i] = 2 * sum_k p[k] * a[k-i] for valid indices
    # Computed via: c[i] = sum_j a[j] * p[i+j] = (a * p_rev)[2n-2-i]
    p_rev = p[::-1].copy()
    conv = cross_fft(a, p_rev)
    grad_a = 2.0 * conv[n - 1:2 * n - 1][::-1]
    # Chain rule through softmax: df/dx[j] = a[j]*(df/da[j] - sum_i df/da[i]*a[i])
    mean_grad = np.dot(grad_a, a)
    grad_x = a * (grad_a - mean_grad)
    return grad_x


def lbfgs_optimize(a_init, tau_list, deadline, max_iter_per_tau=300):
    """Run L-BFGS on smoothed objective with tau annealing and warm starts."""
    n = len(a_init)
    x = np.log(np.maximum(a_init, 1e-15))
    best_a = a_init.copy()
    best_J = compute_J(a_init)
    start_J = best_J

    for tau in tau_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                softmax_obj, x, jac=softmax_grad,
                method='L-BFGS-B', args=(tau,),
                options={'maxiter': max_iter_per_tau, 'ftol': 1e-15, 'gtol': 1e-10}
            )
            x = result.x
            a = np.exp(x - x.max())
            a = a / a.sum()
            J = compute_J(a)
            if J < best_J - 1e-14:
                best_J = J
                best_a = a.copy()
                print(f"    tau={tau}: J={best_J:.10f} (nit={result.nit})", flush=True)
        except Exception as e:
            print(f"    tau={tau}: failed ({e})", flush=True)

    improvement = start_J - best_J
    print(f"    L-BFGS total improvement: {improvement:.10f}", flush=True)
    return best_a, best_J


def build_lp_matrix(n, a):
    """Build sparse constraint matrix A[k,j] = a[k-j] for valid indices."""
    rows = np.empty((2 * n - 1) * n, dtype=np.int32)
    cols = np.empty((2 * n - 1) * n, dtype=np.int32)
    vals = np.empty((2 * n - 1) * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))
    return A


def lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with bilinear line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        A = build_lp_matrix(n, a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 60, 'presolve': True})
            if not (r.success and r.x is not None):
                break
            g = np.maximum(r.x, 0)
            g_sum = g.sum()
            if g_sum < 1e-15:
                break
            g = g / g_sum

            b_ag = cross_fft(a, g)
            b_gg = fast_conv(g)

            def obj_fn(t):
                b = (1.0 - t) ** 2 * b_aa + 2.0 * t * (1.0 - t) * b_ag + t * t * b_gg
                return b.max()

            res = minimize_scalar(obj_fn, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            t_opt = float(res.x)

            a_new = np.maximum((1.0 - t_opt) * a + t_opt * g, 0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new = a_new / s
            J_new = compute_J(a_new)

            if J_new < best_J - 1e-14:
                best_J = J_new
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 80:
                a = a + rng.normal(0, 0.03, n)
                a = np.maximum(a, 0.001)
                a = a / a.sum()
                no_improve = 0

        except Exception:
            break

    return best_a, best_J


def coordinate_descent(a, deadline, max_passes=20, n_cand=10, seed=42):
    """Coordinate descent with multiple candidate values per coordinate."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break
            ai = a[i]

            if ai < 1e-10:
                candidates = np.linspace(0, 0.2, n_cand)
            else:
                lo = np.linspace(0.05, 0.85, n_cand // 2) * ai
                hi = np.linspace(1.15, 3.0, n_cand - n_cand // 2) * ai
                candidates = np.concatenate([[0.0], lo, hi])

            best_i = ai
            best_score = compute_J(a)

            for cand in candidates:
                if abs(cand - ai) < 1e-13:
                    continue
                a[i] = max(cand, 0.0)
                s_chk = a.sum()
                if s_chk < 0.01:
                    a[i] = ai
                    continue
                score = compute_J(a / s_chk)
                if score < best_score - 1e-13:
                    best_score = score
                    best_i = cand

            if best_i != ai:
                a[i] = best_i
                improved = True

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def local_search(a, deadline, n_trials=8000, seed=42):
    """Local search with random perturbations and simulated annealing acceptance."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    T = 0.008

    for it in range(n_trials):
        if time.monotonic() > deadline:
            break

        n_pert = int(rng.choice([1, 1, 2, 2, 3, 5], p=[0.3, 0.2, 0.2, 0.1, 0.1, 0.1]))
        n_pert = min(n_pert, n)
        indices = rng.choice(n, n_pert, replace=False)
        old_vals = a[indices].copy()

        scale = 0.03 + 0.04 * rng.random()
        a[indices] += rng.normal(0, scale, n_pert)
        a[indices] = np.maximum(a[indices], 0)

        s = a.sum()
        if s < 0.01:
            a[indices] = old_vals
            continue
        a_new = a / s
        J_new = compute_J(a_new)

        delta = J_new - current_J if 'current_J' in dir() else J_new - best_J
        if J_new < best_J - 1e-13:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
        elif rng.random() < np.exp(-(J_new - best_J) / max(T, 1e-10)):
            a = a_new
        else:
            a[indices] = old_vals

        T = max(0.0005, T * 0.9995)

    return best_a


def gradient_descent_subgrad(a, deadline, eta=0.01, n_iters=5000, seed=42):
    """Subgradient method on the max of convolution (projected onto simplex)."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        # Active set: indices where b is close to max
        active = np.where(b >= M * 0.999)[0]
        # Uniform distribution over active set
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)

        # Subgradient: 2 * (p cross-correlate a)
        p_rev = p[::-1].copy()
        conv = cross_fft(a, p_rev)
        grad = 2.0 * conv[n - 1:2 * n - 1][::-1]

        # Adaptive step size
        eta_t = eta / (1.0 + 0.01 * it)
        a = a - eta_t * grad
        # Project onto simplex (a >= 0, sum(a) = 1)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 0.01:
            a = best_a.copy()
            continue
        a = a / s
        J = compute_J(a)
        if J < best_J - 1e-14:
            best_J = J
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Phase 1: L-BFGS smoothed optimization (fast local, multiple restarts)
    Phase 2: LP mirror descent (global exploration via best-response)
    Phase 3: Coordinate descent polishing
    Phase 4: Local search + L-BFGS hybrid
    Phase 5: Final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.10f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256.0
        update_best(parent)

    n = parent.size
    a_norm = parent / parent.sum()

    # === PHASE 1: L-BFGS smoothed optimization ===
    p1_deadline = min(t_start + 250, deadline - 500)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: L-BFGS Smoothed (n={n}, budget={p1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Main run from parent
    tau_list_main = [5, 10, 20, 35, 50, 75, 100, 150, 200, 300, 500]
    a_res, J_res = lbfgs_optimize(a_norm, tau_list_main, p1_deadline, max_iter_per_tau=400)
    update_best(a_res)
    print(f"  Main L-BFGS: J={J_res:.10f}", flush=True)

    # Restart from perturbed versions
    for trial in range(4):
        if time.monotonic() > p1_deadline:
            break
        pert_scale = [0.03, 0.08, 0.15, 0.25][trial]
        a_pert = np.maximum(a_res + rng.normal(0, pert_scale * a_res.max(), n), 0.005)
        a_pert = a_pert / a_pert.sum()
        tau_short = [20, 50, 100, 200, 400]
        a_r, J_r = lbfgs_optimize(a_pert, tau_short, p1_deadline, max_iter_per_tau=200)
        update_best(a_r)

    # Try from parent directly with different tau
    if time.monotonic() < p1_deadline - 30:
        tau_alt = [50, 100, 200, 400, 800]
        a_r2, J_r2 = lbfgs_optimize(a_norm, tau_alt, p1_deadline, max_iter_per_tau=300)
        update_best(a_r2)

    # === PHASE 2: LP mirror descent ===
    p2_deadline = min(t_start + 700, deadline - 300)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (budget={p2_deadline - t_start - 250:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # From best L-BFGS result
    a_lp, J_lp = lp_mirror_descent(a_res, 3000, p2_deadline)
    update_best(a_lp)
    print(f"  LP from L-BFGS: J={J_lp:.10f}", flush=True)

    # From parent (different trajectory)
    if time.monotonic() < p2_deadline - 60:
        a_lp2, J_lp2 = lp_mirror_descent(a_norm, 2000, p2_deadline)
        update_best(a_lp2)
        print(f"  LP from parent: J={J_lp2:.10f}", flush=True)

    # === PHASE 3: Coordinate descent + L-BFGS polish ===
    p3_deadline = min(t_start + 1000, deadline - 200)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: CD + L-BFGS Polish (budget={p3_deadline - t_start - 700:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        n_cd = best_seq.size
        a_cd = coordinate_descent(best_seq, p3_deadline, max_passes=25, n_cand=12, seed=7)
        update_best(a_cd)

        # L-BFGS polish on CD result
        if time.monotonic() < p3_deadline - 20:
            a_pol, J_pol = lbfgs_optimize(a_cd, [100, 200, 500, 1000],
                                          min(p3_deadline, deadline - 150), max_iter_per_tau=200)
            update_best(a_pol)

            # One more CD after L-BFGS
            a_cd2 = coordinate_descent(a_pol, p3_deadline, max_passes=10, n_cand=10, seed=13)
            update_best(a_cd2)

    # === PHASE 4: Local search + L-BFGS hybrid ===
    p4_deadline = min(t_start + 1300, deadline - 100)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Local Search + L-BFGS (budget={p4_deadline - t_start - 1000:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        a_ls = local_search(best_seq, p4_deadline, n_trials=15000, seed=99)
        update_best(a_ls)

        # L-BFGS polish on local search result
        if time.monotonic() < p4_deadline - 20:
            a_ls_pol, J_ls_pol = lbfgs_optimize(a_ls, [50, 100, 300, 600],
                                                 min(p4_deadline, deadline - 80), max_iter_per_tau=150)
            update_best(a_ls_pol)

            # CD on polished result
            a_cd3 = coordinate_descent(a_ls_pol, p4_deadline, max_passes=8, n_cand=10, seed=21)
            update_best(a_cd3)

    # === PHASE 5: Final polishing ===
    p5_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Final Polish (budget={p5_deadline - t_start - 1300:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        # Ultimate L-BFGS with very high tau
        a_final, J_final = lbfgs_optimize(best_seq, [200, 500, 1000, 2000],
                                           p5_deadline, max_iter_per_tau=300)
        update_best(a_final)

        # Final CD
        a_final_cd = coordinate_descent(a_final, p5_deadline, max_passes=5, n_cand=14, seed=55)
        update_best(a_final_cd)

        # One more L-BFGS on CD result
        if time.monotonic() < p5_deadline - 10:
            a_ultra, J_ultra = lbfgs_optimize(a_final_cd, [500, 1000, 2000],
                                               p5_deadline, max_iter_per_tau=200)
            update_best(a_ultra)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128.0
        update_best(best_seq)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.10f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
