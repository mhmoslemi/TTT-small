import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx, time_limit=5.0):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val

def smooth_gradient_step(a, tau):
    """Gradient of smooth max approximation (1/tau)*log(sum exp(tau*c))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * pk * seg
    return grad


def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0):
    """Smooth gradient descent with Adam optimizer."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        grad = smooth_gradient_step(a, tau)
        t_count = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        lr = 0.5 / (1.0 + it * 0.01)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        M = compute_M(a)
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
    return best_a


def run_subgradient_averaging(a, max_iters, deadline, init_tau=10.0):
    """Subgradient method with iterate averaging (Nesterov's method)."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    avg_a = np.zeros(n)
    best_a = a.copy()
    best_M = compute_M(a)
    a_smooth = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        tau = init_tau * (1.0 + it * 0.05)
        c = fast_conv(a_smooth)
        c_max = c.max()
        # Subgradient: at argmax positions
        active = np.where(c >= c_max - 1e-10)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            seg = a_smooth[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * seg
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            continue
        grad = grad / g_norm
        # Step size: 2/(t+1) * initial_scale
        step = 2.0 / (1.0 + it * 0.1) * 0.3
        a_smooth = a_smooth - step * grad
        a_smooth = np.maximum(a_smooth, 0.0)
        s = a_smooth.sum()
        if s > 1e-10:
            a_smooth = a_smooth / s
        # Average
        avg_a += a_smooth / (it + 1)
        M = compute_M(avg_a)
        if M < best_M:
            best_M = M
            best_a = avg_a.copy()
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def block_coordinate_descent(a, max_iters, deadline, block_size=32):
    """Optimize blocks of coordinates while keeping others fixed."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        # Pick a random block
        start = rng.integers(0, max(1, n - block_size))
        end = min(start + block_size, n)
        # For this block, try to reduce the max convolution
        # Use a simple gradient step on just these coordinates
        c = fast_conv(a)
        c_max = c.max()
        active = np.where(c >= c_max - 1e-8)[0]
        # Compute gradient for the block coordinates
        grad_block = np.zeros(end - start)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # Which block indices are involved?
            block_lo = max(lo, start) - start
            block_hi = min(hi, end - 1) - start
            if block_hi >= block_lo:
                for bi in range(block_lo, block_hi + 1):
                    gi = bi + start
                    grad_block[bi] += 2.0 * a[k - gi]
        # Try scaling down the block
        current = a[start:end]
        for scale in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
            a_trial = a.copy()
            a_trial[start:end] = current * scale
            # Redistribute the freed/added mass to the rest
            old_sum = a.sum()
            new_sum = a_trial.sum()
            if new_sum > 1e-10:
                a_trial = a_trial / old_sum  # Keep total mass same
            M_trial = compute_M(a_trial)
            if M_trial < best_M:
                best_M = M_trial
                best_a = a_trial.copy()
                a = a_trial

    return best_a


class Tracker:
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining LP mirror descent, subgradient averaging,
    multi-length exploration, block coordinate descent, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent, "(default)")

    # Pre-build LP matrices for common sizes
    lp_cache = {}
    def get_lp_matrix(n):
        if n not in lp_cache:
            lp_cache[n] = build_lp_matrix(n)
        return lp_cache[n]

    # ============ PHASE 1: LP Mirror Descent on parent (blend mode) ============
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent (blend) n={parent.size}", flush=True)
    print(f"{'='*60}", flush=True)

    n0 = parent.size
    r_idx, c_idx = get_lp_matrix(n0)
    a = parent.copy() / parent.sum()
    phase1_deadline = min(t_start + budget_s * 0.20, deadline)
    a_opt = run_lp_mirror_descent(a, 100000, phase1_deadline, r_idx, c_idx,
                                  update_mode='blend', verbose=True, lp_time=5.0)
    tracker.update(a_opt, "(P1 blend)")

    # ============ PHASE 2: LP Mirror Descent (multiplicative mode) ============
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (mult) n={tracker.best.size}", flush=True)
    print(f"{'='*60}", flush=True)

    n_cur = tracker.best.size
    r_cur, c_cur = get_lp_matrix(n_cur)
    a = tracker.best.copy() / tracker.best.sum()
    phase2_deadline = min(t_start + budget_s * 0.32, deadline)
    a_opt = run_lp_mirror_descent(a, 100000, phase2_deadline, r_cur, c_cur,
                                  update_mode='multiplicative', verbose=True, lp_time=4.0)
    tracker.update(a_opt, "(P2 mult)")

    # ============ PHASE 3: Smooth gradient polish ============
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Smooth Gradient (Adam)", flush=True)
    print(f"{'='*60}", flush=True)

    a = tracker.best.copy()
    a_trans = a if a.sum() > 1e-10 else np.ones(n_cur)
    a_trans = a_trans / a_trans.sum()
    phase3_deadline = min(t_start + budget_s * 0.40, deadline)
    a_opt = run_smooth_gradient(a_trans, 5000, phase3_deadline,
                                initial_tau=5.0, final_tau=300.0)
    tracker.update(a_opt, "(P3 smooth)")

    # LP polish after smooth
    r_cur, c_cur = get_lp_matrix(tracker.best.size)
    a = tracker.best.copy() / tracker.best.sum()
    phase3b_deadline = min(t_start + budget_s * 0.48, deadline)
    a_opt = run_lp_mirror_descent(a, 10000, phase3b_deadline, r_cur, c_cur,
                                  update_mode='blend', lp_time=3.0)
    tracker.update(a_opt, "(P3b LP after smooth)")

    # ============ PHASE 4: Multi-length exploration ============
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    base_n = tracker.best.size
    trial_lengths = []
    for mult in [0.5, 0.75, 1.5, 2.0, 3.0, 4.0]:
        n_trial = int(base_n * mult)
        if 64 <= n_trial <= 8192 and n_trial != base_n:
            trial_lengths.append(n_trial)
    # Also try some absolute values
    for n_try in [256, 512, 1024, 2048, 4096]:
        if n_try not in trial_lengths and 64 <= n_try <= 8192:
            trial_lengths.append(n_try)
    trial_lengths = sorted(set(trial_lengths))

    phase4_deadline = min(t_start + budget_s * 0.65, deadline)
    for n_trial in trial_lengths:
        if time.monotonic() > phase4_deadline:
            break
        elapsed_per = (time.monotonic() - t_start)
        remaining = phase4_deadline - time.monotonic()
        if remaining < 10:
            break

        print(f"  Trying n={n_trial}...", flush=True)
        a_init = resample_seq(tracker.best, n_trial)
        a_init = np.maximum(a_init, 1e-8)
        a_init = a_init / a_init.sum()

        r_idx_t, c_idx_t = get_lp_matrix(n_trial)
        # Scale time based on n
        time_per_n = max(10, min(60, 4096 // n_trial * 15))
        sub_dl = min(time.monotonic() + time_per_n, phase4_deadline)

        # Quick smooth gradient first
        a_sg = run_smooth_gradient(a_init, 500, min(time.monotonic() + 5, sub_dl),
                                   initial_tau=5.0, final_tau=50.0)
        tracker.update(a_sg, f"(P4 n={n_trial} smooth)")

        # Then LP
        a_lp = tracker.best if tracker.best.size == n_trial else a_sg
        a_lp = a_lp.copy() / a_lp.sum()
        a_opt = run_lp_mirror_descent(a_lp, 5000, sub_dl, r_idx_t, c_idx_t,
                                      update_mode='blend', lp_time=3.0)
        tracker.update(a_opt, f"(P4 n={n_trial} LP)")

    # ============ PHASE 5: Subgradient averaging ============
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Subgradient Averaging", flush=True)
    print(f"{'='*60}", flush=True)

    phase5_deadline = min(t_start + budget_s * 0.75, deadline)
    a = tracker.best.copy() / tracker.best.sum()
    a_opt = run_subgradient_averaging(a, 10000, phase5_deadline)
    tracker.update(a_opt, "(P5 subgrad avg)")

    # ============ PHASE 6: Random restarts with diverse initializations ============
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Diverse Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    n_best = tracker.best.size
    r_best, c_best = get_lp_matrix(n_best)
    phase6_deadline = min(t_start + budget_s * 0.88, deadline)

    restart_count = 0
    for r in range(50):
        if time.monotonic() > phase6_deadline:
            break
        remaining = phase6_deadline - time.monotonic()
        if remaining < 8:
            break

        # Different initialization strategies
        strategy = r % 5
        a = tracker.best.copy().astype(np.float64) / tracker.best.sum()

        if strategy == 0:
            # Gaussian noise
            noise = rng.normal(0, 0.05, n_best)
            a = a + noise
        elif strategy == 1:
            # Sine perturbation
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            pert = 0.1 * np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
            a = a + pert
        elif strategy == 2:
            # Random spike
            idx = rng.integers(0, n_best)
            a[idx] += 0.3
        elif strategy == 3:
            # Block swap
            bs = n_best // 4
            i1 = rng.integers(0, n_best - bs)
            i2 = rng.integers(0, n_best - bs)
            a[i1:i1+bs], a[i2:i2+bs] = a[i2:i2+bs].copy(), a[i1:i1+bs].copy()
        else:
            # Sharpen/flatten
            power = rng.uniform(0.5, 2.0)
            a = a ** power

        a = np.maximum(a, 1e-6)
        a = a / a.sum()

        # Quick smooth gradient
        sub_dl = min(time.monotonic() + min(5, remaining * 0.3), phase6_deadline)
        a = run_smooth_gradient(a, 300, sub_dl, initial_tau=10, final_tau=100)

        # LP polish
        sub_dl2 = min(time.monotonic() + min(15, remaining * 0.6), phase6_deadline)
        a_opt = run_lp_mirror_descent(a, 2000, sub_dl2, r_best, c_best,
                                      update_mode='blend', lp_time=3.0)
        tracker.update(a_opt, f"(P6 r={r} s={strategy})")
        restart_count += 1

    print(f"  Completed {restart_count} restarts", flush=True)

    # ============ PHASE 7: Final extended LP polish ============
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 7: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    n_final = tracker.best.size
    r_f, c_f = get_lp_matrix(n_final)
    a = tracker.best.copy() / tracker.best.sum()

    # Alternate between blend and multiplicative
    a_opt1 = run_lp_mirror_descent(a, 50000, min(deadline, t_start + budget_s * 0.94),
                                   r_f, c_f, update_mode='blend', verbose=True, lp_time=5.0)
    tracker.update(a_opt1, "(P7a blend)")

    a = tracker.best.copy() / tracker.best.sum()
    a_opt2 = run_lp_mirror_descent(a, 50000, deadline, r_f, c_f,
                                    update_mode='multiplicative', lp_time=4.0)
    tracker.update(a_opt2, "(P7b mult)")

    # Final smooth gradient pass
    a = tracker.best.copy() / tracker.best.sum()
    a_opt3 = run_smooth_gradient(a, 2000, deadline, initial_tau=50, final_tau=500)
    tracker.update(a_opt3, "(P7c final smooth)")

    # One more LP
    if time.monotonic() < deadline - 5:
        a = tracker.best.copy() / tracker.best.sum()
        a_opt4 = run_lp_mirror_descent(a, 20000, deadline, r_f, c_f,
                                       update_mode='blend', lp_time=4.0)
        tracker.update(a_opt4, "(P7d final LP)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
