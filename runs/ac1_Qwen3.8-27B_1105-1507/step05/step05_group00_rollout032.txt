def golden_section_search(f, a, b, tol=1e-6, max_iter=50):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):

            break
        if fc < fd:

            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2

        grad_centered = grad - np.mean(grad)

            if step_size < 1e-6:

        # Decrease temperature
        tau *= 0.95

def spectral_equalization(a, n_iters=50, time_budget=30.0):
    """
    POCS-like spectral equalization: alternate between time-domain projection
    (non-negativity + normalization) and frequency-domain flattening.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    size = 1
    while size < 2 * n:
        size *= 2
    
    gamma = 0.5  # flattening strength
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Forward FFT
        A = fft(a, size)
        
        # Compute power spectrum
        P = np.abs(A) ** 2
        
        # Flattening: reduce peaks in power spectrum
        # P_new = P^alpha for alpha < 1 (softens peaks)
        # Or: P_new = P / (P + epsilon) type compression
        p_max = np.max(P)
        if p_max < 1e-30:
            break
        
        # Apply compression
        alpha = 0.7 + 0.2 * (it / n_iters)  # start with strong compression, gradually reduce
        P_new = (P / p_max) ** alpha * p_max
        
        # Reconstruct with modified magnitude, keeping original phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        
        # IFFT
        a_new = np.real(ifft(A_new, size))[:n]
        
        # Time-domain projection: non-negativity + normalization
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try weaker compression
            a = a_new  # still use it as starting point for next iteration
    
    return best_a

def simulated_annealing(a, time_budget=60.0):
    """
    Simulated annealing: random perturbations with acceptance criterion.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    T = 0.01  # initial temperature
    T_min = 1e-5
    cooling = 0.95
    
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        # Generate perturbation
        # Option 1: Gaussian perturbation
        if rng.random() < 0.5:
            a_pert = a * np.exp(rng.normal(0, 0.05, n))
        else:
            # Option 2: move mass between two random positions
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.1)
            a_pert = a.copy()
            delta = frac * a_pert[i]
            a_pert[i] -= delta
            a_pert[j] += delta
        
        a_pert = np.maximum(a_pert, 0)
        a_pert = normalize(a_pert)
        
        obj_pert = compute_obj(a_pert)
        delta_obj = obj_pert - best_obj
        
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            a = a_pert
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
        
        T *= cooling
        if T < T_min:
            T = T_min
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

def coordinate_descent_v2(a, time_budget=30.0):
    """
    Improved coordinate descent: try pairwise mass transfers.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    rng = np.random.default_rng(123)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        
        # Find the pair that most contributes to the peak
        # The peak at k_star is sum_j a[j]*a[k_star-j]
        # We want to reduce this by moving mass.
        
        # Try moving mass from a[j] to a[j'] where j' is far from the peak
        for trial in range(10):
            if time.monotonic() - start > time_budget:
                break
            
            # Pick a contributor to the peak
            contributors = []
            for j in range(n):
                k2 = k_star - j
                if 0 <= k2 < n:
                    contributors.append((a[j] * a[k2], j, k2))
            contributors.sort(reverse=True)
            
            if not contributors:
                break
            
            # Take the top contributor
            peak_val, j1, j2 = contributors[0]
            if j1 == j2:
                continue
            
            # Try moving mass from j1 to a random far position
            j_target = rng.integers(0, n)
            while j_target == j1 or j_target == j2:
                j_target = rng.integers(0, n)
            
            for frac in [0.01, 0.03, 0.05, 0.1]:
                delta = frac * a[j1]
                a[j1] -= delta
                a[j_target] += delta
                a_try = np.maximum(a, 0)
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)
                
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    break
                else:
                    a[j1] += delta
                    a[j_target] -= delta
                    break
    
    return best_a

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with golden-section line search, spectral equalization (POCS),
soft-max (log-sum-exp) gradient descent, multi-n exploration, random restarts
with diverse perturbation strategies, and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2. Scale-invariant."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1, clipping negatives."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def golden_section_search(f, lo, hi, tol=0.01, max_iter=25):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2.0


def lp_alternate_golden(a, max_iters=50, time_budget=80.0):
    """LP alternating with golden section line search for step size t."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 3:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_max_sum(a, M, time_limit=max(5, int(remaining * 0.6)))
        if g is None:
            break

        # Golden section search for optimal blend t in [0, 1]
        def obj_at_t(t):
            a_try = (1.0 - t) * a + t * g
            a_try = normalize(a_try)
            return compute_obj(a_try)

        t_opt = golden_section_search(obj_at_t, 0.0, 1.0, tol=0.02, max_iter=15)

        a_new = normalize((1.0 - t_opt) * a + t_opt * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Perturb to escape local minimum
            rng = np.random.default_rng((it * 7919 + 12345) % (2**31))
            a = normalize(best_a * np.exp(rng.normal(0, 0.04 + 0.02 * (it % 5), n)))

    return best_a


def soft_max_gradient_step(a, tau_init=0.001, step_size=0.04, max_steps=200, time_budget=40.0):
    """
    Gradient step using soft-max (log-sum-exp) weighting over convolution peaks.
    Provides smoother gradient than hard subgradient by targeting multiple high peaks.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Precompute FFT size for gradient computation
    size = 1
    while size < 3 * n - 1:
        size <<= 1

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Adaptive temperature: start broad, narrow down
        tau = tau_init + 0.5 * cmax / (n * (step + 1))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient via FFT: grad[i] = 2 * sum_j w[i+j] * a[j]
        w_ext = np.zeros(2 * n - 1)
        w_ext[:] = weights
        a_flip = a[::-1]

        Wf = rfft(w_ext, size)
        Af = rfft(a_flip, size)
        corr = irfft(Wf * Af, size)
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Project onto sum=const hyperplane (remove mean)
        grad_centered = grad - np.mean(grad)

        # Adaptive step size
        a_new = normalize(a - step_size * grad_centered)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.6
            if step_size < 1e-7:
                break

        # Cool temperature
        tau_init *= 0.97

    return best_a


def spectral_equalization(a, max_iters=60, time_budget=40.0):
    """
    POCS-style spectral equalization: alternate between frequency-domain
    power spectrum flattening and time-domain non-negativity projection.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    size = 1
    while size < 2 * n:
        size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Forward FFT
        A = fft(a, size)
        P = np.abs(A) ** 2
        p_max = float(np.max(P))
        if p_max < 1e-30:
            break

        # Spectral compression: raise power toward uniform
        # Start aggressive, ease off
        alpha = 0.5 + 0.3 * (it / float(max_iters))
        P_norm = P / p_max
        P_new = (P_norm ** alpha) * p_max

        # Reconstruct with modified magnitude, preserve phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)

        # IFFT to time domain
        a_new = np.real(ifft(A_new, size))[:n]

        # Time-domain projection: non-negativity + normalization
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Blended update to maintain progress
            a = 0.5 * a + 0.5 * a_new
            a = normalize(a)

    return best_a


def coordinate_descent_v2(a, time_budget=30.0):
    """
    Pairwise mass-transfer coordinate descent targeting the convolution peak.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(999)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Identify top contributors to the peak
        contrib = []
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib.append((a[j] * a[k2], j, k2))
        contrib.sort(reverse=True)

        if not contrib:
            break

        improved = False
        for peak_val, j1, j2 in contrib[:5]:
            if time.monotonic() - start > time_budget:
                break
            if j1 == j2:
                continue

            # Try moving mass from peak contributor to random far position
            for _ in range(3):
                j_target = int(rng.integers(0, n))
                if j_target == j1 or j_target == j2:
                    continue

                a_try = a.copy()
                delta = 0.05 * a_try[j1]
                a_try[j1] -= delta
                a_try[j_target] += delta
                a_try = normalize(a_try)
                obj_try = compute_obj(a_try)

                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

        if not improved and rng.random() < 0.1:
            # Occasional random perturbation
            a = normalize(a * np.exp(rng.normal(0, 0.03, n)))

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)
