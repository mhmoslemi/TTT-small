def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """Compute smooth max of conv(a,a) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Numerically stable softmax
    exp_b = np.exp(shifted - shifted.max())
    p = exp_b / exp_b.sum()
    # Gradient: grad[m] = 2 * sum_j p[m+j] * a[j]
    # Use correlate: correlate(p, a, 'valid') gives sum_j p[m+j]*a[j] for m=0..n-1
    grad = 2.0 * np.correlate(p, a, mode='valid')
    smooth_val = b_max + np.log(exp_b.sum()) / tau
    return smooth_val, grad, p, b


def spectral_flatten(a, max_iter=200, flatten_rate=0.1):
    """Iteratively flatten the power spectrum while maintaining non-negativity."""
    a = np.maximum(a.astype(np.float64), 1e-10).copy()
    s = a.sum()
    a = a / s
    n = len(a)
    
    for i in range(max_iter):
        A = np.fft.fft(a)
        mag = np.abs(A)
        # Target: uniform magnitude (keep DC)
        target_mag = np.full(n, mag.mean())
        target_mag[0] = mag[0]  # Keep DC component fixed
        
        # Blend toward target
        new_mag = (1 - flatten_rate) * mag + flatten_rate * target_mag
        # Fix conjugate symmetry
        new_mag[n//2] = new_mag[n//2] if n % 2 == 0 else new_mag[n//2]
        
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        a = a_new
    
    return a

def coordinate_descent(a, deadline, max_rounds=1000, delta=0.01, verbose=False):
    """Coordinate descent: try perturbing individual elements."""
    a = np.maximum(a.astype(np.float64), 0).copy()
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for round_idx in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Random order of coordinates
        rng = np.random.default_rng(round_idx * 7919)
        order = rng.permutation(n)
        
        for idx in order:
            if time.monotonic() > deadline:
                break
            orig = a[idx]
            if orig < 1e-8:
                continue
            
            # Try multiple perturbations
            for frac in [1+delta, 1-delta, 1+2*delta, 1-2*delta, 1+0.5*delta, 1-0.5*delta]:
                a[idx] = max(0, orig * frac)
                s = a.sum()
                if s < 1e-10:
                    a[idx] = orig
                    continue
                a_norm = a / s
                J = compute_J(a_norm)
                if J < best_J - 1e-14:
                    a = a_norm
                    best_J = J
                    best_a = a.copy()
                    improved = True
                    break
            a[idx] = orig  # Reset for next
        
        if not improved and round_idx > 50:
            # Try larger perturbations
            delta *= 1.5
            if delta > 0.5:
                break
        
        if verbose and round_idx % 100 == 0:
            print(f"      CD round={round_idx} J={best_J:.8f} delta={delta:.4f}", flush=True)
    
    return best_a, best_J

def resample_seq(src, dest_n):
    """Resample a 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def spectral_perturb(a, rng, scale=0.1):
    """Perturb in the spectral domain."""
    n = len(a)
    A = np.fft.fft(a)
    mag = np.abs(A)
    phase = np.angle(A)
    # Perturb the phase
    phase += scale * rng.normal(0, 1, n)
    # Slightly perturb magnitude
    mag *= (1 + 0.05 * scale * rng.normal(0, 1, n))
    mag = np.maximum(mag, 0)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(np.fft.ifft(A_new))
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


class ScoreTracker:
    def __init__(self):
        self.best_a = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a, label=""):
        if a is None or len(a) == 0:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = compute_J(a)
        if score < self.best_score:
            self.best_score = score
            self.best_a = a.copy()
            elapsed = time.monotonic() - self.t_start
            msg = f"  ** J={score:.8f} n={len(a)} t={elapsed:.0f}s"
            if label:
                msg += f" [{label}]"
            print(msg, flush=True)
    
    def report(self, label=""):
        if self.best_a is not None:
            print(f"  {label}best: J={self.best_score:.8f} n={len(self.best_a)}", flush=True)

    n = len(a)
    
    # Step 1: Spectral flattening
    a_flat = spectral_flatten(a, max_iter=100, flatten_rate=0.08)
    tracker.update(a_flat, f"{label}_spectral")
    
    # Step 2: Adam optimization
    a_adam, J_adam = adam_optimize(
        a_flat, min(time.monotonic() + 60, deadline - 30),
        tau_init=20, tau_max=150, lr_init=0.03, lr_min=0.001,
        restart_interval=100
    )
    tracker.update(a_adam, f"{label}_adam")
    
    # Step 3: Squared parameterization Adam
    rng = np.random.default_rng(hash(label) % (2**31))
    b_init = np.sqrt(np.maximum(a_adam, 1e-8)) * rng.normal(1, 0.1, n)
    a_sq, J_sq = adam_squared_param(
        b_init, min(time.monotonic() + 40, deadline - 20),
        tau_init=20, tau_max=100, lr_init=0.05, lr_min=0.002
    )
    tracker.update(a_sq, f"{label}_sqr")
    
    # Step 4: Polish best with coordinate descent
    if tracker.best_a is not None:
        a_polish = tracker.best_a.copy() if len(tracker.best_a) == n else resample_seq(tracker.best_a, n)
        a_cd, J_cd = coordinate_descent(
            a_polish, min(time.monotonic() + 30, deadline - 10),
            max_rounds=200, delta=0.005
        )
        tracker.update(a_cd, f"{label}_cd")
    
    return tracker.best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for optimal step function heights:
    Phase 1: Diverse initialization + spectral flattening (200s)
    Phase 2: Adam optimization with adaptive tau (800s)
    Phase 3: Squared-parameter optimization (200s)
    Phase 4: Multi-length exploration (150s)
    Phase 5: Final polishing (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "parent")
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "default")
    
    # === PHASE 1: Diverse initialization ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Diverse Initialization + Spectral Flattening", flush=True)
    print(f"{'='*60}", flush=True)
    
    p1_deadline = time.monotonic() + 200.0
    n_main = 512
    
    candidates = []
    inits = generate_diverse_inits(n_main, rng, parent)
    
    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_deadline:
            break
        a0 = np.maximum(a0, 1e-8).astype(np.float64)
        a0 = a0 / a0.sum()
        
        # Spectral flattening
        a_flat = spectral_flatten(a0, max_iter=80, flatten_rate=0.1)
        # Quick Adam
        a_opt, J_opt = adam_optimize(a_flat, min(time.monotonic() + 15, p1_deadline - 5),
                                       tau_init=20, tau_max=80, lr_init=0.04,
                                       lr_min=0.005, restart_interval=50)
        candidates.append((J_opt, a_opt))
        print(f"  init {idx:2d}: J={J_opt:.8f}", flush=True)
    
    # Sort by quality
    candidates.sort(key=lambda x: x[0])
    
    # Update tracker
    for J_val, a_opt in candidates[:5]:
        tracker.update(a_opt, "phase1")
    
    print(f"Phase 1 best: J={candidates[0][0]:.8f}", flush=True)
    
    # === PHASE 2: Main Adam optimization ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: Main Adam Optimization", flush=True)
    print(f"{'='*60}", flush=True)
    
    p2_deadline = time.monotonic() + 800.0
    p2_start = time.monotonic()
    
    # Allocate time to top candidates
    top_k = min(4, len(candidates))
    time_alloc = np.array([0.45, 0.25, 0.15, 0.15])[:top_k] / sum(np.array([0.45, 0.25, 0.15, 0.15])[:top_k]) * 0.9
    
    for ci in range(top_k):
        if time.monotonic() > p2_deadline:
            break
        
        alloc_time = (p2_deadline - p2_start) * time_alloc[ci]
        alloc_time = max(30.0, alloc_time)
        phase_dl = min(time.monotonic() + alloc_time, p2_deadline)
        
        J_init, a_init = candidates[ci]
        print(f"\n  Candidate {ci} (init J={J_init:.8f}), budget={alloc_time:.0f}s", flush=True)
        
        a_best, J_best = adam_optimize(
            a_init, phase_dl,
            tau_init=15, tau_max=200, lr_init=0.05, lr_min=0.001,
            beta1=0.9, beta2=0.999,
            restart_interval=150, restart_scale=0.15,
            verbose=True
        )
        tracker.update(a_best, f"adam_c{ci}")
        
        # Also try with spectral perturbation
        if time.monotonic() < p2_deadline - 20:
            rng2 = np.random.default_rng(ci * 1000 + seed)
            a_pert = spectral_perturb(a_best, rng2, scale=0.2)
            a_pert_best, _ = adam_optimize(
                a_pert, min(time.monotonic() + 30, p2_deadline - 5),
                tau_init=20, tau_max=150, lr_init=0.03, lr_min=0.002,
                restart_interval=80
            )
            tracker.update(a_pert_best, f"adam_c{ci}_pert")
    
    # === PHASE 3: Squared parameterization ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Squared Parameterization Optimization", flush=True)
    print(f"={'='*59}", flush=True)
    
    p3_deadline = min(time.monotonic() + 200.0, deadline - 100.0)
    
    if tracker.best_a is not None and time.monotonic() < p3_deadline:
        best_n = len(tracker.best_a)
        
        for trial in range(3):
            if time.monotonic() > p3_deadline:
                break
            rng3 = np.random.default_rng(seed + 1000 + trial)
            b_init = np.sqrt(np.maximum(tracker.best_a, 1e-8))
            b_init = b_init * (1 + 0.1 * rng3.normal(0, 1, best_n))
            
            per_trial_dl = min(time.monotonic() + 60, p3_deadline - 10)
            a_sq, J_sq = adam_squared_param(
                b_init, per_trial_dl,
                tau_init=20, tau_max=120, lr_init=0.08, lr_min=0.003,
                restart_interval=100
            )
            tracker.update(a_sq, f"sq_trial{trial}")
    
    tracker.report("Phase 3 complete")
    
    # === PHASE 4: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    p4_deadline = min(time.monotonic() + 150.0, deadline - 50.0)
    
    if tracker.best_a is not None and time.monotonic() < p4_deadline:
        base_n = len(tracker.best_a)
        # Try different lengths
        lengths_to_try = sorted(set([
            base_n, base_n + 64, base_n - 64,
            base_n * 2, base_n // 2 if base_n > 128 else base_n,
            768, 1024, 1536
        ]))
        
        for n_try in lengths_to_try:
            if time.monotonic() > p4_deadline:
                break
            if n_try < 64 or n_try > 2048:
                continue
            if n_try == base_n:
                continue
            
            a_ext = resample_seq(tracker.best_a, n_try)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()
            
            # Quick optimization
            a_opt, _ = adam_optimize(
                a_ext, min(time.monotonic() + 30, p4_deadline - 10),
                tau_init=20, tau_max=100, lr_init=0.04, lr_min=0.003,
                restart_interval=80
            )
            tracker.update(a_opt, f"n={n_try}")
    
    tracker.report("Phase 4 complete")
    
    # === PHASE 5: Final polishing ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 5: Final Polishing", f"(remaining {deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best_a is not None and time.monotonic() < deadline - 5:
        # Coordinate descent
        a_cd, _ = coordinate_descent(
            tracker.best_a, min(time.monotonic() + 40, deadline - 20),
            max_rounds=30
