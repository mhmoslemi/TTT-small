The "spectral equalization" step:
- Compute power spectrum P = |FFT(a)|^2
- Compute autocorrelation c = IFFT(P)
- Find k* = argmax c
- Modify P to reduce contribution to c[k*]:
  - The contribution of P[m] to c[k*] is P[m]*exp(2πimk*/N)/N
  - To reduce c[k*], we want to reduce P[m] where exp(2πimk*/N) aligns with the current phase of c[k*]
  - Simple approach: P_new[m] = P[m] * (1 - eta * cos(2π*m*k*/N)) for small eta
  - This reduces P where the cosine is positive (constructive at k*) and increases where negative
- Take aFFT = sqrt(P_new) * phase(FFT(a)) (keep the phase)
- a_new = IFFT(aFFT)
- Project to non-negative and normalize

This is a form of "iterative reweighting" in the frequency domain.

2. Spectral equalization (frequency-domain peak shaving)

import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_loss(a, tau):
    """Compute smooth max of autocorrelation and its gradient."""
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    # Shift to avoid overflow
    shifted = tau * (c - c_max)
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()
    # Smooth max value
    loss = c_max + np.log(exp_c.sum()) / tau
    
    # Gradient: grad[a][i] = sum_k p[k] * 2*a[k-i] * [0<=k-i<n]
    # This is 2 * (p correlated with a)[i]
    # Cross-correlation: (p ⊛ a)[i] = sum_k p[k] * a[k-i] for valid k-i
    # Using FFT: IFFT(FFT(p) * conj(FFT(a))) gives cross-correlation
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    # Cross-correlation of p and a: sum_k p[k] * a[k-i]
    # This is IFFT(FFT(p) * conj(FFT(a))) ... let me verify
    # IFFT(FP * conj(FA))[i] = sum_k p[k] * a[k-i] (with circular correlation)
    # For linear, we need zero-padding which we have since size >= 2n-1
    cross = np.fft.irfft(fp * np.conj(fa), size)[:n]
    grad = 2.0 * cross
    
    return loss, grad, c, p, c_max

def spectral_equalization(a, n_peaks=3, alpha=0.1):
    """Reduce autocorrelation peaks by reshaping the power spectrum."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    power = np.abs(fa) ** 2
    phase = np.angle(fa)
    
    # Compute autocorrelation to find peaks
    c = np.fft.irfft(power, size)[:2*n-1]
    
    # Find the top peaks
    top_k = np.argsort(c)[-n_peaks:][::-1]
    
    # For each peak, apply a spectral modification
    m = np.arange(size // 2 + 1)
    modified_power = power.copy()
    
    for k in top_k:
        # To reduce c[k], reduce power where exp(2πimk/size) is positive (aligned)
        # c[k] = (1/size) * sum_m power[m] * exp(2πimk/size)
        # So reducing power[m] where cos(2πmk/size) > 0 helps
        cos_term = np.cos(2 * np.pi * m * k / size)
        # Damp power where cosine is positive, boost where negative
        # Use a soft version to avoid negative power
        damping = 1.0 - alpha * cos_term
        damping = np.maximum(damping, 0.1)  # ensure positive
        modified_power = modified_power * damping
    
    # Reconstruct
    new_fa = np.sqrt(np.maximum(modified_power, 0)) * np.exp(1j * phase)
    # Make it real: take real part of ifft and ensure symmetry
    new_a = np.fft.irfft(new_fa, size)[:n]
    
    return np.maximum(new_a, 0)

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def adam_step(params, grad, beta1, beta2, eps, t):
    """One step of Adam optimizer. params is a list of dicts."""
    m = params['m']
    v = params['v']
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    params['m'] = m
    params['v'] = v
    return -m_hat / (np.sqrt(v_hat) + eps)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. FFT-based smooth subgradient with adaptive Adam optimizer
2. Spectral equalization (frequency-domain peak shaving of autocorrelation)
3. Simulated-annealing-style acceptance for escaping local minima
4. Multi-length exploration with diverse initializations
5. Periodic LP-based large moves for global progress
Key innovation: combines time-domain gradient with frequency-domain reshaping,
plus a tempered acceptance criterion to navigate the non-convex landscape.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """Compute smooth max of autocorrelation and gradient w.r.t. a.
    Returns (smooth_max_value, gradient, full_autocorr, softmax_weights, c_max)."""
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    shifted = tau * (c - c_max)
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()
    smooth_val = c_max + np.log(exp_c.sum()) / tau

    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # This is 2 * cross_correlation(p, a)[i]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    cross = np.fft.irfft(fp * np.conj(fa), size)[:n]
    grad = 2.0 * cross

    return smooth_val, grad, c, p, c_max


def spectral_equalize(a, n_peaks=3, alpha=0.1, mode='cosine'):
    """Reduce autocorrelation peaks by reshaping the power spectrum."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    fa = np.fft.rfft(a, size)
    power = np.abs(fa) ** 2
    phase = np.angle(fa)

    c = np.fft.irfft(power, size)[:2 * n - 1]
    top_k = np.argsort(c)[-n_peaks:][::-1]

    m = np.arange(size // 2 + 1)
    modified_power = power.copy()

    for k in top_k:
        if k == 0:
            cos_term = np.ones(size // 2 + 1)
        else:
            cos_term = np.cos(2 * np.pi * m * k / size)
        damping = 1.0 - alpha * cos_term
        damping = np.maximum(damping, 0.05)
        modified_power = modified_power * damping

    # Normalize power to preserve total energy
    total_orig = power.sum()
    total_new = modified_power.sum()
    if total_new > 1e-10:
        modified_power = modified_power * (total_orig / total_new)

    new_fa = np.sqrt(np.maximum(modified_power, 0)) * np.exp(1j * phase)
    new_a = np.fft.irfft(new_fa, size)[:n]
    return np.maximum(new_a, 0.0)


def solve_lp_direction(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, np.inf)] * n
    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.01)


def make_palindrome(a):
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append((p / p.sum(), "parent"))
        inits.append((np.maximum(p + rng.normal(0, 0.1, n), 0.01), "parent+noise"))
        inits.append((make_palindrome(p), "parent_pal"))

    inits.append((np.ones(n), "uniform"))
    inits.append((np.maximum(0.3 + 0.7 * np.cos(np.pi * (2 * t - 1)), 0.01), "cosine_bump"))
    inits.append((np.maximum(np.exp(-4 * (t - 0.5) ** 2), 0.01), "gaussian_center"))
    inits.append((np.maximum(np.exp(-6 * t ** 2) + np.exp(-6 * (1 - t) ** 2) + 0.05, 0.01), "edge_bumps"))

    for f in [2, 3, 4, 5]:
        inits.append((np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * f * t), 0.01), f"freq_{f}"))

    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append((np.maximum(smooth + 0.1, 0.01), "rand_smooth"))

    # Block structure
    n_blocks = max(4, n // 32)
    block_vals = rng.uniform(0.2, 1.0, n_blocks)
    inits.append((np.repeat(block_vals, n // n_blocks + 1)[:n], "blocks"))

    for a, name in inits:
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
        inits[-1] = (a, name)

    return inits


def optimize_ada(a_init, deadline, verbose=False, use_lp=False, anneal_temp=0.0):
    """
    Optimize sequence using adaptive gradient + spectral equalization + optional SA.
    Returns (best_a, best_J).
    """
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    cur_J = best_J

    # Adam state
    beta1, beta2, eps = 0.75, 0.95, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    # Hyperparams
    tau = 15.0
    tau_min = 3.0
    tau_decay = 0.997
    base_lr = 0.8
    spec_interval = 30
    spec_count = 0
    lp_interval = 200 if use_lp else 10**9
    lp_count = 0

    no_improve = 0
    it = 0
    sa_temp = anneal_temp
    sa_min_temp = 1e-10

    while time.monotonic() < deadline:
        it += 1
        t_adam += 1

        # Compute gradient
        loss, grad, c, p, c_max = smooth_max_and_grad(a, tau)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)
        direction = -m_hat / (np.sqrt(v_hat) + eps)

        # Normalize direction (works in simplex since sum of direction doesn't matter after projection)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-12:
            continue
        direction = direction / dir_norm

        # Line search with multiple steps (also handles SA acceptance)
        lr = base_lr / (1.0 + 0.05 * t_adam ** 0.3)
        best_trial_J = cur_J
        best_trial_a = a.copy()

        for scale in [lr, lr * 0.7, lr * 0.5, lr * 0.3, lr * 0.1, lr * 0.05]:
            a_trial = a + scale * direction
            a_trial = np.maximum(a_trial, 1e-10)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)

                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break
            # Simulated annealing: accept worse with probability
            elif sa_temp > 1e-8 and np.random.random() < np.exp(-(J_trial - cur_J) / sa_temp):
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                break

        a = best_trial_a
        cur_J = best_trial_J

        # SA cooling
        sa_temp = max(sa_min_temp, sa_temp * 0.998)

        # Periodic spectral equalization
        if it % spec_interval == 0:
            spec_count += 1
            alpha_spec = 0.25 * (0.97 ** spec_count)
            a_spec = spectral_equalize(a, n_peaks=min(5, 2 + spec_count // 10), alpha=alpha_spec)
            a_spec = np.maximum(a_spec, 1e-8)
            a_spec = a_spec / a_spec.sum()
            J_spec = compute_J(a_spec)
            if J_spec < best_J:
                best_J = J_spec
                best_a = a_spec.copy()
                a = a_spec.copy()
                cur_J = J_spec
                no_improve = 0
            elif J_spec < cur_J:
                a = a_spec.copy()
                cur_J = J_spec
                no_improve = 0

        # Periodic LP-based large move
        if it % lp_interval == 0 and use_lp:
            lp_count += 1
            M = c_max
            g = solve_lp_direction(a, M, time_limit=3)
            if g is not None and g.sum() > 1e-10:
                g = g / g.sum()
                for t_blend in [0.3, 0.2, 0.1, 0.05]:
                    a_lp = (1 - t_blend) * a + t_blend * g
                    a_lp = np.maximum(a_lp, 1e-10)
                    a_lp = a_lp / a_lp.sum()
                    J_lp = compute_J(a_lp)
                    if J_lp < best_J:
                        best_J = J_lp
                        best_a = a_lp.copy()
                        a = a_lp.copy()
                        cur_J = J_lp
                        no_improve = 0
                        break
                    elif J_lp < cur_J:
                        a = a_lp.copy()
                        cur_J = J_lp
                        no_improve = 0
                        break

        # Update best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Decay temperature
        tau = max(tau_min, tau * tau_decay)

        # Restart if stuck
        if no_improve > 150:
            rng = np.random.default_rng(it * 104729 + n * 7)
            scale = 0.15 / (1.0 + no_improve / 300.0)

            # Try multiple perturbation types, pick best
            best_pert_a = a.copy()
            best_pert_J = cur_J
            t_arr = np.linspace(0, 1, n, endpoint=False)

                pert = np.zeros(n)
                if pert_type == 0:

                        pert += rng.uniform(0.3, 1.0) * np.cos(2 * np.pi * f * t_arr + ph)
                elif pert_type == 1:
                    mask = rng.random(n) > 0.7
                    pert[mask] = rng.normal(0, 1, mask.sum())
                elif pert_type == 2:
                    # Shift perturbation

                    pert = np.roll(a, shift) - a
                else:
                    # Random scale of a subset

                    pert = np.zeros(n)
                    pert[idx] = a[idx] * rng.normal(0, 0.5, len(idx))

                a_pert = a + scale * pert
                a_pert = np.maximum(a_pert, 1e-8)
                a_pert = a_pert / a_pert.sum()
                J_pert = compute_J(a_pert)
                if J_pert < best_pert_J:
                    best_pert_J = J_pert
                    best_pert_a = a_pert.copy()

            a = best_pert_a
            cur_J = best_pert_J
            no_improve = 0
            t_adam = 0
            m = np.zeros(n)
            v = np.zeros(n)
            tau = max(tau, 8.0)
            sa_temp = max(sa_temp, 0.05)

        if verbose and it % 300 == 0:
            print(f"      iter={it} J={best_J:.8f} tau={tau:.1f} no_imp={no_improve}", flush=True)

    return best_a, best_J

    print(f"FINAL: J={best_score:.8f} n={best_overall.size}", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_overall]
