    """Build the sparse LP constraint matrix for (a*g)[k] <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})

            # Try smaller steps or perturb
            if no_improve % 5 == 0:
                a = a + 0.01 * np.random.default_rng(it).standard_normal(n)
                a = np.maximum(a, 1e-8)
                a /= a.sum()

        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: M={best_M:.8f}", flush=True)

    return best_a


def coordinate_update_fast(a, i, b, S):
    """
    Golden-section search for optimal a[i] given others fixed.
    b = current convolution, S = current sum.
    Returns optimal value for a[i].
    """
    n = len(a)
    x_cur = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0

    affected_indices = np.arange(k_lo, k_hi + 1)
    c_vals = np.zeros(len(affected_indices))
    slopes = np.zeros(len(affected_indices))
    quad_idx = -1

    for idx, k in enumerate(affected_indices):
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            slopes[idx] = 0.0
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2 * x_cur * a[j]
                slopes[idx] = 2 * a[j]
            else:
                c_vals[idx] = b[k]
                slopes[idx] = 0.0

    def obj(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 1e-15:
            return 1e10
        vals = c_vals + slopes * x_test
        m = base_max
        for idx2 in range(len(vals)):
            if vals[idx2] > m:
                m = vals[idx2]
        if quad_idx >= 0:
            qv = c_vals[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)

    lo = 0.0
    hi = max(x_cur * 3.0, np.sqrt(max(fast_conv(a).max(), 1e-10)) + 0.5, 0.5)
    hi = min(hi, 10.0)

    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)

    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def run_coordinate_descent(a, max_sweeps, deadline, seed=0, verbose=True):
    """Run coordinate descent to minimize max(conv) with sum-normalized a."""
    rng_c = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_c.shuffle(order)

        b = fast_conv(a)
        S = a.sum()
        sweep_improved = False

        for i in order:
            x_opt = coordinate_update_fast(a, i, b, S)
            if abs(x_opt - a[i]) > 1e-10:
                a[i] = x_opt
                sweep_improved = True
                # Update b incrementally (approximate: just mark for recompute)
                # For correctness, recompute periodically
                if sweep % 10 == 0:
                    b = fast_conv(a)
                    S = a.sum()

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 5
        if no_improve > 20:
            break

        if verbose and sweep % 10 == 0:
            print(f"    CD sweep {sweep}: M={M:.8f}", flush=True)

    return best_a


def run_adam_smooth(a, max_iters, deadline, tau_start=5.0, tau_end=200.0, lr=0.03):
    """Adam optimizer on smooth max objective (normalized to sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        lr_t = lr / (1 + 0.05 * it)

        _, grad = smooth_max_grad(a, tau)

        # Project gradient to sum-zero subspace (tangent to constraint sum=1)
        grad_mean = grad.mean()
        grad = grad - grad_mean

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = fast_conv(a).max()
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()

    return best_a


def peak_rebalance(a, eta=0.1, max_iters=50, deadline=None):
    """
    Targeted peak reduction: identify the peak lag and reduce alignment there.
    a[j] <- a[j] * (1 - eta * a[(k*-j) mod n] / max_a) for the peak lag k*.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        M = b.max()

        # Compute the "alignment" pattern for the peak lag
        # contrib[j] = a[j] * a[k_star - j] for valid j
        contrib = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                contrib[j] = a[j] * a[kj]

        max_contrib = contrib.max()
        if max_contrib < 1e-15:
            break

        # Reduce elements that contribute most to the peak
        # Scale factor: reduce a[j] proportional to a[k_star - j]
        scale = np.ones(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                scale[j] = 1.0 - eta * a[kj] / (a.max() + 1e-15)
        scale = np.maximum(scale, 0.0)

        a_new = a * scale
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller eta
            eta *= 0.5
            if eta < 1e-4:
                break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.2, n), 0.01))
        # Shifted versions
        shift = n // 4
        inits.append(np.roll(p, shift).copy())
        inits.append(np.roll(p, -shift).copy())

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * t), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.exp(-3 * (t - 0.5) ** 2) + 0.1)
    inits.append(0.5 + 0.5 * np.sin(np.pi * t))
    # Ramp
    inits.append(np.linspace(0.2, 1.0, n))
    inits.append(np.linspace(1.0, 0.2, n))
    # Bimodal
    inits.append(np.maximum(np.exp(-10 * (t - 0.25) ** 2) + np.exp(-10 * (t - 0.75) ** 2) + 0.05, 0.01))
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def optimize_single(a0, n, deadline, verbose=True):
    """Run full optimization pipeline on a single starting point."""
    a0 = np.maximum(np.asarray(a0, dtype=np.float64), 1e-8)
    a0 = a0 / a0.sum()

    # Phase A: LP mirror descent
    t_lp = min(time.monotonic() + 60, deadline - 100)
    a_lp = run_lp_mirror_descent(a0, 200, t_lp, verbose=verbose)

    # Phase B: Adam smooth gradient
    t_adam = min(time.monotonic() + 40, deadline - 60)
    a_adam = run_adam_smooth(a_lp, 500, t_adam, tau_start=5, tau_end=100, lr=0.03)

    # Phase C: LP again from Adam result
    t_lp2 = min(time.monotonic() + 40, deadline - 40)
    a_lp2 = run_lp_mirror_descent(a_adam, 100, t_lp2, verbose=verbose)

    # Phase D: Coordinate descent
    t_cd = min(time.monotonic() + 30, deadline - 20)
    a_cd = run_coordinate_descent(a_lp2, 30, t_cd, verbose=verbose)

    # Phase E: Peak rebalance
    t_pr = min(time.monotonic() + 15, deadline - 10)
    a_pr = peak_rebalance(a_cd, eta=0.1, max_iters=30, deadline=t_pr)

    # Return best of all
    candidates = [a_lp, a_adam, a_lp2, a_cd, a_pr]
    best = None
    best_M = float('inf')
    for c in candidates:
        c = np.maximum(c, 0)
        s = c.sum()
        if s < 1e-10:
            continue
        c = c / s
        M = fast_conv(c).max()
        if M < best_M:
            best_M = M
            best = c.copy()

    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: LP mirror descent from parent
    Phase 2: Multi-length diverse search
    Phase 3: Gradient + coordinate descent refinement
    Phase 4: Peak rebalancing and final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform n=512, J={tracker.best_score:.6f}", flush=True)

    # === PHASE 1: LP Mirror Descent from parent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent from parent (budget ~400s)", flush=True)
    print(f"{'='*60}", flush=True)

    p1_deadline = min(time.monotonic() + 400, deadline - 800)
    parent_n = parent.size
    a_p1 = run_lp_mirror_descent(parent, 500, p1_deadline, verbose=True)
    tracker.update(a_p1)

    # Also try from parent with different seeds (perturbed)
    for trial in range(3):
        if time.monotonic() > p1_deadline:
            break
        a_pert = parent + rng.normal(0, 0.1, parent_n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert = a_pert / a_pert.sum()
        t_sub = min(time.monotonic() + 60, p1_deadline)
        a_p1b = run_lp_mirror_descent(a_pert, 100, t_sub, verbose=False)
        tracker.update(a_p1b)

    tracker.report("Phase 1 complete")

    # === PHASE 2: Multi-length diverse search ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-length search (budget ~500s)", flush=True)
    print(f"{'='*60}", flush=True)

    p2_deadline = min(time.monotonic() + 500, deadline - 400)
    lengths = [384, 512, 768, 1024, 1280]
    # Allocate time proportionally, favoring lengths near parent
    time_per_length = 80  # seconds per length

    for n in lengths:
        if time.monotonic() > p2_deadline:
            break
        t_alloc = min(time_per_length, p2_deadline - time.monotonic())
        if t_alloc < 30:
            break

        print(f"\n  Trying n={n} (budget {t_alloc:.0f}s) ...", flush=True)
        inits = diverse_inits(n, rng, tracker.best)
        best_n = None
        best_M_n = float('inf')

        for idx, a0 in enumerate(inits[:4]):  # Limit initializations per length
            if time.monotonic() > p2_deadline - 30:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            t_sub = min(time.monotonic() + 25, p2_deadline - 20)

            # Quick optimization: LP + short CD
            a_opt = run_lp_mirror_descent(a0, 50, t_sub, verbose=False)
            M_val = fast_conv(a_opt).max()
            if M_val < best_M_n:
                best_M_n = M_val
                best_n = a_opt.copy()

        if best_n is not None:
            # Further optimize the best for this length
            if time.monotonic() < p2_deadline - 30:
                t_sub2 = min(time.monotonic() + 20, p2_deadline - 10)
                best_n = run_lp_mirror_descent(best_n, 50, t_sub2, verbose=False)
                a_cd_n = run_coordinate_descent(best_n, 15, min(time.monotonic() + 10, p2_deadline - 5), verbose=False)
                M_cd = fast_conv(a_cd_n).max()
                if M_cd < fast_conv(best_n).max():
                    best_n = a_cd_n

            J_val = compute_J(best_n)
            print(f"  n={n:4d}: J={J_val:.6f}", flush=True)
            tracker.update(best_n)

    tracker.report("Phase 2 complete")

    # === PHASE 3: Intensive refinement of best found ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Intensive refinement (budget ~350s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 100:
        p3_deadline = min(time.monotonic() + 350, deadline - 100)
        best_n = tracker.best.size
        print(f"  Refining n={best_n}", flush=True)

        # Run full optimization pipeline
        a_ref = optimize_single(tracker.best, best_n, p3_deadline, verbose=True)
        tracker.update(a_ref)

        # Try from perturbed best
        if time.monotonic() < p3_deadline - 60:
            for pert_scale in [0.05, 0.15, 0.3]:
                if time.monotonic() > p3_deadline - 30:
                    break
                a_pert = tracker.best + rng.normal(0, pert_scale, best_n)
                a_pert = np.maximum(a_pert, 1e-6)
                a_pert = a_pert / a_pert.sum()
                t_sub = min(time.monotonic() + 40, p3_deadline - 20)
                a_pert_opt = run_lp_mirror_descent(a_pert, 80, t_sub, verbose=False)
                a_pert_cd = run_coordinate_descent(a_pert_opt, 20, min(time.monotonic() + 15, p3_deadline - 10), verbose=False)
                tracker.update(a_pert_cd)

        # Peak rebalancing
        if time.monotonic() < p3_deadline - 20:
            a_pr = peak_rebalance(tracker.best, eta=0.05, max_iters=50, deadline=p3_deadline - 20)
            tracker.update(a_pr)
            a_pr2 = peak_rebalance(tracker.best, eta=0.2, max_iters=30, deadline=p3_deadline - 10)
            tracker.update(a_pr2)

    tracker.report("Phase 3 complete")

    # === PHASE 4: Final polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final polish (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 5:
        best_n = tracker.best.size

        # Quick LP
        if time.monotonic() < deadline - 20:
            a_final = run_lp_mirror_descent(tracker.best, 30, deadline - 15, verbose=False)
            tracker.update(a_final)

        # Quick CD
        if time.monotonic() < deadline - 10:
            a_final2 = run_coordinate_descent(tracker.best, 20, deadline - 8, verbose=False)
            tracker.update(a_final2)

        # Peak rebalance
        if time.monotonic() < deadline - 5:
            a_final3 = peak_rebalance(tracker.best, eta=0.03, max_iters=20, deadline=deadline - 3)
            tracker.update(a_final3)

        # Try extending length
        if time.monotonic() < deadline - 30:
            for mult in [2]:
                new_n = best_n * mult
                if new_n > 2048 or new_n < 64:
                    continue
                a_ext = resample_seq(tracker.best, new_n)
                a_ext = np.maximum(a_ext, 1e-6)
                a_ext = a_ext / a_ext.sum()
                t_sub = min(time.monotonic() + 30, deadline - 10)
                a_ext_opt = run_lp_mirror_descent(a_ext, 30, t_sub, verbose=False)
                tracker.update(a_ext_opt)

    tracker.report("Phase 4 complete")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
