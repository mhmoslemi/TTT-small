```python
import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz_sub(a, active_lags):
    """Build submatrix of Toeplitz for given active lags."""
    n = len(a)
    K = len(active_lags)
    A = np.zeros((K, n))
    for row, k in enumerate(active_lags):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[row, j] = a[idx]
    return A

def lp_solve_fast(a, M, active_lags):
    """Fast LP with only active constraints."""
    n = len(a)
    K = len(active_lags)
    A = build_toeplitz_sub(a, active_lags)
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'dual_refine': True})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except:
        pass
    return None
```

```python
"""
Fast active-set LP alternation with subgradient polishing for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj_fast(a):
    """Compute objective without calling evaluate_sequence."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz_sub(a, active_lags):
    """Build K×n submatrix of the convolution operator for given lags."""
    n = len(a)
    K = len(active_lags)
    A = np.zeros((K, n))
    for row_idx in range(K):
        k = int(active_lags[row_idx])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[row_idx, j_start:j_end+1] = a[k - j_end : k - j_start + 1][::-1]
    return A

def solve_lp_direction(a, M, active_lags):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0."""
    n = len(a)
    if len(active_lags) == 0:
        return None
    A = build_toeplitz_sub(a, active_lags)
    c = -np.ones(n)
    b_ub = np.full(len(active_lags), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def lp_alternate(a, time_budget, K=50, verbose=False):
    """Fast LP alternation with active set."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    stall = 0
    it = 0
    diversify_counter = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Find active set: top lags within a fraction of max
        sorted_idx = np.argsort(conv)[::-1]
        # Use lags that are at least 90% of max
        threshold = M * 0.90
        active = sorted_idx[conv[sorted_idx] >= threshold]
        if len(active) > K:
            active = active[:K]
        if len(active) < 5:
            active = sorted_idx[:min(K, 2*n-1)]
        
        # Solve LP
        g = solve_lp_direction(a, M, active)
        
        if g is None:
            stall += 1
        else:
            S = float(np.sum(g))
            if S <= 1.0 + 1e-8:
                stall += 1
            else:
                # Line search
                g_norm = g / S
                improved = False
                for t in [1.0, 0.7, 0.5, 0.3, 0.15]:
                    a_try = (1.0 - t) * a + t * g_norm
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj_fast(a_try)
                    if obj_try < best_obj - 1e-13:
                        a = a_try
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                
                if not improved:
                    stall += 1
        
        # Diversification when stalled
        if stall > 30:
            diversify_counter += 1
            # Try different perturbation strategies
            strategy = diversify_counter % 4
            if strategy == 0:
                # Multiplicative noise
                noise = np.exp(rng.normal(0, 0.08, n))
                a = a * noise
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2
                else:
                    a = best_a.copy()
            elif strategy == 1:
                # Zero out random 10% and renormalize
                mask = rng.random(n) > 0.1
                a = a * mask
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2
                else:
                    a = best_a.copy()
            elif strategy == 2:
                # Add a random wave
                f = rng.uniform(0.5, 5.0)
                wave = 0.1 * np.cos(2 * np.pi * f * np.arange(n) / n)
                a = np.maximum(a + wave, 0)
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2
                else:
                    a = best_a.copy()
            else:
                # Reset to best and try larger perturbation
                a = best_a * np.exp(rng.normal(0, 0.15, n))
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2
                else:
                    a = best_a.copy()
            
            stall = 0
        
        it += 1
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      LP alt: it={it}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s, stall={stall}")
    
    return best_a, best_obj

def subgradient_polish(a, time_budget, verbose=False):
    """Polish using projected subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    lr = 0.005
    patience = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Active set: lags within 2% of max
        threshold = M * 0.98
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        
        # Compute subgradient of max_k conv(a,a)[k]
        # grad[i] = (1/|S|) * sum_{k in S} 2*a[k-i] (if valid)
        grad = np.zeros(n)
        for k in active:
            k_int = int(k)
            for i in range(n):
                j = k_int - i
                if 0 <= j < n:
                    grad[i] += 2.0 * a[j]
        grad /= len(active)
        
        # Projected gradient step
        # Direction: -grad (to decrease max conv)
        # But we need to maintain sum(a) = 1 and a >= 0
        # Simple approach: a_new = a - lr*grad, project
        a_new = a - lr * grad
        # Make non-negative
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj_fast(a_new)
        
        if obj_new < best_obj - 1e-14:
            a = a_new
            best_obj = obj_new
            best_a = a.copy()
            patience = 0
            lr = min(lr * 1.05, 0.05)
        else:
            patience += 1
            lr *= 0.95
            if patience > 200:
                # Reset learning rate and try a perturbation
                lr = 0.01
                rng = np.random.default_rng(int(time.monotonic() * 10000))
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2
                patience = 0
    
    return best_a, best_obj

def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)
```

Actually, the subgradient at index i is:
grad[i] = (1/|S|) * sum_{k in S} 2*a[k-i] (if 0 ≤ k-i < n)

For a fixed i, this is sum_{k in S} 2*a[k-i] where k-i is in [0, n-1], i.e., k is in [i, i+n-1].

So grad[i] = (2/|S|) * sum_{k in S, i ≤ k ≤ i+n-1} a[k-i]

Let me define for each active lag k, the contribution to grad[i] is 2*a[k-i] for i in [max(0, k-n+1), min(k, n-1)].

This is essentially: for each active k, add 2*a reversed to the appropriate slice of grad.

```python
"""
Fast active-set LP alternation with subgradient polishing for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses top-K constraint LP for speed, vectorized subgradients for polishing,
and multi-length exploration to avoid local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj_fast(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz_sub(a, active_lags):
    """Build K x n submatrix of convolution operator for given active lags."""
    n = len(a)
    K = len(active_lags)
    A = np.zeros((K, n))
    for row_idx in range(K):
        k = int(active_lags[row_idx])
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[row_idx, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_dir(a, M, active_lags):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_lags, g >= 0."""
    n = len(a)
    if len(active_lags) == 0:
        return None
    A = build_toeplitz_sub(a, active_lags)
    c = -np.ones(n)
    b_ub = np.full(len(active_lags), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def compute_subgradient(a, active_lags):
    """Compute subgradient of max_k conv(a,a)[k] at given active set."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            grad[j_start:j_end + 1] += 2.0 * a[k - j_end:k - j_start + 1][::-1]
    grad /= max(1, len(active_lags))
    return grad


def lp_alternate(a, time_budget, K=50, verbose=False):
    """Fast LP alternation with adaptive active set and relaxation."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    stall = 0
    it = 0
    relax = 1.0
    rng = np.random.default_rng(42)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))

        # Find active set: lags near the maximum
        sorted_idx = np.argsort(conv)[::-1]
        # Include lags that are at least 85% of max
        threshold = M * 0.85
        active = sorted_idx[conv[sorted_idx] >= threshold]
        if len(active) > K:
            active = active[:K]
        if len(active) < 8:
            active = sorted_idx[:min(K, 2 * n - 1)]

        # Solve LP with current relaxation
        g = solve_lp_dir(a, M * relax, active)

        if g is None:
            stall += 1
        else:
            S = float(np.sum(g))
            if S <= 1.0 + 1e-7:
                stall += 1
            else:
                g_norm = g / S
                improved = False
                # Multi-resolution line search
                for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1]:
                    a_try = (1.0 - t) * a + t * g_norm
                    a_try = np.maximum(a_try, 0)
                    s_try = float(np.sum(a_try))
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj_fast(a_try)
                    if obj_try < best_obj - 1e-14:
                        a = a_try
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        # Reduce relaxation back toward 1
                        relax = max(1.0, relax - 0.03)
                        break

                if not improved:
                    stall += 1

        # Handle stalling
        if stall > 0 and stall % 10 == 0:
            relax = min(relax + 0.05, 1.8)

        if stall > 40:
            # Diversify
            strategy = (stall // 40) % 5
            if strategy == 0:
                a = a * np.exp(rng.normal(0, 0.06, n))
            elif strategy == 1:
                mask = rng.random(n) > 0.08
                a = a * mask
            elif strategy == 2:
                f = rng.uniform(0.5, 6.0)
                wave = 0.15 * np.sin(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 6.28))
                a = np.maximum(a + wave, 0)
            elif strategy == 3:
                # Swap two random segments
                p1 = rng.integers(0, n // 2)
                p2 = rng.integers(n // 2, n)
                length = rng.integers(5, n // 5)
                a = a.copy()
                temp = a[p1:p1 + length].copy()
                a[p1:p1 + length] = a[p2:p2 + length]
                a[p2:p2 + length] = temp
            else:
                a = best_a * np.exp(rng.normal(0, 0.12, n))

            a = np.maximum(a, 0)
            s2 = float(np.sum(a))
            if s2 > 1e-10:
                a = a / s2
            else:
                a = best_a.copy()
            stall = 0
            relax = 1.0

        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"        LP: it={it}, obj={best_obj:.8f}, t={elapsed:.1f}s, stall={stall}, relax={relax:.2f}")

    return best_a, best_obj


def subgradient_polish(a, time_budget, verbose=False):
    """Polish using projected subgradient descent on max convolution."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    lr = 0.003
    patience = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))

        # Active set: lags within 3% of max
        threshold = M * 0.97
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        # Limit active set size for speed
        if len(active) > 80:
            active = active[:80]

        # Compute subgradient
        grad = compute_subgradient(a, active)

        # Projected gradient step: a - lr * grad, project to {a>=0, sum=1}
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj_new = compute_obj_fast(a_new)

        if obj_new < best_obj - 1e-15:
            a = a_new
            best_obj = obj_new
            best_a = a.copy()
            patience = 0
            lr = min(lr * 1.03, 0.03)
        else:
            patience += 1
            lr *= 0.97
            if patience > 300:
                # Perturb to escape
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                s2 = float(np.sum(a))
                if s2 > 1e-10:
                    a = a / s2
                lr = 0.005
                patience = 0

    return best_a, best_obj


def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 7)

    if choice == 0:

    elif choice == 1:

        f2 = rng.uniform(3.0, 10.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 6.28))
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 40)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.5, 8.0)
            amp = rng.uniform(0.05, 0.25)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 6.28))
    elif choice == 4:
        # Block structure
        nblocks = rng.integers(3, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.8)
    elif choice == 5:
        # Asymmetric
        a = np.ones(n)
        a[:n // 3] *= rng.uniform(0.4, 1.0)
        a[n // 3:2 * n // 3] *= rng.uniform(0.8, 1.5)
        a[2 * n // 3:] *= rng.uniform(0.4, 1.0)
    else:
        # Periodic with noise
        period = rng.integers(max(5, n // 20), max(10, n // 5))
        a = np.ones(n)
        for i in range(n):
            a[i] = 0.5 + 0.5 * np.cos(2 * np.pi * i / period)
        a *= np.exp(rng.normal(0, 0.1, n))

    a = np.maximum(a, 0.001)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

        # Combined polish (alternates LP + subgradient)
        a_comb, obj_comb = combined_polish(best_seq, time_budget=remaining * 0.7)
        if obj_comb < best_score:
            best_score = obj_comb
            best_seq = a_comb.copy()
            print(f"  After combined: {best_score:.8f}")

        # Final quick LP with larger active set
        if time.monotonic() < deadline - 20:
            remaining2 = deadline - time.monotonic()
            alloc_final = min(remaining2 * 0.7, 80.0)
            a_fin, obj_fin = lp_alternate(best_seq, time_budget=alloc_final, K=80)
            if obj_fin < best_score:
                best_score = obj_fin
                best_seq = a_fin.copy()
                print(f"  After final LP: {best_score:.8f}")

        # Very final subgradient
        if time.monotonic() < deadline - 10:
            remaining3 = deadline - time.monotonic()
            alloc_sg = min(remaining3 * 0.7, 30.0)
            a_vf, obj_vf = subgradient_polish(best_seq, time_budget=alloc_sg)
            if obj_vf < best_score:
                best_score = obj_vf
                best_seq = a_vf.copy()
                print(f"  After final subgrad: {best_score:.8f}")

    # Fallback if something went wrong
    if best_seq is None:
        best_seq = np.ones(500)
        best_score = compute_obj_fast(best_seq)
        print(f"  FALLBACK: n=500, obj={best_score:.8f}")

    # Final validation using evaluate_sequence
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        best_score = val
    else:
        # Should not happen, but fallback
        result = [0.01] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL RESULT ===")
    print(f"  n = {len(result)}")
    print(f"  score = {best_score:.8f}")
    print(f"  time = {elapsed:.1f}s")

    return result
```
