def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_J(a):

def build_sparse_conv_matrix(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] if valid, for cross-conv constraints."""
    rows = []
    cols = []
    vals = []

        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M, reg=0.0):
    """
    LP/QP: find g that maximizes sum(g) - reg*||g||^2/2
    subject to (a*g)[k] <= M for all k, g >= 0.
    If reg=0, this is a pure LP.

    A = build_sparse_conv_matrix(n, a)

    bounds = [(0, None)] * n

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 60, 'presolve': True})

                g = np.maximum(r.x, 0)
                if g.sum() > 1e-15:
                    return g

            pass
    else:

        # Maximize sum(g) - reg/2 * ||g||^2 subject to A@b_ub... 
        # Equivalent to: minimize -sum(g) + reg/2 * ||g||^2 s.t. A@g <= M, g >= 0
        def obj(g):
            return -g.sum() + 0.5 * reg * np.dot(g, g)
        
        # Constraints: A@g <= M
        from scipy.optimize import LinearConstraint
        constraint = LinearConstraint(A, -np.inf, M)

        g0 = np.maximum(g0 * M * n / max((A @ g0).max(), 1e-15), 0)

        except Exception:

    return None

def resample(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)

    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Vectorized coordinate descent with incremental convolution updates (fast polisher)

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_sparse_conv_matrix(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] if valid."""
    rows = np.empty((2 * n - 1) * n, dtype=np.int32)
    cols = np.empty((2 * n - 1) * n, dtype=np.int32)
    vals = np.empty((2 * n - 1) * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=30):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_sparse_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_bilinear(b_aa, b_ag, b_gg):
    """Find t minimizing max((1-t)^2*b_aa + 2t(1-t)*b_ag + t^2*b_gg)."""
    def obj(t):
        coeff = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
        return coeff.max()
    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                             options={'xatol': 1e-9})
    return float(result.x)

def lp_mirror_descent(a, deadline, max_iters=5000, cd_every=40, lp_time=30):

    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        g = lp_best_response(a, M, time_limit=lp_time)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)
        t_opt = line_search_bilinear(b_aa, b_ag, b_gg)

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        J_new = compute_J(a_new)

        if J_new < best_J - 1e-13:
            best_a = a_new.copy()
            best_J = J_new
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if cd_every > 0 and (it + 1) % cd_every == 0:
            a_cd = cd_incremental(a, min(time.monotonic() + 5, deadline), max_passes=3)

                best_a = a_cd.copy()
            a = a_cd
            no_improve = 0

            a = a + np.random.normal(0, 0.03, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0

            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            j_vals = np.arange(j_min, j_max + 1)
            if len(j_vals) == 0:
                continue
            contribs = a[j_vals] * a[k - j_vals]
            top_idx = np.argmax(contribs)
            top_j = j_vals[top_idx]
            pair_j = k - top_j

            # Reduce the smaller of a[top_j] and a[pair_j]
            if pair_j != top_j and 0 <= pair_j < n:
                reduce_idx = top_j if a[top_j] <= a[pair_j] else pair_j
            else:
                reduce_idx = top_j

            old_val = a[reduce_idx]
            for frac in [0.7, 0.5, 0.3]:
                a[reduce_idx] = old_val * frac
                a_pos = np.maximum(a, 0)
                s_check = a_pos.sum()
                if s_check < 0.01:
                    continue
                a_norm = a_pos / s_check
                J_check = compute_J(a_norm)
                if J_check < best_J - 1e-13:
                    best_J = J_check
                    best_a = a_norm.copy()
                    a = a_norm
                    improved = True
                    break
            if not improved:
                a[reduce_idx] = old_val
                a = np.maximum(a, 0)
                a = a / a.sum()

        if not improved:
            # Try increasing low-valued coordinates
            sort_idx = np.argsort(a)
            for idx in sort_idx[:5]:
                old_val = a[idx]
                a[idx] = old_val + 0.1 * a.mean()
                a_pos = np.maximum(a, 0)
                if a_pos.sum() < 0.01:
                    a[idx] = old_val
                    continue
                a_norm = a_pos / a_pos.sum()
                J_check = compute_J(a_norm)
                if J_check < best_J - 1e-13:
                    best_J = J_check
                    best_a = a_norm.copy()
                    a = a_norm
                    improved = True
                    break
            if not improved:
                a[idx] = old_val
                a = np.maximum(a, 0)
                a = a / a.sum()
                break

    return best_a


def generate_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Geometric
    for ratio in [0.85, 0.75, 0.65, 0.5, 1.15, 1.3]:
        g = ratio ** np.arange(n)
        inits.append(g / g.sum())

    # Exponential from various centers
    for center in [0.25, 0.35, 0.45, 0.5, 0.55, 0.65, 0.75]:
        for c in [3, 6, 12, 24]:
            e = np.exp(-c * (t - center) ** 2) + 0.01
            inits.append(e / e.sum())

    # Power law
    for alpha in [-1.0, -0.5, 0.5, 1.0, 2.0]:
        p = (t + 0.02) ** alpha
        inits.append(p / p.sum())

    # Cosine modulated
    for freq in [1, 2, 3, 5]:
        for phase in [0, 0.25, 0.5, 0.75]:
            c = 0.5 + 0.4 * np.cos(2 * np.pi * freq * t + phase * 2 * np.pi)
            c = np.maximum(c, 0.01)
            inits.append(c / c.sum())

    # Random smooth
    for _ in range(4):
        r = rng.uniform(0, 1, n)
        ks = max(3, n // 25)
        kernel = np.ones(ks) / ks
        sm = np.convolve(r, kernel, mode='same')
        sm = np.maximum(sm + 0.1, 0.01)
        inits.append(sm / sm.sum())

    # Pure random
    r = rng.uniform(0.1, 1.0, n)
    inits.append(r / r.sum())

    # Two-sided (high in middle)
    for width in [0.2, 0.35, 0.5]:
        two = np.exp(-((t - 0.5) / width) ** 2) + 0.02
        inits.append(two / two.sum())

    return inits


def resample(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def sort_by_first(x):
    """Sort key function (top-level to avoid lambda)."""
    return x[0]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Phase 1: LP mirror descent from parent (fast iterations)
    Phase 2: Intensive coordinate descent polishing
    Phase 3: Peak shaving
    Phase 4: Multi-start diverse LP
    Phase 5: Multi-length exploration
    Phase 6: Final combined polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 10

    best_seq_list = [None]
    best_score_list = [float('inf')]

    def update_best(a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score_list[0]:
            best_score_list[0] = score
            best_seq_list[0] = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score_list[0]:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent)

    # === PHASE 1: LP mirror descent from parent ===
    p1_deadline = t_start + 420
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent (n={parent.size}, budget=420s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    a = parent.copy() / parent.sum()
    a_r, J_r = lp_mirror_descent(a, min(p1_deadline, deadline - 300),
                                  max_iters=8000, cd_every=50, lp_time=25)
    update_best(a_r)
    print(f"  Phase 1 done: J={J_r:.8f}", flush=True)

    # === PHASE 2: Intensive coordinate descent ===
    p2_deadline = t_start + 650
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Intensive CD Polishing (budget=230s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq_list[0] is not None:
        a_cd = cd_incremental(best_seq_list[0], min(p2_deadline, deadline - 250), max_passes=50)
        update_best(a_cd)
        # Also polish LP result
        a_cd2 = cd_incremental(a_r, min(p2_deadline, deadline - 250), max_passes=30)
        update_best(a_cd2)
        print(f"  Phase 2 done: best J={best_score_list[0]:.8f}", flush=True)

    # === PHASE 3: Peak shaving ===
    p3_deadline = t_start + 850
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Peak Shaving (budget=200s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq_list[0] is not None:
        a_ps = peak_shaving(best_seq_list[0], min(p3_deadline, deadline - 200), max_iters=300)
        update_best(a_ps)
        # CD after peak shaving
        a_cd = cd_incremental(a_ps, min(p3_deadline + 50, deadline - 150), max_passes=20)
        update_best(a_cd)
        print(f"  Phase 3 done: best J={best_score_list[0]:.8f}", flush=True)

    # === PHASE 4: Multi-start diverse LP ===
    p4_deadline = t_start + 1150
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Multi-start diverse LP (budget=300s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    n_base = best_seq_list[0].size if best_seq_list[0] is not None else 512
    inits = generate_inits(n_base, rng)

    # Add perturbed best
    if best_seq_list[0] is not None:
        for sigma in [0.03, 0.07, 0.15]:
            pert = best_seq_list[0] + rng.normal(0, sigma, n_base)
            pert = np.maximum(pert, 0.001)
            inits.append(pert / pert.sum())

    # Score and sort initializations
    init_scores = []
    for init in inits:
        init_scores.append((compute_J(init), init))
    init_scores.sort(key=sort_by_first)

    max_starts = 12
    for score_init, init in init_scores[:max_starts]:
        if time.monotonic() > p4_deadline:
            break
        remaining = p4_deadline - time.monotonic()
        time_per = max(10, remaining / max(1, max_starts))
        phase_dl = min(time.monotonic() + time_per, p4_deadline)
        a_r, J_r = lp_mirror_descent(init, phase_dl, max_iters=800, cd_every=30, lp_time=15)
        update_best(a_r)
        a_cd = cd_incremental(a_r, min(phase_dl + 15, p4_deadline), max_passes=5)
        update_best(a_cd)
        print(f"  init_J={score_init:.6f} -> J={J_r:.8f}", flush=True)

    print(f"  Phase 4 done: best J={best_score_list[0]:.8f}", flush=True)

    # === PHASE 5: Multi-length exploration ===
    p5_deadline = t_start + 1350
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Multi-length exploration (budget=200s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq_list[0] is not None:
        n_cur = best_seq_list[0].size
        target_lengths = set()
        for mult in [0.75, 1.25, 1.5, 2.0]:
            target_lengths.add(int(n_cur * mult))
        for L in [256, 384, 512, 640, 768, 1024, 1536, 2048]:
            target_lengths.add(L)
        target_lengths = sorted([L for L in target_lengths if 128 <= L <= 2048 and L != n_cur])

        for tgt_n in target_lengths[:8]:
            if time.monotonic() > p5_deadline:
                break
            a_ext = resample(best_seq_list[0], tgt_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()

            phase_dl = min(time.monotonic() + 22, p5_deadline)
            a_r, J_r = lp_mirror_descent(a_ext, phase_dl, max_iters=400, cd_every=25, lp_time=10)
            update_best(a_r)
            a_cd = cd_incremental(a_r, min(phase_dl + 10, p5_deadline), max_passes=5)
            update_best(a_cd)
            print(f"  n={tgt_n}: J={J_r:.8f}", flush=True)

    # === PHASE 6: Final combined polish ===
    p6_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 6: Final Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq_list[0] is not None:
        # One more LP pass
        a_f = best_seq_list[0].copy() / best_seq_list[0].sum()
        remaining_lp = max(5, (t_start + budget_s - 40) - time.monotonic())
        a_r, J_r = lp_mirror_descent(a_f, min(t_start + budget_s - 40, deadline),
                                      max_iters=3000, cd_every=30, lp_time=20)
        update_best(a_r)
        # CD
        a_cd = cd_incremental(a_r, min(t_start + budget_s - 20, deadline), max_passes=30)
        update_best(a_cd)
        # Peak shave
        a_ps = peak_shaving(a_cd, deadline, max_iters=100)
        update_best(a_ps)
        # Final CD
        a_cd2 = cd_incremental(a_ps, deadline, max_passes=10)
        update_best(a_cd2)

    # Fallback
    if best_seq_list[0] is None:
        best_seq_list[0] = np.ones(128) / 128
        update_best(best_seq_list[0])

    final_j = best_score_list[0]
    final_n = best_seq_list[0].size
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={final_j:.8f} n={final_n} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq_list[0]]
