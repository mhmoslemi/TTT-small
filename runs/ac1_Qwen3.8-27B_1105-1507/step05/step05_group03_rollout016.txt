def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)

def cross_conv(a, b):

def line_search(a, g, iters=50):
    """Find best t in [0,1] minimizing max of conv((1-t)*a + t*g)."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    lo = 0.0
    hi = 1.0
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    return t

def peak_shave(a, max_iters, deadline, eta=0.1, topk=10):
    """Multiplicative peak-shaving update."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv = fast_conv(a)
        # Find top peaks
        peak_idx = np.argsort(conv)[-topk:]
        
        # Compute gradient-like direction
        grad = np.zeros(n)
        for ki in peak_idx:
            k = int(ki)
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                grad[j] += a[k - j]
        
        # Multiplicative update: reduce entries that contribute to peaks
        a_new = a * np.exp(-eta * grad / max(grad.max(), 1e-10))
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:

    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive top-k constrained LP mirror descent (much faster than full LP)
2. Multiplicative peak-shaving updates for local refinement
3. Diverse multi-start exploration (gaussian, triangle, U-shape, random, resampled parent)
4. Full-LP final polish with alternating peak-shave
5. Adaptive k schedule: start small, grow as convergence progresses
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_topk_lp(a, M, k):
    """Build LP matrix using only top-k convolution peaks as constraints."""
    n = len(a)
    conv = fast_conv(a)
    k_eff = min(k, 2 * n - 1)
    topk_idx = np.argsort(conv)[-k_eff:]
    topk_idx = np.sort(topk_idx).astype(int)

    rows_list = []
    cols_list = []
    for idx, ki in enumerate(topk_idx):
        kk = int(ki)
        j_min = max(0, kk - n + 1)
        j_max = min(n - 1, kk)
        for j in range(j_min, j_max + 1):
            rows_list.append(idx)
            cols_list.append(j)

    rows_arr = np.array(rows_list, dtype=np.int32)
    cols_arr = np.array(cols_list, dtype=np.int32)
    vals = np.array([a[kk - j] for kk, j in zip(topk_idx, cols_arr)], dtype=np.float64)
    A = csr_matrix((vals, (rows_arr, cols_arr)), shape=(k_eff, n))
    return A


def solve_topk_lp(a, M, k, time_limit=5.0):
    """Solve LP: max 1^T g s.t. (a@g)[topk] <= M, g >= 0."""
    n = len(a)
    try:
        A = build_topk_lp(a, M, k)
        c = -np.ones(n)
        b_ub = np.full(A.shape[0], M)
        bounds = [(0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def solve_full_lp(a, M, row_idx, col_idx, time_limit=8.0):
    """Solve full LP with all convolution constraints."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def build_full_lp_matrix(n):
    """Precompute row/col indices for the full LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def line_search(a, g, iters=50):
    """Find best t in [0,1] minimizing max of conv((1-t)*a + t*g). Both normalized to sum 1."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    return max(0.0, min(1.0, t))


def line_search_multi(a, g):
    """Try multiple t values and return the best one."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    best_t = 0.0
    best_val = np.max(a_coef)

    for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
        val = np.max(a_coef + b_coef * t + c_coef * t * t)
        if val < best_val:
            best_val = val
            best_t = t

    # Refine around best_t with ternary search
    lo = max(0.0, best_t - 0.05)
    hi = min(1.0, best_t + 0.05)
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    val = np.max(a_coef + b_coef * t + c_coef * t * t)
    if val < best_val:
        best_t = t
        best_val = val

    return max(0.0, min(1.0, best_t))


def run_topk_lp_descent(a, max_iters, deadline, k_start=30, k_end=80, verbose=False):
    """LP mirror descent with adaptively growing top-k constraints."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive k: grow from k_start to k_end over iterations
        frac = min(1.0, it / max(1, max_iters))
        k_cur = int(k_start + (k_end - k_start) * frac)
        k_cur = min(k_cur, 2 * n - 1)

        M_cur = float(fast_conv(a).max())
        g = solve_topk_lp(a, M_cur, k=k_cur, time_limit=5.0)
        if g is None:
            # Try larger k as fallback
            g = solve_topk_lp(a, M_cur, k=min(150, 2 * n - 1), time_limit=8.0)
            if g is None:
                break

        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t = line_search_multi(a, g)

        if t < 1e-8:
            no_improve += 1
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                J_new = compute_J(a_new)
                if J_new < best_J - 1e-12:
                    best_J = J_new
                    best_a = a_new.copy()
                    no_improve = 0
                    a = a_new
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 100:
            break

        if verbose and it % 50 == 0:
            print(f"      iter {it}: J={compute_J(a):.10f} best={best_J:.10f} t={t:.6f} k={k_cur}", flush=True)

    return best_a


def run_full_lp_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """Full LP mirror descent (all constraints)."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M_cur = float(fast_conv(a).max())
        g = solve_full_lp(a, M_cur, row_idx, col_idx, time_limit=8.0)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t = line_search_multi(a, g)

        if t < 1e-8:
            no_improve += 1
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                J_new = compute_J(a_new)
                if J_new < best_J - 1e-12:
                    best_J = J_new
                    best_a = a_new.copy()
                    no_improve = 0
                    a = a_new
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 60:
            break

        if verbose and it % 20 == 0:
            print(f"      iter {it}: J={compute_J(a):.10f} best={best_J:.10f} t={t:.6f}", flush=True)

    return best_a


def peak_shave(a, max_iters, deadline, eta=0.05, topk=15):
    """Multiplicative peak-shaving: reduce elements contributing most to top peaks."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        peak_idx = np.argsort(conv)[-topk:]

        grad = np.zeros(n)
        for ki in peak_idx:
            kk = int(ki)
            j_min = max(0, kk - n + 1)
            j_max = min(n - 1, kk)
            for j in range(j_min, j_max + 1):
                grad[j] += a[kk - j]

        gmax = grad.max()
        if gmax < 1e-15:
            break

        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

    return best_a


def smooth_gradient_step(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*(a*a))) at a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Numerical stability
    shifted -= shifted.max()
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[j] gets contribution from 2*pk*a[k-j] for j in [lo, hi]
        for j in range(lo, hi + 1):
            k_minus_j = k - j
            if 0 <= k_minus_j < n:
                grad[j] += 2.0 * pk * a[k_minus_j]

    return grad


def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0):
    """Smooth gradient descent with adaptive tau (log-sum-exp smoothing)."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = 0.5 * (0.01) ** (frac * 0.7)

        grad = smooth_gradient_step(a, tau)
        v = 0.85 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        J_cur = compute_J(a)
        if J_cur < best_J:
            best_J = J_cur
            best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.zeros(dest_n)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_gaussian(n, sigma_frac=0.35):
    """Gaussian-shaped sequence."""
    x = np.linspace(-1, 1, n)
    sigma = sigma_frac
    return np.exp(-x ** 2 / (2 * sigma ** 2))


def make_triangle(n):
    """Triangle-shaped sequence."""
    x = np.linspace(0, 1, n)
    return np.minimum(x, 1 - x) * 2 + 0.01


def make_ushape(n, alpha=0.6):
    """U-shaped (bathtub) sequence - high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    return (1.0 - np.sin(np.pi * x)) ** alpha + 0.01


def make_bump(n, width=0.4):
    """Raised cosine bump."""
    x = np.linspace(0, 1, n)
    center = 0.5
    val = np.zeros(n)
    mask = np.abs(x - center) < width
    x_masked = x[mask]
    val[mask] = 0.5 * (1 + np.cos(np.pi * (x_masked - center) / width))
    return val + 0.01


def make_random_smooth(n, rng, num_harmonics=8):
    """Generate a smooth random non-negative sequence using random Fourier series."""
    t = np.arange(n) / n
    signal = np.zeros(n)
    for i in range(1, num_harmonics + 1):
        amp = rng.uniform(0.3, 1.0) / np.sqrt(i)
        phase = rng.uniform(0, 2 * np.pi)
        signal += amp * np.cos(2 * np.pi * i * t + phase)
    signal = np.abs(signal) + 0.1
    return signal


class Tracker:
    """Tracks the best sequence found."""

    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self):
        if self.best is not None:
            print(f"  best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    1. Diverse multi-start with quick evaluation
    2. Deep top-k LP optimization on best candidates (adaptive k)
    3. Peak shaving + smooth gradient for local refinement
    4. Full LP polish on the single best
    5. Final extended polish with alternating methods
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = make_gaussian(512)
        tracker.update(parent, "(default)")

    # ======== PHASE 1: Diverse Multi-Start ========
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Diverse Multi-Start Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    candidates = []

    # Parent at original length
    if parent is not None and parent.sum() > 0:
        candidates.append(("parent", parent.copy()))

    # Parent resampled to various lengths
    for n_trial in [384, 448, 512, 640, 768, 1024]:
        r = resample_seq(parent, n_trial)
        if r.sum() > 0.01:
            candidates.append((f"parent_res_{n_trial}", r))

    # Different shapes
    for n_val in [384, 512, 640, 768]:
        candidates.append((f"gauss_{n_val}", make_gaussian(n_val, 0.35)))
        candidates.append((f"gauss_w_{n_val}", make_gaussian(n_val, 0.25)))
        candidates.append((f"bump_{n_val}", make_bump(n_val, 0.45)))
        candidates.append((f"u_{n_val}", make_ushape(n_val)))
        candidates.append((f"tri_{n_val}", make_triangle(n_val)))
        candidates.append((f"rand_{n_val}", make_random_smooth(n_val, rng)))

    # Quick evaluation of all candidates
    quick_results = []
    for name, seq in candidates:
        seq = np.maximum(seq, 1e-6)
        s = seq.sum()
        if s < 0.01:
            continue
        seq = seq / s
        J_val = compute_J(seq)
        quick_results.append((J_val, name, seq.copy(), len(seq)))

    # Sort and keep top candidates
    quick_results.sort(key=lambda x: x[0])
    print(f"  Total candidates: {len(quick_results)}", flush=True)
    print("  Top 12 by initial J:", flush=True)
    for J, name, _, n in quick_results[:12]:
        print(f"    {name}: J={J:.8f} n={n}", flush=True)

    # ======== PHASE 2: Deep Top-k LP on top candidates ========
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Deep Top-k LP Optimization (adaptive k)", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase2_deadline = min(t_start + budget_s * 0.55, deadline)
    top_k_candidates = quick_results[:6]

    # Sort by J but ensure diversity in n
    seen_n = set()
    diverse_candidates = []
    for item in top_k_candidates:
        n_val = item[3]
        if n_val not in seen_n:
            diverse_candidates.append(item)
            seen_n.add(n_val)
    # Fill remaining slots from top_k_candidates
    for item in top_k_candidates:
        if item not in diverse_candidates:
            diverse_candidates.append(item)
        if len(diverse_candidates) >= 5:
            break

    for J_init, name, a_init, n_val in diverse_candidates:
        if time.monotonic() > phase2_deadline:
            break

        print(f"\n  [{name}] n={n_val}, init J={J_init:.8f}", flush=True)

        # Sub-phase 2a: Quick peak shave
        sub_dl = min(time.monotonic() + 8, phase2_deadline)
        a_ps = peak_shave(a_init, 150, sub_dl, eta=0.04, topk=12)
        tracker.update(a_ps, f"(P2a {name} shave)")

        # Sub-phase 2b: Top-k LP descent (main workhorse)
        sub_dl2 = min(time.monotonic() + 180, phase2_deadline)
        a_opt = run_topk_lp_descent(a_ps, 2000, sub_dl2, k_start=25, k_end=70, verbose=True)
        tracker.update(a_opt, f"(P2b {name} topk-LP)")

        # Sub-phase 2c: Peak shave + repeat LP
        if time.monotonic() < phase2_deadline - 10:
            sub_dl3 = min(time.monotonic() + 5, phase2_deadline)
            a_ps2 = peak_shave(a_opt, 80, sub_dl3, eta=0.03, topk=20)
            tracker.update(a_ps2, f"(P2c {name} shave2)")

            sub_dl4 = min(time.monotonic() + 60, phase2_deadline)
            a_opt2 = run_topk_lp_descent(a_ps2, 1000, sub_dl4, k_start=40, k_end=70, verbose=False)
            tracker.update(a_opt2, f"(P2d {name} topk-LP2)")

    # ======== PHASE 3: Smooth gradient polish ========
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Smooth Gradient Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase3_deadline = min(t_start + budget_s * 0.70, deadline)
    if tracker.best is not None and time.monotonic() < phase3_deadline:
        a_sg = tracker.best.copy()
        a_sg = a_sg / a_sg.sum()
        a_smooth = run_smooth_gradient(a_sg, 2000, phase3_deadline, initial_tau=10, final_tau=300)
        tracker.update(a_smooth, "(P3 smooth)")

        # Follow up with top-k LP
        if time.monotonic() < phase3_deadline:
            sub_dl = min(time.monotonic() + 60, phase3_deadline)
            a_lp = run_topk_lp_descent(a_smooth, 800, sub_dl, k_start=40, k_end=70, verbose=False)
            tracker.update(a_lp, "(P3 smooth+LP)")

    # ======== PHASE 4: Full LP Polish ========
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Full LP Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    phase4_deadline = min(t_start + budget_s * 0.88, deadline)
    if tracker.best is not None and time.monotonic() < phase4_deadline:
        n_best = tracker.best.size
        r_idx, c_idx = build_full_lp_matrix(n_best)
        a = tracker.best.copy()
        a = a / a.sum()

        a_opt = run_full_lp_descent(a, 3000, phase4_deadline, r_idx, c_idx, verbose=True)
        tracker.update(a_opt, "(P4 full LP)")

    # ======== PHASE 5: Final Extended Polish ========
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Final Extended Polish (alternating)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_idx, c_idx = build_full_lp_matrix(n_best)
        a = tracker.best.copy()
        a = a / a.sum()

        for round_num in range(8):
            if time.monotonic() > deadline:
                break

            # Peak shave
            sub_dl_ps = min(time.monotonic() + 4, deadline)
            a_ps = peak_shave(a, 60, sub_dl_ps, eta=0.015, topk=25)
            tracker.update(a_ps, f"(P5 r{round_num} shave)")

            # Full LP
            a_opt = run_full_lp_descent(a_ps, 1500, deadline, r_idx, c_idx, verbose=False)
            tracker.update(a_opt, f"(P5 r{round_num} LP)")
            a = a_opt

            # Check if we're making progress
            if abs(compute_J(a) - tracker.best_score) < 1e-10:
                print(f"  Converged at round {round_num}", flush=True)
                break

    # Fallback
    if tracker.best is None:
        if parent is not None:
            tracker.best = parent.copy()
        else:
            tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
