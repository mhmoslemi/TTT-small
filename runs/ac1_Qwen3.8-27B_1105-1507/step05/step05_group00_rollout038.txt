The objective is `2*n*max_k (a*a)[k] / (sum(a))^2`. Let's denote `f(a) = max_k (a*a)[k] / (sum(a))^2`. We want to minimize `2n * f(a)`.

The key constraint is `a >= 0`.

Now, `(a*a)[k] = sum_j a[j]*a[k-j]`. This is the (k-th) coefficient of the polynomial `A(x)^2` where `A(x) = sum_j a[j] x^j`.

So we're looking at the maximum coefficient of `A(x)^2` divided by the square of the sum of coefficients of `A(x)`.

This is related to the "Littlewood problem" and "flat polynomials" in algebra.

A polynomial with all coefficients 0 or 1 (or more generally, non-negative) whose square has minimal maximum coefficient... this is a well-studied problem in combinatorics.

For the "Borwein-EB" construction and related work, the minimum maximum coefficient of `A(x)^2` (for A with non-negative coefficients summing to 1) is related to the structure of A.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

# Helper functions
def fft_conv(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n+m-1]

def compute_conv(a):
    return fft_conv(a, a)

def compute_obj(a):
    n = len(a)
    if n == 0: return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01: return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smoothed_obj(a, T):
    """Log-sum-exp smoothed version of the objective."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    lse = cmax + T * np.log(np.sum(np.exp((conv - cmax) / T)))
    sum_a = np.sum(a)
    if sum_a < 0.01: return np.inf
    return 2.0 * n * lse / (sum_a * sum_a)

def smoothed_grad(a, T):
    """Gradient of the smoothed objective."""
    n = len(a)
    conv = compute_conv(a)
    cmax = np.max(conv)
    w = np.exp((conv - cmax) / T)
    w_sum = np.sum(w)
    w = w / w_sum  # normalized weights
    
    # grad of LSE w.r.t. a[i] = 2 * sum_j a[j] * w[j+i] (zero-padded)
    # This is the cross-correlation of a and w
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    fw = rfft(w, size)

    # What we want: sum_j a[j] * w[j+i] = (a * w_flip)[i] where w_flip[j] = w[-j]
    # = IFFT(FFT(a) * FFT(w_flip))[i]
    # w_flip in the zero-padded domain: w_flip[k] = w[(N-k) % N] for circular
    # For linear: we can use the fact that sum_j a[j]*w[j+i] = conv(a_rev, w)[i] where a_rev = a[::-1]

    # = 2*a[k-i] (for k-i != i) or 2*a[i] (for k-i == i, which is the same since a[k-i]=a[i])
    # So d/d a[i] (a*a)[k] = 2*a[k-i] if 0 <= k-i < n, else 0

# LP solver
def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        k_start = j
        k_end = min(j+n, 2*n-1)
        A[k_start:k_end, j] = a[:k_end-k_start]
    return A

def solve_lp_max_sum(a, M, time_limit=15):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def solve_lp_minpeak(a, time_limit=15):
    """LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    # Variables: g[0],...,g[n-1], M
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize M
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g/s, float(res.x[n])
    except Exception:
        pass
    return None, None
