   ```python
   def mw_game_step(a, q, eta, n_inner, N):
       """One step of the MW game."""
       n = len(a)
       
       # Inner QP solve: min a^T Q(q) a s.t. a >= 0, sum(a) = 1
       # Use projected gradient with warm start
       a_qp = a.copy()
       for t in range(n_inner):

           fa = np.fft.fft(a_qp, N)
           fq = np.fft.fft(q, N)

           a_qp = a_qp - lr * grad

       # Compute losses for each lag
       conv = fast_conv(a_qp)  # length 2n-1
       
       # Update q using multiplicative weights
       # q_k <- q_k * exp(-eta * conv_k) / Z
       log_weights = -eta * conv
       log_weights -= log_weights.max()  # for numerical stability
   ```

   ```python
   def proj_simplex(v):
       """Project v onto the simplex {x >= 0, sum(x) = 1}."""
       n = len(v)
       u = np.sort(v)[::-1]
       cv = np.cumsum(u)
       rho = np.nonzero(u * np.arange(1, n+1) > (cv - 1))[0][-1]
       theta = (cv[rho] - 1) / (rho + 1.0)
       return np.maximum(v - theta, 0)
   ```
