Actually, one more thought: the LP `max sum(g) s.t. conv(a,g) <= M, g >= 0` has a special structure. The dual is:
`min M * sum(y) s.t. sum_k A[k,j]*y[k] >= 1 for all j, y >= 0`

where `A[k,j] = a[k-j]` (the Toeplitz matrix). The dual variables `y` tell us which constraints are binding. We can use this for:
- Identifying which constraints to keep (the ones with `y[k] > 0`).
- Determining the optimal step size (KKT conditions).

The KKT condition for the primal-dual pair at optimality:
- `g[j] > 0` implies `sum_k A[k,j]*y[k] = 1` (complementary slackness for dual).
- `y[k] > 0` implies `(conv(a,g))[k] = M` (complementary slackness for primal).

At the optimum, `sum(g) = 1` (we normalize) and `M` is the LP value. The ratio `M / (sum of original conv)` tells us how much the peak can be reduced.

"""
Improved peak-minimization via reduced-constraint LP alternating with lazy
constraint generation, geometric blending, subgradient polishing, and
multi-length exploration. Targets objective 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def fft_convolve(a, b):
    """Linear convolution of a and b via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz_rows(a, indices):
    """Build rows of Toeplitz matrix for given peak indices."""
    n = len(a)
    K = len(indices)
    A = np.zeros((K, n))
    for idx, k in enumerate(indices):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A[idx, j] = a[kj]
    return A


def solve_lp_reduced(a, M, peak_indices, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in peak_indices, g >= 0."""
    n = len(a)
    K = len(peak_indices)
    A = build_toeplitz_rows(a, peak_indices)
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, res
    except Exception:
        pass
    return None, None


def solve_lp_with_lazy_constraints(a, M, initial_top_k=150, max_rounds=5, time_limit=10):
    """Solve LP with lazy constraint generation."""
    n = len(a)
    conv_aa = compute_conv(a)
    peak_indices = np.argsort(conv_aa)[::-1][:initial_top_k].tolist()

    g = None
    for round_idx in range(max_rounds):
        g, res = solve_lp_reduced(a, M, peak_indices, time_limit=time_limit)
        if g is None:
            break

        # Check for violated constraints outside the current set
        conv_ag = fft_convolve(a, g)
        peak_val = float(np.max(conv_ag))
        if peak_val > M * 1.001:
            # Find violated constraints
            violated = np.where(conv_ag > M * 0.999)[0]
            # Add top violated constraints not already included
            current_set = set(peak_indices)
            new_indices = [int(k) for k in violated if int(k) not in current_set]
            # Sort by violation magnitude
            new_indices.sort(key=lambda k: conv_ag[k], reverse=True)
            peak_indices.extend(new_indices[:50])  # Add up to 50 new
            if len(peak_indices) > 500:
                # Keep only top 500 by current conv_ag value
                peak_indices = sorted(peak_indices, key=lambda k: conv_ag[k], reverse=True)[:500]
        else:
            break

    return g


def gradient_maxconv_smooth(a, tau=0.5):
    """Smoothed gradient of max(conv(a,a)) using softmax weighting."""
    n = len(a)
    conv = compute_conv(a)
    cmax = float(np.max(conv))
    shifted = (conv - cmax) / tau
    shifted = np.clip(shifted, -400.0, 400.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= w_sum

    # grad[i] = 2 * sum_j w[i+j] * a[j] = 2 * (flip(w) * a)[len(w)-1-i]
    flip_w = w[::-1]
    conv_fw_a = fft_convolve(flip_w, a)
    # We need indices len(w)-1-n+1 to len(w)-1, i.e., n-1 to 2n-2
    # conv_fw_a has length len(w)+n-1 = 3n-2
    # grad[i] = 2 * conv_fw_a[len(w)-1-i] for i=0..n-1
    # = 2 * conv_fw_a[2n-2-i] for i=0..n-1
    # = 2 * conv_fw_a[n-1 : 2n-1][::-1]
    grad = 2.0 * conv_fw_a[n - 1:2 * n - 1][::-1]
    return grad


def nesterov_subgradient(a, max_iters=300, tau_schedule=None, time_budget=20.0):
    """Nesterov-accelerated subgradient with smooth max."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.85
    step = 0.03
    tau = 0.3

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        grad = gradient_maxconv_smooth(a, tau=tau)
        # Center gradient (project onto sum=const)
        grad_c = grad - np.mean(grad)

        # Nesterov
        v = mu * v - step * grad_c
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert and reduce step
            step *= 0.9

        # Adapt tau
        tau = max(0.1, tau * 0.995)
        # Slowly increase step after some iterations
        if it > 50:
            step = min(0.08, step * 1.005)

    return best_a


def geometric_blend(a, g, t):
    """Geometric (multiplicative) blend: a_new = a * g^t / sum(a * g^t)."""
    # Handle zeros: where g is 0, use small epsilon
    g_safe = np.maximum(g, 1e-15)
    a_safe = np.maximum(a, 1e-15)
    a_new = a_safe * np.power(g_safe, t)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s


def linear_blend(a, g, t):
    """Linear blend."""
    a_new = (1 - t) * a + t * g
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s

def random_swap_optimize(a, max_iters=2000, time_budget=15.0):
    """Random swap optimization: transfer mass between two positions."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if a[i] < 1e-10:
            continue

        # Try transferring a fraction from i to j
        for frac in [0.01, 0.05, 0.1]:
            delta = frac * a[i]
            a[i] -= delta
            a[j] += delta
            obj_new = compute_obj(a)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
                break
            else:
                a[i] += delta
                a[j] -= delta
                break

    return best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

        else:
            # Perturb and try again
            rng = np.random.default_rng(seed + polish_round * 1000)
            a_pert = a_work * np.exp(rng.normal(0, 0.02, len(a_work)))
            a_pert = np.maximum(a_pert, 0)
            sp = np.sum(a_pert)
            if sp > 1e-10:
                a_pert = a_pert / sp
            a_pert = lp_alternate_run(a_pert, time_budget=min(remaining * 0.3, 20.0))
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                a_work = a_pert.copy()
                print(f"  Perturb {polish_round}: {best_score:.8f} **")

    # Phase 3: Explore nearby lengths (20% of time)
    print(f"\n--- Phase 3: Length Exploration (best={best_score:.8f}) ---")
    phase3_end = start_time + (budget_s - 30) * 0.85

    n_options = sorted(set([n_parent - 100, n_parent - 50, n_parent + 50,
                            n_parent + 100, n_parent + 200]))
    n_options = [n for n in n_options if 300 <= n <= 1200]
    # Remove duplicates
    n_options = list(dict.fromkeys(n_options))

    for n_try in n_options:
        if time.monotonic() > phase3_end - 15:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 20:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
            s = np.sum(a_try)
            if s > 1e-10:
                a_try = a_try / s

        print(f"  n={n_try}: budget={remaining:.0f}s")

        # Quick optimization
        a_lp = lp_alternate_run(a_try, time_budget=min(remaining * 0.4, 30.0))
        obj_lp = compute_obj(a_lp)

        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  ** LENGTH IMPROVED: n={n_try}, {best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_lp:.8f} (no improvement)")

    # Phase 4: Final intensive polish (15% of time)
    print(f"\n--- Phase 4: Final Polish (best={best_score:.8f}) ---")
    final_end = deadline

    a_work = best_seq.copy()
    s = np.sum(a_work)
    if s > 1e-10:
        a_work = a_work / s

    for round_idx in range(10):
        if time.monotonic() > final_end - 10:
            break
        remaining = final_end - time.monotonic()
        if remaining < 15:
            break

        a_pol, obj_pol = polish_sequence(a_work, time_budget=min(remaining * 0.4, 30.0),
                                         verbose=True)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            a_work = a_pol.copy()
            print(f"  Final {round_idx}: {best_score:.8f} **")
        else:
            # Small random perturbation + quick LP
            rng = np.random.default_rng(seed * 7 + round_idx * 13)
            a_pert = a_work * np.exp(rng.normal(0, 0.015, len(a_work)))
            a_pert = np.maximum(a_pert, 0)
            sp = np.sum(a_pert)
            if sp > 1e-10:
                a_pert = a_pert / sp
            a_pert = lp_alternate_run(a_pert, time_budget=min(remaining * 0.25, 15.0))
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score:
                best_score = obj_pert
                best_seq = a_pert.copy()
                a_work = a_pert.copy()
                print(f"  Final pert {round_idx}: {best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Ensure all values are valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
