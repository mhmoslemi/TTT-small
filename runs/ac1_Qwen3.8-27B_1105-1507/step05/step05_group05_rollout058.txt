import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def coordinate_update_vectorized(a, i):
    """
    Find optimal value for a[i] given all other coordinates.
    Minimizes max(conv(a_new, a_new)) / (sum(a_new))^2.
    Returns optimal x >= 0.
    """
    n = len(a)
    x_cur = a[i]
    S = a.sum()
    
    # Current convolution
    b = fast_conv(a)
    
    # Affected range in convolution
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0
    
    # Precompute for affected k
    ks = np.arange(k_lo, k_hi + 1)
    c_vals = np.copy(b[ks])
    slopes = np.zeros(len(ks))
    quad_k = -1
    
    for idx, k in enumerate(ks):
        j = k - i
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            slopes[idx] = 0.0
            quad_k = idx
        elif 0 <= j < n:
            c_vals[idx] = b[k] - 2 * x_cur * a[j]
            slopes[idx] = 2 * a[j]
        # else: c_vals stays as b[k], slope = 0
    
    def obj_vec(x):
        """Vectorized objective."""
        new_sum = S - x_cur + x
        if new_sum <= 1e-15:
            return 1e10
        vals = c_vals + slopes * x
        if quad_k >= 0:
            vals[quad_k] = c_vals[quad_k] + x * x
        m = max(base_max, vals.max())
        return m / (new_sum * new_sum)
    
    # Golden section search (vectorized)
    lo = 0.0
    hi = max(x_cur * 3.0, 1.0)
    
    # Expand if needed

            hi *= 2.0
        else:

            fb = fa

    x_opt = (lo + hi) / 2.0

        max_change = 0.0
        for i in order:

            change = abs(x_opt - a[i])
            if change > tol:

            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:

def smooth_max_gradient(a, tau=50.0):
    """Compute gradient of smooth max (log-sum-exp) of convolution, scaled."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Numerical stabilization
    shifted = b - b_max
    max_shifted = shifted.max()
    if max_shifted > 50:
        shifted = shifted - max_shifted
    exp_b = np.exp(tau * shifted)
    exp_sum = exp_b.sum()
    if exp_sum < 1e-300:
        p = np.full(2*n-1, 1.0/(2*n-1))
    else:
        p = exp_b / exp_sum
    
    # Gradient: d(max_b)/da[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2 * pk * a[k-hi : k-lo+1][::-1]
        # a[k-hi], a[k-hi-1], ..., a[k-lo]
        # which is a[k-hi : k-lo-1 : -1] or a[k-lo : k-hi+1][::-1]
        segment = a[k-lo:k-hi+1][::-1]
        grad[lo:hi+1] += 2.0 * pk * segment
    
    return grad

def run_subgradient_descent(a, max_iters, deadline, tau=50.0, lr0=0.1):
    """Subgradient descent with diminishing step size on simplex."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        lr = lr0 / (1 + 0.1 * it)
        grad = smooth_max_gradient(a, tau)
        
        # Project gradient onto tangent space of simplex (sum = 0)
        grad = grad - grad.mean()
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
    
    return best_a

def run_nesterov(a, max_iters, deadline, tau=50.0, alpha=0.5):

    n = len(a)

        beta = (1 - alpha * t_k) / (1 + alpha * it)

        y = np.maximum(y, 0)
        s = y.sum()
        if s < 1e-10:
            y = x.copy()

        if s < 1e-10:
            x_new = y.copy()
        else:

        x = x_new

def peak_shaving(a, n_iters=100, deadline=None):
    """
    Heuristic: find positions where conv is maximal, and reduce products a[i]*a[j]
    at those positions by adjusting a[i] or a[j].
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
        
        # Find positions where b is close to max
        threshold = M * 0.95
        hot_ks = np.where(b >= threshold)[0]
        
        if len(hot_ks) == 0:
            break
        
        # For each hot k, find the (i, j) pairs contributing most
        reduced = False
        for k in hot_ks[:5]:  # Limit to top 5
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # Contributions: a[i] * a[k-i] for i in [lo, hi]
            i_range = np.arange(lo, hi + 1)
            contributions = a[i_range] * a[k - i_range]
            
            # Find the dominant pair
            idx_max = np.argmax(contributions)
            i_dom = i_range[idx_max]
            j_dom = k - i_dom
            
            # Reduce the larger of a[i_dom], a[j_dom] slightly
            if i_dom == j_dom:
                # Can't reduce both, just reduce a[i_dom]
                a[i_dom] *= 0.98
            else:
                # Transfer from larger to smaller
                if a[i_dom] > a[j_dom]:
                    transfer = 0.01 * a[i_dom]
                    a[i_dom] -= transfer
                    # Add to a random position
                    a[np.random.randint(n)] += transfer
                else:
                    transfer = 0.01 * a[j_dom]
                    a[j_dom] -= transfer
                    a[np.random.randint(n)] += transfer
            reduced = True
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        if not reduced:
            break
    
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.2, n), 0.01))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))
    
    # Cosine-based inits
    for freq in [1, 2, 3, 4, 5, 7]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t), 0.01))
    
    # Gaussian
    for sigma in [0.1, 0.2, 0.3, 0.4]:
        inits.append(np.exp(-0.5 * ((t - 0.5) / sigma) ** 2) + 0.1)
    
    # Linear
    inits.append(np.maximum(2.0 * t, 0.01))
    inits.append(np.maximum(2.0 * (1.0 - t), 0.01))
    
    # Random smooth
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # "Optimal shape" heuristic: concentrate in middle
    inits.append(np.sin(np.pi * t) ** 2 + 0.05)
    
    return inits

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1

def compute_J(a):

    s = a.sum()

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    lo = 0.0
    hi = max(x_cur * 3.0, 0.5, min(hi_max, np.sqrt(S) + 0.5))

def run_coord_descent(a, max_sweeps, deadline, seed=0, tol=1e-12):

    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)

        max_change = 0.0
        for i in order:
            x_opt = coordinate_update(a, i)
            ch = abs(x_opt - a[i])
            if ch > tol:

            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if max_change < tol:
            no_improve += 10
        if no_improve > 25:
            break

    return best_a

def run_smooth_descent(a, max_iters, deadline, tau=50.0, lr0=0.05, momentum=0.9):
    """Momentum-based smooth gradient descent on the simplex."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lr = lr0 / (1.0 + 0.05 * it)
        grad = smooth_max_grad(a, tau)

        # Project to tangent space
        grad = grad - grad.mean()

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s
        a = a_new

        M = fast_conv(a).max()
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()

    return best_a


def run_active_set_subgrad(a, max_iters, deadline, top_k=10):
    """Subgradient method using only the top-k active convolution positions."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    lr = 0.1
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()

        # Get top-k positions
        top_idx = np.argsort(b)[-top_k:]
        top_b = b[top_idx]
        # Soft weights
        w = np.exp(50.0 * (top_b - top_b.max()))
        w = w / w.sum()

        # Gradient = sum of weighted gradients of active conv positions
        grad = np.zeros(n)
        for w_k, k in zip(w, top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # d conv[k] / d a[i] = 2 * a[k-i] for i in [lo, hi]
            js = np.arange(lo, hi + 1)
            grad[lo:hi + 1] += w_k * 2.0 * a[k - js]

        # Normalize gradient to unit norm
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad = grad / gn

        # Project to simplex tangent
        grad = grad - grad.mean()

        lr_t = lr / (1.0 + 0.1 * it)
        v = 0.9 * v - lr_t * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.3, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    for freq in [1, 2, 3, 4, 6, 8]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t), 0.05))

    for sig in [0.1, 0.15, 0.2, 0.25, 0.35]:
        inits.append(np.exp(-0.5 * ((t - 0.5) / sig) ** 2) + 0.05)

    inits.append(np.sin(np.pi * t) ** 2 + 0.05)
    inits.append(np.maximum(2.0 * t - 0.3, 0.05))
    inits.append(np.maximum(2.0 * (1.0 - t) - 0.3, 0.05))

    # Smooth random
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(sm + 0.05, 0.01))

    # "Edge-heavy" init (might help with flat convolution)
    inits.append(np.maximum(1.0 + 0.5 * np.cos(np.pi * t), 0.05))
    inits.append(np.maximum(0.5 + 0.8 * np.sin(np.pi * 2 * t) ** 2, 0.05))

    return inits

            print(f"  n={n:4d}: J={Jv:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_n)

    tracker.report("After Phase 1")

    # === PHASE 2: Intensive optimization on best ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: Intensive optimization", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 300:
        n_best = tracker.best.size

        # Long coordinate descent
        d1 = min(time.monotonic() + 150, deadline - 250)
        a1 = run_coord_descent(tracker.best, 100, d1, seed=111)
        tracker.update(a1)

        # Smooth gradient
        if time.monotonic() < deadline - 200:
            d2 = min(time.monotonic() + 60, deadline - 200)
            for tau in [30.0, 80.0, 150.0]:
                a2 = run_smooth_descent(tracker.best, 300, d2, tau=tau, lr0=0.03)
                tracker.update(a2)

        # Active set subgradient
        if time.monotonic() < deadline - 150:
            d3 = min(time.monotonic() + 50, deadline - 150)
            for tk in [5, 10, 20]:
                a3 = run_active_set_subgrad(tracker.best, 200, d3, top_k=tk)
                tracker.update(a3)

    tracker.report("After Phase 2")

    # === PHASE 3: Stochastic restarts ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Stochastic restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 150:
        scales = [0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5]
        n_restarts = min(len(scales), max(3, int((deadline - time.monotonic() - 150) / 30)))
        time_per_r = max(15.0, (deadline - time.monotonic() - 150) / n_restarts)

        for r in range(n_restarts):
            if time.monotonic() > deadline - 150:
                break
            sc = scales[r % len(scales)]
            a_p = tracker.best + rng.normal(0, sc, tracker.best.size)
            a_p = np.maximum(a_p, 0.005)
            a_p = a_p / a_p.sum()

            rd = min(time.monotonic() + time_per_r, deadline - 150)
            a_r = run_coord_descent(a_p, 40, rd, seed=5000 + r * 17)
            tracker.update(a_r)

    tracker.report("After Phase 3")

    # === PHASE 4: Scale-up exploration ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Scale exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 80:
        # Try extending to 2x length
        n_cur = tracker.best.size
        for mult in [2, 3]:
            n_new = n_cur * mult
            if n_new > 4096 or time.monotonic() > deadline - 50:
                break
            a_ext = resample_seq(tracker.best, n_new)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            ed = min(time.monotonic() + 40, deadline - 50)
            a_ext_opt = run_coord_descent(a_ext, 30, ed, seed=8888)
            tracker.update(a_ext_opt)

    tracker.report("After Phase 4")

    # === PHASE 5: Final intensive polish ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 5: Final polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 5:
        # Many coordinate sweeps
        a_f = run_coord_descent(tracker.best, 200, deadline - 3, seed=424242)
        tracker.update(a_f)

        # Quick smooth descent
        if time.monotonic() < deadline - 3:
            a_s = run_smooth_descent(tracker.best, 100, deadline - 3, tau=100.0, lr0=0.02)
            tracker.update(a_s)

    tracker.report("FINAL")

    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
