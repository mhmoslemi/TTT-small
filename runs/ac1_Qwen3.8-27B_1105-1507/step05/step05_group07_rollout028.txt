def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences with emphasis on asymmetry."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        p = p / p.sum() if p.sum() > 0 else p
        inits.append(p.copy())
        # Asymmetric perturbations
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            inits.append(np.maximum(p + rng.normal(0, sigma, n), 0.01))
        # Asymmetric skew
        for alpha in [0.3, 0.5, 0.7]:
            skew = 1 + alpha * (2*t - 1)
            inits.append(np.maximum(p * skew, 0.01))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Asymmetric linear ramp
    for direction in [1, -1]:
        ramp = np.linspace(0.1, 1.0, n)
        if direction == -1:
            ramp = ramp[::-1]
        inits.append(ramp / ramp.sum())
    
    # Chirp-based (quadratic phase in freq -> spread in time)
    for f0 in [0.5, 1, 2, 3]:
        for beta in [1, 2, 4]:
            for alpha in [0.3, 0.5, 0.7]:
                phase = 2*np.pi*f0*t + beta*(t**2)
                sig = np.maximum(1 + alpha*np.cos(phase), 0.01)
                inits.append(sig / sig.sum())
    
    # Asymmetric cosine
    for f in [1, 2, 3, 5]:
        for amp in [0.3, 0.5, 0.7]:
            for phase in [0, np.pi/2, np.pi/3]:
                sig = np.maximum(1 + amp*np.cos(2*np.pi*f*t + phase), 0.01)
                inits.append(sig / sig.sum())
    
    # Gaussian with offset center
    for center in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        for width in [0.1, 0.15, 0.2, 0.3]:
            sig = np.maximum(np.exp(-((t-center)/width)**2) + 0.05, 0.01)
            inits.append(sig / sig.sum())
    
    # Smooth random
    ksize = max(5, n // 15)
    kernel = np.ones(ksize) / ksize
    for _ in range(5):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.05, 0.01) / n)
    
    # Exponential decay (highly asymmetric)
    for rate in [1, 2, 3, 5]:
        sig = np.exp(-rate * t)
        inits.append(sig / sig.sum())
        inits.append(sig[::-1] / sig.sum())
    
    # Power law
    for p in [0.5, 1, 2]:
        sig = t**p + 0.1
        inits.append(sig / sig.sum())
        sig2 = (1-t)**p + 0.1
        inits.append(sig2 / sig2.sum())
    
    # V-shape and inverse V-shape
    inits.append(np.maximum(0.5 + 0.5*np.abs(2*t - 1), 0.01) / n)
    inits.append(np.maximum(1.0 - 0.5*np.abs(2*t - 1), 0.01) / n)
    # Asymmetric V
    for c in [0.3, 0.7]:
        inits.append(np.maximum(0.5 + 0.5*np.abs(t - c)*2, 0.01) / n)
    
    # Normalize all
    result = []
    for a in inits:
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            result.append(a / s)
        else:
            result.append(np.ones(n) / n)
    
    return result

def subgradient_with_momentum(a, max_iters, deadline, lr_init=0.15, momentum=0.9, 
                                annealing=True, noise_scale=0.01):
    """Subgradient descent with momentum and optional simulated annealing."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    running_sum = np.zeros(n)
    count = 0
    
    v = np.zeros(n)  # momentum
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Adaptive learning rate
        lr = lr_init / (1 + it * 0.01) ** 0.5
        
        # Compute subgradient
        b = fast_conv(a)
        M = float(b.max())
        
        # Track best
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
        
        # Find active set (lags near the peak)
        tol = 0.05 * M + 1e-10
        active = np.where(b >= M - tol)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad /= gnorm
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        
        # Simulated annealing: add noise
        if annealing and it % 50 == 49:
            temp = noise_scale * max(0.1, 1.0 - it / max_iters)
            a_new += rng.normal(0, temp, n)
            a_new = np.maximum(a_new, 0.0)
        
        # Normalize
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        running_sum += a_new
        count += 1
        a = a_new
    
    # Check average
    if count > 50:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = float(fast_conv(a_avg).max())
            if M_avg < best_M:
                best_a = a_avg
    
    return best_a

def freq_domain_perturb(a, rng, strength=0.1):
    """Apply a random perturbation in the frequency domain."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    A = np.fft.rfft(a, size)
    # Add random complex perturbation
    noise = (rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A))) * strength
    A_pert = A + noise
    # Ensure DC component is preserved (for sum)
    a_new = np.fft.irfft(A_pert, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def coord_descent_fast(a, max_sweeps, deadline, random_order=False, seed=0):
    """Fast coordinate descent with incremental convolution."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n)
        s = 1.0
    
    conv = fast_conv(a)
    best_conv_max = float(conv.max())
    best_a = a.copy()
    best_obj = 2.0 * n * best_conv_max / (s * s)
    
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        sweep_improved = False
        
        for i in order:
            old = a[i]
            
            if old < 1e-10:
                candidates = [0.0, 1e-4, 1e-3, 1e-2]
            else:
                candidates = [0.0, 0.2*old, 0.4*old, 0.6*old, 0.8*old, 1.2*old, 1.5*old, 2.0*old]
                candidates = [c for c in candidates if abs(c - old) > 1e-3 * old]
            
            if not candidates:
                continue
            
            best_val = old
            best_local = 2.0 * n * float(conv.max()) / (s * s)
            
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2
            
            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue
                
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                
                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * float(conv.max()) / (new_s * new_s)
                
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
            
            if abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                sweep_improved = True
        
        current_M = float(conv.max())
        current_obj = 2.0 * n * current_M / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            break
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, time_limit=20):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def run_lp_mirror_descent(a, max_iters, deadline, time_limit=20):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

def run_averaged_subgradient(a, max_iters, deadline, lr_init=0.1, momentum=0.9):
    """Averaged subgradient with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    running_sum = np.zeros(n)
    count = 0
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        lr = lr_init / (it + 1) ** 0.5
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
        
        # Subgradient at peak
        peak_idx = int(np.argmax(b))
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        grad = np.zeros(n)
        for j in range(lo, hi + 1):
            idx = peak_idx - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]
        
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad /= gnorm
        
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        running_sum += a_new
        count += 1
        a = a_new
    
    if count > 20:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = float(fast_conv(a_avg).max())
            if M_avg < best_M:
                best_a = a_avg
    
    return best_a

Hybrid optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Strategy: Diverse asymmetric initialization, momentum subgradient with 
frequency-domain perturbation (simulated annealing), extended LP mirror 
descent with multiple restarts and adaptive M tightening, multi-length 
exploration, and final coordinate descent polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return False
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

        return False

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences with emphasis on asymmetry."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = p / p.sum() if p.sum() > 0 else p
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1, 0.15]:
            inits.append(np.maximum(p + rng.normal(0, sigma, n), 0.01))
        for alpha in [0.3, 0.5, 0.7, 0.9]:
            skew = 1 + alpha * (2 * t - 1)
            inits.append(np.maximum(p * skew, 0.01))
            skew2 = 1 + alpha * (t - 0.3)
            inits.append(np.maximum(p * skew2, 0.01))

    inits.append(np.ones(n) / n)

    for direction in [1, -1]:
        ramp = np.linspace(0.05, 1.0, n)
        if direction == -1:
            ramp = ramp[::-1]
        inits.append(ramp / ramp.sum())

    for f0 in [0.5, 1, 2, 3, 5]:
        for beta in [1, 2, 4, 6]:
            for alpha in [0.3, 0.5, 0.7]:
                phase = 2 * np.pi * f0 * t + beta * (t ** 2)
                sig = np.maximum(1 + alpha * np.cos(phase), 0.01)
                inits.append(sig / sig.sum())

    for f in [1, 2, 3, 4, 6, 8]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            for phase in [0, np.pi / 3, np.pi / 2, 2 * np.pi / 3]:
                sig = np.maximum(1 + amp * np.cos(2 * np.pi * f * t + phase), 0.01)
                inits.append(sig / sig.sum())

    for center in [0.15, 0.25, 0.35, 0.5, 0.65, 0.75, 0.85]:
        for width in [0.08, 0.12, 0.18, 0.25, 0.35]:
            sig = np.maximum(np.exp(-((t - center) / width) ** 2) + 0.03, 0.01)
            inits.append(sig / sig.sum())

    ksize = max(5, n // 12)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.03, 0.01) / n)

    for rate in [0.5, 1, 2, 3, 5]:
        sig = np.exp(-rate * t)
        inits.append(sig / sig.sum())
        inits.append(sig[::-1] / sig.sum())

    for p in [0.3, 0.5, 1, 2, 3]:
        sig = t ** p + 0.05
        inits.append(sig / sig.sum())
        sig2 = (1 - t) ** p + 0.05
        inits.append(sig2 / sig2.sum())

    inits.append(np.maximum(0.3 + 0.7 * np.abs(2 * t - 1), 0.01) / n)
    inits.append(np.maximum(0.7 - 0.4 * np.abs(2 * t - 1), 0.01) / n)
    for c in [0.2, 0.4, 0.6, 0.8]:
        inits.append(np.maximum(0.3 + 0.7 * np.abs(t - c) / max(c, 1 - c), 0.01) / n)

    result = []
    for a in inits:
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            result.append(a / s)
        else:
            result.append(np.ones(n) / n)

    return result


def freq_domain_perturb(a, rng, strength=0.1):
    """Apply a random perturbation in the frequency domain."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    A = np.fft.rfft(a, size)
    noise = (rng.standard_normal(len(A)) + 1j * rng.standard_normal(len(A))) * strength
    A_pert = A + noise
    a_new = np.fft.irfft(A_pert, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def coord_descent_fast(a, max_sweeps, deadline, random_order=False, seed=0):
    """Fast coordinate descent with incremental convolution."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n)
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * float(conv.max()) / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)

        for i in order:
            old = a[i]
            if old < 1e-12:
                candidates = [0.0, 5e-5, 1e-4, 5e-4, 1e-3]
            else:
                candidates = [0.0, 0.15 * old, 0.35 * old, 0.55 * old, 0.75 * old,
                              1.25 * old, 1.5 * old, 1.8 * old]
                candidates = [c for c in candidates if abs(c - old) > 1e-3 * max(old, 1e-10)]

            if not candidates:
                continue

            best_val = old
            best_local = 2.0 * n * float(conv.max()) / (s * s)

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * float(conv.max()) / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val

        current_M = float(conv.max())
        current_obj = 2.0 * n * current_M / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=20):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, time_limit=20, m_tighten=0.999):
    """
    LP mirror descent with adaptive M tightening.
    At each iteration, solve LP with current M, then try tightening M
    to force more aggressive changes.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_M = float(fast_conv(a).max())
    best_a = a.copy()
    best_M = current_M
    no_improve = 0
    prev_lp_time = 0.1
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_conv = fast_conv(a)
        M = float(b_conv.max())

        # Try slightly tightened M for more aggressive moves
        M_try = M * m_tighten
        if it > 5:
            M_try = M * (m_tighten - 0.0001 * min(it, 50))

        t0 = time.monotonic()
        g = lp_best_response(a, M_try, time_limit=time_limit)
        lp_time = time.monotonic() - t0
        prev_lp_time = 0.6 * prev_lp_time + 0.4 * lp_time

        if g is None:
            # Fall back to less tight M
            g = lp_best_response(a, M, time_limit=time_limit)
            if g is None:
                break
            lp_time = time.monotonic() - t0

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Check if g is too similar to a (stagnation)
        cos_sim = float(np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-15))
        if cos_sim > 0.9999 and it > 10:
            # Add random perturbation to break symmetry
            g = np.maximum(g + rng.normal(0, 0.01, n), 0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g /= g_sum

        improved = False
        base_t = max(0.03, 0.4 * (1.0 - 0.2 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.6, base_t * 0.35, 0.02, 0.01, 0.005]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = float(fast_conv(a_trial).max())
            if M_trial < best_M - 1e-15:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                # Try a larger step or perturbation
                a_pert = freq_domain_perturb(a, rng, strength=0.03)
                a_sg = _quick_subgradient(a_pert, 200, time.monotonic() + 10)
                M_pert = float(fast_conv(a_sg).max())
                if M_pert < best_M:
                    a = a_sg
                    best_M = M_pert
                    best_a = a.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0

        remaining = deadline - time.monotonic()
        if remaining < prev_lp_time * 3:
            break

    return best_a


def _quick_subgradient(a, max_iters, deadline):
    """Quick subgradient descent for escape from local minima."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        lr = 0.1 / (it + 1) ** 0.5
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M:
            best_M = M
            best_a = a.copy()

        peak_idx = int(np.argmax(b))
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        grad = np.zeros(n)
        for j in range(lo, hi + 1):
            idx = peak_idx - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]

        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad /= gnorm

        v = 0.9 * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return best_a


def run_subgradient_momentum(a, max_iters, deadline, lr_init=0.12, momentum=0.9,
                             use_averaging=True):
    """Subgradient descent with momentum and optional averaging."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    running_sum = np.zeros(n)
    count = 0
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lr = lr_init / (it + 1) ** 0.5
        b = fast_conv(a)
        M = float(b.max())

        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

        # Use multiple active lags for richer subgradient
        tol = 0.03 * M + 1e-12
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx]

        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad /= gnorm

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        if use_averaging:
            running_sum += a_new
            count += 1
        a = a_new

    if use_averaging and count > 30:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = float(fast_conv(a_avg).max())
            if M_avg < best_M:
                best_a = a_avg
                best_M = M_avg

    return best_a


def multiplicative_weight_update(a, max_iters, deadline, beta=0.5):
    """
    Multiplicative weight update: iteratively adjust a to flatten the
    convolution profile.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        if M < best_M:
            best_M = M
            best_a = a.copy()

        mean_b = float(b.mean())
        # Compute multiplicative weights: reduce a[j] where it contributes to high-b lags
        log_w = np.zeros(n)
        for k in range(2 * n - 1):
            excess = b[k] - mean_b
            if abs(excess) < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                idx = k - j
                if 0 <= idx < n:
                    log_w[j] += beta * excess * a[idx] / (mean_b + 1e-15)

        # Apply multiplicative update
        a_new = a * np.exp(-np.clip(log_w, -5, 5))
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimum J(a):
    Phase 1: Diverse initialization + coordinate descent (250s)
    Phase 2: Subgradient with momentum + multiplicative weights (250s)
    Phase 3: Extended LP mirror descent with restarts (550s)
    Phase 4: Multi-length exploration (250s)
    Phase 5: Final polish (200s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 8.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        tracker.update(parent)

    current_best = tracker.best.copy()
    n_base = current_best.size

    # === PHASE 1: Coordinate descent on diverse inits (250s) ===
    p1_deadline = min(time.monotonic() + 250, deadline - 800)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Coord Descent n={n_base} (budget {p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_initializations(n_base, rng, current_best)
    # Limit to most promising: parent + diverse
    n_trial = min(10, len(inits))

    for idx in range(n_trial):
        if time.monotonic() > p1_deadline:
            break

        a0 = inits[idx].copy()
        a0 = np.maximum(a0, 0)
        s = a0.sum()
        if s < 0.01:
            continue
        a0 /= s

        t0 = time.monotonic()
        per_init = max(8, (p1_deadline - time.monotonic()) / max(1, n_trial - idx))
        a_opt = coord_descent_fast(a0, max_sweeps=3000,
                                    deadline=time.monotonic() + per_init,
                                    random_order=True, seed=seed + idx)
        J_val = compute_J(a_opt)
        print(f"  Init {idx}: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
        tracker.update(a_opt)

    tracker.report("Phase 1")

    # === PHASE 2: Subgradient + MWU (250s) ===
    p2_deadline = min(time.monotonic() + 250, deadline - 550)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Subgradient+MWU n={tracker.best.size} (budget {p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Run subgradient on best
    if time.monotonic() < p2_deadline - 30:
        t0 = time.monotonic()
        a_sg = run_subgradient_momentum(tracker.best, max_iters=8000,
                                         deadline=time.monotonic() + 80,
                                         lr_init=0.12, momentum=0.9)
        J_val = compute_J(a_sg)
        print(f"  Subgrad: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
        tracker.update(a_sg)

    # MWU on best
    if time.monotonic() < p2_deadline - 30:
        t0 = time.monotonic()
        a_mwu = multiplicative_weight_update(tracker.best, max_iters=5000,
                                              deadline=time.monotonic() + 60,
                                              beta=0.3)
        J_val = compute_J(a_mwu)
        print(f"  MWU: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
        tracker.update(a_mwu)

    # Freq perturb + subgradient
    for pid in range(4):
        if time.monotonic() > p2_deadline - 15:
            break
        a_pert = freq_domain_perturb(tracker.best, rng, strength=0.04 + 0.03 * pid)
        a_opt = run_subgradient_momentum(a_pert, max_iters=3000,
                                          deadline=time.monotonic() + 20,
                                          lr_init=0.08, momentum=0.85)
        # Quick CD polish
        a_cd = coord_descent_fast(a_opt, max_sweeps=500,
                                   deadline=time.monotonic() + 15,
                                   random_order=True, seed=seed + 200 + pid)
        J_val = compute_J(a_cd)
        print(f"  FreqPerturb{pid}+CD: J={J_val:.8f}", flush=True)
        tracker.update(a_cd)

    # CD polish on overall best
    if time.monotonic() < p2_deadline - 10:
        a_cd = coord_descent_fast(tracker.best, max_sweeps=2000,
                                   deadline=p2_deadline,
                                   random_order=True, seed=seed + 99)
        J_val = compute_J(a_cd)
        print(f"  Phase2 CD: J={J_val:.8f}", flush=True)
        tracker.update(a_cd)

    tracker.report("Phase 2")

    # === PHASE 3: Extended LP Mirror Descent (550s) ===
    p3_deadline = min(time.monotonic() + 550, deadline - 200)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent n={tracker.best.size} (budget {p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    n_best = tracker.best.size
    est_lp_time = 0.008 * (n_best / 128) ** 2
    max_iters_lp = max(15, int(450 / max(0.05, est_lp_time)))
    tl = max(5, min(25, int(200 / max(1, max_iters_lp))))
    print(f"  n={n_best}, est_LP={est_lp_time * 1000:.0f}ms, max_iters={max_iters_lp}, tl={tl}s", flush=True)

    # First LP pass on best
    t0 = time.monotonic()
    a_lp = run_lp_mirror_descent(tracker.best, max_iters=max_iters_lp,
                                  deadline=min(time.monotonic() + 350, p3_deadline - 50),
                                  time_limit=tl, m_tighten=0.999)
    J_val = compute_J(a_lp)
    print(f"  LP pass 1: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
    tracker.update(a_lp)

    # Second LP pass with perturbed start
    if time.monotonic() < p3_deadline - 30:
        a_pert = freq_domain_perturb(tracker.best, rng, strength=0.02)
        a_lp2 = run_lp_mirror_descent(a_pert, max_iters=max(10, max_iters_lp // 2),
                                       deadline=p3_deadline,
                                       time_limit=tl, m_tighten=0.998)
        J_val2 = compute_J(a_lp2)
        print(f"  LP pass 2 (perturbed): J={J_val2:.8f}", flush=True)
        tracker.update(a_lp2)

    # Third: CD + LP
    if time.monotonic() < p3_deadline - 20:
        a_cd = coord_descent_fast(tracker.best, max_sweeps=1000,
                                   deadline=time.monotonic() + 30,
                                   random_order=True, seed=seed + 500)
        tracker.update(a_cd)
        if time.monotonic() < p3_deadline - est_lp_time * 5:
            a_lp3 = run_lp_mirror_descent(tracker.best, max_iters=15,
                                           deadline=p3_deadline,
                                           time_limit=tl, m_tighten=0.999)
            J_val3 = compute_J(a_lp3)
            print(f"  LP pass 3: J={J_val3:.8f}", flush=True)
            tracker.update(a_lp3)

    tracker.report("Phase 3")

    # === PHASE 4: Multi-length exploration (250s) ===
    p4_deadline = min(time.monotonic() + 250, deadline - 30)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Multi-length (budget {p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    base_n = tracker.best.size
    target_lengths = []
    for l in [384, 640, 768, 896, 1024, 1280, 1536, 2048]:
        if l != base_n and l >= 384 and l <= 2048:
            target_lengths.append(l)
    target_lengths = sorted(set(target_lengths))[:4]

    for new_n in target_lengths:
        if time.monotonic() > p4_deadline - 40:
            break

        t0 = time.monotonic()
        a_ext = resample_seq(tracker.best, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext /= a_ext.sum()

        # Quick subgradient
        a_sg = run_subgradient_momentum(a_ext, max_iters=2000,
                                         deadline=time.monotonic() + 25,
                                         lr_init=0.08, momentum=0.85)
        # CD
        a_cd = coord_descent_fast(a_sg, max_sweeps=1500,
                                   deadline=min(time.monotonic() + 35, p4_deadline - 20),
                                   random_order=True, seed=seed + 1000 + new_n)
        J_val = compute_J(a_cd)
        print(f"  n={new_n}: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
        tracker.update(a_cd)

        # Brief LP if time allows
        est_lp_n = 0.008 * (new_n / 128) ** 2
        if est_lp_n * 5 < (p4_deadline - time.monotonic()):
            a_lp_ext = run_lp_mirror_descent(tracker.best, max_iters=8,
                                              deadline=p4_deadline,
                                              time_limit=max(5, tl // 2),
                                              m_tighten=0.999)
            J_val_lp = compute_J(a_lp_ext)
            print(f"  n={new_n} LP: J={J_val_lp:.8f}", flush=True)
            tracker.update(a_lp_ext)

    tracker.report("Phase 4")

    # === PHASE 5: Final polish (remaining) ===
    p5_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Final Polish (budget {p5_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None and time.monotonic() < p5_deadline - 5:
        remaining = p5_deadline - time.monotonic()

        # Final CD
        a_cd = coord_descent_fast(tracker.best, max_sweeps=5000,
                                   deadline=time.monotonic() + remaining * 0.35,
                                   random_order=True, seed=seed + 7777)
        J_val = compute_J(a_cd)
        print(f"  Final CD: J={J_val:.8f}", flush=True)
        tracker.update(a_cd)

        # Final subgradient
        if time.monotonic() < p5_deadline - 20:
            a_sg = run_subgradient_momentum(tracker.best, max_iters=3000,
                                             deadline=time.monotonic() + 20,
                                             lr_init=0.05, momentum=0.9)
            J_val = compute_J(a_sg)
            print(f"  Final SG: J={J_val:.8f}", flush=True)
            tracker.update(a_sg)

        # Final LP
        n_est = tracker.best.size
        est_lp = 0.008 * (n_est / 128) ** 2
        if est_lp * 10 < (p5_deadline - time.monotonic()):
            a_final = run_lp_mirror_descent(tracker.best, max_iters=20,
                                             deadline=p5_deadline,
                                             time_limit=10, m_tighten=0.999)
            J_val = compute_J(a_final)
            print(f"  Final LP: J={J_val:.8f}", flush=True)
            tracker.update(a_final)

    tracker.report("Phase 5")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
