import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def golden_section_min(f, a, b, tol=1e-10, max_iter=60):

            d = c
            fd = fc

        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)

    idx = J_vals.argmin()

    # Refine in neighborhood
    lo = max(0.0, x_grid[max(0, idx - 3)])
    hi = min(x_ub, x_grid[min(len(x_grid) - 1, idx + 3)])
    x_opt = golden_section_min(obj, lo, hi, tol=1e-10, max_iter=50)
    return x_opt, obj(x_opt)


def coordinate_descent_fast(a, max_sweeps=200, deadline=None, verbose=False,
                            random_order=False, rng=None, x_ub_mult=3.0):
    """
    Fast cyclic/randomized coordinate descent.
    One FFT per sweep; per-coordinate 1D optimization is vectorized.
    """
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return a, float('inf')

    best_J = compute_J(a)
    best_a = a.copy()
    stall_count = 0

    for sweep in range(max_sweeps):
        if deadline and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        current_max_b = float(b.max())
        current_J = 2.0 * n * current_max_b / (S * S)

        # Determine coordinate order
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)

        sweep_improved = False
        k_vals = np.arange(2 * n - 1)

        for m in order:
            old_val = a[m]
            S_prime = S - old_val

            # Compute base and coefficients
            km = k_vals - m
            valid = (km >= 0) & (km < n)
            mask_cross = valid & (km != m)

            if not mask_cross.any() and (2 * m >= 2 * n - 1):
                continue

            # base[k] = b[k] - old contribution of a[m]
            base = b.copy()
            if mask_cross.any():
                base[mask_cross] -= 2.0 * old_val * a[km[mask_cross]]
            if 2 * m < 2 * n - 1:
                base[2 * m] -= old_val * old_val

            # coeff[k] = 2*a[k-m] for cross terms, 0 otherwise
            coeff = np.zeros(2 * n - 1)
            if mask_cross.any():
                coeff[mask_cross] = 2.0 * a[km[mask_cross]]

            # quad[k] = 1 at k=2m
            quad = np.zeros(2 * n - 1)
            if 2 * m < 2 * n - 1:
                quad[2 * m] = 1.0

            # Bounds for x
            x_ub = max(x_ub_mult * old_val, 2.0 * S / n, 0.01)

            # 1D optimization
            x_opt, J_opt = minimize_1d_hybrid(base, coeff, quad, S_prime, n, x_ub)

            if J_opt < current_J - 1e-12:
                a[m] = x_opt
                S = S_prime + x_opt
                current_J = J_opt
                sweep_improved = True

            elif J_opt < best_J - 1e-12:
                # Even if not better than current sweep J, track best
                pass

        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        if verbose and (sweep % 10 == 0 or sweep == max_sweeps - 1):
            print(f"    CoordDescent sweep {sweep}: J={best_J:.8f} stall={stall_count}", flush=True)

        if stall_count > 20:
            break

    return best_a, best_J


def simulate_annealing(a, budget, rng, T_init=0.01, T_min=1e-6, alpha=0.999):
    """Simulated annealing to escape local minima."""
    a = a.copy().astype(np.float64)
    n = len(a)
    S = float(a.sum())
    best_J = compute_J(a)
    best_a = a.copy()
    T = T_init
    start_time = time.monotonic()
    queries = 0

    while time.monotonic() - start_time < budget:
        # Perturb a random subset of coordinates
        n_pert = max(1, n // 16)
        idx = rng.choice(n, size=n_pert, replace=False)
        a_trial = a.copy()
        scale = 0.1 * S / n
        a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale, n_pert), 0)
        S_trial = a_trial.sum()
        if S_trial < 1e-10:
            continue
        J_trial = compute_J(a_trial)
        queries += 1

        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()

        delta = J_trial - compute_J(a)
        if delta < 0 or rng.random() < np.exp(-max(0, delta) / T):
            a = a_trial

        T = max(T * alpha, T_min)

    return best_a, best_J


def build_lp_matrix(n, a, active_k):
    """Build sparse matrix for cross-convolution with selected rows."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))


def lp_mirror_descent(a, deadline, max_iters=200, top_frac=0.5, verbose=False):
    """LP-based mirror descent for structured improvement."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Select active constraints (top fraction)
        threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
        active_k = np.where(b >= threshold)[0]

        if len(active_k) < 5:
            active_k = np.where(b >= b.max() * 0.9)[0]

        A = build_lp_matrix(n, a, active_k)
        c = -np.ones(n)
        b_ub = np.full(len(active_k), M)
        bounds = [(0, np.inf)] * n

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 5, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0.0)
                if g.sum() > 1e-10:
                    g = g / g.sum()

                    # Verify
                    ag = fast_cross_conv(a, g)
                    if ag.max() <= M * (1.0 + 1e-4):
                        # Line search
                        improved = False
                        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
                            a_trial = (1.0 - t) * a + t * g
                            a_trial = np.maximum(a_trial, 0)
                            s = a_trial.sum()
                            if s < 1e-10:
                                continue
                            a_trial = a_trial / s
                            J_trial = compute_J(a_trial)
                            if J_trial < best_J - 1e-12:
                                a = a_trial
                                best_J = J_trial
                                best_a = a.copy()
                                improved = True
                                no_improve = 0
                                break
                        if not improved:
                            no_improve += 1
                        if verbose and it % 50 == 0:
                            print(f"    LP iter {it}: J={best_J:.8f} no_improve={no_improve}", flush=True)
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_J


def generate_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Parent resampled
    p = resample_seq(parent, n)
    inits.append(np.maximum(p, 0.01))

    # Perturbed parent
    inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
    inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine bump (center)
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t - 1)), 0.01))

    # Gaussian center
    inits.append(np.maximum(np.exp(-4 * (t - 0.5) ** 2), 0.01))

    # Edge concentrated
    inits.append(np.maximum(np.exp(-5 * t ** 2) + np.exp(-5 * (t - 1) ** 2) + 0.05, 0.01))

    # Multi-frequency modulation
    for f in [2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))

    # Block structure
    n_blocks = max(4, n // 64)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    blocks = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(np.maximum(blocks, 0.01))

    return inits


def propose_candidate():
    """
    Multi-phase hybrid search:
    Phase 1: Multi-length exploration with quick coordinate descent (150s)
    Phase 2: Deep coordinate descent on best candidate (800s)
    Phase 3: Simulated annealing for diversification (200s)
    Phase 4: LP polishing + final coordinate descent (remaining time)
    """
    total_budget = 1500.0
    deadline = time.monotonic() + total_budget - 30.0
    rng = np.random.default_rng(42)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        parent = np.maximum(parent, 0.0)
        parent_J = compute_J(parent)
        print(f"Parent loaded: n={parent.size}, J={parent_J:.8f}", flush=True)
    except (NameError, AttributeError, ValueError):
        parent = np.ones(512)
        parent_J = compute_J(parent)
        print(f"Using default parent: n=512, J={parent_J:.8f}", flush=True)

    global_best_a = parent.copy()
    global_best_J = parent_J

    def update_best(a, label=""):
        nonlocal global_best_a, global_best_J
        a = np.maximum(a, 0).astype(np.float64)
        if a.sum() < 0.01:
            return
        J = compute_J(a)
        if J < global_best_J:
            global_best_J = J
            global_best_a = a.copy()
            elapsed = time.monotonic() - (deadline - total_budget + 30)
            print(f"  ** NEW BEST: J={J:.8f} n={a.size} t={elapsed:.0f}s [{label}]", flush=True)
            return J
        return J

    # === PHASE 1: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    p1_deadline = time.monotonic() + 150.0
    n_values = [256, 384, 512, 768, 1024, 1536, 2048]

    phase1_candidates = []
    for n in n_values:
        if time.monotonic() > p1_deadline - 10:
            break
        a0 = resample_seq(parent, n)
        a0 = np.maximum(a0, 1e-6)
        a0 /= a0.sum()

        a_opt, J_opt = coordinate_descent_fast(a0, max_sweeps=30, deadline=p1_deadline - 10,
                                               random_order=True, rng=rng)
        phase1_candidates.append((J_opt, a_opt.copy(), n))
        print(f"  n={n:4d}: J={J_opt:.8f}", flush=True)
        update_best(a_opt, f"phase1_n{n}")

    # Sort by quality
    phase1_candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 best: J={phase1_candidates[0][0]:.8f} n={phase1_candidates[0][2]}", flush=True)

    # === PHASE 2: Deep coordinate descent ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: Deep Coordinate Descent", flush=True)
    print(f"{'='*60}", flush=True)

    p2_deadline = time.monotonic() + 800.0

    # Work on the best candidate from phase 1
    best_init = phase1_candidates[0][1].copy()
    best_n = phase1_candidates[0][2]

    # Deep coordinate descent with alternating random/cyclic order
    rng2 = np.random.default_rng(123)
    a_deep, J_deep = coordinate_descent_fast(
        best_init, max_sweeps=5000, deadline=p2_deadline,
        verbose=True, random_order=False, rng=rng2
    )
    update_best(a_deep, "phase2_cyclic")

    # If time remains, try randomized coordinate descent from a perturbed version
    if time.monotonic() < p2_deadline - 5:
        a_pert = a_deep.copy()
        scale = 0.05 / best_n
        a_pert = np.maximum(a_pert + rng2.normal(0, scale, best_n), 1e-8)
        a_pert /= a_pert.sum()

        a_deep2, J_deep2 = coordinate_descent_fast(
            a_pert, max_sweeps=3000, deadline=p2_deadline,
            verbose=True, random_order=True, rng=rng2
        )
        update_best(a_deep2, "phase2_random")

    # === PHASE 3: Simulated annealing for diversification ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Simulated Annealing", flush=True)
    print(f"{'='*60}", flush=True)

    p3_deadline = time.monotonic() + 150.0
    rng3 = np.random.default_rng(777)

    # Run SA from current best
    a_sa, J_sa = simulate_annealing(global_best_a, budget=min(60, p3_deadline - time.monotonic()),
                                    rng=rng3, T_init=0.005, T_min=1e-5, alpha=0.9995)
    update_best(a_sa, "phase3_sa")

    if time.monotonic() < p3_deadline - 5:
        # Continue with coordinate descent from SA result
        a_after, J_after = coordinate_descent_fast(
            a_sa, max_sweeps=1000, deadline=p3_deadline,
            verbose=False, random_order=True, rng=rng3
        )
        update_best(a_after, "phase3_sa_cd")

    # Also try SA from a different perturbation
    if time.monotonic() < p3_deadline - 10:
        a_alt = global_best_a.copy()
        # Structural perturbation: swap halves
        n_half = len(a_alt) // 2
        a_alt2 = a_alt.copy()
        a_alt2[:n_half], a_alt2[n_half:2*n_half] = a_alt2[n_half:2*n_half].copy(), a_alt2[:n_half].copy()
        a_alt2 = np.maximum(a_alt2, 1e-8)
        a_alt2 /= a_alt2.sum()
        a_alt3, J_alt3 = simulate_annealing(a_alt2, budget=30, rng=rng3, T_init=0.01, alpha=0.999)
        update_best(a_alt3, "phase3_sa_swap")

        if time.monotonic() < p3_deadline - 5:
            a_alt4, _ = coordinate_descent_fast(a_alt3, max_sweeps=500, deadline=p3_deadline,
                                                verbose=False, random_order=True, rng=rng3)
            update_best(a_alt4, "phase3_swap_cd")

    # === PHASE 4: LP polishing + final coordinate descent ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: LP Polishing + Final CD", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"  Current best: J={global_best_J:.8f} n={global_best_a.size}", flush=True)
    print(f"  Time remaining: {deadline - time.monotonic():.0f}s", flush=True)

    if time.monotonic() < deadline - 10:
        # LP mirror descent (short)
        p4_lp_dl = min(time.monotonic() + 80, deadline - 60)
        a_lp, J_lp = lp_mirror_descent(global_best_a, deadline=p4_lp_dl,
                                       max_iters=100, top_frac=0.6, verbose=True)
        update_best(a_lp, "phase4_lp")

    if time.monotonic() < deadline - 5:
        # Final coordinate descent polishing
        rng4 = np.random.default_rng(999)
        a_final, J_final = coordinate_descent_fast(
            global_best_a, max_sweeps=10000, deadline=deadline - 2,
            verbose=True, random_order=False, rng=rng4
        )
        update_best(a_final, "phase4_final_cd")

    # Final check
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={global_best_J:.8f} n={global_best_a.size}", flush=True)
    print(f"{'='*60}", flush=True)

    result = [float(max(0, x)) for x in global_best_a]
    return result
