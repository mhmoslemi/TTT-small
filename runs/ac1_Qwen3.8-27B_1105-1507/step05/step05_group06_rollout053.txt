Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient descent with log-sum-exp smoothing and beta annealing
2. Riemannian-style optimization on the probability simplex
3. Multi-restart with diverse initializations
4. Coordinate descent polishing with exact 1D subproblem solving
5. Multi-length exploration
6. Simulated annealing for final improvements
Key innovation: Fast O(n log n) gradient computation via FFT enables millions of
optimization steps within the time budget, far exceeding the LP-based approach.

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad_and_obj(a, beta):
    """
    Compute smooth objective and gradient using log-sum-exp approximation.
    Returns (J_smooth, grad) where grad = dJ_smooth/d(a).
    """
    n = len(a)
    S = a.sum()
    # Work with normalized a for stable computation
    a_norm = a / S
    
    L = 1
    while L < 4 * n:
        L <<= 1
    
    A = np.fft.rfft(a_norm, L)
    c = np.fft.irfft(A * np.conj(A), L)[:2 * n - 1].real
    
    # Log-sum-exp for numerical stability
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    p = exp_bc / exp_bc.sum()
    f = (c_max + np.log(exp_bc.sum())) / beta
    
    # Compute grad_f = dc_f/d(a) via correlation + convolution
    p_full = np.zeros(L)
    p_full[:2 * n - 1] = p
    P = np.fft.rfft(p_full, L)
    
    # Convolution: (a * p)[i] = sum_j a[j] * p[i-j]
    term_conv = np.fft.irfft(A * P, L)[:n]
    # Correlation: (a ⋆ p)[i] = sum_j a[j] * p[j-i]
    P_conj = np.conj(P)
    term_corr = np.fft.irfft(A * P_conj, L)[:n]
    
    grad_f = (term_conv + term_corr).real
    
    # J = 2n * f / S^2 (with a_norm, S_norm = 1, so J = 2n*f)

    # J(a) = 2n * f(a/S)  (since f is homogeneous of degree 2, f(a/S) = f(a)/S^2)
    # dJ/d(a[i]) = 2n * (1/S^2) * d f(a/S) / d(a[i])
    #            = 2n * (1/S^2) * (1/S) * grad_f[i]  (chain rule: d(a/S)/d(a[i]) = 1/S)
    # Wait, f is quadratic in its argument, so f(c*a) = c^2 * f(a)
    # J(a) = 2n * f(a) / S^2
    # f(a) is quadratic: f(c*a) = c^2*f(a)
    # So f(a) = S^2 * f(a/S)
    # J = 2n * S^2 * f(a/S) / S^2 = 2n * f(a/S)
    # dJ/d(a[i]) = 2n * d f(a/S) / d(a[i]) = 2n * (1/S) * grad_f_norm[i]
    # where grad_f_norm is the gradient of f with respect to a_norm = a/S
    
    J = 2 * n * f
    grad = 2 * n * grad_f / S
    
    return J, grad


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    rho_val = np.max(np.where(cond)[0] + 1) if np.any(cond) else 1
    theta = (np.sum(v[:rho_val]) - 1) / rho_val if rho_val > 0 else 0
    return np.maximum(v - theta, 0)


def gradient_descent_smooth(a_init, beta_schedule, deadline, lr_init=0.1, 
                            max_iters=500000, verbose=True):
    """
    Projected gradient descent with log-sum-exp smoothing and adaptive beta.
    """
    a = simplex_project(a_init.copy())
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    if verbose:
        print(f"    Start: J={best_J:.8f}, n={n}", flush=True)
    
    total_iters = 0
    patience = 0
    max_patience = 5000
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        lr = lr_init / (1.0 + beta / 10.0)
        it_in_beta = 0
        patience = 0
        
        while it_in_beta < max_iters:
            if time.monotonic() > deadline:
                break
            it_in_beta += 1
            total_iters += 1
            
            J_smooth, grad = smooth_grad_and_obj(a, beta)
            
            # Riemannian gradient (project to tangent space of simplex)
            rg = grad - np.sum(a * grad)
            
            # Adaptive step size via backtracking
            step = lr
            a_new = a
            accepted = False
            for _ in range(10):
                candidate = a - step * rg
                candidate = simplex_project(candidate)
                J_check = compute_J(candidate)
                if J_check < J_smooth + 1e-12:
                    a_new = candidate
                    accepted = True
                    break
                step *= 0.5
            
            if not accepted:
                patience += 1
                if patience > max_patience:
                    break
                a_new = a
            else:
                patience = max(patience - 1, 0)
            
            J_true = compute_J(a_new)
            a = a_new
            
            if J_true < best_J - 1e-12:
                best_J = J_true
                best_a = a.copy()
                if verbose and it_in_beta % 5000 == 0:
                    elapsed = time.monotonic()
                    print(f"    beta={beta:.0f} it={it_in_beta} J={best_J:.8f}", flush=True)
        
        # Update actual J after beta change
        J_true = compute_J(a)
        if J_true < best_J:
            best_J = J_true
            best_a = a.copy()
    
    return best_a, best_J, total_iters


def coordinate_descent_fast(a, deadline, max_passes=20):
    """
    Fast coordinate descent: for each coordinate, solve the exact 1D problem.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    if time.monotonic() > deadline:
        return best_a
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        order = np.random.permutation(n)
        b = fast_conv(a)
        S = a.sum()
        improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            ai = a[i]
            
            # Efficiently compute the objective as a function of a[i]
            # When a[i] changes by delta:
            #   b_new[k] = b[k] + 2*delta*a[k-i] for k in [i, i+n-1], k != 2i
            #   b_new[2i] = b[2i] + 2*delta*ai + delta^2
            #   S_new = S + delta
            #   J_new = 2n * max(b_new) / S_new^2
            
            # Find the range of affected k
            k_lo = i  # minimum k where a[i] contributes (k = i, j=0)
            k_hi = min(i + n - 1, 2 * n - 2)  # maximum k
            
            # Values outside [k_lo, k_hi] are unchanged
            outside_max = 0.0
            if k_lo > 0:
                outside_max = max(outside_max, b[:k_lo].max())
            if k_hi < 2 * n - 2:
                outside_max = max(outside_max, b[k_hi + 1:].max())
            
            # Define objective function of delta
            def obj_of_delta(delta):
                S_new = S + delta
                if S_new <= 0.01:
                    return float('inf')
                # Compute affected b values
                # k = 2i is special
                max_val = outside_max
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if k == 2 * i and j == i:
                            new_val = b[k] + 2 * delta * ai + delta * delta
                        else:
                            new_val = b[k] + 2 * delta * a[j]
                        if new_val > max_val:
                            max_val = new_val
                return 2 * n * max_val / (S_new * S_new)
            
            # Try several candidate values
            if ai < 1e-10:
                candidates = [0.0, 0.01, 0.03, 0.05, 0.1, 0.15, 0.2]
            else:
                candidates = [0.0, ai * 0.3, ai * 0.5, ai * 0.7, ai, ai * 1.3, ai * 1.5, ai * 2.0, ai * 3.0]
            
            best_cand = ai
            best_val = obj_of_delta(0.0)
            
            for cand in candidates:
                delta = cand - ai
                if abs(delta) < 1e-14:
                    continue
                val = obj_of_delta(delta)
                if val < best_val - 1e-12:
                    best_val = val
                    best_cand = cand
            
            # Fine-tune around best candidate
            if best_cand != ai:
                lo = max(0, best_cand * 0.7) if best_cand > 0 else 0.0
                hi = best_cand * 1.3 + 0.01
                result = minimize_scalar(lambda d: obj_of_delta(d), 
                                         bounds=(lo - ai, hi - ai),
                                         method='bounded',
                                         options={'xatol': 1e-10})
                if result.success:
                    fine_cand = ai + result.x
                    if fine_cand >= 0:
                        fine_val = obj_of_delta(fine_cand - ai)
                        if fine_val < best_val - 1e-12:
                            best_cand = fine_cand
                            best_val = fine_val
            
            if best_cand != ai:
                delta = best_cand - ai
                a[i] = best_cand
                S += delta
                # Update b incrementally
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if k == 2 * i and j == i:
                            b[k] += 2 * delta * ai + delta * delta
                        else:
                            b[k] += 2 * delta * a[j]
                improved = True
        
        # Renormalize
        s_new = a.sum()
        if s_new > 0.01:
            a = a / s_new
        J = compute_J(a)
        if J < best_J - 1e-12:
            best_J = J
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())
    
    # Random
    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())
    
    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.05, 0.01)
    inits.append(sm / sm.sum())
    
    # Frequency modulated (multiple frequencies)
    for freq in [1, 2, 3, 4, 6, 8]:
        fm = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())
    
    # Double/triple bumps
    centers = rng.uniform(0.1, 0.9, rng.integers(2, 5))
    widths = rng.uniform(0.05, 0.2, len(centers))
    bumps = sum(np.exp(-((t - c) / w) ** 2) for c, w in zip(centers, widths)) + 0.02
    inits.append(bumps / bumps.sum())
    
    # Power law
    for alpha in [0.3, 0.7, 1.5, 2.5]:
        pl = (t + 0.01) ** alpha
        inits.append(pl / pl.sum())
        pl2 = (1 - t + 0.01) ** alpha
        inits.append(pl2 / pl2.sum())
    
    # Polynomial
    for degree in [2, 4, 6]:
        coeffs = rng.normal(0, 1, degree + 1)
        poly = np.polyval(coeffs, t)
        poly = np.maximum(poly - poly.min() + 0.05, 0.01)
        inits.append(poly / poly.sum())
    
    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def simulated_annealing(a, deadline, max_iters=20000, T_start=0.02, T_end=0.001):
    """Simulated annealing with coordinate perturbations."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    current_J = best_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        T = T_start * (T_end / T_start) ** frac
        
        # Perturb 1-3 random coordinates
        n_perturb = np.random.choice([1, 1, 1, 2, 3], p=[0.5, 0.2, 0.1, 0.1, 0.1])
        indices = np.random.choice(n, n_perturb, replace=False)
        old_vals = a[indices].copy()
        
        for idx in indices:
            old = a[idx]
            if old < 0.02:
                new_val = np.random.exponential(0.05)
            else:
                new_val = np.random.normal(old, 0.1 * old + 0.01)
            a[idx] = max(0.0, new_val)
        
        a_new = np.maximum(a, 0)
        s_new = a_new.sum()
        if s_new < 0.01:
            a[indices] = old_vals
            continue
        a_new = a_new / s_new
        new_J = compute_J(a_new)
        
        delta = new_J - current_J
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            current_J = new_J
            if new_J < best_J - 1e-12:
                best_J = new_J
                best_a = a_new.copy()
        else:
            a[indices] = old_vals
    
    return best_a


def beta_anneal_gradient_descent(a_init, deadline, verbose=True):
    """
    Main optimization: gradient descent with annealed beta schedule.
    Start with low beta (exploration), end with high beta (exploitation).
    """
    a = simplex_project(a_init.copy())
    
    # Multi-phase beta schedule
    # Phase A: Low beta for exploration
    betas_low = [20, 50, 100]
    # Phase B: Medium beta for refinement  
    betas_med = [200, 500, 1000]
    # Phase C: High beta for exploitation
    betas_high = [2000, 5000, 10000, 20000]
    
    all_betas = betas_low + betas_med + betas_high
    n_phases = len(all_betas)
    
    # Allocate time proportionally (more time for high beta)
    weights = [1.0, 1.0, 1.5, 1.5, 2.0, 2.0, 3.0, 3.0, 4.0, 4.0]
    total_w = sum(weights)
    
    total_available = deadline - time.monotonic()
    phase_deadlines = []
    cum = time.monotonic()
    for i, w in enumerate(weights):
        cum += total_available * w / total_w
        phase_deadlines.append(cum)
    
    total_iters = 0
    for i, beta in enumerate(all_betas):
        if time.monotonic() > phase_deadlines[i]:
            continue
        
        a, J, iters = gradient_descent_smooth(
            a, [beta], phase_deadlines[i], 
            lr_init=0.5 / (1 + beta / 50),
            max_iters=200000,
            verbose=verbose
        )
        total_iters += iters
    
    return a, compute_J(a), total_iters


def propose_candidate(seed=42, budget_s=1500.0):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    
    best_seq = None
    best_score = float('inf')
    
    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                update_best(parent)
                print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent)
    
    # === PHASE 1: Gradient descent from parent ===
    p1_deadline = min(t_start + 400, deadline - 500)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Beta-annealed GD from parent (n={parent.size})", flush=True)
    print(f"{'='*60}", flush=True)
    
    a_parent = parent / parent.sum()
    a_res, J_res, iters = beta_anneal_gradient_descent(a_parent, p1_deadline, verbose=True)
    update_best(a_res)
    print(f"  Phase 1 done: J={J_res:.8f} iters={iters}", flush=True)
    
    # === PHASE 2: Coordinate descent polishing ===
    p2_deadline = min(t_start + 600, deadline - 400)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Coordinate Descent Polishing", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        a_cd = coordinate_descent_fast(best_seq, p2_deadline, max_passes=15)
        update_best(a_cd)
        print(f"  Phase 2 done: J={best_score:.8f}", flush=True)
    
    # === PHASE 3: Diverse initializations with GD ===
    p3_deadline = min(t_start + 900, deadline - 300)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Diverse Initializations + GD", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        n_best = best_seq.size
        inits = generate_inits(n_best, rng)
        # Add perturbed best
        for noise_scale in [0.05, 0.1, 0.2]:
            pert = np.maximum(best_seq + rng.normal(0, noise_scale, n_best), 0.01)
            inits.append(pert / pert.sum())
        
        # Sort by initial J
        init_scores = [(compute_J(init), init) for init in inits]
        init_scores.sort(key=lambda x: x[0])
        
        time_per_init = 25
        max_inits = min(len(init_scores), int((p3_deadline - time.monotonic()) // time_per_init))
        
        for score_init, init in init_scores[:max(3, max_inits)]:
            if time.monotonic() > p3_deadline:
                break
            phase_dl = min(time.monotonic() + time_per_init, p3_deadline)
            a_r, J_r, _ = beta_anneal_gradient_descent(
                init, phase_dl, verbose=False
            )
            # Quick CD polish
            a_cd = coordinate_descent_fast(a_r, min(phase_dl + 10, p3_deadline), max_passes=3)
            update_best(a_cd)
        
        print(f"  Phase 3 done: J={best_score:.8f}", flush=True)
    
    # === PHASE 4: Multi-length exploration ===
    p4_deadline = min(t_start + 1150, deadline - 250)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = set()
        for L in [384, 448, 512, 576, 640, 768, 896, 1024, 1280]:
            target_lengths.add(L)
        for mult in [0.75, 1.25, 1.5]:
            target_lengths.add(int(n_base * mult))
        
        target_lengths = sorted([L for L in target_lengths if 128 <= L <= 2048])
        
        time_per_len = 20
        for tgt_n in target_lengths:
            if time.monotonic() > p4_deadline:
                break
            if tgt_n == n_base:
                continue
            
            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            
            phase_dl = min(time.monotonic() + time_per_len, p4_deadline)
            a_r, J_r, _ = beta_anneal_gradient_descent(a_ext, phase_dl, verbose=False)
            a_cd = coordinate_descent_fast(a_r, min(phase_dl + 5, p4_deadline), max_passes=3)
            update_best(a_cd)
        
        print(f"  Phase 4 done: J={best_score:.8f}", flush=True)
    
    # === PHASE 5: Focused high-beta GD + CD ===
    p5_deadline = min(t_start + 1350, deadline - 100)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Focused High-Beta Optimization", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        # Run high-beta GD for extended time
        a_hi, J_hi, _ = gradient_descent_smooth(
            best_seq, [5000, 10000, 20000, 50000], p5_deadline,
            lr_init=0.1, max_iters=1000000, verbose=True
        )
        update_best(a_hi)
        
        # CD polish
        a_cd = coordinate_descent_fast(best_seq, p5_deadline, max_passes=10)
        update_best(a_cd)
        
        print(f"  Phase 5 done: J={best_score:.8f}", flush=True)
    
    # === PHASE 6: Simulated Annealing + Final CD ===
    p6_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: SA + Final Polishing", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        sa_dl = min(t_start + budget_s - 60, deadline)
        a_sa = simulated_annealing(best_seq, sa_dl, max_iters=50000, T_start=0.01, T_end=0.0005)
        update_best(a_sa)
        
        # Final CD
        a_cd = coordinate_descent_fast(best_seq, deadline, max_passes=10)
        update_best(a_cd)
        
        # One more high-beta GD on the best
        a_final, J_final, _ = gradient_descent_smooth(
            best_seq, [10000, 20000], deadline,
            lr_init=0.05, max_iters=500000, verbose=False
        )
        update_best(a_final)
        
        # Final CD
        a_cd = coordinate_descent_fast(best_seq, deadline, max_passes=5)
        update_best(a_cd)
        
        print(f"  Phase 6 done: J={best_score:.8f}", flush=True)
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)
    
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_seq]
