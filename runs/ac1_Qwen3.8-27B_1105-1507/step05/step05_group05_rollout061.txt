"""
Multi-phase optimizer for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Uses KL mirror descent with Nesterov acceleration, adaptive smooth-max temperature,
random partial restarts, and coordinate descent polishing.
"""

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_grad(a, tau):
    """Compute soft_max(conv(a,a), tau) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Numerically stable log-sum-exp
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    log_sum = np.log(exp_b.sum())
    soft_m = b_max + log_sum / tau
    p = exp_b / exp_b.sum()
    
    # Gradient: grad_i = 2 * sum_m p[i+m] * a[m]
    # p has length 2n-1, a has length n
    # For each i in [0, n-1]: grad_i = 2 * dot(a, p[i:i+n])
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(a, p[i:i+n])
    
    return soft_m, grad, p, b

def smooth_max_grad_fft(a, tau):
    """Faster version using FFT for the gradient computation."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    log_sum = np.log(exp_b.sum())
    soft_m = b_max + log_sum / tau
    p = exp_b / exp_b.sum()
    
    # grad_i = 2 * sum_{m=0}^{n-1} a[m] * p[i+m]
    # This is a cross-correlation. Use FFT.
    size = 1
    while size < 2 * n:
        size <<= 1
    a_padded = np.zeros(size)
    a_padded[:n] = a
    p_padded = np.zeros(size)
    p_padded[:2*n-1] = p
    
    # We want: grad[i] = 2 * sum_m a[m] * p_padded[i+m]
    # = 2 * (a_reversed * p_padded)[i + n - 1]
    # where * denotes linear convolution
    a_rev = a[::-1]
    conv_result = np.fft.irfft(np.fft.rfft(a_rev, size) * np.fft.rfft(p_padded, size), size)
    grad = 2.0 * conv_result[n-1 : 2*n-1]
    
    return soft_m, grad, p, b

def kl_mirror_descent_nesterov(a, max_iters, deadline, tau_start=20.0, tau_end=200.0,
                                lr=0.01, momentum=0.9):
    """
    KL mirror descent with Nesterov-style momentum on the simplex.
    Optimizes soft_max(conv(a,a), tau) with sum(a)=1, a>=0.
    """
    a = np.maximum(a.copy(), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Nesterov: maintain a velocity and a lookahead point
    v = np.zeros(n)  # velocity in log-space
    y = a.copy()  # lookahead point (Nesterov)
    
    # Adaptive learning rate
    base_lr = lr
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        
        # Nesterov lookahead
        y = a + momentum * v
        
        # Compute gradient at y (projected)
        y = np.maximum(y, 1e-10)
        y = y / y.sum()
        
        soft_m, grad, p, b = smooth_max_grad_fft(y, tau)
        
        # Project gradient onto tangent space of simplex: grad - mean(grad)
        grad_proj = grad - grad.mean()
        
        # KL mirror descent update in log-space:
        # log(y_new) = log(y) - lr * grad_proj
        # a_new = exp(log(y) - lr * grad_proj) / Z
        # = y * exp(-lr * grad_proj) / Z
        lr_t = base_lr / (1.0 + 0.1 * it)
        weights = np.exp(-lr_t * grad_proj)
        a_new = y * weights
        a_new = np.maximum(a_new, 1e-10)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Update velocity
        v = momentum * v + (a_new - a)
        a = a_new
        
        # Check improvement
        M = fast_conv(a).max()
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
    
    return best_a

def coordinate_descent_fast(a, max_sweeps, deadline, seed=0):
    """Fast coordinate descent with exact 1D optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        sweep_improved = False
        order = np.arange(n)
        rng.shuffle(order)
        
        for i in order:
            x_cur = a[i]
            if x_cur < 1e-10 and fast_conv(a).max() < best_M:
                # Try a small positive value
                a[i] = 0.01
                s = a.sum()
                a[i] = 0.01 / s * a.sum()  # keep roughly same scale
                # Actually just set it and let normalization handle it
                a[i] = 0.01
                s = a.sum()
                if s > 0:
                    M_test = fast_conv(a / s).max()
                    if M_test > best_M * 1.1:
                        a[i] = 0.0
                        s = a.sum()
                        if s > 0:
                            a = a / s
                continue
            
            # Golden section to find optimal a[i]
            # Current conv
            b = fast_conv(a)
            S = a.sum()
            
            # Affected range: k from max(0, i) to min(2n-2, i+n-1)
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                if mask.any():
                    base_max = b[mask].max()
                else:
                    base_max = 0.0
            else:
                base_max = 0.0
            
            # Precompute for affected range
            affected = np.arange(k_lo, k_hi + 1)
            c_vals = np.zeros(len(affected))
            slopes = np.zeros(len(affected))
            quad_idx = -1
            
            for idx, k in enumerate(affected):
                if k == 2 * i:
                    c_vals[idx] = b[k] - x_cur * x_cur
                    slopes[idx] = 0.0
                    quad_idx = idx
                else:
                    j = k - i
                    if 0 <= j < n:
                        c_vals[idx] = b[k] - 2 * x_cur * a[j]
                        slopes[idx] = 2 * a[j]
                    else:
                        c_vals[idx] = b[k]
                        slopes[idx] = 0.0
            
            def obj_local(x_test):
                new_sum = S - x_cur + x_test
                if new_sum <= 1e-10:
                    return 1e10
                vals = c_vals + slopes * x_test
                m = base_max
                idx_max = np.argmax(vals)
                m = max(m, vals[idx_max])
                if quad_idx >= 0:
                    qv = c_vals[quad_idx] + x_test * x_test
                    m = max(m, qv)
                return m / (new_sum * new_sum)
            
            # Golden section
            lo = 0.0
            hi = max(x_cur * 2.0 + 0.1, 0.1)
            
            # Check if increasing helps
            if obj_local(hi) > obj_local(x_cur) and x_cur > 0.01:
                hi = x_cur
                lo = 0.0
            else:
                # Try expanding
                while obj_local(hi) < obj_local(hi * 0.5) and hi < 2.0:
                    hi *= 2.0
            
            gr = (np.sqrt(5) - 1) / 2
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_local(a_pt)
            fb = obj_local(b_pt)
            
            for _ in range(40):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_local(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_local(b_pt)
            
            x_opt = (lo + hi) / 2.0
            if obj_local(0.0) < obj_local(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - a[i]) > 1e-10:
                a[i] = max(0.0, x_opt)
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 5
        if no_improve > 20:
            break
    
    return best_a

def partial_perturb(a, rng, frac=0.3):
    """Perturb a random fraction of coordinates."""
    n = len(a)
    a_new = a.copy()
    mask = rng.random(n) < frac
    a_new[mask] = np.maximum(a_new[mask] * np.exp(rng.normal(0, 0.5, mask.sum())), 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Multi-phase optimizer for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Uses KL mirror descent with Nesterov momentum, peak-deflation heuristic,
coordinate descent polishing, and multi-scale random restarts.
Key innovations: FFT-based smooth-max gradient, peak cancellation updates,
and adaptive multi-phase search schedule.
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """
    Compute soft_max(conv(a,a), tau), its gradient, the probability weights,
    and the raw convolution. Uses FFT for the gradient.
    Returns (soft_m, grad, p, b).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    max_shifted = shifted.max()
    exp_b = np.exp(shifted - max_shifted)
    log_sum = max_shifted + np.log(exp_b.sum())
    soft_m = b_max + log_sum / tau
    p = exp_b / exp_b.sum()

    # grad_i = 2 * sum_{m=0}^{n-1} a[m] * p[i+m]
    # Compute via FFT cross-correlation
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    a_padded = np.zeros(size)
    a_padded[:n] = a
    p_padded = np.zeros(size)
    p_padded[:2 * n - 1] = p

    # conv(a_rev, p_padded)[n-1 + i] = sum_k a[n-1-k] * p_padded[n-1+i-k]
    # Let j = n-1-k: = sum_j a[j] * p_padded[i+j]  ... this is what we want!
    a_rev = a[::-1].copy()
    freq_a = np.fft.rfft(a_rev, size)
    freq_p = np.fft.rfft(p_padded, size)
    conv_result = np.fft.irfft(freq_a * freq_p, size)
    grad = 2.0 * conv_result[n - 1: 2 * n - 1]

    return soft_m, grad, p, b


def kl_mirror_descent_nesterov(a, max_iters, deadline, tau_start, tau_end, lr, momentum):
    """
    KL mirror descent with Nesterov-style acceleration on the probability simplex.
    Minimizes soft_max(conv(a,a), tau) with adaptive temperature.
    """
    a = np.maximum(a.astype(np.float64).copy(), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    v = np.zeros(n)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.4)
        lr_t = lr / (1.0 + 0.05 * it)

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 1e-10)
        y = y / y.sum()

        soft_m, grad, p, b = smooth_max_and_grad(y, tau)

        # Project to tangent space (sum-zero)
        grad_proj = grad - grad.mean()

        # KL mirror descent: a_new = y * exp(-lr * grad_proj) / Z
        weights = np.exp(-lr_t * grad_proj)
        a_new = y * weights
        a_new = np.maximum(a_new, 1e-10)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        v = momentum * v + (a_new - a)
        a = a_new

        M = fast_conv(a).max()
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

    return best_a


def peak_deflation(a, max_iters, deadline, eta=0.02):
    """
    Peak cancellation: at each step, find the peak lag and deflate the
    corresponding conjugate pairs. Very cheap per-iteration cost.
    """
    a = np.maximum(a.astype(np.float64).copy(), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_star = int(np.argmax(b))

        # Deflate: reduce a_i where a[k_star - i] is large
        # a_i -= eta * a[k_star - i] for valid i
        deflation = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                deflation[i] = a[j]

        a_new = a - eta * deflation
        a_new = np.maximum(a_new, 1e-10)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        M = fast_conv(a_new).max()
        if M < best_M - 1e-13:
            best_M = M
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller eta
            if no_improve > 50:
                break

    return best_a


def coord_optimize_single(a, i, b, S, base_max, c_vals, slopes, quad_idx, n):
    """Optimize a single coordinate via golden section search."""
    x_cur = a[i]

    def obj_at(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 1e-10:
            return 1e10
        vals = c_vals + slopes * x_test
        m = base_max
        idx_max = np.argmax(vals)
        if vals[idx_max] > m:
            m = vals[idx_max]
        if quad_idx >= 0:
            qv = c_vals[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)

    lo = 0.0
    hi = max(x_cur * 2.5 + 0.05, 0.05)

    # Quick expansion check
    if obj_at(hi * 0.5) < obj_at(hi) and hi < 5.0:
        hi *= 2.0

    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(45):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_at(0.0) < obj_at(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def coordinate_descent_fast(a, max_sweeps, deadline, seed):
    """Fast coordinate descent with exact 1D golden section optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = np.arange(n)
        rng.shuffle(order)
        sweep_improved = False

        for i in order:
            b = fast_conv(a)
            S = a.sum()
            x_cur = a[i]

            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)

            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                if mask.any():
                    base_max = b[mask].max()
                else:
                    base_max = 0.0
            else:
                base_max = 0.0

            affected_len = k_hi - k_lo + 1
            c_vals = np.zeros(affected_len)
            slopes = np.zeros(affected_len)
            quad_idx = -1

            for idx in range(affected_len):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = b[k] - x_cur * x_cur
                    slopes[idx] = 0.0
                    quad_idx = idx
                else:
                    j = k - i
                    if 0 <= j < n:
                        c_vals[idx] = b[k] - 2 * x_cur * a[j]
                        slopes[idx] = 2 * a[j]
                    else:
                        c_vals[idx] = b[k]
                        slopes[idx] = 0.0

            x_opt = coord_optimize_single(a, i, b, S, base_max, c_vals, slopes, quad_idx, n)

            if abs(x_opt - a[i]) > 1e-11:
                a[i] = x_opt
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 5
        if no_improve > 25:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def partial_perturb(a, rng, frac, mode):

    else:

        a_pd = peak_deflation(tracker.best_a, 5000, p2_deadline, eta=0.03)
        tracker.update(a_pd, "(peak-deflate)")

        a_pd2 = peak_deflation(tracker.best_a, 5000, p2_deadline, eta=0.01)
        tracker.update(a_pd2, "(peak-deflate-slow)")

    tracker.report("Phase 2 done")

    # === PHASE 3: Coordinate Descent ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Coordinate Descent", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_deadline = min(time.monotonic() + 250, deadline - 200)
    if tracker.best_a is not None:
        n_best = tracker.best_a.size
        n_sweeps = max(20, int(200000 / max(1, n_best)))

        a_cd = coordinate_descent_fast(tracker.best_a, n_sweeps, min(time.monotonic() + 120, p3_deadline), seed=seed)
        tracker.update(a_cd, "(CD)")

        # From perturbed
        a_pert = partial_perturb(tracker.best_a, rng, 0.15, 1)
        a_cd2 = coordinate_descent_fast(a_pert, max(10, n_sweeps // 2), min(time.monotonic() + 80, p3_deadline), seed=seed + 1)
        tracker.update(a_cd2, "(CD-pert)")

    tracker.report("Phase 3 done")

    # === PHASE 4: Diverse Restarts ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Diverse Restarts", flush=True)
    print(f"{'=' * 60}", flush=True)

    p4_deadline = min(time.monotonic() + 300, deadline - 100)
    if tracker.best_a is not None and time.monotonic() < p4_deadline:
        n_rest = 12
        n_best = tracker.best_a.size
        per_restart = max(8, int((p4_deadline - time.monotonic()) / n_rest))

        for r in range(n_rest):
            if time.monotonic() > p4_deadline - 5:
                break

            mode = r % 4
            frac = 0.2 + 0.1 * (r % 3)
            a_r = partial_perturb(tracker.best_a, rng, frac, mode)

            # Quick KL + peak deflation + CD pipeline
            t_sub = min(time.monotonic() + per_restart * 0.4, p4_deadline - 10)
            a_r = kl_mirror_descent_nesterov(a_r, 300, t_sub, tau_start=50, tau_end=300, lr=0.01, momentum=0.85)
            tracker.update(a_r, f"(restart-{r}-KL)")

            if time.monotonic() < p4_deadline - 8:
                t_sub2 = min(time.monotonic() + per_restart * 0.2, p4_deadline - 8)
                a_r = peak_deflation(a_r, 2000, t_sub2, eta=0.02)
                tracker.update(a_r, f"(restart-{r}-PD)")

            if time.monotonic() < p4_deadline - 5:
                t_sub3 = min(time.monotonic() + per_restart * 0.3, p4_deadline - 5)
                a_r = coordinate_descent_fast(a_r, 15, t_sub3, seed=seed + r * 7 + 100)
                tracker.update(a_r, f"(restart-{r}-CD)")

    tracker.report("Phase 4 done")

    # === PHASE 5: Multi-length Exploration ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Multi-length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    p5_deadline = min(time.monotonic() + 150, deadline - 50)
    if tracker.best_a is not None and time.monotonic() < p5_deadline:
        n_best = tracker.best_a.size
        target_lengths = []
        for mult in [0.75, 1.25, 1.5, 2.0]:
            tl = int(n_best * mult)
            if 192 <= tl <= 2048:
                target_lengths.append(tl)

        # Deduplicate
        seen = set()
        unique_lengths = []
        for tl in target_lengths:
            if tl not in seen:
                seen.add(tl)
                unique_lengths.append(tl)

        per_len = max(10, int((p5_deadline - time.monotonic()) / max(1, len(unique_lengths))))

        for tl in unique_lengths:
            if time.monotonic() > p5_deadline - 5:
                break
            print(f"  Trying n={tl} (~{per_len:.0f}s)", flush=True)

            a_res = resample_seq(tracker.best_a, tl)
            a_res = np.maximum(a_res, 1e-6)
            a_res = a_res / a_res.sum()

            a_opt = kl_mirror_descent_nesterov(
                a_res, int(per_len * 30), min(time.monotonic() + per_len * 0.5, p5_deadline - 10),
                tau_start=30.0, tau_end=250.0, lr=0.012, momentum=0.85
            )
            tracker.update(a_opt, f"(n={tl}-KL)")

            if time.monotonic() < p5_deadline - 5:
                a_cd = coordinate_descent_fast(a_opt, 15, min(time.monotonic() + per_len * 0.4, p5_deadline - 5), seed=seed + tl)
                tracker.update(a_cd, f"(n={tl}-CD)")

    tracker.report("Phase 5 done")

    # === PHASE 6: Final Intensive Polish ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 6: Final Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best_a is not None and time.monotonic() < deadline - 5:
        a_final = tracker.best_a.copy()
        remaining = deadline - time.monotonic()

        n_rounds = max(2, int(remaining / 20))
        for rnd in range(n_rounds):
            if time.monotonic() > deadline - 3:
                break

            # KL with high tau (sharp max approximation)
            a_final = kl_mirror_descent_nesterov(
                a_final, 300, deadline - 15,
                tau_start=200.0, tau_end=1000.0, lr=0.004, momentum=0.92
            )
            tracker.update(a_final, f"(final-KL-{rnd})")

            # Peak deflation
            a_final = peak_deflation(a_final, 3000, deadline - 10, eta=0.015)
            tracker.update(a_final, f"(final-PD-{rnd})")

            # Coordinate descent
            a_final = coordinate_descent_fast(a_final, 10, deadline - 5, seed=seed + 5000 + rnd)
            tracker.update(a_final, f"(final-CD-{rnd})")

    # Fallback
    if tracker.best_a is None:
        tracker.best_a = np.ones(512)
        tracker.best_score = compute_J(tracker.best_a)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best_a.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best_a]
