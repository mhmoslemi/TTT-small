def build_A_values(n, a):
    """Build sparse matrix values for the LP constraint matrix A where (a*g)(k) = (A g)_k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals


def lp_best_response(a, M):
    n = len(a)
    rows, cols, vals = build_A_values(n, a)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def anderson_solve(history_x, history_Fx, m):
    """Solve the Anderson mixing QP: find alpha minimizing ||sum alpha_i * r_i||^2 s.t. sum alpha = 1.
    Returns the mixed point sum(alpha_i * Fx_i)."""
    residuals = []
    for i in range(m):
        residuals.append(history_x[i] - history_Fx[i])
    residuals = np.array(residuals)  # shape (m, n)
    # G_{ij} = r_i . r_j
    G = residuals @ residuals.T  # (m, m)
    # Add small regularization for stability
    G += 1e-12 * np.eye(m)
    ones = np.ones(m)
    try:
        G_inv_ones = np.linalg.solve(G, ones)
        denom = ones @ G_inv_ones
        if abs(denom) < 1e-15:
            alpha = ones / m
        else:
            alpha = G_inv_ones / denom
    except np.linalg.LinAlgError:
        alpha = ones / m
    # Clamp alpha to be in a reasonable range (allow mild extrapolation)
    alpha = np.clip(alpha, -1.0, 1.0)
    alpha = alpha / alpha.sum()
    mixed = np.zeros(len(history_x[0]))
    for i in range(m):
        mixed += alpha[i] * history_Fx[i]
    return np.maximum(mixed, 0)

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.05)
    inits.append(np.exp(-8 * (t - 0.3)**2) + np.exp(-8 * (t - 0.7)**2) + 0.03)
    # Random smooth
    k = max(3, n // 20)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    # GS from random
    a_rand = np.maximum(rng.normal(0.5, 0.3, n), 0.01)
    inits.append(gs_flat_spectral(a_rand, n_iters=100, beta=0.85))
    # GS from parent (if available)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(gs_flat_spectral(p, n_iters=150, beta=0.9))
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    # Determine lengths to explore - focus on longer sequences for better bounds
    base_n = parent.size
    candidate_lengths = set()
    for n_try in [384, 448, 512, 576, 640, 768, 896, 1024, 1152, 1280, 1408, 1536]:
        candidate_lengths.add(n_try)
    lengths = sorted(candidate_lengths)
    print(f"Lengths to try: {lengths}", flush=True)

    # === PHASE 1: Diverse initializations + smooth optimization ===
    p1_budget = 400.0
    p1_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Smooth Optimization ({p1_budget:.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    best_per_length = {}
    # Allocate more time to longer lengths (they have better theoretical bounds)
    # but weight by 1/n for computational cost
    n_candidates = len(lengths)
    time_per_length = p1_budget / n_candidates

    for n in lengths:
        t0 = time.monotonic()
        remaining = deadline - 250 - time.monotonic()
        if remaining < 30:
            break
        allot = min(time_per_length, remaining * 0.5)
        dl = time.monotonic() + allot
        if time.monotonic() >= dl:
            continue

        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_M = float('inf')

        # Run smooth optimization on each initialization
        n_iters_base = max(200, int(8000 * (256.0 / n)))
        for idx, a0 in enumerate(inits):
            if time.monotonic() > dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()
            # Use less time for later inits
            frac = 1.0 / (1.0 + idx * 0.3)
            sub_dl = time.monotonic() + (dl - time.monotonic()) * frac
            a = run_nesterov_smooth(a0, sub_dl, max_iters=n_iters_base,
                                    tau_start=1.0, tau_end=200.0,
                                    lr_init=0.35, lr_end=0.002)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: M={best_M:.7f} J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Anderson-accelerated LP mirror descent ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: Anderson LP Mirror Descent", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        est_lp_ms = 3.0 * (n / 128.0)**2
        print(f"  n={n}, est LP time={est_lp_ms:.0f}ms", flush=True)
        lp_budget = min(900.0, max(100.0, (deadline - 100) - time.monotonic()))
        max_iters = max(50, int(lp_budget / max(0.01, est_lp_ms / 1000.0)))
        print(f"  LP budget={lp_budget:.0f}s, max_iters={max_iters}", flush=True)

        # Run Anderson-accelerated LP on the best sequence
        a = run_lp_anderson(best.seq, min(time.monotonic() + lp_budget * 0.6, deadline - 80),
                            max_iters=max_iters, anderson_window=5,
                            t_base=0.35, min_t=0.005)
        best.update(a)

        # Second pass with more iterations if time remains
        if time.monotonic() < deadline - 60:
            a2 = run_lp_anderson(a, deadline - 40,
                                 max_iters=int(max_iters * 0.7), anderson_window=6,
                                 t_base=0.2, min_t=0.003)
            best.update(a2)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Try top-2 lengths with LP polish ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Multi-length LP Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best_per_length and time.monotonic() < deadline - 50:
        # Sort by J value
        sorted_lengths = sorted(best_per_length.keys(),
                                key=lambda n: compute_J(best_per_length[n]))
        for n_try in sorted_lengths[:2]:
            if time.monotonic() > deadline - 30:
                break
            if n_try == best.seq.size:
                continue
            a_cand = best_per_length[n_try]
            # Quick smooth polish
            a_cand = run_nesterov_smooth(a_cand, min(time.monotonic() + 30, deadline - 25),
                                         max_iters=500, tau_start=5, tau_end=100,
                                         lr_init=0.15, lr_end=0.01)
            best.update(a_cand)
            # LP polish
            if time.monotonic() < deadline - 25:
                a_lp = run_lp_anderson(a_cand, deadline - 20,
                                       max_iters=50, anderson_window=4,
                                       t_base=0.3, min_t=0.01)
                best.update(a_lp)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Final plain LP polish ===
    p4_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Final Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 5:
        a_final = run_lp_mirror_plain(best.seq, deadline, max_iters=80)
        best.update(a_final)

        # One more Anderson pass
        if time.monotonic() < deadline - 3:
            a_final2 = run_lp_anderson(a_final, deadline,
                                       max_iters=30, anderson_window=5,
                                       t_base=0.15, min_t=0.003)
            best.update(a_final2)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Length exploration around best ===
    if best.seq is not None and time.monotonic() < deadline - 10:
        n_best = best.seq.size
        for delta in [-32, -16, 16, 32, 64, 128]:
            if time.monotonic() > deadline - 8:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 2048:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            a_ext = run_nesterov_smooth(a_ext, min(time.monotonic() + 15, deadline - 5),
                                        max_iters=300, tau_start=5, tau_end=100,
                                        lr_init=0.15, lr_end=0.01)
            best.update(a_ext)
            if time.monotonic() < deadline - 5:
                a_lp = run_lp_anderson(a_ext, deadline - 3, max_iters=20,
                                       anderson_window=4, t_base=0.2, min_t=0.01)
                best.update(a_lp)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
