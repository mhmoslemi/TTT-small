the gradient of the smooth max of self-convolution is computed via a single FFT-based
correlation, enabling 10-100x more iterations than the previous O(n^2) per-step approach.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def fft_gradient_smooth_max(a, tau):
    """
    Compute gradient of L(a) = (1/tau)*log(sum_k exp(tau*b[k])) where b = conv(a,a).
    Uses FFT to compute in O(n log n).
    Returns (gradient, b_max, p_weights).
    """
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    # Gradient: dL/da[i] = 2 * (a * p_rev)[2n-2-i]
    p_rev = p[::-1]
    fp_rev = np.fft.rfft(p_rev, size)
    conv_result = np.fft.irfft(fa * fp_rev, size)
    # We need conv_result[2n-2-i] for i=0..n-1
    # That's conv_result[n-1 : 2n-1] reversed
    grad = 2.0 * conv_result[n - 1:2 * n - 1][::-1]
    return grad, b_max, p


def run_smooth_gradient_descent(a, deadline, tau_start=10.0, tau_end=300.0,
                                 lr=0.1, beta1=0.9, beta2=0.999,
                                 coord_every=200, max_iters=5000, seed=42):
    """
    Adam optimizer on smooth max of self-convolution with periodic CD polishing.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(seed)

    m = np.zeros(n)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if time.monotonic() > deadline and it > 10:
            break

        frac = min(1.0, it / max(1, max_iters * 0.8))
        # Smooth annealing of tau
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr_t = lr / (1 + 0.05 * it)

        grad, b_max, p = fft_gradient_smooth_max(a, tau)

        # Project gradient to sum-zero subspace (tangent to sum=1 constraint)
        grad_mean = grad.mean()
        grad = grad - grad_mean

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Check improvement
        M_new = fast_conv(a_new).max()
        if M_new < best_M * (1 - 1e-12):
            best_M = M_new
            best_a = a_new.copy()

        a = a_new

        # Periodic coordinate descent polish
        if coord_every > 0 and it % coord_every == 0 and it > 0:
            a = quick_cd_sweep(a, 3, deadline)
            M_cd = fast_conv(a).max()
            if M_cd < best_M:
                best_M = M_cd
                best_a = a.copy()

    return best_a


def quick_cd_sweep(a, n_sweeps, deadline):
    """Quick coordinate descent with a few sweeps."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        for i in range(n):
            if time.monotonic() > deadline:
                return a
            x_opt = _cd_single_update_fast(a, i)
            a[i] = x_opt
        s = a.sum()
        if s > 1e-10:
            a = a / s
    return a


def _cd_single_update_fast(a, i):
    """
    Fast coordinate update: find optimal a[i] minimizing max(conv(a_new, a_new))
    where a_new = a with a[i] = x, normalized to sum=1.
    Uses binary/golden search on the 1D subproblem.
    """
    n = len(a)
    x_cur = a[i]
    S = a.sum()

    # Precompute: contribution of a[i] to each lag k
    # b[k] = sum_j a[j]*a[k-j]; the part involving a[i] is:
    #   if k-i in [0,n-1]: a[i]*a[k-i] (appears twice unless k=2i)
    # So b[k] = base_k + 2*a[i]*a[k-i] (for k != 2i) or base_k + a[i]^2 (for k=2i)
    # where base_k = b[k] - (contribution of a[i])

    # affected k range: [i, i+n-1] clipped to [0, 2n-2]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    n_affected = k_hi - k_lo + 1

    # Compute base values for affected k
    # For k in [k_lo, k_hi]:
    #   j = k - i, must have 0 <= j < n
    #   base_k = b[k] - 2*a[i]*a[j] (if k != 2i) or b[k] - a[i]^2 (if k == 2i)
    # But we don't have b computed. Let's compute affected b directly.
    # b[k] = sum_j a[j]*a[k-j]
    # This is expensive to compute for all affected k. Instead, use the formula:
    # b[k] for affected k can be computed from the current a.

    # Alternative: use a simpler approach. Just do a few evaluations.
    # Since we're in a "quick" sweep, use a simpler method.

    # Evaluate max_b(x) for a few x values and pick the best
    # max_b(x) = max( base_max, max_{affected k} (c_k + slope_k * x + quad_k * x^2) ) / (S - x_cur + x)^2
    # where base_max = max of b[k] for unaffected k

    # Compute current b for unaffected part (this is the expensive part)
    # For speed in quick sweep, just do a ternary search with a small number of evaluations
    # using the full convolution

    lo = 0.0
    hi = max(3.0 * x_cur, 0.5)

    def eval_max_b(x_test):
        a[i] = x_test
        new_S = S - x_cur + x_test
        if new_S < 1e-10:
            return 1e10
        a_norm = a / new_S
        b = fast_conv(a_norm)
        a[i] = x_cur  # restore
        return b.max()

    # Golden section search (limited iterations for speed)
    gr = (np.sqrt(5) - 1) / 2
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1 = eval_max_b(a1)
    f2 = eval_max_b(a2)

    for _ in range(30):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = hi - gr * (hi - lo)
            f1 = eval_max_b(a1)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = eval_max_b(a2)

    x_opt = (lo + hi) / 2.0
    # Check boundary
    if eval_max_b(0.0) < eval_max_b(x_opt):
        x_opt = 0.0
    a[i] = x_cur  # restore
    return max(0.0, x_opt)


def cd_sweep_accurate(a, max_sweeps, deadline, shuffle=True, seed=0):
    """
    Accurate coordinate descent using precomputed convolution updates.
    Each coordinate update is O(n) using exact 1D analysis.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = np.arange(n)
        if shuffle:
            rng.shuffle(order)

        # Precompute current convolution
        b = fast_conv(a)
        S = a.sum()

        sweep_changed = False
        for i in order:
            if time.monotonic() > deadline:
                break

            x_cur = a[i]
            if x_cur < 1e-15:
                # Try setting to small positive
                a[i] = 1e-4
                b = fast_conv(a)
                S = a.sum()
                a[i] = x_cur
                continue

            # Affected lag range
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)

            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                if mask.any():
                    base_max = float(b[mask].max())
                else:
                    base_max = 0.0
            else:
                base_max = 0.0

            # For affected k: b[k] = c_k + slope_k * x + quad_k * x^2
            # where the quadratic term only exists at k = 2*i
            affected_ks = np.arange(k_lo, k_hi + 1)
            c = np.zeros(len(affected_ks))
            slope = np.zeros(len(affected_ks))
            quad_idx = -1

            for idx, k in enumerate(affected_ks):
                j = k - i
                if 0 <= j < n:
                    if k == 2 * i:
                        c[idx] = b[k] - x_cur * x_cur
                        slope[idx] = 0.0
                        quad_idx = idx
                    else:
                        c[idx] = b[k] - 2.0 * x_cur * a[j]
                        slope[idx] = 2.0 * a[j]
                else:
                    c[idx] = b[k]
                    slope[idx] = 0.0

            def obj_1d(x_test):
                new_S = S - x_cur + x_test
                if new_S <= 1e-10:
                    return 1e10
                # max of (c + slope*x) over affected, and base_max
                m = base_max
                vals = c + slope * x_test
                m = max(m, float(vals.max()))
                if quad_idx >= 0:
                    qv = c[quad_idx] + x_test * x_test
                    if qv > m:
                        m = qv
                return m / (new_S * new_S)

            # Golden section search
            lo = 0.0
            hi = max(2.0 * x_cur, np.sqrt(max(base_max + x_cur * x_cur, 0.0)) + 0.1, 0.1)
            if hi > 10.0:
                hi = 10.0

            gr = (np.sqrt(5) - 1) / 2
            p1 = hi - gr * (hi - lo)
            p2 = lo + gr * (hi - lo)
            f1 = obj_1d(p1)
            f2 = obj_1d(p2)

            for _ in range(40):
                if f1 < f2:
                    hi = p2
                    p2 = p1
                    f2 = f1
                    p1 = hi - gr * (hi - lo)
                    f1 = obj_1d(p1)
                else:
                    lo = p1
                    p1 = p2
                    f1 = f2
                    p2 = lo + gr * (hi - lo)
                    f2 = obj_1d(p2)

            x_opt = (lo + hi) / 2.0
            if obj_1d(0.0) < obj_1d(x_opt):
                x_opt = 0.0

            if abs(x_opt - x_cur) > 1e-10:
                a[i] = x_opt
                # Update b for affected lags
                for idx, k in enumerate(affected_ks):
                    if k == 2 * i:
                        b[k] = c[idx] + x_opt * x_opt
                    else:
                        b[k] = c[idx] + slope[idx] * x_opt
                S = a.sum()
                sweep_changed = True

        # Renormalize
        S = a.sum()
        if S < 1e-10:
            break
        a = a / S
        # Recompute b after normalization (since we divided by S)
        b = fast_conv(a)

        M = float(b.max())
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_changed:
            no_improve += 3
        if no_improve > 40:
            break

    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix A such that (A @ g)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'threads': 1})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a, max_iters, deadline):
    """LP mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        improved = False
        base_t = max(0.05, 0.5 * (1 - 0.3 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = float(fast_conv(a_trial).max())
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Various structured inits
    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t), 0.01))

    # Ramp
    inits.append(0.2 + 0.8 * t)
    inits.append(0.2 + 0.8 * (1 - t))
    # V-shape
    inits.append(0.3 + 0.7 * np.abs(2 * t - 1))
    # Gaussian
    inits.append(np.exp(-8 * (t - 0.5) ** 2) + 0.1)

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Try a "Barker-like" pattern (alternating with some structure)
    bark = np.ones(n)
    bark[::2] = 0.3
    inits.append(bark.copy())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search using FFT-accelerated smooth gradient descent as primary engine,
    coordinate descent for polishing, and LP mirror descent for final refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    # === PHASE 1: FFT Smooth Gradient Descent (main engine) ===
    p1_start = time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: FFT Smooth Gradient Descent (budget=~700s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Try multiple lengths
    if tracker.best is not None:
        n_parent = tracker.best.size
    else:
        n_parent = 512

    # Lengths to try: include parent length and nearby
    lengths = sorted(set([384, 512, 640, n_parent, 768, 1024]))
    # Remove lengths that are too far from parent or too large
    lengths = [n for n in lengths if 256 <= n <= 2048]
    if not lengths:
        lengths = [512]

    # Time allocation per length (more time for parent length)
    total_p1_time = 700
    n_lengths = len(lengths)
    base_time = total_p1_time / n_lengths

    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        t_alloc = base_time * (1.5 if n == n_parent else 1.0)
        phase_deadline = min(time.monotonic() + t_alloc, deadline - 350)

        if time.monotonic() >= phase_deadline:
            continue

        t0 = time.monotonic()
        print(f"  Starting n={n}, allocated {t_alloc:.0f}s", flush=True)

        inits = diverse_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_for_n = None
        best_M_for_n = float('inf')

        # Limit inits based on time
        max_inits = max(3, min(len(inits), int(t_alloc / 30)))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > phase_deadline - 30:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            # Run smooth gradient descent
            remaining_iters = max(100, int((phase_deadline - time.monotonic()) * 3))
            a_opt = run_smooth_gradient_descent(
                a0, phase_deadline - 10,
                tau_start=10.0, tau_end=250.0,
                lr=0.08, coord_every=300,
                max_iters=remaining_iters, seed=seed + idx * 13
            )

            M_val = float(fast_conv(a_opt).max())
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a_opt.copy()

            # Quick CD polish on best so far for this n
            if best_for_n is not None and time.monotonic() < phase_deadline - 20:
                best_for_n = cd_sweep_accurate(best_for_n, 10, phase_deadline - 15,
                                                seed=seed + idx * 7 + 100)
                best_M_for_n = float(fast_conv(best_for_n).max())

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: M={best_M_for_n:.8f} J={J_val:.6f} ({elapsed:.1f}s, "
                  f"inits={min(max_inits, len(inits))})", flush=True)
            tracker.update(best_for_n)

    tracker.report("Phase 1 complete")
    print(f"Phase 1 total: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Coordinate Descent Refinement ===
    p2_start = time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Coordinate Descent Refinement (budget=~300s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 120:
        n_best = tracker.best.size
        # Aggressive CD
        a_cd = cd_sweep_accurate(tracker.best, 50, min(time.monotonic() + 150, deadline - 100),
                                shuffle=True, seed=7777)
        tracker.update(a_cd)

        # Try from perturbed
        a_pert = tracker.best + rng.normal(0, 0.01, tracker.best.size)
        a_pert = np.maximum(a_pert, 0.001)
        a_pert = a_pert / a_pert.sum()
        a_cd2 = cd_sweep_accurate(a_pert, 30, min(time.monotonic() + 80, deadline - 80),
                                  shuffle=True, seed=8888)
        tracker.update(a_cd2)

    tracker.report("Phase 2 complete")
    print(f"Phase 2 total: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: LP Mirror Descent ===
    p3_start = time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (budget=~200s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 40:
        n_best = tracker.best.size
        est_lp_time = 0.004 * (n_best / 128) ** 2
        max_iters = max(10, min(100, int(150 / max(0.01, est_lp_time))))
        print(f"  n={n_best}, est LP time={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)

        a_lp = run_lp_mirror(tracker.best, max_iters, min(time.monotonic() + 150, deadline - 30))
        tracker.update(a_lp)

        # CD after LP
        if time.monotonic() < deadline - 20:
            a_cd3 = cd_sweep_accurate(a_lp, 20, deadline - 15, seed=9999)
            tracker.update(a_cd3)

    tracker.report("Phase 3 complete")
    print(f"Phase 3 total: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Final Polish ===
    p4_start = time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Final Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 5:
        # One more CD pass
        a_final = cd_sweep_accurate(tracker.best, 20, deadline - 3, seed=42424)
        tracker.update(a_final)

        # Try resampling to slightly different length
        if time.monotonic() < deadline - 10:
            for mult in [0.75, 1.5]:
                new_n = int(tracker.best.size * mult)
                if 256 <= new_n <= 2048:
                    a_ext = resample_seq(tracker.best, new_n)
                    a_ext = np.maximum(a_ext, 1e-6)
                    a_ext = a_ext / a_ext.sum()
                    a_ext_opt = cd_sweep_accurate(a_ext, 15, deadline - 5, seed=5555)
                    tracker.update(a_ext_opt)
                    if time.monotonic() > deadline - 10:
                        break

    tracker.report("Phase 4 complete")
    print(f"Phase 4 total: {time.monotonic()-p4_start:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
