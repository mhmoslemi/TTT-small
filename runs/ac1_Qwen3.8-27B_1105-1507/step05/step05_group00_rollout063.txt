    n = len(a)

    conv_len = 2*n - 1
    
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -300, 300)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    weights /= w_sum
    
    # h2[m] = sum_j a[j] * w[m-j] = conv(a, w)[m]
    N = 1
    while N < n + conv_len - 1:
        N <<= 1
    fa = rfft(a, N)
    fw = rfft(weights, N)
    h2 = irfft(fa * fw, N)[:n]

    a_rev = a[::-1]
    fa_rev = rfft(a_rev, N)
    h1_full = irfft(fa_rev * fw, N)
    h1 = h1_full[n-1:2*n-1]
    
    grad = 2.0 * (h1 + h2)
    return grad, weights, cmax

"""
Soft-max gradient optimization with Nesterov momentum, LP-guided corrections,
and adaptive temperature annealing for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def compute_softmax_weights(conv, tau):
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -300.0, 300.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    return weights, cmax

def compute_grad_from_weights(a, weights):
    """Compute grad[m] = 2 * sum_k weights[k] * (a[k-m] + a[m-k]) using FFT."""
    n = len(a)
    conv_len = 2 * n - 1
    
    N = 1
    while N < n + conv_len - 1:
        N <<= 1
    
    fa = rfft(a, N)
    fw = rfft(weights, N)
    
    # h2[m] = conv(a, w)[m] = sum_j a[j] * w[m-j]
    h2 = irfft(fa * fw, N)[:n]
    
    # h1[m] = sum_j a[j] * w[j+m] = conv(a_rev, w)[m+n-1]
    a_rev = a[::-1]
    fa_rev = rfft(a_rev, N)
    h1_full = irfft(fa_rev * fw, N)
    h1 = h1_full[n-1:2*n-1]
    
    return 2.0 * (h1 + h2)

def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s

def softmax_optimize(a, time_budget, tau_init=None, tau_min=None, verbose=False):
    """
    Optimize using soft-max gradient with Nesterov momentum and adaptive temperature.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = project_simplex(a)
    
    if tau_init is None:
        tau_init = 1.0 / n
    if tau_min is None:
        tau_min = 0.1 / n
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)  # momentum
    mu = 0.7  # momentum coefficient (lower = less momentum, more responsive)
    step = 0.5 / n  # initial step size
    
    total_iters = int(time_budget * 50)  # rough estimate
    
    for it in range(total_iters):
        if time.monotonic() - start > time_budget - 1:
            break
        
        # Adaptive temperature
        progress = it / max(total_iters, 1)
        tau = tau_init * (tau_min / tau_init) ** progress
        tau = max(tau, 1e-10 / n)
        
        conv = compute_conv(a)
        weights, cmax = compute_softmax_weights(conv, tau)
        grad = compute_grad_from_weights(a, weights)
        
        # Center gradient (project onto sum=const)
        grad_centered = grad - np.mean(grad)
        
        # Nesterov momentum
        v = mu * v - step * grad_centered
        a_new = a - v
        a_new = project_simplex(a_new)
        
        a = a_new
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        # Restart momentum if we're not making progress
        if it > 0 and it % 100 == 0:
            v = np.zeros(n)
        
        # Decay step size slowly
        step *= 0.9995
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    softmax it{it}: obj={best_obj:.6f}, tau={tau:.2e}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj

def solve_lp_direction(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def lp_alternate(a, time_budget, verbose=False):
    """LP alternating optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = project_simplex(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget - 5:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = solve_lp_direction(a, M, time_limit=5)
        if g is None:
            break
        
        # Try several blend factors
        best_t = 0
        best_t_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = project_simplex(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t
        
        if best_t > 0 and best_t_obj < best_obj - 1e-12:
            a_try = (1 - best_t) * a + best_t * g
            a = project_simplex(a_try)
            best_obj = best_t_obj
            best_a = a.copy()
            if verbose:
                print(f"    LP: obj={best_obj:.6f}, t={best_t:.2f}")
        else:
            # Try perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            a = best_a * np.exp(rng.normal(0, 0.05, n))
            a = project_simplex(a)
    
    return best_a, best_obj

def peak_equalize(a, time_budget, verbose=False):
    """
    Try to equalize the top peaks in the convolution.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = project_simplex(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > time_budget - 1:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Find all peaks within 5% of max
        threshold = cmax * 0.95
        active_peaks = np.where(conv >= threshold)[0]
        
        if len(active_peaks) == 0:
            break
        
        # For each active peak, compute the "pressure" on each coordinate
        pressure = np.zeros(n)
        for k in active_peaks:
            for m in range(n):
                km = k - m
                mk = m - k
                if 0 <= km < n:
                    pressure[m] += a[km]
                if 0 <= mk < n:
                    pressure[m] += a[mk]
        
        # Reduce pressure: move mass from high-pressure to low-pressure coordinates
        pressure_centered = pressure - np.mean(pressure)
        
        step = 0.02
        a_new = a - step * pressure_centered
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        
        step *= 0.99
    
    return best_a, best_obj

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

"""
Soft-max gradient optimization with Nesterov momentum, FFT-based gradient computation,
LP-guided corrections, peak equalization, and adaptive temperature annealing for
minimizing 2*n*max(conv(a,a))/(sum(a))^2. Explores multiple lengths and restarts.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def compute_softmax_weights(conv, tau):
    """Compute softmax weights over convolution values with temperature tau."""
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -300.0, 300.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    return weights, cmax


def compute_grad_from_weights(a, weights):
    """
    Compute grad[m] = 2 * sum_k weights[k] * (a[k-m] + a[m-k]) using FFT.
    This is the gradient of sum_k weights[k] * conv(a,a)[k] w.r.t. a.
    """
    n = len(a)
    conv_len = 2 * n - 1

    N = 1
    while N < n + conv_len - 1:
        N <<= 1

    fa = rfft(a, N)
    fw = rfft(weights, N)

    # h2[m] = conv(a, w)[m] = sum_j a[j] * w[m-j]
    h2 = irfft(fa * fw, N)[:n]

    # h1[m] = sum_j a[j] * w[j+m] = conv(a_rev, w)[m+n-1]
    a_rev = a[::-1]
    fa_rev = rfft(a_rev, N)
    h1_full = irfft(fa_rev * fw, N)
    h1 = h1_full[n - 1:2 * n - 1]

    # Clip tiny negative values from FFT roundoff
    h2 = np.maximum(h2, 0)
    h1 = np.maximum(h1, 0)

    return 2.0 * (h1 + h2)


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_optimize(a, time_budget, tau_init=None, tau_min=None, verbose=False,
                     momentum=0.7, step_scale=1.0):
    """
    Optimize using soft-max gradient with Nesterov momentum and adaptive temperature.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = project_simplex(a)

    if tau_init is None:
        tau_init = 1.0 / n
    if tau_min is None:
        tau_min = 0.05 / n

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    step = step_scale / n

    # Estimate total iterations
    # Each iteration: conv (FFT) + weights + grad (2 FFTs) + obj (FFT) ≈ 4 FFTs of size ~4n
    # For n=700, FFT size ~2048, each ~50us, so ~200us per iter → ~5000 iters/sec
    iters_per_sec = 3000  # conservative estimate
    total_iters = max(int(time_budget * iters_per_sec), 100)

    stall_count = 0
    last_improve_it = 0

    for it in range(total_iters):
        if time.monotonic() - start > time_budget - 0.5:
            break

        # Adaptive temperature (exponential decay)
        progress = it / max(total_iters, 1)
        tau = tau_init * (max(tau_min, 1e-12 / n) / tau_init) ** progress

        weights, cmax = compute_softmax_weights(conv, tau)
        grad = compute_grad_from_weights(a, weights)

        # Center gradient (project onto sum=const hyperplane)
        grad_centered = grad - np.mean(grad)

        # Adaptive step: scale by gradient magnitude
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm > 1e-15:
            adaptive_step = step * min(1.0, 0.1 / grad_norm)
        else:
            adaptive_step = 0.0

        v = momentum * v - adaptive_step * grad_centered
        a_new = a - v
        a_new = project_simplex(a_new)

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall_count = 0
            last_improve_it = it
        else:
            stall_count += 1

        # Restart momentum periodically or when stalled
        if stall_count > 200 or (it > 0 and it % 500 == 0):
            v = np.zeros(n)
            stall_count = 0

        # Slowly decay step
        step *= 0.9998

        # Speed up: skip some computations in later phases
        if it > total_iters * 0.7:
            # Use harder tau (closer to true max)
            tau = max(tau * 0.8, 0.001 / n)

        if verbose and it % 2000 == 0 and it > 0:
            elapsed = time.monotonic() - start
            print(f"      sm it{it}: obj={best_obj:.7f}, tau={tau:.2e}, step={step:.2e}")

def solve_lp_direction(a, M, time_limit=8):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

    n = len(a)

        if g is None:
            break

        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:

            obj_try = compute_obj(a_try)

            a = project_simplex(a)
            it = getattr(lp_alternate, '_it', 0) + 1
            lp_alternate._it = it

def peak_equalize(a, time_budget, verbose=False):
    """Equalize near-peak values in the convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = project_simplex(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(500):
        if time.monotonic() - start > time_budget - 1:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Find active peaks (within 10% of max)
        threshold = cmax * 0.9
        active = np.where(conv >= threshold)[0]

        if len(active) == 0:
            break

        # Compute pressure vector
        pressure = np.zeros(n)
        for k in active:
            for m in range(n):
                km = k - m
                if 0 <= km < n:
                    pressure[m] += a[km]
                mk = m - k
                if 0 <= mk < n:
                    pressure[m] += a[mk]

        pressure_centered = pressure - np.mean(pressure)
        pnorm = np.linalg.norm(pressure_centered)
        if pnorm < 1e-15:
            break

        a_new = a - step * pressure_centered / pnorm
        a_new = project_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.9

        if verbose and it % 100 == 0:
            print(f"      PE it{it}: obj={best_obj:.7f}")

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

def make_cosine_init(n, freq=1, alpha=0.3):
    """Create a cosine-modulated initialization."""
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * x / n)

def make_triangle_init(n, width=0.5):

    return a / np.sum(a)

    print(f"\n=== Phase 1: Soft-max optimization (n={n_parent}) ===")

    schedules = [
        (3.0 / n_parent, 0.05 / n_parent, 0.30),
        (1.0 / n_parent, 0.01 / n_parent, 0.25),
        (0.3 / n_parent, 0.005 / n_parent, 0.20),
        (0.1 / n_parent, 0.001 / n_parent, 0.15),
        (5.0 / n_parent, 0.1 / n_parent, 0.10),
    ]

    for si, (t_init, t_min, frac) in enumerate(schedules):
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        budget = min(remaining * frac, 300.0)
        print(f"  Schedule {si}: tau=[{t_init:.2e}, {t_min:.2e}], budget={budget:.0f}s")
        a_new, obj_new = softmax_optimize(a, budget, t_init, t_min, verbose=True)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  ** UPDATE: {best_score:.8f} **")
        print(f"  After schedule {si}: best={best_score:.8f}")

    # === Phase 2: LP corrections ===
    print(f"\n=== Phase 2: LP alternation ===")
    if time.monotonic() < deadline - 100 and n_parent <= 900:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            budget = min(remaining * 0.25, 150.0)
            a_new, obj_new = lp_alternate(a, budget, verbose=True)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  ** LP UPDATE: {best_score:.8f} **")
    else:
        print("  (skipped)")

    # === Phase 3: Peak equalization ===
    print(f"\n=== Phase 3: Peak equalization ===")
    if time.monotonic() < deadline - 80:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            budget = min(remaining * 0.15, 60.0)
            a_new, obj_new = peak_equalize(a, budget, verbose=True)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  ** PEAK UPDATE: {best_score:.8f} **")

    # === Phase 4: Cycling soft-max + LP ===
    print(f"\n=== Phase 4: Cycling ===")
    for cycle in range(6):
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        # Soft-max with progressively tighter tau
        tau_i = (0.5 / n_parent) * (0.5 ** cycle)
        tau_m = max(0.002 / n_parent, tau_i * 0.01)
        budget_sm = min(remaining * 0.3, 80.0)
        a_new, obj_new = softmax_optimize(a, budget_sm, tau_i, tau_m)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new.copy()

        # LP
        remaining = deadline - time.monotonic()
        if remaining > 20 and n_parent <= 800:
            budget_lp = min(remaining * 0.25, 40.0)
            a_new, obj_new = lp_alternate(a, budget_lp)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()

        # Peak equalize
        remaining = deadline - time.monotonic()
        if remaining > 15:
            budget_pe = min(remaining * 0.15, 20.0)
            a_new, obj_new = peak_equalize(a, budget_pe)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()

        print(f"  Cycle {cycle}: best={best_score:.8f}")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length exploration ===")
    n_base = len(best_seq)
    n_options = sorted(set([
        max(200, n_base - 150),
        max(200, n_base - 75),
        n_base,
        min(1500, n_base + 75),
        min(1500, n_base + 150),
        min(1500, n_base + 250),
    ]))

    for n_try in n_options:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if n_try == n_base:
            a_try = best_seq.copy()
            a_try = project_simplex(a_try)
        else:
            a_try = resample_sequence(best_seq, n_try)
            a_try = project_simplex(a_try)

        budget = min(remaining * 0.35, 50.0)
        a_new, obj_new = softmax_optimize(a_try, budget,
                                          tau_init=1.0 / n_try,
                                          tau_min=0.01 / n_try)
        # Quick LP if time
        remaining = deadline - time.monotonic()
        if remaining > 15 and n_try <= 800:
            a_new2, obj_new2 = lp_alternate(a_new, min(remaining * 0.4, 20.0))
            if obj_new2 < obj_new:
                a_new, obj_new = a_new2, obj_new2

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  ** LENGTH: n={n_try}, obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_new:.8f} (no improvement)")

    # === Phase 6: Alternative initializations ===
    print(f"\n=== Phase 6: Alternative inits ===")
    n_best = len(best_seq)

    inits = [
        ("cos1", make_cosine_init(n_best, 1, 0.3)),
        ("cos2", make_cosine_init(n_best, 2, 0.3)),
        ("cos3", make_cosine_init(n_best, 3, 0.2)),
        ("tri", make_triangle_init(n_best, 0.6)),
    ]

    for name, a_init in inits:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_init = project_simplex(a_init)
        obj_init = compute_obj(a_init)
        if obj_init < best_score:
            print(f"  {name}: init obj={obj_init:.7f} (better than best!)")

        budget = min(remaining * 0.3, 30.0)
        a_new, obj_new = softmax_optimize(a_init, budget,
                                          tau_init=1.0 / n_best,
                                          tau_min=0.01 / n_best)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  ** {name} UPDATE: {best_score:.8f} **")

    # === Phase 7: Final intensive polish with perturbations ===
    print(f"\n=== Phase 7: Final polish ===")
    n_final = len(best_seq)
    a = best_seq.copy().astype(np.float64)
    a = project_simplex(a)

    for r in range(10):
        if time.monotonic() > deadline - 25:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Main soft-max
        budget_sm = min(remaining * 0.3, 40.0)
        tau_i = max(0.1 / n_final, 0.3 / n_final * (0.7 ** r))
        tau_m = max(0.001 / n_final, tau_i * 0.01)
        a_new, obj_new = softmax_optimize(a, budget_sm, tau_i, tau_m)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new.copy()

        # LP
        remaining = deadline - time.monotonic()
        if remaining > 15 and n_final <= 800:
            budget_lp = min(remaining * 0.25, 25.0)
            a_new, obj_new = lp_alternate(a, budget_lp)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()

        # Perturbation restart
        remaining = deadline - time.monotonic()
        if remaining > 25:
            rng = np.random.default_rng(r * 104729 + 7)
            for p in range(3):
                if time.monotonic() > deadline - 20:
                    break
                scale = 0.02 + 0.015 * (r % 5)
                a_pert = a * np.exp(rng.normal(0, scale, n_final))
                a_pert = project_simplex(a_pert)

                budget_p = min(remaining * 0.2, 15.0)
                a_new, obj_new = softmax_optimize(a_pert, budget_p,
                                                  tau_init=0.5 / n_final,
                                                  tau_min=0.005 / n_final)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a = a_new.copy()
                    print(f"  Pert r{r}p{p}: {best_score:.8f}")

        print(f"  Round {r}: best={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
