import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Gauss-Seidel coordinate descent with O(n) incremental convolution updates
2. Projected subgradient global moves to escape local optima
3. Periodic LP mirror descent for large jumps
4. Multi-start with structured initializations (monotone, skewed shapes)
5. Multi-length exploration
Key: coordinate descent pass is O(n^2) with small constant, allowing 100k+ passes.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def paVA_project_decreasing(a):
    """
    Project a non-negative vector onto the set of decreasing sequences with same sum.
    Uses Pool Adjacent Violators Algorithm (isotonic regression with decreasing order).
    """
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    # We want the largest decreasing sequence b such that b <= a (pointwise) is not quite right.
    # Actually, PAVA for decreasing: we want the projection of a onto the cone of decreasing non-negative sequences.
    # Standard PAVA: reverse, do increasing PAVA, reverse back.
    ar = a[::-1].copy()
    # Increasing PAVA
    weights = np.ones(n)
    values = ar.copy()
    stack_vals = []
    stack_wts = []
    i = 0
    while i < n:
        cv = values[i]
        cw = weights[i]
        while stack_vals and stack_vals[-1] > cv:
            pv = stack_vals.pop()
            pw = stack_wts.pop()
            cv = (cv * cw + pv * pw) / (cw + pw)
            cw = cw + pw
        stack_vals.append(cv)
        stack_wts.append(cw)
        i += 1
    result = np.zeros(n)
    idx = 0
    for v, w in zip(stack_vals, stack_wts):
        result[idx:idx + int(w)] = v
        idx += int(w)
    return result[::-1]


def cd_pass(a, b, S, n, deadline, order, use_golden=True):
    """
    One pass of Gauss-Seidel coordinate descent.
    Modifies a and b in place.
    Returns (S, improved).
    """
    improved = False
    two_n = 2.0 * n
    conv_len = 2 * n - 1

    for i in order:
        if time.monotonic() > deadline:
            break

        ai = a[i]
        k_lo = i
        k_hi = i + n - 1  # always <= 2n-2 since i <= n-1

        # a_shift for k in [k_lo, k_hi] is a[0], a[1], ..., a[n-1]
        # b_new[k] = b[k] + 2*delta*a[k-i] + delta^2*[k==2i]

        # Generate candidates
        if ai < 1e-10:
            candidates = [0.001, 0.005, 0.01, 0.02, 0.05]
        else:
            candidates = [0.0, ai * 0.3, ai * 0.6, ai * 0.8,
                          ai * 1.2, ai * 1.5, ai * 2.0, ai * 3.0]

        best_val = two_n * b.max() / (S * S)
        best_new = ai

        # Outside max (computed once, doesn't depend on delta)
        if k_lo > 0:
            out_left = b[:k_lo].max()
        else:
            out_left = 0.0
        if k_hi < conv_len - 1:
            out_right = b[k_hi + 1:].max()
        else:
            out_right = 0.0
        outside_max = max(out_left, out_right)

        b_seg = b[k_lo:k_hi + 1]  # length n

        for cand in candidates:
            delta = cand - ai
            if abs(delta) < 1e-15:
                continue
            S_new = S + delta
            if S_new < 0.01:
                continue
            # inside = b_seg + 2*delta*a[0:n]
            inside = b_seg + 2.0 * delta * a
            if 2 * i >= k_lo and 2 * i <= k_hi:
                inside[2 * i - k_lo] += delta * delta
            new_max = max(outside_max, inside.max())
            val = two_n * new_max / (S_new * S_new)
            if val < best_val - 1e-14:
                best_val = val
                best_new = cand

        # Golden section refinement
        if best_new != ai and use_golden:
            if best_new > ai:
                lo = ai
                hi = best_new * 2.0 + 0.001
            else:
                lo = 0.0
                hi = ai * 1.5

            for _ in range(12):
                if hi - lo < 1e-12 * max(1.0, ai):
                    break
                gr = (lo + hi) / (1.0 + 5.0 ** 0.5)
                gl = (lo + hi) - (hi - lo) / (5.0 ** 0.5)

                delta_l = gl - ai
                S_l = S + delta_l
                if S_l >= 0.01:
                    inside_l = b_seg + 2.0 * delta_l * a
                    if 2 * i >= k_lo and 2 * i <= k_hi:
                        inside_l[2 * i - k_lo] += delta_l * delta_l
                    f_l = two_n * max(outside_max, inside_l.max()) / (S_l * S_l)
                else:
                    f_l = float('inf')

                delta_r = gr - ai
                S_r = S + delta_r
                if S_r >= 0.01:
                    inside_r = b_seg + 2.0 * delta_r * a
                    if 2 * i >= k_lo and 2 * i <= k_hi:
                        inside_r[2 * i - k_lo] += delta_r * delta_r
                    f_r = two_n * max(outside_max, inside_r.max()) / (S_r * S_r)
                else:
                    f_r = float('inf')

                if f_l < f_r:
                    hi = gr
                else:
                    lo = gl

            mid = (lo + hi) / 2.0
            delta_m = mid - ai
            S_m = S + delta_m
            if S_m >= 0.01:
                inside_m = b_seg + 2.0 * delta_m * a
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    inside_m[2 * i - k_lo] += delta_m * delta_m
                f_m = two_n * max(outside_max, inside_m.max()) / (S_m * S_m)
                if f_m < best_val - 1e-14:
                    best_new = mid
                    best_val = f_m

        if best_new != ai:
            delta = best_new - ai
            a[i] = best_new
            S += delta
            # Update b incrementally
            b[k_lo:k_hi + 1] += 2.0 * delta * a[0:n]
            # Wait: we already updated a[i], so a[0:n] now has the new value.
            # But the formula b[k] += 2*delta*a[k-i] uses the OLD a[k-i].
            # Let me fix: save old a before updating.
            # Actually, the formula is:
            # b_new[k] = b_old[k] + 2*delta*a_old[k-i] + delta^2*[k=2i]
            # So I need a_old[k-i]. Since I already updated a[i], let me handle the k=2i case separately.
            # For k != 2i: b[k] += 2*delta*a_old[k-i]
            # For k = 2i: b[2i] += 2*delta*a_old[i] + delta^2
            # Since a_old[i] = ai (the value before update):
            # b[k] += 2*delta*a[k-i] for k in [k_lo,k_hi], k != 2i
            # b[2i] += 2*delta*ai + delta^2
            # But I already added 2*delta*a (with new a[i]) to all of b[k_lo:k_hi+1].
            # The error is: at position k=2i, I added 2*delta*(ai+delta) instead of 2*delta*ai + delta^2.
            # 2*delta*(ai+delta) = 2*delta*ai + 2*delta^2
            # Correct: 2*delta*ai + delta^2
            # Error: +delta^2
            # So I need to subtract delta^2 from b[2i].
            if 2 * i >= k_lo and 2 * i <= k_hi:
                b[2 * i] -= delta * delta
            improved = True

    return S, improved


def run_cd(a, deadline, max_passes=100000, use_pava=False, verbose=False, vinterval=500):
    """Run coordinate descent until convergence or deadline."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 0.01:
        return a
    a = a / S
    S = 1.0
    n = len(a)
    b = fast_conv(a)

    best_a = a.copy()
    best_J = 2.0 * n * b.max() / S ** 2

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = np.random.permutation(n)
        S, improved = cd_pass(a, b, S, n, deadline, order)

        # Renormalize periodically
        if p % 50 == 49:
            S_actual = a.sum()
            if S_actual > 0.01:
                ratio = S_actual / S
                S = 1.0
                a = a / S_actual
                b = b / (S_actual * S_actual)

        # PAVA projection every 200 passes
        if use_pava and p % 200 == 199:
            a_proj = paVA_project_decreasing(a)
            s_proj = a_proj.sum()
            if s_proj > 0.01:
                a_proj = a_proj / s_proj
            b_proj = fast_conv(a_proj)
            J_proj = 2.0 * n * b_proj.max() / s_proj ** 2 if s_proj > 0.01 else float('inf')
            J_cur = 2.0 * n * b.max() / S ** 2
            if J_proj < J_cur:
                a = a_proj
                b = b_proj
                S = 1.0
                improved = True

        # Track best
        J = 2.0 * n * b.max() / S ** 2
        if J < best_J - 1e-13:
            best_J = J
            best_a = a.copy()
            if verbose and p % vinterval == vinterval - 1:
                print(f"      CD p={p}: J={J:.8f}", flush=True)

        if not improved and p > 20:
            break

    return best_a


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    conv_len = 2 * n - 1
    rows = np.empty(conv_len * n, dtype=np.int32)
    cols = np.empty(conv_len * n, dtype=np.int32)
    vals = np.empty(conv_len * n, dtype=np.float64)
    idx = 0
    for k in range(conv_len):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(conv_len, n))
    c = -np.ones(n)
    b_ub = np.full(conv_len, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, max_iters=200):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b_aa = fast_conv(a)
        M = float(b_aa.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Line search
        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)

        def obj(t):
            b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
            return b.max()

        from scipy.optimize import minimize_scalar
        result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-8})
        t_opt = float(result.x)

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)

        if J_new < best_J - 1e-13:
            best_a = a_new.copy()
            best_J = J_new
            a = a_new

    return best_a, best_J


def subgradient_step(a, step_size):
    """Take a projected subgradient step on the objective 2n*max(conv)/(sum)^2."""
    a = np.maximum(a, 0).astype(np.float64)
    S = a.sum()
    if S < 0.01:
        return a
    a = a / S
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    # Find active lags
    active = np.where(np.abs(b - M) < 1e-10 * M)[0]
    # Subgradient of max: average of 2*a shifted by active lags
    subgrad = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                subgrad[i] += 2.0 * a[j]
    subgrad /= len(active)
    # Subgradient of J = 2n*M/S^2: dJ/da[i] = 2n*(subgrad[i]*S - 2*M)/S^3
    # Descent direction
    direction = subgrad * S - 2.0 * M
    # Projected gradient step (project onto simplex {a>=0, sum=1})
    a_new = a - step_size * direction / (2.0 * n)
    # Project onto simplex
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 0.01:
        return a
    a_new = a_new / s
    return a_new


def make_init_decreasing(n, alpha, rng=None):

def make_init_increasing(n, alpha, rng=None):

def make_init_exponential(n, rate, rng=None):
    """Exponential decay initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * t / n)
    return a / a.sum()


def make_init_ramp(n, skew, rng=None):
    """Linear ramp with skew."""
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 + skew * (t - 0.5)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_init_bump_tail(n, bump_frac, tail_rate, rng=None):
    """Bump on one side, exponential tail on the other."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    m = int(n * bump_frac)
    a[m:] = np.exp(-tail_rate * (t[m:] - m) / n)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(make_init_decreasing(n, 1.0))
    inits.append(make_init_decreasing(n, 2.0))
    inits.append(make_init_decreasing(n, 0.5))
    inits.append(make_init_decreasing(n, 3.0))
    inits.append(make_init_increasing(n, 1.0))
    inits.append(make_init_increasing(n, 2.0))
    inits.append(make_init_exponential(n, 1.0))
    inits.append(make_init_exponential(n, 2.0))
    inits.append(make_init_exponential(n, 0.5))
    inits.append(make_init_exponential(n, 3.0))
    inits.append(make_init_ramp(n, 1.0))
    inits.append(make_init_ramp(n, -1.0))
    inits.append(make_init_ramp(n, 0.5))
    inits.append(make_init_ramp(n, -0.5))
    inits.append(make_init_bump_tail(n, 0.5, 2.0))
    inits.append(make_init_bump_tail(n, 0.3, 1.5))
    inits.append(make_init_bump_tail(n, 0.7, 3.0))
    # Random smoothed
    ksize = max(3, n // 16)
    kernel = np.exp(-0.5 * (np.arange(ksize) - ksize // 2) ** 2 / (ksize // 4))
    kernel /= kernel.sum()
    r = rng.uniform(0, 1, n)
    sm = np.convolve(r, kernel, mode='same')
    sm = np.maximum(sm, 0.01)
    inits.append(sm / sm.sum())
    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def perturb_and_refine(a, deadline, rng, noise_scale=0.05):
    """Perturb, then refine with coordinate descent."""
    a_pert = a + rng.normal(0, noise_scale, len(a))
    a_pert = np.maximum(a_pert, 0.0)
    s = a_pert.sum()
    if s < 0.01:
        return a
    a_pert = a_pert / s
    a_ref = run_cd(a_pert, deadline, max_passes=2000, verbose=False)
    return a_ref


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses fast coordinate descent as main workhorse, with LP and subgradient
    for global exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = make_init_decreasing(512, 1.5)
        update_best(parent)

    # === PHASE 1: Coordinate descent from parent (300s) ===
    p1_deadline = t_start + 300
    print(f"\n={'='*58}", flush=True)
    print(f"PHASE 1: CD from parent (n={parent.size}, 300s)", flush=True)
    print(f"={'='*58}", flush=True)

    a1 = run_cd(parent, p1_deadline, max_passes=50000, use_pava=False,
                verbose=True, vinterval=5000)
    update_best(a1)

    # Also try with PAVA
    a1p = run_cd(parent, min(p1_deadline + 50, deadline - 600), max_passes=30000,
                 use_pava=True, verbose=False)
    update_best(a1p)

    # === PHASE 2: Multi-start CD (same n, 250s) ===
    p2_deadline = min(t_start + 550, deadline - 500)
    print(f"\n={'='*58}", flush=True)
    print(f"PHASE 2: Multi-start CD (250s)", flush=True)
    print(f"={'='*58}", flush=True)

    n_best = best_seq.size if best_seq is not None else 512
    inits = generate_inits(n_best, rng)
    # Add perturbed best
    for ns in [0.03, 0.08, 0.15]:
        pert = np.maximum(best_seq + rng.normal(0, ns, n_best), 0.01)
        inits.append(pert / pert.sum())

    init_scores = [(compute_J(init), init) for init in inits]
    init_scores.sort(key=lambda x: x[0])

    for score_init, init in init_scores[:12]:
        if time.monotonic() > p2_deadline:
            break
        phase_dl = min(time.monotonic() + 25, p2_deadline)
        a_r = run_cd(init, phase_dl, max_passes=5000, verbose=False)
        update_best(a_r)

    # === PHASE 3: Multi-length exploration (250s) ===
    p3_deadline = min(t_start + 800, deadline - 400)
    print(f"\n={'='*58}", flush=True)
    print(f"PHASE 3: Multi-length exploration (250s)", flush=True)
    print(f"={'='*58}", flush=True)

    n_base = best_seq.size if best_seq is not None else 512
    target_lengths = sorted(set([
        int(n_base * 0.75), int(n_base * 1.25), int(n_base * 1.5),
        int(n_base * 2.0),
        384, 448, 512, 640, 768, 1024, 1280
    ]))
    target_lengths = [L for L in target_lengths if 128 <= L <= 2048 and L != n_base]

    for tgt_n in target_lengths[:8]:
        if time.monotonic() > p3_deadline:
            break
        a_ext = resample_seq(best_seq, tgt_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()
        phase_dl = min(time.monotonic() + 30, p3_deadline)
        a_r = run_cd(a_ext, phase_dl, max_passes=5000, verbose=False)
        update_best(a_r)
        # Also try decreasing init at this length
        a_dec = make_init_decreasing(tgt_n, 1.5)
        a_r2 = run_cd(a_dec, min(phase_dl + 20, p3_deadline), max_passes=3000, verbose=False)
        update_best(a_r2)

    # === PHASE 4: LP mirror descent on best (200s) ===
    p4_deadline = min(t_start + 1000, deadline - 300)
    print(f"\n={'='*58}", flush=True)
    print(f"PHASE 4: LP Mirror Descent (200s)", flush=True)
    print(f"={'='*58}", flush=True)

    a_lp, J_lp = lp_mirror_descent(best_seq, p4_deadline, max_iters=100)
    update_best(a_lp)
    # Polish LP result with CD
    a_lp_cd = run_cd(a_lp, min(p4_deadline + 100, deadline - 200),
                     max_passes=20000, verbose=False)
    update_best(a_lp_cd)

    # === PHASE 5: Subgradient + CD (200s) ===
    p5_deadline = min(t_start + 1200, deadline - 200)
    print(f"\n={'='*58}", flush=True)
    print(f"PHASE 5: Subgradient + CD (200s)", flush=True)
    print(f"={'='*58}", flush=True)

    a_cur = best_seq.copy()
    step = 0.01
    for sg_iter in range(50):
        if time.monotonic() > p5_deadline:
            break
        a_sg = subgradient_step(a_cur, step)
        # Short CD refinement
        a_ref = run_cd(a_sg, min(time.monotonic() + 10, p5_deadline),
                       max_passes=2000, verbose=False)
        update_best(a_ref)
        if compute_J(ref) if False else compute_J(a_ref) < compute_J(a_cur):
            a_cur = a_ref
        step *= 0.95

    # === PHASE 6: Perturbation + refine (150s) ===
    p6_deadline = min(t_start + 1350, deadline - 100)
    print(f"\n={'='*58}", flush=True)
    print(f"PHASE 6: Perturbation + Refine (150s)", flush=True)
    print(f"={'='*58}", flush=True)

    for ns in [0.02, 0.04, 0.06, 0.08, 0.1, 0.15]:
        if time.monotonic() > p6_deadline:
            break
        a_pert = perturb_and_refine(best_seq, min(time.monotonic() + 20, p6_deadline),
                                    rng, noise_scale=ns)
        update_best(a_pert)

    # === PHASE 7: Final intensive CD (100s) ===
    p7_deadline = deadline
    print(f"\n={'='*58}", flush=True)
    print(f"PHASE 7: Final CD polishing", flush=True)
    print(f"={'='*58}", flush=True)

    a_final = run_cd(best_seq, p7_deadline, max_passes=200000,
                     verbose=True, vinterval=10000)
    update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)

    print(f"\n={'='*58}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"={'='*58}", flush=True)

    return [float(x) for x in best_seq]
