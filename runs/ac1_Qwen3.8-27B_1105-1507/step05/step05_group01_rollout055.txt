import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def subgradient(a, k_star):
    """Compute subgradient of max(conv) at the peak lag k_star.
    Returns vector of length n."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] = 2.0 * a[j]
    return g

def project_simplex(v, total=1.0):
    """Project v onto {x >= 0, sum(x) = total}."""
    n = len(v)
    # Use the standard simplex projection algorithm
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - total))[0][-1]
    theta = (cssv[rho] - total) / (rho + 1.0)
    return np.maximum(v - theta, 0)

Combined LP + subgradient + peak-pair optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def subgradient_at_peak(a, k_star):
    """Subgradient of max(conv) at peak lag k_star."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] = 2.0 * a[j]
    return g

def project_onto_nonneg_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_solve_restricted(a, M, active_k, max_constraints=200):
    """Solve LP with only a subset of constraints.
    max sum(g) s.t. (a*g)(k) <= M for k in active_k, g >= 0
    """
    n = len(a)
    k_list = list(active_k)
    if len(k_list) == 0:
        return None, 0.0
    
    # Build constraint matrix A where A[k_idx, i] = a[k-i] if 0<=k-i<n else 0
    m = len(k_list)
    A = np.zeros((m, n))
    for idx, k in enumerate(k_list):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[idx, i] = a[j]
    
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))

    return None, 0.0

"""
Combined LP + smoothed-subgradient + peak-pair optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses softmax-smoothed gradients for stable convergence, constraint-generation LP,
and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_onto_nonneg_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum} using water-filling."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, tau=2.0):
    """Compute gradient of soft-max approximation to max(conv(a,a)).
    Uses softmax weighting over convolution values for smooth gradients.
    Returns (gradient_vector, soft_max_value, peak_lag).
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Softmax weights
    conv_shifted = conv - conv.max()
    p = np.exp(conv_shifted / tau)
    p = p / p.sum()
    
    # soft_max value
    smax = float(conv.max() + tau * np.log(p.sum()))
    
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # This is 2 * cross-correlation of p and a
    grad = np.zeros(n)
    for i in range(n):
        # sum over k where 0 <= k-i < n, i.e., i <= k <= i+n-1
        k_min = max(0, i)
        k_max = min(2 * n - 2, i + n - 1)
        if k_min <= k_max:
            indices = np.arange(k_min, k_max + 1)
            grad[i] = 2.0 * np.dot(p[indices], a[indices - i])
    
    # Peak lag
    k_star = int(np.argmax(conv))
    
    return grad, smax, k_star


def lp_solve_active(a, M, active_indices):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for k in active_indices, g >= 0."""
    n = len(a)
    m = len(active_indices)
    if m == 0:
        return None, 0.0
    
    # Build sparse constraint matrix
    A = np.zeros((m, n))
    for idx, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A[idx, i] = a[k - i]
    
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, 0.0


def lp_optimize(a, time_budget=100.0, max_constraints=150):
    """LP-based optimization with constraint generation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 15
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Select active constraints (top violators)
        sorted_k = np.argsort(conv)[::-1]
        num_active = min(max_constraints, 2 * n - 1)
        active = sorted_k[:num_active]
        
        g, S = lp_solve_active(a, M, active)
        
        if g is None or S <= 1.0 + 1e-7:
            # Try with more constraints
            if num_active < 2 * n - 1:
                num_active = min(num_active * 2, 2 * n - 1)
                active = sorted_k[:num_active]
                g, S = lp_solve_active(a, M, active)
                if g is None or S <= 1.0 + 1e-7:
                    stall += 1
                    if stall > patience:
                        break
                    continue
            else:
                stall += 1
                if stall > patience:
                    break
                continue
        
        # Line search with multiple steps
        improved = False
        for t in [0.5, 0.7, 1.0, 0.85, 0.3, 0.15, 0.95, 0.6]:
            a_try = (1.0 - t) * a + t * g / S
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
    
    return best_a, best_obj


def smoothed_gradient_descent(a, time_budget=80.0, tau_init=5.0):
    """Polished optimization using softmax-smoothed gradients with Nesterov acceleration."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    x = a.copy()
    y = a.copy()
    k = 0
    tau = tau_init
    stall = 0
    patience = 300
    lr_base = 0.03
    
    while time.monotonic() - start < time_budget:
        k += 1
        lr = lr_base / (1.0 + 0.003 * k)
        tau = max(0.5, tau_init * np.exp(-0.005 * k))
        
        # Compute smoothed gradient
        grad, smax, k_star = smoothed_gradient(y, tau)
        
        # Compute objective gradient (accounting for normalization)
        sum_y = np.sum(y)
        obj_grad = 2.0 * n * (grad * sum_y - 2.0 * smax) / (sum_y ** 3)
        
        # Gradient step with projection
        x_new = project_onto_nonneg_simplex(y - lr * obj_grad, 1.0)
        
        # Nesterov momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (k - 1) ** 2)) / 2.0
        beta = (t_k - 1.0) / t_k
        y_new = x_new + beta * (x_new - x)
        y_new = np.maximum(y_new, 0)
        s_new = np.sum(y_new)
        if s_new > 1e-10:
            y_new = y_new / s_new
        else:
            y_new = x_new.copy()
        
        x = x_new
        y = y_new
        
        # Evaluate every few steps
        if k % 3 == 0:
            obj = compute_obj(x)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = x.copy()
                stall = 0
            else:
                stall += 1
                if stall > patience:
                    # Anneal: perturb and restart
                    rng = np.random.default_rng(k)
                    scale = rng.uniform(0.01, 0.05)
                    x = best_a * np.exp(rng.normal(0, scale, n))
                    x = np.maximum(x, 0)
                    x = x / np.sum(x)
                    y = x.copy()
                    k = 0
                    tau = tau_init
                    stall = 0
                    patience = max(100, patience // 2)
    
    return best_a, best_obj

def spectral_flatten(a, time_budget=20.0):

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final SGD: {best_score:.6f}")
        
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic() - 20
            # Peak pair round
            pp_b = min(remaining * 0.3, 40.0)
            a_f, obj_f = peak_pair_direct(best_seq, time_budget=pp_b)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final PP: {best_score:.6f}")
        
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic() - 10
            # One more LP
            lp_b = min(remaining * 0.5, 50.0)
            a_f, obj_f = lp_optimize(best_seq, time_budget=lp_b)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LP2: {best_score:.6f}")
    
    # Final output
    n_final = len(best_seq)
    print(f"\n=== RESULT: n={n_final}, score={best_score:.6f} ===")
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Safety check
    try:
        val = evaluate_sequence(result)
        if val < np.inf:
            best_score = val
        else:
            result = [1.0 / 100.0] * 100
    except Exception:
        pass
    
    return result
