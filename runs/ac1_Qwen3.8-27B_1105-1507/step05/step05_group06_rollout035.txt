def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J_norm(a):
    """Compute J for a normalized to sum=1, a >= 0. Returns 2n*max(conv)."""
    b = fast_conv(a)
    return float(2 * len(a) * b.max())

def binomial_init(n):

            # binom(2k,k)/4^k = binom(2k-2,k-1)/4^(k-1) * (2k)(2k-1)/(4*k^2)
            # = prev * (2k-1)/(2k)

def diverse_inits(n, rng):

    inits.append(np.ones(n) / n)

    cw = np.cos(np.pi * t / 2) ** 1.5 + 0.05

def build_lp_A(n, a):
    """Build sparse constraint matrix for LP."""
    rows = np.empty((2 * n - 1) * n, dtype=np.int32)
    cols = np.empty((2 * n - 1) * n, dtype=np.int32)
    vals = np.empty((2 * n - 1) * n, dtype=np.float64)
    idx = 0
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        for k in range(k_start, k_end + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    A = build_lp_A(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

                            coeff = a[j]
                            if coeff > 1e-12:
                                d_opt = S * coeff - b[k]
                                # delta = (S - b[k]/coeff) ... let me derive
                                # f(delta) = 2n*(b[k]+2*delta*a[j])/(S+delta)^2
                                # df/ddelta = 0: 2*a[j]*(S+delta) - 2*(b[k]+2*delta*a[j]) = 0
                                # 2*a[j]*S + 2*delta*a[j] - 2*b[k] - 4*delta*a[j] = 0
                                # 2*a[j]*S - 2*b[k] - 2*delta*a[j] = 0
                                # delta = (a[j]*S - b[k]) / a[j] = S - b[k]/a[j]
                                d_opt = S - b[k] / coeff
                                cands.append(max(0.0, ai + d_opt))

            # Deduplicate and filter
            cands = list(set([round(c, 10) for c in cands]))
            cands = [c for c in cands if c >= 0 and c < 5.0]

            # Evaluate candidates
            best_val = 2.0 * n * M / (S * S)
            best_new = ai

            # Precompute outside
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            if k_lo > 0:
                outside_left = float(b[:k_lo].max())
            else:
                outside_left = 0.0
            if k_hi < 2 * n - 2:
                outside_right = float(b[k_hi + 1:].max())
            else:
                outside_right = 0.0
            outside = max(outside_left, outside_right)

            for new_ai in cands:
                new_ai = max(0.0, new_ai)
                delta = new_ai - ai
                if abs(delta) < 1e-14:
                    continue

                S_new = S + delta
                if S_new < 1e-10:
                    continue

                # Compute new max over affected range
                max_in = 0.0
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        nv = b[k] + 2.0 * delta * ai + delta * delta
                    else:
                        j = k - i
                        if 0 <= j < n:
                            nv = b[k] + 2.0 * delta * a[j]
                        else:
                            nv = b[k]
                    if nv > max_in:
                        max_in = nv

                new_max = max(outside, max_in)
                val = 2.0 * n * new_max / (S_new * S_new)

                if val < best_val - 1e-14:
                    best_val = val
                    best_new = new_ai

            if best_new != ai:
                delta = best_new - ai
                a[i] = best_new
                S += delta
                # Incremental b update
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += 2.0 * delta * ai + delta * delta
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * delta * a[j]
                M = float(b.max())
                improved = True

        # Renormalize
        S_final = float(a.sum())
        if S_final > 1e-10:
            a = a / S_final
            J = compute_J_norm(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def subgradient_polish(a, deadline, max_iters=2000, eta_start=0.01):
    """Subgradient method on the max convolution."""
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J_norm(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        k_star = int(np.argmax(b))

        # Subgradient of max_k b_k at k_star
        # db_k/da_i = 2*a[k-i] (if k-i valid, k != 2i) or 2*a_i (if k=2i)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if k_star == 2 * i:
                    grad[i] = 2.0 * a[i]
                else:
                    grad[i] = 2.0 * a[j]
            else:
                grad[i] = 0.0

        # Learning rate
        eta = eta_start / (1.0 + it * 0.01)

        # Update
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        J_new = compute_J_norm(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = eval_seq(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = binomial_init(512)
        update_best(parent)

    # === PHASE 1: LP Mirror Descent from multiple initializations ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent (budget=400s)", flush=True)
    print(f"{'='*60}", flush=True)

    n_parent = parent.size
    p1_end = t_start + 400

    # Try parent and binomial init
    init_list = []
    init_list.append(("parent", parent.copy()))
    init_list.append(("binomial", binomial_init(n_parent)))

    # Add a few diverse inits
    div_inits = diverse_inits(n_parent, rng)[:5]
    for i, di in enumerate(div_inits):
        init_list.append((f"div_{i}", di))

    # Sort by initial J
    init_scores = []
    for name, init in init_list:
        init = np.maximum(init, 0)
        s = init.sum()
        if s > 1e-10:
            init = init / s
            J_init = compute_J_norm(init)
            init_scores.append((J_init, name, init))
    init_scores.sort(key=lambda x: x[0])

    print(f"  Initializations by J: {[f'{n}={s:.4f}' for s, n, _ in init_scores]}", flush=True)

    # Run LP on top 3 initializations
    for rank, (J_init, name, init) in enumerate(init_scores[:3]):
        if time.monotonic() > p1_end - 50:
            break
        phase_dl = min(time.monotonic() + (200 if rank == 0 else 80), p1_end - 30)
        print(f"  LP from '{name}' (J_init={J_init:.6f}, budget={phase_dl - time.monotonic():.0f}s)...", flush=True)
        a_result, J_result = lp_mirror_descent(init, phase_dl, max_iters=3000, cd_every=60)
        update_best(a_result)
        # Quick CD polish
        if time.monotonic() < p1_end - 20:
            a_cd = coordinate_descent(a_result, min(time.monotonic() + 20, p1_end - 10), max_passes=5)
            update_best(a_cd)
        print(f"    Result: J={J_result:.6f}", flush=True)

    # === PHASE 2: Extended coordinate descent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Coordinate Descent (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)

    p2_end = min(t_start + 620, deadline - 300)
    if best_seq is not None:
        a_cd = coordinate_descent(best_seq, p2_end, max_passes=80)
        update_best(a_cd)
        print(f"  CD done: best J={best_score:.8f}", flush=True)

    # === PHASE 3: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration (budget=250s)", flush=True)
    print(f"{'='*60}", flush=True)

    p3_end = min(t_start + 870, deadline - 250)
    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = [int(n_base * m) for m in [0.75, 1.25, 1.5, 2.0]]
        target_lengths += [384, 512, 640, 768, 1024, 1280, 1536]
        target_lengths = sorted(set([L for L in target_lengths if 256 <= L <= 2048]))

        for tgt_n in target_lengths:
            if time.monotonic() > p3_end - 30:
                break
            if tgt_n == n_base:
                continue

            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-10)
            a_ext = a_ext / a_ext.sum()

            # Also try binomial at this length
            a_bin = binomial_init(tgt_n)

            # Pick the better one
            J_ext = compute_J_norm(a_ext)
            J_bin = compute_J_norm(a_bin)
            if J_bin < J_ext:
                a_start = a_bin
            else:
                a_start = a_ext

            phase_dl = min(time.monotonic() + 30, p3_end - 15)
            a_r, J_r = lp_mirror_descent(a_start, phase_dl, max_iters=500, cd_every=40)
            update_best(a_r)
            a_cd = coordinate_descent(a_r, min(phase_dl + 10, p3_end - 5), max_passes=3)
            update_best(a_cd)
            print(f"    n={tgt_n}: J={J_r:.6f}", flush=True)

    # === PHASE 4: More LP from best with perturbations ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: LP with Perturbations (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)

    p4_end = min(t_start + 1070, deadline - 200)
    if best_seq is not None:
        n_b = best_seq.size
        for pert in [0.03, 0.05, 0.08, 0.12]:
            if time.monotonic() > p4_end - 30:
                break
            a_pert = best_seq + rng.normal(0, pert, n_b)
            a_pert = np.maximum(a_pert, 1e-8)
            a_pert = a_pert / a_pert.sum()
            phase_dl = min(time.monotonic() + 40, p4_end - 10)
            a_r, J_r = lp_mirror_descent(a_pert, phase_dl, max_iters=500, cd_every=40)
            update_best(a_r)
            a_cd = coordinate_descent(a_r, min(phase_dl + 5, p4_end - 5), max_passes=3)
            update_best(a_cd)

    # === PHASE 5: Subgradient polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Subgradient Polish (budget=100s)", flush=True)
    print(f"{'='*60}", flush=True)

    p5_end = min(t_start + 1170, deadline - 150)
    if best_seq is not None:
        a_sg = subgradient_polish(best_seq, p5_end, max_iters=3000, eta_start=0.005)
        update_best(a_sg)
        a_cd = coordinate_descent(a_sg, min(p5_end + 10, deadline - 100), max_passes=5)
        update_best(a_cd)

    # === PHASE 6: Final intensive coordinate descent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final Coordinate Descent", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_final = coordinate_descent(best_seq, deadline, max_passes=100)
        update_best(a_final)

        # One more short LP pass
        if time.monotonic() < deadline - 30 and best_seq is not None:
            a_lp = lp_mirror_descent(best_seq, deadline - 5, max_iters=200, cd_every=30)[0]
            update_best(a_lp)
            a_cd = coordinate_descent(a_lp, deadline, max_passes=10)
            update_best(a_cd)

    # Fallback
    if best_seq is None:
        best_seq = binomial_init(512)
        update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
