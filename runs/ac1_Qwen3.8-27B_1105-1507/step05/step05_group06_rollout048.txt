def smooth_max_grad_fast(a, beta):
    """Compute smooth max value and gradient using FFT."""
    n = len(a)
    c = fast_conv(a)  # length 2n-1
    
    # Stable log-sum-exp
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    Z = exp_c.sum()
    w = exp_c / Z
    smooth_val = c_max + np.log(Z) / beta
    
    # Gradient: 2 * (w cross-correlate a)
    # h[i] = Σ_k w[k] * a[k-i] = (w * a_rev)[n-1+i]
    a_rev = a[::-1]
    # Linear convolution of w (length 2n-1) and a_rev (length n)
    conv_len = (2*n-1) + n - 1 = 3*n - 2
    size = 1
    while size < conv_len:
        size <<= 1
    fw = np.fft.rfft(w, size)
    far = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * far, size)[:conv_len]
    
    # h[i] = conv[n-1+i] for i = 0, ..., n-1
    h = conv[n-1 : 2*n-1]
    grad = 2.0 * h
    
    return smooth_val, grad
