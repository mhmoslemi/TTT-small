def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax(b, beta):
    """Numerically stable softmax."""
    shifted = beta * b - beta * b.max()
    exp_vals = np.exp(shifted)
    return exp_vals / exp_vals.sum()

def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    rows = np.empty((2 * n - 1) * n, dtype=np.int32)
    cols = np.empty((2 * n - 1) * n, dtype=np.int32)
    vals = np.empty((2 * n - 1) * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_mirror_descent(a, deadline, max_iters=100):

        bounds = [(0, None)] * n

                        options={'time_limit': 10, 'presolve': True})

            continue

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()

            break
