```python
import numpy as np
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def subgradient_max_conv(a, top_k=5):
    """Compute subgradient of max(conv(a,a)) using top-k peaks."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    
    # Find top-k positions
    k = min(top_k, len(b))
    top_indices = np.argsort(b)[-k:]
    
    # Weight by how close to max
    weights = b[top_indices] / b[top_indices].sum()
    
    g = np.zeros(n)
    for idx in range(k):
        ki = top_indices[idx]
        wi = weights[idx]
        # d(conv[ki])/da[i] = 2*a[ki-i] if 0 <= ki-i < n
        m = ki - np.arange(n)
        valid = (m >= 0) & (m < n)
        g[valid] += 2.0 * wi * a[m[valid]]
    
    return g, M, b
```

```python
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Momentum-based subgradient descent (vectorized, top-k peaks)
2. Exact 1D coordinate descent with precomputed convolution structure
3. Multi-length exploration with diverse starts and periodic restarts
Innovation: backtracking line search on subgradient direction for adaptive
step size, combined with "peak-targeting" that focuses on the top-3 peaks.

import numpy as np
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def get_subgradient(a, b, top_k=5):
    """Compute subgradient of max(conv) using top-k peak positions."""
    n = len(a)
    M = b.max()
    # Find top-k positions
    k = min(top_k, 2 * n - 1)
    # Use partial sort for efficiency
    if k < 2 * n - 1:
        # Get top k indices
        top_idx = np.argpartition(b, -k)[-k:]
    else:
        top_idx = np.arange(2 * n - 1)
    
    g = np.zeros(n)
    for ki in top_idx:
        i_lo = max(0, ki - n + 1)
        i_hi = min(n - 1, ki)
        if i_lo <= i_hi:
            # g[i] += 2 * a[ki - i] for i in [i_lo, i_hi]
            # a[ki - i] for i = i_lo, ..., i_hi gives a[ki-i_lo], ..., a[ki-i_hi]
            # which is a[i_lo:ki+1] in reverse... let me think
            # When i = i_lo: a[ki - i_lo]
            # When i = i_hi: a[ki - i_hi]
            # So the vector is a[ki-i_hi : ki-i_lo+1]
            g[i_lo:i_hi+1] += 2.0 * a[ki - i_hi : ki - i_lo + 1]
    
    g /= k
    return g, M

def momentum_subgradient(a, num_iters, deadline, top_k=5, eta_scale=1.0):
    """Momentum-based subgradient with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)  # velocity (momentum)
    beta = 0.9
    best_a = a.copy()
    b = fast_conv(a)
    best_M = b.max()
    eta = 0.05 * eta_scale
    
    for it in range(num_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        g, _ = get_subgradient(a, b, top_k)
        
        # Project gradient to tangent space (sum-zero)
        g = g - g.mean()
        
        # Nesterov momentum
        v = beta * v - eta * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s_new
        
        M_new = fast_conv(a_new).max()
        
        if M_new < best_M * (1 - 1e-10):
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive eta: decrease if no improvement for a while
        # Simple: eta *= 0.995 each step
        eta *= 0.999
    
    return best_a

def coord_update_fast(a, i, b_cache):
    """
    Find optimal a[i] given all other coordinates fixed.
    Uses golden section search on the 1D objective.
    Returns (optimal_x, new b for affected positions).
    """
    n = len(a)
    x = a[i]
    S = a.sum()
    
    # Precompute: for each k, the "base" contribution (without a[i])
    # b[k] = sum_j a[j]*a[k-j]
    # When we change a[i] to x', the new b[k] = b[k] - old_contrib + new_contrib
    # old_contrib from a[i]:
    #   if k-i is valid and k-i != i: 2*x*a[k-i]
    #   if k == 2i: x^2
    #   (note: when k-i==i, it's x^2, not 2*x*a[i])
    
    # Affected k range: from i to i+n-1 (clipped to [0, 2n-2])
    k_lo = i
    k_hi = min(2*n-2, i+n-1)
    
    # Base max (unaffected positions)
    if k_lo > 0 or k_hi < 2*n-2:
        mask = np.ones(2*n-1, dtype=bool)
        mask[k_lo:k_hi+1] = False
        if mask.any():
            base_max = b_cache[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0
    
    # For affected k, the new b[k] = base_k + f_k(x')
    # where base_k = b[k] - old contribution, f_k(x') = new contribution
    # For k != 2i: f_k(x') = 2*x'*a[k-i]
    # For k == 2i: f_k(x') = x'^2
    
    # Precompute coefficients
    k_range = np.arange(k_lo, k_hi+1)
    c_arr = np.zeros(len(k_range))  # constant term
    slope_arr = np.zeros(len(k_range))  # linear coefficient
    quad_arr = np.zeros(len(k_range))  # quadratic coefficient
    
    for idx, k in enumerate(k_range):
        j = k - i
        if 0 <= j < n:
            if k == 2*i:
                # old: x^2, new: x'^2
                c_arr[idx] = b_cache[k] - x*x
                slope_arr[idx] = 0.0
                quad_arr[idx] = 1.0
            else:
                # old: 2*x*a[j], new: 2*x'*a[j]
                c_arr[idx] = b_cache[k] - 2*x*a[j]
                slope_arr[idx] = 2*a[j]
                quad_arr[idx] = 0.0
        else:
            c_arr[idx] = b_cache[k]
            slope_arr[idx] = 0.0
            quad_arr[idx] = 0.0
    
    def obj_1d(xp):
        """Objective: max of new conv / (S - x + xp)^2"""
        new_S = S - x + xp
        if new_S <= 1e-10:
            return 1e10
        vals = c_arr + slope_arr * xp + quad_arr * xp * xp
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        return m / (new_S * new_S)
    
    # Golden section search
    lo = 0.0
    hi = max(x * 3, np.sqrt(b_cache.max()) + 0.5, 0.1)
    if hi < 0.01:
        hi = 0.01
    
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_1d(a_pt)
    fb = obj_1d(b_pt)
    
    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_1d(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_1d(b_pt)
    
    x_opt = (lo + hi) / 2.0
    
    # Check boundary
    if obj_1d(0.0) < obj_1d(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)

def run_coord_descent(a, max_sweeps, deadline, seed=0):
    """Coordinate descent with FFT-based convolution."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    
    b_cache = fast_conv(a)
    best_a = a.copy()
    best_M = b_cache.max()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = rng.permutation(n)
        sweep_changed = False
        
        b_cache = fast_conv(a)  # Refresh at start of sweep
        for i in order:
            x_opt = coord_update_fast(a, i, b_cache)
            if abs(x_opt - a[i]) > 1e-10:
                a[i] = x_opt
                sweep_changed = True
        
        # Renormalize
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        b_cache = fast_conv(a)
        M = b_cache.max()
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20:
            break
    
    return best_a

def resample(a, n_new):
    """Resample sequence to new length."""
    a = np.asarray(a, dtype=np.float64)
    if len(a) == n_new:
        return a.copy()
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, n_new)
    return np.maximum(np.interp(t_new, t_old, a), 0)

def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Cosine modulated
    for freq in [1, 2, 3]:
        for alpha in [0.2, 0.4, 0.6]:
            inits.append(np.maximum(1.0 + alpha * np.cos(2*np.pi*freq*t), 0.01))
    
    # Gaussian centered
    inits.append(np.exp(-4*(t - 0.5)**2) + 0.1)
    
    # U-shaped
    inits.append(0.3 + 0.7 * (2*t - 1)**2)
    
    # Random smooth
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    if parent is not None:
        p = resample(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))
    
    return inits
```

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov momentum subgradient with backtracking on top-k peaks
2. Vectorized coordinate descent with incremental b-tracking
3. Multi-length diverse exploration with periodic restarts
Innovation: backtracking line search for adaptive step size, peak-targeted
subgradient focusing on top-3 peaks for faster convergence.

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1} via clip + normalize (fast approximate)."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s


def get_subgradient_topk(a, b, top_k=3):
    """Compute subgradient of max(conv) using top-k peak positions. Vectorized."""
    n = len(a)
    k = min(top_k, 2 * n - 1)
    if k < 2 * n - 1:
        top_idx = np.argpartition(b, -k)[-k:]
    else:
        top_idx = np.arange(2 * n - 1)
    
    g = np.zeros(n)
    for ki in top_idx:
        i_lo = max(0, ki - n + 1)
        i_hi = min(n - 1, ki)
        if i_lo <= i_hi:
            g[i_lo:i_hi+1] += 2.0 * a[ki - i_hi : ki - i_lo + 1]
    
    g /= k
    return g


def nesterov_subgradient(a, num_iters, deadline, top_k=3, eta_init=0.08, beta=0.9):
    """Nesterov accelerated projected subgradient with backtracking."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    b = fast_conv(a)
    best_M = b.max()
    eta = eta_init
    no_improve = 0
    
    for it in range(num_iters):
        if time.monotonic() > deadline:
            break
        
        b_y = fast_conv(y)
        g, M_y = get_subgradient_topk(y, b_y, top_k)
        
        # Project gradient to tangent space
        g = g - g.mean()
        
        # Nesterov: v = beta*v - eta*g; y_new = a + v; project
        v = beta * v - eta * g
        a_trial = y + v
        a_trial = simplex_project(a_trial)
        
        M_trial = fast_conv(a_trial).max()
        
        # Backtracking: if worse, reduce eta
        if M_trial > best_M * (1 + 1e-6):
            eta *= 0.85
            v = beta * v
            a_trial = y + v
            a_trial = simplex_project(a_trial)
            M_trial = fast_conv(a_trial).max()
        
        if M_trial < best_M * (1 - 1e-9):
            best_M = M_trial
            best_a = a_trial.copy()
            no_improve = 0
            # Slight eta increase on improvement
            eta = min(eta * 1.02, eta_init * 2)
        else:
            no_improve += 1
        
        a = a_trial
        y = a + beta * (a - y + v)
        # Keep y in feasible set (approximately)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-10:
            y = y / sy
        
        # Periodic eta decay
        if it % 100 == 0:
            eta *= 0.95
        
        if no_improve > 200:
            break
    
    return best_a


def coord_update(a, i, b):
    """Find optimal a[i] via golden section. Returns optimal x >= 0."""
    n = len(a)
    x = a[i]
    S = a.sum()
    
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Base max (unaffected positions) - vectorized
    if k_lo > 0 or k_hi < 2 * n - 2:
        if k_lo > 0 and k_hi < 2 * n - 2:
            base_max = max(b[:k_lo].max() if k_lo > 0 else 0, b[k_hi+1:].max() if k_hi < 2*n-2 else 0)
        elif k_lo > 0:
            base_max = b[:k_lo].max()
        else:
            base_max = b[k_hi+1:].max()
    else:
        base_max = 0.0
    
    # Precompute coefficients for affected positions
    k_range = np.arange(k_lo, k_hi + 1)
    j_vals = k_range - i  # = k - i for each k in range
    
    c_arr = b[k_lo:k_hi+1].copy()
    slope_arr = np.zeros(len(k_range))
    quad_mask = (j_vals == i)  # where k == 2i
    
    valid = (j_vals >= 0) & (j_vals < n)
    
    # For non-quadratic positions: c_k = b[k] - 2*x*a[k-i], slope = 2*a[k-i]
    v_mask = valid & ~quad_mask
    if v_mask.any():
        c_arr[v_mask] -= 2 * x * a[j_vals[v_mask]]
        slope_arr[v_mask] = 2 * a[j_vals[v_mask]]
    
    # For quadratic position (k=2i): c_k = b[k] - x^2, slope=0, quad=1
    if quad_mask.any():
        c_arr[quad_mask] -= x * x
    
    def obj_1d(xp):
        new_S = S - x + xp
        if new_S <= 1e-12:
            return 1e10
        vals = c_arr + slope_arr * xp
        if quad_mask.any():
            vals = vals.copy()
            vals[quad_mask] += xp * xp
        m = max(base_max, vals.max())
        return m / (new_S * new_S)
    
    # Golden section
    lo = 0.0
    hi = max(x * 3.0, np.sqrt(b.max()) + 0.1, 0.05)
    
    gr = (np.sqrt(5) - 1) / 2

    fb = obj_1d(b_pt)

        if fa < fb:
            hi = b_pt

            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_1d(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_1d(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_1d(0.0) < obj_1d(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)
