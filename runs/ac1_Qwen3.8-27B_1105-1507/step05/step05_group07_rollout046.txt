"""
Optimized search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) momentum subgradient descent, (2) LP mirror descent with
constraint reduction, (3) adaptive coordinate descent, (4) multi-length

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()

    return a

def compute_subgradient(a, tol_frac=1e-3):
    """Compute subgradient of f(a) = max_k (a*a)[k]. Returns (grad, active_set, M)."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    if M < 1e-30:
        return np.zeros(n), np.array([n - 1]), 0.0
    tol = tol_frac * M
    active = np.where(b >= M - tol)[0]

    grad = np.zeros(n)
    for k in active:
        # d/d a[j] of sum_i a[i]*a[k-i] = a[k-j] (if valid) + a[j-k] (if valid)
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            j_vals = np.arange(lo1, hi1 + 1)
            idx = k - j_vals
            grad[lo1:hi1 + 1] += a[idx]

        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            j_vals = np.arange(lo2, hi2 + 1)
            idx = j_vals - k
            grad[lo2:hi2 + 1] += a[idx]

    return grad, active, M


def momentum_subgradient(a, max_iters, deadline, lr_init=0.2, momentum=0.9):
    """Momentum-based subgradient descent for min max_k(a*a)[k] s.t. sum(a)=1, a>=0."""
    a = normalize(a.copy())
    n = len(a)
    vel = np.zeros(n)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0
    t = time.monotonic()

    for it in range(max_iters):
        if t > deadline:
            break
        t = time.monotonic()

        grad, active, M = compute_subgradient(a, tol_frac=1e-4)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size: use O(1/sqrt(k))
        lr = lr_init / (1.0 + 0.05 * it)

        # Momentum update
        vel = momentum * vel - lr * grad / gnorm
        a_new = a + vel
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        running_sum += a_new
        count += 1

        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()

        a = a_new

    # Check averaged iterate
    if count > 10:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0.0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg = a_avg / s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return normalize(best_a)


def build_lp_matrix(n, a, active_only=True, tol_frac=0.01):
    """Build sparse constraint matrix for LP. Only include tight constraints if active_only."""
    b = fast_conv(a)
    M = b.max()

    if active_only:
        tol = tol_frac * M
        rows_all = np.where(b >= M - tol)[0]
    else:
        rows_all = np.arange(2 * n - 1)

    rows_list = []
    cols_list = []
    vals_list = []

    for k in rows_all:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(a[k - j])

    return csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n)), rows_all


def lp_best_response(a, M, active_only=True, tol_frac=0.01, time_limit=20):
    """LP: max sum(g) s.t. (a*g)[k] <= M for active k, g >= 0."""
    n = len(a)
    A, rows = build_lp_matrix(n, a, active_only=active_only, tol_frac=tol_frac)
    if A.nnz == 0:
        return None

    # Only keep rows that are actually present
    # Map back to b_ub
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, lr_init=0.3):
    """LP mirror descent: iteratively compute best response and mix."""
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    prev_time = 0.1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Try tight constraints first, fall back to all
        t0 = time.monotonic()
        g = lp_best_response(a, M, active_only=True, tol_frac=0.05,
                             time_limit=min(30, max(5, (deadline - t0) / 2)))
        if g is None:
            g = lp_best_response(a, M, active_only=False, tol_frac=1.0,
                                 time_limit=min(30, max(5, (deadline - t0) / 2)))
        lp_time = time.monotonic() - t0
        prev_time = 0.7 * prev_time + 0.3 * lp_time

        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search with geometric factors
        improved = False
        base_lr = lr_init * (0.95 ** (it // 10))
        for t_val in [base_lr, base_lr * 0.7, base_lr * 0.5, base_lr * 0.3,
                      base_lr * 0.1, 0.05, 0.02, 0.01]:
            a_trial = np.maximum((1 - t_val) * a + t_val * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

        # Check if enough time for another LP
        remaining = deadline - time.monotonic()
        if remaining < prev_time * 2.5:
            break

    return normalize(best_a)


def coord_descent_adaptive(a, max_sweeps, deadline, seed=0):
    """Coordinate descent with adaptive log-spaced grid and incremental conv updates."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s
    s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        sweep_improved = False

        for i in order:
            old = a[i]
            if old < 1e-12:
                candidates = np.array([0.0, 1e-5, 1e-4, 1e-3, 5e-3, 1e-2])
            else:
                # Log-spaced grid around current value
                ratios = np.array([0.0, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0])
                candidates = ratios * old
                candidates = candidates[~np.isclose(candidates, old, rtol=1e-4)]

            if len(candidates) == 0:
                continue

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_M = conv.max()
                new_s = s - old + x
                new_obj = 2.0 * n * new_M / (new_s * new_s) if new_s > 1e-10 else float('inf')

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if best_val != old:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s += delta
                sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

    return normalize(best_a)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = normalize(p)
        inits.append(p)
        inits.append(normalize(np.maximum(p + rng.normal(0, 0.03, n), 0)))
        inits.append(normalize(np.maximum(p + rng.normal(0, 0.1, n), 0)))
        # Cyclic shifts
        for shift in [n // 4, n // 3, n // 5]:
            inits.append(normalize(np.roll(p, shift)))

    inits.append(np.ones(n) / n)

    # Cosine patterns
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            inits.append(normalize(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01)))
            inits.append(normalize(np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t), 0.01)))

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            inits.append(normalize(np.maximum(np.exp(-((t - center) / width) ** 2) + 0.05, 0.01)))

    # U-shape (high at edges)
    inits.append(normalize(np.maximum(0.3 + 0.7 * (2 * t - 1) ** 2, 0.01)))
    inits.append(normalize(np.maximum(0.3 + 0.7 * np.abs(2 * t - 1), 0.01)))

    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(normalize(np.maximum(smooth + 0.05, 0.01)))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search to minimize J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: Diverse initializations + subgradient + coordinate descent
    Phase 2: LP mirror descent (main optimization)
    Phase 3: Fine coordinate descent polishing
    Phase 4: Multi-length exploration + final LP
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError, TypeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        tracker.update(parent)

    # === PHASE 1: Quick exploration with diverse inits ===
    p1_deadline = min(time.monotonic() + 250, deadline - 700)
    n_base = parent.size
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Exploration n={n_base}", flush=True)
    print(f"{'='*60}", flush=True)

    inits = generate_initializations(n_base, rng, parent)

    # Sort inits by quick J estimate, try best ones first
    init_scores = []
    for idx, a0 in enumerate(inits[:12]):
        a0 = normalize(a0)
        J_est = compute_J(a0)
        init_scores.append((J_est, idx, a0))

    init_scores.sort(key=lambda x: x[0])

    for J_est, idx, a0 in init_scores[:6]:
        if time.monotonic() > p1_deadline:
            break

        # Subgradient (short)
        a_sg = momentum_subgradient(a0, max_iters=500,
                                    deadline=min(time.monotonic() + 20, p1_deadline - 50))
        tracker.update(a_sg)

        # Coordinate descent
        remaining_cd = p1_deadline - time.monotonic() - 5
        if remaining_cd > 10:
            a_cd = coord_descent_adaptive(a_sg, max_sweeps=500,
                                          deadline=time.monotonic() + max(5, remaining_cd),
                                          seed=seed + idx)
            tracker.update(a_cd)

    tracker.report("Phase 1")

    # === PHASE 2: LP Mirror Descent (main optimization) ===
    p2_deadline = min(time.monotonic() + 550, deadline - 350)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent n={tracker.best.size}", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a_lp = lp_mirror_descent(tracker.best, max_iters=10000, deadline=p2_deadline)
        tracker.update(a_lp)

        # Second pass with tighter constraints
        if time.monotonic() < p2_deadline - 30:
            a_lp2 = lp_mirror_descent(a_lp, max_iters=10000, deadline=p2_deadline)
            tracker.update(a_lp2)

    tracker.report("Phase 2")

    # === PHASE 3: Fine coordinate descent ===
    p3_deadline = min(time.monotonic() + 250, deadline - 150)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Fine CD n={tracker.best.size}", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p3_deadline:
        # Multiple passes with different random seeds
        a_cur = tracker.best.copy()
        for pass_idx in range(3):
            if time.monotonic() > p3_deadline:
                break
            a_cd = coord_descent_adaptive(a_cur, max_sweeps=3000,
                                          deadline=p3_deadline,
                                          seed=seed + 1000 + pass_idx * 7)
            tracker.update(a_cd)
            a_cur = tracker.best.copy()

        # Subgradient polish
        if time.monotonic() < p3_deadline - 10:
            a_sg = momentum_subgradient(tracker.best, max_iters=2000,
                                        deadline=p3_deadline)
            tracker.update(a_sg)

    tracker.report("Phase 3")

    # === PHASE 4: Multi-length exploration ===
    p4_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p4_deadline - 60:
        base_n = tracker.best.size
        target_lengths = []
        for delta in [-64, -32, 32, 64, 128]:
            cand_n = base_n + delta
            if 128 <= cand_n <= 2048:
                target_lengths.append(cand_n)
        # Also try some standard sizes
        for sn in [384, 480, 512, 544, 640, 768, 1024]:
            if sn not in target_lengths and 128 <= sn <= 2048:
                target_lengths.append(sn)

        target_lengths = sorted(set(target_lengths))

        for new_n in target_lengths:
            if time.monotonic() > p4_deadline - 40:
                break

            a_ext = resample_seq(tracker.best, new_n)
            a_ext = normalize(a_ext)

            # Quick coordinate descent
            t_remaining = p4_deadline - time.monotonic() - 40
            if t_remaining < 20:
                break

            a_opt = coord_descent_adaptive(a_ext, max_sweeps=2000,
                                           deadline=time.monotonic() + min(t_remaining * 0.4, 60),
                                           seed=seed + 5000 + new_n)
            tracker.update(a_opt)

            # Quick LP if time allows
            if time.monotonic() < p4_deadline - 30:
                a_lp = lp_mirror_descent(a_opt, max_iters=200,
                                         deadline=min(time.monotonic() + 30, p4_deadline - 20))
                tracker.update(a_lp)

    tracker.report("Phase 4")

    # === FINAL: One more LP polish on the overall best ===
    if tracker.best is not None and time.monotonic() < deadline - 10:
        print(f"\n  Final LP polish...", flush=True)
        a_final = lp_mirror_descent(tracker.best, max_iters=500,
                                    deadline=deadline)
        tracker.update(a_final)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
