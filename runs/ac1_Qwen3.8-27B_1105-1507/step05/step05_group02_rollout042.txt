3. Smooth gradient descent (log-sum-exp) for robustness

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    n = len(a)

    a /= s

    best_a = a.copy()
    stall_count = 0

        if deadline is not None and time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            
            old_val = a[i]
            if old_val < 1e-10:
                continue

            # Golden section search in delta

            candidates.append(-old_val)

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with 1D golden-section search exploiting O(n) structure
2. LP mirror descent for global refinement
3. Smooth gradient descent (log-sum-exp) for multi-start initialization
4. Length exploration and perturbation restarts
Key insight: When only a[i] changes, b_k changes by 2*delta*a[k-i] for k in [i,i+n-1],

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

            # Compute M_out: max of b outside [i, i+n-1]
            if i == 0:
                M_out = b[n:].max() if n < 2 * n - 1 else -np.inf
            elif i + n - 1 >= 2 * n - 2:
                M_out = b[:i].max() if i > 0 else -np.inf
            else:
                lo_part = b[:i].max() if i > 0 else -np.inf
                hi_part = b[i + n:].max() if i + n < 2 * n - 1 else -np.inf
                M_out = max(lo_part, hi_part)

            aff_b = b[i:i + n]

            # Bound for delta
            lo = -old_val
            hi = min(15.0 * old_val, 0.5)  # cap to avoid extreme values

            if hi <= lo:
                continue

            # Coarse grid search (11 points)
            grid = np.linspace(lo, hi, 11)
            best_f = 2.0 * n * max(M_out, aff_b.max()) / 1.0  # current J (S=1)
            best_delta = 0.0
            for d in grid:
                f_val = eval_coord_delta(aff_b, a, M_out, d, n)
                if f_val < best_f:
                    best_f = f_val
                    best_delta = d

            # Golden section refinement
            if best_delta > lo and best_delta < hi:
                # Find bracket
                idx = int(10.0 * (best_delta - lo) / (hi - lo))
                L = grid[max(0, idx - 1)]
                R = grid[min(10, idx + 1)]
                if R - L < 1e-14:
                    L, R = best_delta - old_val * 0.1, best_delta + old_val * 0.1
                gr = (np.sqrt(5) - 1) / 2
                x1 = R - gr * (R - L)
                x2 = L + gr * (R - L)
                f1 = eval_coord_delta(aff_b, a, M_out, x1, n)
                f2 = eval_coord_delta(aff_b, a, M_out, x2, n)
                for _ in range(25):
                    if f1 < f2:
                        R = x2
                        x2 = x1
                        f2 = f1
                        x1 = R - gr * (R - L)
                        f1 = eval_coord_delta(aff_b, a, M_out, x1, n)
                    else:
                        L = x1
                        x1 = x2
                        f1 = f2
                        x2 = L + gr * (R - L)
                        f2 = eval_coord_delta(aff_b, a, M_out, x2, n)
                best_delta = 0.5 * (L + R)
                best_f = eval_coord_delta(aff_b, a, M_out, best_delta, n)

            # Apply update if beneficial
            if best_delta != 0.0 and best_f < 2.0 * n * max(M_out, aff_b.max()) * (1 - 1e-10):
                a[i] = max(old_val + best_delta, 0.0)
                S_new = a.sum()
                # Update b incrementally
                b[i:i + n] += 2.0 * best_delta * a  # a still has old value at i
                # Actually we need to be careful: a[i] was changed AFTER the b update
                # The formula b[i:i+n] += 2*delta*a_old works because a still has old_val at i
                # But we already set a[i] = old_val + best_delta above
                # So we need to use the OLD a for the slope
                # Let me fix: compute b update BEFORE changing a[i]

        # After each pass, recompute b from scratch to avoid error accumulation
        if (p + 1) % 20 == 0:
            S = a.sum()
            if S < 1e-10:
                break
            a /= S
            b = fast_conv(a)
            b = np.maximum(b, 0)

        # Check improvement
        b_check = fast_conv(a)
        M = b_check.max()
        if M < best_b * (1 - 1e-9):
            best_b = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break

        if verbose and p % 50 == 0:
            print(f"    CD pass {p}: M={M:.8f} J={2*n*M:.8f}", flush=True)

    return best_a


def coord_descent_v2(a, max_passes=1000, deadline=None, verbose=False):
    """Improved coordinate descent with correct incremental b updates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-10:
        return a
    a /= S
    S = 1.0

    b = fast_conv(a).copy()
    best_J = 2.0 * n * b.max() / (S * S)
    best_a = a.copy()
    stall = 0

    for p in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

        pass_changed = False

        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break

            old_val = a[i]
            if old_val < 1e-12:
                continue

            # M_out: max of b outside affected range [i, i+n-1]
            if i + n - 1 >= 2 * n - 2 and i == 0:
                M_out = -1e18
            else:
                parts = []
                if i > 0:
                    parts.append(b[:i].max())
                if i + n < 2 * n - 1:
                    parts.append(b[i + n:].max())
                M_out = max(parts) if parts else -1e18

            aff_b = b[i:i + n].copy()
            # Slopes: slope_j = 2 * a[j] for j = 0..n-1
            # (since b[i+j] changes by 2*delta*a[j])
            slopes = 2.0 * a  # length n

            # Bounds
            lo_d = -old_val
            hi_d = min(20.0 * old_val, 0.3)

            # Coarse evaluation (9 points)
            n_grid = 9
            grid = np.linspace(lo_d, hi_d, n_grid)
            cur_f = 2.0 * n * max(M_out, aff_b.max()) / (S * S)
            best_f = cur_f
            best_d = 0.0
            for d in grid:
                new_S = S + d
                if new_S < 1e-10:
                    continue
                new_max = (aff_b + d * slopes).max()
                if M_out > new_max:
                    new_max = M_out
                f = 2.0 * n * new_max / (new_S * new_S)
                if f < best_f:
                    best_f = f
                    best_d = d

            # Golden section refine
            if best_d > lo_d and best_d < hi_d:
                idx = int(n_grid * (best_d - lo_d) / (hi_d - lo_d))
                L = grid[max(0, idx - 1)]
                R = grid[min(n_grid - 1, idx + 1)]
                if R - L < 1e-14:
                    L = best_d - old_val * 0.1
                    R = best_d + old_val * 0.1
                gr_phi = (np.sqrt(5.0) - 1.0) / 2.0
                x1 = R - gr_phi * (R - L)
                x2 = L + gr_phi * (R - L)

                def _eval(d):
                    ns = S + d
                    if ns < 1e-10:
                        return 1e18
                    nm = (aff_b + d * slopes).max()
                    if M_out > nm:
                        nm = M_out
                    return 2.0 * n * nm / (ns * ns)

                f1 = _eval(x1)
                f2 = _eval(x2)
                for _ in range(20):
                    if f1 < f2:
                        R = x2
                        x2 = x1
                        f2 = f1
                        x1 = R - gr_phi * (R - L)
                        f1 = _eval(x1)
                    else:
                        L = x1
                        x1 = x2
                        f1 = f2
                        x2 = L + gr_phi * (R - L)
                        f2 = _eval(x2)
                best_d = 0.5 * (L + R)

            # Apply if improved
            new_val = old_val + best_d
            if new_val < 0:
                new_val = 0.0
            if new_val != old_val:
                # Update b BEFORE updating a[i] (slopes use old a[i])
                delta = new_val - old_val
                b[i:i + n] += delta * slopes
                a[i] = new_val
                S = a.sum()
                pass_changed = True

        # Periodic recomputation
        if (p + 1) % 30 == 0:
            a = np.maximum(a, 0)
            S = a.sum()
            if S < 1e-10:
                break
            a /= S
            S = 1.0
            b = fast_conv(a).copy()

        # Track best
        S_cur = a.sum()
        M_cur = b.max()
        J_cur = 2.0 * n * M_cur / (S_cur * S_cur)
        if J_cur < best_J * (1 - 1e-9):
            best_J = J_cur
            best_a = (a / a.sum()).copy()
            stall = 0
        else:
            stall += 1
            if not pass_changed:
                stall += 5
            if stall > 20:
                break

        if verbose and p % 100 == 0:
            print(f"    CD pass {p}: J={J_cur:.8f}", flush=True)

    return best_a / best_a.sum()


def smooth_grad_step(a, tau, lr):
    """One smooth gradient step for minimizing max(b) with sum(a)=1."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    a_new = np.maximum(a - lr * grad, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def run_smooth_gradient(a, deadline, max_iters=2000, tau_start=2.0, tau_end=100.0):
    """Smooth gradient descent with Nesterov."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    y = a.copy()
    v = np.zeros(n)
    beta = 0.9
    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = 0.3 * (0.01 / 0.3) ** frac

        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        grad = np.zeros(n)
        active = np.where(p > 1e-10)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * p[k] * y[k - hi:k - lo + 1][::-1]

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()

    return best_a


def build_A_sparse(n, a_old):
    """Build sparse matrix for LP: 2*conv(a_old, a)_k <= M + b_old_k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(2.0 * a_old[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(a,g)<=M, g>=0."""
    n = len(a)
    A = csr_matrix((2 * n - 1, n), dtype=np.float64)
    # Build A: A[k,j] = a[k-j] if valid
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=100):
    """LP mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=8)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-12:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 15:
                break
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
