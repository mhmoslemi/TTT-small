```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frequency-domain initialization (IFT/Gerchberg-Saxton style)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_lp_matrix(n):
    """Precompute the structure (row/col indices) of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)

def solve_lp(a, M, row_idx, col_idx, time_limit=5):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def line_search_analytic(a, g):
    """
    Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2.
    Both a and g must be normalized to sum 1.
    """
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val

def ift_initialize(n, iterations=200, rng=None, target_flat=True):
    """
    Initialize using IFT (Iterative Fourier Transform) / Gerchberg-Saxton style.
    Goal: find a >= 0 such that |DFT(a)| is as flat as possible.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    # Start with random phase
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    # Random initial phase
    phase = rng.uniform(0, 2 * np.pi, size // 2 + 1)
    # Constant magnitude
    mag = np.ones(size // 2 + 1)
    mag[0] = 0  # DC component handled separately
    
    a = np.zeros(n)
    for it in range(iterations):
        # Forward: build spectrum with current magnitude, random/updated phase
        spectrum = mag * np.exp(1j * phase)
        # Make it conjugate symmetric for real signal
        full_spec = np.zeros(size, dtype=complex)
        full_spec[:size//2+1] = spectrum
        for k in range(1, size//2):
            full_spec[size-k] = np.conj(spectrum[k])
        if size % 2 == 0:
            full_spec[size//2] = np.abs(spectrum[size//2])
        
        # Inverse DFT to get time signal
        a_full = np.fft.ifft(full_spec).real
        a = a_full[:n]
        
        # Project onto non-negative
        a = np.maximum(a, 0)
        
        # If all zero, restart
        if a.sum() < 1e-10:
            phase = rng.uniform(0, 2 * np.pi, size // 2 + 1)
            continue
        
        a = a / a.sum()
        
        # Forward DFT
        fa = np.fft.rfft(a, size)
        
        if target_flat:
            # Update magnitude to be flat
            new_mag = np.abs(fa)
            # Average magnitude
            target_mag = new_mag.mean()
            # Blend: keep some of current, push towards flat
            mag = 0.7 * target_mag * np.ones(size // 2 + 1) + 0.3 * new_mag
        else:
            mag = np.abs(fa)
        
        phase = np.angle(fa)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a

def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False, 
                          lp_time_limit=5, no_improve_limit=150):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx, time_limit=lp_time_limit)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > no_improve_limit:
            break

        if verbose and it % 500 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a

def ift_refine(a, iterations=100, mixing=0.5):
    """
    IFT refinement: alternate between time-domain (non-negativity) 
    and frequency-domain (flat spectrum) constraints, blended with current a.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    a_curr = a.copy()
    best_a = a.copy()
    best_M = compute_M(a)
    
    for it in range(iterations):
        # Forward DFT
        fa = np.fft.rfft(a_curr, size)
        
        # Compute target magnitude (flat)
        target_mag = np.full(size // 2 + 1, fa.mean().real if fa.mean().real > 0 else 1.0)
        # Actually, use the mean of |fa|
        mean_mag = np.abs(fa).mean()
        target_mag = np.full(size // 2 + 1, mean_mag)
        
        # Blend magnitude
        current_mag = np.abs(fa)
        phase = np.angle(fa)
        blended_mag = mixing * target_mag + (1 - mixing) * current_mag
        
        # Reconstruct
        fa_new = blended_mag * np.exp(1j * phase)
        # Make conjugate symmetric
        full_spec = np.zeros(size, dtype=complex)
        full_spec[:size//2+1] = fa_new
        for k in range(1, size//2):
            full_spec[size-k] = np.conj(fa_new[k])
        if size % 2 == 0:
            full_spec[size//2] = np.abs(fa_new[size//2])
        
        a_new = np.fft.ifft(full_spec).real[:n]
        a_new = np.maximum(a_new, 0)
        
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        M_new = compute_M(a_new)
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        
        a_curr = a_new
    
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_structured_init(n, rng, method='mixed'):
    """Generate diverse structured initializations."""
    if method == 'gaussian':
        x = np.arange(n)
        center = n / 2
        sigma = n / 4
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif method == 'raised_cosine':
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    elif method == 'triangular':
        x = np.arange(n)
        center = n / 2
        a = np.maximum(0, 1 - np.abs(x - center) / center)
    elif method == 'symmetric_random':
        half = n // 2
        h = rng.random(half + 1)
        if n % 2 == 0:
            a = np.concatenate([h[::-1], h])
        else:
            a = np.concatenate([h[::-1], h])
    elif method == 'bimodal':
        x = np.arange(n)
        c1 = n * 0.25
        c2 = n * 0.75
        sigma = n / 8
        a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    elif method == 'chirp_soft':
        # Softened chirp: |exp(i*pi*j^2/n)| -> always 1, so use amplitude modulation
        x = np.arange(n)
        # Use a window that has good autocorrelation properties
        a = np.ones(n) * 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    elif method == 'mixed':
        methods = ['gaussian', 'raised_cosine', 'triangular', 'symmetric_random', 'bimodal']
        m = methods[rng.integers(len(methods))]
        return generate_structured_init(n, rng, m)
    else:
        a = np.ones(n) / n
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. IFT (Iterative Fourier Transform) based diverse initialization
2. LP mirror descent with analytic line search (primary optimizer)
3. Hybrid LP-MD + IFT alternating for escape from local minima
4. Frequency-domain structured perturbations
5. Aggressive multi-length exploration
6. Final extended LP polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    total_len = len(a) + len(b) - 1
    size = 1
    while size < total_len:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:total_len]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    """Precompute the structure (row/col indices) of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx, time_limit=5):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """
    Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2.
    Both a and g must be normalized to sum 1.
    Uses ternary search on the convex function.
    """
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def ift_initialize(n, iterations=300, rng=None):
    """
    Initialize using IFT / Gerchberg-Saxton style.
    Alternates between time-domain (non-negativity) and frequency-domain (flat spectrum).
    """
    if rng is None:
        rng = np.random.default_rng()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    # Random initial phase
    phase = rng.uniform(0, 2 * np.pi, size // 2 + 1)
    mag = np.ones(size // 2 + 1)

    a = np.ones(n) / n
    for it in range(iterations):
        # Build spectrum with current magnitude and phase
        spectrum = mag * np.exp(1j * phase)
        full_spec = np.zeros(size, dtype=complex)
        full_spec[:size // 2 + 1] = spectrum
        for k in range(1, size // 2):
            full_spec[size - k] = np.conj(spectrum[k])
        if size % 2 == 0:
            full_spec[size // 2] = np.abs(spectrum[size // 2])

        # Inverse DFT
        a_full = np.fft.ifft(full_spec).real
        a = a_full[:n]

        # Project onto non-negative
        a = np.maximum(a, 0)

        if a.sum() < 1e-10:
            phase = rng.uniform(0, 2 * np.pi, size // 2 + 1)
            mag = np.ones(size // 2 + 1)
            continue

        a = a / a.sum()

        # Forward DFT
        fa = np.fft.rfft(a, size)

        # Target: flat magnitude spectrum
        mean_mag = np.abs(fa).mean()
        target_mag = np.full(size // 2 + 1, mean_mag)

        # Blend magnitude (gradually flatten)
        blend = 0.5 + 0.3 * (it / float(iterations))
        mag = blend * target_mag + (1 - blend) * np.abs(fa)

        phase = np.angle(fa)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def ift_refine(a, iterations=100, mixing=0.4):
    """
    IFT refinement: alternate between time-domain and frequency-domain constraints,
    blended to preserve the current solution's structure.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    a_curr = a.copy()
    best_a = a.copy()
    best_M = compute_M(a)

    for it in range(iterations):
        fa = np.fft.rfft(a_curr, size)

        mean_mag = np.abs(fa).mean()
        target_mag = np.full(size // 2 + 1, mean_mag)

        current_mag = np.abs(fa)
        phase = np.angle(fa)
        blended_mag = mixing * target_mag + (1 - mixing) * current_mag

        fa_new = blended_mag * np.exp(1j * phase)
        full_spec = np.zeros(size, dtype=complex)
        full_spec[:size // 2 + 1] = fa_new
        for k in range(1, size // 2):
            full_spec[size - k] = np.conj(fa_new[k])
        if size % 2 == 0:
            full_spec[size // 2] = np.abs(fa_new[size // 2])

        a_new = np.fft.ifft(full_spec).real[:n]
        a_new = np.maximum(a_new, 0)

        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        M_new = compute_M(a_new)
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()

        a_curr = a_new

    return best_a


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False,
                          lp_time_limit=5, no_improve_limit=200):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx, time_limit=lp_time_limit)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > no_improve_limit:
            break

        if verbose and it % 500 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def run_lp_mirror_descent_weighted(a, max_iters, deadline, row_idx, col_idx,
                                    peak_weight=2.0, lp_time_limit=5):
    """
    LP mirror descent with weighted constraints: peak positions get tighter bounds.
    This pushes harder on reducing peaks.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv_a = fast_conv(a)
        M = conv_a.max()
        
        # Weighted bounds: peaks get tighter bound
        threshold = M * 0.95
        b_ub = np.where(conv_a > threshold, M / peak_weight, M)

        vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
        A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        bounds = [(0, None)] * n
        g = None
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': lp_time_limit, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                if g.sum() < 1e-15:
                    g = None
        except Exception:
            g = None

        if g is None:
            # Fallback to standard LP
            g = solve_lp(a, M, row_idx, col_idx, time_limit=lp_time_limit)
            if g is None:
                break

        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-8:
            no_improve += 1
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 150:
            break

    return best_a


def smooth_gradient_step(a, tau):
    """Smooth max approximation gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad


def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0):
    """Smooth gradient descent with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = 0.3 * (0.05 / 0.3) ** frac

        grad = smooth_gradient_step(a, tau)
        v = 0.9 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_structured_init(n, rng, method):
    """Generate diverse structured initializations."""
    if method == 'gaussian':
        x = np.arange(n)
        center = n / 2
        sigma = n / 4
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif method == 'raised_cosine':
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    elif method == 'triangular':
        x = np.arange(n)
        center = n / 2
        a = np.maximum(0, 1 - np.abs(x - center) / center)
    elif method == 'symmetric_random':
        half = n // 2
        h = rng.random(half + 1)
        a = np.concatenate([h[::-1], h])
    elif method == 'bimodal':
        x = np.arange(n)
        c1 = n * 0.25
        c2 = n * 0.75
        sigma = n / 8
        a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    elif method == 'bell_shaped':
        x = np.arange(n)
        a = 1.0 / (1.0 + (2.0 * x / n - 1.0) ** 4)
    elif method == 'power_law':
        x = np.arange(1, n + 1, dtype=np.float64)
        center = (n + 1) / 2.0
        a = 1.0 / (1.0 + 0.5 * np.abs(x - center))
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining:
    1. Diverse initializations (IFT, structured, parent) with LP mirror descent
    2. Weighted LP variant for aggressive peak reduction
    3. Hybrid LP-MD + IFT alternating for escape from local minima
    4. Aggressive multi-length exploration with IFT
    5. Frequency-domain perturbations for diversity
    6. Final extended LP polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    n0 = parent.size
    row_idx0, col_idx0 = build_lp_matrix(n0)

    # =========================================================
    # PHASE 1: Diverse starts + LP mirror descent
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Diverse Starts + LP Mirror Descent (n={n0})", flush=True)
    print(f"{'='*60}", flush=True)

    phase1_deadline = min(t_start + budget_s * 0.28, deadline)

    # Start 1: Parent with standard LP
    a = parent.copy() / parent.sum()
    sub_dl = min(t_start + budget_s * 0.10, phase1_deadline)
    a_opt = run_lp_mirror_descent(a, 20000, sub_dl, row_idx0, col_idx0, verbose=False)
    tracker.update(a_opt, "(P1 parent LP)")

    # Start 2: IFT initialization + LP
    a_ift = ift_initialize(n0, iterations=400, rng=rng)
    tracker.update(a_ift, "(P1 IFT init)")
    sub_dl = min(time.monotonic() + budget_s * 0.06, phase1_deadline)
    a_opt = run_lp_mirror_descent(a_ift, 10000, sub_dl, row_idx0, col_idx0, verbose=False)
    tracker.update(a_opt, "(P1 IFT+LP)")

    # Start 3: Weighted LP on parent
    a = parent.copy() / parent.sum()
    sub_dl = min(time.monotonic() + budget_s * 0.05, phase1_deadline)
    a_opt = run_lp_mirror_descent_weighted(a, 10000, sub_dl, row_idx0, col_idx0,
                                           peak_weight=1.5, lp_time_limit=5)
    tracker.update(a_opt, "(P1 weighted LP)")

    # Start 4: Structured initializations + LP
    for method in ['gaussian', 'raised_cosine', 'bimodal', 'bell_shaped']:
        if time.monotonic() > phase1_deadline:
            break
        a_struct = generate_structured_init(n0, rng, method)
        sub_dl = min(time.monotonic() + budget_s * 0.02, phase1_deadline)
        a_opt = run_lp_mirror_descent(a_struct, 5000, sub_dl, row_idx0, col_idx0, verbose=False)
        tracker.update(a_opt, f"(P1 {method})")

    # =========================================================
    # PHASE 2: Hybrid LP-MD + IFT alternating (escape mechanism)
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Hybrid LP-MD + IFT Alternating", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_cur = tracker.best.size
        if n_cur != n0:
            row_idx_h, col_idx_h = build_lp_matrix(n_cur)
        else:
            row_idx_h, col_idx_h = row_idx0, col_idx0

        phase2_deadline = min(t_start + budget_s * 0.42, deadline)
        a = tracker.best.copy() / tracker.best.sum()

        for round_idx in range(25):
            if time.monotonic() > phase2_deadline:
                break

            # LP-MD step
            sub_dl = min(time.monotonic() + budget_s * 0.012, phase2_deadline)
            a = run_lp_mirror_descent(a, 3000, sub_dl, row_idx_h, col_idx_h, verbose=False)
            tracker.update(a, f"(P2 LP r={round_idx})")

            # IFT refinement with increasing mixing
            mixing = 0.2 + 0.4 * (round_idx / 25.0)
            a = ift_refine(a, iterations=60, mixing=mixing)
            tracker.update(a, f"(P2 IFT r={round_idx} mix={mixing:.2f})")

    # =========================================================
    # PHASE 3: Multi-length exploration
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        phase3_deadline = min(t_start + budget_s * 0.62, deadline)

        length_multipliers = [0.5, 0.6, 0.75, 1.25, 1.5, 1.75, 2.0, 2.5, 3.0, 0.4, 1.1]

        for mult in length_multipliers:
            if time.monotonic() > phase3_deadline:
                break

            n_trial = int(base_n * mult)
            if n_trial < 32 or n_trial > 8192:
                continue
            if n_trial == base_n:
                continue

            print(f"  Trying n={n_trial} (mult={mult:.2f})", flush=True)

            # Resampled parent
            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()

            # IFT initialization at this length
            a_ift_trial = ift_initialize(n_trial, iterations=200, rng=rng)

            r_idx, c_idx = build_lp_matrix(n_trial)

            # LP on resampled
            sub_dl = min(time.monotonic() + budget_s * 0.03, phase3_deadline)
            a_opt = run_lp_mirror_descent(a_init, 5000, sub_dl, r_idx, c_idx, verbose=False)
            tracker.update(a_opt, f"(P3 resample n={n_trial})")

            # LP on IFT
            if time.monotonic() < phase3_deadline:
                sub_dl = min(time.monotonic() + budget_s * 0.025, phase3_deadline)
                a_opt = run_lp_mirror_descent(a_ift_trial, 5000, sub_dl, r_idx, c_idx, verbose=False)
                tracker.update(a_opt, f"(P3 IFT n={n_trial})")

    # =========================================================
    # PHASE 4: Frequency-domain perturbation restarts
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Frequency-Domain Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_idx_b, c_idx_b = build_lp_matrix(n_best)
        temperature = 0.8
        phase4_deadline = min(t_start + budget_s * 0.78, deadline)

        size_b = 1
        while size_b < 2 * n_best - 1:
            size_b <<= 1

        for r in range(35):
            if time.monotonic() > phase4_deadline:
                break

            a = tracker.best.copy().astype(np.float64)

            # Small time-domain noise
            noise_scale = temperature * 0.005
            a = a + rng.normal(0, noise_scale, n_best)
            a = np.maximum(a, 1e-6)
            a = a / a.sum()

            # Frequency-domain phase perturbation
            fa = np.fft.rfft(a, size_b)
            phase_noise = rng.normal(0, 0.3 * temperature, size_b // 2 + 1)
            fa = fa * np.exp(1j * phase_noise)
            a = np.fft.ifft(fa).real[:n_best]
            a = np.maximum(a, 1e-6)
            s = a.sum()
            if s > 1e-10:
                a = a / s

            # Quick LP optimization
            sub_dl = min(time.monotonic() + budget_s * 0.012, phase4_deadline)
            a_opt = run_lp_mirror_descent(a, 3000, sub_dl, r_idx_b, c_idx_b, verbose=False)
            tracker.update(a_opt, f"(P4 r={r} T={temperature:.2f})")

            temperature *= 0.93

    # =========================================================
    # PHASE 5: Final extended polish
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)

        # Extended LP polish
        a = tracker.best.copy() / tracker.best.sum()
        sub_dl = min(t_start + budget_s * 0.93, deadline)
        a_opt = run_lp_mirror_descent(a, 50000, sub_dl, r_f, c_f, verbose=True,
                                      no_improve_limit=300)
        tracker.update(a_opt, "(P5 LP final)")

        # IFT polish
        if time.monotonic() < deadline:
            a_ift = ift_refine(a_opt, iterations=200, mixing=0.3)
            tracker.update(a_ift, "(P5 IFT final)")

            # LP after IFT
            if time.monotonic() < deadline:
                a_final = a_ift / a_ift.sum()
                a_opt2 = run_lp_mirror_descent(a_final, 20000, deadline, r_f, c_f, verbose=False)
                tracker.update(a_opt2, "(P5 LP after IFT)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
```
