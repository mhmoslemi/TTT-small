def block_coordinate_descent(a, block_size, n_sweeps, deadline):
    """Optimize blocks of coordinates jointly."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    rng = np.random.default_rng(42)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        # Random block order
        blocks = list(range(0, n, block_size))
        rng.shuffle(blocks)
        
        sweep_improved = False
        for start in blocks:
            end = min(start + block_size, n)
            x0 = a[start:end]
            
            def block_obj(x):
                a_trial = a.copy()
                a_trial[start:end] = np.maximum(x, 0)
                st = a_trial.sum()
                if st < 1e-10:
                    return 1e10
                a_trial = a_trial / st
                return fast_conv(a_trial).max()
            
            res = minimize(block_obj, x0, method='Nelder-Mead', 
                          options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-12})
            if res.fun < block_obj(x0) - 1e-14:
                a[start:end] = np.maximum(res.x, 0)
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 2
        if no_improve > 20:
            break
    
    return best_a

def spectral_flatten(a, n_iter, deadline):

    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()

        if time.monotonic() > deadline:
            break
        
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = np.fft.rfft(a, size)
        power = np.abs(fa) ** 2
        
        # We want power to be flat. Compute the "flatness error".
        # The mean power is fixed: mean(power) = sum(a^2) * size / n... 
        # Actually by Parseval: sum_k |fa[k]|^2 = size * sum_j a[j]^2
        # So mean power = size * sum(a^2) / (size//2 + 1)
        
        # Instead, let's just work with the time-domain peak.

        tau = 20 + 10 * (it / n_iter)
        grad = lse_gradient(a, tau)

        grad = grad - grad.mean()
        
        # Gradient descent step with adaptive learning rate
        lr = 0.01 / (1 + 0.01 * it)

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def diverse_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            inits.append(np.maximum(p + rng.normal(0, sigma, n), 0.01))
    
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.5, 1.0, n) / n)
    
    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))
    
    # Ramp
    inits.append(np.linspace(0.1, 1.0, n))
    inits.append(np.linspace(1.0, 0.1, n))
    
    # Concentrated
    center = n // 2
    for width in [n//4, n//8]:
        sig = np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
        inits.append(sig + 0.01)
    
    # Block structure
    block_n = n // 4
    for shift in range(4):
        arr = np.zeros(n)
        arr[shift*block_n:(shift+1)*block_n] = 1.0
        arr += 0.1
        inits.append(arr)
    
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def run_multi_scale_cd(a, deadline, base_n=128):
    """
    Multi-scale coordinate descent: work on a coarsened version first,
    then refine at full resolution.
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    
    # Coarsen
    coarse_n = min(base_n, n)
    if coarse_n < n:
        a_coarse = resample_seq(a, coarse_n)
        a_coarse = a_coarse / a_coarse.sum()
        
        # Optimize at coarse scale
        best_M = fast_conv(a_coarse).max()
        for sweep in range(50):
            if time.monotonic() > deadline:
                break
            for i in range(coarse_n):
                # Golden section for a[i]
                b = fast_conv(a_coarse)
                x_cur = a_coarse[i]
                S = a_coarse.sum()
                
                k_lo = i
                k_hi = min(2 * coarse_n - 2, i + coarse_n - 1)
                
                mask = np.ones(2 * coarse_n - 1, dtype=bool)
                mask[k_lo:k_hi+1] = False
                base_max = b[mask].max() if mask.any() else 0.0
                
                def obj(x_t):
                    new_b = b.copy()
                    for k in range(k_lo, k_hi+1):
                        j = k - i
                        if 0 <= j < coarse_n:
                            if k == 2*i:
                                new_b[k] = b[k] - x_cur**2 + x_t**2
                            else:
                                new_b[k] = b[k] - 2*x_cur*a_coarse[j] + 2*x_t*a_coarse[j]
                        else:
                            pass
                    ns = S - x_cur + x_t
                    if ns <= 0:
                        return 1e10
                    return new_b.max() / (ns*ns)
                
                lo, hi = 0.0, max(x_cur * 3, 0.1)
                gr = (np.sqrt(5) - 1) / 2
                a_pt = hi - gr * (hi - lo)
                b_pt = lo + gr * (hi - lo)
                fa, fb = obj(a_pt), obj(b_pt)
                for _ in range(40):
                    if fa < fb:
                        hi, b_pt, fb = b_pt, a_pt, fa
                        a_pt = hi - gr * (hi - lo)
                        fa = obj(a_pt)
                    else:
                        lo, a_pt, fa = a_pt, b_pt, fb
                        b_pt = lo + gr * (hi - lo)
                        fb = obj(b_pt)
                x_opt = (lo + hi) / 2
                if obj(0) < obj(x_opt):
                    x_opt = 0
                a_coarse[i] = max(0, x_opt)
            
            a_coarse = a_coarse / a_coarse.sum()
        
        # Upsample back
        a = resample_seq(a_coarse, n)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
    
    return a

    Phase 1: Multi-scale initialization + coarse optimization
    Phase 2: Peak-targeted local search + coordinate descent
    Phase 3: Gradient-based refinement (log-sum-exp)
    Phase 4: Block coordinate descent for final polish
    Phase 5: Extended length exploration
    """
    rng = np.random.default_mgr = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            parent = parent / parent.sum()
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(256) / 256
        tracker.update(parent)
    
    # === PHASE 1: Multi-scale optimization at various lengths ===
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-scale CD (budget=~400s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    lengths = [256, 384, 512, 640]
    time_alloc = {256: 60, 384: 100, 512: 140, 640: 80}
    
    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        t_alloc = time_alloc.get(n, 60)
        phase_dl = min(time.monotonic() + t_alloc, deadline - 350)
        
        if time.monotonic() >= phase_dl:
            continue
        
        t0 = time.monotonic()
        inits = diverse_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_for_n = None
        best_M_for_n = float('inf')

            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            
            # Multi-scale: coarsen, optimize, upsample
            a_ms = run_multi_scale_cd(a0, phase_dl - 10, base_n=min(64, n // 4))
            
            # Fine coordinate descent
            best_M = fast_conv(a_ms).max()
            for sweep in range(30):
                if time.monotonic() > phase_dl:
                    break
                for i in range(n):
                    b = fast_conv(a_ms)
                    x_cur = a_ms[i]
                    S = a_ms.sum()
                    k_lo = i
                    k_hi = min(2*n-2, i+n-1)
                    
                    mask = np.ones(2*n-1, dtype=bool)
                    mask[k_lo:k_hi+1] = False
                    base_max = b[mask].max() if mask.any() else 0.0
                    
                    def _obj(x_t):
                        new_max = base_max
                        for k in range(k_lo, k_hi+1):
                            j = k - i
                            if 0 <= j < n:
                                if k == 2*i:
                                    v = b[k] - x_cur**2 + x_t**2
                                else:
                                    v = b[k] - 2*x_cur*a_ms[j] + 2*x_t*a_ms[j]
                            else:
                                v = b[k]
                            if v > new_max:
                                new_max = v
                        ns = S - x_cur + x_t
                        if ns <= 0:
                            return 1e10
                        return new_max / (ns*ns)
                    
                    lo, hi = 0.0, max(x_cur * 2.5, 0.05)
                    gr = (np.sqrt(5) - 1) / 2
                    a_pt = hi - gr * (hi - lo)
                    b_pt = lo + gr * (hi - lo)
