"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Fast coordinate descent with paired (symmetric) updates
2. Simulated annealing with adaptive temperature to escape local minima
3. Projected subgradient polish using the structured gradient of self-convolution
4. Multi-length exploration including larger sequences (n up to 2048)
5. Diverse initializations exploiting the structure that max(conv) is typically
   at center lags for non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def conv_gradient(b, a, active_weights):
    """
    Compute gradient of weighted sum of active b[k] values.
    grad[i] = sum_k active_weights[k] * 2*a[k-i] (with zero padding)
    """
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = active_weights[k]
        if wk < 1e-15:
            continue
        # grad[i] += wk * 2 * a[k-i] for valid i
        # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # a[k-i] for i in [i_lo, i_hi]
            idx = k - np.arange(i_lo, i_hi + 1)
            grad[i_lo:i_hi + 1] += wk * 2.0 * a[idx]
    return grad


def smooth_max_grad(a, tau=50.0):
    """Gradient of (1/tau)*log(sum exp(tau*b[k])) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_t = np.exp(tau * (b - b_max))
    p = exp_t / exp_t.sum()
    return conv_gradient(b, a, p)


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def coordinate_update_fast(a, i, b):
    """
    Find optimal value for a[i] given current convolution b.
    Uses the fact that changing a[i] only affects b[k] for k in [i, i+n-1].
    Returns optimal x >= 0 (before renormalization).
    """
    n = len(a)
    x_cur = a[i]
    S = a.sum()

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # Base max outside affected range
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0

    # Precompute for affected lags
    arange = np.arange(k_lo, k_hi + 1)
    c_vals = np.zeros(len(arange))
    slopes = np.zeros(len(arange))
    quad_idx = -1

    for idx, k in enumerate(arange):
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            slopes[idx] = 0.0
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2 * x_cur * a[j]
                slopes[idx] = 2 * a[j]
            else:
                c_vals[idx] = b[k]
                slopes[idx] = 0.0

    def obj_fn(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 0:
            return 1e10
        vals = c_vals + slopes * x_test
        m = base_max
        for idx2 in range(len(vals)):
            v = vals[idx2]
            if v > m:
                m = v
        if quad_idx >= 0:
            qv = c_vals[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)

    lo = 0.0
    hi = max(x_cur * 3.0, 0.5, 0.1)
    f_mid = obj_fn(hi * 0.5)
    f_hi = obj_fn(hi)
    if f_hi < f_mid and hi < 5.0:
        hi *= 2.0

    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_fn(a_pt)
    fb = obj_fn(b_pt)

    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_fn(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_fn(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_fn(0.0) < obj_fn(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_cd(a, max_sweeps, deadline, seed=0, paired=False):
    """Coordinate descent with optional paired (symmetric) updates."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)

        b = fast_conv(a)
        sweep_improved = False

        for i in order:
            if paired and i < n // 2:
                j = n - 1 - i
                # Try paired update: change both a[i] and a[j]
                best_pair = (a[i], a[j])
                best_obj = None
                x_i = a[i]
                x_j = a[j]
                S = a.sum()
                # Try a few candidate values
                candidates_i = [0.0, x_i * 0.5, x_i * 1.5, x_i + 0.01, x_i - 0.01]
                candidates_j = [0.0, x_j * 0.5, x_j * 1.5, x_j + 0.01, x_j - 0.01]
                # Quick evaluation
                for ci in candidates_i[:3]:
                    for cj in candidates_j[:3]:
                        if ci < 0 or cj < 0:
                            continue
                        a[i] = ci
                        a[j] = cj
                        new_sum = S - x_i - x_j + ci + cj
                        if new_sum < 1e-10:
                            a[i] = x_i
                            a[j] = x_j
                            continue
                        a_norm = a / new_sum
                        b_test = fast_conv(a_norm)
                        m_test = b_test.max()
                        if best_obj is None or m_test < best_obj:
                            best_obj = m_test
                            best_pair = (ci, cj)
                a[i] = best_pair[0]
                a[j] = best_pair[1]
                if best_pair[0] != x_i or best_pair[1] != x_j:
                    sweep_improved = True
            else:
                if paired and i >= n // 2:
                    continue  # Already handled as pair
                x_opt = coordinate_update_fast(a, i, b)
                if abs(x_opt - a[i]) > 1e-12:
                    a[i] = x_opt
                    sweep_improved = True
                    b = None  # Invalidate cache

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 3
        if no_improve > 20:
            break

    return best_a


def run_sa_cd(a, deadline, seed=0, n_restarts=5):
    """Simulated annealing wrapping coordinate descent."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = project_simplex(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    n = len(a)

    # Temperature schedule
    t0 = time.monotonic()
    total_time = max(1.0, deadline - t0)

    for restart in range(n_restarts):
        if time.monotonic() > deadline:
            break
        frac = restart / max(1, n_restarts)
        # Temperature decreases: large perturbations early, small later
        temp = 0.15 * (1.0 - 0.7 * frac)

        # Add perturbation
        if restart > 0:
            a = best_a.copy()
            noise = rng.normal(0, temp, n)
            a = a + noise * a  # Multiplicative noise
            a = project_simplex(a)

        # Run CD from here
        phase_deadline = min(deadline, time.monotonic() + total_time / (n_restarts - restart + 1))
        max_sw = max(30, int(5000 / max(1, n // 64)))
        a_opt = run_cd(a, max_sw, phase_deadline, seed=seed + restart * 13)

        M = fast_conv(a_opt).max()
        if M < best_M:
            best_M = M
            best_a = a_opt.copy()

    return best_a


def gradient_polish(a, max_iters, deadline, tau=50.0):
    """Projected subgradient method with backtracking line search."""
    a = project_simplex(a.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    lr = 0.05

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad = smooth_max_grad(a, tau)
        # Project gradient to tangent space of simplex
        grad = grad - grad.mean()

        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(20):
            a_try = a - alpha * grad
            a_try = project_simplex(a_try)
            M_try = fast_conv(a_try).max()
            if M_try < best_M - 1e-15:
                a = a_try
                best_M = M_try
                best_a = a.copy()
                improved = True
                break
            alpha *= 0.5

        if not improved:
            lr *= 0.7
            if lr < 1e-5:
                break
        else:
            lr = min(lr * 1.1, 0.2)
            tau *= 1.5  # Increase smoothness parameter

    return best_a


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = parent.copy()
        if p.size != n:
            t_old = np.linspace(0, 1, p.size)
            p = np.interp(t, t_old, p)
        inits.append(np.maximum(p, 1e-6))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 1e-4))
        inits.append(np.maximum(p + rng.normal(0, 0.3, n), 1e-4))

    # Uniform
    inits.append(np.ones(n))

    # Linear ramp (increasing)
    inits.append(np.linspace(0.1, 1.0, n))
    inits.append(np.linspace(1.0, 0.1, n))

    # Quadratic
    inits.append(0.5 + 0.5 * (2 * t - 1) ** 2)
    inits.append(1.0 - 0.5 * (2 * t - 1) ** 2)

    # Cosine variations
    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.05))

    # Gaussian bump
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + 0.1)
    inits.append(np.exp(-8 * (t - 0.7) ** 2) + 0.1)

    # Power law
    inits.append((t + 0.01) ** 2)
    inits.append((1 - t + 0.01) ** 2)
    inits.append((t + 0.01) ** 0.5)

    # Smooth random
    ksize = max(3, n // 40)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.05))

    # Two-peak
    inits.append(np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.05)

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

        a_sa = run_sa_cd(tracker.best, sa_deadline, seed=seed + 100, n_restarts=8)
        tracker.update(a_sa)

        # Also try from a perturbed version
        a_pert = tracker.best + rng.normal(0, 0.2, tracker.best.size)
        a_pert = project_simplex(a_pert)
        sa_deadline2 = min(time.monotonic() + 100, deadline - 200)
        a_sa2 = run_sa_cd(a_pert, sa_deadline2, seed=seed + 200, n_restarts=5)
        tracker.update(a_sa2)

    tracker.report("Phase 2 complete")
    print(f"Phase 2 total: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Gradient polish ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Gradient Polish (budget=~150s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 100:
        gp_deadline = min(time.monotonic() + 80, deadline - 100)
        a_gp = gradient_polish(tracker.best, 5000, gp_deadline, tau=30.0)
        tracker.update(a_gp)

        # Second pass with different tau
        if time.monotonic() < deadline - 50:
            gp_deadline2 = min(time.monotonic() + 50, deadline - 50)
            a_gp2 = gradient_polish(tracker.best, 3000, gp_deadline2, tau=80.0)
            tracker.update(a_gp2)

    tracker.report("Phase 3 complete")
    print(f"Phase 3 total: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Final CD refinement ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final CD Refinement (budget=~200s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_best = tracker.best.size
        cd_deadline = min(time.monotonic() + 120, deadline - 30)
        max_sw = max(100, int(20000 / max(1, n_best // 64)))
        a_final = run_cd(tracker.best, max_sw, cd_deadline, seed=99999)
        tracker.update(a_final)

        # Paired CD
        if time.monotonic() < deadline - 10:
            cd_deadline2 = min(time.monotonic() + 60, deadline - 10)
            a_paired = run_cd(tracker.best, 100, cd_deadline2, seed=88888, paired=True)
            tracker.update(a_paired)

        # Gradient after CD
        if time.monotonic() < deadline - 5:
            a_gp3 = gradient_polish(tracker.best, 2000, deadline - 3, tau=50.0)
            tracker.update(a_gp3)

    tracker.report("Phase 4 complete")
    print(f"Phase 4 total: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Try larger lengths ===
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Larger Length Exploration (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 20:
        # Try doubling the best length
        for mult in [2, 3]:
            if time.monotonic() > deadline - 20:
                break
            new_n = tracker.best.size * mult
            if new_n > 4096 or new_n < 64:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            ext_deadline = min(time.monotonic() + 40, deadline - 15)
            max_sw = max(20, int(3000 / max(1, new_n // 64)))
            a_ext_opt = run_cd(a_ext, max_sw, ext_deadline, seed=55555 + mult)
            tracker.update(a_ext_opt)

    tracker.report("Phase 5 complete")
    print(f"Phase 5 total: {time.monotonic()-p5_start:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
