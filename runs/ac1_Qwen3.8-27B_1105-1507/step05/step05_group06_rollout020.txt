import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    b = fast_conv(a)
    return float(2 * len(a) * b.max())


def build_lp_matrix(n, a_vals):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a_vals[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_solve(a, M, n):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_multi(b_aa, b_ag, b_gg, n, t_max=2.0):
    """Find best t in [0, t_max] minimizing max((1-t)^2*b_aa + 2t(1-t)*b_ag + t^2*b_gg)."""
    def obj(t):
        if t < 0 or t > t_max:
            return float('inf')
        b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
        return b.max()
    
    # First try bounded minimize
    result = minimize_scalar(obj, bounds=(0.0, t_max), method='bounded',
                             options={'xatol': 1e-7})
    t_bounded = float(result.x)
    
    # Also check a few discrete points
    candidates = [t_bounded, 0.0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, t_max]
    candidates = [c for c in candidates if 0 <= c <= t_max]
    
    best_t = min(candidates, key=obj)
    return best_t


def enhanced_lp_mirror_descent(a, deadline, max_iters=5000):
    """
    Enhanced LP mirror descent with:
    - Multiple M values for diversity
    - Extrapolation (t > 1)
    - Adaptive restart
    - Periodic Frank-Wolfe steps
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    consecutive_failures = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        # Try multiple M values
        m_values = [M, M * 0.995, M * 0.99]
        candidates = []
        for m_val in m_values:
            g = lp_solve(a, m_val, n)
            if g is not None:
                g = g / g.sum()
                candidates.append(g)

        if not candidates:
            consecutive_failures += 1
            if consecutive_failures > 5:
                break
            # Try Frank-Wolfe step instead
            k_star = int(np.argmax(b_aa))
            sg = np.zeros(n)
            for i in range(n):
                v1 = a[k_star - i] if 0 <= k_star - i < n else 0
                v2 = a[i - k_star] if 0 <= i - k_star < n else 0
                sg[i] = v1 + v2
            min_idx = np.argmin(sg)
            g_fw = np.zeros(n)
            g_fw[min_idx] = 1.0
            candidates.append(g_fw)

        # Pick best candidate via line search
        best_t = 0
        best_obj = float('inf')
        for g in candidates:
            b_ag = cross_conv(a, g)
            b_gg = fast_conv(g)
            t_opt = line_search_multi(b_aa, b_ag, b_gg, n, t_max=1.5)
            
            # Check the new point
            a_trial = np.maximum((1 - t_opt) * a + t_opt * g, 0)
            s = a_trial.sum()
            if s < 0.01:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            
            if J_trial < best_obj:
                best_obj = J_trial
                best_t = t_opt
                best_g = g

        if best_obj < best_J - 1e-14:
            a_new = np.maximum((1 - best_t) * a + best_t * best_g, 0)
            a_new = a_new / a_new.sum()
            a = a_new
            best_a = a.copy()
            best_J = best_obj
            no_improve = 0
            consecutive_failures = 0
        else:
            no_improve += 1
            # Accept small improvements to keep moving
            if best_obj < best_J + 1e-10:
                a = np.maximum((1 - best_t * 0.5) * a + best_t * 0.5 * best_g, 0)
                a = a / a.sum()

        if no_improve > 40:
            # Restart with perturbation
            scale = 0.05 + 0.05 * (no_improve / 80.0)
            a = a + np.random.normal(0, scale, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0

        if no_improve > 100:
            # Larger restart
            a = np.maximum(a + np.random.normal(0, 0.15, n), 0.001)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_J

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive LP mirror descent with solver-based M tightening and non-monotone extrapolation
2. Exact 1D coordinate descent using analytical convolution structure

4. Multi-length exploration with interpolation-based initialization
5. Diversity-maintaining restarts with perturbation schedules

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2 with normalization."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    b = fast_conv(a)
    return float(2 * len(a) * b.max())


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP: (a*g)[k] for all k, j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_solve(a, M, time_limit=60):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search(a, g, b_aa, b_ag, b_gg, n, t_max=2.0):
    """Find best t in [0, t_max] minimizing max of interpolated convolution."""
    def obj(t):
        if t < 0 or t > t_max:
            return float('inf')
        b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
        return b.max()

    result = minimize_scalar(obj, bounds=(0.0, t_max), method='bounded',
                             options={'xatol': 1e-7})
    t_best = float(result.x)

    # Check discrete candidates too
    cands = [0.0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 1.75, t_max, t_best]
    cands = [c for c in cands if 0 <= c <= t_max]
    best = min(cands, key=obj)
    return best

    n = len(a)

    cd_counter = 0

        M = float(b_aa.max())

            g = lp_solve(a, M * frac, time_limit=30)
            if g is not None:

        if g is None:
            # Fallback: Frank-Wolfe step
            k_star = int(np.argmax(b_aa))
            sg = np.zeros(n)
            valid_lo = max(0, k_star - n + 1)
            valid_hi = min(n - 1, k_star)
            for i in range(n):
                val = 0.0
                if 0 <= k_star - i < n:
                    val += a[k_star - i]
                if 0 <= i - k_star < n:
                    val += a[i - k_star]
                sg[i] = val
            min_idx = int(np.argmin(sg))
            g = np.zeros(n)
            g[min_idx] = 1.0

        g = g / g.sum()

        # Line search
        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)
        t_opt = line_search(a, g, b_aa, b_ag, b_gg, n, t_max=1.8)

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 0.01:
            no_improve += 1
            if no_improve > 50:
                a = np.maximum(a + np.random.normal(0, 0.05, n), 0.001)
                a = a / a.sum()
                no_improve = 0
            continue

        a_new = a_new / s
        J_new = compute_J(a_new)

        if J_new < best_J - 1e-13:
            best_a = a_new.copy()
            best_J = J_new
            a = a_new
            no_improve = 0
        else:
            # Try half step
            t_half = t_opt * 0.5
            a_half = np.maximum((1 - t_half) * a + t_half * g, 0)
            s_half = a_half.sum()
            if s_half > 0.01:
                a_half = a_half / s_half
                J_half = compute_J(a_half)
                if J_half < best_J:
                    best_a = a_half.copy()
                    best_J = J_half
                    a = a_half
                    no_improve = 0
                    continue
            no_improve += 1

        # Periodic coordinate descent
        cd_counter += 1
        if cd_counter >= 30:
            cd_counter = 0
            a_cd = coordinate_descent_fast(a, min(time.monotonic() + 15, deadline), max_passes=3)
            J_cd = compute_J(a_cd)
            if J_cd < best_J:
                best_J = J_cd
                best_a = a_cd.copy()
            a = a_cd
            no_improve = 0

        # Restart logic
        if no_improve > 50:
            scale = 0.03 + 0.001 * no_improve
            a = np.maximum(a + np.random.normal(0, min(scale, 0.2), n), 0.001)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_J


def coordinate_descent_fast(a, deadline, max_passes=30):
    """
    Fast coordinate descent using analytical convolution update.
    For coordinate i: b_new[k] = b[k] + delta*(a[k-i]+a[i-k]) + delta^2*[k==2i]
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = np.random.permutation(n)
        b = fast_conv(a)
        S = a.sum()
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]

            # Influence: c[k] = a[k-i] (if valid) + a[i-k] (if valid)
            # Simplified: c[k] = 2*a[k-i] for k in [i, i+n-1] when k != 2i
            #              c[2i] = 2*a[i]
            # Actually the correct formula: c[k] = a[k-i]*(0<=k-i<n) + a[i-k]*(0<=i-k<n)
            # For k in [i, i+n-1]: c[k] = a[k-i] (since i-k < 0 for k > i, = 0 for k = i)
            # For k = i: c[i] = a[0] + a[0] = 2*a[0]
            # For k in [i-(n-1), i-1]: c[k] = a[i-k]
            # Wait, 0 <= i-k < n means i-n < k <= i, so k in [i-n+1, i]
            # For k in [i, i+n-1]: 0 <= k-i < n, so c[k] includes a[k-i]
            # For k in [i-n+1, i]: 0 <= i-k < n, so c[k] includes a[i-k]
            # At k = i: both conditions hold, c[i] = a[0] + a[0] = 2*a[0]

            # Build influence vector
            c = np.zeros(2 * n - 1)
            # Contribution from a[k-i]: valid for k in [i, i+n-1]
            k_lo1 = max(0, i)
            k_hi1 = min(2 * n - 2, i + n - 1)
            c[k_lo1:k_hi1 + 1] += a[k_lo1 - i:k_hi1 - i + 1]
            # Contribution from a[i-k]: valid for k in [max(0,i-n+1), i]
            k_lo2 = max(0, i - n + 1)
            k_hi2 = min(2 * n - 2, i)
            c[k_lo2:k_hi2 + 1] += a[i - k_hi2:i - k_lo2 + 1][::-1]

            # Generate candidates
            if ai < 1e-10:
                candidates = [0.0, 0.02, 0.05, 0.1, 0.15, 0.2]
            else:
                candidates = [0.0, ai * 0.25, ai * 0.5, ai * 0.75, ai * 1.25, ai * 1.5, ai * 2.0]
                # Filter out values too close to ai
                candidates = [c for c in candidates if abs(c - ai) > 1e-12]
                if not candidates:
                    continue

            best_cand_J = compute_J(a)
            best_cand = ai

            for cand in candidates:
                delta = cand - ai
                S_new = S + delta
                if S_new < 0.01:
                    continue

                # Compute new max
                b_new_max = 0.0
                for k in range(2 * n - 1):
                    val = b[k] + delta * c[k]
                    if k == 2 * i:
                        val += delta * delta
                    if val > b_new_max:
                        b_new_max = val

                J_cand = 2 * n * b_new_max / (S_new * S_new)
                if J_cand < best_cand_J:
                    best_cand_J = J_cand
                    best_cand = cand

            if best_cand != ai and best_cand_J < best_J - 1e-12:
                delta = best_cand - ai
                a[i] = best_cand
                improved = True
                # Update b incrementally
                for k in range(2 * n - 1):
                    b[k] += delta * c[k]
                    if k == 2 * i:
                        b[k] += delta * delta
                S = S + delta

        if not improved:
            break

        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()

    return best_a


def block_coordinate_descent(a, deadline, block_size=8, max_passes=10):
    """
    Block coordinate descent: optimize groups of coordinates together.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    n_blocks = (n + block_size - 1) // block_size

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        block_order = np.random.permutation(n_blocks)
        improved = False

        for bi in block_order:
            if time.monotonic() > deadline:
                break

            start = bi * block_size
            end = min(start + block_size, n)
            block = a[start:end].copy()
            block_sum = block.sum()

            # Try several perturbations of the block
            best_block = block.copy()
            best_block_J = compute_J(a)

            for trial in range(5):
                if trial == 0:
                    trial_block = block.copy()
                elif trial == 1:
                    trial_block = np.maximum(block * np.random.uniform(0.5, 1.5, end - start), 0)
                elif trial == 2:
                    idx = np.random.randint(end - start)
                    trial_block = block.copy()
                    trial_block[idx] *= np.random.uniform(0.3, 2.0)
                elif trial == 3:
                    trial_block = np.maximum(block + np.random.normal(0, 0.05, end - start), 0)
                else:
                    # Spread mass
                    trial_block = block.copy()
                    trial_block[start % (end - start):(end % (end - start)) + 1] *= 0.7
                    trial_block[(start + 3) % (end - start):(end + 3) % (end - start) + 1] *= 1.3

                trial_block = np.maximum(trial_block, 0)
                if trial_block.sum() < 0.001:
                    continue
                a_trial = a.copy()
                a_trial[start:end] = trial_block
                a_trial = a_trial / a_trial.sum()
                J_trial = compute_J(a_trial)

                if J_trial < best_block_J:
                    best_block_J = J_trial
                    best_block = trial_block.copy()

            if np.any(best_block != block):
                a[start:end] = best_block
                improved = True

        if not improved:
            break

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n) / n)

    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())

    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())

    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())

    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())

    dbl = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())

    for freq in [1, 2, 3, 4, 5, 7]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    for alpha in [0.5, 1.0, 1.5, 2.0]:
        pl = (t + 0.01) ** alpha
        inits.append(pl / pl.sum())

    # Sinusoidal with random phase
    for _ in range(3):
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(1, 5)
        sd = 1.0 + 0.4 * np.sin(2 * np.pi * freq * t + phase)
        sd = np.maximum(sd, 0.01)
        inits.append(sd / sd.sum())

    if parent is not None:
        p_resampled = resample_seq(parent, n)
        p_resampled = np.maximum(p_resampled, 0.001)
        inits.append(p_resampled / p_resampled.sum())
        inits.append(np.maximum(p_resampled + rng.normal(0, 0.1, n), 0.01))
        inits[-1] = inits[-1] / inits[-1].sum()
        inits.append(np.maximum(p_resampled + rng.normal(0, 0.05, n), 0.01))
        inits[-1] = inits[-1] / inits[-1].sum()

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Adaptive LP mirror descent from parent
    Phase 2: Coordinate descent + block CD polishing
    Phase 3: Multi-length exploration
    Phase 4: Diverse LP mirror descent
    Phase 5: Final intensive polishing
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                update_best(parent)
                print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent)

    n_base = parent.size

    # === PHASE 1: LP Mirror Descent from parent (400s) ===
    p1_deadline = t_start + 400
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Adaptive LP Mirror Descent (n={n_base}, budget=400s)", flush=True)
    print(f"{'='*60}", flush=True)

    a = parent.copy() / parent.sum()
    a_result, J_result = adaptive_lp_mirror_descent(a, min(p1_deadline, deadline - 300))
    update_best(a_result)
    print(f"  Phase 1 done: J={J_result:.8f}', flush=True)

    # === PHASE 2: Polishing (100s) ===
    p2_deadline = min(t_start + 530, deadline - 350)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Coordinate Descent + Block CD Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_cd = coordinate_descent_fast(best_seq, min(p2_deadline - 30, deadline - 300), max_passes=15)
        update_best(a_cd)

        a_bcd = block_coordinate_descent(a_cd, min(p2_deadline - 10, deadline - 280), block_size=8, max_passes=8)
        update_best(a_bcd)

        a_cd2 = coordinate_descent_fast(a_bcd, min(p2_deadline, deadline - 270), max_passes=10)
        update_best(a_cd2)
        print(f"  Phase 2 done: best J={best_score:.8f}", flush=True)

    # === PHASE 3: Multi-length exploration (200s) ===
    p3_deadline = min(t_start + 750, deadline - 250)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_best = best_seq.size
        target_lengths = sorted(set([
            max(64, int(n_best * 0.5)),
            max(64, int(n_best * 0.75)),
            n_best,
            int(n_best * 1.25),
            int(n_best * 1.5),
            int(n_best * 2),
            384, 512, 768, 1024, 1536
        ]))
        target_lengths = [L for L in target_lengths if 64 <= L <= 2560]

        per_length_budget = 15
        for tgt_n in target_lengths:
            if time.monotonic() > p3_deadline:
                break
            if tgt_n == n_best:
                continue

            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            phase_dl = min(time.monotonic() + per_length_budget, p3_deadline)
            a_r, J_r = adaptive_lp_mirror_descent(a_ext, phase_dl, max_iters=500)
            update_best(a_r)

            a_cd = coordinate_descent_fast(a_r, min(phase_dl + 5, p3_deadline), max_passes=5)
            update_best(a_cd)

        print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)

    # === PHASE 4: Diverse starts (200s) ===
    p4_deadline = min(t_start + 950, deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Diverse LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_best = best_seq.size
        inits = generate_inits(n_best, rng, parent=best_seq)
        init_scores = [(compute_J(inp), inp) for inp in inits]
        init_scores.sort(key=lambda x: x[0])

        per_init_budget = 18
        for sc, init in init_scores[:10]:
            if time.monotonic() > p4_deadline:
                break
            phase_dl = min(time.monotonic() + per_init_budget, p4_deadline)
            a_r, J_r = adaptive_lp_mirror_descent(init, phase_dl, max_iters=400)
            update_best(a_r)
            a_cd = coordinate_descent_fast(a_r, min(phase_dl + 5, p4_deadline), max_passes=3)
            update_best(a_cd)

        print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Final intensive polishing (remaining time) ===
    p5_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Intensive Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        # Another LP pass on the best
        a_final, J_final = adaptive_lp_mirror_descent(best_seq, min(t_start + 1400, deadline - 30), max_iters=2000)
        update_best(a_final)

        # Final CD
        a_cd = coordinate_descent_fast(a_final, min(p5_deadline - 10, deadline - 10), max_passes=20)
        update_best(a_cd)

        # Final block CD
        a_bc = block_coordinate_descent(a_cd, min(p5_deadline - 5, deadline - 5), block_size=6, max_passes=5)
        update_best(a_bc)

        # One more LP on the polished result
        if time.monotonic() < deadline - 30:
            a_final2, _ = adaptive_lp_mirror_descent(a_bc, min(p5_deadline - 5, deadline - 5), max_iters=500)
            update_best(a_final2)
            a_cd2 = coordinate_descent_fast(a_final2, deadline, max_passes=10)
            update_best(a_cd2)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
