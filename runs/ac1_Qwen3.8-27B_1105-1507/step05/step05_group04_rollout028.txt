```python
def peak_shaving(a, n_steps):
    """Directly reduce the peak of conv(a,a)."""
    for _ in range(n_steps):
        b = conv(a, a)
        peak_idx = np.argmax(b)
        peak_val = b[peak_idx]
        
        # Find which pairs contribute to the peak
        # b[peak_idx] = sum_i a[i] * a[peak_idx - i]
        # We want to decrease a[i] for i where a[i]*a[peak_idx-i] is large
        
        # Compute contributions
        contribs = np.zeros(len(a))
        for i in range(len(a)):
            j = peak_idx - i
            if 0 <= j < len(a):
                contribs[i] = a[i] * a[j]
        
        # Push down high contributors, push up low contributors
        # while maintaining sum(a) constant
        mean_contrib = contribs.mean()
        direction = -(contribs - mean_contrib)  # push down high, up low
        
        # Update
        a_new = a + 0.01 * direction
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum() * a.sum()
        
        if conv(a_new, a_new).max() < peak_val:
            a = a_new
    
    return a
```

```python
        # Sharper parent

        inits.append(p3 / p3.sum())

    inits.append(np.ones(n) / n)

    r = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')

def resample(a, new_n):
    """Resample sequence to new length."""
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(t_new, t_old, a), 1e-8)
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using adaptive smooth gradient
descent with FFT convolution, top-K softmax smoothing, Adam optimizer,
multi-phase search with diverse initializations, frequency-domain restarts,
and multi-length exploration.
"""

import numpy as np
import time


def self_conv(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = self_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau, top_K):
    """
    Compute gradient of J(a) using top-K smoothed max approximation.
    Returns (grad_J, actual_J, b_max_val).
    """
    n = len(a)
    b = self_conv(a)
    b_max = float(b.max())

    if top_K >= len(b):
        top_idx = np.arange(len(b))
        p = np.ones(len(top_idx)) / len(top_idx)
    else:
        top_idx = np.argpartition(b, -top_K)[-top_K:]
        b_top = b[top_idx]
        b_top_max = b_top.max()
        exp_b = np.exp(tau * (b_top - b_top_max))
        p = exp_b / exp_b.sum()

    grad_sm = np.zeros(n)
    for ki in range(len(top_idx)):
        w = p[ki]
        if w < 1e-15:
            continue
        k = int(top_idx[ki])
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        seg = a[k - j_max:k - j_min + 1][::-1]
        grad_sm[j_min:j_max + 1] += 2.0 * w * seg

    s = a.sum()
    s2 = s * s
    s3 = s2 * s

    grad_J = (2.0 * n / s2) * grad_sm - (4.0 * n * b_max / s3)
    actual_J = 2.0 * n * b_max / s2
    return grad_J, float(actual_J), b_max


class AdamOpt:
    def __init__(self, dims, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.t = 0
        self.m = np.zeros(dims)
        self.v = np.zeros(dims)

    def step(self, x, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1.0 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1.0 - self.beta2) * grad * grad
        m_hat = self.m / (1.0 - self.beta1 ** self.t)
        v_hat = self.v / (1.0 - self.beta2 ** self.t)
        return x - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

    def reset(self):
        self.t = 0
        self.m[:] = 0.0
        self.v[:] = 0.0


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        t_old = np.linspace(0, 1, len(parent))
        p = np.interp(t, t_old, parent)
        p = np.maximum(p, 1e-6)
        inits.append(p / p.sum())
        p2 = np.maximum(p + rng.normal(0, 0.1, n), 1e-6)
        inits.append(p2 / p2.sum())
        p3 = np.maximum(p ** 1.5 + 0.05, 1e-6)
        inits.append(p3 / p3.sum())
        p4 = np.maximum(p ** 0.7, 1e-6)
        inits.append(p4 / p4.sum())

    inits.append(np.ones(n) / n)

    c = np.maximum(0.5 * (1.0 + np.cos(2.0 * np.pi * t - np.pi)), 0.01)
    inits.append(c / c.sum())

    e = np.maximum(1.0 / (0.15 + 8.0 * t * (1.0 - t)), 0.01)
    inits.append(e / e.sum())

    g = np.exp(-25.0 * (t - 0.5) ** 2)
    inits.append(g / g.sum())

    for f in [3, 5, 8]:
        m = np.maximum(1.0 + 0.35 * np.cos(2.0 * np.pi * f * t), 0.05)
        inits.append(m / m.sum())

    ksize = max(5, n // 25)
    kernel = np.exp(-0.5 * (np.arange(-ksize, ksize + 1) / (ksize / 3.0)) ** 2)
    kernel /= kernel.sum()
    r = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    r = np.maximum(r, 0.01)
    inits.append(r / r.sum())

    spec = rng.chi(1, n // 2 + 1)
    sig = np.fft.irfft(spec, n)
    sig = np.abs(sig)
    sig = np.maximum(sig, 0.01)
    inits.append(sig / sig.sum())

    alt = np.where(np.arange(n) % 3 == 0, 1.0, 0.25)
    alt = np.convolve(alt, np.ones(9) / 9, mode='same')
    alt = np.maximum(alt, 0.01)
    inits.append(alt / alt.sum())

    # Bimodal
    bm = np.exp(-30 * (t - 0.3) ** 2) + np.exp(-30 * (t - 0.7) ** 2)
    bm = np.maximum(bm, 0.01)
    inits.append(bm / bm.sum())

    return inits


def optimize_with_gradient(a0, deadline, tau_start=3.0, tau_end=80.0,
                           K_start=30, K_end=150, lr=0.02,
                           restart_interval=300, verbose=False, label=""):
    """
    Run adaptive smooth gradient descent with Adam and periodic restarts.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()

    best_a = a.copy()
    best_J = compute_J(a)

    adam = AdamOpt(n, lr=lr)

    it = 0
    no_improve = 0
    restart_count = 0

    while time.monotonic() < deadline:
        it += 1

        progress = min(1.0, it / 8000.0)
        tau = tau_start + (tau_end - tau_start) * (progress ** 0.6)
        K = max(10, int(K_start + (K_end - K_start) * progress))

        grad_J, J_cur, _ = smooth_max_grad(a, tau, K)

        if J_cur < best_J - 1e-13:
            best_J = J_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        a_new = adam.step(a, grad_J)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a = best_a.copy()
            adam.reset()
            no_improve += 200
            continue
        a = a_new / s

        if no_improve >= restart_interval:
            restart_count += 1
            rng_r = np.random.default_rng(it * 2654435761 + restart_count * 7919)

            scale = min(0.6, 0.12 * (1 + 0.3 * restart_count))
            size_fft = 1
            while size_fft < 2 * n:
                size_fft <<= 1
            A = np.fft.rfft(a, size_fft)
            freqs = np.arange(len(A), dtype=np.float64)
            decay = np.exp(-freqs / (n * 0.4))
            filt = 1.0 + scale * decay * rng_r.normal(0, 1, len(A))
            A_new = A * filt
            a_try = np.fft.irfft(A_new, size_fft)[:n]
            a_try = np.maximum(a_try, 1e-6)
            a_try = a_try / a_try.sum()

            J_try = compute_J(a_try)
            if J_try < best_J:
                a = a_try.copy()
                best_a = a_try.copy()
                best_J = J_try
            else:
                a = a_try.copy()

            adam.reset()
            no_improve = 0

            if verbose and restart_count % 5 == 0:
                print(f"    {label}: Restart #{restart_count} iter={it} "
                      f"best={best_J:.8f} trial={J_try:.8f}", flush=True)

        if verbose and it % 30000 == 0:
            print(f"    {label}: it={it} J={best_J:.8f} tau={tau:.1f} "
                  f"K={K} rst={restart_count}", flush=True)

    return best_a, best_J


def resample_seq(a, new_n):
    """Resample 1D sequence to new length."""
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(t_new, t_old, a), 1e-8)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search:
    Phase 1 (180s): Diverse initialization + quick gradient
    Phase 2 (820s): Intensive gradient optimization of top candidates
    Phase 3 (250s): Multi-length exploration
    Phase 4 (remaining): Final polishing with sharp tau
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    rng = np.random.default_rng(seed)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            print(f"Parent: n={parent.size}, J={compute_J(parent):.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        print("No parent, using uniform default.", flush=True)

    best_overall = parent.copy()
    best_J_overall = compute_J(parent)
    print(f"Initial best: J={best_J_overall:.8f} n={best_overall.size}", flush=True)

    # === PHASE 1 ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: Diverse Initialization", flush=True)
    print(f"{'=' * 60}", flush=True)

    n_main = 512
    p1_deadline = time.monotonic() + 180.0

    inits = generate_inits(n_main, rng, parent)
    candidates = []

    for i, a0 in enumerate(inits):
        if time.monotonic() > p1_deadline:
            break
        a0 = np.maximum(a0, 1e-8)
        a0 = a0 / a0.sum()

        a_opt, J_opt = optimize_with_gradient(
            a0, min(time.monotonic() + 12, p1_deadline),
            tau_start=5, tau_end=30, K_start=20, K_end=80,
            lr=0.03, restart_interval=80, verbose=False, label=f"p1_{i}"
        )
        candidates.append((J_opt, a_opt))
        print(f"  init {i:2d}: J={J_opt:.8f}", flush=True)
        if J_opt < best_J_overall:
            best_J_overall = J_opt
            best_overall = a_opt.copy()
            print(f"  ** New best: J={best_J_overall:.8f}", flush=True)

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 best: J={candidates[0][0]:.8f}", flush=True)

    # === PHASE 2 ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 2: Intensive Optimization", flush=True)
    print(f"{'=' * 60}", flush=True)

    p2_deadline = time.monotonic() + 820.0
    n_top = min(5, len(candidates))
    time_per = max(60.0, (p2_deadline - time.monotonic()) / max(1, n_top))

    for ci in range(n_top):
        if time.monotonic() > p2_deadline:
            break
        J_init, a_init = candidates[ci]
        cand_dl = min(time.monotonic() + time_per, p2_deadline)

        print(f"\n  Cand {ci} (J_init={J_init:.8f}), budget={time_per:.0f}s", flush=True)

        a_best, J_best = optimize_with_gradient(
            a_init, cand_dl,
            tau_start=3, tau_end=120, K_start=20, K_end=200,
            lr=0.012, restart_interval=250, verbose=True, label=f"c{ci}"
        )

        if J_best < best_J_overall:
            best_J_overall = J_best
            best_overall = a_best.copy()
            print(f"  ** New overall best: J={best_J_overall:.8f} (cand {ci})",
                  flush=True)

    # Fresh random starts
    if time.monotonic() < p2_deadline - 30:
        fresh_dl = min(time.monotonic() + 120, p2_deadline)
        for fs in range(4):
            if time.monotonic() > fresh_dl:
                break
            rng2 = np.random.default_rng(seed + 7777 + fs * 13)
            a_f = np.maximum(rng2.uniform(0.05, 1.0, n_main), 1e-6)
            a_f = a_f / a_f.sum()

            a_opt, J_opt = optimize_with_gradient(
                a_f, min(time.monotonic() + 28, fresh_dl),
                tau_start=5, tau_end=60, K_start=30, K_end=100,
                lr=0.02, restart_interval=120, verbose=False, label=f"fr{fs}"
            )
            print(f"  fresh {fs}: J={J_opt:.8f}", flush=True)
            if J_opt < best_J_overall:
                best_J_overall = J_opt
                best_overall = a_opt.copy()
                print(f"  ** Fresh {fs} best: J={best_J_overall:.8f}", flush=True)

    # === PHASE 3 ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 3: Multi-Length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_deadline = min(time.monotonic() + 250.0, deadline - 50.0)
    n_best = best_overall.size

    n_list = sorted(set([
        n_best - 128, n_best - 64, n_best + 64,
        n_best + 128, n_best + 256, n_best
    ]))

    for nt in n_list:
        if time.monotonic() > p3_deadline:
            break
        if nt < 128 or nt > 2048:
            continue
        if nt == n_best:
            continue

        a_try = resample_seq(best_overall, nt)
        a_try = a_try / a_try.sum()

        a_opt, J_opt = optimize_with_gradient(
            a_try, min(time.monotonic() + 40, p3_deadline - 10),
            tau_start=5, tau_end=60, K_start=20, K_end=100,
            lr=0.02, restart_interval=150, verbose=False, label=f"n{nt}"
        )
        print(f"  n={nt}: J={J_opt:.8f}", flush=True)

        if J_opt < best_J_overall:
            best_J_overall = J_opt
            best_overall = a_opt.copy()
            print(f"  ** Best at n={nt}: J={best_J_overall:.8f}", flush=True)

    # === PHASE 4 ===
    remaining = deadline - time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Final Polish ({remaining:.0f}s remaining)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if remaining > 15:
        a_pol, J_pol = optimize_with_gradient(
            best_overall, deadline - 5,
            tau_start=30, tau_end=300, K_start=10, K_end=50,
            lr=0.004, restart_interval=600, verbose=True, label="pol"
        )
        if J_pol < best_J_overall:
            best_J_overall = J_pol
            best_overall = a_pol.copy()
            print(f"  ** Polish: J={best_J_overall:.8f}", flush=True)

        # One more pass with slightly different params
        if time.monotonic() < deadline - 10:
            a_pol2, J_pol2 = optimize_with_gradient(
                best_overall, deadline - 5,
                tau_start=15, tau_end=100, K_start=15, K_end=60,
                lr=0.008, restart_interval=400, verbose=False, label="pol2"
            )
            if J_pol2 < best_J_overall:
                best_J_overall = J_pol2
                best_overall = a_pol2.copy()
                print(f"  ** Polish2: J={best_J_overall:.8f}", flush=True)

    final_J = compute_J(best_overall)
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={final_J:.8f} n={best_overall.size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    best_overall = np.clip(best_overall, 0.0, 1000.0)
    if best_overall.sum() < 0.01:
        best_overall = np.ones(128) / 128.0

    return [float(x) for x in best_overall]
```
