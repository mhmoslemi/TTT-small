"""
Hybrid LP + Coordinate Descent optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses top-K constraint LP for direction finding, coordinate descent for polish,
and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz_topK(a, K=50):
    """Build Toeplitz matrix for only the top-K peak positions."""
    n = len(a)
    conv = compute_conv(a)
    # Get top K indices
    top_k = np.argsort(conv)[::-1][:K]
    m = len(top_k)
    A = np.zeros((m, n), dtype=np.float64)
    for idx, k in enumerate(top_k):
        for j in range(n):
            j2 = k - j
            if 0 <= j2 < n:
                A[idx, j] = a[j2]
    return A, top_k


def solve_lp_topK(a, M, K=50, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax for top-K k's, g >= 0."""
    n = len(a)
    A, top_k = build_toeplitz_topK(a, K)
    c = -np.ones(n)
    b_ub = np.full(len(top_k), M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0
