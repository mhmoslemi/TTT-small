Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B with smooth max continuation (multi-temperature)
2. LP-based mirror descent for verification and refinement
3. Multi-start with structured perturbations

Key innovation: combines second-order optimization (L-BFGS) with
first-order LP direction, using adaptive temperature scheduling.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
from scipy.signal import fftconvolve
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_and_grad(a, tau):
    """
    Compute smooth max of autocorrelation and its gradient.
    Returns (smooth_max_value, gradient).
    Assumes a >= 0 (no normalization).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    log_sum_exp = max_shifted + np.log(exp_shifted.sum())
    smooth_val = b_max + log_sum_exp / tau
    
    # Probability weights
    p = exp_shifted / exp_shifted.sum()
    
    # Gradient: grad[m] = 2 * sum_j a[j] * p[m+j]
    # Using FFT: (p ⋆ a)[m] = ifft(fft(p) * conj(fft(a)))[m]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conj(fa), N)
    grad = 2.0 * h[:n]
    
    return smooth_val, grad, p, b

def objective_and_grad(a, tau):
    """
    Objective: J(a) = 2n * smooth_max(a*a) / (sum a)^2
    Gradient computed analytically.
    """
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(a)
    
    n = len(a)
    smooth_val, grad_b, p, b = smooth_max_and_grad(a, tau)
    
    # J = 2n * smooth_val / s^2
    J = 2.0 * n * smooth_val / (s * s)

    #          = 2n * (grad_b[m] * s - 2 * smooth_val) / s^3
    grad_J = 2.0 * n * (grad_b * s - 2.0 * smooth_val) / (s * s * s)
    
    return J, grad_J

def linalg_smooth_max(a, tau_list, max_iter_per_tau, deadline):
    """
    L-BFGS optimization with continuation in tau.
    Starts with small tau (smooth) and increases (sharper).
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()  # Normalize to probability
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for tau in tau_list:
        if time.monotonic() > deadline:
            break
        
        # Run L-BFGS-B with current tau
        def func_and_grad(x):
            x = np.maximum(x, 1e-15)
            x = x / x.sum()  # Project to simplex
            val, grad = objective_and_grad(x, tau)
            # Adjust gradient for the projection (approximate)
            return val, grad
        
        result = minimize(
            fun=func_and_grad,
            x0=a,
            jac=True,
            method='L-BFGS-B',
            bounds=[(1e-15, np.inf)] * n,
            options={'maxiter': max_iter_per_tau, 'gtol': 1e-8, 'ftol': 1e-12}
        )
        
        if result.x is not None:
            a_new = np.maximum(result.x, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
                a = a_new
    
    return best_a, best_J

def lp_direction(a, M, top_frac=0.7):
    """Solve LP to find best direction g."""
    n = len(a)
    b = fast_conv(a)
    
    # Select active constraints
    if top_frac < 1.0:
        threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
        active_k = np.where(b >= threshold)[0]
    else:
        active_k = np.arange(2 * n - 1)
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, np.inf)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                # Verify
                ag = fftconvolve(a, g, mode='full')
                if ag.max() <= M * (1 + 1e-4):
                    return g
                elif top_frac < 1.0:
                    # Retry with full constraints
                    return lp_direction(a, M, top_frac=1.0)

def lp_mirror_descent(a, deadline, verbose=True, top_frac=0.7, max_no_improve=100):
    """LP-based mirror descent."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_direction(a, M, top_frac=top_frac)
        if g is None or g.sum() < 1e-10:
            break
        
        g = g / g.sum()
        
        # Line search
        improved = False
        diff = np.max(np.abs(g - a))
        base_t = min(0.5, max(0.05, diff))
        for t in [base_t, base_t*0.5, base_t*0.25, base_t*0.125, 0.02, 0.01]:
            a_trial = np.maximum((1-t)*a + t*g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
            elif J_trial < compute_J(a) - 1e-12:
                a = a_trial
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Perturb
            rng = np.random.default_rng(it)
            scale = 0.1 / (1 + no_improve / max_no_improve)
            a = a * (1 + scale * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    LP iter {it}: J={best_J:.8f}", flush=True)
    
    return best_a, best_J

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_perturbations(a, n_pert, rng):
    """Generate perturbed versions of a."""
    n = len(a)
    t_arr = np.linspace(0, 1, n, endpoint=False)
    perts = [a.copy()]
    
    for i in range(n_pert):
        scale = 0.05 + 0.1 * rng.random()
        # Different types of perturbation
        ptype = i % 4
        if ptype == 0:
            # Gaussian noise
            p = a * (1 + scale * rng.normal(0, 1, n))
        elif ptype == 1:
            # Smooth perturbation (low frequency)
            freq = rng.integers(1, 5)
            phase = rng.uniform(0, 2*np.pi)
            p = a * (1 + scale * np.cos(2*np.pi*freq*t_arr + phase))
        elif ptype == 2:
            # Edge boost
            edge = np.exp(-10*t_arr**2) + np.exp(-10*(t_arr-1)**2)
            p = a * (1 + scale * (edge - edge.mean()))
        else:
            # Random block perturbation
            block_size = n // rng.integers(4, 16)
            p = a.copy()
            for _ in range(3):
                start = rng.integers(0, n - block_size)
                p[start:start+block_size] *= (1 + scale * rng.normal(0, 1))
        
        p = np.maximum(p, 1e-8)
        p = p / p.sum()
        perts.append(p)
    
    return perts

import numpy as np
from scipy.optimize import linprog, minimize

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """
    Compute smooth max of autocorrelation and its gradient.
    Returns (smooth_val, grad_b, p, b) where grad_b[m] = d(smooth_max)/d(a[m]).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    log_sum_exp = max_shifted + np.log(exp_shifted.sum())
    smooth_val = b_max + log_sum_exp / tau
    p = exp_shifted / exp_shifted.sum()
    # Gradient via FFT cross-correlation
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conj(fa), N)
    grad_b = 2.0 * h[:n]
    return smooth_val, grad_b, p, b


def objective_and_grad_full(a, tau, reg_lambda=1e-8):
    """
    Full objective J(a) = 2n*smooth_max(a*a)/(sum a)^2 + reg*(sum a - 1)^2

    grad_J += 2.0 * reg_lambda * (s - 1.0)

            lr = lr_base / (1.0 + 0.5 * tau_idx)
            # Scale by gradient magnitude for stability
            g_norm = np.sqrt(np.mean(grad_J ** 2))
            if g_norm > 1e-15:
                lr_eff = lr * g_norm / (0.5 + g_norm)
            else:
                lr_eff = lr
            
            # Multiplicative update
            log_update = np.exp(-lr_eff * grad_J)
            a_new = a * log_update
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            
            J_new = compute_J(a_new)
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
            
            a = a_new
            
            if verbose and (it + 1) % 200 == 0:
                print(f"    MD tau={tau} iter={it+1}: J={best_J:.8f}", flush=True)
    
    return best_a, best_J


def lbfgs_smooth_max(a, tau_list, deadline, reg_lambda=1e-8):
    """
    L-BFGS-B optimization with continuation in tau.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for tau_idx, tau in enumerate(tau_list):
        if time.monotonic() > deadline:
            break
        
        def func_grad(x):
            x = np.maximum(x, 1e-15)
            val, grad = objective_and_grad_full(x, tau, reg_lambda)
            return val, grad
        
        remaining = deadline - time.monotonic()
        max_iter = max(10, int(remaining / 0.05))
        
        try:
            result = minimize(
                func_grad, a, jac=True, method='L-BFGS-B',
                bounds=[(1e-15, np.inf)] * n,
                options={'maxiter': max_iter, 'gtol': 1e-10, 'ftol': 1e-14, 'maxls': 50}
            )
            if result.x is not None:
                a_new = np.maximum(result.x, 1e-12)
                a_new = a_new / a_new.sum()
                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
                a = a_new
        except Exception:
            pass
    
    return best_a, best_J


def lp_direction(a, M, top_frac=0.7, time_limit=5):
    """Solve LP to find best direction g maximizing sum(g) s.t. a*g <= M, g>=0."""
    n = len(a)
    b = fast_conv(a)
    
    if top_frac < 1.0:
        threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
        active_k = np.where(b >= threshold)[0]
    else:
        active_k = np.arange(2 * n - 1)
    
    if len(active_k) < 5:
        active_k = np.arange(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, np.inf)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                if top_frac < 1.0:
                    ag = fftconvolve(a, g, mode='full')
                    if ag.max() > M * (1 + 1e-3):
                        return None
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, top_frac=0.7, max_no_improve=80, verbose=True):
    """LP-based mirror descent with adaptive step and restarts."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_direction(a, M, top_frac=top_frac, time_limit=3)
        if g is None or g.sum() < 1e-10:
            break
        
        g = g / g.sum()
        diff = np.max(np.abs(g - a))
        
        improved = False
        base_t = min(0.6, max(0.05, diff * 1.5))
        current_J = compute_J(a)
        
        for t in [base_t, base_t * 0.5, base_t * 0.25, base_t * 0.1, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
            elif J_trial < current_J - 1e-13:
                a = a_trial
                current_J = J_trial
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
        
        if no_improve > max_no_improve:
            scale = 0.1 / (1.0 + 0.3 * (no_improve // max_no_improve))
            t_arr = np.linspace(0, 1, n, endpoint=False)
            freqs = rng.integers(1, 6, size=3)
            phases = rng.uniform(0, 2 * np.pi, size=3)
            pert = np.zeros(n)
            for f, ph in zip(freqs, phases):
                pert += np.cos(2 * np.pi * f * t_arr + ph)
            a = a * (1.0 + scale * pert)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            no_improve = 0
        
        if verbose and it % 1000 == 0:
            print(f"    LP iter {it}: J={best_J:.8f} no_improve={no_improve}", flush=True)
    
    return best_a, best_J


def hybrid_optimize(a, deadline, verbose=True):
    """
    Hybrid: alternate between mirror descent (fast, many iterations)
    and L-BFGS (slower, second-order) with LP verification.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    phase = 0
    phase_time = 60.0
    tau_schedules = [
        [5, 10, 20, 40],
        [20, 50, 100, 200],
        [50, 100, 200, 400, 800],
        [100, 200, 500, 1000],
    ]
    
    while time.monotonic() < deadline:
        phase_deadline = min(time.monotonic() + phase_time, deadline)
        tau = tau_schedules[min(phase, len(tau_schedules) - 1)]
        
        # Mirror descent burst
        a_md, J_md = mirror_descent_smooth(a, tau[:2], phase_deadline, lr_base=0.15, verbose=False)
        if J_md < best_J:
            best_J = J_md
            best_a = a_md.copy()
        a = a_md
        
        if time.monotonic() >= deadline:
            break
        
        # L-BFGS refinement
        phase_deadline2 = min(time.monotonic() + 30, deadline)
        a_lb, J_lb = lbfgs_smooth_max(a, tau[1:], phase_deadline2)
        if J_lb < best_J:
            best_J = J_lb
            best_a = a_lb.copy()
            if verbose:
                print(f"    Hybrid phase {phase}: J={best_J:.8f} [L-BFGS]", flush=True)
        a = a_lb if J_lb < J_md else a_md
        
        # LP verification
        if time.monotonic() < deadline - 10:
            b = fast_conv(a)
            M = float(b.max())
            g = lp_direction(a, M, top_frac=0.8, time_limit=3)
            if g is not None and g.sum() > 1e-10:
                g = g / g.sum()
                for t in [0.3, 0.15, 0.05]:
                    a_trial = np.maximum((1 - t) * a + t * g, 0)
                    s = a_trial.sum()
                    if s > 1e-10:
                        a_trial = a_trial / s
                        J_trial = compute_J(a_trial)
                        if J_trial < best_J:
                            best_J = J_trial
                            best_a = a_trial.copy()
                            a = a_trial
                            if verbose:
                                print(f"    Hybrid phase {phase}: J={best_J:.8f} [LP step t={t}]", flush=True)
                            break
        
        phase += 1
        # Gradually reduce phase time
        phase_time = max(30.0, phase_time * 0.85)
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_candidates(parent, rng, n_cands=10):
    """Generate diverse candidates from parent."""
    n = len(parent)
    t_arr = np.linspace(0, 1, n, endpoint=False)
    cands = [parent.copy()]
    
    for i in range(n_cands):
        scale = 0.03 + 0.12 * rng.random()
        ptype = i % 5
        if ptype == 0:
            c = parent * (1 + scale * rng.normal(0, 1, n))
        elif ptype == 1:
            f = rng.integers(1, 5)
            ph = rng.uniform(0, 2 * np.pi)
            c = parent * (1 + scale * np.cos(2 * np.pi * f * t_arr + ph))
        elif ptype == 2:
            edge = np.exp(-15 * t_arr**2) + np.exp(-15 * (t_arr - 1)**2)
            c = parent * (1 + scale * (edge - edge.mean()) / (edge.std() + 1e-10))
        elif ptype == 3:
            # Block perturbation
            c = parent.copy()
            n_blocks = rng.integers(3, 8)
            block_size = n // n_blocks
            for _ in range(2):
                start = rng.integers(0, max(1, n - block_size))
                c[start:min(start + block_size, n)] *= (1 + scale * rng.normal(0, 2))
        else:
            # Mix with a structural pattern
            patterns = [
                np.ones(n),
                0.5 + 0.5 * np.cos(np.pi * (2 * t_arr - 1)),
                np.exp(-3 * (t_arr - 0.5) ** 2),
                1 + 0.3 * np.sin(2 * np.pi * 4 * t_arr),
            ]
            pat = patterns[rng.integers(0, len(patterns))]
            c = (1 - 0.3) * parent + 0.3 * pat
    
        c = np.maximum(c, 1e-10)
        c = c / c.sum()
        cands.append(c)
    
    return cands


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for optimal step function heights.
    Phase 1: Hybrid optimization on diverse candidates (500s)
    Phase 2: Extended LP mirror descent (400s)
    Phase 3: Multi-length exploration (250s)
    Phase 4: Final high-precision polish (remaining)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0
    
    best_seq = None
    best_J = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_J
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
        s = a.sum()
        if s < 0.01:
            return
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={J:.8f} n={a.size} t={elapsed:.0f}s [{label}]", flush=True)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0)
            if parent.sum() > 0.01:
                parent = parent / parent.sum()
                try_update(parent, "parent")
                print(f"Parent loaded: n={parent.size}, J={best_J:.8f}", flush=True)
            else:
                parent = None
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512)
        parent = parent / parent.sum()
        try_update(parent, "default")
        print("Using default parent (uniform 512)", flush=True)
    
    n_main = parent.size
    
    # === PHASE 1: Hybrid on diverse candidates ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Hybrid Optimization (n={n_main})", flush=True)
    print(f"{'='*60}", flush=True)
    
    p1_deadline = t_start + min(500, (budget_s - 100) * 0.35)
    
    candidates = generate_candidates(parent, rng, n_cands=8)
    # Also add some structural inits
    t_arr = np.linspace(0, 1, n_main, endpoint=False)
    candidates.append(np.maximum(0.7 + 0.3 * np.cos(2 * np.pi * t_arr), 0.01))
    candidates.append(np.maximum(1 + 0.2 * np.cos(2 * np.pi * 3 * t_arr), 0.01))
    
    # Quick evaluate and sort
    cand_data = []
    for c in candidates:
        c = np.maximum(c, 1e-10)
        c = c / c.sum()
        J_c = compute_J(c)
        cand_data.append((J_c, c))
    cand_data.sort(key=lambda x: x[0])
    
    print(f"  {len(cand_data)} candidates, best init J={cand_data[0][0]:.8f}", flush=True)
    
    # Allocate time to top candidates
    top_k = min(5, len(cand_data))
    time_per_cand = (p1_deadline - time.monotonic()) / top_k
    
    for ci in range(top_k):
        if time.monotonic() > p1_deadline:
            break
        J_init, a_init = cand_data[ci]
        cand_deadline = min(time.monotonic() + time_per_cand, p1_deadline)
        
        if J_init >= best_J and ci > 0:
            # Skip if not better than current best and not the first
            continue
        
        a_opt, J_opt = hybrid_optimize(a_init, cand_deadline, verbose=(ci == 0))
        try_update(a_opt, f"hybrid_c{ci}")
    
    # === PHASE 2: Extended LP mirror descent on best ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: Extended LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)
    
    p2_deadline = min(t_start + 900, deadline - 200)
    
    if best_seq is not None and time.monotonic() < p2_deadline:
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        a_opt, J_opt = lp_mirror_descent(a_work, p2_deadline, top_frac=0.7,
                                          max_no_improve=100, verbose=True)
        try_update(a_opt, "lp_ext")
        
        # Also try with different top_frac
        if time.monotonic() < p2_deadline - 30:
            a_opt2, _ = lp_mirror_descent(best_seq / best_seq.sum(),
                                           min(time.monotonic() + 60, p2_deadline),
                                           top_frac=0.5, max_no_improve=50, verbose=False)
            try_update(a_opt2, "lp_ext_t05")
    
    # === PHASE 3: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    p3_deadline = min(t_start + 1200, deadline - 80)
    
    if best_seq is not None and time.monotonic() < p3_deadline:
        base_n = best_seq.size
        # Try lengths: keep, +/- 32, +/- 64, double, half
        n_options = []
        for dn in [-64, -32, 32, 64]:
            nn = base_n + dn
            if 128 <= nn <= 4096:
                n_options.append(nn)
        if base_n // 2 >= 128:
            n_options.append(base_n // 2)
        if base_n * 2 <= 4096:
            n_options.append(base_n * 2)
        # Also try some "natural" lengths
        for nn in [256, 384, 512, 640, 768, 1024]:
            if nn != base_n and 128 <= nn <= 4096:
                if nn not in n_options:
                    n_options.append(nn)
        
        n_options = list(set(n_options))[:6]
        
        for n_try in n_options:
            if time.monotonic() > p3_deadline:
                break
            a_ext = resample_seq(best_seq, n_try)
            a_ext = np.maximum(a_ext, 1e-10)
            a_ext = a_ext / a_ext.sum()
            
            # Quick hybrid
            a_opt, J_opt = hybrid_optimize(a_ext, min(time.monotonic() + 30, p3_deadline - 10),
                                            verbose=False)
            try_update(a_opt, f"n={n_try}")
    
    # === PHASE 4: Final polish ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Final Polish", flush=True)
    print(f"  Remaining: {deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_work = best_seq.copy() / best_seq.sum()
        
        # High-tau L-BFGS for precision
        a_opt, _ = lbfgs_smooth_max(a_work, [200, 500, 1000], deadline - 10)
        try_update(a_opt, "polish_lbfgs")
        
        # Final LP with tight constraints
        if time.monotonic() < deadline - 5:
            a_opt2, _ = lp_mirror_descent(best_seq / best_seq.sum(), deadline - 5,
                                           top_frac=0.9, max_no_improve=300, verbose=False)
            try_update(a_opt2, "polish_lp")
        
        # One more mirror descent with very high tau
        if time.monotonic() < deadline - 5:
            a_opt3, _ = mirror_descent_smooth(best_seq / best_seq.sum(), [500, 1000],
                                               deadline - 5, lr_base=0.05, verbose=False)
            try_update(a_opt3, "polish_md")
    
    # Fallback
    if best_seq is None:
        best_seq = parent
        best_J = compute_J(best_seq)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_J:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_seq]
