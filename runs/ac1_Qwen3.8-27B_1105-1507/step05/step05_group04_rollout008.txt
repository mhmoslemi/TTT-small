Hybrid optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated smooth gradient descent with momentum, coordinate perturbation,
and basin hopping. Key innovation: O(n log n) gradient via FFT cross-correlation,
enabling 50x more optimization steps than previous O(n^2) approach.

import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def update_best(a, label=""):
    global _best_seq, _best_val
    a = np.clip(a.astype(np.float64), 0.0, 1000.0)

def resample(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_grad_fft(a, tau):
    """
    Compute gradient of smooth max of conv(a,a) using FFT.
    Returns (grad, b_max) where b_max = max(conv(a,a)).
    When sum(a)=1, J = 2n * b_max.
    Complexity: O(n log n).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    
    # Softmax weights
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    
    # grad[i] = 2 * sum_m a[m] * p[i+m] for m in [0, n-1], i+m < 2n-1
    # = 2 * (a * p_flip)[2n-2-i]
    p_flip = p[::-1]
    
    size = 1
    while size < n + (2 * n - 1) - 1:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_flip, size)
    conv_result = np.fft.irfft(fa * fp, size)
    
    # Get indices [2n-2, 2n-3, ..., n-1]
    idx = np.arange(2 * n - 2, n - 2, -1)
    grad = 2.0 * conv_result[idx]
    
    return grad, b_max


def make_palindrome(a):
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new

    inits.append(np.maximum(1 - np.abs(2 * t - 1), 0.001))
    
    # Sinc-like
    x = 2 * t - 1
    inits.append(np.maximum(np.sinc(3 * x) + 0.3, 0.001))
    
    # Normalize all
    result = []
    for a in inits:
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
        result.append(a)
    return result


def smooth_optimize(a, deadline, tau_init=50.0, tau_min=3.0,
                    lr=0.005, momentum=0.9, verbose=False):
    """
    Smooth gradient descent with momentum and adaptive tau.
    Normalizes a to sum=1. Uses b_max as proxy for J (since J=2n*b_max when sum=1).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_bmax = fast_conv(a).max()
    v = np.zeros(n)
    tau = tau_init
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        
        # Adaptive tau
        tau = max(tau_min, tau * 0.9995)
        
        # Compute gradient
        grad, b_max = smooth_grad_fft(a, tau)
        
        if b_max < best_bmax:
            best_bmax = b_max
            best_a = a.copy()
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        a_new = np.maximum(a - lr * v, 1e-10)
        a_new = a_new / a_new.sum()
        
        # Check convergence
        if np.max(np.abs(a_new - a)) < 1e-12:
            break
        
        a = a_new
        
        if verbose and it % 2000 == 0:
            print(f"      it={it} b_max={best_bmax:.10f} tau={tau:.1f}", flush=True)
    
    return best_a


def coord_perturb(a, rng, n_tries=50, scale=0.1):
    """
    Try random coordinate perturbations. Accept if J improves.
    Returns the (possibly improved) sequence.
    """
    n = len(a)
    current_bmax = fast_conv(a).max()
    
    for _ in range(n_tries):
        # Pick a random coordinate
        i = rng.integers(0, n)
        
        # Try several multipliers
        best_mult = 1.0
        best_val = 0.0
        for mult in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5]:
            a_try = a.copy()
            a_try[i] *= mult
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            bmax_try = fast_conv(a_try).max()
            if bmax_try < best_val or best_val == 0.0:
                best_val = bmax_try
                best_mult = mult
        
        if best_val < current_bmax * (1 - 1e-10):
            a[i] *= best_mult
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            current_bmax = best_val
    
    return a, current_bmax


def basin_hopping(a, rng, scale=0.1):
    """Generate a perturbed version for basin hopping."""
    n = len(a)
    # Smooth random perturbation
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    noise = np.convolve(rng.normal(0, 1, n), kernel, mode='same')
    
    a_new = a + scale * noise
    a_new = np.maximum(a_new, 1e-6)
    a_new = a_new / a_new.sum()
    return a_new


def multi_coord_perturb(a, rng, n_blocks=10, n_tries_per_block=5, scale=0.2):
    """
    Perturb blocks of coordinates simultaneously.
    More effective than single-coordinate perturbation.
    """
    n = len(a)
    current_bmax = fast_conv(a).max()
    block_size = max(1, n // n_blocks)
    
    for _ in range(n_tries_per_block * n_blocks):
        # Pick a random block
        start = rng.integers(0, n - block_size)
        idx = np.arange(start, min(start + block_size, n))
        
        # Try perturbing the whole block
        best_a = None
        best_bmax = current_bmax
        
        for mult in [0.5, 0.7, 0.8, 1.0, 1.2, 1.3, 1.5]:
            a_try = a.copy()
            a_try[idx] *= mult
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            bmax_try = fast_conv(a_try).max()
            if bmax_try < best_bmax:
                best_bmax = bmax_try
                best_a = a_try
        
        if best_a is not None:
            a = best_a
            current_bmax = best_bmax
    
    return a, current_bmax


def optimize_sequence(a, deadline, verbose=False):
    """
    Main optimization loop combining smooth gradient, coordinate perturbation,
    and basin hopping.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_bmax = fast_conv(a).max()
    
    rng = np.random.default_rng(12345)
    
    # Phase A: Smooth gradient (60% of time)
    smooth_dl = _t_start + 0.6 * (deadline - _t_start)
    a = smooth_optimize(a, smooth_dl, tau_init=50.0, tau_min=5.0,
                        lr=0.005, momentum=0.9, verbose=verbose)
    bmax = fast_conv(a).max()
    if bmax < best_bmax:
        best_bmax = bmax
        best_a = a.copy()
    
    # Phase B: Coordinate perturbation + smooth gradient (30% of time)
    coord_dl = _t_start + 0.9 * (deadline - _t_start)
    it = 0
    while time.monotonic() < coord_dl:
        it += 1
        # Coordinate perturbation
        a, bmax = multi_coord_perturb(a, rng, n_blocks=10, n_tries_per_block=3, scale=0.2)
        if bmax < best_bmax:
            best_bmax = bmax
            best_a = a.copy()
        
        # Short smooth gradient
        a = smooth_optimize(a, time.monotonic() + 1.0, tau_init=20.0, tau_min=5.0,
                           lr=0.003, momentum=0.9)
        bmax = fast_conv(a).max()
        if bmax < best_bmax:
            best_bmax = bmax
            best_a = a.copy()
        
        # Periodic basin hopping
        if it % 10 == 0:
            a = basin_hopping(best_a, rng, scale=0.15)
    
    # Phase C: Fine coordinate perturbation (10% of time)
    while time.monotonic() < deadline:
        a, bmax = multi_coord_perturb(best_a, rng, n_blocks=20, n_tries_per_block=5, scale=0.1)
        if bmax < best_bmax:
            best_bmax = bmax
            best_a = a.copy()
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search for optimal step function heights.
    Phase 1: Diverse initialization + quick optimization (200s)
    Phase 2: Deep optimization of best candidates (900s)
    Phase 3: Multi-length exploration + final polish (350s)
    """
    global _t_start, _best_seq, _best_val
    _t_start = time.monotonic()
    _best_seq = None
    _best_val = float('inf')
    
    deadline = _t_start + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent, "parent")
            print(f"Parent: n={parent.size}, J={_best_val:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(256)
    
    # === PHASE 1: Diverse initialization (200s) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Diverse Initialization", flush=True)
    print(f"{'='*60}", flush=True)
    
    p1_deadline = _t_start + 200.0
    
    n_main = 512
    inits = generate_inits(n_main, rng, parent)
    
    candidates = []
    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_deadline:
            break
        # Quick smooth optimization (20s each, max 10 inits)
        alloc = min(20.0, max(5.0, (p1_deadline - time.monotonic()) / max(1, len(inits) - idx)))
        a_opt = smooth_optimize(a0, time.monotonic() + alloc, 
                                tau_init=50.0, tau_min=10.0,
                                lr=0.005, momentum=0.9)
        bmax = fast_conv(a_opt).max()
        J_val = 2 * n_main * bmax  # since sum=1
        candidates.append((J_val, a_opt))
        print(f"  init {idx:2d}: J={J_val:.8f}", flush=True)
    
    candidates.sort(key=lambda x: x[0])
    
    for J_val, a_opt in candidates[:3]:
        update_best(a_opt, "phase1")
    
    print(f"Phase 1 best: J={candidates[0][0]:.8f}", flush=True)
    
    # === PHASE 2: Deep optimization (900s) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: Deep Optimization", flush=True)
    print(f"{'='*60}", flush=True)
    
    p2_deadline = _t_start + 1100.0
    
    top_k = min(3, len(candidates))
    time_per = (p2_deadline - _t_start - 200) / top_k
    
    for ci in range(top_k):
        if time.monotonic() > p2_deadline:
            break
        
        alloc = min(time_per, p2_deadline - time.monotonic())
        alloc = max(100.0, alloc)
        phase_dl = min(time.monotonic() + alloc, p2_deadline)
        
        print(f"\n  Candidate {ci} (J={candidates[ci][0]:.8f}), budget={alloc:.0f}s", flush=True)
        
        a_opt = optimize_sequence(candidates[ci][1], phase_dl, verbose=(ci == 0))
        update_best(a_opt, f"deep_{ci}")
    
    # === PHASE 3: Multi-length + polish (remaining) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Multi-length + Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    p3_deadline = deadline
    
    if _best_seq is not None and time.monotonic() < p3_deadline:
        base_n = _best_seq.size
