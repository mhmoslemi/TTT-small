"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Hot-set subgradient method (average gradient over active constraints) with backtracking
2. Aggressive LP mirror descent with large steps and line search
3. Non-palindromic structural exploitation (antisymmetric perturbations reduce central peak)
4. Multi-length exploration (shorter n has smaller 2*n factor)
5. Coordinate descent for final polish
Key innovation: exploits the non-convexity by using large aggressive steps with
backtracking, and tries structural initializations based on the insight that
non-palindromic sequences can achieve lower central convolution peaks.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.full_like(v, target_sum / len(v))
    return v * (target_sum / s)


def subgradient_hot_set(a, hot_set):
    """Compute average subgradient of max(b) over hot_set indices."""
    n = len(a)
    grad = np.zeros(n)
    for k in hot_set:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= max(1, len(hot_set))
    return grad


def run_hot_set_subgrad(a, max_iters, deadline, eta_start=0.5, hot_frac=0.05):
    """Hot-set subgradient method with backtracking."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    eta = eta_start
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        S = int(max(1, n * hot_frac))
        hot_set = np.argsort(b)[-S:]

        if len(hot_set) == 0:
            break

        grad = subgradient_hot_set(a, hot_set)
        # Project gradient to sum-zero subspace
        grad = grad - grad.mean()

        # Try step and backtrack
        improved = False
        for trial_eta in [eta, eta * 0.5, eta * 0.25, eta * 0.1, eta * 0.05]:
            a_trial = a - trial_eta * grad
            a_trial = project_simplex(a_trial, 1.0)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                # Increase eta if we've been stuck
                if no_improve > 10:
                    eta = min(eta * 1.2, 2.0)
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            eta *= 0.7
            if no_improve > 50:
                # Shake up
                a = a + np.random.randn(n) * 0.01
                a = project_simplex(a, 1.0)
                no_improve = 0
                eta = eta_start * 0.5

    return best_a


def build_lp_matrix(n, a):
    """Build sparse matrix for LP: (a*g)_k = sum_j a_{k-j} * g_j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, timeout=10):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': timeout, 'feastol': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_aggressive_lp(a, max_iters, deadline, timeout=8):
    """Aggressive LP mirror descent with large steps and backtracking."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, timeout)
        if g is None:
            break
        g = g / g.sum()

        improved = False
        # Try large steps first, then smaller
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

    return best_a


def coordinate_update_fast(a, i, b):
    """Fast 1D coordinate update using precomputed convolution."""
    n = len(a)
    x_cur = a[i]
    S = a.sum()

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0

    affected = np.arange(k_lo, k_hi + 1)
    c_vals = np.zeros(len(affected))
    slopes = np.zeros(len(affected))
    quad_idx = -1

    for idx, k in enumerate(affected):
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            slopes[idx] = 0.0
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2 * x_cur * a[j]
                slopes[idx] = 2 * a[j]
            else:
                c_vals[idx] = b[k]
                slopes[idx] = 0.0

    def obj(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 0:
            return 1e10
        vals = c_vals + slopes * x_test
        m = base_max
        for idx2 in range(len(vals)):
            if vals[idx2] > m:
                m = vals[idx2]
        if quad_idx >= 0:
            qv = c_vals[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)

    lo = 0.0
    hi = max(x_cur * 3.0, 0.5, 0.1)
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)

    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coord_descent(a, max_sweeps, deadline, seed=0):
    """Coordinate descent with periodic re-convolution."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        b = fast_conv(a)
        sweep_improved = False

        for i in order:
            x_opt = coordinate_update_fast(a, i, b)
            if abs(x_opt - a[i]) > 1e-10:
                a[i] = x_opt
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        J = compute_J(a)
        if J < best_J - 1e-10:
            best_J = J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not sweep_improved:
            no_improve += 3
        if no_improve > 20:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == 0:
        return np.ones(dest_n)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def structural_inits(n, rng, parent=None):
    """Generate diverse structural initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(np.maximum(p, 0.01))
        # Antisymmetric perturbation
        d = rng.normal(0, 0.03, n)
        d = d - d[::-1]  # make antisymmetric
        inits.append(np.maximum(p + d, 0.01))
        inits.append(np.maximum(p + 2 * d, 0.01))
        # Random perturbation
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))

    # Uniform
    inits.append(np.ones(n))

    # Valley (dip in middle) - reduces central peak
    for depth in [0.2, 0.4, 0.6]:
        inits.append(np.maximum(1.0 - depth * np.exp(-20 * (t - 0.5)**2), 0.05))

    # Edge-weighted (more at edges)
    for w in [1.0, 2.0, 3.0]:
        inits.append(np.maximum(1.0 + w * (2*t - 1)**2, 0.05))

    # Geometric-ish
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * t), 0.05))
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.05))
    inits.append(np.maximum(0.5 + 0.5 * np.cos(3 * np.pi * t), 0.05))

    # Smooth random
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0.3, 1.0, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.05))

    # "Sawtooth" pattern
    inits.append(np.maximum(0.5 + 0.5 * np.abs(2 * (t * 3 % 1) - 1), 0.05))

    return inits


class Tracker:

                a_mix = a_mix / a_mix.sum()
                cd_dl3 = min(time.monotonic() + 50, deadline - 20)
                a_cd3 = run_coord_descent(a_mix, 50, cd_dl3, seed=333)
                tracker.update(a_cd3)

    tracker.report("Phase 2 complete")
    print(f"Phase 2 total: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Final Polish ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Final Polish (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 5:
        # More LP iterations
        n_best = tracker.best.size
        est_time = 0.003 * (n_best / 128)**2
        n_lp = max(10, min(100, int(100 / max(0.01, est_time))))
        a_final_lp = run_aggressive_lp(tracker.best, n_lp, deadline - 5, timeout=5)
        tracker.update(a_final_lp)

        # Final coordinate descent
        if time.monotonic() < deadline - 3:
            a_final_cd = run_coord_descent(a_final_lp, 30, deadline - 1, seed=999)
            tracker.update(a_final_cd)

        # Try extending length
        if time.monotonic() < deadline - 30:
            for mult in [2]:
                new_n = n_best * mult
                if new_n > 1536:
                    continue
                a_ext = resample_seq(tracker.best, new_n)
                a_ext = np.maximum(a_ext, 1e-4)
                a_ext = a_ext / a_ext.sum()
                a_ext_opt = run_coord_descent(a_ext, 30, deadline - 3, seed=777)
                tracker.update(a_ext_opt)

    tracker.report("Phase 3 complete")
    print(f"Phase 3 total: {time.monotonic()-p3_start:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(384)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
