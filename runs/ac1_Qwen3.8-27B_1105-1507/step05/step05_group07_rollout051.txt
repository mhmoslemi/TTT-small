                    obj = 1e10
                else:
                    obj = 2.0 * n * conv.max() / (new_s * new_s)
                # Rollback
                conv[k_lo:k_hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
                a_slice[idx_self] = old
                return obj

            # Golden section search
            c = hi_b - gr * (hi_b - lo_b)
            d = lo_b + gr * (hi_b - lo_b)
            fc = eval_at(c)
            fd = eval_at(d)

            for _ in range(25):
                if hi_b - lo_b < 1e-12 * max(1.0, old):
                    break
                if fc <= fd:
                    hi_b = d
                    d = c
                    fd = fc
                    c = hi_b - gr * (hi_b - lo_b)
                    fc = eval_at(c)
                else:
                    lo_b = c
                    c = d
                    fc = fd
                    d = lo_b + gr * (hi_b - lo_b)
                    fd = eval_at(d)

            best_x = (lo_b + hi_b) / 2.0
            best_f = min(fc, fd)

            # Check if improvement
            current_f = 2.0 * n * conv.max() / (s * s)
            if best_f < current_f - 1e-13:
                # Apply
                delta = best_x - old
                a_slice[idx_self] = best_x
                conv[k_lo:k_hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_x
                s = s - old + best_x
                a_slice[idx_self] = old
                sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    return best_a


def penalty_lbfgs(a0, deadline, lam_sequence=None, max_outer=10):
    """
    Penalty-based max equalization using L-BFGS.
    Minimizes: t + lam * sum_k max(0, conv[k] - t)^2
    where t = max(conv(a,a)) is updated each outer iteration.
    Inner: L-BFGS on a (with sum=1, a>=0 via projection).
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()

    best_a = a.copy()
    best_val = compute_J(a)

    if lam_sequence is None:
        lam_sequence = [0.1, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]

    for outer in range(min(max_outer, len(lam_sequence))):
        if time.monotonic() > deadline:
            break
        lam = lam_sequence[outer]

        b = fast_conv(a)
        t_target = float(b.max())

        def obj_grad(x):
            a_x = np.maximum(x, 0)
            s_x = a_x.sum()
            if s_x < 1e-10:
                return 1e10, np.zeros(n)
            a_x = a_x / s_x
            b_x = fast_conv(a_x)
            excess = np.maximum(b_x - t_target, 0)
            val = 2.0 * n * t_target + lam * (excess ** 2).sum()

            # Gradient: d/da[i] [lam * sum_k excess[k]^2]
            # = lam * sum_k 2*excess[k] * 2*a[k-i] (for valid k-i)
            w = 4.0 * lam * excess
            n_conv = len(b_x)
            N_fft = 1
            while N_fft < 4 * n + 8:
                N_fft <<= 1
            w_pad = np.zeros(N_fft)
            w_pad[:n_conv] = w
            a_pad = np.zeros(N_fft)
            a_pad[:n] = a_x
            W = rfft(w_pad)
            A = rfft(a_pad)
            grad_raw = irfft(W * np.conj(A), N_fft)[:n]
            # Chain rule for normalization: a_x = x / sum(x)
            # Approximate: just use grad_raw (ignoring the /s chain for speed)
            return val, grad_raw

        try:
            res = minimize(obj_grad, a, method='L-BFGS-B', jac=True,
                          options={'maxiter': 80, 'maxfun': 150,
                                   'ftol': 1e-14, 'gtol': 1e-12, 'disp': False})
            if res.x is not None:
                a_new = np.maximum(res.x, 0)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    J_new = compute_J(a_new)
                    if J_new < best_val:
                        best_val = J_new
                        best_a = a_new.copy()
                    a = a_new
        except Exception as e:
            pass

    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: (a*g)[k] <= M for all k."""
    rows = []
    cols = []
    vals = []
    n_conv = 2 * n - 1
    for k in range(n_conv):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(n_conv, n))


def lp_mirror_step(a, deadline, max_iters=20, time_per_lp=5.0):
    """LP mirror descent: find best response g, blend with a."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()

    best_a = a.copy()
    best_val = compute_J(a)
    no_improve = 0

    # Pre-build A (it only depends on a, which changes, so rebuild each iter)
    for it in range(max_iters):
        if time.monotonic() > deadline - time_per_lp:
            break

        b = fast_conv(a)
        M = float(b.max())

        t0 = time.monotonic()
        A = build_A_sparse(n, a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': time_per_lp, 'presolve': True})
            if not r.success or r.x is None:
                break
            g = np.maximum(r.x, 0)
            if g.sum() < 1e-10:
                break
            g = g / g.sum()
        except Exception:
            break

        lp_time = time.monotonic() - t0
        if lp_time > 30:
            break

        # Line search
        improved = False
        for t in [0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_val - 1e-12:
                best_val = J_trial
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

    return best_a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        if p.sum() > 1e-10:
            p = p / p.sum()
            inits.append(p.copy())
            inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.001))
            inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.001))
            inits.append(np.maximum(p + rng.normal(0, 0.2, n), 0.001))

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine variants
    for beta in [0.3, 0.5, 0.7, 1.0]:
        rc = 0.5 * (1 + np.cos(np.pi * (2 * t - 1) / beta)) if beta < 1 else np.maximum(np.cos(np.pi * (t - 0.5)), 0)
        rc = np.maximum(rc, 0.01)
        inits.append(rc / rc.sum())

    # Cosine bumps
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            sig = np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01)
            inits.append(sig / sig.sum())
            sig2 = np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t + np.pi / 4), 0.01)
            inits.append(sig2 / sig2.sum())

    # Gaussian centered
    for c in [0.3, 0.5, 0.7]:
        for w in [0.1, 0.15, 0.2, 0.3]:
            g = np.exp(-((t - c) / w) ** 2) + 0.05
            inits.append(g / g.sum())

    # Spectral flat init
    inits.append(spectral_init(n, target_max_ratio=0.9, n_iter=100))
    inits.append(spectral_init(n, target_max_ratio=0.7, n_iter=100))

    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        smooth = np.maximum(smooth, 0.01)
        inits.append(smooth / smooth.sum())

    # Watermelon (low at edges, high at center)
    wm = np.maximum(1.0 - 0.8 * (2 * t - 1) ** 2, 0.01)
    inits.append(wm / wm.sum())

    # Inverse watermelon
    iwm = np.maximum(0.2 + 0.8 * (2 * t - 1) ** 2, 0.01)
    inits.append(iwm / iwm.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Multi-phase:
    Phase 1: Diverse coordinate descent on multiple inits (n=512) ~400s
    Phase 2: Penalty L-BFGS refinement ~250s
    Phase 3: Longer lengths (768, 1024, 1536) ~350s
    Phase 4: LP mirror + final polish ~250s
    Phase 5: Final coordinate descent ~remaining
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        tracker.update(parent)

    # === PHASE 1: Coordinate descent on diverse inits ===
    p1_end = min(time.monotonic() + 400, deadline - 550)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Coordinate Descent (budget {p1_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    n_base = parent.size if parent.size > 0 else 512
    if n_base < 256:
        n_base = 512
        parent = resample_seq(parent, n_base)

    inits = generate_initializations(n_base, rng, parent)
    # Sort by initial J (best first)
    init_scores = [(compute_J(a), idx, a) for idx, a in enumerate(inits)]
    init_scores.sort(key=lambda x: x[0])

    for rank, (J_init, idx, a0) in enumerate(init_scores):
        if time.monotonic() > p1_end:
            break
        if J_init < 1.45:
            break  # Diminishing returns

        a0 = np.maximum(a0, 0)
        a0 = a0 / a0.sum()

        t0 = time.monotonic()
        remaining_for_this = max(10, min(80, p1_end - time.monotonic()))
        a_opt = coord_descent_golden(a0, deadline=time.monotonic() + remaining_for_this,
                                      max_sweeps=2000)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_opt)
        print(f"  Init {idx} (rank {rank}, J0={J_init:.6f}): J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
        tracker.update(a_opt)

    tracker.report("Phase 1")

    # === PHASE 2: Penalty L-BFGS ===
    p2_end = min(time.monotonic() + 250, deadline - 400)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Penalty L-BFGS (budget {p2_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        t0 = time.monotonic()
        a_pen = penalty_lbfgs(tracker.best, deadline=p2_end,
                             lam_sequence=[0.5, 1, 2, 5, 10, 20, 50, 100, 200, 500],
                             max_outer=10)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_pen)
        print(f"  Penalty L-BFGS: J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
        tracker.update(a_pen)

        # Follow up with coordinate descent
        if time.monotonic() < p2_end - 10:
            a_cd = coord_descent_golden(a_pen, deadline=p2_end, max_sweeps=1000)
            J_val2 = compute_J(a_cd)
            print(f"  CD after penalty: J={J_val2:.8f}", flush=True)
            tracker.update(a_cd)

    tracker.report("Phase 2")

    # === PHASE 3: Longer lengths ===
    p3_end = min(time.monotonic() + 350, deadline - 250)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Longer Lengths (budget {p3_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    base_n = tracker.best.size if tracker.best is not None else 512
    target_lengths = []
    for L in [768, 1024, 1536, 2048]:
        if L > base_n:
            target_lengths.append(L)

    for new_n in target_lengths[:3]:  # At most 3 longer lengths
        if time.monotonic() > p3_end - 30:
            break

        t0 = time.monotonic()
        a_ext = resample_seq(tracker.best, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        # Quick coord descent
        time_budget = max(10, p3_end - time.monotonic() - 30)
        a_opt_cd = coord_descent_golden(a_ext, deadline=time.monotonic() + time_budget * 0.5,
                                         max_sweeps=1000)
        J_cd = compute_J(a_opt_cd)

        # Penalty L-BFGS
        if time.monotonic() < p3_end - 20:
            a_opt_pen = penalty_lbfgs(a_opt_cd, deadline=p3_end - 10,
                                     lam_sequence=[1, 5, 10, 50, 100, 500],
                                     max_outer=5)
            J_pen = compute_J(a_opt_pen)
            a_opt = a_opt_pen if J_pen < J_cd else a_opt_cd
            J_final = min(J_cd, J_pen)
        else:
            a_opt = a_opt_cd
            J_final = J_cd

        elapsed = time.monotonic() - t0
        print(f"  n={new_n}: J={J_final:.8f} ({elapsed:.1f}s)", flush=True)
        tracker.update(a_opt)

    tracker.report("Phase 3")

    # === PHASE 4: LP mirror + coordinate descent ===
    p4_end = min(time.monotonic() + 250, deadline - 50)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: LP Mirror + CD (budget {p4_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        # LP mirror (only if n is small enough)
        if n_best <= 768 and time.monotonic() < p4_end - 30:
            t0 = time.monotonic()
            a_lp = lp_mirror_step(tracker.best, deadline=time.monotonic() + min(60, p4_end - time.monotonic() - 30),
                                  max_iters=10, time_per_lp=3.0)
            elapsed = time.monotonic() - t0
            J_val = compute_J(a_lp)
            print(f"  LP mirror: J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
            tracker.update(a_lp)

        # Final coordinate descent
        if time.monotonic() < p4_end - 10:
            a_final_cd = coord_descent_golden(tracker.best, deadline=p4_end, max_sweeps=5000)
            J_val = compute_J(a_final_cd)
            print(f"  Final CD: J={J_val:.8f}", flush=True)
            tracker.update(a_final_cd)

    tracker.report("Phase 4")

    # === PHASE 5: Final polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish (budget {deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 10:
        # One more round of penalty + CD
        a_p = penalty_lbfgs(tracker.best, deadline=min(time.monotonic() + 30, deadline - 20),
                           lam_sequence=[100, 500, 1000], max_outer=3)
        tracker.update(a_p)

        if time.monotonic() < deadline - 5:
            a_c = coord_descent_golden(tracker.best, deadline=deadline, max_sweeps=2000)
            tracker.update(a_c)

    tracker.report("Phase 5")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
