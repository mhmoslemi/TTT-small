def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def subgradient_of_max_conv(a, active_indices):
    """Compute subgradient of max_k conv[k] using FFT-based cross-correlation."""
    n = len(a)
    # Build indicator of active set (length 2n-1)
    S = np.zeros(2 * n - 1)
    S[active_indices] = 1.0
    
    # Cross-correlation: subgrad[i] = sum_j S[i+j] * a[j]
    # Use FFT: ifft(fft(S) * conj(fft(a_padded)))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fs = np.fft.rfft(S, size)
    fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(fs * np.conj(fa), size)[:n]
    return 2.0 * corr / len(active_indices)

def subgradient_mirror_descent(a, deadline, eta0=0.1, decay=0.9999, 
                                active_tol=0.99, max_iters=100000,
                                momentum=0.0):

    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)

def spectral_shaping(a, n_iters=100, target='constant'):
    """GS-like spectral shaping to flatten the convolution."""
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(n_iters):
        # Forward FFT
        A = np.fft.rfft(a, N)
        
        # Modify magnitude
        if target == 'constant':
            target_mag = np.abs(A).mean()
        elif target == 'flat_conv':

        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        a = a_new
    
    return a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based subgradient mirror descent with adaptive step size and active set
2. Fast coordinate descent polishing (exact recompute per candidate)
3. Spectral shaping (Gerchberg-Saxton style iterations)
4. Informed MCMC with subgradient-biased proposals
5. Multi-restart with diverse initializations and multi-length exploration
Key innovation: efficient O(n log n) subgradient via cross-correlation FFT,

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def subgradient_via_fft(a, active_indices):
    """Compute subgradient of max_k conv[k] using FFT cross-correlation.
    subgrad[i] = (2/|S|) * sum_{j: i+j in S} a[j]
    """
    n = len(a)
    S = np.zeros(2 * n - 1)
    S[active_indices] = 1.0
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fs = np.fft.rfft(S, size)
    fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(fs * np.conj(fa), size)[:n]
    return 2.0 * corr / len(active_indices)


def adaptive_subgradient_descent(a, deadline, eta0=0.05, max_iters=200000,
                                 active_frac=0.1, momentum=0.3,
                                 explore_every=500):
    """Subgradient mirror descent with adaptive step size and active set."""
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    eta = eta0
    v = np.zeros(n)
    no_improve = 0
    prev_J = best_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = conv.max()

        # Adaptive active set: top fraction of bins
        n_active = max(1, int(n * active_frac))
        sorted_idx = np.argsort(conv)[::-1]
        active_indices = sorted_idx[:n_active]

        # Subgradient
        sg = subgradient_via_fft(a, active_indices)

        # Momentum
        v = momentum * v + (1.0 - momentum) * sg

        # Multiplicative update (mirror map for simplex)
        a = a * np.exp(-eta * v)
        a = np.maximum(a, 1e-15)
        s = a.sum()
        a = a / s

        J = compute_J(a)

        # Adaptive step size
        if J < prev_J - 1e-10:
            eta = min(eta * 1.02, 0.5)
            no_improve = 0
        else:
            eta *= 0.95
            no_improve += 1

        if J < best_J - 1e-14:
            best_J = J
            best_a = a.copy()

        prev_J = J

        # Exploration: periodic large perturbation
        if explore_every > 0 and it % explore_every == explore_every - 1:
            if no_improve > 50:
                a = a + 0.1 * np.random.randn(n) * a
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                no_improve = 0
                eta = eta0

    return best_a, best_J


def coordinate_descent(a, deadline, max_passes=50, n_candidates=7):
    """Fast coordinate descent: recompute J for each candidate value."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    rng = np.random.default_rng(12345)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        any_improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            old_val = a[i]
            candidates = []
            if old_val < 1e-10:
                candidates = [0.01, 0.05, 0.1, 0.2, 0.3]
            else:
                candidates = [0.0, old_val * 0.3, old_val * 0.6, old_val * 0.85,
                              old_val * 1.15, old_val * 1.4, old_val * 1.8]

            best_val = old_val
            best_score = best_J

            for cand in candidates:
                if abs(cand - old_val) < 1e-14:
                    continue
                a[i] = cand
                a_pos = np.maximum(a, 0)
                sc = a_pos.sum()
                if sc < 0.01:
                    continue
                score = compute_J(a_pos / sc)
                if score < best_score - 1e-14:
                    best_score = score
                    best_val = cand

            if best_val != old_val:
                a[i] = best_val
                any_improved = True

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J - 1e-14:
                best_J = J
                best_a = a.copy()

        if not any_improved:
            break

    return best_a


def spectral_shaping(a, n_iters=200, target='constant'):
    """GS-like spectral shaping to encourage flat convolution."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(n_iters):
        # Forward FFT
        A = np.fft.rfft(a, N)

        # Target magnitude
        if target == 'constant':
            target_mag = np.abs(A).mean()
            new_mag = np.full(N // 2 + 1, target_mag)
        elif target == 'flat_conv':
            c = 1.0 / (2 * n - 1)
            new_mag = np.zeros(N // 2 + 1)
            new_mag[0] = np.sqrt(c * (2 * n - 1))
            for k in range(1, N // 2 + 1):
                ratio = np.sin(np.pi * k * (2 * n - 1) / N) / np.sin(np.pi * k / N)
                new_mag[k] = np.sqrt(c * abs(ratio))
        else:
            break

        # Blend: partially move toward target
        alpha = 0.5
        phase = np.angle(A)
        cur_mag = np.abs(A)
        blended_mag = (1 - alpha) * cur_mag + alpha * new_mag
        A_new = blended_mag * np.exp(1j * phase)
        A_new[0] = np.abs(A[0])  # preserve DC

        # Inverse FFT
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        J = compute_J(a_new)
        if J < best_J - 1e-14:
            best_J = J
            best_a = a_new.copy()

        a = a_new

    return best_a


def informed_mcmc(a, deadline, max_iters=50000, T0=0.02, T_end=0.001):
    """MCMC with proposals biased by subgradient information."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    cur_J = best_J
    rng = np.random.default_rng(777)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        T = T0 * (T_end / T0) ** frac

        # Compute gradient info for proposal biasing
        conv = fast_conv(a)
        M = conv.max()
        n_active = max(1, int(n * 0.05))
        sorted_idx = np.argsort(conv)[::-1]
        active_indices = sorted_idx[:n_active]
        sg = subgradient_via_fft(a, active_indices)

        # Choose coordinate: biased toward high subgradient (more impactful)
        weights = np.abs(sg) + 0.1
        weights /= weights.sum()
        i = rng.choice(n, p=weights)

        old_val = a[i]

        # Propose new value: biased to decrease where sg is positive
        direction = -np.sign(sg[i]) if abs(sg[i]) > 0.01 else (1 if rng.random() > 0.5 else -1)
        step_size = T * (1.0 + np.abs(sg[i]))
        delta = direction * rng.exponential(step_size) + rng.normal(0, step_size * 0.5)
        new_val = max(0.0, old_val + delta)

        a[i] = new_val
        a_pos = np.maximum(a, 0)
        s = a_pos.sum()
        if s < 0.01:
            a[i] = old_val
            continue
        a_candidate = a_pos / s
        new_J = compute_J(a_candidate)

        # Metropolis-Hastings acceptance
        delta_J = new_J - cur_J
        if delta_J < 0 or rng.random() < np.exp(-delta_J / max(T, 1e-12)):
            a = a_candidate
            cur_J = new_J
            if new_J < best_J - 1e-14:
                best_J = new_J
                best_a = a.copy()
        else:
            a[i] = old_val

    return best_a


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random
    r = rng.uniform(0.3, 1.7, n)
    inits.append(r / r.sum())

    # Cosine
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(np.maximum(cw, 0.01) / np.maximum(cw, 0.01).sum())

    # Two bumps
    db = np.exp(-6 * (t - 0.3) ** 2) + np.exp(-6 * (t - 0.7) ** 2) + 0.05
    inits.append(db / db.sum())

    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm, 0.01)
    inits.append(sm / sm.sum())

    # Frequency modulated
    for freq in [2, 3, 5, 8]:
        fm = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    # Quasi-periodic (golden ratio)
    golden = (1 + np.sqrt(5)) / 2
    qp = 0.5 + 0.4 * np.cos(2 * np.pi * golden * np.arange(n) / n * 7)
    qp = np.maximum(qp, 0.01)
    inits.append(qp / qp.sum())

    # Power law
    pl = (t + 0.05) ** 0.7 + 0.1
    inits.append(pl / pl.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
