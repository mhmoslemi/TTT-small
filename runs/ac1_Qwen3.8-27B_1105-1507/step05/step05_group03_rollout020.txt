        return float('inf')
    b = fft_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Returns (smooth_max_value, gradient).
    """
    n = len(a)
    b = fft_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # shifted has max 0, so no overflow
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    sm_val = b_max + log_sum / tau

    # p_k = softmax of tau*(b - b_max)
    p = exp_shifted / exp_shifted.sum()

    # Gradient: grad[i] = 2 * sum_j a[j] * p[i+j]
    # = 2 * (reverse(a) * p)[i + n - 1]
    a_rev = a[::-1]
    conv_rp = cross_conv(a_rev, p)
    # conv_rp has length n + (2n-1) - 1 = 3n-2
    # grad[i] = 2 * conv_rp[i + n - 1]
    grad = 2.0 * conv_rp[n - 1:2 * n - 1]

    return sm_val, grad, p, b


def objective_and_gradient(a, tau, n):
    """
    Compute J_smooth(a) = 2*n*smooth_max(conv(a,a))/sum(a)^2 and its gradient.
    """
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)

    sm_val, grad_sm, p, b = smooth_max_and_grad(a, tau)

    # J = 2*n*sm_val / s^2
    J = 2.0 * n * sm_val / (s * s)

    # Gradient: dJ/da = 2*n * [grad_sm / s^2 - 2*sm_val * 1 / s^3]
    # = (2*n / s^2) * grad_sm - (4*n*sm_val / s^3) * ones
    grad_J = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * sm_val / (s * s * s)) * np.ones(n)

    return J, grad_J


def build_lp_matrix(n):
    """Precompute row/col indices for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 15, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Analytic line search for convex combination direction."""
    A = fft_conv(a)
    B = cross_conv(a, g)
    C = fft_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = float(fft_conv(a).max())
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = float(fft_conv(a).max())
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-9:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = float(fft_conv(a).max())
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 150:
            break

        if verbose and it % 100 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def run_lbfgsb(a_init, n, tau_schedule, max_time, deadline):
    """
    Run L-BFGS-B with smooth max objective, increasing tau.
    a_init: initial non-negative vector
    tau_schedule: list of tau values to try in sequence
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s * np.sqrt(n)  # scale so elements are O(1)

    best_a = a.copy()
    best_J = compute_J(a)

    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break
        if time.monotonic() > max_time:
            break

        def fun(x):
            x = np.maximum(x, 0)
            if x.sum() < 1e-10:
                return 1e10
            val, grad, _, _ = objective_and_gradient(x, tau, n)
            return val

        def jac(x):
            x = np.maximum(x, 0)
            if x.sum() < 1e-10:
                return np.zeros(n)
            _, grad, _, _ = objective_and_gradient(x, tau, n)
            return grad

        iterations = 0

        def callback(xk):
            nonlocal iterations
            iterations += 1
            xk_pos = np.maximum(xk, 0)
            if xk_pos.sum() > 0.01:
                j_val = compute_J(xk_pos)
                if j_val < best_J:
                    best_J = j_val
                    best_a = xk_pos.copy()

        bounds = [(0, None)] * n
        try:
            result = minimize(fun, a, jac=jac, method='L-BFGS-B',
                              bounds=bounds, callback=callback,
                              options={'maxiter': 300, 'maxfun': 2000,
                                       'ftol': 1e-12, 'gtol': 1e-8})
        except Exception as e:
            print(f"      L-BFGS-B error at tau={tau}: {e}", flush=True)
            continue

        a = np.maximum(result.x, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s * np.sqrt(n)
            j_val = compute_J(a)
            if j_val < best_J:
                best_J = j_val
                best_a = a.copy()

        if time.monotonic() > deadline:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_tapered(n, alpha=0.5):
    """Create a tapered (raised cosine) sequence."""
    x = np.linspace(0, np.pi, n)
    return np.maximum(0.5 + alpha * np.cos(x), 0.01)


def make_random_positive(n, seed=0):
    """Create a random positive sequence with some structure."""
    rng = np.random.default_rng(seed)
    # Mix of low-frequency and high-frequency components
    base = np.ones(n)
    for freq in [1, 2, 3, 5]:
        base += 0.3 * np.cos(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    base += 0.1 * rng.standard_normal(n)
    return np.maximum(base, 0.01)


class Tracker:
    """Tracks best sequence found."""
    def __init__(self):
        self.best = None
        self.best_score = float('inf')

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} {label}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining L-BFGS-B, LP mirror descent,
    multi-length exploration, and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    # ===== Phase 1: L-BFGS-B on parent length =====
    n0 = parent.size
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: L-BFGS-B n={n0}", flush=True)
    print(f"{'='*60}", flush=True)

    tau_schedule_p1 = [5, 10, 20, 50, 100, 200]
    phase1_deadline = min(t_start + budget_s * 0.25, deadline)
    a_opt = run_lbfgsb(parent, n0, tau_schedule_p1, budget_s * 0.2, phase1_deadline)
    tracker.update(a_opt, "(P1 L-BFGS-B)")

    # LP polish
    row_idx, col_idx = build_lp_matrix(n0)
    a_lp = a_opt.copy()
    a_lp = a_lp / a_lp.sum()
    phase1b_deadline = min(t_start + budget_s * 0.35, deadline)
    a_lp = run_lp_mirror_descent(a_lp, 20000, phase1b_deadline, row_idx, col_idx, verbose=True)
    tracker.update(a_lp, "(P1 LP polish)")

    # ===== Phase 2: Try larger lengths =====
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    trial_lengths = []
    for n_trial in [768, 1024, 1536, 2048]:
        if n_trial != n0:
            trial_lengths.append(n_trial)

    phase2_deadline = min(t_start + budget_s * 0.60, deadline)
    for n_trial in trial_lengths:
        if time.monotonic() > phase2_deadline:
            break

        print(f"  Trying n={n_trial}...", flush=True)
        a_init = resample_seq(tracker.best if tracker.best is not None else parent, n_trial)
        a_init = np.maximum(a_init, 1e-8)

        tau_sched = [5, 10, 30, 80, 150]
        sub_deadline = min(time.monotonic() + budget_s * 0.12, phase2_deadline)
        a_opt = run_lbfgsb(a_init, n_trial, tau_sched, budget_s * 0.1, sub_deadline)
        tracker.update(a_opt, f"(P2 L-BFGS-B n={n_trial})")

        # Quick LP polish
        r_idx, c_idx = build_lp_matrix(n_trial)
        a_lp = a_opt.copy()
        a_lp = a_lp / a_lp.sum()
        sub_dl2 = min(time.monotonic() + 60, phase2_deadline)
        a_lp = run_lp_mirror_descent(a_lp, 10000, sub_dl2, r_idx, c_idx, verbose=False)
        tracker.update(a_lp, f"(P2 LP n={n_trial})")

    # ===== Phase 3: Diverse initializations at best length =====
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Diverse Initializations", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
    else:
        n_best = n0

    r_idx_b, c_idx_b = build_lp_matrix(n_best)
    phase3_deadline = min(t_start + budget_s * 0.80, deadline)

    initializations = [
        ("tapered", make_tapered(n_best, 0.5)),
        ("tapered2", make_tapered(n_best, 0.8)),
        ("random1", make_random_positive(n_best, 123)),
        ("random2", make_random_positive(n_best, 456)),
        ("parent_perturbed", None),  # will be filled
    ]

    # Add perturbed parent
    if tracker.best is not None:
        pert = tracker.best.copy().astype(np.float64) / tracker.best.sum()
        pert += rng.normal(0, 0.02, n_best)
        pert = np.maximum(pert, 1e-6)
        initializations[4] = ("parent_perturbed", pert)

    for name, a_init in initializations:
        if time.monotonic() > phase3_deadline:
            break
        if a_init is None:
            continue

        print(f"  Init: {name} ...", flush=True)
        a_init = np.maximum(a_init, 0.0)
        if a_init.sum() < 0.01:
            a_init = np.ones(n_best)

        tau_sched = [10, 30, 80, 150]
        sub_dl = min(time.monotonic() + budget_s * 0.06, phase3_deadline)
        a_opt = run_lbfgsb(a_init, n_best, tau_sched, budget_s * 0.05, sub_dl)
        tracker.update(a_opt, f"(P3 {name})")

        # Quick LP
        a_lp = a_opt.copy() / a_opt.sum()
        sub_dl2 = min(time.monotonic() + 30, phase3_deadline)
        a_lp = run_lp_mirror_descent(a_lp, 5000, sub_dl2, r_idx_b, c_idx_b, verbose=False)
        tracker.update(a_lp, f"(P3 {name} LP)")

    # ===== Phase 4: Perturbation restarts on best =====
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        if n_best != len(r_idx_b):
            r_idx_b, c_idx_b = build_lp_matrix(n_best)

        phase4_deadline = min(t_start + budget_s * 0.92, deadline)
        temperature = 0.05

        for r in range(20):
            if time.monotonic() > phase4_deadline:
                break

            a = tracker.best.copy().astype(np.float64)
            a = a / a.sum()
            noise = rng.normal(0, temperature * 0.01, n_best)
            a = a + noise
            a = np.maximum(a, 1e-8)
            a = a / a.sum()

            # L-BFGS-B quick
            tau_sched = [30, 80, 150]
            sub_dl = min(time.monotonic() + budget_s * 0.02, phase4_deadline)
            a_opt = run_lbfgsb(a, n_best, tau_sched, budget_s * 0.015, sub_dl)
            tracker.update(a_opt, f"(P4 r={r} T={temperature:.4f})")

            # LP polish
            a_lp = a_opt.copy() / a_opt.sum()
            sub_dl2 = min(time.monotonic() + 20, phase4_deadline)
            a_lp = run_lp_mirror_descent(a_lp, 3000, sub_dl2, r_idx_b, c_idx_b, verbose=False)
            tracker.update(a_lp, f"(P4 r={r} LP)")

            temperature *= 0.85

    # ===== Phase 5: Final extended polish =====
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)

        # Extended LP mirror descent
        a = tracker.best.copy()
        a = a / a.sum()
        a_opt = run_lp_mirror_descent(a, 50000, deadline, r_f, c_f, verbose=True)
        tracker.update(a_opt, "(P5 LP final)")

        # One more L-BFGS-B pass with high tau
        if time.monotonic() < deadline - 30:
            tau_final = [100, 200, 500]
            a_opt2 = run_lbfgsb(a_opt, n_final, tau_final, 60, deadline)
            tracker.update(a_opt2, "(P5 L-BFGS-B final)")

            # Final LP
            a_lp = a_opt2.copy() / a_opt2.sum()
            a_lp = run_lp_mirror_descent(a_lp, 10000, deadline, r_f, c_f, verbose=False)
            tracker.update(a_lp, "(P5 LP final2)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
