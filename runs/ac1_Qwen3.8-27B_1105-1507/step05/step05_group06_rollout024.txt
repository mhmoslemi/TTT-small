        base_eta = 1.0 / (T * g_norm + 1e-10)

        improved = False
        for mult in [2.0, 1.0, 0.5, 0.25, 0.125, 0.0625]:
            eta = base_eta * mult
            a_new = a * np.exp(-eta * v)
            a_new = np.maximum(a_new, 1e-15)
            s = a_new.sum()
            if s < 0.01:
                continue
            a_new = a_new / s
            M_new = float(fast_conv(a_new).max())
            if M_new < actual_M:
                a = a_new
                actual_M = M_new
                improved = True
                break

        it += 1

        # Periodic evaluation
        if it % 20 == 0:
            J = compute_J(a)
            if J < best_J - 1e-13:
                best_J = J
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Anneal temperature
        if it % 100 == 0:
            T_min = T_end ** 0.5
            T = min(T * 1.08, T_end)
            # Reduce momentum as T increases (less smooth -> less momentum)
            if T > T_min:
                momentum = max(0.5, momentum - 0.002)

        # Stagnation: perturb and reduce T
        if no_improve > patience:
            a = np.maximum(a + np.random.normal(0, 0.02, n), 1e-6)
            a = a / a.sum()
            v = np.zeros(n)
            T = max(T * 0.5, T_start)
            no_improve = 0
            patience = max(200, patience // 2)

    return best_a, best_J


def coordinate_descent(a, deadline, max_passes=50):
    """
    Fast coordinate descent with incremental convolution updates.
    For each coordinate, tries several candidate values and picks the best.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = np.random.permutation(n)
        c = fast_conv(a)
        S = a.sum()
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]

            # Determine candidate values
            if ai < 1e-6:
                candidates = [0.0, 0.005, 0.01, 0.02, 0.05]
            else:
                candidates = [0.0, ai * 0.5, ai * 0.75, ai * 1.0,
                              ai * 1.25, ai * 1.5, ai * 2.0, ai * 3.0]

            best_candidate = ai
            best_score = 2.0 * n * c.max() / (S * S)

            for cand in candidates:
                delta = cand - ai
                if abs(delta) < 1e-14:
                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue

                # Compute new max efficiently
                k_lo = max(0, i)
                k_hi = min(i + n - 1, 2 * n - 2)

                # Outside max
                if k_lo > 0 and k_hi < 2 * n - 2:
                    outside = max(c[:k_lo].max(), c[k_hi + 1:].max())
                elif k_lo == 0:
                    outside = c[k_hi + 1:].max() if k_hi < 2 * n - 2 else 0.0
                else:
                    outside = c[:k_lo].max()

                # Inside max (affected by delta)
                inside_max = 0.0
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        new_val = c[k] + 2.0 * delta * ai + delta * delta
                    else:
                        j = k - i
                        if 0 <= j < n:
                            new_val = c[k] + 2.0 * delta * a[j]
                        else:
                            new_val = c[k]
                    if new_val > inside_max:
                        inside_max = new_val

                new_max = max(outside, inside_max)
                score = 2.0 * n * new_max / (S_new * S_new)

                if score < best_score:
                    best_score = score
                    best_candidate = cand

            if abs(best_candidate - ai) > 1e-12:
                delta = best_candidate - ai
                a[i] = best_candidate
                improved = True
                S = S + delta
                # Update c incrementally
                k_lo = max(0, i)
                k_hi = min(i + n - 1, 2 * n - 2)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        c[k] += 2.0 * delta * best_candidate + delta * delta - 2.0 * delta * ai - delta * delta
                        # Actually: old c[k] had contribution 2*ai*ai from this pair; new has 2*best_candidate*best_candidate
                        # Simpler: just recompute affected
                    else:
                        j = k - i
                        if 0 <= j < n:
                            c[k] += 2.0 * delta * a[j]
                # Handle k=2i properly
                if 0 <= 2 * i <= 2 * n - 2:
                    c[2 * i] = c[2 * i]  # already updated above, but let's be safe
                    # Recompute: c[2i] should include a[i]^2 term
                    # This is getting error-prone; just recompute c periodically

        # Recompute c to avoid drift
        c = fast_conv(a)
        S = a.sum()
        if S > 0.01:
            a = a / S
            J = compute_J(a)
            if J < best_J - 1e-13:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def coordinate_descent_recompute(a, deadline, max_passes=30):
    """
    Coordinate descent that recomputes convolution each pass (slower but no drift).
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = a.sum()
        order = np.random.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            if ai < 1e-8:
                candidates = [0.002, 0.008, 0.02, 0.05]
            else:
                candidates = [0.0, ai * 0.4, ai * 0.7, ai * 1.3, ai * 1.7, ai * 2.5]

            best_cand = ai
            best_val = float('inf')

            for cand in candidates:
                a[i] = cand
                a_pos = np.maximum(a, 0)
                s_check = a_pos.sum()
                if s_check < 0.01:
                    continue
                a_norm = a_pos / s_check
                c_check = fast_conv(a_norm)
                val = 2.0 * n * c_check.max() / (1.0)  # sum=1 so J = 2n*max(c)
                if val < best_val:
                    best_val = val
                    best_cand = cand

            if best_cand != ai:
                a[i] = best_cand
                improved = True

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J - 1e-13:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def simulated_annealing_polish(a, deadline, max_iters=20000, T_start=0.02, T_end=0.001):
    """
    SA with coordinate perturbations for final polishing.
    """
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    current_J = best_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        T = T_start * (T_end / T_start) ** frac

        # Pick 1-3 random coordinates to perturb
        n_perturb = 1 + (1 if np.random.random() < 0.3 else 0)
        indices = np.random.choice(n, n_perturb, replace=False)

        old_vals = a[indices].copy()
        for idx in indices:
            old = a[idx]
            if old < 0.02:
                new_val = np.random.exponential(0.03)
            else:
                new_val = np.random.normal(old, 0.1 * old + 0.005)
            a[idx] = max(0.0, new_val)

        a_new = np.maximum(a, 0)
        s = a_new.sum()
        if s < 0.01:
            a[indices] = old_vals
            continue
        a_new = a_new / s
        new_J = compute_J(a_new)

        delta = new_J - current_J
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            current_J = new_J
            if new_J < best_J - 1e-13:
                best_J = new_J
                best_a = a_new.copy()
        else:
            a[indices] = old_vals

    return best_a


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations for restarts."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())

    # Sine squared
    ss = np.sin(np.pi * t) ** 2 + 0.01
    inits.append(ss / ss.sum())

    # Double bump
    dbl = np.exp(-10 * (t - 0.25) ** 2) + np.exp(-10 * (t - 0.75) ** 2) + 0.05
    inits.append(dbl / dbl.sum())

    # Frequency modulated
    for freq in [2, 4, 8]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())

    # Random (less smooth)
    r = rng.uniform(0.2, 2.0, n)
    inits.append(r / r.sum())

    # Triangular
    tri = 1.0 - np.abs(2 * t - 1) + 0.05
    inits.append(tri / tri.sum())

    # Parent perturbations
    if parent is not None:
        for noise_level in [0.02, 0.05, 0.1, 0.2]:
            p = np.maximum(parent + rng.normal(0, noise_level, n), 1e-4)
            inits.append(p / p.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Smoothed max mirror descent from parent
    Phase 2: Coordinate descent polishing
    Phase 3: Diverse restarts with smoothed max
    Phase 4: SA polishing + more CD
    Phase 5: Final combined polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    hard_deadline = t_start + budget_s - 5.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        n_default = 512
        parent = np.ones(n_default) / n_default
        update_best(parent)
        print(f"Using default parent: n={parent.size}, J={best_score:.8f}", flush=True)

    n_base = parent.size

    # === PHASE 1: Smoothed max mirror descent from parent (0-450s) ===
    p1_deadline = t_start + 450
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Smoothed Max Mirror Descent (n={n_base})", flush=True)
    print(f"{'=' * 60}", flush=True)

    a1 = parent.copy() / parent.sum()
    a1, J1 = smoothed_max_mirror_descent(a1, p1_deadline,
                                          T_start=2.0, T_end=300.0,
                                          momentum=0.7, topK_start=16,
                                          topK_end=2 * n_base - 1)
    update_best(a1)
    print(f"  Phase 1 done: J={J1:.8f}", flush=True)

    # === PHASE 2: Coordinate descent polishing (450-650s) ===
    p2_deadline = t_start + 650
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Coordinate Descent Polishing", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        a_cd = coordinate_descent(best_seq, p2_deadline, max_passes=40)
        update_best(a_cd)
        a_cd2 = coordinate_descent_recompute(a_cd, min(p2_deadline + 50, hard_deadline - 400), max_passes=20)
        update_best(a_cd2)
        print(f"  Phase 2 done: best J={best_score:.8f}", flush=True)

    # === PHASE 3: Diverse restarts (650-1050s) ===
    p3_deadline = t_start + 1050
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Diverse Restart Optimization", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        inits = generate_diverse_inits(n_base, rng, parent=best_seq)
        # Also add best_seq itself as a restart point
        inits.append(best_seq.copy() / best_seq.sum())

        # Score initializations
        init_scores = []
        for init in inits:
            init = np.maximum(init, 0)
            s = init.sum()
            if s > 0.01:
                init = init / s
                J_init = compute_J(init)
                init_scores.append((J_init, init))

        init_scores.sort(key=lambda x: x[0])

        # Try top initializations
        tried = 0
        for J_init, init in init_scores:
            if time.monotonic() > p3_deadline:
                break
            if tried >= 8:
                break
            tried += 1

            phase_dl = min(time.monotonic() + 40, p3_deadline)
            print(f"  Trying init {tried}: J_init={J_init:.6f}", flush=True)

            a_r, J_r = smoothed_max_mirror_descent(init, phase_dl,
                                                     T_start=5.0, T_end=400.0,
                                                     momentum=0.6, topK_start=10,
                                                     topK_end=2 * n_base - 1)
            update_best(a_r)

            # Quick CD on result
            phase_dl2 = min(time.monotonic() + 15, p3_deadline)
            a_cd = coordinate_descent(a_r, phase_dl2, max_passes=10)
            update_best(a_cd)

        print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)

    # === PHASE 4: SA + CD on best (1050-1300s) ===
    p4_deadline = t_start + 1300
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: SA Polishing", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        a_sa = simulated_annealing_polish(best_seq, p4_deadline, max_iters=30000,
                                           T_start=0.015, T_end=0.0005)
        update_best(a_sa)
        # CD after SA
        a_cd = coordinate_descent(a_sa, min(p4_deadline + 50, hard_deadline - 150), max_passes=15)
        update_best(a_cd)
        print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Final combined polish (1300-1495s) ===
    p5_deadline = hard_deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Final Polishing", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        # One more smoothed max pass with high T (focused)
        a_f = smoothed_max_mirror_descent(best_seq, min(t_start + 1420, p5_deadline),
                                           T_start=50.0, T_end=500.0,
                                           momentum=0.5, topK_start=5,
                                           topK_end=2 * n_base - 1)
        update_best(a_f)

        # Final CD
        a_cd = coordinate_descent(a_f, min(t_start + 1460, p5_deadline), max_passes=20)
        update_best(a_cd)

        # Tiny SA
        a_sa = simulated_annealing_polish(a_cd, p5_deadline, max_iters=5000,
                                           T_start=0.005, T_end=0.0002)
        update_best(a_sa)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
