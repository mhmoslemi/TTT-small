def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients.
    Returns np.inf if the input is invalid."""
    # Security checks - sequence type and length (above list check is assumed in calling code)
    n = len(sequence)
    if n == 0:
        return np.inf
    
    # Check element validity
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)) or np.isnan(x) or np.isinf(x):
            return np.inf
        if x < 0:
            return np.inf
    
    # Convert to float and clip to non-negative
    a = np.array(sequence, dtype=float)
    a = np.clip(a, 0.0, 1000.0)
