Actually, the cleanest approach: work in the "log-simplex" parameterization. Let a_i = exp(θ_i) / sum(exp(θ_j)). Then optimize over θ. This automatically maintains a >= 0 and sum(a) = 1.

- At each step, the LP gives a direction g.
- Update: a <- normalize((1-t)*a + t*g)
- This maintains a >= 0 and sum = 1 (after normalization).

For the coordinate descent, I'll do:
- Pick random i, j with i ≠ j
- Transfer α from j to i: a[i] += α, a[j] -= α (ensuring a[j] >= 0)
- If J decreases, accept; otherwise reject
- This is like a Markov chain Monte Carlo (Metropolis) step

- Pick random i
- Compute the "subgradient" at i: s_i = sum_{k in active} w_k * a[k-i]

            else:
                no_improve = max(0, no_improve - 1)
        
        # Perturbation to escape local minima
        if no_improve > max_no_improve:
            rng = np.random.default_rng(it * 31337)
            scale = 0.1 / (1.0 + no_improve / max_no_improve)
            # Smooth perturbation
            freqs = rng.integers(1, 4, size=2)
            phases = rng.uniform(0, 2 * np.pi, size=2)
            t_arr = np.linspace(0, 1, n, endpoint=False)
            pert = np.zeros(n)
            for f, ph in zip(freqs, phases):
                pert += np.cos(2 * np.pi * f * t_arr + ph)
            a = a + scale * pert
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            no_improve = 0
        
        if verbose and it % 2000 == 0:
            avg_t = total_lp_time / it * 1000
            print(f"    iter={it} J={best_J:.8f} M={M:.6f} n_act={n_active} "
                  f"lp_avg={avg_t:.1f}ms", flush=True)
        
        if it > 200000:
            break
    
    return best_a, best_J


def coordinate_descent_polish(a, deadline, n_trials=100000, verbose=False):
    """Coordinate descent: try swapping mass between pairs of coordinates."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    rng = np.random.default_rng(12345)
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        if trial % 5000 == 0 and verbose:
            print(f"    CD trial={trial} J={best_J:.8f}", flush=True)
        
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try transferring mass from j to i
        alpha_candidates = [a[j] * 0.01, a[j] * 0.05, a[j] * 0.1,
                            (a[j] - a[i]) * 0.1 if a[j] > a[i] else a[i] * -0.05,
                            a[j] * 0.001]
        
        improved = False
        for alpha in alpha_candidates:
            if alpha == 0:
                continue
            if a[j] + alpha < -1e-12 or a[i] - alpha < -1e-12:
                continue
            # Transfer: a[i] increases by alpha, a[j] decreases by alpha
            a[i] -= alpha
            a[j] += alpha
            if a[i] < 0 or a[j] < 0:
                a[i] += alpha
                a[j] -= alpha
                continue
            
            J_trial = compute_J(a)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a.copy()
                improved = True
                break
            else:
                a[i] += alpha
                a[j] -= alpha
        
        if not improved and trial % 10000 == 0:
            # Larger random perturbation
            delta = 0.02 * a
            a = a + delta * rng.normal(0, 1, n)
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
    
    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP mirror descent, smooth gradient, coordinate descent, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None or len(a) == 0:
            return
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = compute_J(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={len(a)} t={elapsed:.0f}s [{label}]", flush=True)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            parent = p
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512)
    
    print(f"Parent: n={len(parent)}, J={compute_J(parent):.8f}", flush=True)
    try_update(parent, "parent")
    
    # === PHASE 1: Multi-n initialization (120s) ===
    print("\n" + "=" * 60, flush=True)
    print("PHASE 1: Multi-n Initialization", flush=True)
    print("=" * 60, flush=True)
    
    p1_deadline = time.monotonic() + 120.0
    n_values = [512, 1024]
    
    candidates = {}
    for n_try in n_values:
        if time.monotonic() > p1_deadline:
            break
        a_init = resample_seq(parent, n_try)
        a_init = np.maximum(a_init, 1e-6)
        a_init = a_init / a_init.sum()
        
        # Quick smooth gradient
        a_opt = smooth_gradient_descent(a_init, min(time.monotonic() + 30, p1_deadline),
                                         tau=25.0, lr=0.04, max_iters=2000)
        J_val = compute_J(a_opt)
        candidates[n_try] = a_opt
        print(f"  n={n_try}: J={J_val:.8f} (smooth grad)", flush=True)
        try_update(a_opt, f"init_n{n_try}")
    
    # === PHASE 2: Main LP optimization on best n (850s) ===
    print("\n" + "=" * 60, flush=True)
    print("PHASE 2: LP Mirror Descent", flush=True)
    print("=" * 60, flush=True)
    
    p2_deadline = time.monotonic() + 850.0
    
    # Sort candidates by quality
    sorted_cands = sorted(candidates.items(), key=lambda x: compute_J(x[1]))
    
    # Allocate time: best gets 60%, second gets 40% (if exists)
    if len(sorted_cands) >= 2:
        time_splits = [0.6, 0.4]
    elif len(sorted_cands) == 1:
        time_splits = [1.0]
    else:
        time_splits = []
    
    for ci, (n_cand, a_cand) in enumerate(sorted_cands):
        if time.monotonic() > p2_deadline:
            break
        remaining = p2_deadline - time.monotonic()
        alloc = remaining * time_splits[ci]
        alloc = max(alloc, 60.0)
        phase_dl = min(time.monotonic() + alloc, p2_deadline)
        
        print(f"\n  n={n_cand} (J={compute_J(a_cand):.8f}), budget={alloc:.0f}s", flush=True)
        
        # Run LP mirror descent
        a_res, J_res = lp_mirror_descent(a_cand, phase_dl,
                                          max_no_improve=400, top_frac=0.25,
                                          verbose=True)
        try_update(a_res, f"lp_n{n_cand}")
    
    # === PHASE 3: Multi-n exploration (250s) ===
    print("\n" + "=" * 60, flush=True)
    print("PHASE 3: Multi-n Exploration", flush=True)
    print("=" * 60, flush=True)
    
    p3_deadline = min(time.monotonic() + 250.0, deadline - 100.0)
    
    if best_seq is not None:
        base_n = len(best_seq)
        explored_ns = set(candidates.keys())
        
        for n_try in [base_n // 2, base_n + base_n // 4, base_n * 2, 2048, 768, 640]:
            if time.monotonic() > p3_deadline:
                break
            if n_try < 128 or n_try > 4096:
                continue
            if n_try in explored_ns:
                continue
            explored_ns.add(n_try)
            
            a_ext = resample_seq(best_seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            
            # Smooth gradient
            a_ext = smooth_gradient_descent(a_ext, min(time.monotonic() + 20, p3_deadline - 30),
                                             tau=20.0, lr=0.03, max_iters=1000)
            try_update(a_ext, f"expl_n{n_try}_grad")
            
            # Short LP
            if time.monotonic() < p3_deadline - 15:
                a_ext2, _ = lp_mirror_descent(a_ext, min(time.monotonic() + 45, p3_deadline - 5),
                                               max_no_improve=150, top_frac=0.3, verbose=False)
                try_update(a_ext2, f"expl_n{n_try}_lp")
    
    # === PHASE 4: Coordinate descent polishing (100s) ===
    print("\n" + "=" * 60, flush=True)
    print("PHASE 4: Coordinate Descent Polishing", flush=True)
    print("=" * 60, flush=True)
    
    p4_deadline = min(time.monotonic() + 100.0, deadline - 40.0)
    
    if best_seq is not None and time.monotonic() < p4_deadline:
        a_cd, J_cd = coordinate_descent_polish(best_seq, p4_deadline,
                                                n_trials=200000, verbose=True)
        try_update(a_cd, "coord_desc")
    
    # === PHASE 5: Final full-LP polish (remaining time) ===
    print("\n" + "=" * 60, flush=True)
    print("PHASE 5: Final Polishing", flush=True)
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s, current best: J={best_score:.8f}", flush=True)
    print("=" * 60, flush=True)
    
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Full LP polish (all constraints)
        a_final, _ = lp_mirror_descent(best_seq, deadline - 10,
                                        max_no_improve=1000, top_frac=1.0, verbose=True)
        try_update(a_final, "full_lp_polish")
        
        # One more coordinate descent pass
        if time.monotonic() < deadline - 10:
            a_cd2, _ = coordinate_descent_polish(best_seq, deadline - 5,
                                                  n_trials=50000, verbose=False)
            try_update(a_cd2, "final_cd")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = compute_J(best_seq)
    
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={len(best_seq)} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)
    
    return [float(x) for x in best_seq]
