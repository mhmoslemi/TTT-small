    k_hi = i + n - 1

    slopes[linear_mask] = 2 * a[js[linear_mask]]

To reduce b_{k*}, we want to decrease the "contributing" a_j's. The most impactful ones are those with large a_j * a_{k*-j}.

A simple heuristic:
1. Find k* = argmax b_k
2. Find the top m contributors: (j, k*-j) pairs with largest a_j * a_{k*-j}
3. For each such pair, try decreasing a_j (or a_{k*-j}) and see if the overall max decreases
4. Use line search to find the optimal decrease

3. Use adaptive step size

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using an optimized Gauss-Seidel
coordinate descent with in-place convolution updates, Riemannian Adam on
log-sum-exp smoothing, peak-chasing gradient steps, and multi-length exploration.
Key innovations: (1) O(n) in-place convolution updates avoid FFT per coordinate,
(2) exact 1D subproblem solution via golden-section on the piecewise structure,
(3) peak-focused gradient descent targeting top-k autocorrelation peaks.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (max(size, 1) - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def solve_1d(a, b, i, S, old, n_iters=35):
    """Solve 1D subproblem: find optimal x for coordinate i."""
    n = len(a)
    k_lo = i
    k_hi = i + n - 1

    # Unaffected max
    if k_lo > 0:
        bm_parts = [b[:k_lo]]
    else:
        bm_parts = []
    if k_hi < 2 * n - 2:
        bm_parts.append(b[k_hi + 1:])
    if bm_parts:
        base_max = max(np.max(p) for p in bm_parts)
    else:
        base_max = 0.0

    ks = np.arange(k_lo, k_hi + 1)
    js = ks - i

    quad_mask = (ks == 2 * i)
    linear_mask = ~quad_mask

    line_c = b[ks[linear_mask]] - 2.0 * old * a[js[linear_mask]]
    line_s = 2.0 * a[js[linear_mask]]

    if quad_mask.any():
        quad_c = b[2 * i] - old * old
    else:
        quad_c = -1e30

    def obj(x):
        new_S = S - old + x
        if new_S <= 1e-15:
            return 1e15
        lin_max = np.max(line_c + line_s * x) if len(line_c) > 0 else 0.0
        quad_val = quad_c + x * x
        m = base_max
        if lin_max > m:
            m = lin_max
        if quad_val > m:
            m = quad_val
        return m / (new_S * new_S)

    lo = 0.0
    hi = max(old * 3.0, np.sqrt(max(base_max, 0.0) + 1.0) + 1.0, 0.5)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)

    for _ in range(n_iters):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)

def update_b_inplace(b, a, i, old, new):
    """Update convolution in-place after changing a[i] from old to new."""
    n = len(a)
    if abs(new - old) < 1e-15:
        return
    delta = new - old
    for k in range(i, i + n):
        j = k - i
        if k == 2 * i:
            b[k] += new * new - old * old
        else:
            b[k] += 2.0 * delta * a[j]

def coordinate_descent(a, max_sweeps, deadline, shuffle=False, seed=0, recompute_every=15):
    """Fast Gauss-Seidel coordinate descent with in-place b updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    b = fast_conv(a)

    best_a = a.copy()
    best_M = float(b.max())
    rng = np.random.default_rng(seed)
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if shuffle:
            rng.shuffle(order)

        if sweep > 0 and sweep % recompute_every == 0:
            S = a.sum()
            if S > 1e-10:
                scale = 1.0 / S
                a *= scale
                b = fast_conv(a)

        S = a.sum()
        if S < 1e-10:
            break

        for i in order:
            old = a[i]
            x_opt = solve_1d(a, b, i, S, old, n_iters=30)
            if abs(x_opt - old) > 1e-13:
                a[i] = x_opt
                update_b_inplace(b, a, i, old, x_opt)

        S_val = a.sum()
        if S_val < 1e-10:
            break
        a /= S_val
        b *= (1.0 / S_val) ** 2

        M = float(b.max())
        if M < best_M * (1.0 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 25:
            break

    return best_a

def smooth_max_grad(a, tau):
    """Gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2 * pk * a[k-hi : k-lo+1][::-1]
        segment = a[lo:hi + 1]
        grad[lo:hi + 1] += 2.0 * pk * segment[::-1]

    return grad

def riemannian_adam(a, max_iters, deadline, tau_start=5.0, tau_end=100.0, lr=0.05):
    """Riemannian Adam on the simplex for smooth max objective."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = float(fast_conv(a).max())

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        lr_t = lr / (1.0 + 0.05 * it)

        grad = smooth_max_grad(a, tau)
        grad -= grad.mean()  # Project to tangent space of simplex

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new

        M = float(fast_conv(a).max())
        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()

    return best_a

def peak_chase(a, max_iters, deadline, top_k=3):
    """Focused gradient steps targeting the top autocorrelation peaks."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = float(fast_conv(a).max())

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Find top-k peaks
        sorted_idx = np.argsort(b)[::-1]
        top_indices = sorted_idx[:top_k]
        peak_vals = b[top_indices]
        max_peak = peak_vals[0]

        # Weighted gradient: weights based on proximity to max
        weights = np.exp(-2.0 * (max_peak - peak_vals))
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                other = k - j
                if 0 <= other < n:
                    grad[j] += 2.0 * w * a[other]

        grad -= grad.mean()
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        # Adaptive step size with line search
        best_step = 0.0
        best_trial_M = M
        for step in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001]:
            a_trial = a - step * grad / grad_norm
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = float(fast_conv(a_trial).max())
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_step = step

        if best_step == 0.0:
            break

        a = a - best_step * grad / grad_norm
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s

        M = float(fast_conv(a).max())
        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()

    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.001))
        # Symmetric perturbation
        p_sym = (p + p[::-1]) / 2.0
        inits.append(np.maximum(p_sym, 0.001))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.5, 1.0, n))

    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))

    # Triangular
    inits.append(np.maximum(1.0 - np.abs(2 * t - 1), 0.01))
    # Gaussian bump
    inits.append(np.exp(-6 * (t - 0.5) ** 2) + 0.05)
    # Two bumps
    inits.append(np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2) + 0.02)

    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Fast coordinate descent (Gauss-Seidel with in-place conv updates)
    Phase 2: Riemannian Adam on log-sum-exp smoothing
    Phase 3: Peak-chasing gradient steps
    Phase 4: Multi-length exploration
    Phase 5: Final coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    # === PHASE 1: Coordinate Descent (main engine) ===
    p1_deadline = t0 + 550.0
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Coordinate Descent (budget~550s)", flush=True)
    print(f"{'='*60}", flush=True)

    n_parent = parent.size
    lengths_p1 = sorted(set([max(256, n_parent // 2), n_parent, min(1024, n_parent * 2)]))

    for n in lengths_p1:
        if time.monotonic() > p1_deadline - 30:
            break
        phase_dl = min(time.monotonic() + 150, p1_deadline - 20)
        if time.monotonic() >= phase_dl:
            continue

        t_phase = time.monotonic()
        inits = diverse_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_a_n = None
        best_M_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            max_sw = max(50, int(800 - 0.5 * n))
            a_opt = coordinate_descent(a0, max_sw, phase_dl, shuffle=(idx > 0), seed=seed + idx * 13)

            M_val = float(fast_conv(a_opt).max())
            if M_val < best_M_n:
                best_M_n = M_val
                best_a_n = a_opt.copy()

        if best_a_n is not None:
            J_val = compute_J(best_a_n)
            print(f"  n={n:4d}: M={best_M_n:.10f} J={J_val:.8f} ({time.monotonic()-t_phase:.1f}s)", flush=True)
            tracker.update(best_a_n)

    tracker.report("Phase 1 complete")

    # === PHASE 2: Riemannian Adam ===
    p2_deadline = min(t0 + 800.0, deadline - 300.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Riemannian Adam (budget~{p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p2_deadline:
        sub_dl = min(time.monotonic() + 180, p2_deadline - 50)
        a_adam = riemannian_adam(tracker.best, 3000, sub_dl, tau_start=5.0, tau_end=80.0, lr=0.04)
        tracker.update(a_adam)

        if time.monotonic() < p2_deadline:
            a_adam2 = riemannian_adam(tracker.best, 2000, min(time.monotonic() + 100, p2_deadline),
                                       tau_start=10.0, tau_end=200.0, lr=0.06)
            tracker.update(a_adam2)

    tracker.report("Phase 2 complete")

    # === PHASE 3: Peak Chasing ===
    p3_deadline = min(t0 + 1000.0, deadline - 200.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Peak Chasing (budget~{p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p3_deadline:
        sub_dl = min(time.monotonic() + 150, p3_deadline - 30)
        a_peak = peak_chase(tracker.best, 2000, sub_dl, top_k=3)
        tracker.update(a_peak)

        if time.monotonic() < p3_deadline:
            a_peak2 = peak_chase(a_peak, 1000, min(time.monotonic() + 80, p3_deadline), top_k=5)
            tracker.update(a_peak2)

    tracker.report("Phase 3 complete")

    # === PHASE 4: Multi-length exploration ===
    p4_deadline = min(t0 + 1300.0, deadline - 100.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length (budget~{p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p4_deadline:
        n_best = tracker.best.size
        expl_lengths = []
        for candidate_n in [384, 512, 640, 768, 1024, 1280]:
            if candidate_n != n_best and 256 <= candidate_n <= 2048:
                expl_lengths.append(candidate_n)

        time_per_len = max(30, int((p4_deadline - time.monotonic()) / max(1, len(expl_lengths)) * 0.8))

        for n_exp in expl_lengths:
            if time.monotonic() > p4_deadline - 20:
                break
            sub_dl = min(time.monotonic() + time_per_len, p4_deadline - 10)
            if time.monotonic() >= sub_dl:
                continue

            base = resample_seq(tracker.best, n_exp)
            base = np.maximum(base, 1e-8)
            base = base / base.sum()

            a_exp = coordinate_descent(base, 100, sub_dl, shuffle=True, seed=seed + n_exp)
            # Quick Adam polish
            if time.monotonic() < sub_dl - 5:
                a_exp = riemannian_adam(a_exp, 500, sub_dl, tau_start=10.0, tau_end=100.0, lr=0.03)

            tracker.update(a_exp)

    tracker.report("Phase 4 complete")

    # === PHASE 5: Final Polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish (remaining {deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 5:
        # Coordinate descent with many sweeps
        a_final = coordinate_descent(tracker.best, 200, deadline - 3, shuffle=False, seed=77777)
        tracker.update(a_final)

        # Peak chase on the result
        if time.monotonic() < deadline - 3:
            a_final2 = peak_chase(a_final, 500, deadline - 2, top_k=3)
            tracker.update(a_final2)

        # One more coordinate descent
        if time.monotonic() < deadline - 2:
            a_final3 = coordinate_descent(a_final2, 50, deadline - 1, shuffle=False, seed=424242)
            tracker.update(a_final3)

    tracker.report("Phase 5 complete")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    # Final evaluation
    final_score = float(evaluate_sequence([float(x) for x in tracker.best]))
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={tracker.best.size} time={time.monotonic()-t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
