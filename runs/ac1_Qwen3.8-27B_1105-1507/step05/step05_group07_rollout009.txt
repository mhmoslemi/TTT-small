import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import math

def fft_conv(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    """Compute the objective function."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fft_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def normalize(a):
    """Normalize so sum = 1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def build_lp_matrix(a):
    """Build the sparse matrix A where A[k,i] = a[k-i] if 0<=k-i<n."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            rows.append(k)
            cols.append(i)
            vals.append(a[k-i])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except:
        pass
    return None

            y = (1 - beta) * a + beta * y

- y_t = (1 - β_t) x_t + β_t * (x_t - x_{t-1}) + x_t = x_t + β_t (x_t - x_{t-1})

The standard Nesterov acceleration for FW:
- β_t = (1 - sqrt(1 + 4/t^2)) / 2 (this goes to ~1/t)
- Or simply β_t = (t-1)/(t+2) which goes to 1

- y_0 = x_0
- For t >= 1:

"""
Accelerated Frank-Wolfe optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses LP-based best response, Nesterov momentum, multiple restarts, and coordinate descent polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import math


def fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fft_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize so sum = 1, clip to non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def build_lp_matrix(a, cache=None):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = np.zeros(0, dtype=np.int32)
    cols = np.zeros(0, dtype=np.int32)
    vals = np.zeros(0, dtype=np.float64)
    
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        count = i_max - i_min + 1
        if count <= 0:
            continue
        r = np.full(count, k, dtype=np.int32)
        c = np.arange(i_min, i_max + 1, dtype=np.int32)
        v = a[k - i_min:k - i_max + 1][::-1]
        rows = np.concatenate([rows, r])
        cols = np.concatenate([cols, c])
        vals = np.concatenate([vals, v])
    
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. A@ g <= M, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_fw(a, max_iters, deadline, momentum=True, verbose=True):
    """Accelerated Frank-Wolfe with LP best response."""
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    y = a.copy()
    
    best_a = a.copy()
    best_M = float(fft_conv(a).max())
    best_J = 2.0 * n * best_M
    
    no_improve = 0
    max_no_improve = 40
    t0_base = time.monotonic()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        if time.monotonic() - t0_base > 120 and t > 20:
            # Safety: if each iter takes too long, break
            pass
        
        # Nesterov extrapolation
        if momentum and t > 1:
            beta = (t - 1) / (t + 2)
            y_new = (1 - beta) * a + beta * y
            y = normalize(y_new)
        else:
            y = a.copy()
        
        # Compute convolution of y
        conv_y = fft_conv(y)
        M = float(conv_y.max())
        
        # LP best response
        g = lp_best_response(y, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Line search
        gamma_base = 2.0 / (t + 2)
        best_trial = None
        best_trial_M = M
        
        for mult in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            gamma = gamma_base * mult
            trial = (1 - gamma) * a + gamma * g
            trial = normalize(trial)
            M_trial = float(fft_conv(trial).max())
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_trial = trial
        
        if best_trial is not None:
            a = best_trial
            if best_trial_M < best_M - 1e-10:
                best_M = best_trial_M
                best_a = a.copy()
                no_improve = 0
                if verbose and t % 20 == 0:
                    J = 2.0 * n * best_M
                    elapsed = time.monotonic() - t0_base
                    print(f"    FW t={t}: J={J:.8f} (elapsed={elapsed:.0f}s)", flush=True)

                no_improve += 1
        else:
            no_improve += 1

"""
Accelerated Frank-Wolfe optimizer with LP best response, Nesterov momentum,
dual-guided perturbations, length exploration, and coordinate descent polishing.
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences a.

import numpy as np

def fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fft_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def build_lp_matrix(a):
    """Build sparse matrix A where A[k,i] = a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    all_rows = []
    all_cols = []
    all_vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        count = i_max - i_min + 1
        if count <= 0:
            continue
        r = np.full(count, k, dtype=np.int32)
        c = np.arange(i_min, i_max + 1, dtype=np.int32)
        idx = k - np.arange(i_min, i_max + 1)
        v = a[idx]
        all_rows.append(r)
        all_cols.append(c)
        all_vals.append(v)
    rows = np.concatenate(all_rows)
    cols = np.concatenate(all_cols)
    vals = np.concatenate(all_vals)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

            best_local = 2.0 * n * conv.max() / (s * s)

            lo = i
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            for x in candidates:
                x = float(max(x, 0))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
