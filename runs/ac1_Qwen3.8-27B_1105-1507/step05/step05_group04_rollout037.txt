"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _fft_size(n):
    size = 1
    while size < n:
        size <<= 1
    return size


def _fast_conv(a):
    n = len(a)
    size = _fft_size(2 * n - 1)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _fast_cross_conv(a, b):
    n = len(a)
    m = len(b)
    size = _fft_size(n + m - 1)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def _compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = _fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def _grad_smooth(a, tau):
    """Compute gradient direction for smooth max of conv(a,a).
    Returns (direction, b, b_max) where direction[m] = sum_k p[k]*a[k-m]."""
    n = len(a)
    b = _fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    Z = exp_b.sum()
    p = exp_b / Z
    p_rev = p[::-1]
    size = _fft_size(3 * n - 2)
    P = np.fft.rfft(p_rev, size)
    A = np.fft.rfft(a, size)
    conv_pa = np.fft.irfft(P * A, size)
    c = conv_pa[n - 1:2 * n - 1][::-1]
    return c, b, b_max


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def _build_lp_matrix(n, a, active_k):
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))


def _lp_step(a, M, top_frac=0.7, time_limit=3.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for top constraints, g >= 0."""
    n = len(a)
    b = _fast_conv(a)
    threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
    active_k = np.where(b >= threshold)[0]
    if len(active_k) < 20:
        active_k = np.arange(2 * n - 1)

    A = _build_lp_matrix(n, a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, np.inf)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                ag = _fast_cross_conv(a, g)
                if ag.max() <= M * (1.0 + 1e-4):
                    return g
                # Try with full constraints
                A_full = _build_lp_matrix(n, a, np.arange(2 * n - 1))
                b_ub_full = np.full(2 * n - 1, M)
                r2 = linprog(c, A_ub=A_full, b_ub=b_ub_full, bounds=bounds,
                             method='highs', options={'time_limit': time_limit, 'presolve': True})
                if r2.success and r2.x is not None:
                    g = np.maximum(r2.x, 0.0)
                    if g.sum() > 1e-10:
                        return g
    except Exception:
        pass
    return None


def _lp_mirror_descent(a0, deadline, top_frac=0.7, max_no_improve=150):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = _compute_J(a)
    no_improve = 0
    restart_count = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        b = _fast_conv(a)
        M = float(b.max())

        g = _lp_step(a, M, top_frac=top_frac, time_limit=2.0)
        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()
        diff = np.max(np.abs(g - a))

        # Adaptive step size
        base_t = min(0.6, max(0.05, diff * 1.5))
        t_candidates = [base_t, base_t * 0.5, base_t * 0.25, 0.03, 0.01]

        improved = False
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = _compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
            elif J_trial < _compute_J(a) - 1e-12:
                a = a_trial
                no_improve = 0
                break

        if not improved:
            no_improve += 1

        if no_improve > max_no_improve:
            restart_count += 1
            rng = np.random.default_rng(it * 31 + restart_count)
            scale = 0.1 / (1.0 + 0.2 * restart_count)
            freqs = rng.integers(1, 6, size=3)
            phases = rng.uniform(0, 2 * np.pi, size=3)
            t_arr = np.linspace(0, 1, n, endpoint=False)
            pert = np.zeros(n)
            for f, ph in zip(freqs, phases):
                pert += np.cos(2 * np.pi * f * t_arr + ph)
            a = a + scale * pert
            a = _project_simplex(a)
            no_improve = 0

        if it % 500 == 0:
            print(f"    LP iter={it} J={best_J:.8f} M={M:.6f} restarts={restart_count}", flush=True)

    return best_a, best_J

        # Perturb
        scale = 0.1 + 0.05 * r
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amplitudes = rng.uniform(0.3, 1.0, size=5)
        t_arr = np.linspace(0, 1, n, endpoint=False)
        pert = np.zeros(n)
        for f, ph, amp in zip(freqs, phases, amplitudes):
            pert += amp * np.cos(2 * np.pi * f * t_arr + ph)
        a_pert = _project_simplex(a + scale * pert)

        # Quick gradient optimization
        phase_dl = deadline - max(10, (deadline - time.monotonic()) * 0.3 / (n_restarts - r))
        a_opt, J_opt = _smooth_grad_descent(a_pert, phase_dl, tau=20.0)

        if J_opt < best_J:
            best_J = J_opt
            best_a = a_opt.copy()
            print(f"    Basin hop {r}: J={J_opt:.8f} (improved!)", flush=True)
        else:
            print(f"    Basin hop {r}: J={J_opt:.8f} (no improvement)", flush=True)

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search for optimal step function heights.
    Phase 1: Fast gradient descent from multiple starts (45% time)
    Phase 2: LP mirror descent on best candidates (35% time)
    Phase 3: Basin hopping + multi-length + polish (20% time)
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0
    rng = np.random.default_rng(seed)

    best_a = None
    best_score = float('inf')

    def update_tracker(a, label=""):
        nonlocal best_a, best_score
        if a is None or len(a) == 0:
            return
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = _compute_J(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s [{label}]", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_tracker(parent, "parent")
            print(f"Parent: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        update_tracker(parent, "default")

    # === PHASE 1: Fast gradient descent (45% of time) ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: Fast Gradient Descent (multi-start)", flush=True)
    print(f"{'=' * 60}", flush=True)

    p1_deadline = t_start + budget_s * 0.45
    n_main = 512
    candidates = []

    inits = _generate_inits(n_main, rng, parent)
    n_inits = len(inits)
    per_init_time = max(5.0, (p1_deadline - time.monotonic()) / n_inits)

    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_deadline:
            break
        a0 = np.maximum(a0, 1e-8)
        a0 = a0 / a0.sum()
        init_dl = min(time.monotonic() + per_init_time, p1_deadline - 5)
        a_opt, J_val = _smooth_grad_descent(a0, init_dl, tau=30.0, label=f"init{idx}")
        candidates.append((J_val, a_opt))
        update_tracker(a_opt, f"grad_init{idx}")
        print(f"  init {idx:2d}: J={J_val:.8f}", flush=True)

    # Sort and keep top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 best: J={candidates[0][0]:.8f}", flush=True)

    # === PHASE 2: LP Mirror Descent (35% of time) ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'=' * 60}", flush=True)

    p2_deadline = t_start + budget_s * 0.80
    p2_remaining = p2_deadline - time.monotonic()

    if p2_remaining > 30:
        # Allocate time to top candidates
        top_cands = candidates[:min(3, len(candidates))]
        time_alloc = [0.6, 0.25, 0.15]

        for ci, (J_init, a_init) in enumerate(top_cands):
            if time.monotonic() > p2_deadline - 10:
                break
            alloc = p2_remaining * time_alloc[ci] if ci < len(time_alloc) else 30.0
            alloc = max(30.0, alloc)
            phase_dl = min(time.monotonic() + alloc, p2_deadline - 5)

            print(f"\n  LP on candidate {ci} (init J={J_init:.8f}), budget={alloc:.0f}s", flush=True)
            a_best, J_best = _lp_mirror_descent(a_init, phase_dl, top_frac=0.7, max_no_improve=120)
            update_tracker(a_best, f"lp_cand{ci}")

            # Quick gradient polish after LP
            if time.monotonic() < p2_deadline - 10:
                a_pol, J_pol = _smooth_grad_descent(a_best, min(time.monotonic() + 20, p2_deadline - 5), tau=50.0)
                update_tracker(a_pol, f"grad_after_lp{ci}")
            p2_remaining = p2_deadline - time.monotonic()

    # === PHASE 3: Basin hopping + multi-length + final polish (20% of time) ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 3: Basin Hopping + Multi-Length + Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_a is not None and time.monotonic() < deadline - 10:
        # Basin hopping on current best
        bh_dl = min(time.monotonic() + (deadline - time.monotonic()) * 0.4, deadline - 30)
        print(f"  Basin hopping (budget={bh_dl - time.monotonic():.0f}s)", flush=True)
        a_bh, J_bh = _basin_hopping(best_a, bh_dl, n_restarts=5)
        update_tracker(a_bh, "basin_hop")

        # Try different lengths
        if time.monotonic() < deadline - 30:
            base_n = best_a.size
            for n_try in [base_n + 128, base_n - 128, 768, 1024]:
                if time.monotonic() > deadline - 20:
                    break
                if n_try < 128 or n_try > 2048 or n_try == base_n:
                    continue
                a_ext = _resample_seq(best_a, n_try)
                a_ext = np.maximum(a_ext, 1e-8)
                a_ext = a_ext / a_ext.sum()
                ext_dl = min(time.monotonic() + 30, deadline - 10)
                a_opt, J_opt = _smooth_grad_descent(a_ext, ext_dl, tau=25.0)
                update_tracker(a_opt, f"n={n_try}")

        # Final polish: alternate gradient and LP
        if time.monotonic() < deadline - 5:
            print(f"  Final polish (budget={deadline - time.monotonic():.0f}s)", flush=True)
            a = _project_simplex(best_a)
            for polish_round in range(10):
                if time.monotonic() > deadline - 3:
                    break
                # Gradient step
                a_g, _ = _smooth_grad_descent(a, min(time.monotonic() + 15, deadline - 5), tau=80.0)
                a = a_g
                update_tracker(a, f"polish_g{polish_round}")
                # LP step
                if time.monotonic() < deadline - 10:
                    a_l, _ = _lp_mirror_descent(a, min(time.monotonic() + 15, deadline - 5),
                                                 top_frac=0.9, max_no_improve=50)
                    a = a_l
                    update_tracker(a, f"polish_l{polish_round}")

    # Fallback
    if best_a is None:
        best_a = np.ones(128)
        best_score = _compute_J(best_a)

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size} time={elapsed:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_a]
