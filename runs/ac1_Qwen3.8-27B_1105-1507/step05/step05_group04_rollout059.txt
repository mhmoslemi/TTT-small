    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_grad(a, t):
    """Compute gradient of (1/t)*log(sum(exp(t*b))) where b = conv(a,a).
    Returns (gradient, b, M) where M = max(b)."""
    b = fast_conv(a)
    M = b.max()
    shifted = t * (b - M)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    # Gradient: 2 * correlate(p, a, mode='valid')
    grad = 2.0 * correlate(p, a, mode='valid')
    return grad, b, M

def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def nesterov_smooth_opt(a, deadline, t_start=5.0, t_end=200.0, 
                         lr_initial=0.05, momentum=0.9,
                         temp_schedule='exp', verbose=False):
    """Nesterov-accelerated smooth gradient descent with adaptive temperature."""
    a = project_simplex(a.astype(np.float64).copy())
    n = len(a)
    a_prev = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_J = compute_J(a)
    
    t = t_start
    total_iters = 0
    
    while time.monotonic() < deadline:
        grad, b, M = smooth_max_grad(a, t)
        
        # Nesterov step
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        else:
            y = y / sy
        
        # Recompute gradient at extrapolated point for true Nesterov
        grad_y, _, _ = smooth_max_grad(y, t)
        
        # Adaptive learning rate
        lr = lr_initial / (1.0 + 0.01 * total_iters)
        a_new = y - lr * grad_y
        a_new = project_simplex(a_new)
        
        a_prev = a
        a = a_new
        total_iters += 1
        
        # Check improvement
        if total_iters % 50 == 0:
            J_cur = compute_J(a)
            if J_cur < best_J:
                best_J = J_cur
                best_a = a.copy()
                if verbose:
                    print(f"    Nesterov iter={total_iters} J={best_J:.8f} t={t:.1f} M={M:.6f}")
        
        # Adaptive temperature: increase over time

            t = t_start * (t_end / t_start) ** progress
        
        if total_iters > 50000:
            break
    
    # Final check
    J_final = compute_J(a)
    if J_final < best_J:
        best_J = J_final
        best_a = a.copy()

def coordinate_descent(a, deadline, max_rounds=100):

def solve_lp_direction(a, M, top_frac=0.7, time_limit=3):
    """Solve LP to get a direction: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    if top_frac < 1.0:
        threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
        active_k = np.where(b >= threshold)[0]
    else:
        active_k = np.arange(len(b))
    
    m = len(active_k)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0, np.inf)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def lp_polish(a, deadline, max_iters=50):
    """LP-based polishing: use LP direction + line search."""
    a = project_simplex(a.astype(np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = solve_lp_direction(a, M, top_frac=0.7, time_limit=5)
        if g is None or g.sum() < 1e-10:
            break
        
        g = project_simplex(g)
        
        # Line search
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = project_simplex((1 - t) * a + t * g)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_J

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    sr = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(project_simplex(sr + 0.01))

smooth gradient descent with adaptive log-sum-exp temperature annealing,

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, t):
    """Compute gradient of (1/t)*log(sum(exp(t*b))) where b=conv(a,a).
    Returns (gradient, b, M) where M=max(b)."""
    b = fast_conv(a)
    M = b.max()
    shifted = np.clip(t * (b - M), -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = 2.0 * correlate(p, a, mode='valid')
    return grad, b, M


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def solve_lp_direction(a, M, top_frac=0.7, time_limit=5):

    rows = []
    cols = []
    vals = []

        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])

    # Two-bump asymmetric
    inits.append(project_simplex(
        np.exp(-8 * (t - 0.3) ** 2) + 0.7 * np.exp(-8 * (t - 0.7) ** 2) + 0.01))
    inits.append(project_simplex(
        0.7 * np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2) + 0.01))

    return inits


def sort_by_score(items):
    """Sort list of (score, data) tuples by score ascending."""
    return sorted(items, key=operator.itemgetter(0))


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search for optimal step function coefficients:
    Phase 1: Multi-n momentum smooth gradient with diverse inits (550s)
    Phase 2: LP polishing (450s)
    Phase 3: Multi-length exploration from best (100s)
    Phase 4: Coordinate descent + final LP polish (remaining)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    best_a = None
    best_J = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            J_parent = compute_J(parent)
            best_a = parent.copy()
            best_J = J_parent
            print(f"Parent: n={parent.size}, J={J_parent:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if best_a is None:
        best_a = np.ones(256)
        best_J = compute_J(best_a)
        print(f"Default: J={best_J:.8f}", flush=True)

    # === PHASE 1: Multi-n momentum smooth gradient ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: Multi-n Momentum Smooth Gradient", flush=True)
    print(f"{'=' * 60}", flush=True)

    p1_deadline = time.monotonic() + 550.0

    base_n = parent.size if parent is not None else 512
    n_values = [base_n, 384, 448, 512, 576, 640, 768, 896, 1024]
    # Deduplicate preserving order
    seen = set()
    n_unique = []
    for nv in n_values:
        if nv not in seen and 64 <= nv <= 2048:
            seen.add(nv)
            n_unique.append(nv)

    for n in n_unique:
        if time.monotonic() > p1_deadline:
            break
        remaining = p1_deadline - time.monotonic()
        if remaining < 35:
            break

        if n == base_n:
            n_time = min(remaining * 0.35, 180.0)
        else:
            n_time = min(remaining * 0.12, 70.0)
        n_time = max(20.0, n_time)
        phase_dl = min(time.monotonic() + n_time, p1_deadline)

        print(f"  n={n}, budget={n_time:.0f}s", flush=True)

        inits = generate_inits(n, rng, parent)

        # Quick-score initializations
        scored = []
        for a0 in inits:
            if time.monotonic() > phase_dl - 3:
                break
            J0 = compute_J(a0)
            scored.append((J0, a0))

        scored = sort_by_score(scored)

        # Run momentum on top 3
        n_try = min(3, len(scored))
        for ii in range(n_try):
            if time.monotonic() > phase_dl:
                break
            J_init, a_init = scored[ii]

            # Allocate sub-time
            if n_try > 1:
                frac = [0.5, 0.3, 0.2][:n_try]
                sub_dl = min(time.monotonic() + (phase_dl - time.monotonic()) * frac[ii],
                             phase_dl)
            else:
                sub_dl = phase_dl
            sub_dl = max(time.monotonic() + 8, sub_dl)

            a_opt, J_opt = momentum_smooth_opt(
                a_init, sub_dl,
                t_init=10.0, t_final=500.0,
                lr_init=0.03, momentum=0.88
            )
            if J_opt < best_J:
                best_J = J_opt
                best_a = a_opt.copy()
                print(f"    ** J={J_opt:.8f} n={n} [mom_{ii}]", flush=True)
            else:
                print(f"    J={J_opt:.8f} n={n} [mom_{ii}]", flush=True)

    print(f"\nAfter Phase 1: J={best_J:.8f} n={best_a.size}", flush=True)

    # === PHASE 2: LP Polishing ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 2: LP Polishing", flush=True)
    print(f"{'=' * 60}", flush=True)

    p2_deadline = min(time.monotonic() + 450.0, deadline - 130.0)

    if time.monotonic() < p2_deadline:
        a_pol, J_pol = lp_polish(best_a, p2_deadline, max_iters=300)
        if J_pol < best_J:
            best_J = J_pol
            best_a = a_pol.copy()
            print(f"  ** J={J_pol:.8f} [LP]", flush=True)
        else:
            print(f"  J={J_pol:.8f} [LP no improvement]", flush=True)

    # === PHASE 3: Multi-length from best ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 3: Multi-length from Best", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_deadline = min(time.monotonic() + 100.0, deadline - 60.0)

    if time.monotonic() < p3_deadline:
        base_n = best_a.size
        for n_try in [base_n - 32, base_n + 32, base_n + 64, base_n * 2, base_n // 2]:
            if time.monotonic() > p3_deadline:
                break
            if n_try < 64 or n_try > 2048:
                continue
            if n_try == base_n:
                continue

            a_ext = project_simplex(resample_seq(best_a, n_try))
            sub_dl = min(time.monotonic() + 20, p3_deadline - 5)
            a_opt, J_opt = momentum_smooth_opt(
                a_ext, sub_dl,
                t_init=20.0, t_final=400.0, lr_init=0.03, momentum=0.88)
            if J_opt < best_J:
                best_J = J_opt
                best_a = a_opt.copy()
                print(f"  ** J={J_opt:.8f} n={n_try}", flush=True)

    # === PHASE 4: Final refinement ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 4: Final Refinement", flush=True)
    print(f"{'=' * 60}", flush=True)
    remaining = deadline - time.monotonic()
    print(f"  Remaining: {remaining:.0f}s", flush=True)

    if remaining > 40:
        cd_dl = min(time.monotonic() + remaining * 0.25, deadline - 20)
        a_cd, J_cd = coord_descent(best_a, cd_dl)
        if J_cd < best_J:
            best_J = J_cd
            best_a = a_cd.copy()
            print(f"  ** J={J_cd:.8f} [coord]", flush=True)

    if time.monotonic() < deadline - 10:
        a_lp, J_lp = lp_polish(best_a, deadline - 5, max_iters=80)
        if J_lp < best_J:
            best_J = J_lp
            best_a = a_lp.copy()
            print(f"  ** J={J_lp:.8f} [final_LP]", flush=True)

    # Final check
    final_list = [float(x) for x in best_a]
    final_eval = evaluate_sequence(final_list)
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J_internal={best_J:.8f} J_eval={final_eval:.8f} n={best_a.size}", flush=True)
    print(f"Time: {time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return final_list
