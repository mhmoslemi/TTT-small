def golden_section_search_1d(f, a, b, tol=1e-6, max_iter=100):

    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~0.618
    
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    
    x_opt = (a + b) / 2
    return x_opt, f(x_opt)

        if s_new > 1e-10:

            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
        # Decay step
        step *= 0.995
    
    return best_a

def iterated_lp_search(a, n_rounds=30, time_budget=300.0):
    """
    Main iterative search: alternate between LP direction finding and 1D line search.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for rnd in range(n_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        # Try different LP formulations
        used_t = 0.1
        g, M_lp = lp_blend_direction(a, t_approx=used_t, time_limit=20)
        
        if g is None:
            g, M_lp = lp_minpeak(a, time_limit=20)
        
        if g is None:
            break
        
        # Blend with line search
        a_new, obj_new = blend_and_search(a, g)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        
        # Also try smooth gradient descent for a few iterations
        if (rnd + 1) % 5 == 0:
            a_smooth = smooth_grad_descent(a, tau=5.0, n_iters=30, step=0.03)
            obj_smooth = compute_obj(a_smooth)
            if obj_smooth < best_obj:
                best_obj = obj_smooth
                best_a = a_smooth.copy()
                a = a_smooth.copy()
        
        # Periodic perturbation to escape local minima
        if (rnd + 1) % 10 == 0:
            rng = np.random.default_rng(rnd * 137 + 7)
            a_pert = a * np.exp(rng.normal(0, 0.03, n))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / np.sum(a_pert)
            obj_pert = compute_obj(a_pert)
            # Accept with some probability (simulated annealing style)
            if obj_pert < best_obj or (rnd > 5 and np.random.random() < 0.1):
                a = a_pert
    
    return best_a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
