"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive smooth Nesterov gradient descent with two-phase tau annealing
2. Peak-directed stochastic local search (targets coordinates causing peaks)
3. LP mirror descent with aggressive mixing for precise refinement
4. Multi-length exploration and structured initializations
Key improvements: two-phase tau, peak-directed search, longer LP phase,
and smarter time allocation.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[lo:hi + 1]
        rev = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * rev
    return grad, p, b, b_max


def run_smooth_nesterov(a_init, deadline, max_iters=8000,
                        tau_start=1.0, tau_end=300.0,
                        beta=0.9, lr_init=0.5, lr_end=0.003):
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        # Two-phase: slow ramp then faster
        if frac < 0.4:
            tau = tau_start * (50.0 / tau_start) ** (frac / 0.4)
        else:
            tau = 50.0 * (tau_end / 50.0) ** ((frac - 0.4) / 0.6) ** 0.7
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y /= sy
        a = a_new

    # Final check
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_a = a.copy()
    return best_a


def peak_directed_search(a_init, deadline, max_iters=3000, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Find peak positions
        threshold = M - 0.02 * M
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])
        # Limit to top peaks
        if len(peaks) > 10:
            peak_vals = b[peaks]
            top_idx = np.argsort(peak_vals)[-10:]
            peaks = peaks[top_idx]

        # Identify critical coordinates (those with large |a[j]| contributing to peaks)
        critical_scores = np.zeros(n)
        for k in peaks:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            critical_scores[lo:hi + 1] += a[lo:hi + 1] ** 2
        critical_idx = np.where(critical_scores > 0.1 * critical_scores.max())[0]
        if len(critical_idx) < 3:
            critical_idx = np.argsort(critical_scores)[-max(3, n // 10):]

        non_critical = np.setdiff1d(np.arange(n), critical_idx)
        if len(non_critical) == 0:
            non_critical = np.arange(n)

        best_trial_M = M
        best_trial = None

        for attempt in range(15):
            a_trial = a.copy()
            n_src = rng.integers(1, 4)
            n_dst = rng.integers(1, 4)
            srcs = rng.choice(critical_idx, size=min(n_src, len(critical_idx)), replace=False)
            dsts = rng.choice(non_critical, size=min(n_dst, len(non_critical)), replace=False)

            for src in srcs:
                transfer = rng.uniform(0.005, 0.08) * a[src]
                a_trial[src] -= transfer
                dst = dsts[rng.integers(0, len(dsts))]
                a_trial[dst] += transfer * (len(srcs) / len(dsts))

            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 0.01:
                continue
            a_trial /= s

            M_trial = fast_conv(a_trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_trial = a_trial.copy()

        if best_trial is not None and best_trial_M < best_M * (1 - 1e-9):
            a = best_trial
            best_a = a.copy()
            best_M = best_trial_M
            stall = 0
        else:
            stall += 1
            if stall > 300:
                # Try random Gaussian perturbation
                a_trial = a + rng.normal(0, 0.003 * a.max(), n)
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 0.01:
                    a_trial /= s
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M:
                        a = a_trial
                        best_a = a.copy()
                        best_M = M_trial
                        stall = 0
                if stall > 500:
                    break

    return best_a


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a_init, deadline, max_iters=500, aggressive=False):
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    if aggressive:
        t_schedule = [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02, 0.01]
    else:
        t_schedule = [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02, 0.01]

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=10)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        for t in t_schedule:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-13:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 20:
                break

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.005))
        inits.append(np.maximum(p + 0.03 * np.sin(2 * np.pi * 3 * t) * np.outer(p, np.ones(n))[:, 0], 0.005))
    inits.append(np.ones(n))
    # Raised cosine variants
    for alpha in [0.5, 0.8, 1.0, 1.5, 2.0]:
        profile = (1 + np.cos(2 * np.pi * t)) / 2
        inits.append(np.maximum(0.3 + 0.7 * profile ** alpha, 0.01))
    # Asymmetric profiles
    for sk in [0.3, 0.5, 0.7]:
        inits.append(np.maximum(0.2 + 0.8 * np.exp(-((t - sk) ** 2) / 0.05), 0.01))
    # Frequency-based
    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(0.6 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    # Random smooth
    k = max(3, n // 32)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    # Front-loaded (inspired by n=2,3 optimal structures)
    inits.append(np.maximum(0.5 * (1 - 0.5 * t), 0.01))
    inits.append(np.maximum(0.5 * (1 - 0.3 * t) + 0.1 * np.sin(np.pi * t), 0.01))
    return inits

class BestTracker:

        self.score = float('inf')

        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)

            best.update(best_n, f"phase1 n={n}")
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Two-phase smooth on best candidate ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: Two-Phase Smooth + Peak-Directed Search", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        # Two-phase smooth
        a = two_phase_smooth(best.seq, min(time.monotonic() + 120, deadline - 200), best.seq.size)
        best.update(a, "two-phase")

        # Peak-directed local search
        a = peak_directed_search(best.seq, min(time.monotonic() + 150, deadline - 150),
                                 max_iters=3000, rng=rng)
        best.update(a, "peak-directed")

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: LP Mirror Descent (extended) ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (Extended)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        est_lp_ms = 5.0 * (n / 128.0) ** 2
        print(f"  n={n}, est LP time={est_lp_ms:.0f}ms", flush=True)

        # Aggressive LP (try large mixing first)
        lp_dl = min(time.monotonic() + 200, deadline - 100)
        a = run_lp_mirror(best.seq, lp_dl, max_iters=300, aggressive=True)
        best.update(a, "LP aggressive")

        # Standard LP
        if time.monotonic() < deadline - 50:
            a = run_lp_mirror(best.seq, deadline - 50, max_iters=300, aggressive=False)
            best.update(a, "LP standard")

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Iterative refinement loop ===
    p4_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Iterative Refinement", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        for round_i in range(5):
            if time.monotonic() > deadline - 30:
                break
            print(f"  Round {round_i+1}:", flush=True)
            # Smooth polish
            if time.monotonic() < deadline - 25:
                a = run_smooth_nesterov(best.seq, min(time.monotonic() + 30, deadline - 20),
                                        max_iters=2000, tau_start=30.0, tau_end=400.0,
                                        beta=0.95, lr_init=0.1, lr_end=0.002)
                best.update(a, f"round{round_i+1} smooth")
            # Peak search
            if time.monotonic() < deadline - 20:
                a = peak_directed_search(best.seq, min(time.monotonic() + 60, deadline - 15),
                                         max_iters=1500, rng=rng)
                best.update(a, f"round{round_i+1} peak")
            # LP
            if time.monotonic() < deadline - 10:
                a = run_lp_mirror(best.seq, deadline - 10, max_iters=100, aggressive=True)
                best.update(a, f"round{round_i+1} LP")

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Try adjacent lengths ===
    if best.seq is not None and time.monotonic() < deadline - 20:
        p5_start = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"PHASE 5: Length Refinement", flush=True)
        print(f"{'='*50}", flush=True)
        n_best = best.seq.size
        for delta in [-32, -16, 16, 32, 64, -64]:
            if time.monotonic() > deadline - 10:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 3072:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            a_ext = run_smooth_nesterov(a_ext, min(time.monotonic() + 20, deadline - 5),
                                        max_iters=1500, tau_start=20, tau_end=200,
                                        lr_init=0.2, lr_end=0.005)
            best.update(a_ext, f"length {n_try}")
            # Quick peak search
            if time.monotonic() < deadline - 5:
                a2 = peak_directed_search(a_ext, min(time.monotonic() + 20, deadline - 3),
                                          max_iters=500, rng=rng)
                best.update(a2, f"length {n_try} peak")
        best.report("Phase 5")
        print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === Final LP polish ===
    if best.seq is not None and time.monotonic() < deadline - 2:
        a = run_lp_mirror(best.seq, deadline, max_iters=50, aggressive=True)
        best.update(a, "final LP")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    # Final validation with evaluate_sequence
    final_score = evaluate_sequence(best.seq.tolist())
    if not np.isfinite(final_score):
        print("ERROR: final score is not finite, using fallback", flush=True)
        best.seq = np.ones(128)
        final_score = evaluate_sequence(best.seq.tolist())

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
