```python
import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s

def mirror_descent_step(a, a_k, eta):
    """One mirror descent step for minimizing max_k conv(a,a)_k.
    
    a: current point (normalized, sum=1)
    a_k: the subgradient direction (gradient of conv at the argmax k*)
    eta: step size
    """
    # Mirror descent update: a_new_i ∝ a_i * exp(-eta * g_i)
    log_scale = -eta * a_k
    # Numerical stability
    log_scale -= np.max(log_scale)
    a_new = a * np.exp(log_scale)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s

def compute_subgradient(a, k_star):
    """Compute gradient of conv(a,a)_{k_star} w.r.t. a.
    
    The gradient at position m is: 2 * a_{k_star - m} (if 0 <= k_star - m < n), else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    for m in range(n):
        j = k_star - m
        if 0 <= j < n:
            grad[m] = 2.0 * a[j]
    return grad

def mirror_descent_optimize(a, max_iters=10000, eta0=0.1, eta_decay=0.999, 
                            time_budget=60.0, restart_every=500, verbose=False):
    """Mirror descent optimization with restarts."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta0
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        M = conv[k_star]
        
        # Compute subgradient
        grad = compute_subgradient(a, k_star)
        
        # Also try a few nearby peaks for a mixed subgradient
        # (helps when multiple peaks are close to the max)
        threshold = M * 0.9
        peak_indices = np.where(conv >= threshold)[0]
        if len(peak_indices) > 1:
            # Average subgradient over top peaks (weighted by how close they are to max)
            weights = np.exp((conv[peak_indices] - M) / (M * 0.1 + 1e-10))
            weights /= np.sum(weights)
            grad = np.zeros(n)
            for w, k in zip(weights, peak_indices):
                grad += w * compute_subgradient(a, int(k))
        
        # Mirror descent step
        a_new = mirror_descent_step(a, grad, eta)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        eta *= eta_decay
        
        # Periodic restart with perturbation
        if it % restart_every == restart_every - 1:
            if stall > restart_every // 2:
                rng = np.random.default_rng(it)
                perturbation = rng.normal(0, 0.05, n)
                a_pert = a * np.exp(perturbation)
                a = normalize(a_pert)
                eta = eta0
                stall = 0
        
        if verbose and it % 1000 == 0:
            print(f"    MD it={it}, obj={best_obj:.6f}, eta={eta:.6f}")
    
    return best_a, best_obj
```

```python
"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
LP alternation + mirror descent + coordinate descent + spectral shaping.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_minimax(a, norm_sum=1.0):
    """Solve min t s.t. conv(a,g)_k <= t, sum(g)=1, g>=0.
    Returns (g, t) or (None, inf)."""
    n = len(a)
    m = 2 * n - 1
    A_t = build_toeplitz(a)  # (m, n)
    # Variables: g_0, ..., g_{n-1}, t
    # Minimize t -> c = [0,...,0, 1]
    # Constraints: A_t @ g - t <= 0 for each k
    #               sum(g) = 1
    #               g >= 0
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality: A_t @ g - t <= 0
    A_ub = np.hstack([A_t, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.hstack([np.ones(n), [[0.0]]])
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # g_i >= 0, t >= 0
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            t = float(res.x[n])
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s  # Ensure sum = 1
            return g, t
    except Exception:
        pass
    return None, np.inf


def solve_lp_standard(a, M, relax=1.0):
    """Solve max sum(g) s.t. conv(a,g) <= M*relax, g >= 0.
    Returns (g, sum_g) or (None, 0)."""
    n = len(a)
    m = 2 * n - 1
    A_t = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_t, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0


def lp_alternation(a, max_iters=3000, time_budget=120.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 50
    relax = 1.0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Try standard LP
        g, S = solve_lp_standard(a, M, relax=relax)
        
        if g is None or S <= 1.0 + 1e-8:
            # Try minimax LP
            g, t = solve_lp_minimax(a)
            if g is not None:
                S = 1.0
        
        if g is None or S <= 1e-10:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(hash((it, n)) % (2**31))
                a = normalize(a * np.exp(rng.normal(0, 0.1, n)))
                relax = 1.0
                stall = 0
                patience = max(20, patience - 5)
            continue
        
        g_norm = g / S
        
        # Line search
        best_t_obj = best_obj
        best_t = 0
        for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = normalize((1 - t_val) * a + t_val * g_norm)
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t_val
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = normalize((1 - best_t) * a + best_t * g_norm)
            a = best_a.copy()
            stall = 0
            relax = max(1.0, relax - 0.01)
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(hash((it * 7 + 3, n)) % (2**31))
                a = normalize(a * np.exp(rng.normal(0, 0.05, n)))
                relax = min(1.5, relax + 0.1)
                stall = 0
                patience = max(10, patience - 3)
        
        if verbose and it % 500 == 0:
            print(f"    LP it={it}, obj={best_obj:.6f}, relax={relax:.2f}")
    
    return best_a, best_obj


def mirror_descent(a, max_iters=50000, eta0=0.05, time_budget=120.0, verbose=False):
    """Mirror descent for minimizing max_k conv(a,a)_k with sum(a)=1, a>=0."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta0
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        k_star = int(np.argmax(conv))
        
        # Compute subgradient
        grad = np.zeros(n)
        for m in range(n):
            j = k_star - m
            if 0 <= j < n:
                grad[m] = 2.0 * a[j]
        
        # Also incorporate nearby peaks for robustness
        threshold = M * 0.95
        peak_indices = np.where(conv >= threshold)[0]
        if len(peak_indices) > 1:
            weights = np.zeros(len(peak_indices))
            for i, k in enumerate(peak_indices):
                weights[i] = np.exp((conv[k] - M) / (0.05 * M + 1e-15))
            weights /= np.sum(weights)
            grad = np.zeros(n)
            for w, k in zip(weights, peak_indices):
                for m in range(n):
                    j = int(k) - m
                    if 0 <= j < n:
                        grad[m] += w * 2.0 * a[j]
        
        # Mirror descent update
        log_scale = -eta * grad
        log_scale -= np.max(log_scale)  # numerical stability
        a_new = a * np.exp(log_scale)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a = normalize(a + 1e-10 * np.ones(n))
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new  # Still move, but don't update best
            stall += 1
        
        # Adaptive step size
        if stall > 200:
            eta *= 0.9
            if eta < 1e-6:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                a = normalize(best_a * np.exp(rng.normal(0, 0.02, n)))
                eta = eta0
                stall = 0
    
    return best_a, best_obj
```

```python
"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
Combines LP alternation, mirror descent, coordinate descent, and spectral shaping
with adaptive time management and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_standard(a, M, relax=1.0):
    """Solve max sum(g) s.t. conv(a,g) <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    A_t = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_t, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 8})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0


def solve_lp_minimax(a):
    """Solve min t s.t. conv(a,g)_k <= t for all k, sum(g)=1, g>=0."""
    n = len(a)
    m = 2 * n - 1
    A_t = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A_t, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.hstack([np.ones(n), [[0.0]]])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 8})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
            return g
    except Exception:
        pass
    return None

def mirror_descent(a, time_budget=120.0, eta0=0.1, verbose=False):
    """Mirror descent for minimizing max_k conv(a,a)_k with sum(a)=1, a>=0."""

    a = normalize(a.copy())
    best_a = a.copy()

    it = 0

        threshold = M * 0.92

        if len(peak_indices) > 1:
            for k in peak_indices:
                k = int(k)
                weight = np.exp((conv[k] - M) / (0.05 * M + 1e-15))
                for m in range(n):
                    j = k - m
                    if 0 <= j < n:
                        grad[m] += weight * 2.0 * a[j]
            grad /= len(peak_indices)
        else:
            k = k_star
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    grad[m] = 2.0 * a[j]

        # Mirror descent update with line search
        best_obj_local = best_obj
        best_a_local = best_a
        for eta_try in [eta, eta * 0.5, eta * 2.0, eta * 0.25]:
            log_scale = -eta_try * grad
            log_scale -= np.max(log_scale)
            a_try = a * np.exp(log_scale)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_local:
                best_obj_local = obj_try
                best_a_local = a_try.copy()

        if best_obj_local < best_obj:
            best_obj = best_obj_local
            best_a = best_a_local
            a = best_a
            stall = 0
        else:
            # Still move in the direction (helps escape flat regions)
            log_scale = -eta * grad
            log_scale -= np.max(log_scale)

def coordinate_descent(a, time_budget=120.0, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    a = normalize(a.copy())

            if old_val < 1e-15:

                          old_val * 1.25, old_val * 1.5, old_val * 2.0]

            best_peak = np.inf

            for t in candidates:
                if t < 0:
                    t = 0.0

                # Compute new peak
                new_peak = conv.copy()
                new_sum = sum_a - old_val + t

                # Update convolution values affected by changing a[m]
                dm = t - old_val
                for k in range(2 * n - 1):
                    j = k - m
                    if k == 2 * m:
                        # a[m]^2 term
                        new_peak[k] = conv[k] + t * t - old_val * old_val
                    elif 0 <= j < n:
                        new_peak[k] = conv[k] + 2.0 * dm * a[j]

                            pass  # already handled by k == 2m case

                    continue

                    best_t = t

                    elif 0 <= j < n:

                            pass

def spectral_refinement(a, time_budget=30.0, verbose=False):

            tilt = rng.uniform(0.3, 1.5)
            mag_new = mag * np.exp(-tilt * freqs)

        elif choice == 2:

        else:
            target = np.mean(mag)

            mag_new = (1 - alpha) * mag + alpha * target

        s_try = np.sum(a_try)

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, seed):

    elif choice == 1:

        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)

    elif choice == 3:

            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.3)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 5:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 6:
        a = np.ones(n)
        a[:n // 2] *= rng.uniform(0.5, 1.2)
        a[n // 2:] *= rng.uniform(0.5, 1.2)
    else:
        a = rng.random(n) + 0.05

    return normalize(a)
```
