            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix_for_a(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_min_cross(a):
    """Solve LP: min t s.t. conv(a,b)_k <= t, sum(b)=1, b>=0."""
    n = len(a)
    # Variables: b_0, ..., b_{n-1}, t (n+1 variables)
    # Objective: minimize t -> c = [0, ..., 0, 1]
    # Constraints: conv(a,b)_k <= t for k=0,...,2n-2
    #   => sum_j a[k-j] b_j - t <= 0
    # sum(b) = 1
    # b_j >= 0, t free (but t >= 0 implicitly)
    
    # Using standard form for linprog:
    # min c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
    # x = [b_0, ..., b_{n-1}, t]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # A_ub: (2n-1) x (n+1)
    # Row k: sum_j A[k,j] b_j - t <= 0
    A_rows = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n + 1)
        for j in range(max(0, k - n + 1), min(n - 1, k) + 1):
            row[j] = a[k - j]
        row[-1] = -1.0  # -t
        A_rows.append(row)
        b_ub.append(0.0)
    
    A_ub = np.array(A_rows)
    b_ub_arr = np.array(b_ub)
    
    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # b_j >= 0, t free
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub_arr, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs', options={'presolve': True})
        if r.success and r.x is not None:
            b = np.maximum(r.x[:n], 0)
            if b.sum() > 1e-15:
                b /= b.sum()
                return b, float(r.x[-1])
    except Exception:
        pass
    return None, None


def smooth_max_grad(a, tau):
    """Compute smooth approximation of gradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, b, b_max


def run_smooth_gd(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                  lr_init=0.3, lr_end=0.002):
    """Smooth gradient descent with Nesterov acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a

def resample_seq(src, n):
    """Resample a sequence to a new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.001))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.001))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine-based
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    
    # Gaussian bumps
    for center in [0.25, 0.5, 0.75, 0.33, 0.67]:
        inits.append(np.exp(-8 * (t - center)**2) + 0.1)
    
    # Power-law shapes
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        inits.append(np.maximum(0.3 + 0.7 * t**alpha, 0.01))
    
    # Random smooth
    k = max(3, n // 50)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    
    # Symmetric random
    half = np.maximum(rng.uniform(0.2, 1.0, n // 2 + 1), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    
    # "Antibump" - high at edges, low in middle
    inits.append(np.maximum(0.2 + 0.8 * (t**2 + (1-t)**2), 0.01))
    
    return inits

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    """Compute linear auto-convolution of a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective value J(a)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse matrix A such that A @ b = conv(a, b)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns normalized b."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if r.success and r.x is not None:
            x = np.maximum(r.x, 0)
            s = x.sum()
            if s > 1e-15:
                return x / s
    except Exception:
        pass
    return None


def lp_min_cross(a):
    """Solve LP: min t s.t. conv(a,b)_k <= t, sum(b)=1, b>=0. Returns (b, t)."""
    n = len(a)
    # Variables: [b_0,...,b_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    for k in range(n_constraints):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A_ub[k, j] = a[k - j]
        A_ub[k, -1] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs', options={'presolve': True})
        if r.success and r.x is not None:
            b = np.maximum(r.x[:n], 0)
            s = b.sum()
            if s > 1e-15:
                return b / s, float(r.x[-1])
    except Exception:
        pass
    return None, None


def smooth_max_grad(a, tau):
    """Compute gradient of smooth approximation to max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad, b, b_max


def run_smooth_gd(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                  lr_init=0.3, lr_end=0.002):
    """Nesterov-accelerated smooth gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def run_lp_mirror(a, deadline, max_iters=100000, verbose=True):
    """LP mirror descent with adaptive line search and stall detection."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    last_t = 0.2
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Solve LP
        g = lp_best_response(a, M)
        if g is None:
            no_imp += 3
            if no_imp > 50:
                break
            continue

        # Adaptive line search
        best_t = 0.0
        best_M_t = M
        # Try a range of step sizes centered around last successful t
        candidates = [last_t, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
        # Remove duplicates and sort
        candidates = sorted(set(candidates), reverse=True)
        for t in candidates:
            if t <= 0 or t >= 1:
                continue
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M_t:
                best_M_t = M_t
                best_t = t

        if best_t > 0 and best_M_t < M - 1e-14:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            a /= a.sum()
            if best_M_t < best_M:
                best_M = best_M_t
                best_a = a.copy()
            last_t = best_t
            no_imp = 0
            stall_count = 0
        else:
            no_imp += 1
            stall_count += 1
            # Stall detection: perturb and restart
            if no_imp > 80 and stall_count > 80:
                rng_local = np.random.default_rng(hash((n, it)) & 0xFFFFFFFF)
                a = a + 0.01 * rng_local.normal(0, 1, n)
                a = np.maximum(a, 1e-4)
                a /= a.sum()
                no_imp = 0
                stall_count = 0
                last_t = 0.1

        if verbose and (it + 1) % 2000 == 0:
            print(f"    LP iter {it+1}: M={M:.8f} best={best_M:.8f} t={last_t:.3f} noimp={no_imp}", flush=True)

    return best_a


def run_lp_alternating(a, deadline, max_iters=5000, verbose=True):
    """Alternating LP: find b minimizing max(a*b), then mix."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b, t_val = lp_min_cross(a)
        if b is None:
            no_imp += 3
            if no_imp > 40:
                break
            continue

        # Line search
        M_cur = fast_conv(a).max()
        best_t = 0.0
        best_M_t = M_cur
        for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03]:
            a_t = (1.0 - t) * a + t * b
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M_t:
                best_M_t = M_t
                best_t = t

        if best_t > 0 and best_M_t < best_M - 1e-14:
            a = (1.0 - best_t) * a + best_t * b
            a = np.maximum(a, 0.0)
            a /= a.sum()
            best_M = best_M_t
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 60:
                break

        if verbose and (it + 1) % 1000 == 0:
            print(f"    Alt iter {it+1}: M={fast_conv(a).max():.8f} best={best_M:.8f}", flush=True)

    return best_a


def resample_seq(src, n):
    """Resample sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.015, n), 0.001))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.001))

    inits.append(np.ones(n) / n)

    for f in [1, 2, 3, 4]:
        inits.append(np.maximum(0.5 + 0.25 * np.cos(2 * np.pi * f * t), 0.01))

    for center in [0.25, 0.5, 0.75]:
        inits.append(np.exp(-6 * (t - center) ** 2) + 0.1)

    for alpha in [0.5, 2.0, 3.0]:
        inits.append(np.maximum(0.2 + 0.8 * t ** alpha, 0.01))

    k = max(3, n // 60)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))

    half = np.maximum(rng.uniform(0.2, 1.0, n // 2 + 1), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)

    inits.append(np.maximum(0.3 + 0.7 * (t ** 2 + (1 - t) ** 2), 0.01))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kw):
    """
    Main search function. Allocates time across:
    - Phase 1 (150s): Smooth GD warm start across multiple lengths
    - Phase 2 (750s): LP mirror descent on best sequence
    - Phase 3 (300s): Multi-length exploration with LP
    - Phase 4 (remaining): Final dense LP polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_deadline = t_start + budget_s - 10.0

    # Track best
    best_seq = None
    best_score = float('inf')

    def record_candidate(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = evaluate_sequence(a.tolist())
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a.copy()
            el = time.monotonic() - t_start
            print(f"  ** {label} J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                record_candidate(parent, "Parent")
    except (NameError, Exception):
        pass

    if best_seq is None:
        best_seq = np.ones(512)
        record_candidate(best_seq, "Init uniform")

    print(f"Starting: n={best_seq.size}, J={best_score:.8f}", flush=True)

    # === PHASE 1: Smooth GD warm start (150s) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Smooth GD Warm Start", flush=True)
    print(f"{'='*60}", flush=True)

    p1_end = t_start + 150.0
    base_n = best_seq.size
    p1_lengths = sorted(set([
        max(128, min(2048, base_n)),
        max(128, min(2048, base_n + 128)),
        max(128, min(2048, base_n - 64)),
    ]))

    time_alloc = (p1_end - t_start) / len(p1_lengths)
    for n in p1_lengths:
        if time.monotonic() > p1_end - 5:
            break
        dl = min(time.monotonic() + time_alloc - 2, p1_end - 3)
        inits = make_inits(n, rng, best_seq)
        local_best = None
        local_best_M = float('inf')
        n_iters = max(100, int(4000 * (512.0 / n)))
        for a0 in inits:
            if time.monotonic() > dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()
            a = run_smooth_gd(a0, dl, max_iters=n_iters,
                              tau_start=1.0, tau_end=120.0,
                              lr_init=0.3, lr_end=0.002)
            M = fast_conv(a).max()
            if M < local_best_M:
                local_best_M = M
                local_best = a.copy()
        if local_best is not None:
            record_candidate(local_best, f"GD n={n}")

    print(f"Phase 1 done: J={best_score:.8f} t={time.monotonic()-t_start:.0f}s", flush=True)

    # === PHASE 2: LP Mirror Descent (750s) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: LP Mirror Descent (main)", flush=True)
    print(f"{'='*60}", flush=True)

    p2_end = t_start + 900.0
    a = best_seq.copy()
    n = a.size
    print(f"  n={n}, starting LP mirror descent...", flush=True)
    a = run_lp_mirror(a, p2_end, max_iters=200000, verbose=True)
    record_candidate(a, "LP Mirror")

    # Try alternating LP from the improved point
    if time.monotonic() < p2_end - 30:
        print(f"  Trying alternating LP...", flush=True)
        a2 = run_lp_alternating(a.copy(), min(time.monotonic() + 120, p2_end - 10),
                                max_iters=3000, verbose=True)
        record_candidate(a2, "Alt LP")

    print(f"Phase 2 done: J={best_score:.8f} t={time.monotonic()-t_start:.0f}s", flush=True)

    # === PHASE 3: Multi-length exploration (300s) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    p3_end = t_start + 1200.0
    base_n = best_seq.size
    lengths_p3 = []
    for d in [-32, -16, 16, 32, 64, 96]:
        nt = base_n + d
        if 128 <= nt <= 2048 and nt not in lengths_p3:
            lengths_p3.append(nt)
    for nt in [640, 768, 896, 1024, 1152]:
        if nt != base_n and nt not in lengths_p3:
            lengths_p3.append(nt)

    remaining = p3_end - time.monotonic()
    if remaining > 20 and lengths_p3:
        per_len = max(10.0, remaining / len(lengths_p3))
        for n in lengths_p3:
            if time.monotonic() > p3_end - 10:
                break
            dl = min(time.monotonic() + per_len, p3_end - 5)
            a_res = resample_seq(best_seq, n)
            a_res = np.maximum(a_res, 1e-6)
            a_res /= a_res.sum()

            # Quick GD
            a_opt = run_smooth_gd(a_res, min(time.monotonic() + per_len * 0.4, dl - 5),
                                  max_iters=300, tau_start=3, tau_end=80,
                                  lr_init=0.2, lr_end=0.01)
            record_candidate(a_opt, f"Res n={n}")

            # LP polish
            if time.monotonic() < dl:
                a_lp = run_lp_mirror(a_opt, dl, max_iters=3000, verbose=False)
                record_candidate(a_lp, f"LP n={n}")

    print(f"Phase 3 done: J={best_score:.8f} t={time.monotonic()-t_start:.0f}s", flush=True)

    # === PHASE 4: Final dense LP polish ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < total_deadline - 5 and best_seq is not None:
        a = best_seq.copy()
        a = run_lp_mirror(a, total_deadline, max_iters=500000, verbose=True)
        record_candidate(a, "Final LP")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512)
        best_score = compute_J(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
