import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.signal import fftconvolve
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_gradient(a, tau):
    """Compute gradient of (1/tau)*logsumexp(tau * conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Numerical stability
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    # grad[i] = 2 * sum_k p[k] * a[k-i] for valid k,i
    # This is cross-correlation of p (len 2n-1) with a (len n)
    # np.correlate(p, a, mode='valid')[i] = sum_m p[i+m] * a[m]
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad, p, b

def project_normalize(a):
    """Project to a >= 0 and sum(a) = 1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def gradient_descent_smooth(a, deadline, tau_init=10, tau_max=200, lr_init=0.5, 
                              momentum=0.9, verbose=True):
    """Smooth gradient descent with adaptive tau and momentum."""
    a = project_normalize(a.astype(np.float64))
    n = len(a)

    v = np.zeros(n)

    lr = lr_init

            tau = min(tau * 1.05, tau_max)
        if (it + 1) % 50 == 0:
            lr = max(lr * 0.99, 0.01)
    
    return best_a, best_J

def build_sparse_conv_matrix(n, a, active_indices=None):
    """Build sparse matrix for convolution constraints."""
    if active_indices is None:
        active_indices = range(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    return csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))

def solve_lp(a, M, active_indices=None, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    if active_indices is None:
        active_indices = list(range(2 * n - 1))
    
    A = build_sparse_conv_matrix(n, a, active_indices)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-8})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def lp_iteration(a, M, top_k=None, time_limit=3):
    """One LP iteration with adaptive constraint selection."""
    n = len(a)
    b = fast_conv(a)
    
    if top_k is not None and top_k < 2 * n - 1:
        # Select top_k constraints
        top_idx = np.argsort(b)[-top_k:]
        top_idx = np.sort(top_idx)
    else:
        top_idx = np.arange(2 * n - 1)
    
    g = solve_lp(a, M, active_indices=top_idx, time_limit=time_limit)
    
    if g is None:
        return None
    
    # Verify and add violated constraints
    if top_k is not None and top_k < 2 * n - 1:
        ag = fftconvolve(a, g)[:2*n-1]
        violations = np.where(ag > M * 1.001)[0]
        if len(violations) > 0:
            # Add violated constraints and re-solve
            all_idx = np.sort(np.concatenate([top_idx, violations]))
            g = solve_lp(a, M, active_indices=all_idx, time_limit=time_limit)
            if g is None:
                return None
    
    return g

def lp_mirror_descent(a, deadline, use_adaptive_constraints=True, 
                       initial_top_k=300, max_top_k=1023, verbose=True,

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

        if g is None or g.sum() < 1e-10:
            break
        
        g = g / g.sum()
        
        # Line search
        diff = np.max(np.abs(g - a))
        base_t = min(0.5, max(0.1, diff))
        
        improved = False
        for t in [base_t, base_t*0.5, base_t*0.25, 0.05, 0.02]:
            a_trial = (1-t)*a + t*g
            a_trial = project_normalize(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:

        if no_improve > 200:
            rng = np.random.default_rng(it * 7919)

            a = a * (1 + scale * rng.normal(0, 1, n))

            print(f"    LP iter {it}: J={best_J:.8f} M={M:.6f} top_k={top_k}", flush=True)

def resample(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Multi-strategy optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Combines: (1) FFT-based smooth gradient descent with adaptive temperature,
(2) LP-based mirror descent with adaptive constraint management,
(3) spectral (Gerchberg-Saxton) projection for escaping local minima,
(4) multi-length exploration and final full-LP polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_gradient(a, tau):
    """Compute gradient of (1/tau)*logsumexp(tau*conv(a,a)) using FFT."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    # grad[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation)
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad, p, b


def project_normalize(a):
    """Project to a >= 0, sum(a) = 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def gradient_descent(a, deadline, tau_init=10.0, tau_max=300.0, lr_init=0.5,
                     momentum=0.9, verbose=False, label=""):
    """Smooth gradient descent with adaptive tau and momentum."""
    a = project_normalize(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    v = np.zeros(n)
    tau = tau_init
    lr = lr_init
    no_improve = 0

    for it in range(200000):
        if time.monotonic() > deadline:
            break
        grad, p, b = smooth_gradient(a, tau)
        M = b.max()
        J = 2.0 * n * M

        if J < best_J - 1e-13:
            best_J = J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        v = momentum * v + (1.0 - momentum) * grad
        a = a - lr * v
        a = project_normalize(a)

        if no_improve > 200:
            tau = min(tau * 1.08, tau_max)
            if no_improve > 500:
                lr = max(lr * 0.95, 0.05)

        if verbose and (it + 1) % 2000 == 0:
            print(f"    GD iter {it+1}: J={best_J:.8f} tau={tau:.1f} lr={lr:.4f}", flush=True)

    return best_a, best_J


def spectral_projection(a, n_iter=100, deadline=None):
    """Gerchberg-Saxton type spectral projection for flat autocorrelation."""
    a = project_normalize(a.astype(np.float64))
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    # Target: flat autocorrelation b_k = 1/(2n-1) for all k (since sum(a)=1)
    b_target = np.zeros(2 * n - 1)
    b_target[:] = 1.0 / (2 * n - 1)
    b_target_padded = np.zeros(size)
    b_target_padded[:2 * n - 1] = b_target

    B_target = np.fft.rfft(b_target_padded)
    # Clamp negative values
    B_target = np.maximum(B_target, 1e-10)
    desired_mag = np.sqrt(B_target)

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, size)
        phase = np.angle(A)
        A_new = desired_mag * np.exp(1j * phase)
        a = np.fft.irfft(A_new, size)[:n]
        a = project_normalize(a)

    return a


def build_sparse_conv_matrix(n, a, active_indices):
    """Build sparse matrix for convolution constraints at given indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))


def solve_lp(a, M, active_indices, time_limit=3.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_indices, g >= 0."""
    n = len(a)
    A = build_sparse_conv_matrix(n, a, active_indices)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None


def lp_step(a, M, top_k, time_limit=2.0):
    """One LP iteration with adaptive constraint expansion."""
    n = len(a)
    b = fast_conv(a)

    if top_k < 2 * n - 1:
        top_idx = np.sort(np.argsort(b)[-top_k:])
    else:
        top_idx = np.arange(2 * n - 1)

    g = solve_lp(a, M, top_idx, time_limit=time_limit)
    if g is None:
        return None

    if top_k < 2 * n - 1:
        ag = fast_cross_conv(a, g)
        viol = np.where(ag > M * 1.001)[0]
        if len(viol) > 0:
            all_idx = np.sort(np.unique(np.concatenate([top_idx, viol])))
            g = solve_lp(a, M, all_idx, time_limit=time_limit)
            if g is None:
                return None

    return g


def lp_mirror_descent(a, deadline, initial_top_k=300, max_top_k=None,
                      verbose=False, period=100, max_no_improve=300):
    """LP mirror descent with adaptive constraint management."""
    a = project_normalize(a.astype(np.float64))
    n = len(a)
    if max_top_k is None:
        max_top_k = 2 * n - 1
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = float(b.max())

        top_k = min(initial_top_k + it // period * 40, max_top_k)
        g = lp_step(a, M, top_k, time_limit=2.0)

        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()
        diff = np.max(np.abs(g - a))
        base_t = min(0.4, max(0.08, diff * 1.5))

        improved = False
        current_J = compute_J(a)
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02, 0.01]:
            a_trial = project_normalize((1.0 - t) * a + t * g)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
            elif J_trial < current_J - 1e-13:
                a = a_trial
                current_J = J_trial
                no_improve = 0
                break

        if not improved:
            no_improve += 1

        if no_improve > max_no_improve:
            scale = 0.12 / (1.0 + no_improve / max_no_improve)
            a = a * (1.0 + scale * rng.normal(0, 1, n))
            a = project_normalize(a)
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    LP iter {it}: J={best_J:.8f} top_k={top_k} no_imp={no_improve}", flush=True)

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def local_search(a, deadline, n_trials=1000):
    """Random coordinate perturbation search."""
    a = project_normalize(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    rng = np.random.default_rng(999)

    for it in range(n_trials):
        if time.monotonic() > deadline:
            break
        # Perturb a small random subset
        n_pert = rng.integers(3, max(4, n // 32))
        indices = rng.choice(n, size=n_pert, replace=False)
        a_trial = a.copy()
        for idx in indices:
            a_trial[idx] = max(a[idx] * rng.uniform(0.5, 1.5), 0.0)
        a_trial = project_normalize(a_trial)
        J_trial = compute_J(a_trial)
        if J_trial < best_J - 1e-14:
            best_J = J_trial
            best_a = a_trial.copy()
            a = a_trial.copy()

    return best_a, best_J
