def build_conv_matrix_with_a(n, a):
    """Build sparse matrix A(a) such that (A @ g)[k] = (a * g)[k]"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_range_minimize(n, a, M, time_limit=10):
    """
    Find g that minimizes the range of conv(a, g) = max - min.
    Formulation: minimize t + s subject to conv(a,g) <= M + t, conv(a,g) >= M - s,
    sum(g) = 1, g >= 0, t >= 0, s >= 0.
    
    Actually, since a >= 0 and g >= 0, conv(a,g) >= 0, so min is at least 0.
    Since we're looking at the shape, let's just minimize the max directly:
    minimize max_k conv(a,g)_k subject to sum(g)=1, g>=0
    This is: minimize t subject to A@g <= t*1, sum(g)=1, g>=0, t>=0
    """
    A = build_conv_matrix_with_a(n, a)
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: A@g - t <= 0 for each row
    # A is (2n-1) x n, so we need (2n-1) x (n+1)
    A_full = A.tolil()
    # Add -1 column for t
    t_col = -np.ones(2 * n - 1)
    from scipy.sparse import csr_matrix as _csr
    A_full = _csr(np.hstack([A.todense(), t_col.reshape(-1, 1)]))
    
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        r = linprog(c, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x[:n], 0)
    except Exception:
        pass
    return None

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent (log-sum-exp) with Nesterov for initialization
2. LP mirror descent with two LP formulations (peak-min vs sum-max)
3. Adaptive step sizes, periodic perturbations, and multi-start
Key improvement over baseline: dedicated LP phase with peak-minimization LP,
longer LP runtime, and adaptive perturbation strategy.

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def build_A_dense(n, a):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lp_peak_min(n, a, time_limit=5):
    """Minimize max_k (a*g)_k subject to sum(g)=1, g>=0.
    Variables: [g, t]. Minimize t.
    Constraints: A@g - t <= 0, sum(g) = 1, g>=0, t>=0.
    """
    A = build_A_dense(n, a)
    # A_ub: [A | -1], b_ub = 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def lp_sum_max(n, a, M, time_limit=5):
    """Maximize sum(g) subject to (a*g)_k <= M, g>=0."""
    A = build_A_dense(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            g = np.maximum(r.x, 0)
            return g / g.sum()
    except Exception:
        pass
    return None


def smooth_gradient_step(a, tau):
    """Compute gradient of smooth max (log-sum-exp) of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    # Vectorized gradient computation
    # For each k, grad[j] += 2 * p[k] * a[k-j] for j in valid range
    # We can use the matrix A: grad = 2 * A.T @ (p * a_reversed)
    # Actually: (A @ a)[k] = sum_j A[k,j] * a[j] = sum_j a[k-j] * a[j] = (a*a)[k]
    # d/d a[j] of (a*a)[k] = 2 * a[k-j] (for valid j)
    # So grad[j] = 2 * sum_k p[k] * a[k-j] (for valid k,j)
    # = 2 * (A.T @ (p))[j] * ... hmm not quite
    # Let me just use the loop, it's fast enough for n=512
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - idx]
    return grad


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=3.0, tau_end=200.0,
                        lr_init=0.3, lr_end=0.003):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad = smooth_gradient_step(y, tau)
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
        
        if it % 500 == 0:
            M_curr = fast_conv(a).max()
            if M_curr < best_M * (1 - 1e-10):
                best_M = M_curr
                best_a = a.copy()

    if M_curr < best_M:

        best_M = M_curr

def run_lp_mirror(a, deadline, max_iters=50000, use_peak_lp=True, perturb_every=500):

    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    it = 0

        it += 1

        M = float(b.max())

        g = None
        if use_peak_lp:

            g = lp_sum_max(n, a, M, time_limit=3)
        if g is None:

            if s < 1e-10:

                a = np.maximum(a + noise, 0)
                a /= a.sum()
                no_imp = 0
            if no_imp > 200:
                break
        
        if it % 500 == 0:
            print(f"    LP iter {it}: M={best_M:.10f} J={2*n*best_M:.6f}", flush=True)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent (log-sum-exp) with Nesterov for initialization
2. Alternating LP mirror descent with peak-minimization LP as primary method
3. Adaptive step sizes, dual-LP alternating, periodic perturbations, multi-start
Key improvements: alternating LP (solving LP in both a and g directions),
peak-min LP formulation, longer dedicated LP runtime, and structured restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def build_A_dense(n, a):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lp_peak_min(n, a, time_limit=5):
    """Minimize max_k (a*g)_k s.t. sum(g)=1, g>=0. Returns normalized g."""
    A = build_A_dense(n, a)
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_sum_max(n, a, M, time_limit=5):
    """Maximize sum(g) s.t. (a*g)_k <= M, g>=0. Returns normalized g."""
    A = build_A_dense(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-10:
                return g / g.sum()
    except Exception:
        pass
    return None


def smooth_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - idx]
    return grad


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=3.0, tau_end=200.0,
                        lr_init=0.3, lr_end=0.003):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad = smooth_grad(y, tau)
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
        if it % 300 == 0:
            M_curr = fast_conv(a).max()
            if M_curr < best_M * (1 - 1e-10):
                best_M = M_curr
                best_a = a.copy()
    M_curr = fast_conv(a).max()
    if M_curr < best_M:
        best_a = a.copy()
    return best_a


def run_lp_mirror_v2(a, deadline, max_iters=50000, rng=None):
    """Alternating LP mirror descent with peak-min LP and sum-max LP."""
    if rng is None:
        rng = np.random.default_rng(42)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    it = 0
    A_cache = None
    a_cache = None

    while it < max_iters and time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = float(b.max())

        # Build or update A
        if A_cache is None or not np.allclose(a, a_cache, atol=1e-12):
            A_cache = build_A_dense(n, a)
            a_cache = a.copy()

        # Try peak-min LP
        g1 = lp_peak_min(n, a, time_limit=4)

        # Try alternating: solve LP with g1 as fixed
        a_alt = None
        if g1 is not None:
            a_alt = lp_peak_min(n, g1, time_limit=4)

        # Try sum-max LP as fallback/alternative
        g2 = lp_sum_max(n, a, M, time_limit=4)

        # Try all candidate directions with line search
        improved = False
        candidates = []
        if a_alt is not None:
            candidates.append(('alt', a_alt, [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]))
        if g1 is not None:
            candidates.append(('g1', g1, [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]))
        if g2 is not None:
            candidates.append(('g2', g2, [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]))

        for name, g, steps in candidates:
            if time.monotonic() > deadline:
                break
            for t in steps:
                a_t = (1.0 - t) * a + t * g
                a_t = np.maximum(a_t, 0.0)
                s = a_t.sum()
                if s < 1e-10:
                    continue
                a_t /= s
                M_t = float(fast_conv(a_t).max())
                if M_t < M - 1e-15:
                    a = a_t
                    A_cache = None
                    if M_t < best_M:
                        best_M = M_t
                        best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 100 == 0:
                # Perturb to escape local optimum
                scale = 0.001 * min(1.0, no_imp / 200.0)
                noise = rng.normal(0, scale, n)
                a = np.maximum(a + noise, 0)
                a /= a.sum()
                A_cache = None
                no_imp = 0
            if no_imp > 500:
                break

        if it % 1000 == 0:
            J_curr = 2 * n * best_M
            print(f"    LP iter {it}: M={best_M:.10f} J={J_curr:.6f}", flush=True)

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
