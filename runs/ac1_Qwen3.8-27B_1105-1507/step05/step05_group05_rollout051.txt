    n = len(a)

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using fast vectorized coordinate descent
with explicit convolution tracking, multiple initialization strategies (including symmetric
sequences), and adaptive restarts. Key innovation: O(n) vectorized coordinate updates with
golden-section 1D subproblem solvers, enabling thousands of sweeps within the time budget.

import numpy as np
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coord_opt_vec(b_slice, a_slice, x, base_max, has_2i, idx_2i):
    """
    Golden section search for optimal y for one coordinate.
    b_slice: b values in affected range (c_k + m_k*x part already in b)
    a_slice: a[k-i] values for k in affected range
    x: current value of a[i]
    base_max: max of b outside affected range
    has_2i: whether k=2i is in the affected range
    idx_2i: index of k=2i in the slice (if has_2i)
    """
    n_slice = len(b_slice)
    
    def obj(y_val):
        new_S = 1.0 - x + y_val
        if new_S < 1e-12:
            return 1e10
        vals = b_slice + 2.0 * (y_val - x) * a_slice
        if has_2i:
            vals = vals.copy()
            vals[idx_2i] = b_slice[idx_2i] - x * x + y_val * y_val
        m = max(base_max, vals.max())
        return m / (new_S * new_S)
    
    lo = 0.0
    hi = max(x * 2.0 + 0.1, 0.5)
    
    # Expand hi if function still decreasing
    f_hi_half = obj(hi * 0.5)
    f_hi = obj(hi)
    if f_hi < f_hi_half and hi < 5.0:
        hi *= 2.0
        if obj(hi) < f_hi and hi < 20.0:
            hi *= 2.0
    
    gr = (np.sqrt(5) - 1) / 2
    a_gs = hi - gr * (hi - lo)
    b_gs = lo + gr * (hi - lo)
    fa = obj(a_gs)
    fb = obj(b_gs)

            hi = b_gs
            b_gs = a_gs
            fb = fa
            a_gs = hi - gr * (hi - lo)
            fa = obj(a_gs)
        else:
            lo = a_gs
            a_gs = b_gs
            fa = fb
            b_gs = lo + gr * (hi - lo)
            fb = obj(b_gs)

    f_opt = obj(y_opt)
    f_zero = obj(0.0)
    if f_zero < f_opt:
        y_opt = 0.0
    
    return y_opt

    no_improve = 0

                    a[n - 1 - i] = y_opt

            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            break

Let me also think about whether there's a smarter approach to the 1D subproblem.

The objective is: f(y) = max(A, max_j (c_j + m_j * y), c_q + y^2) / (1 - x + y)^2

where A = base_max, c_j + m_j*y are the linear functions, c_q + y^2 is the quadratic.

The minimum of this function occurs where the "envelope" is as flat as possible. We can solve this more efficiently by:
1. Finding which k achieves the max at the current y
2. Checking if the derivative is zero
3. If not, the max will switch to another k at some critical y value

This is essentially a parametric LP / piecewise linear optimization. But golden section is simple and works well enough.

Let me also consider: should I optimize with or without the normalization constraint? 

Option 1: Minimize max(conv(a,a)) subject to sum(a) = 1, a >= 0.
- CD: at each step, change a[i] to y, then renormalize. The 1D subproblem includes the renormalization.

Option 2: Minimize max(conv(a,a)) / (sum(a))^2 directly (scale-invariant).
- CD: at each step, change a[i] to y. The 1D subproblem is max(b_k + delta_k(y)) / (S + delta)^2.
- No need to renormalize.

Option 2 is simpler. Let me use that.

With option 2:
- S = sum(a) (not necessarily 1)
- Changing a[i] from x to y: new_S = S - x + y
- f(y) = max(b_k + delta_k(y)) / (new_S)^2
- delta_k(y) = 2*(y-x)*a[k-i] for k in range, k != 2i
- delta_{2i}(y) = y^2 - x^2

This avoids the renormalization step and is cleaner.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using fast vectorized coordinate descent
with explicit convolution tracking and golden-section 1D solves. Employs multiple diverse
initializations, random restarts, and adaptive length exploration to escape local optima.

import numpy as np
import time

GR = (np.sqrt(5) - 1) / 2.0


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def solve_1d(b_slice, a_slice, x, base_max, has_2i, idx_2i, S_minus_x):
    """
    Golden section search for optimal y for one coordinate.
    Minimizes max(b_k + delta_k(y)) / (S_minus_x + y)^2
    """
    def obj(y_val):
        new_S = S_minus_x + y_val
        if new_S < 1e-12:
            return 1e10
        vals = b_slice + 2.0 * (y_val - x) * a_slice
        if has_2i:
            vals = vals.copy()
            vals[idx_2i] = b_slice[idx_2i] - x * x + y_val * y_val
        m = vals.max()
        if base_max > m:
            m = base_max
        return m / (new_S * new_S)
    
    lo = 0.0
    hi = max(x * 3.0 + 0.1, S_minus_x * 0.5)
    if hi < 0.1:
        hi = 0.1
    
    # Expand if needed
    f_mid = obj(0.5 * hi)
    f_hi = obj(hi)
    if f_hi < f_mid and hi < 50.0:
        hi *= 2.0
    
    a_gs = hi - GR * (hi - lo)
    b_gs = lo + GR * (hi - lo)
    fa = obj(a_gs)
    fb = obj(b_gs)
    
    for _ in range(45):
        if fa < fb:
            hi = b_gs
            b_gs = a_gs
            fb = fa
            a_gs = hi - GR * (hi - lo)
            fa = obj(a_gs)
        else:
            lo = a_gs
            a_gs = b_gs
            fa = fb
            b_gs = lo + GR * (hi - lo)
            fb = obj(b_gs)
    
    y_opt = 0.5 * (lo + hi)
    if obj(0.0) < obj(y_opt):
        y_opt = 0.0
    return y_opt


def run_cd(a_init, max_sweeps, deadline, seed=0, restart_every=30):
    """Fast coordinate descent with periodic random restarts."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    n = len(a)
    b = np.convolve(a, a)
    
    best_b_max = b.max()
    best_a = a.copy()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        # Random restart perturbation if stuck
        if no_improve >= restart_every:
            a = a + rng.normal(0, 0.01 * a.mean(), n)
            a = np.maximum(a, 0)
            b = np.convolve(a, a)
            no_improve = 0
        
        S = a.sum()
        order = np.arange(n)
        rng.shuffle(order)
        
        for i in order:
            x = a[i]
            S_minus_x = S - x
            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            
            # Base max outside
            if k_min == 0:
                base_max = b[k_max + 1:].max() if k_max < 2 * n - 2 else 0.0
            else:
                if k_max < 2 * n - 2:
                    base_max = max(b[:k_min].max(), b[k_max + 1:].max())
                else:
                    base_max = b[:k_min].max()
            
            b_slice = b[k_min:k_max + 1]
            a_slice = a[0:k_max - k_min + 1]
            
            has_2i = (2 * i >= k_min) and (2 * i <= k_max)
            idx_2i = 2 * i - k_min if has_2i else -1
            
            y_opt = solve_1d(b_slice, a_slice, x, base_max, has_2i, idx_2i, S_minus_x)

                delta = y_opt - x
                a[i] = y_opt
                S += delta
                b[k_min:k_max + 1] = b_slice + 2.0 * delta * a_slice
                if has_2i:
                    b[2 * i] = b_slice[idx_2i] - x * x + y_opt * y_opt
        
        b_max = b.max()
        # Compare with normalization: J = 2n * b_max / S^2
        # But since J is scale-invariant, we can just compare b_max / S^2
        current_obj = b_max / (S * S)
        best_obj = best_b_max / (best_a.sum() ** 2)

            best_b_max = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
