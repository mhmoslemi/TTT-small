import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v, target_sum=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * (np.arange(1, n + 1)) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, K_active):
    n = len(a)
    g = np.zeros(n)
    for k in K_active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(K_active)
    return g


def run_subgradient(a, max_iters, deadline, K_top=5, momentum=0.9,
                    init_step=None, step_decay=0.999, min_step=1e-8):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    B = fast_conv(a)
    peak = B.max()
    if init_step is None:
        alpha = peak / (np.linalg.norm(compute_subgradient(a,
                   [np.argmax(B)])) ** 2 + 1e-15)
        alpha = min(alpha, 0.5)
    else:
        alpha = init_step

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        B = fast_conv(a)
        M = B.max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()

        top_k = np.argsort(B)[::-1][:K_top]
        g = compute_subgradient(a, top_k.tolist())
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        v = momentum * v - alpha * g
        a_trial = a + v
        a_trial = project_simplex(a_trial, 1.0)

        B_trial = fast_conv(a_trial)
        M_trial = B_trial.max()

        if M_trial < M:
            a = a_trial
        else:
            alpha *= step_decay
            if alpha < min_step:
                alpha = 0.1 * peak / (g_norm * g_norm + 1e-15)
                step_decay = 0.99
                v = 0.9 * v
            a_trial = a - alpha * g / g_norm
            a_trial = project_simplex(a_trial, 1.0)
            B_trial = fast_conv(a_trial)
            if B_trial.max() < M:
                a = a_trial

        if it % 500 == 0:
            print(f"        SG iter {it}: M={M:.10f} best_M={best_M:.10f} "
                  f"alpha={alpha:.6e}", flush=True)

    return best_a


def pairwise_swap_step(a, B, rng, n_swaps=100, search_iters=30):
    n = len(a)
    best_M = B.max()
    for _ in range(n_swaps):
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if a[j] < 1e-12:
            continue

        d = np.zeros(2 * n - 1)
        lo_i = max(0, 0)
        hi_i = min(n - 1, 2 * n - 2)
        for k in range(2 * n - 1):
            ki = k - i
            kj = k - j
            val = 0.0
            if 0 <= ki < n:
                val += a[ki]
            if 0 <= kj < n:
                val -= a[kj]
            d[k] = val

        d_vec = 2.0 * d
        delta_lo = -a[i]
        delta_hi = a[j]

        lo = delta_lo
        hi = delta_hi
        for _ in range(search_iters):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.max(B + m1 * d_vec)
            f2 = np.max(B + m2 * d_vec)
            if f1 < f2:
                hi = m2
            else:
                lo = m1

        delta = (lo + hi) / 2.0
        delta = max(delta_lo, min(delta_hi, delta))

        if abs(delta) < 1e-12:
            continue

        a[i] += delta
        a[j] -= delta
        a[i] = max(0.0, a[i])
        a[j] = max(0.0, a[j])
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        B = fast_conv(a)
        M_new = B.max()
        if M_new < best_M:
            best_M = M_new

    return a, best_M


def pairwise_sweep(a, B, rng, n_rounds=5, swaps_per_round=200):
    n = len(a)
    best_M = B.max()
    for r in range(n_rounds):
        order = rng.permutation(n)
        for idx in range(0, n, max(1, n // swaps_per_round)):
            i = order[idx]
            j = rng.integers(n)
            if i == j or a[j] < 1e-12:
                continue

            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                ki = k - i
                kj = k - j
                val = 0.0
                if 0 <= ki < n:
                    val += a[ki]
                if 0 <= kj < n:
                    val -= a[kj]
                d[k] = val

            d_vec = 2.0 * d
            delta_lo = -a[i]
            delta_hi = a[j]
            lo = delta_lo
            hi = delta_hi
            for _ in range(25):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(B + m1 * d_vec)
                f2 = np.max(B + m2 * d_vec)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            delta = (lo + hi) / 2.0
            delta = max(delta_lo, min(delta_hi, delta))
            if abs(delta) < 1e-12:
                continue

            a[i] += delta
            a[j] -= delta
            a[i] = max(0.0, a[i])
            a[j] = max(0.0, a[j])
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s
            B = fast_conv(a)
            if B.max() < best_M:
                best_M = B.max()

    return a, best_M


def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx, max_time=10):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': max_time, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def line_search_analytic(a, g):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = fast_conv(a).max()
        g = solve_lp(a, M, row_idx, col_idx, max_time=8)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        t, M_new = line_search_analytic(a, g)
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = fast_conv(a).max()
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 80:
            break
        if verbose and it % 100 == 0:
            print(f"      LP iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_shaped_init(n, shape_type, param):
    x = np.linspace(0, 1, n)
    xc = x - 0.5
    if shape_type == 'gaussian':
        a = np.exp(-param * xc * xc * 4)
    elif shape_type == 'raised_cos':
        a = 0.5 * (1 + np.cos(2 * np.pi * xc)) ** param
    elif shape_type == 'sine':
        a = np.sin(np.pi * x) ** param
    elif shape_type == 'tent':
        a = np.maximum(0, 1 - param * np.abs(xc * 2))
    elif shape_type == 'w_shape':
        a = 0.5 * (1 + np.cos(4 * np.pi * xc)) ** param + 0.3 * np.exp(-8 * param * xc * xc)
    elif shape_type == 'flat_top':
        w = 0.3 * param
        a = np.where(np.abs(xc) < w, 1.0, np.maximum(0, 1 - (np.abs(xc) - w) / (0.5 - w)))
    elif shape_type == 'inverse_gaussian':
        a = 1.0 / (1 + param * xc * xc * 4)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


class Tracker:
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def optimize_sequence(a, deadline, rng, row_idx, col_idx, n_sg_iters=2000,
                      n_swap_rounds=3, swaps_per_round=300, n_lp_iters=2000,
                      K_top=5, verbose=False):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    if time.monotonic() < deadline - 10:
        a = run_subgradient(a, n_sg_iters, deadline, K_top=K_top, verbose=verbose)

    if time.monotonic() < deadline - 10:
        B = fast_conv(a)
        a, _ = pairwise_sweep(a, B, rng, n_rounds=n_swap_rounds,
                              swaps_per_round=swaps_per_round)

    if time.monotonic() < deadline - 5:
        a = run_subgradient(a, 500, min(time.monotonic() + 10, deadline), K_top=K_top)

    if time.monotonic() < deadline - 3:
        a = run_lp_mirror_descent(a, n_lp_iters, deadline, row_idx, col_idx, verbose=verbose)

    return a


def propose_candidate(seed=42, budget_s=1500):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0
    tracker = Tracker(t_start)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    # Phase 0: Parameterized initialization sweep
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 0: Parameterized Initialization Sweep", flush=True)
    print(f"{'='*60}", flush=True)

    n_test = parent.size if parent.size <= 2048 else 512
    best_init = None
    best_init_score = float('inf')

    shapes = [
        ('gaussian', [1, 2, 3, 4, 5, 6, 8, 10, 12, 15, 20]),
        ('raised_cos', [1, 1.5, 2, 3, 4, 5, 8]),
        ('sine', [1, 1.5, 2, 2.5, 3, 4, 5, 6, 8]),
        ('tent', [1, 1.5, 2]),
        ('w_shape', [1, 2, 3, 4]),
        ('flat_top', [0.5, 0.8, 1.0, 1.5, 2.0]),
        ('inverse_gaussian', [1, 2, 3, 5, 8, 12]),
    ]

    for shape_type, params in shapes:
        for param in params:
            a = generate_shaped_init(n_test, shape_type, param)
            score = compute_J(a)
            if score < best_init_score:
                best_init_score = score
                best_init = a.copy()
            if time.monotonic() > t_start + 15:
                break
        if time.monotonic() > t_start + 15:
            break

    tracker.update(best_init, f"(P0 init J={best_init_score:.6f})")
    print(f"  Best init: J={best_init_score:.8f}", flush=True)

    # Also evaluate parent at same n for comparison
    if parent.size != n_test:
        parent_resampled = resample_seq(parent, n_test)
        parent_resampled = parent_resampled / parent_resampled.sum()
        tracker.update(parent_resampled, "(P0 parent resampled)")

    # Use the better of best_init and (resampled) parent as starting point
    start_a = best_init.copy()
    if tracker.best is not None and tracker.best.size == n_test:
        parent_r = resample_seq(parent, n_test)
        parent_r = parent_r / max(parent_r.sum(), 1e-10)
        if compute_J(parent_r) < best_init_score:
            start_a = parent_r.copy()

    # Phase 1: Coarse-to-fine optimization
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Coarse-to-Fine Optimization", flush=True)
    print(f"{'='*60}", flush=True)

    coarse_n = max(32, n_test // 8)
    a_coarse = resample_seq(start_a, coarse_n)
    a_coarse = np.maximum(a_coarse, 1e-10)
    a_coarse = a_coarse / a_coarse.sum()

    r_c, c_c = build_lp_matrix(coarse_n)
    coarse_deadline = min(t_start + 60, deadline - 200)
    a_coarse = optimize_sequence(a_coarse, coarse_deadline, rng, r_c, c_c,
                                 n_sg_iters=3000, n_swap_rounds=5,
                                 swaps_per_round=200, n_lp_iters=3000,
                                 K_top=5, verbose=True)
    tracker.update(a_coarse, f"(P1 coarse n={coarse_n})")

    # Upscale and optimize at full n
    a_mid = resample_seq(a_coarse, max(n_test, 256))
    a_mid = np.maximum(a_mid, 1e-10)
    a_mid = a_mid / a_mid.sum()

    # Also try the direct start
    candidates = [a_mid, start_a]

    for ci, cand in enumerate(candidates):
        if time.monotonic() > deadline - 100:
            break
        n_cand = cand.size
        r_i, c_i = build_lp_matrix(n_cand)
        sub_deadline = min(t_start + 250 + ci * 100, deadline - 100)
        a_opt = optimize_sequence(cand, sub_deadline, rng, r_i, c_i,
                                  n_sg_iters=3000, n_swap_rounds=4,
                                  swaps_per_round=400, n_lp_iters=2000,
                                  K_top=5, verbose=(ci == 0))
        tracker.update(a_opt, f"(P1 full n={n_cand} cand={ci})")

    # Phase 2: Multi-n exploration
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-n Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_ns = set()
        for mult in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
            nt = int(base_n * mult)
            if 128 <= nt <= 2048:
                trial_ns.add(nt)
        trial_ns = sorted(trial_ns)

        phase2_deadline = min(t_start + budget_s * 0.70, deadline - 80)
        for nt in trial_ns:
            if time.monotonic() > phase2_deadline:
                break
            a_init = resample_seq(tracker.best, nt)
            a_init = np.maximum(a_init, 1e-8)
            a_init = a_init / a_init.sum()

            r_i, c_i = build_lp_matrix(nt)
            sub_dl = min(time.monotonic() + 80, phase2_deadline)
            a_opt = optimize_sequence(a_init, sub_dl, rng, r_i, c_i,
                                      n_sg_iters=2000, n_swap_rounds=3,
                                      swaps_per_round=300, n_lp_iters=1500,
                                      K_top=5, verbose=False)
            tracker.update(a_opt, f"(P2 n={nt})")

    # Phase 3: Aggressive local optimization on best
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Aggressive Local Optimization", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_b, c_b = build_lp_matrix(n_best)

        # Multiple perturbation + optimize cycles
        phase3_deadline = min(t_start + budget_s * 0.85, deadline - 50)
        for r in range(15):
            if time.monotonic() > phase3_deadline:
                break

            a = tracker.best.copy().astype(np.float64)
            temp = 0.05 * (0.9 ** r)
            a = a + rng.normal(0, temp * 0.01, n_best)
            a = np.maximum(a, 1e-8)
            a = a / a.sum()

            sub_dl = min(time.monotonic() + 30, phase3_deadline)
            a_opt = optimize_sequence(a, sub_dl, rng, r_b, c_b,
                                      n_sg_iters=1000, n_swap_rounds=2,
                                      swaps_per_round=200, n_lp_iters=1000,
                                      K_top=5, verbose=False)
            tracker.update(a_opt, f"(P3 r={r} T={temp:.4f})")

    # Phase 4: Final extended polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final Extended Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)
        a = tracker.best.copy()
        a = a / a.sum()

        # Subgradient polish
        if time.monotonic() < deadline - 20:
            a_sg = run_subgradient(a, 3000, min(time.monotonic() + 40, deadline - 20),
                                   K_top=7, momentum=0.92, verbose=True)
            tracker.update(a_sg, "(P4 SG)")
            a = a_sg

        # Pairwise sweep
        if time.monotonic() < deadline - 15:
            B = fast_conv(a)
            a_pw, _ = pairwise_sweep(a, B, rng, n_rounds=10, swaps_per_round=500)
            tracker.update(a_pw, "(P4 pairwise)")
            a = a_pw

        # LP polish
        if time.monotonic() < deadline - 5:
            a_lp = run_lp_mirror_descent(a, 50000, deadline, r_f, c_f, verbose=True)
            tracker.update(a_lp, "(P4 LP final)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
