"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smooth gradient descent with log-softmax parameterization
2. LP mirror descent for local refinement
3. Multi-length exploration and diverse restarts
4. Simulated annealing for local minimum escape
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective value."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_gradient_fft(a, p):
    """
    Compute gradient of 2*n*LSE_tau(c) with respect to a.
    a: length n, sum = 1
    p: length 2n-1, softmax(tau * c)
    Returns gradient of length n.
    """
    n = len(a)
    # grad[i] = 4*n * sum_{j=0}^{n-1} p[i+j] * a[j]
    # This equals (reverse(a) * p)[n-1 : 2n-1] * 4*n
    a_rev = a[::-1]
    total_len = n + (2 * n - 1) - 1  # = 3n - 2
    size = 1
    while size < total_len:
        size <<= 1
    fa = np.fft.rfft(a_rev, size)
    fp = np.fft.rfft(p, size)
    conv = np.fft.irfft(fa * fp, size)
    grad = 4.0 * n * conv[n - 1:2 * n - 1]
    return grad


def softmax(x):
    """Numerically stable softmax."""
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    return exp_x / exp_x.sum()


def run_smooth_gradient(a_init, deadline, max_iters=3000, tau_start=2.0, tau_end=300.0,
                        lr_init=0.1, lr_end=0.001, momentum=0.9):
    """
    Smooth gradient descent with log-softmax parameterization.
    a_init: initial sequence (will be normalized)
    Returns best sequence found.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    # Initialize x such that softmax(x) ≈ a
    # x = log(a) works but can have issues with very small values
    x = np.log(a + 1e-15)
    # Center x for numerical stability
    x = x - x.mean()
    
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        # Compute current a from x
        a = softmax(x)
        
        # Compute convolution and softmax
        c = fast_conv(a)
        c_max = c.max()
        
        if c_max < best_M * (1 - 1e-12):
            best_M = c_max
            best_a = a.copy()
        
        # Numerically stable softmax for p
        p = np.exp(tau * (c - c_max))
        p = p / p.sum()
        
        # Compute gradient with respect to a
        grad_a = smooth_gradient_fft(a, p)
        
        # Convert to gradient with respect to x (softmax gradient)
        # grad_x[i] = a[i] * (grad_a[i] - sum_j a[j] * grad_a[j])
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        
        # Momentum update
        v = momentum * v - lr * grad_x
        x = x + v
        
        # Prevent x from drifting too much
        x = x - x.mean()  # recenter for stability
    
    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix A where A @ g = conv(a, g)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):
    """Solve LP to find best response g."""
    n = len(a)
    A = build_A_sparse(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a_init, deadline, max_iters=200):
    """LP mirror descent for local refinement."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = float(c.max())
        
        g = lp_best_response(a, M, time_limit=5)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_t = (1.0 - t) * a + t * g
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            c_t = fast_conv(a_t)
            M_t = c_t.max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                break
    
    return best_a


def resample_seq(src, n):
    """Resample a sequence to length n using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.001))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.001))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Power law shapes
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(t**alpha, 0.001))
        inits.append(np.maximum((1-t)**alpha, 0.001))
    
    # Cosine modulation
    for f in [1, 2, 3, 4]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2*np.pi*f*t), 0.001))
    
    # Gaussian
    inits.append(np.exp(-4*(t-0.5)**2) + 0.05)
    inits.append(np.exp(-8*(t-0.3)**2) + np.exp(-8*(t-0.7)**2) + 0.03)
    
    # Random (smoothed)
    k = max(3, n // 32)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same'), 0.001))
    
    # Random (less smooth)
    k2 = max(2, n // 64)
    kern2 = np.ones(k2) / k2
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern2, 'same'), 0.001))
    
    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.2, 1.0, half_n), 0.001)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    
    # Anti-symmetric (shifted) - large on one side
    inits.append(np.maximum(0.2 + 0.8 * np.exp(-3*t), 0.001))
    inits.append(np.maximum(0.2 + 0.8 * np.exp(-3*(1-t)), 0.001))
    
    return inits
