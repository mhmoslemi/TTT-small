def solve_minimax_lp(a, time_budget=5.0):
    """Solve min t s.t. (a*g)_k <= t, sum(g)=1, g>=0."""
    n = len(a)
    m = 2 * n - 1
    # Variables: g_0, ..., g_{n-1}, t
    # Minimize: t (index n)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Constraints:
    # (a*g)_k - t <= 0 for all k  ->  row k of A, minus t
    # sum(g) = 1 -> equality constraint
    # g >= 0 -> bounds
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = build_toeplitz(a)
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = res.x[:n]
        t = res.x[n]
        return g, t
    return None, None

"""
Minimax LP-based optimization with dual-guided perturbations for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses a novel minimax LP formulation that directly
minimizes the peak, dual-guided perturbations, and adaptive multi-length exploration.

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_minimax_lp(a):
    """Solve min t s.t. (a*g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g, t, dual_info) or (None, None, None)."""
    n = len(a)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = build_toeplitz(a)
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            t = float(res.x[n])
            duals = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
            return g, t, duals
    except Exception:
        pass
    return None, None, None


def solve_maxsum_lp(a, M):
    """Solve max sum(g) s.t. (a*g)_k <= M, g>=0. (Previous formulation for comparison)"""
    n = len(a)

def line_search(a, g, best_obj):
    """Find best t in [0,1] for a_new = (1-t)*a + t*g, both normalized."""
    def obj_at(t):
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            return np.inf
        a_new = a_new / s
        return compute_obj(a_new)
    
    # Use scipy's minimize_scalar

        obj_best = res.fun

        obj_best = obj_at(0.5)

def dual_guided_perturb(a, duals, num_peaks=10, rng=None):
    """Perturb the sequence guided by dual variable information."""
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()
    
    a_new = a.copy()
    if duals is not None and len(duals) > 0:
        # High dual values indicate tight constraints (peaks we can't reduce)
        # Focus perturbations near the lags with high duals
        sorted_duals = np.argsort(duals)[::-1]
        top_lags = sorted_duals[:num_peaks]
        
        for k in top_lags:
            k = int(k)
            # Find indices that contribute to lag k
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # Perturb these indices
            idxs = np.arange(i_start, i_end + 1)
            factors = rng.uniform(0.85, 1.15, len(idxs))
            a_new[idxs] *= factors
    
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s


def spectral_shape(a, rng, mode='flatten'):
    """Apply spectral shaping to the sequence."""
    n = len(a)
    fa = rfft(a)
    mag = np.abs(fa)
    phase = np.angle(fa)
    
    if mode == 'flatten':
        # Reduce high-magnitude frequencies
        threshold = np.percentile(mag[1:], 80)
        mag[1:] = np.where(mag[1:] > threshold, 
                          threshold + (mag[1:] - threshold) * 0.3, mag[1:])
    elif mode == 'smooth':
        # Low-pass filter effect
        freqs = np.linspace(0, 1, len(fa))
        alpha = rng.uniform(0.5, 2.0)
        mag = mag * np.exp(-alpha * freqs)
    elif mode == 'random':
        mag = mag * np.exp(rng.normal(0, 0.1, len(fa)))
    elif mode == 'notch':
        # Remove a random frequency component
        idx = rng.integers(1, len(fa))
        mag[idx] *= 0.5
        if idx + 1 < len(fa):
            mag[idx+1] *= 0.7
    
    fa_new = mag * np.exp(1j * phase)
    a_new = irfft(fa_new, n)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s


def peak_target_perturb(a, num_tries=30, rng=None):
    """Target the top peaks in the convolution and try to reduce them."""
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for _ in range(num_tries):
        conv = compute_conv(best_a)
        sorted_idx = np.argsort(conv)[::-1]
        top_peaks = sorted_idx[:5]
        
        k = int(top_peaks[rng.integers(0, len(top_peaks))])
        
        # Find which a indices contribute to this peak
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        
        if i_end < i_start:
            continue
        
        # Try reducing some of the contributing indices
        a_new = best_a.copy()
        n_reduce = max(1, (i_end - i_start + 1) // 3)
        # Randomly select which to reduce
        candidates = np.arange(i_start, i_end + 1)
        selected = rng.choice(candidates, size=min(n_reduce, len(candidates)), replace=False)
        
        for idx in selected:
            a_new[idx] *= rng.uniform(0.6, 0.9)
        
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

Minimax LP-based optimization with dual-guided perturbations and multi-length
exploration for minimizing 2*n*max(conv(a,a))/(sum(a))^2. Uses a novel minimax
LP formulation that directly minimizes the cross-correlation peak, combined with

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_minimax_lp(a):
    """Solve min t s.t. (a*g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g, t, duals) or (None, None, None)."""
    n = len(a)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = build_toeplitz(a)
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15.0})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            t = float(res.x[n])
            duals = None
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                if hasattr(res.ineqlin, 'marginals') and res.ineqlin.marginals is not None:
                    duals = res.ineqlin.marginals
            return g, t, duals
    except Exception:
        pass
    return None, None, None

def dual_guided_perturb(a, duals, num_peaks=10, rng=None):
    """Perturb the sequence guided by dual variable information."""
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()

    if duals is not None and len(duals) > 0:

        top_lags = sorted_duals[:num_peaks]

        for k in top_lags:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_end < i_start:
                continue
            idxs = np.arange(i_start, i_end + 1)
            factors = rng.uniform(0.85, 1.15, len(idxs))
            a_new[idxs] *= factors

    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s


def spectral_shape(a, rng, mode='flatten'):
    """Apply spectral shaping to the sequence."""
    n = len(a)
    fa = rfft(a)
    mag = np.abs(fa)
    phase = np.angle(fa)

    if mode == 'flatten':
        if len(mag) > 1:
            threshold = np.percentile(mag[1:], 80)
            mag[1:] = np.minimum(mag[1:], threshold + (mag[1:] - threshold) * 0.3)
    elif mode == 'smooth':
        freqs = np.linspace(0, 1, len(fa))
        alpha = rng.uniform(0.5, 2.0)
        mag = mag * np.exp(-alpha * freqs)
    elif mode == 'random':
        mag = mag * np.exp(rng.normal(0, 0.1, len(fa)))
    elif mode == 'notch':
        idx = rng.integers(1, len(fa))
        mag[idx] *= 0.5
        if idx + 1 < len(fa):
            mag[idx + 1] *= 0.7
    elif mode == 'boost_low':
        freqs = np.linspace(0, 1, len(fa))
        alpha = rng.uniform(0.3, 1.0)
        mag = mag * (1.0 + alpha * (1.0 - freqs))

    fa_new = mag * np.exp(1j * phase)
    a_new = irfft(fa_new, n)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s


def peak_target_perturb(a, num_tries=30, rng=None):
    """Target the top peaks in the convolution and try to reduce them."""
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()

    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(num_tries):
        conv = compute_conv(best_a)
        sorted_idx = np.argsort(conv)[::-1]
        top_peaks = sorted_idx[:5]

        k = int(top_peaks[rng.integers(0, len(top_peaks))])

        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)

        if i_end < i_start:
            continue

        a_new = best_a.copy()
        n_reduce = max(1, (i_end - i_start + 1) // 3)
        candidates = np.arange(i_start, i_end + 1)
        n_sel = min(n_reduce, len(candidates))
        if n_sel == 0:
            continue
        selected = rng.choice(candidates, size=n_sel, replace=False)

        for idx in selected:
            a_new[idx] *= rng.uniform(0.6, 0.9)

        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 7)

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 5)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.3)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        nblocks = rng.integers(3, 8)
        block_size = max(1, n // nblocks)
        a = np.zeros(n)
        for b in range(nblocks):
            a[b * block_size:(b + 1) * block_size] = rng.uniform(0.3, 1.5)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 5:
        # Asymmetric
        a = np.ones(n)
        a[:n // 3] *= rng.uniform(0.6, 1.2)
        a[n // 3:2 * n // 3] *= rng.uniform(0.8, 1.3)
        a[2 * n // 3:] *= rng.uniform(0.6, 1.2)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def lp_optimize(a, time_budget=100.0, verbose=False, seed=42):
    """Core optimization using minimax and maxsum LP directions with perturbations."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 15
    rng = np.random.default_rng(seed)

    iter_count = 0
    use_minimax = True

    while time.monotonic() - start < time_budget:
        iter_count += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        conv = compute_conv(best_a)
        M = float(np.max(conv))

        g = None
        t_opt = None
        duals = None

        if use_minimax:
            g, t_opt, duals = solve_minimax_lp(best_a)
            if g is not None:
                g = g / max(np.sum(g), 1e-10)
                # Check if direction helps
                if t_opt is not None and t_opt < M * (1 - 1e-6):
                    t_best, obj_best = line_search(best_a, g, best_obj)
                    if obj_best < best_obj - 1e-10:
                        a_new = (1.0 - t_best) * best_a + t_best * g
                        a_new = np.maximum(a_new, 0)
                        s_new = np.sum(a_new)
                        if s_new > 1e-10:
                            a_new = a_new / s_new
                            best_obj = obj_best
                            best_a = a_new.copy()
                            stall = 0
                            if verbose and iter_count % 30 == 0:
                                print(f"    LP it={iter_count}, obj={best_obj:.6f}, ratio={t_opt/M:.4f}")
                            continue
                # If minimax didn't help, try maxsum
                use_minimax = False
            else:
                use_minimax = False
        else:
            g, S, duals = solve_maxsum_lp(best_a, M)
            if g is not None and S > 1.0 + 1e-8:
                g = g / S
                t_best, obj_best = line_search(best_a, g, best_obj)
                if obj_best < best_obj - 1e-10:
                    a_new = (1.0 - t_best) * best_a + t_best * g
                    a_new = np.maximum(a_new, 0)
                    s_new = np.sum(a_new)
                    if s_new > 1e-10:
                        a_new = a_new / s_new
                        best_obj = obj_best
                        best_a = a_new.copy()
                        stall = 0
                        if verbose and iter_count % 30 == 0:
                            print(f"    MaxSum it={iter_count}, obj={best_obj:.6f}, S={S:.4f}")
                        continue
            # Alternate between the two LP types periodically
            if iter_count % 10 == 0:
                use_minimax = True

        stall += 1

        # Perturb when stalled
        if stall > patience:
            strategy = rng.integers(0, 5)

            if strategy == 0:
                a_pert = dual_guided_perturb(best_a, duals, num_peaks=10, rng=rng)
            elif strategy == 1:
                modes = ['flatten', 'smooth', 'notch', 'boost_low', 'random']
                a_pert = spectral_shape(best_a, rng, mode=modes[rng.integers(0, len(modes))])
            elif strategy == 2:
                a_pert = peak_target_perturb(best_a, num_tries=10, rng=rng)
            elif strategy == 3:
                scale = rng.uniform(0.03, 0.1)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s_pert = np.sum(a_pert)
                if s_pert > 1e-10:
                    a_pert = a_pert / s_pert
                else:
                    a_pert = best_a.copy()
            else:
                # Structured modulation
                k_arr = np.arange(n)
                f = rng.uniform(0.3, 8.0)
                mod = 1.0 + rng.uniform(0.05, 0.2) * np.cos(2 * np.pi * f * k_arr / n + rng.uniform(0, 6.28))
                a_pert = best_a * mod
                a_pert = np.maximum(a_pert, 0)
                s_pert = np.sum(a_pert)
                if s_pert > 1e-10:
                    a_pert = a_pert / s_pert
                else:
                    a_pert = best_a.copy()

            obj_pert = compute_obj(a_pert)
            improved = False
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
                stall = 0
                improved = True
            else:
                # Try smaller perturbation
                small_scale = rng.uniform(0.01, 0.04)
                a_small = best_a * np.exp(rng.normal(0, small_scale, n))
                a_small = np.maximum(a_small, 0)
                s_small = np.sum(a_small)
                if s_small > 1e-10:
                    a_small = a_small / s_small
                    obj_small = compute_obj(a_small)
                    if obj_small < best_obj:
                        best_obj = obj_small
                        best_a = a_small.copy()
                        stall = 0
                        improved = True

            if not improved:
                stall = 0

            patience = max(8, patience - 2)
            if patience <= 8:
                patience = 8

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Minimax LP / MaxSum LP alternating optimization with
    dual-guided perturbations, spectral shaping, peak targeting, and
    adaptive multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.6f}")

    # Phase 1: Intensive LP optimization on parent
    if best_seq is not None:
        n_parent = len(best_seq)
        print(f"\n=== Phase 1: LP Optimize on parent (n={n_parent}) ===")
        polish_budget = min(350.0, (deadline - time.monotonic()) * 0.3)
        a_pol, obj_pol = lp_optimize(best_seq, time_budget=polish_budget, verbose=True, seed=seed)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  After Phase 1: {best_score:.6f}")
        else:
            print(f"  After Phase 1: {obj_pol:.6f} (no improvement)")

    # Phase 2: Explore different n values
    print(f"\n=== Phase 2: Multi-n Exploration ===")

    if best_seq is not None:
        np_ = len(best_seq)
        n_candidates = [max(100, np_ - 200), max(100, np_ - 100), max(100, np_ - 50),
                        np_ + 50, np_ + 100, np_ + 200, np_ + 300, np_ + 400]
    else:
        n_candidates = [300, 500, 700, 900, 1100, 1300]

    n_values = sorted(set(n for n in n_candidates if 100 <= n <= 1500))
    if best_seq is not None:
        n_values = [n for n in n_values if n != len(best_seq)]
    print(f"  n values to try: {n_values}")

    for n in n_values:
        if time.monotonic() > deadline - 250:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        alloc = min(remaining * 0.12, 120.0)
        alloc = max(30.0, alloc)

        print(f"\n  N={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        # Start from resampled best if available
        if best_seq is not None:
            a_start = resample_sequence(best_seq, n)
        else:
            a_start = random_init(n, seed + n * 7919)

        # LP optimization
        a_opt, obj_opt = lp_optimize(a_start, time_budget=alloc, seed=seed + n)
        print(f"  N={n}: obj={obj_opt:.6f}")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Phase 3: Final intensive polish on global best
    print(f"\n=== Phase 3: Final Polish ===")
    print(f"  Best before: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  Polishing n={n_best}, {remaining:.0f}s remaining")

        # Round 1: LP with different seed
        alloc1 = min(remaining * 0.25, 150.0)
        a_cur, obj1 = lp_optimize(best_seq, time_budget=alloc1, seed=seed + 100, verbose=True)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a_cur.copy()
            print(f"  Round 1 (LP): {best_score:.6f}")

        # Round 2: Peak targeting + LP
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            rng2 = np.random.default_rng(seed + 200)
            a_cur = peak_target_perturb(best_seq, num_tries=40, rng=rng2)
            obj2 = compute_obj(a_cur)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_cur.copy()
                print(f"  Round 2 (peak): {best_score:.6f}")

            a_cur, obj2b = lp_optimize(best_seq, time_budget=min(remaining * 0.15, 60.0), seed=seed + 201)
            if obj2b < best_score:
                best_score = obj2b
                best_seq = a_cur.copy()
                print(f"  Round 2b (LP): {best_score:.6f}")

        # Round 3: Spectral + LP
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            rng3 = np.random.default_rng(seed + 300)
            for mode in ['flatten', 'smooth', 'notch', 'boost_low']:
                if time.monotonic() > deadline - 25:
                    break
                a_spec = spectral_shape(best_seq, rng3, mode=mode)
                obj_spec = compute_obj(a_spec)
                if obj_spec < best_score:
                    best_score = obj_spec
                    best_seq = a_spec.copy()
                    print(f"  Round 3 (spec {mode}): {best_score:.6f}")

            a_cur, obj3 = lp_optimize(best_seq, time_budget=min(remaining * 0.12, 40.0), seed=seed + 301)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a_cur.copy()
                print(f"  Round 3b (LP): {best_score:.6f}")

        # Round 4: Multiple perturbation strategies + LP
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            rng4 = np.random.default_rng(seed + 400)
            n_b = len(best_seq)
            k_arr = np.arange(n_b)

            for ptry in range(15):
                if time.monotonic() > deadline - 20:
                    break

                strategy = ptry % 5

                if strategy == 0:
                    # Multi-frequency modulation
                    f1 = rng4.uniform(0.3, 4.0)
                    f2 = rng4.uniform(4.0, 10.0)
                    mod = 1.0 + 0.08 * np.cos(2 * np.pi * f1 * k_arr / n_b) + \
                          0.05 * np.cos(2 * np.pi * f2 * k_arr / n_b)
                    a_asym = best_seq * mod
                elif strategy == 1:
                    # Random per-element scale
                    a_asym = best_seq * rng4.uniform(0.92, 1.08, n_b)
                elif strategy == 2:
                    # Block adjustment
                    nb = rng4.integers(4, 12)
                    a_asym = best_seq.copy()
                    for b in range(nb):
                        s_start = b * n_b // nb
                        s_end = (b + 1) * n_b // nb
                        a_asym[s_start:s_end] *= rng4.uniform(0.88, 1.12)
                elif strategy == 3:
                    # Gaussian with larger scale
                    a_asym = best_seq * np.exp(rng4.normal(0, 0.08, n_b))
                else:
                    # Complementary: try the "mirror"
                    a_asym = best_seq[::-1].copy()
                    a_asym = a_asym * rng4.uniform(0.95, 1.05, n_b)

                a_asym = np.maximum(a_asym, 0)
                s_as = np.sum(a_asym)
                if s_as < 1e-10:
                    continue
                a_asym = a_asym / s_as

                obj_as = compute_obj(a_asym)
                if obj_as < best_score:
                    best_score = obj_as
                    best_seq = a_asym.copy()
                    print(f"  Round 4 (pert {ptry}): {best_score:.6f}")
                    # Quick LP polish
                    rem = deadline - time.monotonic()
                    if rem > 10:
                        a_pol, obj_pol = lp_optimize(a_asym, time_budget=min(15.0, rem * 0.1), seed=seed + 500 + ptry)
                        if obj_pol < best_score:
                            best_score = obj_pol
                            best_seq = a_pol.copy()
                            print(f"  Round 4 (pert {ptry}+LP): {best_score:.6f}")

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
