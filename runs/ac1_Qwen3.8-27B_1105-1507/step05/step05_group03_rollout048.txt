from scipy.optimize import linprog

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_constraint_indices(n, active_k=None):
    """Build row/col indices for LP constraint matrix, optionally restricted to active_k."""
    if active_k is None:
        ks = list(range(2 * n - 1))
    else:
        ks = active_k
    rows = []
    cols = []
    for k in ks:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), ks


def solve_lp_restricted(a, M, active_k, n, time_limit=25):
    """Solve LP: max 1^T g s.t. (a*g)[k] <= M for k in active_k, g >= 0."""
    rows = []
    cols = []
    for k in active_k:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    if len(rows) == 0:
        return np.ones(n) / n
    row_arr = np.array(rows, dtype=np.int32)
    col_arr = np.array(cols, dtype=np.int32)
    vals = np.array([a[k - j] for k, j in zip(row_arr, col_arr)], dtype=np.float64)
    A = csr_matrix((vals, (row_arr, col_arr)), shape=(len(active_k), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

    """Solve LP with all constraints."""
    active_k = list(range(2 * n - 1))

def line_search(a, g, max_iter=80):
    """Find optimal t in [0,1] minimizing max_k conv((1-t)*a + t*g, (1-t)*a + t*g)."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(max_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(a_coef + b_coef * m1 + c_coef * m1 * m1))
        f2 = float(np.max(a_coef + b_coef * m2 + c_coef * m2 * m2))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val

def smooth_gradient_optimize(a, deadline, label=""):
    """
    Nesterov-accelerated gradient descent on smooth max approximation.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)

    total_iters = 5000
    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        frac = it / total_iters
        tau = 5.0 + 200.0 * (frac ** 0.7)
        lr = 0.5 * (0.02 / 0.5) ** (frac ** 0.5)

        # Compute smooth gradient
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # d(b_k)/da_j = a_{k-j} for valid j + a_{j-k} for valid j
            # Simplify: for each j in [lo, hi], contribution is 2*pk*a_{k-j} (approx)
            j_range = np.arange(lo, hi + 1)
            a_vals = a[np.clip(k - j_range, 0, n - 1)]
            grad[j_range] += 2.0 * pk * a_vals

        # Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * frac * frac)) / 2.0
        v = beta * v + (1.0 - beta) * lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = compute_M(a)
        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()
            if label and it % 500 == 0:
                print(f"        {label} iter {it}: M={M:.10f}", flush=True)

    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update(parent, "(parent)")
            print(f"Parent: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        update(parent, "(init)")

    # === Phase 1: LP Mirror Descent on parent (35% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent n={parent.size}", flush=True)
    print(f"{'='*60}", flush=True)

    p1_deadline = t_start + budget_s * 0.35
    a1 = run_lp_mirror_descent(parent.copy(), p1_deadline, initial_tl=20,
                               verbose=True, label="P1")
    update(a1, "(P1 LP-MD)")

    # === Phase 2: Smooth gradient polish (10% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Smooth Gradient Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best is not None:
        p2_deadline = t_start + budget_s * 0.45
        a2 = smooth_gradient_optimize(best.copy(), p2_deadline, label="P2")
        update(a2, "(P2 smooth)")
        # LP polish after smooth
        a2b = run_lp_mirror_descent(a2.copy(), min(t_start + budget_s * 0.50, deadline),
                                    initial_tl=20, label="P2b")
        update(a2b, "(P2 LP polish)")

    # === Phase 3: Multi-length exploration (25% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if best is not None:
        base_n = best.size
        p3_deadline = t_start + budget_s * 0.75
        trial_ns = sorted(set([
            int(base_n * 0.5), int(base_n * 0.75),
            int(base_n * 1.25), int(base_n * 1.5), int(base_n * 2.0),
            256, 512, 1024, 2048
        ]))
        trial_ns = [n for n in trial_ns if 128 <= n <= 4096 and n != base_n]

        for n_trial in trial_ns:
            if time.monotonic() > p3_deadline:
                break
            remaining = p3_deadline - time.monotonic()
            sub_deadline = min(time.monotonic() + max(30, remaining / max(1, len([x for x in trial_ns if x >= n_trial]))), p3_deadline)

            a_init = resample_seq(best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()

            a_opt = run_lp_mirror_descent(a_init, sub_deadline, initial_tl=15,
                                          verbose=False, label=f"P3 n={n_trial}")
            update(a_opt, f"(P3 n={n_trial})")

    # === Phase 4: Symmetric optimization (10% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Symmetric Optimization", flush=True)
    print(f"{'='*60}", flush=True)

    if best is not None:
        p4_deadline = t_start + budget_s * 0.85
        n_sym = best.size
        a_sym = best.copy()
        a_sym = 0.5 * (a_sym + a_sym[::-1].copy())
        a_sym = a_sym / a_sym.sum()
        a_sym_opt = run_lp_symmetric(a_sym, p4_deadline, label="P4")
        update(a_sym_opt, "(P4 symmetric)")

    # === Phase 5: Perturbation restarts (10% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if best is not None:
        n_best = best.size
        p5_deadline = min(t_start + budget_s * 0.95, deadline)
        temp = 0.05

        for r in range(30):
            if time.monotonic() > p5_deadline:
                break
            remaining = p5_deadline - time.monotonic()
            if remaining < 20:
                break
            sub_dl = min(time.monotonic() + max(10, remaining / max(1, 30 - r)), p5_deadline)

            a = best.copy().astype(np.float64)
            noise = 0.01 * temp * a
            a = a + rng.normal(0, 1.0, n_best) * noise
            a = np.maximum(a, 1e-8)
            a = a / a.sum()

            # Quick smooth + LP
            a = smooth_gradient_optimize(a, min(sub_dl, time.monotonic() + remaining * 0.3))
            a_opt = run_lp_mirror_descent(a, sub_dl, initial_tl=10, label=f"P5 r={r}")
            update(a_opt, f"(P5 r={r} T={temp:.4f})")
            temp *= 0.85

    # === Phase 6: Final extended polish (last 5%) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best is not None:
        a_final = run_lp_mirror_descent(best.copy(), deadline, initial_tl=25,
                                        verbose=True, label="P6")
        update(a_final, "(P6 final)")

    # Fallback
    if best is None:
        best = parent.copy()
        best_score = compute_J(best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best]
