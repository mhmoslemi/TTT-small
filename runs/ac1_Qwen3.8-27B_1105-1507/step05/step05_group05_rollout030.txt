def compute_smooth_max_and_grad(a, tau):
    """
    Compute the smooth max of conv(a,a) and its gradient w.r.t. a.
    a: array of length n, non-negative
    tau: smoothing parameter
    Returns: (value, gradient) where gradient has same shape as a.
    """
    n = len(a)
    b = fast_conv(a)  # b has length 2n-1
    
    # Smooth max: (1/tau) * log(sum exp(tau * b_k))
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    log_sum = np.log(exp_b.sum())
    val = b_max + log_sum / tau
    
    # Probabilities
    p = exp_b / exp_b.sum()  # p has length 2n-1
    
    # Gradient: grad[i] = 2 * sum_j p[i+j] * a[j]
    # This is 2 * cross_correlation(p, a)[:n]
    # cross[i] = sum_{j=0}^{n-1} p[i+j] * a[j]
    
    # Via FFT: cross[i] = IFFT(FFT(a) * conj(FFT(p)))[i]
    fft_size = 1
    while fft_size < n + (2*n - 1) - 1:  # 3n - 2
        fft_size <<= 1
    
    fa = np.fft.fft(a, fft_size)
    fp = np.fft.fft(p, fft_size)
    cross = np.fft.ifft(fa * np.conj(fp)).real[:n]
    
    grad = 2.0 * cross
    return val, grad
