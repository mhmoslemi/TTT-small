def full_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = full_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def coord_update_fast(a, b, i, x_cur):
    """
    Fast coordinate update using vectorized evaluation.
    Returns optimal y for a[i] given all other coordinates fixed.
    Uses golden-section with vectorized objective evaluation.
    """
    n = len(a)
    S = a.sum()
    
    # Find affected range in b
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Unaffected max
    if k_lo > 0 and k_hi < 2 * n - 2:
        base_max = max(b[:k_lo].max(), b[k_hi + 1:].max())
    elif k_lo > 0:
        base_max = b[:k_lo].max()
    elif k_hi < 2 * n - 2:
        base_max = b[k_hi + 1:].max()
    else:
        base_max = 0.0
    
    # Precompute coefficients for affected k
    affected_ks = np.arange(k_lo, k_hi + 1)
    # For each affected k, the new b[k] = b[k] + 2*(y-x)*a[k-i] (k != 2i)
    # or b[k] - x^2 + y^2 (k == 2i)
    
    # a[k-i] for valid range
    ki = affected_ks - i
    a_ki = np.zeros(len(affected_ks))
    valid = (ki >= 0) & (ki < n)
    a_ki[valid] = a[ki[valid]]
    
    # Quadratic coefficient (1 where k=2i, 0 elsewhere)
    quad_mask = (affected_ks == 2 * i)
    
    # Linear coefficient: 2*a[k-i] (for non-quad terms)
    lin_coef = 2.0 * a_ki
    lin_coef[quad_mask] = 0.0
    
    # Constant: b[k] - 2*x*a[k-i] (non-quad) or b[k] - x^2 (quad)
    const = b[affected_ks].copy()
    const -= 2.0 * x_cur * a_ki
    const[quad_mask] = b[affected_ks[quad_mask]] - x_cur * x_cur if quad_mask.any() else 0
    
    # We minimize: max(base_max, max(const + lin_coef*y + quad_mask*y^2)) / (S - x + y)^2
    # Let's define f(y) for golden section
    
    def f_of_y(y_val):
        new_sum = S - x_cur + y_val
        if new_sum <= 1e-15:
            return 1e15
        # Affected values
        vals = const + lin_coef * y_val
        # Add quadratic term
        if quad_mask.any():
            vals[quad_mask] += y_val * y_val - x_cur * x_cur  # Wait, we already subtracted x^2
            # Actually const[quad_mask] = b[2i] - x^2, and we need + y^2
            # So total = b[2i] - x^2 + y^2 = const[quad_mask] + y^2
            vals[quad_mask] = const[quad_mask] + y_val * y_val
        m = max(base_max, vals.max())
        return m / (new_sum * new_sum)
    
    # Golden section search
    lo = 0.0
    hi = max(x_cur * 3.0, np.sqrt(max(b.max(), 0.0)) * 1.5 + 0.1, 0.05)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = lo + (1.0 - gr) * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = f_of_y(a_pt)
    fb = f_of_y(b_pt)
    
    for _ in range(25):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = lo + (1.0 - gr) * (hi - lo)
            fa = f_of_y(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = f_of_y(b_pt)
    
    x_opt = (lo + hi) / 2.0
    # Check boundary at 0
    f0 = f_of_y(0.0)
    if f0 < f_of_y(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)

        # For i in [lo, hi]: a[k-i] is valid
        # a[k-hi], a[k-hi+1], ..., a[k-lo]
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1]
    
    return grad


def run_adam(a, max_iters, deadline, tau_start=20.0, tau_end=200.0, lr=0.02):
    """Adam optimizer on smooth max of convolution (normalized to sum=1)."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = full_conv(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        lr_t = lr / (1 + 0.05 * it)
        
        grad = smooth_max_grad(a, tau)
        
        # Project to sum-zero (tangent to sum=1 constraint)
        grad_mean = grad.mean()
        grad = grad - grad_mean
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        M = full_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
    
    return best_a


def solve_lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    # Build constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0
    # A is (2n-1) x n, banded
    
    # Use sparse representation
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a, max_iters, deadline):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = full_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = full_conv(a)
        M = float(b.max())
        g = solve_lp_best_response(a, M)
        if g is None:
            # Try with slightly relaxed M
            g = solve_lp_best_response(a, M * 1.05)
            if g is None:
                break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        improved = False
        base_t = max(0.1, 0.5 * (1.0 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = full_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
    
    return best_a


def apply_tilt(a, gamma):
    """Apply a linear tilt to break symmetry. gamma controls strength."""
    n = len(a)
    t = np.linspace(-1, 1, n)
    tilted = a * (1.0 + gamma * t)
    tilted = np.maximum(tilted, 0.0)
    s = tilted.sum()
    if s < 1e-10:
        return a.copy()
    return tilted / s


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        # Tilted parent
        inits.append(apply_tilt(p, 0.1))
        inits.append(apply_tilt(p, -0.1))
    
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.5, 1.0, n) / n)
    
    # Tapered (Hann window)
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)) + 0.1)
    # Asymmetric taper
    inits.append(0.3 + 0.7 * t)
    inits.append(0.3 + 0.7 * (1 - t))
    # Power taper
    inits.append(t ** 2 + 0.1)
    inits.append((1 - t) ** 2 + 0.1)
    
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Sine-modulated
    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)), 0.05))
    
    # Exponential
    inits.append(np.exp(-2 * t) + 0.2)
    inits.append(np.exp(-2 * (1 - t)) + 0.2)
    
    # Triangular peak off-center
    center = rng.uniform(0.3, 0.7)
    inits.append(1.0 - np.abs(t - center) / max(center, 1 - center) + 0.1)
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimal J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phases:
    1. Fast CD across multiple initializations and lengths
    2. Asymmetry injection + CD refinement
    3. Adam smooth gradient descent
    4. LP mirror descent
    5. Final combined polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = make_tracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            evaluate_and_track(parent, tracker)
            print(f"Parent loaded: n={parent.size}, J={tracker['best_score']:.6f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512)
        evaluate_and_track(parent, tracker)
    
    # === PHASE 1: Fast CD with multiple inits ===
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Fast Coordinate Descent (budget ~500s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    lengths = [384, 512, 640, 768]
    # Allocate time proportionally, favoring lengths near parent
    parent_n = tracker['best'].size if tracker['best'] is not None else 512
    
    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        
        # Time allocation: more time for lengths closer to best
        dist = abs(n - parent_n)
        t_alloc = max(60, 200 - dist // 4)
        phase_deadline = min(time.monotonic() + t_alloc, deadline - 350)
        
        if time.monotonic() >= phase_deadline:
            continue
        
        t0 = time.monotonic()
        inits = diverse_inits(n, rng, tracker['best'])
        best_for_n = None
        best_M_for_n = float('inf')
        
        max_cd_iters = max(30, int(5000 / max(1, n // 64)))
        
        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_deadline:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            
            a_opt = run_cd_fast(a0, max_cd_iters, phase_deadline, seed=seed + idx * 13)
            
            M_val = full_conv(a_opt).max()
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a_opt.copy()
        
        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: M={best_M_for_n:.10f} J={J_val:.6f} ({elapsed:.1f}s)", flush=True)
            evaluate_and_track(best_for_n, tracker)
    
    print(f"Phase 1 total: {time.monotonic()-p1_start:.1f}s", flush=True)
    
    # === PHASE 2: Asymmetry injection + refinement ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Asymmetry Injection + Refinement (budget ~200s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker['best'] is not None and time.monotonic() < deadline - 250:
        best = tracker['best']
        n_best = best.size
        
        for gamma in [0.05, 0.1, 0.2, -0.05, -0.1, -0.2]:
            if time.monotonic() > deadline - 200:
                break
            a_tilt = apply_tilt(best, gamma)
            a_opt = run_cd_fast(a_tilt, 100, min(time.monotonic() + 40, deadline - 180),
                                seed=seed + 1000 + int(abs(gamma) * 100))
            J_val = compute_J(a_opt)
            print(f"  tilt={gamma:+.2f}: J={J_val:.6f}", flush=True)
            evaluate_and_track(a_opt, tracker)
    
    # Also try "shear" transformations
    if tracker['best'] is not None and time.monotonic() < deadline - 200:
        best = tracker['best']
        n_best = best.size
        # Quadratic tilt
        t = np.linspace(-1, 1, n_best)
        for c2 in [0.1, -0.1, 0.2, -0.2]:
            if time.monotonic() > deadline - 180:
                break
            a_q = best * (1.0 + c2 * t**2)
            a_q = np.maximum(a_q, 0.0)
            a_q = a_q / a_q.sum()
            a_opt = run_cd_fast(a_q, 80, min(time.monotonic() + 30, deadline - 160),
                                seed=seed + 2000 + int(abs(c2) * 100))
            J_val = compute_J(a_opt)
            print(f"  quad_tilt={c2:+.2f}: J={J_val:.6f}", flush=True)
            evaluate_and_track(a_opt, tracker)
    
    print(f"Phase 2 total: {time.monotonic()-p2_start:.1f}s", flush=True)
    
    # === PHASE 3: Adam smooth gradient ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Adam Smooth Gradient (budget ~150s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker['best'] is not None and time.monotonic() < deadline - 150:
        # Multiple runs with different hyperparameters
        configs = [
            {'tau_start': 10.0, 'tau_end': 100.0, 'lr': 0.02, 'max_iters': 1500},
            {'tau_start': 20.0, 'tau_end': 300.0, 'lr': 0.01, 'max_iters': 1000},
            {'tau_start': 5.0, 'tau_end': 50.0, 'lr': 0.03, 'max_iters': 2000},
        ]
        
        for cfg in configs:
            if time.monotonic() > deadline - 120:
                break
            a_adam = run_adam(tracker['best'], cfg['max_iters'],
                             min(time.monotonic() + 50, deadline - 100),
                             tau_start=cfg['tau_start'], tau_end=cfg['tau_end'],
                             lr=cfg['lr'])
            J_val = compute_J(a_adam)
            print(f"  Adam (tau={cfg['tau_start']}-{cfg['tau_end']}, lr={cfg['lr']}): J={J_val:.6f}", flush=True)
            evaluate_and_track(a_adam, tracker)
    
    print(f"Phase 3 total: {time.monotonic()-p3_start:.1f}s", flush=True)
    
    # === PHASE 4: LP Mirror Descent ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: LP Mirror Descent (budget ~150s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker['best'] is not None and time.monotonic() < deadline - 100:
        n_best = tracker['best'].size
        # Estimate LP time
        est_lp_time = 0.01 * (n_best / 128.0) ** 2
        max_iters = max(10, min(200, int(130 / max(0.01, est_lp_time))))
        print(f"  n={n_best}, est LP time={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)
        
        a_lp = run_lp_mirror(tracker['best'], max_iters, min(time.monotonic() + 130, deadline - 80))
        J_val = compute_J(a_lp)
        print(f"  LP Mirror: J={J_val:.6f}", flush=True)
        evaluate_and_track(a_lp, tracker)
    
    print(f"Phase 4 total: {time.monotonic()-p4_start:.1f}s", flush=True)
    
    # === PHASE 5: Final combined polish ===
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker['best'] is not None and time.monotonic() < deadline - 20:
        # CD refinement with exact method
        a_final = run_cd_exact(tracker['best'], 30, min(time.monotonic() + 60, deadline - 40),
                               seed=77777)
        evaluate_and_track(a_final, tracker)
        
        # Quick LP if time remains
        if time.monotonic() < deadline - 30:
            a_lp2 = run_lp_mirror(tracker['best'], 20, deadline - 10)
            evaluate_and_track(a_lp2, tracker)
        
        # Try one more tilt + CD
        if time.monotonic() < deadline - 20:
            for gamma in [0.03, -0.03, 0.07]:
                if time.monotonic() > deadline - 15:
                    break
                a_t = apply_tilt(tracker['best'], gamma)
                a_opt = run_cd_fast(a_t, 50, deadline - 10, seed=99999)
                evaluate_and_track(a_opt, tracker)
    
    # Try extended length if time remains
    if tracker['best'] is not None and time.monotonic() < deadline - 30:
        for mult in [2]:
            if time.monotonic() > deadline - 25:
                break
            new_n = tracker['best'].size * mult
            if new_n > 1536:
                continue
            a_ext = resample_seq(tracker['best'], new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_opt = run_cd_fast(a_ext, 50, deadline - 10, seed=55555)
            evaluate_and_track(a_opt, tracker)
    
    # Also try a fresh start with a structured sequence at the best length
    if tracker['best'] is not None and time.monotonic() < deadline - 20:
        n_best = tracker['best'].size
        # Try a "water-filling" style initialization
        t = np.linspace(0, 1, n_best)
        a_struct = 1.0 / (0.5 + np.abs(t - 0.4))  # Peak off-center
        a_struct = np.maximum(a_struct, 0.1)
        a_struct = a_struct / a_struct.sum()
        a_opt = run_cd_fast(a_struct, 80, deadline - 10, seed=11111)
        evaluate_and_track(a_opt, tracker)
    
    print(f"Phase 5 total: {time.monotonic()-p5_start:.1f}s", flush=True)
    
    # Final report
    if tracker['best'] is not None:
        final_score = compute_J(tracker['best'])
        print(f"\n{'='*60}", flush=True)
        print(f"FINAL: J={tracker['best_score']:.8f} (verified: {final_score:.8f}) "
              f"n={tracker['best'].size} time={time.monotonic()-tracker['t_start']:.0f}s", flush=True)
        print(f"{'='*60}", flush=True)
    else:
        # Fallback
        fallback = np.ones(512)
        tracker['best'] = fallback
        tracker['best_score'] = compute_J(fallback)
        print(f"FALLBACK: J={tracker['best_score']:.6f}", flush=True)
    
    return [float(x) for x in tracker['best']]
