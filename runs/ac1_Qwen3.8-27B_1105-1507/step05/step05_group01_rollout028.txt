```python
def lp_minimize_peak(a, M_hint=None):
    """Solve LP: min t s.t. (a*g)[k] <= t, sum(g)=1, g>=0. Returns g, t_opt."""
    n = len(a)
    m = 2*n - 1
    # Variables: g[0..n-1], t
    # Objective: minimize t -> c = [0, 0, ..., 0, 1] (last element is t)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: (a*g)[k] <= t for k=0..m-1
    # In standard form: (a*g)[k] - t <= 0
    A = np.zeros((m, n + 1))
    # Build Toeplitz for a*g
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    # Subtract t (last column)
    A[:, -1] = -1.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    bounds = [(0, None)] * n + [(None, None)]  # g[i] >= 0, t unbounded
    
    res = linprog(c, A_ub=A, b_ub=np.zeros(m), A_eq=A_eq, b_eq=[1.0], 
                  bounds=bounds, method='highs')
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        t = res.x[n]
        return g, t
    return None, None
```

```python
"""
Improved LP-based optimization using "minimize peak" LP formulation,
constraint-based reduction, and refined line search for minimizing
the autocorrelation peak ratio of non-negative sequences.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz_rows(a):
    """Build the matrix for (a*g)[k] as a function of g. Returns (m, n) matrix."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

def lp_minimize_peak(a):
    """Solve: min t s.t. (a*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None)."""
    n = len(a)
    m = 2 * n - 1
    T = build_toeplitz_rows(a)
    
    # x = [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([T, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            t = float(res.x[n])
            return g, t
    except Exception:
        pass
    return None, None

def lp_maximize_sum(a, M):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g>=0."""
    n = len(a)
    m = 2 * n - 1
    T = build_toeplitz_rows(a)
    
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0

def golden_section_line_search(a, g, n_evals=8):
    """Find best t in [0,1] minimizing obj((1-t)*a + t*g)."""
    a_norm = a / np.sum(a)  # ensure normalized
    g_norm = g / np.sum(g) if np.sum(g) > 1e-10 else g
    
    best_t = 0.0
    best_obj = compute_obj(a_norm)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    a_lo, a_hi = 0.0, 1.0
    c1 = a_hi - gr * (a_hi - a_lo)
    c2 = a_hi - c1
    
    f1 = compute_obj(np.maximum((1-c1)*a_norm + c1*g_norm, 0))
    f2 = compute_obj(np.maximum((1-c2)*a_norm + c2*g_norm, 0))
    
    for _ in range(n_evals):
        if f1 < f2:
            a_hi = c2
            c2 = c1
            f2 = f1
            c1 = a_hi - gr * (a_hi - a_lo)
            f1 = compute_obj(np.maximum((1-c1)*a_norm + c1*g_norm, 0))
        else:
            a_lo = c1
            c1 = c2
            f1 = f2
            c2 = a_hi - gr * (a_hi - a_lo)
            f2 = compute_obj(np.maximum((1-c2)*a_norm + c2*g_norm, 0))
    
    t_opt = (a_lo + a_hi) / 2
    result = np.maximum((1-t_opt)*a_norm + t_opt*g_norm, 0)
    s = np.sum(result)
    if s > 1e-10:
        result = result / s
    obj = compute_obj(result)
    
    if obj < best_obj:
        best_obj = obj
        best_t = t_opt
    
    return result, best_obj

def constraint_reduction(a, num_iter=50, damp=0.97):
    """Reduce values at positions contributing most to peak convolution."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    for _ in range(num_iter):
        conv = compute_conv(a)
        k = int(np.argmax(conv))
        
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        indices = np.arange(i_start, i_end + 1)
        partners = k - indices
        products = a[indices] * a[partners]
        
        max_pos = int(np.argmax(products))
        i_star = int(indices[max_pos])
        j_star = int(partners[max_pos])
        
        if a[i_star] >= a[j_star]:
            a[i_star] *= damp
        else:
            a[j_star] *= damp
        
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
    
    return a

def multi_peak_reduction(a, num_iter=30, num_peaks=3, damp=0.95):
    """Target multiple peaks simultaneously."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    for _ in range(num_iter):
        conv = compute_conv(a)
        # Find top peaks
        top_k = np.argsort(conv)[::-1][:num_peaks]
        
        for k in top_k:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            indices = np.arange(i_start, i_end + 1)
            partners = k - indices
            products = a[indices] * a[partners]
            
            max_pos = int(np.argmax(products))
            i_star = int(indices[max_pos])
            j_star = int(partners[max_pos])
            
            if a[i_star] >= a[j_star]:
                a[i_star] *= damp
            else:
                a[j_star] *= damp
        
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
    
    return a

def spectral_flatten(a, alpha=0.1, num_iter=20):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for _ in range(num_iter):
        fa = rfft(a)
        mag = np.abs(fa)
        phase = np.angle(fa)
        
        # Flatten: move towards uniform magnitude
        target = np.mean(mag)
        mag_new = (1 - alpha) * mag + alpha * target
        fa_new = mag_new * np.exp(1j * phase)
        a_new = irfft(fa_new, n)
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
    
    return best_a

def resample_sequence(a, new_n):

    return np.maximum(a_new, 0)
```

```python
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz_rows(a):
    """Build (2n-1, n) matrix T where (a*g)[k] = T[k,:] @ g."""
    n = len(a)
    m = 2 * n - 1
    T = np.zeros((m, n))
    for j in range(n):
        end = min(j + n, m)
        T[j:end, j] = a[:end - j]
    return T


def lp_minimize_peak(a):
    """Solve: min t s.t. (a*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None)."""
    n = len(a)
    m = 2 * n - 1
    T = build_toeplitz_rows(a)

    # Decision vars: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([T, -np.ones((m, 1))])
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            t = float(res.x[n])
            return g, t
    except Exception:
        pass
    return None, None


def lp_maximize_sum(a, M):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g>=0."""
    n = len(a)
    m = 2 * n - 1
    T = build_toeplitz_rows(a)

    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0


def golden_section_line_search(a, g, n_evals=8):
    """Find best t in [0,1] minimizing obj of (1-t)*a + t*g (normalized)."""
    s_a = np.sum(a)
    if s_a < 1e-10:
        return a.copy(), compute_obj(a)
    a_norm = a / s_a

    s_g = np.sum(g)
    if s_g < 1e-10:
        return a_norm.copy(), compute_obj(a_norm)
    g_norm = g / s_g

    best_obj = compute_obj(a_norm)
    best_result = a_norm.copy()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = hi - c1

    def eval_at(t):
        arr = np.maximum((1.0 - t) * a_norm + t * g_norm, 0.0)
        s = np.sum(arr)
        if s < 1e-10:
            return np.inf
        return compute_obj(arr / s)

    f1 = eval_at(c1)
    f2 = eval_at(c2)

    for _ in range(n_evals):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = eval_at(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = hi - gr * (hi - lo)
            f2 = eval_at(c2)

    t_opt = (lo + hi) / 2.0
    result = np.maximum((1.0 - t_opt) * a_norm + t_opt * g_norm, 0.0)
    s = np.sum(result)
    if s > 1e-10:
        result = result / s
    obj = compute_obj(result)

    if obj < best_obj:
        best_obj = obj
        best_result = result.copy()

    return best_result, best_obj


def constraint_reduction(a, num_iter=50, damp=0.97):
    """Reduce values at positions contributing most to peak convolution."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    for _ in range(num_iter):
        conv = compute_conv(a)
        k = int(np.argmax(conv))

        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        indices = np.arange(i_start, i_end + 1)
        partners = k - indices
        products = a[indices] * a[partners]

        max_pos = int(np.argmax(products))
        i_star = int(indices[max_pos])
        j_star = int(partners[max_pos])

        if a[i_star] >= a[j_star]:
            a[i_star] *= damp
        else:
            a[j_star] *= damp

        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

    return a


def multi_peak_reduction(a, num_iter=30, num_peaks=3, damp=0.96):
    """Target multiple peaks simultaneously for reduction."""
    n = len(a)
    a = a.copy()

            indices = np.arange(i_start, i_end + 1)
            partners = k - indices
            products = a[indices] * a[partners]

            max_pos = int(np.argmax(products))
            i_star = int(indices[max_pos])
            j_star = int(partners[max_pos])

            if a[i_star] >= a[j_star]:
                a[i_star] *= damp
            else:
                a[j_star] *= damp

        a = np.maximum(a, 0.0)
        s = np.sum(a)

    return a


def spectral_flatten(a, alpha=0.05, num_iter=20):
    """Flatten the power spectrum iteratively to reduce autocorrelation peaks."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(num_iter):
        fa = rfft(a)
        mag = np.abs(fa)
        phase = np.angle(fa)

        target = np.mean(mag)
        mag_new = (1.0 - alpha) * mag + alpha * target
        fa_new = mag_new * np.exp(1j * phase)
        a_new = irfft(fa_new, n)
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)
```
