```python
def subgradient_mirror_descent(a, deadline, max_iters=10000):
    n = len(a)
    a = a.astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    eta = 0.1  # step size
    tol = 0.05  # tolerance for active set
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        
        # Active set with tolerance
        active_threshold = max_b * (1 - tol)
        active_mask = (b >= active_threshold).astype(np.float64)
        
        if active_mask.sum() == 0:
            active_mask[int(np.argmax(b))] = 1.0
        
        # Compute pressure
        pressure = compute_pressure_fft(a, active_mask)
        avg_pressure = pressure.mean()
        
        # Mirror descent step
        # a_new[i] = a[i] * exp(-eta * (pressure[i] - avg_pressure))
        log_update = -eta * (pressure - avg_pressure)
        # Clip to avoid overflow
        log_update = np.clip(log_update, -5, 5)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 0.01:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            # Maybe decrease eta slowly
        else:
            # Try smaller step
            a = (a + a_new) / 2
            a = a / a.sum()
            J_half = compute_J(a)
            if J_half < best_J:
                best_J = J_half
                best_a = a.copy()
        
        # Adaptive tolerance: tighten as we converge
        if it > 100:
            tol = max(0.001, tol * 0.999)
        
        # Adaptive step size
        if it % 50 == 0:
            eta *= 0.99
    
    return best_a, best_J
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast subgradient mirror descent with FFT-based pressure computation
2. Adaptive active set tolerance
3. Periodic LP-based corrections
4. Efficient coordinate descent
5. Multi-length exploration

import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def compute_pressure(a, active_mask):
    """
    Compute pressure[i] = sum_{m=0}^{n-1} a[m] * active_mask[i+m] / |A|
    using FFT-based convolution.
    """
    n = len(a)
    a_rev = a[::-1]
    size = 1
    while size < n + len(active_mask) - 1:
        size <<= 1
    fa = np.fft.rfft(a_rev, size)
    fm = np.fft.rfft(active_mask.astype(np.float64), size)
    conv_result = np.fft.irfft(fa * fm, size)
    pressure = conv_result[n-1 : 2*n-1]
    count = active_mask.sum()
    if count < 1e-10:
        return np.ones(n) / n
    return pressure / count

def subgradient_mirror_descent(a, deadline, max_iters=20000, 
                                 use_lp_period=100, lp_time_limit=5):
    """
    Fast subgradient mirror descent with adaptive active set and periodic LP.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return a, float('inf')
    a = a / s
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    eta = 0.5  # initial step size
    tol = 0.1  # initial tolerance for active set
    no_improve = 0
    consecutive_worse = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        
        # Adaptive tolerance: start loose, tighten over time
        # Also adapt based on the distribution of b
        active_threshold = max_b * (1.0 - tol)
        active_mask = (b >= active_threshold).astype(np.float64)
        
        if active_mask.sum() < 1:
            active_mask[int(np.argmax(b))] = 1.0
        
        # Compute pressure via FFT
        pressure = compute_pressure(a, active_mask)
        avg_pressure = pressure.mean()
        
        # Mirror descent step: a_new[i] = a[i] * exp(-eta * (pressure[i] - avg_pressure))
        direction = -(pressure - avg_pressure)
        log_update = eta * direction
        log_update = np.clip(log_update, -3, 3)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        a_new = a_new / s_new
        
        J_new = compute_J(a_new)
        
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            consecutive_worse = 0
        else:
            no_improve += 1
            consecutive_worse += 1
            
            # Try half step
            a_half = (a + a_new) / 2
            a_half = np.maximum(a_half, 1e-15)
            a_half = a_half / a_half.sum()
            J_half = compute_J(a_half)
            
            if J_half < best_J:
                best_J = J_half
                best_a = a_half.copy()
                a = a_half
                no_improve = 0
                consecutive_worse = 0
            elif J_half < J_new:
                a = a_half
                consecutive_worse = 0
            else:
                # Decrease step size
                eta *= 0.95
                consecutive_worse += 1
        
        # Adaptive tolerance
        if it % 100 == 0 and it > 0:
            tol = max(0.005, tol * 0.95)
            eta = min(eta * 1.02, 1.0)  # slowly increase step size
        
        # Periodic LP correction
        if use_lp_period > 0 and it > 0 and it % use_lp_period == 0:
            if time.monotonic() < deadline - 10:
                g = lp_best_response_fast(a, max_b, time_limit=lp_time_limit)
                if g is not None:
                    g = g / g.sum()
                    # Line search
                    best_t = 0.0
                    best_tJ = J_new if np.isfinite(J_new) else compute_J(a)
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 1.0]:
                        a_try = (1-t)*a + t*g
                        a_try = np.maximum(a_try, 0)
                        s_try = a_try.sum()
                        if s_try < 0.01:
                            continue
                        a_try = a_try / s_try
                        J_try = compute_J(a_try)
                        if J_try < best_tJ:
                            best_tJ = J_try
                            best_t = t
                    if best_t > 0:
                        a = (1-best_t)*a + best_t*g
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        J_after = compute_J(a)
                        if J_after < best_J:
                            best_J = J_after
                            best_a = a.copy()
                            no_improve = 0
        
        # Restart if stuck
        if no_improve > 200:
            # Perturb
            a = a + np.random.normal(0, 0.01, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0
            eta = 0.5
            tol = 0.1
    
    return best_a, best_J

def lp_best_response_fast(a, M, time_limit=5):
    """Fast LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    # Build sparse constraint matrix
    from scipy.sparse import csr_matrix, lil_matrix
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n-1, k)
        for j in range(j_min, j_max+1):
            A[k, j] = a[k-j]
    A = A.tocsr()
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def coordinate_descent(a, deadline, max_passes=10):
    """Efficient coordinate descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        S = a.sum()
        max_b = b.max()
        improved = False
        
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            ai = a[i]
            # Try several candidate values
            if ai < 1e-10:
                candidates = [0.01/S, 0.02/S, 0.05/S]
            else:
                candidates = [0.0, ai*0.3, ai*0.6, ai*0.9, ai*1.1, ai*1.4, ai*1.8]
            
            best_val = best_J
            best_new = ai
            
            for cand in candidates:
                delta = cand - ai
                if abs(delta) < 1e-14:
                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue
                
                # Compute new max efficiently
                # b[k] changes for k in [i, i+n-1] (within bounds)
                k_lo = max(0, i)
                k_hi = min(i + n - 1, 2*n - 2)
                
                # Outside max
                outside_parts = []
                if k_lo > 0:
                    outside_parts.append(b[:k_lo].max())
                if k_hi < 2*n-2:
                    outside_parts.append(b[k_hi+1:].max())
                outside_max = max(outside_parts) if outside_parts else 0
                
                # Inside max
                inside_max = 0
                for k in range(k_lo, k_hi+1):
                    j = k - i
                    if 0 <= j < n:
                        if k == 2*i:
                            new_val = b[k] + 2*delta*a[i] + delta*delta
                        else:
                            new_val = b[k] + 2*delta*a[j]
                        if new_val > inside_max:
                            inside_max = new_val
                
                new_max = max(outside_max, inside_max)
                val = 2 * n * new_max / (S_new * S_new)
                
                if val < best_val:
                    best_val = val
                    best_new = cand
            
            if best_new != ai:
                a[i] = best_new
                improved = True
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
        
        if not improved:
            break
```

```python
def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_pressure_fft(a, active_mask):
    """
    Compute pressure[i] = sum_m a[m] * active_mask[i+m] / |A|
    using FFT-based convolution.
    """
    n = len(a)
    a_rev = a[::-1]
    am = active_mask.astype(np.float64)
    size = 1
    while size < n + len(am) - 1:
        size <<= 1
    fa = np.fft.rfft(a_rev, size)
    fm = np.fft.rfft(am, size)
    conv_result = np.fft.irfft(fa * fm, size)
    pressure = conv_result[n - 1:2 * n - 1]
    count = am.sum()
    if count < 1e-10:
        return np.ones(n) / n
    return pressure / count


def compute_pressure_softmax(a, b, temperature):

    weight_k = exp(b[k] / temperature)

    total_w = weights.sum()
    if total_w < 1e-10:
        return np.ones(n) / n
    weights = weights / total_w
    return compute_pressure_fft(a, weights)

def lp_solve_fast(a, M, time_limit=8):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    if len(rows) == 0:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def coordinate_descent_fast(a, deadline, max_passes=20):
    """
    Efficient coordinate descent with incremental convolution update.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        S = a.sum()
        improved = False

        order = np.random.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]

            # Generate candidates
            if ai < 1e-10:
                candidates = [0.005 / S, 0.01 / S, 0.02 / S, 0.05 / S]
            else:
                candidates = [0.0, ai * 0.3, ai * 0.6, ai * 0.8,
                              ai * 1.2, ai * 1.5, ai * 2.0]

            best_val = best_J
            best_new = ai

            k_lo = max(0, i)
            k_hi = min(i + n - 1, 2 * n - 2)

            # Outside max (unchanged by this coordinate)
            outside_parts = []
            if k_lo > 0:
                outside_parts.append(b[:k_lo].max())
            if k_hi < 2 * n - 2:
                outside_parts.append(b[k_hi + 1:].max())
            outside_max = max(outside_parts) if outside_parts else 0

            for cand in candidates:
                delta = cand - ai
                if abs(delta) < 1e-14:
                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue

                # Compute new max of affected region
                inside_max = 0.0
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j == i:
                        new_val = b[k] + 2 * delta * ai + delta * delta
                    else:
                        new_val = b[k] + 2 * delta * a[j]
                    if new_val > inside_max:
                        inside_max = new_val

                new_max = max(outside_max, inside_max)
                val = 2 * n * new_max / (S_new * S_new)

                if val < best_val - 1e-14:
                    best_val = val
                    best_new = cand

            if best_new != ai:
                delta = best_new - ai
                a[i] = best_new
                S = S + delta
                # Update b incrementally
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j == i:
                        b[k] += 2 * delta * ai + delta * delta
                    else:
                        b[k] += 2 * delta * a[j]
                improved = True

        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J - 1e-14:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def multi_tolerance_subgradient(a, deadline, max_iters=20000, use_symmetry=True):
    """
    Run subgradient with multiple tolerance schedules and keep the best.
    """
    n = len(a)
    results = []

    # Strategy 1: Start loose, tighten
    tol_schedule_1 = [0.2, 0.15, 0.1, 0.05, 0.02, 0.01, 0.005]
    # Strategy 2: Start medium
    tol_schedule_2 = [0.1, 0.08, 0.05, 0.03, 0.01, 0.005, 0.003]

    for schedule in [tol_schedule_1, tol_schedule_2]:
        if time.monotonic() > deadline:
            break

        a_curr = a.copy()
        a_curr = np.maximum(a_curr, 0)
        a_curr = a_curr / a_curr.sum()
        if use_symmetry:
            a_curr = symmetrize(a_curr)
            a_curr = a_curr / a_curr.sum()

        best_a = a_curr.copy()
        best_J = compute_J(a_curr)

        eta = 0.6
        it = 0
        for tol in schedule:
            stage_deadline = min(deadline, time.monotonic() + 120)
            while it < max_iters and time.monotonic() < stage_deadline:
                b = fast_conv(a_curr)
                max_b = b.max()
                active_threshold = max_b * (1.0 - tol)
                active_mask = (b >= active_threshold).astype(np.float64)
                if active_mask.sum() < 1:
                    active_mask[int(np.argmax(b))] = 1.0

                pressure = compute_pressure_fft(a_curr, active_mask)
                if use_symmetry:
                    pressure = symmetrize(pressure)
                avg_p = pressure.mean()

                direction = -(pressure - avg_p)
                std_d = direction.std()
                if std_d < 1e-15:
                    break
                log_upd = np.clip(eta * direction / std_d, -4, 4)
                a_new = a_curr * np.exp(log_upd)
                a_new = np.maximum(a_new, 1e-15)
                if use_symmetry:
                    a_new = symmetrize(a_new)
                a_new = a_new / a_new.sum()

                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
                    a_curr = a_new
                    eta = min(eta * 1.01, 1.5)
                else:
                    eta *= 0.95
                    a_half = (a_curr + a_new) / 2
                    a_half = np.maximum(a_half, 1e-15)
                    if use_symmetry:
                        a_half = symmetrize(a_half)
                    a_half = a_half / a_half.sum()
                    J_half = compute_J(a_half)
                    if J_half < best_J:
                        best_J = J_half
                        best_a = a_half.copy()
                        a_curr = a_half

                it += 1

        results.append((best_J, best_a.copy()))

    results.sort(key=lambda x: x[0])
    return results[0][1], results[0][0]


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Symmetric random
    half = n // 2
    r_half = rng.uniform(0.5, 1.5, half + 1)
    r = np.concatenate([r_half, r_half[-2::-1]]) if n % 2 == 0 else \
        np.concatenate([r_half, r_half[-1::-1]])
    r = np.maximum(r, 0.01)
    inits.append(r / r.sum())

    # Smooth symmetric
    freqs = [1, 2, 3]
    for f in freqs:
        sm = 1.0 + 0.4 * np.cos(np.pi * f * t)
        sm = np.maximum(sm, 0.01)
        sm = symmetrize(sm)
        sm = np.maximum(sm, 0.01)
        inits.append(sm / sm.sum())

    # Bell (symmetric by construction)
    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())

    # Double bump (symmetric)
    dbl = np.exp(-6 * (t - 0.25) ** 2) + np.exp(-6 * (t - 0.75) ** 2) + 0.03
    dbl = symmetrize(dbl)
    dbl = np.maximum(dbl, 0.01)
    inits.append(dbl / dbl.sum())

    # If parent available, add perturbed versions
    if parent is not None:
        parent_n = parent.size
        if parent_n != n:
            t_old = np.linspace(0, 1, parent_n)
            parent_resampled = np.interp(t, t_old, parent)
        else:
            parent_resampled = parent.copy()
        parent_resampled = np.maximum(parent_resampled, 0.01)
        parent_resampled = symmetrize(parent_resampled)
        parent_resampled = parent_resampled / parent_resampled.sum()

        for scale in [0.02, 0.05, 0.1]:
            perturbed = parent_resampled + rng.normal(0, scale, n)
            perturbed = np.maximum(perturbed, 0.01)
            perturbed = symmetrize(perturbed)
            perturbed = perturbed / perturbed.sum()
            inits.append(perturbed)

    return inits


def resample_sequence(src, dest_n):
    """Resample 1D sequence to new length via interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses fast FFT subgradient method with symmetry, LP corrections,
    coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            parent = symmetrize(parent)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        parent = symmetrize(parent)
        update_best(parent)

    n_parent = parent.size
    print(f"Starting: n={n_parent}, best_score={best_score:.8f}", flush=True)

    # === PHASE 1: Fast subgradient from parent (main workhorse) ===
    p1_deadline = min(t_start + 400, deadline - 400)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Subgradient Mirror Descent (n={n_parent})", flush=True)
    print(f"{'=' * 60}", flush=True)

    a1 = parent.copy()
    a1 = np.maximum(a1, 0)
    a1 = a1 / a1.sum()
    a1 = symmetrize(a1)

    res1, J1 = subgradient_mirror_descent(a1, p1_deadline, max_iters=100000,
                                           use_symmetry=True, lp_period=300,
                                           init_eta=0.8, init_tol=0.15)
    update_best(res1)
    print(f"  Phase 1 done: J={J1:.8f}", flush=True)

    # Also try multi-tolerance
    p1b_deadline = min(t_start + 550, deadline - 300)
    if time.monotonic() < p1b_deadline:
        res1b, J1b = multi_tolerance_subgradient(res1, p1b_deadline,
                                                  max_iters=100000, use_symmetry=True)
        update_best(res1b)
        if J1b < J1:
            res1 = res1b
            J1 = J1b
        print(f"  Phase 1b done: J={J1b:.8f}", flush=True)

    # === PHASE 2: Coordinate descent polishing ===
    p2_deadline = min(t_start + 700, deadline - 250)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Coordinate Descent Polishing", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        a_cd = coordinate_descent_fast(best_seq, p2_deadline, max_passes=30)
        update_best(a_cd)
        # Also polish the subgradient result if different
        if not np.allclose(a_cd, res1):
            a_cd2 = coordinate_descent_fast(res1, min(p2_deadline + 30, deadline - 200),
                                             max_passes=20)
            update_best(a_cd2)
        print(f"  Phase 2 done: best J={best_score:.8f}", flush=True)

    # === PHASE 3: LP-based refinement ===
    p3_deadline = min(t_start + 900, deadline - 200)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent Refinement", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        n_best = best_seq.size
        a_lp = best_seq.copy()
        a_lp = np.maximum(a_lp, 0)
        a_lp = a_lp / a_lp.sum()

        # LP-based mirror descent
        for lp_iter in range(20):
            if time.monotonic() > p3_deadline:
                break
            b = fast_conv(a_lp)
            M = b.max()
            g = lp_solve_fast(a_lp, M, time_limit=15)
            if g is None:
                break
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs < 1e-10:
                break
            g = g / gs
            g = symmetrize(g)
            g = np.maximum(g, 0)
            g = g / g.sum()

            # Line search
            best_t = 0.0
            best_tJ = compute_J(a_lp)
            for t in np.linspace(0.05, 1.0, 10):
                a_try = (1 - t) * a_lp + t * g
                a_try = np.maximum(a_try, 0)
                a_try = symmetrize(a_try)
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st < 0.01:
                    continue
                a_try = a_try / st
                J_try = compute_J(a_try)
                if J_try < best_tJ:
                    best_tJ = J_try
                    best_t = t

            if best_t > 0:
                a_lp = (1 - best_t) * a_lp + best_t * g
                a_lp = np.maximum(a_lp, 0)
                a_lp = symmetrize(a_lp)
                a_lp = a_lp / a_lp.sum()
                update_best(a_lp)

            if best_t < 0.01:
                break

        print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)

    # === PHASE 4: Diverse initializations ===
    p4_deadline = min(t_start + 1150, deadline - 150)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Diverse Initializations", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        n_d = best_seq.size
        inits = generate_inits(n_d, rng, parent=best_seq)

        # Score initializations
        scored_inits = []
        for init in inits:
            init = np.maximum(init, 0)
            init = symmetrize(init)
            init = np.maximum(init, 0)
            if init.sum() > 0.01:
                init = init / init.sum()
                J_init = compute_J(init)
                scored_inits.append((J_init, init))

        scored_inits.sort(key=lambda x: x[0])

        # Run subgradient on top initializations
        for J_init, init in scored_inits[:6]:
            if time.monotonic() > p4_deadline:
                break
            phase_dl = min(time.monotonic() + 35, p4_deadline)
            a_r, J_r = subgradient_mirror_descent(init, phase_dl, max_iters=50000,
                                                   use_symmetry=True, lp_period=0,
                                                   init_eta=0.6, init_tol=0.1)
            update_best(a_r)
            # Quick CD
            a_cd = coordinate_descent_fast(a_r, min(phase_dl + 10, p4_deadline),
                                            max_passes=5)
            update_best(a_cd)

        print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Multi-length exploration ===
    p5_deadline = min(t_start + 1350, deadline - 100)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Multi-length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = set()
        for mult in [0.75, 1.25, 1.5, 2.0, 0.5]:
            L = int(n_base * mult)
            if 64 <= L <= 2048:
                target_lengths.add(L)
        for L in [256, 384, 512, 640, 768, 1024, 1536]:
            target_lengths.add(L)

        target_lengths = sorted([L for L in target_lengths if L != n_base])

        for tgt_n in target_lengths[:6]:
            if time.monotonic() > p5_deadline:
                break
            a_ext = resample_sequence(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = symmetrize(a_ext)
            a_ext = a_ext / a_ext.sum()

            phase_dl = min(time.monotonic() + 30, p5_deadline)
            a_r, J_r = subgradient_mirror_descent(a_ext, phase_dl, max_iters=30000,
                                                   use_symmetry=True, lp_period=0,
                                                   init_eta=0.5, init_tol=0.1)
            update_best(a_r)
            a_cd = coordinate_descent_fast(a_r, min(phase_dl + 10, p5_deadline),
                                            max_passes=5)
            update_best(a_cd)

        print(f"  Phase 5 done: best J={best_score:.8f}", flush=True)

    # === PHASE 6: Final intensive polishing ===
    p6_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 6: Final Polishing", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        # Subgradient with very tight tolerance
        a_fin = best_seq.copy()
        a_fin = np.maximum(a_fin, 0)
        a_fin = a_fin / a_fin.sum()
        a_fin = symmetrize(a_fin)

        res_fin, J_fin = subgradient_mirror_descent(a_fin, p6_deadline,
                                                     max_iters=100000,
                                                     use_symmetry=True, lp_period=500,
                                                     init_eta=0.3, init_tol=0.02)
        update_best(res_fin)

        # Final CD
        a_cd = coordinate_descent_fast(res_fin, p6_deadline, max_passes=20)
        update_best(a_cd)

        # One more LP
        if time.monotonic() < p6_deadline - 10:
            a_lp_fin = res_fin.copy()
            a_lp_fin = np.maximum(a_lp_fin, 0)
            a_lp_fin = a_lp_fin / a_lp_fin.sum()
            b = fast_conv(a_lp_fin)
            M = b.max()
            g = lp_solve_fast(a_lp_fin, M, time_limit=20)
            if g is not None:
                g = np.maximum(g, 0)
                g = g / g.sum()
                g = symmetrize(g)
                g = np.maximum(g, 0)
                g = g / g.sum()
                for t in [0.1, 0.3, 0.5, 0.7, 1.0]:
                    a_try = (1 - t) * a_lp_fin + t * g
                    a_try = np.maximum(a_try, 0)
                    a_try = symmetrize(a_try)
                    a_try = a_try / a_try.sum()
                    update_best(a_try)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256)
        best_seq = symmetrize(best_seq)
        update_best(best_seq)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
```
