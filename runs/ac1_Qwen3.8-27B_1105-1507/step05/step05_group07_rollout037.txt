                conv[lo:hi + 1] += 2.0 * delta * a
                # Need to be careful: a[i] changes, so the slice changes
                # For correctness, recompute the affected part
                # Actually the slice should use the OLD values of other elements
                # Since we're only changing a[i], the formula is:
                # conv[k] += 2*delta*a[k-i] for k in [i, i+n-1] (using old a values)
                # But we already did this above. Let me redo properly.
                # Actually the incremental update is:
                # conv_new[k] = conv_old[k] + 2*delta*a_old[k-i] + delta^2 (at k=2i)
                # where a_old[k-i] is the old value (which equals current a[k-i] since we only change a[i])
                a[i] = best_val
                s = s - old + best_val
                sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

        if verbose and sweep % 50 == 0:
            print(f"    CD sweep {sweep}: J={current_obj:.8f}", flush=True)

        if no_improve_sweeps > 200:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_monotone_inits(n, rng):
    """Generate diverse monotonically decreasing initial sequences."""
    inits = []
    t = np.arange(n, dtype=np.float64)

    # Geometric sequences
    for r in [0.99, 0.98, 0.97, 0.95, 0.93, 0.9, 0.85, 0.8]:
        a = np.array([r ** i for i in range(n)])
        inits.append(a / a.sum())

    # Exponential decay
    for lam in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
        a = np.exp(-lam * t)
        inits.append(a / a.sum())

    # Power law
    for alpha in [0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        a = (t + 1) ** (-alpha)
        inits.append(a / a.sum())

    # Linear decreasing
    a = np.maximum(n - t, 1)
    inits.append(a / a.sum())

    # Quadratic
    a = (n - t) ** 2
    inits.append(a / a.sum())

    # Cosine window (smoothly decreasing from 1 to ~0)
    a = np.cos(np.pi * t / (2 * n)) ** 2
    a = np.maximum(a, 1e-6)
    inits.append(a / a.sum())

    # Mixed: rapid initial decay then slow tail
    for gamma in [1.0, 2.0, 5.0, 10.0]:
        a = np.exp(-gamma * t / n) + 0.1 * np.exp(-0.1 * t / n)
        inits.append(a / a.sum())

    # Smooth random (monotone)
    for _ in range(5):
        raw = rng.uniform(0.5, 1.0, n)
        # Make it monotone decreasing by cumulative product
        ratios = rng.uniform(0.95, 1.0, n - 1)
        a = np.ones(n)
        for i in range(1, n):
            a[i] = a[i - 1] * ratios[i - 1]
        inits.append(a / a.sum())

    return inits


def generate_general_inits(n, rng):
    """Generate general (non-monotone) initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine bumps
    for freq in [1, 2, 3, 4]:
        for phase in [0, 0.5]:
            a = np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * (freq * t + phase)), 0.01)
            inits.append(a / a.sum())

    # Gaussian bumps
    for center in [0.25, 0.5, 0.75]:
        for width in [0.1, 0.2, 0.3]:
            a = np.exp(-((t - center) / width) ** 2) + 0.05
            inits.append(a / a.sum())

    # Smooth random
    ksize = max(5, n // 30)
    kernel = np.ones(ksize) / ksize
    for _ in range(5):
        smooth = np.convolve(rng.uniform(0.5, 1.5, n), kernel, mode='same')
        inits.append(np.maximum(smooth, 0.01) / np.maximum(smooth, 0.01).sum())

    return inits
