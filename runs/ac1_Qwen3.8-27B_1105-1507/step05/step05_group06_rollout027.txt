def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-scale warm start (optimize small n, then upsample)
2. Active-set LP mirror descent (only near-active constraints for speed)
3. Nesterov-accelerated projected subgradient (fast global search)
4. Exact 1D coordinate descent (polishing)
5. Asymmetric initialization (ramps, sigmoids, etc.)
Key insight: optimal sequences are NOT palindromic; asymmetric shapes
can achieve lower J by balancing edge and middle convolution values.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    na, nb = len(a), len(b)
    if na == 0 or nb == 0:
        return np.array([])
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def nesterov_subgradient(a, deadline, max_iters=2000, lr_init=0.01, decay=0.999):
    """
    Nesterov-accelerated projected subgradient descent for min max_k b[k]
    s.t. a >= 0, sum(a) = 1.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    y = a.copy()  # Nesterov momentum variable
    beta = 0.5  # initial momentum
    t_nest = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov step
        t_new = (1 + np.sqrt(1 + 4 * t_nest ** 2)) / 2
        beta = (t_nest - 1) / t_new
        y = a + beta * (a - y)
        y = np.maximum(y, 0)
        y = y / y.sum()

        # Compute convolution and subgradient
        b = fast_conv(y)
        M = b.max()

        # Find near-active constraints (within 20% of max)
        threshold = M * 0.8
        active = np.where(b >= threshold)[0]

        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient: average of gradients at active points
        # grad b[k][i] = a[k-i] + a[i-k] (with boundary handling)
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                ki = k - i
                ik = i - k
                if 0 <= ki < n:
                    grad[i] += y[ki]
                if 0 <= ik < n:
                    grad[i] += y[ik]
        grad /= len(active)

        # Learning rate
        lr = lr_init / (1 + it * 0.01)
        
        # Gradient step
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 0.01:
            continue
        a_new = a_new / s_new

        # Update
        t_nest = t_new
        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

    return best_a

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    na, nb = len(a), len(b)
    if na == 0 or nb == 0:
        return np.array([])
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]


def compute_J(a):

    s = a.sum()
    if s < 0.01:

    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):

    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def active_set_lp(a, M, tolerance=0.85):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for near-active k, g >= 0.
    Only uses constraints where b[k] >= tolerance * M.
    """
    n = len(a)
    b = fast_conv(a)
    threshold = tolerance * M
    active_ks = np.where(b >= threshold)[0]
    
    if len(active_ks) == 0:
        active_ks = np.array([int(np.argmax(b))])

    # Build constraint matrix for only active constraints
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_ks):
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])

    if len(rows) == 0:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(len(active_ks), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_ks), M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None
