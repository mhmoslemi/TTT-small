"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent (log-sum-exp) with Nesterov acceleration for warm-up
2. Aggressive LP Mirror Descent with bisection step-size and periodic perturbation
3. Convolution-flattening heuristic via active constraint analysis
4. Multi-length exploration with adaptive time allocation
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_grad(a, tau):
    """Compute smooth approximation of max(conv(a,a)) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # For numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[j] += 2 * p[k] * a[k-j] for j in [lo, hi]
        # a[k-j] for j in [lo, hi] is a[k-hi : k-lo+1] reversed
        grad[lo:hi+1] += 2.0 * p[k] * a[k-hi:k-lo+1][::-1]
    return grad, p, b, b_max

def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.3, lr_end=0.003):
    """Nesterov-accelerated smooth gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        momentum = y - a
        a_new = np.maximum(y - lr * (grad + beta * momentum), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a

def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[b] = conv(a,b) <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, time_limit=10):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def find_best_blend(a, g, best_M, n_trials=20):
    """Find the best blending parameter t that minimizes max(conv(a_t, a_t))."""
    n = len(a)
    gs = g.sum()
    if gs < 1e-10:
        return None, best_M
    g = g / gs
    best_t = None
    best_M_try = best_M
    # Log-spaced trials
    for t in np.concatenate([np.linspace(0.01, 0.3, n_trials), np.linspace(0.3, 1.0, n_trials)]):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = a_t.sum()
        if s < 1e-10:
            continue
        a_t /= s
        M_t = fast_conv(a_t).max()
        if M_t < best_M_try:
            best_M_try = M_t
            best_t = t
    return best_t, best_M_try

def run_lp_mirror_aggressive(a, deadline, max_iters=500, perturb_every=30, perturb_scale=0.02):
    """Aggressive LP mirror descent with bisection step-size and periodic perturbation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M, time_limit=8)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Find best blend
        best_t, best_M_try = find_best_blend(a, g, M)
        
        if best_t is not None and best_M_try < best_M - 1e-14:
            a_t = (1.0 - best_t) * a + best_t * g
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            a_t /= s
            a = a_t
            best_M = best_M_try
            best_a = a.copy()
            no_imp = 0
        else:
            # Try smaller steps
            improved = False
            for t in [0.05, 0.02, 0.01, 0.005]:
                a_t = (1.0 - t) * a + t * g
                a_t = np.maximum(a_t, 0)
                s = a_t.sum()
                if s < 1e-10:
                    continue
                a_t /= s
                M_t = fast_conv(a_t).max()
                if M_t < best_M - 1e-14:
                    a = a_t
                    best_M = M_t
                    best_a = a.copy()
                    no_imp = 0
                    improved = True
                    break
            if not improved:
                no_imp += 1
                if no_imp > 30:
                    # Perturb to escape local minimum
                    if no_imp > 60:
                        break
                    if no_imp % perturb_every == 0 and time.monotonic() < deadline:
                        a = a + rng.normal(0, perturb_scale * 0.01, n)
                        a = np.maximum(a, 1e-6)
                        a /= a.sum()
                        no_imp = 0

    return best_a

def run_conv_flatten(a, deadline, max_iters=200):
    """Try to flatten the convolution profile by adjusting active coordinates."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(999)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        # Find top few peaks
        top_k = np.argsort(b)[-min(10, 2*n-1):][::-1]
        # For each peak, find which coordinates contribute most
        # and try to reduce them while increasing others
        
        improved = False
        for k in top_k[:5]:
            # Coordinates that contribute to b[k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            contrib = np.zeros(n)
            for j in range(lo, hi + 1):
                contrib[j] = a[k - j]
            # Find the coordinate with largest contribution
            j_max = lo + np.argmax(contrib[lo:hi+1])
            # Reduce it slightly
            delta = a[j_max] * 0.05
            if delta < 1e-10:
                continue
            a_new = a.copy()
            a_new[j_max] -= delta
            # Redistribute to a random coordinate that doesn't create a new peak
            # Try several candidates
            for attempt in range(20):
                j_add = rng.integers(0, n)
                a_try = a_new.copy()
                a_try[j_add] += delta
                a_try = np.maximum(a_try, 0)
                a_try /= a_try.sum()
                M_try = fast_conv(a_try).max()
                if M_try < best_M - 1e-12:
                    a = a_try
                    best_M = M_try
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def make_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.01))
    inits.append(np.ones(n))
    # S-shaped profiles
    for alpha in [0.5, 0.8, 1.2, 2.0, 3.0]:
        inits.append(np.maximum((t**alpha + (1-t)**alpha) / 2, 0.01))
    # Cosine profiles
    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(0.7 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    # Gaussian
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.1)
    # Double Gaussian
    inits.append(np.exp(-8*(t-0.25)**2) + np.exp(-8*(t-0.75)**2) + 0.05)
    # Smooth random
    k = max(3, n // 20)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    # Anti-symmetric peak (high in middle, low at edges)
    inits.append(np.maximum(1.0 - 0.8 * (2*t - 1)**2, 0.01))
    # Edge-peaked
    inits.append(np.maximum(0.3 + 0.7 * (2*t - 1)**4, 0.01))
    return inits

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth Nesterov gradient descent for warm-up and exploration
2. Cutting-plane LP Mirror Descent (much faster per iteration than full LP)
3. Alternating LP+Smooth to escape local optima
4. Targeted perturbation guided by convolution peak analysis
5. Adaptive multi-length search

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_conv_pair(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.3, lr_end=0.003):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        momentum = y - a
        a_new = np.maximum(y - lr * (grad + beta * momentum), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def solve_lp_with_constraints(a, M, constraint_idx, time_limit=5):
    n = len(a)
    if len(constraint_idx) == 0:
        constraint_idx = np.array([0])
    rows = []
    cols = []
    vals = []
    b_ub = []
    for k in constraint_idx:
        ri = len(b_ub)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(ri)
            cols.append(j)
            vals.append(a[k - j])
        b_ub.append(M)
    if len(vals) == 0:

    A = csr_matrix((vals, (rows, cols)), shape=(len(constraint_idx), n))

            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_best_response_cutting(a, M, max_rounds=8, time_limit=5, init_tol=0.4):

def find_best_blend_code(a, g, current_M):

        M_t = fast_conv(a_t).max()

    return best_t, best_M_try

            for t in [0.03, 0.015, 0.007, 0.003]:
                a_t = (1.0 - t) * a + t * g
                a_t = np.maximum(a_t, 0)
                s = a_t.sum()
                if s < 1e-10:
                    continue
                a_t /= s
                M_t = fast_conv(a_t).max()
                if M_t < best_M - 1e-14:
                    a = a_t
                    best_M = M_t
                    best_a = a.copy()
                    no_imp = 0
                    improved = True
                    break
            if not improved:
                no_imp += 1
                if no_imp > 80:
                    break
                if no_imp % perturb_every == 0:
                    # Targeted perturbation: reduce coordinates that contribute to peaks
                    b = fast_conv(a)
                    peak_k = int(np.argmax(b))
                    lo = max(0, peak_k - n + 1)
                    hi = min(n - 1, peak_k)
                    contrib = a[peak_k - hi:peak_k - lo + 1]
                    j_peak = lo + int(np.argmax(contrib))
                    a[j_peak] *= (1.0 - perturb_scale)
                    # Add to a low-contribution coordinate
                    j_add = rng.integers(0, n)
                    while j_add == j_peak:
                        j_add = rng.integers(0, n)
                    a[j_add] += a[j_peak] * perturb_scale / (1.0 - perturb_scale)
                    a = np.maximum(a, 1e-8)
                    a /= a.sum()
                    no_imp = 0

    return best_a


def run_alternating_lp_smooth(a, deadline, total_iters=200, lp_block=30, smooth_block=20):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    it = 0
    no_imp_long = 0

    while it < total_iters and time.monotonic() < deadline:
        # LP block
        for _ in range(lp_block):
            if time.monotonic() > deadline:
                break
            if it >= total_iters:
                break
            M = best_M
            g = lp_best_response_cutting(a, M, max_rounds=5, time_limit=4, init_tol=0.5)
            it += 1
            if g is None:
                break
            gs = g.sum()
            if gs < 1e-10:
                break
            g /= gs
            best_t, best_M_try = find_best_blend_code(a, g, M)
            if best_t is not None and best_M_try < best_M - 1e-14:
                a_t = (1.0 - best_t) * a + best_t * g
                a_t = np.maximum(a_t, 0)
                a_t /= a_t.sum()
                a = a_t
                best_M = best_M_try
                best_a = a.copy()
                no_imp_long = 0
            else:
                no_imp_long += 1

        # Smooth block (to escape LP local optima)
        if time.monotonic() < deadline:
            a = run_smooth_nesterov(a, min(time.monotonic() + 8, deadline),
                                    max_iters=smooth_block * 3,
                                    tau_start=5.0, tau_end=80.0,
                                    lr_init=0.15, lr_end=0.01)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
                no_imp_long = 0
            it += smooth_block

        if no_imp_long > 100:
            break

    return best_a


def run_targeted_perturb(a, deadline, rng, n_trials=10):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        # Analyze convolution peaks
        b = fast_conv(a)
        top_peaks = np.argsort(b)[-5:][::-1]

        # For each peak, find the dominant contributing coordinate
        for k in top_peaks:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            contrib = a[k - hi:k - lo + 1]
            j_dom = lo + int(np.argmax(contrib))

            # Reduce dominant coordinate
            for frac in [0.03, 0.05, 0.08]:
                a_new = a.copy()
                reduction = a_new[j_dom] * frac
                a_new[j_dom] -= reduction
                # Redistribute to a "safe" location (one with low local convolution)
                # Find positions where adding would least increase the max
                candidates = []
                for j_add in range(0, n, max(1, n // 50)):
                    a_try = a_new.copy()
                    a_try[j_add] += reduction
                    a_try = np.maximum(a_try, 0)
                    a_try /= a_try.sum()
                    M_try = fast_conv(a_try).max()
                    candidates.append((M_try, j_add))
                candidates.sort()
                if candidates:
                    best_cand_M, best_j = candidates[0]
                    if best_cand_M < best_M - 1e-14:
                        a_new = a_new.copy()
                        a_new[best_j] += reduction
                        a_new = np.maximum(a_new, 0)
                        a_new /= a_new.sum()
                        a = a_new
                        best_M = best_cand_M
                        best_a = a.copy()
                        break

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.01))
    inits.append(np.ones(n))
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        inits.append(np.maximum((t**alpha + (1 - t)**alpha) / 2, 0.01))
    for f in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(0.7 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.1)
    inits.append(np.exp(-8 * (t - 0.25)**2) + np.exp(-8 * (t - 0.75)**2) + 0.05)
    k = max(3, n // 20)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    inits.append(np.maximum(1.0 - 0.8 * (2 * t - 1)**2, 0.01))
    inits.append(np.maximum(0.3 + 0.7 * (2 * t - 1)**4, 0.01))
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(512)
        best.update(parent)
        print(f"Default parent: n=512, J={best.score:.6f}", flush=True)

    base_n = parent.size

    # === PHASE 1: Smooth gradient warm-up across lengths ===
    p1_budget_end = time.monotonic() + 180.0
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smooth Gradient Warm-up", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = sorted(set(
        [max(256, min(2000, int(base_n * m))) for m in [0.75, 1.0, 1.25, 1.5, 2.0]]
        | {384, 512, 640, 768, 1024, 1280}
    ))

    for n in lengths:
        if time.monotonic() > p1_budget_end:
            break
        t0 = time.monotonic()
        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_M = float('inf')

        n_iters = max(80, int(2500 * (384.0 / n)))
        for a0 in inits[:5]:
            if time.monotonic() > p1_budget_end:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()
            a = run_smooth_nesterov(a0, min(time.monotonic() + 20, p1_budget_end - 5),
                                    max_iters=n_iters, tau_start=1.0, tau_end=100.0,
                                    lr_init=0.3, lr_end=0.005)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Main LP Mirror Descent (cutting-plane) ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (Cutting-Plane)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        # Estimate LP time and set parameters accordingly
        lp_time = max(3, min(10, int(2 * (n / 256)**1.5)))
        print(f"  n={n}, lp_time={lp_time}s", flush=True)

        # Main LP run
        lp_dl = min(time.monotonic() + 350, deadline - 200)
        a = run_lp_mirror_cutting(best.seq, lp_dl, max_iters=2000,
                                  perturb_every=40, perturb_scale=0.012,
                                  init_tol=0.4, max_cp_rounds=8, lp_time=lp_time)
        best.update(a)
        print(f"  After main LP: J={best.score:.8f}", flush=True)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Alternating LP + Smooth ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Alternating LP + Smooth", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 100:
        alt_dl = min(time.monotonic() + 250, deadline - 100)
        a = run_alternating_lp_smooth(best.seq, alt_dl, total_iters=400,
                                      lp_block=25, smooth_block=15)
        best.update(a)
        print(f"  After alternating: J={best.score:.8f}", flush=True)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Targeted perturbation + re-optimization ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Targeted Perturbation", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 60:
        # Run targeted perturbation
        a = run_targeted_perturb(best.seq, min(time.monotonic() + 30, deadline - 50),
                                 rng, n_trials=15)
        best.update(a)

        # Follow with LP
        if time.monotonic() < deadline - 40:
            n = best.seq.size
            lp_time = max(3, min(8, int(2 * (n / 256)**1.5)))
            a = run_lp_mirror_cutting(a, min(time.monotonic() + 60, deadline - 40),
                                      max_iters=300, perturb_every=50,
                                      perturb_scale=0.01, lp_time=lp_time)
            best.update(a)

        # Random restart perturbations
        n = best.seq.size
        for scale in [0.04, 0.07, 0.1]:
            if time.monotonic() > deadline - 30:
                break
            for _ in range(2):
                if time.monotonic() > deadline - 25:
                    break
                noise = rng.normal(0, scale * 0.02, n)
                a = best.seq + noise
                a = np.maximum(a, 1e-5)
                a /= a.sum()
                a = run_smooth_nesterov(a, min(time.monotonic() + 12, deadline - 25),
                                        max_iters=200, tau_start=5.0, tau_end=60.0,
                                        lr_init=0.15, lr_end=0.02)
                best.update(a)
                if time.monotonic() < deadline - 20:
                    a2 = run_lp_mirror_cutting(a, deadline - 20, max_iters=50,
                                               lp_time=3)
                    best.update(a2)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Length refinement ===
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Length Refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 20:
        n_best = best.seq.size
        for delta in [-64, -32, 32, 64, 128, 256]:
            if time.monotonic() > deadline - 15:
                break
            n_try = n_best + delta
            if n_try < 256 or n_try > 2500:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            a_ext = run_smooth_nesterov(a_ext, min(time.monotonic() + 8, deadline - 10),
                                        max_iters=150, tau_start=3, tau_end=50,
                                        lr_init=0.2, lr_end=0.03)
            best.update(a_ext)

    best.report("Phase 5")
    print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === FINAL: Long LP polish ===
    if best.seq is not None and time.monotonic() < deadline - 3:
        print(f"\n{'='*60}", flush=True)
        print(f"FINAL LP POLISH", flush=True)
        print(f"{'='*60}", flush=True)
        n = best.seq.size
        lp_time = max(3, min(8, int(2 * (n / 256)**1.5)))
        a = run_lp_mirror_cutting(best.seq, deadline, max_iters=500,
                                  perturb_every=60, perturb_scale=0.008,
                                  lp_time=lp_time)
        best.update(a)

    if best.seq is None:
        best.seq = np.ones(512)
        best.score = compute_J(best.seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best.seq]
