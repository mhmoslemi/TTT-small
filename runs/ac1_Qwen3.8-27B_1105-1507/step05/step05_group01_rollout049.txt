"""
Asymmetric-seed LP alternating optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: asymmetric initialization (linear/quadratic ramp) starts in a better
basin of attraction than random/symmetric seeds, achieving ~1.34 for n=700 vs
~1.5 for symmetric starts. LP alternation then further refines the solution.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M, time_limit=3.0):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    A = build_toeplitz(a)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            b = np.maximum(res.x, 0)
            s = float(np.sum(b))
            if s > 1e-10:
                b = b / s
            return b, s
    except Exception:
        pass
    return None, 0.0


def lp_alternate(a, time_budget=100.0, verbose=False):
    """LP alternating optimization: iterate between a and LP solution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 50
    relax = 1.0
    rng = np.random.default_rng(12345)

    iter_count = 0
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        iter_count += 1

        conv = compute_conv(a)
        M = float(np.max(conv))
        tl = min(3.0, remaining * 0.3)

        g, S = solve_lp(a, M * relax, time_limit=tl)

        if g is not None and S > 1.0 + 1e-8:
            # Line search
            improved = False
            for t in [0.5, 0.7, 0.3, 0.9, 0.1, 1.0, 0.15, 0.85]:
                a_try = (1.0 - t) * a + t * g
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                stall = 0
                relax = max(1.0, relax - 0.01)
            else:
                stall += 1
        else:
            stall += 1

        if stall >= patience:
            # Escape strategy
            strategy = rng.integers(0, 4)
            if strategy == 0:
                # Multiplicative perturbation
                scale = rng.uniform(0.02, 0.1)
                a = best_a * np.exp(rng.normal(0, scale, n))
            elif strategy == 1:
                # Asymmetric push
                asym = rng.uniform(0.7, 1.3)
                a = best_a.copy()
                a[:n//2] *= asym
                a[n//2:] *= (2.0 - asym)
            elif strategy == 2:
                # Ramp bias
                k = np.arange(n)
                bias = 1.0 + rng.uniform(-0.3, 0.3) * k / n
                a = best_a * bias
            else:
                # Restructure: blend with a fresh ramp
                ramp = np.linspace(0.5, 1.5, n)
                a = rng.uniform(0.7, 1.0) * best_a + (1 - rng.uniform(0.7, 1.0)) * ramp

            a = np.maximum(a, 0)
            s2 = np.sum(a)
            if s2 > 1e-10:
                a = a / s2
            else:
                a = best_a.copy()
            stall = 0
            patience = max(20, patience // 2)
            relax = min(2.0, relax + 0.1)

        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it={iter_count}, obj={best_obj:.6f}, relax={relax:.2f}, "{f"t={elapsed:.0f}s")

    return best_a, best_obj


def subgradient_step(a, num_steps=100, step_scale=0.01, time_budget=20.0):
    """Subgradient descent on the max-convolution objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    for _ in range(num_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_val = float(np.max(conv))
        # Find all positions achieving (near) the max
        threshold = max_val * (1 - 1e-6)
        active = np.where(conv >= threshold)[0]

        # Compute subgradient
        sg = np.zeros(n)
        for k in active:
            for i in range(n):
                j = int(k - i)
                if 0 <= j < n:
                    sg[i] += a[j]
        sg /= len(active)

        # Step
        eta = step_scale / (1.0 + _ * 0.01)  # diminishing step
        a_new = a - eta * sg
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Scale
    v = v.astype(np.float64).copy()
    # Find t such that sum(max(v - t, 0)) = target_sum
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.where(v_sorted > 0)[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    # Binary search for t
    # sum(max(v_i - t, 0)) = target_sum
    # Try: t = (sum(v) - target_sum) / n is the mean adjustment
    lo = 0.0
    hi = float(np.max(v))
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    result = np.maximum(v - t, 0)
    s_result = np.sum(result)
    if s_result > 1e-10:
        result = result * target_sum / s_result
    else:
        result = np.full(n, target_sum / n)
    return result


def make_ramp(n, power=1.0, reverse=False):
    """Create a ramp-like initialization."""
    k = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if reverse:
        k = 1.0 - k
    a = (k + 0.1) ** power
    return a


def make_exp_ramp(n, alpha=1.0, reverse=False):
    """Create an exponential ramp."""
    k = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    if reverse:
        k = 1.0 - k
    a = np.exp(alpha * k)
    return a


def make_bimodal(n, center_frac=0.3, width=0.2):
    """Create a bimodal-ish asymmetric shape."""
    k = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center_frac) / width) ** 2)
    # Add a smaller bump on the other side
    a += 0.2 * np.exp(-0.5 * ((k - (1.0 - center_frac)) / (width * 1.5)) ** 2)
    return a


def resample_to_n(a, n):
    """Resample sequence to length n via interpolation."""
    if len(a) == n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf
