"""
Multi-phase optimizer for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) exact coordinate descent, (2) multiplicative "water-filling" updates,
(3) LP mirror descent, (4) simulated annealing with mutations, (5) multi-length search.
Key innovations: multiplicative updates that directly flatten the convolution profile,
and exploration of larger sequence lengths (up to 2048) for additional degrees of freedom.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coordinate_update_fast(a, i, b_current=None):
    """
    Find optimal value for a[i] given all other coordinates.
    Uses golden-section search on the 1D objective.
    """
    n = len(a)
    x_cur = a[i]
    S = a.sum()
    
    if b_current is None:
        b_current = fast_conv(a)
    
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b_current[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0
    
    # Precompute for affected positions
    affected = np.arange(k_lo, k_hi + 1)
    c_vals = np.zeros(len(affected))
    slopes = np.zeros(len(affected))
    quad_mask = np.zeros(len(affected), dtype=bool)
    
    for idx, k in enumerate(affected):
        if k == 2 * i:
            c_vals[idx] = b_current[k] - x_cur * x_cur
            quad_mask[idx] = True
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b_current[k] - 2 * x_cur * a[j]
                slopes[idx] = 2 * a[j]
            # else: constant, c_vals and slopes already 0 (but c_vals should be b_current[k])
            else:
                c_vals[idx] = b_current[k]
    
    def obj(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 1e-15:
            return 1e10
        scale2 = 1.0 / (new_sum * new_sum)
        # Linear part
        m = base_max
        vals = c_vals + slopes * x_test
        m = max(m, vals.max())
        # Quadratic part
        if quad_mask.any():
            for idx in np.where(quad_mask)[0]:
                qv = c_vals[idx] + x_test * x_test
                if qv > m:
                    m = qv
        return m * scale2
    
    # Golden section search
    lo = 0.0
    hi = max(x_cur * 3.0 + 0.01, 2.0 * np.sqrt(max(b_current.max(), 0)) / S + 0.01, 0.1)
    
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a, max_sweeps, deadline, shuffle=True, seed=0, track_best=False):
    """Run coordinate descent sweeps."""
    rng_c = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        if shuffle:
            rng_c.shuffle(order)
        
        b_current = fast_conv(a)
        sweep_improved = False
        for i in order:
            x_opt = coordinate_update_fast(a, i, b_current)
            if abs(x_opt - a[i]) > 1e-12:
                a[i] = x_opt
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 10
        if no_improve > 20:
            break
    
    return best_a


def multiplicative_update(a, alpha=0.1, n_iter=10):
    """
    Multiplicative 'water-filling' update that directly targets flattening the convolution.
    For each position j in a, reduce it proportionally to how much it contributes to peak
    convolution values.
    """
    n = len(a)
    a = np.maximum(a, 1e-10).copy()
    a = a / a.sum()
    
    for _ in range(n_iter):
        b = fast_conv(a)
        M = b.max()
        if M < 1e-15:
            break
        
        # For each j, compute its "contribution weight" to peaks
        # w[j] = sum over k where b[k] > M/2 of a[j]*a[k-j] / b[k]
        contribution = np.zeros(n)
        threshold = M * 0.5
        peak_mask = b > threshold
        
        for k in np.where(peak_mask)[0]:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                other = k - j
                if 0 <= other < n:
                    contribution[j] += a[j] * a[other] / max(b[k], 1e-15)
        
        # Normalize contribution
        c_max = contribution.max()
        if c_max > 0:
            # Multiplicative update: reduce where contribution is high
            a = a * (1 - alpha * contribution / c_max)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
    
    return a


def lp_best_response(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a, max_iters, deadline, time_per_lp=5):
    """LP mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=time_per_lp)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        improved = False
        for t in [0.3, 0.15, 0.08, 0.04, 0.02]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
    
    return best_a


def simulated_annealing_cd(a, max_iters, deadline, T_start=0.1, T_end=0.001):
    """
    Simulated annealing combined with coordinate descent.
    Randomly accepts worse moves to escape local minima.
    """
    rng = np.random.default_rng(42)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    cur_M = best_M
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        T = T_start * (T_end / T_start) ** (it / max(1, max_iters))
        
        # Random coordinate update with noise
        i = rng.integers(0, n)
        b = fast_conv(a)
        x_opt = coordinate_update_fast(a, i, b)
        
        # Add noise
        noise = rng.normal(0, T * a.max())
        x_trial = max(0, x_opt + noise)
        
        a_trial = a.copy()
        a_trial[i] = x_trial
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s
        
        M_trial = fast_conv(a_trial).max()
        
        # Accept or reject
        if M_trial < cur_M or rng.random() < np.exp(-(M_trial - cur_M) / max(T, 1e-10)):
            a = a_trial
            cur_M = M_trial
            if M_trial < best_M:
                best_M = M_trial
                best_a = a.copy()
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
    
    inits.append(np.ones(n))
    
    # Central concentration (triangular-like)
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2*t - 1), 0.05))
    
    # Edge concentration
    inits.append(0.5 + 0.5 * np.abs(2*t - 1))
    
    # Cosine variations
    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.05))
    
    # Gaussian
    inits.append(np.exp(-3 * (t - 0.5)**2) + 0.05)
    
    # Smooth random
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.05))
    
    # "U-shaped" (high at edges, low in middle)
    inits.append(0.3 + 0.7 * (2*t - 1)**2)
    
    # Alternating (period-2)
    alt = np.ones(n)
    alt[::2] = 0.6
    inits.append(alt)
    
    return inits

"""
Multi-phase optimizer for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) exact coordinate descent with vectorized golden-section, (2) multiplicative
water-filling updates, (3) simulated annealing with mutations, (4) LP mirror descent,
(5) multi-length exploration (up to n=2048). Key innovations: exploring larger sequence
lengths for more degrees of freedom, multiplicative updates that directly flatten the
convolution profile, and aggressive local polishing via repeated CD+LP cycles.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coordinate_update_fast(a, i, b_current=None):
    """Find optimal value for a[i] via golden-section search on 1D piecewise-quad objective."""
    n = len(a)
    x_cur = a[i]
    S = a.sum()

    if b_current is None:
        b_current = fast_conv(a)

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b_current[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0

    affected = np.arange(k_lo, k_hi + 1)
    c_vals = np.zeros(len(affected))
    slopes = np.zeros(len(affected))
    quad_idx = -1

    for idx, k in enumerate(affected):
        if k == 2 * i:
            c_vals[idx] = b_current[k] - x_cur * x_cur
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b_current[k] - 2 * x_cur * a[j]
                slopes[idx] = 2 * a[j]
            else:
                c_vals[idx] = b_current[k]

    def obj(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 1e-15:
            return 1e10
        scale2 = 1.0 / (new_sum * new_sum)
        m = base_max
        vals = c_vals + slopes * x_test
        vmax = vals.max()
        if vmax > m:
            m = vmax
        if quad_idx >= 0:
            qv = c_vals[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m * scale2

    lo = 0.0
    hi = max(x_cur * 3.0 + 0.01, 2.0 * np.sqrt(max(float(b_current.max()), 0.0)) / max(S, 1e-10) + 0.01, 0.1)

    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)

    for _ in range(45):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)

    x_opt = (lo + hi) / 2.0
    f_boundary = obj(0.0)
    f_opt = obj(x_opt)
    if f_boundary < f_opt:
        x_opt = 0.0

    return max(0.0, x_opt)


def run_coordinate_descent(a, max_sweeps, deadline, shuffle=True, seed=0):
    """Run coordinate descent sweeps to minimize max(conv) with sum(a)=1."""
    rng_c = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        if shuffle:
            rng_c.shuffle(order)

        b_current = fast_conv(a)
        sweep_improved = False
        for i in order:
            x_opt = coordinate_update_fast(a, i, b_current)
            if abs(x_opt - a[i]) > 1e-11:
                a[i] = x_opt
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 10
        if no_improve > 25:
            break

    return best_a


def multiplicative_update(a, alpha=0.1, n_iter=15):
    """Multiplicative water-filling: reduce coordinates contributing to peak convolution."""
    n = len(a)
    a = np.maximum(a, 1e-10).copy()
    a = a / a.sum()

    for _ in range(n_iter):
        b = fast_conv(a)
        M = float(b.max())
        if M < 1e-15:
            break

        contribution = np.zeros(n)
        threshold = M * 0.6
        peak_positions = np.where(b > threshold)[0]

        for k in peak_positions:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                other = k - j
                if 0 <= other < n:
                    contribution[j] += a[j] * a[other]

        c_max = contribution.max()
        if c_max > 1e-15:
            a = a * (1.0 - alpha * contribution / c_max)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()

    return a


def lp_best_response(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a, max_iters, deadline, time_per_lp=5):
    """LP mirror descent with adaptive step sizes."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=time_per_lp)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        improved = False
        step_candidates = [0.4, 0.25, 0.15, 0.08, 0.05, 0.03, 0.01]
        for t in step_candidates:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-13:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 12:
                break
        else:
            no_improve = 0

    return best_a


def simulated_annealing_cd(a, max_iters, deadline, T_start=0.05, T_end=0.0005, seed=42):
    """Simulated annealing: randomly accepts worse moves to escape local minima."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    cur_M = best_M

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = it / max(1, max_iters)
        T = T_start * (T_end / T_start) ** progress

        i = int(rng.integers(0, n))
        b = fast_conv(a)
        x_opt = coordinate_update_fast(a, i, b)

        noise_scale = T * max(a.max(), 0.01)
        noise = rng.normal(0, noise_scale)
        x_trial = max(0.0, x_opt + noise)

        a_trial = a.copy()
        a_trial[i] = x_trial
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s

        M_trial = float(fast_conv(a_trial).max())

        if M_trial < cur_M:
            accept = True
        else:
            delta = M_trial - cur_M
            if delta > 50 * T:
                accept = False
            else:
                accept = rng.random() < np.exp(-delta / max(T, 1e-12))

        if accept:
            a = a_trial
            cur_M = M_trial
            if M_trial < best_M:
                best_M = M_trial
                best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))

    inits.append(np.ones(n))
    inits.append(np.maximum(1.0 - 0.4 * np.abs(2 * t - 1), 0.05))
    inits.append(0.4 + 0.6 * np.abs(2 * t - 1)**2)

    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.35 * np.cos(2 * np.pi * freq * t), 0.05))

    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.08)

    ksize = max(5, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.05))

    alt = np.ones(n)
    alt[::2] = 0.65
    inits.append(alt.copy())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Coordinate descent across multiple lengths (512-2048)
    Phase 2: Multiplicative water-filling + CD
    Phase 3: Simulated annealing
    Phase 4: LP mirror descent
    Phase 5: Aggressive final polish (CD+LP cycles)
    Phase 6: Larger length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    base = tracker.best if tracker.best is not None else parent
    base_n = base.size

    # === PHASE 1: Coordinate Descent ===
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Coordinate Descent (target ~450s)", flush=True)
    print(f"{'='*60}", flush=True)

    # Prioritize parent length, then nearby, then larger
    lengths = sorted(set([base_n, base_n, 768, 1024, 1280, 1536]),
                     key=lambda x: abs(x - base_n))
    # Deduplicate while preserving order
    seen = set()
    lengths_dedup = []
    for ln in lengths:
        if ln not in seen and ln <= 2048:
            seen.add(ln)
            lengths_dedup.append(ln)
    lengths = lengths_dedup

    for n in lengths:
        if time.monotonic() > deadline - 350:
            break

        if n == base_n:
            t_alloc = 120
        elif n <= 768:
            t_alloc = 70
        elif n <= 1280:
            t_alloc = 60
        else:
            t_alloc = 50

        phase_deadline = min(time.monotonic() + t_alloc, deadline - 300)
        if time.monotonic() >= phase_deadline:
            continue

        t0 = time.monotonic()
        inits = diverse_inits(n, rng, base)
        best_for_n = None
        best_M_for_n = float('inf')

        num_inits = min(6, len(inits))
        for idx in range(num_inits):
            if time.monotonic() > phase_deadline:
                break
            a0 = np.maximum(inits[idx], 1e-6)
            a0 = a0 / a0.sum()

            max_sw = max(10, int(1500 / max(1, n // 64)))
            a_opt = run_coordinate_descent(a0, max_sw, phase_deadline, seed=seed + idx * 17 + n)

            M_val = float(fast_conv(a_opt).max())
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: M={best_M_for_n:.8f} J={J_val:.6f} ({elapsed:.1f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("Phase 1 complete")
    print(f"Phase 1 total: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Multiplicative Water-Filling ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multiplicative Water-Filling (target ~100s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 200:
        for alpha in [0.08, 0.15, 0.25]:
            if time.monotonic() > deadline - 200:
                break
            a_mf = multiplicative_update(tracker.best, alpha=alpha, n_iter=20)
            a_mf = run_coordinate_descent(a_mf, 25, min(time.monotonic() + 25, deadline - 200), seed=1111)
            tracker.update(a_mf)

    tracker.report("Phase 2 complete")
    print(f"Phase 2 total: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Simulated Annealing ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Simulated Annealing (target ~120s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 150:
        n_sa = tracker.best.size
        max_iters_sa = max(2000, int(800000 / max(1, n_sa)))
        a_sa = simulated_annealing_cd(tracker.best, max_iters_sa,
                                       min(time.monotonic() + 80, deadline - 100),
                                       T_start=0.04, T_end=0.0003, seed=777)
        tracker.update(a_sa)

        if time.monotonic() < deadline - 50:
            a_sa_cd = run_coordinate_descent(a_sa, 40, min(time.monotonic() + 40, deadline - 50), seed=2222)
            tracker.update(a_sa_cd)

    tracker.report("Phase 3 complete")
    print(f"Phase 3 total: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: LP Mirror Descent ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: LP Mirror Descent (target ~200s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 60:
        n_best = tracker.best.size
        time_per_lp = max(1, min(8, int(8 * 512 * 512 / (n_best * n_best))))
        max_iters_lp = max(10, min(80, int(140 / max(0.5, time_per_lp))))
        print(f"  n={n_best}, time_per_lp={time_per_lp}s, max_iters={max_iters_lp}", flush=True)

        a_lp = run_lp_mirror(tracker.best, max_iters_lp,
                             min(time.monotonic() + 140, deadline - 30),
                             time_per_lp=time_per_lp)
        tracker.update(a_lp)

        if time.monotonic() < deadline - 20:
            a_lp_cd = run_coordinate_descent(a_lp, 50, min(time.monotonic() + 25, deadline - 20), seed=3333)
            tracker.update(a_lp_cd)

    tracker.report("Phase 4 complete")
    print(f"Phase 4 total: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Aggressive Final Polish ===
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 10:
        # Cycle of CD + short LP
        for cycle in range(3):
            if time.monotonic() > deadline - 10:
                break

            # CD
            a_pol = run_coordinate_descent(tracker.best, 40,
                                           min(time.monotonic() + 20, deadline - 15), seed=5000 + cycle * 111)
            tracker.update(a_pol)

            # Short LP if n is small enough
            if tracker.best.size <= 1024 and time.monotonic() < deadline - 12:
                t_lp = max(1, min(3, int(3 * 512 * 512 / (tracker.best.size ** 2))))
                a_lp2 = run_lp_mirror(tracker.best, 15,
                                      min(time.monotonic() + 10, deadline - 12), time_per_lp=t_lp)
                tracker.update(a_lp2)

            # Brief perturbation + CD
            if time.monotonic() < deadline - 8:
                a_pert = tracker.best + rng.normal(0, 0.008, tracker.best.size)
                a_pert = np.maximum(a_pert, 0.001)
                a_pert = a_pert / a_pert.sum()
                a_pol2 = run_coordinate_descent(a_pert, 20,
                                                min(time.monotonic() + 10, deadline - 8), seed=9000 + cycle)
                tracker.update(a_pol2)

    tracker.report("Phase 5 complete")
    print(f"Phase 5 total: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === PHASE 6: Larger lengths if time remains ===
    if tracker.best is not None and time.monotonic() < deadline - 40:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 6: Larger lengths", flush=True)
        print(f"{'='*60}", flush=True)

        for mult in [2, 3]:
            if time.monotonic() > deadline - 15:
                break
            new_n = tracker.best.size * mult
            if new_n > 3000 or new_n <= tracker.best.size:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext_opt = run_coordinate_descent(a_ext, 30,
                                               min(time.monotonic() + 30, deadline - 15), seed=8888)
            tracker.update(a_ext_opt)

    tracker.report("Phase 6 complete")

    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
