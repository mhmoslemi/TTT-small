def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def make_palindromic(a):
    """Make a sequence palindromic by averaging with its reverse."""
    n = len(a)
    return (a + a[::-1]) / 2.0


def build_palindromic_lp_matrix(n, a):
    """
    Build the constraint matrix for the palindromic LP.
    Variables: g_0, ..., g_p where p = (n-1)//2
    Constraints: conv(a,g)_k <= M for k = 0, ..., n-1
    """
    # Determine number of free variables
    p = (n - 1) // 2  # max index of free variable
    num_vars = p + 1  # free variables are g_0, ..., g_p
    
    # fold function: fold(i) = min(i, n-1-i)
    # For constraint k, the coefficient of g_free[m] is:
    # sum of a_j for all j where fold(k-j) = m and 0<=k-j<n
    
    # Precompute: for each k and m, A[k,m] = sum_{j: fold(k-j)=m, 0<=k-j<n} a_j
    # fold(k-j) = m means k-j = m or k-j = n-1-m
    # So j = k-m or j = k-(n-1-m) = k-n+1+m
    
    A = np.zeros((n, num_vars))  # only first n constraints (by symmetry)
    
    for k in range(n):
        for m in range(num_vars):
            # j = k - m
            j1 = k - m
            # j = k - n + 1 + m
            j2 = k - n + 1 + m
            
            val = 0.0
            if 0 <= j1 < n:
                val += a[j1]
            if 0 <= j2 < n:
                # Check if j2 != j1 to avoid double counting
                # j2 == j1 iff k-n+1+m == k-m iff n-1 == 2m iff 2m == n-1
                if j2 != j1:
                    val += a[j2]
                else:
                    # They're the same, so we've already counted it
                    val = a[j1]
            
            A[k, m] = val
    
    return A, num_vars


def solve_palindromic_lp(a, M, A_matrix, num_vars):
    """
    Solve LP: max sum(g) s.t. A@g_free <= M, g_free >= 0.
    Then expand to full palindromic g.
    """
    n = len(a)
    p = (n - 1) // 2
    
    c = -np.ones(num_vars)  # minimize -sum
    # But wait: the objective is max sum_{j=0}^{n-1} g_j
    # If g is palindromic: sum = 2*sum_{m=0}^{p-1} g_m + g_p (if n odd)
    # or sum = 2*sum_{m=0}^{p-1} g_m (if n even, where p = n/2 is not a free var)
    
    # Actually, for n odd (n=2p+1): free vars are g_0,...,g_p. 
    #   Full sum = 2*(g_0+...+g_{p-1}) + g_p
    # For n even (n=2p): free vars are g_0,...,g_{p-1}.
    #   Full sum = 2*(g_0+...+g_{p-1})
    
    if n % 2 == 1:
        # n = 2p+1, num_vars = p+1
        obj = -np.concatenate([2*np.ones(p), [1.0]])
    else:
        # n = 2p, num_vars = p
        obj = -2*np.ones(num_vars)
    
    b_ub = np.full(n, M)
    bounds = [(0, None)] * num_vars
    
    try:
        A_sparse = csr_matrix(A_matrix)
        r = linprog(obj, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, 
                    method='highs', options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g_free = np.maximum(r.x, 0)
            # Expand to full palindromic
            g = np.zeros(n)
            for m in range(num_vars):
                g[m] = g_free[m]
                g[n-1-m] = g_free[m]
            return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t minimizing max of ((1-t)*a + t*g)^2 convolution."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_palindromic_lp_descent(a, max_iters, deadline, verbose=False):
    """
    LP mirror descent restricted to palindromic sequences.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = make_palindromic(a)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    A_matrix, num_vars = build_palindromic_lp_matrix(n, a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = compute_M(a)
        # Rebuild A if a changed significantly (actually a changes every iteration)
        A_matrix, _ = build_palindromic_lp_matrix(n, a)
        
        g = solve_palindromic_lp(a, M, A_matrix, num_vars)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        g = make_palindromic(g)  # ensure palindromic
        
        t, M_new = line_search_analytic(a, g)
        
        if t < 1e-10:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = make_palindromic(a_new)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 80:
            break
        
        if verbose and it % 100 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.4f}", flush=True)
    
    return best_a


def create_init_bathtub(n, alpha):
    """Create a bathtub-shaped initialization."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + alpha * x**2
    return np.maximum(a, 0.1)


def create_init_uniform(n):
    return np.ones(n)


def create_init_gaussian(n, sigma=0.3):
    x = np.linspace(-1, 1, n)
    return np.exp(-x**2 / (2*sigma**2))


def create_init_random(n, seed=42):
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.1
    return a


class Tracker:
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start
    
    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)
    
    def report(self):
        if self.best is not None:
            print(f"  best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
