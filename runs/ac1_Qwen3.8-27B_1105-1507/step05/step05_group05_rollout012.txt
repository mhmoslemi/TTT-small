- Pair coordinate updates to break 1D saddle points

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coord_update_fast(a, i, b_cur, base_max_cur):
    """
    Fast coordinate update for a[i] using precomputed convolution.
    Returns optimal x for a[i] (before renormalization).
    """
    n = len(a)
    x_cur = a[i]
    S = a.sum()

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # Precompute c_vals and slopes for affected range
    num_aff = k_hi - k_lo + 1
    c_vals = np.empty(num_aff)
    slopes = np.empty(num_aff)
    quad_pos = -1

    for idx in range(num_aff):
        k = k_lo + idx
        if k == 2 * i:
            c_vals[idx] = b_cur[k] - x_cur * x_cur
            slopes[idx] = 0.0
            quad_pos = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b_cur[k] - 2.0 * x_cur * a[j]
                slopes[idx] = 2.0 * a[j]
            else:
                c_vals[idx] = b_cur[k]
                slopes[idx] = 0.0

    def obj_vec(x_test):
        """Vectorized objective for golden section."""
        new_sum = S - x_cur + x_test
        if new_sum <= 0:
            return 1e10
        vals = c_vals + slopes * x_test
        m = base_max_cur
        vm = vals.max()
        if vm > m:
            m = vm
        if quad_pos >= 0:
            qv = c_vals[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)

    # Golden section search
    lo = 0.0
    hi = max(x_cur * 3.0, np.sqrt(max(b_cur.max(), 0.0)) + 0.5, 0.1)
    if hi > 5.0:
        hi = max(2.0, x_cur * 2.0)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_vec(a_pt)
    fb = obj_vec(b_pt)

    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_vec(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_vec(b_pt)

    x_opt = (lo + hi) / 2.0

    # Check boundary
    f_boundary = obj_vec(0.0)
    f_opt = obj_vec(x_opt)
    if f_boundary < f_opt:
        x_opt = 0.0

    return max(0.0, x_opt)


def pair_update(a, i, j, b_cur):
    """
    Update two coordinates a[i], a[j] simultaneously using a small grid search.
    Returns (x_i, x_j).
    """
    n = len(a)
    xi_cur = a[i]
    xj_cur = a[j]
    S = a.sum()

    # Affected k values for coordinate i
    ki_lo, ki_hi = i, min(2 * n - 2, i + n - 1)
    # Affected k values for coordinate j
    kj_lo, kj_hi = j, min(2 * n - 2, j + n - 1)

    all_lo = min(ki_lo, kj_lo)
    all_hi = max(ki_hi, kj_hi)

    mask = np.ones(2 * n - 1, dtype=bool)
    mask[all_lo:all_hi + 1] = False
    if mask.any():
        base_max = b_cur[mask].max()
    else:
        base_max = 0.0

    # For each affected k, compute the expression as a function of (xi, xj)
    # (a*a)_k = base_k + 2*xi*a[k-i] + 2*xj*a[k-j] + xi*xj*(contributions)
    # This gets complex, so use a coarse grid + refinement

    def obj_2d(xi, xj):
        new_sum = S - xi_cur - xj_cur + xi + xj
        if new_sum <= 0:
            return 1e10
        # Compute affected convolution values
        m = base_max
        for k in range(ki_lo, ki_hi + 1):
            j2 = k - i
            if 0 <= j2 < n:
                if k == 2 * i:
                    val = b_cur[k] - xi_cur * xi_cur - 2 * xi_cur * a[j2] + 2 * xi * a[j2] + xi * xi
                    # Wait, this is getting complicated. Let's just recompute.
                    pass
        # Simpler: just compute the affected values directly
        # For affected k: new_b_k = b_cur[k] + delta
        # delta from i: 2*(xi - xi_cur)*a[k-i] + (xi^2 - xi_cur^2) if k==2i
        # delta from j: 2*(xj - xj_cur)*a[k-j] + (xj^2 - xj_cur^2) if k==2j
        # Cross term: if k-i == j or k-j == i, there's a cross term (xi-xi_cur)*(xj-xj_cur)

        for k in range(all_lo, all_hi + 1):
            new_val = b_cur[k]
            # Contribution from changing a[i]
            ji = k - i
            if 0 <= ji < n:
                if k == 2 * i:
                    new_val += 2.0 * (xi - xi_cur) * a[ji] + (xi - xi_cur) * (xi + xi_cur)
                else:
                    new_val += 2.0 * (xi - xi_cur) * a[ji]
            # Contribution from changing a[j]
            jj = k - j
            if 0 <= jj < n:
                if k == 2 * j:
                    new_val += 2.0 * (xj - xj_cur) * a[jj] + (xj - xj_cur) * (xj + xj_cur)
                else:
                    new_val += 2.0 * (xj - xj_cur) * a[jj]
            # Cross term: a[i] appears as a[k-j] when i = k-j, and a[j] appears as a[k-i] when j = k-i
            # If k - i == j: cross term (xi - xi_cur)(xj - xj_cur)
            if k - i == j and 0 <= j < n and k != 2 * i and k != 2 * j:
                new_val += (xi - xi_cur) * (xj - xj_cur)
            if k - j == i and 0 <= i < n and k != 2 * i and k != 2 * j:
                new_val += (xi - xi_cur) * (xj - xj_cur)

            if new_val > m:
                m = new_val
        return m / (new_sum * new_sum)

    # Coarse grid
    best_val = obj_2d(xi_cur, xj_cur)
    best_xi, best_xj = xi_cur, xj_cur

    scale_i = max(abs(xi_cur), 0.05)
    scale_j = max(abs(xj_cur), 0.05)
    for di in [-2.0, -1.0, -0.5, 0.0, 0.5, 1.0, 2.0]:
        for dj in [-2.0, -1.0, -0.5, 0.0, 0.5, 1.0, 2.0]:
            xi = max(0.0, xi_cur + di * scale_i)
            xj = max(0.0, xj_cur + dj * scale_j)
            v = obj_2d(xi, xj)
            if v < best_val:
                best_val = v
                best_xi, best_xj = xi, xj

    return best_xi, best_xj


def run_annealed_coord_descent(a, deadline, seed=0, max_sweeps=500, T_start=0.02, T_end=0.0001):
    """
    Coordinate descent with simulated annealing and occasional pair updates.
    Works with a normalized to sum=1.

    else:
        a = a / s

    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        frac = sweep / max(1, max_sweeps)
        T = T_start * (T_end / T_start) ** frac

        sweep_improved = False
        b_cur = fast_conv(a)
        S = a.sum()

            if time.monotonic() > deadline:

                x_rand = max(0.0, a[i] + rng.normal(0, 0.1 * max(a[i], 0.05)))
                x_opt = x_rand

            # Occasionally do a pair update
            if rng.random() < 0.1:
                j = order[(i + 1) % n]
                if j != i:
                    xi_new, xj_new = pair_update(a, i, j, b_cur)
                    old_val = b_cur.max()
                    a[i] = xi_new
                    a[j] = xj_new

                    if s < 1e-10:
                        a[i] = 0.0
                        a[j] = 0.0
                        continue
                    a = a / s
                    new_val = fast_conv(a).max()
                    if new_val > old_val and rng.random() > np.exp(-(new_val - old_val) / T):
                        # Revert
                        a[i] = 0.0
                        a[j] = 0.0
                        # Restore old values (approximate)
                        pass  # Can't easily restore, just skip
                    else:
                        sweep_improved = True
                        b_cur = fast_conv(a)
                        continue

            if abs(x_opt - a[i]) > 1e-12:
                a[i] = x_opt
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        b_cur = fast_conv(a)
        M = b_cur.max()

        if M < best_M * (1 - 1e-10):

        else:

            a = a + rng.normal(0, 0.03, n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bnds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_best_response(a, max_iters, deadline):
    """LP best-response iteration."""
    a = np.maximum(a.astype(np.float64), 1e-8)

        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        improved = False
        base_t = max(0.03, 0.5 * (1 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.01]:

def smooth_max_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*(a*a)_k))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Numerical stability
    shifted_max = shifted.max()
    exp_b = np.exp(shifted - shifted_max)
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2 * p[k] * a[hi-lo:-1:-1]... 
        # (a*a)_k = sum_{j=lo}^{hi} a[j] * a[k-j]
        # d/d a[i] = 2 * a[k-i] for i in [lo, hi] (with the k=2i correction)
        # Simplified: use the formula d(a*a)_k/d(a_i) ≈ 2*a[k-i]
        idx = k - np.arange(lo, hi + 1)
        contrib = 2.0 * a[idx]
        # Correction for k=2i case
        center = k // 2
        if k % 2 == 0 and lo <= center <= hi:
            pos = center - lo
            contrib[pos] = a[center]
        grad[lo:hi + 1] += p[k] * contrib

    return grad


def run_adam_smooth(a, max_iters, deadline, tau_start=5.0, tau_end=100.0, lr=0.04):
    """Adam on smooth max objective."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        lr_t = lr / (1 + 0.005 * it)

        grad = smooth_max_grad(a, tau)
        grad = grad - grad.mean()  # Project to sum-zero

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        step = m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = a - lr_t * step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def structured_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []

        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))
        inits.append(np.maximum(p * (1 + rng.normal(0, 0.1, n)), 0.01))

    inits.append(rng.uniform(0.3, 1.0, n))

    # Structured: periodic

        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t), 0.01))

    # Gaussian
    inits.append(np.exp(-6 * (t - 0.5) ** 2) + 0.05)
    # Arc
    inits.append(np.maximum(np.sin(np.pi * t) ** 0.5, 0.05))
    # Two peaks
    inits.append(np.maximum(np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.05, 0.01))
    # Smooth random
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # "Anti-peak" structure: try to make convolution flat
    inits.append(np.maximum(1.0 - 0.3 * np.cos(4 * np.pi * t), 0.05))
    inits.append(np.maximum(1.0 + 0.2 * np.sin(2 * np.pi * 3 * t + 0.5), 0.05))

    return inits

    inits = structured_inits(n_main, rng, tracker.best if tracker.best is not None else parent)
    best_for_n = None
    best_M_for_n = float('inf')

    per_init_time = 0.40 * (p1_deadline - time.monotonic()) / max(1, len(inits))

    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_deadline - 60:
            break
        a0 = np.maximum(a0, 1e-6)
        a0 = a0 / a0.sum()

        local_deadline = min(time.monotonic() + per_init_time, p1_deadline - 30)
        if time.monotonic() >= local_deadline:
            local_deadline = time.monotonic() + 5

        a_opt = run_annealed_coord_descent(a0, local_deadline, seed=seed + idx * 13)
        M_val = fast_conv(a_opt).max()
        if M_val < best_M_for_n:
            best_M_for_n = M_val
            best_for_n = a_opt.copy()

        J_val = compute_J(a_opt)
        print(f"  init {idx}: M={M_val:.8f} J={J_val:.6f} t={time.monotonic()-tracker.t_start:.0f}s", flush=True)
        tracker.update(a_opt)

    if best_for_n is not None:
        tracker.update(best_for_n)

    # Try alternative length
    if time.monotonic() < p1_deadline - 30:
        inits_alt = structured_inits(alt_n, rng, tracker.best)
        for idx, a0 in enumerate(inits_alt[:4]):
            if time.monotonic() > p1_deadline - 20:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            local_deadline = min(time.monotonic() + 20, p1_deadline - 15)
            a_opt = run_annealed_coord_descent(a0, local_deadline, seed=seed + 100 + idx)
            tracker.update(a_opt)

    tracker.report("Phase 1 complete")

    # === PHASE 2: LP Best Response ===
    p2_start = time.monotonic()
    p2_deadline = p2_start + 0.25 * (deadline - p2_start)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Best Response (budget={p2_deadline - p2_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p2_deadline - 10:
        n_best = tracker.best.size
        est_lp_time = 0.008 * (n_best / 128) ** 2
        max_iters = max(10, min(200, int((p2_deadline - time.monotonic() - 5) / max(0.01, est_lp_time))))
        print(f"  n={n_best}, est_lp_time={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)

        a_lp = run_lp_best_response(tracker.best, max_iters, p2_deadline - 5)
        tracker.update(a_lp)

        # Second pass
        if time.monotonic() < p2_deadline - 5:
            a_lp2 = run_lp_best_response(a_lp, max(10, max_iters // 3), p2_deadline - 2)
            tracker.update(a_lp2)

    tracker.report("Phase 2 complete")

    # === PHASE 3: Smooth Gradient (Adam) ===
    p3_start = time.monotonic()
    p3_deadline = p3_start + 0.15 * (deadline - p3_start)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Adam Smooth Gradient (budget={p3_deadline - p3_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p3_deadline - 5:
        a_adam = run_adam_smooth(tracker.best, 3000, min(p3_deadline - 20, time.monotonic() + 80),
                                  tau_start=5.0, tau_end=120.0, lr=0.04)
        tracker.update(a_adam)

        if time.monotonic() < p3_deadline - 10:
            a_adam2 = run_adam_smooth(tracker.best, 2000, min(p3_deadline - 5, time.monotonic() + 40),
                                       tau_start=10.0, tau_end=60.0, lr=0.06)
            tracker.update(a_adam2)

    tracker.report("Phase 3 complete")

    # === PHASE 4: Final Coordinate Descent Polish ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final Coordinate Descent Polish (remaining={deadline - p4_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 5:
        # Multiple rounds of coordinate descent with different seeds
        for rnd in range(3):
            if time.monotonic() > deadline - 5:
                break
            a_polish = run_annealed_coord_descent(tracker.best, deadline - 10 - rnd * 10,
                                                  seed=77000 + rnd * 111,
                                                  max_sweeps=200,
                                                  T_start=0.005, T_end=0.0001)
            tracker.update(a_polish)

        # Quick LP if time
        if time.monotonic() < deadline - 10:
            a_final_lp = run_lp_best_response(tracker.best, 30, deadline - 5)
            tracker.update(a_final_lp)

    tracker.report("Phase 4 complete")

    # === PHASE 5: Try extending length if time remains ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        for mult in [2]:
            new_n = tracker.best.size * mult
            if new_n > 1500:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext_opt = run_annealed_coord_descent(a_ext, deadline - 5, seed=55555, max_sweeps=50)
            tracker.update(a_ext_opt)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
