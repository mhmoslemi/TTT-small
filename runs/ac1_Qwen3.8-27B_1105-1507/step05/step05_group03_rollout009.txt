"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp_standard(a, M, row_idx, col_idx, time_limit=10):
    """Original LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def solve_lp_targeted(a, M, active_k, row_idx, col_idx, time_limit=10):
    """
    Peak-targeting LP: min t s.t. (a*g)[k] - (a*a)[k] <= t for k in active_k,
    g >= 0, sum(g) = 1.
    
    Variables: [g_0, ..., g_{n-1}, t]
    Minimize: t
    Subject to:
      For each k in active_k: sum_j g_j * a_{k-j} - t <= (a*a)[k]
      sum(g) = 1
      g_j >= 0
      t >= -M
    """
    n = len(a)
    aa = fast_conv(a)
    
    # Build constraint matrix for active peaks
    # Each constraint: sum_j g_j * a_{k-j} - t <= aa[k]
    # In standard form: A_ub * x <= b_ub where x = [g, t]
    
    rows_ub = []
    cols_ub = []
    vals_ub = []
    b_ub_list = []
    
    for k in active_k:
        row = len(rows_ub)  # current row index
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows_ub.append(row)
            cols_ub.append(j)  # column for g_j
            vals_ub.append(a[k - j])
        # -t coefficient
        rows_ub.append(row)
        cols_ub.append(n)  # column for t
        vals_ub.append(-1.0)
        b_ub_list.append(aa[k])
    
    A_ub = csr_matrix((vals_ub, (rows_ub, cols_ub)), shape=(len(active_k), n + 1))
    b_ub = np.array(b_ub_list)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds: g_j >= 0, t >= -M
    bounds = [(0, None)] * n + [(-M, None)]
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x[:n], 0)
            sg = g.sum()
            if sg > 1e-15:
                return g / sg
    except Exception:
        pass
    return None


def line_search_analytic(a, g, n_iters=80):
    """Find best t in [0,1] minimizing max of ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    lo = 0.0
    hi = 1.0
    for _ in range(n_iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def get_active_set(a, fraction=0.2):
    """Get indices where self-convolution is in the top fraction."""
    aa = fast_conv(a)
    M = aa.max()
    threshold = M * (1.0 - fraction)
    active = np.where(aa >= threshold)[0]
    # Always include the global max
    max_idx = np.argmax(aa)
    if max_idx not in active:
        active = np.concatenate([active, [max_idx]])
    return active

def smooth_gradient_step(a, tau):
    """Gradient of smooth max approximation."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    return grad


def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0):
    """Smooth gradient descent with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = 0.5 * (0.02 / 0.5) ** frac
        
        grad = smooth_gradient_step(a, tau)
        v = 0.9 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP mirror descent.
Key improvements over previous version:
- Peak-targeting LP: directly minimizes excess correlation at active peaks
- Adaptive tight-M strategy: LP constraints use M*(1-eps) to force peak reduction
- Multiple LP directions per iteration (standard tight, standard normal, targeted)
- Better line search accuracy (80 ternary iterations)

import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp_standard(a, M, row_idx, col_idx, time_limit=10):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)

            g = np.maximum(r.x, 0)

        pass
    return None

def line_search_analytic(a, g, n_iters=80):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(n_iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))

            break
        if verbose and it % 100 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={best_t:.6f} eps={eps:.5f}", flush=True)
    return best_a


def smooth_gradient_step(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad


def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = 0.5 * (0.02 / 0.5) ** frac
        grad = smooth_gradient_step(a, tau)
        v = 0.9 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    n0 = parent.size
    row_idx, col_idx = build_lp_matrix(n0)

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Enhanced LP Mirror Descent n={n0}", flush=True)
    print(f"{'='*60}", flush=True)

    a = parent.copy()

    phase1_dl = t_start + budget_s * 0.40

                                      verbose=True, use_targeted=True, tight_eps=0.01)

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Smooth Gradient + LP Polish", flush=True)

    if tracker.best is not None:
        n_cur = tracker.best.size
        if n_cur == n0:
            r_idx2, c_idx2 = row_idx, col_idx
        else:
            r_idx2, c_idx2 = build_lp_matrix(n_cur)

        a_sg = tracker.best.copy()
        a_sg = a_sg / a_sg.sum()
        a_sg = run_smooth_gradient(a_sg, 3000, t_start + budget_s * 0.50,
                                    initial_tau=10.0, final_tau=300.0)
        tracker.update(a_sg, "(P2 smooth)")

        a_lp = a_sg.copy()
        a_lp = a_lp / a_lp.sum()
        a_lp = run_lp_mirror_descent_v2(a_lp, 10000, t_start + budget_s * 0.55,
                                         r_idx2, c_idx2, verbose=False,
                                         use_targeted=True, tight_eps=0.005)
        tracker.update(a_lp, "(P2 LP polish)")

    # Phase 3: Multi-length exploration
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.5, 0.625, 0.75, 1.25, 1.5, 2.0, 3.0]:
            n_trial = int(base_n * mult)
            if 32 <= n_trial <= 4096 and n_trial != base_n:

        for p in [499, 503, 509, 521, 541, 547, 557, 563, 569, 571, 577, 587, 593, 599,

                trial_lengths.append(p)
        trial_lengths = sorted(set(trial_lengths))
        trial_lengths = trial_lengths[:12]

        phase3_dl = t_start + budget_s * 0.72
        for n_trial in trial_lengths:
            if time.monotonic() > phase3_dl:

            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()
            r_idx, c_idx = build_lp_matrix(n_trial)

                                              tight_eps=0.005)

    print(f"PHASE 4: Perturbation Restarts", flush=True)

            a = a / a.sum()
            sub_dl = min(time.monotonic() + 10, phase4_dl)

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Extended LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)

                                          verbose=True, use_targeted=True,
                                          tight_eps=0.002)
        tracker.update(a_opt, "(P5 final)")

    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
