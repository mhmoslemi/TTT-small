import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coord_update_fast(a, i, b):
    """
    Fast coordinate update using precomputed convolution.
    Returns optimal value for a[i] given all others fixed.
    """
    n = len(a)
    x_cur = a[i]
    S = a.sum()
    
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0
    
    affected_arange = np.arange(k_lo, k_hi + 1)
    c_vals = np.zeros(len(affected_arange))
    slopes = np.zeros(len(affected_arange))
    quad_idx = -1
    
    for idx, k in enumerate(affected_arange):
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            slopes[idx] = 0.0
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2 * x_cur * a[j]
                slopes[idx] = 2 * a[j]
            else:
                c_vals[idx] = b[k]
                slopes[idx] = 0.0
    
    def obj(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 0:
            return 1e10
        vals = c_vals + slopes * x_test
        m = base_max
        for idx2 in range(len(vals)):
            v = vals[idx2]
            if v > m:
                m = v
        if quad_idx >= 0:
            qv = c_vals[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)
    
    lo = 0.0
    hi = max(x_cur * 3.0, np.sqrt(max(b.max(), 0)) + 0.5, 0.1)
    
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(60):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def coord_update_vec(a, i, b):

    n = len(a)
    x_cur = a[i]

        base_max = 0.0
    
    affected_arange = np.arange(k_lo, k_hi + 1)
    k_eq_2i = (affected_arange == 2 * i)

    # c_k = b[k] - 2*x_cur*a[k-i] (when k != 2i)
    # slope_k = 2*a[k-i] (when 0 <= k-i < n and k != 2i)
    
    j_indices = affected_arange - i  # k - i for each affected k
    valid_j = (j_indices >= 0) & (j_indices < n) & ~k_eq_2i
    
    # For quadratic term at k=2i: b[2i] - x_cur^2 + x^2
    
    c_vals = np.where(k_eq_2i, b[affected_arange] - x_cur**2, b[affected_arange])
    slopes = np.zeros(len(affected_arange))
    if np.any(valid_j):
        slopes[valid_j] = 2 * a[j_indices[valid_j]]
        c_vals[valid_j] = b[affected_arange[valid_j]] - 2 * x_cur * a[j_indices[valid_j]]
    
    # Handle the quadratic term separately
    quad_mask = k_eq_2i
    quad_pos = np.where(quad_mask)[0]
    quad_c = b[affected_arange[quad_mask]].item() - x_cur**2 if np.any(quad_mask) else 0.0
    
    lo = 0.0
    hi = max(x_cur * 3.0, np.sqrt(max(b.max(), 0)) + 0.5, 0.1)
    
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    
    def obj_v(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 0:
            return 1e10

        if np.any(quad_mask):

            if qv > m:

    fa = obj_v(a_pt)
    fb = obj_v(b_pt)

            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_v(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj_v(0.0) < obj_v(x_opt):
        x_opt = 0.0

def run_coord_descent(a, max_sweeps, deadline, shuffle=True, seed=0):
    """Run coordinate descent sweeps."""
    rng_c = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        if shuffle:
            rng_c.shuffle(order)
        
        b = fast_conv(a)
        sweep_improved = False
        
        for i in order:
            x_opt = coord_update_vec(a, i, b)
            if abs(x_opt - a[i]) > 1e-10:
                a[i] = x_opt
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 3
        if no_improve > 20:
            break
    
    return best_a

def run_block_cd(a, max_sweeps, deadline, block_size=2, seed=0):

    if s < 1e-10:

            block = blocks[bi]
            if len(block) == 1:
                x_opt = coord_update_vec(a, block[0], b)
                if abs(x_opt - a[block[0]]) > 1e-10:
                    a[block[0]] = x_opt

            elif len(block) == 2:
                i, j = block
                # Alternating coordinate updates within the block
                for _ in range(5):
                    b = fast_conv(a)
                    ni = coord_update_vec(a, i, b)
                    a[i] = ni
                    b = fast_conv(a)
                    nj = coord_update_vec(a, j, b)
                    a[j] = nj
                improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not improved:
            no_improve += 3
        if no_improve > 15:
            break
    
    return best_a


def fourier_subspace_optimize(a, deadline, n_iters=200, subspace_dim=16, seed=0):
    """
    Optimize in random Fourier subspaces.
    At each step, pick a random subspace of Fourier modes, project the gradient,
    and take a step.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    # Precompute FFT of a
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute smooth max gradient
        b = fast_conv(a)
        b_max = b.max()
        tau = 30.0
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
        
        # Project to sum-zero (tangent to simplex)
        grad = grad - grad.mean()
        
        # Pick random Fourier subspace
        freqs = rng.integers(1, n // 2, size=min(subspace_dim, n // 2 - 1))
        # Build subspace basis vectors
        sub_basis = np.zeros((len(freqs), n))
        for idx, f in enumerate(freqs):
            sub_basis[idx, :] = np.cos(2 * np.pi * f * np.arange(n) / n)
            sub_basis[idx, :] = sub_basis[idx, :] - sub_basis[idx, :].mean()  # center
        
        # Project gradient onto subspace
        # v = sub_basis^T * grad, then direction = sub_basis * v
        coeffs = sub_basis @ grad
        direction = sub_basis.T @ coeffs
        direction = direction - direction.mean()  # re-center
        
        # Normalize direction
        d_norm = np.linalg.norm(direction)
        if d_norm < 1e-12:
            continue
        direction /= d_norm
        
        # Line search
        best_t = 0
        best_M_trial = best_M
        for t in [0.05, 0.03, 0.01, 0.005, 0.001, -0.01, -0.005]:
            a_trial = a + t * direction
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_t = t
        
        if best_t != 0:
            a = a + best_t * direction
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 30:
            break
    
    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=10):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_cycles(a, max_iters, deadline, time_limit=5):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Slightly relax M to help LP find better directions
        M_relaxed = M * 1.001
        g = lp_best_response(a, M_relaxed, time_limit=time_limit)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Adaptive line search
        improved = False
        base_t = max(0.1, 0.5 * (1 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, base_t * 0.1, 0.05, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
    
    return best_a


def resample_seq(src, dest_n):
    """Resample via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.2 * rng.normal(0, 1, n)), 0.01))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))
    
    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))
    
    # Gaussian
    inits.append(np.exp(-3 * (t - 0.5) ** 2) + 0.1)
    
    return inits

def smooth_gradient(a, tau=50.0):
    """Compute gradient of smooth max of convolution."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    return grad


def run_adam(a, max_iters, deadline, tau_start=20.0, tau_end=200.0, lr=0.02):
    """Adam on smooth max."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        lr_t = lr / (1 + 0.005 * it)
        
        grad = smooth_gradient(a, tau)
        grad = grad - grad.mean()
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
    
    return best_a


def perturbation_escape(a, rng, n_perturbations=10, strength=0.05):
    """
    Try to escape local minima by perturbing and doing quick local search.
    Returns best found.
    """
    n = len(a)
    best = a.copy()
    best_M = fast_conv(a).max()
    
    for p in range(n_perturbations):
        # Random perturbation
        noise = rng.normal(0, strength, n)
        a_pert = np.maximum(a + noise, 0.0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s
        
        # Quick coordinate descent
        a_opt = run_coord_descent(a_pert, 5, time.monotonic() + 5, seed=p)
        M = fast_conv(a_opt).max()
        if M < best_M:
            best = a_opt.copy()
            best_M = M
    
    return best

So the peak is at k=n-1, and its value is sum_i a[i]*a[n-1-i], which is the inner product of a with its reverse. By Cauchy-Schwarz, this is at most sqrt(sum a_i^2 * sum a_i^2) = sum a_i^2, and at least... well, it can be small if a is "skewed" (e.g., concentrated at one end).

- If a is concentrated at index 0: conv(a,a)[n-1] = 0 (for large n)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coord_update_vec(a, i, b, S):
    """Vectorized coordinate update. b and S precomputed."""
    n = len(a)
    x_cur = a[i]

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0

    affected_arange = np.arange(k_lo, k_hi + 1)
    j_indices = affected_arange - i
    is_quad = (affected_arange == 2 * i)
    valid_linear = (j_indices >= 0) & (j_indices < n) & ~is_quad

    c_vals = b[affected_arange].copy()
    slopes = np.zeros(len(affected_arange))

    if np.any(valid_linear):
        jj = j_indices[valid_linear]
        slopes[valid_linear] = 2.0 * a[jj]
        c_vals[valid_linear] = b[affected_arange[valid_linear]] - 2.0 * x_cur * a[jj]

    if np.any(is_quad):
        quad_c = b[affected_arange[is_quad]].item() - x_cur * x_cur
    else:
        quad_c = None

    lo = 0.0
    hi = max(x_cur * 3.0, 0.5)

    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)

    def obj_fn(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 0:
            return 1e10
        vals = c_vals + slopes * x_test
        m = base_max
        vm = vals.max()
        if vm > m:
            m = vm
        if quad_c is not None:
            qv = quad_c + x_test * x_test
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)

    fa = obj_fn(a_pt)
    fb = obj_fn(b_pt)

    for _ in range(55):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_fn(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_fn(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_fn(0.0) < obj_fn(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coord_descent(a, max_sweeps, deadline, seed=0):
    """Fast coordinate descent with precomputed sums."""
    rng_c = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_c.shuffle(order)

        b = fast_conv(a)
        S = a.sum()
        improved = False

        for i in order:
            x_opt = coord_update_vec(a, i, b, S)
            if abs(x_opt - a[i]) > 1e-12:
                S = S - a[i] + x_opt
                a[i] = x_opt
                improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not improved:
            no_improve += 5
        if no_improve > 20:
            break

    return best_a


def smooth_gradient(a, tau=50.0):
    """Gradient of smooth max (log-sum-exp) of convolution values."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad


def run_nesterov_smooth(a, max_iters, deadline, tau_start=20.0, tau_end=300.0, lr=0.015):
    """Nesterov-accelerated gradient descent on smooth max."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    v = a.copy()  # Nesterov momentum variable
    gamma = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.6)
        lr_t = lr / (1 + 0.003 * it)

        # Nesterov lookahead
        v_new = a + (gamma - 1.0) / (1.0 + gamma) * (a - v)
        gamma = 1.0 + 2.0 / np.sqrt(it + 6.0)

        grad = smooth_gradient(v_new, tau)
        grad = grad - grad.mean()

        step = -lr_t * grad
        a_new = a + step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        v = a_new
        a = a_new

        M = fast_conv(a).max()
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()

    return best_a


def fourier_subspace_optimize(a, deadline, n_iters=300, seed=0):
    """Optimize using random Fourier subspace directions with adaptive step."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        tau = 40.0
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

        grad = grad - grad.mean()

        # Random Fourier direction
        n_modes = rng.integers(4, min(32, n // 4))
        freqs = rng.choice(np.arange(1, n // 2), size=n_modes, replace=False)
        coeffs = rng.normal(0, 1, n_modes)

        direction = np.zeros(n)
        for c, f in zip(coeffs, freqs):
            direction += c * np.cos(2.0 * np.pi * f * np.arange(n) / n)
        direction = direction - direction.mean()

        # Also add sine components
        direction2 = np.zeros(n)
        for c, f in zip(coeffs, freqs):
            direction2 += c * np.sin(2.0 * np.pi * f * np.arange(n) / n)
        direction2 = direction2 - direction2.mean()

        d_dir = direction + direction2
        d_dir = d_dir - d_dir.mean()
        d_norm = np.linalg.norm(d_dir)
        if d_norm < 1e-12:
            continue
        d_dir /= d_norm

        # Align with or against gradient
        dot = d_dir @ grad
        if dot > 0:
            d_dir = -d_dir

        # Adaptive line search
        base_t = 0.03 / (1 + 0.01 * it)
        best_t = 0.0
        best_M_t = best_M

        for t in [base_t, base_t * 2, base_t * 0.5, base_t * 4, base_t * 0.25]:
            a_trial = a + t * d_dir
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            M_t = fast_conv(a_trial).max()
            if M_t < best_M_t:
                best_M_t = M_t
                best_t = t

        if best_t != 0.0:
            a = a + best_t * d_dir
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        M = fast_conv(a).max()
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 40:
            break

    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix for LP constraints."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=8):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_cycles(a, max_iters, deadline, time_limit=5):
    """LP mirror descent with refined line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()

    for it in range(max_iters):

            break

        b = fast_conv(a)
        M = float(b.max())
        M_relaxed = M * 1.0005

        g = lp_best_response(a, M_relaxed, time_limit=time_limit)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        improved = False
        base_t = max(0.05, 0.4 * (1 - 0.4 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 2.0, base_t * 0.25, 0.03, 0.01]:

def structured_perturb_escape(a, rng, deadline, base_strength=0.04):

            break

        if att < n_attempts // 2:
            # Sinusoidal perturbation

            phase = rng.uniform(0, 2 * np.pi)
            strength = base_strength * (0.5 + rng.random())
            noise = strength * np.sin(2 * np.pi * freq * t_scale + phase)
        else:
            # Random smooth perturbation
            ksize = max(5, n // 20)
            raw = rng.normal(0, 1, n)
            kernel = np.ones(ksize) / ksize
            noise = convolve_smooth(raw, kernel) * base_strength * (0.5 + rng.random())

        a_pert = a + noise
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-10:
            continue

        M = fast_conv(a_opt).max()
        if M < best_M - 1e-14:

def convolve_smooth(x, kernel):
    """Fast convolution with given kernel (same mode)."""
    n = len(x)
    size = 1
    while size < n + len(kernel) - 1:
        size <<= 1
    fx = np.fft.rfft(x, size)
    fk = np.fft.rfft(kernel, size)
    result = np.fft.irfft(fx * fk, size)
    offset = len(kernel) // 2
    return result[offset:offset + n]


def resample_seq(src, dest_n):
    """Resample via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
