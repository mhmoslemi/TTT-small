import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    return result


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx, timeout=8):
    """Solve LP: max 1^T g s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': timeout, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max of ((1-t)*a + t*g)^2 convolution."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = max(0.0, min(1.0, (lo + hi) / 2.0))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_J(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = float(fast_conv(a).max())
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        
        t, M_new = line_search_analytic(a, g)
        
        if t < 1e-8:
            no_improve += 1
        else:
            a_new = np.maximum((1.0 - t) * a + t * g, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:

                J_new = compute_J(a_new)
                if J_new < best_M - 1e-12:
                    a = a_new
                    best_M = J_new
                    best_a = a.copy()
                    no_improve = 0
                else:

            else:

def run_adam_smooth(a, max_iters, deadline, tau_start=3.0, tau_end=200.0, lr_start=0.05, lr_end=0.001):
    """Adam optimization on smoothed max-convolution."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        grad = softmax_gradient(a, tau)
        
        # Center the gradient to handle sum(a)=1 constraint
        grad = grad - grad.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    
    return best_a

    """
    Exact 1D coordinate descent on the max-convolution objective.
    For each coordinate i, find the optimal x that minimizes max_k (a*a)[k]
    when a[i] = x (with renormalization to sum=1).
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        B = fast_conv(a)  # Current convolution
        improved = False
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            old_val = a[i]
            
            # Compute C[k] = B[k] with a[i] removed
            # C[k] = B[k] - 2*old_val*a[k-i] (if valid, k!=2i) - old_val^2 (if k==2i)
            C = B.copy()
            k_min = max(0, i)  # k-i >= 0 => k >= i
            k_max = min(2*n-2, i + n - 1)  # k-i <= n-1 => k <= i+n-1
            
            if k_max >= k_min:
                # For k in [k_min, k_max], k-i is in [0, n-1]
                if 2*i <= 2*n-2 and 2*i >= k_min:
                    # k = 2i is in range
                    C[2*i] -= old_val * old_val
                    if k_min < 2*i:
                        C[k_min:2*i] -= 2 * old_val * a[0:k_min-i+1][::-1][:2*i-k_min]
                    if 2*i < k_max:
                        C[2*i+1:k_max+1] -= 2 * old_val * a[1:k_max-i+1]
                else:
                    C[k_min:k_max+1] -= 2 * old_val * a[0:k_max-k_min+1]
            
            # Now minimize f(x) = max_k (C[k] + 2*x*a[k-i] + x^2*[k==2i]) over x >= 0
            # Subject to: after setting a[i]=x, renormalize to sum=1
            # S(x) = (sum(a) - old_val + x) / (sum(a) - old_val + x) = 1 (after normalization)
            # Actually if we maintain sum=1: S_other = 1 - old_val, S(x) = S_other + x
            # After normalization: a_new[j] = a[j]/(S_other + x) for j!=i, a_new[i] = x/(S_other + x)
            # The normalized convolution: (a_new * a_new)[k] = conv_unnorm[k] / (S_other + x)^2
            # where conv_unnorm[k] = C[k] + 2*x*a[k-i] + x^2*[k==2i]
            # So max_k normalized = max_k conv_unnorm[k] / (S_other + x)^2
            # J = 2*n * max_k conv_unnorm[k] / (S_other + x)^2
            # (the (sum)^2 in denominator becomes (S_other+x)^2)
            
            S_other = 1.0 - old_val  # sum of all other coordinates
            
            # Define f(x) = 2*n * max_k g_k(x) / (S_other + x)^2
            # where g_k(x) = C[k] + 2*x*a[k-i] + x^2 (if k==2i)
            # This is convex in x (for x >= 0) since each g_k is convex quadratic and
            # the ratio of convex to increasing quadratic... hmm not obviously convex.
            # But in practice it should be unimodal. Use ternary search.
            
            # Search range: [0, 3*old_val + 0.1] (cap at reasonable value)
            x_lo = 0.0
            x_hi = max(3.0 * old_val + 0.05, 0.01)
            
            # Ternary search
            for _ in range(45):
                m1 = x_lo + (x_hi - x_lo) / 3.0
                m2 = x_hi - (x_hi - x_lo) / 3.0
                
                # Evaluate f at m1 and m2
                denom1 = S_other + m1
                denom2 = S_other + m2
                
                if denom1 <= 1e-15:
                    f1 = float('inf')
                else:
                    # g_k(m1) = C[k] + 2*m1*a[k-i] + m1^2 (if k==2i)
                    g1 = C.copy()
                    if k_max >= k_min:
                        g1[k_min:k_max+1] += 2 * m1 * a[0:k_max-k_min+1]
                    if 2*i < 2*n and 2*i >= k_min and 2*i <= k_max:
                        # Need to be careful: we already added 2*m1*a[2i-i]=2*m1*a[i]
                        # But actually the formula is g_k(x) = C[k] + 2*x*a[k-i] (valid) + x^2 (if k==2i)
                        # When k==2i: g_k(x) = C[2i] + 2*x*a[i] + x^2
                        # We already added 2*m1*a[i] to g1[2i] (since 2i is in [k_min, k_max])
                        # So we need to add m1^2 to g1[2i]
                        g1[2*i] += m1 * m1
                    f1 = 2 * n * g1.max() / (denom1 * denom1)
                
                if denom2 <= 1e-15:
                    f2 = float('inf')
                else:
                    g2 = C.copy()
                    if k_max >= k_min:
                        g2[k_min:k_max+1] += 2 * m2 * a[0:k_max-k_min+1]
                    if 2*i < 2*n and 2*i >= k_min and 2*i <= k_max:
                        g2[2*i] += m2 * m2
                    f2 = 2 * n * g2.max() / (denom2 * denom2)
                
                if f1 < f2:
                    x_hi = m2
                else:
                    x_lo = m1
            
            x_opt = (x_lo + x_hi) / 2.0
            x_opt = max(0.0, min(x_opt, x_hi))
            
            # Update
            a[i] = x_opt
            # Renormalize
            s_new = a.sum()
            if s_new > 1e-10:
                a = a / s_new
            
            # Update B incrementally
            # Actually, let's just mark for recomputation
            improved = True
        
        # Record
        J = compute_J(a)
        if J < best_J - 1e-14:
            best_J = J
            best_a = a.copy()
        
        if verbose and sweep % 10 == 0:
            print(f"      CD sweep {sweep}: J={J:.10f} best={best_J:.10f}", flush=True)
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
