"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using LP mirror descent with
adaptive active set, subgradient refinement, frequency-domain initialization,
and multi-scale exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def fast_cross_conv(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_lp_matrix(n, a, constraint_indices):
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))

def solve_lp(a, M, constraint_indices):
    n = len(a)
    A = build_lp_matrix(n, a, constraint_indices)
    c = -np.ones(n)
    b_ub = np.full(len(constraint_indices), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 3, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None

def smooth_max_grad(a, tau):
    """Gradient of smooth max of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Clip to avoid overflow
    scaled = tau * (b - b_max)
    scaled = np.clip(scaled, -100, 100)
    exp_b = np.exp(scaled)
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[j] += 2 * p[k] * a[k-j] for j in [lo, hi]
        indices = np.arange(lo, hi + 1)
        a_vals = a[lo:k - lo + 1][::-1]  # a[k-lo], a[k-lo-1], ..., a[k-hi] = a[0]
        grad[indices] += 2.0 * pk * a_vals
    return grad, b, b_max

def subgradient_step(a, step_size):
    """One subgradient step on the max of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Active set: where b is within 1% of max
    active = b >= b_max * 0.999
    
    # Subgradient
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        indices = np.arange(lo, hi + 1)
        a_vals = a[lo:k - lo + 1][::-1]
        grad[indices] += 2.0 * a_vals
    
    # Normalize gradient
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return a.copy()
    
    # Projected subgradient step (project onto sum(a)=1, a>=0)
    a_new = a - step_size * grad / (n * grad_norm)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def freq_domain_init(n, rng):
    """Generate initial sequence with flat-ish power spectrum."""
    # Random phases, flat magnitude
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    freqs = np.fft.rfftfreq(size)
    # Flat magnitude with some structure
    mag = np.ones(n)
    phases = rng.uniform(0, 2 * np.pi, n)
    spectrum = mag * np.exp(1j * phases)
    # Inverse FFT
    a = np.fft.irfft(spectrum, size)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def smooth_gradient_descent(a, max_iters, deadline, tau=20.0, lr=0.05, momentum=0.9):
    """Smooth gradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, b, _ = smooth_max_grad(a, tau)
        M = b.max()
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
        
        # Momentum
        v = momentum * v + grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm
        
        # Adaptive step size
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        J_curr = compute_J(a)
        if J_new < J_curr:
            a = a_new
        else:
            lr *= 0.8
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def make_palindrome(a):
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new

def lp_mirror_descent(a, deadline, verbose=False, epsilon=0.001):
    """
    LP mirror descent with adaptive active set.
    epsilon: fraction of max to use as threshold for "active" constraints.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    max_no_improve = 100
    it = 0
    step_scale = 1.0
    
    while time.monotonic() < deadline:
        it += 1
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive active set: constraints within (1-epsilon) of max
        threshold = M * (1.0 - epsilon)
        active_k = np.where(b >= threshold)[0]
        
        # Also always include the exact max
        max_k = np.argmax(b)
        if max_k not in active_k:
            active_k = np.append(active_k, max_k)
        
        # If too many constraints, use top fraction
        if len(active_k) > 2 * n:
            # Use top n constraints
            top_n = len(active_k) // 2
            order = np.argsort(b[active_k])[::-1][:top_n]
            active_k = active_k[order]
        
        # Solve LP
        g = solve_lp(a, M, active_k)
        
        if g is None or g.sum() < 1e-10:
            # Fall back to subgradient step
            a = subgradient_step(a, 0.01 / it)
            continue
        
        g = g / g.sum()
        
        # Line search
        current_J = compute_J(a)
        improved = False
        base_t = min(0.5, max(0.05, (1.0 - step_scale) * 0.1))
        t_candidates = [0.3, 0.15, 0.07, 0.03, 0.01, 0.005]
        
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            
            if J_trial < best_J - 1e-12:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                step_scale = min(1.0, step_scale * 1.1)
                break
            elif J_trial < current_J - 1e-12:
                a = a_trial
                current_J = J_trial
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
            step_scale = max(0.1, step_scale * 0.9)
        
        # Periodic perturbation
        if no_improve > max_no_improve:
            rng = np.random.default_rng(hash((it, no_improve)) % (2**32))
            scale = 0.1 / (1.0 + no_improve / max_no_improve)
            
            # Try different perturbation types
            pert_type = it % 3
            if pert_type == 0:
                a = a + scale * rng.normal(0, 1, n)
            elif pert_type == 1:
                # Frequency domain perturbation
                size = 1
                while size < 2 * n - 1:
                    size <<= 1
                spectrum = np.fft.rfft(a, size)
                spectrum *= (1.0 + scale * rng.normal(0, 0.5, len(spectrum)))
                a = np.fft.irfft(spectrum, size)[:n]
            else:
                # Peak-directed perturbation
                peaks = np.where(b >= M * 0.99)[0]
                for k in peaks[:5]:
                    lo = max(0, k - n + 1)
                    hi = min(n - 1, k)
                    # Reduce a[j] where j contributes to peak k
                    for j in range(lo, hi + 1):
                        a[j] -= scale * 0.01 * a[k - j]
            
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            no_improve = 0
        
        # Subgradient refinement when LP is not improving
        if no_improve > max_no_improve // 2:
            for _ in range(3):
                a = subgradient_step(a, 0.005)
            s = a.sum()
            if s > 1e-10:
                a = a / s
                J_sub = compute_J(a)
                if J_sub < best_J:
                    best_J = J_sub
                    best_a = a.copy()
        
        if verbose and it % 1000 == 0:
            print(f"      iter={it} J={best_J:.8f} M={M:.6f} "
                  f"active={len(active_k)} no_improve={no_improve}", flush=True)
    
    return best_a, best_J

Actually, a better approach: use the fact that the problem is to minimize a piecewise-linear function (max of linear functions in the "dual" sense). We can use a cutting-plane method:

1. Start with a small set of constraints
2. At each iteration, evaluate the current point and add the most violated constraint
3. Solve the resulting convex program (which is an LP after the max is "cut")

Wait, the objective is max_k c[k] where c[k] is quadratic. So it's max of quadratic functions, which is convex. We can use a cutting-plane method where each cut is a tangent to one of the quadratic functions.

The tangent to c[k] at point a is: c[k](a) + ∇c[k](a) · (a' - a) = c[k](a) + 2a[k-n+1:k+1] · (a' - a) (approximately).

So the cutting-plane method would be:
1. Start with a few cuts
2. Solve the LP: min t subject to t >= c[k_i](a) + ∇c[k_i](a) · (a - a_i) for all cuts i, sum(a) = 1, a >= 0
3. Evaluate the true objective at the solution
4. Add the most violated cut
5. Repeat

This is a valid approach for convex problems! And since max_k c[k] is convex (max of convex functions), the cutting-plane method will converge.

But wait, is max_k c[k] actually convex? c[k] = a^T Q_k a where Q_k is the matrix for the k-th convolution slice. For c[k] to be convex, Q_k must be PSD.

Let me check: Q_k has entries (Q_k)_{i,j} = 1/2 if i+j = k (or i+j = k+2n-2 for the reflected part... actually it's simpler than that).

For the full convolution c[k] = sum_i a[i] * a[k-i] (with a defined as 0 outside [0,n-1]):
c[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i] * a[k-i]

As a quadratic form: c[k] = a^T Q_k a where (Q_k)_{i,j} = 1 if i+j=k or j+i=k (i.e., the anti-diagonal at position k), and 0 otherwise. But we need to be careful about the factor of 2.

Actually, a^T Q a = sum_{i,j} a[i] Q[i,j] a[j].
For c[k] = sum_i a[i] a[k-i], we need Q[i,j] = 1 if j = k-i, i.e., i+j = k.
But then a^T Q a = sum_{i+j=k} a[i] a[j] which is exactly c[k].

So Q_k has 1s on the anti-diagonal i+j=k. This matrix is a permutation matrix (shifted), which has eigenvalues on the unit circle. It's NOT positive semidefinite unless n=1.

So c[k] is NOT convex in general! The max of non-convex functions is not necessarily convex.

This means the cutting-plane approach doesn't directly apply. The problem is genuinely non-convex.
