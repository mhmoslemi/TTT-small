def compute_conv(a):
    """ Compute linear self-convolution for full mode, returns array of length 2*n-1 """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    """ Compute objective: 2*n * max(convolve(a,a,mode='full')) / (sum(a))^2 """
    if not isinstance(a, (list, np.ndarray)):
        return np.inf
    try:
        a_arr = np.asarray(a)
    except:
        a_arr = np.array(a)
    if not np.issubdtype(a_arr.dtype, np.floating):
        a_arr = np.array(a_arr, dtype=np.float64)
    if not np.isfinite(a_arr).all():
        return np.inf
    n = len(a_arr)
    if n == 0:
        return np.inf
    conv = compute_conv(a_arr)
    max_val = float(np.max(conv))
    sum_a = np.sum(a_arr)
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_val / (sum_a ** 2)

def project_simplex(v):
    """ Project a vector to the simplex (sum=1, nonnegative) """
    n = len(v)
    if n == 0:
        return v
    s = np.sum(v)
    if np.abs(s - 1) < 1e-10:
        return v
    # According to Moreau decomposition
    temp = []
    for i in range(n):
        candidate = v[i] - (s - 1) / n
        if candidate < 0:
            candidate = 0
        temp.append(candidate)
    s_new = np.sum(temp)
    if s_new == 0:
        return np.ones(n) / n
    return temp

def gradient_descent_step(a, time_budget=1000.0):
    """ Optimize a sequence of fixed length using gradient descent and FFT convolution """
    n = len(a)

    eta = eta_max
    iter_count = 0
    improvement_made = True
    
    while time.monotonic() < time_limit - 1 and iter_count < 3000:
        iter_count += 1
        # Compute gradient using Moreau's step projection method
        conv_a = compute_conv(a_old)
        max_val_current = np.max(conv_a) if len(conv_a) > 0 else 0.0
        
        # Gradient calculation: use the subgradient of the max function
        grad_val = np.zeros(n)
        for i in range(n):
            # Collect positive contributions from all indices
            contrib = 0.0
            for k in range(2*n-2):
                if k - i >= 0 and k - i < n:  # Check if the term is part of the convolution
                    contrib += conv_a[k] * a_old[k-i] * (compute_obj(a_old) < 1.0) + (compute_obj(a_old) >= 1.0)*0
                else:
                    # Handle filter boundaries - no contribution
                    pass
            grad_val[i] = contrib * eta
            
        # Update step
        a_new_task = a_old + grad_val
        
        # Project to simplex to maintain norm
        a_new = project_simplex(a_new_task)
        
        # Check if improved
        obj_new = compute_obj(a_new)
        if isinstance(obj_new, float) and obj_new < best_value:
            improvement_made = True
            best_sequence = a_new.copy()
            best_value = obj_new
        
        # Decay learning rate if no improvement or stable iterations
        if not improvement_made:
            eta = eta * eta_decay
        
        # Periodically revert direction to avoid getting stuck
        current_time = time.monotonic()

def aggressive_optimization(a, time_budget=1000.0):
