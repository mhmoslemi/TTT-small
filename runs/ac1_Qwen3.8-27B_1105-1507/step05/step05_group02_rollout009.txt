"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov smooth gradient descent (log-sum-exp)
2. LP mirror descent with line search (primary local optimizer)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_and_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(2 * n - 1)  # We'll compute the gradient of the smooth max w.r.t. a
    # Actually, the gradient of sum_k p_k * b_k w.r.t. a_i is:
    # sum_k p_k * db_k/da_i = sum_k p_k * (a_{k-i} + a_{k-i} if k-i != ...) 
    # More carefully: b_k = sum_j a_j * a_{k-j}
    # db_k/da_i = a_{k-i} (if 0 <= k-i <= n-1)
    # So grad_i = sum_k p_k * a_{k-i} (summed over valid k)
    # Which is: (p * a_reversed) convolved... 
    # Actually: grad_i = sum_k p_k * a_{k-i} for k where 0 <= k-i <= n-1
    # = sum_{k=i}^{i+n-1} p_k * a_{k-i}
    # This is a cross-correlation of p with a, evaluated at offset i
    # Let me just compute it directly for clarity
    grad_a = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad_a[lo:hi + 1] += p[k] * a[k - hi:k - lo + 1][::-1]
    return 2 * grad_a, p, b, b_max

def run_smooth_nesterov(a, deadline, max_iters=5000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.4, lr_end=0.003):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        beta = 0.9

        grad, p, b, b_max = smooth_max_and_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        momentum = y - a
        a_new = np.maximum(y - lr * (grad + beta * momentum), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a

def build_A_sparse(n, a):
    """Build sparse matrix A such that A @ g = convolve(a, g)"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0"""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def run_lp_mirror(a, deadline, max_iters=500, line_search=True):
    """LP Mirror Descent with line search"""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP
        t0 = time.monotonic()
        g = lp_best_response(a, M)
        lp_time = time.monotonic() - t0
        
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Line search for optimal mixing
        improved = False
        if line_search:
            # Try a range of t values
            best_t = None
            best_M_t = M
            for t in np.linspace(0.01, 0.5, 20):
                a_t = (1.0 - t) * a + t * g
                s = a_t.sum()
                if s < 1e-10:
                    continue
                a_t = a_t / s
                M_t = fast_conv(a_t).max()
                if M_t < best_M_t:
                    best_M_t = M_t
                    best_t = t
            if best_t is not None:
                a = (1.0 - best_t) * a + best_t * g
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a /= s
                M = fast_conv(a).max()
                if M < best_M - 1e-14:
                    best_M = M
                    best_a = a.copy()
                    improved = True
        else:
            for t in [0.3, 0.2, 0.1, 0.05]:
                a_t = (1.0 - t) * a + t * g
                s = a_t.sum()
                if s < 1e-10:
                    continue
                a_t /= s
                M_t = fast_conv(a_t).max()
                if M_t < best_M - 1e-14:
                    a = a_t
                    best_M = M_t
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 15:
                break

    return best_a

def spectral_relaxation(a, deadline, max_iters=200, alpha_init=0.5, alpha_end=0.05):
    """Spectral relaxation: try to flatten the power spectrum"""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Target: flat power spectrum
    # |F(a)|^2 should be as flat as possible
    # The flat target has value S^2/(2n-1) at each frequency bin
    # (since sum of |F|^2 = n * sum a^2 by Parseval, and sum of autocorrelation = S^2)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        alpha = alpha_init * (alpha_end / alpha_init) ** frac

        # Compute FFT (full, for the autocorrelation)
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        F = np.fft.fft(a, size)
        power = np.abs(F) ** 2
        
        # Target flat power
        # By Parseval: sum_k conv(a,a)_k = (sum a)^2 = S^2
        # In frequency domain: sum_j |F_j|^2 / size = S^2 (for circular)
        # For linear: sum of autocorrelation = S^2
        # Target: each power bin = S^2 / (2n-1) approximately
        S = a.sum()
        target_power = np.full(size, S * S / (2 * n - 1))
        
        # Blend: new_power = alpha * power + (1-alpha) * target_power
        # But we need to preserve the DC component
        new_power = alpha * power + (1 - alpha) * target_power
        new_power[0] = power[0]  # Preserve DC
        
        # Take square root and inverse FFT
        new_F = np.sqrt(new_power) * F / np.abs(F + 1e-30)
        a_new = np.fft.ifft(new_F, size)[:n].real
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new /= s
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
        a = a_new

    return best_a

def coordinate_descent(a, deadline, max_sweeps=10):
    """Coordinate descent: try adjusting each coordinate individually"""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(123)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        # Random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            for delta_frac in [0.01, -0.01, 0.05, -0.05, 0.005, -0.005]:
                a_new = a.copy()
                a_new[i] = max(0, a[i] * (1 + delta_frac))
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new /= s
                M_new = fast_conv(a_new).max()
                if M_new < best_M - 1e-12:
                    best_M = M_new
                    best_a = a_new.copy()
                    a = a_new
                    break
    
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov smooth gradient descent (log-sum-exp) for initialization
2. Aggressive LP Mirror Descent with adaptive line search (primary optimizer)
3. Spectral relaxation (Gerchberg-Saxton style power spectrum flattening)
4. Multi-length exploration with resampling
5. Coordinate descent for final polish
Key improvements: better line search, more LP iterations, spectral step,
adaptive time allocation, and multi-restart strategy.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad_a = np.zeros(n)
    active = np.where(p > 1e-14)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad_a[lo:hi + 1] += p[k] * a[k - hi:k - lo + 1][::-1]
    return 2.0 * grad_a, p, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=5000, tau_start=2.0, tau_end=200.0,
                        lr_init=0.35, lr_end=0.005):

    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        beta = 0.9

        grad, p, b, b_max = smooth_max_and_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        momentum = y - a
        a_new = np.maximum(y - lr * (grad + beta * momentum), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def spectral_relaxation(a, deadline, max_iters=300, alpha_init=0.6, alpha_end=0.1):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        alpha = alpha_init * (alpha_end / alpha_init) ** frac

        size = 1
        while size < 2 * n - 1:
            size <<= 1
        F = np.fft.fft(a, size)
        power = np.abs(F) ** 2

        S = a.sum()
        target_power = np.full(size, S * S / (2 * n - 1))

        new_power = alpha * power + (1.0 - alpha) * target_power
        new_power[0] = power[0]
        new_power = np.maximum(new_power, 0)

        phases = F / (np.abs(F) + 1e-30)
        new_F = np.sqrt(new_power) * phases
        a_new = np.fft.ifft(new_F, size)[:n].real
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new /= s

        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
        a = a_new

        if no_imp > 30:
            break

    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    if time.monotonic() < p2_deadline - 50:
        a_opt = run_lp_mirror(a_opt, p2_deadline, max_iters=1000, use_line_search=True)
        best.update(a_opt)
        best.report("LP pass 2")

    # === PHASE 3: Spectral Relaxation ===
    p3_deadline = time.monotonic() + 180
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Spectral Relaxation + LP", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        a_spec = spectral_relaxation(best.seq, min(time.monotonic() + 100, p3_deadline - 80),
                                     max_iters=200, alpha_init=0.5, alpha_end=0.1)
        best.update(a_spec)

        # Follow up with LP
        if time.monotonic() < p3_deadline - 30:
            a_spec = run_lp_mirror(a_spec, p3_deadline, max_iters=200, use_line_search=True)
            best.update(a_spec)

        best.report("Phase 3")

    # === PHASE 4: Multi-length exploration ===
    p4_deadline = time.monotonic() + 200
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n_best = best.seq.size
        lengths_to_try = []
        for delta in [-64, -32, -16, 16, 32, 64, 128]:
            n_try = n_best + delta
            if 64 <= n_try <= 4096:
                lengths_to_try.append(n_try)
        lengths_to_try.extend([n_best // 2, n_best * 2])
        lengths_to_try = sorted(set(l for l in lengths_to_try if 64 <= l <= 4096))

        for n_try in lengths_to_try:
            if time.monotonic() > p4_deadline - 20:
                break
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            # Quick smooth GD
            a_ext = run_smooth_nesterov(a_ext, min(time.monotonic() + 30, p4_deadline - 20),
                                        max_iters=500, tau_start=3, tau_end=80,
                                        lr_init=0.2, lr_end=0.02)
            best.update(a_ext)
            # Quick LP
            if time.monotonic() < p4_deadline - 15:
                a_ext = run_lp_mirror(a_ext, min(time.monotonic() + 40, p4_deadline - 10),
                                      max_iters=50, use_line_search=False)
                best.update(a_ext)
            print(f"  n={n_try}: J={best.score:.8f}", flush=True)

        best.report("Phase 4")

    # === PHASE 5: Perturbation + re-optimization ===
    p5_deadline = time.monotonic() + 150
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Perturbation + Re-optimization", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        for scale in [0.03, 0.05, 0.08]:
            if time.monotonic() > p5_deadline - 20:
                break
            for trial in range(3):
                if time.monotonic() > p5_deadline - 15:
                    break
                noise = rng.normal(0, scale * 0.01, n)
                a_pert = np.maximum(best.seq + noise, 1e-5)
                a_pert /= a_pert.sum()
                # Quick smooth
                a_pert = run_smooth_nesterov(a_pert, min(time.monotonic() + 20, p5_deadline - 15),
                                             max_iters=500, tau_start=5, tau_end=100,
                                             lr_init=0.2, lr_end=0.01)
                # Quick LP
                if time.monotonic() < p5_deadline - 10:
                    a_pert = run_lp_mirror(a_pert, min(time.monotonic() + 25, p5_deadline - 10),
                                           max_iters=50, use_line_search=True)
                best.update(a_pert)

        best.report("Phase 5")

    # === PHASE 6: Final intensive LP polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final Intensive LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        a_final = best.seq.copy()
        a_final = a_final / a_final.sum()

        # Multiple LP passes with remaining time
        remaining = deadline - time.monotonic()
        n_passes = 3
        per_pass = remaining / n_passes

        for p in range(n_passes):
            dl = time.monotonic() + per_pass
            a_final = run_lp_mirror(a_final, min(dl, deadline),
                                    max_iters=500, use_line_search=True)
            best.update(a_final)
            best.report(f"Final LP {p+1}")

    # === PHASE 7: Coordinate descent if time remains ===
    if best.seq is not None and time.monotonic() < deadline - 10:
        print(f"\nPHASE 7: Coordinate Descent", flush=True)
        a_cd = coordinate_descent(best.seq, deadline, max_sweeps=5, rng=rng)
        best.update(a_cd)
        best.report("Phase 7")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = 2.0

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best.seq]
