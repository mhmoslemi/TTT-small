def fast_conv(a):
    """Fast convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_M(a):
    """Compute max of linear self-convolution."""
    return float(fast_conv(a).max())

def compute_J(a):
    """Compute the objective function."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    M = compute_M(a)
    return float(2 * len(a) * M / (s * s))

def smooth_max_and_grad(a, tau):
    """Compute smoothed max of conv(a,a) and its gradient w.r.t. a."""
    b = fast_conv(a)
    b_max = b.max()
    # For numerical stability
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(len(a))
    n = len(a)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, float(b_max)

def run_smooth_nesterov(a0, deadline, max_iters=5000, tau_start=1.0, tau_end=200.0,

def lp_solve(a, M, time_limit=0.010):
    """Solve the LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    # Build sparse matrix A where (A @ g)[k] = (a * g)[k] for k = 0..2n-2
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and np.sum(r.x) > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def run_lp_mirror(a0, deadline, max_iters=500, lp_time=0.010):
    """LP mirror descent with golden-section line search."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break
        
        # Solve LP for best response
        g = lp_solve(a, M, time_limit=lp_time)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Golden section line search
        phi = (1 + np.sqrt(5)) / 2
        gr = (2 - phi) / phi  # golden ratio conjugate ≈ 0.382
        
        # Search in t ∈ [0, 1]
        ta, tb = 0.0, 1.0
        tc = ta + gr * (tb - ta)
        td = ta + (1 - gr) * (tb - ta)
        
        def obj_at_t(t):
            a_t = (1 - t) * a + t * g
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-10:
                return float('inf')
            a_t /= s
            return compute_M(a_t)
        
        fc = obj_at_t(tc)
        fd = obj_at_t(td)
        
        for _ in range(25):  # ~25 iterations of golden section
            if time.monotonic() > deadline + 2:
                break
            if fc < fd:
                tb = td
                td = tc
                fd = fc
                tc = ta + gr * (tb - ta)
                fc = obj_at_t(tc)
            else:
                ta = tc
                tc = td
                fc = fd
                td = ta + gr * (tb - ta)
                fd = obj_at_t(td)
        
        t_opt = (ta + tb) / 2
        a = (1 - t_opt) * a + t_opt * g
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
        else:
            a /= s
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov smooth gradient descent for warm-start
2. Aggressive LP mirror descent with golden-section line search and double-LP steps
3. Coordinate descent polish with incremental updates
4. Multi-start with structured perturbations and periodic restarts
Key improvements: more LP iterations, golden-section line search, double-LP fictitious play,
and coordinate descent as a final polish step.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    M = compute_M(a)
    return float(2 * len(a) * M / (s * s))


def smooth_max_and_grad(a, tau):
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for stability
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()

    grad = np.zeros(len(a))
    n = len(a)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, float(b_max)


def run_smooth_nesterov(a0, deadline, max_iters=5000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.4, lr_end=0.003):
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b_max = smooth_max_and_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def precompute_lp_structure(n):
    """Precompute row/col structure for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def lp_solve_with_structure(a, M, rows, cols, time_limit=0.020):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0 using precomputed structure."""
    n = len(a)
    # Build values: for entry (k, j), value is a[k-j]
    vals = a[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and np.sum(r.x) > 1e-15:
            return np.maximum(r.x, 0)
        elif r.x is not None and np.sum(r.x) > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def golden_section_1d(f, lo, hi, max_iters=30, tol=1e-8):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5) - 1) / 2  # ≈ 0.618
    a = lo
    b = hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)

    for _ in range(max_iters):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)

    return (a + b) / 2


def run_lp_mirror_aggressive(a0, deadline, max_iters=10000, lp_time=0.020,
                            use_double_lp=False, perturbation_interval=200):
    """Aggressive LP mirror descent with golden-section line search."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)

    # Precompute LP structure
    rows, cols = precompute_lp_structure(n)

    no_improve = 0
    iter_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        iter_count += 1

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M * (1 - 1e-14):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                # Try perturbation to escape
                if perturbation_interval > 0 and it % perturbation_interval == 0:
                    noise = np.random.default_rng(it).normal(0, 0.01, n)
                    a_pert = np.maximum(a + noise, 1e-6)
                    a_pert /= a_pert.sum()
                    M_pert = compute_M(a_pert)
                    if M_pert < best_M:
                        best_M = M_pert
                        best_a = a_pert.copy()
                        a = a_pert.copy()
                        no_improve = 0
                        continue
                break

        # Solve LP for best response
        g = lp_solve_with_structure(a, M, rows, cols, time_limit=lp_time)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Double LP: solve LP with g as the fixed vector
        if use_double_lp:
            h = lp_solve_with_structure(g, M, rows, cols, time_limit=lp_time)
            if h is not None and h.sum() > 1e-10:
                h /= h.sum()
                direction = 0.5 * g + 0.5 * h
            else:
                direction = g
        else:
            direction = g

        # Golden section line search for optimal t
        def obj_t(t):
            a_t = (1 - t) * a + t * direction
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-10:
                return float('inf')
            a_t /= s
            return compute_M(a_t)

        # First check t=1 (full step)
        f1 = obj_t(1.0)
        if f1 < M * (1.001):
            t_opt = 1.0
        else:
            # Golden section on [0, 1]
            def obj_t_inner(t):
                return obj_t(t)
            t_opt = golden_section_1d(obj_t_inner, 0.0, 1.0, max_iters=20, tol=1e-6)

        a = (1 - t_opt) * a + t_opt * direction
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
        else:
            a /= s

    return best_a


def coord_descent_polish(a0, deadline, max_passes=500):
    """Coordinate descent: for each coordinate, find optimal value via 1D search."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    rng = np.random.default_rng(123)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        # Shuffle coordinate order
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]

            # Try several multipliers
            best_local_M = compute_M(a)
            best_local_val = old_val

            for mult in [0.3, 0.5, 0.7, 1.3, 1.6, 2.0]:
                v = old_val * mult
                if v < 1e-12:
                    continue
                a[i] = v
                M_try = compute_M(a)
                if M_try < best_local_M:
                    best_local_M = M_try
                    best_local_val = v

            a[i] = best_local_val

        M_new = compute_M(a)
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.005))
    inits.append(np.ones(n))
    for alpha in [0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.2 + 0.8 * (t**alpha + (1 - t)**alpha) / 2, 0.01))
    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.05)
    inits.append(np.exp(-8*(t-0.3)**2) + np.exp(-8*(t-0.7)**2) + 0.03)
    inits.append(np.maximum(rng.uniform(0, 1, n), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(512)
        best.update(parent)
        print(f"No parent found, using ones(512), J={best.score:.6f}", flush=True)

    base_n = parent.size

    # === Phase 1: Smooth GD warm-start (250s) ===
    p1_deadline = time.monotonic() + 250.0
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Smooth GD Warm-start (250s)", flush=True)
    print(f"{'='*50}", flush=True)

    # Try the parent's length and a few others
    lengths_to_try = [base_n]
    for mult in [0.5, 1.5, 2.0]:
        n_try = int(base_n * mult)
        n_try = max(128, min(1024, (n_try // 16) * 16))
        if n_try not in lengths_to_try:
            lengths_to_try.append(n_try)

    for n_try in lengths_to_try:
        if time.monotonic() > p1_deadline - 30:
            break
        t0 = time.monotonic()
        inits = make_inits(n_try, rng, best.seq if best.seq is not None else parent)

        best_n = None
        best_M = float('inf')

        # Run smooth GD from multiple starts
        time_per_init = max(5, 200.0 / len(inits))
        for a0 in inits:
            if time.monotonic() > p1_deadline - 20:
                break
            dl = min(time.monotonic() + time_per_init, p1_deadline - 20)
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()
            a = run_smooth_nesterov(a0, dl, max_iters=3000,
                                    tau_start=1.0, tau_end=150.0,
                                    lr_init=0.35, lr_end=0.003)
            M = compute_M(a)
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n_try:4d}: M={best_M:.8f} J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)

    best.report("Phase 1 complete")

    # === Phase 2: Aggressive LP Mirror Descent (700s) ===
    p2_deadline = time.monotonic() + 700.0
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: Aggressive LP Mirror Descent (700s)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        print(f"  n={n}", flush=True)

        # First: single LP mirror descent
        a = run_lp_mirror_aggressive(best.seq, min(time.monotonic() + 400, p2_deadline - 300),
                                     max_iters=50000, lp_time=0.020,
                                     use_double_lp=False, perturbation_interval=300)
        best.update(a)

        # Second: double LP (fictitious play style)
        if time.monotonic() < p2_deadline - 100:
            a = run_lp_mirror_aggressive(best.seq, min(time.monotonic() + 250, p2_deadline - 100),
                                         max_iters=50000, lp_time=0.015,
                                         use_double_lp=True, perturbation_interval=500)
            best.update(a)

    best.report("Phase 2 complete")

    # === Phase 3: Multi-start perturbations + LP (300s) ===
    p3_deadline = time.monotonic() + 300.0
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Multi-start Perturbations (300s)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < p3_deadline - 30:
        n = best.seq.size
        perturb_schemes = [
            (0.01, 'gauss'),
            (0.02, 'gauss'),
            (0.03, 'gauss'),
            (0.02, 'lowfreq'),
            (0.03, 'lowfreq'),
            (0.05, 'gauss'),
        ]
        for scale, ptype in perturb_schemes:
            if time.monotonic() > p3_deadline - 60:
                break
            for trial in range(3):
                if time.monotonic() > p3_deadline - 30:
                    break
                if ptype == 'gauss':
                    noise = rng.normal(0, scale * 0.005, n)
                else:
                    freqs = rng.integers(1, 8, 3)
                    phases = rng.uniform(0, 2 * np.pi, 3)
                    t_arr = np.linspace(0, 1, n)
                    noise = sum(scale * 0.005 * np.cos(2 * np.pi * f * t_arr + ph)
                                for f, ph in zip(freqs, phases))
                a = best.seq + noise
                a = np.maximum(a, 1e-4)
                a /= a.sum()

                # Quick smooth GD
                dl_gd = min(time.monotonic() + 30, p3_deadline - 150)
                a = run_smooth_nesterov(a, dl_gd, max_iters=500,
                                        tau_start=5.0, tau_end=100.0,
                                        lr_init=0.1, lr_end=0.01)
                best.update(a)

                # LP polish
                if time.monotonic() < p3_deadline - 60:
                    dl_lp = min(time.monotonic() + 80, p3_deadline - 40)
                    a = run_lp_mirror_aggressive(a, dl_lp, max_iters=2000,
                                                lp_time=0.015, use_double_lp=False,
                                                perturbation_interval=0)
                    best.update(a)

    best.report("Phase 3 complete")

    # === Phase 4: Coordinate Descent Polish (150s) ===
    p4_deadline = min(time.monotonic() + 150.0, deadline - 10)
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Coordinate Descent Polish (150s)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        a = coord_descent_polish(best.seq, p4_deadline, max_passes=1000)
        best.update(a)

    best.report("Phase 4 complete")

    # === Phase 5: Final LP polish ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Final LP Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 5:
        a = run_lp_mirror_aggressive(best.seq, deadline, max_iters=20000,
                                     lp_time=0.020, use_double_lp=False,
                                     perturbation_interval=0)
        best.update(a)

    best.report("Phase 5 complete")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
