def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 0:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n

    result = np.maximum(v - tau, 0)
    s = np.sum(result)
    if s < 1e-12:

    return result / s

    start = time.monotonic()
    prev_grad = None
    prev_x = None
    stall = 0
    
    for it in range(10000):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute subgradient
        grad = smooth_gradient(x, top_k=20)
        
        # BB step size
        if prev_grad is not None and prev_x is not None:
            s_vec = x - prev_x
            y_vec = grad - prev_grad
            yy = np.dot(y_vec, y_vec)
            if yy > 1e-14:
                alpha = np.dot(s_vec, y_vec) / yy
                alpha = max(1e-6, min(alpha, 1.0))
            else:
                alpha = 1.0 / (np.linalg.norm(grad) + 1e-10)
        else:
            norm_g = np.linalg.norm(grad)
            alpha = 0.1 / (norm_g + 1e-10)
            alpha = min(alpha, 0.5)
        
        # Gradient step
        x_new = x - alpha * grad
        x_new = project_simplex(x_new)
        
        obj_new = compute_obj(x_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
        
        prev_x = x.copy()
        prev_grad = grad.copy()
        x = x_new
        
        if verbose and it % 500 == 0:
            print(f"    BBPG it={it}, obj={best_obj:.6f}, alpha={alpha:.6f}")
    
    return best_a, best_obj


def lse_gradient_descent(a, time_budget=50.0, top_k=30, verbose=False):
    """Gradient descent on log-sum-exp smooth approximation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    x = a.copy()
    best_a = x.copy()
    best_obj = compute_obj(x)
    
    start = time.monotonic()
    lr = 0.01
    stall = 0
    
    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        
        grad = smooth_gradient(x, top_k=top_k)
        
        # Adaptive step size
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-10:
            break
        step = lr / norm_g
        
        x_new = x - step * grad
        x_new = project_simplex(x_new)
        
        obj_new = compute_obj(x_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = x_new.copy()
            stall = 0
            # Increase lr slightly
            lr = min(lr * 1.05, 0.5)
        else:
            stall += 1
            lr = lr * 0.9
            if stall > 50:
                break
        
        x = x_new
    
    return best_a, best_obj


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)
    
    if choice == 0:
        # Cosine with random frequency
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        # Two cosines
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2))
    elif choice == 2:
        # Smoothed random
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        # Asymmetric ramp + noise
        a = np.linspace(0.3, 1.5, n) + 0.2 * rng.standard_normal(n)
    elif choice == 4:
        # Two bumps (asymmetric)
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w1 = n * rng.uniform(0.05, 0.15)
        w2 = n * rng.uniform(0.05, 0.15)
        a = 0.3 + np.exp(-0.5 * ((k - c1) / w1) ** 2) + np.exp(-0.5 * ((k - c2) / w2) ** 2)
    elif choice == 5:
        # Blocks
        nblocks = rng.integers(3, 8)
        a = np.zeros(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.2, 1.5)
    elif choice == 6:
        # Sine power
        f = rng.uniform(1, 5)
        p = rng.uniform(1, 4)
        a = np.abs(np.sin(np.pi * f * k / n)) ** p + 0.1
    elif choice == 7:
        # Geometric decay + oscillation
        base = 0.5 ** (k / (n * 0.3))
        osc = 1 + 0.4 * np.cos(2 * np.pi * rng.uniform(2, 8) * k / n)
        a = base * osc + 0.1
    elif choice == 8:
        # Random walk (smoothed)
        steps = rng.standard_normal(n) * 0.05
        a = np.cumsum(steps) + 1.0
        a = a - np.min(a) + 0.2
    else:
        a = rng.uniform(0.1, 2.0, n)
    
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def combined_local_search(a, time_budget=100.0, verbose=False):
    """Run multiple local search methods and return the best."""
    n = len(a)
    best_a = a.copy().astype(np.float64)
    s = np.sum(best_a)
    if s < 1e-10:
        best_a = np.ones(n) / n
    else:
        best_a = best_a / s
    best_obj = compute_obj(best_a)
    
    start = time.monotonic()
    
    # Phase A: Accelerated FW
    alloc_a = time_budget * 0.4
    if time_budget > 20:
        a_fv, obj_fv = accelerated_fw(best_a, time_budget=alloc_a, verbose=verbose)
        if obj_fv < best_obj:
            best_obj = obj_fv
            best_a = a_fv.copy()
    
    # Phase B: LSE gradient descent
    remaining = time_budget - (time.monotonic() - start)
    alloc_b = remaining * 0.3
    if alloc_b > 5:
        a_lse, obj_lse = lse_gradient_descent(best_a, time_budget=alloc_b, top_k=25, verbose=verbose)
        if obj_lse < best_obj:
            best_obj = obj_lse
            best_a = a_lse.copy()
    
    # Phase C: More FW
    remaining = time_budget - (time.monotonic() - start)
    alloc_c = remaining * 0.35
    if alloc_c > 5:
        a_fv2, obj_fv2 = accelerated_fw(best_a, time_budget=alloc_c, verbose=verbose)
        if obj_fv2 < best_obj:
            best_obj = obj_fv2
            best_a = a_fv2.copy()
    
    # Phase D: BB gradient
    remaining = time_budget - (time.monotonic() - start)
    alloc_d = remaining * 0.5
    if alloc_d > 5:
        a_bb, obj_bb = bb_projected_gradient(best_a, time_budget=alloc_d, verbose=verbose)
        if obj_bb < best_obj:
            best_obj = obj_bb
            best_a = a_bb.copy()
    
    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: Accelerated Frank-Wolfe with smooth gradient escape,
    BB projected gradient, adaptive constraints, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    print(f"Initial best: {best_score:.6f}")
    
    # Phase 1: Intensive optimization on parent
    if best_seq is not None:
        n_parent = len(best_seq)
        print(f"\n=== Phase 1: Optimize parent (n={n_parent}) ===")
        remaining = deadline - time.monotonic()
        alloc1 = min(remaining * 0.35, 400.0)
        a_opt, obj_opt = combined_local_search(best_seq, time_budget=alloc1)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  After optimization: {best_score:.6f}")
        else:
            print(f"  After optimization: {obj_opt:.6f} (no improvement)")
    
    # Phase 2: Multi-n exploration
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    
    if best_seq is not None:
        np_ = len(best_seq)
        n_candidates = sorted(set([
            max(100, np_ - 100),
            max(100, np_ - 50),
            np_,
            np_ + 50,
            np_ + 100,
            np_ + 200,
            np_ + 300,
        ]))
    else:
        n_candidates = [400, 600, 700, 800, 1000, 1200]
    
    n_candidates = [n for n in n_candidates if 50 <= n <= 2000]
    print(f"  n values: {n_candidates}")
    
    for n in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        # Skip if we already tried this n with parent
        if best_seq is not None and n == len(best_seq):
            # Still do a quick attempt with different init
            alloc = min(remaining * 0.12, 60.0)
        else:
            alloc = min(remaining * 0.15, 100.0)
        alloc = max(15.0, alloc)
        
        print(f"\n  N={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")
        
        # Multiple random starts for this n
        best_n = None
        best_n_obj = np.inf
        
        n_starts = max(1, int(alloc // 30))
        for si in range(n_starts):
            if time.monotonic() > deadline - 60:
                break
            sub_alloc = (deadline - 60 - time.monotonic()) / (n_starts - si)
            sub_alloc = min(sub_alloc, alloc / n_starts + 5)
            sub_alloc = max(10.0, sub_alloc)
            
            if best_seq is not None and si == 0:
                a_start = resample_sequence(best_seq, n)
            else:
                a_start = random_init(n, seed + n * 1000 + si * 77)
            
            a_res, obj_res = combined_local_search(a_start, time_budget=sub_alloc)
            
            if obj_res < best_n_obj:
                best_n_obj = obj_res
                best_n = a_res.copy()
        
        if best_n is not None and best_n_obj < best_score:
            best_score = best_n_obj
            best_seq = best_n.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")
        elif best_n is not None:
            print(f"  N={n}: best={best_n_obj:.6f} (no update)")
    
    # Phase 3: Final intensive polish
    print(f"\n=== Phase 3: Final Polish ===")
    print(f"  Best before: {best_score:.6f}")
    
    if best_seq is not None and time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  Polishing n={n_best}, {remaining:.0f}s remaining")
        
        # Round 1: Extended AFW
        alloc_r1 = min(remaining * 0.35, 300.0)
        a_r1, obj_r1 = accelerated_fw(best_seq, time_budget=alloc_r1)
        if obj_r1 < best_score:
            best_score = obj_r1
            best_seq = a_r1.copy()
            print(f"  Round 1 (AFW): {best_score:.6f}")
        
        # Round 2: LSE + AFW
        if time.monotonic() < deadline - 40:
            remaining = deadline - time.monotonic()
            a_r2, obj_r2 = lse_gradient_descent(best_seq, time_budget=min(remaining * 0.2, 60.0), top_k=20)
            if obj_r2 < best_score:
                best_score = obj_r2
                best_seq = a_r2.copy()
                print(f"  Round 2a (LSE): {best_score:.6f}")
            
            if time.monotonic() < deadline - 30:
                remaining = deadline - time.monotonic()
                a_r2b, obj_r2b = accelerated_fw(best_seq, time_budget=min(remaining * 0.4, 120.0))
                if obj_r2b < best_score:
                    best_score = obj_r2b
                    best_seq = a_r2b.copy()
                    print(f"  Round 2b (AFW): {best_score:.6f}")
        
        # Round 3: Perturbation + AFW
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            rng = np.random.default_rng(seed + 4242)
            for ptry in range(4):
                if time.monotonic() > deadline - 25:
                    break
                pert_alloc = min(remaining * 0.2, 60.0)
                
                # Different perturbation strategies
                strat = ptry % 4
                a_pert = best_seq.copy()
                if strat == 0:
                    # Gaussian perturbation
                    scale = rng.uniform(0.02, 0.06)
                    a_pert = a_pert * np.exp(rng.normal(0, scale, n_best))
                elif strat == 1:
                    # Asymmetric perturbation
                    asym = rng.uniform(0.7, 1.3)
                    a_pert[:n_best // 2] *= asym
                    a_pert[n_best // 2:] *= (2.0 - asym)
                elif strat == 2:
                    # Peak-breaking: reduce values that contribute to the max
                    conv_p = compute_conv(a_pert)
                    max_idx = int(np.argmax(conv_p))
                    n_half = n_best
                    # Reduce values that contribute to this peak
                    for j in range(n_half):
                        idx = max_idx - j
                        if 0 <= idx < n_half:
                            a_pert[j] *= 0.85
                    a_pert = np.maximum(a_pert, 0)
                else:
                    # Spectral perturbation
                    fa = rfft(a_pert)
                    mag = np.abs(fa)
                    phase = np.angle(fa)
                    mag *= np.exp(rng.normal(0, 0.1, len(mag)))
                    fa_new = mag * np.exp(1j * phase)
                    a_pert = irfft(fa_new, n_best)
            
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    a_r3, obj_r3 = accelerated_fw(a_pert, time_budget=pert_alloc)
                    if obj_r3 < best_score:
                        best_score = obj_r3
                        best_seq = a_r3.copy()
                        print(f"  Round 3 (pert {ptry}): {best_score:.6f}")
        
        # Round 4: BB gradient
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            a_r4, obj_r4 = bb_projected_gradient(best_seq, time_budget=min(remaining * 0.5, 80.0))
            if obj_r4 < best_score:
                best_score = obj_r4
                best_seq = a_r4.copy()
                print(f"  Round 4 (BB): {best_score:.6f}")
    
    # Phase 4: Try a slightly different n with the best pattern
    if best_seq is not None and time.monotonic() < deadline - 50:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        # Try n+50 and n-50 with resampled best
        for dn in [50, -50, 100, -100]:
            n_try = n_best + dn
            if n_try < 100 or n_try > 2000:
                continue
            if time.monotonic() > deadline - 30:
                break
            sub_budget = min(remaining * 0.3, 80.0)
            remaining = deadline - time.monotonic()
            a_try = resample_sequence(best_seq, n_try)
            a_res, obj_res = combined_local_search(a_try, time_budget=sub_budget)
            if obj_res < best_score:
                best_score = obj_res
                best_seq = a_res.copy()
                print(f"  n-shift +{dn}: {best_score:.6f}")
    
    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    
    return result
