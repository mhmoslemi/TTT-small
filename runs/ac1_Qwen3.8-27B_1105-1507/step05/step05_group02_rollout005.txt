import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Compute linear auto-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective value (scale-invariant)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def spectral_shaping(a, n_iters=30, alpha=0.25):
    """Apply spectral baton-style shaping to reduce peak auto-conv."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for _ in range(n_iters):
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        c = np.fft.irfft(P, N)[:2 * n - 1]
        c_max = c.max()
        c_target = c.sum() / len(c)

        # Flatten the power spectrum toward uniform level
        # This reduces concentration in any frequency bin
        target = np.full_like(P, c_target * N / max(c_max, 1e-15) * 0 + P.mean())
        P_new = P + alpha * (target - P)
        P_new = np.maximum(P_new, 1e-20)

        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new

    return a


def smooth_max_grad(a, tau):
    """Compute gradient of soft-max of auto-conv. Returns (grad, b, b_max)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[lo:k - hi + 1][::-1]
    return grad, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.35, lr_end=0.003):
    """Smooth gradient descent with Nesterov momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def build_conv_matrix(n, a):
    """Build the (2n-1) x n convolution matrix A where A@x gives a*x (linear conv)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_best_response(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-7})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=5000):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    last_imp_time = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        # Check for timeout on no improvement
        if no_imp > 500 and (time.monotonic() - last_imp_time) > 60:
            break

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_imp = 0
            last_imp_time = time.monotonic()

        # Solve LP for best response
        g = solve_best_response(a, M, time_limit=5)
        if g is None:
            break

        # Adaptive step size: try multiple t values
        improved = False
        best_t = 0.1
        best_M_t = M
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M_t:
                best_M_t = M_t
                best_t = t

        if best_M_t < M - 1e-14:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            a /= a.sum()
            improved = True
        else:
            no_imp += 1
            # Try smaller steps
            for t in [0.02, 0.01, 0.005]:
                a_t = (1.0 - t) * a + t * g
                a_t = np.maximum(a_t, 0.0)
                s = a_t.sum()
                if s < 1e-10:
                    continue
                a_t /= s
                M_t = fast_conv(a_t).max()
                if M_t < M - 1e-14:
                    a = a_t
                    improved = True
                    break
            if not improved:
                no_imp += 1
                continue

    return best_a


def run_subgradient_polish(a, deadline, max_iters=2000):
    """Subgradient descent with Polyak averaging for final polish."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

        # Subgradient of max: uniform over active set
        tol = 1e-8 * M
        active = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[lo:k - hi + 1][::-1]
        grad /= len(active)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        lr = 0.15 / np.sqrt(it + 1)
        a_new = np.maximum(a - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        running += a_new
        count += 1
        a = a_new

    if count > 100:
        avg = running / count
        avg = np.maximum(avg, 0.0)
        s = avg.sum()
        if s > 1e-10:
            avg /= s
            M_avg = fast_conv(avg).max()
            if M_avg < best_M:
                best_a = avg

    return best_a


def resample_seq(src, n_target):
    """Resample a sequence to a new length using interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n_target:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n_target)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 1e-4))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 1e-4))

    # Flat-top window
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))

    # Raised cosine variants
    for beta in [0.5, 1.0, 1.5]:
        w = 0.5 + 0.5 * np.cos(np.pi * (2 * t - 1) / beta) if beta > 0 else np.ones(n)
        inits.append(np.maximum(w, 0.01))

    # Gaussian
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.05)

    # Double Gaussian
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03)

    # Cosine-modulated
    for f in [2, 3, 5]:
        inits.append(np.maximum(0.7 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))

    # Random smoothed
    k = max(3, n // 50)
    kern = np.hanning(k)
    kern /= kern.sum()
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))

    # Symmetric random
    half = np.maximum(rng.uniform(0.2, 1.0, n // 2 + 1), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)

    return inits

                    if time.monotonic() > p4_deadline - 10:
                        break
                    noise = rng.normal(0, scale, n_b)
                    a = np.maximum(best.seq + noise, 1e-6)
                    a /= a.sum()
                    a = run_lp_mirror(a, min(time.monotonic() + 20, p4_deadline - 5),
                                      max_iters=500)
                    best.update(a)

    best.report("Phase 4")

    # === PHASE 5: Try adjacent lengths (60s) ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Length Exploration", flush=True)
    print(f"{'='*50}", flush=True)
    p5_deadline = deadline - 5.0

    if best.seq is not None:
        n_cur = best.seq.size
        for delta in [-16, -8, 8, 16, 32]:
            if time.monotonic() > p5_deadline:
                break
            n_try = n_cur + delta
            if n_try < 64 or n_try > 4096:
                continue
            a = resample_seq(best.seq, n_try)
            a = np.maximum(a, 1e-6)
            a /= a.sum()
            a = run_smooth_nesterov(a, min(time.monotonic() + 15, p5_deadline - 5),
                                    max_iters=500, tau_start=5, tau_end=100,
                                    lr_init=0.15, lr_end=0.01)
            best.update(a)
            if best.seq is not None and best.seq.size == n_try:
                # Quick LP on this length
                a = run_lp_mirror(best.seq, min(time.monotonic() + 15, p5_deadline - 5),
                                  max_iters=300)
                best.update(a)

    best.report("Phase 5")

    # === FINAL: One more LP pass if time remains ===
    if best.seq is not None and time.monotonic() < deadline:
        a = run_lp_mirror(best.seq, deadline, max_iters=2000)
        best.update(a)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
