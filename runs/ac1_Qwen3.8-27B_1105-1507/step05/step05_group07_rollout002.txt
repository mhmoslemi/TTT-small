    for i, k in enumerate(active_indices):
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[k - j])
    if len(rows) == 0:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))


def solve_lp(a, M, active_indices, time_limit=3.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_indices, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a, active_indices)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, constraint_tol=0.12, full_lp_every=8):
    """
    LP mirror descent with adaptive constraint selection.
    Uses only near-active constraints for speed, with periodic full LP checks.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_obj = 2.0 * n * best_M
    no_improve = 0
    iter_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Determine active constraints adaptively
        if it % full_lp_every == 0:
            active_mask = np.ones(2 * n - 1, dtype=bool)
        else:
            tol = M * constraint_tol
            active_mask = b >= (M - tol)
            if active_mask.sum() < 8:
                active_mask = np.ones(2 * n - 1, dtype=bool)

        active_idx = np.where(active_mask)[0]
        iter_count += 1

        # Solve LP
        tl = min(3.0, max(0.5, (deadline - time.monotonic()) / 3.0))
        g = solve_lp(a, M, active_idx, time_limit=tl)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Verify no constraint violations for dropped constraints
        if not active_mask.all():
            ag_conv = np.convolve(a, g)
            if len(ag_conv) == 2 * n - 1 and ag_conv.max() > M * (1 + 1e-5):
                # Re-solve with all constraints
                g = solve_lp(a, M, np.arange(2 * n - 1), time_limit=tl * 2)
                if g is None:
                    break
                g = np.maximum(g, 0)
                gs = g.sum()
                if gs < 1e-10:
                    break
                g = g / gs

        # Line search with multiple step sizes
        improved = False
        progress = it / max(1, max_iters)
        base_t = 0.4 * (1.0 - 0.4 * progress)
        step_sizes = [base_t, base_t * 0.5, base_t * 0.25, base_t * 0.125, 0.05, 0.02]

        for t in step_sizes:
            if t <= 0:
                continue
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            obj_trial = 2.0 * n * M_trial
            if obj_trial < best_obj - 1e-12:
                a = a_trial
                best_M = M_trial
                best_obj = obj_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Try even smaller steps
            for t in [0.01, 0.005, 0.001]:
                a_trial = np.maximum((1 - t) * a + t * g, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                M_trial = fast_conv(a_trial).max()
                if 2.0 * n * M_trial < best_obj - 1e-14:
                    a = a_trial
                    best_a = a.copy()
                    best_obj = 2.0 * n * M_trial
                    no_improve = 0
                    improved = True
                    break

        if no_improve > 60:
            break

    return best_a


def coord_descent_fast(a, max_sweeps, deadline, random_order=True, seed=0):
    """
    Fast coordinate descent with incremental convolution updates.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = a / max(a.sum(), 1e-10)
        s = a.sum()

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    no_improve_sweeps = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order and sweep % 2 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        sweep_improved = False

        for i in order:
            old = a[i]
            if old < 1e-10:
                candidates = [0.0, 1e-4, 1e-3, 1e-2]
            else:
                candidates = np.array([0.0, 0.3, 0.5, 0.7, 1.0, 1.4, 1.8]) * old
                candidates = candidates[~np.isclose(candidates, old, rtol=1e-3)]

            if len(candidates) == 0:
                continue

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            for x in candidates:
                x = float(max(0, x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

        if no_improve_sweeps > 100:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def freq_perturb(a, rng, strength=0.1):
    """Perturb a sequence in frequency domain for diversity."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    # Add random phase perturbation to higher frequencies
    noise = np.zeros_like(fa)
    for k in range(1, len(fa)):
        noise[k] = strength * (rng.normal() + 1j * rng.normal()) / (1 + k * 0.1)
    fa_pert = fa + noise
    a_pert = np.fft.irfft(fa_pert, size)[:n]
    a_pert = np.maximum(a_pert, 0)
    if a_pert.sum() > 1e-10:
        a_pert /= a_pert.sum()
    return a_pert


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_polish(a, max_iters, deadline, lr_init=0.1):
    """Subgradient descent with averaging for final polish."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lr = lr_init / (it + 1) ** 0.6
        b = fast_conv(a)
        M = b.max()
        tol = 0.01 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]

        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad /= gnorm

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        running_sum += a_new
        count += 1
        a = a_new

        M_cur = fast_conv(a).max()
        if M_cur < best_M:
            best_M = M_cur
            best_a = a.copy()

    if count > 20:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0.0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return best_a


def focused_cd(a, deadline, max_sweeps=200):
    """
    Focused coordinate descent: only modify coordinates that affect
    the top convolution peaks.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a / max(a.sum(), 1e-10)

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Find top peak positions
        M = conv.max()
        peak_idx = np.where(conv >= M * 0.95)[0]

        # Find coordinates that contribute to peaks
        active_coords = set()
        for k in peak_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                active_coords.add(j)

        if not active_coords:
            break

        coords = sorted(active_coords)
        improved = False

        for i in coords:
            old = a[i]
            if old < 1e-10:
                candidates = [0.0, 1e-4, 1e-3]
            else:
                candidates = np.array([0.0, 0.4, 0.6, 0.8, 1.0, 1.3]) * old
                candidates = candidates[~np.isclose(candidates, old, rtol=1e-3)]

            if len(candidates) == 0:
                continue

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            for x in candidates:
                x = float(max(0, x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase LP mirror descent search with constraint selection,
    frequency-domain perturbation restarts, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        if a is None or len(a) == 0:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
            return True
        return False

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        parent = parent / parent.sum()
        update_best(parent)

    # === PHASE 1: Quick coordinate descent (150s) ===
    p1_deadline = min(t_start + 150, deadline - 600)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Coordinate Descent n={len(parent)} (budget={p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    n_base = len(parent)
    a_cd = coord_descent_fast(parent, max_sweeps=3000, deadline=p1_deadline,
                              random_order=True, seed=seed)
    update_best(a_cd)

    # === PHASE 2: LP Mirror Descent with constraint selection (500s) ===
    p2_deadline = min(t_start + 650, deadline - 300)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent n={n_base} (budget={p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Multiple constraint tolerances - start tight, loosen if stuck
    for ctol in [0.08, 0.12, 0.18, 0.25]:
        if time.monotonic() > p2_deadline - 30:
            break
        a_lp = lp_mirror_descent(best_seq if best_seq is not None else a_cd,
                                 max_iters=500, deadline=p2_deadline,
                                 constraint_tol=ctol, full_lp_every=10)
        update_best(a_lp)
        print(f"  LP ctol={ctol}: J={compute_J(best_seq):.8f}", flush=True)

    # === PHASE 3: Multi-length exploration (300s) ===
    p3_deadline = min(t_start + 950, deadline - 150)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Multi-length (budget={p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    target_lengths = [l for l in [768, 1024, 1536] if l > n_base]
    for new_n in target_lengths:
        if time.monotonic() > p3_deadline - 60:
            break
        print(f"  Trying n={new_n}...", flush=True)
        a_ext = resample_seq(best_seq, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        # Quick CD first
        a_cd2 = coord_descent_fast(a_ext, max_sweeps=1000,
                                   deadline=min(time.monotonic() + 60, p3_deadline - 60),
                                   random_order=True, seed=seed + new_n)
        update_best(a_cd2)

        # LP mirror descent
        if time.monotonic() < p3_deadline - 30:
            a_lp2 = lp_mirror_descent(a_cd2, max_iters=200,
                                      deadline=p3_deadline - 10,
                                      constraint_tol=0.15, full_lp_every=10)
            update_best(a_lp2)

    # === PHASE 4: Annealing restarts + final polish (remaining) ===
    p4_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Annealing + Polish (budget={p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if time.monotonic() < p4_deadline - 30:
        # Frequency-domain perturbation restarts
        base_n = best_seq.size
        for restart in range(5):
            if time.monotonic() > p4_deadline - 30:
                break
            strength = 0.05 + 0.03 * restart
            a_pert = freq_perturb(best_seq, rng, strength=strength)
            # Quick LP on perturbed
            a_lr = lp_mirror_descent(a_pert, max_iters=100,
                                     deadline=min(time.monotonic() + 60, p4_deadline - 20),
                                     constraint_tol=0.15, full_lp_every=8)
            update_best(a_lr)

        # Subgradient polish
        if time.monotonic() < p4_deadline - 20:
            a_sg = subgradient_polish(best_seq, max_iters=2000,
                                      deadline=min(time.monotonic() + 40, p4_deadline - 10),
                                      lr_init=0.08)
            update_best(a_sg)

        # Focused CD
        if time.monotonic() < p4_deadline - 15:
            a_fc = focused_cd(best_seq, deadline=min(time.monotonic() + 30, p4_deadline - 5),
                              max_sweeps=500)
            update_best(a_fc)

        # Final LP
        if time.monotonic() < p4_deadline - 5:
            a_final = lp_mirror_descent(best_seq, max_iters=50,
                                        deadline=p4_deadline,
                                        constraint_tol=0.1, full_lp_every=5)
            update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
