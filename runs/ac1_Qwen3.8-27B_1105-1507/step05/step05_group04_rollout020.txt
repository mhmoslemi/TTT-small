def fast_conv(a):

class Adam:
    def __init__(self, dim, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros(dim)
        self.v = np.zeros(dim)
        self.t = 0
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        
    def step(self, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad ** 2)
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Variance-of-autocorrelation smooth gradient pre-optimization
2. Adaptive log-sum-exp peak optimization with Adam
3. LP-based mirror descent with multi-temperature M targeting
4. Palindromic constraint exploration
5. Multi-length search (n=512, n=1024) and targeted perturbation

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize_simplex(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def grad_variance(a):
    n = len(a)
    b = fast_conv(a)
    grad = np.zeros(n)
    for i in range(n):
        j_max = min(n, 2 * n - 1 - i)
        if j_max > 0:
            grad[i] = 4.0 * np.dot(b[i:i + j_max], a[:j_max])
    return grad


def grad_peak_smooth(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    scaled = tau * (b - b_max)
    exp_scaled = np.exp(scaled - scaled.max())
    p = exp_scaled / exp_scaled.sum()
    grad = np.zeros(n)
    for i in range(n):
        j_max = min(n, 2 * n - 1 - i)
        if j_max > 0:
            grad[i] = 2.0 * np.dot(p[i:i + j_max], a[:j_max])
    return grad


class AdamOpt:
    def __init__(self, dim, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros(dim)
        self.v = np.zeros(dim)
        self.t = 0
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps

    def step(self, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad * grad)
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return self.lr * m_hat / (np.sqrt(v_hat) + self.eps)


def build_lp_matrix(n, a, active_k=None):
    if active_k is None:
        active_k = list(range(2 * n - 1))
    rows, cols, vals = [], [], []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))


def solve_lp(a, M, top_frac=1.0, time_limit=5):
    n = len(a)
    b = fast_conv(a)
    if top_frac < 1.0:
        threshold = np.percentile(b, (1.0 - top_frac) * 100)
        active_k = np.where(b >= threshold)[0].tolist()
    else:
        active_k = list(range(2 * n - 1))
    if len(active_k) < 5:
        active_k = list(range(2 * n - 1))
    A = build_lp_matrix(n, a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, np.inf)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-12:
                if top_frac < 1.0:
                    ag = fast_cross_conv(a, g)
                    if ag.max() <= M * (1 + 1e-4):
                        return g
            return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a_init, deadline, top_frac=0.6, verbose=False,
                      max_no_improve=300, use_multi_M=True):
    a = a_init.copy()
    a = normalize_simplex(a)
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    it = 0
    M_offsets = [1.0, 1.0, 1.0, 0.999, 0.998, 1.001] if use_multi_M else [1.0]
    m_idx = 0

    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = float(b.max())

        offset = M_offsets[m_idx % len(M_offsets)]
        M_target = M * offset
        m_idx += 1

        g = solve_lp(a, M_target, top_frac=top_frac, time_limit=4)
        if g is None or g.sum() < 1e-12:
            continue
        g = g / g.sum()

        diff = np.max(np.abs(g - a))
        base_t = min(0.6, max(0.05, diff * 1.5))
        t_cands = [base_t, base_t * 0.5, base_t * 0.25, base_t * 0.1, 0.03, 0.01]

        current_J = compute_J(a)
        improved = False
        for t in t_cands:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-12:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
            elif J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                no_improve = 0
                break

        if not improved:
            no_improve += 1

        if no_improve > max_no_improve:
            rng = np.random.default_rng(it * 31 + no_improve)
            scale = 0.08 / (1.0 + 0.2 * (no_improve // max_no_improve))
            # Use gradient-aligned perturbation
            grad = grad_variance(a)
            grad = grad / (np.linalg.norm(grad) + 1e-10)
            noise = rng.normal(0, 1, n)
            a = a - scale * 0.5 * grad + scale * 0.5 * noise
            a = normalize_simplex(a)
            no_improve = 0

        if verbose and it % 1000 == 0:
            print(f"    LP iter={it} J={best_J:.8f} M={M:.8f} diff={diff:.4f} "
                  f"no_imm={no_improve}", flush=True)

    return best_a, best_J


def smooth_optimize(a_init, deadline, objective='var', lr=0.05, verbose=False):
    a = a_init.copy()
    a = normalize_simplex(a)
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    adam = AdamOpt(n, lr=lr)
    tau = 5.0
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        if objective == 'var':
            grad = grad_variance(a)
        elif objective == 'peak':
            grad = grad_peak_smooth(a, tau)
        elif objective == 'mixed':
            if it % 4 < 2:
                grad = grad_variance(a)
            else:
                grad = grad_peak_smooth(a, tau)
                tau = min(tau * 1.01, 80.0)
        else:
            grad = grad_variance(a)

        step = adam.step(grad)
        a_new = a - step
        a_new = normalize_simplex(a_new)
        J_new = compute_J(a_new)

        if J_new < best_J - 1e-13:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new

        if no_improve > 500:
            rng = np.random.default_rng(it * 17)
            a = a + 0.03 * rng.normal(0, 1, n)
            a = normalize_simplex(a)
            adam = AdamOpt(n, lr=lr)
            no_improve = 0
            tau = 5.0

        if verbose and it % 3000 == 0:
            print(f"    smooth iter={it} J={best_J:.8f} tau={tau:.1f}", flush=True)

    return best_a, best_J


def make_palindrome(a):
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(normalize_simplex(p.copy()))
        inits.append(normalize_simplex(p + rng.normal(0, 0.03, n)))
        inits.append(normalize_simplex(p + rng.normal(0, 0.1, n)))
        inits.append(normalize_simplex(make_palindrome(p)))

    inits.append(np.ones(n) / n)
    inits.append(normalize_simplex(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t_arr - 1)), 0.01)))
    inits.append(normalize_simplex(np.maximum(0.3 + 0.7 * np.exp(-4 * (t_arr - 0.5) ** 2), 0.01)))
    inits.append(normalize_simplex(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 2 * t_arr), 0.01)))
    inits.append(normalize_simplex(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * 3 * t_arr), 0.01)))
    edge = np.maximum(np.exp(-4 * t_arr**2) + np.exp(-4 * (t_arr - 1)**2) + 0.05, 0.01)
    inits.append(normalize_simplex(edge))

    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(normalize_simplex(smooth + 0.1))

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a, label=""):
        if a is None or a.size == 0:
            return
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = compute_J(a)
        if score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s [{label}]", flush=True)
