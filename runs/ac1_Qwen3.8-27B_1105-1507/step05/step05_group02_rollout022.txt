def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=2.0, tau_end=200.0,

def peak_tracking_refine(a, deadline, max_iters=200):
    """Refine by targeting the peak coordinates of the autocorrelation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):

            break
        b = fast_conv(a)
        M = b.max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

        # Find the peak location
        k_peak = np.argmax(b)
        # Find coordinates that contribute most to the peak
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        contrib = a[lo:hi + 1] * a[k_peak - hi:k_peak - lo + 1][::-1]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        # Find the coordinate with highest contribution
        idx_max = lo + np.argmax(contrib)
        # Reduce this coordinate slightly and redistribute
        old_val = a[idx_max]
        if old_val < 1e-6:
            break

        # Try reducing
        for factor in [0.85, 0.7, 0.5]:
            a[idx_max] = old_val * factor
            # Redistribute to reduce peak: increase coordinates opposite to peak
            s = a.sum()
            a /= s
            b_new = fast_conv(a)
            M_new = b_new.max()
            if M_new < M:
                M = M_new
                break
            a[idx_max] = old_val
        else:
            # Try increasing a different coordinate
            # Find coordinate that most reduces the peak when increased
            for idx_try in range(n):
                if idx_try == idx_max:
                    continue
                old = a[idx_try]
                a[idx_try] = old * 1.3 + 0.01
                s = a.sum()
                a /= s
                b_new = fast_conv(a)
                M_new = b_new.max()
                if M_new < M * (1 - 1e-9):
                    M = M_new
                    break
                a[idx_try] = old
            else:
                a[idx_max] = old_val
                break

    return best_a

def freq_domain_perturb(a, rng, amount=0.05):
    """Perturb in frequency domain for diversity."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    # Randomly scale frequency components
    magnitudes = np.abs(A)
    phases = np.angle(A)
    # Low-pass emphasis: reduce high frequency noise
    freqs = np.fft.rfftfreq(size, d=1.0 / size)
    mask = np.exp(-freqs * 2.0 / n)
    scale = 1.0 + amount * (rng.standard_normal(len(A)) * mask)
    scale = np.maximum(scale, 0.1)
    A_new = magnitudes * scale * np.exp(1j * phases)
    a_new = np.fft.irfft(A_new, size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def get_parent():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            return np.maximum(p, 0.0)
    except (NameError, AttributeError, Exception):
        pass
    return None


def relax_schedule_linear(it, max_iters):
    """Start with 10% relaxation, decay to 0."""
    frac = it / max_iters
    if frac < 0.2:
        return 0.10 * (1.0 - frac / 0.2)
    elif frac < 0.5:
        return 0.05 * (1.0 - (frac - 0.2) / 0.3)
    elif frac < 0.8:
        return 0.02 * (1.0 - (frac - 0.5) / 0.3)
    else:
        return 0.0


def relax_schedule_staircase(it, max_iters):
    """Step relaxation: 5%, 3%, 1%, 0%"""
    frac = it / max_iters
    if frac < 0.25:
        return 0.05
    elif frac < 0.5:
        return 0.03
    elif frac < 0.75:
        return 0.01
    else:
        return 0.0


def make_diverse_inits(parent, n, rng):
    """Generate diverse initializations for multi-start."""
    inits = []
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        # Small perturbations
        for sigma in [0.01, 0.02, 0.04]:
            inits.append(np.maximum(p + rng.normal(0, sigma * 0.01, n), 0.005))
        # Frequency domain perturbations
        for _ in range(2):
            inits.append(freq_domain_perturb(p, rng, amount=0.08))
        # Multiplicative perturbation
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.005))
    # Some global initializations
    t = np.linspace(0, 1, n, endpoint=False)
    for alpha in [0.5, 0.8, 1.0, 1.5, 2.0]:
        inits.append(np.maximum(0.3 + 0.7 * (t**alpha + (1 - t)**alpha) / 2, 0.01))
    # Cosine
    for f in [1, 2]:
        inits.append(np.maximum(0.6 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    # Random smooth
    k = max(4, n // 25)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    return inits


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0
    best = BestTracker()

    # Get parent
    parent = get_parent()
    if parent is not None:
        best.update(parent)
        print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
        base_n = parent.size
    else:
        parent = np.ones(256)
        best.update(parent)
        base_n = 256

    # Determine lengths to explore
    lengths = set()
    for delta in [-64, -48, -32, -16, 0, 16, 32, 48, 64, 96, 128]:
        n_try = base_n + delta
        if 128 <= n_try <= 2048:
            n_try = (n_try // 16) * 16
            lengths.add(n_try)
    lengths = sorted(lengths)
    print(f"Lengths to explore: {lengths}", flush=True)

    # === PHASE 1: Diverse smooth gradient descent (25% of time) ===
    p1_budget = 0.25 * (deadline - time.monotonic())
    p1_deadline = time.monotonic() + p1_budget
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Diverse Smooth GD (budget={p1_budget:.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    phase1_best = None
    phase1_best_M = float('inf')

    # Focus on the most promising lengths
    priority_lengths = [base_n]
    for n in lengths:
        if n != base_n and n not in priority_lengths:
            priority_lengths.append(n)
    priority_lengths = priority_lengths[:8]  # Limit to 8 lengths

    for i, n in enumerate(priority_lengths):
        if time.monotonic() > p1_deadline - 30:
            break
        # Allocate more time to the base length
        if n == base_n:
            allot = p1_budget * 0.4
        else:
            allot = p1_budget * 0.6 / (len(priority_lengths) - 1)

        dl = min(time.monotonic() + allot, p1_deadline - 10)
        if time.monotonic() >= dl:
            continue

        t0 = time.monotonic()
        inits = make_diverse_inits(parent, n, rng)
        # For base length, use more inits
        if n != base_n:
            inits = inits[:4]

        n_iters = max(100, int(3000 * (256.0 / n)))
        for a0 in inits:
            if time.monotonic() > dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            s = a0.sum()
            if s < 1e-10:
                continue
            a0 /= s
            a = run_smooth_nesterov(a0, dl, max_iters=n_iters,
                                    tau_start=2.0, tau_end=180.0,
                                    lr_init=0.3, lr_end=0.003)
            M = fast_conv(a).max()
            if M < phase1_best_M:
                phase1_best_M = M
                phase1_best = a.copy()

        if phase1_best is not None and (phase1_best.size == n or i == 0):
            best.update(phase1_best)

        if n == base_n and phase1_best is not None:
            print(f"  n={n}: M={phase1_best_M:.7f} J={compute_J(phase1_best):.6f} "
                  f"({time.monotonic()-t0:.1f}s)", flush=True)

    best.report("Phase 1 end")

    # === PHASE 2: LP Mirror Descent with annealing (45% of time) ===
    p2_budget = 0.45 * (deadline - time.monotonic())
    p2_deadline = time.monotonic() + p2_budget
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: LP Mirror with Annealing (budget={p2_budget:.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        # Run LP from the best with annealing schedule
        a_start = best.seq.copy()
        a = run_lp_mirror_anneal(a_start, min(time.monotonic() + p2_budget * 0.6, p2_deadline - 30),
                                 max_iters=10000, relax_schedule=relax_schedule_linear)
        best.update(a)

        # Run LP from a perturbed version
        if time.monotonic() < p2_deadline - 30:
            a_pert = best.seq + rng.normal(0, 0.003, best.seq.size) * 0.01
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert /= a_pert.sum()
            a = run_lp_mirror_anneal(a_pert, min(time.monotonic() + 60, p2_deadline - 20),
                                     max_iters=3000, relax_schedule=relax_schedule_staircase)
            best.update(a)

        # Peak tracking refinement
        if time.monotonic() < p2_deadline - 10:
            a = peak_tracking_refine(best.seq, p2_deadline - 5, max_iters=100)
            best.update(a)

    best.report("Phase 2 end")

    # === PHASE 3: Multi-start LP (20% of time) ===
    p3_budget = 0.20 * (deadline - time.monotonic())
    p3_deadline = time.monotonic() + p3_budget
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Multi-Start LP (budget={p3_budget:.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        # Generate diverse starting points for LP
        starts = []
        starts.append(best.seq.copy())
        for sigma in [0.01, 0.03, 0.06]:
            a_s = best.seq + rng.normal(0, sigma * 0.01, n)
            a_s = np.maximum(a_s, 1e-6)
            a_s /= a_s.sum()
            starts.append(a_s)
        sf = freq_domain_perturb(best.seq, rng, amount=0.1)
        starts.append(sf)

        # First run smooth GD on each start to improve, then LP
        for i, a_s in enumerate(starts):
            if time.monotonic() > p3_deadline - 20:
                break
            allot = max(10, p3_budget / len(starts) - 10)
            # Quick smooth GD
            a_gd = run_smooth_nesterov(a_s, min(time.monotonic() + allot * 0.4, p3_deadline - 15),
                                       max_iters=500, tau_start=5.0, tau_end=100.0,
                                       lr_init=0.15, lr_end=0.01)
            # Then LP
            if time.monotonic() < p3_deadline - 10:
                a_lp = run_lp_mirror_anneal(a_gd, min(time.monotonic() + allot * 0.6, p3_deadline - 5),
                                            max_iters=2000, relax_schedule=relax_schedule_staircase)
                best.update(a_lp)

    best.report("Phase 3 end")

    # === PHASE 4: Try nearby lengths (5% of time) ===
    p4_budget = 0.05 * (deadline - time.monotonic())
    p4_deadline = time.monotonic() + p4_budget
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Length Refinement (budget={p4_budget:.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < p4_deadline:
        n_best = best.seq.size
        for delta in [-16, -8, 8, 16]:
            if time.monotonic() > p4_deadline - 5:
                break
            n_try = n_best + delta
            if n_try < 64 or n_try > 2048:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            a_ext = run_smooth_nesterov(a_ext, min(time.monotonic() + 8, p4_deadline - 3),
                                        max_iters=200, tau_start=3, tau_end=80,
                                        lr_init=0.2, lr_end=0.02)
            best.update(a_ext)
            if time.monotonic() < p4_deadline - 3:
                a_lp = run_lp_mirror_anneal(a_ext, p4_deadline - 2, max_iters=100,
                                            relax_schedule=relax_schedule_staircase)
                best.update(a_lp)

    best.report("Phase 4 end")

    # === PHASE 5: Final extended LP polish (remaining time) ===
    remain = deadline - time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Final LP Polish (budget={remain:.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and remain > 10:
        # Extended LP without relaxation
        a = run_lp_mirror_anneal(best.seq, deadline, max_iters=50000, relax_schedule=None)
        best.update(a)

        # One more with slight perturbation
        if time.monotonic() < deadline - 5:
            a2 = best.seq + rng.normal(0, 0.001, best.seq.size) * 0.005
            a2 = np.maximum(a2, 1e-8)
            a2 /= a2.sum()
            a2 = run_lp_mirror_anneal(a2, deadline, max_iters=20000, relax_schedule=None)
            best.update(a2)

    best.report("Phase 5 end")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
