Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent (alternating best-response)

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_lp_matrix(n, a):
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # (a*g)[k] = sum_j A[k,j] * g[j]
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, time_limit=10):

    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',

        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

        x = x / s
        b = fast_conv(x)
        return 2 * len(x) * b.max() / (s * s)
    
    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                             options={'xatol': 1e-6})
    return g, result.x

def coordinate_descent(a, deadline, max_passes=50, verbose=False):
    """
    Fast coordinate descent with exact 1D optimization.
    For each coordinate i, find the optimal value by analyzing the 1D problem.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        order = np.random.permutation(n)
        b = fast_conv(a)
        S = a.sum()
        M = b.max()
        improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            ai = a[i]
            
            # Find optimal delta for coordinate i
            # b'[k] = b[k] + 2*delta*a[k-i] + delta^2*delta(k, 2i)
            # J(delta) = 2*n*max_k b'[k] / (S+delta)^2
            
            # The affected k range is [i, i+n-1] intersect [0, 2n-2]
            k_lo = i  # k-i >= 0 => k >= i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Unchanged max
            if k_lo > 0:
                outside_left = b[:k_lo].max()
            else:
                outside_left = 0.0
            if k_hi < 2 * n - 2:
                outside_right = b[k_hi + 1:].max()
            else:
                outside_right = 0.0
            C = max(outside_left, outside_right)
            
            # For the affected range, b'[k] = b[k] + 2*delta*a[k-i] (+ delta^2 if k==2i)
            # We want to find delta >= -ai (to keep a[i] >= 0) and delta > -S (to keep S+delta > 0)
            
            # Use golden section search on delta in [-ai, some_upper_bound]
            # Upper bound: we can try a few heuristics
            # The optimal delta for a linear constraint b[k]+2*delta*a[k-i] is at delta* = S - b[k]/a[k-i]
            # So a reasonable upper bound is max over k of (S - b[k]/a[k-i]) clipped
            
            if ai > 1e-10:
                # Find a reasonable upper bound
                # For each affected k, the unconstrained optimum is delta_k = S - b[k]/a[k-i]
                # (ignoring the delta^2 term for k==2i)
                affected_k = np.arange(k_lo, k_hi + 1)
                affected_aj = a[np.clip(affected_k - i, 0, n - 1)]
                # Avoid division by zero
                mask = affected_aj > 1e-15
                if mask.any():
                    delta_candidates = (S - b[affected_k][mask] / affected_aj[mask])
                    upper = max(float(np.percentile(delta_candidates, 95)), 0.5 * S / n)
                else:
                    upper = 0.1
            else:
                upper = 0.1
            
            upper = min(upper, S)  # Can't make sum negative
            lower = -ai  # Keep a[i] >= 0
            
            if upper - lower < 1e-10:
                continue
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
            
            def J_of_delta(d):
                S_new = S + d
                if S_new < 0.01:
                    return float('inf')
                # Compute new max
                # Unchanged: C
                # Changed: b[k] + 2*d*a[k-i] (+ d^2 if k==2i)
                new_vals = b[k_lo:k_hi+1] + 2 * d * a[np.clip(np.arange(k_lo, k_hi + 1) - i, 0, n - 1)]
                # Special case k==2i
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    new_vals[2 * i - k_lo] += d * d
                new_max = max(C, float(new_vals.max()))
                return 2 * n * new_max / (S_new * S_new)
            
            # Golden section search
            lo = lower
            hi = upper
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = J_of_delta(c1)
            f2 = J_of_delta(c2)
            
            for _ in range(60):  # ~60 iterations gives ~1e-12 precision
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = J_of_delta(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = J_of_delta(c2)
            
            opt_delta = (lo + hi) / 2
            new_ai = max(ai + opt_delta, 0.0)
            
            if abs(new_ai - ai) > 1e-12:
                a[i] = new_ai
                S += opt_delta
                improved = True
                # Update b incrementally
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j >= 0 and j < n:
                        if k == 2 * i:
                            b[k] += 2 * opt_delta * ai + opt_delta * opt_delta
                        else:
                            b[k] += 2 * opt_delta * a[j]
            
            # Renormalize to keep sum = 1 (do it periodically to avoid drift)
            if (p + 1) % 10 == 0:
                a = np.maximum(a, 0)
                S = a.sum()
                if S < 0.01:
                    break
                a = a / S
                b = fast_conv(a)
                S = 1.0
        
        # End of pass: check improvement
        a = np.maximum(a, 0)
        S = a.sum()
        if S < 0.01:
            break
        a = a / S
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
        
        if not improved:
            break
        
        if verbose and p % 10 == 0:
            print(f"    CD pass {p}: J={J:.8f}")
    
    return best_a

def lp_mirror_descent(a, max_iters, deadline, verbose=False):
    """LP mirror descent with periodic CD."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10).copy()
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b_aa = fast_conv(a)
        M = float(b_aa.max())
        
        g = lp_best_response(a, M, time_limit=5)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Line search
        a_norm = a / a.sum()
        
        def obj(t):
            x = (1 - t) * a_norm + t * g
            x = np.maximum(x, 0)
            s_val = x.sum()
            if s_val < 0.01:
                return float('inf')
            x = x / s_val
            b_val = fast_conv(x)
            return 2 * len(x) * b_val.max() / (s_val * s_val)
        
        result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-7})
        t_opt = float(result.x)
        
        a_new = (1 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0)
        s_val = a_new.sum()
        if s_val < 1e-10:
            break
        a_new = a_new / s_val
        
        J_new = compute_J(a_new)
        
        if J_new < best_J - 1e-12:
            best_a = a_new.copy()
            best_J = J_new
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic CD
        if it % 30 == 29 and no_improve < 20:
            cd_deadline = min(time.monotonic() + 10, deadline)
            a_cd = coordinate_descent(a, cd_deadline, max_passes=3)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-12:
                best_a = a_cd.copy()
                best_J = J_cd
                a = a_cd
                no_improve = 0
        
        if no_improve > 50:
            # Perturb
            a = a + np.random.normal(0, 0.01, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0
        
        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: J={best_J:.8f}")
    
    return best_a, best_J

def symmetric_coordinate_descent(a, deadline, max_passes=30):
    """
    Coordinate descent exploiting symmetry: a[i] = a[n-1-i].
    Only optimize the first half; mirror to the second half.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    a = a / a.sum()
    n = len(a)
    half = (n + 1) // 2
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        # Enforce symmetry
        a_sym = (a + a[::-1].copy()) / 2
        a_sym = a_sym / a_sym.sum()
        
        order = np.random.permutation(half)
        b = fast_conv(a_sym)
        S = a_sym.sum()
        improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            # The mirror index
            i_m = n - 1 - i
            if i_m < half:
                # Both i and i_m are in the first half (only when n is odd and i is the middle)
                continue
            
            ai = a_sym[i]
            ai_m = a_sym[i_m]
            
            # When we change a_sym[i] by delta, a_sym[i_m] also changes by delta (symmetry)
            # So the change in b is:
            # b'[k] = b[k] + 2*delta*(a_sym[k-i] + a_sym[k-i_m]) + delta^2*(delta(k,2i) + delta(k,2*i_m))
            # (since both a[i] and a[i_m] change by delta)
            
            # Affected k range: union of [i, i+n-1] and [i_m, i_m+n-1] intersected with [0, 2n-2]
            k_lo1 = i
            k_hi1 = min(i + n - 1, 2 * n - 2)
            k_lo2 = i_m
            k_hi2 = min(i_m + n - 1, 2 * n - 2)
            k_lo = min(k_lo1, k_lo2)
            k_hi = max(k_hi1, k_hi2)
            
            # Unchanged max
            C = 0.0
            if k_lo > 0:
                C = max(C, b[:k_lo].max())
            if k_hi < 2 * n - 2:
                C = max(C, b[k_hi + 1:].max())
            
            # For the search, define the affected change
            # delta is in [-min(ai, ai_m), upper]
            lower = -min(ai, ai_m)
            upper = min(0.5 * S / n, S)
            
            if upper - lower < 1e-10:
                continue
            
            gr = (np.sqrt(5) - 1) / 2
            
            def J_of_delta(d):
                S_new = S + 2 * d  # Both a[i] and a[i_m] change by d
                if S_new < 0.01:
                    return float('inf')
                new_max = C
                for k in range(k_lo, k_hi + 1):
                    # Change in b[k] from changing a[i] by d:
                    change = 0.0
                    # From a[i] -> a[i]+d:
                    j1 = k - i
                    if 0 <= j1 < n:
                        change += 2 * d * a_sym[j1]
                        if k == 2 * i:
                            change += d * d
                    # From a[i_m] -> a[i_m]+d:
                    j2 = k - i_m
                    if 0 <= j2 < n:
                        change += 2 * d * a_sym[j2]
                        if k == 2 * i_m:
                            change += d * d
                    new_val = b[k] + change
                    if new_val > new_max:
                        new_max = new_val
                return 2 * n * new_max / (S_new * S_new)
            
            # Golden section
            lo = lower
            hi = upper
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = J_of_delta(c1)
            f2 = J_of_delta(c2)
            
            for _ in range(50):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = J_of_delta(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = J_of_delta(c2)
            
            opt_delta = (lo + hi) / 2
            new_val = max(ai + opt_delta, 0.0)
            new_val_m = max(ai_m + opt_delta, 0.0)
            
            if abs(new_val - ai) > 1e-12 or abs(new_val_m - ai_m) > 1e-12:
                a_sym[i] = new_val
                a_sym[i_m] = new_val_m
                improved = True
        
        # Renormalize
        a_sym = np.maximum(a_sym, 0)
        s_val = a_sym.sum()
        if s_val < 0.01:
            break
        a_sym = a_sym / s_val
        J = compute_J(a_sym)
        if J < best_J:
            best_J = J
            best_a = a_sym.copy()
        
        if not improved:
            break
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()

    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

= 4n * [a[k-i]*S + a[k-i]*delta - b[k] - 2*delta*a[k-i]] / (S+delta)^3
= 4n * [a[k-i]*S - b[k] - delta*a[k-i]] / (S+delta)^3

Setting to zero: a[k-i]*S - b[k] - delta*a[k-i] = 0
delta* = S - b[k]/a[k-i]

So the optimal delta for constraint k is delta_k* = S - b[k]/a[k-i].

For constraint k = 2i: f(delta) = 2n*(b[2i] + 2*delta*a[i] + delta^2) / (S+delta)^2
This is a rational function. The optimal is found by solving the derivative = 0, which gives a quadratic.

For the search bounds, delta* = S - b[k]/a[k-i] can be negative (if b[k] > S*a[k-i]) or positive. Since we need delta >= -a[i] (to keep a[i] >= 0), the lower bound is -a[i].

For the upper bound, I'll use a heuristic: the maximum of delta_k* over all affected k, clipped to something reasonable.

Actually, I realize there might be an issue with my golden section search. The function J_of_delta is the max of several functions, each of which is unimodal. The max of unimodal functions is not necessarily unimodal. However, in practice, for this problem, it's usually close to unimodal.

A safer approach would be to do a coarse grid search first to bracket the minimum, then do golden section within that bracket.
