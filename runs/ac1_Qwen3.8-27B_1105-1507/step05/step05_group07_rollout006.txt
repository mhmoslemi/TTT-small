def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=5.0):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, verbose=True):
    """LP mirror descent with adaptive blending."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    no_improve = 0
    lp_times = []

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_conv = fast_conv(a)
        M = float(b_conv.max())

        # Adaptive time limit: shorter as we approach deadline
        remaining = deadline - time.monotonic()
        avg_lp = np.mean(lp_times[-10:]) if lp_times else 0.1
        tl = max(0.5, min(10.0, remaining * 0.1))
        if avg_lp > remaining * 0.3:
            break

        t0 = time.monotonic()
        g = lp_best_response(a, M, time_limit=tl)
        lp_time = time.monotonic() - t0
        lp_times.append(lp_time)

        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive blending: start aggressive, decay
        base_t = 0.3 * (1.0 + 0.5 * np.cos(np.pi * it / max(1, max_iters)))
        base_t = max(0.05, min(0.4, base_t))

        improved = False
        for t in [base_t, base_t * 0.7, base_t * 0.5, base_t * 0.3, 0.05, 0.02]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-10):
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 40:
                break
        else:
            no_improve = 0

        if verbose and (it % 20 == 0 or improved):
            J_val = 2.0 * n * best_M
            print(f"    LP iter {it}: M={best_M:.8f} J={J_val:.8f} t={lp_time:.2f}s", flush=True)

    return best_a


def gradient_directed_coord_descent(a, max_sweeps, deadline, seed=0):
    """
    Coordinate descent where coordinates are visited in order of their
    contribution to the convolution peak. Uses finer candidate grid and
    gradient information for direction.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve_sweeps = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Find peak and compute gradient-based priority
        peak_idx = conv.argmax()
        # Gradient of (a*a)_{peak} w.r.t. a[j] = 2*a[peak-j] if valid
        grad_priority = np.zeros(n)
        for j in range(n):
            k_minus_j = peak_idx - j
            if 0 <= k_minus_j < n:
                grad_priority[j] = 2.0 * a[k_minus_j]

        # Visit in order of decreasing priority (shuffled for diversity)
        if sweep % 5 == 0:
            order = np.argsort(-grad_priority)
            # Partial shuffle
            shuffle_range = min(n, 64)
            perm = rng.permutation(shuffle_range)
            order[:shuffle_range] = order[perm]
        else:
            order = range(n)

        sweep_improved = False

        for i in order:
            old = a[i]

            # Generate candidates: finer grid
            if old < 1e-8:
                candidates = [0.0, 1e-4, 1e-3, 5e-3, 0.01, 0.02]
            else:
                # Finer grid + gradient-directed candidates
                ratios = np.array([0.0, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
                candidates = ratios * old
                # Add gradient-suggested value: reduce if contributes to peak
                if grad_priority[i] > 0:
                    candidates.append(old * 0.6)  # More aggressive reduction
                candidates = list(set(round(float(c), 12) for c in candidates))
                candidates = [c for c in candidates if c >= 0 and not np.isclose(c, old, rtol=1e-4)]

            if len(candidates) == 0:
                continue

            best_val = old
            best_local_obj = 2.0 * n * conv.max() / (s * s)

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2
            idx_in_slice = i - lo

            for x in candidates:
                x = float(x)
                if x < 0:
                    x = 0.0
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                # Apply
                a_slice[idx_in_slice] = x
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local_obj:
                    best_local_obj = new_obj
                    best_val = x

                # Rollback
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
                a_slice[idx_in_slice] = old

            if best_val != old and abs(best_val - old) > 1e-12:
                delta = best_val - old
                a_slice[idx_in_slice] = best_val
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                sweep_improved = True

        current_M = conv.max()
        current_obj = 2.0 * n * current_M / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

        if no_improve_sweeps > 50:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def subgradient_descent(a, max_iters, deadline, lr_init=0.2):
    """Subgradient descent with averaging for max(conv)."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lr = lr_init / (1 + it) ** 0.5
        b = fast_conv(a)
        M = b.max()
        tol = 0.01 * M
        active = np.where(b >= M - tol)[0]

        # Subgradient: average over active peaks
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # grad[j] += 2 * a[k-j] for j in [lo, hi]
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * a[k - j]
        grad /= len(active)

        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad /= gnorm

        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        running_sum += a_new
        count += 1
        a = a_new

    if count > 20:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0.0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return best_a


def freq_domain_init(n, rng, parent=None):
    """Initialize using frequency domain flat-spectrum idea."""
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []

    if parent is not None:
        p = parent / parent.sum()
        inits.append(p.copy())

    # Flat magnitude spectrum with random phase
    for _ in range(3):
        freq = np.arange(0, n)
        phase = rng.uniform(0, 2 * np.pi, n)
        if n % 2 == 0:
            phase[0] = 0
            phase[n // 2] = 0
        A = np.exp(1j * phase)
        a = np.fft.ifft(A).real
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            inits.append(a)

    # Cosine chirp (quadratic phase)
    for freq_scale in [1, 2, 3, 5]:
        phase = np.pi * freq_scale * (t ** 2)
        A = np.exp(1j * 2 * np.pi * freq_scale * t)
        a = np.fft.ifft(A).real
        a = np.maximum(a + 0.5, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            inits.append(a)

    # Gaussian-modulated carriers
    for center in [0.2, 0.5, 0.8]:
        for width in [0.1, 0.2, 0.3]:
            a = np.exp(-((t - center) / width) ** 2)
            a = np.maximum(a, 0)
            a /= a.sum()
            inits.append(a)

    # Random smooth
    for _ in range(3):
        ksize = max(3, n // 32)
        kernel = np.ones(ksize) / ksize
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        a = np.maximum(smooth + 0.1, 0)
        a /= a.sum()
        inits.append(a)

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_reoptimize(a, sigma_frac, deadline, seed=0):
    """Smooth the sequence and re-optimize - helps escape local minima."""
    n = len(a)
    ksize = max(3, int(n * sigma_frac))
    if ksize % 2 == 0:
        ksize += 1
    kernel = np.exp(-0.5 * (np.arange(ksize) - ksize // 2) ** 2 / (ksize / 4) ** 2)
    kernel /= kernel.sum()
    a_smooth = np.convolve(a, kernel, mode='same')
    a_smooth = np.maximum(a_smooth, 0)
    s = a_smooth.sum()
    if s > 1e-10:
        a_smooth /= s
    a_opt = gradient_directed_coord_descent(a_smooth, max_sweeps=500,
                                             deadline=deadline, seed=seed)
    return a_opt

        elapsed = time.monotonic() - t0
        J_val = compute_J(a_opt)
        print(f"  n={new_n}: J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
        tracker.update(a_opt)

    tracker.report("Phase 3")

    # === PHASE 4: Subgradient + smoothing restarts ===
    p4_deadline = time.monotonic() + min(200, (deadline - time.monotonic()) * 0.2)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Subgradient + Smoothing", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        # Subgradient
        sg_deadline = min(time.monotonic() + 50, p4_deadline - 50)
        a_sg = subgradient_descent(tracker.best, max_iters=5000,
                                   deadline=sg_deadline, lr_init=0.15)
        J_val = compute_J(a_sg)
        print(f"  Subgradient: J={J_val:.8f}", flush=True)
        tracker.update(a_sg)

        # Smoothing restarts
        for sigma in [0.01, 0.02, 0.03]:
            if time.monotonic() > p4_deadline - 30:
                break
            t0 = time.monotonic()
            a_smooth = smooth_reoptimize(tracker.best, sigma,
                                         min(time.monotonic() + 40, p4_deadline - 20),
                                         seed=seed + int(sigma * 1000))
            J_val = compute_J(a_smooth)
            print(f"  Smooth({sigma}): J={J_val:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(a_smooth)

    tracker.report("Phase 4")

    # === PHASE 5: Final intensive LP + CD on best ===
    p5_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p5_deadline - 20:
        # Try LP again on the current best
        lp_time_budget = min(120, p5_deadline - time.monotonic() - 60)
        if lp_time_budget > 20:
            a_final_lp = lp_mirror_descent(tracker.best, max_iters=200,
                                           deadline=time.monotonic() + lp_time_budget,
                                           verbose=True)
            J_val = compute_J(a_final_lp)
            print(f"  Final LP: J={J_val:.8f}", flush=True)
            tracker.update(a_final_lp)

        # Final CD
        if time.monotonic() < p5_deadline - 10:
            a_final_cd = gradient_directed_coord_descent(tracker.best, max_sweeps=5000,
                                                         deadline=p5_deadline,
                                                         seed=seed + 9999)
            J_val = compute_J(a_final_cd)
            print(f"  Final CD: J={J_val:.8f}", flush=True)
            tracker.update(a_final_cd)

    tracker.report("Phase 5")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
