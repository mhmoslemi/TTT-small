3. a_new = max(a_new, 0)
4. a_new = a_new / sum(a_new)
5. If J(a_new) < J(a), accept; else reduce eta

This is O(n log n) per iteration (dominated by the FFT for the convolution).

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Accelerated coordinate descent with gradient-guided candidates
2. Projected gradient descent with backtracking line search
3. Peak-flattening heuristic for targeted improvement
4. Random perturbations + CD for global exploration
5. Multi-length exploration with resampling
Key improvements: smarter CD candidates (gradient + peak-flattening guided),
faster per-iteration (no LP), and more thorough exploration.
"""

import numpy as np
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * len(a) * b.max() / (s * s))


def smoothed_subgradient(a, tau=0.01):
    """Compute smoothed subgradient of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp((b - b_max) / tau)
    w = exp_b / exp_b.sum()
    
    # s[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # = 2 * conv(w, a_rev)[n-1+i]
    a_rev = a[::-1]
    size = 1
    while size < len(w) + n - 1:
        size <<= 1
    fw = np.fft.rfft(w, size)
    far = np.fft.rfft(a_rev, size)
    c = np.fft.irfft(fw * far, size)[:len(w) + n - 1]
    s = 2.0 * c[n-1:n-1+n]
    return s


def cd_pass(a, deadline, max_cands=10):
    """One pass of coordinate descent with smart candidates."""
    n = len(a)
    b = fast_conv(a).copy()
    S = a.sum()
    if S < 0.01:
        return a, float('inf')
    
    current_J = 2.0 * n * b.max() / (S * S)
    
    for i in range(n):
        if time.monotonic() > deadline:
            break
        
        ai = a[i]
        
        # Generate candidates
        cands = set()
        cands.add(0.0)
        if ai > 1e-12:
            for mult in [0.25, 0.5, 0.75, 1.25, 1.5, 2.0, 3.0]:
                cands.add(ai * mult)
        else:
            for v in [0.005, 0.01, 0.02, 0.05]:
                cands.add(v)
        
        # Peak-flattening candidate
        k_star = b.argmax()
        if 0 <= k_star - i < n and k_star != 2 * i:
            # Try to reduce b[k_star]
            # b[k_star] new = b[k_star] + 2*(x-ai)*a[k_star-i]
            # We want b[k_star]_new = b[k_star] * 0.9 (reduce by 10%)
            a_ks = a[k_star - i]
            if a_ks > 1e-15:
                x_flat = ai - 0.1 * b[k_star] / (2.0 * a_ks)
                if x_flat > 0:
                    cands.add(x_flat)
        
        best_x = ai
        best_J = current_J
        
        for x in cands:
            x = max(0.0, x)
            if abs(x - ai) < 1e-14:
                continue
            delta = x - ai
            S_new = S + delta
            if S_new < 0.01:
                continue
            
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Outside max
            outside_max = 0.0
            if k_lo > 0:
                outside_max = max(outside_max, b[:k_lo].max())
            if k_hi < 2 * n - 2:
                outside_max = max(outside_max, b[k_hi + 1:].max())
            
            # Inside
            j_idx = np.arange(k_hi - k_lo + 1)
            b_mod = b[k_lo:k_hi + 1] + 2.0 * delta * a[j_idx]
            
            # Fix 2i
            if k_lo <= 2 * i <= k_hi:
                idx = 2 * i - k_lo
                b_mod[idx] = b[2 * i] + (x * x - ai * ai)
            
            inside_max = b_mod.max()
            new_max = max(outside_max, inside_max)
            J_cand = 2.0 * n * new_max / (S_new * S_new)
            
            if J_cand < best_J - 1e-15:
                best_J = J_cand
                best_x = x
        
        if best_x != ai:
            delta = best_x - ai
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            j_idx = np.arange(k_hi - k_lo + 1)
            b[k_lo:k_hi + 1] += 2.0 * delta * a[j_idx]
            if k_lo <= 2 * i <= k_hi:
                b[2 * i] += delta * delta
            a[i] = best_x
            S += delta
            current_J = best_J
        
        # Recompute b periodically to avoid drift
        if (i + 1) % 128 == 0:
            b = fast_conv(a).copy()
            S = a.sum()
    
    return a, current_J


def gradient_descent_pass(a, deadline, tau=0.05, base_lr=0.1):
    """Projected gradient descent with backtracking."""
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(500):
        if time.monotonic() > deadline:
            break
        
        s = smoothed_subgradient(a, tau)
        s_norm = np.linalg.norm(s)
        if s_norm < 1e-15:
            break
        
        eta = base_lr
        improved = False
        for _ in range(20):  # Backtracking
            a_try = a - eta * s
            a_try = np.maximum(a_try, 0.0)
            S_try = a_try.sum()
            if S_try < 0.01:
                eta *= 0.5
                continue
            a_try = a_try / S_try
            J_try = compute_J(a_try)
            if J_try < best_J - 1e-12:
                a = a_try
                best_J = J_try
                best_a = a_try.copy()
                improved = True
                break
            eta *= 0.5
        
        if not improved:
            # Try smaller tau
            tau *= 0.5
            if tau < 1e-6:
                break
        
    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
