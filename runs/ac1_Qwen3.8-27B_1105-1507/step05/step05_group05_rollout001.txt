    ks = np.arange(k_lo, k_hi + 1)
    ks_off = ks - i
    valid = (ks_off >= 0) & (ks_off < n)

    c = np.zeros(len(ks), dtype=np.float64)
    slope = np.zeros(len(ks), dtype=np.float64)
    is_quad = (ks == 2 * i)

    mask_nq = ~is_quad & valid
    if mask_nq.any():
        js = ks_off[mask_nq]
        c[mask_nq] = b[ks[mask_nq]] - 2.0 * x_cur * a[js]
        slope[mask_nq] = 2.0 * a[js]

    if is_quad.any():
        idx_q = np.where(is_quad)[0][0]
        c[idx_q] = b[ks[idx_q]] - x_cur * x_cur

    def obj(x):
        new_S = S - x_cur + x
        if new_S <= 1e-15:
            return 1e10
        vals = c + slope * x
        if is_quad.any():
            vals = vals.copy()
            vals[is_quad] = c[is_quad] + x * x
        m = vals.max()
        if base_max > m:
            m = base_max
        return m / (new_S * new_S)

    lo = 0.0
    hi = max(x_cur * 2.5, np.sqrt(max(b.max(), 0.0)) + 0.3, 0.05)
    f_mid = obj(hi * 0.5)
    f_hi = obj(hi)
    if f_hi < f_mid and hi < 20.0:
        hi *= 2.0

    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)

    for _ in range(45):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)

    x_opt = (lo + hi) * 0.5
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coord_descent(a, max_sweeps, deadline, seed=0):
    """Run coordinate descent with vectorized updates."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = len(a)
    a = a / s
    n = len(a)

    best_a = a.copy()
    b = fast_conv(a)
    best_M = b.max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        S = a.sum()
        improved = False

        for i in order:
            x_opt = coordinate_update_vec(a, i, b, S)
            if abs(x_opt - a[i]) > 1e-13:
                a[i] = x_opt
                S = S - x_prev + x_opt if False else a.sum()
                x_prev = a[i]
                # Recompute b incrementally: just recompute full conv (fast with FFT)
                b = fast_conv(a)
                improved = True

        s2 = a.sum()
        if s2 < 1e-10:
            break
        a = a / s2
        b = fast_conv(a)
        M = b.max()
        if M < best_M * (1 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 25:
            break

    return best_a


def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (A @ g)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, tol=1e-8):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'feasibility_tolerance': tol})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-14:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a, max_iters, deadline, use_geo=True, momentum_beta=0.0):
    """LP mirror descent with optional geometric mean and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    g_momentum = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Try slightly tighter M for more aggressive direction
        M_tight = M * 0.995
        g = lp_best_response(a, M_tight)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        # Momentum
        if momentum_beta > 0:
            g_momentum = momentum_beta * g_momentum + (1 - momentum_beta) * g
            g_momentum = g_momentum / g_momentum.sum()
            g_dir = g_momentum
        else:
            g_dir = g

        # Try step sizes
        improved = False
        t_candidates = [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
        for t in t_candidates:
            if use_geo and np.all(a > 1e-10) and np.all(g_dir > 1e-10):
                a_trial = a ** (1 - t) * g_dir ** t
            else:
                a_trial = (1 - t) * a + t * g_dir
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-13:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if it % 50 == 0:
            J = 2 * n * best_M / 1.0  # sum=1 so sum^2=1
            print(f"    LP it={it}, M={best_M:.10f}, J={J:.6f}", flush=True)

    return best_a


def run_subgradient(a, max_iters, deadline, lr0=0.005):
    """Subgradient method with Polyak-like step sizes."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_sum_sq = 1.0  # since sum=1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        k_star = int(b.argmax())

        # Subgradient of max_k (a*a)[k] at the active index k_star
        # d/d a_i [(a*a)[k_star]] = a[k_star - i] + (a[i - k_star] if i != k_star - i)
        # More precisely: (a*a)[k] = sum_j a_j * a[k-j]
        # d/d a_i = a[k-i] + a[i-k] (if both in range and i != k-i)
        #           = 2*a[k-i] (if i == k-i, i.e., 2i = k)
        #           = a[k-i] (if only one term is valid)
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            j2 = i - k_star
            val = 0.0
            if 0 <= j1 < n:
                val += a[j1]
            if 0 <= j2 < n:
                val += a[j2]
            grad[i] = val

        # Project gradient to tangent space (sum=0) and a>=0
        grad_mean = grad.mean()
        grad = grad - grad_mean

        # Polyak step: eta = (M - M_lower) / ||grad||^2
        M_lower = 1.0 / (2 * n - 1)  # theoretical lower bound on max conv when sum=1
        grad_sq = grad.dot(grad)
        if grad_sq < 1e-15:
            break
        eta = lr0 / (1.0 + 0.05 * it)
        # Also try Polyak
        eta_polyak = max(0.001, (M - M_lower) / (grad_sq + 1e-15))
        eta = max(eta, min(eta_polyak, 0.02))

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        b_new = fast_conv(a_new)
        M_new = b_new.max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        from_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        inits.append(np.maximum(from_interp, 0.01))
        inits.append(np.maximum(from_interp + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(from_interp + rng.normal(0, 0.1, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # U-shaped (high at edges)
    for alpha in [0.5, 0.8, 1.0, 1.2]:
        inits.append(np.maximum(1.0 + alpha * np.cos(np.pi * 2 * t), 0.01))

    # Cosine modes
    for freq in [1, 2, 3, 4, 6]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t), 0.01))
        inits.append(np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * freq * t), 0.01))

    # Gaussian bump
    inits.append(np.exp(-8 * (t - 0.5) ** 2) + 0.2)
    # Two bumps
    inits.append(np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.1)
    # Edge peaks
    inits.append(np.exp(-15 * t ** 2) + np.exp(-15 * (t - 1) ** 2) + 0.1)

    # Alternating (smoothed)
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    alt = (1.0 + 0.6 * (-1) ** np.arange(n)).astype(float)
    inits.append(np.maximum(np.convolve(alt, kernel, 'same') + 0.2, 0.01))

    # Random smooth
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: best J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Coordinate descent from diverse inits (multiple lengths)
    Phase 2: LP mirror descent with geometric mean steps
    Phase 3: Extended coordinate descent + periodic LP
    Phase 4: Final LP polish with momentum
    Phase 5: Length adaptation
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    # === PHASE 1: Coordinate Descent from diverse inits ===
    p1_deadline = time.monotonic() + 300
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Coordinate Descent (300s)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [256, 384, 512, 640]
    time_per_len = {256: 40, 384: 80, 512: 100, 640: 60}

    for n in lengths:
        phase_dl = min(time.monotonic() + time_per_len.get(n, 50), p1_deadline)
        if time.monotonic() >= phase_dl:
            continue
        t0 = time.monotonic()
        inits = diverse_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_for_n = None
        best_M = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            max_sw = max(10, int(2000 // max(1, n // 64)))
            a_opt = run_coord_descent(a0, max_sw, phase_dl, seed=seed + idx * 13)
            M_val = fast_conv(a_opt).max()
            if M_val < best_M:
                best_M = M_val
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n:4d}: M={best_M:.8f} J={J_val:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("Phase 1 done")

    # === PHASE 2: LP Mirror Descent ===
    p2_deadline = time.monotonic() + 400
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (400s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p2_deadline:
        a_lp = run_lp_mirror(tracker.best, 500, min(time.monotonic() + 250, p2_deadline - 80),
                             use_geo=True, momentum_beta=0.0)
        tracker.update(a_lp)

        # Second pass with momentum
        if time.monotonic() < p2_deadline - 60:
            a_lp2 = run_lp_mirror(a_lp, 300, p2_deadline,
                                  use_geo=True, momentum_beta=0.3)
            tracker.update(a_lp2)

    tracker.report("Phase 2 done")

    # === PHASE 3: Coordinate Descent + Periodic LP ===
    p3_deadline = time.monotonic() + 300
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: CD + Periodic LP (300s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p3_deadline:
        a_cur = tracker.best.copy()
        a_cur = a_cur / a_cur.sum()
        n_cur = a_cur.size

        # Interleave CD and LP
        chunk_cd = 40
        chunk_lp = time.monotonic() + 20
        while time.monotonic() < p3_deadline - 30:
            # CD chunk
            a_cur = run_coord_descent(a_cur, chunk_cd, min(time.monotonic() + 30, p3_deadline - 60),
                                      seed=rng.integers(100000))
            M_cur = fast_conv(a_cur).max()
            J_cur = 2 * n_cur * M_cur  # sum=1
            if J_cur < tracker.best_score - 1e-9:
                tracker.update(a_cur)

            # LP chunk (small)
            if time.monotonic() < p3_deadline - 30:
                a_lp_small = run_lp_mirror(a_cur, 30, min(time.monotonic() + 25, p3_deadline - 20),
                                           use_geo=True, momentum_beta=0.2)
                if fast_conv(a_lp_small).max() < M_cur:
                    a_cur = a_lp_small
                    tracker.update(a_cur)

    tracker.report("Phase 3 done")

    # === PHASE 4: Final LP Polish ===
    p4_deadline = time.monotonic() + 200
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final LP Polish (200s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p4_deadline:
        # Run LP with decreasing step concept (more iterations, momentum)
        a_polish = run_lp_mirror(tracker.best, 400, min(time.monotonic() + 100, p4_deadline - 50),
                                 use_geo=True, momentum_beta=0.4)
        tracker.update(a_polish)

        # Final CD
        if time.monotonic() < p4_deadline - 10:
            a_final_cd = run_coord_descent(a_polish, 50, p4_deadline - 5, seed=424242)
            tracker.update(a_final_cd)

    tracker.report("Phase 4 done")

    # === PHASE 5: Length Adaptation ===
    p5_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Length Adaptation (remaining)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p5_deadline - 30:
        n_best = tracker.best.size
        # Try nearby lengths
        for new_n in [n_best - 64, n_best + 64, n_best + 128, n_best * 2]:
            if new_n < 128 or new_n > 2048:
                continue
            if time.monotonic() > p5_deadline - 20:
                break
            t0 = time.monotonic()
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 1e-6)
            a_res = a_res / a_res.sum()
            a_opt = run_coord_descent(a_res, 30, min(time.monotonic() + 40, p5_deadline - 10),
                                      seed=88888)
            tracker.update(a_opt)
            print(f"  n={new_n}: J={compute_J(a_opt):.6f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
