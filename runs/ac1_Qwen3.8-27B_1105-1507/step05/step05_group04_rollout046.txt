def fast_conv(a):

def fast_cross_conv(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def smooth_max_grad_fft(a, tau):
    """
    Compute gradient of 2n * (1/tau) * log(sum_k exp(tau * R_k))
    where R = conv(a, a).

    n = len(a)
    R = fast_conv(a)
    
    # Compute softmax weights
    R_shifted = tau * R - tau * R.max()  # numerical stability
    exp_R = np.exp(R_shifted)
    p = exp_R / exp_R.sum()
    smooth_val = (R.max() + np.log(exp_R.sum()) / tau)

def lp_mirror_descent(a, deadline, verbose=False, top_frac=0.6, max_no_improve=200):

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append((p.copy(), 'parent'))
        inits.append((np.maximum(p + rng.normal(0, 0.1, n), 0.01), 'parent_pert'))
    
    inits.append((np.ones(n), 'uniform'))
    inits.append((np.maximum(0.5 + 0.5 * np.cos(np.pi * (2*t_arr - 1)), 0.01), 'cosine'))
    inits.append((np.maximum(np.exp(-3*(t_arr-0.5)**2), 0.01), 'gaussian_center'))
    
    # Multi-frequency
    for f in [2, 3, 7]:
        vals = np.maximum(1.0 + 0.4 * np.cos(2*np.pi*f*t_arr + rng.uniform(0, 2*np.pi)), 0.01)
        inits.append((vals, f'f{f}'))
    
    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append((np.maximum(smooth, 0.01), 'rand_smooth'))
    
    # Random with structure
    half = n // 2
    left = rng.uniform(0.2, 1.0, half)
    right = left[::-1]
    inits.append((np.concatenate([left, right]) if n % 2 == 0 else np.concatenate([left, [left[0]], right]), 'palindrome'))
    
    # Edge concentrated
    inits.append((np.maximum(np.exp(-2*t_arr**2) + np.exp(-2*(t_arr-1)**2) + 0.05, 0.01), 'edges'))
    
    # Sine wave
    inits.append((np.maximum(1.0 + np.sin(2*np.pi*t_arr), 0.01), 'sine1'))
    inits.append((np.maximum(1.0 + 0.7*np.sin(4*np.pi*t_arr + 0.5), 0.01), 'sine2'))
    
    return inits

conv[2] = sum_i a[i]*a[2-i] = a[0]*a[2] + a[1]*a[1] + a[2]*a[0] = 3 + 4 + 3 = 10.

Autocorrelation lag 0 = sum_i a[i]^2 = 1 + 4 + 9 = 14.

So they're different! The middle convolution value is sum a[i]*a[n-1-i], not sum a[i]^2.

But the gradient formula is the same: d(conv[k])/da[m] = 2*a[k-m] (when valid).

So my gradient computation is correct. The key difference from before is just in the interpretation: we're minimizing the peak of the convolution, not the autocorrelation.

Now, the sum of all convolution values is S^2, so the average is S^2/(2n-1). The minimum possible max is S^2/(2n-1) (if perfectly flat), giving J_min = 2n/(2n-1).

For the uniform sequence a=(1,...,1): conv = (1,2,...,n,n-1,...,1), max = n, S = n, J = 2n*n/n^2 = 2.

To get below 2, we need to make the convolution more flat. The uniform sequence has a triangular convolution. We need to "spread out" the energy differently.

One approach: make a sequence where a[i] is small at the edges and large in the center (or vice versa). Let's try a = (1, 2, 3, 2, 1) (palindrome, peak in center):
conv = (1, 4, 10, 12, 14, 12, 10, 4, 1)
max = 14, S = 9, J = 2*5*14/81 = 140/81 ≈ 1.728

Better than 2!

Wow, that's much better! So concentrating energy at the edges helps a lot!

Even better!

So the trend is: concentrating mass at the edges (making a "U-shaped" or "bimodal" sequence) gives lower J.

This makes sense: the convolution of a U-shaped sequence will be more flat because the "peaks" at the edges create a broad convolution pattern.
