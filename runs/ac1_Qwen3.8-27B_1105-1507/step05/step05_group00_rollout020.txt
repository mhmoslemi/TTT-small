"""
LP direction-finding method with smoothed gradient fallback for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses linearized QCQP direction finding via LP,
log-sum-exp smoothed gradient descent, and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_lp_matrices(a, c):
    """Build A_ub and b_ub for the direction-finding LP.
    
    LP: min M'
        s.t. 2*(C_k a).d + c[k] - M' <= 0  for k=0,...,2n-2
             sum(d) = 0
             d >= -a
    
    Variables: [d_0, ..., d_{n-1}, M']
    """
    n = len(a)
    m = 2 * n - 1
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for k in range(m):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            # (C_k a)_i = a[k-i] for i in [i_start, i_end]
            # k-i ranges from k-i_start to k-i_end
            indices = np.arange(i_start, i_end + 1)
            A_ub[k, indices] = 2.0 * a[k - indices]
        A_ub[k, n] = -1.0
        b_ub[k] = -c[k]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-a[i], None) for i in range(n)] + [(-np.inf, None)]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # Minimize M'
    
    return c_obj, A_ub, b_ub, A_eq, b_eq, bounds


def solve_direction_lp(a, c, M):
    """Solve LP to find descent direction d and new M'.
    
    Returns (d, M_prime) or (None, None) on failure.
    """
    n = len(a)
    c_obj, A_ub, b_ub, A_eq, b_eq, bounds = build_lp_matrices(a, c)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            d = res.x[:n]
            M_prime = float(res.x[n])
            return d, M_prime
    except Exception:
        pass
    return None, None


def line_search(a, d, obj_current, max_steps=20):

    feasible_t = np.inf
    for i in range(n):

    feasible_t = min(feasible_t, 1.0)  # Cap at 1
    
    # Try several step sizes
    candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]

            stall = 0
            a_new = smoothed_gradient_descent(a, time_budget=min(remaining * 0.1, 10.0),
                                              max_iters=int(min(remaining * 0.1, 10.0) * 20))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1000) + it)
                scale = 0.02 + 0.01 * min(stall // 10, 5)
                a_pert = a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a = a_pert / s_p
    
    return best_a


def coordinate_descent_polish(a, time_budget=60.0):
    """Coordinate descent: try pairwise mass transfers to reduce peak."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    
    for it in range(500):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        k_star = int(np.argmax(conv))
        
        # Try moving mass: for pairs (i, j), move small amount from i to j
        # Focus on indices that contribute to the peak
        peak_indices = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                peak_indices.append(i)
        
        # Shuffle for diversity
        rng = np.random.default_rng(it)
        rng.shuffle(peak_indices)
        
        for i in peak_indices[:30]:  # Try top 30 contributors
            if time.monotonic() - start > time_budget:
                break
            j = k_star - i
            if j < 0 or j >= n or j == i:
                continue
            
            # Try moving mass from i to a random position far from peak
            for target in [0, n - 1, (n + 1) // 2]:
                if target == i:
                    continue
                for frac in [0.01, 0.02, 0.05]:
                    delta = frac * a[i]
                    if delta < 1e-15:
                        continue
                    a[i] -= delta
                    a[target] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * np.max(new_conv) / (np.sum(a) ** 2)
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        improved = True
                    else:
                        a[i] += delta
                        a[target] -= delta
                        break
        
        if not improved and it % 50 == 0:
            # Try random pairs
            for _ in range(10):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                for frac in [0.01, 0.05]:
                    delta = frac * max(a[i], a[j])
                    src, dst = (i, j) if a[i] > a[j] else (j, i)
                    if a[src] < delta:
                        continue
                    a[src] -= delta
                    a[dst] += delta
                    new_conv = compute_conv(a)
                    new_obj = 2.0 * n * np.max(new_conv) / (np.sum(a) ** 2)
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                    else:
                        a[src] += delta
                        a[dst] -= delta
                        break
    
    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP direction-finding (linearized QCQP) as the
    primary optimization method, with smoothed gradient fallback, coordinate
    descent polishing, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    if best_seq is None:
        # Fallback: use a structured initial guess
        n_init = 500
        # Use a raised cosine window (reduces edge effects in convolution)
        x = np.linspace(0, 2 * np.pi, n_init)
        best_seq = 0.5 + 0.5 * np.cos(x)
        best_score = compute_obj(best_seq)
        print(f"Using fallback initial: n={n_init}, obj={best_score:.6f}")
    
    n_parent = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n={n_parent}")
    
    # === Phase 1: LP direction-finding optimization ===
    print("\n=== Phase 1: LP Direction Finding ===")
    phase1_end = start_time + min(800.0, (deadline - start_time) * 0.55)
    
    a = best_seq.copy()
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    
    # Run LP optimization in chunks to allow monitoring
    chunk_size = 200.0
    current = start_time
    while current < phase1_end:
        chunk_end = min(phase1_end, current + chunk_size)
        alloc = chunk_end - current
        if alloc < 10:
            break
        
        a_new = lp_direction_optimize(a, time_budget=alloc, max_iters=200)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            a = a_new.copy()
            print(f"  LP: {best_score:.8f}")
        else:
            # Try smoothed gradient from best
            a_sg = smoothed_gradient_descent(best_seq, time_budget=alloc * 0.3,
                                             max_iters=int(alloc * 0.3 * 15))
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                a = a_sg.copy()
                print(f"  SG: {best_score:.8f}")
            else:
                a = a_new  # Still use the LP result as current
        
        current = time.monotonic()
    
    # === Phase 2: Coordinate descent polishing ===
    print(f"\n=== Phase 2: Coordinate Descent Polish ===")
    print(f"Best before CD: {best_score:.8f}")
    
    if time.monotonic() < deadline - 60:
        cd_budget = min(120.0, deadline - time.monotonic() - 60)
        a_cd = coordinate_descent_polish(best_seq, time_budget=cd_budget)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  CD: {best_score:.8f}")
        else:
            print(f"  CD: {obj_cd:.8f} (no improvement)")
    
    # === Phase 3: Additional LP rounds on polished sequence ===
    print(f"\n=== Phase 3: Post-CD LP ===")
    if time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic() - 40
        alloc = min(remaining * 0.5, 150.0)
        a_lp2 = lp_direction_optimize(best_seq, time_budget=alloc, max_iters=100)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  Post-CD LP: {best_score:.8f}")
    
    # === Phase 4: Try nearby sequence lengths ===
    print(f"\n=== Phase 4: Length Exploration ===")
    if time.monotonic() < deadline - 30:
        n_options = [n_parent - 50, n_parent + 50, n_parent + 100]
        n_options = [n for n in n_options if 200 <= n <= 2000]
        
        for n_try in n_options:
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic() - 20
            if remaining < 15:
                break
            
            a_res = resample_sequence(best_seq, n_try)
            s_res = np.sum(a_res)
            if s_res > 1e-10:
                a_res = a_res / s_res
            
            # Quick optimization
            alloc = min(remaining * 0.4, 60.0)
            a_opt = lp_direction_optimize(a_res, time_budget=alloc, max_iters=50)
            obj_opt = compute_obj(a_opt)
            
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: {best_score:.8f} ** UPDATE **")
            else:
                print(f"  n={n_try}: {obj_opt:.8f}")
    
    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 10
        
        # Multiple short polishing rounds
        for round_idx in range(10):
            if time.monotonic() > deadline - 5:
                break
            r = deadline - time.monotonic() - 5
            if r < 5:
                break
            
            alloc = min(r * 0.15, 30.0)
            
            # Alternate between LP and smoothed gradient
            if round_idx % 2 == 0:
                a_new = lp_direction_optimize(best_seq, time_budget=alloc, max_iters=30)
            else:
                a_new = smoothed_gradient_descent(best_seq, time_budget=alloc,
                                                  max_iters=int(alloc * 20))
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Round {round_idx}: {best_score:.8f}")
        
        # Final CD
        if time.monotonic() < deadline - 5:
            alloc = min(deadline - time.monotonic() - 5, 30.0)
            if alloc > 5:
                a_cd2 = coordinate_descent_polish(best_seq, time_budget=alloc)
                obj_cd2 = compute_obj(a_cd2)
                if obj_cd2 < best_score:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    print(f"  Final CD: {best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
