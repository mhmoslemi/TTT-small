def fft_conv(a):

    n = len(a)
    size = 1
    while size < 2 * n - 1:

    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fft_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

                        bi = ai
                        f_bi = f_ai
                        ai = hi_i - gr * (hi_i - lo_i)
                        f_ai, _ = obj_1d_outer(ai)
                    else:
                        lo_i = ai
                        ai = bi
                        f_ai = f_bi
                        bi = lo_i + gr * (hi_i - lo_i)
                        f_bi, _ = obj_1d_outer(bi)
                
                xi_opt = (lo_i + hi_i) / 2
                _, xj_opt = obj_1d_outer(xi_opt)
                
                if abs(xi_opt - xi_cur) > 1e-8 or abs(xj_opt - xj_cur) > 1e-8:
                    self.a[i] = max(0, xi_opt)
                    self.a[j] = max(0, xj_opt)
                    self.recompute_b()
                    self.update_best()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                # For larger blocks, just do sequential CD within the block
                for i in block:
                    x_opt = self.coordinate_update(i)
                    if abs(x_opt - self.a[i]) > 1e-10:
                        self.a[i] = x_opt
                self.recompute_b()
                self.update_best()
                no_improve += 1  # Conservative
            
            if no_improve > 50:
                break
        
        return self.best.copy()
    
    def lp_step(self, time_limit=5):
        """One LP mirror descent step."""
        n = self.n
        b = self.b
        M = b.max()
        
        # Build constraint matrix: A[i,j] = a[k-j] where k=i, valid j
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(self.a[k - j])
        
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                       options={'time_limit': time_limit})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    # Line search
                    best_t = 0
                    best_M_trial = self.best_M
                    for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
                        a_trial = (1 - t) * self.a + t * g
                        a_trial = np.maximum(a_trial, 0)
                        s = a_trial.sum()
                        if s < 1e-10:
                            continue
                        a_trial = a_trial / s
                        M_trial = fft_conv(a_trial).max()
                        if M_trial < best_M_trial:
                            best_M_trial = M_trial
                            best_t = t
                    if best_t > 0:
                        self.a = (1 - best_t) * self.a + best_t * g
                        self.a = np.maximum(self.a, 0)
                        s = self.a.sum()
                        if s > 1e-10:
                            self.a = self.a / s
                        self.recompute_b()
                        self.update_best()
                        return True
        except Exception:
            pass
        return False


def resample(src, dest_n):
    """Resample sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_new = np.linspace(0, 1, dest_n)
    t_old = np.linspace(0, 1, src.size)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(np.maximum(p * (1 + rng.normal(0, 0.1, n)), 0.01))
    
    # Uniform
    inits.append(np.ones(n) * (1.0/n))
    
    # Sine perturbations
    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))
    
    # Random smooth
    ksize = max(3, n // 50)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Random (less smooth)
    ksize2 = max(2, n // 200)
    kernel2 = np.ones(ksize2) / ksize2
    smooth2 = np.convolve(rng.uniform(0, 1, n), kernel2, mode='same')
    inits.append(np.maximum(smooth2 + 0.1, 0.01))
    
    # Triangular
    inits.append(1.0 - 0.5 * np.abs(2 * t - 1))
    
    # Exponential
    inits.append(np.maximum(np.exp(-3 * (t - 0.5)**2) + 0.2, 0.01))
    
    # Normalize all
    normalized = []
    for a in inits:
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            normalized.append(a / s)
    return normalized

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
1. Multi-n coordinate descent with exact 1D subproblem solving via FFT-accelerated conv
2. Subgradient descent with adaptive learning rates
3. LP-based mirror descent (for n <= 768)
4. Random perturbations to escape local minima
5. Multi-resolution: optimize at multiple sequence lengths, keep the best
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fft_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coord_update_fast(a, i, b, x_cur=None):
    """
    Find optimal a[i] given all other coordinates fixed.
    Uses precomputed convolution b. Returns optimal x >= 0.
    """
    n = len(a)
    if x_cur is None:
        x_cur = a[i]
    S = a.sum()

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # Unaffected max
    base_max = 0.0
    if k_lo > 0:
        base_max = max(base_max, b[:k_lo].max())
    if k_hi < 2 * n - 2:
        base_max = max(base_max, b[k_hi + 1:].max())

    # Precompute for affected k values
    k_vals = np.arange(k_lo, k_hi + 1)
    c_arr = np.empty(len(k_vals))
    s_arr = np.empty(len(k_vals))
    quad_idx = -1

    for idx, k in enumerate(k_vals):
        if k == 2 * i:
            c_arr[idx] = b[k] - x_cur * x_cur
            s_arr[idx] = 0.0
            quad_idx = idx
        else:
            ji = k - i
            if 0 <= ji < n and ji != i:
                c_arr[idx] = b[k] - 2.0 * x_cur * a[ji]
                s_arr[idx] = 2.0 * a[ji]
            else:
                c_arr[idx] = b[k]
                s_arr[idx] = 0.0

    def obj(x):
        new_sum = S - x_cur + x
        if new_sum <= 1e-10:
            return 1e10
        m = base_max
        # Vectorized max of linear terms
        linear_max = np.max(c_arr + s_arr * x)
        if linear_max > m:
            m = linear_max
        if quad_idx >= 0:
            qv = c_arr[quad_idx] + x * x
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)

    # Golden section with warm start
    lo = 0.0
    hi = max(x_cur * 2.0 + 0.05, x_cur + 0.01, 0.01)

    gr = (np.sqrt(5) - 1) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)

    for _ in range(35):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)

    x_opt = (lo + hi) / 2.0

    # Check boundaries
    f0 = obj(0.0)
    if f0 < obj(x_opt):
        x_opt = 0.0
    if obj(hi) < obj(x_opt):
        x_opt = hi

    return max(0.0, x_opt)


def run_cd(a, deadline, max_sweeps=100, seed=0):
    """Run coordinate descent with periodic full convolution updates."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    b = fft_conv(a)
    best_a = a.copy()
    best_M = b.max()
    no_improve = 0

    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)

        sweep_changed = False
        for i in order:
            x_opt = coord_update_fast(a, i, b)
            if abs(x_opt - a[i]) > 1e-11:
                a[i] = x_opt
                sweep_changed = True

        # Recompute convolution every sweep
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        b = fft_conv(a)
        M = b.max()

        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_changed:
            no_improve += 5
        if no_improve > 15:
            break

    return best_a, best_M


def smooth_max_grad(a, b, tau=10.0):
    """Gradient of log-sum-exp approximation to max(conv)."""
    n = len(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        # grad[j] += 2 * p[k] * a[k-j] for j in [j_lo, j_hi]
        # Vectorized: for each j in range, a[k-j]
        indices = np.arange(j_lo, j_hi + 1)
        grad[indices] += 2.0 * pk * a[k - indices]

    return grad


def run_smooth_grad(a, deadline, max_iters=2000, tau0=5.0, tau1=80.0, lr0=0.05):
    """Adam on smooth max of convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_M = fft_conv(a).max()
    b = fft_conv(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau0 + (tau1 - tau0) * (frac ** 0.5)
        lr = lr0 / (1.0 + 0.05 * it)

        grad = smooth_max_grad(a, b, tau)
        # Project to sum-zero
        grad = grad - grad.mean()

        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        m_hat = m / (1.0 - 0.9 ** (it + 1))
        v_hat = v / (1.0 - 0.999 ** (it + 1))

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        if it % 10 == 0:
            b = fft_conv(a)
            M = b.max()
            if M < best_M:
                best_M = M
                best_a = a.copy()

    return best_a


def run_lp_mirror(a, deadline, max_iters=50, lp_time_limit=10):
    """LP mirror descent: solve LP for best response, then interpolate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fft_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fft_conv(a)
        M = b.max()

        # Build LP constraint matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])

        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'time_limit': lp_time_limit})
            if not r.success or r.x is None:
                no_improve += 1
                if no_improve > 5:
                    break
                continue

            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs < 1e-10:
                no_improve += 1
                if no_improve > 5:
                    break
                continue
            g = g / gs

            # Line search
            improved = False
            for t in [0.4, 0.25, 0.15, 0.1, 0.05, 0.03]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0)
                s2 = a_trial.sum()
                if s2 < 1e-10:
                    continue
                a_trial = a_trial / s2
                M_trial = fft_conv(a_trial).max()
                if M_trial < best_M - 1e-14:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 8:
                    break

        except Exception:
            no_improve += 1
            if no_improve > 5:
                break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.005))
        inits.append(np.maximum(p * (1.0 + rng.normal(0, 0.08, n)), 0.005))
        # Asymmetric perturbation
        shift = np.roll(p, rng.integers(1, max(2, n // 20)))
        inits.append(np.maximum(0.7 * p + 0.3 * shift, 0.005))

    # Uniform
    inits.append(np.ones(n) * (1.0 / n))

    # Cosine perturbations (different frequencies and phases)
    for freq in [1, 2, 3, 5, 7]:
        for phase in [0, np.pi / 2]:
            inits.append(np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * freq * t + phase), 0.01))

    # Smooth random
    for ksize_frac in [30, 100, 300]:
        ksize = max(2, n // ksize_frac)
        kernel = np.ones(ksize) / ksize
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.05, 0.01))

    # Triangular
    inits.append(1.0 - 0.4 * np.abs(2.0 * t - 1.0))

    # Gaussian bump
    inits.append(np.maximum(np.exp(-5.0 * (t - 0.5) ** 2) + 0.15, 0.01))

    #Normalize
    result = []
    for a in inits:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            result.append(a / s)
    return result


def random_perturb_optimize(a, deadline, seed=0, n_trials=100):
    """Try random perturbations of a, keeping improvements."""
    rng = np.random.default_rng(seed)
    n = len(a)
    best = a.copy()
    best_J = compute_J(best)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        # Random perturbation with decreasing amplitude
        amp = 0.05 * (1.0 - trial / n_trials) + 0.005
        a_pert = best + rng.normal(0, amp, n)
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        J_pert = compute_J(a_pert)
        if J_pert < best_J:
            best = a_pert.copy()
            best_J = J_pert

        # Also try small CD refinement from perturbation
        if trial % 10 == 5:
            a_cd, _ = run_cd(a_pert, min(time.monotonic() + 5, deadline), max_sweeps=5, seed=trial)
            J_cd = compute_J(a_cd)
            if J_cd < best_J:
                best = a_cd.copy()
                best_J = J_cd

    return best


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search:
    Phase 1: Quick CD at multiple n values (multi-start)
    Phase 2: Detailed CD + smooth gradient at best n
    Phase 3: LP mirror descent (if n small enough)
    Phase 4: Random perturbation escape + final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    best_a = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            ps = parent.sum()
            if ps > 0:
                parent = parent / ps
            best_a = parent.copy()
            best_score = compute_J(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = np.ones(512) / 512
        best_a = parent.copy()
        best_score = compute_J(parent)
        print(f"No parent found, using uniform: n=512, J={best_score:.6f}", flush=True)

    def try_update(a, label=""):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01 or a.size == 0:
            return
        score = compute_J(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # === PHASE 1: Multi-n, multi-start CD ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Multi-n Multi-start Coordinate Descent", flush=True)
    print(f"{'='*60}", flush=True)

    p1_deadline = t_start + 350.0

    # Choose n values to try
    n_base = best_a.size if best_a is not None else 512
    n_list = sorted(set([
        max(256, n_base // 2),
        n_base,
        int(n_base * 1.25),
        int(n_base * 1.5),
        int(n_base * 2),
    ]))
    n_list = [n for n in n_list if 256 <= n <= 2048]

    # Time allocation: more time for n near the parent
    total_time = p1_deadline - time.monotonic()
    weights = {n: 1.0 / (1.0 + abs(n - n_base) / n_base) for n in n_list}
    w_sum = sum(weights.values())
    time_alloc = {n: max(20.0, total_time * weights[n] / w_sum) for n in n_list}

    for n in n_list:
        remaining = p1_deadline - time.monotonic()
        if remaining < 20:
            break
        phase_dl = time.monotonic() + min(time_alloc[n], remaining)

        inits = generate_inits(n, rng, best_a)
        # Limit number of inits based on n
        max_inits = max(3, min(8, int(40000 / n)))
        inits = inits[:max_inits]

        best_M_n = float('inf')
        best_a_n = None

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            n_sweeps = max(5, min(40, 20000 // n))
            a_opt, M_opt = run_cd(a0, phase_dl, max_sweeps=n_sweeps, seed=seed + idx * 31)

            if M_opt < best_M_n:
                best_M_n = M_opt
                best_a_n = a_opt.copy()

        if best_a_n is not None:
            J_val = compute_J(best_a_n)
            print(f"  n={n:4d}: best J={J_val:.6f} (t={time.monotonic()-t_start:.0f}s)", flush=True)
            try_update(best_a_n, f"P1-n{n}")

    print(f"Phase 1 complete: J={best_score:.8f} t={time.monotonic()-t_start:.0f}s", flush=True)

    # === PHASE 2: Detailed optimization at best n ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: Detailed CD + Gradient at Best n", flush=True)
    print(f"{'='*60}", flush=True)

    p2_deadline = t_start + 750.0
    n_best = best_a.size if best_a is not None else 512

    # Extended CD from best
    a_work = best_a.copy()
    n_sweeps2 = max(20, min(100, 80000 // n_best))
    a_cd2, M_cd2 = run_cd(a_work, min(time.monotonic() + 200, p2_deadline - 100),
                          max_sweeps=n_sweeps2, seed=5000)
    try_update(a_cd2, "P2-CD")

    # Smooth gradient from best
    if time.monotonic() < p2_deadline - 50:
        a_grad = run_smooth_grad(best_a, min(time.monotonic() + 100, p2_deadline - 30),
                                 max_iters=1000, tau0=5.0, tau1=100.0, lr0=0.03)
        try_update(a_grad, "P2-Grad")

    # CD from gradient result
    if time.monotonic() < p2_deadline - 20:
        a_cd3, _ = run_cd(a_grad, p2_deadline - 10, max_sweeps=30, seed=9000)
        try_update(a_cd3, "P2-CD2")

    # Multiple random restarts from best with CD
    for restart in range(5):
        if time.monotonic() > p2_deadline - 10:
            break
        a_r = best_a + rng.normal(0, 0.02 + 0.01 * restart, best_a.size)
        a_r = np.maximum(a_r, 0.001)
        a_r = a_r / a_r.sum()
        a_r_cd, _ = run_cd(a_r, time.monotonic() + 30, max_sweeps=15, seed=20000 + restart)
        try_update(a_r_cd, f"P2-Restart{restart}")

    print(f"Phase 2 complete: J={best_score:.8f} t={time.monotonic()-t_start:.0f}s", flush=True)

    # === PHASE 3: LP Mirror Descent ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    p3_deadline = t_start + 1050.0

    if best_a is not None and best_a.size <= 768 and time.monotonic() < p3_deadline - 30:
        n_lp = best_a.size
        lp_tl = max(3, min(15, 3000 // n_lp))  # seconds per LP solve
        max_lp_iters = max(10, min(50, int(200 / (n_lp / 256))))

        a_lp = run_lp_mirror(best_a, p3_deadline - 10, max_iters=max_lp_iters, lp_time_limit=lp_tl)
        try_update(a_lp, "P3-LP")

        # CD polish after LP
        if time.monotonic() < p3_deadline - 5:
            a_lp_cd, _ = run_cd(a_lp, p3_deadline, max_sweeps=20, seed=30000)
            try_update(a_lp_cd, "P3-LP+CD")

    elif best_a is not None:
        # n too large for LP, do more CD
        print("  n too large for LP, doing extended CD instead", flush=True)
        a_ext, _ = run_cd(best_a, p3_deadline, max_sweeps=50, seed=31000)
        try_update(a_ext, "P3-CD")

    print(f"Phase 3 complete: J={best_score:.8f} t={time.monotonic()-t_start:.0f}s", flush=True)

    # === PHASE 4: Random perturbation escape + final polish ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Perturbation Escape + Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    # Random perturbations
    if best_a is not None and time.monotonic() < deadline - 30:
        a_pert = random_perturb_optimize(best_a, min(time.monotonic() + 100, deadline - 20),
                                         seed=seed + 999, n_trials=200)
        try_update(a_pert, "P4-Perturb")

    # Final CD
    if best_a is not None and time.monotonic() < deadline - 5:
        a_final, _ = run_cd(best_a, deadline - 2, max_sweeps=30, seed=77777)
        try_update(a_final, "P4-Final")

    # Try slightly different n if time permits
    if best_a is not None and time.monotonic() < deadline - 30:
        for mult in [1.5, 2.0]:
            if time.monotonic() > deadline - 20:
                break
            new_n = int(best_a.size * mult)
            if new_n > 2048 or new_n == best_a.size:
                continue
            a_ext = resample_seq(best_a, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()
            a_ext_opt, _ = run_cd(a_ext, deadline - 5, max_sweeps=15, seed=55555)
            try_update(a_ext_opt, f"P4-n{new_n}")

    # Final evaluation with the actual evaluate_sequence
    if best_a is not None:
        list_a = [float(x) for x in best_a]
        final_score = evaluate_sequence(list_a)
        print(f"\n{'='*60}", flush=True)
        print(f"FINAL: evaluate_sequence J={final_score:.8f}, n={best_a.size}", flush=True)
        print(f"  (internal J={best_score:.8f})", flush=True)
        print(f"  Total time: {time.monotonic()-t_start:.0f}s / {budget_s}s", flush=True)
        print(f"{'='*60}", flush=True)

    if best_a is None:
        best_a = np.ones(512) / 512

    return [float(x) for x in best_a]
