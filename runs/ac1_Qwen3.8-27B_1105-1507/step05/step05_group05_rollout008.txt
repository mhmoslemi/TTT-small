The function f(x') is the max of several functions:
- For unaffected k: g_k(x') = b[k] / (1 + x' - x)^2 (decreasing in x')
- For affected k (k = 2i): g_k(x') = (b[k] + (x'-x)^2 + 2*x*(x'-x)) / (1 + x' - x)^2
  = (b[2i] - x^2 + x'^2) / (1 + x' - x)^2
- For affected k (k = i+j, j != i, 0 <= j < n): g_k(x') = (b[k] + 2*x'[a[k-i]] - 2*x*a[k-i]) / (1 + x' - x)^2
  = (c_k + s_k * x') / (1 + x' - x)^2 where c_k = b[k] - 2*x*a[k-i], s_k = 2*a[k-i]

So f(x') = max of a quadratic / quadratic and linear / quadratic functions. This is a piecewise smooth function. We can find its minimum numerically.

The key insight is that the minimum of f(x') occurs either at:
1. x' = 0 (boundary)
2. A point where two of the g_k(x') functions cross (the "kink" points)
3. A point where the derivative of one of the g_k is zero (critical point)

For the linear/quadratic functions g_k(x') = (c_k + s_k*x') / (1 + x' - x)^2, the critical point is where:
d/dx' [(c + s*x') / (1 + x' - x)^2] = 0
(s*(1+x'-x)^2 - (c+s*x')*2*(1+x'-x)) / (1+x'-x)^4 = 0
s*(1+x'-x) - 2*(c+s*x') = 0
s + s*x' - s*x - 2c - 2s*x' = 0
-2c + s*(1 - x) - s*x' = 0
x' = (s*(1-x) - 2c) / s = (1-x) - 2c/s = (1-x) - 2*(b[k] - 2*x*a[k-i]) / (2*a[k-i])
= (1-x) - (b[k] - 2*x*a[k-i]) / a[k-i]
= (1-x) - b[k]/a[k-i] + 2*x
= 1 + x - b[k]/a[k-i]

For the quadratic/quadratic (k=2i case): g(x') = (b[2i] - x^2 + x'^2) / (1 + x' - x)^2
This is more complex to differentiate. Let u = x' - x (so x' = x + u):
g = (b[2i] - x^2 + (x+u)^2) / (1+u)^2 = (b[2i] + 2*x*u + u^2) / (1+u)^2

dg/du = [ (2x + 2u)(1+u)^2 - (b[2i] + 2xu + u^2)*2(1+u) ] / (1+u)^4
= [ (2x+2u)(1+u) - 2(b[2i] + 2xu + u^2) ] / (1+u)^3
= [ 2x + 2xu + 2u + 2u^2 - 2b[2i] - 4xu - 2u^2 ] / (1+u)^3
= [ 2x + 2u - 2xu - 2b[2i] ] / (1+u)^3
= 2[x + u - xu - b[2i]] / (1+u)^3

Setting to zero: x + u - xu = b[2i], i.e., x + u(1-x) = b[2i], so u = (b[2i] - x)/(1-x) (if x != 1).

OK so we can find all critical points and the kink points, then evaluate f at all of them to find the minimum. But there could be O(n) kink points (where two g_k's cross), which is a lot.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using vectorized coordinate descent
with annealed restarts, multi-start initialization, and multi-length exploration.
Key improvements over previous: (1) fully vectorized 1D subproblem solving,
(2) simulated annealing to escape local minima, (3) structured initializations
exploiting edge-heavy patterns, (4) adaptive sweep count based on convergence rate.

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    b = fast_conv(a)
    return float(2 * len(a) * b.max())


def coord_update_fast(a, i, b, base_max_unaffected):
    """
    Find optimal a[i] given all other coordinates.
    a is normalized (sum=1), b = conv(a,a), base_max_unaffected = max of b outside affected range.
    Returns optimal x >= 0.
    """
    n = len(a)
    x = a[i]
    
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Precompute affected b values and their structure
    ks = np.arange(k_lo, k_hi + 1)
    # For each k in affected range, the change in b[k] when a[i] goes from x to x':
    # If k == 2*i: delta = x'^2 - x^2 (quadratic)
    # If k != 2*i: delta = 2*x'*a[k-i] - 2*x*a[k-i] (linear in x')
    
    # Linear part: c_k + s_k * x'
    # c_k = b[k] - 2*x*a[k-i] (for k != 2i)
    # s_k = 2*a[k-i]
    
    js = ks - i  # j = k - i
    # Valid j are those in [0, n-1]
    valid = (js >= 0) & (js < n)
    
    a_j = np.where(valid, a[js[valid]], 0.0)  # a[k-i] for valid k
    
    c = b[ks].copy()
    s = np.zeros(len(ks))
    quad_k = -1  # index in ks where k == 2*i
    
    for idx in range(len(ks)):
        if ks[idx] == 2 * i:
            quad_k = idx
            c[idx] = b[ks[idx]] - x * x  # b[2i] - x^2 (the non-x^2 part)
            s[idx] = 0.0
        else:
            j = ks[idx] - i
            if 0 <= j < n:
                c[idx] = b[ks[idx]] - 2 * x * a[j]
                s[idx] = 2 * a[j]
            # else: c stays as b[k], s = 0 (no change)
    
    def f_x(xp):
        """Evaluate max of convolution after setting a[i]=xp, normalized."""
        S = 1.0 + xp - x  # new sum
        if S <= 0:
            return 1e10
        inv_S2 = 1.0 / (S * S)
        # Affected values
        vals = c + s * xp
        if quad_k >= 0:
            vals[quad_k] = c[quad_k] + xp * xp
        m = max(base_max_unaffected, vals.max())
        return m * inv_S2
    
    # Golden section search
    lo = 0.0
    hi = max(x * 3.0, 0.5 / n, 0.01)
    
    # Ensure hi is large enough
    while hi < 1.0 / n * 10 and hi < 1.0:
        hi *= 2.0
    hi = min(hi, 2.0 / n)  # cap at reasonable value
    
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = f_x(a_pt)
    fb = f_x(b_pt)
    
    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = f_x(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = f_x(b_pt)
    
    x_opt = (lo + hi) / 2.0
    
    # Check boundary
    f0 = f_x(0.0)
    f_opt = f_x(x_opt)
    if f0 < f_opt:
        x_opt = 0.0
    
    return max(0.0, x_opt)


def compute_base_max(a, i):
    """Compute max of conv(a,a) outside the affected range of coordinate i."""
    n = len(a)
    b = fast_conv(a)
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[k_lo:k_hi + 1] = False
    if mask.any():
        return b[mask].max()
    return 0.0


def run_coord_descent(a, max_sweeps, deadline, temperature=0.0, seed=0):
    """
    Run coordinate descent with optional annealing.
    a should be normalized (sum=1).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    T = temperature
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        # Compute full convolution for this sweep
        b = fast_conv(a)
        
        # Annealing: add noise to temperature schedule
        if temperature > 0:
            T = temperature * (0.95 ** sweep)
        
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[k_lo:k_hi + 1] = False
            base_max = b[mask].max() if mask.any() else 0.0
            
            x_opt = coord_update_fast(a, i, b, base_max)
            
            # Annealing: with probability proportional to T, accept a worse move
            if T > 1e-10:
                # Sometimes perturb instead of taking exact optimum
                if rng.random() < 0.1:
                    x_opt = max(0.0, x_opt + rng.normal(0, T * 2.0 / n))
            
            if abs(x_opt - a[i]) > 1e-14:
                a[i] = x_opt
                improved = True
        
        # Renormalize
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not improved:
            no_improve += 10
        if no_improve > 20:
            break
    
    return best_a


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n (normalized to sum=1)."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    def normalize(a):
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            return np.ones(n) / n
        return a / s
    
    if parent is not None:
        p = np.maximum(parent, 0)
        p = p / p.sum()
        # Resample to n if needed
        if p.size != n:
            t_old = np.linspace(0, 1, p.size)
            t_new = np.linspace(0, 1, n)
            p = np.interp(t_new, t_old, p)
        inits.append(normalize(p.copy()))
        inits.append(normalize(p + rng.normal(0, 0.02, n)))
        inits.append(normalize(p + rng.normal(0, 0.05, n)))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Edge-heavy (large at edges)
    edge = np.ones(n)
    edge[:n//4] = 2.0
    edge[-(n//4):] = 2.0
    inits.append(normalize(edge))
    
    # Strong edge-heavy
    edge2 = np.exp(-4 * (2*t - 1)**2 * 0 + 3*(t**2 + (1-t)**2))
    inits.append(normalize(edge2))
    
    # Gaussian (peak in middle)
    inits.append(normalize(np.exp(-20*(t-0.5)**2)))
    
    # Triangular
    inits.append(normalize(np.abs(2*t - 1)))
    
    # Inverted triangular (peak at edges)
    inits.append(normalize(1 - np.abs(2*t - 1)))
    
    # Cosine variations
    for freq in [1, 2, 3, 5, 8]:
        inits.append(normalize(np.maximum(1.0 + 0.5*np.cos(2*np.pi*freq*t), 0.01)))
    
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(normalize(smooth + 0.1))
    
    # Another smooth random
    smooth2 = np.convolve(rng.uniform(0, 1, n), np.ones(max(3, n//10))/max(3, n//10), mode='same')
    inits.append(normalize(smooth2 + 0.1))
    
    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
