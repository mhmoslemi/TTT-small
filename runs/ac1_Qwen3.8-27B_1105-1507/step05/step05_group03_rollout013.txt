            if f_new < f_old * (1.0 - 1e-12):
                accept = True
            elif anneal_temp > 1e-10:
                ratio = (f_old - f_new) / max(anneal_temp, 1e-15)
                if rng.random() < np.exp(-max(ratio, 0.0)):
                    accept = True
            
            if accept:
                delta = x_opt - old_ai
                if abs(delta) < 1e-15:
                    continue
                a[i] = x_opt
                total += delta
                
                # Update c: for k in [lo_k, hi_k], k != k_ii: c[k] += 2*delta*a[k-i]
                # For k = k_ii: c[k_ii] += x_opt^2 - old_ai^2
                if lo_k <= hi_k:
                    if 0 <= k_ii <= hi_k and k_ii >= lo_k:
                        # Exclude k_ii from the linear update
                        mask = np.arange(lo_k, hi_k + 1)
                        mask = mask[mask != k_ii]
                        if mask.size > 0:
                            c[mask] += 2.0 * delta * a[mask - i]
                    else:
                        c[lo_k:hi_k+1] += 2.0 * delta * a[lo_k-i:hi_k-i+1]
                
                if 0 <= k_ii <= 2 * n - 2:
                    c[k_ii] += x_opt * x_opt - old_ai * old_ai
                
                improved = True
        
        current_J = 2.0 * n * c.max() / (total * total)
        if current_J < best_J * (1.0 - 1e-11):
            best_J = current_J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 15:
            break
        
        if verbose and p % 200 == 0:
            print(f"      pass {p}: J={current_J:.10f} best={best_J:.10f} "
                  f"improved={improved}", flush=True)
    
    return best_a, best_J


def build_chirp(n, alpha=0.4, beta=1.0):
    """Create a chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    seq = 1.0 + alpha * np.cos(2.0 * np.pi * beta * t * t / n)
    return np.maximum(seq, 0.01)


def build_random(n, rng, kind='uniform'):
    """Create a random initialization."""
    if kind == 'uniform':
        return rng.uniform(0.1, 1.0, n)
    elif kind == 'binary':
        return rng.choice([0.0, 1.0], n)
    elif kind == 'exponential':
        return np.maximum(rng.exponential(0.5, n), 0.01)
    elif kind == 'bimodal':
        s = np.zeros(n)
        mask = rng.random(n) < 0.5
        s[mask] = rng.uniform(0.0, 0.3, mask.sum())
        s[~mask] = rng.uniform(0.7, 1.0, (~mask).sum())
        return s
    else:
        return rng.uniform(0.0, 1.0, n)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max of ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    return t, float(np.max(a_coef + b_coef * t + c_coef * t * t))


def lp_polish(a, deadline, max_iters=5000):
    """LP mirror descent polish."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0
    
    row_idx, col_idx = build_lp_matrix(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = float(fast_conv(a).max())
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        t, M_new = line_search_analytic(a, g)
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = float(fast_conv(a).max())
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 80:
            break
    
    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    1. Coordinate descent on parent
    2. Multiple structured initializations
    3. Multi-length exploration
    4. Annealed restarts
    5. LP polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s [{label}]", flush=True)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            try_update(parent, "parent")
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512)
        try_update(parent, "default")
    
    n_parent = parent.size
    
    # === Phase 1: Coordinate descent on parent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Coordinate Descent on parent (n={n_parent})", flush=True)
    print(f"{'='*60}", flush=True)
    
    phase1_dl = min(t_start + budget_s * 0.25, deadline)
    a_opt, J_opt = coordinate_descent(parent, max_passes=100000, deadline=phase1_dl,
                                       verbose=True, seed=seed)
    try_update(a_opt, "P1 CD parent")
    
    # === Phase 2: Multiple initializations at same length ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multiple Initializations (n={n_parent})", flush=True)
    print(f"{'='*60}", flush=True)
    
    phase2_dl = min(t_start + budget_s * 0.40, deadline)
    inits = []
    inits.append(('chirp1', build_chirp(n_parent, 0.4, 1.0)))
    inits.append(('chirp2', build_chirp(n_parent, 0.3, 0.7)))
    inits.append(('random1', build_random(n_parent, rng, 'uniform')))
    inits.append(('bimodal1', build_random(n_parent, rng, 'bimodal')))
    inits.append(('chirp3', build_chirp(n_parent, 0.5, 1.5)))
    inits.append(('random2', build_random(n_parent + rng.integers(-20, 20), rng, 'uniform')[:n_parent]))
    
    for name, init in inits:
        if time.monotonic() > phase2_dl:
            break
        sub_dl = min(time.monotonic() + (phase2_dl - time.monotonic()) / max(1, len(inits) - 1), phase2_dl)
        a_opt, J_opt = coordinate_descent(init, max_passes=50000, deadline=sub_dl,
                                           verbose=False, seed=seed + hash(name) % 10000)
        try_update(a_opt, f"P2 {name}")
    
    # === Phase 3: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    phase3_dl = min(t_start + budget_s * 0.65, deadline)
    
    if best_seq is not None:
        base_n = best_seq.size
        trial_lengths = []
        for mult in [0.75, 1.25, 1.5, 2.0, 2.5, 0.5]:
            n_trial = int(base_n * mult)
            if 128 <= n_trial <= 8192 and n_trial != base_n:
                trial_lengths.append(n_trial)
        trial_lengths = sorted(set(trial_lengths))
        
        remaining = phase3_dl - time.monotonic()
        time_per = max(remaining / max(1, len(trial_lengths)), 5.0)
        
        for n_trial in trial_lengths:
            if time.monotonic() > phase3_dl:
                break
            sub_dl = min(time.monotonic() + time_per, phase3_dl)
            a_init = resample_seq(best_seq, n_trial)
            a_init = np.maximum(a_init, 1e-4)
            a_opt, J_opt = coordinate_descent(a_init, max_passes=50000, deadline=sub_dl,
                                               verbose=False, seed=seed + n_trial)
            try_update(a_opt, f"P3 n={n_trial}")
    
    # === Phase 4: Annealed restarts from best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Annealed Restarts", flush=True)
    print(f"{'='*60}", flush=True)
    
    phase4_dl = min(t_start + budget_s * 0.82, deadline)
    
    if best_seq is not None:
        n_best = best_seq.size
        temp = 0.005
        num_restarts = 20
        remaining = phase4_dl - time.monotonic()
        time_per = max(remaining / max(1, num_restarts), 3.0)
        
        for r in range(num_restarts):
            if time.monotonic() > phase4_dl:
                break
            sub_dl = min(time.monotonic() + time_per, phase4_dl)
            
            a = best_seq.copy().astype(np.float64)
            noise_scale = temp * 0.01
            a = a + rng.normal(0, noise_scale, n_best)
            a = np.maximum(a, 1e-6)
            
            a_opt, J_opt = coordinate_descent(a, max_passes=20000, deadline=sub_dl,
                                               verbose=False, seed=seed + 1000 + r,
                                               anneal_temp=temp, anneal_cool=0.9)
            try_update(a_opt, f"P4 r={r} T={temp:.5f}")
            temp *= 0.85
    
    # === Phase 5: LP polish on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: LP Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_final = best_seq.size
        a = best_seq.copy()
        a = a / a.sum()
        a_opt = lp_polish(a, deadline, max_iters=3000)
        try_update(a_opt, "P5 LP")
        
        # Coordinate descent after LP
        if time.monotonic() < deadline - 10:
            a_opt2, J_opt2 = coordinate_descent(a_opt, max_passes=50000, deadline=deadline,
                                                 verbose=False, seed=seed + 9999)
            try_update(a_opt2, "P5 CD after LP")
    
    # === Phase 6: Final coordinate descent polish on absolute best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_opt, J_opt = coordinate_descent(best_seq, max_passes=100000, deadline=deadline,
                                           verbose=True, seed=seed + 7777)
        try_update(a_opt, "P6 final CD")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = compute_J(best_seq)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_seq]
