```python
"""
LP-based optimization with QP refinement and spectral shaping for minimizing
the normalized peak autocorrelation of non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    k = np.arange(m)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    mask = (idx >= 0) & (idx < n)
    T = np.zeros((m, n), dtype=np.float64)
    T[mask] = a[idx[mask]]
    return T

def solve_lp(a, M, relax=1.0):
    n = len(a)
    m = 2 * n - 1
    T = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g)), res
    except Exception:
        pass
    return None, 0.0, None

def normalize(a):
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lp_alternation(a, time_budget=100.0, max_relax=1.5):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    relax = 1.0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, S, _ = solve_lp(a, M, relax=relax)
        if g is None or S <= 1.0 + 1e-8:
            stall += 1
            if stall > 20:
                relax = min(relax + 0.02, max_relax)
                if relax >= max_relax:
                    # Perturb and reset
                    rng = np.random.default_rng(int(time.monotonic()*1000))
                    a = a * np.exp(rng.normal(0, 0.03, n))
                    a = np.maximum(a, 0)
                    a = normalize(a)
                    relax = 1.0
                    stall = 0
            continue
        
        # Line search
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.2]:
            a_try = (1-t)*a + t*g/S
            a_try = np.maximum(a_try, 0)
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-10:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                stall = 0
                relax = max(1.0, relax - 0.01)
                break
        
        if not improved:
            stall += 1
            if stall > 30:
                relax = min(relax + 0.05, max_relax)
    
    return best_a, best_obj

def qp_refine(a, time_budget=30.0, K=3):
    """Refine using quadratic subproblems for top K peaks."""
    from scipy.optimize import minimize
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        # Top K peaks
        top_k = np.argsort(conv)[::-1][:K]
        k_star = int(top_k[0])
        
        # Build D matrix for the active peak
        # D[k_star][i,j] = 1 if i+j = k_star (valid)
        # The quadratic form a^T D a = (a*a)[k_star]
        # Gradient: 2 * D @ a = 2 * (correlation vector)
        
        # Use projected gradient on the quadratic
        # Minimize a^T D_{k*} a subject to a >= 0, sum(a) = 1
        # Gradient = 2 * (D_{k*} @ a) = 2 * conv_with_a_at_k_star
        
        # The gradient of (a*a)[k] wrt a[j] = 2*a[k-j] if 0<=k-j<n else 0
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]
        
        # Projected gradient step: a_new = proj(a - alpha * grad / ||grad||^2)
        # where proj clips to [0,inf) and normalizes to sum=1
        
        alpha = 0.1
        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step or different peak
            alpha = alpha * 0.5
            if alpha < 0.001:
                # Perturb
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = normalize(a)
                alpha = 0.1
    
    return best_a, best_obj

def spectral_flatten(a, time_budget=10.0):
    """Try to flatten the spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(7)
    
    for _ in range(100):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a)
        mag = np.abs(fa)
```

```python
"""
LP-based optimization with multi-level peak constraints, QP refinement,
U-shaped initialization, and spectral shaping for minimizing the normalized
peak autocorrelation of non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize so sum = 1."""
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix T where T[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    m = 2 * n - 1
    k = np.arange(m)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    mask = (idx >= 0) & (idx < n)
    T = np.zeros((m, n), dtype=np.float64)
    T[mask] = a[idx[mask]]
    return T


def solve_lp(a, M, relax=1.0, top_k=None, epsilon=0.0):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0.
    Optionally tighten constraints for top_k peaks (multi-level).
    """
    n = len(a)
    m = 2 * n - 1
    T = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    if top_k is not None and len(top_k) > 0 and epsilon > 0:
        for k_idx in top_k[1:]:
            b_ub[k_idx] = M * (1.0 - epsilon) * relax
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g)), res
    except Exception:
        pass
    return None, 0.0, None


def lp_alternation(a, time_budget=200.0, max_relax=1.5, multi_level=True,

    """LP alternating optimization with multi-level peak constraints."""
    n = len(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    relax = 1.0
    rng = np.random.default_rng(12345)

    iter_count = 0
    while time.monotonic() - start < time_budget:
        iter_count += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        top_k = np.argsort(conv)[::-1][:K]

        if multi_level:
            g, S, _ = solve_lp(a, M, relax=relax, top_k=top_k, epsilon=epsilon)
        else:
            g, S, _ = solve_lp(a, M, relax=relax)

        if g is None or S <= 1.0 + 1e-8:
            stall += 1
            if stall > 15:

def qp_refine(a, time_budget=30.0, K=3):
    """
    Refine using subgradient steps targeting top K peaks.
    Uses the gradient of (a*a)[k] wrt a[J].
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    alpha = 0.1

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_k = np.argsort(conv)[::-1][:K]

        # Compute combined gradient for top peaks
        grad = np.zeros(n)
        weights = np.zeros(K)
        for i, k in enumerate(top_k):
            # Weight by how much this peak exceeds the mean
            mean_val = (np.sum(a) ** 2) / (2 * n - 1)
            excess = max(0, conv[k] - mean_val)
            weights[i] = excess

        total_w = np.sum(weights)
        if total_w < 1e-10:
            break

        for i, k in enumerate(top_k):
            w = weights[i] / total_w
            # Gradient of (a*a)[k] wrt a[j] = 2*a[k-j] if 0<=k-j<n
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * w * a[idx]

        # Projected gradient descent
        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            alpha = min(alpha * 1.1, 0.5)
        else:
            alpha *= 0.7
            if alpha < 0.005:
                # Perturb to escape
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 0)
                a = normalize(a)
                alpha = 0.1

    return best_a, best_obj


def spectral_flatten(a, time_budget=10.0):

    n = len(a)

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        # QP step
        if remaining > 5:
            a_qp, obj_qp = qp_refine(best_a, time_budget=min(3.0, remaining * 0.1))
            if obj_qp < best_obj - 1e-10:
                best_obj = obj_qp
                best_a = a_qp.copy()
                stall = 0
                continue

        # Peak perturbation
        if remaining > 5:
            a_pk, obj_pk = peak_target_perturb(best_a, time_budget=min(2.0, remaining * 0.05),
                                               num_peaks=3)
            if obj_pk < best_obj - 1e-10:
                best_obj = obj_pk
                best_a = a_pk.copy()
                stall = 0
                continue

        # Spectral
        if remaining > 5:
            a_sp, obj_sp = spectral_flatten(best_a, time_budget=min(2.0, remaining * 0.05))
            if obj_sp < best_obj - 1e-10:
                best_obj = obj_sp
                best_a = a_sp.copy()
                stall = 0
                continue

        # Random perturbation to escape
        stall += 1
        if stall > 5:
            scale = rng.uniform(0.02, 0.06)
            a_rand = best_a * np.exp(rng.normal(0, scale, n))
            a_rand = np.maximum(a_rand, 0)
            a_rand = normalize(a_rand)
            obj_rand = compute_obj(a_rand)
            if obj_rand < best_obj:
                best_obj = obj_rand
                best_a = a_rand.copy()
            stall = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-level LP alternation with QP refinement, spectral shaping,
    U-shaped initialization, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent: n={parent.size}, obj={obj:.6f}")
            best_seq = parent.copy()
            best_score = obj
    except Exception:
        pass

    # Phase 1: U-shaped initialization and quick LP
    print("\n=== Phase 1: Initialization ===")
    if best_seq is not None:
        n_parent = len(best_seq)
        n_candidates = sorted(set(
            [n_parent, n_parent - 50, n_parent + 50, n_parent + 100,
             n_parent + 200, 1000, 1200]
        ))
        n_candidates = [n for n in n_candidates if 100 <= n <= 1500]
    else:
        n_candidates = [600, 700, 800, 1000]

    for n in n_candidates:
        if time.monotonic() > deadline - 300:
            break
        for c in [0.3, 0.5, 0.8, 1.2]:
            for p in [2, 3, 4]:
                a_init = u_shape_init(n, c, p)
                # Quick LP (short budget)
                remaining = deadline - time.monotonic()
                budget = min(15.0, remaining * 0.05)
                if budget < 3:
                    break
                a_opt, obj_opt = lp_alternation(a_init, time_budget=budget,
                                                multi_level=True, K=5, epsilon=0.02)
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  n={n}, c={c}, p={p}: obj={obj_opt:.6f}")

    if best_seq is None:
        best_seq = u_shape_init(700, 0.5, 2)
        best_score = compute_obj(best_seq)
        print(f"  Fallback init: n={len(best_seq)}, obj={best_score:.6f}")

    # Phase 2: Intensive LP on current best
    print(f"\n=== Phase 2: Intensive LP (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.3, 300.0)
    a_opt, obj_opt = lp_alternation(best_seq, time_budget=alloc,
                                    multi_level=True, K=7, epsilon=0.03)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  After LP: obj={best_score:.6f}")

    # Phase 3: QP + Polish
    print(f"\n=== Phase 3: QP Refinement ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.15, 80.0)
    a_opt, obj_opt = qp_refine(best_seq, time_budget=alloc, K=5)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  After QP: obj={best_score:.6f}")

    # Phase 4: Aggressive polish (combines all methods)
    print(f"\n=== Phase 4: Aggressive Polish ===")
    remaining = deadline - time.monotonic()
    alloc = min(remaining * 0.25, 200.0)
    a_opt, obj_opt = aggressive_polish(best_seq, time_budget=alloc)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  After polish: obj={best_score:.6f}")

    # Phase 5: Try resampling to larger n
    print(f"\n=== Phase 5: Larger n Exploration ===")
    n_current = len(best_seq)
    for n_new in [n_current + 100, n_current + 200, n_current + 400]:
        if time.monotonic() > deadline - 60:
            break
        if n_new > 1500:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_resample = resample_sequence(best_seq, n_new)
        a_resample = normalize(a_resample)
        alloc = min(remaining * 0.3, 80.0)
        a_opt, obj_opt = lp_alternation(a_resample, time_budget=alloc,
                                        multi_level=True, K=5, epsilon=0.02)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  n={n_new}: obj={obj_opt:.6f} ** UPDATE **")
        else:
            print(f"  n={n_new}: obj={obj_opt:.6f} (no improvement)")

    # Phase 6: Final intensive polish
    print(f"\n=== Phase 6: Final Polish ===")
    print(f"  Best before: {best_score:.6f}")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        alloc = min(remaining * 0.8, 150.0)
        a_opt, obj_opt = aggressive_polish(best_seq, time_budget=alloc)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  Final polish: obj={best_score:.6f}")

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val < np.inf:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
```
