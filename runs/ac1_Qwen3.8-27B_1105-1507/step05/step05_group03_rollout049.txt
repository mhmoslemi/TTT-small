def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) without calling evaluate_sequence."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(a, M, row_idx, col_idx, time_limit=5):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max of ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(a_coef + b_coef * m1 + c_coef * m1 * m1))
        f2 = float(np.max(a_coef + b_coef * m2 + c_coef * m2 * m2))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def smooth_grad_fft(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*(a*a))) using FFT cross-correlation."""
    n = len(a)
    b = fast_conv(a)
    b_shifted = b - b_max
    if b_max < 0:
        b_max = 0.0
    exp_b = np.exp(tau * b_shifted)
    p = exp_b / exp_b.sum()
    # grad[j] = (2/tau) * sum_m p[j+m] * a[m] = (2/tau) * (p cross-corr a)[j]
    # Compute via FFT: (p cross-corr a)[j] = convolve(p, a_rev)[j + n - 1]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    p_pad = np.zeros(size)
    p_pad[:len(p)] = p
    a_rev = a[::-1]
    a_pad = np.zeros(size)
    a_pad[:n] = a_rev
    fa = np.fft.rfft(p_pad)
    fb = np.fft.rfft(a_pad)
    c = np.fft.irfft(fa * fb, size)
    # (p cross-corr a)[j] = c[j + n - 1]
    grad = (2.0 / tau) * c[n - 1:n - 1 + n]
    return grad

        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.7)
        lr = 0.5 * (0.02 / 0.5) ** frac
        grad = smooth_grad_fft(a, tau)
        v = 0.85 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)

def tap_equalization(a, n_taps=5, n_iters=50):
    """Reduce spread of top convolution taps via iterative adjustment."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_spread = float('inf')

    for it in range(n_iters):
        b = fast_conv(a)
        M = b.max()
        # Find top taps
        top_k = np.argsort(b)[::-1][:n_taps]
        threshold = b[top_k[-1]] * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            break

        # Compute exposure of each element to active taps
        exposure = np.zeros(n)
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                exposure[j] += b[k] * abs(a[k - j])

        # Normalize exposure
        if exposure.max() > 0:
            exposure_norm = exposure / exposure.max()
            # Elements with high exposure should be reduced, low exposure increased
            # Use a gentle adjustment
            delta = 0.02 * (1.0 - it / n_iters)
            target = np.ones(n) / n  # uniform target
            a_new = a + delta * (target - a * exposure_norm)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                b_new = fast_conv(a_new)
                spread_new = b_new.max() - b_new.min()
                if spread_new < best_spread:
                    best_spread = spread_new
                    best_a = a_new.copy()
                a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_structured_init(n, kind='raised_cosine'):
    """Create structured initial sequences."""
    if kind == 'raised_cosine':
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif kind == 'gaussian':
        t = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * t * t)
    elif kind == 'triangle':
        t = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2 * np.abs(t - 0.5))
    elif kind == 'tophat_smooth':
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        edge = max(1, n // 10)
        a[:edge] = np.linspace(0.3, 1, edge)
        a[-edge:] = np.linspace(1, 0.3, edge)
    elif kind == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    return np.maximum(a, 0.0)

        parent = make_structured_init(512, 'raised_cosine')
        tracker.update(parent, "(default)")

    # === PHASE 1: Multi-length structured initialization ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    # Determine candidate lengths
    base_n = tracker.best.size if tracker.best is not None else 512
    candidate_lengths = sorted(set([
        int(base_n * m) for m in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]
        if 64 <= int(base_n * m) <= 4096
    ]))
    # Also add some fixed beneficial lengths
    for fixed_n in [256, 384, 512, 640, 768]:
        if 64 <= fixed_n <= 4096:
            candidate_lengths.append(fixed_n)
    candidate_lengths = sorted(set(candidate_lengths))

    phase1_deadline = t_start + budget_s * 0.30
    best_init = None
    best_init_score = float('inf')

    for n_trial in candidate_lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"  Trying n={n_trial}...", flush=True)

        # Try multiple initializations
        for kind in ['raised_cosine', 'gaussian', 'uniform', 'tophat_smooth']:
            if time.monotonic() > phase1_deadline:
                break
            a_init = make_structured_init(n_trial, kind)
            a_init = a_init / a_init.sum()

            # Quick smooth gradient
            sub_dl = min(time.monotonic() + 20, phase1_deadline)
            a_opt = run_smooth_gradient(a_init, 500, sub_dl, initial_tau=10.0, final_tau=80.0)

            # Quick LP
            r_idx, c_idx = build_lp_matrix(n_trial)
            sub_dl2 = min(time.monotonic() + 30, phase1_deadline)
            a_opt = run_lp_mirror_descent(a_opt, 2000, sub_dl2, r_idx, c_idx,
                                          aggressive=True, time_limit=3)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_init_score:
                best_init_score = score
                best_init = a_opt.copy()
                print(f"    n={n_trial} {kind}: J={score:.8f}", flush=True)

            tracker.update(a_opt, f"(P1 n={n_trial} {kind})")

    # Use best initialization if found, else parent
    if best_init is not None:
        start_a = best_init
        start_n = best_init.size
    else:
        start_a = parent
        start_n = parent.size

    tracker.report("(after P1)")

    # === PHASE 2: Main LP mirror descent (aggressive) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (Aggressive) n={start_n}", flush=True)
    print(f"{'='*60}", flush=True)

    phase2_deadline = t_start + budget_s * 0.60
    r_idx, c_idx = build_lp_matrix(start_n)
    a = start_a.copy() / start_a.sum()
    a_opt = run_lp_mirror_descent(a, 50000, phase2_deadline, r_idx, c_idx,
                                  verbose=True, aggressive=True, time_limit=5)
    tracker.update(a_opt, "(P2 LP-MD aggr)")

    # === PHASE 3: Tap equalization + smooth polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Tap Equalization + Smooth Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_cur = tracker.best.size
        a_te = tap_equalization(tracker.best, n_taps=5, n_iters=100)
        tracker.update(a_te, "(P3 tap-eq)")

        # Smooth polish from tap-eq result
        a_sp = run_smooth_gradient(a_te, 2000, min(t_start + budget_s * 0.72, deadline),
                                   initial_tau=20.0, final_tau=300.0)
        tracker.update(a_sp, "(P3 smooth)")

        # LP after smooth
        r_cur, c_cur = build_lp_matrix(n_cur)
        a_lp = run_lp_mirror_descent(a_sp, 5000, min(t_start + budget_s * 0.78, deadline),
                                     r_cur, c_cur, aggressive=True, time_limit=3)
        tracker.update(a_lp, "(P3 LP after smooth)")

    # === PHASE 4: Length perturbation ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length Perturbation", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        perturb_lengths = []
        for delta in [-64, -32, 32, 64, 128]:
            n_p = n_best + delta
            if 64 <= n_p <= 4096 and n_p != n_best:
                perturb_lengths.append(n_p)

        phase4_deadline = min(t_start + budget_s * 0.88, deadline)
        for n_p in sorted(set(perturb_lengths)):
            if time.monotonic() > phase4_deadline:
                break
            a_r = resample_seq(tracker.best, n_p)
            a_r = np.maximum(a_r, 1e-8)
            a_r = a_r / a_r.sum()
            r_p, c_p = build_lp_matrix(n_p)
            sub_dl = min(time.monotonic() + 40, phase4_deadline)
            a_opt = run_lp_mirror_descent(a_r, 3000, sub_dl, r_p, c_p,
                                          aggressive=True, time_limit=4)
            tracker.update(a_opt, f"(P4 n={n_p})")

    # === PHASE 5: Perturbation restarts ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_b, c_b = build_lp_matrix(n_best)
        phase5_deadline = min(t_start + budget_s * 0.94, deadline)
        temperature = 0.05

        for r in range(30):
            if time.monotonic() > phase5_deadline:
                break
            a = tracker.best.copy().astype(np.float64)
            noise_scale = temperature * 0.01
            a = a + rng.normal(0, noise_scale, n_best)
            a = np.maximum(a, 1e-8)
            a = a / a.sum()

            sub_dl = min(time.monotonic() + 20, phase5_deadline)
            a = run_smooth_gradient(a, 150, sub_dl, initial_tau=15, final_tau=100)
            sub_dl2 = min(time.monotonic() + 35, phase5_deadline)
            a_opt = run_lp_mirror_descent(a, 2000, sub_dl2, r_b, c_b,
                                          aggressive=True, time_limit=3)
            tracker.update(a_opt, f"(P5 r={r})")
            temperature *= 0.85

    # === PHASE 6: Final extended LP polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)
        a = tracker.best.copy() / tracker.best.sum()
        a_opt = run_lp_mirror_descent(a, 50000, deadline, r_f, c_f,
                                      verbose=True, aggressive=False, time_limit=5)
        tracker.update(a_opt, "(P6 final)")

        # One more tap equalization + short LP
        a_te2 = tap_equalization(a_opt, n_taps=3, n_iters=50)
        tracker.update(a_te2, "(P6 final tap-eq)")
        a_lp2 = run_lp_mirror_descent(a_te2, 2000, deadline, r_f, c_f,
                                      aggressive=False, time_limit=3)
        tracker.update(a_lp2, "(P6 final LP)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
