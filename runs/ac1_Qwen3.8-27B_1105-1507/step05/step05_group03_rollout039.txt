We want to minimize the max of the ifft of |FFT(a)|^2. This is equivalent to minimizing the peak of the "power spectrum" after inverse FFT.

In the frequency domain, we're trying to make |FFT(a)|^2 as "flat" as possible (to minimize the peak after inverse FFT). This is related to "flat spectrum" design in signal processing.

The constraint is that a >= 0 in the time domain, which translates to a constraint on the inverse FFT being non-negative. This is a hard constraint in the frequency domain.

def fast_conv(a):
    """FFT-based self-convolution."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

Hmm, the Nesterov method with projection onto the simplex is a bit tricky. The standard Nesterov method works for unconstrained problems or problems with simple projections. The projection onto {x >= 0, sum(x) = 1} (the simplex) is:

```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u * np.arange(1, n+1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):

    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)

def cross_conv(a, b):
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):

        return float('inf')

def gradient_smooth_max(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    log_b = tau * (b - b_max)
    # Numerically stable softmax
    log_b -= log_b.max()
    exp_b = np.exp(log_b)
    p = exp_b / exp_b.sum()
    
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    cross_corr = np.fft.irfft(fa * np.conj(fp), size)[:n]
    
    grad = 2.0 * cross_corr
    return grad


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = 1 - u * np.arange(1, n + 1) > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.argmax(~cond)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def nesterov_smooth(a, max_iters, deadline, tau_start=5.0, tau_end=200.0):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    y = a.copy()
    beta = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, (it + 1) / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        step = 2.0 / (np.sqrt(it + 1) + 1.0)
        
        y = project_simplex(y)
        grad = gradient_smooth_max(y, tau)
        
        a_new = project_simplex(y - step * grad)
        
        # Update momentum
        y = a_new + ((beta / (beta + 1.0)) * (a_new - a))
        beta += 1.0
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
    
    return best_a


def line_search_analytic(a, g):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * B
    c_coef = C - 2.0 * B + A
    # Actually: ((1-t)a + tg)^2 = (1-t)^2 A + 2t(1-t) B_cross + t^2 C
    # where B_cross = cross_conv(a, g)
    # = (1 - 2t + t^2) A + (2t - 2t^2) B + t^2 C
    # = A + t(2B - 2A) + t^2(A - 2B + C)
    
    # Let me redo:
    # f(t) = ((1-t)a + tg) * ((1-t)a + tg)
    # = (1-t)^2 (a*a) + 2t(1-t)(a*g) + t^2 (g*g)
    # = (1-t)^2 A + 2t(1-t) B + t^2 C
    # where A = a*a, B = a*g, C = g*g
    
    # Expanding:
    # = (1 - 2t + t^2) A + (2t - 2t^2) B + t^2 C
    # = A + t(-2A + 2B) + t^2(A - 2B + C)
    
    a_coef = A
    b_coef = -2.0 * A + 2.0 * B
    c_coef = A - 2.0 * B + C
    
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val

def lp_mirror_descent(a, max_iters, deadline, top_k=None, lp_time=5, verbose=False):
    """LP mirror descent with optional reduced constraints."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = fast_conv(a).max()
        
        if top_k is not None and top_k < 2 * n - 1:
            g = solve_lp_reduced(a, M, top_k, lp_time)
        else:
            g = solve_lp_full(a, M, lp_time)
        
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        
        t, M_new = line_search_analytic(a, g)
        
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = fast_conv(a).max()
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            break
        
        if verbose and it % 100 == 0:
            print(f"    LP-MD iter {it}: M={M:.10f} best={best_M:.10f} t={t:.4f}", flush=True)
    
    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

1. Compute Ag = conv(a, g) - O(n log n)
2. Compute A^T max(0, Ag - M) - O(n log n)
3. Update: g = g * exp(-eta * A^T max(0, Ag - M)) - O(n)
4. Renormalize g to sum to 1

This gives a feasible direction (approximately) in O(n log n) per iteration, with O(log n / epsilon^2) total iterations. This is much faster than solving the LP exactly!

        excess = np.maximum(Ag - M, 0)
        # Compute A^T excess
        # (A^T excess)[j] = sum_k a[k-j] * excess[k]
        # = cross_correlation(a, excess)[j]

        g = np.maximum(g, 1e-30)
        g = g / g.sum()
    
    return g

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

2. LP mirror descent with multiplicative weights approximation and exact LP
3. Multi-length exploration with resampling
4. Perturbation restarts to escape local minima
5. Final extended polish
Key improvements over previous: MW-based fast LP approximation, Nesterov acceleration,
adaptive constraint reduction, better time allocation.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """FFT-based self-convolution."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """FFT-based cross-convolution."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gradient_smooth_max(a, tau):
    """Compute gradient of smooth max approximation of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted -= shifted.max()
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()

    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    cross_corr = np.fft.irfft(fa * np.conj(fp), size)[:n]

    grad = 2.0 * cross_corr
    return grad


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = 1.0 - u * np.arange(1, n + 1) > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.argmin(~cond)
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 1.0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, (it + 1) / max(1, max_iters * 0.8))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        step = 2.5 / (np.sqrt(it + 1) + 1.0)

        y = project_simplex(np.maximum(y, 0))
        grad = gradient_smooth_max(y, tau)

        a_new = project_simplex(y - step * grad)

        y = a_new + ((beta / (beta + 1.0)) * (a_new - a))
        beta += 1.0
        a_prev = a
        a = a_new

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000:
            break

    return best_a


def mw_lp_approx(a, M, n_iters=150, eta=None):
    """Approximate LP solution using multiplicative weights. Returns g (sum=1, >= 0)."""
    n = len(a)
    g = np.ones(n) / n

    if eta is None:
        eta = 2.0 / np.sqrt(n)

    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)

    for _ in range(n_iters):
        Ag = cross_conv(a, g)
        excess = np.maximum(Ag - M, 0)
        if excess.max() < 1e-14:
            break
        fe = np.fft.rfft(excess, size)
        A_t_excess = np.fft.irfft(fa * np.conj(fe), size)[:n]
        g = g * np.exp(-eta * A_t_excess)
        g = np.maximum(g, 1e-30)
        g = g / g.sum()

    return g


def solve_lp_full(a, M, time_limit=5):
    """Solve the full LP: max 1^T g s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def solve_lp_reduced(a, M, top_k, time_limit=5):
    """Solve LP using only top_k most active constraints."""
    n = len(a)
    b = fast_conv(a)
    top_positions = np.argsort(b)[::-1][:top_k]

    rows = []
    cols = []
    vals = []
    for i, k in enumerate(top_positions):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[k - j])

    A = csr_matrix((vals, (rows, cols)), shape=(len(top_positions), n))
    c = -np.ones(n)
    b_ub = np.full(len(top_positions), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max of ((1-t)a + tg)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = -2.0 * A + 2.0 * B
    c_coef = A - 2.0 * B + C

    lo = 0.0
    hi = 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def lp_mirror_descent(a, max_iters, deadline, mode="mw", top_k=None, lp_time=5,
                      mw_iters=150, verbose=False):
    """LP mirror descent. mode: 'mw' for multiplicative weights, 'lp' for exact, 'hybrid'."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    mw_it = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = fast_conv(a).max()

        if mode == "mw":
            g = mw_lp_approx(a, M, n_iters=mw_iters)
        elif mode == "lp":
            g = solve_lp_full(a, M, lp_time)
        elif mode == "hybrid":
            if it % 5 == 0:
                g = solve_lp_full(a, M, lp_time)
                if g is None:
                    g = mw_lp_approx(a, M, n_iters=mw_iters)
            else:
                g = mw_lp_approx(a, M, n_iters=mw_iters)
        else:
            g = mw_lp_approx(a, M, n_iters=mw_iters)

        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-9:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = fast_conv(a).max()
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > (80 if mode == "mw" else 40):
            break

        if verbose and it % 200 == 0:
            print(f"      iter {it}: M={M:.10f} best={best_M:.10f} t={t:.4f} mode={mode}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class BestTracker:
    """Tracks the best sequence found."""
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining Nesterov smooth descent, LP mirror descent
    (both MW-approximate and exact), multi-length exploration, and perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0
    tracker = BestTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    # === Phase 1: Nesterov smooth (50s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Nesterov Smooth (n={parent.size})", flush=True)
    print(f"{'='*60}", flush=True)
    a1 = parent.copy()
    a1 = a1 / a1.sum()
    dl1 = min(t_start + 50, deadline)
    a_opt = nesterov_smooth(a1, 100000, dl1, tau_start=3.0, tau_end=400.0)
    tracker.update(a_opt, "(P1 Nesterov)")

    # === Phase 2: LP-MD with MW approximation (400s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP-MD MW Approx (n={tracker.best.size})", flush=True)
    print(f"{'='*60}", flush=True)
    a2 = tracker.best.copy()
    a2 = a2 / a2.sum()
    dl2 = min(t_start + 450, deadline)
    a_opt = lp_mirror_descent(a2, 100000, dl2, mode="mw", mw_iters=200, verbose=True)
    tracker.update(a_opt, "(P2 MW)")

    # === Phase 3: LP-MD with exact LP (300s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP-MD Exact (n={tracker.best.size})", flush=True)
    print(f"{'='*60}", flush=True)
    a3 = tracker.best.copy()
    a3 = a3 / a3.sum()
    dl3 = min(t_start + 750, deadline)
    a_opt = lp_mirror_descent(a3, 5000, dl3, mode="lp", lp_time=6, verbose=True)
    tracker.update(a_opt, "(P3 Exact LP)")

    # === Phase 4: Multi-length exploration (300s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)
    if tracker.best is not None:
        base_n = tracker.best.size
        trial_ns = set()
        for mult in [0.5, 0.75, 1.25, 1.5, 2.0, 0.625, 1.875]:
            nt = int(base_n * mult)
            if 64 <= nt <= 4096 and nt != base_n:
                trial_ns.add(nt)
        for nt in [192, 256, 320, 384, 448, 512, 576, 640, 768, 896, 1024, 1280, 1536, 2048]:
            if 64 <= nt <= 4096 and nt != base_n:
                trial_ns.add(nt)
        trial_ns = sorted(trial_ns)

        dl4 = min(t_start + 1050, deadline)
        tried = 0
        for nt in trial_ns:
            if time.monotonic() > dl4:
                break
            if tried >= 12:
                break
            tried += 1
            a_init = resample_seq(tracker.best, nt)
            a_init = np.maximum(a_init, 1e-8)
            a_init = a_init / a_init.sum()

            sub_dl = min(time.monotonic() + 25, dl4)
            a_mid = nesterov_smooth(a_init, 5000, sub_dl, tau_start=5, tau_end=150)

            sub_dl2 = min(time.monotonic() + 45, dl4)
            a_opt = lp_mirror_descent(a_mid, 3000, sub_dl2, mode="hybrid", lp_time=4,
                                      mw_iters=150, verbose=False)
            tracker.update(a_opt, f"(P4 n={nt})")

    # === Phase 5: Perturbation restarts (150s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)
    if tracker.best is not None:
        n_best = tracker.best.size
        temperature = 0.05
        dl5 = min(t_start + 1200, deadline)
        for r in range(20):
            if time.monotonic() > dl5:
                break
            a = tracker.best.copy().astype(np.float64)
            noise_scale = temperature * 0.02
            a = a + rng.normal(0, noise_scale, n_best)
            a = np.maximum(a, 1e-8)
            a = a / a.sum()

            sub_dl = min(time.monotonic() + 10, dl5)

            tracker.update(a_opt, f"(P5 r={r} T={temperature:.4f})")

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final Polish (n={tracker.best.size})", flush=True)
    print(f"{'='*60}", flush=True)
    if tracker.best is not None:
        a6 = tracker.best.copy()
        a6 = a6 / a6.sum()

        # Nesterov round for potential escape
        sub_dl = min(time.monotonic() + 40, deadline)
        a6 = nesterov_smooth(a6, 5000, sub_dl, tau_start=20, tau_end=600)
        tracker.update(a6, "(P6 Nesterov)")

        # Extended hybrid LP
        sub_dl2 = min(time.monotonic() + 200, deadline)
        a_opt = lp_mirror_descent(a6, 50000, sub_dl2, mode="hybrid", lp_time=8,

        # Final Nesterov
        a_opt2 = nesterov_smooth(a_opt, 5000, deadline, tau_start=100, tau_end=2000)
        tracker.update(a_opt2, "(P6 Final Nesterov)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
