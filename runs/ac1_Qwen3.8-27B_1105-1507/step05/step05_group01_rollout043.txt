"""
Gradient-based optimization with FFT-accelerated smooth-max gradients,
combined with LP polishing and multi-length exploration for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_max_and_grad(a, beta):
    """Compute smooth max and its gradient using FFT cross-correlation.
    Returns (L, L_prime) where L = (1/beta)*logsumexp(beta*c) and
    L_prime[j] = dL/da[j]."""
    n = len(a)
    c = compute_conv(a)
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    p = exp_bc / exp_bc.sum()
    L = (bc_max + np.log(exp_bc.sum())) / beta

    # L_prime[j] = 2 * sum_i a[i] * p[i+j]  (cross-correlation)
    size = 1
    while size < 2 * n:
        size <<= 1
    A = np.fft.rfft(a, size)
    P = np.fft.rfft(p, size)
    corr = np.fft.irfft(np.conj(A) * P, size)
    L_prime = 2.0 * corr[:n]
    return L, L_prime, p, c


def full_gradient(a, beta):
    """Full gradient of 2n*L/(sum a)^2."""
    n = len(a)
    L, L_prime, p, c = smooth_max_and_grad(a, beta)
    S = float(np.sum(a))
    if S < 1e-15:
        return np.zeros(n)
    grad = 2.0 * n * (L_prime * S - 2.0 * L) / (S ** 3)
    return grad

                  beta_power=0.5, verbose=False):

    a = a.copy().astype(np.float64)

    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve = 0
    patience = 500

        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, num_iters - 1)
        beta = start_beta * (end_beta / start_beta) ** frac

        t += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        a_new = a - step

        a = a_new

        # Periodic perturbation if stuck
        if no_improve > patience:
            rng = np.random.default_rng(it * 104729 + 7)
            scale = rng.uniform(0.01, 0.05) * (1 + no_improve / patience)
            a = a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            S = np.sum(a)
            if S > 1e-10:
                a = a / S
            no_improve = 0
            patience = max(100, patience // 2)

        if verbose and it % 1000 == 0:
            print(f"    Adam it={it}, obj={best_obj:.6f}, beta={beta:.2f}")

    return best_a, best_obj


def mirror_descent_optimize(a, start_beta, end_beta, num_iters, lr=0.01,
                            time_budget=100.0, verbose=False):
    """Mirror descent (multiplicative weight) on the smoothed objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve = 0

    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, num_iters - 1)
        beta = start_beta * (end_beta / start_beta) ** frac

        grad = full_gradient(a, beta)

        # Multiplicative update: a_j <- a_j * exp(-lr * grad_j)
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / S_new

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7)
            scale = rng.uniform(0.01, 0.04)
            a = a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            S = np.sum(a)
            if S > 1e-10:
                a = a / S
            no_improve = 0

        if verbose and it % 2000 == 0:
            print(f"    Mirror it={it}, obj={best_obj:.6f}, beta={beta:.2f}")

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_direction(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0


def lp_polish(a, time_budget=60.0, max_iters=500):
    """LP alternating polishing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, S_g = solve_lp_direction(a, M, relax=1.0)
        if g is None or S_g <= 1.0 + 1e-8:
            if stall > 20:
                # Perturb
                rng = np.random.default_rng(it * 7919 + 12345)
                scale = rng.uniform(0.02, 0.06)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                S = np.sum(a)
                if S > 1e-10:
                    a = a / S
                stall = 0
            stall += 1
            continue

        # Line search
        improved = False
        for t in [0.5, 0.7, 0.85, 1.0, 0.3, 0.9, 0.6]:
            a_try = (1.0 - t) * a + t * g / S_g
            a_try = np.maximum(a_try, 0)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                rng = np.random.default_rng(it * 104729 + 42)
                scale = rng.uniform(0.03, 0.08)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                S = np.sum(a)
                if S > 1e-10:
                    a = a / S
                stall = 0

    return best_a, best_obj


def peak_reducing_perturb(a, num_peaks=5, num_tries=30, time_budget=10.0):
    """Targeted perturbations to reduce top peaks in convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    for _ in range(num_tries):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        sorted_idx = np.argsort(conv)[::-1]
        top_peaks = sorted_idx[:num_peaks]

        for _ in range(3):
            k = int(top_peaks[rng.integers(0, len(top_peaks))])

            # Find which a-indices contribute to peak k
            affected = []
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    affected.append(i)

            if len(affected) < 2:
                continue

            # Reduce at affected positions, increase elsewhere
            a_new = a.copy()
            idx = np.array(affected)
            factors = rng.uniform(0.7, 1.0, len(idx))
            a_new[idx] *= factors
            # Compensate at other positions
            other_mask = np.ones(n)
            other_mask[idx] = 0
            other_idx = np.where(other_mask > 0)[0]
            if len(other_idx) > 0:
                comp = rng.uniform(1.0, 1.5, len(other_idx))
                a_new[other_idx] *= comp

            a_new = np.maximum(a_new, 0)
            S = np.sum(a_new)
            if S < 1e-10:
                continue
            a_new = a_new / S
            obj_new = compute_obj(a_new)

            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                break

    return best_a, best_obj


def spectral_perturb(a, num_tries=20, time_budget=10.0):
    """Spectral domain perturbations to flatten autocorrelation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(123)

    for _ in range(num_tries):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, 2 * n)
        mag = np.abs(fa)
        phase = np.angle(fa)

        choice = rng.integers(0, 5)
        if choice == 0:
            # Soft-clip magnitude peaks
            thresh = np.percentile(mag, rng.uniform(80, 95))
            mag_new = np.where(mag > thresh, thresh + (mag - thresh) * 0.5, mag)
            fa_new = mag_new * np.exp(1j * phase)
        elif choice == 1:
            # Random spectral perturbation
            mag_new = mag * np.exp(rng.normal(0, 0.05, len(fa)))
            fa_new = mag_new * np.exp(1j * phase)
        elif choice == 2:
            # Spectral tilt
            freqs = np.linspace(0, 1, len(fa))
            tilt = rng.uniform(-1.0, 1.0)
            mag_new = mag * np.exp(tilt * freqs)
            fa_new = mag_new * np.exp(1j * phase)
        elif choice == 3:
            # Targeted: reduce frequencies that contribute to peak
            conv = compute_conv(a)
            peak_k = int(np.argmax(conv))
            # The peak at k is influenced by all frequencies
            # Reduce high frequency content
            cutoff = rng.uniform(0.3, 0.7) * len(fa)
            mag_new = mag.copy()
            mag_new[cutoff:] *= 0.8
            fa_new = mag_new * np.exp(1j * phase)
        else:
            # Equalization
            target = np.mean(mag)
            alpha = rng.uniform(0.1, 0.4)
            mag_new = (1 - alpha) * mag + alpha * target
            fa_new = mag_new * np.exp(1j * phase)

        a_new = irfft(fa_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        S = np.sum(a_new)
        if S < 1e-10:
            continue
        a_new = a_new / S
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Block pattern
        nblocks = rng.integers(2, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 3:
        # Bimodal
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((k - c1) / w) ** 2) + np.exp(-0.5 * ((k - c2) / w) ** 2)
    elif choice == 4:
        # Smoothed random
        a = rng.random(n)
        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 5:
        # Alternating with noise
        a = 1.0 + 0.3 * np.cos(2 * np.pi * k / (n // rng.integers(2, 10)))
        a += rng.normal(0, 0.1, n)
    elif choice == 6:
        # Power law
        a = (k + 1) ** rng.uniform(-0.5, 0.5)
    elif choice == 7:
        # Gaussian window
        center = n / 2
        width = n * rng.uniform(0.2, 0.5)
        a = np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 8:
        # Two-tone
        f1 = rng.uniform(1, 5)
        f2 = f1 + rng.uniform(1, 5)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.01)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S
    return a


def dual_guided_perturb(a, num_tries=20, time_budget=5.0):
    """Use LP dual info to guide perturbations toward reducing active constraints."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(99)

    for _ in range(num_tries):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, S_g = solve_lp_direction(a, M, relax=1.0)
        if g is None:
            # Fallback: perturb at peak locations
            peak_idx = int(np.argmax(conv))
            a_new = a.copy()
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    a_new[i] *= rng.uniform(0.8, 1.0)
            a_new = np.maximum(a_new, 0)
            S = np.sum(a_new)
            if S > 1e-10:
                a_new = a_new / S
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
                    a = a_new
            continue

        # Use g as a direction: positions where g is large relative to a
        # are "expansion" directions
        ratio = g / (a + 1e-10)
        # Find positions where the LP wants to increase
        expand_mask = ratio > 1.0
        contract_mask = ratio < 0.5

        a_new = a.copy()
        t = rng.uniform(0.1, 0.5)
        a_new = (1 - t) * a + t * g / S_g
        a_new = np.maximum(a_new, 0)
        S = np.sum(a_new)
        if S > 1e-10:
            a_new = a_new / S
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_single_n(a_start, time_budget, seed=42):
    """Optimize a sequence of fixed length using combined methods."""
    n = len(a_start)
    a = a_start.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget - 2:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break

        # Strategy selection based on remaining time and progress
        phase = (time.monotonic() - start) / time_budget

        if phase < 0.4:
            # Exploration phase: gradient + perturbations
            # Adam with moderate beta
            beta_s = max(10.0, 5.0 / max(1.0, n / 100))
            beta_e = max(100.0, 20.0 / max(1.0, n / 100))
            alloc = min(remaining * 0.3, 30.0)
            a_try, obj_try = adam_optimize(best_a, beta_s, beta_e,
                                           num_iters=10000, lr=0.03,
                                           time_budget=alloc)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                print(f"      Adam improved: {best_obj:.6f}")
        elif phase < 0.7:
            # Mid phase: mirror descent + LP
            beta_s = max(50.0, 10.0 / max(1.0, n / 100))
            beta_e = max(200.0, 50.0 / max(1.0, n / 100))
            alloc = min(remaining * 0.2, 20.0)
            a_try, obj_try = mirror_descent_optimize(best_a, beta_s, beta_e,
                                                     num_iters=10000, lr=0.01,
                                                     time_budget=alloc)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                print(f"      Mirror improved: {best_obj:.6f}")

            # LP polish
            alloc_lp = min(remaining * 0.2, 30.0)
            a_try, obj_try = lp_polish(best_a, time_budget=alloc_lp)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                print(f"      LP improved: {best_obj:.6f}")
        else:
            # Final phase: intensive LP + fine gradient
            alloc_lp = min(remaining * 0.5, 40.0)
            a_try, obj_try = lp_polish(best_a, time_budget=alloc_lp)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                print(f"      Final LP improved: {best_obj:.6f}")

            # Fine gradient with high beta
            if time.monotonic() - start < time_budget - 5:
                beta = max(500.0, 100.0 / max(1.0, n / 100))
                alloc_g = min(remaining * 0.2, 15.0)
                a_try, obj_try = adam_optimize(best_a, beta, beta * 2,
                                               num_iters=50000, lr=0.005,
                                               time_budget=alloc_g)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    print(f"      Fine grad improved: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: FFT-accelerated gradient optimization with Adam and mirror descent,
    combined with LP polishing, peak-targeted perturbations, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.6f}")

    # === Phase 1: Intensive gradient optimization on parent ===
    if best_seq is not None:
        n_parent = len(best_seq)
        print(f"\n=== Phase 1: Gradient optimization on parent (n={n_parent}) ===")

        # Round 1: Adam exploration
        alloc1 = min(120.0, (deadline - time.monotonic()) * 0.15)
        beta_s = max(5.0, 2.0 / max(1.0, n_parent / 200))
        beta_e = max(50.0, 10.0 / max(1.0, n_parent / 200))
        a_try, obj_try = adam_optimize(best_seq, beta_s, beta_e,
                                       num_iters=50000, lr=0.05,
                                       time_budget=alloc1)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  After Adam round 1: {best_score:.6f}")

        # Round 2: Mirror descent
        if time.monotonic() < deadline - 60:
            alloc2 = min(60.0, (deadline - time.monotonic()) * 0.1)
            beta_s = max(10.0, 5.0 / max(1.0, n_parent / 200))
            beta_e = max(100.0, 20.0 / max(1.0, n_parent / 200))
            a_try, obj_try = mirror_descent_optimize(best_seq, beta_s, beta_e,
                                                     num_iters=50000, lr=0.01,
                                                     time_budget=alloc2)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  After Mirror: {best_score:.6f}")

        # Round 3: Peak reduction + LP
        if time.monotonic() < deadline - 60:
            a_try, obj_try = peak_reducing_perturb(best_seq, num_peaks=8,
                                                   num_tries=50, time_budget=10.0)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  After peak perturb: {best_score:.6f}")

            alloc3 = min(60.0, (deadline - time.monotonic()) * 0.1)
            a_try, obj_try = lp_polish(best_seq, time_budget=alloc3)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  After LP: {best_score:.6f}")

        # Round 4: Fine gradient
        if time.monotonic() < deadline - 30:
            n_p = len(best_seq)
            beta = max(100.0, 20.0 / max(1.0, n_p / 200))
            alloc4 = min(30.0, (deadline - time.monotonic()) * 0.08)
            a_try, obj_try = adam_optimize(best_seq, beta, beta * 3,
                                           num_iters=100000, lr=0.01,
                                           time_budget=alloc4)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  After fine gradient: {best_score:.6f}")

    # === Phase 2: Multi-length exploration ===
    print(f"\n=== Phase 2: Multi-length exploration ===")
    print(f"  Current best: {best_score:.6f}")

    if best_seq is not None:
        np_best = len(best_seq)
        n_values = sorted(set([
            max(200, np_best - 200),
            np_best,
            np_best + 100,
            np_best + 200,
            np_best + 400,
            np_best + 600,
        ]))
    else:
        n_values = [400, 600, 800, 1000, 1200, 1500]

    print(f"  n values: {n_values}")

    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Time allocation: more time for promising n values
        alloc = min(remaining * 0.25, 80.0)
        alloc = max(15.0, alloc)

        print(f"\n  N={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        # Start from resampled best or random
        if best_seq is not None and n != len(best_seq):
            a_start = resample_sequence(best_seq, n)
        elif best_seq is not None:
            a_start = best_seq.copy()
        else:
            a_start = random_init(n, seed + n * 104729)

        a_opt, obj_opt = optimize_single_n(a_start, time_budget=alloc, seed=seed + n)

        # Additional perturbation + polish
        if time.monotonic() < deadline - 30:
            a_opt, obj_opt = peak_reducing_perturb(a_opt, num_peaks=5,
                                                   num_tries=20, time_budget=5.0)
            if time.monotonic() < deadline - 20:
                a_opt, obj_opt = lp_polish(a_opt, time_budget=min(10.0,
                                                                 (deadline - time.monotonic()) * 0.1))

        print(f"  N={n}: obj={obj_opt:.6f}")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # === Phase 3: Random restart exploration ===
    print(f"\n=== Phase 3: Random restarts ===")
    print(f"  Current best: {best_score:.6f}")

    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq) if best_seq is not None else 700

        for restart in range(5):
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            # Try a random initialization at the best n or slightly different
            n_try = n_best + int(restart * 50 - 100) if restart > 0 else n_best
            n_try = max(200, min(2000, n_try))

            a_rand = random_init(n_try, seed + 10000 + restart * 7777)
            alloc = min(remaining * 0.15, 40.0)
            alloc = max(10.0, alloc)

            a_opt, obj_opt = optimize_single_n(a_rand, time_budget=alloc,
                                               seed=seed + 20000 + restart)
            print(f"  Restart {restart} (n={n_try}): obj={obj_opt:.6f}")

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** UPDATE from restart: n={n_try}, obj={best_score:.6f} **")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final polish ===")
    print(f"  Best before: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()

        # LP polish
        alloc = min(remaining * 0.3, 60.0)
        a_try, obj_try = lp_polish(best_seq, time_budget=alloc)
        if obj_try < best_score:
            best_score = obj_try
            best_seq = a_try.copy()
            print(f"  LP: {best_score:.6f}")

        # Fine gradient
        if time.monotonic() < deadline - 20:
            n_f = len(best_seq)
            remaining = deadline - time.monotonic()
            beta = max(200.0, 50.0 / max(1.0, n_f / 200))
            alloc = min(remaining * 0.3, 40.0)
            a_try, obj_try = adam_optimize(best_seq, beta, beta * 2,
                                           num_iters=200000, lr=0.003,
                                           time_budget=alloc)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Fine grad: {best_score:.6f}")

        # One more LP
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.3, 30.0)
            a_try, obj_try = lp_polish(best_seq, time_budget=alloc)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Final LP: {best_score:.6f}")

        # Peak reduction
        if time.monotonic() < deadline - 10:
            a_try, obj_try = peak_reducing_perturb(best_seq, num_peaks=10,
                                                   num_tries=50,
                                                   time_budget=min(10, (deadline - time.monotonic()) * 0.2))
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Peak: {best_score:.6f}")

    # === Final validation ===
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
