        if verbose and it % 100 == 0:
            elapsed = time.monotonic()
            print(f"        LP iter {it}: M={M:.10f} best={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.best_n = 0
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            self.best_n = a.size
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s [{label}]", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining smooth optimization, LP mirror descent with
    active sets, multi-length exploration, and structured perturbations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "parent")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent, "default-ones")

    # =========================================================
    # PHASE 1: Fast smooth optimization on parent (20% budget)
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smooth Nesterov on parent n={parent.size}", flush=True)
    print(f"{'='*60}", flush=True)

    n0 = parent.size
    a1 = parent.copy()
    a1 = a1 / a1.sum()

    p1_deadline = min(t_start + budget_s * 0.20, deadline)
    a_opt1 = run_smooth_nesterov(a1, 50000, p1_deadline,
                                  tau_start=10.0, tau_end=300.0,
                                  lr_start=0.4, lr_end=0.03, momentum=0.92)
    tracker.update(a_opt1, "P1-nesterov")

    # Also try vanilla for comparison
    a_opt1b = run_smooth_vanilla(a1, 20000, p1_deadline,
                                  tau_start=5.0, tau_end=150.0,
                                  lr_start=0.3, lr_end=0.02)
    tracker.update(a_opt1b, "P1-vanilla")

    # =========================================================
    # PHASE 2: LP Mirror Descent (35% budget) - main workhorse
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent n={tracker.best_n}", flush=True)
    print(f"{'='*60}", flush=True)

    a2 = tracker.best.copy()
    a2 = a2 / a2.sum()
    n2 = len(a2)

    # Start with fewer active constraints, gradually increase
    p2_deadline = min(t_start + budget_s * 0.55, deadline)
    phase2_time = p2_deadline - time.monotonic()

    # Sub-phase 2a: LP with top 80 constraints (fast, many iterations)
    sub2a_dl = min(time.monotonic() + phase2_time * 0.3, p2_deadline)
    n_act_2a = max(40, int(0.15 * n2))
    a2a = run_lp_mirror(a2, 20000, sub2a_dl, n_active=n_act_2a, verbose=True)
    tracker.update(a2a, f"P2a LP-act{n_act_2a}")

    # Sub-phase 2b: LP with top 200 constraints
    a2b = tracker.best.copy()
    a2b = a2b / a2b.sum()
    sub2b_dl = min(time.monotonic() + phase2_time * 0.4, p2_deadline)
    n_act_2b = max(100, int(0.4 * n2))
    a2b = run_lp_mirror(a2b, 20000, sub2b_dl, n_active=n_act_2b, verbose=True)
    tracker.update(a2b, f"P2b LP-act{n_act_2b}")

    # Sub-phase 2c: Full LP (all constraints)
    a2c = tracker.best.copy()
    a2c = a2c / a2c.sum()
    a2c = run_lp_mirror(a2c, 20000, p2_deadline, n_active=None, verbose=True)
    tracker.update(a2c, "P2c LP-full")

    # =========================================================
    # PHASE 3: Multi-length exploration (20% budget)
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    p3_deadline = min(t_start + budget_s * 0.75, deadline)
    base_n = tracker.best_n

    # Try various lengths
    trial_lengths = []
    for n_try in [256, 320, 384, 448, 480, 512, 544, 576, 640, 768, 896, 1024]:
        if n_try != base_n and 128 <= n_try <= 2048:
            trial_lengths.append(n_try)

    # Sort by closeness to current best
    trial_lengths.sort(key=lambda x: abs(x - base_n))

    for n_trial in trial_lengths[:8]:
        if time.monotonic() > p3_deadline:
            break

        a_init = resample_seq(tracker.best, n_trial)
        a_init = np.maximum(a_init, 1e-6)
        a_init = a_init / a_init.sum()

        # Quick smooth
        sub_dl = min(time.monotonic() + 8, p3_deadline)
        a_s = run_smooth_nesterov(a_init, 3000, sub_dl,
                                   tau_start=10, tau_end=100,
                                   lr_start=0.3, lr_end=0.05, momentum=0.9)

        # Quick LP polish
        sub_dl2 = min(time.monotonic() + 15, p3_deadline)
        n_act = max(30, int(0.3 * n_trial))
        a_lp = run_lp_mirror(a_s, 5000, sub_dl2, n_active=n_act, verbose=False)
        tracker.update(a_lp, f"P3 n={n_trial}")

    # =========================================================
    # PHASE 4: Structured perturbation restarts (15% budget)
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Structured Perturbations", flush=True)
    print(f"{'='*60}", flush=True)

    p4_deadline = min(t_start + budget_s * 0.90, deadline)
    n_best = tracker.best_n

    perturb_types = [
        ("noise", 0.0),
        ("noise", 0.0),
        ("shift", 1),
        ("shift", -1),
        ("shift", 2),
        ("shift", -2),
        ("local_scale", 0),
        ("local_scale", 1),
        ("bimodal_mix", 0),
        ("noise", 0.0),
        ("shift", 3),
        ("shift", -3),
        ("local_scale", 2),
        ("noise", 0.0),
        ("bimodal_mix", 1),
    ]

    temperature = 0.1
    for r, (ptype, param) in enumerate(perturb_types):
        if time.monotonic() > p4_deadline:
            break

        a = tracker.best.copy().astype(np.float64)
        noise_scale = temperature * 0.01

        if ptype == "noise":
            a = a + rng.normal(0, noise_scale, n_best)
        elif ptype == "shift":
            a = np.roll(a, param)
            a[0] *= 0.5
            a[-1] *= 0.5
            a = a + rng.normal(0, noise_scale * 0.5, n_best)
        elif ptype == "local_scale":
            # Scale a random middle segment
            seg_start = rng.integers(0, n_best // 2)
            seg_end = seg_start + rng.integers(n_best // 4, n_best // 2)
            scale = 1.0 + rng.uniform(-0.3, 0.3)
            a[seg_start:seg_end] *= scale
            a = a + rng.normal(0, noise_scale * 0.3, n_best)
        elif ptype == "bimodal_mix":
            # Mix with a bimodal distribution
            x = np.arange(n_best) / (n_best - 1)
            bimodal = np.exp(-8.0 * (x - 0.2) ** 2) + np.exp(-8.0 * (x - 0.8) ** 2)
            mix_frac = 0.1 + 0.1 * param
            a = (1 - mix_frac) * a / a.sum() + mix_frac * bimodal / bimodal.sum()

        a = np.maximum(a, 1e-6)
        a = a / a.sum()

        # Quick smooth
        sub_dl = min(time.monotonic() + 5, p4_deadline)
        a = run_smooth_nesterov(a, 1000, sub_dl,
                                 tau_start=15, tau_end=80,
                                 lr_start=0.3, lr_end=0.05, momentum=0.9)

        # LP polish
        sub_dl2 = min(time.monotonic() + 20, p4_deadline)
        n_act = max(50, int(0.4 * n_best))
        a_opt = run_lp_mirror(a, 5000, sub_dl2, n_active=n_act, verbose=False)
        tracker.update(a_opt, f"P4 r={r} {ptype}")

        temperature *= 0.90

    # =========================================================
    # PHASE 5: Final aggressive LP polish (10% budget)
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        a_final = tracker.best.copy()
        a_final = a_final / a_final.sum()

        # Round 1: LP with 60% active
        sub_f1 = min(time.monotonic() + (deadline - time.monotonic()) * 0.4, deadline)
        n_act_f1 = max(80, int(0.6 * n_final))
        a_f1 = run_lp_mirror(a_final, 30000, sub_f1, n_active=n_act_f1, verbose=True)
        tracker.update(a_f1, "P5-r1")

        # Round 2: Full LP
        a_final2 = tracker.best.copy()
        a_final2 = a_final2 / a_final2.sum()
        a_f2 = run_lp_mirror(a_final2, 30000, deadline, n_active=None, verbose=True)
        tracker.update(a_f2, "P5-r2-full")

        # Try one more round if time permits
        if time.monotonic() < deadline - 10:
            a_final3 = tracker.best.copy()
            a_final3 = a_final3 / a_final3.sum()
            n_act_f3 = max(100, int(0.8 * n_final))
            a_f3 = run_lp_mirror(a_final3, 20000, deadline, n_active=n_act_f3, verbose=True)
            tracker.update(a_f3, "P5-r3")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
