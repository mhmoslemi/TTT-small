```python
"""
Smooth optimization via log-sum-exp annealing combined with LP alternation
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix for conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_direction(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=build_toeplitz(a), b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0


def compute_grad_smooth(a, t):
    """
    Compute gradient of 2n * logsumexp(conv(a,a)/t) with respect to a.
    Returns (objective_value, gradient).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = float(np.max(conv))
    
    # Log-sum-exp for numerical stability
    shifted = (conv - max_c) / t
    max_shifted = np.max(shifted)
    lse = t * (max_c / t + max_shifted + np.log(np.sum(np.exp(shifted - max_shifted))))
    obj = 2.0 * n * lse
    
    # Softmax weights
    w = np.exp(shifted - max_shifted)
    w /= np.sum(w)
    
    # Gradient: d lse / da_j = sum_k w[k] * (a[k-j] + a[j-k]) with bounds
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        # Term 1: a[k-j] valid when 0 <= k-j < n => j <= k <= j+n-1
        k_lo = max(j, 0)
        k_hi = min(j + n, 2 * n - 1)  # exclusive
        if k_hi > k_lo:
            s += float(np.dot(w[k_lo:k_hi], a[k_lo - j:k_hi - j]))
        # Term 2: a[j-k] valid when 0 <= j-k < n => j-n+1 <= k <= j
        k_lo = max(j - n + 1, 0)
        k_hi = min(j + 1, 2 * n - 1)  # exclusive
        if k_hi > k_lo:
            a_idx = j - np.arange(k_lo, k_hi)
            s += float(np.dot(w[k_lo:k_hi], a[a_idx]))
        grad[j] = 2.0 * n * s
    
    return obj, grad


def smooth_optimize(a, t_schedule=None, max_iters=200, time_budget=60.0):
    """
    Optimize using log-sum-exp smoothed objective with L-BFGS-B.
    Temperature annealing: start high, decrease over iterations.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if t_schedule is None:
        # Default: geometric schedule from n/4 down to 1/(4n)
        t_start = n / 4.0
        t_end = 1.0 / (4.0 * n)
        t_schedule = np.geomspace(t_start, t_end, max_iters)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        t = t_schedule[min(it, len(t_schedule) - 1)]
        
        def fun_and_grad(a_flat):
            a_local = np.maximum(a_flat, 1e-15)
            a_local = a_local / np.sum(a_local)  # normalize
            obj, grad = compute_grad_smooth(a_local, t)
            return obj, grad
        
        res = minimize(fun_and_grad, a, method='L-BFGS-B',
                      bounds=[(1e-15, None)] * n,
                      options={'maxiter': 50, 'maxfun': 200, 'gtol': 1e-8})
        
        a_new = np.maximum(res.x, 1e-15)
        a_new = a_new / np.sum(a_new)
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_obj


def lp_alternation(a, max_iters=3000, time_budget=100.0):
    """LP alternating optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 50
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, S = solve_lp_direction(a, M, relax=1.0)
        
        if g is not None and S > 1.0 + 1e-8:
            # Line search
            improved = False
            for t in [1.0, 0.7, 0.5, 0.3, 0.85, 0.15]:
                a_try = (1.0 - t) * a + t * g / S
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            
            if improved:
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        # Perturb when stalled
        if stall > patience:
            rng = np.random.default_rng(it * 31 + 7)
            scale = rng.uniform(0.05, 0.15)
            a = a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s2 = np.sum(a)
            if s2 > 1e-10:
                a = a / s2
            else:
                a = best_a.copy()
            stall = 0
            patience = max(15, patience // 2)
    
    return best_a


def peak_perturb(a, num_tries=30, time_budget=10.0):
    """Targeted perturbation to reduce peaks."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    for _ in range(num_tries):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        # Find the peak position
        peak_k = int(np.argmax(conv))
        
        # Determine which a indices contribute to this peak
        affected = []
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                affected.append(i)
        
        # Reduce affected positions
        a_new = a.copy()
        for pos in affected:
            a_new[pos] *= rng.uniform(0.7, 1.0)
        
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
    
    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)
    
    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 5)):
            f = rng.uniform(0.5, 6.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        # Multi-scale
        a = np.ones(n)
        for scale in [0.3, 1.0, 3.0, 7.0]:
            f = scale * rng.uniform(0.8, 1.2)
            amp = rng.uniform(0.05, 0.2)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    else:
        a = rng.random(n) + 0.05
    
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
```
