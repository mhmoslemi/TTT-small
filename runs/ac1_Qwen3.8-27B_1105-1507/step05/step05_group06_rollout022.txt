import numpy as np
from scipy.optimize import linprog, minimize, minimize_scalar
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv_fft(a, b):
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    b = fast_conv(a)
    return float(2 * len(a) * b.max())

def softmax_stable(x):
    x = x - x.max()
    ex = np.exp(x)
    return ex / ex.sum()

def lse_objective_and_grad(x, tau):
    """Compute F(x) = 2n/tau * log(sum_k exp(tau * b[k])) and its gradient wrt x.
    where a = softmax(x) and b = conv(a, a)."""
    n = len(x)
    a = softmax_stable(x)
    b = fast_conv(a)  # length 2n-1
    
    # Log-sum-exp with stability
    tb = tau * b
    tb_max = tb.max()
    log_sum_exp = tb_max + np.log(np.sum(np.exp(tb - tb_max)))
    F = 2.0 * n / tau * log_sum_exp
    
    # Softmax weights
    w = np.exp(tb - tb_max)
    w = w / w.sum()
    
    # grad F/da[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * h[i + n - 1] where h = conv(w, a_reversed)
    a_rev = a[::-1]
    h = cross_conv_fft(w, a_rev)  # length (2n-1) + n - 1 = 3n-2
    grad_da = 2.0 * h[n-1:2*n-1]  # length n
    
    # grad F/dx = a * (grad_da - dot(grad_da, a))
    dot = np.dot(grad_da, a)
    grad_dx = a * (grad_da - dot)
    
    return F, grad_dx

def lse_optimize(a_init, tau_start, tau_end, max_iters, deadline):
    """Optimize using L-BFGS on the log-sum-exp smoothed objective.
    Uses temperature annealing from tau_start to tau_end."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / a_init.sum()
    
    # Initialize x such that softmax(x) = a_init
    x = np.log(a_init)
    
    best_a = a_init.copy()
    best_J = compute_J(a_init)
    
    # Current tau
    tau = tau_start
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        # Increase tau progressively
        progress = it / max(1, max_iters)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Run a few L-BFGS iterations at current tau
        def obj_and_grad(x_local):
            F, g = lse_objective_and_grad(x_local, tau)
            return F, g
        
        try:
            result = minimize(obj_and_grad, x, method='L-BFGS-B',
                            jac=True, options={'maxiter': 20, 'gtol': 1e-8})
            if result.success or result.fun < 1e6:
                x = result.x
                a_new = softmax_stable(x)
                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
        except Exception:
            pass
        
        it += 1
    
    return best_a, best_J

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent (fast precomputed-convolution line search)
2. Exact coordinate descent with 1D bounded optimization per coordinate
3. LSE-smoothed mirror descent with progressive temperature annealing
4. Multi-length exploration and diverse initializations
5. Crossover and perturbation strategies for escaping local optima
Key novelty: exact 1D coordinate optimization (minimize_scalar) combined with
LSE-smoothed mirror descent for navigating the complex landscape.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv_fft(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J_local(a):
    """Compute J(a) = 2n*max(conv(a,a))/(sum(a))^2 with local normalization."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    b = fast_conv(a)
    return float(2 * len(a) * b.max())


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with fast line search."""
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J_local(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        g = lp_best_response(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Line search using precomputed convolutions
        b_ag = cross_conv_fft(a, g)
        b_gg = fast_conv(g)

        def obj_t(t):
            b_blend = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
            return b_blend.max()

        result = minimize_scalar(obj_t, bounds=(0.01, 0.99), method='bounded',
                                 options={'xatol': 1e-8, 'maxiter': 30})
        t_opt = float(result.x)

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J_local(a_new)

        if J_new < best_J - 1e-13:
            best_a = a_new.copy()
            best_J = J_new
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 40:
            a = a + np.random.normal(0, 0.02, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_J

            elif k_lo == 0:

            else:
