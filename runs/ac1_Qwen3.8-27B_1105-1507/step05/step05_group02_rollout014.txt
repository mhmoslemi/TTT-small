        # Adaptive LP time limit: longer early, shorter later
        frac = it / max_iters
        if frac < 0.3:
            lp_tl = max(3.0, 8.0 * (1 - frac))
        elif frac < 0.7:
            lp_tl = max(2.0, 5.0 * (1 - frac))
        else:
            lp_tl = max(1.5, 3.0 * (1 - frac))

        g = lp_best_response(a, M, time_limit=lp_tl)
        if g is None:
            no_imp += 1
            if no_imp > 20:
                break
            continue

        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Geometric line search with finer granularity
        improved = False
        best_t = None
        best_M_t = M
        t = 1.0
        for _ in range(20):
            a_t = (1 - t) * a + t * g
            a_t = np.maximum(a_t, 0)
            s_t = a_t.sum()
            if s_t < 1e-10:
                t *= 0.5
                continue
            a_t /= s_t
            M_t = fast_conv(a_t).max()
            if M_t < best_M_t - 1e-15:
                best_M_t = M_t
                best_t = t
                best_a_t = a_t.copy()
            if t < 0.01:
                break
            t *= 0.7

        if best_t is not None:
            a = best_a_t
            no_imp = 0
        else:
            no_imp += 1

        # Periodic perturbation to escape local minima
        if no_imp > 0 and it % perturb_every < no_imp and no_imp > 5:
            scale = perturb_scale_init * (perturb_decay ** (no_imp // perturb_every))
            noise = rng.normal(0, scale, n)
            a = np.maximum(a + noise, 1e-6)
            a /= a.sum()
            no_imp = 0
            if verbose:
                print(f"    Perturbed at iter {it}, scale={scale:.4f}")

        if no_imp > 60:
            if verbose:
                print(f"    LP stuck at iter {it}, M={M:.10f}")
            break

        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: M={M:.10f}, best={best_M:.10f}")

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 0.01:
            return
        # Normalize to avoid issues
        a_norm = a / s
        score = compute_J(a_norm)
        if np.isfinite(score) and score < self.score:
            self.score = score
            self.seq = a_norm.copy()
            el = time.monotonic() - self.t0
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={el:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    rng = np.random.default_rng(seed)

        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                parent = parent / parent.sum()
                best.update(parent)
                print(f"Parent loaded: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:

    print(f"  After main LP: J={best.score:.8f}", flush=True)

    p3_budget = remaining * 0.15
    p3_start = time.monotonic()
    p3_deadline = time.monotonic() + p3_budget

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration ({p3_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n_cur = best.seq.size
        # Try lengths around the current best
        length_candidates = []
        for delta in [-64, -32, 32, 64, 128, -128]:
            n_try = n_cur + delta
            if 64 <= n_try <= 4096:
                n_try = (n_try // 8) * 8  # round to multiple of 8
                if n_try >= 64:
                    length_candidates.append(n_try)
        # Also try some absolute values
        for n_try in [384, 448, 512, 576, 640, 768, 896, 1024]:
            if n_try != n_cur and n_try not in length_candidates:
                length_candidates.append(n_try)

        length_candidates = sorted(set(length_candidates))
        print(f"  Trying lengths: {length_candidates}", flush=True)

        for n_try in length_candidates:

                              verbose=(round_idx == 0),
                              perturb_every=100,
                              perturb_scale_init=0.03 * (round_idx + 1),
                              perturb_decay=0.9)
            best.update(a)

    # === PHASE 5: Subgradient polishing ===
    if best.seq is not None and time.monotonic() < deadline - 5:
        a = run_subgradient_avg(best.seq, deadline - 3, max_iters=1000, lr_const=0.2)
        best.update(a)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(512) / 512
        best.score = compute_J(best.seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best.seq]
