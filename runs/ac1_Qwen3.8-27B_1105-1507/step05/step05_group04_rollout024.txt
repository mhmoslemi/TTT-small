```python
def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    # Project onto simplex with sum = s
    # Equivalent to projecting v/s onto standard simplex, then multiply by s
    v = v / s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0) * s
```

```python
import numpy as np
import cvxpy as cp
from scipy.sparse import csr_matrix
from scipy.signal import fftconvolve
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def build_M_k(n, k):
    """Build sparse matrix M_k where a^T M_k a = (a*a)[k]."""
    rows, cols, vals = [], [], []
    j_min = max(0, k - n + 1)
    j_max = min(n - 1, k)
    for j in range(j_min, j_max + 1):
        l = k - j
        if l < 0 or l >= n:
            continue
        if j == l:
            rows.append(j)
            cols.append(j)
            vals.append(1.0)
        else:
            rows.append(j)
            cols.append(l)
            vals.append(0.5)
            rows.append(l)
            cols.append(j)
            vals.append(0.5)
    return csr_matrix((vals, (rows, cols)), shape=(n, n))

def solve_subproblem(a, top_m=5, solver='ECOS'):
    """Solve: min t s.t. (a'*a')[k_i] <= t, sum(a')=1, a'>=0."""
    n = len(a)
    b = fast_conv(a)
    top_k = np.argsort(b)[-top_m:]
    
    a_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1, a_var >= 0]
    for k in top_k:
        M_k = build_M_k(n, k)
        constraints.append(cp.quad_form(a_var, M_k) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=solver)
    
    if a_var.value is not None:
        return np.maximum(a_var.value, 0)
    return a  # fallback

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))
```

```python
def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - s)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
def find_minimax(x_max, m, c):
    """Find x in [0, x_max] minimizing max_k (m_k*x + c_k)."""
    if x_max <= 0:
        return 0.0
    
    def g(x):
        return np.max(m * x + c)
    
    # Binary search for the point where the subgradient contains 0
    lo = 0.0
    hi = x_max
    
    # Check endpoints
    if g(0.0) <= g(x_max):
        # Minimum might be at 0 or in between
        pass
    
    for _ in range(60):  # ~1e-18 precision
        mid = (lo + hi) / 2
        vals = m * mid + c
        gmid = vals.max()
        
        # Subgradient: the active m_k values (where vals[k] == gmid)
        active = np.abs(vals - gmid) < 1e-12
        if active.any():
            sg = m[active]
            if sg.max() < 0:
                # Function is decreasing at mid, go right
                lo = mid
            elif sg.min() > 0:
                # Function is increasing at mid, go left
                hi = mid
            else:
                # 0 is in the subgradient, we're at the optimum
                break
        else:
            # Shouldn't happen, but just go right
            lo = mid
    
    return (lo + hi) / 2
```
