The key insight is that we want `conv(a,a)[k] ≈ (sum_a)^2/(2n-1)` for all k. This is like a "flat spectrum" design.

In the frequency domain, `conv(a,a)` is the inverse FFT of `|FFT(a)|^2` (the power spectrum). So making `conv(a,a)` flat is equivalent to making `|FFT(a)|^2` flat, i.e., making the power spectrum flat (white).

But we also need `a >= 0`! This is the constraint that makes it hard.

**Approach: Iterative spectral flattening with non-negativity projection**

1. Compute `A = FFT(a)`.
2. Modify `|A|` to be flatter (e.g., equalize magnitudes).
3. Compute `a_new = IFFT(modified_A)`.
4. Project to non-negativity: `a_new = max(a_new, 0)`.
5. Normalize: `a_new = a_new / sum(a_new)`.
6. Repeat.

This is similar to the "spectral step" in the previous code, but I'll make it more aggressive and systematic.

Since we want to minimize `max_k conv(a,a)[k]` (with `sum(a)=1`), and the sum of all `conv` values is fixed at 1, we want to make them all close to `1/(2n-1)`.

This is equivalent to minimizing the variance of `conv(a,a)`, or equivalently, minimizing `sum_k (conv(a,a)[k])^2` (since the mean is fixed).

`sum_k (conv(a,a)[k])^2 = ||conv(a,a)||^2 = ||a||^4` ... wait, is that right?

Actually, `||conv(a,a)||^2 = sum_k (sum_i a[i]*a[k-i])^2`. This is a 4th order form in a, not simply related to `||a||^4`.

Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

def build_toeplitz_subset(a, rows):
    """Build only the specified rows of the Toeplitz matrix."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((len(rows), n), dtype=np.float64)
    for row_idx, k in enumerate(rows):
        # conv(a,g)[k] = sum_i a[i]*g[k-i] for valid i
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i  # index into g
            A[row_idx, j] += a[i]
    return A

def solve_lp_with_active_set(a, M, active_rows, relax=1.0):
    """Solve LP using only active constraints."""
    n = len(a)
    k = len(active_rows)
    if k == 0:
        return None, 0.0
    
    A = build_toeplitz_subset(a, active_rows)
    b_ub = np.full(k, M * relax)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = float(np.sum(g))
            duals = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
            return g, S, duals
    except Exception:
        pass
    return None, 0.0, None

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = np.sum(w)
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def gradient_subgradient(a, active_peaks, top_k=5):
    """Compute subgradient of max(conv(a,a)[k]) over active peaks."""
    n = len(a)
    conv = compute_conv(a)
    # Get top_k peaks (or specified active peaks)
    if active_peaks is None:
        sorted_idx = np.argsort(conv)[::-1][:top_k]
    else:
        sorted_idx = active_peaks[:top_k]
    
    # Subgradient: average of gradients at active peaks
    # grad_k[j] = 2*a[k-j] if 0 <= k-j < n else 0
    grad = np.zeros(n)
    for k in sorted_idx:
        k = int(k)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]
    grad /= len(sorted_idx)
    return grad, conv

def line_search(a, direction, obj_cur, n_steps=10):
    """Line search: find best t in [0,1] such that a + t*(direction - a) improves."""
    best_t = 0.0
    best_obj = obj_cur
    best_a = a.copy()
    
    for i in range(n_steps):
        t = (i + 1) / (n_steps + 1)
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = np.sum(a_try)
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
            best_a = a_try.copy()
    
    return best_a, best_obj, best_t

def golden_section_line_search(a, direction, obj_cur):
    """Golden section search for optimal step size."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    def f(t):
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = np.sum(a_try)
        if s < 1e-10:
            return 1e10
        a_try = a_try / s
        return compute_obj(a_try)
    
    lo, hi = 0.0, 1.0
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc, fd = f(c), f(d)
    
    for _ in range(20):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = lo + gr * (hi - lo)
            fc = f(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + (1 - gr) * (hi - lo)
            fd = f(d)
    
    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * direction
    a_best = np.maximum(a_best, 0)
    s = np.sum(a_best)
    if s < 1e-10:
        return a.copy(), obj_cur, 0.0
    a_best = a_best / s
    obj_best = f(t_best)
    if obj_best > obj_cur:
        return a.copy(), obj_cur, 0.0
    return a_best, obj_best, t_best

def resample_sequence(a, new_n):

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) LP-based direction finding with active-set constraint selection,
(2) smooth log-sum-exp gradient polishing, (3) spectral flattening initialization,
and (4) multi-length exploration with adaptive time allocation.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build (2n-1) x n Toeplitz matrix where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def build_toeplitz_rows(a, rows):
    """Build only specific rows of the Toeplitz matrix."""
    n = len(a)
    A = np.zeros((len(rows), n), dtype=np.float64)
    for idx, k in enumerate(rows):
        k = int(k)
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            A[idx, k - i] += a[i]
    return A


def spectral_flatten_init(n, seed=42, n_iters=80):
    """Create initial sequence with flat-ish autocorrelation."""
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.1

        fa = rfft(a)
        mag = np.abs(fa)
        phase = np.angle(fa)

        # Compress magnitude toward mean
        target_mag = np.mean(mag)
        alpha = 0.08 / (1.0 + it * 0.05)
        mag_new = (1 - alpha) * mag + alpha * target_mag
        fa_new = mag_new * np.exp(1j * phase)
        a_new = irfft(fa_new, n)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        obj_cur = compute_obj(a)
        if obj_new < obj_cur:
            a = a_new

    return a


def solve_lp_active(a, M, active_rows, relax=1.0, timeout=30):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M*relax for k in active_rows, g >= 0."""
    n = len(a)
    if len(active_rows) == 0:
        return None, 0.0, None
    A = build_toeplitz_rows(a, active_rows)
    b_ub = np.full(len(active_rows), M * relax)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = float(np.sum(g))
            duals = None
            if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
                duals = res.ineqlin.marginals
            return g, S, duals
    except Exception:
        pass
    return None, 0.0, None


def solve_lp_full(a, M, relax=1.0, timeout=60):
    """Solve LP with all constraints."""
    n = len(a)
    m = 2 * n - 1
    A = build_toeplitz(a)
    b_ub = np.full(m, M * relax)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:

            S = float(np.sum(g))

def smooth_gradient(a, tau=0.01):
    """Compute gradient of smooth approximation tau*log(sum(exp(conv[k]/tau)))."""
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights
    max_cv = np.max(conv)
    exp_cv = np.exp((conv - max_cv) / tau)
    w = exp_cv / np.sum(exp_cv)

    # Gradient: sum_k w[k] * 2*a[k-j] for valid k-j
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i
            if 0 <= j < n:
                grad[j] += 2.0 * w[k] * a[i]
    return grad, conv, w


def smooth_encobj(a, tau=0.01):
    """Compute smooth objective: 2*n*tau*log(sum(exp(conv/tau)))/(sum(a))^2."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    max_cv = np.max(conv)
    log_sum = max_cv / tau + np.log(np.sum(np.exp((conv - max_cv) / tau)))
    return 2.0 * n * tau * log_sum / (sum_a * sum_a)


def golden_section_line_search(a, direction, obj_cur, max_evals=15):
    """Golden section search for best step along direction."""
    gr = (np.sqrt(5) - 1) / 2

    def f_of_t(t):
        a_try = (1 - t) * a + t * direction
        a_try = np.maximum(a_try, 0)
        s = np.sum(a_try)
        if s < 1e-10:
            return 1e10
        a_try = a_try / s
        return compute_obj(a_try)

    lo, hi = 0.0, 1.0
    c = lo + gr * (hi - lo)
    d = lo + (1 - gr) * (hi - lo)
    fc = f_of_t(c)
    fd = f_of_t(d)
    evals = 2

    while hi - lo > 1e-6 and evals < max_evals:
        if fc < fd:
            hi, d, fd = d, c, fc
            c = lo + gr * (hi - lo)
            fc = f_of_t(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + (1 - gr) * (hi - lo)
            fd = f_of_t(d)
        evals += 2

    t_best = (lo + hi) / 2
    a_best = (1 - t_best) * a + t_best * direction
    a_best = np.maximum(a_best, 0)
    s = np.sum(a_best)
    if s < 1e-10:
        return a.copy(), obj_cur, 0.0
    a_best = a_best / s
    obj_best = f_of_t(t_best)
    if obj_best >= obj_cur:
        return a.copy(), obj_cur, 0.0
    return a_best, obj_best, t_best


def smooth_polish(a, time_budget=30.0, tau_start=0.05, tau_end=0.005):
    """Polish using smooth gradient descent with decreasing tau."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = tau_start
    step_size = 0.1

    for it in range(5000):
        if time.monotonic() - start > time_budget:
            break
        # Decrease tau over time
        progress = min(1.0, it / 2000.0)
        tau = tau_start * (tau_end / tau_start) ** progress

        grad, conv, w = smooth_gradient(a, tau)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        # Normalized gradient step
        direction = a - step_size * grad / grad_norm
        direction = np.maximum(direction, 0)
        s = np.sum(direction)
        if s < 1e-10:
            step_size *= 0.5
            continue
        direction = direction / s

        a_try, obj_try, t_used = golden_section_line_search(a, direction, compute_obj(a))
        if obj_try < best_obj - 1e-10:
            best_obj = obj_try
            best_a = a_try.copy()
            a = a_try
            step_size = min(step_size * 1.2, 0.5)
        else:
            step_size *= 0.7

    return best_a, best_obj

        # Periodic smooth gradient step
        if it % use_smooth_every == use_smooth_every - 1 and elapsed < time_budget - 5:
            budget_left = time_budget - elapsed
            smooth_budget = min(5.0, budget_left * 0.1)
            if smooth_budget > 1.0:
                a_sm, obj_sm = smooth_polish(best_a, time_budget=smooth_budget,
                                             tau_start=0.02, tau_end=0.005)
                if obj_sm < best_obj - 1e-10:
                    best_obj = obj_sm
                    best_a = a_sm.copy()
                    stall = 0

        # Perturbation when stalled
        if stall > patience:
            rng = np.random.default_rng(it * 104729 + 7 + n)
            improved = False

            # Strategy A: Peak-targeted reduction
            peak_k = int(sorted_idx[0])
            a_pert = best_a.copy()
            for i in range(n):
                j = peak_k - i
                if 0 <= j < n:
                    a_pert[i] *= 0.88
            a_pert = np.maximum(a_pert, 0)
            s_p = np.sum(a_pert)
            if s_p > 1e-10:
                a_pert = a_pert / s_p
                obj_p = compute_obj(a_pert)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_pert.copy()
                    improved = True

            # Strategy B: Spectral perturbation
            if not improved:
                fa = rfft(best_a)
                mag = np.abs(fa)
                phase = np.angle(fa)
                noise_scale = rng.uniform(0.1, 0.3)
                mag_pert = mag * np.exp(rng.normal(0, noise_scale, len(mag)))
                a_pert2 = irfft(mag_pert * np.exp(1j * phase), n)
                a_pert2 = np.maximum(a_pert2, 0)
                s_p2 = np.sum(a_pert2)
                if s_p2 > 1e-10:
                    a_pert2 = a_pert2 / s_p2
                    obj_p2 = compute_obj(a_pert2)
                    if obj_p2 < best_obj:
                        best_obj = obj_p2
                        best_a = a_pert2.copy()
                        improved = True

            # Strategy C: Random multiplicative perturbation
            if not improved:
                scale = rng.uniform(0.03, 0.12)
                a_rand = best_a * np.exp(rng.normal(0, scale, n))
                a_rand = np.maximum(a_rand, 0)
                s_r = np.sum(a_rand)
                if s_r > 1e-10:
                    a_rand = a_rand / s_r
                    obj_r = compute_obj(a_rand)
                    if obj_r < best_obj:
                        best_obj = obj_r
                        best_a = a_rand.copy()
                        improved = True

            if not improved:
                # Accept the perturbation anyway to escape
                a_pert = best_a * np.exp(rng.normal(0, 0.05, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    best_a = a_pert / s_p

            stall = 0
            patience = max(15, patience * 3 // 4)

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid LP+smooth-gradient optimization with active-set LP,
    spectral initialization, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.6f}")

    # Phase 1: Hybrid polish on parent
    if best_seq is not None:
        n_p = len(best_seq)
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.2, 250.0)
        print(f"\n=== Phase 1: Hybrid polish (n={n_p}, budget={alloc:.0f}s) ===")
        a_pol, obj_pol = lp_gradient_hybrid(best_seq, max_iters=5000,
                                            time_budget=alloc, n_top_peaks=5)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  After hybrid: {best_score:.6f}")
        else:
            print(f"  After hybrid: {obj_pol:.6f} (no improvement)")

    # Phase 2: Multi-n exploration
    print(f"\n=== Phase 2: Multi-n Exploration ===")

    n_candidates = []
    if best_seq is not None:
        np_ = len(best_seq)
        n_candidates = list(range(max(100, np_ - 100), np_ + 400, 50))
    else:
        n_candidates = list(range(300, 1100, 100))

    # Add specific values
    n_candidates += [500, 600, 700, 800, 900, 1000, 1100]
    n_candidates = sorted(set(max(100, min(1200, n)) for n in n_candidates))

    # Prefer n values close to current best
    if best_seq is not None:
        np_ = len(best_seq)
        n_candidates.sort(key=lambda x: abs(x - np_))

    print(f"  Candidates: {n_candidates[:15]}...")

    tried_n = set()
    for n in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        if n in tried_n:
            continue
        tried_n.add(n)

        # Time allocation: give more time to n values close to current best
        if best_seq is not None:
            proximity = 1.0 - min(1.0, abs(n - len(best_seq)) / 300.0)
        else:
            proximity = 0.5
        alloc = min(remaining * 0.12 * (0.5 + proximity), 120.0)
        alloc = max(25.0, alloc)

        # Starting point
        if best_seq is not None:
            a_start = resample_sequence(best_seq, n)
        else:
            a_start = spectral_flatten_init(n, seed=seed + n * 7919)

        print(f"  N={n} ({proximity:.2f}): {remaining:.0f}s left, alloc={alloc:.0f}s")

        a_opt, obj_opt = lp_gradient_hybrid(
            a_start, max_iters=3000, time_budget=alloc * 0.75, n_top_peaks=5
        )

        # Smooth polish
        if time.monotonic() < deadline - 30 and obj_opt < np.inf:
            rem = deadline - time.monotonic()
            smooth_b = min(10.0, rem * 0.1)
            if smooth_b > 2.0:
                a_sm, obj_sm = smooth_polish(a_opt, time_budget=smooth_b,
                                             tau_start=0.03, tau_end=0.005)
                if obj_sm < obj_opt:
                    a_opt, obj_opt = a_sm, obj_sm

        print(f"  N={n}: obj={obj_opt:.6f}")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Phase 3: Final intensive polish
    print(f"\n=== Phase 3: Final Polish ===")
    print(f"  Best before: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()

        # Round 1: Long hybrid
        alloc1 = min(remaining * 0.35, 200.0)
        a_r1, obj_r1 = lp_gradient_hybrid(best_seq, max_iters=5000,
                                          time_budget=alloc1, n_top_peaks=7)
        if obj_r1 < best_score:
            best_score = obj_r1
            best_seq = a_r1.copy()
            print(f"  Round 1: {best_score:.6f}")
        else:
            print(f"  Round 1: {obj_r1:.6f} (no improvement)")

        # Round 2: Smooth polish
        if time.monotonic() < deadline - 20:
            rem2 = deadline - time.monotonic()
            alloc2 = min(rem2 * 0.3, 100.0)
            a_r2, obj_r2 = smooth_polish(best_seq, time_budget=alloc2,
                                         tau_start=0.02, tau_end=0.003)
            if obj_r2 < best_score:
                best_score = obj_r2
                best_seq = a_r2.copy()
                print(f"  Round 2 (smooth): {best_score:.6f}")
            else:
                print(f"  Round 2 (smooth): {obj_r2:.6f}")

        # Round 3: Another hybrid with different params
        if time.monotonic() < deadline - 20:
            rem3 = deadline - time.monotonic()
            alloc3 = min(rem3 * 0.3, 100.0)
            a_r3, obj_r3 = lp_gradient_hybrid(best_seq, max_iters=5000,
                                              time_budget=alloc3, n_top_peaks=10,
                                              use_smooth_every=10)
            if obj_r3 < best_score:
                best_score = obj_r3
                best_seq = a_r3.copy()
                print(f"  Round 3: {best_score:.6f}")
            else:
                print(f"  Round 3: {obj_r3:.6f}")

    # Phase 4: Spectral micro-polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        print(f"\n=== Phase 4: Spectral Micro-Polish ===")
        a_cur = best_seq.copy()
        n = len(a_cur)
        rng = np.random.default_rng(seed + 4242)
        improved_count = 0

        for _ in range(int(min(30, remaining / 1.5))):
            if time.monotonic() > deadline - 10:
                break
            fa = rfft(a_cur)
            mag = np.abs(fa)
            phase = np.angle(fa)

            choice = rng.integers(0, 4)
            if choice == 0:
                target = np.mean(mag)
                alpha = rng.uniform(0.03, 0.1)
                mag_new = (1 - alpha) * mag + alpha * target
            elif choice == 1:
                freqs = np.linspace(0, 1, len(mag))
                tilt = rng.uniform(0.3, 1.0)
                mag_new = mag * np.exp(-tilt * freqs ** 2)
                mag_new = mag_new / mag_new[0]
            elif choice == 2:
                target = np.median(mag)
                alpha = rng.uniform(0.05, 0.15)
                mag_new = (1 - alpha) * mag + alpha * target
            else:
                # Log compression
                mag_log = np.log(mag + 1e-12)
                tl = np.mean(mag_log)
                alpha = rng.uniform(0.1, 0.25)
                mag_log_new = (1 - alpha) * mag_log + alpha * tl
                mag_new = np.exp(mag_log_new)

            fa_new = mag_new * np.exp(1j * phase)
            a_new = irfft(fa_new, n)
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_score:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    a_cur = a_new
                    improved_count += 1

        print(f"  Improvements: {improved_count}")
        print(f"  Final score: {best_score:.6f}")

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s")

    return result
