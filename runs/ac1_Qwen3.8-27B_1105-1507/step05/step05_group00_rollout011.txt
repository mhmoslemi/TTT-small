from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

    """Solve LP: min t s.t. (a*g)[k] <= t for all k, sum(g)=1, g>=0."""
    n = len(a)

    # Variables: g_0, ..., g_{n-1}, t
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Constraints: (a*g)[k] - t <= 0 for k = 0, ..., 2n-2
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A

    b_ub = np.zeros(2 * n - 1)

    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

"""
Min-peak LP alternation with smoothed gradient refinement for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses direct peak-minimizing LP directions,
log-sum-exp smoothed gradient descent, golden-section line search, and

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_minpeak_lp(a, time_limit=25):
    """
    Solve LP: min t s.t. (a*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None).
    """
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'mip_rel_gap': 1e-6})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                M_opt = float(res.x[n])
                return g, M_opt
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def smooth_gradient_step(a, beta=50.0, step_size=0.03):

def smooth_gradient_descent(a, max_iters=300, time_budget=30.0, beta_start=30.0, beta_end=200.0):

    Iterative smoothed gradient descent with adaptive beta and step size.
    """

            a = a_new
            step = min(step * 1.02, 0.1)
        else:

            step *= 0.9

    return best_a


def golden_section_line_search(a, g, f_func, tol=1e-8, max_iter=50):
    """
    Golden section search for optimal blend: a_new = (1-t)*a + t*g, t in [0,1].
    Returns (best_t, best_a).
    """
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f_func((1 - c) * a + c * g)
    fd = f_func((1 - d) * a + d * g)

    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_func((1 - c) * a + c * g)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f_func((1 - d) * a + d * g)

    best_t = (lo + hi) / 2.0
    a_best = np.maximum((1 - best_t) * a + best_t * g, 0)
    s = np.sum(a_best)
    if s > 1e-10:
        a_best = a_best / s
    return best_t, a_best

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

def structured_perturbation(a, seed):
    """Apply structured perturbation: mix of sinusoids and noise."""
    n = len(a)
    rng = np.random.default_rng(seed)
    # Add small sinusoidal perturbation
    freqs = rng.integers(1, 6, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.01, 0.03, size=3)
    x = np.linspace(0, 2 * np.pi, n)
    pert = np.zeros(n)
    for f, p, amp in zip(freqs, phases, amps):
        pert += amp * np.sin(f * x + p)
    # Add small noise
    pert += rng.normal(0, 0.01, n)
    a_new = a + pert
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s

            print(f"  ** NEW BEST: n={n_try}, obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj:.8f} (no improvement)")

    print(f"\nAfter Phase 2: best={best_score:.8f}, n={len(best_seq)}")

    # === Phase 3: Final intensive polish on best ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Final polish (budget: remaining)")
    print(f"{'='*60}")
    phase3_end = deadline

    a_final = best_seq.copy().astype(np.float64)
    s = np.sum(a_final)
    if s > 1e-10:
        a_final = a_final / s

    polish_round = 0
    while time.monotonic() < phase3_end - 20:
        remaining = phase3_end - time.monotonic()
        if remaining < 15:
            break
        polish_round += 1
        print(f"\n  Polish round {polish_round} ({remaining:.0f}s left), current={best_score:.8f}")

        a_pol, obj_pol = polish_sequence(a_final, time_budget=min(remaining * 0.4, 50.0),
                                         verbose=True)
        if obj_pol < best_score - 1e-12:
            best_score = obj_pol
            best_seq = a_pol.copy()
            a_final = a_pol.copy()
            print(f"  ** POLISH improved: {best_score:.8f} **")
        else:
            # Try perturbation + LP
            seed_p = polish_round * 31337 + seed
            a_pert = structured_perturbation(a_final, seed_p)
            a_pert = lp_optimize(a_pert, max_iters=20, time_budget=min(remaining * 0.3, 30.0))
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_score - 1e-12:
                best_score = obj_pert
                best_seq = a_pert.copy()
                a_final = a_pert.copy()
                print(f"  ** PERT improved: {best_score:.8f} **")
            else:
                # Try a different perturbation scale
                n = len(a_final)
                rng = np.random.default_rng(seed_p + 1)
                a_pert2 = a_final * np.exp(rng.normal(0, 0.015, n))
                a_pert2 = np.maximum(a_pert2, 0)
                a_pert2 = a_pert2 / np.sum(a_pert2)
                a_pert2 = smooth_gradient_descent(a_pert2, max_iters=100,
                                                   time_budget=min(remaining * 0.2, 15.0),
                                                   beta_start=50, beta_end=200)
                obj_pert2 = compute_obj(a_pert2)
                if obj_pert2 < best_score - 1e-12:
                    best_score = obj_pert2
                    best_seq = a_pert2.copy()
                    a_final = a_pert2.copy()
                    print(f"  ** NOISE improved: {best_score:.8f} **")

    # === Phase 4: Final validation ===
    print(f"\n{'='*60}")
    print(f"Final validation")
    print(f"{'='*60}")

    if best_seq is not None:
        # Clamp and validate
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        result = [float(x) for x in best_seq]
        val = evaluate_sequence(result)
        if val < np.inf:
            if val < best_score:
                best_score = val
            print(f"evaluated: {val:.8f}")
        else:
            print("WARNING: evaluate returned inf, using fallback")
            result = [1.0] * 100
            best_score = 2.0

    else:
        result = [1.0] * 700
        best_score = compute_obj(np.ones(700))

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
