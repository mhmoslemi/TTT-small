"""
Hybrid optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2:
1. Log-sum-exp smoothed optimization with L-BFGS continuation method (main innovation)
2. Peak-directed coordinate descent with golden-section line search
3. Multi-length exploration and multi-start diversification
4. Spectral-aware initialization and shake-and-bake restarts

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lse_obj_grad(z, tau):
    """
    Log-sum-exp smoothed objective and gradient w.r.t. z (softmax parameterization).
    a = softmax(z), c = a*a (full conv), 
    f = (1/tau)*log(sum_k exp(tau*c_k))
    """
    # Softmax
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    n = len(a)

    # Convolution
    c = fast_conv(a)

    # Log-sum-exp weights
    tc = tau * c
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    w = exp_tc / exp_tc.sum()

    # Objective
    obj = (tc_max + np.log(exp_tc.sum())) / tau

    # S2 = weighted average of c
    S2 = np.dot(w, c)

    # S1_k = sum_{m=0}^{n-1} w[m+k] * a[m] for k=0..n-1
    # Use FFT for efficiency
    size = 1
    while size < 2 * n:
        size <<= 1
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    # cross-correlation: S1_k = sum_m w[m+k]*a[m]
    # = IFFT(FFT(w) * conj(FFT(a)))[k] ... need to verify
    # Actually: circular cross-correlation (a ⊛ w)[k] = sum_m a[m] * w[m+k] (mod size)
    # With zero-padding, this equals the linear cross-correlation for k < n
    corr = np.fft.irfft(fw * np.conj(fa), size)
    S1 = corr[:n]

    # Gradient: df/dz_k = 2 * a_k * (S1_k - S2)
    grad_z = 2.0 * a * (S1 - S2)

    return obj, grad_z


def lse_continuation(a0, tau_min=2.0, tau_max=200.0, n_tau=35, max_iter_per_tau=300, deadline=None):
    """
    Continuation method: optimize LSE-smoothed objective with increasing tau.
    Uses L-BFGS-B on the softmax parameterization.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    # Initialize z from a
    z = np.log(a)
    z = z - z.mean()  # center for numerical stability

    # Tau schedule (geometric)
    taus = np.geomspace(tau_min, tau_max, n_tau)

    best_a = a.copy()
    best_score = compute_J(a)

    for ti, tau in enumerate(taus):
        if deadline is not None and time.monotonic() > deadline:
            break

        t0 = time.monotonic()

        # L-BFGS optimization
        def fun_g(z_vec):
            obj, grad = lse_obj_grad(z_vec, tau)
            return obj, grad

        result = minimize(fun_g, z, method='L-BFGS-B',
                          jac=True, options={'maxiter': max_iter_per_tau,
                                             'ftol': 1e-12, 'gtol': 1e-8,
                                             'maxfun': max_iter_per_tau * 3})

        if result.success or result.fun < 1e6:
            z = result.x
            a = np.exp(z - z.max())
            a = a / a.sum()
            a = np.maximum(a, 0)
            if a.sum() > 0:
                a = a / a.sum()

            score = compute_J(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        elapsed = time.monotonic() - t0
        if ti % 7 == 0:
            print(f"    tau={tau:.1f}: J={best_score:.8f} (iter time {elapsed:.2f}s)", flush=True)

    return best_a


def golden_section_line_search(a, conv, i, s, n, max_iters=30):
    """
    Golden section search to find optimal value for a[i].
    Returns (best_val, best_obj).
    """
    old = a[i]
    lo = max(0, i)
    hi = min(2 * n - 2, i + n - 1)
    a_slice = a[lo - i:hi + 1 - lo].copy()
    kq = 2 * i
    valid_kq = kq <= 2 * n - 2

    # Determine search range
    if old < 1e-10:
        x_lo, x_hi = 0.0, 4.0 * max(np.median(a) * 2, 1e-4)
    else:
        x_lo, x_hi = 0.0, 2.5 * old

    def evaluate_at(x):
        delta = x - old
        conv[lo:hi + 1] += 2.0 * delta * a_slice
        if valid_kq:
            conv[kq] += delta * delta
        new_s = s - old + x
        if new_s < 1e-10:
            val = float('inf')
        else:
            val = 2.0 * n * conv.max() / (new_s * new_s)
        conv[lo:hi + 1] -= 2.0 * delta * a_slice
        if valid_kq:
            conv[kq] -= delta * delta
        return val

    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = x_hi - gr * (x_hi - x_lo)
    d = x_lo + gr * (x_hi - x_lo)
    fc = evaluate_at(c)
    fd = evaluate_at(d)

    for _ in range(max_iters):
        if fc < fd:
            x_hi = d
            d = c
            fd = fc
            c = x_hi - gr * (x_hi - x_lo)
            fc = evaluate_at(c)
        else:
            x_lo = c
            c = d
            fc = fd
            d = x_lo + gr * (x_hi - x_lo)
            fd = evaluate_at(d)

    best_x = (x_lo + x_hi) / 2.0
    best_val = min(fc, fd)
    # Also check endpoints and current
    best_x, best_val = old, evaluate_at(old)
    for x_test in [c, d, 0.0, best_x]:
        v = evaluate_at(x_test)
        if v < best_val:
            best_val = v
            best_x = x_test

    return best_x, best_val


def coord_descent_ls(a, max_sweeps=200, deadline=None, seed=42):
    """
    Coordinate descent with golden-section line search.
    Much more precise than discrete candidates.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = a / a.sum() if a.sum() > 0 else np.ones(n) / n
        s = a.sum()

    conv = fast_conv(a)
    current_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    best_obj = current_obj
    rng = np.random.default_rng(seed)

    no_improve = 0
    max_no_improve = 50

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            new_val, new_obj = golden_section_line_search(a, conv, i, s, n)
            old = a[i]

            if abs(new_val - old) > 1e-12:
                delta = new_val - old
                lo = max(0, i)
                hi = min(2 * n - 2, i + n - 1)
                a_slice = a[lo - i:hi + 1 - lo].copy()
                # Recompute a_slice with old value (should be fine since we haven't changed a[i] yet)
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                kq = 2 * i
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                a[i] = new_val
                s = s - old + new_val
                sweep_improved = True

        if not sweep_improved:
            break

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    return a


def peak_directed_cd(a, max_sweeps=100, deadline=None, seed=0):

    """

        if deadline is not None and time.monotonic() > deadline:
            break

        # Find peak position
        k_peak = int(np.argmax(conv))

        # Compute contribution of each coordinate to the peak
        # Contribution of a[i] to conv[k_peak] is proportional to a[k_peak - i]
        contrib = np.zeros(n)
        for i in range(n):

        order = np.argsort(-contrib)

        for i in order:

            old = a[i]

                lo = max(0, i)
                hi = min(2 * n - 2, i + n - 1)

                conv[lo:hi + 1] += 2.0 * delta * a_slice

                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                a[i] = new_val
                s = s - old + new_val
                sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
        elif not sweep_improved:

    a = best_a.copy()
    a = np.maximum(a, 0)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_init(parent, n, rng, variant=0):
    """Generate diverse initial sequences."""

        if p.sum() > 0:
            p = p / p.sum()
        noise_scale = [0.0, 0.05, 0.15][variant]
        a = np.maximum(p + rng.normal(0, noise_scale, n), 1e-6)
        return a / a.sum()

    variants = [
        np.ones(n) / n,
        np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 2 * t), 0.01),
        np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * 3 * t), 0.01),
        np.maximum(np.exp(-((t - 0.5) / 0.2) ** 2) + 0.2, 0.01),

    ]

    idx = variant % len(variants)
    a = variants[idx].copy()
    a = np.maximum(a + rng.normal(0, 0.02, n), 1e-6)
    return a / a.sum()


def spectral_flatten(a, n_iters=50, deadline=None):
    """
    Iteratively flatten the power spectrum to reduce convolution peaks.
    In frequency domain: try to make |A(w)|^2 as flat as possible.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0)
    a = a / a.sum()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_score = compute_J(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Compute power spectrum
        fa = np.fft.rfft(a, size)
        power = np.abs(fa) ** 2

        # Target: flat spectrum
        target = np.full(len(power), power.mean())

        # Compute "error" spectrum
        error = power - target

        # Time-domain correction: modulate a by the inverse FFT of the error
        # a_new = a * (1 - alpha * IFFT(error / target)) ... this is approximate
        # Better: a_new = a - alpha * IFFT(error * A_conj / |A|^2) ... 
        # Simplest: reduce a where it contributes to high-power frequencies
        
        # Use a simpler approach: a_new_j = a_j * exp(-alpha * (error_conv)_j)
        # where error_conv is the IFFT of (error / target)
        err_norm = error / (target + 1e-10)
        correction = np.fft.irfft(np.fft.rfft(err_norm, size), size)[:n]
        correction = np.maximum(correction, -1.0)  # clip for safety

        alpha = 0.1 * (1.0 - it / n_iters)
        a_new = a * np.exp(-alpha * correction)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        score = compute_J(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new

    return best_a

    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        a = a / a.sum()
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - (deadline - float(budget_s) + 15.0)
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} [{label}] t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0:
                parent = parent / parent.sum()
            try_update(parent, "parent")
            print(f"Parent: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        try_update(parent, "default")

    t_start = time.monotonic()

    # === PHASE 1: LSE continuation for multiple lengths ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: LSE Continuation Optimization", flush=True)
    print(f"{'=' * 60}", flush=True)

    target_lengths = [512, 768, 1024, 1536]
    # Filter to reasonable range
    base_n = parent.size
    target_lengths = [l for l in target_lengths if l >= min(384, base_n) and l <= 2048]
    # Ensure we always include the parent length if reasonable
    if base_n not in target_lengths and 256 <= base_n <= 2048:
        target_lengths.insert(0, int(base_n))
        target_lengths = sorted(set(target_lengths))

    # Budget allocation: give more time to larger n (harder problem)
    total_p1_budget = 700.0
    n_items = len(target_lengths)
    # Give first length less time, later ones more
    weights = np.linspace(0.5, 1.5, n_items)
    weights = weights / weights.sum()
    p1_budgets = weights * total_p1_budget

    for li, n_target in enumerate(target_lengths):
        remaining = deadline - time.monotonic()
        alloc = min(p1_budgets[li], remaining - 200)  # Leave 200s for later phases
        if alloc < 30:
            break

        phase_deadline = time.monotonic() + alloc
        print(f"\n  n={n_target} (budget {alloc:.0f}s)", flush=True)

        # Try a few initializations
        n_inits = min(4, max(2, int(alloc / 60)))
        for vi in range(n_inits):
            if time.monotonic() > phase_deadline:
                break

            a0 = generate_init(parent, n_target, rng, variant=vi)
            t0 = time.monotonic()

            # LSE continuation
            a_lse = lse_continuation(a0, tau_min=2.0, tau_max=150.0,
                                     n_tau=30, max_iter_per_tau=200,
                                     deadline=phase_deadline)
            elapsed = time.monotonic() - t0
            score = compute_J(a_lse)
            print(f"    LSE init{vi}: J={score:.8f} ({elapsed:.1f}s)", flush=True)
            try_update(a_lse, f"LSE n={n_target} v{vi}")

            # Quick coordinate descent polish
            if time.monotonic() < phase_deadline - 5:
                a_cd = coord_descent_ls(a_lse, max_sweeps=50,
                                        deadline=phase_deadline, seed=seed + li * 10 + vi)
                try_update(a_cd, f"CD n={n_target} v{vi}")

    print(f"\n  Phase 1 best: J={best_score:.8f}", flush=True)

    # === PHASE 2: Peak-directed CD on best ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Peak-Directed Coordinate Descent", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        p2_deadline = min(time.monotonic() + 200, deadline - 200)
        a_peak = peak_directed_cd(best_seq, max_sweeps=200,
                                  deadline=p2_deadline, seed=seed)
        try_update(a_peak, "peak-CD")

        # Multiple rounds with restarts
        for rnd in range(3):
            if time.monotonic() > p2_deadline:
                break
            a_pert = np.maximum(best_seq + rng.normal(0, 0.03, len(best_seq)), 0)
            a_pert = a_pert / a_pert.sum()
            a_p2 = peak_directed_cd(a_pert, max_sweeps=100,
                                    deadline=p2_deadline, seed=seed + 100 + rnd)
            try_update(a_p2, f"peak-CD r{rnd}")

    print(f"\n  Phase 2 best: J={best_score:.8f}", flush=True)

    # === PHASE 3: Spectral flattening ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Spectral Flattening", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        p3_deadline = min(time.monotonic() + 100, deadline - 100)
        a_spec = spectral_flatten(best_seq, n_iters=100, deadline=p3_deadline)
        try_update(a_spec, "spectral")

        # CD after spectral
        if time.monotonic() < p3_deadline:
            a_spec_cd = coord_descent_ls(a_spec, max_sweeps=30,
                                         deadline=p3_deadline, seed=seed + 500)
            try_update(a_spec_cd, "spectral+CD")

    print(f"\n  Phase 3 best: J={best_score:.8f}", flush=True)

    # === PHASE 4: LSE continuation with higher tau on best ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: High-precision LSE on best", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        p4_deadline = min(time.monotonic() + 300, deadline - 60)
        a_hl = lse_continuation(best_seq, tau_min=50.0, tau_max=500.0,
                                n_tau=25, max_iter_per_tau=300,
                                deadline=p4_deadline)
        try_update(a_hl, "LSE-high-tau")

        # CD polish
        if time.monotonic() < p4_deadline:
            a_hl_cd = coord_descent_ls(a_hl, max_sweeps=50,
                                       deadline=p4_deadline, seed=seed + 800)
            try_update(a_hl_cd, "LSE-high+CD")

    print(f"\n  Phase 4 best: J={best_score:.8f}", flush=True)

    # === PHASE 5: Final intensive polish ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        # Extended coordinate descent
        a_final = coord_descent_ls(best_seq, max_sweeps=300,
                                   deadline=deadline - 20, seed=seed + 999)
        try_update(a_final, "final-CD")

        # Peak-directed on the result
        if time.monotonic() < deadline - 30:
            a_fp = peak_directed_cd(a_final, max_sweeps=100,
                                    deadline=deadline - 20, seed=seed + 1111)
            try_update(a_fp, "final-peak")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_seq.tolist()))

    total_time = time.monotonic() - t_start
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={total_time:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Ensure valid output
    result = np.maximum(best_seq.astype(np.float64), 0.0)
    if result.sum() < 0.01:
        result = result / result.sum() if result.sum() > 0 else np.ones(128) / 128
    result = np.minimum(result, 1000.0)

    return [float(x) for x in result]
