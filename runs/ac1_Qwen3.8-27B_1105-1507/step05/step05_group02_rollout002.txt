The subgradient of max_k (a*a)[k] at point a is:

The KKT condition is: there exists lambda >= 0 such that for all j:
  2 * sum_k p[k] * a_conj[k, j] = lambda (for a[j] > 0)
  2 * sum_k p[k] * a_conj[k, j] >= lambda (for a[j] = 0)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def conv_grad(a, k):
    """Compute gradient of (a*a)[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    for j in range(lo, hi + 1):
        grad[j] += 2.0 * a[k - j]
    return grad

def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-8)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, b, b_max

def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=2.0, tau_end=150.0,
                        lr_init=0.3, lr_end=0.003):

    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.85

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a

def lp_best_response(a, M, n, A=None, time_limit=10):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0"""
    if A is None:
        A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def line_search_step(a, g, M_current, max_trials=15):
    """Find best step size by trying various t values."""
    s_g = g.sum()
    if s_g < 1e-10:
        return a, M_current
    g_norm = g / s_g
    best_a = a.copy()
    best_M = M_current
    
    for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005, 0.003]:
        a_t = (1.0 - t) * a + t * g_norm
        a_t = np.maximum(a_t, 0.0)
        st = a_t.sum()
        if st < 1e-10:
            continue
        a_t /= st
        M_t = fast_conv(a_t).max()
        if M_t < best_M:
            best_M = M_t
            best_a = a_t.copy()
    
    return best_a, best_M

        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive LP mirror descent with multi-M ensemble and ultra-fine line search
2. Nesterov smooth gradient descent for exploration and escape from plateaus
3. Momentum-based LP direction averaging
4. Multi-start with diverse structured initializations and length exploration
Key innovations: 7-value M ensemble, momentum LP directions, adaptive kick frequency,
ultra-fine step sizes (down to 0.0005), and extended LP refinement phase.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=2.0, tau_end=120.0,
                        lr_init=0.3, lr_end=0.003, beta=0.85):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def lp_best_response(a, M, n, A=None, time_limit=8):
    if A is None:
        A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def fine_line_search(a, g, M_current):
    s_g = g.sum()
    if s_g < 1e-10:
        return a, M_current
    g_norm = g / s_g
    best_a = a.copy()
    best_M = M_current
    steps = [0.4, 0.25, 0.15, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03, 0.02,
             0.015, 0.01, 0.008, 0.006, 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001]
    for t in steps:
        a_t = (1.0 - t) * a + t * g_norm
        a_t = np.maximum(a_t, 0.0)
        st = a_t.sum()
        if st < 1e-10:
            continue
        a_t /= st
        M_t = fast_conv(a_t).max()
        if M_t < best_M:
            best_M = M_t
            best_a = a_t.copy()
    return best_a, best_M

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.015, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.04, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.01))

    inits.append(np.ones(n) * 0.5)
    inits.append(np.exp(-((t - 0.5) ** 2) / (2 * 0.12 ** 2)))
    inits.append(np.exp(-((t - 0.3) ** 2) / 0.02) + np.exp(-((t - 0.7) ** 2) / 0.02) + 0.05)

    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t + rng.uniform(0, 6.28)), 0.05))

    for alpha in [0.5, 1.0, 1.5, 2.5, 4.0]:
        inits.append(np.maximum(0.1 + 0.9 * (t ** alpha + (1 - t) ** alpha) / 2.0, 0.01))

    k = max(3, n // 25)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))

    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.2, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)

    # Quadratic residue pattern
    q = n
    inits.append(np.maximum(np.array([1.0 if (i * i % q) < q // 3 else 0.2 for i in range(n)]), 0.01))

    # Triangular
    inits.append(np.maximum(1.0 - 2.0 * np.abs(t - 0.5), 0.05))

    return inits
