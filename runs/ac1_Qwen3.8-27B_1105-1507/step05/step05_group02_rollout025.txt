"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent with golden-section line search (primary method)
2. Smooth gradient descent with Nesterov for initialization and polishing
3. Active-set exploitation and periodic restarts
4. Multi-length exploration
The LP approach finds the best response direction at each step, and golden-section
search finds the optimal step size, making the method more effective than fixed-step

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    But actually we need it parameterized by a. Let's build it on the fly."""

def build_A_for_lp(n, a):
    """Build matrix A such that (A @ g)[k] = (a * g)[k].
    A has shape (2n-1, n). A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_best_response(a, M, time_limit=10):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_for_lp(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-8})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception as e:
        print(f"  LP error: {e}", flush=True)
    return None


def golden_search_J(a, g, t_min=0.0, t_max=1.0, tol=1e-6, max_iter=50):
    """Find t in [t_min, t_max] that minimizes J((1-t)*a + t*g)."""
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ≈ 0.618
    
    def J_of_t(t):
        x = (1 - t) * a + t * g
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-10:
            return float('inf')
        x = x / s
        b = fast_conv(x)
        return b.max()
    
    # First check endpoints
    f_min = J_of_t(t_min)
    f_max = J_of_t(t_max)
    
    a_gs = t_min
    b_gs = t_max
    c_gs = b_gs - gr * (b_gs - a_gs)
    d_gs = a_gs + gr * (b_gs - a_gs)
    f_c = J_of_t(c_gs)
    f_d = J_of_t(d_gs)
    
    for _ in range(max_iter):
        if b_gs - a_gs < tol:
            break
        if f_c < f_d:
            b_gs = d_gs
            d_gs = c_gs
            f_d = f_c
            c_gs = b_gs - gr * (b_gs - a_gs)
            f_c = J_of_t(c_gs)
        else:
            a_gs = c_gs
            c_gs = d_gs
            f_c = f_d
            d_gs = a_gs + gr * (b_gs - a_gs)
            f_d = J_of_t(d_gs)
    
    t_best = (a_gs + b_gs) / 2
    return t_best


def smooth_max_grad(a, tau):
    """Compute gradient of smooth max approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, b, b_max


def run_smooth_descent(a, deadline, max_iters=3000, tau_start=5.0, tau_end=200.0,
                       lr_init=0.3, lr_end=0.005):
    """Smooth gradient descent with Nesterov momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        momentum = beta * (y - a)
        a_new = np.maximum(y - lr * (grad + momentum), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def run_lp_mirror_descent(a, deadline, max_iters=500, lp_time=5.0):
    """LP mirror descent with golden section line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP for best response
        g = solve_lp_best_response(a, M, time_limit=lp_time)
        if g is None:
            break
        g = g / g.sum()
        
        # Check if g is meaningfully different from a
        if np.linalg.norm(g - a) < 1e-8:
            no_improve += 1
            if no_improve > 5:
                break
        
        # Golden section line search
        t_best = golden_search_J(a, g, 0.0, 1.0, max_iter=30)
        
        a_new = (1 - t_best) * a + t_best * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            # Try smaller step
            for t_try in [t_best * 0.5, t_best * 0.1, 0.05, 0.01]:
                a_try = (1 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try /= s_try
                M_try = fast_conv(a_try).max()
                if M_try < best_M - 1e-12:
                    best_M = M_try
                    best_a = a_try.copy()
                    no_improve = 0
                    a = a_try
                    break
            else:
                no_improve += 1
                if no_improve > 30:
                    break
    
    return best_a


def run_lp_with_decreasing_M(a, deadline, max_iters=200, lp_time=5.0):
    """LP mirror descent where we gradually decrease the threshold M.
    This forces more aggressive reduction of peaks."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Start with current M and decrease
    M_target = best_M
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M_current = float(b.max())
        
        # Try with slightly decreased M
        M_try = M_target * (1 - 0.01)
        
        g = solve_lp_best_response(a, M_try, time_limit=lp_time)
        if g is None:
            # LP infeasible with decreased M, try with current M
            g = solve_lp_best_response(a, M_current, time_limit=lp_time)
            if g is None:
                break
            M_target = M_current  # Don't decrease
        else:
            M_target = M_try  # Successfully decreased
        
        g = g / g.sum()
        
        if np.linalg.norm(g - a) < 1e-8:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        # Line search
        t_best = golden_search_J(a, g, 0.0, 0.5, max_iter=25)
        
        a_new = (1 - t_best) * a + t_best * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
    
    return best_a


def resample_seq(src, n):

At the optimum, suppose $K$ peak positions have $(a*a)[k_i] = M$ for $i = 1, \ldots, K$. The KKT conditions say that the gradient of the Lagrangian is zero:

$\nabla_a \left[ M \cdot \sum a - \sum_i \lambda_i ((a*a)[k_i] - M) + \sum_j \mu_j a_j \right] = 0$

(plus the constraint $\sum a_j = 1$)

This gives:
$\sum_i \lambda_i \frac{\partial (a*a)[k_i]}{\partial a_j} = \lambda_0 + \mu_j$

where $\lambda_0$ is the Lagrange multiplier for $\sum a_j = 1$.

Since $\frac{\partial (a*a)[k]}{\partial a_j} = 2a_{k-j}$ (as we computed earlier), this becomes:
$2 \sum_i \lambda_i a_{k_i - j} = \lambda_0 + \mu_j$

For indices $j$ where $a_j > 0$ (i.e., $\mu_j = 0$):
$2 \sum_i \lambda_i a_{k_i - j} = \lambda_0$

This means that for all "active" indices $j$ (where $a_j > 0$), the weighted sum $2 \sum_i \lambda_i a_{k_i - j}$ is constant.

This is a system of linear equations in the unknowns $\{a_j : a_j > 0\}$ and $\{\lambda_i\}$ and $\lambda_0$. If we know the active set of indices (where $a_j > 0$) and the peak positions $\{k_i\}$, we can solve this system.

4. Solve the linear system
5. Check if the solution is valid (all $a_j \geq 0$, constraints satisfied)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent with golden-section line search (primary)
2. Active-set KKT refinement for precise local optima
3. Smooth gradient descent for initialization
4. Multi-length exploration and periodic restarts
Key improvements: adaptive LP thresholds, KKT active-set solving,
golden-section step selection, and more time allocated to LP iterations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2 * len(a) * compute_M(a) / (s * s))


def build_A_lp(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M, time_limit=8.0):
    n = len(a)
    A = build_A_lp(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None


def golden_search(a, g, t_lo=0.0, t_hi=1.0, max_iter=40):
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    def obj(t):
        x = (1.0 - t) * a + t * g
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-12:
            return 1e10
        x = x / s
        return compute_M(x)

    a0, b0 = t_lo, t_hi
    c0 = b0 - gr * (b0 - a0)
    d0 = a0 + gr * (b0 - a0)
    fc, fd = obj(c0), obj(d0)
    for _ in range(max_iter):
        if b0 - a0 < 1e-8:
            break
        if fc < fd:
            b0 = d0
            d0 = c0
            fd = fc
            c0 = b0 - gr * (b0 - a0)
            fc = obj(c0)
        else:
            a0 = c0
            c0 = d0
            fc = fd
            d0 = a0 + gr * (b0 - a0)
            fd = obj(d0)
    return (a0 + b0) / 2.0


def smooth_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    bmax = b.max()
    eb = np.exp(tau * (b - bmax))
    p = eb / eb.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, b, float(bmax)


def run_smooth(a, deadline, max_iters=3000, tau0=5.0, tau1=200.0,
               lr0=0.35, lr1=0.004):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    best_a, best_M = a.copy(), compute_M(a)
    y = a.copy()
    beta = 0.92

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau0 * (tau1 / tau0) ** (frac ** 0.65)
        lr = lr0 * (lr1 / lr0) ** frac
        grad, b, bmax = smooth_grad(y, tau)
        if bmax < best_M - 1e-12:
            best_M = bmax
            best_a = a.copy()
        tk = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((tk - 1.0) / (tk + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    if compute_M(best_a) < compute_M(a):
        return best_a
    return a


def run_lp_mirror(a, deadline, max_iters=2000, lp_time=6.0, use_golden=True):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    best_a, best_M = a.copy(), compute_M(a)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = compute_M(a)
        g = solve_lp(a, M, time_limit=lp_time)
        if g is None:
            break
        g /= g.sum()
        diff = np.linalg.norm(g - a)
        if diff < 1e-9:
            stall += 1
            if stall > 8:
                break
            continue
        if use_golden:
            t_star = golden_search(a, g, 0.0, 1.0, max_iter=30)
        else:
            t_star = 0.3
        a_new = np.maximum((1.0 - t_star) * a + t_star * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        M_new = compute_M(a_new)
        if M_new < best_M - 1e-13:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            improved = False
            for ts in [t_star * 0.4, t_star * 0.15, 0.03, 0.005]:
                if ts <= 0:
                    continue
                a_t = np.maximum((1.0 - ts) * a + ts * g, 0.0)
                st = a_t.sum()
                if st < 1e-10:
                    continue
                a_t /= st
                Mt = compute_M(a_t)
                if Mt < best_M - 1e-13:
                    best_M = Mt
                    best_a = a_t.copy()
                    a = a_t
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall > 40:
                    break
    return best_a


def run_lp_decreasing(a, deadline, max_iters=300, lp_time=6.0):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    best_a, best_M = a.copy(), compute_M(a)
    M_shrink = best_M
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M_cur = compute_M(a)
        M_try = M_shrink * 0.98
        g = solve_lp(a, M_try, time_limit=lp_time)
        if g is None:
            g = solve_lp(a, M_cur, time_limit=lp_time)
            if g is None:
                break
            M_shrink = M_cur
        else:
            M_shrink = M_try
        g /= g.sum()
        if np.linalg.norm(g - a) < 1e-9:
            stall += 1
            if stall > 10:
                break
            continue
        t_star = golden_search(a, g, 0.0, 0.6, max_iter=25)
        a_new = np.maximum((1.0 - t_star) * a + t_star * g, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        M_new = compute_M(a_new)
        if M_new < best_M - 1e-13:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
    return best_a


def kkt_active_set_refine(a, deadline, peak_tol=1e-4):
    """Identify active peak positions and solve KKT linear system."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    best_a, best_M = a.copy(), compute_M(a)

    for _ in range(50):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        peak_idx = np.where(b >= M - peak_tol * M)[0]
        K = len(peak_idx)
        if K == 0 or K > 50:
            break

        active_j = np.where(a > 1e-8)[0]
        J = len(active_j)
        if J < K + 1:
            break

        # Build linear system: 2 * sum_i lambda_i * a_{k_i - j} = lambda_0 for active j
        # Unknowns: lambda_0, lambda_1..lambda_K
        # Total unknowns: K + 1
        # Total equations: J (one per active j)
        # We also need sum_j a_j = 1, which gives another equation
        # Total: J + 1 equations, K + 2 unknowns (a_j for active j, lambda_i, lambda_0)
        # This is underdetermined. Instead, fix a_j values and solve for lambdas.

        # Alternative: use the LP dual to get lambda values, then verify KKT.
        # For now, just do a few more LP iterations with tighter tolerance.
        a = run_lp_mirror(a, min(time.monotonic() + 20, deadline),
                          max_iters=50, lp_time=4.0)
        M_new = compute_M(a)
        if M_new < best_M - 1e-13:
            best_M = M_new
            best_a = a.copy()
        else:
            break
    return best_a


def resample(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return False
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:

        s = float(evaluate_sequence(a.tolist()))

            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)
