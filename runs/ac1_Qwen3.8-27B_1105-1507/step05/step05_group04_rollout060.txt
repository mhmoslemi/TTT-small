import numpy as np
from scipy.signal import fftconvolve
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def smooth_max_and_grad(a, tau):
    """Compute smooth max of conv(a,a) and its gradient."""
    b = fast_conv(a)
    b_max = b.max()
    # Shift for numerical stability
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    phi = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: grad[i] = 2 * real(ifft(fft(a) * conj(fft(p))))
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    grad = 2.0 * np.fft.irfft(fa * fp, size)[:n]
    
    return phi, grad, p, b

def project_simplex(a, target_sum=1.0):
    """Project a onto the simplex {x : sum(x) = target_sum, x >= 0}."""
    # First project onto {x : sum(x) = 1, x >= 0}, then scale
    if a.sum() <= 0:
        return np.full_like(a, 1.0/len(a))
    a = a / a.sum()
    # Now project onto probability simplex
    n = len(a)
    if np.all(a >= 0):
        return a
    # Use water-filling
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0)

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_grad(a, tau):

    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Returns: (phi, grad, b) where phi ≈ max(b), grad = ∂phi/∂a, b = conv(a,a)

    exp_sum = exp_b.sum()
    p = exp_b / exp_sum
    phi = b_max + np.log(exp_sum) / tau
    
    # Gradient: 2 * cross_correlation(p, a)
    # (p *⋆ a)[i] = sum_m p[i+m] * a[m]

    conv_len = 2 * n - 1
    size = 1

    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    grad = 2.0 * np.fft.irfft(fp * np.conj(fa), size)[:n]
    
    return phi, grad, b, p

    n = len(a)

def project_simplex(v):
    """Project v onto {x : sum(x) = 1, x >= 0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find largest rho such that u[rho] - (cssv[rho] - 1)/(rho+1) > 0
    # i.e., u[rho]*(rho+1) > cssv[rho] - 1
    # i.e., u[rho]*(rho+1) - cssv[rho] + 1 > 0
    inds = np.arange(1, n + 1)

    rho = np.max(np.where(cond)[0]) + 1  # 1-indexed
    theta = (cssv[rho - 1] - 1.0) / rho

    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    grad = 2.0 * np.fft.irfft(fp * np.conj(fa), size)[:n]
    
    return phi, grad, b

def project_simplex(v):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    inds = np.arange(1, n + 1)
    cond = u * inds > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = np.max(np.where(cond)[0]) + 1
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def adam_smooth_descent(a_init, deadline, tau_schedule, lr_base=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    """Run Adam optimization with homotopy on tau."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    for tau, n_iters in tau_schedule:
        if time.monotonic() > deadline:
            break
        for _ in range(n_iters):
            if time.monotonic() > deadline:
                break
            t += 1
            
            _, grad, b = smooth_max_grad(a, tau)
            M = b.max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            lr = lr_base / np.sqrt(t + 1)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            
            a_new = a - step
            a_new = project_simplex(a_new)
            a = a_new
    
    return best_a, 2 * n * best_M

def make_gaussian(n, center, width):
    x = np.arange(n) / n
    return np.exp(-0.5 * ((x - center) / width) ** 2)

def make_cosine(n, freq, phase=0):
    x = np.arange(n) / n
    return np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * freq * x + phase), 0.01)

def make_polynomial(n, power):
    x = np.arange(n) / (n - 1) if n > 1 else np.array([1.0])
    return np.maximum(x**power + (1-x)**power + 0.1, 0.01)

def make_bimodal(n, centers, widths):
    result = np.zeros(n)
    x = np.arange(n) / n
    for c, w in zip(centers, widths):
        result += np.exp(-0.5 * ((x - c) / w) ** 2)
    return result

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    if parent is not None:

        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    inits.append(make_cosine(n, 2))
    inits.append(make_cosine(n, 3))

    inits.append(make_polynomial(n, 2.0))
    inits.append(make_bimodal(n, [0.25, 0.75], [0.1, 0.1]))

    n_freq = 3
    freqs = rng.integers(1, 8, size=n_freq)

    amps = rng.uniform(0.2, 1.0, size=n_freq)

    for f, ph, am in zip(freqs, phases, amps):
        rand_smooth += am * np.cos(2 * np.pi * f * x + ph)
    inits.append(np.maximum(rand_smooth, 0.01))

def simulate_annealing_restart(a, rng, scale=0.2):
    """Perturb a solution for restart."""
    n = len(a)
    # Mix with random smooth pattern
    x = np.arange(n) / n
    freqs = rng.integers(1, 6, size=2)
    phases = rng.uniform(0, 2 * np.pi, size=2)
    pert = np.zeros(n)
    for f, ph in zip(freqs, phases):
        pert += np.cos(2 * np.pi * f * x + ph)
    a_new = a + scale * pert
    a_new = np.maximum(a_new, 0.001)
    a_new = a_new / a_new.sum()
    return a_new

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based smooth maximum
gradient with Nesterov momentum, adaptive temperature homotopy, diverse structured
initializations, and multi-length exploration. Key difference from LP-based parent:
O(n log n) per step (FFT gradient) vs O(n^2) (LP solve), enabling 50-100x more
iterations for substantially better landscape exploration.
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    """Compute smooth max of conv(a,a) and gradient w.r.t. a via FFT."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_b = np.exp(shifted)
    exp_sum = exp_b.sum()
    p = exp_b / exp_sum
    phi = b_max + np.log(exp_sum) / tau

    # Gradient: 2 * cross_correlation(p, a)[i] = 2 * sum_m p[i+m]*a[m]
    # = 2 * irfft(rfft(p) * conj(rfft(a)))[:n]
    conv_len = 2 * n - 1
    size = 1
    while size < conv_len + n - 1:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    grad = 2.0 * np.fft.irfft(fp * np.conj(fa), size)[:n]

    return phi, grad, b


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    inds = np.arange(1, n + 1, dtype=np.float64)
    cond = u * inds > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = int(np.max(np.where(cond)[0])) + 1
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def nesterov_smooth_descent(a_init, deadline, tau_schedule, lr_base=0.1,
                            momentum=0.9, max_no_improve=200):
    """
    Nesterov-momentum smooth gradient descent with simplex projection.
    tau_schedule: list of (tau, n_iters) pairs.
    Returns (best_a, best_J_unnormalized) where best_J_unnormalized = 2*n*max(b).
    """
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    s = a.sum()
    if s < 1e-10:
        return a_init.copy(), float('inf')
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    v_mom = np.zeros(n)
    no_improve = 0
    global_it = 0

    for tau, n_iters in tau_schedule:
        lr = lr_base / max(1.0, tau / 10.0)
        for _ in range(n_iters):
            if time.monotonic() > deadline:
                break
            global_it += 1

            _, grad, b = smooth_max_grad(a, tau)
            M = b.max()
            if M < best_M - 1e-12:
                best_M = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Nesterov-style momentum
            v_mom = momentum * v_mom - lr * grad
            a_new = a + v_mom
            a_new = project_simplex(a_new)
            a = a_new

            # Adaptive restart if stuck
            if no_improve > max_no_improve:
                _adaptive_restart_inplace(a, global_it)
                v_mom = np.zeros(n)
                no_improve = 0

    J_val = 2.0 * n * best_M
    return best_a, J_val


def _adaptive_restart_inplace(a, it):
    """Perturb solution in-place to escape local minimum."""
    n = len(a)
    rng = np.random.default_rng(it * 6151 + 7)
    scale = max(0.05, 0.2 / (1.0 + 0.1 * np.log1p(it / 100.0)))
    x = np.arange(n) / n
    freqs = rng.integers(1, 6, size=2)
    phases = rng.uniform(0, 2 * np.pi, size=2)
    pert = np.zeros(n)
    for f, ph in zip(freqs, phases):
        pert += np.cos(2 * np.pi * f * x + ph)
    a += scale * pert
    a = np.maximum(a, 0.001)
    a = a / a.sum()

    if s < 1e-10:

    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    v = np.zeros(n)
    t = 0
    no_improve = 0

        for _ in range(n_iters):
            if time.monotonic() > deadline:

            t += 1

            _, grad, b = smooth_max_grad(a, tau)
            M = b.max()
            if M < best_M - 1e-12:
                best_M = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr_base / (1.0 + 0.01 * t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a_new = project_simplex(a_new)
            a = a_new

            if no_improve > 150:
                _adaptive_restart_inplace(a, t)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0

    J_val = 2.0 * n * best_M
    return best_a, J_val


def make_gaussian(n, center, width):
    x = np.arange(n, dtype=np.float64) / n
    return np.exp(-0.5 * ((x - center) / width) ** 2)


def make_cosine(n, freq, phase=0.0, offset=0.5):
    x = np.arange(n, dtype=np.float64) / n
    return np.maximum(offset + (1.0 - offset) * np.cos(2 * np.pi * freq * x + phase), 0.01)


def make_polynomial(n, p1, p2=1.0):
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    return np.maximum(x ** p1 + (1.0 - x) ** p2 + 0.05, 0.01)


def make_bimodal(n, centers, widths):
    result = np.zeros(n)
    x = np.arange(n, dtype=np.float64) / n
    for c, w in zip(centers, widths):
        result += np.exp(-0.5 * ((x - c) / w) ** 2)
    return result


def make_windowed(n, center, width, freq):
    x = np.arange(n, dtype=np.float64) / n
    window = np.exp(-0.5 * ((x - center) / width) ** 2)
    carrier = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x)
    return np.maximum(window * carrier + 0.05, 0.01)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        t_old = np.linspace(0, 1, len(parent))
        t_new = np.linspace(0, 1, n)
        p = np.interp(t_new, t_old, parent)
        inits.append(np.maximum(p, 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.2, n), 0.01))

    inits.append(np.ones(n, dtype=np.float64))
    inits.append(make_gaussian(n, 0.5, 0.25))
    inits.append(make_gaussian(n, 0.5, 0.4))
    inits.append(make_gaussian(n, 0.3, 0.15) + 0.7 * make_gaussian(n, 0.7, 0.15))
    inits.append(make_cosine(n, 1))
    inits.append(make_cosine(n, 2))
    inits.append(make_cosine(n, 3))
    inits.append(make_cosine(n, 5, offset=0.3))
    inits.append(make_polynomial(n, 0.5))
    inits.append(make_polynomial(n, 2.0))
    inits.append(make_bimodal(n, [0.25, 0.75], [0.08, 0.08]))
    inits.append(make_bimodal(n, [0.2, 0.5, 0.8], [0.06, 0.06, 0.06]))
    inits.append(make_windowed(n, 0.5, 0.3, 4))
    inits.append(make_windowed(n, 0.5, 0.3, 7))

    # Random smooth
    n_freq = 3
    freqs = rng.integers(1, 8, size=n_freq)
    phases = rng.uniform(0, 2 * np.pi, size=n_freq)
    amps = rng.uniform(0.2, 0.8, size=n_freq)
    x = np.arange(n, dtype=np.float64) / n
    rand_smooth = np.full(n, 0.5)
    for f, ph, am in zip(freqs, phases, amps):
        rand_smooth += am * np.cos(2 * np.pi * f * x + ph)
    inits.append(np.maximum(rand_smooth, 0.01))

    # Block structure
    n_blocks = max(4, n // 64)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    inits.append(np.repeat(block_vals, n // n_blocks + 1)[:n].astype(np.float64))

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.01)


class ScoreTracker:
    def __init__(self):
        self.best_a = None
        self.best_J = float('inf')
        self.t_start = time.monotonic()

    def update(self, a, label=""):
        if a is None or a.size == 0:
            return
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        J = compute_J(a)
        if J < self.best_J:
            self.best_a = a.copy()
            self.best_J = J
            t = time.monotonic() - self.t_start
            print(f"  ** {label}: J={J:.8f} n={a.size} t={t:.0f}s", flush=True)

    def report(self):
        if self.best_a is not None:
            print(f"  Best: J={self.best_J:.8f} n={self.best_a.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search for optimal step function heights:
    Phase 1: Diverse initialization with quick Nesterov (90s)
    Phase 2: Extended Nesterov + Adam hybrid with tau homotopy (1080s)
    Phase 3: Multi-length exploration (180s)
    Phase 4: Final high-tau polish (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent, "parent")
                print(f"Parent: n={parent.size}, J={tracker.best_J:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "default")

    # === PHASE 1: Diverse initialization (90s) ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: Diverse Initialization + Quick Nesterov", flush=True)
    print(f"{'=' * 60}", flush=True)

    p1_deadline = time.monotonic() + 90.0
    n_main = 512

    candidates = []
    inits = generate_inits(n_main, rng, parent)

    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_deadline:
            break
        a0 = np.maximum(a0, 1e-6)
        a0 = a0 / a0.sum()

        # Quick Nesterov with moderate tau
        tau_sched = [(5, 80), (15, 80), (40, 80)]
        a_opt, val = nesterov_smooth_descent(
            a0, min(time.monotonic() + 7, p1_deadline), tau_sched,
            lr_base=0.08, momentum=0.85, max_no_improve=100
        )
        candidates.append((val, a_opt))
        print(f"  init {idx:2d}: est_J={val:.8f}", flush=True)

    candidates.sort(key=lambda x: x[0])
    for val, a_opt in candidates[:4]:
        tracker.update(a_opt, "phase1")

    print(f"Phase 1 best: est_J={candidates[0][0]:.8f}", flush=True)

    # === PHASE 2: Extended optimization (1080s) ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 2: Extended Nesterov + Adam with Homotopy", flush=True)
    print(f"{'=' * 60}", flush=True)

    p2_deadline = min(time.monotonic() + 1080.0, deadline - 200.0)
    top_k = min(3, len(candidates))
    time_alloc = [0.45, 0.30, 0.25][:top_k]

    for ci in range(top_k):
        if time.monotonic() > p2_deadline:
            break
        remaining_p2 = p2_deadline - time.monotonic()
        alloc = remaining_p2 * time_alloc[ci]
        alloc = max(40.0, alloc)
        phase_dl = min(time.monotonic() + alloc, p2_deadline)

        a_init = candidates[ci][1]
        print(f"\n  Candidate {ci + 1} (est_J={candidates[ci][0]:.6f}), budget={alloc:.0f}s", flush=True)

        # Sub-phase 2a: Nesterov with increasing tau
        tau_sched_n = [
            (5, 200), (10, 200), (20, 250), (50, 250),
            (100, 250), (200, 250), (400, 200)
        ]
        half_dl = min(time.monotonic() + alloc * 0.5, phase_dl - 30)
        a_opt, val = nesterov_smooth_descent(
            a_init, half_dl, tau_sched_n,
            lr_base=0.06, momentum=0.9, max_no_improve=150
        )
        tracker.update(a_opt, f"phase2a_c{ci + 1}")

        # Sub-phase 2b: Adam from Nesterov result
        tau_sched_a = [
            (50, 200), (100, 200), (200, 200),
            (400, 200), (800, 200), (1500, 150)
        ]
        a_opt2, val2 = adam_smooth_descent(
            a_opt, phase_dl, tau_sched_a,
            lr_base=0.03, beta1=0.9, beta2=0.999, eps=1e-5
        )
        tracker.update(a_opt2, f"phase2b_c{ci + 1}")

        # Sub-phase 2c: Restart from best and continue
        if time.monotonic() < phase_dl - 20 and tracker.best_a is not None:
            a_restart = tracker.best_a + rng.normal(0, 0.05, tracker.best_a.size)
            a_restart = np.maximum(a_restart, 0.001)
            a_restart = a_restart / a_restart.sum()
            tau_sched_r = [(100, 200), (300, 200), (600, 200)]
            a_opt3, val3 = nesterov_smooth_descent(
                a_restart, phase_dl, tau_sched_r,
                lr_base=0.04, momentum=0.92, max_no_improve=100
            )
            tracker.update(a_opt3, f"phase2c_restart_c{ci + 1}")

    # === PHASE 3: Multi-length exploration (180s) ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_deadline = min(time.monotonic() + 180.0, deadline - 60.0)

    if tracker.best_a is not None and time.monotonic() < p3_deadline:
        base_n = tracker.best_a.size
        lengths_to_try = [128, 256, 384, 512, 640, 768, 1024, 1536, 2048]
        lengths_to_try = [l for l in lengths_to_try if l != base_n and 64 <= l <= 4096]
        lengths_to_try.sort(key=lambda l: abs(l - base_n))

        for n_try in lengths_to_try[:5]:
            if time.monotonic() > p3_deadline:
                break
            a_try = resample_seq(tracker.best_a, n_try)
            a_try = a_try / a_try.sum()

            per_time = max(20.0, (p3_deadline - time.monotonic()) / max(1, len(lengths_to_try[:5])))
            sub_dl = min(time.monotonic() + per_time, p3_deadline - 10)

            tau_sched = [(10, 100), (50, 100), (200, 100), (500, 80)]
            a_opt, val = nesterov_smooth_descent(
                a_try, sub_dl, tau_sched,
                lr_base=0.06, momentum=0.9, max_no_improve=80
            )
            tracker.update(a_opt, f"n={n_try}")

    # === PHASE 4: Final polish ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 4: Final High-Tau Polish", flush=True)
    print(f"{'=' * 60}", flush=True)
    remaining = deadline - time.monotonic()
    print(f"  Remaining time: {remaining:.0f}s", flush=True)

    if tracker.best_a is not None and remaining > 30:
        # High tau for precision
        tau_sched_polish = [
            (200, 300), (500, 300), (1000, 300),
            (2000, 300), (4000, 200)
        ]
        a_polish, val = nesterov_smooth_descent(
            tracker.best_a, deadline - 5, tau_sched_polish,
            lr_base=0.02, momentum=0.95, max_no_improve=100
        )
        tracker.update(a_polish, "polish_nesterov")

        if time.monotonic() < deadline - 20:
            tau_sched_polish2 = [(500, 300), (1500, 300), (3000, 200)]
            a_polish2, _ = adam_smooth_descent(
                tracker.best_a, deadline - 5, tau_sched_polish2,
                lr_base=0.01, beta1=0.95, beta2=0.999, eps=1e-5
            )
            tracker.update(a_polish2, "polish_adam")

    # Fallback
    if tracker.best_a is None:
        tracker.best_a = np.ones(128)
        tracker.best_J = compute_J(tracker.best_a)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_J:.8f} n={tracker.best_a.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best_a]
