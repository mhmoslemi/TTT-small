def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2*n-1]

    b = fast_conv(a)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Nesterov-accelerated smooth gradient descent for fast exploration
2. LP mirror descent with warm-starting for precise optimization
3. Multi-length and multi-initialization exploration
4. Targeted peak reduction for final polishing
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def compute_M(a):
    return float(fast_conv(a).max())


def nesterov_gradient(a, max_time, tau_start=5.0, tau_end=300.0, lr_start=0.8, name=""):
    """Nesterov-accelerated gradient descent on smooth max approximation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_M = compute_M(a)
    
    x = a.copy()
    v = np.zeros(n)
    
    for it in range(500000):
        if time.monotonic() - t0 > max_time:
            break
        
        frac = min(it / 500.0, 1.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start / (1.0 + 0.02 * it)
        
        # Compute gradient at x (using momentum look-ahead)
        y = x + v  # Nesterov look-ahead
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        b = fast_conv(y)
        b_max = b.max()
        shifted = tau * (b - b_max)
        # Clip to avoid overflow
        shifted = np.clip(shifted, -500, 500)
        exp_b = np.exp(shifted)
        p = exp_b / exp_b.sum()
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-13:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * pk * y[k - hi:k - lo + 1][::-1]
        
        # Nesterov update
        v = 0.9 * v - lr * grad
        x_new = x + v
        x_new = np.maximum(x_new, 0)
        sx = x_new.sum()
        if sx < 1e-10:
            x_new = x
            v = np.zeros(n)
        else:
            x_new = x_new / sx
        v = x_new - x
        x = x_new
        
        M = compute_M(x)
        if M < best_M:
            best_M = M
            best_a = x.copy()
    
    return best_a


def line_search(a, g):
    """Analytic line search: find t in [0,1] minimizing max_k conv((1-t)a+tg, (1-t)a+tg)."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    # Coarse ternary search
    lo, hi = 0.0, 1.0
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    # Fine ternary search
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = max(0.0, min(1.0, (lo + hi) / 2.0))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def solve_lp(a, M, row_idx, col_idx, time_limit=5.0):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.fun < 0:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_time, row_idx, col_idx, verbose=False, lp_time=5.0):
    """LP mirror descent with analytic line search."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    it = 0
    while time.monotonic() - t0 < max_time:
        if no_improve > 200:
            break
        
        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx, time_limit=lp_time)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        
        t, M_new = line_search(a, g)
        it += 1
        
        if t < 1e-10:
            no_improve += 1
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                actual_M = compute_M(a_new)
                if actual_M < best_M - 1e-15:
                    best_M = actual_M
                    best_a = a_new.copy()
                    a = a_new
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if verbose and it % 500 == 0:
            print(f"    LP iter {it}: M={M:.10f} best={best_M:.10f} t={t:.6f} time={time.monotonic()-t0:.0f}s", flush=True)
    
    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def peak_reduction(a, max_time, step_scale=0.01):
    """Targeted reduction of autocorrelation peaks."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_M = compute_M(a)
    
    for it in range(100000):
        if time.monotonic() - t0 > max_time:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Find peaks (within 0.5% of max)
        threshold = 0.995 * M
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Compute gradient of average active peaks
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        grad /= len(active)
        
        # Try a few step sizes
        best_t = 0
        best_val = M
        for t_mult in [0.5, 1.0, 2.0, 5.0]:
            t = step_scale * t_mult
            a_try = np.maximum(a - t * grad, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            M_try = compute_M(a_try)
            if M_try < best_val:
                best_val = M_try
                best_t = t
        
        if best_t == 0:
            step_scale *= 0.5
            if step_scale < 1e-8:
                break
        else:
            a = np.maximum(a - best_t * grad, 0)
            a = a / a.sum()
            step_scale = max(step_scale * 1.1, 1e-8)
            M = compute_M(a)
            if M < best_M:
                best_M = M
                best_a = a.copy()
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated smooth gradient for fast diverse exploration
2. LP mirror descent with tight per-solve time limits (many iterations)

4. Multi-length and multi-initialization exploration
5. Aggressive final LP polish on the best candidate

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def nesterov_gradient(a, max_time, tau_start=5.0, tau_end=500.0, lr_start=1.0):
    """Nesterov-accelerated gradient descent on smooth max approximation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    t0 = time.monotonic()
    best_a = a.copy()
    best_M = compute_M(a)

    x = a.copy()
    v = np.zeros(n)
    momentum = 0.92

    for it in range(1000000):
        if time.monotonic() - t0 > max_time:
            break

        frac = min(it / 300.0, 1.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start / (1.0 + 0.015 * it)

        y = x + momentum * v
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-10:
            y = x.copy()
        else:
            y = y / sy

        b = fast_conv(y)
        b_max = b.max()
        shifted = np.clip(tau * (b - b_max), -400, 400)
        exp_b = np.exp(shifted)
        p = exp_b / exp_b.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-13:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * pk * y[k - hi:k - lo + 1][::-1]

        v = momentum * v - lr * grad
        x_new = x + v
        x_new = np.maximum(x_new, 0)
        sx = x_new.sum()
        if sx < 1e-10:
            x_new = x
            v = np.zeros(n)
        else:
            x_new = x_new / sx
        v = x_new - x
        x = x_new

        M = compute_M(x)
        if M < best_M:
            best_M = M
            best_a = x.copy()

    return best_a


def line_search(a, g):
    """Find best t in [0,1] minimizing max of conv((1-t)a+tg, (1-t)a+tg)."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = max(0.0, min(1.0, (lo + hi) / 2.0))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def solve_lp(a, M, row_idx, col_idx, time_limit=3.0):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.fun < -1e-10:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_time, row_idx, col_idx, lp_time=3.0, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    t0 = time.monotonic()
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < max_time:
        if no_improve > 300:
            break

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx, time_limit=lp_time)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search(a, g)
        it += 1

        if t < 1e-10:
            no_improve += 1
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                actual_M = compute_M(a_new)
                if actual_M < best_M - 1e-16:
                    best_M = actual_M
                    best_a = a_new.copy()
                    a = a_new
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if verbose and it % 1000 == 0:
            print(f"    LP iter {it}: M={M:.10f} best={best_M:.10f} t={t:.6f} "
                  f"time={time.monotonic()-t0:.0f}s", flush=True)

    if verbose:
        print(f"    LP done: {it} iters, best_M={best_M:.10f} "
              f"time={time.monotonic()-t0:.0f}s", flush=True)

    return best_a


def peak_reduction(a, max_time):
    """Directly reduce autocorrelation peaks using gradient on active set."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    t0 = time.monotonic()
    best_a = a.copy()
    best_M = compute_M(a)
    step_scale = 0.005

    for it in range(200000):
        if time.monotonic() - t0 > max_time:
            break

        b = fast_conv(a)
        M = b.max()

        threshold = 0.99 * M
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            break

        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        grad /= len(active)

        best_t = 0.0
        best_val = M
        for mult in [0.3, 0.7, 1.0, 2.0, 4.0]:
            t = step_scale * mult
            a_try = np.maximum(a - t * grad, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            M_try = compute_M(a_try)
            if M_try < best_val:
                best_val = M_try
                best_t = t

        if best_t == 0.0:
            step_scale *= 0.5
            if step_scale < 1e-9:
                break
        else:
            a = np.maximum(a - best_t * grad, 0)
            a = a / a.sum()
            step_scale = min(step_scale * 1.2, 0.05)
            M = compute_M(a)
            if M < best_M:
                best_M = M
                best_a = a.copy()

    return best_a

        tracker.update(a_pk, "P7 peak", t_start)

    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_J = compute_J(tracker.best)
        print(f"  Fallback: J={tracker.best_J:.8f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_J:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
