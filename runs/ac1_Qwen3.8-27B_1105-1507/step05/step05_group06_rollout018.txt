**Refined approach for coordinate descent:**

For each coordinate i:
1. Compute the current c = conv(a, a)
2. The peak value is M = max(c)
3. Identify which k values have c_k close to M (within some tolerance)
4. For a new value x of a_i, compute the new c'_k for affected k
5. The new peak is max(max(unchanged c_k), max(affected c'_k))
6. Minimize 2n * new_peak / (s + x - a_i)^2

The key optimization: since we know which c_k are at the peak, we can focus on those. If changing a_i reduces the peak, great. If it increases some other c_k above the peak, that's bad.

```python
def coord_optim_exact(a, i, c, n, s):
    """Find optimal value of a[i] given current a, c=conv(a,a), n, s=sum(a)."""
    ai = a[i]
    # Affected k range
    k_lo = max(0, i)  # Actually k ranges from i to i+n-1, but let's just use the full range
    k_hi = min(2*n-2, i+n-1)
    
    # Precomputation: for k in [k_lo, k_hi], c'_k(x) = c[k] + 2*(x-ai)*a[k-i] (if k != 2i)
    # or c[k] + (x-ai)*(x+ai) (if k == 2i)
    
    # We want to minimize 2n * max(c'_k for all k) / (s + x - ai)^2
    # Note: c'_k for k outside [k_lo, k_hi] is unchanged
    
    # The unchanged max:
    if k_lo > 0:
        unchanged_lo_max = c[:k_lo].max()
    else:
        unchanged_lo_max = 0
    if k_hi < 2*n-2:
        unchanged_hi_max = c[k_hi+1:].max()
    else:
        unchanged_hi_max = 0
    unchanged_max = max(unchanged_lo_max, unchanged_hi_max)
    
    # For the affected range, we need to compute max of:
    # - c[k] + 2*(x-ai)*a[k-i] for k != 2i in range
    # - c[2i] + (x-ai)*(x+ai) for k = 2i (if in range)
    # - unchanged_max
    
    # This is a convex function (max of linear/quadratic functions) in x
    # We can minimize 2n * max(affected_max(x), unchanged_max) / (s+x-ai)^2
    
    # Let's do a bounded search in x
    # x should be in [0, x_max] where x_max is reasonable (like 2*a.mean() or something)
    
    x_max = max(4 * a.mean(), 2*ai + 0.1)  # reasonable upper bound
    
    # Use scipy minimize_scalar
    from scipy.optimize import minimize_scalar
    
    def obj(x):
        if x < 0:
            return float('inf')
        s_new = s + x - ai
        if s_new < 0.01:
            return float('inf')
        # Compute max of affected c'_k
        max_affected = unchanged_max
        for k in range(k_lo, k_hi+1):
            if k == 2*i and 2*i <= 2*n-2:
                c_new = c[k] + (x-ai)*(x+ai)
            else:
                idx = k - i
                if 0 <= idx < n:
                    c_new = c[k] + 2*(x-ai)*a[idx]
                else:
                    c_new = c[k]
            if c_new > max_affected:
                max_affected = c_new
        return 2*n*max_affected / (s_new*s_new)
    
    result = minimize_scalar(obj, bounds=(0, x_max), method='bounded', options={'xatol': 1e-10})
    return result.x, result.fun
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Brief LP mirror descent for initial improvement
2. Intensive vectorized coordinate descent with exact 1D grid optimization
3. Peak-targeted coordinate updates prioritizing peak-contributing coordinates
4. Multi-restart with structured perturbations and re-optimization
5. Adaptive multi-resolution search (coarse-to-fine on coordinate values)

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    total = (2 * n - 1) * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=30):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def lp_mirror_descent(a, max_iters, deadline, use_cd_every=30):
    """LP mirror descent with fast line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        g = lp_best_response(a, M, time_limit=20)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)

        def obj(t):
            b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
            return b.max()

        result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-9})
        t_opt = float(result.x)

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)

        if J_new < best_J - 1e-13:
            best_a = a_new.copy()
            best_J = J_new
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if use_cd_every > 0 and it % use_cd_every == use_cd_every - 1:
            a_cd = coordinate_descent_intensive(a, deadline, max_passes=3)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-13:
                best_J = J_cd
                best_a = a_cd.copy()
            a = a_cd
            no_improve = 0

        if no_improve > 40:
            a = a + np.random.normal(0, 0.01, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_J


def coordinate_descent_intensive(a, deadline, max_passes=100, verbose=False):
    """
    Intensive coordinate descent with vectorized 1D optimization.
    For each coordinate, finds optimal value via grid search + refinement.
    """
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    c = fast_conv(a)
    current_s = a.sum()

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = np.arange(n)
        np.random.shuffle(order)
        improved = False
        total_improvement = 0.0

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)

            # Unchanged max
            umax = 0.0
            if k_lo > 0:
                umax = max(umax, float(np.max(c[:k_lo])))
            if k_hi < 2 * n - 2:
                umax = max(umax, float(np.max(c[k_hi + 1:])))

            # Precompute affected portion
            k_range = np.arange(k_lo, k_hi + 1)
            idx = k_range - i
            valid = (idx >= 0) & (idx < n)
            a_vals = np.zeros(len(k_range))
            c_vals = c[k_range].copy()
            a_vals[valid] = a[idx[valid]]
            is_quad = valid & (k_range == 2 * i)

            # Determine search range
            mean_val = current_s / n
            if ai > 1e-10:
                x_lo = max(0, ai * 0.2)
                x_hi = ai * 3.0 + 0.5 * mean_val
            else:
                x_lo = 0.0
                x_hi = 4.0 * mean_val

            # Coarse grid: 25 points
            x_grid = np.linspace(x_lo, x_hi, 25)
            best_x = ai
            best_val = 2 * n * max(umax, float(np.max(c_vals))) / (current_s * current_s)

            for x in x_grid:
                if x < 0:
                    continue
                s_new = current_s + x - ai
                if s_new < 0.01:
                    continue
                c_new = c_vals + 2.0 * (x - ai) * a_vals
                if is_quad.any():
                    c_new[is_quad] = c_vals[is_quad] + (x - ai) * (x + ai)
                max_aff = max(umax, float(np.max(c_new)))
                val = 2.0 * n * max_aff / (s_new * s_new)
                if val < best_val:
                    best_val = val
                    best_x = x

            # Fine grid around best: 15 points
            if best_x != ai and best_x > 0:
                span = max((x_hi - x_lo) / 25.0 * 1.5, mean_val * 0.05)
                x_fine = np.linspace(max(0, best_x - span), best_x + span, 15)
                for x in x_fine:
                    if x < 0:
                        continue
                    s_new = current_s + x - ai
                    if s_new < 0.01:
                        continue
                    c_new = c_vals + 2.0 * (x - ai) * a_vals
                    if is_quad.any():
                        c_new[is_quad] = c_vals[is_quad] + (x - ai) * (x + ai)
                    max_aff = max(umax, float(np.max(c_new)))
                    val = 2.0 * n * max_aff / (s_new * s_new)
                    if val < best_val:
                        best_val = val
                        best_x = x

            if best_x != ai:
                # Update a and c
                delta = best_x - ai
                a[i] = best_x
                current_s += delta
                # Update c
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j == i and k < 2 * n - 1:
                        c[k] += 2.0 * delta * ai + delta * delta
                    elif 0 <= j < n and j != i:
                        c[k] += 2.0 * delta * a[j]
                    elif j == i:
                        c[k] += 2.0 * delta * ai + delta * delta
                improved = True
                total_improvement += (best_val - compute_J(a))

            # Periodic recompute to avoid drift
            if (i + 1) % 128 == 0:
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 0.01:
                    a = a / s
                    current_s = s
                    c = fast_conv(a)

        # End of pass: recompute and check
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()
            c = fast_conv(a)
            current_s = a.sum()

        if verbose and p % 10 == 0:
            elapsed = time.monotonic() - (deadline - (max_passes - p) * 0.1)
            print(f"    CD pass {p}: J={best_J:.10f}", flush=True)

        if not improved:
            break

    return best_a


def cd_with_peak_focusing(a, deadline, max_passes=50):
    """Coordinate descent that prioritizes coordinates affecting the peak."""
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()

        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        current_s = a.sum()
        M = float(np.max(c))
        c_sorted = np.argsort(c)[::-1]

        # Process coordinates in order of their impact on peak
        # A coordinate i affects c[k] for k in [i, i+n-1]
        # Coordinates that affect the peak positions are most important
        peak_positions = set(c_sorted[:min(50, 2 * n - 1)].tolist())

        # Sort coordinates by how many peak positions they affect
        coord_impact = []
        for i in range(n):
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            impact = sum(1 for k in range(k_lo, k_hi + 1) if k in peak_positions)
            coord_impact.append((-impact, i))

        coord_impact.sort()
        order = [x[1] for x in coord_impact]

        improved = False
        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)

            umax = 0.0
            if k_lo > 0:
                umax = max(umax, float(np.max(c[:k_lo])))
            if k_hi < 2 * n - 2:
                umax = max(umax, float(np.max(c[k_hi + 1:])))

            k_range = np.arange(k_lo, k_hi + 1)
            idx = k_range - i
            valid = (idx >= 0) & (idx < n)
            a_vals = np.zeros(len(k_range))
            c_vals = c[k_range].copy()
            a_vals[valid] = a[idx[valid]]
            is_quad = valid & (k_range == 2 * i)

            mean_val = current_s / n
            if ai > 1e-10:
                x_lo = max(0, ai * 0.3)
                x_hi = ai * 2.5
            else:
                x_lo = 0.0
                x_hi = 3.0 * mean_val

            # Grid search
            x_grid = np.linspace(x_lo, x_hi, 20)
            best_x = ai
            best_val = 2 * n * max(umax, float(np.max(c_vals))) / (current_s * current_s)

            for x in x_grid:
                if x < 0:
                    continue
                s_new = current_s + x - ai
                if s_new < 0.01:
                    continue
                c_new = c_vals + 2.0 * (x - ai) * a_vals
                if is_quad.any():
                    c_new[is_quad] = c_vals[is_quad] + (x - ai) * (x + ai)
                max_aff = max(umax, float(np.max(c_new)))
                val = 2.0 * n * max_aff / (s_new * s_new)
                if val < best_val:
                    best_val = val
                    best_x = x

            if abs(best_x - ai) > 1e-12:
                delta = best_x - ai
                a[i] = best_x
                current_s += delta
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if j == i:
                            c[k] += 2.0 * delta * ai + delta * delta
                        else:
                            c[k] += 2.0 * delta * a[j]
                improved = True

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def pairwise_cd(a, deadline, max_passes=20):
    """Try pairwise coordinate updates for pairs that both affect the peak."""
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        current_s = a.sum()
        M = float(np.max(c))
        c_sorted = np.argsort(c)[::-1]
        peak_positions = set(c_sorted[:30].tolist())

        # Find pairs of coordinates that both affect some peak position
        # and are close to each other (for efficiency)
        rng = np.random.default_rng(p * 12345 + n)
        n_pairs = min(50, n // 2)
        improved = False

        for _ in range(n_pairs):
            if time.monotonic() > deadline:
                break

            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue

            ai = a[i]
            aj = a[j]

            # Try a 2D grid of (x_i, x_j)
            mean_val = current_s / n
            xi_range = np.linspace(max(0, ai * 0.5), ai * 1.5 + 0.5 * mean_val, 8) if ai > 1e-10 else np.linspace(0, 2 * mean_val, 8)
            xj_range = np.linspace(max(0, aj * 0.5), aj * 1.5 + 0.5 * mean_val, 8) if aj > 1e-10 else np.linspace(0, 2 * mean_val, 8)

            best_score = 2 * n * M / (current_s * current_s)
            best_xi, best_xj = ai, aj

            for xi in xi_range:
                for xj in xj_range:
                    if xi < 0 or xj < 0:
                        continue
                    s_new = current_s + (xi - ai) + (xj - aj)
                    if s_new < 0.01:
                        continue

                    # Compute new c at affected positions
                    # Affected by i: k in [i, i+n-1]
                    # Affected by j: k in [j, j+n-1]
                    k_lo = max(0, min(i, j))
                    k_hi = min(2 * n - 2, max(i + n - 1, j + n - 1))

                    c_new = c[k_lo:k_hi + 1].copy()
                    # Apply changes for i
                    for k in range(max(0, i), min(2 * n - 2, i + n - 1) + 1):
                        jk = k - i
                        if 0 <= jk < n:
                            if jk == i:
                                c_new[k - k_lo] += 2 * (xi - ai) * ai + (xi - ai) ** 2
                            else:
                                c_new[k - k_lo] += 2 * (xi - ai) * a[jk]
                    # Apply changes for j
                    for k in range(max(0, j), min(2 * n - 2, j + n - 1) + 1):
                        jk = k - j
                        if 0 <= jk < n:
                            if jk == j:
                                c_new[k - k_lo] += 2 * (xj - aj) * aj + (xj - aj) ** 2
                            else:
                                c_new[k - k_lo] += 2 * (xj - aj) * a[jk]

                    # Also need unchanged max
                    unchanged_max = 0.0
                    if k_lo > 0:
                        unchanged_max = max(unchanged_max, float(np.max(c[:k_lo])))
                    if k_hi < 2 * n - 2:
                        unchanged_max = max(unchanged_max, float(np.max(c[k_hi + 1:])))

                    new_M = max(unchanged_max, float(np.max(c_new)))
                    score = 2 * n * new_M / (s_new * s_new)
                    if score < best_score:
                        best_score = score
                        best_xi, best_xj = xi, xj

            if best_xi != ai or best_xj != aj:
                a[i] = best_xi
                a[j] = best_xj
                improved = True

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses intensive coordinate descent as the main workhorse, with LP for
    initial improvement, peak-focused CD, pairwise updates, and multi-restart.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent)

    n_parent = parent.size

    # === PHASE 1: Brief LP mirror descent ===
    p1_deadline = min(t_start + 180, deadline - 400)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent (n={n_parent}, budget~180s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    a = parent.copy()
    s = a.sum()
    if s > 0.01:
        a = a / s
    a_result, J_result = lp_mirror_descent(a, 3000, p1_deadline, use_cd_every=20)
    update_best(a_result)
    print(f"  Phase 1 done: J={J_result:.8f}", flush=True)

    # Use the best of parent and LP result
    work_seq = a_result if J_result < compute_J(parent) else parent
    work_seq = np.maximum(work_seq, 0)
    ws = work_seq.sum()
    if ws > 0.01:
        work_seq = work_seq / ws

    # === PHASE 2: Intensive coordinate descent ===
    p2_deadline = min(t_start + 700, deadline - 300)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Intensive Coordinate Descent (budget~500s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    a_cd = coordinate_descent_intensive(work_seq, p2_deadline, max_passes=500, verbose=True)
    update_best(a_cd)
    print(f"  Phase 2 done: J={compute_J(a_cd):.8f}", flush=True)

    # Peak-focused CD
    p2b_deadline = min(t_start + 850, deadline - 200)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2b: Peak-Focused CD (budget~150s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    a_pf = cd_with_peak_focusing(a_cd, p2b_deadline, max_passes=100)
    update_best(a_pf)
    print(f"  Phase 2b done: J={compute_J(a_pf):.8f}", flush=True)

    # === PHASE 3: Pairwise CD ===
    p3_deadline = min(t_start + 1000, deadline - 150)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Pairwise CD (budget~150s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    a_pw = pairwise_cd(a_pf, p3_deadline, max_passes=30)
    update_best(a_pw)
    print(f"  Phase 3 done: J={compute_J(a_pw):.8f}", flush=True)

    # === PHASE 4: Multi-restart with perturbations ===
    p4_deadline = min(t_start + 1250, deadline - 100)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Multi-Restart + CD (budget~250s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    base = best_seq if best_seq is not None else a_pf
    n_base = base.size

    # Try several perturbation scales
    perturb_scales = [0.02, 0.05, 0.1, 0.2]
    n_restarts = 4

    for scale_idx, scale in enumerate(perturb_scales):
        p4_sub = min(time.monotonic(), p4_deadline)
        remaining = p4_deadline - p4_sub
        if remaining < 30:
            break

        per_restart = remaining / (n_restarts - scale_idx) if scale_idx < len(perturb_scales) else remaining
        per_restart = max(30, min(50, per_restart))

        for r in range(n_restarts - scale_idx):
            if time.monotonic() > p4_deadline:
                break

            a_pert = base + rng.normal(0, scale, n_base)
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp < 0.01:
                continue
            a_pert = a_pert / sp

            # Quick CD on perturbed
            a_r = coordinate_descent_intensive(a_pert, time.monotonic() + per_restart, max_passes=50)
            update_best(a_r)

    print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Multi-length exploration ===
    p5_deadline = min(t_start + 1400, deadline - 60)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Multi-Length Exploration (budget~150s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        target_lengths = [int(n_base * m) for m in [0.75, 1.25, 1.5]]
        target_lengths += [384, 512, 640, 768, 1024]
        target_lengths = sorted(set([L for L in target_lengths if 128 <= L <= 2048 and L != n_base]))

        for tgt_n in target_lengths[:4]:
            if time.monotonic() > p5_deadline:
                break
            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()
            rem = max(20, p5_deadline - time.monotonic()) / max(1, len(target_lengths[:4]) - target_lengths[:4].index(tgt_n) - 1)
            a_r = coordinate_descent_intensive(a_ext, time.monotonic() + rem, max_passes=30)
            update_best(a_r)

    print(f"  Phase 5 done: best J={best_score:.8f}", flush=True)

    # === PHASE 6: Final intensive polishing ===
    p6_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 6: Final Polishing", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        rem = max(10, p6_deadline - time.monotonic())
        a_final = coordinate_descent_intensive(best_seq, p6_deadline, max_passes=200, verbose=True)
        update_best(a_final)
        # One more peak-focused pass
        rem2 = max(10, p6_deadline - time.monotonic())
        if rem2 > 20:
            a_pf2 = cd_with_peak_focusing(a_final, p6_deadline, max_passes=50)
            update_best(a_pf2)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        best_score = compute_J(best_seq)
        update_best(best_seq)

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={elapsed:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
```
