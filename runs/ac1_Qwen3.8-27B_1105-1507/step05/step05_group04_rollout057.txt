    """Initialize by iteratively enforcing flat spectrum and non-negativity."""
    b = rng.random(n) + 0.1
    b = b / b.sum()
    
    target_mag = 1.0 / np.sqrt(2*n - 1)  # target magnitude for flat spectrum
    
    for _ in range(num_iters):
        B = np.fft.rfft(b, 2*n-1)
        # Enforce flat magnitude (keeping phase)
        phase = np.angle(B)
        B_new = target_mag * np.exp(1j * phase)

        B_new[0] = 1.0
        b = np.fft.irfft(B_new, 2*n-1)[:n]
        # Enforce non-negativity
        b = np.maximum(b, 0)
        s = b.sum()
        if s < 1e-10:

        b = b / s
    
    return b

    C_max = C.max()

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
