def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_and_gradient(a, tau):
    """Compute smooth max value and its gradient.
    Returns (smooth_max_value, gradient, exact_max, p_weights)
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    b_shifted = b - b_max
    exp_b = np.exp(tau * b_shifted)
    p = exp_b / exp_b.sum()
    smooth_val = (1.0 / tau) * np.log(np.sum(np.exp(tau * b_shifted)) * np.exp(tau * b_max)) - b_max
    # Actually: smooth_max = (1/tau) * log(sum(exp(tau*b)))
    # = (1/tau) * log(exp(tau*b_max) * sum(exp(tau*(b-b_max))))
    # = b_max + (1/tau) * log(sum(exp(tau*(b-b_max))))
    smooth_val = b_max + (1.0 / tau) * np.log(exp_b.sum())
    
    # Gradient: 2 * (p cross-correlated with a)
    # (p ⊛ a)[i] = (p * a_flip)[n-1+i]
    flipped_a = a[::-1]
    conv_pa = fftconvolve(p, flipped_a, mode='full')
    gradient = 2.0 * conv_pa[n-1:n-1+n]
    
    return smooth_val, gradient, b_max, p, b

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def adam_smooth_gradient_descent(a, deadline, tau_init=10.0, tau_max=200.0,
                                   lr_init=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                                   restart_interval=2000, verbose=True):
    """Adam optimization of smooth max of conv(a,a) on the simplex."""
    n = len(a)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    # Adam state
    m = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    
    tau = tau_init
    no_improve = 0
    restart_count = 0
    
    while time.monotonic() < deadline:
        t += 1
        smooth_val, grad, b_max, p, b = smooth_max_and_gradient(a, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat_hat = v_hat / (1 - beta2 ** t)
        
        lr = lr_init / np.sqrt(1 + 0.001 * t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat_hat) + eps)
        a_new = project_simplex(a_new)
        
        # Check improvement
        J_new = compute_J(a_new)
        if J_new < best_J - 1e-10:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive temperature
        if t % 100 == 0:
            tau = min(tau_max, tau * 1.05)
        
        # Restart if stuck
        if no_improve > restart_interval:
            restart_count += 1
            rng = np.random.default_rng(t * 31 + restart_count)
            scale = 0.1 / (1 + 0.1 * restart_count)
            a = a + scale * rng.normal(0, 1, n)
            a = project_simplex(a)
            m = np.zeros(n)
            v_hat = np.zeros(n)
            no_improve = 0
            t = 0
            if verbose:
                print(f"    Restart #{restart_count} at step, best_J={best_J:.8f}, tau={tau:.1f}", flush=True)
        
        if verbose and t % 5000 == 0:
            print(f"    step={t} J={best_J:.8f} bmax={b_max:.6f} tau={tau:.1f} restarts={restart_count}", flush=True)
    
    return best_a, best_J

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def make_palindrome(a):
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        inits.append(p / p.sum())
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01) / 1.0)
        inits.append(np.maximum(p + rng.normal(0, 0.3, n), 0.01) / 1.0)
        pal = make_palindrome(p)
        inits.append(pal / pal.sum())
    
    inits.append(np.ones(n) / n)
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t - 1)), 0.01))
    inits.append(np.maximum(0.3 + 0.7 * np.exp(-4 * (t - 0.5) ** 2), 0.01))
    
    for f in [2, 3, 5, 7]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    
    # Spectral init: random in frequency domain
    for _ in range(3):
        freq = rng.uniform(0, 2 * np.pi, n)
        amp = rng.uniform(0.5, 1.5, n)
        spec = amp * np.exp(1j * freq)
        spec[0] = np.abs(spec[0])  # DC component positive
        a_spec = np.fft.irfft(spec, n)
        inits.append(np.maximum(a_spec, 0.01))
    
    # Block structure
    n_blocks = max(4, n // 32)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    inits.append(np.repeat(block_vals, n // n_blocks + 1)[:n])
    
    # Normalize all
    result = []
    for a in inits:
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
        result.append(a)
    return result

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using adaptive smooth gradient descent
with Adam optimizer, progressive temperature scheduling, multi-length exploration,
palindrome exploitation, and structured restarts.
Key improvements over LP-based approach: (1) O(n log n) FFT-based gradients,
(2) Adam for per-coordinate adaptive steps, (3) 100x more iterations in same time,
(4) multi-scale length search, (5) progressive tau sharpening.
"""

import numpy as np
from scipy.signal import fftconvolve
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_gradient(a, tau):
    """Compute smooth max of conv(a,a) and its gradient.
    Returns (smooth_val, gradient, exact_max, p_weights, conv_values).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    b_shifted = b - b_max
    exp_b = np.exp(tau * b_shifted)
    exp_sum = exp_b.sum()
    p = exp_b / exp_sum
    smooth_val = b_max + np.log(exp_sum) / tau

    # Gradient: 2 * (p cross-correlated with a)
    # (p ⊛ a)[i] = (p * a_flip)[n-1+i]
    flipped_a = a[::-1]
    conv_pa = fftconvolve(p, flipped_a, mode='full')
    gradient = 2.0 * conv_pa[n - 1:n - 1 + n]

    return smooth_val, gradient, b_max, p, b


def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def adam_smooth_descent(a, deadline, tau_init=10.0, tau_max=500.0,
                        lr_init=0.005, beta1=0.9, beta2=0.999, eps=1e-8,
                        restart_after=3000, tau_step=200, tau_mult=1.12,
                        verbose=True, label=""):
    """Adam optimization of smooth max of conv(a,a) on the simplex."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    m = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    tau = tau_init
    no_improve = 0
    restart_count = 0

    while time.monotonic() < deadline:
        t += 1
        smooth_val, grad, b_max, p, b = smooth_max_gradient(a, tau)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v_hat = beta2 * v_hat + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat_hat = v_hat / (1.0 - beta2 ** t)

        lr = lr_init / (1.0 + 0.0005 * t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat_hat) + eps)
        a_new = project_simplex(a_new)

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Progressive temperature increase
        if t % tau_step == 0:
            tau = min(tau_max, tau * tau_mult)

        # Restart if stuck
        if no_improve > restart_after:
            restart_count += 1
            rng = np.random.default_rng(t * 31 + restart_count * 97)
            # Structured perturbation: mix noise with a smooth pattern
            scale = 0.15 / (1.0 + 0.2 * restart_count)
            t_arr = np.linspace(0, 1, n, endpoint=False)
            freqs = rng.integers(1, 6, size=3)
            phases = rng.uniform(0, 2 * np.pi, size=3)
            amp = rng.uniform(0.3, 1.0, size=3)
            pert = np.zeros(n)
            for f, ph, am in zip(freqs, phases, amp):
                pert += am * np.cos(2 * np.pi * f * t_arr + ph)
            noise = rng.normal(0, 0.5, n)
            a = a + scale * (pert + noise)
            a = project_simplex(a)
            m = np.zeros(n)
            v_hat = np.zeros(n)
            no_improve = 0
            t = 0
            tau = max(tau_init, tau * 0.5)  # Reset tau somewhat
            if verbose:
                print(f"    {label} restart#{restart_count} J={best_J:.8f} tau={tau:.1f}", flush=True)

        if verbose and t % 10000 == 0:
            elapsed_info = f"  {label} t={t} J={best_J:.8f} bmax={b_max:.6f} tau={tau:.1f} r={restart_count}"
            print(elapsed_info, flush=True)

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences for optimization."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        inits.append(p / p.sum())
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.2, n), 0.01))
        pal = make_palindrome(p)
        inits.append(pal / pal.sum())

    inits.append(np.ones(n))
    # Cosine bell
    inits.append(np.maximum(0.3 + 0.7 * np.cos(np.pi * (2 * t - 1)) ** 2, 0.01))
    # Gaussian bump
    inits.append(np.maximum(0.2 + 0.8 * np.exp(-6 * (t - 0.5) ** 2), 0.01))
    # Multi-frequency
    for f in [2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))
    # Edge concentration
    inits.append(np.maximum(np.exp(-4 * t ** 2) + np.exp(-4 * (t - 1) ** 2) + 0.05, 0.01))
    # Spectral inits
    for _ in range(3):
        freq = rng.uniform(0, 2 * np.pi, n)
        amp = rng.uniform(0.5, 1.5, n)
        spec = amp * np.exp(1j * freq)
        spec[0] = abs(spec[0])
        a_spec = np.fft.irfft(spec, n)
        inits.append(np.maximum(a_spec, 0.01))
    # Block structure
    n_blocks = max(4, n // 32)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    inits.append(np.repeat(block_vals, n // n_blocks + 1)[:n])
    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Normalize all
    result = []
    for a in inits:
        a = np.maximum(a, 1e-6).astype(np.float64)
        s = a.sum()
        if s > 1e-10:
            result.append(a / s)
    return result
