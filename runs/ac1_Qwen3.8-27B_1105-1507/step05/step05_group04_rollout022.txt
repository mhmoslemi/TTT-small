```python
def compute_grad(a, p):
    """
    a: sequence of length n
    p: probability vector over lags, length 2n-1
    Returns gradient of smooth max w.r.t. a
    """
    n = len(a)
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if p[k] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2 * p[k] * a[k - lo : k - hi + 1][::-1]
        # k - lo to k - hi, reversed
        # When i = lo: k - lo
        # When i = hi: k - hi
        # So a[k-hi : k-lo+1] reversed = a[k-hi : k-lo+1][::-1]
        seg = a[k-hi : k-lo+1][::-1]
        grad[lo:hi+1] += 2.0 * p[k] * seg
    return grad
```

```python
def adam_optimize_smooth_max(a, deadline, tau_start=50, tau_end=500, beta1=0.9, beta2=0.999, lr_start=0.01, lr_end=0.0001):
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    n = len(a)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    best_a = a.copy()
    best_val = fast_conv(a).max()
    
    total_steps = 10000  # estimated
    while time.monotonic() < deadline:
        t += 1
        # Anneal tau
        frac = min(1.0, t / total_steps)
        tau = tau_start * (tau_end / tau_start) ** frac
        # Anneal lr
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Compute smooth max and gradient
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        grad = compute_grad(a, p)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Project: non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s
        
        # Track best
        cur_val = fast_conv(a).max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
```

```python
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Adam on smooth max as complementary method for escaping local minima

import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2 * len(a) * fast_conv(a).max() / (s * s))

def build_conv_matrix(a):
    """Build (2n-1) × n matrix where A[k,i] = a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    k_idx = np.arange(2*n-1)[:, None]
    i_idx = np.arange(n)[None, :]
    diff = k_idx - i_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2*n-1, n))
    A[mask] = a[diff[mask]]
    return A

def solve_lp(a, M):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        return np.maximum(r.x, 0)
    return None

def compute_smooth_max_grad(a, tau):
    """Compute gradient of smooth max approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if p[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k-hi : k-lo+1][::-1]
        grad[lo:hi+1] += 2.0 * p[k] * seg
    return grad, p, b

def adam_smooth_max(a, deadline, tau_start=10, tau_end=500, lr=0.05, 
                     beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimization on smooth max of convolution."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_val = fast_conv(a).max()
    
    # Estimate total steps for annealing
    # We'll anneal over the course of the available time
    while time.monotonic() < deadline:
        t += 1
        frac = min(1.0, t / 5000.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr_t = lr * (1.0 - 0.9 * frac)
        
        grad, p, b = compute_smooth_max_grad(a, tau)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s
        
        cur_val = fast_conv(a).max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        
        if t % 5000 == 0:
            print(f"    Adam: t={t}, val={cur_val:.8f}, best={best_val:.8f}, tau={tau:.1f}", flush=True)
    
    return best_a
```

```python
import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2 * len(a) * fast_conv(a).max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix where A[k,i] = a[k-i] if 0<=k-i<n else 0."""
    n = len(a)
    k_idx = np.arange(2 * n - 1)[:, None]
    i_idx = np.arange(n)[None, :]
    diff = k_idx - i_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = a[diff[mask]]
    return A


def solve_lp_full(a, M):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        return np.maximum(r.x, 0)
    return None


def solve_lp_reduced(a, M, top_frac=0.5):
    """Solve LP with only top constraints for speed. Returns (g, is_full)."""
    n = len(a)
    b = fast_conv(a)
    threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
    active_k = np.where(b >= threshold)[0]
    
    if len(active_k) < 5 or len(active_k) >= 2 * n - 1:
        return solve_lp_full(a, M), False
    
    k_idx = active_k[:, None]
    i_idx = np.arange(n)[None, :]
    diff = k_idx - i_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((len(active_k), n))
    A[mask] = a[diff[mask]]
    
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0)
        if g.sum() > 1e-12:
            return g, True
    return solve_lp_full(a, M), False


def compute_smooth_max_grad(a, tau):
    """Compute gradient of smooth max approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[k - hi: k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * seg
    return grad, p, b

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new

            a_best3, _ = lp_mirror_descent(
                a_pal, min(time.monotonic() + 40, p3_deadline - 5),
                max_no_improve=50, use_full_lp=True, verbose=False, seed=seed + 777
            )
            update_best(a_best3, "palindrome")
    
    # === PHASE 4: Final polishing ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Final Polishing", flush=True)
    print(f"  Remaining: {deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_a is not None and time.monotonic() < deadline - 10:
        # Full LP polish with many iterations
        a_polish, _ = lp_mirror_descent(
            best_a, deadline - 10, max_no_improve=200,
            use_full_lp=True, verbose=True, seed=seed + 12345
        )
        update_best(a_polish, "polish_full")
        
        # One more pass with reduced LP for speed
        if time.monotonic() < deadline - 10:
            a_polish2, _ = lp_mirror_descent(
                best_a, deadline - 10, max_no_improve=150,
                use_full_lp=False, top_frac=0.4, verbose=False, seed=seed + 54321
            )
            update_best(a_polish2, "polish_red")
        
        # Adam polish with very high tau (sharp focus on max)
        if time.monotonic() < deadline - 10:
            a_polish3 = adam_smooth_max(
                best_a, deadline - 10,
                tau_start=200, tau_end=2000, lr=0.02
            )
            update_best(a_polish3, "polish_adam_sharp")
    
    # Fallback
    if best_a is None:
        best_a = np.ones(128)
        best_J = compute_J(best_a)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_J:.10f} n={best_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_a]
```
