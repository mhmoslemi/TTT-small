import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_pattern(n):
    """Pre-compute the sparse pattern for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)

def build_A(n, a, row_idx, col_idx):
    """Build the sparse constraint matrix with current a values."""
    vals = a[row_idx - col_idx]  # A[k,j] = a[k-j]
    return csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))

def lp_solve(n, a, M, row_idx, col_idx):
    """Solve the LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    A = build_A(n, a, row_idx, col_idx)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A.tocsc(), b_ub=b_ub, bounds=bounds,
                    method='highs', options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

"""
LP Mirror Descent optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous:
1. Aggressive LP mirror descent with adaptive step sizes and dual-gap monitoring
2. Faster coordinate descent with subgradient-guided candidate generation
3. Perturbation-based restarts to escape local minima
4. Multi-length exploration (512, 768, 1024) with resampling
5. Pre-computed sparse LP structure for speed
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, nonnegative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


class ScoreTracker:
    def __init__(self):

        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:

            return True
        return False

def build_lp_structure(n):
    """Pre-compute sparse pattern for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(n, a, M, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    # Build values: A[k,j] = a[k-j]
    vals = a[row_idx - col_idx]
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A.tocsc(), b_ub=b_ub, bounds=bounds,
                    method='highs', options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, row_idx, col_idx, max_iters=50000, aggressive=True):
    """LP mirror descent with adaptive step sizes and convergence monitoring."""
    n = len(a)
    a = normalize(a.copy())
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    best_obj = 2.0 * n * best_M  # since sum=1

    t_lr = 0.4  # initial learning rate
    no_improve_count = 0
    max_no_improve = 50
    prev_obj = best_obj

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_conv = fast_conv(a)
        M = float(b_conv.max())
        obj = 2.0 * n * M

        # Convergence check: if M barely changed
        if obj < best_M * (2.0 * n) - 1e-12:
            best_M = obj / (2.0 * n)
            best_obj = obj
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > max_no_improve:
            # Reduce step size and try a few more
            t_lr *= 0.5
            if t_lr < 0.001:
                break
            no_improve_count = 0

        # Solve LP
        g = solve_lp(n, a, M, row_idx, col_idx)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search with backtracking
        improved = False
        if aggressive:
            t_candidates = [t_lr, t_lr * 0.7, t_lr * 0.5, t_lr * 0.25, t_lr * 0.1, 0.05, 0.01]
        else:
            t_candidates = [t_lr, t_lr * 0.5, t_lr * 0.25]

        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial
            M_trial = fast_conv(a_trial).max()
            obj_trial = 2.0 * n * M_trial
            if obj_trial < best_obj - 1e-14:
                a = a_trial
                best_obj = obj_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                # Slightly increase LR if we made big progress
                if aggressive and it % 50 == 0:
                    t_lr = min(0.8, t_lr * 1.05)
                break

        if not improved:
            t_lr *= 0.7
            no_improve_count += 1
            if no_improve_count > max_no_improve * 3:
                break

    return best_a, best_obj / (2.0 * n)


def coord_descent_fast(a, max_sweeps, deadline, seed=0):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    conv = fast_conv(a)
    s = 1.0  # sum after normalization
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if sweep % 5 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        sweep_improved = False

        for i in order:
            old = a[i]
            if old < 1e-10:
                candidates = np.array([0.0, 1e-5, 1e-4, 1e-3])
            else:
                # Finer grid including the "analytical" bound
                candidates = np.array([0.0, 0.25, 0.5, 0.75, 0.9, 1.0, 1.1, 1.25, 1.5]) * old
                # Remove ones too close to old
                mask = ~np.isclose(candidates, old, rtol=1e-3)
                candidates = candidates[mask]

            if len(candidates) == 0:
                continue

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            best_val = old
            best_local = 2.0 * n * conv.max()

            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                # Apply
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_M = conv.max()
                new_s = s - old + x
                new_obj = 2.0 * n * new_M / (new_s * new_s) if new_s > 1e-10 else float('inf')

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                # Rollback
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-13:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                sweep_improved = True

        # Check improvement
        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            break

    best_a = np.maximum(best_a, 0.0)
    return normalize(best_a)


def perturb(a, sigma, rng):
    """Add multiplicative log-normal perturbation preserving sum=1."""
    n = len(a)
    noise = rng.lognormal(0, sigma, n)
    a_pert = a * noise
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
