"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. FFT-based smooth gradient descent with Barzilai-Borwein step sizes
2. LP-based direction finding with adaptive active sets
3. Multi-length exploration and basin hopping
Key improvements: faster gradient via FFT, BB step sizes, adaptive tau,
smaller LP active sets, and diverse multi-length search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    r = np.arange(1, n + 1)
    cond = sv - (cv - 1) / r > 0
    i = np.where(cond)[0][-1] if np.any(cond) else 0
    mu = (sv[i] - 1) / (i + 1)
    return np.maximum(v - mu, 0)


def compute_smooth_grad(a, tau):
    """Compute gradient of 2n * smooth_max(conv(a,a)) using FFT."""
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    exponents = tau * (c - c_max)
    p = np.exp(exponents)
    p = p / p.sum()

    # grad[i] = 4n * sum_k p[k] * a[k-i]
    # Using FFT cross-correlation
    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    r = np.fft.irfft(P * np.conj(A), size)
    grad = 4.0 * n * r[:n]
    return grad, p, c


def bb_step_size(s, y, prev_alpha):
    """Barzilai-Borwein step size (averaged)."""
    sy = np.dot(s, y)
    yy = np.dot(y, y)
    ss = np.dot(s, s)
    if yy < 1e-16 or ss < 1e-16:
        return prev_alpha
    alpha1 = ss / sy if abs(sy) > 1e-16 else prev_alpha
    alpha2 = sy / yy if abs(sy) > 1e-16 else prev_alpha
    alpha1 = np.clip(alpha1, 1e-5, 50.0)
    alpha2 = np.clip(alpha2, 1e-5, 50.0)
    return 0.5 * (alpha1 + alpha2)


def pgd_optimize(a, deadline, max_iters=5000, tau_start=8.0, tau_end=40.0, verbose=False):
    """Projected gradient descent with BB step size and adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    grad, _, c = compute_smooth_grad(a, tau_start)
    alpha = 0.1 / (np.max(np.abs(grad)) + 1e-10)
    prev_s = None
    prev_y = None
    prev_grad = grad.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive tau
        progress = min(1.0, it / max(100, max_iters * 0.5))
        tau = tau_start + (tau_end - tau_start) * progress

        # Take step
        a_new = a - alpha * grad
        a_new = project_simplex(a_new)

        # Compute new gradient
        new_grad, _, new_c = compute_smooth_grad(a_new, tau)

        # BB step size update
        s = a_new - a
        y = new_grad - prev_grad
        if prev_s is not None:
            alpha = bb_step_size(s, y, alpha)
        prev_s = s
        prev_y = y
        prev_grad = new_grad

        a = a_new
        grad = new_grad

        # Check improvement
        J_val = 2.0 * n * new_c.max()
        if J_val < best_J - 1e-14:
            best_J = J_val
            best_a = a.copy()

        # Convergence check
        if np.linalg.norm(grad) < 1e-12:
            break

        if verbose and it % 500 == 0:
            print(f"    PGD iter={it} J={best_J:.8f} tau={tau:.1f} alpha={alpha:.6f}", flush=True)

    return best_a, best_J


def build_lp_rows(a, k_list):
    """Build sparse rows for LP constraints at given convolution indices."""
    n = len(a)
    rows_data = []
    cols_data = []
    vals_data = []
    for idx, k in enumerate(k_list):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows_data.append(idx)
            cols_data.append(j)
            vals_data.append(a[k - j])
    return csr_matrix((vals_data, (rows_data, cols_data)), shape=(len(k_list), n))


def solve_lp_direction(a, M, top_k=80, time_limit=3.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for top constraints, g >= 0."""
    n = len(a)
    b = fast_conv(a)

    # Select top constraints
    if top_k < len(b):
        top_indices = np.argsort(b)[::-1][:top_k]
    else:
        top_indices = np.arange(len(b))
    top_indices = np.sort(top_indices)

    A = build_lp_rows(a, top_indices)
    c_obj = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    bounds = [(0, np.inf)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': time_limit})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            if g.sum() > 1e-10:
                # Verify against all constraints
                ag = fast_cross_conv(a, g)
                if ag.max() <= M * 1.001:
                    return g
                else:
                    # Scale down to satisfy constraints
                    violation = ag.max() / M
                    if violation > 1:
                        g = g / violation
                        return g
                    return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, max_iters=10000, top_k=80, verbose=False):
    """LP mirror descent with adaptive step and basin hopping."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = float(b.max())

        g = solve_lp_direction(a, M, top_k=top_k, time_limit=3.0)
        if g is None or g.sum() < 1e-12:
            continue

        g = g / g.sum()
        diff = np.max(np.abs(g - a))

        # Line search with multiple step sizes
        current_J = compute_J(a)
        base_t = min(0.7, max(0.05, diff * 1.5))
        t_candidates = [base_t, base_t * 0.6, base_t * 0.3, 0.05, 0.02, 0.01, 0.005]

        improved = False
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = project_simplex(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
            elif J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                no_improve = 0
                break

        if not improved:
            no_improve += 1

        # Basin hopping: structured perturbation
        if no_improve > 100:
            scale = min(0.4, 0.1 + 0.05 * (no_improve // 100))
            freqs = rng.integers(1, 8, size=4)
            phases = rng.uniform(0, 2 * np.pi, size=4)
            amps = rng.uniform(0.2, 0.8, size=4)
            t_arr = np.linspace(0, 1, n, endpoint=False)
            pert = np.zeros(n)
            for f, ph, amp in zip(freqs, phases, amps):
                pert += amp * np.cos(2 * np.pi * f * t_arr + ph)
            a = a + scale * pert
            a = project_simplex(a)
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    LP iter={it} J={best_J:.8f} M={M:.6f} no_imp={no_improve}", flush=True)

    return best_a, best_J


def generate_inits(n, parent=None, rng=None):
    """Generate diverse initial sequences of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []

    if parent is not None:
        if len(parent) != n:
            t_old = np.linspace(0, 1, len(parent))
            t_new = np.linspace(0, 1, n)
            p_resampled = np.maximum(np.interp(t_new, t_old, parent), 0.0)
        else:
            p_resampled = parent.copy()
        p_resampled = np.maximum(p_resampled, 1e-6)
        p_resampled = p_resampled / p_resampled.sum()
        inits.append(p_resampled)
        inits.append(project_simplex(p_resampled + 0.05 * rng.normal(0, 1, n)))

    # Hann window
    inits.append(project_simplex(0.5 * (1 + np.cos(2 * np.pi * t))))
    # Raised cosine (flat top)
    inits.append(project_simplex(np.maximum(np.cos(np.pi * (2 * t - 1)), 0) ** 2 + 0.1))
    # Gaussian
    inits.append(project_simplex(np.exp(-8 * (t - 0.5) ** 2)))
    # U-shaped
    inits.append(project_simplex(1 + 0.7 * np.cos(2 * np.pi * t)))
    # W-shaped
    inits.append(project_simplex(1 + 0.5 * np.cos(4 * np.pi * t)))
    # Uniform
    inits.append(np.ones(n) / n)
    # Bimodal
    inits.append(project_simplex(np.exp(-30 * (t - 0.25) ** 2) + np.exp(-30 * (t - 0.75) ** 2)))
    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0.1, 1, n), kernel, mode='same')
    inits.append(project_simplex(sm))

    return inits


def resample_to_length(a, target_n):
    """Resample sequence to target length."""
    if len(a) == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, target_n)
    return project_simplex(np.maximum(np.interp(t_new, t_old, a), 0.0))


def coordinate_descent(a, max_sweeps=50, deadline=None, verbose=False):
    """Coordinate descent: optimize one element at a time."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            old_val = a[i]
            # Try scaling a[i] by different factors
            best_local = old_val
            best_local_J = best_J
            for factor in [0.5, 0.75, 0.9, 1.0, 1.1, 1.25, 1.5, 2.0]:
                trial = a.copy()
                trial[i] = old_val * factor
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-10:
                    continue
                trial_J = compute_J(trial)
                if trial_J < best_local_J:
                    best_local_J = trial_J
                    best_local = trial[i]
            a[i] = best_local
            a = a / a.sum()
            if best_local_J < best_J - 1e-14:
                best_J = best_local_J
                best_a = a.copy()

        if verbose and sweep % 10 == 0:
            print(f"    CD sweep={sweep} J={best_J:.8f}", flush=True)

    return best_a
