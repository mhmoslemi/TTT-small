"""
Smooth-max gradient descent with momentum, adaptive temperature annealing,
line search, and multi-length exploration for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based convolution and gradient computation, with spectral restarts
and peak-targeted perturbations to escape local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj_from_conv(conv, n):
    """Compute objective given precomputed convolution and n (assuming sum(a)=1)."""
    return 2.0 * n * float(np.max(conv))


def compute_smooth_gradient(w, a, L, N_fft):
    """Compute gradient of smooth_max(conv(a,a), t) w.r.t. a.
    grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j in [0, n-1].
    w has length 2n-1, a has length n.
    """
    n = len(a)
    c = np.zeros(L)
    c[:n] = a
    c_flipped = c[::-1]
    W = np.fft.rfft(w, N_fft)
    CF = np.fft.rfft(c_flipped, N_fft)
    conv_result = np.fft.irfft(W * CF, N_fft)
    grad = 2.0 * conv_result[L - 1: L - 1 + n]
    return grad


def smooth_max_descent(a, time_budget=60.0, t_start=2.0, t_end=200.0,
                       max_iters=10000, use_momentum=True, verbose=False):
    """Core smooth max gradient descent with annealing and momentum."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    L = 2 * n - 1
    N_fft = 1
    while N_fft < 2 * L - 1:
        N_fft <<= 1

    best_a = a.copy()
    best_obj = compute_obj_from_conv(compute_conv_fft(a), n)
    start = time.monotonic()

    momentum = np.zeros(n)
    beta = 0.9

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = min(1.0, it / max_iters)
        t = t_start * (t_end / t_start) ** (frac ** 0.7)

        conv = compute_conv_fft(a)

        # Softmax weights (numerically stable)
        conv_shifted = t * (conv - conv.max())
        exp_conv = np.exp(conv_shifted)
        exp_sum = exp_conv.sum()
        if exp_sum < 1e-300:
            break
        w = exp_conv / exp_sum

        # Compute gradient
        grad = compute_smooth_gradient(w, a, L, N_fft)

        # Center gradient (project onto sum=0 hyperplane)
        grad_centered = grad - np.mean(grad)

        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-14:
            break

        # Line search
        improved = False
        base_alpha = 0.5 / (grad_norm * (1.0 + 0.001 * it))
        base_alpha = min(base_alpha, 2.0 / n)

        for mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            alpha = base_alpha * mult

            if use_momentum:
                momentum = beta * momentum + (1 - beta) * grad_centered
                step = alpha * momentum
            else:
                step = alpha * grad_centered

            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            s_new = np.sum(a_new)
            if s_new < 1e-12:
                continue
            a_new = a_new / s_new

            obj_new = compute_obj_from_conv(compute_conv_fft(a_new), n)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break

        if not improved:
            # Reduce momentum and try smaller step
            beta = max(0.5, beta * 0.95)
            a_new = a - (base_alpha * 0.03) * grad_centered
            a_new = np.maximum(a_new, 0.0)
            s_new = np.sum(a_new)
            if s_new > 1e-12:
                a_new = a_new / s_new
                obj_new = compute_obj_from_conv(compute_conv_fft(a_new), n)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    improved = True

    if verbose:
        print(f"    smooth descent: obj={best_obj:.8f}, iters={it}")

    return best_a


def peak_directed_step(a, num_top=3, num_trials=20, time_budget=10.0, seed=0):
    """Directly perturb positions that contribute to peak convolution values."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_from_conv(compute_conv_fft(a), n)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for _ in range(num_trials):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv_fft(a)
        # Get top peak positions
        top_idx = np.argsort(conv)[::-1][:num_top]

        # Pick one peak
        k = int(top_idx[rng.integers(0, len(top_idx))])

        # Find which a indices contribute to this peak
        # conv[k] = sum_i a[i]*a[k-i] for valid i
        contributors = []
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                contributors.append(i)

        if len(contributors) == 0:
            continue

        # Reduce contributions: decrease some contributors
        a_new = a.copy()
        num_perturb = max(1, len(contributors) // 3)
        chosen = rng.choice(contributors, size=min(num_perturb, len(contributors)), replace=False)
        for idx in chosen:
            a_new[idx] *= rng.uniform(0.6, 0.95)

        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-12:
            continue
        a_new = a_new / s_new

        obj_new = compute_obj_from_conv(compute_conv_fft(a_new), n)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def spectral_restart(a, num_trials=10, time_budget=8.0, seed=0):
    """Perturb the phase of the FFT and project back to feasible set."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj_from_conv(compute_conv_fft(a), n)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    for _ in range(num_trials):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a)
        mag = np.abs(fa)
        phase = np.angle(fa)

        # Random phase perturbation
        scale = rng.uniform(0.05, 0.5)
        phase_new = phase + rng.normal(0, scale, len(phase))
        # Keep DC component phase fixed
        phase_new[0] = 0.0

        fa_new = mag * np.exp(1j * phase_new)
        a_new = irfft(fa_new, n)
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-12:
            continue
        a_new = a_new / s_new

        obj_new = compute_obj_from_conv(compute_conv_fft(a_new), n)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)

            elif init_i == 2:
                # Highly asymmetric init
                a_start = np.exp(-np.linspace(0, 3, n_best))
                a_start += 0.1 * np.random.default_rng(seed + 6000 + init_i).random(n_best)
                a_start = a_start / np.sum(a_start)
            elif init_i == 3:
                # Resampled with noise
                a_start = resample_sequence(best_seq, n_best)
                a_start += 0.05 * np.random.default_rng(seed + 7000 + init_i).standard_normal(n_best)
                a_start = np.maximum(a_start, 0)
                a_start = a_start / np.sum(a_start)
            else:
                a_start = random_init(n_best, seed + 8000 + init_i * 31)

            a_opt, obj_opt = combined_optimize(a_start, time_budget=alloc,
                                               seed=seed + 10000 + init_i)
            print(f"  Init {init_i}: obj={obj_opt:.8f}")

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: init {init_i}, obj={best_score:.8f} **")

    # Phase 3: Final intensive polish on the global best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Phase 3: Final Polish, n={n_best}, {remaining:.0f}s remaining ===")

        # Round 1: High-temperature smooth descent (to escape)
        alloc1 = min(remaining * 0.25, 100.0)
        a_r1 = smooth_max_descent(best_seq, time_budget=alloc1,
                                  t_start=0.5, t_end=50.0, use_momentum=True)
        obj_r1 = compute_obj_from_conv(compute_conv_fft(a_r1), n_best)
        if obj_r1 < best_score:
            best_score = obj_r1
            best_seq = a_r1.copy()
            print(f"  Round 1: {best_score:.8f}")

        # Round 2: Low-temperature precision polish
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.3, 120.0)
            a_r2 = smooth_max_descent(best_seq, time_budget=alloc2,
                                      t_start=20.0, t_end=500.0, use_momentum=True)
            obj_r2 = compute_obj_from_conv(compute_conv_fft(a_r2), n_best)
            if obj_r2 < best_score:
                best_score = obj_r2
                best_seq = a_r2.copy()
                print(f"  Round 2: {best_score:.8f}")

        # Round 3: Peak breaking + polish
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            a_r3 = peak_directed_step(best_seq, num_top=8, num_trials=80,
                                      time_budget=min(20.0, remaining * 0.15), seed=seed + 4242)
            obj_r3 = compute_obj_from_conv(compute_conv_fft(a_r3), n_best)
            if obj_r3 < best_score:
                best_score = obj_r3
                best_seq = a_r3.copy()
                print(f"  Round 3 (peak): {best_score:.8f}")

            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_r3b = smooth_max_descent(best_seq, time_budget=min(remaining * 0.4, 80.0),
                                           t_start=30.0, t_end=400.0, use_momentum=True)
                obj_r3b = compute_obj_from_conv(compute_conv_fft(a_r3b), n_best)
                if obj_r3b < best_score:
                    best_score = obj_r3b
                    best_seq = a_r3b.copy()
                    print(f"  Round 3b: {best_score:.8f}")

        # Round 4: Try slight length changes
        if time.monotonic() < deadline - 30:
            for dn in [-20, 20, -50, 50]:
                if time.monotonic() > deadline - 15:
                    break
                n_trial = n_best + dn
                if n_trial < 100:
                    continue
                remaining = deadline - time.monotonic()
                a_trial = resample_sequence(best_seq, n_trial)
                alloc_t = min(remaining * 0.2, 40.0)
                a_opt, obj_t = combined_optimize(a_trial, time_budget=alloc_t, seed=seed + dn)
                if obj_t < best_score:
                    best_score = obj_t
                    best_seq = a_opt.copy()
                    print(f"  Round 4 (n={n_trial}): {best_score:.8f}")

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_from_conv(compute_conv_fft(best_seq / np.sum(best_seq)), 200)

    # Normalize to sum=1 for the final answer (the evaluate function normalizes anyway
    # but let's keep values reasonable)
    s_final = np.sum(best_seq)
    result_list = [float(max(0.0, min(1000.0, x))) for x in (best_seq / s_final)]

    val = evaluate_sequence(result_list)
    if np.isfinite(val) and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result_list)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result_list
