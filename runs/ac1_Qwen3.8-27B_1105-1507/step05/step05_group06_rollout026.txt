"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed log-sum-exp gradient descent with FISTA acceleration (fast FFT-based iterations)
2. Tau annealing schedule for progressively sharpening the max approximation
3. Exact coordinate descent for local refinement
4. LP mirror descent for targeted final polish
5. Multi-start with diverse initializations and multi-length exploration
Key innovation: replaces expensive LP as primary optimizer with fast O(n log n) smoothed
gradient method, allowing 10-100x more iterations in same time.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    Compute the smoothed objective value and its gradient.

    S = a.sum()
    b = fast_conv(a)
    x = b / (S * S)  # normalized autocorrelation values, shape (2n-1,)

    Z = exp_vals.sum()
    w = exp_vals / Z  # softmax weights, shape (2n-1,)
    
    f_val = float(2 * n * (x_max + np.log(Z)) / tau)
    J_val = float(2 * n * x_max)
    
    # Compute g[i] = sum_k w[k] * (a[k-i]*1[0<=k-i<n] + a[i-k]*1[0<=i-k<n])
    # Using FFT decomposition:
    # term2[i] = sum_j a[j] * w[i-j] = conv(a, w)[i]  (first n elements)
    # term1[i] = sum_j a[j] * w[i+j] = conv(a, w_rev)[2n-2-i]
    
    w_rev = w[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    conv_a_w = np.fft.irfft(fa * fw, size)[:3 * n - 2]
    term2 = conv_a_w[:n]
    
    fw_rev = np.fft.rfft(w_rev, size)
    conv_a_wrev = np.fft.irfft(fa * fw_rev, size)[:3 * n - 2]
    term1 = conv_a_wrev[n - 1:2 * n - 1][::-1]
    
    g = term1 + term2
    
    C = float(np.dot(w, b))  # sum_k w[k] * B_k
    
    # grad[i] = (2n/tau) / S^2 * (g[i] - 2*C/S)
    grad = (2.0 * n / tau) / (S * S) * (g - 2.0 * C / S)
    
    return f_val, grad, J_val


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smoothed_fista(a, tau, max_iters, deadline, lr_init=None):
    """
    FISTA for the smoothed objective with simplex constraint.
    a is assumed normalized to sum=1.
    """
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    n = len(a)
    
    # Estimate Lipschitz constant
    if lr_init is None:
        # The gradient of the log-sum-exp is tau-Lipschitz in the Euclidean metric
        # with constant proportional to tau * max(a) * 2n
        # Use a conservative estimate
        lr = 1.0 / (tau * n * 4)
    else:
        lr = lr_init
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    a_curr = a.copy()
    y = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        f_y, grad_y, J_y = compute_smoothed_grad(y, tau)
        
        # FISTA update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * ((it + 1) + 0.5) ** 2)) / 2.0 if it > 0 else 1.0
        t_prev = (1.0 + np.sqrt(1.0 + 4.0 * (it) ** 2)) / 2.0 if it > 0 else 1.0
        
        a_new = project_simplex(y - lr * grad_y)
        
        # Momentum
        momentum = (t_prev - 1) / t_k
        y = a_new + momentum * (a_new - a_curr)
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum > 1e-10:
            y = y / y_sum
        a_curr = a_new
        
        # Check improvement every few iterations
        if it % 20 == 0 or it == max_iters - 1:
            J_check = compute_J(a_curr)
            if J_check < best_J:
                best_J = J_check
                best_a = a_curr.copy()
    
    J_final = compute_J(a_curr)
    if J_final < best_J:
        best_J = J_final
        best_a = a_curr.copy()
    
    return best_a


def smoothed_mirror_descent_schedule(a, deadline, tau_schedule=None, steps_per_tau=None):
    """
    Smoothed mirror descent with tau annealing.
    Uses multiplicative weight updates at each tau level.
    """
    a = np.maximum(a.astype(np.float64), 1e-15)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(len(a)) / len(a)
        S = 1.0
    a = a / S
    n = len(a)
    
    if tau_schedule is None:
        tau_schedule = [10, 25, 50, 100, 200, 400, 800, 1500]
    if steps_per_tau is None:
        # More steps at lower tau
        total = sum(tau_schedule)
        steps_per_tau = [max(5, int(300 * t / total)) for t in tau_schedule]
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for idx_tau, tau in enumerate(tau_schedule):
        if time.monotonic() > deadline:
            break
        
        n_steps = steps_per_tau[idx_tau]
        lr = 2.0 / tau
        
        for step in range(n_steps):
            if time.monotonic() > deadline:
                break
            
            f_val, grad, J_val = compute_smoothed_grad(a, tau)
            
            # Multiplicative weight update with adaptive step
            log_update = -lr * grad
            # Clip for numerical stability
            log_update = np.clip(log_update, -10, 10)
            a_new = a * np.exp(log_update)
            a_new = np.maximum(a_new, 1e-30)
            a_new_sum = a_new.sum()
            if a_new_sum < 1e-10:
                break
            a_new = a_new / a_new_sum
            
            # Check if this helped (compare smoothed objectives)
            f_new, _, _ = compute_smoothed_grad(a_new, tau)
            
            if f_new < f_val + 1e-12:
                a = a_new
            else:
                # Halve step and retry once
                lr = lr * 0.5
                log_update = -lr * grad
                log_update = np.clip(log_update, -10, 10)
                a_new = a * np.exp(log_update)
                a_new = np.maximum(a_new, 1e-30)
                a_new_sum = a_new.sum()
                if a_new_sum > 1e-10:
                    a_new = a_new / a_new_sum
                    a = a_new
            
            # Periodically check true J
            if step % 50 == 0:
                J_check = compute_J(a)
                if J_check < best_J:
                    best_J = J_check
                    best_a = a.copy()
        
        # End of tau level: check J
        J_check = compute_J(a)
        if J_check < best_J:
            best_J = J_check
            best_a = a.copy()
        a = best_a.copy()  # Restart from best for next tau level
    
    return best_a, best_J


def coordinate_descent_exact(a, deadline, max_passes=30):
    """
    Coordinate descent with exact 1D optimization per coordinate.
    For each coordinate i, finds the best value by scanning candidates
    and using the structure of how convolution changes.
    """
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 0.01:
        return a
    a = a / S
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        order = np.random.permutation(n)
        b = fast_conv(a)
        S = a.sum()
        any_improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            ai = a[i]
            
            # Determine affected range of b
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Candidates for new value
            if ai < 1e-8:
                candidates = [0.005, 0.02, 0.05, 0.1, 0.2 / n]
            else:
                candidates = [0.0, ai * 0.3, ai * 0.6, ai * 1.4, ai * 1.8, ai * 2.5]
                # Add some absolute values
                mean_val = 1.0 / n
                candidates.extend([mean_val * 0.5, mean_val * 1.5, mean_val * 2.0])
            
            best_cand = ai
            best_score = 2.0 * n * b.max() / (S * S)
            
            for cand in candidates:
                cand = max(0.0, cand)
                delta = cand - ai
                if abs(delta) < 1e-14:
                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue
                
                # Compute new max of b efficiently
                # b_new[k] = b[k] + 2*delta*a[k-i] for k in [k_lo, k_hi], k != 2i
                # b_new[2i] = b[2i] + 2*delta*ai + delta^2 (if 2i in range)
                
                # Changed indices
                max_changed = 0.0
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if k == 2 * i:
                            new_val = b[k] + 2 * delta * ai + delta * delta
                        else:
                            new_val = b[k] + 2 * delta * a[j]
                        if new_val > max_changed:
                            max_changed = new_val
                
                # Unchanged max
                if k_lo > 0:
                    outside_lo = b[:k_lo].max()
                else:
                    outside_lo = 0.0
                if k_hi < 2 * n - 2:
                    outside_hi = b[k_hi + 1:].max()
                else:
                    outside_hi = 0.0
                outside_max = max(outside_lo, outside_hi, 0.0)
                
                new_max = max(max_changed, outside_max)
                new_score = 2.0 * n * new_max / (S_new * S_new)
                
                if new_score < best_score:
                    best_score = new_score
                    best_cand = cand
            
            if best_cand != ai:
                a[i] = best_cand
                any_improved = True
                # Update b
                delta = best_cand - ai
                S = S + delta
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if k == 2 * i:
                            b[k] += 2 * delta * ai + delta * delta
                        else:
                            b[k] += 2 * delta * a[j]
        
        # Renormalize and check
        a = np.maximum(a, 0)
        S = a.sum()
        if S > 0.01:
            a = a / S
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
        
        if not any_improved:
            break
    
    return best_a


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(a[k - j])
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    vals = np.array(vals_list, dtype=np.float64)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_mirror_descent(a, deadline, max_iters=200):
    """LP mirror descent for final polish."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b_aa = fast_conv(a)
        M = float(b_aa.max())
        
        # Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0
        A = build_lp_matrix(n, a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 30, 'presolve': True})
            if not (r.success and r.x is not None):
                break
            g = np.maximum(r.x, 0)
            g_sum = g.sum()
            if g_sum < 1e-12:
                break
            g = g / g_sum
        except Exception:
            break
        
        # Line search
        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)
        
        def obj_t(t):
            b_t = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
            return b_t.max()
        
        result = minimize_scalar(obj_t, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-7})
        t_opt = float(result.x)
        
        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for t_try in [t_opt * 0.5, t_opt * 0.25, 0.1]:
                a_try = np.maximum((1 - t_try) * a + t_try * g, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                J_try = compute_J(a_try)
                if J_try < best_J - 1e-14:
                    best_J = J_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # No improvement, perturb
                a = a + np.random.normal(0, 0.01, n)
                a = np.maximum(a, 0.001)
                a = a / a.sum()
    
    return best_a, best_J


def generate_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Raised cosine
    rc = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(rc / rc.sum())
    
    # Hamming window
    ham = np.sin(np.pi * t) ** 2
    inits.append(np.maximum(ham, 0.01) / ham.sum())
    
    # Gaussian
    for sig in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((t - 0.5) / sig) ** 2)
        inits.append(g / g.sum())
    
    # Double bump
    db = np.exp(-12 * (t - 0.3) ** 2) + np.exp(-12 * (t - 0.7) ** 2) + 0.05
    inits.append(db / db.sum())
    
    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm, 0.01)
    inits.append(sm / sm.sum())
    
    # Frequency modulated
    for freq in [1, 2, 3]:
        fm = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())
    
    # Power law
    for alpha in [0.5, -0.5, 1.0]:
        pl = (t + 0.01) ** alpha
        pl = np.maximum(pl, 0.01)
        inits.append(pl / pl.sum())
    
    # Random (unstructured)
    r = rng.exponential(1.0, n)
    inits.append(r / r.sum())
    
    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a).
    Phase 1: Smoothed FISTA with tau annealing (fast, ~45% time)
    Phase 2: Smoothed mirror descent with different schedule (~15% time)
    Phase 3: Exact coordinate descent (~20% time)
    Phase 4: LP mirror descent for final polish (~15% time)
    Phase 5: Multi-length and multi-start exploration (~5% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0
    
    best_seq = None
    best_score = float('inf')
    
    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent)
    
    n_parent = parent.size
    
    # === PHASE 1: Smoothed FISTA with tau annealing ===
    p1_deadline = t_start + float(budget_s) * 0.40
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smoothed FISTA (n={n_parent}, budget={float(budget_s)*0.40:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    a = parent.copy() / parent.sum()
    
    # Tau schedule: start smooth, get progressively sharper
    tau_levels = [10, 30, 80, 200, 500, 1200]
    
    for tau in tau_levels:
        if time.monotonic() > p1_deadline:
            break
        # Estimate steps we can afford
        remaining = p1_deadline - time.monotonic()
        est_time_per_step = 0.005  # ~5ms per step for n=512
        n_steps = int(remaining / (len(tau_levels) * est_time_per_step))
        n_steps = max(50, min(n_steps, 2000))
        
        print(f"  tau={tau}, ~{n_steps} steps", flush=True)
        a = smoothed_fista(a, tau, n_steps, min(time.monotonic() + remaining / 3, p1_deadline),
                           lr_init=2.0 / (tau * n_parent))
        update_best(a)
        print(f"    J after tau={tau}: {compute_J(a):.8f}", flush=True)
    
    # === PHASE 1b: Smoothed mirror descent with multiplicative weights ===
    p1b_deadline = min(t_start + float(budget_s) * 0.55, p1_deadline + 120)
    print(f"\n  Phase 1b: Mirror descent schedule", flush=True)
    tau_schedule2 = [50, 100, 300, 800, 2000]
    a2, J2 = smoothed_mirror_descent_schedule(a, p1b_deadline, tau_schedule=tau_schedule2)
    update_best(a2)
    print(f"  Phase 1b done: J={J2:.8f}", flush=True)
    
    # === PHASE 2: Exact coordinate descent ===
    p2_deadline = min(t_start + float(budget_s) * 0.72, p1b_deadline + float(budget_s) * 0.17)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Coordinate Descent (budget to {p2_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        # CD on best
        for cd_iter in range(3):
            if time.monotonic() > p2_deadline:
                break
            a_cd = coordinate_descent_exact(best_seq, min(time.monotonic() + 60, p2_deadline),
                                            max_passes=20)
            update_best(a_cd)
            print(f"  CD pass {cd_iter+1}: J={compute_J(a_cd):.8f}", flush=True)
    
    # === PHASE 3: LP mirror descent ===
    p3_deadline = min(t_start + float(budget_s) * 0.88, p2_deadline + float(budget_s) * 0.16)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (budget to {p3_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        a_lp, J_lp = lp_mirror_descent(best_seq, p3_deadline, max_iters=100)
        update_best(a_lp)
        print(f"  LP done: J={J_lp:.8f}", flush=True)
        
        # CD after LP
        if time.monotonic() < p3_deadline - 30:
            a_cd = coordinate_descent_exact(a_lp, p3_deadline, max_passes=15)
            update_best(a_cd)
    
    # === PHASE 4: Multi-length exploration ===
    p4_deadline = min(t_start + float(budget_s) * 0.95, p3_deadline + float(budget_s) * 0.07)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = sorted(set([
            int(n_base * 0.75), int(n_base * 1.25), int(n_base * 1.5),
            384, 448, 512, 640, 768, 1024, 1280
        ]))
        target_lengths = [L for L in target_lengths if 128 <= L <= 2048 and L != n_base]
        
        for tgt_n in target_lengths[:6]:
            if time.monotonic() > p4_deadline:
                break
            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            
            # Quick smoothed FISTA
            a_r = smoothed_fista(a_ext, 200, 300, min(time.monotonic() + 30, p4_deadline),
                                 lr_init=2.0 / (200 * tgt_n))
            update_best(a_r)
            # Quick CD
            a_cd = coordinate_descent_exact(a_r, min(time.monotonic() + 15, p4_deadline), max_passes=5)
            update_best(a_cd)
            print(f"  n={tgt_n}: J={compute_J(a_cd):.8f}", flush=True)
    
    # === PHASE 5: Diverse starts (if time allows) ===
    p5_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Diverse Starts & Final Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        n_best = best_seq.size
        inits = generate_diverse_inits(n_best, rng)
        # Add perturbed best
        for noise_scale in [0.03, 0.08, 0.15]:
            pert = np.maximum(best_seq + rng.normal(0, noise_scale, n_best), 0.01)
            inits.append(pert / pert.sum())
        
        # Sort by initial J
        init_scores = []
        for init in inits:
            init = np.maximum(init, 0)
            S = init.sum()
            if S > 0.01:
                init = init / S
                init_scores.append((compute_J(init), init))
        init_scores.sort(key=lambda x: x[0])
        
        for score_init, init in init_scores[:5]:
            if time.monotonic() > p5_deadline:
                break
            phase_dl = min(time.monotonic() + 25, p5_deadline)
            a_r = smoothed_fista(init, 100, 200, phase_dl, lr_init=2.0 / (100 * n_best))
            update_best(a_r)
            a_cd = coordinate_descent_exact(a_r, min(phase_dl + 10, p5_deadline), max_passes=8)
            update_best(a_cd)
        
        # Final polish on best
        if best_seq is not None and time.monotonic() < p5_deadline - 20:
            a_final, J_final = lp_mirror_descent(best_seq, p5_deadline, max_iters=50)
            update_best(a_final)
            a_cd = coordinate_descent_exact(a_final, p5_deadline, max_passes=10)
            update_best(a_cd)
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)
    
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_seq]
