"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with log-sum-exp approximation (mirror descent on simplex)
2. Exact coordinate descent with golden-section 1D search

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    rows = np.empty((2 * n - 1) * n, dtype=np.int32)
    cols = np.empty((2 * n - 1) * n, dtype=np.int32)
    vals = np.empty((2 * n - 1) * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def smooth_gradient(a, sigma):
    """
    Compute gradient of log-sum-exp approximation of max(b).
    f = sigma * log(sum exp(b[k]/sigma))
    Returns (gradient, f_value, b_array).
    """
    n = len(a)
    b = fast_conv(a)
    # Clamp for numerical stability
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted / sigma)
    Z = exp_b.sum()
    p = exp_b / Z  # softmax weights
    
    # Gradient: df/da_i = 2*c[i] - p[2i]*a[i]
    # where c[i] = sum_j p[i+j]*a[j] for valid j
    c = np.zeros(n)
    for i in range(n):
        # c[i] = sum_{j: 0<=i+j<2n-1, 0<=j<n} p[i+j] * a[j]
        j_start = 0
        j_end = min(n, 2 * n - 1 - i)
        if j_end > j_start:
            c[i] = np.dot(p[i:i + j_end], a[:j_end])
    
    grad = 2.0 * c - np.zeros(n)
    for i in range(n):
        if 2 * i < 2 * n - 1:
            grad[i] -= p[2 * i] * a[i]
    
    f_val = sigma * (b.max() + np.log(Z))  # = sigma * logsumexp(b/sigma)
    return grad, f_val, b, p


def smooth_mirror_descent(a, deadline, max_iters=2000, sigma_start=0.01, sigma_end=0.001):
    """Mirror descent (multiplicative weights) on simplex with smooth objective."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    current_J = best_J
    
    # Precompute index pairs for gradient
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        sigma = sigma_start * (sigma_end / sigma_start) ** frac
        
        grad, f_val, b, p = smooth_gradient(a, sigma)
        
        # Project gradient onto tangent space of simplex
        # Mirror descent: a_new[i] = a[i] * exp(-eta * (grad[i] - <grad, a>))
        eta = 0.5 / (np.max(np.abs(grad)) + 1e-10)
        grad_centered = grad - np.dot(grad, a)
        
        a_new = a * np.exp(-eta * grad_centered)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
        
        # Accept if improves current or random acceptance for exploration
        if J_new < current_J + 1e-10:
            a = a_new
            current_J = J_new
        else:
            # Occasionally accept worse solutions for exploration
            if np.random.random() < 0.05 and J_new < current_J + 0.01:
                a = a_new
                current_J = J_new
    
    return best_a, best_J

        if time.monotonic() > deadline:
            break
        
        order = np.random.permutation(n)
        improved = False
        b = fast_conv(a)
        S = a.sum()
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            ai = a[i]
            
            # Set a[i] = 0 and compute b0
            a[i] = 0.0
            b0 = fast_conv(a)
            S0 = S - ai
            a[i] = ai
            
            # Build the coefficient array for the linear terms
            # coeff[k] = 2*a[k-i] if 0<=k-i<n and k!=2i, else 0
            coeff = np.zeros(2 * n - 1)
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            for k in range(k_lo, k_hi + 1):
                j = k - i
                if j != i:
                    coeff[k] = 2.0 * a[j]
            
            # Objective: f(x) = 2n * max(b0 + coeff*x, b0[2i] + x^2) / (S0+x)^2
            # We search over x >= 0
            
            def ev_f(x):
                Sx = S0 + x
                if Sx < 0.01:
                    return 1e10
                # max of (b0 + coeff*x) and (b0[2i] + x^2) for k=2i
                b_linear = b0 + coeff * x
                if 2 * i < 2 * n - 1:
                    b_quad = b0[2 * i] + x * x
                    b_linear[2 * i] = max(b_linear[2 * i], b_quad)
                return 2.0 * n * b_linear.max() / (Sx * Sx)
            
            # Golden section search
            x_max = max(ai * 10, 1.0 / n)
            
            a_ls = 0.0
            b_ls = x_max
            invphi = (np.sqrt(5) - 1) / 2
            c_ls = b_ls - invphi * (b_ls - a_ls)
            d_ls = a_ls + invphi * (b_ls - a_ls)
            fc = ev_f(c_ls)
            fd = ev_f(d_ls)
            
            for _ in range(35):
                if b_ls - a_ls < 1e-11 * max(1, ai):
                    break
                if fc < fd:
                    b_ls = d_ls
                    d_ls = c_ls
                    fd = fc
                    c_ls = b_ls - invphi * (b_ls - a_ls)
                    fc = ev_f(c_ls)
                else:
                    a_ls = c_ls
                    c_ls = d_ls
                    fc = fd
                    d_ls = a_ls + invphi * (b_ls - a_ls)
                    fd = ev_f(d_ls)
            
            x_opt = (a_ls + b_ls) / 2
            f_opt = ev_f(x_opt)
            
            # Compare with current
            f_cur = ev_f(ai)
            
            if f_opt < f_cur - 1e-12:
                a[i] = max(0, x_opt)
                improved = True
        
        # Post-pass: renormalize and check
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J - 1e-13:
                best_J = J
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a


def lp_mirror_descent(a, deadline, max_iters=2000):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b_aa = fast_conv(a)
        M = float(b_aa.max())
        
        g = lp_best_response(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Line search
        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)
        
        def line_obj(t):
            b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
            return b.max()
        
        result = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-9})
        t_opt = float(result.x)
        
        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)
        
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
        else:
            break
    
    return best_a, best_J


def kkt_polish(a, deadline, max_iters=100):
    """
    KKT-based polishing: identify active set and solve the resulting system.
    At optimum, for active k values (where b[k] = max), the gradient condition holds.
    """
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Find active set (within tolerance)
        active = np.where(b >= M * (1 - 1e-6))[0]
        
        # Gradient condition: for i with a[i] > 0,
        # sum_{k in active} p_k * (2*a[k-i] if valid) should equal lambda
        # Simplified: use the max-k gradient
        k_star = b.argmax()
        
        # Compute gradient of b[k_star] w.r.t. a
        grad_k = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if 2 * i == k_star:
                    grad_k[i] = a[i]
                else:
                    grad_k[i] = 2.0 * a[j]
        
        # At KKT: grad_k[i] = lambda for all i with a[i] > 0
        # So lambda should be approximately constant on support
        support = a > 1e-10
        if support.sum() < 2:
            break
        
        lambda_val = np.mean(grad_k[support])
        
        # Direction: move a to make grad more uniform on support
        # a_new[i] proportional to (something that makes grad_k[i] = lambda)
        # Use a small step: a[i] += eta * (lambda - grad_k[i]) for i in support
        eta = 0.01
        a_new = a.copy()
        a_new[support] += eta * (lambda_val - grad_k[support])
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 0.01:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
        else:
            eta *= 0.5
            a_new = a.copy()
            a_new[support] += eta * (lambda_val - grad_k[support])
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 0.01:
                a_new = a_new / s
                J_new = compute_J(a_new)
                if J_new < best_J - 1e-14:
                    best_J = J_new
                    best_a = a_new.copy()
                    a = a_new
                else:
                    break
    
    return best_a


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    inits.append(np.ones(n) / n)
    
    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())
    
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())
    
    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())
    
    dbl = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())
    
    for freq in [1, 2, 3]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())
    
    # One-sided
    one_sided = np.exp(-3 * t) + 0.01
    inits.append(one_sided / one_sided.sum())
    
    # Ramp
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())
    
    # V-shaped
    v = np.abs(t - 0.5) * 2 + 0.1
    inits.append(v / v.sum())
    
    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    t_start = time.monotonic()
    
    best_seq = None
    best_score = float('inf')
    
    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent)
    
    # === PHASE 1: Coordinate descent on parent (fast, reliable) ===
    p1_dl = t_start + 200
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Coordinate Descent (n={parent.size}, budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    a1 = coord_descent_vectorized(parent, p1_dl, max_passes=100)
    update_best(a1)
    
    # === PHASE 2: Smooth gradient descent ===
    p2_dl = min(t_start + 400, deadline - 400)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Smooth Gradient Descent (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    a2, J2 = smooth_mirror_descent(a1, p2_dl, max_iters=5000, sigma_start=0.02, sigma_end=0.002)
    update_best(a2)
    
    # Coordinate descent after smooth
    a2_cd = coord_descent_vectorized(a2, min(p2_dl + 50, deadline - 300), max_passes=30)
    update_best(a2_cd)
    
    # === PHASE 3: LP mirror descent ===
    p3_dl = min(t_start + 600, deadline - 300)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        a3, J3 = lp_mirror_descent(best_seq, p3_dl, max_iters=100)
        update_best(a3)
        a3_cd = coord_descent_vectorized(a3, min(p3_dl + 50, deadline - 200), max_passes=20)
        update_best(a3_cd)
    
    # === PHASE 4: KKT polishing + more CD ===
    p4_dl = min(t_start + 800, deadline - 300)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: KKT Polish + CD (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        a4 = kkt_polish(best_seq, p4_dl, max_iters=200)
        update_best(a4)
        a4_cd = coord_descent_vectorized(a4, min(p4_dl + 50, deadline - 200), max_passes=30)
        update_best(a4_cd)
    
    # === PHASE 5: Multi-length exploration ===
    p5_dl = min(t_start + 1050, deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Multi-length Exploration (budget=250s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = sorted(set([
            int(n_base * 0.75), int(n_base * 1.25), int(n_base * 1.5),
            int(n_base * 2.0),
            256, 320, 384, 448, 512, 576, 640, 768, 896, 1024, 1280
        ]))
        target_lengths = [L for L in target_lengths if 128 <= L <= 2048 and L != n_base]
        
        for tgt_n in target_lengths[:6]:
            if time.monotonic() > p5_dl:
                break
            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            
            phase_dl = min(time.monotonic() + 35, p5_dl)
            a_r = coord_descent_vectorized(a_ext, phase_dl, max_passes=20)
            update_best(a_r)
            
            # Quick smooth GD
            a_s, _ = smooth_mirror_descent(a_r, min(phase_dl + 15, p5_dl), max_iters=500)
            update_best(a_s)
            a_s_cd = coord_descent_vectorized(a_s, p5_dl, max_passes=10)
            update_best(a_s_cd)
    
    # === PHASE 6: Diverse starts at best length ===
    p6_dl = min(t_start + 1250, deadline - 150)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Diverse Starts (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        n_best = best_seq.size
        inits = generate_inits(n_best, rng)
        inits.append(np.maximum(best_seq + rng.normal(0, 0.03, n_best), 0.001))
        inits[-1] = inits[-1] / inits[-1].sum()
        inits.append(np.maximum(best_seq + rng.normal(0, 0.08, n_best), 0.001))
        inits[-1] = inits[-1] / inits[-1].sum()
        
        # Sort by initial J
        init_scores = [(compute_J(init), init) for init in inits]
        init_scores.sort(key=lambda x: x[0])
        
        for score_init, init in init_scores[:5]:
            if time.monotonic() > p6_dl:
                break
            phase_dl = min(time.monotonic() + 30, p6_dl)
            a_r = coord_descent_vectorized(init, phase_dl, max_passes=15)
            update_best(a_r)
    
    # === PHASE 7: Final intensive polishing ===
    p7_dl = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 7: Final Polishing (budget=remaining)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        # Multiple rounds of CD with different random orderings
        a_final = best_seq.copy()
        for round_i in range(3):
            phase_dl = min(time.monotonic() + (deadline - time.monotonic()) / 3, deadline)
            a_final = coord_descent_vectorized(a_final, phase_dl, max_passes=30)
            update_best(a_final)
        
        # Smooth GD one more time
        a_sm, _ = smooth_mirror_descent(best_seq, min(time.monotonic() + 30, deadline), max_iters=1000)
        update_best(a_sm)
        a_sm_cd = coord_descent_vectorized(a_sm, deadline, max_passes=20)
        update_best(a_sm_cd)
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_seq]
