Let me think about the frequency domain more carefully.

If A(f) = FFT(a), then the circular autocorrelation is IFFT(|A(f)|^2). The aperiodic autocorrelation (which is what we compute with conv) is related but not identical.

However, for large n, the difference between circular and aperiodic autocorrelation is small (the "wrap-around" effects are only at the edges). So we can use the frequency domain as an approximation.

In the frequency domain, the peak of the autocorrelation is related to the peak of |A(f)|^2. Specifically:
max_k R(k) ≤ max_f |A(f)|^2 (approximately)

And by Parseval: sum_k R(k) = sum_j a_j^2 = (1/N) sum_f |A(f)|^2

So to minimize the peak of R, we want to make |A(f)|^2 as flat as possible, which means making |A(f)| as constant as possible.

This is the "constant modulus" problem! The ideal solution would have |A(f)| = constant for all f, which gives a "flat spectrum" or "white noise" sequence.

For a real, non-negative sequence, achieving a perfectly flat spectrum is impossible, but we can get close.

**Frequency-domain approach:**
1. Compute A(f) = FFT(a)
2. Apply a "flattening" operation: A_new(f) = A(f) / |A(f)| * sqrt(|A(f)|^2 + target) or use gradient descent to flatten |A(f)|
3. Transform back: a_new = IFFT(A_new(f))
4. Clip to non-negative
5. Renormalize

The gradient of max_f |A(f)|^2 with respect to a is related to the IFFT of the phase of A(f) at the peak frequency. But this is getting complicated.

Simpler: just apply a "soft" flattening:
A_new(f) = A(f) * (1 - λ) + exp(1j * angle(A(f))) * |target| * λ

where |target| is a constant (e.g., the geometric mean of |A(f)|).

This gradually makes the spectrum flatter.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast exact coordinate descent with O(n) per-coordinate optimization
2. Frequency-domain spectrum flattening for global smoothing
3. Pair coordinate updates to escape local optima
4. Multi-start with diverse initializations
5. Adaptive simulated annealing with block moves

import numpy as np
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def exact_coord_update(a, c, S, n, i, deadline):
    """
    Find optimal value for coordinate i given all others fixed.
    Returns (new_value, new_delta, new_max, score).
    """
    v = a[i]
    k_lo = max(0, i)
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        M_out = c[mask].max() if mask.any() else 0.0
    else:
        M_out = 0.0
    
    # Affected indices and their a[k-i] values
    k_range = np.arange(k_lo, k_hi + 1)
    j_range = k_range - i  # = k - i, should be in [0, n-1]
    valid = (j_range >= 0) & (j_range < n)
    k_valid = k_range[valid]
    j_valid = j_range[valid]
    
    c_aff = c[k_valid]
    a_aff = a[j_valid]
    
    # Check if k=2i is in the affected range
    has_2i = (2 * i >= k_lo) and (2 * i <= k_hi)
    pos_2i = -1
    if has_2i:
        pos_2i = np.where(k_valid == 2 * i)[0]
        pos_2i = pos_2i[0] if len(pos_2i) > 0 else -1
    
    # Find candidate delta values
    candidates = set()
    candidates.add(0.0)
    candidates.add(-v)  # a_i = 0
    
    # Where linear terms cross M_out
    if M_out > 1e-15:
        for idx in range(len(a_aff)):
            if a_aff[idx] > 1e-15:
                d = (M_out - c_aff[idx]) / (2.0 * a_aff[idx])
                if -v * 1.1 - 0.01 <= d <= max(v * 5, 0.5):
                    candidates.add(d)
    
    # Where quadratic term (at 2i) crosses M_out
    if has_2i and pos_2i >= 0 and M_out > 1e-15:
        # c[2i] + 2*delta*v + delta^2 = M_out
        # delta^2 + 2v*delta + (c[2i] - M_out) = 0
        disc = 4 * v * v - 4 * (c[2 * i] - M_out)
        if disc >= 0:
            sq = np.sqrt(disc)
            for d in [(-2 * v + sq) / 2, (-2 * v - sq) / 2]:
                if -v * 1.1 - 0.01 <= d <= max(v * 5, 0.5):
                    candidates.add(d)
    
    # Heuristic candidates
    for mult in [0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5]:
        candidates.add(v * mult)
    for val in [0.01, 0.02, 0.05, 0.1, 0.2]:
        candidates.add(val)
    
    # Evaluate and find best
    current_max = c.max()
    best_score = 2.0 * n * current_max / (S * S)
    best_delta = 0.0
    
    for delta in candidates:
        new_v = v + delta
        if new_v < -1e-14:
            continue
        new_v = max(new_v, 0.0)
        delta = new_v - v
        if abs(delta) < 1e-14:
            continue
        S_new = S + delta
        if S_new < 0.01:
            continue
        
        # New affected values
        new_c_aff = c_aff + 2.0 * delta * a_aff

            new_c_aff[pos_2i] = c[2 * i] + 2.0 * delta * v + delta * delta

            new_max = 0.0

def coordinate_descent(a, deadline, max_passes=200, recompute_every=20):
    """
    Exact coordinate descent with periodic recomputation.
    """
    a = np.maximum(a.copy(), 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    c = fast_conv(a)
    S = a.sum()
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        # Recompute c periodically to avoid drift
        if p % recompute_every == 0 and p > 0:
            c = fast_conv(a)
            S = a.sum()
        
        improved = False
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            delta, score = exact_coord_update(a, c, S, n, i, deadline)
            
            if abs(delta) > 1e-14:
                v = a[i]
                new_v = max(v + delta, 0.0)
                delta_actual = new_v - v
                a[i] = new_v
                S += delta_actual
                
                # Update c incrementally
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if k == 2 * i:
                            c[k] += 2.0 * delta_actual * v + delta_actual * delta_actual
                        else:
                            c[k] += 2.0 * delta_actual * a[j]
                improved = True

        if p % 5 == 0:
            J_current = compute_J(a)
            if J_current < best_J - 1e-12:
                best_J = J_current
                best_a = a.copy()
        
        if not improved:
            break
    
    # Final check
    J_final = compute_J(best_a)
    if J_final < best_J:
        best_J = J_final
    
    return best_a

def freq_smooth(a, lambda_smooth=0.1, n_iters=50):
    """
    Frequency-domain smoothing: flatten |A(f)| to reduce autocorrelation peak.
    """
    a = a.copy().astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    
    for it in range(n_iters):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target magnitude: geometric mean (for flat spectrum)
        log_mag = np.log(mag + 1e-15)
        target_mag = np.exp(np.mean(log_mag))
        
        # Blend: move magnitude toward target
        new_mag = mag + lambda_smooth * (target_mag - mag)
        new_mag = np.maximum(new_mag, 0)
        
        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 0.01:
            a = a_new / s_new
    
    return a

def pair_update(a, i, j, c, S, n):
    """
    Try updating coordinates i and j together.
    Returns (new_i, new_j, improved).
    """
    vi, vj = a[i], a[j]
    best_score = compute_J(a)
    best_vi, best_vj = vi, vj
    
    # Try a few combinations
    for di_mult in [0.5, 0.7, 1.0, 1.3, 1.5]:
        for dj_mult in [0.5, 0.7, 1.0, 1.3, 1.5]:
            new_vi = vi * di_mult
            new_vj = vj * dj_mult
            a[i] = new_vi
            a[j] = new_vj
            score = compute_J(a)
            if score < best_score - 1e-13:
                best_score = score
                best_vi, best_vj = new_vi, new_vj
    
    a[i] = best_vi
    a[j] = best_vj
    return best_vi != vi or best_vj != vj

def diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random
    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())
    
    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())
    
    # Bell
    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())
    
    # Double bump
    dbl = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())
    
    # Frequency modulated
    for freq in [1, 2, 3, 5, 7]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())
    
    # Smooth random
    ksize = max(3, n // 10)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())
    
    # Power law
    for alpha in [0.3, 0.7, 1.3, 2.0]:
        pl = (t + 0.01) ** alpha
        inits.append(pl / pl.sum())
    
    # Low PAPR style: random with frequency flattening
    r = rng.uniform(0, 1, n)
    for _ in range(20):
        A = np.fft.rfft(r)
        mag = np.abs(A)
        phase = np.angle(A)
        target = np.exp(np.mean(np.log(mag + 1e-15)))
        mag = mag + 0.3 * (target - mag)
        r = np.fft.irfft(mag * np.exp(1j * phase), n)
        r = np.maximum(r, 0.01)
    inits.append(r / r.sum())
    
    return inits

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        nonlocal best_seq, best_score
