    a_scaled = resample_seq(a_final, int(J_best_str * 0.7 or use parent's length))
    
    # Apply one final optimized scaling
    J_scaled = evaluate_sequence(a_scaled)
    if J_scaled < J_best_str:
        J_best_str = J_scaled
        
    return a_final.tolist()
