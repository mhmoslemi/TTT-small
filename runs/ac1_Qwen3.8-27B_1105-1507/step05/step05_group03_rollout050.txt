import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J_from_parts(n, M, s):
    """Compute J from precomputed parts."""
    if s < 0.01:
        return float('inf')
    return float(2 * n * M / (s * s))


def build_lp_structure(n):
    """Precompute row/col indices for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp_direction(a, M, row_idx, col_idx, time_limit=20):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_convex(a, g):
    """
    Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2.
    Uses ternary search on the convex piecewise-quadratic function.
    """
    A_self = fast_conv(a)
    B_cross = cross_conv(a, g)
    C_self = fast_conv(g)

    a_coef = A_self
    b_coef = 2.0 * (B_cross - A_self)
    c_coef = C_self - 2.0 * B_cross + A_self

    lo = 0.0
    hi = 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(a_coef + b_coef * m1 + c_coef * m1 * m1))
        f2 = float(np.max(a_coef + b_coef * m2 + c_coef * m2 * m2))
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    arange = np.arange(1, n + 1, dtype=np.float64)
    cond = u * arange > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_maxconv(a):
    """
    Compute subgradient of max_k (a*a)[k] at point a.
    Returns (gradient array, index of max).

    b = fast_conv(a)
    k_star = int(np.argmax(b))
    grad = np.zeros(n)
    m_vals = np.arange(n)
    idx = k_star - m_vals
    valid = (idx >= 0) & (idx < n)
    grad[valid] = 2.0 * a[idx[valid]]
    return grad, k_star


def run_projected_gradient(a, max_iters, deadline, lr_init=0.05, momentum=0.9):
    """
    Projected gradient descent with momentum on max(conv(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)
    lr = lr_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, _ = compute_subgradient_maxconv(a)
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = project_simplex(a_new)
        a = a_new

        M = compute_M(a)
        if M < best_M:
            best_M = M
            best_a = a.copy()

        # Decay learning rate
        lr = lr_init * 0.999 ** it

    return best_a


def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0):
    """Smooth gradient descent with adaptive tau for max-conv approximation."""
    a = np.maximum(a.astype(np.float64), 1e-8)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_structural_perturbation(a, rng, mode='gaussian'):
    """Generate a perturbed version of a using different strategies."""
    n = len(a)
    a = a.copy()

    if mode == 'gaussian':
        noise = rng.normal(0, 0.02, n)
        a = a + noise
    elif mode == 'structural':
        # Multiply by a smooth random function
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 1, n)
        mod = 1.0 + 0.1 * np.cos(2 * np.pi * freq * x + phase)
        a = a * mod
    elif mode == 'dropout':
        # Zero out a small fraction of coordinates
        frac = rng.uniform(0.05, 0.15)
        mask = rng.random(n) > frac
        a = a * mask
    elif mode == 'mirror':
        # Try mirrored version
        a = a[::-1].copy()
    elif mode == 'shift':
        # Circular shift
        k = rng.integers(1, max(2, n // 4))
        a = np.roll(a, k)
    else:
        noise = rng.normal(0, 0.03, n)
        a = a + noise

    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining projected gradient, LP mirror descent,
    multi-length exploration, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    n0 = parent.size
    a0 = parent.copy()
    s0 = a0.sum()
    if s0 > 0:
        a0 = a0 / s0

    # =========================================================
    # Phase 1: Quick smooth gradient to get close to optimum
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smooth Gradient (n={n0})", flush=True)
    print(f"{'='*60}", flush=True)

    p1_deadline = t_start + budget_s * 0.15
    a_p1 = run_smooth_gradient(a0, 500, p1_deadline, initial_tau=3.0, final_tau=80.0)
    tracker.update(a_p1, "(P1 smooth)")

    # Also try projected gradient
    a_pg = run_projected_gradient(a0, 2000, t_start + budget_s * 0.20,
                                  lr_init=0.02, momentum=0.9)
    tracker.update(a_pg, "(P1 proj-grad)")

    # Use the best from phase 1
    if tracker.best is None:
        a_cur = a0.copy()
    else:
        a_cur = tracker.best.copy()
        a_cur = a_cur / a_cur.sum()
    n_cur = a_cur.size

    # =========================================================
    # Phase 2: LP Mirror Descent (main workhorse)
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (n={n_cur})", flush=True)
    print(f"{'='*60}", flush=True)

    row_idx, col_idx = build_lp_structure(n_cur)
    p2_deadline = t_start + budget_s * 0.45
    a_p2 = run_lp_mirror_descent(a_cur, 100000, p2_deadline, row_idx, col_idx,
                                 verbose=True, m_relax=1.0)
    tracker.update(a_p2, "(P2 LP-MD)")

    # Run v2 (tighter, no relaxation)
    if tracker.best is not None:
        a_cur2 = tracker.best.copy()
        a_cur2 = a_cur2 / a_cur2.sum()
        n_cur2 = a_cur2.size
        if n_cur2 != n_cur:
            row_idx2, col_idx2 = build_lp_structure(n_cur2)
        else:
            row_idx2, col_idx2 = row_idx, col_idx
        p2b_deadline = t_start + budget_s * 0.55
        a_p2b = run_lp_mirror_descent_v2(a_cur2, 50000, p2b_deadline,
                                         row_idx2, col_idx2, verbose=True)
        tracker.update(a_p2b, "(P2b LP-MD-v2)")

    # =========================================================
    # Phase 3: Multi-length exploration
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for delta in [-32, -16, 16, 32, 48, 64, 96]:
            n_trial = base_n + delta
            if 64 <= n_trial <= 4096:
                trial_lengths.append(n_trial)
        # Also try some ratios
        for ratio in [0.75, 1.25, 1.5]:
            n_trial = int(base_n * ratio)
            if 64 <= n_trial <= 4096:
                trial_lengths.append(n_trial)
        trial_lengths = sorted(set(trial_lengths))

        p3_deadline = t_start + budget_s * 0.72
        for n_trial in trial_lengths:
            if time.monotonic() > p3_deadline:
                break

            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()

            r_idx, c_idx = build_lp_structure(n_trial)
            # Quick smooth first
            sub_dl1 = min(time.monotonic() + 20, p3_deadline)
            a_init = run_smooth_gradient(a_init, 200, sub_dl1,
                                         initial_tau=5, final_tau=50)
            # Then LP
            sub_dl2 = min(time.monotonic() + 100, p3_deadline)
            a_opt = run_lp_mirror_descent(a_init, 5000, sub_dl2, r_idx, c_idx,
                                          verbose=False)
            tracker.update(a_opt, f"(P3 n={n_trial})")

    # =========================================================
    # Phase 4: Diverse perturbation restarts
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Diverse Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_idx_b, c_idx_b = build_lp_structure(n_best)

        modes = ['gaussian', 'structural', 'dropout', 'mirror', 'shift',
                 'gaussian', 'structural', 'dropout', 'gaussian', 'structural']
        temperatures = [1.0, 0.8, 0.65, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.12]

        p4_deadline = t_start + budget_s * 0.88
        for r in range(len(modes)):
            if time.monotonic() > p4_deadline:
                break

            mode = modes[r]
            a_pert = generate_structural_perturbation(tracker.best, rng, mode)

            # Quick smooth
            sub_dl1 = min(time.monotonic() + 10, p4_deadline)
            a_pert = run_smooth_gradient(a_pert, 100, sub_dl1,
                                         initial_tau=10, final_tau=60)
            # Then LP
            sub_dl2 = min(time.monotonic() + 60, p4_deadline)
            a_opt = run_lp_mirror_descent(a_pert, 3000, sub_dl2, r_idx_b, c_idx_b,
                                          verbose=False)
            tracker.update(a_opt, f"(P4 r={r} mode={mode})")

    # =========================================================
    # Phase 5: Final extended LP polish
    # =========================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_structure(n_final)
        a_final = tracker.best.copy()
        a_final = a_final / a_final.sum()

        # Run with very generous settings
        a_opt_f = run_lp_mirror_descent(a_final, 100000, deadline, r_f, c_f,
                                        verbose=True, m_relax=1.0,
                                        no_improve_limit=300)
        tracker.update(a_opt_f, "(P5 final)")

        # One more pass with v2
        if tracker.best is not None:
            a_final2 = tracker.best.copy()
            a_final2 = a_final2 / a_final2.sum()
            a_opt_f2 = run_lp_mirror_descent_v2(a_final2, 50000, deadline,
                                                r_f, c_f, verbose=False)
            tracker.update(a_opt_f2, "(P5b final-v2)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J_from_parts(parent.size, compute_M(parent), parent.sum())

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
