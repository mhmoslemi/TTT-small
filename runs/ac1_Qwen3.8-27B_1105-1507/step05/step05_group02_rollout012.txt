import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective J(a)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse matrix for A @ b where A represents convolution with a."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=1.0):
    """Solve LP to find best response: maximize sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_min_max(a, time_limit=1.0):
    """Solve LP to find minimum t such that (a*b)_k <= t for all k, b>=0, sum(b)=1.
    Returns (t_star, b_star)."""

    # Variables: b_0, ..., b_{n-1}, t

    # subject to (a*b)_k - t <= 0 for all k
    #            sum(b) = 1
    #            b >= 0
    #            t free (but will be positive)
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: (a*b)_k - t <= 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
        rows.append(k)
        cols.append(n)  # t component
        vals.append(-1.0)
    
    A_ineq = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ineq = np.zeros(2 * n - 1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # b >= 0, t free
    
    try:

                    options={'time_limit': time_limit, 'presolve': True})

def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute objective J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse convolution matrix A where (A @ x)_k = (a*x)_k."""
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))

        try_update(a)

    print(f"Phase 1 done: J={best_J:.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    # === PHASE 2: Intensive LP mirror descent on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Intensive LP Mirror Descent (n={best_seq.size})", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    # Allocate most of the remaining time here
    p2_deadline = min(time.monotonic() + 500.0, deadline - 400.0)
    a = run_lp_mirror(best_seq, p2_deadline, max_iters=100000,
                      initial_lp_time=0.3, use_min_max=False, restart_every=80)
    try_update(a)

    # Second LP pass with min_max formulation
    if time.monotonic() < deadline - 350.0:
        p2b_deadline = min(time.monotonic() + 200.0, deadline - 350.0)
        a = run_lp_mirror(best_seq, p2b_deadline, max_iters=50000,
                          initial_lp_time=0.5, use_min_max=True, restart_every=60)
        try_update(a)

    print(f"Phase 2 done: J={best_J:.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    # === PHASE 3: Try different lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    base_n = best_seq.size
    lengths_to_try = set()
    for mult in [0.5, 0.75, 1.25, 1.5, 2.0, 2.5, 3.0]:
        n_try = int(base_n * mult)
        n_try = max(128, min(4096, (n_try // 16) * 16))
        lengths_to_try.add(n_try)
    # Add some specific lengths
    for n_try in [256, 384, 512, 768, 1024, 1536, 2048]:
        if n_try != base_n and 128 <= n_try <= 4096:
            lengths_to_try.add(n_try)

    # Sort by expected benefit (prefer lengths close to current best, then longer)
    lengths_sorted = sorted(lengths_to_try, key=lambda x: abs(x - base_n))
    lengths_sorted = lengths_sorted[:8]  # Limit to 8 lengths
    print(f"  Trying lengths: {lengths_sorted}", flush=True)

    p3_per_length = max(30.0, (deadline - time.monotonic() - 300.0) / max(1, len(lengths_sorted)))
    p3_per_length = min(p3_per_length, 150.0)

    for n_try in lengths_sorted:
        if time.monotonic() > deadline - 250.0:
            break
        print(f"  --- n={n_try} (budget {p3_per_length:.0f}s) ---", flush=True)
        p3_dl = min(time.monotonic() + p3_per_length, deadline - 200.0)

        # Initialize from best
        a0 = resample_seq(best_seq, n_try)
        a0 = np.maximum(a0, 1e-6)
        a0 = a0 / a0.sum()

        # Quick gradient
        grad_dl = min(time.monotonic() + 15.0, p3_dl - 10.0)
        a = run_smooth_gradient(a0, grad_dl, max_iters=500,
                                 tau_start=3.0, tau_end=150.0,
                                 lr_init=0.3, lr_end=0.01)

        # LP polish
        lp_dl = p3_dl
        a = run_lp_mirror(a, lp_dl, max_iters=50000,
                          initial_lp_time=0.2, use_min_max=False, restart_every=60)
        try_update(a)

        # Alternative init for this length
        if time.monotonic() < p3_dl - 20.0:
            a0b = resample_seq(best_seq, n_try)
            noise = rng.normal(0, 0.03, n_try)
            a0b = np.maximum(a0b + noise, 1e-6)
            a0b = a0b / a0b.sum()
            a = run_smooth_gradient(a0b, min(time.monotonic() + 10, p3_dl - 10.0),
                                     max_iters=400, tau_start=5.0, tau_end=100.0,
                                     lr_init=0.2, lr_end=0.01)
            a = run_lp_mirror(a, p3_dl, max_iters=20000,
                              initial_lp_time=0.2, use_min_max=False, restart_every=50)
            try_update(a)

    print(f"Phase 3 done: J={best_J:.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    # === PHASE 4: Extended LP on the absolute best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Extended LP on best (n={best_seq.size})", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    p4_deadline = deadline - 60.0
    a = run_lp_mirror(best_seq, p4_deadline, max_iters=200000,
                      initial_lp_time=0.3, use_min_max=False, restart_every=100)
    try_update(a)

    # Final min_max LP pass
    if time.monotonic() < deadline - 30.0:
        a = run_lp_mirror(best_seq, deadline - 30.0, max_iters=100000,
                          initial_lp_time=0.5, use_min_max=True, restart_every=80)
        try_update(a)

    print(f"Phase 4 done: J={best_J:.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    # === PHASE 5: Final perturbation attempts ===
    if time.monotonic() < deadline - 10.0:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 5: Final perturbations", flush=True)
        print(f"{'='*60}", flush=True)
        n_best = best_seq.size
        for scale in [0.05, 0.02, 0.01]:
            if time.monotonic() > deadline - 10.0:
                break
            a_pert = np.maximum(best_seq + rng.normal(0, scale, n_best), 1e-6)
            a_pert = a_pert / a_pert.sum()
            a = run_lp_mirror(a_pert, min(time.monotonic() + 30, deadline - 5.0),
                              max_iters=20000, initial_lp_time=0.2,
                              use_min_max=False, restart_every=50)
            try_update(a)

    # Final validation
    final_j = evaluate_sequence([float(x) for x in best_seq])
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J_eval={final_j:.8f} J_computed={compute_J(best_seq):.8f} "
          f"n={best_seq.size} total_time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    # Fallback
    if best_seq is None or not np.all(np.isfinite(best_seq)):
        best_seq = np.ones(128)

    return [float(x) for x in best_seq]
