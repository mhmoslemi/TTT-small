def evaluate_sequence(sequence):
    """Evaluates a sequence of coefficients and returns the J value.
    Returns inf if invalid input."""
    if not isinstance(sequence, list):
        return np.inf
    if len(sequence) == 0:
        return np.inf

    # Ensure all elements are numbers and within bounds
    valid_types = (int, float)
    for elem in sequence:
        if isinstance(elem, bool):
            return np.inf
        if not isinstance(elem, valid_types):
            return np.inf
    try:
        seq = np.array(sequence, dtype=np.float64)
    except:
        return np.inf

    n = len(seq)
    seq = np.maximum(seq, 0)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf

    # Protect against very large numbers (though the problem says nonnegative, the bounds might help escape local minima)
    seq = np.clip(seq, 0, 1000)

    b_seq = np.convolve(seq, seq, mode='full')
    
    max_b_val = np.max(b_seq)
    print(f"Raw evaluation: J={2 * n * max_b_val / (sum_a ** 2)}, n={n}, max_b={max_b_val}")
    return float(2 * n * max_b_val / (sum_a ** 2))

def create_uniform_sequence(length):
    """Generate a uniform sequence of length `length`."""
    return np.ones(length) * 0.01 / length

def create_pulsed_sequence(length, pulse_count):
    """Generate a sequence with `pulse_count` pulses spread optimally."""
    positions = np.linspace(0, length-1, pulse_count, endpoint=False)
    seq = np.zeros(length)
    center_vals = 1.0 / pulse_count
    for pos in positions:
        if pos < 0 or pos >= length:
            continue
        seq[pos] = center_vals
    return np.clip(seq, 0, None)

def mutate_sequence(seq, method='comb', comb_strength=0.2):
    """Apply a mutation to the sequence."""
    n = len(seq)
    if method == 'comb':
        mutation_points = np.arange(n) + int(comb_strength * n)
        mutation_points = np.mod(mutation_points, n)
        for point in mutation_points[:n//2]:  # Mutate half as many points
            seq[point] *= (0.5 + np.random.random())
    elif method == 'cut':
        cut_length = int(comb_strength * n)
        seq = seq[:-cut_length]
    return seq
