def frank_wolfe_step(a, n):
    # Compute autocorrelation
    ac = fast_conv(a)  # length 2n-1
    M = ac.max()
    
    # Find active set (taps within epsilon of max)
    tol = 1e-8 * M
    active_k = np.where(ac >= M - tol)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in active_k:
        # grad += 2 * a[k-l] for valid l
        # a[k-l] is valid when 0 <= k-l < n, i.e., max(0, k-n+1) <= l <= min(n-1, k)
        l_min = max(0, k - n + 1)
        l_max = min(n - 1, k)
        grad[l_min:l_max+1] += 2.0 * a[k - l_max : k - l_min + 1][::-1]
    
    grad /= len(active_k)
    
    # Find best vertex
    v = np.argmin(grad)
    
    # Line search
    # f(t) = max_k ((1-t)*a + t*e_v)^T M_k ((1-t)*a + t*e_v)
    # This is convex in t, use ternary search
    e_v = np.zeros(n)
    e_v[v] = 1.0
    
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        a1 = (1-m1)*a + m1*e_v
        a2 = (1-m2)*a + m2*e_v
        f1 = fast_conv(a1).max()
        f2 = fast_conv(a2).max()
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2
    return t, v

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with log-sum-exp (handles non-convexity)
2. LP mirror descent for convex relaxation polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def compute_smooth_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*(a*a)[k])).
    Returns (gradient, smooth_value).
    """
    n = len(a)
    b = fast_conv(a)
    b_shifted = b - b.max()
    exp_b = np.exp(tau * b_shifted)
    p = exp_b / exp_b.sum()
    smooth_val = (b.max() + np.log(exp_b.sum())) / tau

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[l] += 2 * pk * a[k-l] for l in [lo, hi]
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad, smooth_val


def smooth_gradient_descent(a, deadline, max_iters=50000, tau_start=5.0, tau_end=200.0,
                            lr_start=0.5, decay=0.9995):
    """
    Smooth gradient descent with adaptive tau and learning rate.
    a is normalized to sum 1. Returns optimized sequence (sum 1).
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)
    lr = lr_start

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / (max_iters * 0.8))
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.7)
        lr = lr_start * (decay ** it)

        grad, smooth_val = compute_smooth_gradient(a, tau)

        # Gradient step with momentum
        v = 0.85 * v + lr * grad
        a_new = a - v

        # Project to simplex {a >= 0, sum(a) = 1}
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        a = a_new
        M = compute_M(a)

        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()

    return best_a


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    A_pre = fast_conv(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        # Analytic line search
        B = cross_conv(a, g)
        C = fast_conv(g)
        a_coef = A_pre
        b_coef = 2.0 * (B - A_pre)
        c_coef = C - 2.0 * B + A_pre

        lo = 0.0
        hi = 1.0
        for _ in range(50):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
            f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1

        t = (lo + hi) / 2.0
        t = max(0.0, min(1.0, t))

        if t < 1e-8:
            no_improve += 1
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                A_pre = fast_conv(a)
                actual_M = compute_M(a)
                if actual_M < best_M - 1e-14:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 80:
            break

        if verbose and it % 100 == 0:
            print(f"      LP iter {it}: M={M:.10f} best={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def frank_wolfe(a, deadline, max_iters=20000, tol=1e-10):
    """
    Frank-Wolfe (conditional gradient) for minimizing max_k (a*a)[k]
    subject to a >= 0, sum(a) = 1.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        ac = fast_conv(a)
        M = ac.max()

        # Find active set
        tol_active = max(1e-10, 1e-6 * M)
        active_k = np.where(ac >= M - tol_active)[0]

        # Compute subgradient
        grad = np.zeros(n)
        for k in active_k:
            l_min = max(0, k - n + 1)
            l_max = min(n - 1, k)
            grad[l_min:l_max + 1] += 2.0 * a[k - l_max:k - l_min + 1][::-1]
        grad /= len(active_k)

        # Best vertex (minimum gradient component)
        v_best = np.argmin(grad)

        # Line search: minimize f(t) = max_k ((1-t)*a + t*e_v)[k]
        # f(t) = max_k [(1-t)^2*ac[k] + 2t(1-t)*cross[k] + t^2*delta[k]]
        e_v = np.zeros(n)
        e_v[v_best] = 1.0
        cross = cross_conv(a, e_v)  # (a * e_v)[k] = a[k-v_best] if valid
        delta = np.zeros(2 * n - 1)
        if 2 * v_best < 2 * n - 1:
            delta[2 * v_best] = 1.0

        def f_of_t(t):
            return np.max((1 - t) ** 2 * ac + 2 * t * (1 - t) * cross + t ** 2 * delta)

        lo, hi = 0.0, 1.0
        for _ in range(40):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if f_of_t(m1) < f_of_t(m2):
                hi = m2
            else:
                lo = m1

        t_opt = (lo + hi) / 2.0

        if t_opt < 1e-10:
            no_improve += 1
        else:
            a_new = (1.0 - t_opt) * a + t_opt * e_v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                new_M = compute_M(a)
                if new_M < best_M - tol:
                    best_M = new_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 50:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_symmetric(a):
    """Make a sequence palindromic: a[j] = a[n-1-j]."""
    n = len(a)
    half = (n + 1) // 2
    s = np.zeros(n)
    for j in range(half):
        s[j] = a[j]
        s[n - 1 - j] = a[j]
    return s


def symmetrize_and_normalize(a):
    """Make symmetric and normalize to sum 1."""
    a_sym = make_symmetric(a)
    s = a_sym.sum()
    if s < 1e-10:
        return a / a.sum()
    return a_sym / s

                a_l = run_lp_mirror_descent(a_l, 5000, sub_dl2, r_i, c_i, verbose=False)
                tracker.update(a_l, f"(multi n={n_t} LP)")

    # Phase 6: Perturbation restarts
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_idx_b, c_idx_b = build_lp_matrix(n_best)
        phase6_deadline = deadline
        temps = [0.1, 0.05, 0.02, 0.01, 0.005]

        for temp in temps:
            if time.monotonic() > phase6_deadline:
                break

            for trial in range(3):
                if time.monotonic() > phase6_deadline:
                    break

                a = tracker.best.copy().astype(np.float64)
                noise = rng.normal(0, temp * 0.01, n_best)
                a = a + noise
                a = np.maximum(a, 1e-8)
                a = a / a.sum()

                # Quick SGD (short)
                sub_dl = min(time.monotonic() + 20, phase6_deadline)
                a = smooth_gradient_descent(a, sub_dl, max_iters=3000,
                                            tau_start=10.0, tau_end=100.0,
                                            lr_start=0.3, decay=0.999)
                tracker.update(a, f"(pert T={temp} r={trial})")

                # Quick LP (short)
                if time.monotonic() < phase6_deadline - 15:
                    a2 = a / a.sum()
                    sub_dl2 = min(time.monotonic() + 20, phase6_deadline)
                    a2 = run_lp_mirror_descent(a2, 3000, sub_dl2, r_idx_b, c_idx_b, verbose=False)
                    tracker.update(a2, f"(pert T={temp} r={trial}+LP)")

    # Phase 7: Final extended polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 7: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)

        # Final SGD
        a_sg = tracker.best / tracker.best.sum()
        sub_dl = min(time.monotonic() + 60, deadline)
        a_sg = smooth_gradient_descent(a_sg, sub_dl, max_iters=20000,
                                       tau_start=50.0, tau_end=500.0,
                                       lr_start=0.2, decay=0.9995)
        tracker.update(a_sg, "(final SGD)")

        # Final LP
        if time.monotonic() < deadline - 10:
            a_lp = tracker.best / tracker.best.sum()
            a_lp = run_lp_mirror_descent(a_lp, 50000, deadline, r_f, c_f, verbose=True)
            tracker.update(a_lp, "(final LP)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
