Let me implement the projection onto the simplex. The standard algorithm:
```
def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient descent with tau annealing (continuation method)
2. Subgradient method with iterate averaging for the true max objective  
3. Incremental coordinate descent with exact 1D line search per coordinate
4. Multi-length exploration with diverse initializations (DC window, spectral, random)
Key innovations: O(n log n) gradient via FFT cross-correlation, continuation on tau
for smooth-to-sharp transition, and O(n) incremental convolution updates in CD.
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * n * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u * rho) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    k = np.max(np.where(cond)[0])
    theta = (cssv[k] - 1.0) / (k + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_from_weights(w, a):
    """Compute grad[i] = 2 * sum_j a[j] * w[i+j].
    w has length 2n-1, a has length n.
    Uses FFT for efficiency."""
    n = len(a)
    if n <= 1:
        return np.array([2.0 * a[0] * w[0]])
    # FFT cross-correlation: (w cross-corr a)[i] = sum_n w[n+i]*a[n]
    # = IFFT(conj(FFT(w)) * FFT(a))
    size = 1
    while size < 3 * n:
        size <<= 1
    W = np.fft.fft(w, size)
    A = np.fft.fft(a, size)
    xcorr = np.fft.ifft(np.conj(W) * A).real[:n]
    return 2.0 * xcorr


def lse_obj(a, tau):
    """Compute LSE objective: (1/tau)*log(sum exp(tau*b_k)) where b=conv(a,a)."""
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip to avoid underflow
    shifted = np.maximum(shifted, -50)
    return float(b_max + np.log(np.exp(shifted).sum()) / tau)


def lse_grad(a, tau):
    """Compute gradient of LSE objective."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.maximum(shifted, -50)
    e = np.exp(shifted)
    w = e / e.sum()
    return grad_from_weights(w, a), b


def annealed_lse_gd(a, deadline, max_tau_mult=100.0, min_iters_per_tau=50, max_iters_per_tau=500):
    """
    Annealed LSE gradient descent: starts with small tau, increases tau.
    a should be on the simplex (sum=1, a>=0).
    Returns optimized a on simplex.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    tau_start = n / 10.0
    tau_end = n * max_tau_mult
    n_tau_steps = 25
    tau_vals = np.logspace(np.log10(tau_start), np.log10(tau_end), n_tau_steps)
    
    eta = 1.0  # initial step size
    
    for tau in tau_vals:
        if time.monotonic() > deadline:
            break
        
        # Run GD for this tau
        F_current = lse_obj(a, tau)
        
        for it in range(min_iters_per_tau, max_iters_per_tau + 1):
            if time.monotonic() > deadline:
                break
            if it < min_iters_per_tau:
                break
                
            grad, b = lse_grad(a, tau)
            
            # Backtracking line search
            g_norm = np.linalg.norm(grad)
            if g_norm < 1e-12:
                break
            
            step = eta
            x_trial = a - step * grad
            for bt in range(20):
                x_new = proj_simplex(x_trial)
                F_new = lse_obj(x_new, tau)
                if F_new <= F_current + 1e-14:
                    break
                step *= 0.5
                x_trial = a - step * grad
            else:
                x_new = proj_simplex(a - step * grad)
                F_new = lse_obj(x_new, tau)
            
            if F_new < F_current - 1e-14:
                a = x_new
                F_current = F_new
            else:
                # No improvement, reduce step size permanently
                eta = step
            
            # Check convergence
            if it > min_iters_per_tau and np.linalg.norm(x_new - a + step * grad) < 1e-10:
                break
        
        eta = max(step, 0.1)  # carry over step size
    
    return a


def subgradient_polish(a, deadline, max_iters=5000, eta_start=0.1):
    """
    Subgradient method with iterate averaging for the true max objective.
    a should be on the simplex.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    # Keep running average of iterates (Polyak averaging)
    a_avg = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute subgradient of max(conv(a,a))
        b = fast_conv(a)
        M = b.max()
        active = np.where(b >= M - 1e-12)[0]
        
        # Subgradient: (2/|active|) * sum_{k in active} a[k-j] (boundary handled)
        # This equals (2/|active|) * (indicator_active cross-corr a)
        w_active = np.zeros(2 * n - 1)
        w_active[active] = 1.0 / len(active)
        grad = grad_from_weights(w_active, a)
        
        # Adaptive step size
        eta = eta_start / (1.0 + 0.1 * it)
        
        # Update
        a_new = proj_simplex(a - eta * grad)
        
        # Polyak average: average of all iterates
        a_avg = (it * a_avg + a_new) / (it + 1)
        
        a = a_new
    
    # Return the averaged iterate (often better than last iterate)
    a_avg = a_avg / a_avg.sum()
    a_last = a / a.sum()
    
    # Pick the better one
    J_avg = compute_J(a_avg)
    J_last = compute_J(a_last)
    if J_avg < J_last:
        return a_avg
    else:
        return a_last


def coord_descent_incremental(a, deadline, max_passes=30):
    """
    Coordinate descent with incremental convolution updates.
    Maintains b = conv(a,a) and s = sum(a) incrementally.
    For each coordinate, tries several candidate values and picks the best.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return a
    # Work without normalization for simplicity (J is homogeneous)
    n = len(a)
    b = fast_conv(a)
    
    best_a = a.copy()
    best_J = 2.0 * n * b.max() / (s * s)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        order = np.random.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            ai = a[i]
            if ai < 1e-14:
                candidates = [0.0, s / n * 0.1, s / n * 0.5, s / n * 2.0]
            else:
                candidates = [0.0, ai * 0.3, ai * 0.6, ai * 0.9, ai * 1.1, ai * 1.5, ai * 2.0, ai * 3.0]
            
            best_val = 2.0 * n * b.max() / (s * s)
            best_new = ai
            
            for cand in candidates:
                delta = cand - ai
                if abs(delta) < 1e-15:
                    continue
                
                s_new = s + delta
                if s_new < 0.01:
                    continue
                
                # Update b incrementally
                # b[k] changes for k in [i, i+n-1]
                # b_new[k] = b[k] + 2*delta*a[k-i] + delta^2 (if k=2i)
                k_lo = max(0, i)
                k_hi = min(i + n - 1, 2 * n - 2)
                
                # Compute new max
                # Outside [k_lo, k_hi]: unchanged
                outside_max = 0.0
                if k_lo > 0:
                    outside_max = max(outside_max, b[:k_lo].max())
                if k_hi < 2 * n - 2:
                    outside_max = max(outside_max, b[k_hi + 1:].max())
                
                # Inside [k_lo, k_hi]:
                # b_new[k] = b[k] + 2*delta*a[k-i] (+ delta^2 if k==2i)
                # We need max of these
                inner_max = 0.0
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        new_bk = b[k] + 2.0 * delta * a[j]
                        if k == 2 * i:
                            new_bk += delta * delta
                        if new_bk > inner_max:
                            inner_max = new_bk
                
                new_max = max(outside_max, inner_max)
                val = 2.0 * n * new_max / (s_new * s_new)
                
                if val < best_val:
                    best_val = val
                    best_new = cand
            
            if best_new != ai:
                # Apply update
                delta = best_new - ai
                a[i] = best_new
                s += delta
                # Update b
                k_lo = max(0, i)
                k_hi = min(i + n - 1, 2 * n - 2)
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        b[k] += 2.0 * delta * a[j]
                    if k == 2 * i:
                        b[k] += delta * delta
                improved = True
        
        # Check overall improvement
        current_J = 2.0 * n * b.max() / (s * s)
        if current_J < best_J - 1e-12:
            best_J = current_J
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, best=None):
    """Generate diverse initializations for length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.5))
    inits.append(np.maximum(cw, 0.01) / np.maximum(cw, 0.01).sum())
    
    # Bell
    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())
    
    # Double bump
    dbl = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())
    
    # Frequency modulated
    for freq in [1, 2, 3, 5, 7]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2*np.pi))
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())
    
    # Random smooth
    ksize = max(3, n // 15)
    raw = rng.uniform(0, 1, n)
    sm = np.convolve(raw, np.ones(ksize) / ksize, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())
    
    # Power law
    for alpha in [0.3, 0.7, 1.5]:
        pl = (t + 0.01) ** alpha
        inits.append(pl / pl.sum())
        pl2 = ((1 - t) + 0.01) ** alpha
        inits.append(pl2 / pl2.sum())
    
    # If we have a best, add perturbed versions
    if best is not None and len(best) == n:
        for sigma in [0.05, 0.1, 0.2, 0.3]:
            pert = best + rng.normal(0, sigma, n)
            pert = np.maximum(pert, 0.01)
            inits.append(pert / pert.sum())
    
    return inits


def optimize_sequence(a, deadline, method='lse_gd', **kwargs):

    elif method == 'subgrad':

    elif method == 'cd':

        a1 = annealed_lse_gd(a, dl1)
        dl2 = time.monotonic() + (deadline - time.monotonic()) * 0.5
        a2 = subgradient_polish(a1, dl2, max
