Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Variance warm-up (minimize ||conv||_2^2)
2. Log-sum-exp smooth gradient descent with Nesterov acceleration
3. LP mirror descent for precise local refinement
4. Adaptive multi-start with structured perturbations

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]


def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):

        grad[i_min:i_max+1] += wk * a[k-i_max:k-i_min+1][::-1]
    
    return grad, b, b_max

    for k in range(2 * n - 1):

        if abs(wk) < 1e-16:

        i_max = min(n - 1, k)

def project_normalize(a):
    """Project onto {x >= 0, sum(x) = 1}"""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def run_variance_warmup(a, deadline, max_iters=2000, lr=0.5):
    """Minimize ||a*a||_2^2 using gradient descent"""
    a = project_normalize(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_obj = (fast_conv(a) ** 2).sum()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, b = variance_grad(a)
        obj = (b ** 2).sum()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Gradient descent with normalization
        lr_k = lr / (1.0 + 0.1 * it)
        a_new = a - lr_k * grad
        a = project_normalize(a_new)
    
    return best_a


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=0.5, tau_end=300.0,
                        lr_init=0.5, lr_end=0.005):
    a = project_normalize(a.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * 0.9 * 0.9)) / 2.0  # beta=0.9
        a_new = project_normalize(y - lr * (grad + 0.9 * (y - a)))
        momentum = (t_k - 1.0) / (t_k + 1.0)
        y = project_normalize(a_new + momentum * (a_new - a))
        a = a_new
    
    return best_a
