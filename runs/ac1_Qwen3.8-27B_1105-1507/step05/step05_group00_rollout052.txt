"""
Optimized LP-alternating method with backtracking line search, dual-based
subgradient steps, and structured perturbations for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses parent sequence as starting point
and explores multiple sequence lengths.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
from numpy.fft import fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]. Shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g_normalized, dual_weights) or (None, None).
    """
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                g_norm = g / s
                # Extract dual variables (yub) as peak weights
                dual = np.zeros(2 * n - 1)
                if res.yub is not None:
                    dual = np.asarray(res.yub, dtype=np.float64)
                    dsum = float(np.sum(dual))
                    if dsum > 1e-15:
                        dual = dual / dsum
                return g_norm, dual
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=10):
    """Solve LP: min t s.t. (a*g)[k] <= t, sum(g)=1, g>=0.
    Returns (g_normalized, t_opt) or (None, None).
    """

    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(-np.inf, np.inf)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',

    except Exception:
        pass
    return None, None


def compute_gradient(a, weights):
    """Compute gradient of sum_k weights[k]*conv_k(a) w.r.t. a.
    Uses FFT cross-correlation. O(n log n).
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = fft(a, N)
    fw = fft(weights, N)
    R = np.real(ifft(np.conj(fa) * fw))
    return 2.0 * R[:n]


def softmax_weights(conv, tau):
    """Compute softmax weights over convolution peaks."""
    cmax = float(np.max(conv))
    shifted = np.clip((conv - cmax) / tau, -300.0, 0.0)
    w = np.exp(shifted)
    s = float(np.sum(w))
    if s < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= s
    return w


def lp_alternate_backtrack(a, max_iters=100, time_budget=200.0):
    """LP alternating with backtracking line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, dual = solve_lp_max_sum(a, M, time_limit=8)
        if g is None:
            break

        # Backtracking line search: start with t=1, reduce until improvement
        t = 1.0
        improved = False
        for _ in range(12):  # max 12 halvings
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = float(np.sum(a_try))
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                stall_count = 0
                break
            t *= 0.5

        if not improved:
            # Try subgradient step using dual weights
            if dual is not None and np.sum(dual) > 1e-10:
                grad = compute_gradient(a, dual)
                grad_centered = grad - np.mean(grad)
                # Normalize gradient for step size
                gnorm = float(np.linalg.norm(grad_centered))
                if gnorm > 1e-15:
                    grad_centered /= gnorm
                    for step in [0.01, 0.02, 0.05, 0.1, 0.2]:
                        a_try = a - step * grad_centered
                        a_try = np.maximum(a_try, 0)
                        s_try = float(np.sum(a_try))
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-14:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            a = a_try
                            improved = True
                            break

            if not improved:
                stall_count += 1
                if stall_count > 8:
                    break

    return best_a, best_obj


def subgradient_momentum(a, max_iters=300, time_budget=30.0):
    """Subgradient with Nesterov momentum and adaptive softmax temperature."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.85
    step = 0.03

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        # Adaptive temperature: start high (spread), decrease (focus)
        tau = max(0.01, 2.0 * (1.0 - it / max_iters))
        weights = softmax_weights(conv, tau)

        grad = compute_gradient(a, weights)
        grad_centered = grad - np.mean(grad)

        v = mu * v - step * grad_centered
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        # Adapt step size
        step = max(0.001, step * 0.997)
        mu = min(0.95, 0.85 + 0.001 * it)

    return best_a, best_obj


def structured_initialization(n, kind='gaussian'):
    """Create a structured initial sequence."""
    x = np.linspace(0, 1, n)
    if kind == 'gaussian':
        sigma = 1.0 / (3.0 * n ** 0.5)
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif kind == 'binomial':
        # Approximate binomial using convolution of uniform
        base = np.ones(max(3, n // 50)) / max(3, n // 50)
        a = base.copy()
        for _ in range(min(10, n // 10)):
            a = np.convolve(a, base, 'full')
            # Resize to n
            if len(a) > n:
                a = a[:n]
            elif len(a) < n:
                a = np.pad(a, (0, n - len(a)))
    elif kind == 'raised_cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif kind == 'inverse_sqrt':
        a = 1.0 / (np.sqrt(x + 0.1) + np.sqrt(1 - x + 0.1))
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def coordinate_descent_fast(a, max_sweeps=20, time_budget=30.0):
    """Fast coordinate descent: try pairwise swaps to reduce peak."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # For each position contributing to the peak, try moving mass
        for i in range(0, n, max(1, n // 100)):
            if time.monotonic() - start > time_budget:
                break

            j = k_star - i
            if 0 <= j < n and j != i and a[i] > 1e-10:
                for frac in [0.05, 0.1, 0.2]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[j] += delta
                    obj_try = compute_obj(a)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break
            if not (0 <= k_star - i < n):
                # Try moving mass TO position i FROM a position contributing to peak
                pass

        if not improved and sweep > 5:
            break

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_sequence(a, time_budget=60.0):
    """Full optimization pipeline for a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Round 1: LP alternating
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10 and n <= 1000:
        a_new, obj_new = lp_alternate_backtrack(a, max_iters=50,
                                                 time_budget=min(remaining * 0.5, 60.0))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    # Round 2: Subgradient momentum
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        a_new, obj_new = subgradient_momentum(best_a, max_iters=200,
                                               time_budget=min(remaining * 0.3, 20.0))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    # Round 3: Another LP pass
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10 and n <= 900:
        a_new, obj_new = lp_alternate_backtrack(best_a, max_iters=30,
                                                 time_budget=min(remaining * 0.4, 40.0))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    # Round 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_new, obj_new = coordinate_descent_fast(best_a, max_sweeps=10,
                                                  time_budget=min(remaining * 0.3, 15.0))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP alternating with backtracking,
    dual-based subgradient, structured perturbations, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Parent loaded: n={parent.size}, obj={best_score:.8f}")

    n_parent = len(best_seq) if best_seq is not None else 700

    # Phase 1: Intensive optimization of parent
    print(f"\n=== Phase 1: Optimize parent (n={n_parent}) ===")
    if best_seq is not None:
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        # Multiple rounds of LP + subgradient
        for round_idx in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            alloc = min(remaining * 0.25, 80.0)
            a_new, obj_new = optimize_sequence(a, time_budget=alloc)

            if obj_new < best_score - 1e-12:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"  Round {round_idx}: {best_score:.8f}")
            else:
                # Perturb and try again
                rng = np.random.default_rng(seed + round_idx * 7919)
                for pert_scale in [0.02, 0.05, 0.1]:
                    remaining = deadline - time.monotonic()
                    if remaining < 15:
                        break
                    a_pert = a * np.exp(rng.normal(0, pert_scale, len(a)))
                    a_pert = np.maximum(a_pert, 0)
                    s_p = float(np.sum(a_pert))
                    if s_p < 1e-10:
                        continue
                    a_pert = a_pert / s_p

                    # Quick optimize
                    a_q, obj_q = optimize_sequence(a_pert, time_budget=min(remaining * 0.3, 30.0))
                    if obj_q < best_score - 1e-12:
                        best_score = obj_q
                        best_seq = a_q.copy()
                        a = a_q.copy()
                        print(f"  Round {round_idx} pert({pert_scale}): {best_score:.8f}")
                        break

    # Phase 2: Explore different n values
    print(f"\n=== Phase 2: Explore n values ===")
    n_candidates = [500, 600, 700, 800, 900, 1000]
    # Add nearby values to parent
    n_candidates = sorted(set(n_candidates + [n_parent - 100, n_parent + 100]))
    n_candidates = [n for n in n_candidates if 300 <= n <= 1200]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if n_try == n_parent and best_seq is not None:
            a_try = best_seq.copy()
        elif best_seq is not None:
            a_try = resample_sequence(best_seq, n_try)
            # Normalize to same "energy" scale
            s_try = float(np.sum(a_try))
            if s_try > 1e-10:
                a_try = a_try / s_try
        else:
            a_try = structured_initialization(n_try, 'gaussian')

        alloc = min(remaining * 0.2, 50.0)
        a_new, obj_new = optimize_sequence(a_try, time_budget=alloc)

        if obj_new < best_score - 1e-12:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  n={n_try}: {best_score:.8f} ** IMPROVED **")
        else:
            print(f"  n={n_try}: {obj_new:.8f}")

    # Phase 3: Try structured initializations
    print(f"\n=== Phase 3: Structured initializations ===")
    n_best = len(best_seq) if best_seq is not None else 700

    for kind in ['gaussian', 'binomial', 'raised_cosine', 'inverse_sqrt']:
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        a_struct = structured_initialization(n_best, kind)
        # Blend with current best
        if best_seq is not None:
            for blend_t in [0.3, 0.5, 0.7]:
                a_blend = (1 - blend_t) * best_seq + blend_t * a_struct
                a_blend = np.maximum(a_blend, 0)
                s_b = float(np.sum(a_blend))
                if s_b < 1e-10:
                    continue
                a_blend = a_blend / s_b

                alloc = min(remaining * 0.15, 30.0)
                a_new, obj_new = optimize_sequence(a_blend, time_budget=alloc)
                if obj_new < best_score - 1e-12:
                    best_score = obj_new
                    best_seq = a_new.copy()
                    print(f"  {kind} blend({blend_t}): {best_score:.8f} ** IMPROVED **")
                    break

    # Phase 4: Final intensive polish
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"Best before final: {best_score:.8f}")

    if best_seq is not None:
        a_final = best_seq.copy()
        s = np.sum(a_final)
        if s > 1e-10:
            a_final = a_final / s

        for round_idx in range(10):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            alloc = min(remaining * 0.15, 40.0)
            a_new, obj_new = optimize_sequence(a_final, time_budget=alloc)

            if obj_new < best_score - 1e-14:
                best_score = obj_new
                best_seq = a_new.copy()
                a_final = a_new.copy()
                print(f"  Final {round_idx}: {best_score:.8f}")
            else:
                # Try small perturbation
                rng = np.random.default_rng(seed + 10000 + round_idx)
                pert_scale = 0.01 + 0.005 * round_idx
                a_pert = a_final * np.exp(rng.normal(0, pert_scale, len(a_final)))
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    alloc2 = min(remaining * 0.1, 20.0)
                    a_new2, obj_new2 = optimize_sequence(a_pert, time_budget=alloc2)
                    if obj_new2 < best_score - 1e-14:
                        best_score = obj_new2
                        best_seq = a_new2.copy()
                        a_final = a_new2.copy()
                        print(f"  Final {round_idx} pert: {best_score:.8f}")

        # Final coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_new, obj_new = coordinate_descent_fast(a_final, max_sweeps=30,
                                                      time_budget=min(remaining * 0.5, 30.0))
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Final CD: {best_score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
