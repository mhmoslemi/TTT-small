            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = (kq <= 2 * n - 2) and (kq != lo) and (kq <= hi)
            # kq is in [lo, hi] range if lo <= 2i <= hi

            conv_base = conv[lo:hi + 1].copy()
            kq_val_base = conv[kq] if valid_kq else 0.0

            def obj_func(x):
                x = float(np.clip(x, 0.0, 1000.0))
                delta = x - old
                if abs(delta) < 1e-14:
                    M = conv.max()
                else:
                    updated_slice = conv_base + 2.0 * delta * a_slice
                    # The max could be in the slice or outside
                    M_in = updated_slice.max()
                    # Values outside [lo, hi] are unchanged
                    if lo > 0:
                        M_out = max(M_out, conv[:lo].max()) if lo > 0 else 0
                    else:
                        M_out = 0.0
                    if hi < 2 * n - 2:
                        M_out = max(M_out, conv[hi + 1:].max())
                    M = max(M_in, M_out)
                    # kq correction (it's within the slice, already included in updated_slice)
                    # Actually kq = 2i, and the slice is [lo, hi], so kq is in the slice
                    # The updated_slice already includes the 2*delta*a_slice correction
                    # but we also need the delta^2 at position kq
                    kq_idx = kq - lo
                    if valid_kq and 0 <= kq_idx < len(updated_slice):
                        # The a_slice already has a[i] = old at position kq_idx
                        # so 2*delta*a_slice[kq_idx] = 2*delta*old
                        # but we need delta*old + delta*old + delta^2 = 2*delta*old + delta^2
                        # The 2*delta*a_slice already gives 2*delta*old at kq_idx
                        # So we just need to add delta^2
                        pass  # Already handled: the formula conv + 2*delta*a_slice + delta^2 at kq
                        # But wait, we only applied 2*delta*a_slice. We need delta^2 at kq.
                        # Let me redo: the correct update is:
                        # new_conv[k] = old_conv[k] + 2*delta*a[j] for k in [lo,hi] where j=k-i
                        # PLUS delta^2 at k=2i (if 2i is in [lo,hi])
                        # So the slice update should be: conv_base + 2*delta*a_slice
                        # and then add delta^2 at position kq_idx
                        updated_slice = conv_base + 2.0 * delta * a_slice
                        updated_slice[kq_idx] += delta * delta
                        M_in = updated_slice.max()
                        if lo > 0:
                            M_out = conv[:lo].max()
                        else:
                            M_out = 0.0
                        if hi < 2 * n - 2:
                            M_out = max(M_out, conv[hi + 1:].max())
                        M = max(M_in, M_out)

                new_s = s - old + x
                if new_s < 1e-10:
                    return 1e10
                return 2.0 * n * M / (new_s * new_s)

            # Determine bounds for search
            if old < 1e-6:
                search_lo, search_hi = 0.0, max(0.5 * s / n, 1e-3)
            else:
                search_lo = 0.0
                search_hi = max(3.0 * old, 2.0 * s / n)

            try:
                result = minimize_scalar(obj_func, bounds=(search_lo, search_hi),
                                         method='brent',
                                         options={'maxiter': max_per_coord, 'xatol': 1e-9})
                new_val = float(np.clip(result.x, 0.0, 1000.0))
            except Exception:
                new_val = old

            if abs(new_val - old) > 1e-10:
                delta = new_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = new_val
                s = s - old + new_val
                sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

    a = best_a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def coord_descent_fast(a, max_sweeps, deadline, seed=0):
    """
    Fast coordinate descent with a small grid (for quick sweeps).
    Each coordinate tries a few candidate values.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = a / max(a.sum(), 1e-10)
        s = a.sum()

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    no_improve = 0
    rng = np.random.default_rng(seed)
    max_no_improve = 50

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            if old < 1e-8:
                candidates = [0.0, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2]
            else:
                candidates = [0.0, 0.3 * old, 0.5 * old, 0.7 * old,
                              1.0 * old, 1.3 * old, 1.5 * old, 2.0 * old]
                candidates = [x for x in candidates if abs(x - old) > old * 0.1 or x == 0.0]

            best_x = old
            best_local = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = float(np.clip(x, 0.0, 1000.0))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = 1e10
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_x = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if abs(best_x - old) > 1e-12:
                delta = best_x - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_x
                s = s - old + best_x
                sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

    a = best_a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, method='highs'):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=method,
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, verbose=True):
    """
    LP mirror descent with adaptive line search.
    Key improvement: better step size selection and more iterations.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    no_improve = 0

    if verbose:
        print(f"    LP MD start: M={current_M:.8f}, n={n}", flush=True)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_conv = fast_conv(a)
        M = float(b_conv.max())

        t0 = time.monotonic()
        g = lp_best_response(a, M)
        lp_time = time.monotonic() - t0

        if g is None:
            if verbose:
                print(f"    LP failed at iter {it}", flush=True)
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive line search: try large steps first, then shrink
        improved = False
        # Start with a relatively large step
        steps = [0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]
        # Adjust based on iteration count (smaller steps later)
        if it > max_iters * 0.7:
            steps = [0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.02, 0.01]

        for t in steps:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial
            M_trial = fast_conv(a_trial).max()
            # We need M_trial < M (the old M) for improvement
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                if verbose and it % 10 == 0:
                    print(f"    LP iter {it}: M={best_M:.8f} (t={t}, lp_time={lp_time:.2f}s)", flush=True)
                break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                if verbose:
                    print(f"    LP MD stopped at iter {it} (no improvement)", flush=True)
                break
        else:
            no_improve = 0

        # Check remaining time
        remaining = deadline - time.monotonic()
        if remaining < max(lp_time * 3, 5.0):
            break

    if verbose:
        print(f"    LP MD done: best_M={best_M:.8f}", flush=True)
    return best_a


def run_lp_mirror_descent_v2(a, max_iters, deadline, verbose=True):
    """
    LP mirror descent variant: instead of fixed M, update M each iteration
    and use a smaller LP (only active constraints).
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    if verbose:
        print(f"    LP MD v2 start: M={best_M:.8f}, n={n}", flush=True)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_conv = fast_conv(a)
        M = float(b_conv.max())

        # Solve LP
        t0 = time.monotonic()
        g = lp_best_response(a, M)
        lp_time = time.monotonic() - t0

        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search with geometric sequence
        improved = False
        t = 0.5
        for _ in range(10):
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s_trial
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-12):
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                if verbose and it % 5 == 0:
                    print(f"    LP v2 iter {it}: M={best_M:.8f} (t={t:.3f})", flush=True)
                break
            t *= 0.5

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        remaining = deadline - time.monotonic()
        if remaining < max(lp_time * 2.5, 3.0):
            break

    if verbose:
        print(f"    LP MD v2 done: best_M={best_M:.8f}", flush=True)
    return best_a


def pair_coord_descent(a, max_sweeps, deadline, seed=0):
    """
    Coordinate descent on anti-palindromic pairs (i, n-1-i).
    This directly exploits the structure that reduces the center peak.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = a / max(a.sum(), 1e-10)
        s = a.sum()

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    # Pairs are (i, n-1-i) for i = 0, ..., floor((n-1)/2)
    n_pairs = (n + 1) // 2

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n_pairs)
        sweep_improved = False

        for pi in order:
            i = pi
            j = n - 1 - pi
            if i == j:
                # Middle element (only when n is odd)
                old = a[i]
                lo = max(0, i)
                hi = min(2 * n - 2, i + n - 1)
                a_slice = a[lo - i:hi - i + 1].copy()
                kq = 2 * i
                valid_kq = kq <= 2 * n - 2

                candidates = [0.0, 0.5 * old, old, 1.5 * old, 2.0 * old] if old > 1e-8 \
                    else [0.0, 1e-4, 1e-3, 1e-2, 1e-1]

                best_x = old
                best_local = 2.0 * n * conv.max() / (s * s)

                for x in candidates:
                    x = float(np.clip(x, 0, 1000))
                    delta = x - old
                    if abs(delta) < 1e-14:
                        continue
                    conv[lo:hi + 1] += 2.0 * delta * a_slice
                    if valid_kq:
                        conv[kq] += delta * delta
                    new_s = s - old + x
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s) if new_s > 1e-10 else 1e10
                    if new_obj < best_local:
                        best_local = new_obj
                        best_x = x
                    conv[lo:hi + 1] -= 2.0 * delta * a_slice
                    if valid_kq:
                        conv[kq] -= delta * delta

                if abs(best_x - old) > 1e-12:
                    delta = best_x - old
                    conv[lo:hi + 1] += 2.0 * delta * a_slice
                    if valid_kq:
                        conv[kq] += delta * delta
                    a[i] = best_x
                    s = s - old + best_x
                    sweep_improved = True
                continue

            # Pair (i, j) with i < j
            old_i, old_j = a[i], a[j]
            if old_i < 1e-8 and old_j < 1e-8:
                continue

            # Try a small 2D grid
            ri = [0.0, 0.5, 1.0, 1.5, 2.0] if old_i > 1e-8 else [0.0, 1e-4, 1e-3, 5e-3]
            rj = [0.0, 0.5, 1.0, 1.5, 2.0] if old_j > 1e-8 else [0.0, 1e-4, 1e-3, 5e-3]

            # Pre-compute slices
            lo_i = max(0, i)
            hi_i = min(2 * n - 2, i + n - 1)
            lo_j = max(0, j)
            hi_j = min(2 * n - 2, j + n - 1)
            a_slice_i = a[lo_i - i:hi_i - i + 1].copy()
            a_slice_j = a[lo_j - j:hi_j - j + 1].copy()
            kq_i = 2 * i
            kq_j = 2 * j
            vkq_i = kq_i <= 2 * n - 2
            vkq_j = kq_j <= 2 * n - 2

            best_xi, best_xj = old_i, old_j
            best_local = 2.0 * n * conv.max() / (s * s)

            for xi in ri:
                xi = xi * old_i if old_i > 1e-8 else xi
                for xj in rj:
                    xj = xj * old_j if old_j > 1e-8 else xj
                    xi = float(np.clip(xi, 0, 1000))
                    xj = float(np.clip(xj, 0, 1000))
                    di = xi - old_i
                    dj = xj - old_j
                    if abs(di) < 1e-14 and abs(dj) < 1e-14:
                        continue

                    # Apply both updates
                    if abs(di) > 1e-14:
                        conv[lo_i:hi_i + 1] += 2.0 * di * a_slice_i
                        if vkq_i:
                            conv[kq_i] += di * di
                    if abs(dj) > 1e-14:
                        conv[lo_j:hi_j + 1] += 2.0 * dj * a_slice_j
                        if vkq_j:
                            conv[kq_j] += dj * dj

                    new_s = s - old_i - old_j + xi + xj
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s) if new_s > 1e-10 else 1e10

                    if new_obj < best_local:
                        best_local = new_obj
                        best_xi, best_xj = xi, xj

                    # Rollback
                    if abs(di) > 1e-14:
                        conv[lo_i:hi_i + 1] -= 2.0 * di * a_slice_i
                        if vkq_i:
                            conv
