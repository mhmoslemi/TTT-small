"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines LP-based alternating directions with Nesterov momentum, adaptive
relaxation, dual-guided perturbations, spectral exploration, and a novel
peak-splitting heuristic. Uses aggressive time management across multiple

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M, relax=1.0, x0=None):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax for all k, g >= 0.
    Returns (g, sum_g) or (None, 0)."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    A_ub = build_toeplitz(a)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    kwargs = {'method': 'highs', 'options': {'presolve': True}}
    if x0 is not None:
        kwargs['x0'] = np.maximum(x0, 0)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, **kwargs)
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, 0.0


def solve_lp_direct(a, relax=1.0):
    """Solve LP directly: minimize t s.t. (a*g)_k <= t, sum(g)=1, g>=0.
    Returns (g, t) or (None, inf)."""
    n = len(a)
    m = 2 * n - 1
    # Variables: g_0, ..., g_{n-1}, t
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = build_toeplitz(a)
    # (a*g)_k <= t  =>  (a*g)_k - t <= 0
    A_full = np.hstack([A_ub, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_full, b_ub=b_ub,
                      A_eq=np.ones((1, n + 1)), b_eq=np.array([1.0]),
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            t = float(res.x[n])
            return g, t
    except Exception:
        pass
    return None, np.inf


def nesterov_lp_polish(a, time_budget=100.0, momentum_start=0.5,
                       momentum_end=0.0, verbose=False):
    """LP-based polynomial with Nesterov momentum and adaptive relaxation."""

    v = np.zeros(n)
    it = 0
    total_iters_est = int(time_budget / 0.5)  # rough estimate
    stall = 0
    relax = 1.0

    while time.monotonic() - start < time_budget - 1:
        it += 1
        # Adaptive momentum
        progress = min(1.0, it / max(1, total_iters_est))
        momentum = momentum_start * (1.0 - progress) * momentum_start + \
                   momentum_end * progress
        # Actually let's use a simpler schedule
        momentum = momentum_start * np.exp(-0.02 * it)

        # Nesterov lookahead
        y = a + momentum * v

        conv = compute_conv(y)
        M = float(np.max(conv))

        g, S = solve_lp(y, M, relax=relax, x0=best_a)

        if g is None or S <= 1.0 + 1e-6:
            stall += 1
            if stall > 20:
                relax = min(relax + 0.03, 1.5)
                if relax > 1.3:
                    rng = np.random.default_rng(it * 104729 + 7)
                    a = a * np.exp(rng.normal(0, 0.04, n))
                    a = np.maximum(a, 0)
                    s2 = np.sum(a)
                    if s2 > 1e-10:
                        a = a / s2
                    else:
                        a = best_a.copy()
                    v = np.zeros(n)
                    relax = 1.0
                    stall = 0
            v = 0.5 * v
            continue

        g = g / S  # normalize to sum 1

        # Line search with multiple t values
        best_t_obj = best_obj
        best_a_new = best_a.copy()
        best_t = 0.0

        # Geometric line search
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]
        for t in t_candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_new = a_try.copy()
                best_t = t

        if best_t_obj < best_obj - 1e-10:
            # Update momentum
            v = momentum * v + best_t * (best_a_new - a)
            a = best_a_new.copy()
            best_obj = best_t_obj
            best_a = best_a_new.copy()
            stall = 0
            relax = max(1.0, relax - 0.01)

            if stall > 30:

                scale = rng.uniform(0.02, 0.06)

                    a = a / s2
                else:
                    a = best_a.copy()
                    v = np.zeros(n)
                relax = min(1.2, relax + 0.05)
                stall = 0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    Nesterov LP it={it}, obj={best_obj:.6f}, "
                  f"relax={relax:.2f}, stall={stall}, time={elapsed:.1f}s")

    return best_a, best_obj


def peak_split_perturb(a, num_peaks=5, num_tries=30, time_budget=10.0):
    """Targeted perturbation that splits reduce dominant peaks."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    for _ in range(num_tries):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        sorted_idx = np.argsort(conv)[::-1]
        top_peaks = sorted_idx[:num_peaks]

        k = int(top_peaks[rng.integers(0, len(top_peaks))])

        # Find positions contributing to this peak
        affected = []
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                affected.append((i, j, a[i] * a[j]))

        if not affected:
            continue

        # Sort by contribution
        affected.sort(key=lambda x: -x[2])
        top_affected = affected[:min(10, len(affected))]

        # Strategy: reduce some contributors, increase others
        a_new = a.copy()
        num_reduce = max(1, len(top_affected) // 2)

        for idx in range(num_reduce):
            i, j, _ = top_affected[idx]
            factor = rng.uniform(0.6, 0.9)
            # Reduce a[i] and compensate by increasing a[j] (if j != i)
            if j != i:
                a_new[i] *= factor
                a_new[j] += a[i] * (1 - factor) * 0.5
            else:
                a_new[i] *= factor

        # Also try: swap mass between peak contributors and non-contributors
        non_affected = [i for i in range(n) if i not in dict([(x[0], 1) for x in top_affected])]
        if non_affected:
            src = top_affected[0][0]
            dst = int(rng.choice(non_affected))
            transfer = a[src] * rng.uniform(0.05, 0.15)
            a_new[src] -= transfer
            a_new[dst] += transfer

        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a

def subgradient_step(a, num_steps=50, time_budget=10.0):
    """Subgradient descent on the peak of the autocorrelation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    step_size = 0.01
    for step in range(num_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        S = float(np.sum(a))
        K = np.where(np.abs(conv - M) < 1e-10)[0]

        if len(K) == 0:
            break

        # Subgradient
        grad = np.zeros(n)
        for k in K:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]

        grad /= len(K)
        grad = (4.0 * n / (S * S)) * grad - (4.0 * n * M / (S ** 3)) * np.ones(n)

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = step_size / (1.0 + 0.01 * step)

        a_new = a - step * grad / gnorm
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.95

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimization combining Nesterov-momentum LP,
    peak-splitting perturbations, spectral optimization, subgradient steps,
    and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")

    # Phase 1: Nesterov LP polish on parent
    if best_seq is not None and time.monotonic() < deadline - 300:
        n_parent = len(best_seq)
        print(f"\n=== Phase 1: Nesterov LP Polish (n={n_parent}) ===")
        polish_budget = min(250.0, (deadline - time.monotonic()) * 0.25)
        a_pol, obj_pol = nesterov_lp_polish(best_seq, time_budget=polish_budget,
                                            momentum_start=0.3, momentum_end=0.0)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  After Nesterov LP: {best_score:.6f}")
        else:
            print(f"  After Nesterov LP: {obj_pol:.6f} (no improvement)")

        # Peak splitting
        if time.monotonic() < deadline - 200:
            a_ps = peak_split_perturb(best_seq, num_peaks=5, num_tries=40,
                                      time_budget=15.0)
            obj_ps = compute_obj(a_ps)
            if obj_ps < best_score:
                best_score = obj_ps
                best_seq = a_ps.copy()
                print(f"  After peak split: {best_score:.6f}")

            # LP polish after peak split
            if time.monotonic() < deadline - 150:
                a_pol2, obj_pol2 = nesterov_lp_polish(best_seq, time_budget=60.0,
                                                      momentum_start=0.2)
                if obj_pol2 < best_score:
                    best_score = obj_pol2
                    best_seq = a_pol2.copy()
                    print(f"  After LP polish 2: {best_score:.6f}")

    # Phase 2: Varying relaxation LP on best
    if best_seq is not None and time.monotonic() < deadline - 200:
        print(f"\n=== Phase 2: Varying Relaxation LP ===")
        vr_budget = min(60.0, (deadline - time.monotonic()) * 0.1)
        a_vr, obj_vr = lp_with_varying_relax(best_seq, time_budget=vr_budget)
        if obj_vr < best_score:
            best_score = obj_vr
            best_seq = a_vr.copy()
            print(f"  After varying relax: {best_score:.6f}")

    # Phase 3: Multi-n exploration
    print(f"\n=== Phase 3: Multi-n Exploration ===")

    if best_seq is not None:
        np_ = len(best_seq)
        n_values = [max(200, np_ - 150), max(200, np_ - 50), np_,
                    np_ + 50, np_ + 100, np_ + 200, np_ + 400]
    else:
        n_values = [300, 500, 700, 1000, 1200]

    n_values = sorted(set(max(100, min(2000, nn)) for nn in n_values))
    print(f"  n values: {n_values}")

    for n in n_values:
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        alloc = min(remaining * 0.15, 80.0)
        alloc = max(15.0, alloc)

        print(f"\n  N={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        # Start from resampled best or random init
        if best_seq is not None:
            a_start = resample_sequence(best_seq, n)
        else:
            a_start = random_init(n, seed + n * 7919)

        # LP polish
        a_opt, obj_opt = nesterov_lp_polish(a_start, time_budget=alloc * 0.5,
                                            momentum_start=0.3)

        # Spectral step
        if time.monotonic() < deadline - 60:
            a_opt = spectral_optimize(a_opt, num_iters=15,
                                      time_budget=min(5.0, alloc * 0.15))
            obj_opt = compute_obj(a_opt)

        # Peak split
        if time.monotonic() < deadline - 60:
            a_opt = peak_split_perturb(a_opt, num_peaks=4, num_tries=20,
                                       time_budget=min(5.0, alloc * 0.15))
            obj_opt = compute_obj(a_opt)

        # Final LP
        if time.monotonic() < deadline - 40:
            a_opt, obj_opt = nesterov_lp_polish(a_opt,
                                                time_budget=min(alloc * 0.2, 20.0),
                                                momentum_start=0.1)

        print(f"  N={n}: obj={obj_opt:.6f}")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Phase 4: Subgradient refinement
    if best_seq is not None and time.monotonic() < deadline - 100:
        print(f"\n=== Phase 4: Subgradient Refinement ===")
        a_sg = subgradient_step(best_seq, num_steps=100, time_budget=20.0)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  After subgradient: {best_score:.6f}")

    # Phase 5: Final intensive polish
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"  Best before: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  Polishing n={n_best}, {remaining:.0f}s remaining")

        a_cur = best_seq.copy()

        # Round 1: Aggressive Nesterov LP
        alloc1 = min(remaining * 0.35, 150.0)
        a_cur, obj1 = nesterov_lp_polish(a_cur, time_budget=alloc1,
                                         momentum_start=0.4, momentum_end=0.0)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a_cur.copy()
            print(f"  Round 1 (Nesterov LP): {best_score:.6f}")

        # Round 2: Peak split + LP
        if time.monotonic() < deadline - 30:
            a_cur = peak_split_perturb(best_seq, num_peaks=8, num_tries=60,
                                       time_budget=min(15.0, (deadline - time.monotonic()) * 0.05))
            obj2 = compute_obj(a_cur)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_cur.copy()
                print(f"  Round 2 (peak): {best_score:.6f}")

            a_cur, obj2b = nesterov_lp_polish(best_seq,
                                              time_budget=min(60.0, (deadline - time.monotonic()) * 0.15),
                                              momentum_start=0.3)
            if obj2b < best_score:
                best_score = obj2b
                best_seq = a_cur.copy()
                print(f"  Round 2b (LP): {best_score:.6f}")

        # Round 3: Spectral + LP
        if time.monotonic() < deadline - 30:
            a_cur = spectral_optimize(best_seq, num_iters=20,
                                      time_budget=min(10.0, (deadline - time.monotonic()) * 0.05))
            obj3 = compute_obj(a_cur)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a_cur.copy()
                print(f"  Round 3 (spec): {best_score:.6f}")

            a_cur, obj3b = nesterov_lp_polish(best_seq,
                                              time_budget=min(40.0, (deadline - time.monotonic()) * 0.1),
                                              momentum_start=0.2)
            if obj3b < best_score:
                best_score = obj3b
                best_seq = a_cur.copy()
                print(f"  Round 3b (LP): {best_score:.6f}")

        # Round 4: Try varying n around best
        if time.monotonic() < deadline - 30:
            n_cur = len(best_seq)
            for dn in [-20, 20, -50, 50]:
                if time.monotonic() > deadline - 30:
                    break
                n_try = n_cur + dn
                if n_try < 100:
                    continue
                a_try = resample_sequence(best_seq, n_try)
                a_try, obj_try = nesterov_lp_polish(a_try, time_budget=20.0,
                                                    momentum_start=0.2)
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    print(f"  Round 4 (n={n_try}): {best_score:.6f}")

        # Round 5: Asymmetric perturbation + LP
        if time.monotonic() < deadline - 20:
            rng = np.random.default_rng(seed + 7777)
            for ptry in range(5):
                if time.monotonic() > deadline - 20:
                    break
                n_b = len(best_seq)
                factor = rng.uniform(0.85, 1.15)
                split = int(n_b * rng.uniform(0.4, 0.6))
                a_asym = best_seq.copy()
                a_asym[:split] *= factor
                a_asym[split:] *= (2.0 - factor) * 0.5
                a_asym = np.maximum(a_asym, 0)
                s_as = np.sum(a_asym)
                if s_as > 1e-10:
                    a_asym = a_asym / s_as
                    a_asym, obj_as = nesterov_lp_polish(a_asym, time_budget=15.0,
                                                        momentum_start=0.15)
                    if obj_as < best_score:
                        best_score = obj_as
                        best_seq = a_asym.copy()
                        print(f"  Round 5 (asym {ptry}): {best_score:.6f}")

    # Phase 6: Ultra-fine polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\n=== Phase 6: Ultra-fine polish ({remaining:.0f}s) ===")
        a_final, obj_final = nesterov_lp_polish(best_seq, time_budget=remaining * 0.8,
                                                momentum_start=0.1, momentum_end=0.0)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Ultra-fine: {best_score:.6f}")

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
