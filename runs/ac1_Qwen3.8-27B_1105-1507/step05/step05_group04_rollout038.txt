"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using adaptive LP mirror descent
with gradient warm start, multi-length exploration, and Nesterov-style momentum.
Key improvements: (1) faster gradient warm start for better initial points,
(2) adaptive constraint selection in LP (top_frac varies with convergence),
(3) Nesterov momentum for faster convergence, (4) multi-length search,
(5) more aggressive final polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# --- Utility functions ---

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def fast_cross_conv(a, b):
    """Cross-convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_lp_matrix(n, a, active_k=None):
    """Build sparse convolution matrix for LP."""
    if active_k is None:
        active_k = np.arange(2 * n - 1)
    rows, cols, vals = [], [], []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))

def solve_lp(a, M, top_frac=1.0, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    
    if top_frac < 1.0:
        threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
        active_k = np.where(b >= threshold)[0]
    else:
        active_k = np.arange(2 * n - 1)
    
    A = build_lp_matrix(n, a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, np.inf)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if top_frac < 1.0 and g.sum() > 1e-10:
                # Verify against all constraints
                ag = fast_cross_conv(a, g)
                if ag.max() <= M * (1.0 + 1e-4):
                    return g
            return g
    except Exception:
        pass
    return None

def smooth_gradient_step(a, tau=20.0):
    """Compute gradient of smooth max approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    b_shifted = b - b_max
    exp_b = np.exp(tau * b_shifted)
    p = exp_b / exp_b.sum()
    
    # Gradient: 2 * sum_k p[k] * a[k-j] for each j
    # This is 2 * cross_correlate(p, a) computed via FFT
    size = 1
    while size < (2*n-1) + n:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a[::-1], size)
    conv_full = np.fft.irfft(fp * fa, size)
    grad = 2.0 * conv_full[n-1:n-1+n]
    
    return grad

def run_gradient_descent(a, max_iters, deadline, tau=20.0, lr=0.1, momentum=0.85):
    """Quick gradient descent with smooth max approximation."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad = smooth_gradient_step(a, tau)
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a = a_new / s
        if it % 5 == 0:
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
    
    J = compute_J(a)
    if J < best_J:
        best_J = J
        best_a = a.copy()
    return best_a, best_J

def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def lp_mirror_descent(a, deadline, verbose=True, max_no_improve=300):

    - Periodic structured perturbation for escaping local minima

    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    restart_count = 0
    it = 0
    prev_a = a.copy()
    
    while time.monotonic() < deadline:
        it += 1
        t0 = time.monotonic()
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive top_frac: start with fewer constraints, add more as we converge
        if it < 100:
            top_frac = 0.5
        elif it < 500:
            top_frac = 0.7
        elif it < 2000:
            top_frac = 0.85
        else:
            top_frac = 1.0
        
        # Nesterov lookahead (small momentum)
        if it > 1:
            y = a + 0.15 * (a - prev_a)
            y = np.maximum(y, 1e-10)
            y = y / y.sum()
        else:
            y = a
        
        # Solve LP with y
        g = solve_lp(y, M, top_frac=top_frac, time_limit=3.0)
        
        if g is None or g.sum() < 1e-10:
            # Try with full constraints
            g = solve_lp(a, M, top_frac=1.0, time_limit=5.0)
            if g is None or g.sum() < 1e-10:
                break
        g = g / g.sum()
        
        # Line search
        diff = np.max(np.abs(g - a))
        base_t = min(0.7, max(0.05, diff * 1.5))
        
        # Backtracking line search
        t = base_t
        improved = False
        for _ in range(8):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            
            if J_trial < best_J - 1e-14:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                break
            elif J_trial < compute_J(a) - 1e-14:
                a = a_trial
                improved = True
                break
            else:
                t *= 0.5
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        prev_a = a.copy()
        
        # Periodic perturbation
        if no_improve > max_no_improve:
            restart_count += 1
            rng = np.random.default_rng(it * 7919 + restart_count * 31)
            
            # Different perturbation strategies
            strategy = restart_count % 3
            if strategy == 0:
                # Smooth random perturbation
                freqs = rng.integers(1, 5, size=3)
                phases = rng.uniform(0, 2 * np.pi, size=3)
                t_arr = np.linspace(0, 1, n, endpoint=False)
                pert = np.zeros(n)
                for f, ph in zip(freqs, phases):
                    pert += np.cos(2 * np.pi * f * t_arr + ph)
                scale = 0.1 / (1.0 + 0.2 * restart_count)
                a = a + scale * pert
            elif strategy == 1:
                # Element-wise perturbation
                scale = 0.15 / (1.0 + 0.2 * restart_count)
                a = a * (1.0 + scale * rng.normal(0, 1, n))
            else:
                # Block perturbation
                n_blocks = max(4, n // 128)
                block_scale = 0.2 / (1.0 + 0.2 * restart_count)
                block_mult = rng.uniform(1 - block_scale, 1 + block_scale, n_blocks)
                a = a * np.repeat(block_mult, n // n_blocks + 1)[:n]
            
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            prev_a = a.copy()
            no_improve = 0
            
            if verbose and restart_count % 10 == 0:
                print(f"    Restart #{restart_count} at iter {it}, best_J={best_J:.8f}", flush=True)
        
        if verbose and it % 1000 == 0:

            print(f"    iter={it} J={best_J:.8f} M={M:.6f} diff={diff:.4f} "

def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy() / p.sum())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
    
    inits.append(np.ones(n) / n)
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t_arr - 1)), 0.01))
    inits.append(np.maximum(0.3 + 0.7 * np.exp(-4 * (t_arr - 0.5) ** 2), 0.01))
    
    # Multi-frequency
    for f in [2, 3, 5, 7]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t_arr), 0.01))
    
    # Edge-concentrated
    inits.append(np.maximum(np.exp(-3 * t_arr ** 2) + np.exp(-3 * (t_arr - 1) ** 2) + 0.1, 0.01))
    
    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Power law
    inits.append(np.maximum(t_arr ** 0.5 + (1-t_arr) ** 0.5, 0.01))
    
    # "Anti-peak" - low in center, high at edges
    inits.append(np.maximum(1.0 - 0.5 * np.exp(-4 * (t_arr - 0.5) ** 2) + 0.2, 0.01))
    
    # Normalize all
    inits = [np.maximum(x, 1e-6) / np.maximum(x, 1e-6).sum() for x in inits]
    return inits

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using adaptive LP mirror descent
with fast reduced-constraint LPs, gradient warm start, multi-length search,
and adaptive restarts. Key improvements: (1) aggressive constraint reduction

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n, a, active_k):
    """Build sparse convolution matrix for LP constraints."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        count = j_max - j_min + 1
        if count > 0:
            rows.extend([idx] * count)
            cols.extend(range(j_min, j_max + 1))
            vals.extend(a[k - j_max:k - j_min + 1][::-1])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))


def solve_lp(a, M, top_frac=0.6, time_limit=3.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for active k, g >= 0."""
    n = len(a)
    b = fast_conv(a)

    if top_frac < 1.0:
        threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
        active_k = np.where(b >= threshold)[0]
        if len(active_k) < 10:
            active_k = np.arange(2 * n - 1)
    else:
        active_k = np.arange(2 * n - 1)

    A = build_lp_matrix(n, a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, np.inf)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if top_frac < 1.0 and g.sum() > 1e-10:
                ag = fast_cross_conv(a, g)
                if ag.max() <= M * (1.0 + 1e-4):
                    return g
            return g
    except Exception:
        pass
    return None


def solve_lp_full(a, M, time_limit=5.0):
    """Solve LP with all constraints."""
    return solve_lp(a, M, top_frac=1.0, time_limit=time_limit)


def smooth_gradient_step(a, tau=20.0):
    """Compute gradient of smooth max approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    b_shifted = b - b_max
    exp_b = np.exp(tau * b_shifted)
    p = exp_b / exp_b.sum()

    size = 1
    while size < (2 * n - 1) + n:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a[::-1], size)
    conv_full = np.fft.irfft(fp * fa, size)
    grad = 2.0 * conv_full[n - 1:n - 1 + n]
    return grad


def run_gradient_descent(a, max_iters, deadline, tau=20.0, lr=0.1, momentum=0.85):

    v = np.zeros(n)
    current_tau = tau

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad = smooth_gradient_step(a, current_tau)
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a = a_new / s
        if it % 5 == 0:

        best_a = a.copy()
    return best_a, best_J


def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        state.update(a_opt, "phase1")
    print(f"Phase 1 best: J={candidates[0][0]:.8f}", flush=True)

    # === PHASE 2: LP Mirror Descent ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'=' * 60}", flush=True)

    p2_deadline = time.monotonic() + 1000.0
    p2_start = time.monotonic()

    top_candidates = candidates[:min(3, len(candidates))]
    time_alloc = [0.55, 0.30, 0.15]

    for ci, (J_init, a_init) in enumerate(top_candidates):
        if time.monotonic() > p2_deadline:
            break
        remaining = p2_deadline - time.monotonic()
        if remaining < 60:
            break
        alloc_time = remaining * time_alloc[ci]
        alloc_time = max(60.0, alloc_time)
        phase_dl = min(time.monotonic() + alloc_time, p2_deadline)

        print(f"\n  Candidate {ci} (init J={J_init:.8f}), budget={alloc_time:.0f}s",
              flush=True)

        a_best_c, J_best_c = lp_mirror_descent(
            a_init, phase_dl, verbose=True, max_no_improve=200
        )
        state.update(a_best_c, f"lp_cand{ci}")

    # === PHASE 3: Multi-length exploration ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 3: Multi-Length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_deadline = min(time.monotonic() + 150.0, deadline - 60.0)

    if state.best_a is not None and time.monotonic() < p3_deadline:
        base_n = state.best_a.size
        n_tries = list(set([
            base_n + 64, base_n - 64, base_n + 128,
            base_n * 2 if base_n * 2 <= 2048 else base_n + 256,
            max(128, base_n - 128)
        ]))
        n_tries = [n for n in n_tries if 128 <= n <= 2048 and n != base_n]
        n_tries = n_tries[:4]

        for n_try in n_tries:
            if time.monotonic() > p3_deadline:
                break

            a_ext = resample_seq(state.best_a, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            a_ext, _ = run_gradient_descent(
                a_ext, 40, min(time.monotonic() + 20, p3_deadline - 20),
                tau=15, lr=0.1, momentum=0.8)
            state.update(a_ext, f"n={n_try}_grad")

            if time.monotonic() < p3_deadline - 10:
                a_ext2, _ = lp_mirror_descent(
                    a_ext, min(time.monotonic() + 30, p3_deadline - 5),
                    verbose=False, max_no_improve=80)
                state.update(a_ext2, f"n={n_try}_lp")

        # Palindrome variant
        if time.monotonic() < p3_deadline - 20:
            a_pal = make_palindrome(state.best_a)
            a_pal = np.maximum(a_pal, 1e-6)
            a_pal = a_pal / a_pal.sum()
            a_pal, _ = lp_mirror_descent(
                a_pal, min(time.monotonic() + 40, p3_deadline - 5),
                verbose=False, max_no_improve=100)
            state.update(a_pal, "palindrome")

    state.report("Phase 3 complete")

    # === PHASE 4: Final polishing ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 4: Final Polishing", flush=True)
    remaining = deadline - time.monotonic()
    print(f"Remaining: {remaining:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    if state.best_a is not None and remaining > 30:
        a_polish, _ = lp_mirror_descent(
            state.best_a, deadline - 5, verbose=True, max_no_improve=400)
        state.update(a_polish, "polish")

    # Fallback
    if state.best_a is None:
        state.best_a = np.ones(128)
        state.best_J = compute_J(state.best_a)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={state.best_J:.8f} n={state.best_a.size} "
          f"time={time.monotonic() - state.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in state.best_a]
