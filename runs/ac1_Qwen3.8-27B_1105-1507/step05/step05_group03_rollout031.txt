import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# Helper functions

def fft_convolve(a, b=None):
    """FFT-based convolution."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fft_convolve(a)
    return 2.0 * n * b.max() / (s * s)

def cd_sweeps(a, max_sweeps=1000, deadline=None, verbose=False, seed=42):
    """
    Coordinate descent on J(a). Returns optimized a and best J.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a, float('inf')
    
    b = fft_convolve(a).copy()
    best_a = a.copy()
    best_J = compute_J(a)
    rng = np.random.default_rng(seed)
    
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Shuffle order each sweep
        rng.shuffle(order)
        
        for i in order:
            old_x = a[i]
            S_0 = S - old_x
            
            if S_0 < 1e-15:
                # All other elements are ~0
                # J = 2n for any x > 0
                continue
            
            # Precompute affected terms
            # Affected k: i <= k <= i+n-1, intersected with [0, 2n-2]
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # b[k] = d_k + c_k * x (linear) or d_k + x^2 (quadratic if k=2i)
            affected_ks = np.arange(k_start, k_end + 1)
            # a[k-i] for k in affected_ks: indices k-i range from 0 to n-1
            a_vals = a[affected_ks - i]  # a[k-i] for each affected k

            # k = 2i means the j=i and j=k-i terms are the same
            quad_mask = (affected_ks == 2 * i)
            lin_mask = ~quad_mask

            D_lin = b[affected_ks[lin_mask]] - 2.0 * old_x * a_vals[lin_mask]
            C_lin = 2.0 * a_vals[lin_mask]
            
            # For quadratic term (if exists): b[k] = (b[k] - old_x^2) + x^2
            if quad_mask.any():
                D_quad = b[affected_ks[quad_mask]] - old_x**2
            else:
                D_quad = None
            
            # Unaffected max
            # Unaffected k: k < i or k > i+n-1
            if k_start > 0:
                unaff_left = b[:k_start].max()
            else:
                unaff_left = -np.inf
            if k_end < 2 * n - 2:
                unaff_right = b[k_end+1:].max()
            else:
                unaff_right = -np.inf
            U = max(unaff_left, unaff_right)

            x_max = max(S_0 * 5.0, old_x * 3.0, 1e-6)

                S_x = S_0 + x
                if S_x < 1e-15:
                    return float('inf')
                return 2.0 * n * M_val / (S_x * S_x)
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~ 0.618
            lo = 0.0
            hi = x_max
            m1 = hi - gr * (hi - lo)
            m2 = lo + gr * (hi - lo)
            f1 = J_of_x(m1)
            f2 = J_of_x(m2)
            for _ in range(40):
                if f1 < f2:
                    hi = m2
                    m2 = m1
                    f2 = f1
                    m1 = hi - gr * (hi - lo)
                    f1 = J_of_x(m1)
                else:
                    lo = m1
                    m1 = m2
                    f1 = f2
                    m2 = lo + gr * (hi - lo)
                    f2 = J_of_x(m2)
            
            x_opt = (lo + hi) / 2.0
            x_opt = max(0.0, min(x_opt, x_max))
            
            # Update a[i] and b
            a[i] = x_opt
            S = S_0 + x_opt

def fft_convolve(a, b=None):
    """FFT-based full convolution."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fft_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def cd_sweeps(a, max_sweeps=2000, deadline=None, verbose=False, seed=42, 
              gi_iters=35, x_max_factor=5.0):
    """
    Fast coordinate descent on J(a).
    Each coordinate update uses golden-section search on the 1D subproblem.
    Returns (best_a, best_J).
    """
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a, float('inf')

    b = fft_convolve(a).copy()
    best_a = a.copy()
    best_J = 2.0 * n * b.max() / (S * S)
    rng = np.random.default_rng(seed)

    order = np.arange(n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    no_improve = 0

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        rng.shuffle(order)
        improved = False

        for i in order:
            old_x = a[i]
            S_0 = S - old_x

            if S_0 < 1e-15:
                continue

            # Affected k range: k from i to i+n-1, clipped to [0, 2n-2]
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            n_aff = k_end - k_start + 1

            if n_aff <= 0:
                continue

            # Indices a[k-i] for k in [k_start, k_end]: range [0, n_aff-1]
            a_slice = a[:n_aff]  # a[k-i] for k = k_start..k_end => a[0..n_aff-1]

            # Quadratic at k=2i (if in range)
            quad_offset = 2 * i - k_start  # position of k=2i in affected range
            has_quad = 0 <= quad_offset < n_aff

            if has_quad:
                lin_indices = np.concatenate([np.arange(quad_offset), np.arange(quad_offset + 1, n_aff)])
            else:
                lin_indices = np.arange(n_aff)

            # Linear coefficients: C = 2*a[k-i], D = b[k] - 2*old_x*a[k-i]
            if len(lin_indices) > 0:
                C_lin = 2.0 * a_slice[lin_indices]
                ks_lin = k_start + lin_indices
                D_lin = b[ks_lin] - 2.0 * old_x * a_slice[lin_indices]
            else:
                C_lin = np.array([])
                D_lin = np.array([])

            # Quadratic: D_quad = b[2i] - old_x^2
            if has_quad:
                D_quad = b[2 * i] - old_x * old_x
            else:
                D_quad = -np.inf

            # Unaffected max
            if k_start > 0:
                U_l = b[:k_start].max()
            else:
                U_l = -np.inf
            if k_end < 2 * n - 2:
                U_r = b[k_end + 1:].max()
            else:
                U_r = -np.inf
            U = max(U_l, U_r) if k_start > 0 or k_end < 2 * n - 2 else -np.inf

            # Golden section search on x in [0, x_max]
            x_max = max(S_0 * x_max_factor, old_x * 3.0, 1e-6)

            # Precompute for fast evaluation
            c_lin_arr = C_lin if len(C_lin) > 0 else None
            d_lin_arr = D_lin if len(D_lin) > 0 else None

            lo = 0.0
            hi = x_max
            m1 = hi - gr * (hi - lo)
            m2 = lo + gr * (hi - lo)

            # Inline J evaluation for speed
            def eval_J(x):
                Sv = S_0 + x
                if Sv < 1e-15:
                    return 1e30
                if c_lin_arr is not None:
                    Mv = np.maximum.reduce([d_lin_arr + c_lin_arr * x])
                else:
                    Mv = -np.inf
                if has_quad:
                    Mv = max(Mv, D_quad + x * x)
                Mv = max(Mv, U)
                if Mv < 0:
                    Mv = 0.0
                return 2.0 * n * Mv / (Sv * Sv)

            f1 = eval_J(m1)
            f2 = eval_J(m2)
            for _ in range(gi_iters):
                if f1 < f2:
                    hi = m2
                    m2 = m1
                    f2 = f1
                    m1 = hi - gr * (hi - lo)
                    f1 = eval_J(m1)
                else:
                    lo = m1
                    m1 = m2
                    f1 = f2
                    m2 = lo + gr * (hi - lo)
                    f2 = eval_J(m2)

            x_opt = (lo + hi) * 0.5
            x_opt = max(0.0, min(x_opt, x_max))

            if abs(x_opt - old_x) < 1e-12:
                continue

            # Update a[i] and b
            a[i] = x_opt
            S += (x_opt - old_x)

            # Update b for affected positions
            if len(lin_indices) > 0:
                ks_lin = k_start + lin_indices
                b[ks_lin] = D_lin + C_lin * x_opt
            if has_quad:
                b[2 * i] = D_quad + x_opt * x_opt

            improved = True

        # End of sweep
        J_now = 2.0 * n * b.max() / (S * S)
        if J_now < best_J - 1e-12:
            best_J = J_now
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and sweep % 200 == 0:
            print(f"    CD sweep {sweep}: J={J_now:.10f} best={best_J:.10f}", flush=True)

        if no_improve > 200:
            break

    return best_a, best_J

def build_lp_matrix(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(a, M, row_idx, col_idx, time_limit=5):
    """Solve LP: max sum(g) s.t. (a@g)[k] <= M for all k, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max((1-t)a+tg * ((1-t)a+tg))."""
    A = fft_convolve(a)
    B = fft_convolve(a, g)
    C = fft_convolve(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters=2000, deadline=None, row_idx=None, col_idx=None,

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 1:
        return np.full(dest_n, src[0])
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_ramp(n, reverse=False):
    """Asymmetric ramp sequence."""
    a = np.arange(1, n + 1, dtype=np.float64)
    if reverse:
        a = a[::-1]
    return a / a.sum()


def make_exponential(n, alpha=0.5, reverse=False):
    """Exponential decay sequence."""
    t = np.arange(n, dtype=np.float64)
    if reverse:
        t = t[::-1]
    a = np.exp(-alpha * t)
    return a / a.sum()


def make_spectral_flat(n, rng):
    """Initialize using spectral flatness approach."""
    # Start with random phase
    freqs = np.fft.rfftfreq(n)
    # Create a target power spectrum that's as flat as possible
    # Keep DC = 1, make other bins roughly equal
    power = np.ones(n // 2 + 1)
    power[0] = 1.0  # DC
    # Add some randomness to avoid perfect flatness (which is unachievable)
    noise = rng.normal(0, 0.3, power.shape)
    noise[0] = 0
    power = np.maximum(power + noise, 0.01)
    # Create complex FFT with appropriate magnitude
    phases = rng.uniform(0, 2 * np.pi, power.shape)
    phases[0] = 0  # DC is real
    if n % 2 == 0 and power.shape[0] > n // 2:
        phases[-1] = 0  # Nyquist is real
    FFT_target = power * np.exp(1j * phases)
    # Inverse FFT
    a = np.fft.irfft(FFT_target, n)
    # Take absolute value to make non-negative (this destroys some structure but gives a start)
    a = np.abs(a)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def make_symmetric_asymmetric_mix(n, rng):
    """Create an asymmetric sequence by mixing a symmetric base with an asymmetric perturbation."""
    base = np.ones(n) / n
    # Add asymmetric perturbation
    t = np.linspace(0, 1, n)
    asym = np.sin(np.pi * t) * (1 - 2 * t)  # asymmetric
    a = base + 0.3 * asym
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return base
    return a / s

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            self.best_n = a.size
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self):
        if self.best is not None:
            print(f"  best: J={self.best_score:.8f} n={self.best_n}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining coordinate descent, LP mirror descent,
    subgradient descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = make_ramp(256)
        tracker.update(parent, "(default ramp)")

    # Initialize multiple candidates
    n_base = parent.size
    candidates = []
    candidates.append(("parent", parent.copy() / parent.sum()))

    # Structured initializations at base length
    candidates.append(("ramp_f", make_ramp(n_base)))
    candidates.append(("ramp_b", make_ramp(n_base, reverse=True)))
    candidates.append(("exp_f", make_exponential(n_base, alpha=0.3)))
    candidates.append(("exp_b", make_exponential(n_base, alpha=0.3, reverse=True)))
    candidates.append(("spectral", make_spectral_flat(n_base, rng)))
    candidates.append(("asym_mix", make_symmetric_asymmetric_mix(n_base, rng)))
    # Random asymmetric
    a_rand = np.maximum(rng.random(n_base), 0.01)
    a_rand = a_rand / a_rand.sum()
    candidates.append(("random", a_rand))

    # Phase 1: Coordinate descent from all candidates (quick, ~30s total)
