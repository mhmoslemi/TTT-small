"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = next_fast_len(n + m - 1)
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def normalize(a):
    """Normalize a to sum=1, clip to non-negative."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        a_j = a[j]
        # (a * g)[k] includes a[j] * g[k-j] for valid k-j
        # So column j of A has a_j in positions k where k-j is in [0, n-1]
        # i.e., k in [j, j+n-1], clipped to [0, m-1]
        k_start = j
        k_end = min(j + n, m)
        if k_start < k_end:
            A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None."""
    n = len(a)
    if n > 1200:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=20):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None)."""
    n = len(a)
    if n > 1000:
        return None, None
    A = build_toeplitz(a)
    # Variables: g[0..n-1], M
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # (a*g)[k] - M <= 0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

def softmax_subgradient(a, time_budget=30.0, max_iters=500):
    """Smooth softmax-based subgradient with Nesterov momentum."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.85
    step = 0.03

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        # Softmax temperature: start small (sharp), increase over time
        tau = 0.1 + 0.5 * it / max_iters
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500, 0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # conv(a,a)[k] = sum_i a[i] * a[k-i]

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines min-peak LP alternating, softmax subgradient with Nesterov momentum,
spectral shaping, multi-start with structured initializations, and adaptive
resampling across sequence lengths to escape local optima.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, next_fast_len
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = next_fast_len(2 * n - 1)
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0).astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        if k_start < k_end:
            A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a, time_limit=20):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None)."""
    n = len(a)
    if n > 1200:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_maxsum(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns normalized g or None."""
    n = len(a)
    if n > 1200:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-10,
                               'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def softmax_subgrad(a, time_budget=40.0, max_iters=600):
    """Smooth softmax subgradient with Nesterov momentum."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.88
    step = 0.025
    step_min = 0.002

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Softmax weights over convolution positions
        tau = 0.05 + 0.4 * (it / max_iters) ** 0.5
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -400.0, 0.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient: grad[j] = 2 * sum_{m=0}^{n-1} a[m] * weights[j+m]
        # (where j+m < 2n-1)
        grad = np.zeros(n)
        for j in range(n):
            m_max = min(n, 2 * n - 1 - j)
            if m_max > 0:
                grad[j] = 2.0 * np.dot(a[:m_max], weights[j:j + m_max])

        # Center gradient (project onto sum=const hyperplane)
        grad_centered = grad - np.mean(grad)

        # Nesterov momentum
        v = mu * v - step * grad_centered
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        # Decay step
        step = max(step * 0.997, step_min)

    return best_a, best_obj


def peak_chase(a, n_steps=100, step_size=0.015):
    """Chase the peak: iteratively reduce the maximum convolution entry."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_steps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

        # Subgradient direction: reduce a[i] where i contributes to peak at k_star
        # d conv[k_star]/da[i] = 2*a[k_star - i] (if 0<=k_star-i<n)
        direction = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                direction[i] = 2.0 * a[j]

        # Center to preserve sum
        direction_centered = direction - np.mean(direction)

        a_new = a - step_size * direction_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step size
            step_size *= 0.8
            if step_size < 0.001:
                break

    return best_a, best_obj

        # Current spectrum
        fa = rfft(a, size)
        power = np.abs(fa) ** 2
        p_mean = np.mean(power)

        # Target amplitude: sqrt(p_mean) for all frequencies
        target_amp = np.sqrt(p_mean)

        # Adjust phase-preserving amplitude
        new_amp = np.abs(fa)
        # Blend toward target
        blend = 0.3
        new_amp = (1 - blend) * new_amp + blend * target_amp

        # Reconstruct with original phase
        phase = np.angle(fa)
        fa_new = new_amp * np.exp(1j * phase)

        # Transform back
        a_new = irfft(fa_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
        else:
            break

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def coordinate_refine(a, time_budget=20.0, max_rounds=20):
    """Coordinate-wise refinement: try small adjustments to individual elements."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        improved = False

        # Try moving mass from high-contribution to low-contribution positions
        perm = np.random.default_rng(rnd * 31 + n).permutation(n)
        for i in perm[:min(n, 200)]:
            if time.monotonic() - start > time_budget:
                break

            # Contribution of a[i] to the peak
            j = k_star - i
            if 0 <= j < n:
                contrib = a[i] * a[j]
            else:
                contrib = 0.0

            # Try moving some mass to a random other position
            for attempt in range(3):
                other = np.random.default_rng(hash((i, rnd, attempt)) % 2**31).integers(n)
                if other == i:
                    continue

                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[i]
                    a[i] -= delta
                    a[other] += delta
                    conv_new = compute_conv(a)
                    new_obj = 2.0 * n * float(np.max(conv_new)) / (np.sum(a) ** 2)

                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = conv_new
                        improved = True
                    else:
                        a[i] += delta
                        a[other] -= delta
                        break

        if not improved and rnd > 5:
            break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def generate_chirp_init(n, seed=0):
    """Generate a chirp-like initialization (good initial autocorrelation properties)."""
    rng = np.random.default_rng(seed)
    t = np.arange(n) / n
    # Real-valued chirp: cos(2*pi*beta*t^2) shifted to be non-negative
    beta = rng.uniform(1, 5)
    sig = np.cos(2 * np.pi * beta * t ** 2)
    a = np.maximum(sig, 0) + 0.1
    # Add some randomness
    a *= np.exp(rng.normal(0, 0.1, n))
    return normalize(a)


def generate_block_init(n, seed=0):
    """Generate a block-constant initialization."""
    rng = np.random.default_rng(seed)
    n_blocks = rng.integers(3, 10)
    block_size = n // n_blocks
    blocks = rng.uniform(0.2, 1.0, n_blocks)
    a = np.zeros(n)
    for i in range(n_blocks):
        start = i * block_size
        end = min(start + block_size + (1 if i < n % n_blocks else 0), n)
        a[start:end] = blocks[i]
    return normalize(a)


def generate_random_init(n, seed=0):
    """Generate a random initialization with structured correlation."""
    rng = np.random.default_rng(seed)
    # Use a mixture of uniform and peaked distributions
    a = rng.uniform(0.1, 1.0, n)
    # Add a cosine modulation
    freq = rng.uniform(0.5, 3.0)
    a *= 1.0 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n)
    a = np.maximum(a, 0.01)
    return normalize(a)


def polish_sequence(a, time_budget=60.0):
    """Full polishing pipeline for a single sequence."""
    n = len(a)
    start = time.monotonic()
    best_a = normalize(a.copy())
    best_obj = compute_obj(best_a)

    while time.monotonic() - start < time_budget - 3:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # LP alternating
        if n <= 900 and remaining > 25:
            alloc = min(remaining * 0.3, 45.0)
            a_new, obj_new = lp_alternation(best_a, time_budget=alloc)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        # Softmax subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            alloc = min(remaining * 0.25, 30.0)
            a_new, obj_new = softmax_subgrad(best_a, time_budget=alloc)
            if obj_new < best_opt if False else obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        # Peak chase
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_new, obj_new = peak_chase(best_a, n_steps=50, step_size=0.012)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        # Spectral shaping
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_new, obj_new = spectral_shaping(best_a, time_budget=min(remaining * 0.2, 15.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        # Coordinate refinement
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            a_new, obj_new = coordinate_refine(best_a, time_budget=min(remaining * 0.2, 15.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-start optimization combining LP alternating,
    softmax subgradient, peak chasing, spectral shaping, and coordinate refinement.
    Explores multiple sequence lengths and initialization strategies.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}, budget={budget_s}s")

    # === Phase 1: Intensive optimization of parent ===
    print(f"\n=== Phase 1: Optimize parent (n={n_parent}) ===")
    phase1_end = start_time + min(400.0, (deadline - start_time) * 0.35)

    if best_seq is not None:
        a = normalize(best_seq.copy().astype(np.float64))

        # Multiple optimization cycles
        for cycle in range(6):
            if time.monotonic() > phase1_end - 30:
                break
            remaining = phase1_end - time.monotonic()
            if remaining < 20:
                break

            # LP alternating
            alloc_lp = min(remaining * 0.35, 50.0)
            a_l, obj_l = lp_alternation(a, time_budget=alloc_lp)
            if obj_l < best_score:
                best_score = obj_l
                best_seq = a_l.copy()
                a = a_l.copy()
                print(f"  P1 cycle{cycle} LP: {best_score:.8f}")
            else:
                a = a_l

            # Softmax subgradient
            remaining = phase1_end - time.monotonic()
            if remaining > 15:
                alloc_sg = min(remaining * 0.25, 30.0)
                a_s, obj_s = softmax_subgrad(a, time_budget=alloc_sg)
                if obj_s < best_score:
                    best_score = obj_s
                    best_seq = a_s.copy()
                    a = a_s.copy()
                    print(f"  P1 cycle{cycle} SG: {best_score:.8f}")
                else:
                    a = a_s

            # Peak chase
            remaining = phase1_end - time.monotonic()
            if remaining > 10:
                a_p, obj_p = peak_chase(a, n_steps=80, step_size=0.01)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a = a_p.copy()
                    print(f"  P1 cycle{cycle} PC: {best_score:.8f}")

            # Spectral
            remaining = phase1_end - time.monotonic()
            if remaining > 10:
                a_sp, obj_sp = spectral_shaping(a, time_budget=min(remaining * 0.2, 15.0))
                if obj_sp < best_score:
                    best_score = obj_sp
                    best_seq = a_sp.copy()
                    a = a_sp.copy()
                    print(f"  P1 cycle{cycle} SPEC: {best_score:.8f}")

    # === Phase 2: Try different lengths ===
    print(f"\n=== Phase 2: Explore lengths ===")
    n_options = []
    for dn in [-100, -75, -50, -25, 0, 25, 50, 75, 100, 150, 200]:
        n_try = n_parent + dn
        if 150 <= n_try <= 1200:
            n_options.append(n_try)
    # Deduplicate and sort
    n_options = sorted(set(n_options))

    for n_try in n_options:
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_parent and best_seq is not None:
            a_try = normalize(best_seq.copy())
        elif best_seq is not None:
            a_try = normalize(resample_sequence(best_seq, n_try))
        else:
            a_try = generate_random_init(n_try, seed=seed + n_try)

        print(f"  n={n_try}: {remaining:.0f}s remaining")

        # Quick optimization cycle
        alloc_total = min(remaining * 0.25, 80.0)

        # Subgradient (fast)
        a1, obj1 = softmax_subgrad(a_try, time_budget=alloc_total * 0.35)

        # LP (slower but more effective)
        alloc_lp = min(remaining * 0.12, 40.0)
        if time.monotonic() < deadline - 60 and n_try <= 900:
            a2, obj2 = lp_alternation(a1, time_budget=alloc_lp)
            if obj2 < obj1:
                a_try = a2
                obj1 = obj2
            else:
                a_try = a1
        else:
            a_try = a1

        # Peak chase
        if time.monotonic() < deadline - 40:
            a3, obj3 = peak_chase(a_try, n_steps=60, step_size=0.01)
            if obj3 < obj1:
                a_try = a3
                obj1 = obj3

        if obj1 < best_score:
            best_score = obj1
            best_seq = a_try.copy()
            print(f"  ** UPDATE: n={n_try}, obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj1:.8f} (no improvement)")

    # === Phase 3: Multi-start with different initializations ===
    print(f"\n=== Phase 3: Multi-start ===")
    n_best = len(best_seq) if best_seq is not None else 700

    inits = []
    for s in range(5):
        inits.append((generate_chirp_init(n_best, seed=1000 + s), f"chirp_{s}"))
        inits.append((generate_block_init(n_best, seed=2000 + s), f"block_{s}"))
        inits.append((generate_random_init(n_best, seed=3000 + s), f"rand_{s}"))

    # Shuffle
    rng_main = np.random.default_rng(seed)
    rng_main.shuffle(inits)

    for a_init, name in inits:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        print(f"  Try {name}: {remaining:.0f}s left")

        # Quick polish
        a_res, obj_res = polish_sequence(a_init, time_budget=min(remaining * 0.2, 40.0))
        if obj_res < best_score:
            best_score = obj_res
            best_seq = a_res.copy()
            print(f"  ** UPDATE from {name}: obj={best_score:.8f} **")
        else:
            print(f"  {name}: obj={obj_res:.8f}")

    # === Phase 4: Final intensive polish on best ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"Best before final: {best_score:.8f}")

    if best_seq is not None:
        a_final = normalize(best_seq.copy().astype(np.float64))

        for round_idx in range(8):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_pol, obj_pol = polish_sequence(a_final, time_budget=min(remaining * 0.2, 40.0))
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol.copy()
                print(f"  Final r{round_idx}: {best_score:.8f}")
            else:
                # Try perturbation and re-optimize
                rng_p = np.random.default_rng(round_idx * 777 + seed)
                for pert in range(2):
                    if time.monotonic() > deadline - 10:
                        break
                    scale = 0.03 + 0.02 * round_idx
                    a_pert = a_final * np.exp(rng_p.normal(0, scale, len(a_final)))
                    a_pert = normalize(a_pert)
                    a_pert, obj_pert = polish_sequence(a_pert, time_budget=min(remaining * 0.1, 20.0))
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_final = a_pert.copy()
                        print(f"  Final pert r{round_idx}p{pert}: {best_score:.8f}")
                    break  # Only one perturbation per round to save time

        # Very final coordinate refinement
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            a_cd, obj_cd = coordinate_refine(a_final, time_budget=min(remaining * 0.4, 20.0))
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    # Clamp values
    result = [float(np.clip(x, 0.0, 1000.0)) for x in best_seq.astype(np.float64)]

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        # Fallback
        result = [1.0] * 200
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
