"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed mirror descent on the simplex (main workhorse)
   - Soft-max smoothing of the max operator for differentiability
   - Multiplicative weight updates (mirror descent with entropy potential)
   - Adaptive step size and temperature annealing
2. Coordinate descent polishing (exact 1D optimization per coordinate)
3. Multi-scale curriculum (optimize small n first, then upscale)
4. Multi-length exploration and diverse initializations
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_gradient_smoothed(a, tau):
    """
    Compute the gradient of the smoothed objective f_tau(a) = 2n*tau*log(sum exp(b_k/tau)/(2n-1))
    where b = conv(a,a) and sum(a) = 1.
    
    Returns gradient vector of length n.
    """
    n = len(a)
    b = fast_conv(a)
    num_lags = 2 * n - 1
    
    # Softmax weights (numerically stable)
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted / tau)
    p = exp_b / exp_b.sum()
    
    # Gradient: g_i = (4n/(2n-1)) * sum_k p[k] * a[k-i]
    # Use cross-correlation
    g = np.correlate(p, a, mode='full')[:n]
    g *= 4 * n / num_lags
    
    return g, b


def mirror_descent(a_init, deadline, tau_init=0.005, tau_min=0.0005,
                   eta_init=1.0, cooling=0.9999, max_iters=100000,
                   restart_threshold=200):
    """
    Smoothed mirror descent on the simplex.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    eta = eta_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal temperature
        frac = min(it / (max_iters * 0.8), 1.0)
        tau = tau_init * (tau_min / tau_init) ** frac
        
        # Compute gradient
        g, b = compute_gradient_smoothed(a, tau)
        
        # Mirror descent update: a_i <- a_i * exp(-eta * g_i)
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        # Evaluate
        b_new = fast_conv(a_new)
        M_new = b_new.max()
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        eta *= cooling
        
        # Restart if stuck
        if no_improve > restart_threshold:
            # Small perturbation
            a = a + np.random.normal(0, 0.01 / n, n)
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            no_improve = 0
            eta = eta_init * 0.5
    
    return best_a, best_M


def coordinate_descent(a_init, deadline, max_passes=20):
    """
    Coordinate descent: for each coordinate, try to find the best value
    that reduces the objective.
    """
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        order = np.random.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            ai = a[i]
            # Try several candidate values
            if ai < 1e-6:
                candidates = [0.01, 0.03, 0.05, 0.1]
            else:
                candidates = [0.0, ai * 0.3, ai * 0.6, ai * 1.4, ai * 2.0, ai * 3.0]
            
            best_val = compute_J(a)
            best_cand = ai
            
            for cand in candidates:
                if abs(cand - ai) < 1e-14:
                    continue
                a[i] = cand
                a_clip = np.maximum(a, 0)
                s_check = a_clip.sum()
                if s_check < 0.01:
                    a[i] = ai
                    continue
                score = compute_J(a_clip / s_check)
                if score < best_val:
                    best_val = score
                    best_cand = cand
            
            a[i] = best_cand
            if best_cand != ai:
                improved = True
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random
    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())
    
    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())
    
    # Bell
    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())
    
    # Double bump
    dbl = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())
    
    # Frequency modulated
    for freq in [2, 3, 5, 7]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())
    
    # Zigzag (high-frequency)
    for freq in [10, 20, 50]:
        zz = 1.0 + 0.2 * np.cos(2 * np.pi * freq * t)
        zz = np.maximum(zz, 0.01)
        inits.append(zz / zz.sum())
    
    # Edge concentration
    edge = np.exp(-2 * (t - 0.5)**2 * 20) 
    edge = 1.0 / (edge + 0.1)  # inverted bell = edge concentration
    inits.append(edge / edge.sum())
    
    return inits

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed mirror descent on the simplex (main workhorse)
   - Soft-max smoothing of the max operator for differentiability
   - Multiplicative weight updates (mirror descent with entropy potential)
   - Adaptive step size and temperature annealing
   - Backtracking on non-improvement
2. Coordinate descent polishing (candidate-based 1D optimization)
3. Multi-scale curriculum (optimize at various n, upscale)
4. Diverse initializations and multi-length exploration
Key difference from previous: direct smoothed gradient approach allows
much faster iterations (no LP solves), enabling more exploration.
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_smoothed_gradient(a, tau):
    """
    Compute gradient of smoothed objective f_tau(a) on the simplex (sum(a)=1).
    f_tau(a) = 2n * tau * log(sum_k exp(b_k/tau) / (2n-1))
    Gradient: g_i = (4n/(2n-1)) * sum_k p_k * a[k-i]
    where p_k = softmax(b/tau).
    """
    n = len(a)
    b = fast_conv(a)
    num_lags = 2 * n - 1

    # Numerically stable softmax
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted / tau)
    p = exp_b / exp_b.sum()

    # Cross-correlation: g_i = sum_k p[k] * a[k-i]
    # np.correlate(p, a, mode='full')[i] = sum_j p[j+i] * a[j]
    g = np.correlate(p, a, mode='full')[:n]
    g *= 4.0 * n / num_lags

    return g, b, p


def mirror_descent(a_init, deadline, tau_init=0.003, tau_min=0.0003,
                   eta_init=2.0, cooling=0.99995, max_iters=100000,
                   restart_threshold=300, use_adaptive_eta=True):
    """
    Smoothed mirror descent on the simplex.
    Iteratively updates a_i <- a_i * exp(-eta * g_i), normalizes, anneals tau.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    current_M = best_M
    no_improve = 0
    eta = eta_init
    tau = tau_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal temperature over 80% of iterations
        frac = min(it / (max_iters * 0.8), 1.0)
        tau = tau_init * (tau_min / tau_init) ** frac

        # Compute gradient
        g, b, p = compute_smoothed_gradient(a, tau)

        # Mirror descent update with adaptive step size
        if use_adaptive_eta:
            # Scale eta based on current entropy level
            entropy = -np.sum(a * np.log(a + 1e-30))
            max_entropy = np.log(n)
            # Reduce eta when solution is peaked (low entropy)
            eta_scale = 0.5 + 0.5 * (entropy / max_entropy)
            effective_eta = eta * eta_scale
        else:
            effective_eta = eta

        # Multiplicative update
        a_new = a * np.exp(-effective_eta * g)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Evaluate
        b_new = fast_conv(a_new)
        M_new = b_new.max()

        if M_new < current_M - 1e-14:
            current_M = M_new
            if M_new < best_M - 1e-14:
                best_M = M_new
                best_a = a_new.copy()
                no_improve = 0
            no_improve = 0
        else:
            no_improve += 1
            # Backtrack: if no improvement, reduce eta
            if no_improve > 50:
                eta *= 0.9
                if no_improve > 150:
                    # Revert and perturb
                    a = best_a.copy() + np.random.normal(0, 0.02 / n, n)
                    a = np.maximum(a, 1e-6)
                    a = a / a.sum()
                    current_M = fast_conv(a).max()
                    no_improve = 0
                    eta = eta_init * 0.7

        a = a_new
        eta *= cooling

        # Hard restart if completely stuck
        if no_improve > restart_threshold:
            a = best_a.copy()
            # Random coordinate swap perturbation
            n_perturb = max(5, n // 20)
            indices = np.random.choice(n, n_perturb, replace=False)
            for idx in indices:
                a[idx] *= np.random.uniform(0.5, 1.5)
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            current_M = fast_conv(a).max()
            no_improve = 0
            eta = eta_init

    return best_a, best_M


def mirror_descent_dual(a_init, deadline, tau_init=0.005, tau_min=0.0005,
                       eta_init=1.0, cooling=0.9999, max_iters=50000):
    """
    Alternative: use top-K lags instead of full softmax for the gradient.
    More aggressive focus on the worst lags.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    num_lags = 2 * n - 1

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    eta = eta_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(it / (max_iters * 0.85), 1.0)
        tau = tau_init * (tau_min / tau_init) ** frac

        b = fast_conv(a)

        # Use top-K weighted gradient (sharper than full softmax)
        # Weight by exp(b_k / tau) but only for top K lags
        K = max(5, num_lags // 10)
        top_k_idx = np.argsort(b)[::-1][:K]
        b_top = b[top_k_idx]
        b_top_shifted = b_top - b_top.max()
        weights = np.exp(b_top_shifted / tau)
        weights = weights / weights.sum()

        # Build sparse p vector
        p = np.zeros(num_lags)
        p[top_k_idx] = weights

        # Gradient via cross-correlation
        g = np.correlate(p, a, mode='full')[:n]
        g *= 4.0 * n / num_lags

        # Update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        b_new = fast_conv(a_new)
        M_new = b_new.max()

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                eta *= 0.95
                if no_improve > 300:
                    a = best_a.copy()
                    a += np.random.normal(0, 0.01 / n, n)
                    a = np.maximum(a, 1e-6)
                    a = a / a.sum()
                    no_improve = 0
                    eta = eta_init * 0.5

        a = a_new
        eta *= cooling

    return best_a, best_M


def coordinate_descent(a_init, deadline, max_passes=20):

    Coordinate descent: for each coordinate, try candidate values
    that reduce the objective.
    """
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    b = fast_conv(a)
    S = a.sum()

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = np.random.permutation(n)
        improved = False

            if time.monotonic() > deadline:
                break

            # Generate candidates based on current value
            if ai < 1e-8:
                candidates = [0.005, 0.02, 0.05, 0.1, 0.2 / n]
            else:
                candidates = [0.0, ai * 0.2, ai * 0.5, ai * 0.8,
                              ai * 1.2, ai * 1.5, ai * 2.0, ai * 3.0]

            # Use current b for fast evaluation
            best_val = 2.0 * n * b.max() / (S * S)
            best_cand = ai

                if abs(cand - ai) < 1e-14:
                    continue
                delta = cand - ai
                S_new = S + delta
                if S_new < 0.01:

                else:
                    outside_right = 0.0
                outside_max = max(outside_left, outside_right)

                # Inside changes
                max_inside = 0.0
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j == i:
                        new_val = b[k] + 2.0 * delta * ai + delta * delta
                    else:
                        new_val = b[k] + 2.0 * delta * a[j]
                    if new_val > max_inside:
                        max_inside = new_val

                new_max = max(outside_max, max_inside)
                new_val_obj = 2.0 * n * new_max / (S_new * S_new)

                if new_val_obj < best_val - 1e-14:
                    best_val = new_val_obj
                    best_cand = cand

            if best_cand != ai:
                delta = best_cand - ai
                a[i] = best_cand
                S = S + delta
                # Update b incrementally
                k_lo = i
                k_hi = min(i + n - 1, 2 * n - 2)
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j == i:
                        b[k] += 2.0 * delta * ai + delta * delta
                    else:
                        b[k] += 2.0 * delta * a[j]
                improved = True

        # Renormalize
        a = np.maximum(a, 0)
        S = a.sum()
        if S > 0.01:
            a = a / S
            b = fast_conv(a)
            J = 2.0 * n * b.max() / (S * S)
            if J < best_J - 1e-14:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


defCoordinate_descent_global(a_init, deadline, max_passes=10):
    """Coordinate descent with full recomputation (slower but more robust)."""
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = np.random.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            if ai < 1e-8:
                candidates = [0.01, 0.03, 0.08, 0.15]
            else:
                candidates = [0.0, ai * 0.4, ai * 0.7, ai * 1.3, ai * 2.0]

            best_score = compute_J(a)
            best_cand = ai

            for cand in candidates:
                if abs(cand - ai) < 1e-14:
                    continue
                a[i] = cand
                a_clip = np.maximum(a, 0)
                s_chk = a_clip.sum()
                if s_chk < 0.01:
                    a[i] = ai
                    continue
                score = compute_J(a_clip / s_chk)
                if score < best_score:
                    best_score = score
                    best_cand = cand

            a[i] = best_cand
            if best_cand != ai:
                improved = True

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random
    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())

    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())

    # Bell
    bell = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())

    # Double bump
    dbl = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())

    # Frequency modulated
    for freq in [2, 3, 5, 8, 13]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    # High frequency
    for freq in [20, 50, 100]:
        hf = 1.0 + 0.15 * np.cos(2 * np.pi * freq * t)
        hf = np.maximum(hf, 0.05)
        inits.append(hf / hf.sum())

    # Edge concentration
    edge = 1.0 / (np.exp(-10.0 * (2 * t - 1) ** 2) + 0.1)
    inits.append(edge / edge.sum())

    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.05, 0.01)
    inits.append(sm / sm.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses smoothed mirror descent with multiple strategies.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256.0
        update_best(parent)

    # === PHASE 1: Smoothed mirror descent from parent (full softmax) ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Smoothed Mirror Descent (n={parent.size})", flush=True)
    print(f"{'=' * 60}", flush=True)

    p1_deadline = t_start + 350
    a = parent.copy() / parent.sum()
    a_result, M_result = mirror_descent(a, p1_deadline, tau_init=0.003,
                                         tau_min=0.0003, eta_init=2.0,
                                         cooling=0.99995, max_iters=200000)
    update_best(a_result)

    # CD polish
    a_cd = coordinate_descent(a_result, min(t_start + 420, deadline - 400), max_passes=10)
    update_best(a_cd)
    print(f"  Phase 1 done: best J={best_score:.8f}", flush=True)

    # === PHASE 2: Top-K mirror descent (sharper focus) ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Top-K Mirror Descent (n={best_seq.size if best_seq is not None else '?'})", flush=True)
    print(f"{'=' * 60}", flush=True)

    p2_deadline = t_start + 600
    if best_seq is not None:
        a = best_seq.copy() / best_seq.sum()
        a_result2, _ = mirror_descent_dual(a, p2_deadline, tau_init=0.003,
                                           tau_min=0.0003, eta_init=1.5,
                                           cooling=0.99995, max_iters=100000)
        update_best(a_result2)
        a_cd = coordinate_descent(a_result2, min(t_start + 650, deadline - 350), max_passes=8)
        update_best(a_cd)
        print(f"  Phase 2 done: best J={best_score:.8f}", flush=True)

    # === PHASE 3: Multi-length exploration ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Multi-length exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_deadline = t_start + 950
    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = sorted(set([
            int(n_base * 0.5), int(n_base * 0.75), int(n_base * 1.25),
            int(n_base * 1.5), int(n_base * 2.0),
            256, 384, 512, 768, 1024, 1536, 2048
        ]))
        target_lengths = [L for L in target_lengths if 128 <= L <= 2560 and L != n_base]

        for tgt_n in target_lengths:
            if time.monotonic() > p3_deadline:
                break

            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            # Quick optimization
            phase_dl = min(time.monotonic() + 45, p3_deadline)
            a_opt, _ = mirror_descent(a_ext, phase_dl, tau_init=0.003,
                                       tau_min=0.0003, eta_init=2.0,
                                       cooling=0.99995, max_iters=50000)
            update_best(a_opt)
            a_cd = coordinate_descent(a_opt, min(phase_dl + 10, p3_deadline), max_passes=4)
            update_best(a_cd)

        print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)

    # === PHASE 4: Diverse initializations ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Diverse initializations", flush=True)
    print(f"{'=' * 60}", flush=True)

    p4_deadline = t_start + 1200
    if best_seq is not None:
        n_best = best_seq.size
        inits = generate_inits(n_best, rng)
        # Add perturbed versions of best
        for sigma in [0.03, 0.08, 0.15]:
            pert = np.maximum(best_seq + rng.normal(0, sigma / n_best, n_best), 0.005)
            inits.append(pert / pert.sum())

        # Sort by initial quality
        init_scores = [(compute_J(init), init) for init in inits]
        init_scores.sort(key=lambda x: x[0])

        for score_init, init in init_scores[:5]:
            if time.monotonic() > p4_deadline:
                break
            phase_dl = min(time.monotonic() + 45, p4_deadline)
            a_opt, _ = mirror_descent(init, phase_dl, tau_init=0.002,
                                       tau_min=0.0002, eta_init=1.5,
                                       cooling=0.99997, max_iters=30000)
            a_cd = coordinate_descent(a_opt, min(phase_dl + 10, p4_deadline), max_passes=4)
            update_best(a_cd)

        # Also try top-K from best
        if time.monotonic() < p4_deadline:
            a = best_seq.copy() / best_seq.sum()
            phase_dl = min(time.monotonic() + 40, p4_deadline)
            a_opt, _ = mirror_descent_dual(a, phase_dl, tau_init=0.001,
                                           tau_min=0.0001, eta_init=1.0,
                                           cooling=0.99998, max_iters=30000)
            update_best(a_opt)

        print(f"  Phase 5 done: best J={best_score:.8f}", flush=True)

    # === PHASE 6: Final fine-tuning ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 6: Final fine-tuning", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        # Low-temperature mirror descent
        a_fine = best_seq.copy() / best_seq.sum()
        p6_deadline = min(t_start + 1400, deadline - 60)
        a_opt, _ = mirror_descent(a_fine, p6_deadline, tau_init=0.0008,
                                   tau_min=0.00005, eta_init=0.8,
                                   cooling=0.99999, max_iters=100000,
                                   restart_threshold=500)
        update_best(a_opt)

        # Final CD
        a_cd = coordinate_descent(a_opt, deadline, max_passes=20)
        update_best(a_cd)

        # One more CD on best
        if time.monotonic() < deadline:
            a_cd2 = coordinate_descent_global(best_seq, deadline, max_passes=5)
            update_best(a_cd2)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128.0
        update_best(best_seq)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
