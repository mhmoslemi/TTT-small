"""
Convex optimization approach to minimize J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses FISTA with Nesterov acceleration, LP mirror descent with restricted constraints,
and adaptive coordinate descent. Exploits the convexity of the problem (after normalization).

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, tol_frac=1e-10):
    """Compute subgradient of max_k (a*a)_k at point a.
    Returns (gradient, active_set, M)."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    if M <= 0:
        return np.zeros(n), np.array([0]), 0.0
    
    tol = tol_frac * M
    active = np.where(b >= M - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2 * a[k-hi : k-lo+1][::-1]
        # a[k-hi : k-lo+1] reversed gives a[k-lo], a[k-lo-1], ..., a[k-hi]

        # So grad[lo:hi+1] += 2 * a[k-hi : k-lo-1 : -1]
        idx = k - np.arange(hi, lo - 1, -1)
        grad[lo:hi + 1] += 2.0 * a[idx]
    
    grad /= len(active)
    return grad, active, M

    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    x = a.copy()
    y = a.copy()
    x_prev = a.copy()
    t_k = 1.0

    best_M = fast_conv(a).max()
    no_improve = 0

    # The gradient of (a*a)_k has norm <= 2*||a||_2
    # The Hessian (of each (a*a)_k) has spectral norm <= 2*||a||_1 = 2
    # So L = 2 is valid for the individual functions
    # For the max, we use L = 2 (conservative)
    L = 2.0
    eta = 1.0 / L
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute subgradient at y
        grad, active, M_y = compute_subgradient(y, tol_frac=1e-14)

        # Track best

def fista_adaptive(a, max_iters=100000, deadline=None, verbose=False):

def build_A_restricted(n, a, M, frac=0.15):
    """Build sparse constraint matrix using only active constraints.
    Includes constraints where (a*a)_k >= M*(1-frac)."""
    b = fast_conv(a)
    threshold = M * (1 - frac)
    active_ks = np.where(b >= threshold)[0]
    
    rows = []
    cols = []
    vals = []
    for k in active_ks:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    if not rows:
        return None, None, active_ks
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_ks), n))
    b_ub = b[active_ks].copy()  # Use current values as bounds (they're all <= M)
    # Actually we want b_ub = M for all active constraints
    b_ub = np.full(len(active_ks), M)
    
    return A, b_ub, active_ks


def lp_best_response(a, M, constrained_frac=0.2, time_limit=10):
    """Solve restricted LP: max sum(g) s.t. (a*g)[k] <= M for active k, g >= 0."""
    n = len(a)
    A, b_ub, active_ks = build_A_restricted(n, a, M, frac=constrained_frac)
    
    if A is None or A.shape[0] == 0:
        return None
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-10:
                # Verify all constraints are satisfied
                conv_check = fast_conv(g)
                if conv_check.max() <= M * 1.001:
                    return g
                else:
                    # Scale down to satisfy constraints
                    scale = M / conv_check.max()
                    g *= scale
                    if g.sum() > 1e-10:
                        return g
    except Exception:
        pass
    return None

def coord_descent_fast(a, max_sweeps=2000, deadline=None, seed=0, verbose=False):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            old = a[i]
            
            if old < 1e-12:
                candidates = [0.0, old * 10, old * 100, old * 1000]
                candidates = [c for c in candidates if c < 0.1]
            else:
                # Adaptive grid: focus on values near current
                candidates = [old * r for r in [0.0, 0.2, 0.5, 0.8, 1.25, 1.5, 2.0]]
                # Also try some absolute values
                candidates.append(0.0)
                # Remove values too close to old
                candidates = [c for c in candidates if abs(c - old) > 1e-10 * max(old, 1e-3)]
            
            if not candidates:
                continue
            
            best_val = old
            best_local_obj = 2.0 * n * conv.max() / (s * s)
            
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2
            
            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue
                
                # Apply
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                
                new_M = conv.max()
                new_s = s - old + x
                new_obj = 2.0 * n * new_M / (new_s * new_s) if new_s > 1e-10 else float('inf')
                
                if new_obj < best_local_obj:
                    best_local_obj = new_obj
                    best_val = x
                
                # Rollback
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
            
            if best_val != old and abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                sweep_improved = True
        
        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            break
        
        if verbose and sweep % 100 == 0:
            print(f"    CD sweep={sweep}, J={best_obj:.10f}", flush=True)
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, parent=None, rng=None):

    if rng is None:

        for sigma in [0.01, 0.05, 0.1]:

    inits.append(np.ones(n) / n)

    for center in [0.25, 0.5, 0.75]:

    for alpha in [0.5, 1.0, 2.0]:
        inits.append(np.maximum(t ** alpha + 0.01, 0.01) / n)

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, tol_frac=1e-10):
    """Compute subgradient of max_k (a*a)_k at point a.
    Returns (gradient vector, active_set array, M value)."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    if M <= 0:
        return np.zeros(n), np.array([0]), 0.0

    tol = tol_frac * M
    active = np.where(b >= M - tol)[0]

    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = k - np.arange(hi, lo - 1, -1)
        if len(idx) > 0 and np.all(idx >= 0) and np.all(idx < n):
            grad[lo:hi + 1] += 2.0 * a[idx]

    grad /= len(active)
    return grad, active, M


def fista(a, max_iters=100000, deadline=None, L=2.0, verbose=False):
    """FISTA (Nesterov accelerated proximal gradient) for min max_k(a*a)_k s.t. a>=0, sum(a)=1."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    x = a.copy()
    y = a.copy()
    x_prev = a.copy()
    t_k = 1.0
    eta = 1.0 / L

    best_x = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        grad, active, M_y = compute_subgradient(y, tol_frac=1e-14)

        z = y - eta * grad
        x_new = project_simplex(z)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        y = x_new + momentum * (x_new - x_prev)
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum < 1e-10:
            y = x_new.copy()
        elif y_sum > 2.0:
            y = y / y_sum

        t_k = t_new
        x_prev = x_new.copy()

        M_new = fast_conv(x_new).max()
        if M_new < best_M * (1.0 - 1e-13):

            best_x = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it > 0 and it % 20000 == 0:
            J_val = 2 * n * best_M
            print(f"      FISTA it={it}, M={best_M:.12f}, J={J_val:.10f}", flush=True)

        if no_improve > 80000 and it > 20000:
            break

    return best_x

def build_A_restricted(n, a, M, frac=0.2):
    """Build sparse constraint matrix using only near-active constraints."""
    b = fast_conv(a)
    threshold = M * (1.0 - frac)
    active_ks = np.where(b >= threshold)[0]

    if len(active_ks) == 0:
        return None, None

    rows = []
    cols = []
    vals = []
    for k in active_ks:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    if not rows:
        return None, None

    A = csr_matrix((vals, (rows, cols)), shape=(len(active_ks), n))
    b_ub = np.full(len(active_ks), M)
    return A, b_ub


def lp_best_response(a, M, constrained_frac=0.25, time_limit=5):
    """Solve restricted LP: max sum(g) s.t. (a*g)[k] <= M for active k, g >= 0."""
    n = len(a)
    A, b_ub = build_A_restricted(n, a, M, frac=constrained_frac)

    if A is None or A.shape[0] == 0:
        return None

    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-10:
                conv_check = fast_conv(g)
                max_check = conv_check.max()
                if max_check > M * 1.0001:
                    scale = M / max_check
                    g *= scale
                    if g.sum() > 1e-10:
                        return g
                else:
                    return g
    except Exception:
        pass
    return None
