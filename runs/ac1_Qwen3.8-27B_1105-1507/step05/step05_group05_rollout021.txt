"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Smooth gradient descent (Adam with adaptive tau) for global exploration
2. Exact coordinate descent with incremental convolution updates for refinement
3. Block coordinate descent for escaping local minima
4. Multi-scale optimization (try larger n by resampling)

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_gradient(a, tau):
    """Compute gradient of (1/tau)*logsumexp(tau*conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    return grad, b.max()

def run_adam_smooth(a, max_iters, deadline, tau_start=2.0, tau_end=100.0,
                    lr_start=0.05, lr_end=0.001, beta1=0.9, beta2=0.999):
    """Adam optimizer on smooth max objective with adaptive tau and lr."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        # Cosine schedule for tau
        tau = tau_start + (tau_end - tau_start) * (1 - np.cos(np.pi * frac)) / 2
        # Linear decrease for lr
        lr_t = lr_start + (lr_end - lr_start) * frac
        
        grad, _ = smooth_gradient(a, tau)
        
        # Project gradient to tangent space (sum = 0)
        grad_mean = grad.mean()
        grad = grad - grad_mean
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
    
    return best_a

def coordinate_update_incremental(a, i, conv_cached):
    """
    Find optimal value for a[i] using golden section search.
    Uses incremental convolution update.
    Returns optimal x >= 0 and the updated convolution.
    """
    n = len(a)
    x_cur = a[i]
    S = a.sum()
    
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = conv_cached[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0
    
    # Precompute for affected range
    affected_k = np.arange(k_lo, k_hi + 1)
    c_vals = np.zeros(len(affected_k))
    slopes = np.zeros(len(affected_k))
    quad_idx = -1
    
    for idx, k in enumerate(affected_k):
        if k == 2 * i:
            c_vals[idx] = conv_cached[k] - x_cur * x_cur
            slopes[idx] = 0.0
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = conv_cached[k] - 2 * x_cur * a[j]
                slopes[idx] = 2 * a[j]
            else:
                c_vals[idx] = conv_cached[k]
                slopes[idx] = 0.0
    
    def obj(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 0:
            return 1e10
        vals = c_vals + slopes * x_test
        m = base_max
        for idx2 in range(len(vals)):
            v = vals[idx2]
            if v > m:
                m = v
        if quad_idx >= 0:
            qv = c_vals[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)
    
    lo = 0.0
    hi = max(x_cur * 2.0 + 0.1, np.sqrt(max(conv_cached.max(), 0)) * 0.5, 0.05)
    
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)

def run_coordinate_descent_fast(a, max_sweeps, deadline, seed=0):
    """Coordinate descent with incremental convolution updates."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:

    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    order = np.arange(n)
    conv_cached = fast_conv(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        if sweep % 3 == 0:
            rng.shuffle(order)
        
        sweep_improved = False
        for i in order:
            x_opt = coordinate_update_incremental(a, i, conv_cached)
            if abs(x_opt - a[i]) > 1e-10:
                # Update a[i] and conv incrementally
                dx = x_opt - a[i]
                a[i] = x_opt
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        conv_cached[k] += dx * (a[i] + a[i] - dx)  # = (x_new^2 - x_old^2)
                    else:
                        j = k - i
                        if 0 <= j < n:
                            conv_cached[k] += 2 * dx * a[j]
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        conv_cached = conv_cached / (s * s)
        
        M = conv_cached.max()
        if M < best_M * (1 - 1e-8):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 3
        if no_improve > 20:
            break
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def diverse_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
    
    inits.append(np.ones(n))
    
    # Raised cosine
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * t)), 0.01))
    
    # Sine window
    inits.append(np.maximum(np.sin(np.pi * (t + 0.5/n)), 0.01))
    
    # Gaussian
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.1)
    
    # Linear ramp
    inits.append(t + 0.1)
    inits.append(1 - t + 0.1)
    
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Wedge
    inits.append(np.maximum(1 - 2*np.abs(t - 0.5), 0.01))
    
    return inits
