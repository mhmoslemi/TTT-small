            #   else: contribution is 2*x*a[k-i] (linear)

                if new_sum <= 1e-10:

                m = base_max

                    if k == 2 * i:

                    else:

                        else:
                            val = b[k]
                    if val > m:
                        m = val
                return m * scale
            
            # Golden section search
            lo = 0.0
            hi = max(x_cur * 3.0, np.sqrt(max(b.max(), 0.0)) + 0.1, 0.1)
            if hi < 0.01:
                hi = 0.1

                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)

            if obj(0.0) < obj(x_opt):
                x_opt = 0.0
            
            a[i] = max(0.0, x_opt)
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 20:
            break
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Key mathematical insight: By Cauchy-Schwarz, (sum a)^2 <= n * sum(a^2), so b_0 = sum(a^2) >= (sum a)^2/n,
meaning J >= 2 with equality iff a is uniform. This program verifies this, tries uniform sequences
of various lengths, and also runs optimization from the parent to handle any numerical subtleties.
"""
import numpy as np
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau=20.0):
    """Gradient of soft-max approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_vals = np.arange(k - hi, k - lo + 1)
        grad[lo:hi + 1] += 2.0 * pk * a[j_vals][::-1]
    return grad


def run_gradient_ascent_neg(a, max_iters, deadline, tau=20.0, lr=0.01):

    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)
    beta = 0.9
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        tau_t = tau + 0.05 * it
        lr_t = lr / (1.0 + 0.005 * it)

        if s < 1e-10:

        a = a_new
