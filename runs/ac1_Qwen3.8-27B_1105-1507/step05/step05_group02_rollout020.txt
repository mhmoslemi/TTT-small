import numpy as np

import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    s = a.sum()

        return float('inf')
    b = fast_conv(a)

def compute_grad_smooth(a, p):
    """Compute gradient of smooth max approximation w.r.t. a.
    p is the softmax distribution over convolution values."""
    n = len(a)
    grad = np.zeros(n)
    # grad_j = 2 * sum_{m} p[j+m] * a[m] - correction
    # Vectorized: for each j, dot p[j:j+valid_len] with a[:valid_len]
    for j in range(n):
        m_max = min(n, 2 * n - 1 - j)
        if m_max > 0:
            grad[j] = 2.0 * np.dot(p[j:j + m_max], a[:m_max])
        if 2 * j < 2 * n - 1:
            grad[j] -= p[2 * j] * a[j]
    return grad


def smooth_gd_nesterov(a, deadline, max_iters=3000, tau_start=5.0, tau_end=400.0,
                       lr_init=0.4, min_lr=0.01):
    """Smooth gradient descent with Nesterov acceleration and backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    y = a.copy()
    t = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = (it + 1) / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)

        b = fast_conv(y)
        b_max = b.max()
        shifted = tau * (b - b_max)
        shifted = np.clip(shifted, -50, 0)
        exp_b = np.exp(shifted)
        p = exp_b / exp_b.sum()

        grad = compute_grad_smooth(y, p)

        # Backtracking line search from current lr
        lr = lr_init * (min_lr / lr_init) ** frac
        best_trial_M = best_M
        best_trial = a.copy()

        for attempt in range(15):
            a_trial = np.maximum(y - lr * grad, 0)
            s = a_trial.sum()
            if s < 1e-10:
                lr *= 0.5
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_trial_M:
                best_trial_M = M_trial
                best_trial = a_trial.copy()
                break
            lr *= 0.5

        a_new = best_trial
        M_new = best_trial_M

        if M_new < best_M * (1 - 1e-10):
            best_M = M_new
            best_a = a_new.copy()

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / (t_new + 1.0)
        y = a_new + beta * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-10:
            y /= sy
        else:
            y = a_new.copy()
        a = a_new
        t = t_new

    return best_a

        M = float(b.max())

            if g is None:

                continue

        gs = g.sum()
        if gs < 1e-10:

        improved = False

def resample_seq(src, n):

    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_structural_inits(n, rng, parent=None):
    """Create diverse initial sequences with U-shape structure."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        # Perturbations of parent
        for scale in [0.005, 0.01, 0.02, 0.03]:
            inits.append(np.maximum(p + rng.normal(0, scale, n), 0.001))
        # Multiplicative perturbation
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.001))

    # Parametric U-shaped families
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        inits.append(np.maximum((t**alpha + (1 - t)**alpha), 0.01))
    # Asymmetric U-shapes
    for asym in [0.1, 0.2, 0.3, 0.5]:
        inits.append(np.maximum((t**0.5 + (1 - t + asym * (2 * t - 1))**0.5), 0.01))
    # Cosine-based
    for k in [1, 2, 3]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(np.pi * k * t), 0.01))
    # Exponential U-shapes
    for gamma in [2, 4, 6, 10]:
        inits.append(np.exp(-gamma * (t - 0.5) ** 2) * 0.3 + (t ** 2 + (1 - t) ** 2))
    # Random smooth
    k_half = max(3, n // 32)
    kern = np.ones(k_half) / k_half
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same'), 0.01))
    # Two-peak
    for x1, x2, w in [(0.2, 0.8, 0.1), (0.1, 0.9, 0.15), (0.15, 0.85, 0.08)]:
        inits.append(np.maximum(
            np.exp(-((t - x1) / w) ** 2) + np.exp(-((t - x2) / w) ** 2) + 0.1, 0.01))

    # Normalize all
    result = []
    for init in inits:
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 1e-10:
            result.append(init / s)
    return result


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def targeted_perturbation(a, rng, n_perturb=5, scale=0.02):
    """Perturb elements near the peaking positions of the convolution."""
    b = fast_conv(a)
    n = len(a)
    # Find peaking positions
    top_k = np.argsort(b)[-min(10, 2 * n - 1):][::-1]
    # For each peaking position k, the elements involved are a[max(0,k-n+1):min(n,k+1)]
    perturb_mask = np.zeros(n)
    for k in top_k[:5]:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        perturb_mask[j_min:j_max + 1] += 1.0
    perturb_mask = perturb_mask / perturb_mask.max()
    # Apply perturbation weighted by mask
    noise = rng.normal(0, scale, n) * perturb_mask
    a_new = np.maximum(a + noise, 0.001)
    a_new /= a_new.sum()
    return a_new


def coarse_to_fine(parent, target_n, deadline):
    """Optimize at coarse resolution, then refine."""
    n = target_n
    # Start at n//8 resolution
    coarse_n = max(16, n // 8)
    a_coarse = resample_seq(parent, coarse_n)
    a_coarse = np.maximum(a_coarse, 1e-6)
    a_coarse /= a_coarse.sum()

    # Optimize at coarse level
    dl_coarse = time.monotonic() + min(20, (deadline - time.monotonic()) * 0.2)
    a_coarse = smooth_gd_nesterov(a_coarse, dl_coarse, max_iters=500, tau_end=100)

    # Progressively increase resolution
    current_n = coarse_n
    while current_n < n:
        current_n = min(current_n * 2, n)
        a_current = resample_seq(a_coarse, current_n)
        a_current = np.maximum(a_current, 1e-6)
        a_current /= a_current.sum()
        dl_step = time.monotonic() + min(15, (deadline - time.monotonic()) * 0.3)
        a_current = smooth_gd_nesterov(a_current, dl_step, max_iters=300, tau_end=80)
        a_coarse = a_current

    return a_coarse


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0
    best = BestTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)
        print(f"Using default parent: n={parent.size}", flush=True)

    base_n = parent.size

    # === PHASE 1: Smooth GD at base length (40% of time) ===
    p1_budget = 0.35 * budget_s
    p1_deadline = time.monotonic() + p1_budget
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Smooth GD at n={base_n} ({p1_budget:.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    inits = make_structural_inits(base_n, rng, parent)
    best_p1 = parent.copy()
    best_M1 = fast_conv(parent).max()

    # Sort inits by initial quality
    init_scores = []
    for i, init in enumerate(inits[:12]):  # Limit to 12 inits
        M_init = fast_conv(init).max()
        init_scores.append((M_init, i, init))
    init_scores.sort()

    n_per_init = max(300, int(4000 * (256.0 / base_n)))
    for rank, (M_init, idx, init) in enumerate(init_scores[:8]):
        if time.monotonic() > p1_deadline:
            break
        allot = p1_budget / max(1, len(init_scores) - rank)
        dl = min(time.monotonic() + allot, p1_deadline)
        if time.monotonic() >= dl:
            continue

        a = smooth_gd_nesterov(init, dl, max_iters=n_per_init,
                                tau_start=5.0, tau_end=300.0,
                                lr_init=0.35, min_lr=0.01)
        M = fast_conv(a).max()
        if M < best_M1:
            best_M1 = M
            best_p1 = a.copy()
            J = compute_J(a)
            print(f"  init {rank}: M={M:.7f} J={J:.6f}", flush=True)

    best.update(best_p1)
    best.report("Phase 1")

    # === PHASE 2: Active-set LP polishing (25% of time) ===
    p2_budget = 0.25 * budget_s
    p2_deadline = time.monotonic() + p2_budget
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: Active-set LP ({p2_budget:.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        a = lp_mirror_descent(best.seq, p2_deadline, max_iters=300,
                             n_active=25, expand_threshold=12)
        best.update(a)

        # Second pass with different params
        if time.monotonic() < p2_deadline - 10:
            a2 = lp_mirror_descent(a, p2_deadline, max_iters=100,
                                  n_active=40, expand_threshold=20)
            best.update(a2)

    best.report("Phase 2")

    # === PHASE 3: Multi-length exploration (20% of time) ===
    p3_budget = 0.20 * budget_s
    p3_deadline = time.monotonic() + p3_budget
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Multi-length ({p3_budget:.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n_best = best.seq.size
        # Try nearby lengths
        lengths_to_try = []
        for delta in [-64, -32, 32, 64, 128, -128]:
            n_try = n_best + delta
            if 128 <= n_try <= 2048:
                lengths_to_try.append(n_try)
        # Also try some specific lengths
        for n_try in [384, 448, 512, 576, 640, 768]:
            if 128 <= n_try <= 2048 and n_try not in lengths_to_try:
                lengths_to_try.append(n_try)

        lengths_to_try = sorted(set(lengths_to_try))[:6]
        allot_per = p3_budget / max(1, len(lengths_to_try))

        for n_try in lengths_to_try:
            if time.monotonic() > p3_deadline:
                break
            dl = min(time.monotonic() + allot_per, p3_deadline)
            if time.monotonic() >= dl:
                continue

            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()

            # Quick smooth GD
            a_ext = smooth_gd_nesterov(a_ext, min(time.monotonic() + allot_per * 0.5, dl),
                                       max_iters=200, tau_start=5, tau_end=100,
                                       lr_init=0.25, min_lr=0.02)
            # Quick LP
            if time.monotonic() < dl - 2:
                a_ext = lp_mirror_descent(a_ext, dl, max_iters=30, n_active=20)

            best.update(a_ext)
            if best.seq.size == n_try:
                print(f"  n={n_try}: improved!", flush=True)

    best.report("Phase 3")

    # === PHASE 4: Targeted perturbations + re-optimization (15% of time) ===
    p4_budget = 0.15 * budget_s
    p4_deadline = time.monotonic() + p4_budget
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Targeted Perturbations ({p4_budget:.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        n_perturbs = 6
        allot_per = p4_budget / n_perturbs

        for i in range(n_perturbs):
            if time.monotonic() > p4_deadline:
                break
            dl = min(time.monotonic() + allot_per, p4_deadline)

            # Different perturbation strategies
            if i % 3 == 0:
                a_pert = targeted_perturbation(best.seq, rng, n_perturb=5, scale=0.015)
            elif i % 3 == 1:
                # Safer local perturbation
                a_pert = np.maximum(best.seq + rng.normal(0, 0.008, n), 0.001)
                a_pert /= a_pert.sum()
            else:
                # Shift mass from peak to trough
                b = fast_conv(best.seq)
                peak_k = np.argmax(b)
                # Elements contributing to peak
                j_min = max(0, peak_k - n + 1)
                j_max = min(n - 1, peak_k)
                a_pert = best.seq.copy()
                # Reduce peak contributors slightly, increase others
                mask = np.ones(n)
                mask[j_min:j_max + 1] = 0.95
                a_pert = a_pert * mask
                a_pert = np.maximum(a_pert, 0.001)
                a_pert /= a_pert.sum()

            # Quick optimization
            a_pert = smooth_gd_nesterov(a_pert, min(time.monotonic() + allot_per * 0.4, dl),
                                         max_iters=150, tau_start=10, tau_end=150,
                                         lr_init=0.15, min_lr=0.02)
            if time.monotonic() < dl - 1:
                a_pert = lp_mirror_descent(a_pert, dl, max_iters=20, n_active=20)

            best.update(a_pert)

    best.report("Phase 4")

    # === PHASE 5: Final intensive LP polish ===
    final_deadline = deadline
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        # Many LP iterations with growing active set
        a = lp_mirror_descent(best.seq, final_deadline, max_iters=200,
                             n_active=50, expand_threshold=10)
        best.update(a)

    # Final fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in best.seq]
