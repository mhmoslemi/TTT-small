"""
Optimized step function search using:
1. Fine-grid coordinate descent with incremental convolution updates
2. Nesterov-accelerated subgradient descent
3. LP mirror descent with cvxpy (cutting-plane for efficiency)
4. Multi-length exploration with smart time allocation
"""

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_M(a):
    """Compute subgradient of max_k conv(a,a)[k] at point a."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = 1e-10 * M + 1e-15
    active = np.where(b >= M - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[i] += a[k-i] for i in [lo, hi]
        # a[k-i] for i in [lo, hi] is a[k-hi : k-lo+1][::-1]
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    
    return grad, b, M


def nesterov_subgradient(a, max_iters, deadline, lr=0.1):
    """Nesterov accelerated subgradient method with simplex projection."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    
    a_current = a.copy()
    a_ave = a.copy()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        
        # Nesterov look-ahead
        alpha_t = 2.0 / (t + 1.5)  # or 2/(t+1)
        a_look = (1 - alpha_t) * a_current + alpha_t * a_ave
        
        # Subgradient
        grad, b, M = subgradient_M(a_look)
        
        # Update
        lr_t = lr / np.sqrt(t)
        a_current = project_simplex(a_look - lr_t * grad)
        
        # Running average
        a_ave = (a_current + (t - 1) * a_ave) / t
        
        # Track best
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a_look.copy()
        
        # Also check the average iterate
        if t % 100 == 0:
            M_ave = fast_conv(a_ave).max()
            if M_ave < best_M * (1 - 1e-10):
                best_M = M_ave
                best_a = a_ave.copy()
    
    return np.maximum(best_a, 0)

    for sweep in range(max_sweeps):

        sweep_improved = False

            if len(candidates) == 0:

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)
            
            lo = max(0, i)

            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i

                    conv[kq] += delta * delta

                    new_obj = float('inf')

                    new_M = conv.max()
                    new_obj = 2.0 * n * new_M / (new_s * new_s)

                    best_val = x
                
                conv[lo:hi + 1] -= 2.0 * delta * a_slice

                    conv[kq] -= delta * delta

                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                sweep_improved = True
        
        current_M = conv.max()
        current_obj = 2.0 * n * current_M / (s * s)

            no_improve = 0

            no_improve += 1

def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (A*g)[k] = conv(a,g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_mirror(a, M, max_cutting_iters=50, deadline=None):
    """LP mirror descent step using cutting plane method."""
    n = len(a)
    
    # Start with top constraints
    b = fast_conv(a)
    k_all = np.argsort(-b)[:min(200, 2 * n - 1)]
    
    active_k = list(k_all)
    
    for cp_iter in range(max_cutting_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Build LP with current active constraints
        A_active = np.array([[a[k - j] if 0 <= k - j < n else 0 for j in range(n)] 
                            for k in active_k])
        
        # LP: max sum(g) s.t. A_active * g <= M, g >= 0
        prob = cp.Problem(
            cp.Maximize(cp.sum(cp.Variable(n))),

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    out = np.maximum(np.interp(t_new, t_old, src), 0.0)
    return out


def project_simplex(v, s=1.0):
    """Project v onto simplex {x >= 0, sum(x) = s} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]
    if len(rho_candidates) == 0:
        return np.full(n, s / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Compute subgradient of max_k conv(a,a)[k] at point a. Returns (grad, b, M)."""

        hi = min(n - 1, k)
        segment = a[k - hi:k - lo + 1][::-1]

            c2 = lo_x + (hi_x - lo_x) / gr
            obj2 = eval_at(c2)
    best_x = c1 if obj1 < obj2 else c2
    return best_x


def build_A_matrix(n, a):
    """Build dense convolution matrix A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp_mirror_step(a, M, max_cp_iters=80):
    """Solve LP mirror step using cutting-plane: max sum(g) s.t. A*g <= M, g>=0."""
    n = len(a)
    b_check = np.zeros(2 * n - 1)
    conv_a = fast_conv(a)
    top_k = np.argsort(-conv_a)[:min(100, 2 * n - 1)]
    active_k = list(top_k)
    g = None
    for cp_iter in range(max_cp_iters):
        A_sub = np.array([[a[k - j] if 0 <= k - j < n else 0.0 for j in range(n)]
                         for k in active_k])
        c = -np.ones(n)
        b_ub = np.full(len(active_k), M)
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
            if not res.success or res.x is None:
                break
            g = np.maximum(res.x, 0.0)
        except Exception:
            break
        if g is None or g.sum() < 1e-12:
            break
        conv_ag = np.convolve(a, g)
        violated = np.where(conv_ag > M + 1e-8 * max(M, 1e-10))[0]
        if len(violated) == 0:
            break
        most_v = violated[np.argmax(conv_ag[violated])]
        if most_v not in active_k:
            active_k.append(most_v)
    if g is not None and g.sum() > 1e-12:
        return g / g.sum()
    return None


def lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b_conv = fast_conv(a)
        M = float(b_conv.max())
        t0 = time.monotonic()
        g = solve_lp_mirror_step(a, M, max_cp_iters=50)
        lp_time = time.monotonic() - t0
        if g is None:
            break
        improved = False
        base_t = max(0.05, 0.4 / (1.0 + 0.05 * it))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.03, 0.01]:
            a_trial = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 25:
                break
        remaining = deadline - time.monotonic()
        if remaining < max(0.5, lp_time * 3):
            break
    return best_a


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = [parent.copy()]
    t = np.linspace(0, 1, n, endpoint=False)
    for sigma in [0.03, 0.08, 0.15, 0.25, 0.4]:
        a_p = np.maximum(parent + rng.normal(0, sigma, n), 0)
        if a_p.sum() > 1e-10:
            inits.append(a_p / a_p.sum())
    for freq in [2, 3, 5, 8, 13]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            a_c = np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)), 0.005)
            inits.append(a_c / a_c.sum())
    for center in [0.2, 0.5, 0.8]:
        for width in [0.05, 0.1, 0.2]:
            a_g = np.maximum(np.exp(-((t - center) / width) ** 2) + 0.05, 0.005)
            inits.append(a_g / a_g.sum())
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        a_r = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        a_r = np.maximum(a_r + 0.02, 0.005)
        inits.append(a_r / a_r.sum())
    inits.append(np.ones(n) / n)
    inits.append(np.maximum(1.0 - 0.8 * np.abs(2 * t - 1), 0.01) / n)
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01) / n)
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: Fine coordinate descent (40% time)
    Phase 2: Nesterov subgradient (15% time)
    Phase 3: LP mirror descent + CD polish (30% time)
    Phase 4: Length exploration + final polish (15% time)
    """
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 12.0
    rng = np.random.default_rng(seed)

    best_a = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < best_score:
            best_a = a.copy()
            best_score = s
            elapsed = time.monotonic() - t_start
            print(f"  ** J={s:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0:
                parent = parent / parent.sum()
            try_update(parent)
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        try_update(parent)

    base_n = parent.size
    print(f"Base length: n={base_n}", flush=True)

    # Phase 1: Fine coordinate descent
    p1_frac = 0.40
    p1_deadline = min(t_start + p1_frac * total_budget, deadline - (1.0 - p1_frac) * total_budget + 10)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Fine CD n={base_n} (budget {p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    all_inits = generate_inits(base_n, rng, parent)
    max_inits = 8
    inits = all_inits[:max_inits]
    time_per_init = (p1_deadline - time.monotonic()) / max(1, len(inits))

    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_deadline:
            break
        init_deadline = min(time.monotonic() + time_per_init * 1.8, p1_deadline)
        t0 = time.monotonic()
        a_opt = coord_descent_fine(a0, max_sweeps=20000, deadline=init_deadline,
                                   seed=seed + idx * 7)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_opt)
        print(f"  CD[{idx}] J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
        try_update(a_opt)

    if best_a is not None and time.monotonic() < p1_deadline - 20:
        t0 = time.monotonic()
        a_cd = coord_descent_fine(best_a, max_sweeps=10000, deadline=p1_deadline,
                                  seed=seed + 500)
        print(f"  CD-refine J={compute_J(a_cd):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        try_update(a_cd)

    # Phase 2: Nesterov subgradient
    p2_deadline = min(time.monotonic() + 0.15 * total_budget, deadline - 0.40 * total_budget + 10)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Nesterov Subgradient (budget {p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None:
        a_nest = nesterov_subgradient(best_a, max_iters=50000, deadline=p2_deadline, lr_init=0.4)
        print(f"  Nesterov J={compute_J(a_nest):.8f}", flush=True)
        try_update(a_nest)
        if time.monotonic() < p2_deadline - 10:
            a_nest2 = nesterov_subgradient(a_nest, max_iters=20000, deadline=p2_deadline, lr_init=0.2)
            print(f"  Nesterov-2 J={compute_J(a_nest2):.8f}", flush=True)
            try_update(a_nest2)

    # Phase 3: LP mirror descent
    p3_deadline = min(time.monotonic() + 0.30 * total_budget, deadline - 0.10 * total_budget)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (budget {p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None:
        n_b = best_a.size
        a_lp = lp_mirror_descent(best_a, max_iters=200, deadline=p3_deadline)
        print(f"  LP J={compute_J(a_lp):.8f}", flush=True)
        try_update(a_lp)

        if time.monotonic() < p3_deadline - 15:
            a_cd_after = coord_descent_fine(a_lp, max_sweeps=8000, deadline=p3_deadline,
                                            seed=seed + 700)
            print(f"  CD-after-LP J={compute_J(a_cd_after):.8f}", flush=True)
            try_update(a_cd_after)

    # Phase 4: Length exploration
    p4_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length Exploration (budget {p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None and time.monotonic() < p4_deadline - 30:
        cur_n = best_a.size
        candidates_n = [l for l in [256, 320, 384, 448, 512, 640, 768, 896, 1024, 1280]
                       if l != cur_n and 128 <= l <= 2048]
        candidates_n.sort(key=lambda x: abs(x - cur_n))
        tried = 0
        for new_n in candidates_n:
            if time.monotonic() > p4_deadline - 25:
                break
            if tried >= 4:
                break
            a_ext = resample_seq(best_a, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            t0 = time.monotonic()
            alloc = min(80.0, (p4_deadline - time.monotonic()) / max(1, 4 - tried) - 10)
            a_opt = coord_descent_fine(a_ext, max_sweeps=5000,
                                       deadline=time.monotonic() + alloc,
                                       seed=seed + 800 + new_n)
            elapsed = time.monotonic() - t0
            J_val = compute_J(a_opt)
            print(f"  n={new_n}: J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
            try_update(a_opt)
            tried += 1

    # Final polish
    if best_a is not None and time.monotonic() < deadline - 5:
        a_fin = coord_descent_fine(best_a, max_sweeps=5000, deadline=deadline,
                                   seed=seed + 999)
        try_update(a_fin)

    if best_a is None:
        best_a = np.ones(128)
        best_score = float(evaluate_sequence(best_a.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_a]
