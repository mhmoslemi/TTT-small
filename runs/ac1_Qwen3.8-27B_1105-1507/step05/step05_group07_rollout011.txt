"""
Mirror descent with Nesterov acceleration for minimizing max(full_conv(a,a))/(sum(a))^2.
Key innovation: exploits the structure of the subgradient (sum of shifted a's) for
efficient computation, uses entropy-based mirror for natural non-negativity handling,
and Nesterov momentum for fast convergence. Combines with coordinate descent polishing
and multi-length exploration (512, 640, 768).
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_subgrad(a, active_set):
    """
    Compute subgradient of max_k conv(a,a)[k].
    grad[m] = (2/|S|) * sum_{k in S} a[k-m] (where a[j]=0 for j outside [0,n))
    """
    n = len(a)
    grad = np.zeros(n)
    inv_s = 2.0 / len(active_set)
    for k in active_set:
        # a[k-m] is valid when 0 <= k-m < n, i.e., k-n+1 <= m <= k
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        if m_min <= m_max:
            # a[k-m] for m = m_min..m_max: indices in a go from (k-m_min) down to (k-m_max)
            grad[m_min:m_max + 1] += a[k - m_max:k - m_min + 1][::-1] * inv_s
    return grad


def mirror_descent_nesterov(a, max_iters, deadline, tol_frac=0.01, eta0=0.5,
                            momentum0=0.3, verbose=True):
    """
    Mirror descent with negative entropy + Nesterov acceleration.
    Minimizes max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_J = 2.0 * n * best_M  # since sum=1

    # Nesterov state
    a_nest = a.copy()
    momentum = momentum0

    # Adaptive parameters
    eta = eta0
    no_improve = 0
    max_no_improve = 200
    tol = tol_frac

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Compute autocorrelation at nesterov point
        conv = fast_conv(a_nest)
        M = conv.max()

        # Find active set (within tolerance)
        threshold = M - tol * M
        active_set = list(np.where(conv >= threshold)[0])
        if len(active_set) == 0:
            active_set = [int(np.argmax(conv))]

        # Limit active set size for efficiency (keep top peaks)
        if len(active_set) > 40:
            conv_at_active = np.array([conv[k] for k in active_set])
            top_idx = np.argsort(conv_at_active)[-20:]
            active_set = [active_set[i] for i in top_idx]

        # Compute subgradient
        grad = compute_subgrad(a_nest, active_set)

        # Nesterov update: a_{t+1} ∝ a_nest * exp(-eta * grad)
        log_update = -eta * grad
        # Clip for numerical stability
        log_update = np.clip(log_update, -50, 50)
        a_new = a_nest * np.exp(log_update)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new /= s

        # Check improvement
        conv_new = fast_conv(a_new)
        M_new = conv_new.max()
        J_new = 2.0 * n * M_new

        if J_new < best_J - 1e-12:
            best_J = J_new
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Backtrack: reduce eta
            if no_improve > 20:
                eta *= 0.7
                no_improve = 0
                if eta < 1e-6:
                    break

        # Nesterov momentum update
        # a_nest_{t+1} = (a_{t+1} - (1-beta)*a_t) / beta
        beta = momentum + (1 - momentum) * (it / max(it, 10))**0.5
        beta = np.clip(beta, 0.1, 0.9)
        a_nest = (a_new - (1 - beta) * a) / beta
        a_nest = np.maximum(a_nest, 1e-12)
        a_nest /= a_nest.sum()

        a = a_new

        # Periodic report
        if verbose and (it % 5000 == 0 or it == 0):
            elapsed = time.monotonic()
            print(f"    MD iter {it}: J={J_new:.8f}, M={M_new:.8f}, eta={eta:.4f}, |S|={len(active_set)}", flush=True)

        if no_improve > max_no_improve:
            break

    return best_a, best_J


def coord_descent_fast(a, max_sweeps, deadline, random_order=False, seed=0):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    no_improve_sweeps = 0
    max_no_improve = 300

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order and sweep % 5 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        sweep_improved = False

        for i in order:
            old = a[i]
            if old < 1e-12:
                candidates = [0.0, 0.01, 0.05]
            else:
                candidates = [0.0, old * 0.25, old * 0.5, old * 0.75, old * 1.5, old * 2.0]

            best_val = old
            best_local = conv.max()

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            for x in candidates:
                if x < 0:
                    x = 0.0
                if abs(x - old) < 1e-14:
                    continue
                delta = x - old

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_M = conv.max()
                if new_M < best_local:
                    best_local = new_M
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-14:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                sweep_improved = True

        if s != 1.0 and s > 1e-10:
            a /= s
            s = 1.0
            conv = fast_conv(a)

        current_M = conv.max()
        current_obj = 2.0 * n * current_M
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

        if no_improve_sweeps > max_no_improve:
            break

    return best_a, best_obj


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a, max_iters, deadline):
    """LP mirror descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = float(conv.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        base_t = max(0.03, 0.3 * (1.0 - 0.3 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        remaining = deadline - time.monotonic()
        if remaining < 2:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 0:
            p /= ps
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        # Tapered versions
        taper = np.cos(np.pi * t / 2) ** 0.5
        inits.append(np.maximum(p * taper, 0.001))
        taper2 = 1.0 - 0.3 * np.sin(np.pi * t)
        inits.append(np.maximum(p * taper2, 0.001))

    inits.append(np.ones(n) / n)

    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01) / n)

    # Power-law shapes
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        shape = np.maximum((t + 0.01) ** alpha, 0.01)
        inits.append(shape / shape.sum())
        shape2 = np.maximum((1 - t + 0.01) ** alpha, 0.01)
        inits.append(shape2 / shape2.sum())

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.15, 0.25]:
            g = np.exp(-((t - center) / width) ** 2)
            inits.append((g + 0.1) / (g + 0.1).sum())

    return inits

    """
    Multi-phase search combining mirror descent with Nesterov acceleration,
    coordinate descent, and LP-based refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return False
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} ({label}) t={elapsed:.0f}s", flush=True)
            return True
        return False

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            ps = parent.sum()
            if ps > 0.01:
                try_update(parent, "parent")
                print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512
        try_update(parent, "uniform512")

    # === PHASE 1: Quick exploration with diverse inits + short CD (n=512) ===
    n_base = 512
    p1_deadline = min(time.monotonic() + 200, deadline - 900)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Quick exploration n={n_base} (budget {p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    inits = generate_inits(n_base, rng, parent)
    best_phase1 = parent.copy() if parent.size == n_base else resample_seq(parent, n_base)
    best_phase1_J = compute_J(best_phase1)

    for idx, a0 in enumerate(inits[:10]):
        if time.monotonic() > p1_deadline:
            break
        a0 = np.maximum(a0, 0)
        s0 = a0.sum()
        if s0 < 1e-10:
            continue
        a0 = a0 / s0

        # Short coordinate descent
        sw_budget = max(5, int((p1_deadline - time.monotonic()) / (len(inits) - idx) * 0.8))
        a_opt, J_opt = coord_descent_fast(a0, max_sweeps=sw_budget,
                                          deadline=p1_deadline, random_order=True,
                                          seed=seed + idx)
        try_update(a_opt, f"CD1-{idx}")
        if J_opt < best_phase1_J:
            best_phase1 = a_opt.copy()
            best_phase1_J = J_opt

    print(f"  Phase 1 best: J={best_phase1_J:.8f}", flush=True)

    # === PHASE 2: Mirror descent with Nesterov (main workhorse) ===
    p2_deadline = min(time.monotonic() + 500, deadline - 400)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Mirror Descent + Nesterov n={n_base} (budget {p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    # Multiple mirror descent runs with different parameters
    md_configs = [
        {'eta0': 0.3, 'momentum0': 0.3, 'tol_frac': 0.05},
        {'eta0': 0.1, 'momentum0': 0.5, 'tol_frac': 0.02},
        {'eta0': 0.5, 'momentum0': 0.2, 'tol_frac': 0.10},
        {'eta0': 0.2, 'momentum0': 0.4, 'tol_frac': 0.03},
    ]

    a_start = best_phase1.copy()
    a_start = np.maximum(a_start, 1e-10)
    a_start /= a_start.sum()

    for ci, cfg in enumerate(md_configs):
        if time.monotonic() > p2_deadline - 30:
            break
        print(f"  MD run {ci}: eta0={cfg['eta0']}, mom={cfg['momentum0']}, tol={cfg['tol_frac']}", flush=True)
        a_md, J_md = mirror_descent_nesterov(a_start, max_iters=50000,
                                             deadline=p2_deadline - 10,
                                             tol_frac=cfg['tol_frac'],
                                             eta0=cfg['eta0'],
                                             momentum0=cfg['momentum0'],
                                             verbose=(ci == 0))
        try_update(a_md, f"MD{ci}")
        if J_md < best_phase1_J:
            best_phase1 = a_md.copy()
            best_phase1_J = J_md

    # Polish with coordinate descent
    if time.monotonic() < p2_deadline - 20:
        a_cd, J_cd = coord_descent_fast(best_phase1, max_sweeps=5000,
                                        deadline=p2_deadline, random_order=True,
                                        seed=seed + 500)
        try_update(a_cd, "CD-polish-2")
        if J_cd < best_phase1_J:
            best_phase1 = a_cd.copy()
            best_phase1_J = J_cd

    print(f"  Phase 2 best: J={best_phase1_J:.8f}", flush=True)

    # === PHASE 3: Try longer lengths ===
    p3_deadline = min(time.monotonic() + 350, deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Longer lengths (budget {p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    target_lengths = [640, 768]
    for new_n in target_lengths:
        if time.monotonic() > p3_deadline - 60:
            break

        a_ext = resample_seq(best_phase1, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext /= a_ext.sum()

        # Mirror descent on extended
        a_md_ext, J_md_ext = mirror_descent_nesterov(a_ext, max_iters=30000,
                                                     deadline=p3_deadline - 30,
                                                     tol_frac=0.03,
                                                     eta0=0.2,
                                                     momentum0=0.4,
                                                     verbose=True)
        try_update(a_md_ext, f"MD-n{new_n}")

        # Quick CD polish
        if time.monotonic() < p3_deadline - 10:
            a_cd_ext, J_cd_ext = coord_descent_fast(a_md_ext, max_sweeps=3000,
                                                    deadline=p3_deadline,
                                                    random_order=True,
                                                    seed=seed + 1000 + new_n)
            try_update(a_cd_ext, f"CD-n{new_n}")

    print(f"  Phase 3 best: J={best_score:.8f} n={best_seq.size if best_seq is not None else 0}", flush=True)

    # === PHASE 4: LP refinement + final polish ===
    p4_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: LP + Final Polish (budget {p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None and time.monotonic() < p4_deadline - 20:
        # LP mirror descent on best
        n_best = best_seq.size
        if n_best <= 768:
            a_lp = run_lp_mirror(best_seq, max_iters=50, deadline=p4_deadline - 60)
            try_update(a_lp, "LP-refine")

        # Final mirror descent with tight tolerance
        if time.monotonic() < p4_deadline - 30:
            a_final, J_final = mirror_descent_nesterov(best_seq, max_iters=50000,
                                                       deadline=p4_deadline - 30,
                                                       tol_frac=0.005,
                                                       eta0=0.1,
                                                       momentum0=0.5,
                                                       verbose=True)
            try_update(a_final, "MD-final")

        # Final coordinate descent
        if time.monotonic() < p4_deadline - 5:
            a_fin_cd, J_fin_cd = coord_descent_fast(best_seq, max_sweeps=10000,
                                                    deadline=p4_deadline,
                                                    random_order=True,
                                                    seed=seed + 9999)
            try_update(a_fin_cd, "CD-final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
