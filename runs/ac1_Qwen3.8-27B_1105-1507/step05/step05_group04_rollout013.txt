"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L2-spectrum-flatness gradient descent (fast, smooth, FFT-based)
2. LP-based mirror descent for polishing
3. Multi-resolution refinement (optimize at smaller n, upscale, refine)
4. Diverse initialization with frequency-domain insights

Key insight: Minimizing the L2 deviation of the convolution from flat
is equivalent to minimizing sum|A(f)|^4 (PAPR in frequency domain),
which has a smooth gradient computable via FFT in O(n log n).
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def l2_gradient(a, mu):
    """
    Compute gradient of F(a) = sum_k (c_k - mu)^2 where c = conv(a,a).
    Uses FFT-based computation: grad = 4 * cross_corr(a_rev, c - mu)
    """
    n = len(a)
    c = fast_conv(a)
    d = c - mu  # length 2n-1
    a_rev = a[::-1]
    # grad_m = 4 * sum_{l=0}^{n-1} d_{m+l} * a_l
    # This is elements [n-1, 2n-3] of conv(a_rev, d)
    cross = fast_cross_conv(a_rev, d)
    grad = 4.0 * cross[n-1:2*n-2]  # length n
    return grad, c


def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def l2_gradient_descent(a, max_iters, deadline, lr_init=0.1, verbose=False):
    """
    L2 gradient descent on F(a) = sum(c_k - mu)^2 with simplex projection.
    Alternates with smooth-max gradient for the final phase.
    Returns best sequence found.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    mu = 1.0 / (2 * n - 1)  # mean of c when sum(a)=1
    
    best_a = a.copy()
    best_J = compute_J(a)
    lr = lr_init
    
    # Adam-like momentum
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t += 1
        
        grad, c = l2_gradient(a, mu)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_trial = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex
        a_trial = project_simplex(a_trial, 1.0)
        
        # Check improvement
        J_trial = compute_J(a_trial)
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            lr = min(lr * 1.05, 1.0)  # slightly increase lr
        else:
            # Backtrack
            a_trial = a - 0.5 * lr * m_hat / (np.sqrt(v_hat) + eps)
            a_trial = project_simplex(a_trial, 1.0)
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()
            lr *= 0.8
        
        a = best_a.copy()
        
        if verbose and it % 500 == 0:
            print(f"    L2-GD iter={it} J={best_J:.8f} lr={lr:.4f}", flush=True)
    
    return best_a


def l2_gradient_descent_momentum(a, max_iters, deadline, lr=0.05, verbose=False):
    """Simpler momentum-based L2 gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    mu = 1.0 / (2 * n - 1)
    
    best_a = a.copy()
    best_J = compute_J(a)
    vel = np.zeros(n)
    momentum = 0.9
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, c = l2_gradient(a, mu)
        
        # Project gradient onto tangent space of simplex
        grad = grad - np.mean(grad)
        
        vel = momentum * vel - lr * grad
        a_trial = a + vel
        a_trial = project_simplex(a_trial, 1.0)
        
        J_trial = compute_J(a_trial)
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            a = a_trial
        else:
            vel *= 0.5
            # Try smaller step
            a_trial = a - 0.5 * lr * grad
            a_trial = project_simplex(a_trial, 1.0)
            if compute_J(a_trial) < best_J:
                best_J = compute_J(a_trial)
                best_a = a_trial.copy()
                a = a_trial
            lr *= 0.95
        
        if verbose and it % 1000 == 0:
            print(f"    L2-Mom iter={it} J={best_J:.8f} lr={lr:.4f}", flush=True)
    
    return best_a
