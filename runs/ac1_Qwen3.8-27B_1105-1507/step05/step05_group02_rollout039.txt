"""

import numpy as np

def fast_conv(a):
    """Full convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax(w):
    """Numerically stable softmax."""
    w = w - w.max()
    e = np.exp(w)
    return e / e.sum()


def smooth_obj_and_grad_w(w, tau):
    """
    Objective and gradient with respect to w (logit) parameters.
    a = softmax(w), objective = (1/tau)*log(sum exp(tau * conv(a,a)))
    Gradient computed via chain rule through softmax.
    """
    a = softmax(w)
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # For numerical stability, shift by b_max
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    # g[m] = sum_k p[k] * a[k-m] (cross-correlation)
    # Compute via FFT
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    # Cross-correlation: g[m] = sum_k p[k] * a[k-m]
    # This is conv(p, flip(a)) shifted
    # Or: IFFT(conj(P) * A) gives sum_k p[k+m] * a[k] ... 
    # Let me think more carefully.
    # corr[m] = sum_k p[k] * a[k-m]
    # In terms of convolution: corr = conv(p, a_flipped) where a_flipped[i] = a[n-1-i]
    # Actually, let's use the direct formula for the gradient.
    
    # Direct computation: grad_a[m] = 2 * sum_k p[k] * a[k-m] for valid (k,m)
    # For each k, a[k-m] is valid when 0 <= k-m < n, i.e., k-n+1 <= m <= k
    # And m must be in [0, n-1]
    
    # Use FFT: g[m] = sum_k p[k] * a[k-m]
    # Let's define: extend a to length 2n-1 by zero padding (it's already length n)
    # g[m] for m in [0, n-1] = sum_{k=m}^{m+n-1} p[k] * a[k-m]
    # This is the cross-correlation of p and a at lag m.
    # Using FFT: corr = ifft(conj(fft(p, N)) * fft(a, N)) where N >= 2n-1
    # corr[m] = sum_k p[k] * a[k-m+n] ... hmm, the indexing is tricky.
    
    # Let me just do it the direct way for correctness, using the same approach as the previous code.
    grad_a = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # For m in [lo, hi]: a[k-m] is valid
        # grad_a[m] += 2 * p[k] * a[k-m]
        grad_a[lo:hi+1] += 2.0 * p[k] * a[k-hi:k-lo+1][::-1]
    
    # Now convert to grad_w via chain rule
    # da[di]/dw[j] = a[di] * (delta(i,j) - a[j])
    # df/dw[j] = sum_m grad_a[m] * a[m] * (delta(m,j) - a[j])
    # = grad_a[j] * a[j] - a[j] * sum_m grad_a[m] * a[m]
    # = a[j] * (grad_a[j] - dot(grad_a, a))
    
    dot_ga_a = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot_ga_a)
    
    # Objective value (normalized)
    obj = (1.0 / tau) * np.log(np.sum(exp_b)) + b_max
    
    return obj, grad_w


def run_lbfgs_softmax(w_init, deadline, tau_start=5.0, tau_end=500.0, max_iters=100):
    """L-BFGS optimization using softmax parametrization with annealed tau."""
    n = len(w_init)
    best_a = softmax(w_init)
    best_J = compute_J(best_a)
    
    # Multi-stage L-BFGS with increasing tau
    tau_schedule = np.geomspace(tau_start, tau_end, max_iters)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        tau = tau_schedule[it]
        
        def obj_grad(w):
            return smooth_obj_and_grad_w(w, tau)
        
        try:
            result = minimize(obj_grad, w_init, jac=True, method='L-BFGS-B',
                            options={'maxiter': 30, 'gtol': 1e-8})
            w_init = result.x
        except Exception:
            break
        
        a = softmax(w_init)
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    
    return best_a


def smooth_max_grad_direct(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*conv(a,a))) w.r.t. a.
    Returns (gradient, p, b, b_max) where p are softmax weights over b.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    
    return grad, p, b, b_max


def run_smooth_nesterov(a_init, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,
                        lr_init=0.4, lr_end=0.005, beta=0.9):
    """Smooth gradient descent with Nesterov acceleration on the simplex."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_max_grad_direct(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        # Gradient step with Nesterov momentum
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y = y / sy
        a = a_new
    
    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid (k,j)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):
    """Solve LP: maximize sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    # Build A matrix: A[k, j] = a[k-j] for valid j
    # We need: for each k, sum_j A[k,j] * b[j] <= M
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_alternating_lp(a_init, deadline, max_rounds=100):
    """
    Alternating LP: alternate between optimizing a given b, and b given a.
    Each subproblem: maximize sum(x) s.t. conv(other, x) <= M, x >= 0.
    Then rescale to sum = 1.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    b = a.copy()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Update b: maximize sum(b) s.t. conv(a, b) <= M_a, b >= 0
        M_a = fast_conv(a).max()
        b_new = lp_best_response(a, M_a, time_limit=5)
        if b_new is not None:
            b_new = b_new / b_new.sum()
            # Check if this improves
            M_check = fast_conv(b_new).max()
            if M_check < best_M:
                best_M = M_check
                best_a = b_new.copy()
            b = b_new
        
        # Update a: maximize sum(a) s.t. conv(b, a) <= M_b, a >= 0
        M_b = fast_conv(b).max()
        a_new = lp_best_response(b, M_b, time_limit=5)
        if a_new is not None:
            a_new = a_new / a_new.sum()
            M_check = fast_conv(a_new).max()
            if M_check < best_M:
                best_M = M_check
                best_a = a_new.copy()
            a = a_new
        
        # Also try averaging a and b
        avg = (a + b) / 2
        avg = avg / avg.sum()
        M_avg = fast_conv(avg).max()
        if M_avg < best_M:
            best_M = M_avg
            best_a = avg.copy()
    
    return best_a

def resample_seq(src, n):
    """Resample a sequence to length n using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.005))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Various parametric shapes
    for alpha in [0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        shape = (t**alpha + (1-t)**alpha) / 2
        inits.append(np.maximum(0.2 + 0.8 * shape, 0.01))
    
    # Cosine-based
    for f in [1, 2, 3]:
        for phase in [0, np.pi/2]:
            shape = 0.7 + 0.3 * np.cos(2*np.pi*f*t + phase)
            inits.append(np.maximum(shape, 0.01))
    
    # Gaussian-like
    inits.append(np.exp(-4*(t-0.5)**2) + 0.1)
    inits.append(np.exp(-8*(t-0.3)**2) + np.exp(-8*(t-0.7)**2) + 0.05)
    
    # "Bowl" shape (high at edges, low in middle)
    for c in [1, 2, 4, 8]:
        shape = 1 + c * (t - 0.5)**2
        inits.append(np.maximum(shape, 0.01))
    
    # Random smooth
    k = max(3, n // 30)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    
    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    
    # Fejer-like (triangular in frequency → sinc-like in time)
    inits.append(np.maximum(np.sinc((t - 0.5) * 4) + 0.5, 0.01))
    
    return inits

"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_fn(w):
    """Numerically stable softmax."""
    w = w - w.max()
    e = np.exp(w)
    return e / e.sum()


def smooth_obj_grad_w(w, tau):
    """
    Objective and gradient w.r.t. logit parameters w.
    a = softmax(w), obj = b_max + (1/tau)*log(sum exp(tau*(b - b_max)))
    """
    a = softmax_fn(w)
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    # grad_a[m] = 2 * sum_k p[k] * a[k-m] for valid (k,m)
    grad_a = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad_a[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]

    # Chain rule through softmax: df/dw[j] = a[j] * (grad_a[j] - <grad_a, a>)
    dot_ga_a = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot_ga_a)

    obj = b_max + (1.0 / tau) * np.log(np.sum(exp_b))
    return obj, grad_w


def run_lbfgs_softmax(w_init, deadline, tau_start=5.0, tau_end=500.0, max_rounds=150):
    """L-BFGS with softmax parametrization and annealed tau."""
    n = len(w_init)
    best_a = softmax_fn(w_init)
    best_J = compute_J(best_a)
    w = w_init.copy()

    tau_schedule = np.geomspace(tau_start, tau_end, max_rounds)

    for it in range(max_rounds):
        if time.monotonic() > deadline:
            break
        tau = tau_schedule[it]

        def obj_grad(w_local):
            return smooth_obj_grad_w(w_local, tau)

        try:
            result = minimize(obj_grad, w, jac=True, method='L-BFGS-B',
                              options={'maxiter': 20, 'gtol': 1e-7})
            w = result.x
        except Exception:
            break

        a = softmax_fn(w)
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()

    return best_a


def smooth_max_grad_direct(a, tau):
    """
    Gradient of (1/tau)*log(sum exp(tau*conv(a,a))) w.r.t. a.
    Returns (grad, p, b, b_max).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]

    return grad, p, b, b_max


def run_smooth_nesterov(a_init, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,

    y = a.copy()

        if time.monotonic() > deadline:
            break

        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        sy = y.sum()
        if sy < 1e-10:

def build_A_sparse(n, a):
    """Sparse matrix A where A[k,j] = a[k-j] for valid (k,j)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=8):
    """Solve LP: maximize sum(b) s.t. conv(a,b)(k) <= M for all k, b >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_alternating_lp(a_init, deadline, max_rounds=100):
    """
    Alternating LP: alternate between optimizing a given b, and b given a.
    Each subproblem is an LP. Both are rescaled to sum=1 after each step.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    b = a.copy()
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    stall = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Update b: max sum(b) s.t. conv(a,b) <= M_a, b >= 0
        M_a = float(fast_conv(a).max())
        b_new = lp_best_response(a, M_a, time_limit=5)
        if b_new is not None and b_new.sum() > 1e-10:
            b_new = b_new / b_new.sum()
            b = b_new

        # Update a: max sum(a) s.t. conv(b,a) <= M_b, a >= 0
        M_b = float(fast_conv(b).max())
        a_new = lp_best_response(b, M_b, time_limit=5)
        if a_new is not None and a_new.sum() > 1e-10:
            a_new = a_new / a_new.sum()
            a = a_new

        # Track best among a, b, and their mixture
        for candidate in [a, b, (a + b) / 2.0]:
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-10:
                continue
            candidate = candidate / cs
            M_c = float(fast_conv(candidate).max())
            if M_c < best_M:
                best_M = M_c
                best_a = candidate.copy()
                stall = 0
            else:
                stall += 1

        if stall > 4:
            stall = 0

    return best_a


def run_lp_mirror(a_init, deadline, max_iters=200):

    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        improved = False
        for t in [0.5, 0.35, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:

        else:

def resample_seq(src, n):
    """Resample sequence to length n via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    for c in [1, 2, 4, 8, 16]:
        shape = 1.0 + c * (t - 0.5)**2
        inits.append(np.maximum(shape, 0.01))

    # Random smooth
    k = max(3, n // 30)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))

    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)

    # Sinc-like
    inits.append(np.maximum(np.sinc((t - 0.5) * 3.0) + 0.5, 0.01))

    return inits


class BestTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def length_priority(n, base_n):
    """Priority score for a length (lower = more important)."""
    proximity = abs(n - base_n) / max(base_n, 1)
    efficiency = 1.0 / np.sqrt(n)
    return proximity + 0.3 * efficiency


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Searches for the best sequence minimizing
    J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    base_n = parent.size

    # === PHASE 1: Multi-length smooth gradient descent ===
    p1_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Smooth Gradient Descent", flush=True)
    print(f"{'='*50}", flush=True)

    p1_budget = 320.0
    lengths = set()
    for mult in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 2.5, 3.0, 4.0]:
        n_try = int(base_n * mult)
        n_try = max(128, min(4096, (n_try // 16) * 16))
        lengths.add(n_try)
    for n_try in [192, 256, 320, 384, 448, 512, 576, 640, 768, 896, 1024, 1280, 1536, 2048, 2560, 3072]:
        if 128 <= n_try <= 4096:
            lengths.add(n_try)

    # Sort by priority
    lengths = sorted(lengths, key=lambda x: length_priority(x, base_n))
    print(f"Lengths ({len(lengths)}): {lengths[:10]}...", flush=True)

    for n in lengths:
        if time.monotonic() > deadline - 350:
            break
        if time.monotonic() > p1_start + p1_budget + 50:
            break

        # Time allocation based on length
        allot = min(50.0, max(10.0, 280.0 * (256.0 / n) ** 0.6))
        dl = min(time.monotonic() + allot, deadline - 300, p1_start + p1_budget + 50)
        if time.monotonic() >= dl:
            continue

        t_n = time.monotonic()
        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n_seq = None
        best_n_M = float('inf')

        n_max_iters = max(80, int(4000 * (256.0 / n)))

        for a0 in inits:
            if time.monotonic() > dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()
            a = run_smooth_nesterov(a0, dl, max_iters=n_max_iters,
                                    tau_start=1.0, tau_end=200.0,
                                    lr_init=0.35, lr_end=0.003)
            M = float(fast_conv(a).max())
            if M < best_n_M:
                best_n_M = M
                best_n_seq = a.copy()

        if best_n_seq is not None:
            J = compute_J(best_n_seq)
            print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-t_n:.1f}s)", flush=True)
            best.update(best_n_seq)

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: L-BFGS with softmax parametrization ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: L-BFGS Softmax", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        a_norm = best.seq / best.seq.sum()
        w_init = np.log(a_norm + 1e-12)

        lbfgs_budget = min(120.0, max(30.0, (deadline - 200) - time.monotonic()))
        dl = time.monotonic() + lbfgs_budget
        print(f"  n={n}, budget={lbfgs_budget:.0f}s", flush=True)

        a_lbfgs = run_lbfgs_softmax(w_init, dl, tau_start=5.0, tau_end=400.0, max_rounds=200)
        best.update(a_lbfgs)

        # Try from a few other inits
        inits2 = make_inits(n, rng, best.seq)
        for a0 in inits2[1:4]:
            if time.monotonic() > dl - 10:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()
            w0 = np.log(a0 + 1e-12)
            a_l = run_lbfgs_softmax(w0, min(time.monotonic() + 30, dl),
                                    tau_start=3.0, tau_end=300.0, max_rounds=60)
            best.update(a_l)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Alternating LP ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Alternating LP", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        est_lp_ms = 5.0 * (n / 128.0) ** 1.5
        lp_budget = min(130.0, max(20.0, (deadline - 120) - time.monotonic()))
        max_rounds = max(10, int(lp_budget / max(0.02, 2 * est_lp_ms / 1000.0)))
        print(f"  n={n}, LP budget={lp_budget:.0f}s, rounds={max_rounds}", flush=True)

        a_alt = run_alternating_lp(best.seq, min(time.monotonic() + lp_budget, deadline - 100),
                                   max_rounds=max_rounds)
        best.update(a_alt)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: LP Mirror Descent ===
    p4_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: LP Mirror Descent", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        est_lp_ms = 5.0 * (n / 128.0) ** 1.5
        lp_budget = min(100.0, max(20.0, (deadline - 60) - time.monotonic()))
        max_iters = max(10, int(lp_budget / max(0.02, est_lp_ms / 1000.0)))
        print(f"  n={n}, budget={lp_budget:.0f}s, iters={max_iters}", flush=True)

        a_mirror = run_lp_mirror(best.seq, min(time.monotonic() + lp_budget, deadline - 50),
                                 max_iters=max_iters)
        best.update(a_mirror)

        # Second pass
        if time.monotonic() < deadline - 25:
            a_mirror2 = run_lp_mirror(a_mirror, deadline - 25, max_iters=40)
            best.update(a_mirror2)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: High-tau fine-tuning ===
    p5_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Fine-tuning", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 20:
        # Very high tau smooth gradient
        a_fine = run_smooth_nesterov(best.seq, deadline - 15, max_iters=2000,
                                     tau_start=50.0, tau_end=500.0,
                                     lr_init=0.05, lr_end=0.001, beta=0.5)
        best.update(a_fine)

        # L-BFGS at very high tau
        if time.monotonic() < deadline - 15:
            n = best.seq.size
            a_norm = best.seq / best.seq.sum()
            w_init = np.log(a_norm + 1e-12)
            a_lbf = run_lbfgs_softmax(w_init, deadline - 5, tau_start=100.0,
                                      tau_end=1000.0, max_rounds=40)
            best.update(a_lbf)

    best.report("Phase 5")
    print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === PHASE 6: Subgradient averaging on best ===
    if best.seq is not None and time.monotonic() < deadline - 15:
        a_sg = run_subgradient_avg(best.seq, min(time.monotonic() + 30, deadline - 10),
                                   max_iters=1500, lr_const=0.2)
        best.update(a_sg)

    # === PHASE 7: Multi-start perturbations ===
    p7_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 7: Perturbations", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 15:
        n = best.seq.size
        n_perturb = 0
        max_perturb = 25
        while time.monotonic() < deadline - 12 and n_perturb < max_perturb:
            n_perturb += 1
            scale = rng.uniform(0.002, 0.012)
            # Mix Gaussian and low-frequency noise
            if rng.random() < 0.5:
                noise = rng.normal(0, scale, n)
            else:
                t_arr = np.linspace(0, 1, n)
                f1 = rng.integers(1, 5)
                f2 = rng.integers(1, 8)
                ph1 = rng.uniform(0, 2 * np.pi)
                ph2 = rng.uniform(0, 2 * np.pi)
                noise = scale * (np.cos(2 * np.pi * f1 * t_arr + ph1) +
                                 0.5 * np.cos(2 * np.pi * f2 * t_arr + ph2))
            a = best.seq + noise
            a = np.maximum(a, 1e-4)
            a = a / a.sum()

            a_opt = run_smooth_nesterov(a, min(time.monotonic() + 15, deadline - 10),
                                        max_iters=250, tau_start=20.0, tau_end=200.0,
                                        lr_init=0.1, lr_end=0.01)
            best.update(a_opt)

            if time.monotonic() < deadline - 8:
                a_lp = run_lp_mirror(a_opt, deadline - 5, max_iters=12)
                best.update(a_lp)

    best.report("Phase 7")
    print(f"Phase 7 time: {time.monotonic()-p7_start:.1f}s", flush=True)

    # === PHASE 8: Try nearby lengths ===
    if best.seq is not None and time.monotonic() < deadline - 12:
        n_best = best.seq.size
        for delta in [-64, -32, -16, 16, 32, 64, 128]:
            if time.monotonic() > deadline - 10:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 4096:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext = run_smooth_nesterov(a_ext, min(time.monotonic() + 12, deadline - 5),
                                        max_iters=200, tau_start=10, tau_end=150,
                                        lr_init=0.15, lr_end=0.02)
            best.update(a_ext)

    # Final LP polish if time
    if best.seq is not None and time.monotonic() < deadline - 3:
        a_final = run_lp_mirror(best.seq, deadline, max_iters=30)
        best.update(a_final)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in best.seq]
