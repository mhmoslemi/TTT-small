"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based gradient descent with soft-max, peak-directed coordinate
descent, spectral domain optimization, and LP polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Efficient linear convolution using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_wgrad(a, alpha):
    """
    Compute gradient of smooth objective f(a) = 2n*M_soft/S^2
    where M_soft = soft_max_alpha(conv(a,a)).
    Returns (grad, M_soft, conv).
    """
    n = len(a)
    c = compute_conv(a)
    S = float(np.sum(a))
    if S < 1e-15:
        return np.zeros(n), 0.0, c

    # Soft max weights (numerically stable)
    c_max = float(np.max(c))
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / float(np.sum(exp_c))
    M_soft = float(np.dot(w, c))

    # dM_soft/da[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # = 2 * cross_correlation(w, a)[i]
    # cross[i] = sum_j w[i+j]*a[j] = conv(w_rev, a)[2n-2-i]
    w_rev = w[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fw = rfft(w_rev, size)
    fa = rfft(a, size)
    corr = irfft(fw * fa, size)
    cross = corr[n - 1:2 * n - 1][::-1]

    dM_da = 2.0 * cross

    # f = 2n * M_soft / S^2
    # df/da[i] = 2n * dM_da[i] / S^2 - 4n * M_soft / S^3
    grad = (2.0 * n * dM_da / (S * S)) - (4.0 * n * M_soft / (S ** 3))

    return grad, M_soft, c


def normalize_nonneg(a):
    """Project to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def adam_optimize(a, max_iters=50000, time_budget=300.0, alpha_start=50.0,
                  alpha_end=500.0, lr_start=0.05, lr_end=0.001, seed=42):
    """
    Adam gradient descent with adaptive soft-max temperature.
    """
    n = len(a)
    a = normalize_nonneg(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    patience = 500
    stall = 0

    while t < max_iters:
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        # Adaptive parameters
        frac = min(1.0, t / max(1, max_iters * 0.5))
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lr = lr_start + (lr_end - lr_start) * frac
        if t > max_iters * 0.8:
            lr = lr_end * 0.5

        grad, M_soft, c = compute_wgrad(a, alpha)

        # Adam update
        t += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad ** 2
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = normalize_nonneg(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
            if stall > patience:
                # Perturbation to escape local minimum
                scale = rng.uniform(0.01, 0.05)
                a = a * np.exp(rng.normal(0, scale, n))
                a = normalize_nonneg(a)
                obj_check = compute_obj(a)
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a.copy()
                stall = 0
                patience = max(100, patience // 2)
                lr = max(lr * 2, lr_start)

        if t % 5000 == 0:
            print(f"    Adam it={t}, obj={best_obj:.6f}, alpha={alpha:.0f}, lr={lr:.5f}")

    return best_a, best_obj


def peak_coordinate_descent(a, time_budget=100.0, max_rounds=20000, seed=123):
    """
    Coordinate descent focusing on peak positions in the convolution.
    For each peak, try adjusting the contributing coordinates.
    """
    n = len(a)
    a = normalize_nonneg(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    round_count = 0

    while time.monotonic() - start < time_budget:
        round_count += 1
        c = compute_conv(a)
        max_b = float(np.max(c))
        S = float(np.sum(a))
        obj = 2.0 * n * max_b / (S * S)

        # Find peak positions
        peak_threshold = max_b * 0.95
        peak_indices = np.where(c >= peak_threshold)[0]

        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(c)])

        # Pick a random peak
        k = int(peak_indices[rng.integers(0, len(peak_indices))])

        # Find contributing indices: a[i]*a[k-i] where both valid
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)

        # Try adjusting a random contributing coordinate
        if i_max < i_min:
            continue
        i = int(rng.integers(i_min, i_max + 1))
        j = k - i

        # Try different multipliers for a[i]
        best_local_obj = obj
        best_factor = 1.0
        for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
            a_try = a.copy()
            a_try[i] = a[i] * factor
            a_try = normalize_nonneg(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_factor = factor

        if best_factor != 1.0:
            a[best_factor and int(rng.integers(i_min, i_max + 1))]  # dummy
            a[i] = a[i] * best_factor
            a = normalize_nonneg(a)
            obj_new = compute_obj(a)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()

    return best_a, best_obj


def spectral_lmp(a, time_budget=100.0, max_iters=5000, seed=777):
    """
    Local Minimization based PAPR reduction (LMP) adapted for our problem.
    Iteratively adjusts the phase of the spectrum to reduce time-domain peaks,
    then projects to non-negativity.
    """
    n = len(a)
    a = normalize_nonneg(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    # Use FFT size that's a power of 2 >= 2n-1
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute FFT of zero-padded a
        a_pad = np.zeros(fft_size)
        a_pad[:n] = a
        A = rfft(a_pad)

        # Compute time-domain signal (the convolution)
        t_signal = irfft(A * A, fft_size)[:2 * n - 1]
        peak_val = float(np.max(t_signal))
        peak_idx = int(np.argmax(t_signal))

        # Try to reduce the peak by adjusting the spectrum
        # At the peak location, the contribution from frequency k is |A_k|^2 * exp(2j*k*peak_idx/N)
        # We want to change the phase of A to reduce the peak
        # Simple approach: for each frequency bin, try rotating the phase to reduce the peak

        # Use a subset of frequency bins for efficiency
        num_bins = min(fft_size // 2, 200)
        bins = rng.choice(fft_size // 2, size=min(num_bins, fft_size // 2), replace=False)

        improved = False
        for b in bins:
            if time.monotonic() - start > time_budget:
                break
            # Try rotating phase of A[b] (and A[fft_size-b] for conjugate)
            A_orig_b = A[b].copy()
            phase_orig = np.angle(A_orig_b)
            mag = np.abs(A_orig_b)

            # The contribution to t_signal[peak_idx] from this bin pair:
            # 2 * |A_b|^2 * cos(2*b*peak_idx/fft_size + 2*phase_orig) + (DC and Nyquist special)
            # We want to minimize the total, so set phase to make this term negative
            target_phase = np.pi - 2.0 * b * peak_idx / fft_size
            # Try the target phase
            A[b] = mag * np.exp(1j * target_phase)
            if b != 0 and b != fft_size // 2:
                A[fft_size - b] = np.conj(A[b])

            t_try = irfft(A * A, fft_size)[:2 * n - 1]
            peak_try = float(np.max(t_try))
            a_try = t_try  # dummy, we need to get a from A
            # Actually we need a from A, not t_signal
            a_try = irfft(A, fft_size)[:n]
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                A[b] = A_orig_b
                if b != 0 and b != fft_size // 2:
                    A[fft_size - b] = np.conj(A_orig_b)
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)

            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                # Recompute A from new a
                a_pad[:n] = a
                A = rfft(a_pad)
                improved = True
                break
            else:
                A[b] = A_orig_b
                if b != 0 and b != fft_size // 2:
                    A[fft_size - b] = np.conj(A_orig_b)

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix for the LP."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_direction(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        A = build_toeplitz(a)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0


def lp_polish(a, time_budget=200.0, seed=42):
    """LP-based polishing with adaptive relaxation and line search."""
    n = len(a)
    a = normalize_nonneg(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - start < time_budget - 2:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(best_a)
        M = float(np.max(conv))
        S = float(np.sum(best_a))

        # Adaptive relaxation
        relax = 1.0 + 0.03 * (stall // 10)
        relax = min(relax, 1.3)

        g, S_g = solve_lp_direction(best_a, M, relax=relax)

        if g is not None and S_g > 1.0 + 1e-8:
            # Line search with golden section
            best_t = 0.0
            best_t_obj = best_obj
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
                a_try = (1.0 - t) * best_a + t * g / S_g
                a_try = normalize_nonneg(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_t = t

            if best_t > 0 and best_t_obj < best_obj - 1e-12:
                a_try = (1.0 - best_t) * best_a + best_t * g / S_g
                best_a = normalize_nonneg(a_try)
                best_obj = best_t_obj
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        # Perturbation when stalled
        if stall > 20:
            scale = rng.uniform(0.02, 0.06)
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = normalize_nonneg(a_pert)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert
            stall = 0

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def resample_sequence(a, new_n):
    """Resample sequence to new length via cubic interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, kind='cubic')
    return np.maximum(a_new, 0)


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)

    if choice == 0:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * k / n))
        a = a + 0.1 * rng.random(n)
    elif choice == 1:
        # Blackman-like window
        a = 0.42 - 0.5 * np.cos(2 * np.pi * k / n) + 0.08 * np.cos(4 * np.pi * k / n)
        a = np.maximum(a, 0) + 0.05
    elif choice == 2:
        # Gaussian bump
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.35)
        a = np.exp(-0.5 * ((k - center) / width) ** 2) + 0.05
    elif choice == 3:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.5, 4.0)
            a += rng.uniform(0.1, 0.3) * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        # Block structure
        nblocks = rng.integers(3, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.8)
    else:
        # Smoothed random
        a = rng.random(n)
        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
        a = a + 0.1

    a = np.maximum(a, 0.01)
    return normalize_nonneg(a)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization combining gradient descent,
    coordinate descent, spectral methods, and LP polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Initial best: {best_score:.6f} (n={len(best_seq)})")

    # Phase 1: Adam gradient descent on parent (if available)
    if best_seq is not None and time.monotonic() < deadline - 300:
        n_p = len(best_seq)
        alloc = min(250.0, (deadline - time.monotonic()) * 0.2)
        print(f"\n=== Phase 1: Adam GD (n={n_p}, budget={alloc:.0f}s) ===")
        a_opt, obj_opt = adam_optimize(best_seq, max_iters=80000,
                                       time_budget=alloc, alpha_start=30,
                                       alpha_end=300, lr_start=0.03, lr_end=0.001,
                                       seed=seed)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  After Adam: {best_score:.6f}")
        else:
            print(f"  After Adam: {obj_opt:.6f} (no improvement)")

    # Phase 2: Peak-directed coordinate descent
    if best_seq is not None and time.monotonic() < deadline - 250:
        alloc = min(100.0, (deadline - time.monotonic()) * 0.15)
        print(f"\n=== Phase 2: Peak Coord Descent (budget={alloc:.0f}s) ===")
        a_opt, obj_opt = peak_coordinate_descent(best_seq, time_budget=alloc, seed=seed + 1)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  After peak CD: {best_score:.6f}")
        else:
            print(f"  After peak CD: {obj_opt:.6f}")

    # Phase 3: Spectral LMP
    if best_seq is not None and time.monotonic() < deadline - 200:
        alloc = min(80.0, (deadline - time.monotonic()) * 0.1)
        print(f"\n=== Phase 3: Spectral LMP (budget={alloc:.0f}s) ===")
        a_opt, obj_opt = spectral_lmp(best_seq, time_budget=alloc, max_iters=3000, seed=seed + 2)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  After spectral: {best_score:.6f}")
        else:
            print(f"  After spectral: {obj_opt:.6f}")

    # Phase 4: LP Polish
    if best_seq is not None and time.monotonic() < deadline - 100:
        alloc = min(250.0, (deadline - time.monotonic()) * 0.25)
        print(f"\n=== Phase 4: LP Polish (budget={alloc:.0f}s) ===")
        a_opt, obj_opt = lp_polish(best_seq, time_budget=alloc, seed=seed + 3)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  After LP: {best_score:.6f}")
        else:
            print(f"  After LP: {obj_opt:.6f}")

    # Phase 5: Multi-n exploration
    print(f"\n=== Phase 5: Multi-n Exploration ===")
    if best_seq is not None:
        n_base = len(best_seq)
        n_candidates = sorted(set([max(200, n_base - 100), n_base,
                                    n_base + 100, n_base + 200, n_base + 400]))
    else:
        n_candidates = [400, 600, 700, 800, 1000]

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.2, 120.0)
        alloc = max(15.0, alloc)

        print(f"  n={n_try}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        if best_seq is not None and len(best_seq) != n_try:
            a_start = resample_sequence(best_seq, n_try)
        elif best_seq is not None:
            a_start = best_seq.copy()
        else:
            a_start = random_init(n_try, seed + n_try)

        # Quick Adam
        a_cur, obj_cur = adam_optimize(a_start, max_iters=30000,
                                       time_budget=alloc * 0.5,
                                       alpha_start=50, alpha_end=200,
                                       lr_start=0.02, lr_end=0.005,
                                       seed=seed + n_try)
        # Quick LP
        if time.monotonic() < deadline - 30:
            a_cur, obj_cur = lp_polish(a_cur, time_budget=min(alloc * 0.3, 40.0),
                                       seed=seed + n_try + 1)

        print(f"  n={n_try}: obj={obj_cur:.6f}")
        if obj_cur < best_score:
            best_score = obj_cur
            best_seq = a_cur.copy()
            print(f"  ** UPDATE: n={n_try}, obj={best_score:.6f} **")

    # Phase 6: Final intensive polish
    print(f"\n=== Phase 6: Final Polish ===")
    print(f"  Best before: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  Polishing n={n_best}, {remaining:.0f}s remaining")

        # Round 1: Adam with lower temperature
        alloc1 = min(remaining * 0.3, 150.0)
        a_cur, obj1 = adam_optimize(best_seq, max_iters=50000,
                                    time_budget=alloc1, alpha_start=100,
                                    alpha_end=800, lr_start=0.01, lr_end=0.0005,
                                    seed=seed + 100)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a_cur.copy()
            print(f"  Final Adam: {best_score:.6f}")

        # Round 2: LP
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.4, 150.0)
            a_cur, obj2 = lp_polish(best_seq, time_budget=alloc2, seed=seed + 200)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_cur.copy()
                print(f"  Final LP: {best_score:.6f}")

        # Round 3: Peak CD + LP
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            a_cur, obj3 = peak_coordinate_descent(best_seq,
                                                  time_budget=min(remaining * 0.2, 30.0),
                                                  seed=seed + 300)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a_cur.copy()
            a_cur, obj3b = lp_polish(best_seq, time_budget=min(remaining * 0.3, 60.0),
                                     seed=seed + 301)
            if obj3b < best_score:
                best_score = obj3b
                best_seq = a_cur.copy()
                print(f"  Final CD+LP: {best_score:.6f}")

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(500)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
