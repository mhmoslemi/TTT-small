def lp_mirror_descent(a, deadline, max_iters=50000):
    """LP Mirror Descent with cutting-plane LP and adaptive step size."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    M_current = fast_conv(a).max()
    best_a = a.copy()
    best_M = M_current
    no_improve = 0
    prev_lp_time = 0.001

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M_current = fast_conv(a).max()

        t0 = time.monotonic()
        g = solve_lp_cutting_plane(a, M_current)
        lp_time = time.monotonic() - t0
        prev_lp_time = 0.8 * prev_lp_time + 0.2 * lp_time

        if g is None or g.sum() < 1e-12:
            break
        g = g / g.sum()

        # Check if g is too similar to a (stuck)
        similarity = np.dot(a, g)
        if similarity > 0.9999:
            no_improve += 1
            if no_improve > 50:
                break

        # Line search
        improved = False
        # Try larger steps first
        for t in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial = a_trial / st
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-12):
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break

        # Time guard: stop if LP is taking too long
        remaining = deadline - time.monotonic()
        if remaining < prev_lp_time * 3:
            break

    return best_a


def perturbation(a, rng, sigma=0.05):
    """Add structured perturbation to escape local minima."""
    n = len(a)
    t = np.linspace(0, 1, n, endpoint=False)

    # Random smooth perturbation
    ksize = max(3, n // 50)
    kernel = np.ones(ksize) / ksize
    noise = np.convolve(rng.normal(0, sigma, n), kernel, mode='same')

    # Also try a sinusoidal perturbation
    freq = rng.integers(1, 10)
    sin_noise = sigma * 0.5 * np.sin(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))

    a_pert = np.maximum(a + noise + sin_noise, 0)
    s = a_pert.sum()
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Quick CD polish of parent (100s)
    Phase 2: LP Mirror Descent - main optimization (900s)
    Phase 3: CD + LP alternating polish (300s)
    Phase 4: Final squeeze (remaining time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, verbose=False)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent, verbose=False)

    n_best = len(parent)

    # === PHASE 1: Quick CD to ensure local optimum ===
    p1_deadline = min(time.monotonic() + 80, deadline - 1100)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Coordinate Descent polish n={n_best}", flush=True)
    print(f"{'='*60}", flush=True)

    a_cd = coord_descent_fast(parent, max_sweeps=500, deadline=p1_deadline, seed=seed)
    tracker.update(a_cd, verbose=False)
    if tracker.best is not None:
        current_best = tracker.best.copy()
    else:
        current_best = a_cd

    # Also try a few perturbations + short CD
    for pidx in range(3):
        if time.monotonic() > p1_deadline:
            break
        a_pert = perturbation(current_best, rng, sigma=0.03)
        a_pert_cd = coord_descent_fast(a_pert, max_sweeps=100, deadline=min(time.monotonic() + 20, p1_deadline), seed=seed + 10 + pidx)
        tracker.update(a_pert_cd, verbose=False)

    tracker.report("Phase 1") if hasattr(tracker, 'report') else None
    print(f"  After Phase 1: J={tracker.best_score:.8f} n={tracker.best.size}", flush=True)

    # === PHASE 2: LP Mirror Descent (main phase) ===
    p2_deadline = min(time.monotonic() + 850, deadline - 250)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent n={tracker.best.size} (budget ~{p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    a_lp = lp_mirror_descent(tracker.best, deadline=p2_deadline, max_iters=100000)
    J_lp = compute_J(a_lp)
    print(f"  LP result: J={J_lp:.8f}", flush=True)
    tracker.update(a_lp, verbose=False)

    # If improvements are slow, try a perturbation and restart
    if time.monotonic() < p2_deadline - 60:
        a_pert2 = perturbation(tracker.best, rng, sigma=0.02)
        a_lp2 = lp_mirror_descent(a_pert2, deadline=p2_deadline, max_iters=50000)
        J_lp2 = compute_J(a_lp2)
        print(f"  LP restart: J={J_lp2:.8f}", flush=True)
        tracker.update(a_lp2, verbose=False)

    print(f"  After Phase 2: J={tracker.best_score:.8f} n={tracker.best.size}", flush=True)

    # === PHASE 3: CD + LP alternating ===
    p3_deadline = min(time.monotonic() + 250, deadline - 50)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: CD+LP alternating (budget ~{p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    alternating_deadline = p3_deadline
    for alt_round in range(10):
        if time.monotonic() > alternating_deadline:
            break

        # CD step
        cd_dl = min(time.monotonic() + 30, alternating_deadline - 30)
        a_cd2 = coord_descent_fast(tracker.best, max_sweeps=200, deadline=cd_dl, seed=seed + 100 + alt_round)
        tracker.update(a_cd2, verbose=False)

        if time.monotonic() > alternating_deadline:
            break

        # LP step
        lp_dl = min(time.monotonic() + 60, alternating_deadline)
        a_lp3 = lp_mirror_descent(tracker.best, deadline=lp_dl, max_iters=20000)
        J_val = compute_J(a_lp3)
        print(f"  Alt round {alt_round}: J={J_val:.8f}", flush=True)
        tracker.update(a_lp3, verbose=False)

    print(f"  After Phase 3: J={tracker.best_score:.8f} n={tracker.best.size}", flush=True)

    # === PHASE 4: Try different length ===
    p4_deadline = deadline
    if time.monotonic() < p4_deadline - 30:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 4: Try n=768 (budget ~{p4_deadline - time.monotonic():.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        new_n = 768
        a_ext = resample_seq(tracker.best, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        # Quick CD first
        a_ext_cd = coord_descent_fast(a_ext, max_sweeps=100, deadline=min(time.monotonic() + 30, p4_deadline - 20), seed=seed + 500)
        tracker.update(a_ext_cd, verbose=False)

        if time.monotonic() < p4_deadline - 10:
            a_ext_lp = lp_mirror_descent(a_ext_cd, deadline=p4_deadline, max_iters=30000)
            J_ext = compute_J(a_ext_lp)
            print(f"  n=768 LP: J={J_ext:.8f}", flush=True)
            tracker.update(a_ext_lp, verbose=False)

    # === Final: one more CD squeeze ===
    if time.monotonic() < p4_deadline - 5 and tracker.best is not None:
        a_final_cd = coord_descent_fast(tracker.best, max_sweeps=100, deadline=p4_deadline, seed=seed + 999)
        tracker.update(a_final_cd, verbose=False)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
