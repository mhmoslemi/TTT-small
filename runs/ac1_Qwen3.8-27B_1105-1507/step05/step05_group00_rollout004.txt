"""
Convex optimization of step function heights to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated smooth subgradient descent with adaptive smoothing,
warm-started LP direction finding, and pair-coordinate descent. Explores
multiple sequence lengths around the parent length.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= 1 for all k, g >= 0.
    Returns normalized direction g/sum(g) and the optimal M (max conv of a with g)."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, -res.fun
    except Exception:
        pass
    return None, None


def nesterov_smooth_subgrad(a, max_iters=2000, time_budget=100.0, tau_start=0.1, tau_end=0.001):
    """
    Nesterov-accelerated smooth subgradient descent on f_tau(a) = tau*log(sum exp(conv(j)/tau)).
    This is a smooth convex surrogate for max_j conv(j).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Nesterov parameters
    v = np.zeros(n)
    y = a.copy()
    beta_prev = 1.0

    for k in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive beta (Nesterov)
        beta_k = (1.0 + np.sqrt(1.0 + 4.0 * beta_prev * beta_prev)) / 2.0
        t_k = (1.0 - beta_prev) / beta_k

        # Extrapolation step
        y = a + t_k * (a - y)
        y = np.maximum(y, 0)
        s_y = np.sum(y)
        if s_y > 1e-10:
            y = y / s_y

        # Adaptive tau
        progress = k / max(max_iters - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** progress

        # Compute smooth gradient
        conv_y = compute_conv(y)
        cmax = float(np.max(conv_y))
        shifted = (conv_y - cmax) / tau
        shifted = np.clip(shifted, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv_y))
            weights = np.zeros(len(conv_y))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient: grad[i] = 2 * sum_j weights[j] * a[y-index-i] (if valid)
        # Actually, the gradient of conv(j) w.r.t. a[i] is 2*a[j-i] (if valid)
        # So the weighted gradient is 2 * sum_j w_j * a[j-i] (if 0<=j-i<n)
        # This is 2 * (weights reversed) convolved with a, evaluated at the right positions
        w_rev = weights[::-1]
        # grad_a[i] = 2 * sum_j w[j] * a[j-i] for valid j-i
        # = 2 * (w * a)[i] where (w*a)[i] = sum_j w[j]*a[i+j]... 
        # Let me think differently.
        # sum_j w[j] * a[j-i] = sum_m w[m+i] * a[m] where m = j-i
        # = (w shifted by i) dotted with a
        # This is the cross-correlation of w and a at lag i
        # = (w_rev * a)[n-1-i] where * denotes convolution... 
        # Actually, let's just compute it directly for efficiency using FFT
        # grad[i] = 2 * sum_{j: 0<=j-i<n} w[j] * a[j-i]
        # Let m = j-i, then j = m+i, and we need 0 <= m < n, 0 <= m+i < 2n-1
        # So m ranges from max(0, -(i)) to min(n-1, 2n-2-i)
        # Since i >= 0: m from 0 to min(n-1, 2n-2-i)
        # grad[i] = 2 * sum_{m=0}^{min(n-1, 2n-2-i)} w[m+i] * a[m]
        # This is the cross-correlation!
        # cross_corr[i] = sum_m a[m] * w[m+i]
        # We can compute this via FFT: cross_corr = ifft(fft(a) * conj(fft(w)))
        # But we need to be careful about the definition.
        
        # Simpler approach: use the relation
        # grad[i] = 2 * (w * a_reversed)[n-1-i] (with appropriate padding)
        # Or just: the gradient of sum_j w[j]*conv(j) w.r.t. a[i]
        # = 2 * sum_j w[j] * a[j-i] (valid range)
        # = 2 * (full cross-correlation of a and w) evaluated at lag i
        
        # Let's compute cross-correlation: xcorr[i] = sum_m a[m] * w[m+i] for m where both are valid
        # Using FFT: xcorr = ifft(conj(fft(a, N)) * fft(w, N)) ... hmm this is getting complicated
        # Let me just use a direct convolution approach
        
        # Actually the simplest: grad = 2 * (w_full_convolved_with_a) at the right indices
        # where w_full has length 2n-1
        # conv_w_a[k] = sum_j w[j] * a[j + k - (n-1)] ... no this is confusing
        
        # Let me just use the fact that:
        # d/da[i] of (a*a)[j] = 2*a[j-i] if 0<=j-i<n else 0
        # So grad_smooth[i] = 2 * sum_j w[j] * a[j-i] * I[0<=j-i<n]
        # = 2 * sum_{j=i}^{i+n-1} w[j] * a[j-i] (with w[j]=0 for j outside [0, 2n-2])
        # Let m = j-i: = 2 * sum_{m=0}^{n-1} w[m+i] * a[m]
        # = 2 * (w[i:i+n] . a)  (dot product of w shifted by i with a)
        
        # This is the cross-correlation of w and a!
        # xcorr[i] = sum_{m=0}^{n-1} a[m] * w[m+i]
        # We can compute this efficiently:
        # xcorr = np.correlate(w, a, mode='valid') ... no, that's the wrong direction
        # Actually: xcorr[i] = sum_m a[m] * w[m+i] for i = 0, ..., (2n-2)-(n-1) = n-1
        # Wait, w has length 2n-1, a has length n
        # For i from 0 to 2n-1-n = n-1: xcorr[i] = sum_{m=0}^{n-1} a[m] * w[m+i]
        
        # Using FFT: pad both to size >= 2n-1
        N = 1
        while N < 2 * n - 1:
            N <<= 1
        A_fft = rfft(a, N)
        W_fft = rfft(weights, N)
        # xcorr[i] = sum_m a[m] * w[m+i] = (a flipped) convolved with w, at position...
        # Let's use: xcorr = ifft(fft(a_flipped) * fft(w))[n-1 : 2n-1]
        a_flip = a[::-1]
        af = rfft(a_flip, N)
        conv_af_w = irfft(af * W_fft, N)
        xcorr = conv_af_w[n - 1:2 * n - 1]
        
        # Verify: xcorr[i] should = sum_m a[m] * w[m+i]
        # conv_af_w[k] = sum_j a_flip[j] * w[k-j] = sum_j a[n-1-j] * w[k-j]
        # Let m = n-1-j: = sum_m a[m] * w[k-n+1+m]
        # So conv_af_w[k] = sum_m a[m] * w[m + (k-n+1)]
        # We want sum_m a[m] * w[m+i], so k-n+1 = i, k = n-1+i
        # xcorr[i] = conv_af_w[n-1+i] ✓
        
        grad_smooth = 2.0 * xcorr

        # Project gradient onto sum=const (center it)
        grad_centered = grad_smooth - np.mean(grad_smooth)

        # Step size: for f_tau, the Lipschitz constant is roughly 2*||a||_inf / tau
        # Use step = c / (||grad||_2 * (k+1)) or similar
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break
        step = 1.0 / (grad_norm * (k + 1))
        # Scale to keep updates reasonable
        step = min(step, 0.1 / max(beta_k, 1.0))

        # Nesterov update
        v = t_k * v + step * grad_centered
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
            v = np.zeros(n)

        a = a_new
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        beta_prev = beta_k

    return best_a


def pair_coordinate_descent(a, time_budget=30.0, max_sweeps=10):
    """
    For each pair (i, j), try transferring mass to reduce the peak.
    Focus on pairs that contribute most to the current peak.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        M = float(conv[k_star])

        # Find which pairs contribute most to conv[k_star]
        # conv[k_star] = sum_i a[i] * a[k_star - i]
        # The contribution of pair (i, k_star-i) is a[i]*a[k_star-i]
        contributions = {}
        for i in range(n):
            j = k_star - i
            if 0 <= j < n and j > i:  # avoid double counting
                contributions[(i, j)] = a[i] * a[j]
            elif 0 <= j < n and j == i:
                contributions[(i, i)] = a[i] * a[j]

        # Sort by contribution (descending) and try to reduce top contributors
        sorted_pairs = sorted(contributions.items(), key=lambda x: -x[1])
        
        improved = False
        for (i, j), contrib in sorted_pairs[:20]:  # Try top 20 pairs
            if time.monotonic() - start > time_budget:
                break
            if contrib < 1e-10:
                break

            # Try reducing a[i] and a[j] (and adding to other positions)
            # Or try moving mass from i to a position that doesn't contribute to k_star
            # Option 1: Move mass from i to a position p that minimizes contribution to k_star
            # Option 2: Move mass from i to j (if j != i) to make the product smaller
            
            if i != j and a[i] > 0 and a[j] > 0:
                # Move some mass from the larger to the smaller (equalization reduces product)
                if a[i] > a[j]:
                    src, dst = i, j
                else:
                    src, dst = j, i
                for frac in [0.1, 0.3, 0.5]:
                    delta = frac * (a[src] - a[dst]) / 2
                    if delta <= 0:
                        break
                    a[src] -= delta
                    a[dst] += delta
                    new_conv = compute_conv(a)
                    new_max = float(np.max(new_conv))
                    new_obj = 2.0 * n * new_max  # sum is still 1
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                    else:
                        a[src] += delta
                        a[dst] -= delta
                        break
            else:
                # i == j: move mass from i to a position that doesn't contribute to k_star
                # Find position p where k_star - p is out of range or a[k_star-p] is small
                best_p = None
                best_val = np.inf
                for p in range(n):
                    if p == i:
                        continue
                    q = k_star - p
                    if 0 <= q < n:
                        val = a[p] * a[q]
                    else:
                        val = 0
                    if val < best_val:
                        best_val = val
                        best_p = p
                if best_p is not None and a[i] > 0:
                    for frac in [0.05, 0.1, 0.2]:
                        delta = frac * a[i]
                        a[i] -= delta
                        a[best_p] += delta
                        new_conv = compute_conv(a)
                        new_max = float(np.max(new_conv))
                        new_obj = 2.0 * n * new_max
                        if new_obj < best_obj - 1e-12:
                            best_obj = new_obj
                            best_a = a.copy()
                            improved = True
                        else:
                            a[i] += delta
                            a[best_p] -= delta
                            break

        if not improved:
            break

    return best_a


def lp_direction_step(a, time_budget=20.0):
    """Use LP to find a direction that reduces the peak, then take a step."""
    n = len(a)
    start = time.monotonic()

    g, _ = solve_lp_direction(a, time_limit=min(time_budget * 0.5, 15))
    if g is None:
        return a

    best_a = a.copy()
    best_obj = compute_obj(a)

    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        if time.monotonic() - start > time_budget:
            break
        a_try = (1 - t) * a + t * g
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def stochastic_restart(a, scale=0.05, seed=None):
    """Perturb the sequence multiplicatively."""
    rng = np.random.default_rng(seed)
    a_pert = a * np.exp(rng.normal(0, scale, len(a)))
    a_pert = np.maximum(a_pert, 0)
    s = np.sum(a_pert)
    if s > 1e-10:
        a_pert = a_pert / s
    return a_pert
