Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using fast coordinate descent with
incremental convolution updates (O(n^2) per sweep vs O(n^2 log n)), spectral nudging
for local-minimum escape, softmax-parameterized Adam, and multi-length exploration.
Key improvements: vectorized incremental b updates eliminate redundant FFT calls,
active-set tracking reduces per-evaluation cost, and spectral domain perturbations

import numpy as np

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def eval_coord_obj(c_vals, sl_vals, quad_pos, base_max, S, x_old, x):
    """Evaluate the normalized max-conv objective for a candidate coordinate value x."""
    ns = S - x_old + x
    if ns < 1e-15:
        return 1e10
    vals = c_vals + sl_vals * x
    mx = base_max
    if vals.size > 0:
        vm = vals.max()
        if vm > mx:
            mx = vm
    if quad_pos >= 0:
        qv = c_vals[quad_pos] + x * x
        if qv > mx:
            mx = qv
    return mx / (ns * ns)


def precompute_coord_data(a, i, b):
    """Precompute c_vals, sl_vals, quad_pos, base_max for coordinate i."""
    n = len(a)
    x_old = a[i]
    k_lo = i

    sl_vals = np.zeros(m)

    else:
        base_max = 0.0

    return c_vals, sl_vals, quad_pos, base_max

def spectral_nudge(a, strength=0.08, rng=None):
    """Nudge power spectrum towards flatness, then project to non-negative simplex."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    mag = np.abs(fa)
    phase = np.angle(fa)
    # Target: uniform magnitude (scaled to match current energy)
    target_mag = np.ones_like(mag) * mag.mean()
    new_mag = (1.0 - strength) * mag + strength * target_mag
    new_fa = new_mag * np.exp(1j * phase)
    new_a = np.fft.irfft(new_fa, size)[:n]
    # Project to non-negative simplex
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s < 1e-10:
        return a.copy()
    return new_a / s


def spectral_init(n, rng, variant=0):

    else:

    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-8)

def smooth_max_grad(a, tau):
    """Compute gradient of log-sum-exp(tau * conv(a)) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[i] += 2 * p[k] * a[k-i] for i in [lo, hi]
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * pk * a[k - i]

    return grad


def softmax_adam(a, max_iters, deadline, lr=0.02, tau_start=10.0, tau_end=100.0):
    """Adam optimizer in softmax parameterization on the simplex."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_M = float(fast_conv(a).max())
    best_a = a.copy()
    no_improve = 0

    m = np.zeros(n)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)

        grad = smooth_max_grad(a, tau)
        # Riemannian gradient: project to tangent space of simplex
        grad = grad - grad.mean()

        # Adam
        beta1, beta2 = 0.9, 0.999
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        lr_t = lr / (1.0 + 0.005 * it)
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        a = a_new

        M = float(fast_conv(a).max())
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 100:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        if S_cur < 1e-10:
            break
        a = a / S_cur

        # Periodic spectral nudge
        if sweep > 0 and sweep % nudge_every == 0:
            a_nudged = spectral_nudge(a, strength=nudge_str, rng=rng)
            # Check if nudge helps (after a few CD sweeps)
            a_trial = coord_descent_fast(a_nudged, 3, min(time.monotonic() + 5, deadline), seed=seed + sweep)
            M_trial = float(fast_conv(a_trial).max())
            if M_trial < best_M:
                a = a_trial
            else:
                a = a  # Keep original

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not any_change:
            no_improve += 5
        if no_improve > 35:
            break

    return best_a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: fast coordinate descent with incremental updates,
    spectral nudging for escape, softmax Adam, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        tracker.update(parent)
        print("No parent; using uniform init", flush=True)

    # === PHASE 1: Multi-length coordinate descent with spectral init ===
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length CD with spectral init", flush=True)
    print(f"{'='*60}", flush=True)

    n_parent = parent.size if parent is not None else 512
    lengths = sorted(set([n_parent, max(256, n_parent // 2), n_parent * 2, n_parent + 128]))
    lengths = [n for n in lengths if 128 <= n <= 2048]

    time_per_length = max(40, min(180, 500 // len(lengths)))

    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        phase_dl = min(time.monotonic() + time_per_length, deadline - 250)
        if time.monotonic() >= phase_dl:
            continue

        t0 = time.monotonic()
        best_for_n = None
        best_M_for_n = float('inf')

        # Generate diverse inits
        inits = []
        if tracker.best is not None:
            inits.append(resample_seq(tracker.best, n))
            inits.append(np.maximum(resample_seq(tracker.best, n) + rng.normal(0, 0.03, n), 0.01))
            inits.append(np.maximum(resample_seq(tracker.best, n) + rng.normal(0, 0.1, n), 0.01))
        inits.append(np.ones(n))
        for v in range(4):
            inits.append(spectral_init(n, rng, variant=v))

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            # Run coordinate descent
            max_sw = max(30, int(5000 // max(1, n // 128)))
            a_opt = coord_descent_fast(a0, max_sw, phase_dl, seed=seed + idx * 13)

            M_val = float(fast_conv(a_opt).max())
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: M={best_M_for_n:.10f} J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("Phase 1 complete")
    print(f"Phase 1 total: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Continued CD with nudge on best ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: CD with spectral nudging", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 200:
        n_best = tracker.best.size
        phase_dl = min(time.monotonic() + 150, deadline - 180)

        # Run nudge-CD from best
        a_nudged = periodic_nudge_cd(tracker.best, 100, phase_dl, seed=seed + 100,
                                     nudge_every=10, nudge_str=0.04)
        tracker.update(a_nudged)

        # Try stronger nudge
        a_nudged2 = periodic_nudge_cd(tracker.best, 60, min(time.monotonic() + 60, phase_dl),
                                      seed=seed + 200, nudge_every=8, nudge_str=0.1)
        tracker.update(a_nudged2)

    tracker.report("Phase 2 complete")
    print(f"Phase 2 total: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Softmax Adam ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Softmax Adam", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 100:
        n_best = tracker.best.size
        # Adam is expensive for large n; limit iterations
        max_it = max(50, min(500, int(80.0 / (n_best / 256.0))))
        phase_dl = min(time.monotonic() + 80, deadline - 80)

        a_adam = softmax_adam(tracker.best, max_it, phase_dl, lr=0.015, tau_start=8.0, tau_end=80.0)
        tracker.update(a_adam)

        # Second Adam run with different params
        if time.monotonic() < deadline - 60:
            a_adam2 = softmax_adam(tracker.best, max(30, max_it // 2),
                                   min(time.monotonic() + 40, deadline - 50),
                                   lr=0.03, tau_start=20.0, tau_end=60.0)
            tracker.update(a_adam2)

    tracker.report("Phase 3 complete")
    print(f"Phase 3 total: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Final CD polish ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final CD polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 10:
        phase_dl = min(time.monotonic() + 80, deadline - 5)
        max_sw = max(50, int(10000 // max(1, tracker.best.size // 128)))

        # Direct CD polish
        a_polished = coord_descent_fast(tracker.best, max_sw, phase_dl, seed=seed + 999)
        tracker.update(a_polished)

        # Perturbed restart + CD
        a_pert = tracker.best + rng.normal(0, 0.01, tracker.best.size)
        a_pert = np.maximum(a_pert, 0.001)
        a_pert = a_pert / a_pert.sum()
        a_polished2 = coord_descent_fast(a_pert, max(30, max_sw // 2),
                                         min(time.monotonic() + 30, deadline - 5), seed=seed + 7777)
        tracker.update(a_polished2)

    tracker.report("Phase 4 complete")
    print(f"Phase 4 total: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Quick multi-length exploration if time remains ===
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Additional exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 15:
        # Try 2x length
        n2 = tracker.best.size * 2
        if n2 <= 2048:
            a_ext = resample_seq(tracker.best, n2)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext_opt = coord_descent_fast(a_ext, 30, min(time.monotonic() + 30, deadline - 10),
                                           seed=seed + 5555)
            tracker.update(a_ext_opt)

        # Try another spectral nudge round
        if time.monotonic() < deadline - 10:
            a_sn = spectral_nudge(tracker.best, strength=0.03, rng=rng)
            a_sn = coord_descent_fast(a_sn, 20, min(time.monotonic() + 15, deadline - 5),
                                      seed=seed + 8888)
            tracker.update(a_sn)

    tracker.report("Phase 5 complete")
    print(f"Phase 5 total: {time.monotonic()-p5_start:.1f}s", flush=True)

    # Final fallback
    if tracker.best is None:
        tracker.best = np.ones(512) / 512.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
