from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]

def compute_M(a):
    return float(fast_conv(a).max())

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def solve_lp(a, M, row_idx, col_idx, x0=None):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def line_search_analytic(a, g):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val

def gradient_of_conv_at_k(a, k):
    """Gradient of (a*a)[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[k - j]
    return grad

def fista_smooth(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0):
    """FISTA on smooth approximation of max conv."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    v = a.copy()
    t_k = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        
        b = fast_conv(v)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * p[k] * a[k - j]
        
        # Step size
        L = 2.0 * n * tau  # simple upper bound on Lipschitz constant
        step = 1.0 / L
        
        a_new = v - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        v = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        v = np.maximum(v, 0.0)
        v = v / v.sum()
        t_k = t_new
        a = a_new
        
        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
    
    return best_a

def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = compute_M(a)
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        t, M_new = line_search_analytic(a, g)
        if t < 1e-10:
            no_improve += 1
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        if no_improve > 80:
            break
        if verbose and it % 100 == 0:
            print(f"    iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)
    
    return best_a

def active_set_polish(a, max_iters=50, tol=1e-10):
    """
    Active set refinement: identify active constraints, solve KKT system,
    and check if we're at the optimum.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    for it in range(max_iters):
        b = fast_conv(a)
        M = b.max()
        
        # Active set: coefficients within 1% of max
        active = np.where(b >= M * (1 - 0.01))[0]
        
        if len(active) < 2:
            break
        
        # Build KKT system
        # Unknowns: alpha[0..len(active)-1] and lambda
        # Equations: 2 * sum_k alpha[k] * a[k-j] = lambda for j=0..n-1
        #           sum(alpha) = 1
        
        # Matrix: n+1 rows, len(active)+1 cols
        A_kkt = np.zeros((n + 1, len(active) + 1))
        b_kkt = np.zeros(n + 1)
        
        for idx, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A_kkt[j, idx] = 2.0 * a[k - j]
            A_kkt[n, idx] = 1.0  # sum(alpha) = 1
        
        A_kkt[:, len(active)] = -1.0  # = -lambda
        b_kkt[n] = 1.0
        
        # Least squares solve
        try:
            sol, residuals, rank, sv = np.linalg.lstsq(A_kkt, b_kkt, rcond=None)
        except:
            break
        
        alpha = sol[:len(active)]
        lambda_val = sol[len(active)]
        
        # Check: are all alpha >= 0?
        if np.any(alpha < -1e-8):
            # Remove negative alphas from active set
            # This is complex; just break for now
            break
        
        # Compute gradient
        grad = np.zeros(n)
        for idx, k in enumerate(active):
            if alpha[idx] > 1e-14:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    grad[j] += 2.0 * alpha[idx] * a[k - j]
        
        # Check KKT: grad[j] should be <= lambda where a[j] = 0
        # And grad[j] should be ~= lambda where a[j] > 0
        support = a > 1e-10
        if np.any(support):
            violation = np.max(np.abs(grad[support] - lambda_val))
        else:
            violation = 0
        
        # For complementarity: grad[j] >= lambda where a[j] = 0
        if np.any(~support):
            comp_violation = np.max(np.maximum(0, lambda_val - grad[~support]))
        else:
            comp_violation = 0
        
        total_violation = violation + comp_violation
        
        if total_violation < tol:
            break
        
        # If not at optimum, we can try to adjust
        # Simple approach: add a small correction in the direction that reduces violation
        # grad should be = lambda on support
        # correction = lambda - grad (on support), projected to be orthogonal to 1 (to preserve sum)
        correction = np.zeros(n)
        correction[support] = lambda_val - grad[support]
        # Remove mean to preserve sum(a) = 1
        correction -= correction[support].mean() * support
        # Also ensure we don't make a negative
        # Take a small step
        step_size = 0.01
        a_new = a + step_size * correction
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
            if compute_M(a_new) <= compute_M(a) * (1 + 1e-6):
                a = a_new
    
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def flat_top_window(n):
    """Generate a flat-top window of length n."""
    j = np.arange(n)
    if n == 1:
        return np.ones(1)
    x = 2 * np.pi * j / (n - 1)
    a0, a1, a2, a3, a4 = 0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368
    w = a0 - a1 * np.cos(x) + a2 * np.cos(2*x) - a3 * np.cos(3*x) + a4 * np.cos(4*x)
    return np.maximum(w, 0)

def raised_cosine_window(n):
    if n == 1:
        return np.ones(1)
    j = np.arange(n)
    return 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1)))

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse initialization (flat-top, windows, random, parent)
2. FFT-accelerated smooth gradient screening (FISTA)
3. LP mirror descent with analytic line search (main optimization)
4. Active-set KKT refinement
5. Multi-length exploration via resampling
6. Structured and random perturbation restarts
7. Final extended LP polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    """Precompute row/col indices for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    vals = a[row_idx - col_idx]  # a[k-j] for each (k,j) pair
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2 convolution."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def fft_cross_correlation(p, a, size):
    """
    Compute cross-correlation: xc[j] = sum_k a[k]*p[k+j] for j=0..n-1
    using FFT. p has length 2n-1, a has length n.
    """
    A = np.fft.rfft(a, size)
    P = np.fft.rfft(p, size)
    xc = np.fft.irfft(np.conj(A) * P, size)
    return xc[:len(a)]


def smooth_gradient_fft(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*(a*a)_k)) using FFT.
    Returns grad of length n.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    # Need size >= 3n-2 to avoid circular aliasing
    size = 1
    while size < 3 * n:
        size <<= 1

    xc = fft_cross_correlation(p, a, size)
    grad = 2.0 * xc
    return grad, p


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    best_M = compute_M(a)

    t_k = 1.0

        step = 1.0 / (2.0 * tau + 1e-10)

        a_new = v - step * grad
        # Project onto simplex
        a_new = project_simplex(a_new)
        if a_new.sum() < 1e-10:
            a_new = a.copy()

        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        v = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        v = np.maximum(v, 0.0)
        sv = v.sum()
        if sv > 1e-10:
            v = v / sv
        t_k = t_new
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a


def subgradient_screen(a, max_iters, deadline):
    """
    Simple subgradient method for quick screening.
    Minimize max_k (a*a)_k over {a>=0, sum(a)=1}.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        # Argmax (or average of argmaxes for robustness)
        threshold = M * 0.999
        active = b >= threshold
        n_active = active.sum()
        if n_active == 0:
            active = np.zeros_like(b, dtype=bool)
            active[np.argmax(b)] = True
            n_active = 1

        # Subgradient: average of gradients at active points
        grad = np.zeros(n)
        size = 1
        while size < 3 * n:
            size <<= 1
        A = np.fft.rfft(a, size)
        for k in range(2 * n - 1):
            if not active[k]:
                continue
            # grad_k[j] = 2*a[k-j] for valid j
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi+1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        grad /= n_active

        # Step size: Polylak
        step = 0.5 / (it + 1.0)
        a_new = a - step * grad
        a_new = project_simplex(a_new)
        if a_new.sum() < 1e-10:
            a_new = a.copy()
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = compute_M(a)
        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_improve = 0

        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        t, M_new = line_search_analytic(a, g)
        if t < 1e-10:
            no_improve += 1
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        if no_improve > 80:
            break
        if verbose and it % 200 == 0:
            print(f"    iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def active_set_polish(a, max_iters=50, tol=1e-10):
    """
    Active set KKT refinement: identify active constraints,
    solve KKT system, iterate to convergence.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    for it in range(max_iters):
        b = fast_conv(a)
        M = b.max()

        # Active set: coefficients close to max
        active_mask = b >= M * (1 - 0.02)
        active_indices = np.where(active_mask)[0]

        if len(active_indices) < 2:
            break

        # Build KKT system
        # Unknowns: alpha (len(active)) + lambda
        # Equations: 2*sum_k alpha[k]*a[k-j] = lambda for j=0..n-1
        #            sum(alpha) = 1

        n_act = len(active_indices)
        A_kkt = np.zeros((n + 1, n_act + 1))
        b_kkt = np.zeros(n + 1)

        for idx, k in enumerate(active_indices):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A_kkt[j, idx] = 2.0 * a[k - j]
            A_kkt[n, idx] = 1.0

        A_kkt[:, n_act] = -1.0
        b_kkt[n] = 1.0

        try:
            sol, _, _, _ = np.linalg.lstsq(A_kkt, b_kkt, rcond=None)
        except Exception:
            break

        alpha = sol[:n_act]
        lambda_val = sol[n_act]

        # Check non-negativity of alpha
        if np.min(alpha) < -1e-6:
            break

        # Compute gradient
        grad = np.zeros(n)
        for idx, k in enumerate(active_indices):
            if alpha[idx] > 1e-14:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    grad[j] += 2.0 * alpha[idx] * a[k - j]

        # Check KKT: on support, grad ~= lambda; off support, grad <= lambda
        support = a > 1e-10
        if np.any(support):
            viol_support = np.max(np.abs(grad[support] - lambda_val))
        else:
            viol_support = 0.0
        if np.any(~support):
            viol_off = np.max(np.maximum(0, lambda_val - grad[~support]))
        else:
            viol_off = 0.0

        total_viol = viol_support + viol_off
        if total_viol < tol:
            break

        # Correction step
        correction = np.zeros(n)
        correction[support] = lambda_val - grad[support]
        # Zero out correction where a is already zero (to avoid making negative)
        correction[~support] = 0.0
        # Remove mean on support to preserve sum
        if np.sum(support) > 0:
            correction[support] -= correction[support][support].mean()

        step_size = min(0.05, 0.5 / (total_viol + 1e-10))
        a_new = a + step_size * correction
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
            if compute_M(a_new) <= compute_M(a) * (1 + 1e-8):
                a = a_new

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def flat_top_window(n):
    """Generate a flat-top window of length n."""
    if n == 1:
        return np.ones(1)
    j = np.arange(n)
    x = 2.0 * np.pi * j / (n - 1)
    a0 = 0.21557895
    a1 = 0.41663158
    a2 = 0.277263158
    a3 = 0.083578947
    a4 = 0.006947368
    w = a0 - a1 * np.cos(x) + a2 * np.cos(2 * x) - a3 * np.cos(3 * x) + a4 * np.cos(4 * x)
    return np.maximum(w, 0.0)


def raised_cosine_window(n):
    if n == 1:
        return np.ones(1)
    j = np.arange(n)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * j / (n - 1)))


def cosine_bell_window(n):
    if n == 1:
        return np.ones(1)
    j = np.arange(n)
    x = np.pi * j / (n - 1)
    return np.sin(x) ** 2


def tracker_update(tracker, a, label=""):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0 or a.sum() < 0.01:
        return
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < tracker['best_score']:
        tracker['best'] = a.copy()
        tracker['best_score'] = score
        elapsed = time.monotonic() - tracker['t_start']
        print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Diverse initialization + quick screening (FISTA/subgradient)
    Phase 2: LP mirror descent on top candidates
    Phase 3: Active set polish
    Phase 4: Multi-length exploration
    Phase 5: Perturbation restarts (random + structured)
    Phase 6: Final extended LP polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    budget_s = float(budget_s)
    deadline = t_start + budget_s - 20.0

    tracker = {'best': None, 'best_score': float('inf'), 't_start': t_start}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker_update(tracker, parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker['best_score']:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker_update(tracker, parent, "(default)")

    n0 = parent.size

    # Phase 1: Diverse initialization + screening
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Diverse Init + Screening (n0={n0})", flush=True)
    print(f"{'=' * 60}", flush=True)

    candidates = []

    # Parent
    p_norm = parent / parent.sum()
    candidates.append(('parent', p_norm.copy()))

    # Windows at same length
    for wname, wfunc in [('flat-top', flat_top_window),
                         ('raised-cos', raised_cosine_window),
                         ('cos-bell', cosine_bell_window)]:
        w = wfunc(n0)
        if w.sum() > 1e-10:
            w = w / w.sum()
            candidates.append((wname, w.copy()))

    # Random
    rand = rng.uniform(0.3, 1.5, n0)
    rand = rand / rand.sum()
    candidates.append(('random', rand))

    # Pareto-like: emphasize edges
    j = np.arange(n0)
    edge = 1.0 / (1.0 + 0.5 * np.exp(-((j - n0 / 2) / (n0 / 4)) ** 2))
    edge = np.maximum(edge, 0.01)
    edge = edge / edge.sum()
    candidates.append(('edge-weight', edge))

    # Windows at different lengths
    for mult, wfunc in [(0.75, flat_top_window), (1.25, flat_top_window),
                        (1.5, raised_cosine_window), (2.0, flat_top_window)]:
        n_trial = int(n0 * mult)
        if 64 <= n_trial <= 4096:
            w = wfunc(n_trial)
            if w.sum() > 1e-10:
                w = w / w.sum()
                candidates.append((f'{wfunc.__name__[:4]}-n{n_trial}', w))

    # Screen with FISTA and subgradient
    screen_deadline = min(t_start + budget_s * 0.18, deadline)
    per_cand_time = max(8.0, (screen_deadline - time.monotonic()) / max(1, len(candidates)))

    screened = []
    for name, cand in candidates:
        if time.monotonic() > screen_deadline:
            screened.append((name, cand))
            continue
        sub_dl = min(time.monotonic() + per_cand_time * 0.6, screen_deadline)
        opt1 = fista_smooth(cand, 300, sub_dl, initial_tau=5.0, final_tau=100.0)
        sub_dl2 = min(time.monotonic() + per_cand_time * 0.4, screen_deadline)
        opt2 = subgradient_screen(opt1, 200, sub_dl2)
        best_screen = opt1 if compute_M(opt1) <= compute_M(opt2) else opt2
        screened.append((name, best_screen))
        j_val = compute_J(best_screen)
        print(f"  Screened {name} (n={len(best_screen)}): J={j_val:.8f}", flush=True)
        tracker_update(tracker, best_screen, f"(screen {name})")

    # Phase 2: LP mirror descent on top candidates
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'=' * 60}", flush=True)

    screened.sort(key=lambda x: compute_J(x[1]))
    top_k = min(4, len(screened))
    top_candidates = screened[:top_k]

    lp_deadline = min(t_start + budget_s * 0.50, deadline)
    per_lp_time = max(30.0, (lp_deadline - time.monotonic()) / max(1, top_k))

    for name, cand in top_candidates:
        if time.monotonic() > lp_deadline:
            break
        n_c = len(cand)
        r_idx, c_idx = build_lp_matrix(n_c)
        sub_dl = min(time.monotonic() + per_lp_time, lp_deadline)
        print(f"  LP on {name} (n={n_c}), time_budget={per_lp_time:.0f}s", flush=True)
        opt = run_lp_mirror_descent(cand, 20000, sub_dl, r_idx, c_idx, verbose=True)
        tracker_update(tracker, opt, f"(LP {name})")

    # Phase 3: Active set polish
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Active Set Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker['best'] is not None:
        n_best = tracker['best'].size
        polished = active_set_polish(tracker['best'], max_iters=100)
        tracker_update(tracker, polished, "(AS polish)")

        # LP after AS polish
        r_idx_b, c_idx_b = build_lp_matrix(n_best)
        sub_dl = min(t_start + budget_s * 0.58, deadline)
        opt = run_lp_mirror_descent(polished, 10000, sub_dl, r_idx_b, c_idx_b, verbose=False)
        tracker_update(tracker, opt, "(LP after AS)")

    # Phase 4: Multi-length exploration
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker['best'] is not None:
        base_n = tracker['best'].size
        trial_mults = [0.5, 0.6, 0.75, 1.25, 1.4, 1.5, 2.0]
        phase4_deadline = min(t_start + budget_s * 0.72, deadline)

        for mult in trial_mults:
            if time.monotonic() > phase4_deadline:
                break
            n_trial = int(base_n * mult)
            if 64 <= n_trial <= 4096 and n_trial != base_n:
                a_init = resample_seq(tracker['best'], n_trial)
                a_init = np.maximum(a_init, 1e-8)
                a_init = a_init / a_init.sum()
                r_idx, c_idx = build_lp_matrix(n_trial)
                sub_dl = min(time.monotonic() + 50, phase4_deadline)
                opt = run_lp_mirror_descent(a_init, 5000, sub_dl, r_idx, c_idx, verbose=False)
                tracker_update(tracker, opt, f"(ML n={n_trial})")

    # Phase 5: Perturbation restarts
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Perturbation Restarts", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker['best'] is not None:
        n_best = tracker['best'].size
        r_idx_b, c_idx_b = build_lp_matrix(n_best)
        phase5_deadline = min(t_start + budget_s * 0.88, deadline)
        temperature = 0.1

        for r in range(60):
            if time.monotonic() > phase5_deadline:
                break

            a = tracker['best'].copy().astype(np.float64)
            a = a / a.sum()
            rtype = r % 4

            if rtype == 0:
                # Gaussian noise
                noise = rng.normal(0, temperature * 0.012, n_best)
                a = a + noise
            elif rtype == 1:
                # Random mask (zero out ~15%)
                mask = rng.random(n_best) > 0.15
                a = a * mask
            elif rtype == 2:
                # Circular shift
                shift = int(rng.integers(1, max(2, n_best // 6)))
                a = np.roll(a, shift)
            else:
                # Hybrid: noise + local emphasis
                noise = rng.normal(0, temperature * 0.008, n_best)
                emphasis = np.exp(-0.5 * ((np.arange(n_best) - n_best * rng.uniform(0.2, 0.8)) / (n_best * 0.1)) ** 2)
                a = a + noise + 0.01 * emphasis

            a = np.maximum(a, 1e-8)
            s_a = a.sum()
            if s_a < 1e-10:
                continue
            a = a / s_a

            # Quick smooth gradient
            sub_dl = min(time.monotonic() + 12, phase5_deadline)
            a = fista_smooth(a, 150, sub_dl, initial_tau=10, final_tau=80)

            # LP polish
            sub_dl2 = min(time.monotonic() + 25, phase5_deadline)
            opt = run_lp_mirror_descent(a, 2500, sub_dl2, r_idx_b, c_idx_b, verbose=False)
            tracker_update(tracker, opt, f"(Perturb r={r} type={rtype})")
            temperature *= 0.93

    # Phase 6: Final extended LP polish
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 6: Final LP Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker['best'] is not None:
        n_final = tracker['best'].size
        r_f, c_f = build_lp_matrix(n_final)
        a = tracker['best'].copy()
        a = a / a.sum()
        a_opt = run_lp_mirror_descent(a, 50000, deadline, r_f, c_f, verbose=True)
        tracker_update(tracker, a_opt, "(Final LP)")

        # Final AS polish
        polished = active_set_polish(a_opt, max_iters=50)
        tracker_update(tracker, polished, "(Final AS)")

        # One more short LP
        if deadline - time.monotonic() > 10:
            a2 = tracker['best'].copy() / tracker['best'].sum()
            opt2 = run_lp_mirror_descent(a2, 5000, deadline, r_f, c_f, verbose=False)
            tracker_update(tracker, opt2, "(Final LP2)")

    if tracker['best'] is None:
        tracker['best'] = parent.copy()
        tracker['best_score'] = compute_J(tracker['best'])

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker['best_score']:.8f} n={tracker['best'].size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker['best']]
