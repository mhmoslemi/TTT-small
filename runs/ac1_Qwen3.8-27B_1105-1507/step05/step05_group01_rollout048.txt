"""
LP-based peak minimization with flattened autocorrelation targeting for
minimizing 2*n*max(conv(a,a))/(sum(a))^2. Uses direct peak-minimizing LP
and flatness-targeting LP with adaptive line search and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix where row k gives coefficients for (a*g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        # (a*g)[k] includes term a[j]*g[k-j] for valid k-j
        # So row k, col j is a[j] if 0 <= k-j < n, i.e., j <= k < j+n
        start_row = j
        end_row = min(j + n, m)
        A[start_row:end_row, j] = a[j]
    return A


def solve_min_peak_lp(a, M=None):
    """
    Solve LP: min t s.t. (a*g)[k] <= t for all k, sum(g) = 1, g >= 0.
    Returns (g, t_opt) or (None, inf).
    """
    n = len(a)
    m = 2 * n - 1
    
    # Variables: g[0..n-1], t (index n)
    # Objective: minimize t → c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: (a*g)[k] - t <= 0 for each k
    # Row k: [A[k,0], ..., A[k,n-1], -1]
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = build_toeplitz(a)
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g[i] >= 0, t unbounded below (but will be positive)
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            t_opt = float(res.x[n])
            return g, t_opt
    except Exception:
        pass
    return None, np.inf


def solve_flatten_lp(a, target=None):

    m = 2 * n - 1
    
    if target is None:

    # Variables: g[0..n-1], t (index n)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_toep = build_toeplitz(a)

    # (a*g)[k] - t <= target  →  (a*g)[k] - t <= target
    # -(a*g)[k] - t <= -target  →  (a*g)[k] + t >= target

"""
LP-based peak minimization with direct peak-targeting and flatness-targeting LPs,
adaptive line search, gradient refinement, and multi-length exploration for
minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

def gradient_step(a, lr=0.01, n_peaks=3):
    """One gradient descent step on the peak of the autocorrelation."""
    n = len(a)
    conv = compute_conv(a)
    peak_idx = np.argsort(conv)[::-1][:n_peaks]

    grad = np.zeros(n)
    for k in peak_idx:
        k = int(k)
        for m in range(n):
            val = k - m
            if 0 <= val < n:
                if val == m:
                    grad[m] += 2.0 * a[val]
                else:
                    grad[m] += a[val]

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a
    return a_new / s

            if g2 is not None:
                candidates.append(('flatten', g2, t2))

        # 3. Max-sum LP with slight relaxation (every 5th iteration)
        if iter_count % 5 == 0:
            relax = 1.0 + 0.02 * (stall // 10)
            relax = min(relax, 1.2)
            g3, S3 = solve_max_sum_lp(best_a, M, relax=relax, time_limit=min(10, remaining * 0.05))
            if g3 is not None:
                candidates.append(('maxsum', g3, S3))

        improved = False
        for name, g_cand, lp_val in candidates:
            g_cand = np.maximum(g_cand, 0)
            sg = np.sum(g_cand)
            if sg < 1e-10:
                continue
            g_cand = g_cand / sg

            a_new, obj_new, alpha = line_search(best_a, g_cand, n_trials=8)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1

        # Gradient step periodically
        if iter_count % 10 == 0 and remaining > 10:
            a_grad = gradient_step(best_a, lr=0.005 * max(1, stall // 5), n_peaks=5)
            obj_grad = compute_obj(a_grad)
            if obj_grad < best_obj:
                best_obj = obj_grad
                best_a = a_grad.copy()
                improved = True
                stall = 0

        # Perturbation when stalled
        if stall > patience:
            strategy = rng.integers(0, 4)
            if strategy == 0:
                # Gaussian multiplicative noise
                scale = 0.03 + 0.02 * (stall // patience)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                sp = np.sum(a_pert)
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
            elif strategy == 1:
                # Peak-targeted reduction
                conv_p = compute_conv(best_a)
                peak_pos = np.argsort(conv_p)[::-1][0]
                a_pert = best_a.copy()
                for m in range(n):
                    val = peak_pos - m
                    if 0 <= val < n:
                        a_pert[m] *= 0.95
                        a_pert[val] *= 0.95
                a_pert = np.maximum(a_pert, 0)
                sp = np.sum(a_pert)
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
            elif strategy == 2:
                # Spectral perturbation
                fa = rfft(best_a)
                mag = np.abs(fa)
                phase = np.angle(fa)
                mag *= np.exp(rng.normal(0, 0.05, len(fa)))
                a_pert = irfft(mag * np.exp(1j * phase), n)
                a_pert = np.maximum(a_pert, 0)
                sp = np.sum(a_pert)
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()
            else:
                # Larger random restart
                a_pert = best_a * np.exp(rng.normal(0, 0.1, n))
                a_pert = np.maximum(a_pert, 0)
                sp = np.sum(a_pert)
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert.copy()

            stall = 0
            patience = max(15, patience // 2 + 5)

        if verbose and iter_count % 50 == 0:
            print(f"    LP polish it={iter_count}, obj={best_obj:.8f}, stall={stall}")

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)

    if choice == 0:
        f1 = rng.uniform(0.5, 3.0)
        f2 = rng.uniform(3.0, 8.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 6.28))
    elif choice == 1:
        a = rng.uniform(0.3, 1.5, n)
    elif choice == 2:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * rng.uniform(1, 5) * k / n)
        a += 0.2 * np.cos(2 * np.pi * rng.uniform(5, 15) * k / n + rng.uniform(0, 6.28))
    elif choice == 3:
        nblocks = rng.integers(3, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 4:
        a = np.ones(n)
        a[:n // 2] *= rng.uniform(0.6, 1.1)
        a[n // 2:] *= rng.uniform(0.6, 1.1)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-based peak minimization with min-peak LP, flatten LP,
    gradient refinement, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.8f}")

    # Phase 1: Intensive LP polishing on parent
    if best_seq is not None:
        n_parent = len(best_seq)
        print(f"\n=== Phase 1: LP Polish on parent (n={n_parent}) ===")
        polish_budget = min(400.0, (deadline - time.monotonic()) * 0.35)
        a_norm = best_seq / np.sum(best_seq)
        a_pol, obj_pol = lp_polish(a_norm, time_budget=polish_budget, verbose=True)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol * np.sum(best_seq)
            print(f"  After LP polish: {best_score:.8f}")
        else:
            print(f"  After LP polish: {obj_pol:.8f} (no improvement)")

    # Phase 2: Explore different n values
    print(f"\n=== Phase 2: Multi-n Exploration ===")

    if best_seq is not None:
        np_ = len(best_seq)
        n_values = sorted(set([np_ - 50, np_, np_ + 50, np_ + 100, np_ + 150, np_ + 200]))
    else:
        n_values = [500, 600, 700, 800]

    n_values = [max(100, min(1000, n)) for n in n_values]
    print(f"  n values to try: {n_values}")

    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        alloc = min(remaining * 0.2, 100.0)
        alloc = max(30.0, alloc)

        print(f"\n  N={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        if best_seq is not None and len(best_seq) != n:
            a_start = resample_sequence(best_seq, n)
            a_start = a_start / np.sum(a_start)
        elif best_seq is not None:
            a_start = best_seq / np.sum(best_seq)
        else:
            a_start = random_init(n, seed + n * 7919)

        # LP polish
        a_opt, obj_opt = lp_polish(a_start, time_budget=alloc, verbose=False)

        print(f"  N={n}: obj={obj_opt:.8f}")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt * np.sum(best_seq)
            print(f"  ** UPDATE: n={n}, obj={best_score:.8f} **")

    # Phase 3: Final intensive polish on global best
    print(f"\n=== Phase 3: Final Polish ===")
    print(f"  Best before: {best_score:.8f}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  Polishing n={n_best}, {remaining:.0f}s remaining")

        a_cur = best_seq / np.sum(best_seq)

        # Multiple rounds with different strategies
        for rnd in range(5):
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            alloc_rnd = min(remaining * 0.15, 120.0)
            alloc_rnd = max(20.0, alloc_rnd)

            # Add perturbation before each round to escape local minima
            rng = np.random.default_rng(seed + rnd * 31 + 7)
            a_cur = a_cur * np.exp(rng.normal(0, 0.02, n_best))
            a_cur = np.maximum(a_cur, 0)
            s_c = np.sum(a_cur)
            if s_c > 1e-10:
                a_cur = a_cur / s_c

            a_cur, obj_cur = lp_polish(a_cur, time_budget=alloc_rnd, verbose=False)

            if obj_cur < best_score:
                best_score = obj_cur
                best_seq = a_cur * np.sum(best_seq)
                print(f"  Round {rnd+1}: {best_score:.8f}")
            else:
                print(f"  Round {rnd+1}: {obj_cur:.8f} (no improvement)")

    # Phase 4: Final gradient refinement
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_fin = best_seq / np.sum(best_seq)
            n_fin = len(a_fin)
            obj_fin = compute_obj(a_fin)
            for _ in range(int(min(remaining * 50, 500))):
                a_fin = gradient_step(a_fin, lr=0.002, n_peaks=5)
                obj_new = compute_obj(a_fin)
                if obj_new < obj_fin:
                    obj_fin = obj_new
                else:
                    break
            if obj_fin < best_score:
                best_score = obj_fin
                best_seq = a_fin * np.sum(best_seq)
                print(f"  After gradient refine: {best_score:.8f}")

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        final_score = val
    else:
        result = [1.0 / 100.0] * 100
        final_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={final_score:.8f}, time={elapsed:.1f}s ===")

    return result
