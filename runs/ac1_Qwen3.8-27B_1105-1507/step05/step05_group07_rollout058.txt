def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u - (cssv - 1) / ind > 0
    rho = np.max(ind[cond])
    tau = (cssv[rho - 1] - 1) / rho
    return np.maximum(v - tau, 0)

def subgradient_max_conv(a, tol_frac=0.01):
    """Compute subgradient of max_k conv(a,a)[k].
    Returns (subgradient, max_val, active_set)."""
    n = len(a)
    c = fast_conv(a)
    M = c.max()
    tol = tol_frac * M
    active = np.where(c >= M - tol)[0]
    
    sg = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        sg[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    
    # Average over active set (minimal norm subgradient)
    sg /= len(active)
    
    return sg, M, active

def coord_descent(a, deadline, max_sweeps=500):

    n = len(a)

            break

            # The optimal value of a[i] given all others fixed:
            # We want to minimize max_k c_k where c_k changes as:

            break

    a = np.maximum(a, 0)
    if a.sum() > 1e-15:

"""
FISTA-based optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. FFT-based subgradient computation with adaptive active-set tolerance
2. Nesterov-accelerated projected gradient (FISTA) with O(n log n) per iteration
3. Multi-length exploration (512 -> 1024 -> 2048) with spectral initialization
4. Adaptive restart mechanism and Polyak averaging
5. Final coordinate-descent polish with incremental convolution updates
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u - (cssv - 1) / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(ind[cond])
    tau = (cssv[rho - 1] - 1) / rho
    return np.maximum(v - tau, 0)


def compute_subgradient(a, tol_frac=0.01):
    """Compute subgradient of max_k conv(a,a)[k].
    Returns (subgradient, max_value, active_indices)."""
    n = len(a)
    c = fast_conv(a)
    M = c.max()
    if M <= 0:
        return np.ones(n) / n, M, np.array([n - 1])
    tol = tol_frac * M
    active = np.where(c >= M - tol)[0]

    sg = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        sg[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    sg /= len(active)
    return sg, M, active


def fista(a0, deadline, tol_frac_init=0.1, tol_frac_min=0.001, lr=None, max_iters=100000):
    """FISTA for minimizing max(conv(a,a)) subject to sum(a)=1, a>=0.
    Uses adaptive tolerance and warm restarts."""
    n = len(a0)
    x = np.maximum(a0, 0)
    x_sum = x.sum()
    if x_sum < 1e-15:
        x = np.ones(n) / n
    else:
        x = x / x_sum
    y = x.copy()
    t_k = 1.0

    if lr is None:
        lr = 1.0 / (2.0 * np.sqrt(n))  # conservative initial step

    best_x = x.copy()
    best_M = fast_conv(x).max()
    stall_count = 0
    max_stall = 200

    # Polyak average
    x_avg = x.copy()
    avg_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive tolerance
        tol_frac = max(tol_frac_min, tol_frac_init * (1.0 - it / (max_iters * 0.5)))

        # Subgradient at y
        sg, M_y, active = compute_subgradient(y, tol_frac=tol_frac)

        # Gradient step + projection
        v = y - lr * sg
        x_new = proj_simplex(v)

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y / y_sum
        else:
            y = x_new.copy()
        t_k = t_new
        x = x_new

        # Track best
        M_x = fast_conv(x).max()
        if M_x < best_M - 1e-15:
            best_M = M_x
            best_x = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Polyak averaging
        x_avg = (avg_count * x_avg + x) / (avg_count + 1)
        avg_count += 1

        # Check averaged solution periodically
        if avg_count > 50 and avg_count % 100 == 0:
            M_avg = fast_conv(x_avg).max()
            if M_avg < best_M:
                best_M = M_avg
                best_x = x_avg.copy()
                stall_count = 0

        # Warm restart if stalled
        if stall_count > max_stall:
            t_k = 1.0
            y = x.copy()
            stall_count = 0
            # Slightly perturb
            y = y + 0.01 * np.random.standard_normal(n)
            y = proj_simplex(y)

        # Adaptive step size: increase if function decreasing
        if it > 10 and it % 100 == 0:
            # Estimate local Lipschitz constant
            c = fast_conv(x)
            M = c.max()
            if M > 0:
                # L ~ 2 * ||a||_2 for our problem
                L_est = 2.0 * np.linalg.norm(x)
                lr_new = 1.0 / max(L_est, 1e-10)
                lr = np.clip(lr_new, 0.01, 1.0)

    # Final check of average
    if avg_count > 100:
        M_avg = fast_conv(x_avg).max()
        if M_avg < best_M:
            best_x = x_avg.copy()

    return best_x


def coord_descent_fast(a, deadline, max_sweeps=100):
    """Fast coordinate descent with incremental convolution for local refinement."""
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    a_sum = a.sum()
    if a_sum < 1e-15:
        return a
    a = a / a_sum

    conv = fast_conv(a)
    best_M = conv.max()
    best_a = a.copy()

    order = np.random.default_rng(42).permutation(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        sweep_improved = False
        for i in order:
            old = a[i]
            if old < 1e-10:
                continue

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            # Try candidates: 0, half, 3/4
            candidates = [0.0, old * 0.5, old * 0.75]
            best_val = old
            best_local_M = conv.max()

            for x in candidates:
                if abs(x - old) < 1e-12:
                    continue
                delta = x - old
                # Temporarily set a[i] = x in slice for correct computation
                a_slice[i - (lo - i)] = x
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_M = conv.max()
                if new_M < best_local_M:
                    best_local_M = new_M
                    best_val = x

                # Rollback
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if best_val != old:
                delta = best_val - old
                a_slice[i - (lo - i)] = best_val
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                sweep_improved = True
                if best_M > conv.max():
                    best_M = conv.max()
                    best_a = a.copy()

        if not sweep_improved:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 1e-15:
        a = a / a.sum()
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def spectral_init(n, parent, rng):
    """Create initialization via spectral modification of parent."""
    p = resample_seq(parent, n)
    p = np.maximum(p, 0)
    p_sum = p.sum()
    if p_sum < 1e-15:
        return np.ones(n) / n
    p = p / p_sum

    # Modify power spectrum to be flatter
    fa = np.fft.rfft(p, 2 * n)
    mag = np.abs(fa)
    phase = np.angle(fa)

    # Flatten magnitude (push toward uniform spectrum)
    target_mag = np.ones_like(mag)
    blend = 0.3
    new_mag = (1 - blend) * mag + blend * target_mag * mag.mean()
    new_fa = new_mag * np.exp(1j * phase)
    new_p = np.fft.irfft(new_fa, 2 * n)[:n]
    new_p = np.maximum(new_p, 0)
    s = new_p.sum()
    if s > 1e-15:
        new_p /= s
    else:
        new_p = p
    return new_p

        print(f"  FISTA n=2048: J={J_val:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        try_update(a_opt3, "P3-n2048")

    print(f"  Phase 3 best: J={best_score:.8f}", flush=True)

    # === PHASE 4: Final polishing ===
    p4_deadline = deadline
    remaining = p4_deadline - time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Final Polish (budget ~{remaining:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None and remaining > 20:
        # Coordinate descent on the best
        t0 = time.monotonic()
        a_cd = coord_descent_fast(best_seq, deadline=p4_deadline - 5, max_sweeps=100)
        try_update(a_cd, "P4-CD")
        print(f"  CD: J={compute_J(a_cd):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        # One more FISTA pass with tight tolerance
        if time.monotonic() < p4_deadline - 30:
            t0 = time.monotonic()
            a_final = fista(best_seq, deadline=p4_deadline - 5,
                            tol_frac_init=0.01, tol_frac_min=0.0005)
            a_final = a_final / a_final.sum()
            try_update(a_final, "P4-FISTA-tight")
            print(f"  FISTA tight: J={compute_J(a_final):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: J={best_score:.6f}", flush=True)

    total_time = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={total_time:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
