**Algorithm: "Flat Cross-Correlation" (FCC)**

1. Initialize a (e.g., from parent or random)
2. Repeat:
   a. Compute c = conv(a, a)
   b. Find the active set S = {k : c[k] >= c_max - ε}
   c. Set λ_k = 1/|S| for k in S, 0 otherwise
   d. Solve the linear system: T*a = μ/2 * 1, sum(a) = 1
      where T[i][l] = λ_{l+i}
   e. If the solution a_new has negative components, project to the non-negative cone
   f. Update a = a_new (or a blend)
   g. Check convergence
3. Return best a found

The key question is: how to handle the case where the linear system gives negative a's? In that case, we need to identify which components should be zero (active inequality constraints) and solve a reduced system.

This is essentially the KKT conditions for the problem! The active set for the non-negativity constraints determines which a[i] = 0.

So the full algorithm is:
1. Guess the set Z = {i : a[i] = 0} (zero set)
2. Guess the active set S = {k : c[k] = c_max}
3. Solve the KKT system for the remaining variables
4. Check if the solution is consistent (a[i] > 0 for i not in Z, c[k] <= c_max for k not in S)
5. If not, update the guesses

This is similar to the active set method for QP, but the problem is non-convex so convergence isn't guaranteed.

def fcc_solve(lam, n):
    """
    Solve the KKT system: T*a = (mu/2)*1, sum(a) = 1
    where T[i][l] = lam[l+i]
    
    Returns (a, mu) or (None, None) if the system is singular.
    """
    # Build augmented system
    A = np.zeros((n+1, n+1))
    b = np.zeros(n+1)
    b[n] = 1.0
    
    for i in range(n):
        for l in range(n):
            A[i, l] = lam[l + i]
        A[i, n] = -0.5
    
    A[n, :n] = 1.0
    
    try:
        sol = np.linalg.solve(A, b)
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(A, b, rcond=None)[0]
        except:
            return None, None
    
    a = sol[:n]
    mu = sol[n]
    return a, mu

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Flat Cross-Correlation (FCC) method - solves KKT conditions directly
2. Smooth gradient descent for exploration
3. LP mirror descent for polishing
4. Multi-length and multi-start exploration

import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def fcc_step(a, active_frac=0.1, max_active=None):
    """
    One FCC (Flat Cross-Correlation) step.
    Solves the KKT conditions: T*x = 1, then normalizes.
    """
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    
    # Determine active set
    if max_active is not None:
        # Top-k approach
        k = min(max_active, 2*n-1)
        threshold = np.sort(c)[-k]
        S = np.where(c >= threshold)[0]
    else:
        # Fraction-based
        threshold = np.quantile(c, 1 - active_frac)
        S = np.where(c >= threshold)[0]
    
    lam = np.zeros(2*n-1)
    lam[S] = 1.0 / len(S)
    
    # Build Hankel matrix T: T[i][l] = lam[l+i]
    # T is n x n
    # T[i][l] = lam[l+i] for i=0..n-1, l=0..n-1
    # This means row i is lam[i:i+n]
    T = np.empty((n, n))
    for i in range(n):
        T[i, :] = lam[i:i+n]
    
    # Solve T*x = 1
    ones = np.ones(n)
    try:
        x = np.linalg.solve(T, ones)
    except np.linalg.LinAlgError:
        # Try with regularization
        try:
            x = np.linalg.solve(T + 1e-10 * np.eye(n), ones)
        except:
            return None
    
    s = x.sum()
    if s < 1e-10 or abs(s) < 1e-10:
        return None
    
    a_new = x / s
    return a_new

def fcc_with_nonneg(a, active_frac=0.1, max_iters=100):
    """
    FCC with active set method for non-negativity.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    zero_set = set()
    
    for it in range(max_iters):
        # Get FCC solution
        a_new = fcc_step(a, active_frac=active_frac)
        if a_new is None:
            break
        
        # Check non-negativity
        neg = a_new < -1e-10
        if not neg.any():
            # All non-negative (within tolerance)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            return a_new
        
        # Add to zero set
        for i in np.where(neg)[0]:
            zero_set.add(i)
        
        # Solve reduced system
        free = [i for i in range(n) if i not in zero_set]
        if len(free) < 2:
            break
        
        # Build reduced system
        # T_reduced * x_free = 1 (for the free variables)
        # But we also need: sum(x_free) = 1 - sum(x_zero) = 1 (since x_zero = 0)
        # Wait, the KKT condition for free variables is:
        # T[i,l]*x[l] = mu/2 for i in free set
        # And sum(x) = 1
        # And x[i] = 0 for i in zero set

        #     sum(x_free) = 1
        
        n_free = len(free)
        A = np.zeros((n_free + 1, n_free + 1))

        # Need lambda
        c = fast_conv(a)

        for ii, i in enumerate(free):

def smooth_gradient_step(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    
    return grad

    v = np.zeros(n)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Flat Cross-Correlation (FCC) method - solves KKT conditions via Hankel system
2. Smooth gradient descent with adaptive softmax for exploration
3. Hybrid FCC+gradient iterations for robustness
4. Multi-length and multi-start exploration
5. Active-set non-negativity handling
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J_fast(a):
    """Compute J(a) quickly."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_hankel(lam, n):
    """Build Hankel matrix T where T[i,l] = lam[l+i] for i,l in 0..n-1."""
    T = np.empty((n, n), dtype=np.float64)
    for i in range(n):
        T[i, :] = lam[i:i + n]
    return T


def fcc_solve(lam, n):
    """
    Solve T*x = 1 where T is the Hankel matrix from lam.
    Returns normalized a = x/sum(x) or None on failure.
    """
    ones = np.ones(n)
    T = build_hankel(lam, n)
    
    # Check condition number quickly
    try:
        # Use LU factorization
        x = np.linalg.solve(T, ones)
    except np.linalg.LinAlgError:
        # Add small regularization
        try:
            x = np.linalg.solve(T + 1e-12 * np.eye(n), ones)
        except np.linalg.LinAlgError:
            return None
    
    s = x.sum()
    if abs(s) < 1e-14:
        return None
    
    a_new = x / s
    return a_new


def fcc_step(a, active_frac=0.1, top_k=None):
    """
    One FCC step: determine active set, solve KKT system, return new a.
    """
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    
    if top_k is not None:
        k = min(top_k, 2 * n - 1)
        # Use top-k entries
        threshold = np.partition(c, -k)[-k]
        S = np.where(c >= threshold - 1e-15)[0]
    else:
        threshold = np.quantile(c, 1.0 - active_frac)
        S = np.where(c >= threshold)[0]
    
    if len(S) == 0:
        return None
    if len(S) > 2 * n - 1:
        S = np.arange(2 * n - 1)
    
    lam = np.zeros(2 * n - 1)
    lam[S] = 1.0 / len(S)
    
    return fcc_solve(lam, n)


def fcc_with_active_set(a, active_frac=0.1, max_iters=50, tol=1e-10):
    """
    FCC with active set method for non-negativity constraints.
    Returns optimized non-negative sequence (sum=1).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-14:
        return a
    a = a / s
    
    zero_set = set()
    
    for outer in range(max_iters):
        # Compute current convolution and active set
        c = fast_conv(a)
        c_max = c.max()
        threshold = np.quantile(c, 1.0 - active_frac)
        S = np.where(c >= threshold)[0]
        if len(S) == 0:
            S = np.array([int(np.argmax(c))])
        
        lam = np.zeros(2 * n - 1)
        lam[S] = 1.0 / len(S)
        
        # Determine free variables
        free = np.array([i for i in range(n) if i not in zero_set])
        if len(free) < 2:
            break
        
        n_free = len(free)
        
        # Build reduced system: T_reduced * x_free = (mu/2)*1, sum(x_free) = 1
        A = np.zeros((n_free + 1, n_free + 1))
        b_rhs = np.zeros(n_free + 1)
        b_rhs[n_free] = 1.0
        
        for ii in range(n_free):
            i = free[ii]
            for ll in range(n_free):
                l = free[ll]
                A[ii, ll] = lam[l + i]
            A[ii, n_free] = -0.5
        A[n_free, :n_free] = 1.0
        
        try:
            sol = np.linalg.solve(A, b_rhs)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(A, b_rhs, rcond=None)[0]
            except Exception:
                break
        
        x_free = sol[:n_free]
        
        # Check non-negativity of free variables
        neg_mask = x_free < -tol
        if neg_mask.any():
            # Add most negative to zero set
            worst = free[np.argmin(x_free)]
            zero_set.add(int(worst))
            continue
        
        # Clip small negatives
        x_free = np.maximum(x_free, 0.0)
        s_free = x_free.sum()
        if s_free < 1e-14:
            break
        
        # Build full solution
        a_new = np.zeros(n)
        a_new[free] = x_free
        a_new = a_new / a_new.sum()
        
        # Check convergence
        if np.max(np.abs(a_new - a)) < tol:
            a = a_new
            break
        
        a = a_new
    
    return a


def fcc_blend(a, a_fcc, alpha=0.5):
    """Blend current a with FCC solution for stability."""
    a_new = (1 - alpha) * a + alpha * a_fcc
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-14:
        return a
    return a_new / s


def run_fcc_optimization(a, max_iters=500, deadline=None, active_frac=0.1, alpha=0.3, verbose=False):
    """
    Run FCC optimization with blending and active set handling.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-14:
        return a
    a = a / s
    
    best_a = a.copy()
    best_M = compute_M(a)
    stall = 0
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # FCC step with active set method
        a_new = fcc_with_active_set(a, active_frac=active_frac, max_iters=20)
        
        if a_new is None or np.allclose(a_new, a, atol=1e-14):
            stall += 1
        else:
            # Blend for stability
            a_new = fcc_blend(a, a_new, alpha=alpha)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            
            M_new = compute_M(a_new)
            if M_new < best_M - 1e-14:
                best_M = M_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
            
            a = a_new
        
        if verbose and it % 50 == 0:
            print(f"        FCC iter {it}: M={compute_M(a):.10f} best={best_M:.10f}", flush=True)
        
        if stall > 50:
            break
    
    return best_a


def smooth_grad_step(a, tau):
    """Compute gradient of soft-max approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (where valid)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo..hi] += 2*pk * a[k-hi : k-lo+1][::-1]
        idx = k - np.arange(hi - lo + 1)
        grad[lo:hi + 1] += 2.0 * pk * a[idx]
    
    return grad


def run_smooth_gradient(a, max_iters=500, deadline=None, tau0=10.0, tau1=300.0, momentum=0.9):
    """Smooth gradient descent with adaptive tau and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        tau = tau0 + (tau1 - tau0) * (frac ** 0.7)
        lr = 0.8 * (0.005 / 0.8) ** frac
        
        grad = smooth_grad_step(a, tau)
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-14:
            a_new = a.copy()
            v *= 0.5
        else:
            a_new = a_new / s
        a = a_new
        
        M = compute_M(a)
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
    
    return best_a
