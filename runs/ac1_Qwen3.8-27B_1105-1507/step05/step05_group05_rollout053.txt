    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_at_peak(a, b, top_k=3):
    """
    Compute a (sub)gradient of max(conv(a,a)) at a.
    Uses top-k peaks for robustness. Returns gradient vector of length n.
    """
    n = len(a)
    g = np.zeros(n)
    # Get top-k indices
    k = min(top_k, 2 * n - 1)
    top_idx = np.argpartition(b, -k)[-k:]
    top_idx = top_idx[np.argsort(b[top_idx])[::-1]]
    # Weights: based on how close to max
    b_top = b[top_idx]
    w = b_top - b_top.min() + 0.1
    w = w / w.sum()
    for wi, ki in zip(w, top_idx):
        m_min = max(0, ki - n + 1)
        m_max = min(n - 1, ki)
        if m_max >= m_min:
            # g[m] += wi * 2 * a[ki - m] for m in [m_min, m_max]
            seg = a[ki - m_max:ki - m_min + 1][::-1]
            g[m_min:m_max + 1] += wi * 2.0 * seg
    return g


def run_subgrad_search(a, max_iters, deadline, top_k=3, init_lr=0.5):
    """
    Subgradient descent with backtracking line search on the simplex.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    lr = init_lr
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        g = subgradient_at_peak(a, b, top_k)
        # Project to tangent space (sum-zero)
        g = g - g.mean()

        # Backtracking line search
        improved = False
        for t in [lr, lr * 0.5, lr * 0.25, lr * 0.125, lr * 0.0625]:
            a_trial = a - t * g
            a_trial = project_simplex(a_trial)
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * 0.9999:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
            lr = min(lr * 1.05, 2.0)
        else:
            no_improve += 1
            lr *= 0.7
            if lr < 1e-5 or no_improve > 100:
                break

    return best_a


def run_fista_subgrad(a, max_iters, deadline, top_k=5):
    """
    FISTA (accelerated proximal gradient) with damping for the non-convex case.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    y = a.copy()
    t_k = 1.0
    lr_base = 0.3
    damping_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Compute gradient at y
        b_y = fast_conv(y)
        g = subgradient_at_peak(y, b_y, top_k)
        g = g - g.mean()

        # Learning rate
        lr = lr_base / (np.sqrt(it + 1) + 1.0)

        # Gradient step
        z = y - lr * g

        # Proximal: project to simplex
        a_new = project_simplex(z)

        # Track best
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            damping_count = 0
        else:
            damping_count += 1

        # FISTA momentum update with damping
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        if damping_count > 20:
            # Damp momentum
            momentum = 0.5
        else:
            momentum = 1.0
        y = a_new + momentum * ((t_k - 1.0) / t_new) * (a_new - a)
        y = project_simplex(y)
        a = a_new
        t_k = t_new if damping_count <= 20 else 1.0

    return best_a


def coordinate_update_fast(a, i, b):
    """
    Golden-section search for optimal a[i] given all other coordinates fixed.
    Uses precomputed convolution b for efficiency.
    """
    n = len(a)
    x_cur = a[i]
    S = a.sum()

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=np.float64)
        mask[k_lo:k_hi + 1] = -np.inf
        base_max = mask.max()
        if np.isneginf(base_max):
            base_max = 0.0
    else:
        base_max = 0.0

    # Precompute for affected positions
    affected = np.arange(k_lo, k_hi + 1)
    c_vals = np.empty(len(affected))
    slopes = np.empty(len(affected))
    quad_at = -1

    for idx, k in enumerate(affected):
        j = k - i
        if 0 <= j < n:
            if k == 2 * i:
                c_vals[idx] = b[k] - x_cur * x_cur
                slopes[idx] = 0.0
                quad_at = idx
            else:
                c_vals[idx] = b[k] - 2.0 * x_cur * a[j]
                slopes[idx] = 2.0 * a[j]
        else:
            c_vals[idx] = b[k]
            slopes[idx] = 0.0

    def obj(x):
        new_sum = S - x_cur + x
        if new_sum <= 1e-10:
            return 1e10
        vals = c_vals + slopes * x
        m = base_max
        for idx2 in range(len(vals)):
            vv = vals[idx2]
            if vv > m:
                m = vv
        if quad_at >= 0:
            qv = c_vals[quad_at] + x * x
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)

    lo = 0.0
    hi = max(x_cur * 3.0, 0.5 / n + x_cur, 0.01)

    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)

    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coordinate_descent(a, max_sweeps, deadline, seed=0):
    """Coordinate descent with exact 1D minimization per coordinate."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        b = fast_conv(a)
        sweep_improved = False
        for i in order:
            x_opt = coordinate_update_fast(a, i, b)
            if abs(x_opt - a[i]) > 1e-10:
                a[i] = x_opt
                sweep_improved = True
                b = fast_conv(a)  # Update convolution

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 20:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy() + 1e-8)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    # Uniform
    inits.append(np.ones(n) * 0.01)

    # Cosine modes
    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t), 0.01))

    # Chirp (quadratic phase)
    inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * t * t * n / 2), 0.01))

    # Anti-symmetric: high on one side, low on other
    inits.append(np.maximum(0.5 + 0.5 * np.sin(np.pi * t), 0.01))
    inits.append(np.maximum(0.3 + 0.7 * t, 0.01))  # linear ramp
    inits.append(np.maximum(0.3 + 0.7 * (1 - t), 0.01))  # linear ramp down

    # Two humps
    inits.append(np.maximum(0.5 + 0.5 * np.cos(4 * np.pi * t), 0.01))

    # Gaussian bump (off-center to break symmetry)
    inits.append(np.maximum(0.1 + 0.9 * np.exp(-8 * (t - 0.3) ** 2), 0.01))
    inits.append(np.maximum(0.1 + 0.9 * np.exp(-8 * (t - 0.7) ** 2), 0.01))

    # Smoothed random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # "V" shape (minimum in middle)
    inits.append(0.2 + 0.8 * np.abs(2 * t - 1))

    # Periodic binary-like (smoothed)
    period = max(4, n // 16)
    binary = (np.arange(n) % period) < period // 2
    b_smooth = np.convolve(binary.astype(float), np.ones(5) / 5, mode='same')
    inits.append(np.maximum(b_smooth + 0.05, 0.01))

    return inits


class Tracker:
    """Tracks best sequence found."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining subgradient descent (O(n) gradient via shift structure),
    FISTA acceleration, coordinate descent polish, and diverse structured initialization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) * 0.5
        tracker.update(parent)

    best_n = tracker.best.size if tracker.best is not None else 512

    # === PHASE 1: Subgradient search from diverse inits ===
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Subgradient Search (budget=~400s)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [best_n, best_n // 2, best_n * 2]
    lengths = [l for l in lengths if 128 <= l <= 2048]
    # Add some variety
    for alt in [384, 512, 768, 1024]:
        if alt not in lengths and 128 <= alt <= 2048:
            lengths.append(alt)
    lengths = sorted(set(lengths))

    time_per_n = max(30, min(120, 350 / len(lengths)))

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break
        phase_dl = min(time.monotonic() + time_per_n, deadline - 280)
        if time.monotonic() >= phase_dl:
            continue

        t0 = time.monotonic()
        inits = make_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_for_n = None
        best_M = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            # Subgradient search
            a_opt = run_subgrad_search(a0, 2000, phase_dl, top_k=3, init_lr=0.5)

            M_val = fast_conv(a_opt).max()
            if M_val < best_M:
                best_M = M_val
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: M={best_M:.8f} J={J_val:.6f} ({elapsed:.1f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("Phase 1")
    print(f"Phase 1 total: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: FISTA on best ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: FISTA Acceleration (budget=~200s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 200:
        a_fista = run_fista_subgrad(tracker.best, 3000, min(time.monotonic() + 100, deadline - 180), top_k=5)
        tracker.update(a_fista)

        # Second FISTA with different params
        if time.monotonic() < deadline - 100:
            a_fista2 = run_fista_subgrad(a_fista, 2000, min(time.monotonic() + 80, deadline - 80), top_k=3)
            tracker.update(a_fista2)

    tracker.report("Phase 2")
    print(f"Phase 2 total: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Coordinate Descent Polish ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Coordinate Descent Polish (budget=~200s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 150:
        a_cd = run_coordinate_descent(tracker.best, 100, min(time.monotonic() + 100, deadline - 130), seed=42)
        tracker.update(a_cd)

        # Perturbation + CD
        if time.monotonic() < deadline - 60:
            a_pert = tracker.best + rng.normal(0, 0.02, tracker.best.size)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            a_cd2 = run_coordinate_descent(a_pert, 80, min(time.monotonic() + 50, deadline - 50), seed=123)
            tracker.update(a_cd2)

    tracker.report("Phase 3")
    print(f"Phase 3 total: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Alternating subgradient + CD ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Alternating Refinement (budget=~150s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 50:
        a_cur = tracker.best.copy()
        for round_idx in range(10):
            if time.monotonic() > deadline - 30:
                break

            # Subgradient step
            a_sub = run_subgrad_search(a_cur, 500, min(time.monotonic() + 20, deadline - 40), top_k=3)
            # CD step
            a_cd = run_coordinate_descent(a_sub, 30, min(time.monotonic() + 15, deadline - 35), seed=round_idx * 111)
            tracker.update(a_cd)

            # Check if improved
            if tracker.best is not a_cd and fast_conv(a_cd).max() > fast_conv(a_cur).max():
                a_cur = a_cur  # keep old
            else:
                a_cur = a_cd

    tracker.report("Phase 4")
    print(f"Phase 4 total: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Try larger dimension ===
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Dimension Exploration (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        for mult in [2, 3]:
            new_n = tracker.best.size * mult
            if new_n > 2048:
                continue
            if time.monotonic() > deadline - 20:
                break
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext_opt = run_subgrad_search(a_ext, 1000, min(time.monotonic() + 20, deadline - 15), top_k=3)
            a_ext_cd = run_coordinate_descent(a_ext_opt, 30, min(time.monotonic() + 10, deadline - 10), seed=777)
            tracker.update(a_ext_cd)

    # === PHASE 6: Final coordinate descent polish ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        a_final = run_coordinate_descent(tracker.best, 50, deadline - 3, seed=99999)
        tracker.update(a_final)

    tracker.report("FINAL")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512) * 0.5
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
