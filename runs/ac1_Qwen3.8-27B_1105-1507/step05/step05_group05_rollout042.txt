def spectral_flatten(a, n_iter=100, target='flat', alpha=1.0):
    """Spectral flattening: iteratively make |A| flat in frequency domain."""
    a = np.maximum(a, 0).copy()
    a = a / a.sum()
    n = len(a)
    
    # FFT size
    size = 1
    while size < 2 * n:
        size <<= 1
    
    for it in range(n_iter):
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target magnitude
        if target == 'flat':
            # Flat magnitude (all equal)
            target_mag = np.full_like(mag, mag.mean())
        elif target == 'smooth':
            # Allow some low-frequency content
            freqs = np.fft.rfftfreq(size) * size
            target_mag = mag.mean() * (1 + 0.5 * np.exp(-freqs / 10))
        
        # Blend current and target
        new_mag = alpha * target_mag + (1 - alpha) * mag
        new_mag = np.maximum(new_mag, 1e-10)
        
        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]
        
        # Project: non-negative, sum=1
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
    
    return a
```

def fista_smooth_max(a, max_iter, tau, deadline):
    """FISTA on smooth max approximation of max_k (a*a)[k] on the simplex."""
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-10: a = np.ones_like(a) / len(a)
    else: a = a / s
    n = len(a)
    
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_val = fast_conv(a).max()
    
    for it in range(max_iter):
        if time.monotonic() > deadline: break
        
        # Compute smooth max and gradient
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        smooth_val = b_max + np.log(exp_b.sum()) / tau
        
        # Gradient
        grad = np.zeros(n)
        for k in range(2*n-1):
            pk = p[k]
            if pk < 1e-12: continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # grad[i] += 2 * pk * y[k-i] for i in [lo, hi]
            grad[lo:hi+1] += 2.0 * pk * y[k-hi:k-lo+1][::-1]
        
        # Step size (Lipschitz constant)
        L = 4.0 * tau * n  # upper bound
        step = 1.0 / L
        
        # FISTA update
        x_new = y - step * grad
        x_new = proj_simplex(x_new)
        
        t_new = (1 + np.sqrt(1 + 4 * t_k**2)) / 2
        y = x_new + ((t_k - 1) / t_new) * (x_new - a)
        t_k = t_new
        a = x_new
        
        # Track best
        M = fast_conv(a).max()
        if M < best_val:
            best_val = M
            best_a = a.copy()
    
    return best_a

def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a novel spectral flattening
approach combined with FISTA (accelerated smooth gradient descent), exact coordinate
descent, and LP mirror descent. The key innovation is working in the frequency domain
to iteratively flatten the magnitude spectrum, which directly reduces the peak
autocorrelation in a manner that single-coordinate or simple gradient methods cannot.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} in O(n) time."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def spectral_flatten(a, n_iter=500, target='flat', alpha_min=0.05, alpha_max=0.5,
                     deadline=None, verbose=False):
    """
    Spectral flattening: iteratively flatten |A| in frequency domain to reduce
    peak autocorrelation. Uses adaptive blending factor.
    """
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(n_iter):
        if deadline and time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)
        mag2 = mag ** 2
        
        # Target magnitude profile
        if target == 'flat':
            target_mag = np.full_like(mag, mag.mean())
        elif target == 'flat_no_dc':
            target_mag = np.full_like(mag, mag[1:].mean() if len(mag) > 1 else mag[0])
            target_mag[0] = mag[0]  # keep DC
        elif target == 'smooth_low':
            freqs = np.arange(len(mag))
            target_mag = mag[1:].mean() if len(mag) > 1 else mag[0]
            target_mag_arr = np.full_like(mag, target_mag)
            # Allow more energy at low frequencies
            target_mag_arr[0] = mag[0]
            for j in range(1, min(10, len(mag))):
                target_mag_arr[j] = mag[j] * 0.5 + target_mag * 0.5
            target_mag = target_mag_arr
        else:
            target_mag = np.full_like(mag, mag.mean())
        
        # Adaptive blending: more aggressive early, less aggressive later
        frac = it / max(1, n_iter)
        alpha = alpha_max * (1 - frac) ** 2 + alpha_min
        
        new_mag = alpha * target_mag + (1 - alpha) * mag
        new_mag = np.maximum(new_mag, 1e-10)
        
        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]
        
        # Project
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            break
    
    return best_a


def spectral_oscillation(a, n_iter=300, deadline=None):

    s = a.sum()

        a = np.ones_like(a)

def fista_smooth(a, max_iters, tau_start=20.0, tau_end=200.0, deadline=None):
    """
    FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) on the
    log-sum-exp smooth approximation of max_k (a*a)[k].
    """
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    # Estimate Lipschitz constant
    L = 4.0 * np.sqrt(n)
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        
        # Compute smooth max gradient at y
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient: grad[i] = sum_k p[k] * 2 * y[k-i] (with boundary)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-13:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # For i in [lo, hi]: grad[i] += 2*pk*y[k-i]
            # y[k-lo : k-hi+1] in reverse
            grad[lo:hi+1] += 2.0 * pk * y[k-hi:k-lo+1][::-1]
        
        # FISTA step
        step = 1.0 / L
        x_new = y - step * grad
        x_new = proj_simplex(x_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        t_k = t_new
        a = x_new
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            # Restart FISTA momentum
            y = a.copy()
            t_k = 1.0
            no_improve = 0
    
    return best_a


def coordinate_update_fast(a, i, b):
    """
    Fast coordinate update using precomputed convolution b.
    Returns optimal value for a[i] (with normalization).
    """
    n = len(a)
    x_cur = a[i]
    S = a.sum()
    
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0
    
    # Affected coefficients
    affected = np.arange(k_lo, k_hi + 1)
    c_vals = np.zeros(len(affected))
    slopes = np.zeros(len(affected))
    quad_idx = -1
    
    for idx, k in enumerate(affected):
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2 * x_cur * a[j]
                slopes[idx] = 2.0 * a[j]
            else:
                c_vals[idx] = b[k]
    
    def obj(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 1e-15:
            return 1e10
        # Max of affected
        vals = c_vals + slopes * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if quad_idx >= 0:
            qv = c_vals[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)
    
    # Golden section search
    lo = 0.0
    hi = max(x_cur * 3.0, np.sqrt(max(b.max(), 0)) + 0.5, 0.1)
    
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a, max_sweeps, deadline, seed=0):
    """Run coordinate descent with exact 1D subproblem solves."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    rng = np.random.default_rng(seed)
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        
        b = fast_conv(a)
        sweep_changed = False
        
        for i in order:
            x_opt = coordinate_update_fast(a, i, b)
            if abs(x_opt - a[i]) > 1e-10:
                a[i] = x_opt
                sweep_changed = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_changed:
            no_improve += 5
        if no_improve > 30:
            break
    
    return best_a


def run_block_cd(a, block_size, max_iters, deadline, seed=0):
    """
    Block coordinate descent: update a random block of coordinates simultaneously
    using a small 1D line search over the block parameters.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    rng = np.random.default_rng(seed)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Pick a random block
        start = rng.integers(0, n)
        end = min(start + block_size, n)
        if end - start < 2:
            end = start + 2 if start + 2 <= n else n
            start = end - 2
        
        block = a[start:end]
        S_rest = s - block.sum()
        
        # The block contributes to the convolution in a structured way
        # Instead of full optimization, do a few random perturbations
        # and keep the best
        best_block = block.copy()
        best_val = best_M  # approximate
        
        # Generate perturbations
        for trial in range(5):
            if trial == 0:
                perturbation = np.zeros_like(block)
            else:
                perturbation = rng.normal(0, 0.05 * block.max(), block.size)
            
            new_block = np.maximum(block + perturbation, 0)
            new_block_sum = new_block.sum()
            if new_block_sum < 1e-10:
                continue
            
            # Compute new total sum
            new_total = S_rest + new_block_sum
            if new_total < 1e-10:
                continue
            
            # Build new sequence
            a_new = a.copy()
            a_new[start:end] = new_block
            a_new = a_new / new_total
            
            M_new = fast_conv(a_new).max()
            if M_new < best_val:
                best_val = M_new
                best_block = new_block
                a = a_new
                s = new_total
        
        if best_val < best_M * (1 - 1e-10):
            best_M = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            break
    
    return best_a


def build_A_dense(n, a):
    """Build dense constraint matrix for LP (small n only)."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 2048:
        return None
    A = build_A_dense(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'max_iter': 100000})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        improved = False
        base_t = max(0.05, 0.5 * (1 - 0.2 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01, 0.005]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    
    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t), 0.01))
    
    # "Comb" patterns (alternate high/low)
    for period in [2, 3, 4]:
        comb = np.zeros(n)
        comb[::period] = 1.0
        inits.append(np.maximum(comb + 0.05 * rng.uniform(0, 1, n), 0.01))
    
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Zeros at center (to break the autocorrelation peak)
    center_zero = np.ones(n)
    center_zero[n//4:3*n//4] = 0.1
    inits.append(center_zero + 0.01 * rng.uniform(0, 1, n))
    
    return inits

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a novel spectral flattening
approach combined with FISTA (accelerated smooth gradient descent), exact coordinate
descent, and LP mirror descent. The key innovation is working in the frequency domain
to iteratively flatten the magnitude spectrum, which directly reduces the peak
autocorrelation in a manner that single-coordinate or simple gradient methods cannot.
Additionally uses adaptive target profiles and multiple blending strategies.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} in O(n log n) time."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho_info = np.where(cond)[0]
    if len(rho_info) == 0:
        return np.full(n, s / n)
    rho = rho_info[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def spectral_flatten(a, n_iter=500, target='flat', alpha_min=0.05, alpha_max=0.5,
                     deadline=None, verbose=False):

    Spectral flattening: iteratively flatten |A| in frequency domain to reduce
    peak autocorrelation. Uses adaptive blending factor.
    """
    a = np.maximum(a, 0).copy()

    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s

    while size < 2 * n - 1:
        size <<= 1

    best_M = fast_conv(a).max()

        if deadline and time.monotonic() > deadline:
            break

        mag = np.abs(A)
        phase = np.angle(A)

        if target == 'flat':
            target_mag = np.full_like(mag, mag.mean())
        elif target == 'flat_no_dc':
            if len(mag) > 1:
                target_mag = np.full_like(mag, mag[1:].mean())
                target_mag[0] = mag[0]
            else:
                target_mag = mag.copy()
        elif target == 'smooth_low':
            target_mag = np.full_like(mag, mag[1:].mean() if len(mag) > 1 else mag[0])
            target_mag[0] = mag[0]
            for j in range(1, min(10, len(mag))):
                target_mag[j] = 0.3 * mag[j] + 0.7 * target_mag[j]
        elif target == 'bump':
            # Target: lower at mid-frequencies, higher at low freq
            freqs = np.arange(len(mag))
            target_mag = mag.mean() * (1.0 + 0.8 * np.exp(-freqs.astype(float) / max(1, len(mag) // 8)))
        else:
            target_mag = np.full_like(mag, mag.mean())

        alpha = alpha_max * (1.0 - frac) ** 1.5 + alpha_min

        new_mag = np.maximum(new_mag, 1e-10)

        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            # Try a kick: random perturbation
            a = a + 0.02 * np.random.default_rng(it).normal(0, 1, n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0

    return best_a


def spectral_targeted(a, n_iter=300, deadline=None):
    """
    Targeted spectral reduction: at each step, identify the peak lag in the
    autocorrelation and reduce |A| at frequencies that constructively interfere
    at that lag.
    """
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(n_iter):
        if deadline and time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)

        r = fast_conv(a)
        r_max = r.max()

        # Find top peaks
        top_k = np.argsort(r)[-min(5, len(r)):]

        # Compute which frequency bins contribute constructively to each peak
        freqs = np.arange(len(mag))
        reduction = np.zeros_like(mag)
        for k in top_k:
            # exp(2*pi*i*m*k/size) is the phase at frequency m for lag k
            cos_vals = np.cos(2 * np.pi * freqs * k / size)
            # Reduce where cos_vals > 0 (constructive)
            reduction += np.maximum(cos_vals, 0)

        # Apply reduction inversely proportional to magnitude
        eps = 1e-10
        new_mag = mag - 0.05 * reduction * (mag / (mag.mean() + eps))
        new_mag = np.maximum(new_mag, 0.1 * mag.min() + 1e-10)

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new

        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()

    return best_a


def fista_smooth(a, max_iters, tau_start=20.0, tau_end=200.0, deadline=None):
    """
    FISTA on the log-sum-exp smooth approximation of max_k (a*a)[k].
    Uses Nesterov acceleration for O(1/k^2) convergence rate.
    """
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    # Lipschitz constant: L = 2 * max_k ||Q_k||_2 <= 2*sqrt(n)
    # But with the smooth max, effective L is larger; use a safe bound
    L = 4.0 * np.sqrt(n)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)

        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        # Gradient via direct computation
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-13:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * pk * y[k - hi:k - lo + 1][::-1]

        # Adaptive step size: start small, grow if no change in objective
        step = 1.0 / L
        x_new = y - step * grad
        x_new = proj_simplex(x_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        t_k = t_new
        a = x_new

        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            y = a.copy()
            t_k = 1.0
            no_improve = 0

    return best_a


def coordinate_update_fast(a, i, b):
    """
    Fast coordinate update using precomputed convolution b.
    Returns optimal value for a[i] (maintaining normalization).
    """
    n = len(a)
    x_cur = a[i]
    S = a.sum()

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0

    affected = np.arange(k_lo, k_hi + 1)
    c_vals = np.zeros(len(affected))
    slopes = np.zeros(len(affected))
    quad_idx = -1

    for idx, k in enumerate(affected):
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2.0 * x_cur * a[j]
                slopes[idx] = 2.0 * a[j]
            else:
                c_vals[idx] = b[k]

    def obj(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 1e-15:
            return 1e10
        vals = c_vals + slopes * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if quad_idx >= 0:
            qv = c_vals[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)

    lo = 0.0
    hi = max(x_cur * 3.0, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.1)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)

    for _ in range(35):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)


def run_coordinate_descent(a, max_sweeps, deadline, seed=0):
    """Run coordinate descent with exact 1D subproblem solves."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    rng = np.random.default_rng(seed)
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)

        b = fast_conv(a)
        sweep_changed = False

        for i in order:
            x_opt = coordinate_update_fast(a, i, b)
            if abs(x_opt - a[i]) > 1e-10:
                a[i] = x_opt
                sweep_changed = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_changed:
            no_improve += 5
        if no_improve > 40:
            break

    return best_a


def build_A_dense(n, a):
    """Build dense constraint matrix for LP."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 2048:
        return None
    A = build_A_dense(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'max_iter': 200000})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        improved = False
        base_t = max(0.05, 0.5 * (1.0 - 0.2 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.03, 0.01, 0.005]:
            a_trial = np.maximum((1.0 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 25:
                break
        else:
            no_improve = 0

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t), 0.01))

    # Comb patterns
    for period in [2, 3, 4]:
        comb = np.zeros(n)
        comb[::period] = 1.0
        inits.append(np.maximum(comb + 0.05 * rng.uniform(0, 1, n), 0.01))

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Center dip
    center_dip = 1.0 - 0.5 * np.exp(-((t - 0.5) ** 2) / 0.05)
    inits.append(np.maximum(center_dip + 0.01 * rng.uniform(0, 1, n), 0.01))

    return inits


def score_tracker_update(tracker, a, label=""):
    """Update score tracker with new candidate."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0 or a.sum() < 0.01:
        return
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < tracker['best_score']:
        tracker['best'] = a.copy()
        tracker['best_score'] = score
        elapsed = time.monotonic() - tracker['t_start']
        print(f"  ** {label} J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)


def run_combined_pipeline(a, deadline, seed=0):
    """Run a combined spectral + FISTA + CD pipeline on a single sequence."""
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_M = fast_conv(a).max()
    n = len(a)

    # Step 1: Spectral flattening (fast)
    if time.monotonic() < deadline - 5:
        a1 = spectral_flatten(a, n_iter=200, target='flat',
                              alpha_min=0.05, alpha_max=0.4, deadline=deadline)
        M1 = fast_conv(a1).max()
        if M1 < best_M:
            best_M = M1
            best_a = a1.copy()
            a = a1

        if time.monotonic() < deadline - 3:
            a1b = spectral_flatten(a1, n_iter=100, target='flat_no_dc',
                                   alpha_min=0.03, alpha_max=0.3, deadline=deadline)
            M1b = fast_conv(a1b).max()
            if M1b < best_M:
                best_M = M1b
                best_a = a1b.copy()
                a = a1b

    # Step 2: FISTA
    if time.monotonic() < deadline - 3:
        a2 = fista_smooth(a, 1500, tau_start=20, tau_end=150, deadline=deadline)
        M2 = fast_conv(a2).max()
        if M2 < best_M:
            best_M = M2
            best_a = a2.copy()
            a = a2

    # Step 3: Coordinate descent (polish)
    if time.monotonic() < deadline - 1:
        a3 = run_coordinate_descent(a, 100, deadline, seed=seed)
        M3 = fast_conv(a3).max()
        if M3 < best_M:
            best_M = M3
            best_a = a3.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining spectral flattening, FISTA, coordinate descent,
    and LP mirror descent. Key innovation: spectral domain operations for fast
    coarse optimization of the autocorrelation peak.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = {'best': None, 'best_score': float('inf'), 't_start': time.monotonic()}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            score_tracker_update(tracker, parent, "parent")
            print(f"Parent: n={parent.size}, J={tracker['best_score']:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        score_tracker_update(tracker, parent, "init")

    # === PHASE 1: Spectral Flattening (fast, novel) ===
    p1_start = time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Spectral Flattening (budget=~300s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    lengths = [384, 512, 640, 768]
    time_alloc = {384: 60, 512: 120, 640: 60, 768: 60}

    for n in lengths:
        if time.monotonic() > deadline - 1000:
            break
        t_alloc = time_alloc.get(n, 50)
        phase_deadline = min(time.monotonic() + t_alloc, deadline - 950)

        if time.monotonic() >= phase_deadline:
            continue

        t0 = time.monotonic()
        inits = diverse_inits(n, rng, tracker['best'] if tracker['best'] is not None else parent)
        best_for_n = None
        best_M_for_n = float('inf')

        for idx, a0 in enumerate(inits[:6]):
            if time.monotonic() > phase_deadline:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            # Try multiple spectral targets
            for target in ['flat', 'flat_no_dc']:
                if time.monotonic() > phase_deadline:
                    break
                a_spec = spectral_flatten(a0, n_iter=400, target=target,
                                          alpha_min=0.04, alpha_max=0.5,
                                          deadline=phase_deadline)
                M_val = fast_conv(a_spec).max()
                if M_val < best_M_for_n:
                    best_M_for_n = M_val
                    best_for_n = a_spec.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: M={best_M_for_n:.8f} J={J_val:.6f} ({elapsed:.1f}s)", flush=True)
            score_tracker_update(tracker, best_for_n, f"spec n={n}")

    print(f"Phase 1 done: J={tracker['best_score']:.6f} ({time.monotonic() - p1_start:.1f}s)", flush=True)

    # === PHASE 2: FISTA (accelerated smooth gradient) ===
    p2_start = time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: FISTA (budget=~300s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker['best'] is not None and time.monotonic() < deadline - 700:
        a_base = tracker['best'].copy()

        for tau_s, tau_e, label in [(10, 80, 'low'), (30, 200, 'mid'), (60, 300, 'high')]:
            if time.monotonic() > deadline - 700:
                break
            a_fista = fista_smooth(a_base, 3000, tau_start=tau_s, tau_end=tau_e,
                                   deadline=min(time.monotonic() + 90, deadline - 650))
            score_tracker_update(tracker, a_fista, f"fista-{label}")
            if a_fista.sum() > 0.01:
                a_base = a_fista

    print(f"Phase 2 done: J={tracker['best_score']:.6f} ({time.monotonic() - p2_start:.1f}s)", flush=True)

    # === PHASE 3: Coordinate Descent (exact, main workhorse) ===
    p3_start = time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Coordinate Descent (budget=~500s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker['best'] is not None and time.monotonic() < deadline - 200:
        # Main long CD run
        a_cd = run_coordinate_descent(tracker['best'], 300,
                                      min(time.monotonic() + 350, deadline - 150),
                                      seed=seed)
        score_tracker_update(tracker, a_cd, "CD main")

        # CD from perturbed starting point
        if time.monotonic() < deadline - 150:
            a_pert = tracker['best'] + rng.normal(0, 0.03, tracker['best'].size)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            a_cd2 = run_coordinate_descent(a_pert, 150,
                                           min(time.monotonic() + 150, deadline - 100),
                                           seed=99999)
            score_tracker_update(tracker, a_cd2, "CD pert")

    print(f"Phase 3 done: J={tracker['best_score']:.6f} ({time.monotonic() - p3_start:.1f}s)", flush=True)

    # === PHASE 4: LP Mirror Descent ===
    p4_start = time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: LP Mirror Descent (budget=~150s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker['best'] is not None and time.monotonic() < deadline - 50:
        n_best = tracker['best'].size
        if n_best <= 2048:
            a_lp = run_lp_mirror_descent(tracker['best'], 200,
                                         min(time.monotonic() + 120, deadline - 40))
            score_tracker_update(tracker, a_lp, "LP")

            if time.monotonic() < deadline - 40:
                a_lp2 = run_lp_mirror_descent(a_lp, 100, deadline - 20)
                score_tracker_update(tracker, a_lp2, "LP2")

    print(f"Phase 4 done: J={tracker['best_score']:.6f} ({time.monotonic() - p4_start:.1f}s)", flush=True)

    # === PHASE 5: Final Polish (combined) ===
    p5_start = time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker['best'] is not None and time.monotonic() < deadline - 10:
        # Quick combined pipeline
        a_final = run_combined_pipeline(tracker['best'], deadline - 3, seed=12345)
        score_tracker_update(tracker, a_final, "combined")

        # Quick CD
        if time.monotonic() < deadline - 5:
            a_final2 = run_coordinate_descent(a_final, 30, deadline - 2, seed=77777)
            score_tracker_update(tracker, a_final2, "final CD")

        # Try extended length if time permits
        if time.monotonic() < deadline - 30:
            new_n = tracker['best'].size * 2
            if new_n <= 2048:
                a_ext = resample_seq(tracker['best'], new_n)
                a_ext = np.maximum(a_ext, 1e-6)
                a_ext = a_ext / a_ext.sum()
                a_ext_opt = run_combined_pipeline(a_ext, deadline - 5, seed=55555)
                score_tracker_update(tracker, a_ext_opt, f"ext n={new_n}")

    print(f"Phase 5 done: J={tracker['best_score']:.6f} ({time.monotonic() - p5_start:.1f}s)", flush=True)

    # Fallback
    if tracker['best'] is None:
        tracker['best'] = np.ones(512)
        tracker['best_score'] = compute_J(tracker['best'])

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker['best_score']:.8f} n={tracker['best'].size} "
          f"time={time.monotonic() - tracker['t_start']:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker['best']]
