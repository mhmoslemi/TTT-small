def _fft_convolution(a):
    """Compute full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _softmax_obj_grad(x, T, n_fix=None):
    """
    Compute smoothed objective and gradient w.r.t. x (log-parameters).
    a = softmax(x), objective = 2n * T * log(sum exp(conv(a,a)/T))
    Gradient computed via chain rule: dx = a*(g_a - a.dot(g_a))
    """
    a = np.exp(x - x.max())
    a_sum = a.sum()
    a = a / a_sum
    n = len(a)
    conv = _fft_convolution(a)
    conv_scaled = conv / T
    max_cs = conv_scaled.max()
    exp_cs = np.exp(conv_scaled - max_cs)
    sum_exp = exp_cs.sum()
    p = exp_cs / sum_exp
    obj = 2.0 * n * T * (np.log(sum_exp) + max_cs)
    # Gradient w.r.t. a: g_a[i] = 4n * sum_k p[k] * a[k-i] * 1_{0<=k-i<n}
    g_a = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-18:
            continue
        w = 2.0 * pk
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            g_a[i_lo:i_hi + 1] += w * a[k - i_hi:k - i_lo + 1][::-1]
    g_a *= 2.0 * n
    # Chain rule: grad_x = a * (g_a - a.dot(g_a))
    ag = a.dot(g_a)
    grad_x = a * (g_a - ag)
    return obj, grad_x


def _true_objective(a):
    """Compute the true (non-smoothed) objective."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    conv = _fft_convolution(a)
    return float(2.0 * n * conv.max() / (s * s))


def _softmax(x):
    """Compute softmax of x."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def _resample(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            old = a[i]
            # Golden section search for optimal value of a[i]
            # Objective as function of a[i]: J(x) = 2n * max(conv') / (s')^2
            # where conv' is conv with a[i] changed to x
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            # Define objective for a single coordinate
            def obj_single(x_val):
                delta = x_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                new_s = s - old + x_val
                if new_s < 1e-10:
                    result = float('inf')
                else:
                    result = 2.0 * n * conv.max() / (new_s * new_s)
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
                return result

            # Try a few candidate regions
            candidates = [0.0]
            if old > 1e-10:
                candidates += [old * 0.3, old * 0.6, old * 0.8, old * 1.2, old * 1.5]
            else:
                candidates += [1e-4, 1e-3, 5e-3]

            best_val = old
            best_val_obj = obj_single(old)

            for x_try in candidates:
                x_try = max(0.0, x_try)
                if abs(x_try - old) < 1e-12:
                    continue
                obj_try = obj_single(x_try)
                if obj_try < best_val_obj:
                    best_val = x_try
                    best_val_obj = obj_try

            # Golden section around best_val
            if best_val > 0:
                gs_lo = max(0, best_val * 0.5)
                gs_hi = best_val * 1.5
                invphi2 = (np.sqrt(5.0) - 1.0) / 2.0
                for _ in range(20):
                    c1 = gs_hi - invphi2 * (gs_hi - gs_lo)
                    c2 = gs_lo + invphi2 * (gs_hi - gs_lo)
                    f1 = obj_single(c1)
                    f2 = obj_single(c2)
                    if f1 < f2:
                        gs_hi = c2
                    else:
                        gs_lo = c1
                    if abs(gs_hi - gs_lo) < 1e-12:
                        break
                best_gs = (gs_lo + gs_hi) / 2.0
                obj_gs = obj_single(best_gs)
                if obj_gs < best_val_obj:
                    best_val = best_gs
                    best_val_obj = obj_gs

            # Apply best
            if abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

    a = np.maximum(best_a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    return a


def _subgradient_optimize(a, max_iters, deadline, seed=0):
    """
    Projected subgradient with Nesterov acceleration and backtracking.
    Works on the simplex {a >= 0, sum(a) = 1}.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _true_objective(a)
    rng = np.random.default_rng(seed)

    # Nesterov momentum
    y = a.copy()
    momentum = 0.0
    step = 0.1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        conv = _fft_convolution(y)
        M = conv.max()
        # Active set: top K or within tolerance
        tol = max(1e-10, 0.01 * M)
        active = np.where(conv >= M - tol)[0]
        # Also include some random near-max for diversity
        if len(active) < 5 and len(conv) > 5:
            sorted_idx = np.argsort(conv)[::-1]
            active = np.unique(np.concatenate([active, sorted_idx[:5]]))

        # Subgradient
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += 2.0 * y[k - i_hi:k - i_lo + 1][::-1]
        grad /= len(active)

        # Nesterov update
        momentum = (it + 1) / (it + 1 + 2.5)  # decaying momentum ~ 1 - 2/(t+2.5)
        a_new = y - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            a_new = y.copy()
        else:
            a_new = a_new / a_new_sum

        # Backtracking line search
        obj_current = 2.0 * n * M  # proportional to J since sum=1
        t = step
        for _ in range(20):
            a_trial = np.maximum(y - t * grad, 0.0)
            a_trial_s = a_trial.sum()
            if a_trial_s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / a_trial_s
            conv_trial = _fft_convolution(a_trial)
            obj_trial = 2.0 * n * conv_trial.max()
            if obj_trial < obj_current - 1e-12 * t * np.linalg.norm(grad)**2:
                break
            t *= 0.5

        a_new = np.maximum(y - t * grad, 0.0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-10:
            a_new = y.copy()
        else:
            a_new = a_new / a_new_s

        # Nesterov: y = a + momentum * (a - y_old)
        y_old = y.copy()
        y = a_new + momentum * (a_new - y_old)
        y = np.maximum(y, 0.0)
        y_s = y.sum()
        if y_s < 1e-10:
            y = a_new.copy()
        else:
            y = y / y_s

        a = a_new
        step *= 0.995  # slow decay

        conv_new = _fft_convolution(a)
        score = 2.0 * n * conv_new.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score / (1.0)  # score already accounts for sum=1


def _generate_mseq_init(n, seed=0):
    """Generate initialization based on m-sequence autocorrelation structure."""
    rng = np.random.default_rng(seed)
    # Use a smooth random signal with specific spectral shaping
    freqs = np.fft.rfftfreq(n)
    # Shape spectrum to be relatively flat (white noise like)
    spectrum = np.ones(n // 2 + 1)
    # Add some structure
    for f in [0.1, 0.3, 0.5, 0.7]:
        spectrum *= (1.0 + 0.3 * np.cos(2 * np.pi * f * np.arange(n // 2 + 1)))
    spectrum = np.maximum(spectrum, 0.1)
    # Random phase
    phase = rng.uniform(0, 2 * np.pi, n // 2 + 1)
    phase[0] = 0
    spectrum_complex = spectrum * np.exp(1j * phase)
    # Inverse FFT
    sig = np.fft.irfft(spectrum_complex, n)
    # Make non-negative
    sig = np.maximum(sig + abs(sig).min() + 0.1, 0.01)
    sig = sig / sig.sum()
    return sig


def _generate_structured_init(n, seed=0):
    """Generate various structured initializations."""
    rng = np.random.default_rng(seed)
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []

    # Flat
    inits.append(np.ones(n) / n)

    # Raised cosine
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01) / n)

    # Tukey window (various r)
    for r in [0.1, 0.3, 0.5, 0.7]:
        w = np.ones(n)
        m = int(r * n // 2)
        if m > 0:
            for i in range(m):
                val = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
                w[i] = val
                w[n - 1 - i] = val
        inits.append(w / w.sum())

    # Gaussian bump
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.1, 0.2, 0.3]:
            g = np.exp(-((t - center) / width) ** 2)
            inits.append(np.maximum(g + 0.05, 0.01) / n)

    # Fourier-based (flat spectrum)
    inits.append(_generate_mseq_init(n, seed))
    inits.append(_generate_mseq_init(n, seed + 1))

    # V-shape and inverted
    inits.append(np.maximum(0.5 + 0.5 * np.abs(2 * t - 1), 0.01) / n)
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2 * t - 1), 0.01) / n)

    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for s in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.05, 0.01) / n)

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Phase 1: L-BFGS-B with softmax parameterization and temperature annealing
    Phase 2: Refined coordinate descent with golden section
    Phase 3: Subgradient with Nesterov acceleration
    Phase 4: Final polish
    """
    t_global_start = time.monotonic()
    deadline = t_global_start + budget_s - 20.0

    best_a = None
    best_score = float('inf')

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            ps = parent.sum()
            if ps > 0.01:
                best_a = parent / ps
                best_score = float(evaluate_sequence(best_a.tolist()))
                print(f"Parent loaded: n={best_a.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if best_a is None:
        best_a = np.ones(256) / 256
        best_score = _true_objective(best_a)
        print(f"No parent, using uniform n=256, J={best_score:.8f}", flush=True)

    def update_best(a):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01 or a.size == 0:
            return
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
            return True
        return False

    # === PHASE 1: L-BFGS-B with temperature annealing ===
    p1_deadline = min(t_global_start + 600, deadline - 300)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smoothed L-BFGS-B optimization (budget {p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    target_lengths = [best_a.size]
    # Add nearby lengths
    for l in [384, 512, 640, 768, 1024, 1536]:
        if l not in target_lengths and l <= 2048:
            target_lengths.append(l)
    target_lengths = sorted(set(target_lengths))

    # Limit to at most 4 lengths
    if len(target_lengths) > 4:
        # Keep parent length and spread others
        target_lengths = [target_lengths[0], target_lengths[len(target_lengths)//3],
                          target_lengths[2*len(target_lengths)//3], target_lengths[-1]]
        target_lengths = sorted(set(target_lengths))

    for new_n in target_lengths:
        if time.monotonic() > p1_deadline - 60:
            break
        remaining = p1_deadline - time.monotonic()
        print(f"\n  Optimizing n={new_n} (remaining {remaining:.0f}s)...", flush=True)

        # Generate initializations
        inits = []
        # Resample best
        a_resample = _resample(best_a, new_n)
        a_resample = np.maximum(a_resample, 1e-6)
        a_resample = a_resample / a_resample.sum()
        inits.append(a_resample)

        # Structured inits
        struct_inits = _generate_structured_init(new_n, seed=seed + new_n)
        inits.extend(struct_inits[:4])

        for idx, a_init in enumerate(inits):
            if time.monotonic() > p1_deadline - 30:
                break
            a_init = np.maximum(a_init, 1e-30)
            a_init = a_init / a_init.sum()

            # Run smoothed optimization
            a_opt, score = _optimize_smoothed(a_init, p1_deadline - 30, verbose=(idx == 0))
            update_best(a_opt)

    update_best(best_a)

    # === PHASE 2: Refined coordinate descent ===
    p2_deadline = min(t_global_start + 900, deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Fine Coordinate Descent (budget {p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None and time.monotonic() < p2_deadline - 30:
        a_cd = _coord_descent_fine(best_a, max_sweeps=2000, deadline=p2_deadline, seed=seed)
        update_best(a_cd)

        # Second pass with different seed
        if time.monotonic() < p2_deadline - 30:
            a_cd2 = _coord_descent_fine(best_a, max_sweeps=1000, deadline=p2_deadline, seed=seed + 12345)
            update_best(a_cd2)

    # === PHASE 3: Subgradient with Nesterov ===
    p3_deadline = min(t_global_start + 1200, deadline - 100)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Nesterov Subgradient (budget {p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None and time.monotonic() < p3_deadline - 30:
        a_sg, _ = _subgradient_optimize(best_a, max_iters=5000, deadline=p3_deadline, seed=seed)
        update_best(a_sg)
        # Polish after subgradient
        if time.monotonic() < p3_deadline - 20:
            a_polish = _coord_descent_fine(a_sg, max_sweeps=500, deadline=p3_deadline, seed=seed + 777)
            update_best(a_polish)

    # === PHASE 4: Final L-BFGS-B polish at lower temperature ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final Polish (budget {deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None and time.monotonic() < deadline - 30:
        # Short L-BFGS-B from current best
        a_final, _ = _optimize_smoothed(best_a, deadline - 10, verbose=True)
        update_best(a_final)

        # Very short coordinate descent
        if time.monotonic() < deadline - 10:
            a_final_cd = _coord_descent_fine(best_a, max_sweeps=500, deadline=deadline - 5, seed=seed + 9999)
            update_best(a_final_cd)

    # Final validation
    if best_a is None:
        best_a = np.ones(128)
        best_score = _true_objective(best_a)

    final_score = float(evaluate_sequence(best_a.tolist()))
    elapsed = time.monotonic() - t_global_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={len(best_a)} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    if not np.isfinite(final_score) or final_score > best_score:
        final_score = best_score

    return [float(x) for x in best_a]
