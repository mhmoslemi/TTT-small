def compute_convolve_2n_safe(a_val):
    """ Computes the full linear convolution with safety precautions """
    n_val = len(a_val)
    convolution_output = np.zeros(2 * n_val - 1)
    for i in range(2 * n_val - 1):
        # Only take defined sums with non-negative indices
        total_val = 0.0
        for j in range(max(0, i - n_val + 1), min(i+1, n_val)):

        convolution_output[i] = total_val
    return convolution_output
