    n = len(a)
    rng = np.random.default_rng(seed)
    a = a.astype(np.float64).copy()
    b = fast_conv(a)
    
    best_obj = b.max()
    best_a = a.copy()
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        
        # Build A[k] = 2*(a[k-i] - a[k-j])
        a_ki = np.zeros(2*n-1)
        a_ki[i:i+n] = a
        a_kj = np.zeros(2*n-1)
        a_kj[j:j+n] = a
        A = 2.0 * (a_ki - a_kj)

        B2i = 1 if 0 <= 2*i < 2*n-1 else 0
        Bij = -2 if 0 <= i+j < 2*n-1 else 0
        B2j = 1 if 0 <= 2*j < 2*n-1 else 0

        lo_t = -a[j]
        hi_t = a[i]

            m1 = lo_t + (hi_t - lo_t) / 3
            m2 = hi_t - (hi_t - lo_t) / 3
            # max of b[k] + A[k]*t + B[k]*t^2
            f1 = np.max(b + A*m1)
            if B2i:
                f1 = max(f1, b[2*i] + A[2*i]*m1 + B2i*m1*m1)
            if Bij:
                f1 = max(f1, b[i+j] + A[i+j]*m1 + Bij*m1*m1)
            if B2j:
                f1 = max(f1, b[2*j] + A[2*j]*m1 + B2j*m1*m1)
            
            f2 = np.max(b + A*m2)
            if B2i:
                f2 = max(f2, b[2*i] + A[2*i]*m2 + B2i*m2*m2)
            if Bij:
                f2 = max(f2, b[i+j] + A[i+j]*m2 + Bij*m2*m2)
            if B2j:
                f2 = max(f2, b[2*j] + A[2*j]*m2 + B2j*m2*m2)
            
            if f1 < f2:
                hi_t = m2
            else:
                lo_t = m1

        if abs(t_opt) < 1e-15:
            continue

        a[i] += t_opt
        a[j] -= t_opt
        if a[i] < 0: a[i] = 0
        if a[j] < 0: a[j] = 0

        if B2i:
            b[2*i] += B2i*t_opt*t_opt
        if Bij:
            b[i+j] += Bij*t_opt*t_opt
        if B2j:
            b[2*j] += B2j*t_opt*t_opt
        
        cur_obj = b.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast pairwise coordinate descent with directed/random pair selection (main workhorse)
2. Smooth subgradient descent with Nesterov momentum
3. LP mirror descent with analytic line search for polishing
4. Multi-length exploration with resampling
5. Perturbation-based restarts
6. Final extended LP polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
