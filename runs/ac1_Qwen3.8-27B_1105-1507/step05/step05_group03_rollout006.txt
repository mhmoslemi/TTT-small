"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. SQP with LP subproblem (linearized quadratic constraints) + analytic line search
2. Multi-length exploration with diverse initializations
3. Perturbation-based restarts with adaptive cooling
4. Final extended polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_structure(n):
    """Precompute sparsity pattern for convolution constraints."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
        rows.append(k)
        cols.append(n)  # M variable
    return np.array(rows), np.array(cols)


def solve_sqP_lp(a, row_idx, col_idx, time_limit=3):
    """
    Solve SQP LP: min M s.t. 2*(a*b)[k] - (a*a)[k] <= M, sum(b)=1, b>=0.
    """
    n = len(a)
    A_a = fast_conv(a)
    vals_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            vals_list.append(2.0 * a[k - j])
        vals_list.append(-1.0)
    vals = np.array(vals_list, dtype=np.float64)

    A_ub = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n + 1))
    b_ub = A_a
    A_eq = csr_matrix((np.ones(n), (list(range(n)), list(range(n)))), shape=(1, n + 1))
    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, None)] * n + [(None, None)]

    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            b_sol = np.maximum(r.x[:n], 0)
            if b_sol.sum() > 1e-10:
                b_sol = b_sol / b_sol.sum()
                return b_sol, float(r.x[n])
    except Exception:
        pass
    return None, None


def line_search(a, b, A_a, A_b, B_b):
    """Find t in [0,1] minimizing max_k f_k(t) where f_k is quadratic."""
    a_coef = A_a
    b_coef = 2.0 * (A_b - A_a)
    c_coef = A_a - 2.0 * A_b + B_b

    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(a_coef + b_coef * m1 + c_coef * m1 * m1))
        f2 = float(np.max(a_coef + b_coef * m2 + c_coef * m2 * m2))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_smooth_init(n, rng):
    """Create a smooth random initialization."""
    center = (n - 1) / 2.0
    sigma = n * (0.2 + 0.3 * rng.random())
    base = np.exp(-((np.arange(n) - center) ** 2) / (2 * sigma ** 2))
    noise = rng.normal(0, 0.05, n)
    a = np.maximum(base + noise, 0.01)
    return a / a.sum()


def make_binary_init(n, rng):
    """Create a random binary-like initialization."""
    a = rng.binomial(1, 0.5, n).astype(np.float64)
    if a.sum() < n * 0.1:
        a = rng.binomial(1, 0.5, n).astype(np.float64)
    a = np.maximum(a + 0.01, 0.01)
    return a / a.sum()


def make_cosine_init(n, rng):
    """Raised cosine with random phase perturbation."""
    x = np.linspace(0, 1, n)
    phase = rng.uniform(-0.3, 0.3)
    a = 0.5 * (1 + np.cos(np.pi * (x - 0.5) + phase))
    a = np.maximum(a + rng.normal(0, 0.02, n), 0.001)
    return a / a.sum()

                # Random block flip
                block_size = max(1, n_best // 8)
                start = rng.integers(0, max(1, n_best - block_size))
                a[start:start+block_size] *= rng.uniform(0.3, 1.5)
            elif strategy == 2:
                # Sparse large perturbation
                idx = rng.choice(n_best, size=max(1, n_best // 16), replace=False)
                a[idx] += rng.normal(0, temperature * 0.05, len(idx))
            else:
                # Smoothed noise
                noise = rng.normal(0, temperature * 0.03, n_best)
                # Simple moving average smoothing
                k = max(1, n_best // 32)
                kernel = np.ones(k) / k
                noise = np.convolve(noise, kernel, mode='same')
                a = a + noise

            a = np.maximum(a, 0.001)
            a = a / a.sum()

            sub_dl = min(time.monotonic() + per_restart, phase3_deadline)
            a_opt = run_sqP(a, 1000, sub_dl, verbose=False, lp_time=2)
            tracker.update(a_opt, f"(P3 r={r} s={strategy} T={temperature:.3f})")

            temperature *= 0.92

    # ============================================================
    # PHASE 4: Length changes from best
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length Adjustments", flush=True)
    print(f"{'='*60}", flush=True)

    phase4_deadline = min(t_start + budget_s * 0.90, deadline)

    if tracker.best is not None:
        n_best = tracker.best.size
        trial_mults = [0.75, 0.875, 1.125, 1.25, 1.5]
        for mult in trial_mults:
            if time.monotonic() > phase4_deadline:
                break
            n_try = int(n_best * mult)
            if n_try < 32 or n_try > 4096 or n_try == n_best:
                continue
            a0 = resample_seq(tracker.best, n_try)
            a0 = np.maximum(a0, 0.001)
            a0 = a0 / a0.sum()
            sub_dl = min(time.monotonic() + 40, phase4_deadline)
            a_opt = run_sqP(a0, 3000, sub_dl, verbose=False, lp_time=3)
            tracker.update(a_opt, f"(P4 n={n_try})")

    # ============================================================
    # PHASE 5: Final extended polish
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a = tracker.best.copy().astype(np.float64)
        a = a / a.sum()
        # Alternate SQP and LP-MD for robustness
        for round_num in range(4):
            if time.monotonic() > deadline:
                break
            sub_dl = min(time.monotonic() + 30, deadline)
            if round_num % 2 == 0:
                a = run_sqP(a, 3000, sub_dl, verbose=False, lp_time=2)
            else:
                a = run_lp_mirror(a, 2000, sub_dl, verbose=False, lp_time=2)
            tracker.update(a, f"(P5 round={round_num})")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
