import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_obj_grad(a, tau):
    """Compute smooth max of self-convolution and its gradient."""
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    f = b_max + np.log(exp_b.sum()) / tau
    weights = exp_b / exp_b.sum()
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(weights[i:i + n], a)
    return f, grad, weights, b


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_palindrome(a):

def build_conv_matrix(n, a, constraint_indices=None):
    """Build sparse matrix A where (A@g)[k] = (a*g)[k]."""
    if constraint_indices is not None:
        ks = constraint_indices
    else:
        ks = range(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(ks):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    n_constraints = len(ks) if constraint_indices is not None else (2 * n - 1)
    return csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))


def solve_lp_direction(a, M, top_frac=1.0, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    if top_frac < 1.0:
        threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
        active_k = np.where(b >= threshold)[0]
        if len(active_k) < 5:
            active_k = np.arange(2 * n - 1)
    else:
        active_k = np.arange(2 * n - 1)

    A = build_conv_matrix(n, a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, np.inf)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                if top_frac < 1.0:
                    ag = fast_cross_conv(a, g)
                    if ag.max() > M * (1.0 + 1e-4):
                        return None
                return g
    except Exception:
        pass
    return None

            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
            elif J_trial < current_J - 1e-12:
                a = a_trial
                current_J = J_trial
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Structured perturbation
            rng = np.random.default_rng(hash((it, no_improve)) % (2**32))
            scale = 0.2 / (1.0 + 0.1 * (it // 100))
            freqs = rng.integers(1, 6, size=4)
            phases = rng.uniform(0, 2 * np.pi, size=4)
            t_arr = np.linspace(0, 1, n, endpoint=False)
            pert = np.zeros(n)
            for f, ph in zip(freqs, phases):
                pert += np.cos(2 * np.pi * f * t_arr + ph)
            a = a + scale * pert
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            no_improve = 0
            top_frac = min(1.0, top_frac + 0.05)
        
        if it % 500 == 0 and time.monotonic() < deadline:
            pass
    
    return best_a, best_J


def generate_equiripple_init(n):
    """Generate initial guess based on equi-ripple recurrence (first n terms)."""
    a = np.zeros(n)
    a[0] = 1.0
    C = 1.0  # target convolution value
    for k in range(1, n):
        # (a*a)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j]*a[k-j] = C
        # Solve for a[k] (the largest index that appears)
        # a[k] appears in: j=k gives a[k]*a[0], and j=0 gives a[0]*a[k]
        # So 2*a[k]*a[0] + sum_{j=1}^{k-1} a[j]*a[k-j] = C
        # a[k] = (C - sum_{j=1}^{k-1} a[j]*a[k-j]) / (2*a[0])
        s = 0.0
        for j in range(1, k):
            if k - j < n and a[k - j] > 0:
                s += a[j] * a[k - j]
        a[k] = max(0.0, (C - s) / (2.0 * a[0]))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(make_palindrome(p))
    
    inits.append(np.ones(n))
    inits.append(generate_equiripple_init(n))
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t_arr - 1)), 0.01))
    inits.append(np.maximum(0.3 + 0.7 * np.exp(-4 * (t_arr - 0.5) ** 2), 0.01))
    
    # Multi-frequency
    for f in [2, 3]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * f * t_arr), 0.01))
    
    # Edge-concentrated
    inits.append(np.maximum(np.exp(-4 * t_arr ** 2) + np.exp(-4 * (t_arr - 1) ** 2) + 0.1, 0.01))
    
    # Triangular
    tri = np.maximum(1.0 - 2.0 * np.abs(t_arr - 0.5), 0.0)
    inits.append(np.maximum(tri + 0.05, 0.01))
    
    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a, label=""):
        if a is None or len(a) == 0:
            return
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = compute_J(a)
        if score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={len(a)} t={elapsed:.0f}s [{label}]", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={len(self.best)}", flush=True)


def run_coordinate_descent(a, deadline, passes=5):
    """Run cyclic coordinate descent for polishing."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(passes * 10):
        if time.monotonic() > deadline:
            break
        for i in range(n):
            if time.monotonic() > deadline:
                break
            if a[i] < 1e-10:
                continue
            # Quick 1D optimization
            a_orig = a[i]
            S_rest = a.sum() - a[i]
            
            def obj_1d(t):
                a_trial = a.copy()
                a_trial[i] = max(t, 0.0)
                st = a_trial.sum()
                if st < 1e-10:
                    return 1e10
                bt = fast_conv(a_trial)
                return float(2 * n * bt.max() / (st * st))
            
            # Try a few values
            best_t = a_orig
            best_val = obj_1d(a_orig)
            for scale in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]:
                t_try = a_orig * scale
                val = obj_1d(t_try)
                if val < best_val:
                    best_val = val
                    best_t = t_try
            
            if best_val < best_J - 1e-12:
                a[i] = best_t
                best_J = best_val
                best_a = a.copy()
    
    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase hybrid search:
    Phase 1: Diverse initialization + quick smooth gradient (80s)
    Phase 2: Multi-resolution warm start (100s)
    Phase 3: Main optimization - alternating smooth grad + LP (900s)
    Phase 4: Coordinate descent polishing (350s)
    Phase 5: Final check (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    n_main = 512

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "parent")
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(n_main)
        tracker.update(parent, "default")

    # === PHASE 1: Diverse initialization ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: Diverse Initialization", flush=True)
    print(f"{'=' * 60}", flush=True)

    p1_deadline = time.monotonic() + 80.0
    inits = generate_diverse_inits(n_main, rng, parent)
    
    candidates = []
    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_deadline:
            break
        a0 = np.maximum(a0, 1e-6).astype(np.float64)
        a0 = a0 / a0.sum()
        # Quick smooth gradient
        a_opt, J_val = run_smooth_gradient(a0, min(time.monotonic() + 10, p1_deadline - 5),
                                            tau=60.0, lr_init=0.03, patience=100)
        J_val = compute_J(a_opt)
        candidates.append((J_val, a_opt))
        print(f"  init {idx:2d}: J={J_val:.8f}", flush=True)
        tracker.update(a_opt, f"init{idx}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 best: J={candidates[0][0]:.8f}", flush=True)

    # === PHASE 2: Multi-resolution warm start ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 2: Multi-resolution Warm Start", flush=True)
    print(f"{'=' * 60}", flush=True)

    p2_deadline = time.monotonic() + 100.0
    best_a = candidates[0][1].copy()
    
    # Downscale to 128, optimize, then upscale back
    for n_small in [128, 256]:
        if time.monotonic() > p2_deadline:
            break
        if n_small >= best_a.size:
            continue
        a_small = resample_seq(best_a, n_small)
        a_small = np.maximum(a_small, 1e-6)
        a_small = a_small / a_small.sum()
        
        a_small, J_small = run_smooth_gradient(a_small, min(time.monotonic() + 25, p2_deadline - 10),
                                                tau=50.0, lr_init=0.04, patience=80)
        J_small = compute_J(a_small)
        print(f"  n={n_small}: J={J_small:.8f}", flush=True)
        
        # Upscale and continue
        a_up = resample_seq(a_small, best_a.size)
        a_up = np.maximum(a_up, 1e-6)
        a_up = a_up / a_up.sum()
        a_up, J_up = run_smooth_gradient(a_up, min(time.monotonic() + 20, p2_deadline - 5),
                                          tau=50.0, lr_init=0.02, patience=60)
        J_up = compute_J(a_up)
        print(f"  upscaled to n={best_a.size}: J={J_up:.8f}", flush=True)
        if J_up < compute_J(best_a):
            best_a = a_up
            tracker.update(best_a, f"multires_{n_small}")

    # === PHASE 3: Main optimization ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 3: Main Optimization (Smooth Grad + LP + CD)", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_deadline = min(time.monotonic() + 900.0, deadline - 350.0)
    a = best_a.copy()
    a = a / a.sum()
    n = len(a)
    best_a_main = a.copy()
    best_J_main = compute_J(a)
    tracker.update(best_a_main, "phase3_start")
    
    no_improve_global = 0
    cycle = 0
    
    while time.monotonic() < p3_deadline:
        cycle += 1
        
        # --- Smooth gradient phase ---
        sg_deadline = min(time.monotonic() + 60, p3_deadline - 60)
        if time.monotonic() < sg_deadline:
            a_sg, J_sg = run_smooth_gradient(a, sg_deadline, tau=80.0, lr_init=0.03, patience=150)
            if J_sg < best_J_main - 1e-12:
                best_J_main = J_sg
                best_a_main = a_sg.copy()
                a = a_sg.copy()
                tracker.update(a, f"cycle{cycle}_sg")
                no_improve_global = 0
            else:
                no_improve_global += 1
        
        # --- LP phase ---
        lp_deadline = min(time.monotonic() + 80, p3_deadline - 30)
        if time.monotonic() < lp_deadline:
            a_lp, J_lp = run_lp_mirror(a, lp_deadline, top_frac=0.7, max_no_improve=100)
            if J_lp < best_J_main - 1e-12:
                best_J_main = J_lp
                best_a_main = a_lp.copy()
                a = a_lp.copy()
                tracker.update(a, f"cycle{cycle}_lp")
                no_improve_global = 0
            else:
                no_improve_global += 1
        
        # --- Coordinate descent phase (every 3 cycles) ---
        if cycle % 3 == 0 and time.monotonic() < p3_deadline - 30:
            cd_deadline = min(time.monotonic() + 30, p3_deadline - 10)
            a_cd = run_coordinate_descent(a, cd_deadline, passes=2)
            J_cd = compute_J(a_cd)
            if J_cd < best_J_main - 1e-12:
                best_J_main = J_cd
                best_a_main = a_cd.copy()
                a = a_cd.copy()
                tracker.update(a, f"cycle{cycle}_cd")
                no_improve_global = 0
        
        # --- Periodic perturbation ---
        if no_improve_global > 6:
            no_improve_global = 0
            rng_pert = np.random.default_rng(cycle * 104729)
            scale = 0.15 / (1.0 + 0.05 * (cycle // 10))
            
            # Mix of approaches: random smooth, random sparse, palindrome
            pert_type = cycle % 3
            if pert_type == 0:
                # Smooth random
                ksize = max(3, n // 32)
                kernel = np.ones(ksize) / ksize
                pert = np.convolve(rng_pert.normal(0, 1, n), kernel, mode='same')
            elif pert_type == 1:
                # Sparse random
                pert = np.zeros(n)
                n_spikes = n // 8
                idx = rng_pert.integers(0, n, n_spikes)
                pert[idx] = rng_pert.normal(0, 1, n_spikes)
            else:
                # Palindrome perturbation
                half = n // 2
                pert_half = rng_pert.normal(0, 1, half)
                pert = np.zeros(n)
                pert[:half] = pert_half
                pert[half:] = pert_half[::-1]
            
            a = a + scale * pert
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            print(f"  Perturbation at cycle {cycle}, scale={scale:.3f}", flush=True)
    
    tracker.update(best_a_main, "phase3_final")

    # === PHASE 4: Coordinate descent polishing ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 4: Coordinate Descent Polishing", flush=True)
    print(f"{'=' * 60}", flush=True)
    print(f"  Starting J={compute_J(best_a_main):.8f}", flush=True)

    p4_deadline = min(time.monotonic() + 350.0, deadline - 30.0)
    
    # Multiple passes of coordinate descent with decreasing aggressiveness
    a_polish = best_a_main.copy()
    a_polish = a_polish / a_polish.sum()
    
    for pass_num in range(20):
        if time.monotonic() > p4_deadline:
            break
        cd_deadline = min(time.monotonic() + 15, p4_deadline - 5)
        a_cd = run_coordinate_descent(a_polish, cd_deadline, passes=1)
        J_cd = compute_J(a_cd)
        if J_cd < compute_J(a_polish) - 1e-12:
            a_polish = a_cd
            tracker.update(a_polish, f"cd_pass{pass_num}")
        elif pass_num > 5:
            # If stuck for a while, try a small perturbation and continue
            rng_cd = np.random.default_rng(pass_num * 31337)
            a_polish = a_polish + 0.01 * rng_cd.normal(0, 1, len(a_polish))
            a_polish = np.maximum(a_polish, 1e-6)
            a_polish = a_polish / a_polish.sum()

    # === PHASE 5: Final short optimization ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 5: Final Optimization", flush=True)
    print(f"{'=' * 60}", flush=True)

    if time.monotonic() < deadline - 10:
        # Quick smooth gradient on best
        a_final, J_final = run_smooth_gradient(tracker.best, min(time.monotonic() + 60, deadline - 10),
                                                tau=100.0, lr_init=0.01, patience=50)
        tracker.update(a_final, "final_sg")
        
        # Quick LP
        if time.monotonic() < deadline - 10:
            a_lp_final, _ = run_lp_mirror(tracker.best, min(time.monotonic() + 60, deadline - 10),
                                           top_frac=1.0, max_no_improve=50)
            tracker.update(a_lp_final, "final_lp")
        
        # Final CD
        if time.monotonic() < deadline - 10:
            a_cd_final = run_coordinate_descent(tracker.best, deadline - 5, passes=3)
            tracker.update(a_cd_final, "final_cd")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    # Verify with evaluate_sequence
    final_score = evaluate_sequence([float(x) for x in tracker.best])
    if np.isinf(final_score):
        print("WARNING: Final evaluation returned inf, using fallback", flush=True)
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
    else:
        tracker.best_score = final_score

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={len(tracker.best)} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)
    
    return [float(x) for x in tracker.best]
