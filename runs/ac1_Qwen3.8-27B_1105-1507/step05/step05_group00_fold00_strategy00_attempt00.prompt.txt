def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):

def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / np.sum(e)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def chase_peak_step(a, step_size=0.05):
    """
    Chase-the-peak: find the max peak in conv(a,a), compute subgradient,
    and take a step to reduce it while maintaining sum and non-negativity.
    """
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    
    # Subgradient of max_conv w.r.t. a: g[i] = 2*a[k_star - i] if valid
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    
    # Project gradient onto hyperplane sum(a) = const (remove mean)
    grad_centered = grad - np.mean(grad)
    
    # Take step
    a_new = a - step_size * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s

def coordinate_descent(a, max_iters=200, time_budget=30.0):
    """
    Coordinate descent: for each coordinate, try small adjustments to reduce objective.
    """

    a = a.copy().astype(np.float64)

            break

        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            # Try moving mass from i to another position
            # Direction: reduce contribution to max peak from this coordinate
            k_star = int(np.argmax(conv))
            j = k_star - i
            if 0 <= j < n and j != i:
                # Moving mass from i to j (or vice versa) might help
                # Try: decrease a[i], increase a[j] (maintain sum)
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = frac * a[i]
                    if delta > a[i]:
                        continue
                    a[i] -= delta
                    a[j] += delta
                    # Recompute only affected convolution entries
                    # For efficiency, recompute full conv for small n
                    new_conv = compute_conv(a)
                    new_max = float(np.max(new_conv))
                    sum_a = float(np.sum(a))
                    new_obj = 2.0 * n * new_max / (sum_a * sum_a)

                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_a = a.copy()
                        conv = new_conv
                        max_b = new_max
                        improved = True
                    else:
                        a[i] += delta
                        a[j] -= delta
                        break
                else:
                    # Revert if we didn't break
                    pass
            else:
                continue

        if not improved:
            # Try reverse: move mass from j to i for peak-reducing moves
            for i in range(0, n, 2):
                if time.monotonic() - start > time_budget:
                    break
                k_star = int(np.argmax(conv))
                j = k_star - i
                if 0 <= j < n and j != i and a[j] > 0:

                            best_obj = new_obj
                            best_a = a.copy()
                            conv = new_conv
                        else:
                            a[j] += delta
                            a[i] -= delta
                            break

def subgradient_momentum(a, max_iters=500, time_budget=30.0):
    """
    Subgradient optimization with Nesterov momentum on the smoothed max objective.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    step = 0.05

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Soft-max weighting over convolution peaks
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        tau = 1.0 + 2.0 * it / max_iters
        shifted = np.clip((conv - cmax) / tau, -300.0, 300.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i]
        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

        # Center gradient (project onto sum=const hyperplane)
        grad_centered = grad_a - np.mean(grad_a)

        # Nesterov momentum
        v = mu * v - step * grad_centered
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Decay step size
        step *= 0.999

    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new
