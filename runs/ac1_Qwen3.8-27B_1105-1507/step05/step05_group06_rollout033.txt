3. LP mirror descent with bilinear line search (retained from parent)
4. Multi-start with diverse initializations and multi-length exploration
5. Adaptive simulated annealing with block moves for global exploration
Key novelty: L-BFGS on smooth objective provides second-order information for rapid polishing,
complementing the first-order LP method. Coordinate descent uses incremental b updates for speed.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar, fmin_bfgs
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    rows = np.empty((2 * n - 1) * n, dtype=np.int32)
    cols = np.empty((2 * n - 1) * n, dtype=np.int32)
    vals = np.empty((2 * n - 1) * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=60):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_bilinear(b_aa, b_ag, b_gg):
    """Find t in [0,1] minimizing max((1-t)^2*b_aa + 2t(1-t)*b_ag + t^2*b_gg)."""
    def obj(t):
        b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
        return b.max()
    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                             options={'xatol': 1e-9})
    return float(result.x)


def smooth_objective_and_grad(a, beta):
    """
    Compute smooth J and its gradient.
    F_smooth = (1/beta)*log(sum_k exp(beta*b_k)) approximates max(b_k)
    J = 2n * F_smooth / S^2 where S = sum(a)
    """
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 0.01:
        return 1e10, np.zeros_like(a)
    n = len(a)
    b = fast_conv(a)
    # Numerically stable log-sum-exp
    b_max = b.max()
    log_sum_exp = b_max + np.log(np.sum(np.exp(beta * (b - b_max))))
    F = log_sum_exp / beta
    J_val = 2.0 * n * F / (S * S)
    # Gradient of F w.r.t. a_i:
    # dF/da_i = sum_k w_k * db_k/da_i / beta * beta = sum_k w_k * 2*a_{k-i}
    # where w_k = exp(beta*b_k) / sum_j exp(beta*b_j)
    w = np.exp(beta * (b - b_max))
    w = w / w.sum()
    # partner[i] = sum_k w_k * a_{k-i} (where k-i is valid)
    # = sum_j a[j] * w[j+i] (substituting k=j+i)
    partner = np.zeros(n)
    for i in range(n):
        partner[i] = np.dot(a, w[i:i + n])
    grad_F = 2.0 * partner
    # dJ/da_i = 2n * [grad_F_i / S^2 - 2*F / S^3]
    grad_J = 2.0 * n * (grad_F / (S * S) - 2.0 * F / (S ** 3))
    return J_val, grad_J


def lbfgs_smooth(a_init, deadline, beta_start=30.0, beta_end=200.0, max_iter=300):
    """
    L-BFGS optimization using smooth max approximation with adaptive beta.
    Optimize over a (non-negative), with clipping after each step.
    """
    a = np.maximum(a_init.astype(np.float64), 0)
    S = a.sum()
    if S < 0.01:
        return a
    a = a / S
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    t_start = time.monotonic()
    total_time = max(deadline - t_start, 10)

    # Run L-BFGS with increasing beta
    stages = [
        (beta_start, 0.3),    # 30% of time
        (beta_start * 3, 0.3), # 30%
        (beta_end, 0.4),       # 40%
    ]

    for beta, time_frac in stages:
        stage_deadline = time.monotonic() + time_frac * total_time
        if time.monotonic() > stage_deadline:
            continue

        # Define objective and gradient for this beta
        def obj_wrapper(x):
            a_clip = np.maximum(x, 0)
            s = a_clip.sum()
            if s < 0.01:
                return 1e10
            a_n = a_clip / s
            val, _ = smooth_objective_and_grad(a_n, beta)
            return val

        def grad_wrapper(x):
            a_clip = np.maximum(x, 0)
            s = a_clip.sum()
            if s < 0.01:
                return np.zeros_like(x)
            a_n = a_clip / s
            _, grad = smooth_objective_and_grad(a_n, beta)
            # Transform gradient from a_n space to x space
            # a_n = x / sum(x), so da_n_i/dx_j = (delta_ij * S - x_j) / S^2
            # dx_j = sum_i (dJ/da_n_i) * (delta_ij*S - x_j)/S^2 ... 
            # Actually for L-BFGS we just need the gradient w.r.t. x
            # dJ/dx_j = sum_i dJ/da_n_i * da_n_i/dx_j
            # da_n_i/dx_j = (delta_ij * S - x_j) / S^2
            # dJ/dx_j = (dJ/da_n_j * S - sum_i dJ/da_n_i * x_i) / S^2
            dJ_da = grad
            S2 = s * s
            grad_x = (dJ_da[j] * s - np.dot(dJ_da, a_clip)) / S2
            return grad_x

        # Actually, let's simplify: optimize over the normalized a directly
        # and just clip at the end. The normalization is implicit in the objective.
        
        def obj_direct(x):
            a_clip = np.maximum(x, 0)
            s = a_clip.sum()
            if s < 0.01:
                return 1e10
            a_n = a_clip / s
            val, _ = smooth_objective_and_grad(a_n, beta)
            return val

        def grad_direct(x):
            a_clip = np.maximum(x, 0)
            s = a_clip.sum()
            if s < 0.01:
                return np.zeros_like(x)
            a_n = a_clip / s
            _, dJ_da = smooth_objective_and_grad(a_n, beta)
            # Chain rule: dJ/dx_j = sum_i dJ/da_i * da_i/dx_j
            # a_i = x_i / S, so da_i/dx_j = (delta_ij * S - x_j) / S^2
            # dJ/dx_j = dJ/da_j / S - (sum_i dJ/da_i * x_i) / S^2
            grad_x = dJ_da / s - np.dot(dJ_da, a_clip) / (s * s) * a_clip / a_clip
            # Actually: dJ/dx_j = (dJ/da_j * S - sum_i dJ/da_i * x_i) / S^2
            # = dJ/da_j / S - (dot(dJ_da, x) / S^2)
            # But this gives a vector where each component is different only through dJ/da_j
            num = dJ_da * s - np.dot(dJ_da, a_clip)
            grad_x = num / (s * s)
            return grad_x

        # Use scipy's L-BFGS-B which handles bounds
        from scipy.optimize import minimize
        res = minimize(
            obj_direct, a.copy(),
            jac=grad_direct,
            method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': 200, 'gtol': 1e-8, 'maxls': 50}
        )

        if res.x is not None:
            a_new = np.maximum(res.x, 0)
            aa = a_new.sum()
            if aa > 0.01:
                a_new = a_new / aa
                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
                a = a_new

        if time.monotonic() > deadline:
            break

    return best_a


def coordinate_descent_fast(a, deadline, max_passes=50):
    """
    Fast coordinate descent with incremental convolution updates.
    For each coordinate i, try candidate values and pick the one that minimizes J.
    Uses the fact that changing a[i] only affects b[k] for k in [i, i+n-1].
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    b = fast_conv(a)
    S = a.sum()  # = 1 after normalization

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = np.random.permutation(n)
        improved = False
        any_change = False

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            # Determine affected lag range
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Precompute unaffected max
            if k_lo > 0:
                unaff_lo_max = b[:k_lo].max()
            else:
                unaff_lo_max = 0.0
            if k_hi < 2 * n - 2:
                unaff_hi_max = b[k_hi + 1:].max()
            else:
                unaff_hi_max = 0.0
            unaff_max = max(unaff_lo_max, unaff_hi_max)

            # Current J
            cur_max_b = max(unaff_max, b[k_lo:k_hi + 1].max())
            cur_J = 2.0 * n * cur_max_b / (S * S)

            # Generate candidates
            if ai < 1e-6:
                candidates = [0.0, 0.005, 0.01, 0.02, 0.05, 0.1]
            else:
                candidates = [0.0, ai * 0.3, ai * 0.5, ai * 0.7, ai * 1.3, ai * 1.5, ai * 2.0]
                # Add a candidate based on the ratio of affected to unaffected max
                if unaff_max > 0 and b[k_lo:k_hi + 1].max() > unaff_max:
                    candidates.append(ai * 0.1)

            best_cand = ai
            best_cand_J = cur_J

            for cand in candidates:
                delta = cand - ai
                if abs(delta) < 1e-14:
                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue

                # Compute new max over affected lags
                new_affected_max = 0.0
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if k == 2 * i:
                        new_val = b[k] + 2.0 * delta * ai + delta * delta
                    else:
                        new_val = b[k] + 2.0 * delta * a[j]
                    if new_val > new_affected_max:
                        new_affected_max = new_val

                new_max_b = max(unaff_max, new_affected_max)
                new_J = 2.0 * n * new_max_b / (S_new * S_new)

                if new_J < best_cand_J - 1e-13:
                    best_cand_J = new_J
                    best_cand = cand

            if best_cand != ai:
                # Apply the change
                delta = best_cand - ai
                a[i] = best_cand
                S += delta
                # Update b incrementally
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if k == 2 * i:
                        b[k] += 2.0 * delta * ai + delta * delta
                    else:
                        b[k] += 2.0 * delta * a[j]
                improved = True
                any_change = True

        # Renormalize
        if S > 0.01:
            a = a / S
            S = 1.0
        # Recompute b to avoid drift (every few passes)
        if p % 3 == 2 or not any_change:
            b = fast_conv(a)
            S = a.sum()
            if S > 0.01:
                J = 2.0 * n * b.max() / (S * S)
                if J < best_J - 1e-12:
                    best_J = J
                    best_a = a.copy()

        if not improved and not any_change:
            break
        if not improved:
            # Check if we should continue
            b = fast_conv(a)
            J = compute_J(a)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()
            else:
                break

    return best_a


def coordinate_descent_exact(a, deadline, max_passes=30):
    """
    Coordinate descent using golden section search for exact optimal value per coordinate.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    b = fast_conv(a)
    S = 1.0

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = np.random.permutation(n)
        improved_count = 0

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Precompute coefficients for each affected lag
            # b_new[k] = b[k] + 2*delta*a[k-i] for k != 2i
            # b_new[2i] = b[2i] + 2*delta*a[i] + delta^2
            coeffs = np.zeros(k_hi - k_lo + 1)
            for idx, k in enumerate(range(k_lo, k_hi + 1)):
                j = k - i
                if k == 2 * i:
                    coeffs[idx] = 2.0 * ai  # linear coefficient (quadratic handled separately)
                else:
                    coeffs[idx] = 2.0 * a[j]

            b_affected = b[k_lo:k_hi + 1].copy()

            # Unaffected max
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, b[:k_lo].max())
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, b[k_hi + 1:].max())

            # Define objective for golden section
            def obj_delta(delta):
                S_new = S + delta
                if S_new < 0.01 or delta < -ai + 1e-15:
                    return 1e10
                new_b = b_affected + delta * coeffs
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    idx_2i = 2 * i - k_lo
                    new_b[idx_2i] += delta * delta
                new_max = max(unaff_max, new_b.max())
                return 2.0 * n * new_max / (S_new * S_new)

            # Golden section search in [-ai, ai*5 + 0.05]
            lo = max(-ai, 0.01 - S - 0.9 * ai)  # ensure a[i]+delta >= 0 and S+delta > 0.01
            hi = ai * 5.0 + 0.05
            if hi <= lo:
                hi = lo + 0.01

            # Evaluate at a few points first to narrow range
            test_points = [lo, (lo + hi) / 2, hi, max(0, ai * 0.5), max(0, ai * 1.5)]
            test_points = [t for t in test_points if lo <= t <= hi]
            if not test_points:
                test_points = [lo, hi]

            min_val = min(obj_delta(t) for t in test_points)
            min_t = min(test_points, key=lambda t: obj_delta(t))

            # Local golden section around min_t
            search_lo = max(lo, min_t - (hi - lo) * 0.2)
            search_hi = min(hi, min_t + (hi - lo) * 0.2)

            if search_hi - search_lo > 1e-10:
                from scipy.optimize import minimize_scalar
                res = minimize_scalar(obj_delta, bounds=(search_lo, search_hi),
                                      method='bounded', options={'xatol': 1e-10})
                best_delta = float(res.x)
            else:
                best_delta = min_t

            # Compare with current
            cur_J = 2.0 * n * max(unaff_max, b[k_lo:k_hi + 1].max()) / (S * S)
            new_J = obj_delta(best_delta)

            if new_J < cur_J - 1e-12:
                delta = best_delta
                a[i] = ai + delta
                S += delta
                # Update b
                for idx, k in enumerate(range(k_lo, k_hi + 1)):
                    j = k - i
                    if k == 2 * i:
                        b[k] += 2.0 * delta * ai + delta * delta
                    else:
                        b[k] += 2.0 * delta * a[j]
                improved_count += 1

        # Renormalize periodically
        if S > 0.01:
            a = a / S
            S = 1.0
        if p % 5 == 4 or improved_count == 0:
            b = fast_conv(a)
            J = compute_J(a)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()
        if improved_count == 0:
            break

    return best_a


def lp_mirror_descent(a, max_iters, deadline, use_cd_every=30, cd_type='fast'):
    """LP mirror descent with line search and periodic coordinate descent."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        g = lp_best_response(a, M, time_limit=30)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)
        t_opt = line_search_bilinear(b_aa, b_ag, b_gg)

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)

        if J_new < best_J - 1e-13:
            best_a = a_new.copy()
            best_J = J_new
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if use_cd_every > 0 and it % use_cd_every == use_cd_every - 1:
            if cd_type == 'fast':
                a_cd = coordinate_descent_fast(a, min(time.monotonic() + 15, deadline), max_passes=3)
            else:
                a_cd = coordinate_descent_exact(a, min(time.monotonic() + 20, deadline), max_passes=3)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-13:
                best_J = J_cd
                best_a = a_cd.copy()
            a = a_cd
            no_improve = 0

        if no_improve > 40:
            a = a + np.random.normal(0, 0.03, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def block_simulated_annealing(a, deadline, max_iters=2000, T_start=0.02, T_end=0.001):
    """Simulated annealing with block coordinate moves."""
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    current_J = best_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        T = T_start * (T_end / T_start) ** frac

        # Block move: change a random block of size 1-5
        block_size = np.random.choice([1, 1, 1, 2, 2, 3, 4, 5])
        start = np.random.randint(0, max(1, n - block_size + 1))
        indices = range(start, min(start + block_size, n))

        old_vals = [a[i] for i in indices]

        for i in indices:
            if a[i] < 0.05:
                a[i] = max(0, np.random.exponential(0.05))
            else:
                a[i] = max(0, np.random.normal(a[i], 0.1 * a[i] + 0.01))

        a_new = np.maximum(a, 0)
        s = a_new.sum()
        if s < 0.01:
            for i, v in zip(indices, old_vals):
                a[i] = v
            continue
        a_new = a_new / s
        new_J = compute_J(a_new)

        delta = new_J - current_J
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            current_J = new_J
            if new_J < best_J - 1e-12:
                best_J = new_J
                best_a = a_new.copy()
        else:
            for i, v in zip(indices, old_vals):
                a[i] = v

    return best_a


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    rc = 0.5 - 0.5 * np.cos(2 * np.pi * t)
    inits.append(rc / rc.sum())

    # Gaussian
    g = np.exp(-4 * (t - 0.5) ** 2) + 0.02
    inits.append(g / g.sum())

    # Double bump
    db = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(db / db.sum())

    # Smooth random
    ksize = max(3, n // 50)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.05, 0.01)
    inits.append(sm / sm.sum())

    # Frequency modulated
    for freq in [2, 3, 5]:
        fm = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    # End-heavy (to reduce boundary effects)
    eh = 0.3 + 0.7 * np.sin(np.pi * t)
    inits.append(eh / eh.sum())

    # Synchronized bursts
    sb = np.ones(n)
    for center in range(0, n, max(4, n // 10)):
        width = max(2, n // 30)
        mask = np.abs(np.arange(n) - center) < width
        sb[mask] += 1.0
    inits.append(sb / sb.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining L-BFGS (smooth), coordinate descent (exact),
    LP mirror descent, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent)

    # === PHASE 1: L-BFGS with smooth max from parent ===
    p1_deadline = t_start + 250
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: L-BFGS smooth (n={parent.size}, budget=250s)", flush=True)
    print(f"{'='*60}", flush=True)

    a1 = lbfgs_smooth(parent, min(p1_deadline, deadline - 400), beta_start=30, beta_end=200)
    update_best(a1)
    print(f"  Phase 1 done: J={compute_J(a1):.8f}", flush=True)

    # === PHASE 2: Fast coordinate descent ===
    p2_deadline = t_start + 450
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Coordinate Descent Fast (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)

    a2 = coordinate_descent_fast(a1, min(p2_deadline, deadline - 300), max_passes=50)
    update_best(a2)
    # Also polish the original parent
    if best_seq is not None:
        a2b = coordinate_descent_fast(best_seq, min(p2_deadline, deadline - 300), max_passes=30)
        update_best(a2b)
    print(f"  Phase 2 done: best J={best_score:.8f}", flush=True)

    # === PHASE 3: LP mirror descent ===
    p3_deadline = t_start + 700
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (budget=250s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a3, J3 = lp_mirror_descent(best_seq, 3000, min(p3_deadline, deadline - 250),
                                    use_cd_every=25, cd_type='fast')
        update_best(a3)
        print(f"  Phase 3 LP done: J={J3:.8f}, best={best_score:.8f}", flush=True)

    # === PHASE 4: L-BFGS again from LP result ===
    p4_deadline = t_start + 900
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: L-BFGS refine (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        # Perturb slightly to escape potential local opt
        a4_init = np.maximum(best_seq + rng.normal(0, 0.01, best_seq.size), 0.001)
        a4_init = a4_init / a4_init.sum()
        a4 = lbfgs_smooth(a4_init, min(p4_deadline, deadline - 200), beta_start=50, beta_end=300)
        update_best(a4)
        print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Exact coordinate descent ===
    p5_deadline = t_start + 1100
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Coordinate Descent Exact (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a5 = coordinate_descent_exact(best_seq, min(p5_deadline, deadline - 150), max_passes=30)
        update_best(a5)
        print(f"  Phase 5 done: best J={best_score:.8f}", flush=True)

    # === PHASE 6: Multi-start exploration ===
    p6_deadline = t_start + 1350
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Multi-Start Exploration (budget=250s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_base = best_seq.size
        # Try different lengths
        target_lengths = sorted(set([
            int(n_base * 0.75), int(n_base * 1.25), int(n_base * 1.5),
            384, 448, 512, 640, 768, 1024
        ]))
        target_lengths = [L for L in target_lengths if 128 <= L <= 1536]

        for tgt_n in target_lengths:
            if time.monotonic() > p6_deadline:
                break
            if tgt_n == n_base:
                continue

            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-5)
            a_ext = a_ext / a_ext.sum()

            # Quick L-BFGS + CD
            sub_dl = min(time.monotonic() + 20, p6_deadline)
            a_r = lbfgs_smooth(a_ext, sub_dl, beta_start=50, beta_end=150)
            update_best(a_r)
            sub_dl2 = min(time.monotonic() + 10, p6_deadline)
            a_cd = coordinate_descent_fast(a_r, sub_dl2, max_passes=5)
            update_best(a_cd)

        # Also try different inits at the best length
        n_best = best_seq.size
        inits = generate_inits(n_best, rng)
        # Add perturbed best
        for sigma in [0.05, 0.1, 0.2]:
            pert = np.maximum(best_seq + rng.normal(0, sigma, n_best), 0.001)
            inits.append(pert / pert.sum())

        # Sort by initial J, try best few
        init_scores = sorted([(compute_J(init), init) for init in inits])
        for score_init, init in init_scores[:4]:
            if time.monotonic() > p6_deadline:
                break
            sub_dl = min(time.monotonic() + 25, p6_deadline)
            a_r = lbfgs_smooth(init, sub_dl, beta_start=50, beta_end=150)
            update_best(a_r)
            sub_dl2 = min(time.monotonic() + 8, p6_deadline)
            a_cd = coordinate_descent_fast(a_r, sub_dl2, max_passes=5)
            update_best(a_cd)

        print(f"  Phase 6 done: best J={best_score:.8f}", flush=True)

    # === PHASE 7: Final polishing ===
    p7_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 7: Final Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        # L-BFGS with high beta
        a7 = lbfgs_smooth(best_seq, min(t_start + 1480, deadline - 30),
                          beta_start=100, beta_end=500)
        update_best(a7)

        # Final CD
        a7b = coordinate_descent_fast(a7, min(t_start + 1490, deadline - 10), max_passes=10)
        update_best(a7b)

        # SA with block moves
        a7c = block_simulated_annealing(a7b, deadline, max_iters=3000,
                                         T_start=0.01, T_end=0.0005)
        update_best(a7c)
        a7d = coordinate_descent_fast(a7c, deadline, max_passes=5)
        update_best(a7d)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
