"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive smooth gradient descent (Nesterov) with gap-based tau scheduling
2. LP mirror descent with restricted active set and dual convergence checking
3. Multi-length exploration and structured perturbations
4. Coordinate-descent polishing for final refinement
Key improvements: adaptive tau, restricted LP with growing constraint set,
dual LP for optimality gap estimation, and tighter time management.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max


def run_smooth_nesterov(a_init, deadline, max_iters=5000, tau_start=0.5, tau_end=500.0,
                        lr_init=0.5, lr_end=0.003, tau_adaptive=True):
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        if tau_adaptive:
            b_curr = fast_conv(y)
            sorted_b = np.sort(b_curr)[::-1]
            gap = max(b_curr.max() - sorted_b[min(5, len(sorted_b)-1)], 1e-12)
            tau = max(tau_start, min(tau_end, 3.0 / gap))
        else:
            tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def run_smooth_standard(a_init, deadline, max_iters=3000, tau=80.0, lr=0.08):
    """Standard gradient descent without Nesterov (more stable at high tau)."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, p, b, b_max = smooth_max_grad(a, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        # Adaptive learning rate based on gradient norm
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        lr_adapt = lr * 0.1 / max(gnorm, 0.01)
        a_new = np.maximum(a - lr_adapt * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new

    return best_a


def build_A_sparse(n, a, k_indices=None):
    """Build sparse convolution matrix, optionally for a subset of k indices."""
    if k_indices is None:
        k_indices = range(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))


def lp_best_response(a, M, time_limit=5.0, top_k=None):
    """Solve LP best response, optionally restricting to top_k constraints."""
    n = len(a)
    b = fast_conv(a)
    if top_k is not None and top_k < 2 * n - 1:
        top_idx = np.argsort(b)[::-1][:top_k]
        top_idx = np.sort(top_idx)
        A = build_A_sparse(n, a, k_indices=top_idx)
        b_ub = b[top_idx].copy()
    else:
        A = build_A_sparse(n, a)
        b_ub = b.copy()

    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_dual_bound(a, M, time_limit=5.0):
    """Solve dual LP to get upper bound on optimal value."""
    n = len(a)
    b = fast_conv(a)
    A = build_A_sparse(n, a)
    # Dual: minimize M * sum(y) subject to A^T y >= 1, y >= 0
    # Convert to standard form: minimize M * sum(y) subject to -A^T y <= -1, y >= 0
    AT = A.T.tocsr()
    c = M * np.ones(AT.shape[0])
    A_ub = -AT
    b_ub = -np.ones(AT.shape[1])
    bounds = [(0, None)] * AT.shape[0]
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success:
            return float(r.fun)
    except Exception:
        pass
    return None


def run_lp_mirror(a_init, deadline, max_iters=500, start_top_k=100, end_top_k=None,
                  lp_time=5.0):
    """LP mirror descent with adaptive active set size."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    if end_top_k is None:
        end_top_k = 2 * n - 1
    no_imp = 0
    no_imp_threshold = 30

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        # Adaptive top_k: start small, grow over iterations
        frac = min(1.0, it / max(50, max_iters * 0.3))
        top_k = int(start_top_k + frac * (end_top_k - start_top_k))
        top_k = min(top_k, 2 * n - 1)

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=lp_time, top_k=top_k)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        # Geometric step size search
        for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_t = (1.0 - t) * a + t * g
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-15:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
            elif M_t < M - 1e-15:
                # Accept even if not new best (helps escape plateaus)
                if t <= 0.1 and no_imp > 5:
                    a = a_t
                    improved = True
                    break

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > no_imp_threshold:
                # Try with full constraint set
                g = lp_best_response(a, M, time_limit=lp_time, top_k=None)
                if g is not None and g.sum() > 1e-10:
                    g /= g.sum()
                    for t in [0.1, 0.05, 0.02, 0.01]:
                        a_t = (1.0 - t) * a + t * g
                        s = a_t.sum()
                        if s < 1e-10:
                            continue
                        a_t /= s
                        if fast_conv(a_t).max() < best_M - 1e-15:
                            a = a_t
                            best_M = fast_conv(a).max()
                            best_a = a.copy()
                            no_imp = 0
                            break
                if no_imp > no_imp_threshold + 20:
                    break

    return best_a


def run_coord_polish(a_init, deadline, max_iters=2000):
    """Coordinate-style polishing: cyclically update each coordinate."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        active = np.where(b >= M - 1e-6 * M)[0]

        # Pick a random coordinate to update
        i = it % n
        # Find max allowed a_i given current other coordinates
        # conv(a,a)_k involving a_i:
        # For each active k, the contribution of a_i is:
        #   a_i * a_{k-i} + a_{k-i} * a_i (if k != 2i) or a_i^2 (if k == 2i)
        # We want: sum_j a_j a_{k-j} <= M for all active k
        # Fix all a_j for j != i, find max a_i

        max_ai = a[i] * 2.0  # allow doubling as upper bound
        for k in active:
            # Compute conv(a,a)_k as function of a_i
            # conv_k = sum_{j != i} a_j * a_{k-j} + (terms with a_i)
            # Terms with a_i: a_i * a_{k-i} (if 0<=k-i<n) + a_{k-i} * a_i (if 0<=k-i<n, same term)
            # But if k-i == i (i.e., k=2i), it's a_i^2
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # Sum over j != i
            base = 0.0
            coeff = 0.0  # coefficient of a_i (linear part)
            quad = False
            for j in range(lo, hi + 1):
                other = k - j
                if j == i:
                    if other == i:
                        quad = True
                    else:
                        coeff += a[other]
                elif other == i:
                    coeff += a[j]
                else:
                    base += a[j] * a[other]
            if quad:
                # a_i^2 + coeff * a_i + base <= M
                # a_i <= [-coeff + sqrt(coeff^2 + 4*(M-base))]/2
                disc = coeff * coeff + 4.0 * (M - base)
                if disc >= 0:
                    val = (-coeff + np.sqrt(disc)) / 2.0
                    max_ai = min(max_ai, max(0.0, val))
            else:
                if coeff > 1e-15:
                    val = (M - base) / coeff
                    max_ai = min(max_ai, max(0.0, val))

        if max_ai < 1e-15:
            max_ai = 1e-10
        # Update with damping
        a[i] = 0.5 * a[i] + 0.5 * max_ai
        a = np.maximum(a, 1e-12)
        s = a.sum()
        if s > 0:
            a /= s

        if it % 50 == 0:
            M_curr = fast_conv(a).max()
            if M_curr < best_M * (1 - 1e-10):
                best_M = M_curr
                best_a = a.copy()

    M_final = fast_conv(a).max()
    if M_final < best_M * (1 - 1e-10):
        best_a = a.copy()
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        p /= p.sum()
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.001))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.001))
        inits.append(np.maximum(p * (1 + 0.2 * rng.normal(0, 1, n)), 0.001))
    inits.append(np.ones(n) / n)
    # U-shaped (Barker-like)
    for alpha in [0.5, 0.7, 1.0, 1.3, 1.7, 2.5, 3.0]:
        inits.append(np.maximum((t**alpha + (1 - t)**alpha) / 2 + 0.05, 0.01))
    # Cosine windows
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.7 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    # Gaussian bumps
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.1)
    inits.append(np.exp(-8*(t-0.3)**2) + np.exp(-8*(t-0.7)**2) + 0.05)
    inits.append(np.exp(-2*(t-0.5)**2) + 0.3)
    # Triangle
    inits.append(1.0 - 2.0 * np.abs(t - 0.5) + 0.1)
    # Random smooth
    k = max(5, n // 20)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.2, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    # Flat with edges emphasized
    for boost in [0.2, 0.5, 1.0]:
        inits.append(np.maximum(1.0 + boost * (t**4 + (1-t)**4), 0.01))

    # Normalize all
    result = []
    for x in inits:
        x = np.maximum(x, 1e-8)
        x /= x.sum()
        result.append(x)
    return result

    print(f"  Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Smooth GD refinement at high tau ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: High-Tau Refinement", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 80:
        p3_budget = min(150.0, deadline - 80 - time.monotonic())
        if p3_budget > 30:
            # Standard GD at high tau (more stable than Nesterov at high tau)
            a_ref = run_smooth_standard(best.seq, time.monotonic() + p3_budget * 0.5,
                                        max_iters=2000, tau=200.0, lr=0.05)
            best.update(a_ref)

            # Nesterov with very high tau
            a_ref2 = run_smooth_nesterov(a_ref, time.monotonic() + p3_budget * 0.5,
                                         max_iters=1500, tau_start=200.0, tau_end=1000.0,
                                         lr_init=0.2, lr_end=0.001, tau_adaptive=False)
            best.update(a_ref2)

    best.report("Phase 3")
    print(f"  Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Coordinate polishing ===
    p4_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Coordinate Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 60:
        p4_budget = min(120.0, deadline - 60 - time.monotonic())
        if p4_budget > 20:
            a_coord = run_coord_polish(best.seq, time.monotonic() + p4_budget,
                                       max_iters=5000)
            best.update(a_coord)

    best.report("Phase 4")
    print(f"  Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Final LP polish ===
    p5_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Final LP Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 10:
        n_best = best.seq.size
        a_final = run_lp_mirror(best.seq, deadline,
                                max_iters=200, start_top_k=2*n_best-1,
                                end_top_k=2*n_best-1, lp_time=5.0)
        best.update(a_final)

    best.report("Phase 5")
    print(f"  Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === PHASE 6: Multi-start perturbations if time remains ===
    if best.seq is not None and time.monotonic() < deadline - 20:
        p6_start = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"PHASE 6: Perturbation Search", flush=True)
        print(f"{'='*50}", flush=True)

        n = best.seq.size
        perturb_schemes = [
            (0.01, 'gauss'),
            (0.005, 'gauss'),
            (0.02, 'lowfreq'),
            (0.015, 'lowfreq'),
            (0.03, 'edge'),
        ]
        for scale, ptype in perturb_schemes:
            if time.monotonic() > deadline - 15:
                break
            if ptype == 'gauss':
                noise = rng.normal(0, scale * 0.01, n)
            elif ptype == 'lowfreq':
                freqs = rng.integers(1, 5, 3)
                phases = rng.uniform(0, 2 * np.pi, 3)
                t_arr = np.linspace(0, 1, n)
                noise = sum(scale * 0.01 * np.cos(2 * np.pi * f * t_arr + ph)
                            for f, ph in zip(freqs, phases))
            else:  # edge
                t_arr = np.linspace(0, 1, n)
                noise = scale * 0.01 * (t_arr**6 + (1-t_arr)**6)

            a_pert = best.seq + noise
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert /= a_pert.sum()

            # Quick smooth GD
            a_opt = run_smooth_nesterov(a_pert, min(time.monotonic() + 20, deadline - 15),
                                        max_iters=300, tau_start=5.0, tau_end=200.0,
                                        lr_init=0.3, lr_end=0.01, tau_adaptive=True)
            best.update(a_opt)

            # Quick LP
            if time.monotonic() < deadline - 10:
                a_lp3 = run_lp_mirror(a_opt, deadline - 10, max_iters=30,
                                      start_top_k=2*n-1, end_top_k=2*n-1, lp_time=3.0)
                best.update(a_lp3)

        best.report("Phase 6")
        print(f"  Phase 6 time: {time.monotonic()-p6_start:.1f}s", flush=True)

    # === PHASE 7: Try nearby lengths ===
    if best.seq is not None and time.monotonic() < deadline - 15:
        p7_start = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"PHASE 7: Length Adjustment", flush=True)
        print(f"{'='*50}", flush=True)

        n_best = best.seq.size
        for delta in [-32, -16, 16, 32, -64, 64]:
            if time.monotonic() > deadline - 10:
                break
            n_try = n_best + delta
            if n_try < 64 or n_try > 4096:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext /= a_ext.sum()
            a_ext = run_smooth_nesterov(a_ext, min(time.monotonic() + 25, deadline - 10),
                                        max_iters=400, tau_start=3, tau_end=150,
                                        lr_init=0.3, lr_end=0.01, tau_adaptive=True)
            best.update(a_ext)
            if time.monotonic() < deadline - 8:
                a_ext_lp = run_lp_mirror(a_ext, deadline - 8, max_iters=20,
                                         start_top_k=2*n_try-1, end_top_k=2*n_try-1,
                                         lp_time=3.0)
                best.update(a_ext_lp)

        best.report("Phase 7")
        print(f"  Phase 7 time: {time.monotonic()-p7_start:.1f}s", flush=True)

    # Final verification
    if best.seq is not None:
        final_score = float(evaluate_sequence(best.seq.tolist()))
        print(f"\n  Final verified score: {final_score:.8f}", flush=True)
        best.score = min(best.score, final_score)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128) / 128
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
