```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a):
    """Solve LP: min M s.t. (a*g)_k <= M for all k, sum(g)=1, g>=0"""
    n = len(a)
    # Build A matrix: (2n-1) x (n+1)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # A[k, n] = -1 (for the M variable)
    
    num_cons = 2*n - 1
    A = np.zeros((num_cons, n+1))
    A[:, n] = -1.0
    
    # Fill in the convolution matrix
    for k in range(num_cons):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize M
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    result = linprog(c, A_ub=A, b_ub=np.zeros(num_cons),
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                     method='highs')
    
    if result.success:
        g = result.x[:n]
        M = result.x[n]
        return np.maximum(g, 0), M
    return None, None
```

"""
Optimizes J(a) = 2n * max(self_convolution(a)) / (sum(a))^2 using a theoretically-
grounded LP best-response game iteration as the primary engine, combined with smooth
subgradient descent and exact coordinate descent for final polish. The game iteration
exploits the bilinear structure of convolution to guarantee convergence to the optimal
solution via the minimax theorem.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] ** 2])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2n * max(conv(a,a)) / (sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lp_best_response(a):
    """
    Solve LP: min M s.t. (a*g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_value, duality_gap_info).
    """
    n = len(a)
    num_cons = 2 * n - 1
    
    # Build sparse A matrix: (2n-1) x (n+1)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # A[k, n] = -1
    rows = []
    cols = []
    vals = []
    
    for k in range(num_cons):
        # M column
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
        
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(num_cons, n + 1))
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # M >= 0
    
    result = linprog(c, A_ub=A, b_ub=np.zeros(num_cons),
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                     method='highs')
    
    if result.success and result.x is not None:
        g = np.maximum(result.x[:n], 0)
        g = g / g.sum()  # renormalize for numerical safety
        M_val = max(result.x[n], 0)
        return g, M_val
    return None, None


def lp_game_iteration(a0, max_iters, deadline, verbose=True):
    """
    Run LP best-response game iteration.
    Returns (best_a, best_value, running_avg).
    """
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Running average (theoretical convergence guarantee)
    avg_a = a.copy()
    avg_M = best_M
    
    for t in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Solve LP best response
        g, M_lp = lp_best_response(a)
        if g is None:
            break
        
        # Track running average
        avg_a = (t * avg_a + a) / (t + 1)
        avg_M = fast_conv(avg_a).max()
        
        # Check convergence via duality gap
        current_M = fast_conv(a).max()
        gap = current_M - M_lp
        
        if verbose and (t % 50 == 0 or t < 5):
            print(f"    iter {t}: M={current_M:.8f}, LP_val={M_lp:.8f}, "
                  f"gap={gap:.2e}, avg_M={avg_M:.8f}", flush=True)
        
        # Update: use adaptive step size
        # For the game, the average strategy converges; individual iterates may oscillate
        # Use tau = 1 (pure best response) but track the average
        tau = 1.0
        a_new = (1 - tau) * a + tau * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        
        # Update best
        M_now = fast_conv(a).max()
        if M_now < best_M:
            best_M = M_now
            best_a = a.copy()
        
        # Check if average is better
        M_avg = fast_conv(avg_a).max()
        if M_avg < best_M:
            best_M = M_avg
            best_a = avg_a.copy()
        
        # Early stopping if gap is very small
        if gap < 1e-10:
            if verbose:
                print(f"    Converged at iter {t}: gap={gap:.2e}", flush=True)
            break
    
    return best_a, best_M


def smooth_subgradient_descent(a0, max_iters, deadline, tau_start=10, tau_end=200, lr=0.01):
    """
    Subgradient descent on smooth approximation of max(conv(a,a)).
    """
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    
    for t in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = t / max(1, max_iters)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Compute smooth gradient
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient: grad_j = sum_k p_k * d(b_k)/da_j
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-12:
                continue
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            # d(b_k)/da_j = a[k-j] for j in [j_min, j_max]
            # But if k-j == j (i.e., k=2j), it's 2*a_j
            js = np.arange(j_min, j_max + 1)
            contributions = a[k - js]
            # Double the diagonal
            diag_mask = (js == k - js)  # j == k-j, i.e., 2j == k
            contributions[diag_mask] *= 2
            grad[js] += p[k] * contributions
        
        # Project gradient to tangent space (sum=0)
        grad = grad - grad.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        
        lr_t = lr / (1 + 0.01 * t)
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        
        M_now = fast_conv(a).max()
        if M_now < best_M:
            best_M = M_now
            best_a = a.copy()
    
    return best_a, best_M


def coordinate_update_fast(a, i, b_prev):
    """
    Find optimal a[i] given all other coordinates.
    Uses the precomputed convolution b_prev for efficiency.
    """
    n = len(a)
    x = a[i]
    S = a.sum()
    
    # Affected k range
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Unaffected max
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[k_lo:k_hi + 1] = False
    if mask.any():
        base_max = b_prev[mask].max()
    else:
        base_max = 0.0
    
    # Precompute for affected positions
    affected_k = np.arange(k_lo, k_hi + 1)
    c_vals = np.zeros(len(affected_k))
    slopes = np.zeros(len(affected_k))
    has_quad = False
    quad_offset = 0
    
    for idx, k in enumerate(affected_k):
        j = k - i
        if 0 <= j < n:
            if j == i:  # k = 2i, quadratic
                c_vals[idx] = b_prev[k] - x * x
                slopes[idx] = 0.0
                has_quad = True
                quad_offset = idx
            else:
                c_vals[idx] = b_prev[k] - 2 * x * a[j]
                slopes[idx] = 2 * a[j]
        else:
            c_vals[idx] = b_prev[k]
            slopes[idx] = 0.0
    
    def obj(x_test):
        new_sum = S - x + x_test
        if new_sum <= 1e-15:
            return 1e10
        vals = c_vals + slopes * x_test
        m = base_max
        if has_quad:
            qv = c_vals[quad_offset] + x_test * x_test
            m = max(m, qv)
        m = max(m, vals.max())
        return m / (new_sum * new_sum)
    
    # Golden section
    lo = 0.0
    hi = max(x * 3.0, 2.0 * np.sqrt(b_prev.max() + 1e-10), 0.01)
    
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def coordinate_descent(a0, max_sweeps, deadline, seed=0):
    """Coordinate descent with exact 1D subproblem solving."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        rng.shuffle(order)
        
        b_prev = fast_conv(a)
        sweep_changed = False
        
        for i in order:
            x_opt = coordinate_update_fast(a, i, b_prev)
            if abs(x_opt - a[i]) > 1e-12:
                a[i] = x_opt
                # Update b_prev incrementally
                # b[k] changes for k in [i, i+n-1]
                # delta_b[k] = (new_x - old_x) * (a[k-i] + (a[i-(k-i)] if valid)) + ...
                # Actually, let's just recompute for the affected range
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    old_contrib = a[i] * a[k - i] if 0 <= k - i < n else 0
                    # Wait, this is getting complicated. Let me just recompute b periodically.
                sweep_changed = True
        
        # Recompute b after each sweep
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        b = fast_conv(a)
        M = b.max()
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20:
            break
    
    return best_a, best_M


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 0.01)
        p = p / p.sum()
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
    
    inits.append(np.ones(n) / n)
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * t), 0.01) / n)
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4 * np.pi * t), 0.01) / n)
    inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * 3 * t), 0.01) / n)
    inits.append(np.exp(-3 * (t - 0.5) ** 2) + 0.1)
    inits[-1] = inits[-1] / inits[-1].sum()
    
    # Smoother random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n + ksize - 1), kernel, mode='valid')[:n]
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # Raised cosine
    raised = 0.3 + 0.7 * np.cos(np.pi * (2 * t - 1)) ** 2
    inits.append(raised / raised.sum())
    
    # Han window
    han = 0.5 * (1 + np.cos(2 * np.pi * t))
    inits.append(np.maximum(han + 0.1, 0.01) / (han + 0.1).sum() * n)
    inits[-1] = inits[-1] / inits[-1].sum()
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
