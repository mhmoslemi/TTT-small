import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_grad(a, tau):
    """Compute gradient of smooth max of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # For numerical stability, subtract max before exp
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad contribution from lag k: 2 * p[k] * a reflected
        grad[lo:hi+1] += 2.0 * p[k] * a[k-hi:k-lo+1][::-1]
    return grad, p, b, b_max

def nesterov_smooth(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,
                     lr_init=0.4, lr_end=0.005, beta=0.9):
    """Nesterov-accelerated smooth gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()

    y = a.copy()

        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0

        sy = y.sum()

        a = a_new

def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (A @ g) = conv(a, g)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None

def peak_directed_descent(a, deadline, max_iters=2000):

        hi = min(n - 1, k_peak)

            lo_k = max(0, k - n + 1)

            weight = b[k] / M if M > 0 else 0
            grad[lo_k:hi_k+1] += weight * a[k-hi_k:k-lo_k+1][::-1]

            lr *= 0.8
            a_new = a.copy()
        else:
            lr = min(lr * 1.05, 0.3)
        
        a = a_new
    
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max

def build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                                tau_start=1.0, tau_end=200.0,
                                lr_init=0.35, lr_end=0.003)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            best.update(best_n)

    best.report("Phase 1")
    print(f"Phase 1 elapsed: {time.monotonic() - (time.monotonic() - p1_budget):.0f}s", flush=True)

    # === PHASE 2: Peak repair on best (10% of time) ===
    if best.seq is not None:
        print(f"\n{'=' * 50}", flush=True)
        print("PHASE 2: Peak Repair", flush=True)
        print(f"{'=' * 50}", flush=True)
        p2_end = min(time.monotonic() + budget_s * 0.10, deadline - 200)
        a = peak_repair(best.seq, p2_end, max_iters=3000)
        best.update(a)
        best.report("Phase 2")

    # === PHASE 3: LP Mirror Descent - MAIN PHASE (50% of time) ===
    if best.seq is not None:
        print(f"\n{'=' * 50}", flush=True)
        print("PHASE 3: LP Mirror Descent (Main)", flush=True)
        print(f"{'=' * 50}", flush=True)
        n = best.seq.size

        # Calibrate LP time
        a_cal = best.seq.copy()
        M_cal = fast_conv(a_cal).max()
        t0 = time.monotonic()
        _ = lp_best_response(a_cal, M_cal, time_limit=30)
        lp_time = max(0.02, time.monotonic() - t0)
        print(f"  n={n}, LP time={lp_time * 1000:.0f}ms", flush=True)

        remaining = deadline - 60 - time.monotonic()
        p3_budget = min(remaining * 0.55, budget_s * 0.50)
        p3_budget = max(60, p3_budget)
        max_lp_iters = max(20, int(p3_budget / (lp_time * 1.3)))
        print(f"  Budget={p3_budget:.0f}s, est iters={max_lp_iters}", flush=True)

        a = lp_mirror_descent(best.seq, time.monotonic() + p3_budget,
                              max_iters=max_lp_iters)
        best.update(a)
        best.report("Phase 3a")

        # === PHASE 4: Post-LP smooth + LP squeeze ===
        if best.seq is not None and time.monotonic() < deadline - 60:
            print(f"\n{'=' * 50}", flush=True)
            print("PHASE 4: Post-LP Polish", flush=True)
            print(f"{'=' * 50}", flush=True)

            # Smooth gradient to "smooth out" any LP artifacts
            p4a_end = min(time.monotonic() + 40, deadline - 80)
            a = nesterov_smooth(best.seq, p4a_end, max_iters=2000,
                                tau_start=5.0, tau_end=300.0,
                                lr_init=0.15, lr_end=0.001)
            best.update(a)

            # More LP
            if time.monotonic() < deadline - 30:
                remaining2 = deadline - 25 - time.monotonic()
                max_iters2 = max(10, int(remaining2 / (lp_time * 1.3)))
                a2 = lp_mirror_descent(best.seq, deadline - 20, max_iters=max_iters2)
                best.update(a2)

            best.report("Phase 4")

        # === PHASE 5: Perturbation restarts ===
        if best.seq is not None and time.monotonic() < deadline - 25:
            print(f"\n{'=' * 50}", flush=True)
            print("PHASE 5: Perturbation Restarts", flush=True)
            print(f"{'=' * 50}", flush=True)
            n = best.seq.size
            for scale in [0.005, 0.01, 0.015, 0.02]:
                if time.monotonic() > deadline - 15:
                    break
                noise = rng.normal(0, scale, n)
                a = np.maximum(best.seq + noise, 1e-6)
                a /= a.sum()
                # Quick smooth
                dl = min(time.monotonic() + 12, deadline - 12)
                a = nesterov_smooth(a, dl, max_iters=500,
                                    tau_start=10.0, tau_end=200.0,
                                    lr_init=0.1, lr_end=0.01)
                best.update(a)
                # Quick LP if time
                if time.monotonic() < deadline - 8:
                    a2 = lp_mirror_descent(a, deadline - 5, max_iters=10)
                    best.update(a2)
            best.report("Phase 5")

    # Final LP polish if any time left
    if best.seq is not None and time.monotonic() < deadline - 3:
        a = lp_mirror_descent(best.seq, deadline, max_iters=20)
        best.update(a)

    if best.seq is None:
        best.seq = np.ones(512) / 512.0
        best.score = compute_J(best.seq)

    print(f"\n{'=' * 50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic() - best.t0:.0f}s", flush=True)
    print(f"{'=' * 50}", flush=True)
    return [float(x) for x in best.seq]
