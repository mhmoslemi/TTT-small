import numpy as np

from scipy.sparse import csr_matrix

def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective function value."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse matrix A where (A @ b)[k] = (a*b)[k] for full convolution."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, n_constraints=None, time_limit=10):
    """
    Find b >= 0 that maximizes sum(b) subject to (a*b)[k] <= M for all k.
    If n_constraints is specified, only use the top n_constraints tightest constraints.
    """
    n = len(a)
    b_conv = fast_conv(a)  # This is (a*a), but we need (a*b) constraint
    # Actually we need: for each k, sum_j a[k-j] * b[j] <= M
    # This is (A @ b)[k] <= M where A is the convolution matrix of a
    
    if n_constraints is not None and n_constraints < 2 * n - 1:
        # Only use the most binding constraints
        A_full = build_conv_matrix(n, a)
        # We don't know b yet, so use all constraints but limit
        # Actually, let's just use all constraints for correctness
        A = A_full
    else:
        A = build_conv_matrix(n, a)
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:

        if r.success and r.x is not None and r.x.sum() > 1e-15:

    except Exception:
        pass
    return None

def spectral_nudge(a, n_steps=100, deadline=None):
    """
    Nudge the sequence in frequency domain to flatten the autocorrelation.
    Idea: compute the DFT, adjust phases to reduce peak autocorrelation.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Zero-pad to 2n for linear convolution via circular
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        
        F = np.fft.rfft(a, size)
        # Compute current autocorrelation
        R = np.fft.irfft(F * np.conj(F), size)[:2*n-1]
        M = R.max()
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
        
        # Gradient of max R with respect to phases
        # R = IFFT(|F|^2), so dR/dphase[k] involves cross terms
        # Simpler: use subgradient approach
        # The peak of R is at some lag k*, and we want to reduce R[k*]
        # dR[k*]/dF[m] = conj(F[m]) * exp(2πi*m*k*/size)
        # For real a, we adjust the real part
        
        peak_idx = int(np.argmax(R))
        # Subgradient direction in frequency domain
        # We want to decrease R[peak_idx], so we want to decrease |F|^2 at 
        # frequencies that constructively interfere at lag peak_idx
        
        # Simpler approach: compute gradient in time domain
        # d(a*a)[k]/da[j] = a[k-j] + a[j-k] (for valid indices)
        grad = np.zeros(n)
        k = peak_idx
        for j in range(n):
            if 0 <= k - j < n:
                grad[j] += a[k - j]
            if 0 <= j - k < n:
                grad[j] += a[j - k]
        
        # Take a small step against the gradient (to reduce the peak)
        lr = 0.01 / (np.linalg.norm(grad) + 1e-10)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        # Check if this helps
        M_new = fast_conv(a_new).max()
        if M_new < M:
            a = a_new
        # Otherwise try smaller step
        else:
            a_new2 = a - 0.1 * lr * grad
            a_new2 = np.maximum(a_new2, 1e-8)
            a_new2 /= a_new2.sum()
            if fast_conv(a_new2).max() < M:
                a = a_new2
    
    return best_a


def smooth_descent(a, deadline, max_iters=3000, tau_start=2.0, tau_end=200.0,
                   lr_init=0.3, lr_end=0.005):
    """Smooth max gradient descent with Nesterov momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        grad = np.zeros(n)
        active = np.where(p > 1e-10)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
        
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    
    return best_a


def resample_seq(src, n):
    """Resample a sequence to a new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):

    c = -np.ones(n)

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP mirror descent
with momentum, adaptive step sizes, periodic spectral reheating, and multi-length

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(a, M, time_limit=10):
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def smooth_descent(a, deadline, max_iters=5000, tau_start=1.0, tau_end=200.0,
                   lr_init=0.35, lr_end=0.004):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.88

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(np.clip(tau * (b - b_max), -50, 0))
        p = exp_b / exp_b.sum()

        grad = np.zeros(n)
        active = np.where(p > 1e-10)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]

        if b_max < best_M * (1.0 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def soft_penalty_step(a, M_target, n_iter=80, lr=0.08):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(n_iter):
        b = fast_conv(a)
        excess = np.maximum(b - M_target, 0)
        if excess.max() < 1e-13:
            break

        grad = np.zeros(n)
        active = np.where(excess > 1e-12)[0]
        for k in active:
            coeff = 2.0 * excess[k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += coeff * a[k - hi:k - lo + 1][::-1]

        a_new = np.maximum(a - lr * grad, 1e-8)
        a_new /= a_new.sum()
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        a = a_new

    return best_a


def resample_seq(src, n):

    a = run_lp_mirror(best_seq, p5_dl, max_iters=1500, seed=seed + 999)
    try_update(a)

    if best_seq is None:
        best_seq = np.ones(128)
        best_score = compute_J(best_seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best_seq]
