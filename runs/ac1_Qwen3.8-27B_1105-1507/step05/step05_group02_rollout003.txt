"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Accelerated LP best-response iteration (Nesterov-accelerated mirror descent)
2. Smooth gradient descent with adaptive tau
3. Frequency-domain spectral flattening
4. Multi-start with diverse initializations
Key ideas: Nesterov acceleration on LP iterations, multiple M levels,

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    """Build sparse matrix for LP: rows are lag k, cols are b_j"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, x0=None, time_limit=10):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0"""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                            'mip_rel_gap': 0.0})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def smooth_max_grad(a, tau):
    """Compute gradient of log-sum-exp approximation to max autocorrelation"""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max

def run_smooth_nesterov(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,
                        lr_init=0.4, lr_end=0.005):
    """Smooth gradient descent with Nesterov acceleration"""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    v = np.zeros(n)
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a

def run_accelerated_lp(a, deadline, max_iters=200, m_factors=None):
    """
    Nesterov-accelerated LP best-response iteration.
    At each step, solve LP with the "lookahead" point y, then update.
    Also try multiple M levels for more aggressive descent.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    no_imp = 0
    
    if m_factors is None:
        m_factors = [1.0, 0.99, 0.98]
    
    t_prev = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Try multiple M levels
        b_y = fast_conv(y)
        M_y = float(b_y.max())
        
        g_best = None
        M_used = None
        for mf in m_factors:
            M_try = M_y * mf
            g = lp_best_response(y, M_try, time_limit=8)
            if g is not None:
                g_sum = g.sum()
                if g_sum > 1e-10:
                    g = g / g_sum
                    g_best = g
                    M_used = M_try
                    break
        
        if g_best is None:
            # Fallback: use current M
            g = lp_best_response(y, M_y, time_limit=8)
            if g is not None and g.sum() > 1e-10:
                g_best = g / g.sum()
            else:
                no_imp += 1
                if no_imp > 10:
                    break
                continue
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
        gamma = (t_prev - 1.0) / t_new
        
        # Blend: a_new = (1-t)*a + t*g, with Nesterov momentum
        # Standard: y_new = a_new + gamma * (a_new - a)
        # a_new = (1 - step)*a + step*g
        step = 2.0 / (t_new + 1.0)
        a_new = np.maximum((1.0 - step) * a + step * g_best, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + gamma * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-10:
            y /= sy
        else:
            y = a_new.copy()
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break
        
        a = a_new
        t_prev = t_new
    
    return best_a

def run_lp_mirror_simple(a, deadline, max_iters=200):
    """Simple LP mirror descent without acceleration (for robustness)"""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=8)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        for t in [0.5, 0.35, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 25:
                break
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.005))
    inits.append(np.ones(n))
    for alpha in [0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.2 + 0.8 * (t**alpha + (1 - t)**alpha) / 2, 0.01))
    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.05)
    inits.append(np.exp(-8*(t-0.3)**2) + np.exp(-8*(t-0.7)**2) + 0.03)
    # Triangular
    inits.append(1.0 - 2.0 * np.abs(t - 0.5))
    # Sigmoid-like
    inits.append(1.0 / (1.0 + np.exp(-20*(t-0.5))))
    return inits

def run_subgradient_avg(a, deadline, max_iters=3000, lr_const=0.3):
    """Subgradient descent with Polyak averaging"""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

        tol = 1e-6 * M
        active = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad /= gn

        lr = lr_const / np.sqrt(it + 1)
        a_new = np.maximum(a - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        running += a_new
        count += 1
        a = a_new

    if count > 50:
        avg = running / count
        avg = np.maximum(avg, 0.0)
        s = avg.sum()
        if s > 1e-10:
            avg /= s
            M_avg = fast_conv(avg).max()
            if M_avg < best_M:
                best_a = avg
    return best_a

    best_M = fast_conv(a).max()

        if time.monotonic() > deadline:

        circ_corr = np.fft.ifft(power).real[:n]  # first n values
        M = max(circ_corr.max(), abs(circ_corr.min()))
        # Actually for real a, corr is real and we want max
        M = circ_corr.max()
