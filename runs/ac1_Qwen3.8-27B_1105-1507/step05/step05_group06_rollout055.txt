def fast_conv(a):

    n = len(a)
    if n == 0:

        size <<= 1
    fa = np.fft.rfft(a, size)

    if s < 0.01:

    Compute c[i] = sum_k p[k] * a[k-i] for i = 0, ..., n-1.
    where p has length 2n-1 and a has length n.
    Uses FFT: c[i] = conv(a[::-1], p)[n-1+i]
    """

    while size < n + np_ - 1:

    fa = np.fft.rfft(a[::-1], size)
    fp = np.fft.rfft(p, size)
    cp = np.fft.irfft(fa * fp, size)
    return cp[n - 1:n - 1 + n]

    n = len(a)
    np_len = len(p)
    size = 1
    while size < n + np_len - 1:
        size <<= 1
    fa = np.fft.rfft(a[::-1], size)
    fp = np.fft.rfft(p, size)
    cp = np.fft.irfft(fa * fp, size)
    return cp[n - 1:n - 1 + n]


def soft_max_obj_grad(a, tau):
    """
    Compute soft-max smoothed J and its gradient.
    J(a) = 2n * sm(b) / s^2 where sm is soft-max of b = conv(a,a).
    Returns (J, grad) tuple.
    """
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = b - b_max
    shifted = np.maximum(shifted, -50.0 / tau)
    exp_s = np.exp(tau * shifted)
    exp_sum = exp_s.sum()
    sm = b_max + np.log(exp_sum) / tau
    p = exp_s / exp_sum
    c = compute_cross_corr(p, a)
    J = 2.0 * n * sm / (s * s)
    grad = 4.0 * n * (c * s - sm) / (s ** 3)
    return J, grad


def softmax_lbfgs(a, deadline, tau_start=20.0, tau_mult=1.5, max_phases=15, max_iter_per_phase=100):
    """
    Soft-max smoothed L-BFGS with increasing tau schedule.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a, float('inf')
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    current_tau = tau_start
    no_improve = 0

    for phase in range(max_phases):
        if time.monotonic() > deadline:
            break

        tau = current_tau

        def obj(x):
            x = np.maximum(x, 0)
            sx = x.sum()
            if sx < 0.01:
                return 1e10
            b = fast_conv(x)
            b_max = b.max()
            shifted = np.maximum(b - b_max, -50.0 / tau)
            sm = b_max + np.log(np.exp(tau * shifted).sum()) / tau
            return 2.0 * n * sm / (sx * sx)

        def grad_fn(x):
            x = np.maximum(x, 0)
            sx = x.sum()
            if sx < 0.01:
                return np.zeros_like(x)
            b = fast_conv(x)
            b_max = b.max()
            shifted = np.maximum(b - b_max, -50.0 / tau)
            exp_s = np.exp(tau * shifted)
            exp_sum = exp_s.sum()
            sm = b_max + np.log(exp_sum) / tau
            p = exp_s / exp_sum
            c = compute_cross_corr(p, x)
            return 4.0 * n * (c * sx - sm) / (sx ** 3)

        res = minimize(obj, a, method='L-BFGS-B', jac=grad_fn,
                       bounds=[(0, None)] * n,
                       options={'maxiter': max_iter_per_phase, 'ftol': 1e-18, 'gtol': 1e-10})

        a_new = np.maximum(res.x, 0)
        s_new = a_new.sum()
        if s_new < 0.01:
            break
        a_new = a_new / s_new

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-13:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        current_tau *= tau_mult

        if no_improve > 3:
            break

    return best_a, best_J


def exact_coordinate_descent(a, deadline, max_passes=50):
    """
    Exact coordinate descent: for each coordinate, find optimal value via 1D minimization.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = np.random.permutation(n)
        b = fast_conv(a)
        S = a.sum()
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            # Precompute the "direction" vector for this coordinate
            # b_new[k] = b[k] + (v-ai) * d[k] for k != 2i
            # b_new[2i] = b[2i] + v^2 - ai^2
            # where d[k] = a[k-i] + a[i-k] (0 if index out of range)

            d = np.zeros(2 * n - 1)
            for k in range(i, i + n):
                if k <= 2 * n - 2:
                    j1 = k - i  # = 0, 1, ..., n-1
                    j2 = i - (k - i)  # wait, this is wrong

            # Let me recompute. b[k] = sum_j a[j]*a[k-j]
            # ∂b[k]/∂a[i] = a[k-i] (if 0<=k-i<=n-1) + a[i-k] (if 0<=i-k<=n-1)
            # But a[i-k] = a[k-i] only if we interpret it as a[|k-i|]... no.
            # Actually: the terms involving a[i] in b[k] are:
            #   j=i: a[i]*a[k-i] (valid if 0<=k-i<=n-1)
            #   j=k-i: a[k-i]*a[i] (valid if 0<=k-i<=n-1, i.e., same condition)
            # If k==2i: only one term (j=i), derivative = 2*a[i]
            # If k!=2i and 0<=k-i<=n-1: two terms, derivative = 2*a[k-i]
            # If k-i out of range: derivative = 0

            # So d[k] = 2*a[k-i] if 0<=k-i<=n-1 else 0
            # But for k=2i: d[2i] should be special since b_new[2i] = b[2i] + v^2 - ai^2
            # Let's handle k=2i separately.
            # For k != 2i: b_new[k] = b[k] + (v-ai)*2*a[k-i] (if 0<=k-i<=n-1)
            # For k = 2i: b_new[2i] = b[2i] + v^2 - ai^2

            # Precompute linear coefficients
            lin_coef = np.zeros(2 * n - 1)
            valid_range = range(max(0, i), min(2 * n - 1, i + n - 1) + 1)
            for k in valid_range:
                ki = k - i
                if 0 <= ki <= n - 1 and k != 2 * i:
                    lin_coef[k] = 2.0 * a[ki]

            S_ai = S - ai  # S minus a[i]

            def obj_v(v):
                S_new = S_ai + v
                if S_new < 0.01:
                    return 1e10
                # Compute new max
                # Linear terms
                new_b = b + (v - ai) * lin_coef
                # Quadratic term at k=2i
                if 2 * i < 2 * n - 1:
                    new_b[2 * i] = b[2 * i] + v * v - ai * ai
                return 2.0 * n * new_b.max() / (S_new * S_new)

            # Find optimal v
            v_max = max(2.0 * S, 10.0 * max(a), 1.0)
            result = minimize_scalar(obj_v, bounds=(0.0, v_max), method='bounded',
                                     options={'xatol': 1e-12})
            v_opt = max(0.0, result.x)

            if abs(v_opt - ai) > 1e-12:
                a[i] = v_opt
                improved = True

        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J - 1e-13:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_bilinear(b_aa, b_ag, b_gg):
    """Find t in [0,1] minimizing max((1-t)^2*b_aa + 2t(1-t)*b_ag + t^2*b_gg)."""
    def obj(t):
        b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
        return b.max()
    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                             options={'xatol': 1e-9})
    return float(result.x)


def lp_mirror_descent(a, deadline, max_iters=200, use_cd_every=50):
    """LP mirror descent with fast line search and periodic coordinate descent."""
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()

        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)
        t_opt = line_search_bilinear(b_aa, b_ag, b_gg)

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)

        if J_new < best_J - 1e-13:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if use_cd_every > 0 and it % use_cd_every == use_cd_every - 1:
            a_cd = exact_coordinate_descent(a, min(deadline, time.monotonic() + 30), max_passes=3)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-13:
                best_J = J_cd
                best_a = a_cd.copy()
            a = a_cd
            no_improve = 0

        if no_improve > 40:
            a = a + np.random.normal(0, 0.03, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_J


def generate_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n) / n)

    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())

    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())

    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())

    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())

    dbl = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())

    for freq in [1, 2, 3]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    # Geometric
    geom = 0.95 ** np.arange(n)
    inits.append(geom / geom.sum())

    # Two-scale
    ts = 1.0 + 0.2 * np.cos(2 * np.pi * t) + 0.1 * np.cos(2 * np.pi * 5 * t)
    ts = np.maximum(ts, 0.01)
    inits.append(ts / ts.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses soft-max L-BFGS, exact coordinate descent, LP mirror descent,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent)

    n_base = parent.size

    # === PHASE 1: Soft-max L-BFGS from parent ===
    p1_deadline = min(t_start + 400, deadline - 500)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Soft-max L-BFGS (n={n_base}, budget={p1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    a = parent.copy() / parent.sum()
    a_r, J_r = softmax_lbfgs(a, p1_deadline, tau_start=10.0, tau_mult=1.4,
                              max_phases=12, max_iter_per_phase=80)
    update_best(a_r)
    print(f"  Phase 1 L-BFGS done: J={J_r:.8f}", flush=True)

    # Also try with different tau schedules
    for tau_s in [5.0, 30.0]:
        if time.monotonic() > p1_deadline:
            break
        a_r2, J_r2 = softmax_lbfgs(parent / parent.sum(), min(time.monotonic() + 60, p1_deadline),
                                    tau_start=tau_s, tau_mult=1.5,
                                    max_phases=8, max_iter_per_phase=50)
        update_best(a_r2)
        if J_r2 < J_r:
            a_r = a_r2
            J_r = J_r2

    # === PHASE 2: Exact coordinate descent polishing ===
    p2_deadline = min(t_start + 650, deadline - 400)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Exact Coordinate Descent (budget={p2_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_cd = exact_coordinate_descent(best_seq, p2_deadline, max_passes=30)
        update_best(a_cd)

        a_cd2 = exact_coordinate_descent(a_r, min(time.monotonic() + 60, p2_deadline), max_passes=15)
        update_best(a_cd2)
        print(f"  Phase 2 done: best J={best_score:.8f}", flush=True)

    # === PHASE 3: More L-BFGS with higher tau + CD interleave ===
    p3_deadline = min(t_start + 900, deadline - 350)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: High-tau L-BFGS + CD (budget={p3_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_high_tau = softmax_lbfgs(best_seq, p3_deadline, tau_start=100.0, tau_mult=1.3,
                                    max_phases=10, max_iter_per_phase=60)
        update_best(a_high_tau[0])
        a_cd3 = exact_coordinate_descent(a_high_tau[0], min(time.monotonic() + 60, p3_deadline), max_passes=20)
        update_best(a_cd3)
        print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)

    # === PHASE 4: LP mirror descent for global moves ===
    p4_deadline = min(t_start + 1150, deadline - 250)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: LP Mirror Descent (budget={p4_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_lp, J_lp = lp_mirror_descent(best_seq, p4_deadline, max_iters=100, use_cd_every=40)
        update_best(a_lp)
        a_cd4 = exact_coordinate_descent(a_lp, min(time.monotonic() + 30, p4_deadline), max_passes=10)
        update_best(a_cd4)
        print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Multi-length exploration ===
    p5_deadline = min(t_start + 1350, deadline - 120)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Multi-length Exploration (budget={p5_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_b = best_seq.size
        target_lengths = set()
        for mult in [0.75, 1.25, 1.5, 2.0, 0.5]:
            target_lengths.add(int(n_b * mult))
        for L in [256, 384, 448, 512, 640, 768, 1024]:
            target_lengths.add(L)
        target_lengths = sorted([L for L in target_lengths if 128 <= L <= 2048 and L != n_b])

        for tgt_n in target_lengths[:8]:
            if time.monotonic() > p5_deadline:
                break

            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()

            # Quick L-BFGS
            phase_dl = min(time.monotonic() + 25, p5_deadline)
            a_r5, J_r5 = softmax_lbfgs(a_ext, phase_dl, tau_start=10.0, tau_mult=1.5,
                                        max_phases=5, max_iter_per_phase=40)
            update_best(a_r5)
            # Quick CD
            a_cd5 = exact_coordinate_descent(a_r5, min(phase_dl + 10, p5_deadline), max_passes=5)
            update_best(a_cd5)

        print(f"  Phase 5 done: best J={best_score:.8f}", flush=True)

    # === PHASE 6: Diverse initializations ===
    p6_deadline = min(t_start + 1420, deadline - 60)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Diverse Inits (budget={p6_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_d = best_seq.size
        inits = generate_diverse_inits(n_d, rng)
        # Add perturbed best
        inits.append(np.maximum(best_seq + rng.normal(0, 0.05, n_d), 0.001))
        inits[-1] = inits[-1] / inits[-1].sum()

        init_scores = [(compute_J(init), init) for init in inits]
        init_scores.sort(key=lambda x: x[0])

        for sc, init in init_scores[:5]:
            if time.monotonic() > p6_deadline:
                break
            phase_dl = min(time.monotonic() + 20, p6_deadline)
            a_r6, J_r6 = softmax_lbfgs(init, phase_dl, tau_start=10.0, tau_mult=1.5,
                                        max_phases=5, max_iter_per_phase=30)
            update_best(a_r6)
            a_cd6 = exact_coordinate_descent(a_r6, min(phase_dl + 10, p6_deadline), max_passes=5)
            update_best(a_cd6)

        print(f"  Phase 6 done: best J={best_score:.8f}", flush=True)

    # === PHASE 7: Final intensive polishing ===
    p7_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 7: Final Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        # Final L-BFGS with high tau
        a_f1, J_f1 = softmax_lbfgs(best_seq, min(t_start + budget_s - 50, p7_deadline),
                                    tau_start=200.0, tau_mult=1.2,
                                    max_phases=5, max_iter_per_phase=50)
        update_best(a_f1)
        # Final CD
        a_f2 = exact_coordinate_descent(a_f1, p7_deadline, max_passes=20)
        update_best(a_f2)
        # One more L-BFGS on CD result
        a_f3, J_f3 = softmax_lbfgs(a_f2, min(time.monotonic() + 30, p7_deadline),
                                    tau_start=300.0, tau_mult=1.2,
                                    max_phases=3, max_iter_per_phase=40)
        update_best(a_f3)
        a_f4 = exact_coordinate_descent(a_f3, p7_deadline, max_passes=10)
        update_best(a_f4)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
