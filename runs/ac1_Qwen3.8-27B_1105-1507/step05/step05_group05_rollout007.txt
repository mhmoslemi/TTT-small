Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using fast coordinate descent with

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def update_b_incremental(b, a, i, old_val, new_val, n):
    """Update convolution b when a[i] changes from old_val to new_val."""
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return
    # Affected k range: k where both i and k-i are in [0, n-1]
    k_lo = max(0, i)  # i >= 0 always
    k_hi = min(2 * n - 2, i + n - 1)
    
    # For each k in [k_lo, k_hi]:
    # If k == 2*i: b[k] += 2*old_val*delta + delta^2
    # Else: b[k] += 2*delta*a[k-i]
    for k in range(k_lo, k_hi + 1):
        j = k - i
        if 0 <= j < n:
            if k == 2 * i:
                b[k] += 2 * old_val * delta + delta * delta
            else:
                b[k] += 2 * delta * a[j]


def find_optimal_x(a, i, b, n, base_max_uneff):
    """Find optimal a[i] given all other coordinates and current b.
    Uses golden section search on the 1D objective."""
    old_x = a[i]
    S = a.sum()  # current sum
    
    k_lo = max(0, i)
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Precompute for affected positions

    affected = np.arange(k_lo, k_hi + 1)
    c = np.zeros(len(affected))
    slope = np.zeros(len(affected))
    quad_pos = -1
    
    for idx, k in enumerate(affected):
        j = k - i
        if 0 <= j < n:
            if k == 2 * i:
                c[idx] = b[k] - old_x * old_x
                slope[idx] = 0.0
                quad_pos = idx
            else:
                c[idx] = b[k] - 2 * old_x * a[j]
                slope[idx] = 2 * a[j]
        else:
            c[idx] = b[k]
            slope[idx] = 0.0
    
    def obj(x_test):
        new_sum = S - old_x + x_test
        if new_sum <= 1e-10:
            return 1e10
        # max of convolution / new_sum^2
        m = base_max_uneff
        vals = c + slope * x_test
        m = max(m, np.max(vals) if len(vals) > 0 else 0)
        if quad_pos >= 0:
            qv = c[quad_pos] + x_test * x_test
            m = max(m, qv)
        return m / (new_sum * new_sum)
    
    # Golden section search
    lo = 0.0
    hi = max(old_x * 3.0, 0.1, np.sqrt(max(b[k_lo:k_hi+1].max(), 0)) * 1.5)

    return max(0.0, x_opt)

    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]

    theta = (cssv[rho] - 1) / (rho + 1.0)

def fista_smooth(a, max_iters, deadline, tau_start=10.0, tau_end=200.0, lr_base=0.01):
    """FISTA on log-sum-exp smoothed max(conv) with simplex constraint."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_val = fast_conv(a).max()
    
    y = a.copy()
    t = 1.0
    y_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        # Lipschitz constant upper bound: roughly 4*tau*max(a) ... use adaptive
        # For simplicity, use a fixed learning rate that works
        lr = lr_base / (1 + 0.1 * it)

        # Project onto simplex
        y_new = project_simplex(y_new)
        
        # FISTA update

        y = y_new + ((t - 1) / t_new) * (y_new - y_prev)
        y = project_simplex(y)
        y_prev = y_new
        t = t_new
        
        # Check actual objective
        M = fast_conv(y).max()
        if M < best_val - 1e-12:
            best_val = M
            best_a = y.copy()
    
    return best_a

    # Precompute
    affected_k = np.arange(k_lo, k_hi + 1)
    c = np.empty(len(affected_k))
    slope = np.zeros(len(affected_k))
    quad_idx = -1
    
    for idx, k in enumerate(affected_k):
        j = k - i
        if 0 <= j < n:
            if k == 2 * i:
                c[idx] = b[k] - old_x * old_x
                quad_idx = idx
            else:
                c[idx] = b[k] - 2 * old_x * a[j]
                slope[idx] = 2.0 * a[j]
        else:
            c[idx] = b[k]
    
    # Base max outside affected range
    if k_lo > 0:
        base = b[:k_lo].max()
    else:
        base = 0.0
    if k_hi < 2 * n - 2:
        base = max(base, b[k_hi + 1:].max())
    
    # Golden section
    lo = 0.0
    hi_v = max(old_x * 2.0 + 0.1, 0.1)
    
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi_v - gr * hi_v
    b_pt = gr * hi_v
    fa = _eval_obj(a_pt, old_x, S, c, slope, quad_idx, base)
    fb = _eval_obj(b_pt, old_x, S, c, slope, quad_idx, base)
    
    for _ in range(30):
        if fa < fb:
            hi_v = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi_v - gr * (hi_v - lo)
            fa = _eval_obj(a_pt, old_x, S, c, slope, quad_idx, base)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi_v - lo)
            fb = _eval_obj(b_pt, old_x, S, c, slope, quad_idx, base)
    
    x_opt = (lo + hi_v) / 2.0
    f0 = _eval_obj(0.0, old_x, S, c, slope, quad_idx, base)
    fx = _eval_obj(x_opt, old_x, S, c, slope, quad_idx, base)
    if f0 < fx:
        x_opt = 0.0
    return x_opt


def _eval_obj(x, old_x, S, c, slope, quad_idx, base):
    new_sum = S - old_x + x
    if new_sum <= 1e-10:
        return 1e10
    m = base
    vals = c + slope * x
    mv = vals.max()
    if mv > m:
        m = mv
    if quad_idx >= 0:
        qv = c[quad_idx] + x * x
        if qv > m:
            m = qv
    return m / (new_sum * new_sum)


def run_cd(a, max_sweeps, deadline, seed=0):
    """Coordinate descent with FFT-based convolution (recomputed each sweep)."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        
        b = fast_conv(a)
        S = a.sum()
        
        for i in order:
            x_opt = find_opt_x(a, i, n, b)
            old_x = a[i]
            if abs(x_opt - old_x) > 1e-12:
                a[i] = x_opt
                # Update b incrementally
                delta = x_opt - old_x
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if k == 2 * i:
                            b[k] += 2 * old_x * delta + delta * delta
                        else:
                            b[k] += 2 * delta * a[j]
        
        s = a.sum()
        if s < 1e-10:
            break
        if abs(s - 1.0) > 1e-8:
            a = a / s
        
        M = fast_conv(a).max()
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20:
            break
    
    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
