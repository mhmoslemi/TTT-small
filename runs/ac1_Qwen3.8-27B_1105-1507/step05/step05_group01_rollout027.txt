"""
Enhanced LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive peak selection, multi-scale search, and intensive polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_conv_matrix(a):
    """Build the convolution matrix A where (A @ g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # g[j] contributes a[i] to position k = i + j for valid i
        # (conv(a,g))[k] = sum_i a[i] * g[k-i]
        # So column j of A has A[k,j] = a[k-j] for valid k-j
        for k in range(j, j + n):
            if k < m:
                A[k, j] = a[k - j]
    return A

def solve_lp(a, M, constraint_indices=None, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax for selected k, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    
    if constraint_indices is None:
        constraint_indices = np.arange(m)
    
    A_full = build_conv_matrix(a)
    A = A_full[constraint_indices, :]
    b = np.full(len(constraint_indices), M * relax)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, 0.0

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def conv_matrix_times(a, g):
    """Compute (a * g) via FFT (convolution of a and g)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    conv = irfft(fa * fg, size)
    return conv[:2 * n - 1]


def build_conv_matrix(a):
    """Build matrix A where (A @ g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For column j: rows j to j+n-1 have values a[0], a[1], ..., a[n-1]
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M, indices=None, relax=1.0):
    """
    Solve: max sum(g) s.t. (a*g)[k] <= M*relax for k in indices, g >= 0.
    Returns (g, sum_g) or (None, 0).
    """
    n = len(a)
    m = 2 * n - 1
    
    if indices is None:
        indices = np.arange(m)
    
    idx = np.asarray(indices)
    A = build_conv_matrix(a)[idx, :]
    b = np.full(len(idx), M * relax)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-15:
                return g, s
    except Exception:
        pass
    return None, 0.0

def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.01)

Enhanced LP-based alternating optimization with adaptive peak selection,
multi-length exploration, dual-guided perturbations, and intensive polishing
to minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build matrix A where (A @ g)[k] = (conv(a, g))[k]. Shape: (2n-1, n)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M, indices=None, relax=1.0):
    """
    Solve: max sum(g) s.t. (a*g)[k] <= M*relax for k in indices, g >= 0.
    Returns (g, sum_g) or (None, 0).
    """
    n = len(a)
    m = 2 * n - 1

    if indices is None:
        indices = np.arange(m)

    idx = np.asarray(indices, dtype=int)
    if len(idx) == 0:
        return None, 0.0

    A = build_conv_matrix(a)
    A_sub = A[idx, :]
    b = np.full(len(idx), M * relax)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A_sub, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0


def lp_optimize(a, time_budget=100.0, verbose=False):
    """
    Core LP alternating optimization with adaptive strategies.
    Input a should be non-negative. Will normalize internally.
    Returns optimized (normalized) a.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    iteration = 0
    patience = 30

    while time.monotonic() - start < time_budget:
        iteration += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.05:
            break

        conv = compute_conv(best_a)
        M = float(np.max(conv))

        # Adaptive peak selection
        m = 2 * n - 1
        if n <= 600:
            K = m
        elif n <= 1000:
            K = int(1.5 * n)
        else:
            K = int(1.2 * n)
        K = min(K, m)

        # Adaptive relaxation based on stall
        relax = 1.0
        if stall > 10:
            relax = min(1.0 + 0.05 * (stall - 10), 1.5)
        if stall > 30:
            relax = min(1.0 + 0.1 * (stall - 10), 2.0)

        sorted_idx = np.argsort(conv)[::-1][:K]

        g, S = solve_lp(best_a, M, indices=sorted_idx, relax=relax)

        if g is None:
            stall += 1
            if stall > patience:
                _handle_stall(best_a, n, iteration, relax)
            continue

        if S < 1e-15:
            stall += 1
            continue

        # Line search: try convex combination of best_a and g/sum(g)
        g_norm = g / S
        improved = False
        best_t_obj = best_obj
        best_a_try = best_a

        t_values = [1.0, 0.5, 0.7, 0.3, 0.85, 0.9, 0.6, 0.4, 0.15, 0.2]
        for t in t_values:
            a_try = (1.0 - t) * best_a + t * g_norm
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_try = a_try.copy()

        if best_t_obj < best_obj - 1e-15:
            best_obj = best_t_obj
            best_a = best_a_try
            stall = 0
            patience = max(15, min(50, patience + 1))
        else:
            stall += 1
            if stall > patience:
                # Try harder perturbation
                a_pert = _perturb_escape(best_a, n, stall, iteration)
                if a_pert is not None:
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_obj:
                        best_obj = obj_pert
                        best_a = a_pert
                    else:
                        # Still use LP from perturbed point
                        best_a = a_pert
                    stall = 0
                    patience = max(10, patience // 2)

        if verbose and iteration % 50 == 0:
            print(f"    LP it={iteration}, obj={best_obj:.8f}, stall={stall}, relax={relax:.2f}, S={S:.4f}")

    if verbose:
        print(f"    LP done: iters={iteration}, obj={best_obj:.8f}")

    return best_a

def _perturb_escape(a, n, stall, iteration):
    """Generate a perturbation to escape local optimum."""
    rng = np.random.default_rng(iteration * 31 + 7 + stall)
    choice = rng.integers(0, 4)

    if choice == 0:
        # Gaussian perturbation
        scale = min(0.1, 0.02 + 0.005 * stall)
        a_new = a * np.exp(rng.normal(0, scale, n))
    elif choice == 1:
        # Peak-targeted reduction
        conv = compute_conv(a)
        top_peaks = np.argsort(conv)[::-1][:min(20, 2 * n - 1)]
        a_new = a.copy()
        for k in top_peaks[:5]:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    factor = rng.uniform(0.8, 0.95)
                    if a_new[i] >= a_new[j]:
                        a_new[i] *= factor
                    else:
                        a_new[j] *= factor
    elif choice == 2:
        # Random block modification
        nblocks = rng.integers(3, 10)
        a_new = a.copy()
        for _ in range(rng.integers(1, 4)):
            start = rng.integers(0, n - 10)
            end = min(start + rng.integers(10, n // 5), n)
            a_new[start:end] *= rng.uniform(0.7, 1.3)
    else:
        # Spectral perturbation
        size = 1
        while size < 2 * n:
            size <<= 1
        fa = rfft(a, size)
        mag = np.abs(fa)
        phase = np.angle(fa)
        mag_new = mag * np.exp(rng.normal(0, 0.05, len(mag)))
        fa_new = mag_new * np.exp(1j * phase)
        a_new = irfft(fa_new, size)[:n]

    a_new = np.maximum(a_new, 0)
    s_new = np.sum(a_new)
    if s_new < 1e-10:
        return None
    return a_new / s_new


def peak_break(a, num_peaks=15, time_budget=10.0):
    """Targeted perturbation to break top peaks in the convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(54321)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(best_a)
        top_k = np.argsort(conv)[::-1][:num_peaks]

        # Pick a peak to attack
        k = int(top_k[rng.integers(0, len(top_k))])

        # Find contributing pairs
        contributions = []
        for i in range(n):
            j = k - i
            if 0 <= j < n and i != j:
                contributions.append((i, j, best_a[i] * best_a[j]))

        if not contributions:
            continue

        contributions.sort(key=lambda x: x[2], reverse=True)

        # Reduce top contributors
        a_new = best_a.copy()
        top_c = contributions[:min(15, len(contributions))]
        for i, j, val in top_c:
            factor = rng.uniform(0.75, 0.95)
            if a_new[i] >= a_new[j]:
                a_new[i] *= factor
            else:
                a_new[j] *= factor

        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

    return best_a


def spectral_optimize(a, time_budget=10.0):
    """Spectral domain optimization to flatten autocorrelation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(77777)

    size = 1
    while size < 2 * n:
        size <<= 1

    while time.monotonic() - start < time_budget:
        fa = rfft(best_a, size)
        mag = np.abs(fa)
        phase = np.angle(fa)

        choice = rng.integers(0, 3)
        if choice == 0:
            # Low-pass
            freqs = np.linspace(0, np.pi, len(fa))
            alpha = rng.uniform(0.5, 3.0)
            lp = np.exp(-alpha * (freqs / np.pi) ** 2)
            mag_new = mag * lp
        elif choice == 1:
            # Equalize magnitude
            target = np.median(mag[mag > 0]) if np.any(mag > 0) else 1.0
            alpha = rng.uniform(0.1, 0.5)
            mag_new = (1 - alpha) * mag + alpha * target
        else:
            # Random spectral perturbation
            mag_new = mag * np.exp(rng.normal(0, 0.05, len(mag)))

        fa_new = mag_new * np.exp(1j * phase)
        a_new = irfft(fa_new, size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

    return best_a


def random_init(n, seed):
    """Generate diverse random initialization, normalized to sum=1."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.3, 3.0)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 1:
        a = rng.random(n)
        win = max(3, n // 25)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 2:
        a = np.ones(n)
        f1 = rng.uniform(0.5, 2.0)
        f2 = rng.uniform(3.0, 8.0)
        a += 0.3 * np.cos(2 * np.pi * f1 * k / n)
        a += 0.2 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 3:
        nblocks = rng.integers(3, 12)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 4:
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.08, 0.3)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 5:
        # Two-lobe
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * 0.1
        a = 1.0 + 0.4 * np.exp(-0.5 * ((k - c1) / w) ** 2) + 0.4 * np.exp(-0.5 * ((k - c2) / w) ** 2)
    elif choice == 6:
        # Sawtooth-like
        period = n // rng.integers(3, 8)
        a = 1.0 + 0.3 * np.abs(2 * ((k % period) / period) - 1)
    else:
        a = rng.uniform(0.5, 1.5, n)

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s


def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return np.maximum(a.copy(), 0.01)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.01)

    deadline = start_time + float(budget_s) - 45.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.8f}")

    # === Phase 1: Intensive LP on current best ===
    if best_seq is not None:
        print("\n=== Phase 1: LP on Best ===")
        n_best = len(best_seq)
        budget1 = min(350.0, (deadline - time.monotonic()) * 0.30)
        print(f"  n={n_best}, budget={budget1:.0f}s")

        a_opt = lp_optimize(best_seq, time_budget=budget1, verbose=True)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  After LP: {best_score:.8f}")
        else:
            print(f"  After LP: {obj_opt:.8f} (no improvement)")

        # Peak break + short LP
        remaining = deadline - time.monotonic()

            a_pb = peak_break(best_seq, num_peaks=20, time_budget=min(20.0, remaining * 0.08))
