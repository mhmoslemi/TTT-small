           running += a_new
           count += 1
           a = a_new
       
       if count > 50:
           avg = running / count
           avg = np.maximum(avg, 0.0)
           s = avg.sum()
           if s > 1e-10:
               avg /= s
               M_avg = float(fast_conv(avg).max())
               if M_avg < best_M:
                   best_a = avg
       return best_a
   
   
   def resample_seq(src, n):
       src = np.asarray(src, dtype=np.float64)
       if src.size == n:
           return src.copy()
       t_old = np.linspace(0, 1, src.size)
       t_new = np.linspace(0, 1, n)
       return np.maximum(np.interp(t_new, t_old, src), 0.0)
   
   
   def make_inits(n, rng, parent=None):
       inits = []
       t = np.linspace(0, 1, n, endpoint=False)
       if parent is not None:
           p = resample_seq(parent, n)
           inits.append(p.copy())
           inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.005))
           inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.005))
       inits.append(np.ones(n))
       for alpha in [0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
           inits.append(np.maximum(0.2 + 0.8 * (t**alpha + (1 - t)**alpha) / 2, 0.01))
       for f in [1, 2, 3, 4, 5]:
           inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
       inits.append(np.exp(-4 * (t - 0.5)**2) + 0.05)
       half_n = n // 2 + 1
       half = np.maximum(rng.uniform(0.2, 1.0, half_n), 0.01)
       full = np.concatenate([half, half[-2::-1]])[:n]
       inits.append(full)
       return inits
