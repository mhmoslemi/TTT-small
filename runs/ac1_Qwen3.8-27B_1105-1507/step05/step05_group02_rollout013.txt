from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def build_A_matrix(n, a):
    """Build the matrix A where A[k,j] = a[k-j] for valid k-j."""
    # A is (2n-1) x n, largely sparse
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_solve(a, M):

    n = len(a)
    A = build_A_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10.0})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except:
        pass
    return None

    cols_list = []

    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Polyak averaging
    avg_a = np.zeros(n)
    avg_count = 0
    
    # Lower bound on M*
    M_lb = 1.0 / (2 * n - 1)  # from sum of all lags = 1
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
        
        # Check convergence
        if M - M_lb < 1e-10:
            break
        
        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        A = build_A_sparse(n, a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5.0})
        
        if not r.success or r.x is None:
            break
        
        g = np.maximum(r.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum  # normalize g to sum to 1
        
        # Polyak step size
        diff = g - a
        diff_norm_sq = np.sum(diff**2)
        if diff_norm_sq < 1e-20:
            break
        t = (M - M_lb) / diff_norm_sq
        t = min(t, 1.0)  # cap at 1
        
        # Update
        a_new = np.maximum((1 - t) * a + t * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Polyak averaging
        avg_a += a_new
        avg_count += 1
        avg_a_current = avg_a / avg_count
        
        # Check if average is better
        M_avg = fast_conv(avg_a_current).max()
        if M_avg < best_M:
            best_M = M_avg
            best_a = avg_a_current.copy()
        
        a = a_new
    
    return best_a

def build_A_sparse(n, a):
    """Build sparse convolution matrix A where A[k,j] = a[k-j] for 0<=k-j<n."""
    rows = np.empty(n * n + n * (n - 1), dtype=np.int32)
    cols = np.empty_like(rows)
    vals = np.empty(n * n + n * (n - 1), dtype=np.float64)
    count = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        width = j_max - j_min + 1
        rows[count:count+width] = k
        cols[count:count+width] = np.arange(j_min, j_max + 1)
        vals[count:count+width] = a[k - np.arange(j_min, j_max + 1)]
        count += width
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(2*n-1, n))


def lp_solve(a, M, time_limit=10.0):
    """Solve max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            g = np.maximum(r.x, 0)
            return g / g.sum()
    except Exception:
        pass
    return None


def lp_mirror_descent(a_init, deadline, max_iters=5000):
    """LP Mirror Descent (Frank-Wolfe) with Polyak step size."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Polyak averaging
    avg_a = np.zeros(n)
    avg_start = 0  # start averaging after this iteration
    avg_count = 0
    
    # Lower bound
    M_lb = 1.0 / (2 * n - 1)
    
    # Track M_lb improvements (use the best M seen as a practical lower bound)
    # Actually, M_lb should be a valid lower bound on the OPTIMUM, not current M.
    # 1/(2n-1) is valid since sum of all convolution values = (sum a)^2 = 1.
    
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp > 500:
            # Try to escape: add small noise
            a = a + np.random.default_rng(it).normal(0, 0.01/n, n)
            a = np.maximum(a, 0)
            a /= a.sum()
            no_imp = 0
            continue
        
        if M - M_lb < 1e-10:
            break
        
        # Solve LP
        g = lp_solve(a, M, time_limit=8.0)
        if g is None:
            break
        
        # Polyak step size
        diff = g - a
        diff_norm_sq = np.dot(diff, diff)
        if diff_norm_sq < 1e-20:
            break
        t_polyak = (M - M_lb) / diff_norm_sq
        t = min(t_polyak, 1.0)
        
        # Also try standard FW step size
        t_fw = 2.0 / (it + 2.0)
        
        # Try both and pick the better one
        a_p = np.maximum((1 - t) * a + t * g, 0)
        s_p = a_p.sum()
        if s_p > 1e-10:
            a_p /= s_p
            M_p = fast_conv(a_p).max()
        else:
            M_p = float('inf')
        
        a_fw = np.maximum((1 - t_fw) * a + t_fw * g, 0)
        s_fw = a_fw.sum()
        if s_fw > 1e-10:
            a_fw /= s_fw
            M_fw = fast_conv(a_fw).max()
        else:
            M_fw = float('inf')
        
        if M_p <= M_fw:
            a = a_p
        else:
            a = a_fw
        
        # Start averaging after 50 iterations
        if it >= 50:
            avg_a += a
            avg_count += 1
            if avg_count % 10 == 0 or it == max_iters - 1:
                avg_current = avg_a / avg_count
                M_avg = fast_conv(avg_current).max()
                if M_avg < best_M:
                    best_M = M_avg
                    best_a = avg_current.copy()
    
    # Final check of average
    if avg_count > 0:
        avg_final = avg_a / avg_count
        M_final = fast_conv(avg_final).max()
        if M_final < best_M:
            best_a = avg_final.copy()
            best_M = M_final
    
    return best_a


def coordinate_descent(a_init, deadline, max_sweeps=50):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a_init.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        any_improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            old_val = a[i]
            # Search range: [0, min(2*old_val + 0.5/n, 1)]
            hi = min(2 * old_val + 0.5 / n, 1.0)
            if hi < 1e-12:
                hi = 1.0 / n
            
            # Evaluate at a few points
            candidates = [0.0, old_val, hi * 0.25, hi * 0.5, hi * 0.75, hi, old_val * 0.5]
            # Add the point that would equalize the top constraint
            b = fast_conv(a)
            top_k = np.argmax(b)
            # (a*a)_{top_k} as function of x = a[i]:
            # If top_k == 2i: (a*a)_{2i} = B + x^2
            # Else: (a*a)_k = B + 2*x*a[k-i]
            
            best_x = old_val
            best_obj = float('inf')
            
            for x in candidates:
                if x < -1e-12:
                    continue
                x = max(0.0, x)
                a[i] = x
                s = a.sum()
                if s < 1e-10:
                    continue
                a_norm = a / s
                b_new = fast_conv(a_norm)
                obj = b_new.max()
                if obj < best_obj:
                    best_obj = obj
                    best_x = x
            
            # Also try a few more points around the best
            if best_x > 0:
                for frac in [0.8, 0.9, 1.1, 1.2]:
                    x = best_x * frac
                    if x > 1.0:
                        continue
                    a[i] = x
                    s = a.sum()
                    if s < 1e-10:
                        continue
                    a_norm = a / s
                    b_new = fast_conv(a_norm)
                    obj = b_new.max()
                    if obj < best_obj:
                        best_obj = obj
                        best_x = x
            
            a[i] = best_x
            s = a.sum()
            if s > 1e-10:
                a = a / s
                M_new = fast_conv(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    any_improved = True
        
        if not any_improved:
            break
    
    return best_a


def smooth_gd_nesterov(a_init, deadline, max_iters=5000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.4, lr_end=0.005):
    """Smooth gradient descent with Nesterov acceleration."""
    a = np.maximum(a_init.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        grad = np.zeros(n)
        active = np.where(p > 1e-10)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * p
