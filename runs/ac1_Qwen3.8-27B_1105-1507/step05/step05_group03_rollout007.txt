        return 1
    x -= 1
    x |= x >> 1
    x |= x >> 2
    x |= x >> 4
    x |= x >> 8
    x |= x >> 16
    x |= x >> 32
    return x + 1


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = next_power_of_2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = next_power_of_2(n + m - 1)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def cross_corr(a, p):
    """
    Compute cross-correlation g[i] = sum_j a[j] * p[j+i] for i=0,...,len(a)-1.
    a has length n, p has length 2n-1.
    """
    n = len(a)
    size = next_power_of_2(n + len(p) - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    result = np.fft.irfft(fa * np.conj(fp), size)
    return result[:n]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np
