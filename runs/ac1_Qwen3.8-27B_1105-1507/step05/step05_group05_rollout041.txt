def full_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = full_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    return ex / ex.sum()


def update_conv_single(a, b, i, new_val):
    """Update convolution b after changing a[i] to new_val. Returns new b (modified in place)."""
    n = len(a)

    if abs(new_val - old_val) < 1e-15:

    delta = new_val - old_val
    k_lo = max(0, i)
    k_hi = min(2 * n - 2, i + n - 1)

    """Coordinate descent with incremental convolution updates."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        a = np.ones_like(a)
        S = a.sum()
    a = a / S
    S = 1.0
    n = len(a)

    b = full_conv(a)
    best_M = b.max()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        order = np.arange(n)
        rng.shuffle(order)

        for i in order:
            S = a.sum()
            if S < 1e-10:
                break
            x_opt = coord_update_fast(a, b, i, S)
            if abs(x_opt - a[i]) > 1e-14:
                b = update_conv_single(a, b, i, x_opt)

        S = a.sum()
        if S < 1e-10:
            break
        a_old_sum = a.sum()
        a = a / a_old_sum
        S = 1.0
        b = b / (a_old_sum * a_old_sum)

        M = b.max()
        if M < best_M:
            best_M = M

    return a, best_M


def smooth_obj_and_grad(x, tau):
    """Objective and gradient for L-BFGS using softmax parameterization.
    x is the log-parameter, a = softmax(x).
    Returns (f, grad_x) where f = smooth_max(conv(a,a)) and grad_x is w.r.t. x.
    """
    n = len(x)
    a = softmax(x)
    b = full_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    weights = exp_shifted / Z

    # g[i] = sum_k weights[k] * db[k]/da[i] = 2 * sum_k weights[k] * a[k-i] (when valid)
    # This is 2 * (cross-correlation of weights with a) evaluated appropriately
    # (weights * a)[i] = sum_m weights[i+m] * a[m] for valid m
    # Use FFT for efficiency
    size = 1
    while size < 2 * n:
        size <<= 1
    W = np.fft.rfft(weights, size)
    A = np.fft.rfft(a, size)
    # We want: g[i] = 2 * sum_{m=0}^{n-1} weights[i+m] * a[m]
    # This is the cross-correlation: (weights ⋆ a)[i] = sum_m weights[i+m] * a[m]
    # = (flip(weights) * a)[2n-2-i] ... let me just use direct computation for clarity
    # Actually: sum_m weights[i+m] * a[m] is the (i-th) element of the convolution
    # of weights with a, where we define (w*a)[i] = sum_m w[i+m]*a[m]
    # This is equivalent to: ifft(fft(weights, size) * conj(fft(a, size))) ... no.
    # Let me just use: for each i, sum over m=0..n-1 of weights[i+m]*a[m]
    # This is a "sliding dot product" which is a cross-correlation.
    # np.correlate(weights, a, mode='valid') gives sum_m weights[m+i]*a[m] for i=0..len(weights)-len(a)
    # = sum_m weights[m+i]*a[m] which is what we want!
    cross = np.correlate(weights[:2*n-1], a, mode='valid')
    # cross[i] = sum_m weights[i+m] * a[m] for i = 0, ..., (2n-1) - n = n-1
    g = 2.0 * cross

    # Project to tangent space of simplex:
    # grad_x[m] = a[m] * (g[m] - sum_i g[i]*a[i])
    weighted_sum = np.dot(g, a)
    grad_x = a * (g - weighted_sum)

    f = b_max + np.log(Z) / tau
    return f, grad_x


def run_lbfgs_smooth(a_init, deadline, tau_start=3.0, tau_end=80.0, max_iter=500, max_tau_iterations=50):
    """L-BFGS on smooth objective with increasing tau."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-10)
    a_init = a_init / a_init.sum()

    best_a = a_init.copy()
    best_M = full_conv(a_init).max()

    x = np.log(a_init + 1e-10)

    # Anneal tau from tau_start to tau_end over max_tau_iterations
    it_per_tau = max(1, max_iter // max_tau_iterations)

    for tau_idx in range(max_tau_iterations):
        if time.monotonic() > deadline:
            break
        frac = tau_idx / max(1, max_tau_iterations - 1)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)

        # Run L-BFGS for a few iterations at this tau
        it_count = 0
        while it_count < it_per_tau and time.monotonic() < deadline:
            f0, g0 = smooth_obj_and_grad(x, tau)
            if not np.all(np.isfinite(g0)):
                break
            result = minimize(
                smooth_obj_and_grad, x, args=(tau,),
                jac=True, method='L-BFGS-B',
                options={'maxiter': 5, 'gtol': 1e-6, 'maxls': 20}
            )
            x = result.x
            it_count += 1

        a = softmax(x)
        a = a / a.sum()
        M = full_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()

    return best_a


def peak_target_search(a, deadline, n_trials=200, seed=0):
    """Perturb coordinates near peak positions to reduce them."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = full_conv(a).max()

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        b = full_conv(a)
        peaks = np.where(b > 0.95 * b.max())[0]
        peak_set = set(int(p) for p in peaks)

        # Pick random peak positions and try to reduce them
        a_trial = a.copy()
        n_perturb = rng.integers(2, min(8, n // 4) + 1)
        for _ in range(n_perturb):
            # Pick a coordinate that contributes to a peak
            k = int(peak_set[rng.integers(0, len(peak_set))]) if len(peak_set) > 0 else n // 2
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            j = int(rng.integers(j_lo, j_hi + 1))
            # Perturb: add or subtract a small amount
            direction = rng.choice([-1, 1])
            magnitude = rng.uniform(0.01, 0.05) * a.max()
            a_trial[j] = max(0.0, a_trial[j] + direction * magnitude * rng.uniform(0.5, 2.0))

        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s
        M = full_conv(a_trial).max()
        if M < best_M * (1 - 1e-8):
            best_M = M
            best_a = a_trial.copy()
            a = a_trial.copy()

    return best_a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        from scipy.interpolate import interp1d
        if len(parent) != n:
            p_resampled = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        else:
            p_resampled = parent.copy()
        p_resampled = np.maximum(p_resampled, 0.01)
        p_resampled = p_resampled / p_resampled.sum()
        inits.append(p_resampled.copy())
        inits.append(np.maximum(p_resampled + rng.normal(0, 0.03, n), 0.001))
        inits.append(np.maximum(p_resampled + rng.normal(0, 0.1, n), 0.001))

    inits.append(np.ones(n) / n)
    inits.append(np.maximum(rng.uniform(0.3, 1.0, n), 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()

    # Window functions
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01))
    inits.append(np.maximum(0.42 + 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t), 0.01))  # Blackman
    for w in [0.5, 0.8, 1.0, 1.5, 2.0]:
        inits.append(np.maximum(np.exp(-w * (t - 0.5) ** 2 * 4), 0.01))
    for i in range(len(inits)):
        inits[i] = inits[i] / inits[i].sum()

    # Multi-peak structures
    for n_peaks in [2, 3, 5]:
        centers = np.linspace(0.1, 0.9, n_peaks)
        sig = 0.08
        val = sum(np.exp(-0.5 * ((t - c) / sig) ** 2) for c in centers)
        inits.append(np.maximum(val, 0.01))
    for i in range(len(inits)):
        inits[i] = inits[i] / inits[i].sum()

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.05, 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()

    return inits


def resample_sequence(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = float(score)
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Multi-start coordinate descent (fast incremental updates)
    Phase 2: L-BFGS on smoothed objective
    Phase 3: Refined coordinate descent + peak targeting
    Phase 4: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        tracker.update(parent)

    # === PHASE 1: Multi-start coordinate descent ===
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-start Coordinate Descent", flush=True)
    print(f"{'='*60}", flush=True)

    lengths_to_try = [384, 512, 640, 768, 896]
    # Allocate time proportional to what's likely to help
    time_per_n = {384: 60, 512: 120, 640: 80, 768: 70, 896: 60}
    total_cd_time = 400  # seconds for phase 1

    for n in lengths_to_try:
        if time.monotonic() > deadline - (total_cd_time - (time.monotonic() - p1_start)) - 500:
            break
        t_alloc = time_per_n.get(n, 60)
        phase_dl = min(time.monotonic() + t_alloc, deadline - 400)
        if time.monotonic() >= phase_dl:
            continue

        t0 = time.monotonic()
        inits = generate_initializations(n, rng,
                                         tracker.best if tracker.best is not None else parent)

        best_for_n = None
        best_M_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_dl - 2:
                break
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            # Determine max sweeps based on time
            elapsed_init = time.monotonic() - t0
            remaining = phase_dl - time.monotonic()
            # Estimate: each sweep takes roughly n * n * 5 / 1e9 seconds (rough)
            est_sweep_time = n * n * 3.0 / 1e9
            max_sw = max(5, min(200, int(remaining / max(est_sweep_time, 0.01))))

            a_opt, M_opt = run_coordinate_descent_fast(
                a0, max_sw, phase_dl, seed=seed + idx * 13)

            if M_opt < best_M_for_n:
                best_M_for_n = M_opt
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: M={best_M_for_n:.8f} J={J_val:.6f} ({elapsed:.1f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("Phase 1 complete")
    print(f"Phase 1 total: {time.monotonic() - p1_start:.1f}s", flush=True)

    # === PHASE 2: L-BFGS on smoothed objective ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: L-BFGS Smoothed Optimization", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 250:
        n_best = tracker.best.size
        a_for_lbfgs = tracker.best.copy()

        # Try different tau schedules
        for tau_sched in [
            (2.0, 60.0, 40),
            (5.0, 100.0, 30),
            (3.0, 150.0, 25),
        ]:
            if time.monotonic() > deadline - 200:
                break
            tau_s, tau_e, max_tau_it = tau_sched
            a_lbfgs = run_lbfgs_smooth(
                a_for_lbfgs,
                deadline=min(time.monotonic() + 60, deadline - 180),
                tau_start=tau_s, tau_end=tau_e,
                max_iter=300, max_tau_iterations=max_tau_it)
            tracker.update(a_lbfgs)
            if tracker.best is not a_for_lbfgs:
                a_for_lbfgs = tracker.best.copy()

    tracker.report("Phase 2 complete")
    print(f"Phase 2 total: {time.monotonic() - p2_start:.1f}s", flush=True)

    # === PHASE 3: Refined coordinate descent + peak targeting ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Refined CD + Peak Targeting", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 100:
        # More coordinate descent sweeps
        n_best = tracker.best.size
        a_cd, M_cd = run_coordinate_descent_fast(
            tracker.best, 300, min(time.monotonic() + 80, deadline - 80), seed=77777)
        tracker.update(a_cd)

        # Peak targeting
        if time.monotonic() < deadline - 50:
            a_peak = peak_target_search(
                tracker.best, deadline=min(time.monotonic() + 40, deadline - 40),
                n_trials=500, seed=4242)
            tracker.update(a_peak)

        # Another CD pass on the peak-targeted result
        if time.monotonic() < deadline - 20:
            a_cd2, M_cd2 = run_coordinate_descent_fast(
                tracker.best, 150, deadline - 10, seed=12345)
            tracker.update(a_cd2)

    tracker.report("Phase 3 complete")
    print(f"Phase 3 total: {time.monotonic() - p3_start:.1f}s", flush=True)

    # === PHASE 4: Try different lengths / extensions ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        # Try resampling to slightly different lengths
        for mult in [1.25, 1.5, 0.75]:
            if time.monotonic() > deadline - 15:
                break
            new_n = int(tracker.best.size * mult)
            if new_n < 128 or new_n > 2048:
                continue
            a_resamp = resample_sequence(tracker.best, new_n)
            a_resamp = np.maximum(a_resamp, 1e-8)
            a_resamp = a_resamp / a_resamp.sum()
            a_opt, _ = run_coordinate_descent_fast(
                a_resamp, 100, deadline - 10, seed=99999)
            tracker.update(a_opt)

    tracker.report("Phase 4 complete")
    print(f"Phase 4 total: {time.monotonic() - p4_start:.1f}s", flush=True)

    # === PHASE 5: Final polish ===
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 3:
        # Quick CD
        a_final, _ = run_coordinate_descent_fast(
            tracker.best, 50, deadline - 2, seed=55555)
        tracker.update(a_final)

    # Fallback
    if tracker.best is None:
        fallback = np.ones(512) / 512.0
        tracker.best = fallback
        tracker.best_score = compute_J(fallback)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
