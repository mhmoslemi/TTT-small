    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.01)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(np.maximum(p * rng.uniform(0.7, 1.3, n), 0.01))

    inits.append(np.ones(n, dtype=np.float64))
    inits.append(rng.uniform(0.5, 1.0, n).astype(np.float64))

    for freq in [1, 2, 3, 4, 5, 7, 10]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.05))

    inits.append(np.exp(-3 * (t - 0.5) ** 2) + 0.1)
    inits.append(0.3 + 0.7 * np.abs(2 * t - 1))
    inits.append(np.maximum(0.5 + 0.5 * np.sin(2 * np.pi * 3 * t + 1), 0.05))

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.05))

    # Power law shapes
    inits.append((1.0 - t) ** 0.5 + 0.1)
    inits.append(t ** 0.5 + 0.1)
    inits.append(1.0 / (0.5 + t) + 0.1)

    return inits


def _eval_1d(t, base_max, c_lin, s_lin, c_quad, has_quad, x_old):
    """Evaluate the 1D subproblem objective at t."""
    S = 1.0 - x_old + t
    if S <= 1e-15:
        return 1e10
    m = base_max
    if c_lin.size > 0:
        lin_max = np.max(c_lin + s_lin * t)
        if lin_max > m:
            m = lin_max
    if has_quad:
        qv = c_quad + t * t
        if qv > m:
            m = qv
    return m / (S * S)


def _solve_1d_subproblem(b, a, i, n):
    """
    Solve the 1D subproblem: find optimal a[i] given all other coords.
    a is normalized to sum=1. Returns optimal t >= 0.
    """
    x_old = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # Compute base_max (unaffected entries)
    affected_mask = np.zeros(2 * n - 1, dtype=bool)
    affected_mask[k_lo:k_hi + 1] = True
    if not affected_mask.all():
        base_max = b[~affected_mask].max()
    else:
        base_max = 0.0

    # Build linear and quadratic parts for affected entries
    k_range = np.arange(k_lo, k_hi + 1)
    has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
    quad_idx = -1

    lin_ks = []
    c_lin_list = []
    s_lin_list = []

    for idx, k in enumerate(k_range):
        if k == 2 * i:
            quad_idx = idx
            continue
        j = k - i
        if 0 <= j < n:
            lin_ks.append(k)
            c_lin_list.append(b[k] - 2.0 * x_old * a[j])
            s_lin_list.append(2.0 * a[j])

    c_lin = np.array(c_lin_list, dtype=np.float64) if c_lin_list else np.array([], dtype=np.float64)
    s_lin = np.array(s_lin_list, dtype=np.float64) if s_lin_list else np.array([], dtype=np.float64)
    c_quad = b[2 * i] - x_old * x_old if has_quad else 0.0

    # Evaluate at current point
    f_cur = _eval_1d(x_old, base_max, c_lin, s_lin, c_quad, has_quad, x_old)

    # Golden section search
    lo = 0.0
    hi = max(x_old * 3.0, 0.1, np.sqrt(max(base_max, f_cur - base_max, 0.0)) + 0.1)
    # Cap hi
    hi = min(hi, 1.0 + 0.5)  # Since sum=1, a[i] can't exceed 1

    gr = (np.sqrt(5) - 1) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = _eval_1d(a_pt, base_max, c_lin, s_lin, c_quad, has_quad, x_old)
    fb = _eval_1d(b_pt, base_max, c_lin, s_lin, c_quad, has_quad, x_old)

    for _ in range(30):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = _eval_1d(a_pt, base_max, c_lin, s_lin, c_quad, has_quad, x_old)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = _eval_1d(b_pt, base_max, c_lin, s_lin, c_quad, has_quad, x_old)

    x_opt = (lo + hi) / 2.0

    # Check boundary t=0
    f_zero = _eval_1d(0.0, base_max, c_lin, s_lin, c_quad, has_quad, x_old)
    f_opt = _eval_1d(x_opt, base_max, c_lin, s_lin, c_quad, has_quad, x_old)
    if f_zero < f_opt:
        x_opt = 0.0

    return max(0.0, x_opt)


def run_cd_fast(a, max_sweeps, deadline, rng, momentum=0.0):
    """
    Fast coordinate descent with incremental convolution updates.
    a is normalized to sum=1 on input and output.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s

    b = fast_conv(a)
    best_M = b.max()
    best_a = a.copy()
    no_improve = 0
    update_count = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            x_old = a[i]
            t_opt = _solve_1d_subproblem(b, a, i, n)

            if momentum > 0:
                t_opt = (1.0 - momentum) * x_old + momentum * t_opt

            if abs(t_opt - x_old) > 1e-12:
                # Incrementally update b
                delta = t_opt - x_old
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += t_opt * t_opt - x_old * x_old
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * delta * a[j]
                a[i] = t_opt
                sweep_improved = True
                update_count += 1

        # Renormalize
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        # Rescale b: conv(a/s) = conv(a)/s^2, but a was already the normalized version
        # Actually: we modified a in place without normalizing between updates.
        # Let's recompute b from the normalized a to avoid drift.
        # Only do full recompute every few sweeps for efficiency
        if sweep % 5 == 0 or update_count > 4 * n:
            b = fast_conv(a)
            update_count = 0

        M = b.max()
        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 3
        if no_improve > 20:
            break

    return best_a


def run_cd_with_restarts(a, max_sweeps, deadline, rng, n_restarts=3, perturb_scale=0.05):
    """Coordinate descent with periodic random perturbations."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s

    b = fast_conv(a)
    best_M = b.max()
    best_a = a.copy()

    sweeps_per_restart = max(20, max_sweeps // (n_restarts + 1))

    for restart in range(n_restarts + 1):
        if time.monotonic() > deadline:
            break

        if restart > 0:
            # Perturb from best
            scale = perturb_scale * (1.0 + 0.5 * restart)
            a = best_a + rng.normal(0, scale, n)
            a = np.maximum(a, 1e-4)
            a = a / a.sum()
            b = fast_conv(a)

        a = run_cd_fast(a, sweeps_per_restart, deadline, rng)
        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()

    return best_a


def smooth_max_grad(a, tau):
    """Gradient of soft-max approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2 * pk * a[hi-lo:-1 or all] reversed
        # (a*a)_k = sum_j a_j * a_{k-j}, for j in [lo, hi]
        # d/d a_i of (a*a)_k = 2 * a_{k-i} (if valid) or a_{k-i}+a_{i} (if k-i==i)
        # Actually: d/d a_i (sum_j a_j a_{k-j}) = a_{k-i} + a_i (if 2i != k) or 2a_i (if 2i==k)
        # Wait, more carefully:
        # (a*a)_k = sum_{j: 0<=j<n, 0<=k-j<n} a_j * a_{k-j}
        # d/d a_i = sum of contributions where a_i appears
        # a_i appears as a_j when j=i: contributes a_{k-i} (if 0<=k-i<n)
        # a_i appears as a_{k-j} when k-j=i, i.e., j=k-i: contributes a_{k-i} (if 0<=k-i<n)
        # So d/d a_i = 2*a_{k-i} if 0<=k-i<n (and i != k-i)
        # Or d/d a_i = 2*a_i if 2*i==k (the term a_i^2 gives derivative 2*a_i)
        # Actually for the j=i and j=k-i terms:
        # If i != k-i: two terms each give a_{k-i}, total 2*a_{k-i}
        # If i == k-i (2i=k): one term a_i^2 gives 2*a_i
        # Both cases: 2 * a[k-i] (when k-i valid), since if 2i=k then a[k-i]=a[i]
        if lo <= i <= hi:
            j = k - i
            if 0 <= j < n:
                if 2 * i == k:
                    grad[i] += 2.0 * pk * a[i]
                else:
                    grad[i] += 2.0 * pk * a[j]

    return grad


def run_adam_smooth(a, max_iters, deadline, tau_start=5.0, tau_end=100.0, lr=0.03):
    """Adam optimizer on smooth max objective."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    m = np.zeros(n)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        lr_t = lr / (1.0 + 0.005 * it)

        grad = smooth_max_grad(a, tau)

        # Project to sum-zero subspace
        grad = grad - grad.mean()

        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad * grad
        m_hat = m / (1.0 - 0.9 ** (it + 1))
        v_hat = v / (1.0 - 0.999 ** (it + 1))

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 1e-8)
        s = a_new.sum()
        a_new = a_new / s
        a = a_new

        if it % 20 == 0:
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()

    M = fast_conv(a).max()
    if M < best_M:
        best_M = M
        best_a = a.copy()

    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix where (A @ g)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=20):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_refinement(a, max_iters, deadline):
    """LP-based mirror descent refinement."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=10)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        improved = False
        base_t = max(0.03, 0.4 * (1.0 - 0.2 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.01]:
            a_trial = np.maximum((1.0 - t) * a + t * g, 1e-8)
            a_trial = a_trial / a_trial.sum()
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for the sequence minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses fast coordinate descent with incremental convolution as the primary engine,
    combined with diverse multi-scale initialization, Adam smoothing for basin-hopping,
    and LP-based refinement for final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    # === PHASE 1: Multi-scale initialization (budget ~150s) ===
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-scale init + quick CD (~150s)", flush=True)
    print(f"{'='*60}", flush=True)

    # Start small, build up
    small_lengths = [64, 128, 256]
    for n_small in small_lengths:
        if time.monotonic() > deadline - 1000:
            break
        t0 = time.monotonic()
        alloc = min(40, max(10, int((deadline - 1000 - time.monotonic()) / 2)))
        phase_dl = time.monotonic() + alloc

        inits = generate_diverse_inits(n_small, rng, tracker.best)
        best_for_n = None
        best_M_n = float('inf')

        for a0 in inits:
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-4)
            a0 = a0 / a0.sum()
            a_opt = run_cd_fast(a0, max(30, 2000 // n_small), phase_dl, rng)
            M_val = fast_conv(a_opt).max()
            if M_val < best_M_n:
                best_M_n = M_val
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            print(f"  n={n_small:4d}: J={J_val:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("Phase 1")
    print(f"Phase 1 total: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Main CD at target lengths (budget ~700s) ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Main CD at target lengths (~700s)", flush=True)
    print(f"{'='*60}", flush=True)

    target_n = tracker.best.size if tracker.best is not None else 512
    # Try a range of lengths around the best
    candidate_lengths = sorted(set([
        max(128, target_n - 128),
        target_n,
        target_n + 64,
        target_n + 128,
        min(1024, target_n * 2),
    ]))

    remaining = deadline - p2_start - 250  # reserve 250s for phases 3-4
    per_length = max(30, int(remaining / len(candidate_lengths)))

    for n in candidate_lengths:
        if time.monotonic() > deadline - 250:
            break
        t0 = time.monotonic()
        phase_dl = min(time.monotonic() + per_length, deadline - 250)

        if time.monotonic() >= phase_dl:
            continue

        # Multiple diverse initializations for this length
        inits = generate_diverse_inits(n, rng, tracker.best)
        # Limit number of inits to fit in time
        max_inits = max(3, int(per_length / 15))
        inits = inits[:max_inits]

        best_for_n = None
        best_M_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-4)
            a0 = a0 / a0.sum()

            max_sw = max(50, int(500 / (n / 128)))
            a_opt = run_cd_fast(a0, max_sw, phase_dl, rng, seed_offset=seed + idx)

            M_val = fast_conv(a_opt).max()
            if M_val < best_M_n:
                best_M_n = M_val
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: J={J_val:.6f} ({elapsed:.1f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("Phase 2")
    print(f"Phase 2 total: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Adam smoothing + diverse restarts (budget ~300s) ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Adam + restarts (~300s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        # Adam from current best
        if time.monotonic() < deadline - 150:
            a_adam = run_adam_smooth(tracker.best, 3000,
                                     min(time.monotonic() + 80, deadline - 140),
                                     tau_start=3.0, tau_end=80.0, lr=0.02)
            tracker.update(a_adam)

            # CD after Adam
            if time.monotonic() < deadline - 100:
                a_cd = run_cd_fast(a_adam, 200, min(time.monotonic() + 60, deadline - 90), rng)
                tracker.update(a_cd)

        # Diverse restarts with CD
        if time.monotonic() < deadline - 80:
            n_best = tracker.best.size
            for restart in range(4):
                if time.monotonic() > deadline - 80:
                    break
                scale = 0.03 + 0.04 * restart
                a_pert = tracker.best + rng.normal(0, scale, n_best)
                a_pert = np.maximum(a_pert, 1e-4)
                a_pert = a_pert / a_pert.sum()
                a_opt = run_cd_fast(a_pert, 150, min(time.monotonic() + 40, deadline - 70), rng)
                tracker.update(a_opt)

    tracker.report("Phase 3")
    print(f"Phase 3 total: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: LP refinement + final polish (budget ~remaining) ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: LP refinement + final polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        n_best = tracker.best.size
        # LP refinement (only for moderate sizes)
        if n_best <= 768 and time.monotonic() < deadline - 60:
            est_time = 0.01 * (n_best / 256) ** 2
            max_iters = max(10, min(50, int(40 / max(0.01, est_time))))
            print(f"  LP: n={n_best}, est_iter_time={est_time*1000:.0f}ms, iters={max_iters}", flush=True)
            a_lp = run_lp_refinement(tracker.best, max_iters,
                                     min(time.monotonic() + 50, deadline - 50))
            tracker.update(a_lp)

        # Final CD polish
        if time.monotonic() < deadline - 10:
            a_final = run_cd_fast(tracker.best, 300, deadline - 5, rng)
            tracker.update(a_final)

        # One more short LP pass if time
        if tracker.best is not None and n_best <= 512 and time.monotonic() < deadline - 5:
            a_final2 = run_lp_refinement(a_final, 15, deadline - 1)
            tracker.update(a_final2)

    tracker.report("Phase 4")
    print(f"Phase 4 total: {time.monotonic()-p4_start:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
