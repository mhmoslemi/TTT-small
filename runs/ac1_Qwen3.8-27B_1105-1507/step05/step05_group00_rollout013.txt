Improved optimization combining subgradient descent with adaptive temperature,
active-set KKT refinement, pairwise swap search, and simulated annealing restarts.
Targets minimization of 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)
