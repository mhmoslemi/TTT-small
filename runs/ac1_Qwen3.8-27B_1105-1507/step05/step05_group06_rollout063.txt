    n = len(v)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    # This is 2 * (p * a) reversed... actually it's a cross-correlation

    # Actually, this is just the "sliding dot product" of p with a, which is
    # the cross-correlation at lag i.
    
    # cross_corr(p, a)[i] = sum_k p[k] * a[k-i] (with a[j] = 0 for j outside [0,n-1])
    
    # We can compute this via FFT:
    # cross_corr(p, a) = ifft(FFT(p) * conj(FFT(a)))

def fista_smoothed(a_init, max_iters, deadline, beta=50.0):
    """
    FISTA for minimizing f(a) = (1/beta)*log(sum exp(beta * conv(a,a)_k))
    subject to a in simplex.
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)
    
    y = a.copy()
    t = 1.0
    L = 2.0  # Initial Lipschitz estimate
    best_a = a.copy()
    best_f = compute_J(a)
    current_f = best_f
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        b_y = fast_conv(y)
        g = smooth_gradient(y, b_y, beta)
        
        # FISTA update
        z = y - g / L
        x_new = project_simplex(z)
        
        # Update t and y
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        t = t_new
        a = x_new
        
        # Evaluate
        if it % 10 == 0:
            f_val = compute_J(a)
            if f_val < best_f - 1e-12:
                best_f = f_val
                best_a = a.copy()
        
    # Final evaluation
    f_val = compute_J(a)
    if f_val < best_f:
        best_f = f_val
        best_a = a.copy()
    
    return best_a, best_f

def mirror_descent_smoothed(a_init, max_iters, deadline, beta=50.0, lr=0.1):

    Mirror descent (exponentiated gradient) with log-sum-exp smoothing.
    Uses the dual averaging / mirror descent on the simplex.
    """

    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()

    m = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

            break
        
        b = fast_conv(a)
        g = smooth_gradient(a, b, beta)
        
        # Adam update on the "log" parameter
        # We're optimizing in log-space: theta_i = log(a_i)
        # Gradient in theta-space: g_theta_i = a_i * g_i (chain rule)

def lp_frank_wolfe(a_init, max_iters, deadline):
    """
    Frank-Wolfe with LP best response and backtracking step size.
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_f = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0
        # Build constraint matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True})
            if not r.success or r.x is None:
                break
            g = np.maximum(r.x, 0)
            g_sum = g.sum()
            if g_sum < 1e-15:
                break
            g = g / g_sum
        except Exception:
            break
        
        # Backtracking step size
        best_t = 0
        best_J = compute_J(a)
        for t in [0.5, 0.4, 0.3, 0.2, 0.1, 0.7, 0.8, 0.9, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 0.01:
                continue
            a_try = a_try / s
            J_try = compute_J(a_try)
            if J_try < best_J:
                best_J = J_try
                best_t = t
        
        if best_t == 0:
            # Try smaller steps
            for t in [0.05, 0.02, 0.01]:
                a_try = (1 - t) * a + t * g
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 0.01:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < best_J:
                    best_J = J_try
                    best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_J < best_f - 1e-12:
                best_f = best_J
                best_a = a.copy()
        else:
            break
    
    return best_a, best_f

    Gerchberg-Saxton style algorithm for flat power spectrum design.
    Aims to make conv(a,a) as flat as possible.
    """

    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_f = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Step 1: Compute DFT
        A = np.fft.fft(a)
        
        # Step 2: Flatten the magnitude spectrum
        # We want |A(k)| = C for all k
        # But A(0) = sum(a) = 1 (DC component)
        # For k != 0, set |A(k)| = C
        # Choose C to maintain the same energy: sum|A(k)|^2 = n * sum(a^2)
        # n * sum(a^2) = 1^2 + (n-1) * C^2
        # C^2 = (n * sum(a^2) - 1) / (n-1)
        
        energy = np.sum(np.abs(A) ** 2)
        C_sq = (energy - np.abs(A[0])**2) / (n - 1)

        A_new = A.copy()

                A_new[k] = C * A[k] / np.abs(A[k])

        a_new = np.fft.ifft(A_new).real

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton style flat-spectrum design (key innovation)
2. LP-based Frank-Wolfe with backtracking
3. Efficient incremental coordinate descent
4. Mirror descent with log-sum-exp smoothing
5. Multi-length exploration and diverse initializations
Key insight: flat power spectrum => flat autocorrelation => low peak.
The problem is NON-CONVEX (quadratic forms are indefinite), so global
exploration with diverse initializations is essential.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def flat_spectrum_design(a_init, max_iters, deadline, alpha=1.0):
    """
    Gerchberg-Saxton style algorithm to flatten the power spectrum.
    Flat spectrum => flat autocorrelation => minimized peak.
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_f = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Compute DFT
        A = np.fft.fft(a)

        # Target: equalize non-DC magnitudes
        energy = np.sum(np.abs(A) ** 2)
        dc_mag = np.abs(A[0])
        C_sq = (energy - dc_mag ** 2) / (n - 1)
        C = np.sqrt(max(C_sq, 0))

        # Modify spectrum: relax towards flat magnitude
        A_new = A.copy()
        mags = np.abs(A)
        for k in range(1, n):
            if mags[k] > 1e-15:
                target = C * A[k] / mags[k]
                A_new[k] = (1 - alpha) * A[k] + alpha * target
            elif it % 50 == 0:
                # Occasionally add small random phase to escape zeros
                phase = np.random.uniform(0, 2 * np.pi)
                A_new[k] = C * 0.01 * np.exp(1j * phase)

        # Inverse DFT
        a_new = np.fft.ifft(A_new).real

        # Enforce non-negativity and normalization
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 0.01:
            break
        a_new = a_new / s_new
        a = a_new

        # Evaluate periodically
        if it % 20 == 0:
            f_val = compute_J(a)
            if f_val < best_f - 1e-12:
                best_f = f_val
                best_a = a.copy()

    f_val = compute_J(a)
    if f_val < best_f:
        best_f = f_val
        best_a = a.copy()

    return best_a, best_f


def mirror_descent_smoothed(a_init, max_iters, deadline, beta=30.0, lr=0.05):
    """
    Mirror descent (exponentiated gradient) with log-sum-exp smoothing.
    Update: a_i <- a_i * exp(-lr * (g_i - <a,g>)) then normalize.
    """
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_f = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        # Compute smoothed gradient
        b_max = b.max()
        log_weights = beta * (b - b_max)
        log_weights -= log_weights.max()
        p = np.exp(log_weights)
        p = p / p.sum()

        # Gradient: g_i = 2 * sum_k p_k * a_{k-i}
        # This is 2 * cross-correlation of p and a
        size = 1
        while size < 2 * n:
            size <<= 1
        fp = np.fft.rfft(p, size)
        fa = np.fft.rfft(a, size)
        corr = np.fft.irfft(fp * np.conj(fa), size)
        g = 2.0 * corr[:n]

        # Mirror descent update on simplex
        mean_g = np.dot(a, g)
        updates = g - mean_g
        # Adaptive step size
        max_update = np.max(np.abs(updates))
        if max_update > 1e-15:
            step = lr / (1.0 + it / 1000.0)
            log_a = np.log(a) - step * updates
            log_a -= log_a.max()  # numerical stability
            a = np.exp(log_a)
            a = a / a.sum()
        else:
            break

        if it % 50 == 0:
            f_val = compute_J(a)
            if f_val < best_f - 1e-12:
                best_f = f_val
                best_a = a.copy()

    f_val = compute_J(a)
    if f_val < best_f:
        best_f = f_val
        best_a = a.copy()

    return best_a, best_f


def lp_frank_wolfe(a_init, max_iters, deadline):
    """
    Frank-Wolfe with LP best response and backtracking step size.
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_f = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        # Build LP: max sum(g) s.t. (a*g)_k <= M, g >= 0
        rows = np.empty((2 * n - 1) * n, dtype=np.int32)
        cols = np.empty((2 * n - 1) * n, dtype=np.int32)
        vals = np.empty((2 * n - 1) * n, dtype=np.float64)
        idx = 0
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows[idx] = k
                cols[idx] = j
                vals[idx] = a[k - j]
                idx += 1
        rows = rows[:idx]
        cols = cols[:idx]
        vals = vals[:idx]
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True})
            if not r.success or r.x is None:
                break
            g = np.maximum(r.x, 0)
            g_sum = g.sum()
            if g_sum < 1e-15:
                break
            g = g / g_sum
        except Exception:
            break

        # Backtracking step size
        best_t = 0
        current_f = best_f
        for t in [0.4, 0.3, 0.2, 0.5, 0.6, 0.1, 0.7, 0.8, 0.15]:
            a_try = np.maximum((1 - t) * a + t * g, 0)
            s = a_try.sum()
            if s < 0.01:
                continue
            a_try = a_try / s
            J_try = compute_J(a_try)
            if J_try < current_f - 1e-14:
                current_f = J_try
                best_t = t

        if best_t > 0:
            a = np.maximum((1 - best_t) * a + best_t * g, 0)
            a = a / a.sum()
            if current_f < best_f - 1e-12:
                best_f = current_f
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            break

    return best_a, best_f


def efficient_coord_descent(a_init, deadline, max_passes=100):
    """
    Efficient coordinate descent with incremental convolution updates.
    For each coordinate, try several candidates and pick the best.
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)

    b = fast_conv(a)
    S = 1.0
    max_b = b.max()
    best_f = 2.0 * n * max_b / (S * S)
    best_a = a.copy()
    rng = np.random.default_rng(777)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            ai = a[i]
            if ai < 1e-12:
                candidates = [0.001, 0.005, 0.01, 0.02, 0.05]
            else:
                candidates = [0.0, ai * 0.3, ai * 0.6, ai * 1.4, ai * 1.8, 2.0 * ai]

            best_cand = ai
            current_obj = 2.0 * n * max_b / (S * S)

            k_lo = max(0, i)
            k_hi = min(i + n - 1, 2 * n - 2)

            for cand in candidates:
                delta = cand - ai
                if abs(delta) < 1e-15:
                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue

                # Compute new max efficiently
                # Outside [k_lo, k_hi]
                if k_lo == 0 and k_hi == 2 * n - 2:
                    outside_max = 0.0
                elif k_lo == 0:
                    outside_max = b[k_hi + 1:].max() if k_hi < 2 * n - 2 else 0.0
                elif k_hi == 2 * n - 2:
                    outside_max = b[:k_lo].max()
                else:
                    outside_max = max(b[:k_lo].max(), b[k_hi + 1:].max())

                # Inside [k_lo, k_hi]: b[k] changes by 2*delta*a[k-i] (+ delta^2 if k=2i)
                js = np.arange(k_lo, k_hi + 1) - i
                delta_b = 2.0 * delta * a[js]
                delta_b[js == i] += delta * delta
                new_vals = b[k_lo:k_hi + 1] + delta_b
                inside_max = new_vals.max()

                new_max_b = max(outside_max, inside_max)
                obj = 2.0 * n * new_max_b / (S_new * S_new)

                if obj < current_obj - 1e-14:
                    current_obj = obj
                    best_cand = cand

            if best_cand != ai:
                delta = best_cand - ai
                a[i] = best_cand
                S = S + delta
                js = np.arange(k_lo, k_hi + 1) - i
                delta_b = 2.0 * delta * a[js]
                # For k=2i, use old ai
                old_ai = ai
                delta_b[js == i] = 2.0 * delta * old_ai + delta * delta
                b[k_lo:k_hi + 1] = b[k_lo:k_hi + 1] + delta_b
                max_b = b.max()
                improved = True

        # Renormalize
        if S > 0:
            a = a / S
            S = 1.0
            b = fast_conv(a)  # Recompute to avoid drift
            max_b = b.max()
            f = 2.0 * n * max_b
            if f < best_f - 1e-12:
                best_f = f
                best_a = a.copy()

        if not improved:
            break

    return best_a


def fista_simplex(a_init, max_iters, deadline, beta=50.0):
    """
    FISTA for smoothed objective on the simplex.
    f_beta(a) = (1/beta)*log(sum exp(beta * conv(a,a)_k))
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)

    y = a.copy()
    t = 1.0
    L = 8.0
    best_a = a.copy()
    best_f = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_y = fast_conv(y)
        # Smooth gradient
        b_max = b_y.max()
        log_w = beta * (b_y - b_max)
        log_w -= log_w.max()
        p = np.exp(log_w)
        p = p / p.sum()

        size = 1
        while size < 2 * n:
            size <<= 1
        fp = np.fft.rfft(p, size)
        fy = np.fft.rfft(y, size)
        corr = np.fft.irfft(fp * np.conj(fy), size)
        g = 2.0 * corr[:n]

        z = y - g / L
        x_new = project_simplex(z)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        t = t_new
        a = x_new

        if it % 100 == 0:
            f_val = compute_J(a)
            if f_val < best_f - 1e-12:
                best_f = f_val
                best_a = a.copy()

    f_val = compute_J(a)
    if f_val < best_f:
        best_f = f_val
        best_a = a.copy()

    return best_a, best_f


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    rc = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(rc / rc.sum())

    # Gaussian
    g = np.exp(-3.0 * (t - 0.5) ** 2) + 0.01
    inits.append(g / g.sum())

    # Triangle
    tri = np.maximum(1.0 - np.abs(2 * t - 1), 0.01)
    inits.append(tri / tri.sum())

    # Power law
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        pl = (t + 0.01) ** alpha
        inits.append(pl / pl.sum())
        pl2 = (1 - t + 0.01) ** alpha
        inits.append(pl2 / pl2.sum())

    # Sinusoidal modulations
    for freq in [1, 2, 3, 5, 8]:
        sm = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t)
        sm = np.maximum(sm, 0.01)
        inits.append(sm / sm.sum())

    # Random smooth
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    r = rng.uniform(0, 1, n)
    sm = np.convolve(r, kernel, mode='same')
    sm = np.maximum(sm + 0.05, 0.01)
    inits.append(sm / sm.sum())

    # Double bump
    for pos in [0.3, 0.5, 0.7]:
        db = np.exp(-8 * (t - pos) ** 2) + np.exp(-8 * (t - (1 - pos)) ** 2) + 0.02
        inits.append(db / db.sum())

    # Cosine (Zadoff-Chu real part)
    if n % 2 == 0:
        zc = np.cos(np.pi * np.arange(n) * (np.arange(n) + 1) / n)
        zc = np.abs(zc) + 0.1
        inits.append(zc / zc.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        if time.monotonic() > p2_dl:
            break
        sub_dl = min(time.monotonic() + 50, p2_dl)
        print(f"  Init {idx}: J_init={f_init:.6f}", flush=True)

        # Flat spectrum design
        a_gs, f_gs = flat_spectrum_design(init, 30000, sub_dl, alpha=0.9)
        update_best(a_gs)

        # Quick mirror descent
        a_md, f_md = mirror_descent_smoothed(a_gs, 5000, min(time.monotonic() + 20, p2_dl),
                                             beta=30, lr=0.03)
        update_best(a_md)

        # Coordinate descent polish
        if time.monotonic() < p2_dl - 20:
            a_cd = efficient_coord_descent(a_md, min(time.monotonic() + 20, p2_dl), max_passes=50)
            update_best(a_cd)

    print(f"  Phase 2 done: best J={best_score:.8f}", flush=True)

    # === PHASE 3: LP refinement on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Frank-Wolfe Refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        p3_dl = min(t_start + 1050, deadline - 150)
        a, f = lp_frank_wolfe(best_seq, 50000, p3_dl)
        update_best(a)

        # Follow with coordinate descent
        if time.monotonic() < p3_dl:
            a_cd = efficient_coord_descent(a, min(time.monotonic() + 60, p3_dl), max_passes=200)
            update_best(a_cd)

        # Another LP pass
        if time.monotonic() < p3_dl:
            a2, f2 = lp_frank_wolfe(a_cd, 20000, p3_dl)
            update_best(a2)
            a_cd2 = efficient_coord_descent(a2, min(time.monotonic() + 30, p3_dl), max_passes=100)
            update_best(a_cd2)

    print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)

    # === PHASE 4: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        p4_dl = min(t_start + 1300, deadline - 100)
        n_best = best_seq.size
        target_lengths = set()
        for mult in [0.75, 1.25, 1.5, 2.0, 0.5]:
            target_lengths.add(int(n_best * mult))
        for L in [256, 384, 512, 640, 768, 1024, 1280, 1536]:
            target_lengths.add(L)
        target_lengths = sorted([L for L in target_lengths if 128 <= L <= 3072 and L != n_best])

        for tgt_n in target_lengths[:6]:
            if time.monotonic() > p4_dl:
                break
            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(time.monotonic() + 30, p4_dl)

            # Flat spectrum on new length
            a_gs, f_gs = flat_spectrum_design(a_ext, 20000, sub_dl, alpha=0.9)
            update_best(a_gs)
            # Quick polish
            a_cd = efficient_coord_descent(a_gs, sub_dl, max_passes=30)
            update_best(a_cd)

    print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Final intensive polishing ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        # Multiple short GS runs with different alpha
        for alpha in [0.5, 0.7, 0.95]:
            if time.monotonic() > deadline:
                break
            sub_dl = min(time.monotonic() + 30, deadline)
            a, f = flat_spectrum_design(best_seq, 10000, sub_dl, alpha=alpha)
            update_best(a)

        # Final CD
        if time.monotonic() < deadline:
            a_cd = efficient_coord_descent(best_seq, deadline, max_passes=500)
            update_best(a_cd)

        # Final LP
        if time.monotonic() < deadline - 10:
            a, f = lp_frank_wolfe(a_cd, 5000, deadline)
            update_best(a)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
