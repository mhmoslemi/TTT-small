    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J_internal(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2 internally."""
    a = np.maximum(a, 0)

    if s < 0.01:

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    k_max = min(2 * n - 2, i + n - 1)

    # Constant max outside affected range
    const_max = 0.0
    if k_min > 0:
        m = b[0:k_min].max()
        if m > const_max:
            const_max = m
    if k_max < 2 * n - 2:
        m = b[k_max + 1:2 * n - 1].max()
        if m > const_max:
            const_max = m

    # Affected range: compute c_k and slope for each k
    affected_k = np.arange(k_min, k_max + 1)
    c_vals = np.zeros(len(affected_k))
    slopes = np.zeros(len(affected_k))
    quad_idx = -1

    for idx, k in enumerate(affected_k):
        j = k - i
        if j == i:  # k = 2i, quadratic
            c_vals[idx] = b[k] - x_old * x_old
            slopes[idx] = 0.0
            quad_idx = idx
        elif 0 <= j < n:
            c_vals[idx] = b[k] - 2.0 * x_old * a[j]
            slopes[idx] = 2.0 * a[j]
        else:
            c_vals[idx] = b[k]
            slopes[idx] = 0.0

    return const_max, c_vals, slopes, quad_idx


def coord_obj(x, const_max, c_vals, slopes, quad_idx, quad_c, C):
    """Evaluate the coordinate objective at x."""
    vals = c_vals + slopes * x
    m = const_max
    if vals.size > 0:
        vm = vals.max()
        if vm > m:
            m = vm
    if quad_idx >= 0:
        qv = quad_c + x * x
        if qv > m:
            m = qv
    denom = C + x
    if denom < 1e-20:
        return 1e10
    return m / (denom * denom)


def find_optimal_coord(a, b, i):
    """Find optimal x for coordinate i via golden section search."""
    n = len(a)
    C = a.sum() - a[i]
    x_old = a[i]

    const_max, c_vals, slopes, quad_idx = coord_subproblem_precompute(a, b, i)
    quad_c = c_vals[quad_idx] if quad_idx >= 0 else 0.0

    # Bounds
    lo = 0.0
    hi = max(3.0 * x_old, 0.5, 1.0 / max(C, 1e-10))

    # Expand hi if objective still decreasing
    f_hi = coord_obj(hi, const_max, c_vals, slopes, quad_idx, quad_c, C)
    f_hi_half = coord_obj(hi * 0.5, const_max, c_vals, slopes, quad_idx, quad_c, C)
    while f_hi < f_hi_half and hi < 100.0:
        hi *= 2.0
        f_hi_half = f_hi
        f_hi = coord_obj(hi, const_max, c_vals, slopes, quad_idx, quad_c, C)

    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    fa1 = coord_obj(a1, const_max, c_vals, slopes, quad_idx, quad_c, C)
    fa2 = coord_obj(a2, const_max, c_vals, slopes, quad_idx, quad_c, C)

    for _ in range(45):
        if fa1 < fa2:
            hi = a2
            a2 = a1
            fa2 = fa1
            a1 = hi - gr * (hi - lo)
            fa1 = coord_obj(a1, const_max, c_vals, slopes, quad_idx, quad_c, C)
        else:
            lo = a1
            a1 = a2
            fa1 = fa2
            a2 = lo + gr * (hi - lo)
            fa2 = coord_obj(a2, const_max, c_vals, slopes, quad_idx, quad_c, C)

    x_opt = (lo + hi) / 2.0

    # Check boundary
    f0 = coord_obj(0.0, const_max, c_vals, slopes, quad_idx, quad_c, C)
    fopt = coord_obj(x_opt, const_max, c_vals, slopes, quad_idx, quad_c, C)
    if f0 < fopt:
        x_opt = 0.0

    return max(0.0, x_opt)


def update_conv_inplace(b, a, i, old_x, new_x):
    """Update b = conv(a,a) after changing a[i] from old_x to new_x. Modifies b in place."""
    n = len(a)
    delta = new_x - old_x
    if abs(delta) < 1e-15:
        return
    k_min = i
    k_max = min(2 * n - 2, i + n - 1)
    for k in range(k_min, k_max + 1):
        j = k - i
        if j == i:
            b[k] += new_x * new_x - old_x * old_x
        elif 0 <= j < n:
            b[k] += 2.0 * delta * a[j]


def run_coord_descent(a, deadline, seed=0, max_sweeps=200):
    """Run coordinate descent with incremental convolution updates."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    b = fast_conv(a)
    best_M = b.max()
    best_a = a.copy()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)

        sweep_changed = False
        for i in order:
            x_opt = find_optimal_coord(a, b, i)
            if abs(x_opt - a[i]) > 1e-12:
                old_x = a[i]
                update_conv_inplace(b, a, i, old_x, x_opt)
                a[i] = x_opt
                sweep_changed = True

        # Renormalize
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n)
            b = fast_conv(a)
        else:
            a = a / s
            b = fast_conv(a)

        M = b.max()
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_changed:
            no_improve += 3
        if no_improve >= 20:
            break

    return best_a


def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (A @ g)_k = (a*g)_k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_descent(a, deadline, max_iters=100):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        improved = False
        t0 = max(0.1, 0.5 * (1.0 - 0.5 * it / max(1, max_iters)))
        for t in [t0, t0 * 0.5, t0 * 0.25, t0 * 0.1, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve >= 10:
                break
        else:
            no_improve = 0

    return best_a


def mwu_step(a, b, eta, top_k=20):
    """Multiplicative weight update step based on active convolution terms."""
    n = len(a)
    b_max = b.max()
    # Find top-k terms
    top_indices = np.argsort(b)[-top_k:]
    g = np.zeros(n)
    for k in top_indices:
        weight = (b[k] - 0.5 * b_max) / (b_max + 1e-10)
        if weight <= 0:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            g[j] += weight * a[k - j]
    # Multiplicative update
    a_new = a * np.exp(-eta * g)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.2 * rng.standard_normal(n)), 0.01))

    # Standard windows
    inits.append(np.ones(n))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(np.maximum(0.4 + 0.6 * np.cos(np.pi * t), 0.01))

    # Gaussian
    for sigma in [0.15, 0.25, 0.4]:
        inits.append(np.exp(-0.5 * ((t - 0.5) / sigma) ** 2) + 0.05)

    # Polynomials
    for deg in [2, 4]:
        x = 2 * t - 1
        inits.append(np.maximum(1.0 - x ** deg, 0.01))

    # Random smooth
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.1, 0.01))

    # Sawtooth / triangular
    inits.append(np.abs(2 * t - 1) + 0.1)
    inits.append(1.0 - np.abs(2 * t - 1) + 0.1)

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if self.best is None or a is None:
            if a is not None:
                a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
                if a.size > 0 and a.sum() >= 0.01:
                    score = float(evaluate_sequence(a.tolist()))
                    if np.isfinite(score) and score < self.best_score:
                        self.best = a.copy()
                        self.best_score = score
                        elapsed = time.monotonic() - self.t_start
                        print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: best J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search combining coordinate descent, LP mirror descent,
    MWU, and diverse initialization to minimize J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform n=512, J={tracker.best_score:.8f}", flush=True)

    # Phase 1: Coordinate descent across multiple lengths
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Coordinate Descent (multi-length)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [384, 512, 640, 768, 1024]
    time_p1 = 550.0
    p1_deadline = time.monotonic() + time_p1

    for n in lengths:
        if time.monotonic() > p1_deadline - 30:
            break
        if time.monotonic() >= deadline - 300:
            break

        # Time allocation: give more time to lengths near parent
        if parent is not None:
            dist = abs(n - parent.size)
            weight = 1.0 / (1.0 + dist / 128.0)
        else:
            weight = 1.0

        t_alloc = time_p1 * weight / (len(lengths) * 0.8)
        t_alloc = max(30, min(120, t_alloc))
        phase_dl = min(time.monotonic() + t_alloc, p1_deadline, deadline - 250)

        if time.monotonic() >= phase_dl:
            continue

        t0 = time.monotonic()
        inits = diverse_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            max_sw = 80
            a_opt = run_coord_descent(a0, phase_dl, seed=seed + idx * 13, max_sweeps=max_sw)

            J_val = compute_J_internal(a_opt)
            if J_val < best_J_for_n:
                best_J_for_n = J_val
                best_for_n = a_opt.copy()
                tracker.update(a_opt)

        if best_for_n is not None:
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: best J={best_J_for_n:.8f} ({elapsed:.1f}s)", flush=True)

    tracker.report("Phase 1")

    # Phase 2: LP mirror descent polish
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    p2_deadline = min(time.monotonic() + 300, deadline - 200)

    if tracker.best is not None and time.monotonic() < p2_deadline:
        n_best = tracker.best.size
        a_lp = run_lp_descent(tracker.best, p2_deadline, max_iters=200)
        tracker.update(a_lp)

        # Second LP pass from result
        if time.monotonic() < p2_deadline:
            a_lp2 = run_lp_descent(a_lp, min(time.monotonic() + 100, p2_deadline), max_iters=50)
            tracker.update(a_lp2)

    # Also try LP from a different starting point
    if tracker.best is not None and time.monotonic() < p2_deadline:
        pert = tracker.best + rng.normal(0, 0.05, tracker.best.size)
        pert = np.maximum(pert, 0.01)
        pert = pert / pert.sum()
        a_lp3 = run_lp_descent(pert, min(time.monotonic() + 50, p2_deadline), max_iters=30)
        tracker.update(a_lp3)

    tracker.report("Phase 2")

    # Phase 3: More coordinate descent with annealing
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Coordinate Descent with Annealing", flush=True)
    print(f"{'='*60}", flush=True)

    p3_deadline = min(time.monotonic() + 250, deadline - 80)

    if tracker.best is not None and time.monotonic() < p3_deadline:
        current = tracker.best.copy()

        for restart in range(20):
            if time.monotonic() > p3_deadline:
                break

            # Annealing: perturb less over time
            scale = 0.1 * (0.85 ** restart)
            perturbed = current + rng.normal(0, scale, current.size)
            perturbed = np.maximum(perturbed, 0.0)
            s = perturbed.sum()
            if s < 0.01:
                perturbed = current.copy()
            else:
                perturbed = perturbed / s

            # Run coordinate descent
            a_opt = run_coord_descent(perturbed, min(time.monotonic() + 40, p3_deadline),
                                      seed=seed + 1000 + restart, max_sweeps=50)
            J_val = compute_J_internal(a_opt)
            tracker.update(a_opt)

            if J_val < compute_J_internal(current):
                current = a_opt.copy()
            else:
                # Keep current (annealing: sometimes accept worse)
                if rng.random() < 0.1 * (0.9 ** restart):
                    current = a_opt.copy()

    tracker.report("Phase 3")

    # Phase 4: MWU + Coord Descent hybrid
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: MWU + Coord Descent Hybrid", flush=True)
    print(f"{'='*60}", flush=True)

    p4_deadline = min(time.monotonic() + 150, deadline - 30)

    if tracker.best is not None and time.monotonic() < p4_deadline:
        current = tracker.best.copy()
        eta = 0.1
        for it in range(100):
            if time.monotonic() > p4_deadline:
                break
            b = fast_conv(current)
            current = mwu_step(current, b, eta, top_k=min(30, len(b) // 4))
            eta = eta * 0.98

            # Periodic coordinate descent polish
            if it % 5 == 4:
                current = run_coord_descent(current, min(time.monotonic() + 10, p4_deadline),
                                            seed=seed + 5000 + it, max_sweeps=10)
                tracker.update(current)

    tracker.report("Phase 4")

    # Phase 5: Final polish
    print(f"\n{'='*60}", flush=True)
    print("PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 5:
        a_final = run_coord_descent(tracker.best, deadline - 3, seed=99999, max_sweeps=50)
        tracker.update(a_final)

        # Try LP one more time if enough time
        if time.monotonic() < deadline - 10:
            a_final2 = run_lp_descent(a_final, deadline - 2, max_iters=10)
            tracker.update(a_final2)

    tracker.report("Phase 5")

    # Try extending length if time remains
    if tracker.best is not None and time.monotonic() < deadline - 10:
        for mult in [2]:
            new_n = tracker.best.size * mult
            if new_n > 2048:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext_opt = run_coord_descent(a_ext, deadline - 5, seed=77777, max_sweeps=30)
            tracker.update(a_ext_opt)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J_internal(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
