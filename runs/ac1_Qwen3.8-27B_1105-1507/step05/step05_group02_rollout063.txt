```python
        size <<= 1

    s = a.sum()

    return float(2 * len(a) * b.max() / (s * s))

def conv_diff_update(b, a, i, new_val):

    n = len(a)

    b_new = b.copy()
    # For k where k-i is valid (0 <= k-i <= n-1)
    k_min = i  # k - i >= 0 => k >= i

        if k == 2 * i:
            continue
        j = k - i
        if 0 <= j < n and j != i:
            b_new[k] += 2 * delta * a[j]
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent (log-sum-exp) with Nesterov acceleration
2. LP mirror descent with active-set reduction and precomputed sparse structure
3. Coordinate descent with O(n) convolution updates
4. Multi-length exploration with resampling
5. Final polish combining LP + coordinate descent
Key innovations: precomputed LP matrix structure, active-set LP reduction,
vectorized coordinate descent, and adaptive time allocation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective J = 2n*max_b/(sum_a)^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_structure(n):
    """Precompute row/col indices for the convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp_best_response(a, M, rows, cols, max_constraints=None):
    """
    Solve LP: maximize sum(g) subject to (a*g)[k] <= M for all k, g >= 0.
    If max_constraints is set, only use the most active constraints.
    """
    n = len(a)
    if max_constraints is not None and max_constraints < 2 * n - 1:
        b = fast_conv(a)
        # Get the most active constraints (those closest to M)
        active = np.argsort(b)[-max_constraints:]
        # Filter rows/cols to only these constraints
        active_set = set(active.tolist())
        mask = np.array([k in active_set for k in rows])
        r = rows[mask]
        c = cols[mask]
        b_ub = np.array([M] * len(r))
    else:
        r = rows
        c = cols
        b_ub = np.full(2 * n - 1, M)
    
    vals = a[r - c]
    A = csr_matrix((vals, (r, c)), shape=(len(b_ub), n))
    obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None and res.x.sum() > 1e-15:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, deadline, max_iters=500, active_frac=0.3):
    """
    LP mirror descent: iteratively find best response g via LP,
    then update a = (1-t)*a + t*g.
    """
    a = np.maximum(a.copy(), 1e-10)
    a /= a.sum()
    n = len(a)
    rows, cols = build_A_structure(n)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
        
        # Determine active set size
        n_active = max(10, int(active_frac * (2 * n - 1)))
        
        g = solve_lp_best_response(a, M, rows, cols, max_constraints=n_active)
        if g is None:
            break
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Line search for step size
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < M * (1 - 1e-10):
                a = a_t
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Try smaller active set or break
                active_frac = max(0.1, active_frac * 0.7)
                if active_frac < 0.1:
                    break
    
    return best_a


def conv_update_single(a, b, i, x):
    """
    Efficiently update convolution b when a[i] changes from a[i] to x.
    Returns new b (does not modify a).
    """
    n = len(a)
    old = a[i]
    delta = x - old
    if abs(delta) < 1e-15:
        return b.copy()
    
    b_new = b.copy()
    # k ranges from i to i+n-1 (clamped to [0, 2n-2])
    k_min = i
    k_max = min(2 * n - 2, i + n - 1)
    
    if k_min > k_max:
        return b_new
    
    ks = np.arange(k_min, k_max + 1)
    js = ks - i  # j = k - i, all in [0, n-1]
    
    # General update: b[k] += 2 * delta * a[j] for all k in range
    b_new[k_min:k_max + 1] += 2.0 * delta * a[js]
    
    # Fix the double-count at k = 2i
    if k_min <= 2 * i <= k_max:
        # We added 2*delta*a[i] at k=2i, but the correct contribution is:
        # new^2 - old^2 = (old+delta)^2 - old^2 = 2*old*delta + delta^2
        # So we need to replace 2*delta*a[i] (= 2*delta*old) with 2*old*delta + delta^2
        # Difference: delta^2
        b_new[2 * i] += delta * delta
    
    return b_new

    n = len(a)

    for p in range(n_passes):
        order = rng.permutation(n)
        for i in order:
            old = a[i]
            # Candidate values
            if old < 1e-8:
                candidates = np.array([0.0, 0.5 / n, 1.0 / n, 2.0 / n, 3.0 / n])
            else:
                candidates = np.array([0.0, old * 0.3, old * 0.6, old * 0.85,
                                       old, old * 1.15, old * 1.4, old * 1.8, 2.0 / n])
            
            best_x = old
            best_obj = best_M  # Use current max as threshold
            
            for x in candidates:
                x = float(max(0.0, min(x, 0.5)))  # x can't be too large
                b_new = conv_update_single(a, b, i, x)
                # Account for renormalization
                s_new = s - old + x  # Wait, s should be 1 since we normalized
                # Actually after normalization s=1, so s_new = 1 - old + x
                s_new = 1.0 - old + x
                if s_new < 1e-10:
                    continue
                obj = b_new.max() / (s_new * s_new)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_x = x
            
            if best_x != old:
                a[i] = best_x
                b = conv_update_single(a, b, i, best_x)
                # Renormalize
                s = a.sum()
                if s > 1e-10:
                    a /= s
                    # Recompute b after normalization (since all elements changed)
                    b = fast_conv(a)
                M = b.max()
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
    
    return best_a


def smooth_max_grad(a, tau):
    """Compute gradient of smooth max (log-sum-exp) of convolution."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Softmax weights
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.3, lr_end=0.003):
    """Smooth gradient descent with Nesterov momentum."""
    a = np.maximum(a.astype(np.float64).copy(), 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        momentum = beta * (y - a)
        a_new = np.maximum(y - lr * (grad + momentum), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y /= sy
        a = a_new
    
    return best_a


def resample_seq(src, n):
    """Resample a sequence to a new length using interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.01))
    
    inits.append(np.ones(n))
    for alpha in [0.5, 0.8, 1.2, 1.6, 2.5]:
        inits.append(np.maximum(0.2 + 0.8 * (t**alpha + (1.0-t)**alpha) / (1.0 + (1.0-t)**alpha) / 2, 0.01))
    for f in [1, 2, 3, 4]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t + rng.uniform(0, 2*np.pi)), 0.01))
    
    # Gaussian-like
    inits.append(np.exp(-5 * (t - 0.5)**2) + 0.05)
    # Two peaks
    inits.append(np.exp(-8*(t-0.3)**2) + np.exp(-8*(t-0.7)**2) + 0.03)
    # Smooth random
    k = max(3, n // 32)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    
    for a_init in inits:
        a_init = np.maximum(a_init, 0.0)
        s = a_init.sum()
        if s > 0:
            a_init /= s
    return inits
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent (log-sum-exp) with Nesterov acceleration
2. LP mirror descent with adaptive active-set reduction
3. Coordinate descent with O(n) convolution updates
4. Multi-length exploration with resampling and random restarts
5. Final combined LP + coordinate descent polish
Key innovations: precomputed LP structure, threshold-based active set,
vectorized coordinate descent with efficient conv updates, adaptive time.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective J = 2n*max_b/(sum_a)^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_structure(n):
    """Precompute row/col indices for the convolution constraint matrix."""
    rows_list = np.empty(n * n, dtype=np.int32)
    cols_list = np.empty(n * n, dtype=np.int32)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        count = j_max - j_min + 1
        rows_list[idx:idx + count] = k
        cols_list[idx:idx + count] = np.arange(j_min, j_max + 1)
        idx += count
    return rows_list[:idx], cols_list[:idx]


def solve_lp_response(a, M, rows, cols, threshold=None):
    """
    LP: maximize sum(g) s.t. (a*g)[k] <= M for active k, g >= 0.
    threshold: only include constraints where b[k] >= threshold.
    """
    n = len(a)
    if threshold is not None:
        b = fast_conv(a)
        mask = b[rows] >= threshold
        r = rows[mask]
        c = cols[mask]
    else:
        r = rows
        c = cols
    nrows = len(r)
    if nrows == 0:
        return np.ones(n) / n

    vals = a[r - c]
    A = csr_matrix((vals, (r, c)), shape=(nrows, n))
    obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    b_ub = np.full(nrows, M)

    try:
        res = linprog(obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 8})
        if res.success and res.x is not None and res.x.sum() > 1e-15:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=500, thresh_frac=0.85):
    """LP mirror descent with adaptive active-set threshold."""
    a = np.maximum(a.astype(np.float64).copy(), 1e-10)
    a /= a.sum()
    n = len(a)
    rows, cols = build_A_structure(n)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()

        threshold = M * thresh_frac
        g = solve_lp_response(a, M, rows, cols, threshold=threshold)
        if g is None:
            break

        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        for t in [0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < M * (1.0 - 1e-10):
                a = a_t
                improved = True
                break

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 25:
                thresh_frac = max(0.7, thresh_frac * 0.92)
                if thresh_frac < 0.7:
                    break

    return best_a


def conv_update_single(a, b, i, x):
    """Update convolution b when a[i] changes to x. O(n) operation."""
    n = len(a)
    old = a[i]
    delta = x - old
    if abs(delta) < 1e-15:
        return b.copy()

    b_new = b.copy()
    k_min = i
    k_max = min(2 * n - 2, i + n - 1)
    if k_min > k_max:
        return b_new

    js = np.arange(k_min, k_max + 1) - i
    b_new[k_min:k_max + 1] += 2.0 * delta * a[js]

    if k_min <= 2 * i <= k_max:
        b_new[2 * i] += delta * delta

    return b_new


def coordinate_descent(a, n_passes=10, seed=42):
    """Coordinate descent with O(n) conv updates and 1D line search."""
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    n = len(a)
    b = fast_conv(a)
    best_M = b.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)

    for p in range(n_passes):
        order = rng.permutation(n)
        for i in order:
            old = a[i]
            if old < 1e-8:
                candidates = np.array([0.0, 0.3/n, 0.7/n, 1.2/n, 2.0/n, 3.0/n])
            else:
                candidates = np.array([0.0, old*0.4, old*0.7, old*0.9,
                                       old, old*1.1, old*1.3, old*1.6, 2.0/n, 3.0/n])

            best_x = old
            best_obj = best_M

            for x in candidates:
                x = float(max(0.0, min(x, 0.4)))
                b_new = conv_update_single(a, b, i, x)
                s_new = 1.0 - old + x
                if s_new < 1e-8:
                    continue
                obj = b_new.max() / (s_new * s_new)
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_x = x

            if best_x != old:
                a[i] = best_x
                b = conv_update_single(a, b, i, best_x)
                s = a.sum()
                if s > 1e-10:
                    a /= s
                    b = fast_conv(a)
                M = b.max()
                if M < best_M:
                    best_M = M
                    best_a = a.copy()

    return best_a


def smooth_max_grad(a, tau):
    """Gradient of log-sum-exp smoothing of max convolution."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.3, lr_end=0.003):
    """Smooth GD with Nesterov momentum."""
    a = np.maximum(a.astype(np.float64).copy(), 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1.0 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y /= sy
        a = a_new

    return best_a


def resample_seq(src, n):
    """Resample sequence to new length via interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.1f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization:
    Phase 1: Smooth GD warm-up (40s)
    Phase 2: LP Mirror Descent (700s)
    Phase 3: Coordinate Descent polish (100s)
    Phase 4: Multi-length + restarts (250s)
    Phase 5: Final refinement (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0
    best = BestTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.8f}", flush=True)
    except (NameError, AttributeError, Exception) as e:
        print(f"No parent: {e}", flush=True)

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    n_base = parent.size

    # Lengths to explore
    lengths = set()
    for mult in [0.75, 0.875, 1.0, 1.125, 1.25, 1.5]:
        n_try = int(n_base * mult)
        n_try = max(128, min(1024, (n_try // 16) * 16))
        lengths.add(n_try)
    for n_try in [384, 448, 512, 576, 640, 768, 896]:
        if 128 <= n_try <= 1024:
            lengths.add(n_try)
    lengths = sorted(lengths)
    n_focus = n_base if n_base in lengths else 512
    print(f"Lengths: {lengths}, focus={n_focus}", flush=True)

    # === PHASE 1: Smooth GD (40s) ===
    p1_end = min(time.monotonic() + 40.0, deadline - 250.0)
    print(f"\n{'='*50}\nPHASE 1: Smooth GD (budget {p1_end - time.monotonic():.0f}s)\n{'='*50}", flush=True)

    a = run_smooth_nesterov(resample_seq(parent, n_focus), p1_end,
                            max_iters=1500, tau_start=1.0, tau_end=120.0,
                            lr_init=0.3, lr_end=0.01)
    best.update(a)

    for n in lengths:
        if n == n_focus or time.monotonic() > p1_end:
            continue
        a0 = resample_seq(parent, n)
        a = run_smooth_nesterov(a0, min(time.monotonic() + 12, p1_end),
                                max_iters=400, tau_start=2.0, tau_end=80.0,
                                lr_init=0.25, lr_end=0.02)
        best.update(a)
    best.report("Phase 1")

    # === PHASE 2: LP Mirror Descent (700s) ===
    p2_end = min(time.monotonic() + 700.0, deadline - 250.0)
    if time.monotonic() > deadline - 100:
        p2_end = deadline - 30
    est_time = p2_end - time.monotonic()
    print(f"\n{'='*50}\nPHASE 2: LP Mirror Descent (budget {est_time:.0f}s)\n{'='*50}", flush=True)

    if best.seq is not None:
        a = run_lp_mirror(best.seq, p2_end, max_iters=1000, thresh_frac=0.85)
        best.update(a)

        # Second pass with tighter threshold
        if time.monotonic() < p2_end - 20:
            a2 = run_lp_mirror(a, p2_end, max_iters=500, thresh_frac=0.75)
            best.update(a2)
    best.report("Phase 2")

    # === PHASE 3: Coordinate Descent (100s) ===
    p3_end = min(time.monotonic() + 100.0, deadline - 100.0)
    est_time = p3_end - time.monotonic()
    print(f"\n{'='*50}\nPHASE 3: Coordinate Descent (budget {est_time:.0f}s)\n{'='*50}", flush=True)

    if best.seq is not None:
        n_cd = best.seq.size
        est_per_pass = n_cd * 10 * 0.0008
        n_passes = max(3, min(40, int(est_time / max(0.1, est_per_pass))))
        print(f"  n={n_cd}, passes={n_passes}", flush=True)
        a = coordinate_descent(best.seq, n_passes=n_passes, seed=seed)
        best.update(a)
    best.report("Phase 3")

    # === PHASE 4: Multi-length + Restarts (250s) ===
    p4_end = min(time.monotonic() + 250.0, deadline - 80.0)
    est_time = p4_end - time.monotonic()
    print(f"\n{'='*50}\nPHASE 4: Multi-length/Restarts (budget {est_time:.0f}s)\n{'='*50}", flush=True)

    if best.seq is not None:
        n_best = best.seq.size
        # Try other lengths
        for n in lengths:
            if time.monotonic() > p4_end - 20:
                break
            if abs(n - n_best) < 32:
                continue
            allot = min(45.0, (p4_end - time.monotonic()) * 0.4)
            a0 = resample_seq(best.seq, n)
            a = run_smooth_nesterov(a0, min(time.monotonic() + allot * 0.3, p4_end - 15),
                                    max_iters=400, tau_start=2, tau_end=80,
                                    lr_init=0.2, lr_end=0.02)
            a = run_lp_mirror(a, min(time.monotonic() + allot * 0.6, p4_end - 10),
                              max_iters=40, thresh_frac=0.85)
            best.update(a)

        # Random restarts from best
        if time.monotonic() < p4_end - 30:
            for restart in range(3):
                if time.monotonic() > p4_end - 20:
                    break
                noise_scale = [0.02, 0.04, 0.08][restart]
                a0 = best.seq + rng.normal(0, noise_scale * 0.01, n_best)
                a0 = np.maximum(a0, 1e-4)
                a0 /= a0.sum()
                a = run_smooth_nesterov(a0, min(time.monotonic() + 30, p4_end - 20),
                                        max_iters=500, tau_start=3, tau_end=100,
                                        lr_init=0.2, lr_end=0.01)
                a = run_lp_mirror(a, min(time.monotonic() + 30, p4_end - 15),
                                  max_iters=50, thresh_frac=0.85)
                best.update(a)
    best.report("Phase 4")

    # === PHASE 5: Final Refinement ===
    p5_end = deadline
    est_time = p5_end - time.monotonic()
    print(f"\n{'='*50}\nPHASE 5: Final Refinement (budget {est_time:.0f}s)\n{'='*50}", flush=True)

    if best.seq is not None and est_time > 10:
        # LP with tight threshold
        t_lp = est_time * 0.5
        a = run_lp_mirror(best.seq, time.monotonic() + t_lp,
                          max_iters=300, thresh_frac=0.7)
        best.update(a)

        # Coordinate descent
        if time.monotonic() < p5_end - 5:
            a = coordinate_descent(a, n_passes=8, seed=seed + 100)
            best.update(a)

        # Final LP
        if time.monotonic() < p5_end - 3:
            a = run_lp_mirror(a, p5_end - 1, max_iters=100, thresh_frac=0.65)
            best.update(a)

        # Final CD
        if time.monotonic() < p5_end - 2:
            a = coordinate_descent(a, n_passes=5, seed=seed + 200)
            best.update(a)

    best.report("FINAL")

    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"RESULT: J={best.score:.8f} n={best.seq.size} elapsed={time.monotonic()-best.t0:.1f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
```
