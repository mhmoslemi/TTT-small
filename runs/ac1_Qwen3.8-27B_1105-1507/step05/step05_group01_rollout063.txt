import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def simplex_project(v, target_sum=1.0):
    """Project v onto {a >= 0, sum(a) = target_sum} in L2 sense."""
    # Standard simplex projection
    v = v.copy()
    n = len(v)
    # Find threshold tau such that sum(max(v_i - tau, 0)) = target_sum
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    # Find the support
    cssv = np.cumsum(u)
    cond = u > (cssv - target_sum) / np.arange(1, n+1)
    if not np.any(cond):
        # All would be negative, return uniform
        return np.full(n, target_sum / n)
    rho = np.max(np.where(cond)[0]) + 1
    tau = (cssv[rho-1] - target_sum) / rho
    return np.maximum(v - tau, 0)

def gradient_of_peak(a, k):
    """Compute gradient of (a*a)_k w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    # (a*a)_k = sum_i a_i * a_{k-i} for valid i
    # d/d(a_j) = 2 * a_{k-j} if 0 <= k-j < n
    valid_j = np.arange(max(0, k-n+1), min(n, k+1))
    grad[valid_j] = 2.0 * a[k - valid_j]
    return grad

def subgradient_step(a, alpha, top_k=5):
    """Take a subgradient step targeting top-k peaks."""
    n = len(a)
    conv = compute_conv(a)
    # Get top k peak positions
    top_idx = np.argsort(conv)[::-1][:top_k]
    # Weight by how prominent each peak is
    peak_vals = conv[top_idx]
    max_val = peak_vals[0]
    weights = np.maximum(peak_vals / max_val - 0.5, 0)
    if np.sum(weights) < 1e-10:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights = weights / np.sum(weights)
    
    # Compute combined gradient
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        grad += w * gradient_of_peak(a, int(k))
    
    # Take step and project
    a_new = a - alpha * grad
    a_new = simplex_project(a_new)
    return a_new, conv

def lp_direction_constrained(a, M, active_idx, relax=1.0):
    """Solve LP with only active constraints. Returns g or None."""
    n = len(a)
    m = len(active_idx)
    if m == 0:
        return None
    
    # Build constraint matrix for active constraints only
    # (a*g)_{k} = sum_i a_i * g_{k-i} for valid i
    # For constraint k: sum over valid i of a_i * g_{k-i} <= M
    # Let's build the matrix row by row
    A = np.zeros((m, n))
    for row, k in enumerate(active_idx):
        # (a*g)_k = sum_i a_i * g_{k-i}
        # For each j (index of g), the coefficient is a_{k-j} if 0 <= k-j < n
        valid_j = np.arange(max(0, k-n+1), min(n, k+1))
        A[row, valid_j] = a[k - valid_j]
    
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    
    try:

"""
Combined spectral-flattening, subgradient, and LP optimization for minimizing
the peak-to-sum ratio of autocorrelation of non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def simplex_project(v, target_sum=1.0):
    """Project v onto {a >= 0, sum(a) = target_sum} (Euclidean projection)."""
    n = len(v)
    v = v.astype(np.float64).copy()
    
    # Binary search for threshold
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    tau = (lo + hi) / 2.0
    result = np.maximum(v - tau, 0.0)
    
    # Fix rounding
    s = np.sum(result)
    if s > 1e-15:
        result *= target_sum / s
    else:
        result = np.full(n, target_sum / n)
    
    return result


def gradient_peak(a, k):
    """Gradient of (a*a)_k w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        valid_j = np.arange(lo, hi)
        grad[valid_j] = 2.0 * a[k - valid_j]
    return grad


def subgradient_step(a, alpha, top_k=5, momentum=0.9):
    """Take a momentum subgradient step targeting top-k peaks."""
    n = len(a)
    conv = compute_conv(a)
    top_idx = np.argsort(conv)[::-1][:top_k]
    
    # Weight peaks by prominence
    peak_vals = conv[top_idx].astype(float)
    max_val = peak_vals[0]
    if max_val < 1e-15:
        return a, conv
    weights = np.maximum(peak_vals - 0.8 * max_val, 0)
    if np.sum(weights) < 1e-15:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights = weights / np.sum(weights)
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        grad += w * gradient_peak(a, int(k))
    
    return grad, conv


def spectral_flatten(a, alpha=0.3, n_iter=1):
    """Apply spectral flattening to reduce autocorrelation peaks."""
    n = len(a)
    # Find FFT size
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    a_padded = np.zeros(N)
    a_padded[:n] = a
    
    A = fft(a_padded)
    mag = np.abs(A)
    phase = np.angle(A)
    
    for _ in range(n_iter):
        # Keep DC component fixed (preserves sum)
        # Flatten the rest
        target_mag = np.sqrt(np.mean(mag[1:]**2)) if N > 1 else mag[0]
        
        # Geometric mean blending
        mag_new = mag.copy()
        mag_new[1:] = mag[1:] * (target_mag / np.maximum(mag[1:], 1e-15)) ** alpha
        mag_new[1:] = np.sqrt(mag[1:]**2 * (target_mag / np.maximum(mag[1:], 1e-15)) ** (2*alpha))
        # Simpler: mag_new[1:] = mag[1:] ** (1-alpha) * target_mag ** alpha
        mag_new[1:] = np.maximum(mag[1:], 1e-15) ** (1 - alpha) * max(target_mag, 1e-15) ** alpha
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a = a_new / s
        else:
            break
    
    return a


def lp_direction(a, M, relax=1.0, max_constraints=50, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M*relax for top constraints, g >= 0."""
    n = len(a)
    conv = compute_conv(a)
    # Get top constraints
    k_active = np.argsort(conv)[::-1][:max_constraints]
    
    # Build constraint matrix
    m = len(k_active)
    A_mat = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(k_active):
        lo = max(0, int(k) - n + 1)
        hi = min(n, int(k) + 1)
        if lo < hi:
            A_mat[row, lo:hi] = a[int(k) - np.arange(lo, hi)]
    
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g, s
    except Exception:
        pass
    return None, 0.0

"""
Combined spectral-flattening, subgradient, and LP-optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses FFT-based convolution, active-set subgradient methods, spectral shaping,
constrained LP direction finding, and multi-length exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft, rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def simplex_project(v, target_sum=1.0):
    """Euclidean projection onto {a >= 0, sum(a) = target_sum}."""
    n = len(v)
    v = v.astype(np.float64)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    if u[0] <= 0:
        return np.full(n, target_sum / n)
    cssv = np.cumsum(u)
    cond = u > (cssv - target_sum) / np.arange(1, n + 1)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = int(np.max(np.where(cond)[0])) + 1
    tau = (cssv[rho - 1] - target_sum) / rho
    result = np.maximum(v - tau, 0.0)
    s = np.sum(result)
    if s > 1e-15:
        result *= target_sum / s
    return result


def gradient_peak(a, k):
    """Exact gradient of (a*a)_k with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        valid_j = np.arange(lo, hi)
        grad[valid_j] = 2.0 * a[k - valid_j]
    return grad


def active_set_gradient(a, conv, tol=0.01):
    """Compute gradient using all near-active peaks (within tol of max)."""
    n = len(a)
    max_val = float(np.max(conv))
    if max_val < 1e-15:
        return np.zeros(n)
    # Find all peaks within (1-tol) of max
    active = np.where(conv >= (1.0 - tol) * max_val)[0]
    n_active = len(active)
    if n_active == 0:
        active = np.array([int(np.argmax(conv))])
        n_active = 1
    # Weight by prominence
    weights = conv[active] / (max_val * n_active)
    grad = np.zeros(n)
    for w, k in zip(weights, active):
        grad += w * gradient_peak(a, int(k))
    return grad


def spectral_flatten(a, alpha=0.3):
    """Apply one step of spectral magnitude flattening."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    mag = np.abs(A).copy()
    phase = np.angle(A)
    # Keep DC fixed, flatten others toward geometric mean
    if N > 1:
        target = np.sqrt(np.mean(mag[1:] ** 2))
        target = max(target, 1e-10)
        mag[1:] = np.maximum(mag[1:], 1e-15) ** (1.0 - alpha) * target ** alpha
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new /= s
    else:
        a_new = np.ones(n) / n
    return a_new


def spectral_flatten_aggressive(a, alpha=0.5, n_refine=3):
    """More aggressive spectral flattening with iterative refinement."""
    for _ in range(n_refine):
        a = spectral_flatten(a, alpha=alpha)
    return a


def lp_direction_gen(a, M, relax=1.0, initial_constraints=15, max_rounds=8,
                     time_limit_single=3.0):
    """LP direction finding with constraint generation (cutting plane)."""
    n = len(a)
    conv = compute_conv(a)
    # Start with top constraints
    active = np.argsort(conv)[::-1][:initial_constraints].tolist()
    
    for _round in range(max_rounds):
        m = len(active)
        A_mat = np.zeros((m, n), dtype=np.float64)
        for row, k in enumerate(active):
            lo = max(0, int(k) - n + 1)
            hi = min(n, int(k) + 1)
            if lo < hi:
                A_mat[row, lo:hi] = a[int(k) - np.arange(lo, hi)]
        
        c = -np.ones(n)
        b_ub = np.full(m, M * relax)
        bounds = [(0.0, None)] * n
        g = None
        try:
            res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True,
                                                   'time_limit': time_limit_single})
            if res.status == 0:
                g = np.maximum(res.x, 0.0)
        except Exception:
            break
        
        if g is None:
            break
        
        S = float(np.sum(g))
        if S <= 1e-10:
            break
        
        # Check for violated constraints
        conv_g = compute_conv(a * g)
        # Actually we need (a*g)_k, not (a*a*_g)_k
        # (a*g)_k = sum_i a_i * g_{k-i}
        # Let's compute this properly
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = rfft(a, size)
        fg = rfft(g, size)
        ag = irfft(fa * fg, size)[:2 * n - 1]
        
        violations = ag - M * relax
        if np.max(violations) < 1e-8:
            # No violations, we have the solution
            return g, S
        
        # Add most violated constraint
        worst = int(np.argmax(violations))
        if worst not in active:
            active.append(worst)
        else:
            break
    
    if g is not None:
        return g, S
    return None, 0.0


def lp_polish(a, time_budget=100.0, max_constraints=25):
    """LP-based alternating optimization with constraint generation."""
    n = len(a)
    a = simplex_project(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, S = lp_direction_gen(a, M, relax=1.0,
                                initial_constraints=max_constraints,
                                max_rounds=5,
                                time_limit_single=min(3.0, remaining * 0.3))
        
        if g is None or S <= 1.0 + 1e-6:
            # Try with relaxation
            g, S = lp_direction_gen(a, M, relax=1.1,
                                    initial_constraints=max_constraints,
                                    max_rounds=3,
                                    time_limit_single=min(3.0, remaining * 0.3))
        
        if g is None or S <= 1.0 + 1e-6:
            stall += 1
            if stall > 8:
                break
            continue
        
        improved = False
        for t in [0.6, 0.8, 1.0, 0.4, 0.9, 0.5]:
            a_try = (1.0 - t) * a + t * g / S
            a_try = simplex_project(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-13:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            stall += 1
            if stall > 12:
                break
    
    return best_a, best_obj

def resample_seq(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def random_init(n, seed):
    """Generate a diverse random initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)
    
    if choice == 0:
        f1 = rng.uniform(0.3, 3.0)
        f2 = rng.uniform(3.0, 10.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 6.28))
    elif choice == 1:
        # Incommensurate frequencies (golden ratio)
        phi = (1 + np.sqrt(5)) / 2
        f1 = 1.0 / phi
        f2 = 1.0 / (phi * phi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f1 * k) + 0.3 * np.cos(2 * np.pi * f2 * k + 1.0)
    elif choice == 2:
        # Block structure
        nblocks = rng.integers(3, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.4, 1.6)
    elif choice == 3:
        # Smooth random
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        # Multi-sinusoid with random phases
        a = np.ones(n)
        for _ in range(3):
            f = rng.uniform(0.5, 8.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 6.28))
    else:
        a = rng.uniform(0.5, 1.5, n)
    
    a = np.maximum(a, 0.01)
    a = simplex_project(a)
    return a

                a_try2, obj_try2 = subgradient_optimize(a_try, time_budget=quick_budget,
                                                        top_k=5, use_spectrum=True)
                if obj_try2 < best_score:
                    best_score = obj_try2
                    best_seq = a_try2.copy()
                    print(f"  ** n={n_try} improved: {best_score:.6f} **")

    # Final LP polish on the best found
    print("\n=== Final Polish ===")
    if time.monotonic() < deadline - 10:
        final_budget = min(50.0, deadline - time.monotonic() - 5)
        if final_budget > 5:
            a_final, obj_final = lp_polish(best_seq, time_budget=final_budget,
                                           max_constraints=30)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final LP: {best_score:.6f}")

    # Validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 700
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    return result
