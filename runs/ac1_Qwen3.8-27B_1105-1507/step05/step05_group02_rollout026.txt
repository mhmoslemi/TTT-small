```python
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent with warm starts and adaptive step size
2. Coordinate descent refinement

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix_sparse(n, a):
    """Build the (2n-1) x n convolution matrix A where (A*g)_k = sum_j A[k,j]*g[j] = conv(a,g)_k"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_best_response(n, a, M):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0"""
    A = build_conv_matrix_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception as e:
        print(f"  LP error: {e}")
    return None


def run_lp_mirror(a, deadline, max_iters=500, verbose=True):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    prev_g = None
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        g = solve_lp_best_response(n, a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Adaptive step size: try multiple values, pick best
        improved = False
        for t in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.02, 0.01]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-12:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break
        
        if verbose and (it % 50 == 0 or improved and it < 20):
            print(f"  LP iter {it}: M={M:.8f} best_M={best_M:.8f}", flush=True)
    
    return best_a


def coordinate_descent(a, deadline, max_sweeps=20):
    """Refine using coordinate descent. For each coordinate, try to optimize it."""
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

            break
        improved = False
        # Randomize order
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            # Try several values for a[j]
            old_val = a[j]
            # Scale: try multiplying by factors
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.0, 0.1]:
                new_val = old_val * factor if factor > 0 else 0.0
                a[j] = new_val
                s = a.sum()
                if s < 1e-10:

                a_norm = a / s
                M_new = fast_conv(a_norm).max()
                if M_new < best_M - 1e-12:

                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    improved = True
                    break
            else:
                a[j] = old_val  # revert if no improvement found
        
        if not improved:
            break
    
    return best_a
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage approach:
1. Diverse initialization with parametric forms and parent resampling
2. Smooth Nesterov gradient descent for warm starts

4. LP mirror descent with adaptive step size (primary refinement)

6. Multi-length exploration (256-1024)

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_matrix(n, a):
    """Build (2n-1) x n matrix where (A*g)_k = conv(a,g)_k"""
    rows = np.concatenate([np.arange(max(0, k-n+1), min(n-1, k)+1) for k in range(2*n-1)])
    cols = np.concatenate([[k] * (min(n-1, k) - max(0, k-n+1) + 1) for k in range(2*n-1)])
    vals = np.array([a[k - j] for k in range(2*n-1) for j in range(max(0, k-n+1), min(n-1, k)+1)])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def lp_solve(n, a, M, time_limit=10):
    """Solve max sum(g) s.t. A*g <= M, g >= 0"""
    A = build_A_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def smooth_grad(a, tau):
    """Compute gradient of soft-max approximation of max(conv(a,a))"""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += 2.0 * p[k] * a[k-hi:k-lo+1][::-1]
    return grad, p, b


def run_nesterov(a, deadline, max_iters=3000, tau_start=2.0, tau_end=300.0,
                 lr_init=0.3, lr_end=0.003):
    """Smoothesur gradient descent with Nesterov acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b = smooth_grad(y, tau)
        b_max = b.max()
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 0.25 * (it + 1) ** 2)) / (it + 1)
        a_new = np.maximum(y - lr * (grad + 0.9 * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    
    return best_a


def run_lp_mirror(a, deadline, max_iters=1000, verbose=False):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        t0 = time.monotonic()
        g = lp_solve(n, a, M)
        lp_time = time.monotonic() - t0
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        improved = False
        for t in [0.5, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = float(fast_conv(a_t).max())
            if M_t < best_M - 1e-13:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 50:
                break
        
        if verbose and (it % 100 == 0):
            print(f"  LP iter {it}: M={M:.8f} best={best_M:.8f} lp_time={lp_time*1000:.0f}ms", flush=True)
    
    return best_a


def run_coord_descent(a, deadline, max_sweeps=30, n_trials=8):
    """Coordinate descent with incremental updates."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    rng = np.random.default_rng(123)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            old = a[j]
            if old < 1e-15:
                # Try setting to small positive values
                trials = [0.001, 0.005, 0.01, 0.02, 0.05]
            else:
                trials = [old * f for f in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]] + [0.0]
            
            found = False
            for nv in trials:
                a[j] = max(0.0, nv)
                s = a.sum()
                if s < 1e-10:
                    continue
                an = a / s
                Mn = float(fast_conv(an).max())
                if Mn < best_M - 1e-13:
                    best_M = Mn
                    best_a = an.copy()
                    a = an.copy()
                    found = True
                    improved = True
                    break
            if not found:
                a[j] = old
            else:
                # After finding improvement, a has been updated and normalized
                # old value is stale, but that's ok for coordinate descent
        
        if not improved:
            break
    
    return best_a


def resample(src, n):
    src = np.asarray(src, dtype=np.float64)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
```

```python
        if best_t is not None:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a /= a.sum()
            best_M = best_new_M
            best_a = a.copy()
            plates = 0
        else:
            plates += 1
            if plates > max_plateau:
                # Restart with perturbation
                pert = rng.normal(0, 0.01, n)
                a = np.maximum(a + pert, 1e-6)
                a /= a.sum()
                a = run_nesterov(a, min(time.monotonic() + 10, deadline), max_iters=500, tau_start=5, tau_end=50, lr_init=0.2, lr_end=0.05)
                M = float(fast_conv(a).max())
                if M < best_M:
                    best_M = M
                    best_a = a.copy()
                plates = 0
                max_plateau = max(20, max_plateau // 2)
        
        if verbose and it % 200 == 0:
            print(f"    LP iter {it}: M={M:.8f} best={best_M:.8f} plateau={plateaus}", flush=True)
    
    return best_a
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length exploration with diverse parametric initializations
2. Smooth Nesterov gradient descent for warm starts
3. LP mirror descent with adaptive step size and restarts (primary method)
4. Coordinate descent with full re-evaluation for final refinement
Key differences from previous: more aggressive time allocation to LP mirror
descent, restart mechanism on plateaus, better step-size search, and
coordinated length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# Cache for convex matrix structure
_A_structure_cache = {}

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_A_structure(n):
    """Get or build the (rows, cols) structure for the convolution matrix."""
    if n in _A_structure_cache:
        return _A_structure_cache[n]
    rows_parts = []
    cols_parts = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        count = j_max - j_min + 1
        rows_parts.append(np.full(count, k))
        cols_parts.append(np.arange(j_min, j_max + 1))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    _A_structure_cache[n] = (rows, cols)
    return rows, cols


def build_A_matrix(n, a):
    """Build (2n-1) x n sparse matrix where (A*g)_k = conv(a,g)_k."""
    rows, cols = get_A_structure(n)
    vals = np.array([a[k - j] for k, j in zip(rows, cols)], dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(n, a, M, time_limit=8):
    """Solve max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    A = build_A_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def smooth_grad(a, tau):
    """Gradient of soft-max approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, b, b_max


def run_nesterov(a, deadline, max_iters=3000, tau_start=2.0, tau_end=300.0,
                 lr_init=0.3, lr_end=0.003):
    """Smooth gradient descent with Nesterov acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    y = a.copy()

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b, b_max = smooth_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 0.25 * (it + 1) ** 2)) / (it + 1)
        a_new = np.maximum(y - lr * (grad + 0.9 * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def run_lp_mirror(a, deadline, max_iters=5000, verbose=False):
    """LP mirror descent with adaptive step size and restart on plateau."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    rng = np.random.default_rng(777)
    no_imp = 0
    max_no_imp = 60

    step_candidates = [0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01, 0.005]

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        g = lp_solve(n, a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        for t in step_candidates:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = float(fast_conv(a_t).max())
            if M_t < best_M - 1e-13:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > max_no_imp:
                # Restart with perturbation + short smooth GD
                pert_scale = 0.005 + 0.01 * (no_imp / max_no_imp)
                pert = rng.normal(0, pert_scale, n)
                a_pert = np.maximum(a + pert, 1e-8)
                a_pert /= a_pert.sum()
                a_pert = run_nesterov(a_pert, min(time.monotonic() + 15, deadline),
                                      max_iters=800, tau_start=5.0, tau_end=100.0,
                                      lr_init=0.2, lr_end=0.02)
                M_pert = float(fast_conv(a_pert).max())
                if M_pert < best_M:
                    best_M = M_pert
                    best_a = a_pert.copy()
                a = a_pert.copy()
                no_imp = 0
                max_no_imp = max(20, max_no_imp - 10)

        if verbose and it % 200 == 0:
            print(f"    LP iter {it}: M={M:.8f} best={best_M:.8f} no_imp={no_imp}", flush=True)

    return best_a


def run_coord_descent(a, deadline, max_sweeps=40):
    """Coordinate descent: try scaling each coordinate."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    rng = np.random.default_rng(42)

    factors = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0, 0.0]

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            old = a[j]
            found = False
            for f in factors:
                nv = old * f if f > 0 else 0.0
                a[j] = nv
                s = a.sum()
                if s < 1e-10:
                    continue
                an = a / s
                Mn = float(fast_conv(an).max())
                if Mn < best_M - 1e-13:
                    best_M = Mn
                    best_a = an.copy()
                    a = an.copy()
                    found = True
                    improved = True
                    break
            if not found:
                a[j] = old
        if not improved:
            break

    return best_a


def resample(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample(parent, n)
        inits.append(np.maximum(p, 1e-6))
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 1e-6))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 1e-6))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 1e-6))
        inits.append(np.maximum(p * (1 + 0.2 * rng.normal(0, 1, n)), 1e-6))

    # Parametric forms
    inits.append(np.ones(n).copy())
    for alpha in [0.5, 0.7, 1.0, 1.4, 2.0, 3.0]:
        inits.append(np.maximum(0.2 + 0.8 * (t ** alpha + (1 - t) ** alpha) / 2, 0.01))
    for f in [1, 2, 3, 4, 6]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.05)
    inits.append(np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2) + 0.03)
    for p_exp in [1.5, 2.0, 2.5, 3.0]:
        inits.append(np.abs(np.cos(np.pi * t)) ** p_exp + 0.03)
    # Random smooth
    k = max(3, n // 32)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    # Symmetric
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    # Periodic-like
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * 3 * t) ** 2 + 0.05)

    return inits
```
