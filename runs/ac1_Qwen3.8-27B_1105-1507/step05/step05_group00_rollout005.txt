    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def normalize(a):
    """Normalize sequence to sum 1, clip negatives."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones_like(a) / len(a)
    return a / s


def active_set_gradient_descent(a, max_iters=2000, time_budget=60.0,
                                 top_k_frac=0.1, momentum=0.9,
                                 step_init=0.05, verbose=False):
    """
    Accelerated gradient descent on the mean of top-K convolution peaks.
    Uses FFT for gradient computation and Nesterov momentum.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    step = step_init
    x = a.copy()  # Nesterov lookahead

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 1:
            break

        # Evaluate at lookahead point for gradient
        conv_x = compute_conv(x)
        cmax = float(np.max(conv_x))

        # Select top-K fraction of peaks
        k_count = max(1, int(len(conv_x) * top_k_frac))
        # Use threshold: peaks within 5% of max
        threshold = cmax * 0.95
        S = np.where(conv_x >= threshold)[0]
        if len(S) == 0:
            S = np.array([int(np.argmax(conv_x))])

        # Build weight vector for active set (uniform over S)
        w = np.zeros(2 * n - 1)
        w[S] = 1.0 / len(S)

        # Gradient: grad_j = 2 * sum_k w[k] * a[k-j]
        # This is convolution of a with reversed w, taking appropriate slice
        w_rev = w[::-1]
        conv_grad = fft_convolve(a, w_rev)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1][::-1]

        # Project gradient onto sum=0 hyperplane
        grad -= np.mean(grad)

        # Nesterov momentum update
        v = momentum * v - step * grad
        a_new = a - v

        # Line search (backtracking)
        alpha = 1.0
        obj_new = np.inf
        while alpha > 1e-4:
            a_try = a + alpha * (a_new - a)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < compute_obj(a) * (1 - 1e-5 * alpha):
                obj_new = obj_try
                break
            alpha *= 0.5

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_try.copy()
            a = a_try.copy()
            # Update Nesterov lookahead
            x = a + momentum * (a - (a + v))  # x = a - momentum * v
        else:
            # No improvement, shrink step
            step *= 0.9
            a = a  # stay
            x = a.copy()

        # Decay step size slowly
        step = max(step * 0.9995, 0.001)

        if verbose and it % 100 == 0:
            print(f"    ASGD it{it}: obj={best_obj:.8f}, step={step:.5f}, |S|={len(S)}")

    return best_a


def solve_lp_max_sum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M for all k, g >= 0."""
    n = len(a)
    # Build Toeplitz matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_alternate_step(a, time_limit=15, line_search_points=8):
    """
    One step of LP alternation: solve LP given a, then blend towards solution.
    Returns improved a or None if no improvement.
    """
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    obj_cur = compute_obj(a)

    g = solve_lp_max_sum(a, M, time_limit=time_limit)
    if g is None:
        return None, obj_cur

    best_a = a.copy()
    best_obj = obj_cur

    for t in np.linspace(0.1, 1.0, line_search_points):
        a_try = (1 - t) * a + t * g
        a_try = normalize(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a, best_obj


def give_and_take(a, max_rounds=50, time_budget=10.0, verbose=False):
    """
    Targeted give-and-take: identify peak-contributing pairs and redistribute mass.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        peak_val = float(conv[k_star])

        # Find contributing pairs (i, k_star - i)
        best_delta_obj = 0
        best_move = None

        for i in range(n):
            j = k_star - i
            if j < 0 or j >= n or j == i:
                continue
            if a[i] < 1e-6 or a[j] < 1e-6:
                continue

            # Try moving small mass from larger to smaller
            if a[i] > a[j]:
                src, dst = i, j
            else:
                src, dst = j, i

            for frac in [0.01, 0.02, 0.05]:
                delta = frac * a[src]
                a[src] -= delta
                a[dst] += delta
                new_conv = compute_conv(a)
                new_obj = 2.0 * n * float(np.max(new_conv)) / (a.sum() ** 2)
                if new_obj < best_obj - 1e-12:
                    best_delta_obj = new_obj
                    best_move = (src, dst, delta)
                    best_a = a.copy()
                    best_obj = new_obj
                else:
                    a[src] += delta
                    a[dst] -= delta
                    break

        if best_move is None:
            break

    return best_a


def multiplicative_weights_step(a, M, iterations=50, eta=0.1):
    """
    Multiplicative weights update for the LP (faster than full LP solve).
    Approximates the LP solution using exponential weights.
    """
    n = len(a)

    # Build Toeplitz
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]

    g = np.ones(n) / n

    for it in range(iterations):
        conv_ag = A @ g  # conv(a, g)
        # Multiplicative update: g_j *= exp(-eta * sum_k conv_ag[k] / M)
        # Actually: g_j *= product_k exp(-eta * A[k,j] / M)
        # = exp(-eta/M * sum_k A[k,j])
        # sum_k A[k,j] = sum_k conv(a,g)[k] contribution from j
        # This is sum_k sum_i a[i]*delta(k-i-j) = sum_i a[i] = sum(a) = 1 (if normalized)
        # Hmm, that's not quite right. Let me use the standard MW update:
        # w_j <- w_j * exp(-eta * (A^T u)_j) where u_k = 1/M for all k
        # (A^T u)_j = sum_k A[k,j] * (1/M) = (sum_k a[k-j]) / M (for valid k)

        # Simpler: use the constraint violation
        violations = np.maximum(conv_ag - M, 0)
        # Gradient of constraint violation w.r.t. g
        grad = A.T @ (violations / M)
        g = g * np.exp(-eta * grad)
        g = g / g.sum()

    return g


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def diverse_starts(n, seed=42):
    """Generate diverse initial sequences of length n."""
    rng = np.random.default_rng(seed)
    starts = []

    # Uniform
    starts.append(np.ones(n) / n)

    # Gaussian bump
    x = np.linspace(0, 1, n)
    starts.append(np.exp(-10 * (x - 0.5) ** 2))
    starts[-1] = starts[-1] / starts[-1].sum()

    # Two bumps
    starts.append(np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2))
    starts[-1] = starts[-1] / starts[-1].sum()

    # Cosine window
    c = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    starts.append(c / c.sum())

    # Random smooth (low-pass filtered random)
    r = rng.random(n)
    # Simple moving average
    kernel_size = max(1, n // 20)
    kernel = np.ones(kernel_size) / kernel_size
    r_smooth = np.convolve(r, kernel, mode='same')
    starts.append(r_smooth / r_smooth.sum())

    # Triangle
    tri = np.minimum(x, 1 - x) * 2
    starts.append(tri / tri.sum())

    # Random with structure (periodic + noise)
    freq = rng.integers(2, 5)
    periodic = 0.5 + 0.3 * np.cos(2 * np.pi * freq * x)
    noise = 0.1 * rng.standard_normal(n)
    p = periodic + noise
    p = np.maximum(p, 0)
    if p.sum() > 0:
        starts.append(p / p.sum())

    return starts


def pollish_sequence(a, time_budget=60.0, verbose=False):
    """
    Full polishing pipeline for a single sequence.
    Alternates between active-set gradient, LP, and give-and-take.
    """
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    if verbose:
        print(f"  Polish start: n={n}, obj={best_obj:.8f}")

    while time.monotonic() - start < time_budget - 2:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        improved = False

        # Active set gradient descent
        if remaining > 10:
            alloc = min(remaining * 0.3, 25.0)
            a_new = active_set_gradient_descent(best_a, max_iters=int(alloc * 50),
                                                 time_budget=alloc, top_k_frac=0.08,
                                                 momentum=0.9, step_init=0.03)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                improved = True
                if verbose:
                    print(f"    ASGD: {best_obj:.8f}")

        # LP alternation (less frequent, more expensive)
        if n <= 800 and remaining > 20:
            alloc = min(remaining * 0.2, 15.0)
            a_new, obj_new = lp_alternate_step(best_a, time_limit=int(alloc * 0.8),
                                                line_search_points=6)
            if a_new is not None and obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                improved = True
                if verbose:
                    print(f"    LP: {best_obj:.8f}")

        # Give and take
        if remaining > 5:
            alloc = min(remaining * 0.15, 8.0)
            a_new = give_and_take(best_a, max_rounds=int(alloc * 10),
                                   time_budget=alloc)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                improved = True
                if verbose:
                    print(f"    G&T: {best_obj:.8f}")

        if not improved:
            # Try perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            for scale in [0.01, 0.03, 0.05]:
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = normalize(a_pert)
                # Quick ASGD on perturbed
                a_pert = active_set_gradient_descent(a_pert, max_iters=100,
                                                      time_budget=3.0,
                                                      top_k_frac=0.1,
                                                      momentum=0.9, step_init=0.02)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj - 1e-10:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    improved = True
                    if verbose:
                        print(f"    Pert(scale={scale}): {best_obj:.8f}")
                    break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Intensive optimization of parent + diverse starts at parent length
    Phase 2: Length exploration
    Phase 3: Final intensive polishing
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting search: best={best_score:.8f}, n_parent={n_parent}")

    # Phase 1: Optimize at parent length with diverse starts
    print(f"\n=== Phase 1: Intensive optimization (n={n_parent}) ===")
    phase1_deadline = start_time + (deadline - start_time) * 0.40

    # First: optimize the parent
    if best_seq is not None:
        a = normalize(best_seq.copy())
        a_new, obj_new = pollish_sequence(a, time_budget=80.0, verbose=True)
        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  Parent polished: {best_score:.8f}")

    # Try diverse starts at parent length
    starts = diverse_starts(n_parent, seed=seed)
    for idx, start_seq in enumerate(starts):
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 15:
            break

        print(f"  Start {idx}: quick eval...")
        # Quick evaluation
        a_try = normalize(start_seq.copy())
        obj_try = compute_obj(a_try)
        print(f"    Initial: {obj_try:.8f}")

        if obj_try < best_score + 0.5:
            # Worth optimizing
            alloc = min(remaining * 0.4, 40.0)
            a_new, obj_new = pollish_sequence(a_try, time_budget=alloc, verbose=False)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"    ** NEW BEST: {best_score:.8f} **")

    # Additional ASGD passes on the best
    if best_seq is not None and time.monotonic() < phase1_deadline - 30:
        remaining = phase1_deadline - time.monotonic()
        print(f"  Extra ASGD passes: {remaining:.0f}s available")
        a = normalize(best_seq.copy())
        for pass_idx in range(5):
            if time.monotonic() > phase1_deadline - 10:
                break
            alloc = min(remaining * 0.15, 15.0)
            a_new = active_set_gradient_descent(
                a, max_iters=int(alloc * 100), time_budget=alloc,
                top_k_frac=0.05 + 0.02 * pass_idx,
                momentum=0.9 - 0.05 * pass_idx,
                step_init=0.02 * (0.8 ** pass_idx))
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a = a_new.copy()
                print(f"    ASGD pass {pass_idx}: {best_score:.8f}")
            else:
                a = a_new  # still use it as starting point

    print(f"\nPhase 1 complete: best={best_score:.8f}")

    # Phase 2: Length exploration
    print(f"\n=== Phase 2: Length exploration ===")
    phase2_deadline = start_time + (deadline - start_time) * 0.70

    if best_seq is not None:
        n_best = len(best_seq)
        # Try nearby lengths
        length_factors = [0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.4, 1.5, 1.7, 2.0]
        n_options = sorted(set([int(n_best * f) for f in length_factors]))
        n_options = [n for n in n_options if 100 <= n <= 2000]
        # Remove the parent length (already optimized)
        n_options = [n for n in n_options if n != n_best]

        for n_try in n_options:
            if time.monotonic() > phase2_deadline:
                break
            remaining = phase2_deadline - time.monotonic()
            if remaining < 20:
                break

            # Resample best sequence to new length
            a_try = resample_sequence(best_seq, n_try)
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)

            # Also try a fresh random start at this length
            rng = np.random.default_rng(seed + n_try)
            a_rand = np.maximum(rng.standard_normal(n_try), 0)
            a_rand = normalize(a_rand)
            obj_rand = compute_obj(a_rand)

            # Pick the better starting point
            if obj_try <= obj_rand:
                a_start = a_try
            else:
                a_start = a_rand

            # Quick optimization
            alloc = min(remaining * 0.3, 30.0)
            a_new, obj_new = pollish_sequence(a_start, time_budget=alloc, verbose=False)

            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  ** NEW BEST: n={n_try}, obj={best_score:.8f} **")
            else:
                print(f"  n={n_try}: obj={obj_new:.8f} (no improvement)")

    print(f"\nPhase 2 complete: best={best_score:.8f}")

    # Phase 3: Final intensive polishing
    print(f"\n=== Phase 3: Final polishing ===")

    if best_seq is not None:
        n_final = len(best_seq)
        remaining = deadline - time.monotonic()
        print(f"  n={n_final}, {remaining:.0f}s remaining")

        a_final = normalize(best_seq.copy())

        # Multiple polishing rounds with decreasing tolerance
        for round_idx in range(8):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            alloc = min(remaining * 0.2, 30.0)
            a_new, obj_new = pollish_sequence(a_final, time_budget=alloc,
                                               verbose=(round_idx == 0))
            if obj_new < best_score - 1e-8:
                best_score = obj_new
                best_seq = a_new.copy()
                a_final = a_new.copy()
                print(f"  Round {round_idx}: {best_score:.8f}")
            else:
                # Try perturbation
                rng = np.random.default_rng(round_idx * 1000 + seed)
                best_pert = None
                for scale in [0.005, 0.01, 0.02, 0.03, 0.05]:
                    if time.monotonic() > deadline - 10:
                        break
                    a_pert = a_final * np.exp(rng.normal(0, scale, len(a_final)))
                    a_pert = normalize(a_pert)
                    a_pert = active_set_gradient_descent(
                        a_pert, max_iters=200, time_budget=5.0,
                        top_k_frac=0.05, momentum=0.9, step_init=0.01)
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_final = a_pert.copy()
                        best_pert = obj_pert
                        print(f"  Round {round_idx} pert(scale={scale}): {best_score:.8f}")
                        break

                if best_pert is None:
                    a_final = pollish_sequence(a_final, time_budget=min(alloc, 20.0))[0]

        # Final LP attempt on the very best
        if n_final <= 800 and time.monotonic() < deadline - 20:
            a_lp, obj_lp = lp_alternate_step(a_final, time_limit=10,
                                              line_search_points=10)
            if a_lp is not None and obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  Final LP: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
