        # Symmetric version
        sym = (p[:n // 2] + p[n // 2:][::-1]) / 2.0
        if n % 2 == 0:
            full_sym = np.concatenate([sym, sym[::-1]])
        else:
            full_sym = np.concatenate([sym, sym[-2::-1]])
        inits.append(np.maximum(full_sym, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.05))

    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(0.3 + 0.7 * np.abs(2.0 * t - 1.0))

    # Bump functions
    for center in [0.25, 0.5, 0.75]:
        inits.append(np.exp(-20.0 * (t - center) ** 2) + 0.05)

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Alternating pattern
    alt = np.ones(n)
    alt[1::2] = 0.7
    inits.append(alt)

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_a = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_a, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        update_best(parent)

    target_n = parent.size if parent.size >= 256 else 512

    # === PHASE 1: Multi-scale initialization ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-scale init", flush=True)
    print(f"{'='*60}", flush=True)
    p1_deadline = min(t_start + 80, deadline - 400)
    a_ms = multi_scale_init(parent, target_n, p1_deadline, rng)
    if a_ms is not None:
        update_best(a_ms)
    print(f"Phase 1 done: {time.monotonic()-t_start:.1f}s", flush=True)

    # === PHASE 2: Extended CD from multiple starts ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: CD from multiple starts (n={target_n})", flush=True)
    print(f"{'='*60}", flush=True)
    p2_deadline = min(t_start + 350, deadline - 300)

    inits = diverse_inits(target_n, rng, best_a if best_a is not None else parent)
    best_cd_M = float('inf')
    best_cd_a = None

    for idx, a0 in enumerate(inits):
        if time.monotonic() > p2_deadline:
            break
        a0 = np.maximum(a0, 1e-6)
        a0 = a0 / a0.sum()

        # Allocate more time to first few initializations
        remaining = p2_deadline - time.monotonic()
        n_remaining = len(inits) - idx
        per_init_time = max(20, remaining / max(1, n_remaining))
        init_deadline = min(time.monotonic() + per_init_time, p2_deadline)

        # CD with many sweeps
        max_sw = max(50, int(min(500, 1000000 // target_n)))
        a_opt = run_cd(a0, max_sw, init_deadline, shuffle_rng=rng)

        M_val = float(fast_conv(a_opt).max())
        if M_val < best_cd_M:
            best_cd_M = M_val
            best_cd_a = a_opt.copy()
            J_val = compute_J(a_opt)
            print(f"  init {idx}: M={M_val:.8f} J={J_val:.6f}", flush=True)
            update_best(a_opt)

    print(f"Phase 2 done: {time.monotonic()-t_start:.1f}s, best_cd_J={compute_J(best_cd_a):.6f}", flush=True)

    # === PHASE 3: LP diversification ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)
    p3_deadline = min(t_start + 700, deadline - 200)

    if best_a is not None:
        n_best = best_a.size
        # Try different K values
        for K_val in [10, 20, 40]:
            if time.monotonic() > p3_deadline:
                break
            a_lp = run_lp_mirror(best_a, max_iters=100, deadline=p3_deadline, K=K_val)
            if a_lp is not None:
                update_best(a_lp)

    print(f"Phase 3 done: {time.monotonic()-t_start:.1f}s", flush=True)

    # === PHASE 4: CD refinement after LP ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: CD Refinement", flush=True)
    print(f"{'='*60}", flush=True)
    p4_deadline = min(t_start + 1000, deadline - 150)

    if best_a is not None and time.monotonic() < p4_deadline:
        a_ref = run_cd(best_a, 200, p4_deadline, shuffle_rng=rng)
        update_best(a_ref)

        # Perturb and re-optimize
        for pert_scale in [0.01, 0.05, 0.1]:
            if time.monotonic() > p4_deadline:
                break
            a_pert = best_a + rng.normal(0, pert_scale, best_a.size)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            a_ref2 = run_cd(a_pert, 100, min(time.monotonic() + 60, p4_deadline), shuffle_rng=rng)
            update_best(a_ref2)

    print(f"Phase 4 done: {time.monotonic()-t_start:.1f}s", flush=True)

    # === PHASE 5: More LP with adapted K ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final LP + CD", flush=True)
    print(f"{'='*60}", flush=True)
    p5_deadline = min(t_start + 1200, deadline - 80)

    if best_a is not None and time.monotonic() < p5_deadline:
        for K_val in [15, 30, 50]:
            if time.monotonic() > p5_deadline:
                break
            a_lp = run_lp_mirror(best_a, max_iters=50, deadline=min(time.monotonic() + 40, p5_deadline), K=K_val)
            if a_lp is not None:
                update_best(a_lp)
                a_cd = run_cd(a_lp, 50, min(time.monotonic() + 30, p5_deadline), shuffle_rng=rng)
                update_best(a_cd)

    print(f"Phase 5 done: {time.monotonic()-t_start:.1f}s", flush=True)

    # === PHASE 6: Length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Length exploration", flush=True)
    print(f"{'='*60}", flush=True)
    p6_deadline = deadline - 40

    if best_a is not None and time.monotonic() < p6_deadline:
        n_curr = best_a.size
        candidate_lengths = [n_curr // 2, n_curr * 2]
        # Filter reasonable lengths
        candidate_lengths = [nl for nl in candidate_lengths if 128 <= nl <= 2048]
        # Add a few nearby lengths
        for delta in [-128, 128, -64, 64]:
            nl = n_curr + delta
            if 128 <= nl <= 2048 and nl not in candidate_lengths:
                candidate_lengths.append(nl)

        for nl in candidate_lengths[:4]:
            if time.monotonic() > p6_deadline:
                break
            a_ext = resample_seq(best_a, nl)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            max_sw = max(30, int(min(200, 500000 // nl)))
            a_opt = run_cd(a_ext, max_sw, min(time.monotonic() + 60, p6_deadline), shuffle_rng=rng)
            update_best(a_opt)

            if time.monotonic() < p6_deadline:
                a_lp = run_lp_mirror(a_opt, max_iters=30, deadline=min(time.monotonic() + 30, p6_deadline), K=20)
                if a_lp is not None:
                    update_best(a_lp)

    print(f"Phase 6 done: {time.monotonic()-t_start:.1f}s", flush=True)

    # === PHASE 7: Final polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 7: Final polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None:
        a_final = run_cd(best_a, 100, deadline - 5, shuffle_rng=rng)
        update_best(a_final)

    # Fallback
    if best_a is None:
        best_a = np.ones(512)
        best_score = compute_J(best_a)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_a]
