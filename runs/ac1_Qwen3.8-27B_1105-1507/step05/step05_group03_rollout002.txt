import numpy as np

from scipy.sparse import csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v, s=1.0):
    """Project vector v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []

        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False, lp_timeout=10):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_J(a)
        # Scale M by sum^2 for the LP (we work on normalized a)
        s_curr = a.sum()
        M_real = M * (s_curr ** 2) / (2 * n)

        g = solve_lp(a, M_real, row_idx, col_idx, time_limit=lp_timeout)
        if g is None:
            # Try with a slightly relaxed M
            g = solve_lp(a, M_real * 1.001, row_idx, col_idx, time_limit=lp_timeout)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-8:
            no_improve += 1
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                J_new = compute_J(a)
                if J_new < best_M:
                    best_M = J_new
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 150:
            break

        if verbose and it % 100 == 0:
            print(f"      LP iter {it}: J={compute_J(a):.10f} best_J={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def smooth_grad_logsumexp(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*(a*a)))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    # Gradient of (a*a)[k] w.r.t. a is: for each m, 2*a[k-m] if 0<=k-m<n
    # So grad[m] = 2 * sum_k p[k] * a[k-m] (for valid k-m)
    # This is 2 * cross-correlation of p with a, appropriately indexed.
    # (conv(p, a))[k+m] involves sum_j p[j]*a[k+m-j], not quite what we want.
    # Let's compute directly:
    # grad_a[m] = 2 * sum_k p[k] * a[k-m]  (where 0<=k-m<n and 0<=k<2n-1)
    # Let i = k-m, so k = i+m, and i ranges in [0, n-1], k in [m, m+n-1]
    # grad_a[m] = 2 * sum_{i=0}^{n-1} p[i+m] * a[i]  (where i+m < 2n-1)
    # This is 2 * (p * a)[m] in the "upconv" sense... let me just do it with FFT.
    
    # Actually, sum_k p[k]*a[k-m] for valid indices is the cross-correlation.
    # Let's use: grad_a = 2 * np.array([sum_k p[k]*a[k-m] for m in range(n)])
    # This is the same as the convolution of p with a, taking appropriate slices.
    # conv(p, a)[m + i] = sum_j p[j]*a[m+i-j]... not quite.
    # Let me just use a direct approach for correctness, or FFT.
    
    # Using FFT: if we define c[m] = sum_{i=0}^{n-1} p[i+m]*a[i],
    # this is the cross-correlation of p and a (with p indexed starting from m).
    # c = iffft(fft(p_extended) * conj(fft(a_extended)))... 
    
    # Simpler: c[m] = sum_i p[i+m]*a[i] for i in [0,n-1].
    # Let's build a matrix or use the fact that this is a Toeplitz product.
    # For efficiency with n up to ~1024, direct computation in a loop is O(n^2)
    # but we can use FFT: c = ifft(fft(p, 2n) * fft(a_padded, 2n)) won't work directly.
    
    # Actually, c[m] = <p[m:m+n], a> = sum_{i=0}^{n-1} p[m+i] * a[i]
    # This is the cross-correlation (p circ a)[m] in a specific sense.
    # Let's just compute it as: for each m, dot product of p[m:m+n] with a.
    # For n up to 1024, this is 1024 dot products of length 1024 = ~1M ops, fine.
    
    # More efficient: c = np.zeros(n)
    # for i in range(n):
    #     c[:n-i] += a[i] * p[i:n]  # not quite right either
    
    # Let me think again. c[m] = sum_{i=0}^{n-1} p[i+m] * a[i]
    # = sum_{i=0}^{min(n-1, 2n-2-m)} p[i+m] * a[i]
    # For m=0: sum_i p[i]*a[i] (but p has 2n-1 elements, so p[0:n] dot a)
    # For m=n-1: sum_i p[i+n-1]*a[i] = p[n-1:2n-1] dot a (p has indices 0..2n-2)
    
    # So c[m] = np.dot(p[m:m+n], a) for m in [0, n-1]
    # This is just slicing p and taking dot product with a.
    
    c = np.array([np.dot(p[m:m+n], a) for m in range(n)])
    grad = 2.0 * c
    return grad


def run_nesterov_smooth(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0):
    """Nesterov-accelerated smooth gradient descent on log-sum-exp of conv."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    
    # Nesterov variables
    y = a.copy()
    y_prev = a.copy()
    beta = 0.9

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.3)
        lr = 0.5 / (1.0 + it * 0.01)

        # Nesterov step
        y_new = y + beta * (y - y_prev)
        y_prev = y
        y = y_new

        # Compute gradient at y
        grad = smooth_grad_logsumexp(y, tau)
        
        # Update
        a_new = y - lr * grad
        a_new = proj_simplex(a_new, 1.0)
        a = a_new

        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)
        elif np.isfinite(score):
            pass  # Not better, no print

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_differential_evolution(best_seq, tracker, deadline, rng, pop_size=12, max_gens=60):
    """Differential evolution with LP polishing for top candidates."""
    n = best_seq.size
    best_norm = best_seq / best_seq.sum()
    
    # Initialize population around best
    population = []
    for i in range(pop_size):
        if i == 0:
            pop = best_norm.copy()
        else:
            noise_scale = 0.02 * (1.0 + i * 0.3)
            pop = best_norm + rng.normal(0, noise_scale, n)
            pop = np.maximum(pop, 0)
            s = pop.sum()
            if s > 1e-10:
                pop = pop / s
            else:
                pop = best_norm.copy()
        population.append({"a": pop, "J": compute_J(pop)})
    
    population.sort(key=lambda x: x["J"])
    row_idx, col_idx = build_lp_matrix(n)
    
    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break
        
        new_population = []
        # Keep the top half
        keep = pop_size // 2
        for i in range(keep):
            new_population.append(population[i])
        
        # Generate new candidates
        for i in range(pop_size - keep):
            if time.monotonic() > deadline:
                break
            
            # DE/rand/1 mutation
            r1, r2, r3 = rng choisir(3, keep) if False else tuple(rng.choice(keep, size=3, replace=False))
            F = 0.6 + 0.3 * rng.random()
            CR = 0.4 + 0.3 * rng.random()
            
            mutant = population[r1[0]]["a"] + F * (population[r1[1]]["a"] - population[r1[2]]["a"])
            
            # Binomial crossover
            trial = population[r1[0]]["a"].copy()
            cross_mask = rng.random(n) < CR
            trial[cross_mask] = mutant[cross_mask]
            trial = np.maximum(trial, 0)
            s = trial.sum()
            if s > 1e-10:
                trial = trial / s
            
            J_trial = compute_J(trial)
            
            # Competition with a random member of the kept set
            j = rng.integers(keep, pop_size)
            if J_trial < population[min(j, len(population)-1)]["J"]:
                new_population.append({"a": trial, "J": J_trial})
            else:
                # Add a perturbed version of the best
                pert = best_norm + rng.normal(0, 0.005, n)
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                new_population.append({"a": pert, "J": compute_J(pert)})
        
        population = new_population[:pop_size]
        population.sort(key=lambda x: x["J"])
        
        # Every 8 generations, polish top 2 with LP
        if gen % 8 == 7:
            for i in range(min(2, len(population))):
                sub_dl = min(time.monotonic() + 20, deadline)
                polished = run_lp_mirror_descent(
                    population[i]["a"], 200, sub_dl, row_idx, col_idx,
                    verbose=False, lp_timeout=5
                )
                J_pol = compute_J(polished)
                if J_pol < population[i]["J"]:
                    population[i]["a"] = polished
                    population[i]["J"] = J_pol
                tracker.update(polished, f"(DE gen={gen} i={i})")
        
        if gen % 15 == 0:
            print(f"      DE gen {gen}: best_J={population[0]['J']:.10f}", flush=True)
    
    # Return best from DE
    return population[0]["a"]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    1. LP mirror descent (primary)
    2. Nesterov smooth gradient (exploration)
    3. Differential evolution population (global search)
    4. Multi-length exploration
    5. Final LP polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    n0 = parent.size
    print(f"\n{'='*60}", flush=True)
    print(f"Starting: n={n0}, J={tracker.best_score:.8f}", flush=True)
    print(f"{'='*60}", flush=True)

    # ===== PHASE 1: LP Mirror Descent =====
    print(f"\n--- PHASE 1: LP Mirror Descent ---", flush=True)
    row_idx0, col_idx0 = build_lp_matrix(n0)
    a1 = parent.copy()
    a1 = a1 / a1.sum()
    
    phase1_end = min(t_start + budget_s * 0.30, deadline)
    a_opt = run_lp_mirror_descent(a1, 50000, phase1_end, row_idx0, col_idx0,
                                   verbose=True, lp_timeout=15)
    tracker.update(a_opt, "(P1 LP-MD)")

    # ===== PHASE 2: Nesterov Smooth Gradient =====
    print(f"\n--- PHASE 2: Nesterov Smooth Gradient ---", flush=True)
    if tracker.best is not None:
        a2 = tracker.best.copy()
        a2 = a2 / a2.sum()
        n2 = a2.size
        phase2_end = min(t_start + budget_s * 0.40, deadline)
        a_opt2 = run_nesterov_smooth(a2, 5000, phase2_end, initial_tau=8.0, final_tau=300.0)
        tracker.update(a_opt2, "(P2 Nesterov)")
        
        # LP polish after Nesterov
        if tracker.best is not None:
            n_cur = tracker.best.size
            r_idx_p, c_idx_p = build_lp_matrix(n_cur)
            a_pol = tracker.best.copy() / tracker.best.sum()
            sub_dl = min(t_start + budget_s * 0.45, deadline)
            a_opt2b = run_lp_mirror_descent(a_pol, 5000, sub_dl, r_idx_p, c_idx_p,
                                             verbose=False, lp_timeout=10)
            tracker.update(a_opt2b, "(P2 LP polish)")

    # ===== PHASE 3: Differential Evolution =====
    print(f"\n--- PHASE 3: Differential Evolution ---", flush=True)
    if tracker.best is not None:
        phase3_end = min(t_start + budget_s * 0.70, deadline)
        de_best = run_differential_evolution(
            tracker.best, tracker, phase3_end, rng,
            pop_size=14, max_gens=80
        )
        tracker.update(de_best, "(P3 DE final)")

    # ===== PHASE 4: Multi-length Exploration =====
    print(f"\n--- PHASE 4: Multi-length Exploration ---", flush=True)
    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.5, 0.625, 0.75, 1.25, 1.5, 1.75, 2.0, 2.5]:
            n_trial = int(base_n * mult)
            if 64 <= n_trial <= 4096 and n_trial != base_n:
                trial_lengths.append(n_trial)
        trial_lengths = sorted(set(trial_lengths))

        phase4_end = min(t_start + budget_s * 0.85, deadline)
        time_per_trial = max(20, (phase4_end - time.monotonic()) / max(1, len(trial_lengths)))
        
        for n_trial in trial_lengths:
            if time.monotonic() > phase4_end:
                break
            
            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()

            r_idx_t, c_idx_t = build_lp_matrix(n_trial)
            sub_dl = min(time.monotonic() + time_per_trial, phase4_end)
            a_opt_t = run_lp_mirror_descent(a_init, 3000, sub_dl, r_idx_t, c_idx_t,
                                             verbose=False, lp_timeout=8)
            tracker.update(a_opt_t, f"(P4 n={n_trial})")

    # ===== PHASE 5: Final LP Polish =====
    print(f"\n--- PHASE 5: Final LP Polish ---", flush=True)
    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)
        a_f = tracker.best.copy() / tracker.best.sum()
        # Use longer LP timeouts for final polish
        a_opt_f = run_lp_mirror_descent(a_f, 50000, deadline, r_f, c_f,
                                         verbose=True, lp_timeout=20)
        tracker.update(a_opt_f, "(P5 final)")
        
        # One more round: Nesterov + LP
        if tracker.best is not None:
            n_f2 = tracker.best.size
            a_n2 = tracker.best.copy() / tracker.best.sum()
            sub_dl2 = min(time.monotonic() + 30, deadline)
            a_n2 = run_nesterov_smooth(a_n2, 2000, sub_dl2, initial_tau=20, final_tau=500)
            tracker.update(a_n2, "(P5 Nesterov2)")
            
            if tracker.best is not None:
                r_f2, c_f2 = build_lp_matrix(tracker.best.size)
                a_f2 = tracker.best.copy() / tracker.best.sum()
                a_f2 = run_lp_mirror_descent(a_f2, 20000, deadline, r_f2, c_f2,
                                              verbose=False, lp_timeout=15)
                tracker.update(a_f2, "(P5 final2)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
