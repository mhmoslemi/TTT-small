"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    """Full cross-convolution via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M, time_limit=5):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def line_search_bilinear(b_aa, b_ag, b_gg):
    """Find t in [0,1] minimizing max((1-t)^2*b_aa + 2t(1-t)*b_ag + t^2*b_gg)."""
    def obj(t):
        b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
        return b.max()
    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                             options={'xatol': 1e-9})
    return float(result.x)

def lp_mirror_descent(a, max_iters, deadline, use_cd_every=30, epsilon=1e-10):
    """LP mirror descent with fast line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b_aa = fast_conv(a)
        M = float(b_aa.max())
        g = lp_best_response(a, M, time_limit=5)
        if g is None:
            break
        g = g / g.sum()
        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)
        t_opt = line_search_bilinear(b_aa, b_ag, b_gg)
        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)
        if J_new < best_J - 1e-13:
            best_a = a_new.copy()

            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80:
            a = a + np.random.normal(0, 0.01, n)
            a = np.maximum(a, epsilon)
            a = a / a.sum()
            no_improve = 0
    return best_a, best_J

def compute_gradient_r(a, k):
    """Compute gradient of r[k] with respect to a. Returns vector of length n."""
    n = len(a)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 2.0 * a
    elif k > 0:
        # r[k] = sum_i a[i] * a[i+k] for i = 0 to n-1-k
        # d/d a[j]: a[j-k] (if 0<=j-k<n) + a[j+k] (if 0<=j+k<n)
        # For j in 0..n-1:
        #   a[j-k] valid if j >= k, i.e., j in k..n-1
        #   a[j+k] valid if j+k < n, i.e., j in 0..n-1-k
        grad[k:] = a[:n-k]  # a[j-k] for j=k..n-1
        grad[:n-k] += a[k:]  # a[j+k] for j=0..n-1-k
    # For k < 0, r[k] = r[|k|], so gradient is the same
    return grad

def identify_active_set(a, tol=1e-6):
    """Identify active lags (where r[k] is close to max)."""
    r = fast_conv(a)
    max_r = r.max()
    # r is indexed 0 to 2n-2, where r[k] corresponds to lag k-(n-1)
    # The "center" is at index n-1 (lag 0)
    n = len(a)
    active = []
    for idx in range(len(r)):
        if r[idx] >= max_r - tol * max_r:
            lag = idx - (n - 1)
            active.append(lag)
    return active, max_r

def newton_kkt(a, active_set, max_newton_iters=50, tol=1e-10):
    """
    Solve KKT system for the given active set using Newton's method.
    a: current solution (normalized, sum=1)
    active_set: list of lags k where r[k] is at the maximum
    Returns: improved a, success flag
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    
    # Convert lags to non-negative indices for the convolution
    # r[k] for lag k (can be negative) = r[|k|] in our convention
    # In our fast_conv output, index i corresponds to lag i-(n-1)
    # So lag k corresponds to index k + (n-1)
    
    def get_r_vals(a_vec):
        """Get r values for the active set."""
        r_full = fast_conv(a_vec)
        return np.array([r_full[k + n - 1] for k in active_set])
    
    def get_all_r(a_vec):
        return fast_conv(a_vec)
    
    # KKT system:
    # Let S = active_set, m = |S|
    # Unknowns: a (n), lambda (m), mu (1)
    # Equations:
    #   (1) r[k_i] - r[k_0] = 0 for i=1..m-1  [m-1 eqs]
    #   (2) sum(a) - 1 = 0                      [1 eq]
    #   (3) sum_i lambda_i * grad_{k_i}(a) - mu * 1 = 0  [n eqs]
    #   (4) sum(lambda) - 1 = 0                 [1 eq]
    
    m = len(active_set)
    if m < 2:
        return a, False
    
    k0 = active_set[0]
    k_rest = active_set[1:]
    
    def residual(a_vec, lam, mu):
        """Compute the residual of the KKT system."""
        r_full = fast_conv(a_vec)
        # (1): r[k_i] - r[k_0] for i in k_rest
        res1 = np.array([r_full[k + n - 1] - r_full[k0 + n - 1] for k in k_rest])
        # (2): sum(a) - 1
        res2 = a_vec.sum() - 1.0
        # (3): sum_i lam_i * grad_{k_i}(a) - mu
        grad_sum = np.zeros(n)
        for i, k in enumerate(active_set):
            if lam[i] > 1e-15:
                grad_sum += lam[i] * compute_gradient_r(a_vec, k)
        res3 = grad_sum - mu
        # (4): sum(lam) - 1
        res4 = lam.sum() - 1.0
        return np.concatenate([res1, [res2], res3, [res4]])
    
    def jacobian(a_vec, lam, mu):
        """Compute the Jacobian of the residual."""
        # Size: (m-1 + 1 + n + 1) x (n + m + 1)
        # = (n + m) x (n + m + 1)
        
        # Precompute gradients for active set
        grads = []
        for k in active_set:
            grads.append(compute_gradient_r(a_vec, k))
        grads = np.array(grads)  # shape (m, n)
        
        # Block (1, a): (m-1) x n
        J_1a = np.zeros((m - 1, n))
        for i in range(m - 1):
            J_1a[i] = grads[i + 1] - grads[0]  # grad_{k_{i+1}} - grad_{k_0}
        
        # Block (2, a): 1 x n
        J_2a = np.ones(n)
        
        # Block (3, a): n x n
        # (3) is: sum_i lam_i * grad_{k_i}(a) - mu * 1 = 0
        # d/d a_j [sum_i lam_i * grad_{k_i}(a)_j] = sum_i lam_i * d(grad_{k_i}(a)_j)/da_j
        # grad_k(a)_j = a[j-k] + a[j+k] (for k>0)
        # d grad_k(a)_j / da_l = delta_{l, j-k} + delta_{l, j+k}
        # So d/d a_l [sum_i lam_i * grad_{k_i}(a)_j] = sum_i lam_i * (delta_{l, j-k_i} + delta_{l, j+k_i})
        J_3a = np.zeros((n, n))
        for i, k in enumerate(active_set):
            if abs(lam[i]) < 1e-15:
                continue
            if k == 0:
                # grad_0(a)_j = 2*a_j, d/d a_l = 2*delta_{j,l}
                J_3a += 2.0 * lam[i] * np.eye(n)
            else:
                kk = abs(k)
                for j in range(n):
                    # delta_{l, j-kk}
                    l1 = j - kk
                    if 0 <= l1 < n:
                        J_3a[j, l1] += lam[i]
                    l2 = j + kk
                    if 0 <= l2 < n:
                        J_3a[j, l2] += lam[i]
        
        # Block (3, lambda): n x m
        J_3lam = grads.T  # J_3lam[j, i] = grads[i, j] = grad_{k_i}(a)_j
        
        # Block (3, mu): n x 1
        J_3mu = -np.ones(n)
        
        # Assemble full Jacobian
        total_rows = (m - 1) + 1 + n + 1
        total_cols = n + m + 1
        J = np.zeros((total_rows, total_cols))
        row = 0
        for i in range(m - 1):
            J[row, :n] = J_1a[i]
            row += 1
        J[row, :n] = J_2a
        row += 1
        J[row:row+n, :n] = J_3a
        J[row:row+n, n:n+m] = J_3lam
        J[row:row+n, n+m] = J_3mu
        row += n
        J[row, n:n+m] = 1.0
        return J
    
    # Newton iteration
    lam = np.ones(m) / m  # initial guess
    mu = 1.0
    
    for nit in range(max_newton_iters):
        F = residual(a, lam, mu)
        if np.max(np.abs(F)) < tol:
            break
        
        J = jacobian(a, lam, mu)
        try:
            delta = np.linalg.solve(J, -F)
        except np.linalg.LinAlgError:
            # Use least squares as fallback
            delta, _, _, _ = np.linalg.lstsq(J, -F, rcond=None)
        
        # Damping: find alpha in (0,1] such that ||F(new)|| < ||F(old)||
        F_norm = np.linalg.norm(F)
        alpha = 1.0
        for _ in range(20):
            # Split delta
            da = delta[:n]
            dlam = delta[n:n+m]
            dm = delta[n+m]
            
            a_trial = a + alpha * da
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            lam_trial = np.maximum(lam + alpha * dlam, 0)
            if lam_trial.sum() > 1e-10:
                lam_trial = lam_trial / lam_trial.sum()
            mu_trial = mu + alpha * dm
            
            F_trial = residual(a_trial, lam_trial, mu_trial)
            if np.linalg.norm(F_trial) < F_norm:
                break
            alpha *= 0.5
        
        a = np.maximum(a + alpha * delta[:n], 0)
        a = a / a.sum()
        lam = np.maximum(lam + alpha * delta[n:n+m], 0)
        lam = lam / lam.sum()
        mu = mu + alpha * delta[n+m]
    
    # Verify: check that r[k] <= max_r for all k
    r_full = fast_conv(a)
    max_r = r_full.max()
    active_max = max(r_full[k + n - 1] for k in active_set)
    if max_r > active_max * (1 + 1e-6):
        # Some non-active constraint became active - signal to update active set
        return a, False
    
    return a, True

def softmax_gradient_descent(a, deadline, beta_start=100, beta_end=5000, max_iters=2000):
    """
    Minimize smooth max: (1/beta)*log(sum_k exp(beta*r[k]))
    using gradient descent with momentum and adaptive beta.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    momentum = np.zeros(n)
    beta = beta_start
    lr = 0.01
    beta_step = (beta_end / beta_start) ** (1.0 / max_iters)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        beta = beta_start * (beta_step ** it)
        
        r = fast_conv(a)
        # Softmax weights (shift by max for numerical stability)
        r_shifted = beta * r - beta * r.max()
        exp_r = np.exp(r_shifted)
        p = exp_r / exp_r.sum()
        
        # Gradient of smooth max
        grad = np.zeros(n)
        for idx in range(2 * n - 1):
            if p[idx] > 1e-10:
                lag = idx - (n - 1)
                grad += p[idx] * compute_gradient_r(a, lag)
        
        # Project gradient onto tangent space of simplex (sum(a) = 1)
        grad -= np.mean(grad)
        
        # Momentum
        momentum = 0.9 * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new
        
        # Adaptive learning rate
        if J_new > best_J + 0.01:
            lr *= 0.5
        elif it % 50 == 49:
            lr = min(lr * 1.1, 0.05)
    
    return best_a

def coordinate_descent_pass(a, deadline, max_passes=10):
    """Coordinate descent for final polishing."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        order = np.random.permutation(n)
        improved = False
        r = fast_conv(a)
        S = a.sum()
        
        for i in order:
            if time.monotonic() > deadline:
                break
            ai = a[i]
            candidates = [0.0, ai * 0.5, ai * 0.75, ai * 1.25, ai * 1.5, ai * 2.0]
            if ai < 1e-6:
                candidates = [0.01, 0.05, 0.1]
            
            best_cand = ai
            best_score = compute_J(a)
            
            for cand in candidates:
                if abs(cand - ai) < 1e-10:
                    continue
                a[i] = cand
                a_clip = np.maximum(a, 0)
                if a_clip.sum() < 0.01:
                    a[i] = ai
                    continue
                score = compute_J(a_clip)
                if score < best_score:
                    best_score = score
                    best_cand = cand
            
            if best_cand != ai:
                a[i] = best_cand
                improved = True
        
        if not improved:
            break
    
    final_J = compute_J(a)
    if final_J < best_J:
        best_J = final_J
        best_a = a.copy()
    return best_a

def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random
    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())
    
    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())
    
    # Bell
    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())
    
    # Double bump
    dbl = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())
    
    # Edge concentration
    edge = 0.1 + np.exp(-10 * t**2) + np.exp(-10 * (1-t)**2)
    inits.append(edge / edge.sum())
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.5
    inits.append(alt / alt.sum())
    
    # Frequency modulated
    for freq in [2, 3, 5]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())
    
    return inits

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0
