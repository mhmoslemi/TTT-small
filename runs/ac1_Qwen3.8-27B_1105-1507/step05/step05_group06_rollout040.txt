def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(p, beta):

    b = fast_conv(p)
    # Shift for numerical stability
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    w = exp_b / exp_b.sum()  # softmax weights

    t = 0

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    v_prev = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(it / max(1, max_iters), 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac

        grad, b = smooth_max_grad_fast(p, beta)

        if it % 50 == 0:
            J = compute_J(p)
            if J < best_J:

                best_p = p.copy()
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # J = 2n * logsumexp(beta*b) / (beta * (sum p)^2)

        # For the projected method, we just use the gradient direction

def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=30):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_bilinear(b_aa, b_ag, b_gg):
    """Find t in [0,1] minimizing max((1-t)^2*b_aa + 2t(1-t)*b_ag + t^2*b_gg)."""
    def obj(t):
        b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
        return b.max()
    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                             options={'xatol': 1e-10})
    return float(result.x)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with log-sum-exp (Adam + animated beta)
2. LP mirror descent with bilinear line search
3. Fast vectorized coordinate descent with exact 1D ternary search

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution via FFT."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(p, beta):
    """
    Compute gradient of logsumexp(beta * conv(p,p)) w.r.t. p.
    Returns (grad, b) where b = conv(p,p).
    """
    n = len(p)
    b = fast_conv(p)
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    w = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(w[i:i + n], p)
    
    return grad, b


def projected_smooth_grad_descent(p, deadline, max_iters=30000, beta_start=3.0, beta_end=500.0):

    beta1 = 0.9

        beta = beta_start * (beta_end / beta_start) ** frac

def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows = np.empty((2 * n - 1) * n, dtype=np.int32)
    cols = np.empty((2 * n - 1) * n, dtype=np.int32)
    vals = np.empty((2 * n - 1) * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=25):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_bilinear(b_aa, b_ag, b_gg):
    """Find t in [0,1] minimizing max((1-t)^2*b_aa + 2t(1-t)*b_ag + t^2*b_gg)."""
    def obj(t):
        b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
        return b.max()
    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                             options={'xatol': 1e-10})
    return float(result.x)

                k_lo = i
                k_hi = min(i + n - 1, 2 * n - 2)
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j == i:
                        b[k] += 2.0 * delta * p[i] + delta * delta
                    else:
                        b[k] += 2.0 * delta * p[j]
                improved = True
                improvements += 1
        
        if improved:
            # Renormalize
            p = np.maximum(p, 0)
            s = p.sum()
            if s > 0.01:
                p = p / s
                J = compute_J(p)
                if J < best_J:
                    best_J = J
                    best_p = p.copy()
        else:
            break
    
    return best_p, best_J


def coordinate_descent_2d(p, deadline, max_passes=5, num_pairs=200):
    """
    2D coordinate descent: update pairs of coordinates simultaneously.
    """
    n = len(p)
    p = np.maximum(p.astype(np.float64), 0)
    s = p.sum()
    if s < 0.01:
        return p, compute_J(p)
    p = p / s
    
    best_p = p.copy()
    best_J = compute_J(p)
    
    for pass_num in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(p)
        S = p.sum()
        improved = False
        
        # Random pairs
        for _ in range(num_pairs):
            if time.monotonic() > deadline:
                break
            
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            
            pi = p[i]
            pj = p[j]
            
            # Grid search in 2D: try a few values for each
            best_xi, best_xj = pi, pj
            best_val = compute_J(p)
            
            candidates_i = [0.0, pi * 0.5, pi * 1.5, pi * 2.0]
            candidates_j = [0.0, pj * 0.5, pj * 1.5, pj * 2.0]
            
            for xi in candidates_i:
                for xj in candidates_j:
                    if abs(xi - pi) < 1e-14 and abs(xj - pj) < 1e-14:
                        continue
                    p[i] = xi
                    p[j] = xj
                    s_check = p.sum()
                    if s_check < 0.01:
                        p[i] = pi
                        p[j] = pj
                        continue
                    val = compute_J(p / s_check)
                    if val < best_val:
                        best_val = val
                        best_xi = xi
                        best_xj = xj
            
            p[i] = pi
            p[j] = pj
            
            if abs(best_xi - pi) > 1e-14 or abs(best_xj - pj) > 1e-14:
                p[i] = best_xi
                p[j] = best_xj
                improved = True
        
        if improved:
            p = np.maximum(p, 0)
            s = p.sum()
            if s > 0.01:
                p = p / s
                J = compute_J(p)
                if J < best_J:
                    best_J = J
                    best_p = p.copy()
        else:
            break
    
    return best_p, best_J


def generate_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear ramp (asymmetric)
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())
    
    # Inverse ramp
    inv_ramp = np.linspace(1.0, 0.1, n)
    inits.append(inv_ramp / inv_ramp.sum())
    
    # Exponential decay
    for rate in [2, 5, 10]:
        exp_d = np.exp(-rate * t)
        inits.append(exp_d / exp_d.sum())
    
    # Double exponential
    for rate in [3, 7]:
        dbl = np.exp(-rate * t) + np.exp(-rate * (1 - t))
        inits.append(dbl / dbl.sum())
    
    # Sigmoid shapes
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.4]:
            sig = 1.0 / (1.0 + np.exp(-(t - center) / width))
            inits.append(sig / sig.sum())
    
    # Frequency modulated
    for freq in [1, 2, 4, 8]:
        for amp in [0.3, 0.6, 0.9]:
            fm = 1.0 + amp * np.cos(2 * np.pi * freq * t)
            fm = np.maximum(fm, 0.01)
            inits.append(fm / fm.sum())
    
    # Power law
    for alpha in [0.3, 0.5, 1.0, 2.0, 3.0]:
        pl = (t + 0.01) ** alpha
        inits.append(pl / pl.sum())
    
    # Random smooth
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm, 0.01)
    inits.append(sm / sm.sum())
    
    # Random (noisy)
    r = rng.uniform(0.1, 1.0, n)
    inits.append(r / r.sum())
    
    # W-shaped
    w = np.exp(-20 * (t - 0.2) ** 2) + np.exp(-20 * (t - 0.8) ** 2) + 0.1 * np.exp(-20 * (t - 0.5) ** 2)
    inits.append(w / w.sum())
    
    # Asymmetric bell
    asym = np.exp(-4 * ((t - 0.4) / 0.6) ** 2) * np.exp(-2 * ((t - 0.4) / 0.3) ** 2 * (t - 0.4))
    inits.append(asym / asym.sum())
    
    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Smooth gradient descent (350s)
    Phase 2: LP mirror descent (400s)
    Phase 3: Coordinate descent + 2D CD (250s)
    Phase 4: Diverse starts + multi-length (300s)
    Phase 5: Final polishing (100s)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0
    
    best_seq = None
    best_score = float('inf')
    
    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.10f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)
    
    n_parent = parent.size
    p_norm = parent / parent.sum()
    
    # === PHASE 1: Smooth gradient descent ===
    p1_deadline = t_start + 350
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smooth Gradient Descent (n={n_parent}, budget=350s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    a_sg = np.maximum(p_norm + rng.normal(0, 0.01, n_parent), 0.01)
    a_sg = a_sg / a_sg.sum()
    a_sg_result, J_sg = projected_smooth_grad_descent(
        a_sg, min(p1_deadline, deadline - 500),
        max_iters=50000, beta_start=3.0, beta_end=400.0)
    update_best(a_sg_result)
    print(f"  Phase 1 done: J={J_sg:.8f}", flush=True)
    
    # Also try from parent directly
    if time.monotonic() < p1_deadline - 100:
        a_sg2 = p_norm.copy()
        a_sg2_result, J_sg2 = projected_smooth_grad_descent(
            a_sg2, min(p1_deadline, deadline - 500),
            max_iters=30000, beta_start=5.0, beta_end=300.0)
        update_best(a_sg2_result)
        print(f"  Phase 1b done: J={J_sg2:.8f}", flush=True)
    
    # === PHASE 2: LP mirror descent ===
    # Start from best of (parent, smooth gradient results)
    candidates_lp = [p_norm]
    if a_sg_result is not None:
        candidates_lp.append(a_sg_result)
    try:
        candidates_lp.append(a_sg2_result)
    except NameError:
        pass
    
    # Pick best candidate
    lp_start = min(candidates_lp, key=lambda x: compute_J(x))
    
    p2_deadline = t_start + 750
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (n={n_parent}, budget=400s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    a_lp, J_lp = lp_mirror_descent(lp_start, 3000, min(p2_deadline, deadline - 350),
                                    use_cd_every=40, lp_time_limit=20)
    update_best(a_lp)
    print(f"  Phase 2 done: J={J_lp:.8f}", flush=True)
    
    # === PHASE 3: Coordinate descent + 2D CD ===
    p3_deadline = min(t_start + 1000, deadline - 250)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Coordinate Descent (budget=250s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        a_cd1, J_cd1 = coordinate_descent_fast(best_seq, min(time.monotonic() + 80, p3_deadline), max_passes=20)
        update_best(a_cd1)
        
        if time.monotonic() < p3_deadline - 30:
            a_cd2, J_cd2 = coordinate_descent_fast(a_lp, min(time.monotonic() + 60, p3_deadline - 30), max_passes=15)
            update_best(a_cd2)
        
        # 2D CD
        if time.monotonic() < p3_deadline - 30 and best_seq is not None:
            a_2d, J_2d = coordinate_descent_2d(best_seq, min(time.monotonic() + 60, p3_deadline),
                                                max_passes=5, num_pairs=100)
            update_best(a_2d)
    
    print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)
    
    # === PHASE 4: Diverse starts + multi-length ===
    p4_deadline = min(t_start + 1300, deadline - 150)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Diverse Starts + Multi-Length (budget=300s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        n_best = best_seq.size
        best_p = best_seq / best_seq.sum()
        
        # Try different lengths
        target_lengths = sorted(set([
            int(n_best * 0.8), int(n_best * 1.2),
            384, 448, 512, 640, 768, 896
        ]))
        target_lengths = [L for L in target_lengths if 128 <= L <= 1536]
        
        for tgt_n in target_lengths:
            if time.monotonic() > p4_deadline:
                break
            if tgt_n == n_best:
                continue
            
            a_ext = resample_seq(best_p, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            
            phase_dl = min(time.monotonic() + 40, p4_deadline)
            a_r, J_r = lp_mirror_descent(a_ext, 500, phase_dl, use_cd_every=30, lp_time_limit=15)
            update_best(a_r)
            
            a_cd = coordinate_descent_fast(a_r, min(time.monotonic() + 10, p4_deadline), max_passes=5)
            update_best(a_cd)
        
        # Diverse initializations at best length
        if time.monotonic() < p4_deadline - 50:
            inits = generate_diverse_inits(n_best, rng)
            # Add perturbed best
            for scale in [0.05, 0.1, 0.2]:
                pert = np.maximum(best_p + rng.normal(0, scale, n_best), 0.01)
                inits.append(pert / pert.sum())
            
            # Sort by initial J
            init_scores = []
            for init in inits:
                j = compute_J(init)
                if np.isfinite(j) and j < best_score * 1.5:
                    init_scores.append((j, init))
            init_scores.sort(key=lambda x: x[0])
        
        for score_init, init in init_scores[:6]:
            if time.monotonic() > p4_deadline:
                break
            phase_dl = min(time.monotonic() + 25, p4_deadline)
            a_r, J_r = lp_mirror_descent(init, 300, phase_dl, use_cd_every=25, lp_time_limit=10)
            update_best(a_r)
            a_cd = coordinate_descent_fast(a_r, min(time.monotonic() + 5, p4_deadline), max_passes=3)
            update_best(a_cd)
    
    print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)
    
    # === PHASE 5: Final polishing ===
    p5_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polishing", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        # One more LP pass
        a_f, J_f = lp_mirror_descent(best_seq, 200, min(time.monotonic() + 60, p5_deadline),
                                      use_cd_every=20, lp_time_limit=15)
        update_best(a_f)
        
        # Final CD
        a_cd_final, J_cd_final = coordinate_descent_fast(a_f, p5_deadline, max_passes=15)
        update_best(a_cd_final)
        
        # Quick smooth gradient polish
        if time.monotonic() < p5_deadline - 10:
            a_polish = a_cd_final.copy()
            a_polish_r, J_polish = projected_smooth_grad_descent(
                a_polish, p5_deadline, max_iters=5000, beta_start=100.0, beta_end=300.0)
            update_best(a_polish_r)
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128.0
        update_best(best_seq)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.10f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_seq]
