def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

        pass

        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g)), res
    except Exception:
        pass
    return None, 0.0, None

def lp_constraint_generation(a, M, relax=1.0, max_rounds=3, max_cons=300):
    """LP with constraint generation."""
    n = len(a)
    m = 2 * n - 1
    # Initialize with top peaks of conv(a,a) as proxy for active constraints
    conv_a = compute_conv(a)
    top_idx = np.argsort(conv_a)[::-1][:min(max_cons, m)]
    active = list(top_idx)
    
    g, S = None, 0.0
    for round_num in range(max_rounds):
        g, S, res = solve_lp_subset(a, M, active, relax)
        if g is None:
            break
        
        # Check violations
        conv_g = convolve_ab(a, g)
        violated = np.where(conv_g > M * relax + 1e-8)[0]
        
        if len(violated) == 0:
            break
        
        # Add most violated
        if len(active) < m:
            violations = conv_g[violated]
            worst = violated[np.argsort(violations)[::-1]][:30]
            for w in worst:
                if w not in active:
                    active.append(int(w))
            active = active[:max_cons * 3]
    
    return g, S

def lp_alternation(a, time_budget=100.0, max_iters=5000):
    """LP alternation with constraint generation and momentum."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    mu = 0.7  # momentum
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Solve LP with constraint generation
        g, S = lp_constraint_generation(a, M, relax=1.0)
        
        if g is not None and S > 1.0 + 1e-6:
            g_norm = g / S
            # Momentum step
            v = mu * v + (1 - mu) * (g_norm - a)
            
            # Try multiple step sizes
            improved = False
            for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2]:
                a_try = a + t * v
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            
            if not improved:
                # Try without momentum
                for t in [1.0, 0.7, 0.5, 0.3]:
                    a_try = (1 - t) * a + t * g_norm
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        v = np.zeros(n)
                        improved = True
                        break
            
            if improved:
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 15:
            # Perturb to escape
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            scale = rng.uniform(0.03, min(0.1, 0.02 + stall * 0.005))
            a = a * np.exp(rng.normal(0, scale, n))
            a = normalize(a)
            v = np.zeros(n)
            stall = 0
    
    return best_a, best_obj
