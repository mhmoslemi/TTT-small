from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

def solve_lp_active(a, M, active_idx, max_rounds=5):
    """Solver LP with adaptive active set (cutting plane method)."""
    n = len(a)
    m = 2*n - 1
    c = -np.ones(n)
    
    # Start with initial active set
    constraints = list(active_idx)
    
    for round_ in range(max_rounds):
        b_ub = np.full(len(constraints), M)
        # Build constraint matrix
        A_ub = np.zeros((len(constraints), n))
        for i, k in enumerate(constraints):
            # conv(a, g)[k] = sum_j a[j] * g[k-j] for valid j
            # = sum_j A[k,j] * g[j]
            # A[k,j] = a[j] if 0 <= k-j < n, else 0
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            A_ub[i, lo:hi+1] = a[lo:hi+1]
        
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.status != 0:
                return None, 0.0, constraints
            g = np.maximum(res.x, 0)

            return None, 0.0, constraints

        conv_ag = compute_conv_ag(a, g)  # Need helper for conv(a,g)

        violations = np.setdiff1d(violations, constraints)

            return g, float(np.sum(g)), constraints
        
        # Add most violated constraints (up to 20 per round)
        most_violated = violations[np.argsort(conv_ag[violations])[::-1][:20]]
        constraints.extend(most_violated.tolist())
    
    return None, 0.0, constraints

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv_sq(a):
    """Compute a*a (linear auto-convolution) via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_conv_ag(a, g):
    """Compute a*g (linear cross-convolution) via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    fg = rfft(g, size)
    conv = irfft(fa * fg, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv_sq(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

    """Build the constraint matrix A where conv(a,g)[k] = sum_j A[k,j]*g[j].
    If indices given, only build those rows."""
    n = len(a)
    m = 2 * n - 1
    if indices is None:
        indices = range(m)
    indices = list(indices)
    A = np.zeros((len(indices), n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[i, lo:hi+1] = a[lo:hi+1]
    return A, indices

        conv = compute_conv_sq(a)
        M = float(np.max(conv))

        g, S, res = solve_lp_full(a, M, relax=relax)

        if g is None or S <= 1.0 + 1e-8:
            stall += 1
            if stall > 20:
                # Gradient-based perturbation
                a_pert = gradient_shake(a, conv, M, strength=0.03)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    a = a_pert
                else:
                    rng = np.random.default_rng(int(time.monotonic() * 1000))
                    a = a * np.exp(rng.normal(0, 0.04, n))
                    a = np.maximum(a, 0)
                    s2 = np.sum(a)
                    if s2 > 1e-10:
                        a = a / s2
                        if compute_obj(a) < best_obj:
                            best_obj = compute_obj(a)
                            best_a = a.copy()
                    else:
                        a = best_a.copy()
                relax = min(relax + 0.1, 1.5)
                stall = 0
            continue

        # Line search
        improved = False
        best_t_obj = best_obj
        best_t_a = None
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_try = (1.0 - t) * a + t * g / S
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj - 1e-12:
                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_a is not None:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = a_try.copy() if False else best_t_a
            stall = 0
            relax = max(1.0, relax - 0.02)
            if verbose:
                print(f"    LP it: obj={best_obj:.8f}, S={S:.4f}")
        else:
            stall += 1
            if stall > 15:
                a = a * np.exp(np.random.default_rng().normal(0, 0.03, n))
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2
                relax = min(relax + 0.1, 1.4)
                stall = 0

    return best_a


def gradient_shake(a, conv, M, strength=0.03):
    """Perturb a to reduce peak contributions using gradient information."""
    n = len(a)
    # Find peak positions
    peak_threshold = M * 0.95
    peak_indices = np.where(conv >= peak_threshold)[0]

    # Compute subgradient of max(conv) w.r.t. a
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            partner = k - j
            if 0 <= partner < n:
                grad[j] += a[partner]

    # Normalize gradient
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-10:
        grad = grad / gnorm

    # Apply perturbation: subtract from peak contributors, add to others
    a_new = a - strength * grad
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    a_new = a_new / s
    return a_new


def spectral_flat(a, iterations=20, time_budget=10.0):
    """Try to flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    for _ in range(iterations):
        if time.monotonic() - start > time_budget:
            break

        fa = rfft(a, n)
        mag = np.abs(fa)
        phase = np.angle(fa)

        # Soft-clip high magnitude components
        threshold = np.percentile(mag, rng.uniform(80, 95))
        mag_new = np.where(mag > threshold,
                           threshold + (mag - threshold) * 0.3,
                           mag)
        # Preserve DC component
        mag_new[0] = mag[0]

        fa_new = mag_new * np.exp(1j * phase)
        a_new = irfft(fa_new, n)
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new

    return best_a


def two_level_init(n, seed=42):
    """Create a two-level sequence with alternating pattern."""
    rng = np.random.default_rng(seed)
    period = rng.integers(3, 15)
    amp = rng.uniform(0.1, 0.4)
    phase = rng.uniform(0, period)
    k = np.arange(n)
    # Create periodic +-1 pattern
    pattern = np.sign(np.sin(2 * np.pi * (k + phase) / period))
    a = 1.0 + amp * pattern
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def random_structured_init(n, seed=42):
    """Generate a diverse random structured initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)

    if choice == 0:
        f1 = rng.uniform(0.5, 3.0)
        f2 = rng.uniform(3.0, 10.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 1:
        # Block structure
        nblocks = rng.integers(3, 10)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.4, 1.4)
    elif choice == 2:
        # Exponentially decaying blocks
        output = np.zeros(n)
        pos = 0
        while pos < n:
            length = rng.integers(5, 30)
            level = rng.uniform(0.3, 1.5)
            output[pos:pos + length] = level
            pos += length
        a = output
    elif choice == 3:
        # Periodic with random duty cycle
        period = rng.integers(4, 20)
        duty = rng.uniform(0.3, 0.7)
        a = np.ones(n)
        for i in range(n):
            frac = (i % period) / period
            if frac < duty:
                a[i] = rng.uniform(1.2, 2.0)
            else:
                a[i] = rng.uniform(0.2, 0.6)
    elif choice == 4:
        # Smoothed random
        a = rng.uniform(0.3, 1.5, n)
        kernel_size = rng.integers(3, 15) | 1
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    else:
        a = rng.uniform(0.5, 1.5, n)

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.01)


def aggressive_polish(a, time_budget=100.0):
    """Intensive LP polishing with adaptive strategies."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget - 1:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        conv = compute_conv_sq(a)
        M = float(np.max(conv))

        # Try full LP
        g, S, _ = solve_lp_full(a, M)

        if g is not None and S > 1.0 + 1e-8:
            # Line search with finer steps
            improved = False
            for t in [1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
                a_try = (1.0 - t) * a + t * g / S
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
            if improved:
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        # When stuck, try different perturbations
        if stall > 10:
            pert_type = stall % 4
            if pert_type == 0:
                # Gradient shake
                a_new = gradient_shake(a, conv, M, strength=rng.uniform(0.02, 0.06))
            elif pert_type == 1:
                # Random multiplicative
                a_new = a * np.exp(rng.normal(0, rng.uniform(0.02, 0.06), n))
                a_new = np.maximum(a_new, 0)
                s2 = np.sum(a_new)
                if s2 > 1e-10:
                    a_new = a_new / s2
            elif pert_type == 2:
                # Targeted reduction at peak contributors
                a_new = a.copy()
                peak_k = int(np.argmax(conv))
                lo = max(0, peak_k - n + 1)
                hi = min(peak_k, n - 1)
                for j in range(lo, hi + 1):
                    partner = peak_k - j
                    if 0 <= partner < n:
                        a_new[j] *= rng.uniform(0.85, 0.98)
                s2 = np.sum(a_new)
                if s2 > 1e-10:
                    a_new = a_new / s2
            else:
                # Spectral nudge
                a_new = spectral_flat(a, iterations=5, time_budget=min(2.0, remaining * 0.1))

            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
            stall = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternation with gradient perturbation, spectral shaping,
    two-level constructions, and multi-length exploration.
    """
    start_time = time.monotonic()

    best_seq = None
    best_score = np.inf

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and np.sum(parent) > 0.01:
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
    except Exception:
        pass

    # Phase 1: Intensive LP polish on parent
    if best_seq is not None:
        n_par = len(best_seq)
        print(f"\n=== Phase 1: LP Polish (n={n_par}) ===")
        alloc1 = min(250.0, (deadline - time.monotonic()) * 0.2)
        a_pol, obj_pol = aggressive_polish(best_seq, time_budget=alloc1)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  After polish: {best_score:.8f}")
        else:
            print(f"  After polish: {obj_pol:.8f}")

    # Phase 2: Multi-length exploration
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    if best_seq is not None:
        np_ = len(best_seq)

            np_ + 200,

            np_ + 500,
            np_ + 800,
        ]))
    else:
        n_values = [400, 600, 800, 1000, 1200, 1500]

        remaining = deadline - time.monotonic()
        if remaining < 60:

        for start_idx in range(3):

                a_start = resample_sequence(best_seq, n)
            elif start_idx == 1:
                a_start = two_level_init(n, seed + n * 31 + start_idx)
            else:
                a_start = random_structured_init(n, seed + n * 97 + start_idx * 13)

            # LP alternation
            a_opt = lp_alternate(a_start, time_budget=sub_alloc * 0.7)
            obj_opt = compute_obj(a_opt)

            # Gradient shake + LP
            if time.monotonic() < deadline - 100:
                conv = compute_conv_sq(a_opt)
                M = float(np.max(conv))
                a_shake = gradient_shake(a_opt, conv, M, strength=0.04)
                a_shake = lp_alternate(a_shake, time_budget=sub_alloc * 0.2)
                obj_shake = compute_obj(a_shake)
                if obj_shake < obj_opt:
                    a_opt = a_shake
                    obj_opt = obj_shake

            # Spectral flat
            if time.monotonic() < deadline - 80:
                a_spec = spectral_flat(a_opt, iterations=15,
                                       time_budget=min(5.0, sub_alloc * 0.15))
                obj_spec = compute_obj(a_spec)
                if obj_spec < obj_opt:
                    a_opt = a_spec
                    obj_opt = obj_spec

            print(f"    start={start_idx}: obj={obj_opt:.8f}")

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"    ** NEW BEST: n={n}, obj={best_score:.8f} **")

    # Phase 3: Final intensive polish on global best
    print(f"\n=== Phase 3: Final Polish ===")
    print(f"  Best before: {best_score:.8f}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  Polishing n={n_best}, remaining={remaining:.0f}s")

        # Round 1: Aggressive LP
        alloc_r1 = min(remaining * 0.35, 200.0)
        a_r1, obj_r1 = aggressive_polish(best_seq, time_budget=alloc_r1)
        if obj_r1 < best_score:
            best_score = obj_r1
            best_seq = a_r1.copy()
            print(f"  R1 (LP): {best_score:.8f}")

            obj_spec = compute_obj(a_spec)

                best_seq = a_spec.copy()
            a_r3, obj_r3 = aggressive_polish(best_seq, time_budget=min(remaining * 0.2, 60.0))

            n_cur = len(best_seq)
            for dn in [-50, 50, -100, 100]:
                if time.monotonic() > deadline - 30:
                    break
                n_try = n_cur + dn
                if n_try < 100:

                a_try = resample_sequence(best_seq, n_try)
                a_try, obj_try = aggressive_polish(a_try, time_budget=min(remaining * 0.15, 40.0))
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    print(f"  R4 (n={n_try}): {best_score:.8f}")

    # Phase 4: Final micro-polish
    if best_seq is not None and time.monotonic() < deadline - 5:

        if remaining > 10:

    # Final validation

    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

        best_score = val
    elif val == np.inf:

        result = fallback
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
