def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_J_normalized(a):
    """Compute J assuming sum(a)=1: 2*n*max(conv(a,a))."""
    return float(2 * len(a) * fast_conv(a).max())


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_max_conv(a, b):
    """
    Compute subgradient of max_k (a*a)[k] at point a.
    Uses average of gradients over active (near-max) k values.
    """
    n = len(a)
    M = b.max()
    # Active set: all k where b[k] >= M * 0.999 (or within small epsilon)
    active = np.where(b >= M * 0.999 - 1e-12)[0]
    
    sg = np.zeros(n)
    for k in active:
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                if j == kj:  # k = 2j
                    sg[j] += 2 * a[j]
                else:
                    sg[j] += 2 * a[kj]
    sg /= len(active)
    return sg


def gradient_descent_momentum(a, deadline, max_iters=50000, lr_init=0.01, momentum=0.9):
    """
    Subgradient descent with Nesterov momentum for minimizing max(conv(a,a))
    subject to a >= 0, sum(a) = 1.
    """
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    v = np.zeros(n)  # momentum
    lr = lr_init
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 0.01:
            y = a.copy()
            sy = s
        y = y / sy
        
        b = fast_conv(y)
        M = b.max()
        
        # Subgradient
        sg = subgradient_max_conv(y, b)
        
        # Update
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        a_new = project_simplex(a_new)
        
        J_new = compute_J(a_new)
        
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        s = a.sum()
        
        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.8
            no_improve = 0
        elif no_improve > 200:
            lr *= 0.5
            v *= 0.5
            no_improve = 0
            if lr < 1e-6:
                break
    
    return best_a

        return a

                    best_v = cand
