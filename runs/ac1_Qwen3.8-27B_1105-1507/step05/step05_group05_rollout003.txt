def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def get_max_and_active(a, tol=1e-4):
    """Return max conv value and indices of near-max lags."""
    b = fast_conv(a)
    M = b.max()
    active = np.where(b >= M - tol * M)[0]
    return M, active, b

def coordinate_update(a, i):
    """Find optimal a[i] given all other coordinates fixed."""
    n = len(a)
    x_cur = a[i]
    S = a.sum()
    b = fast_conv(a)
    
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    affected_k = np.arange(k_lo, k_hi + 1)
    j_vals = affected_k - i
    valid = (j_vals >= 0) & (j_vals < n)
    is_quad = (affected_k == 2 * i)
    
    c = b[affected_k].copy()
    slopes = np.zeros(len(affected_k))
    
    linear_mask = valid & ~is_quad
    if linear_mask.any():
        c[linear_mask] -= 2 * x_cur * a[j_vals[linear_mask]]
        slopes[linear_mask] = 2 * a[j_vals[linear_mask]]
    
    if is_quad.any():
        c[is_quad] -= x_cur * x_cur
    
    has_quad = is_quad.any()
    quad_c = c[is_quad].item() if has_quad else 0.0
    
    def obj(x):
        ns = S - x_cur + x
        if ns <= 1e-10:
            return 1e10
        m = base_max
        vm = (c + slopes * x).max()
        if vm > m:
            m = vm
        if has_quad:
            qv = quad_c + x * x
            if qv > m:
                m = qv
        return m / (ns * ns)
    
    # Golden section
    lo = 0.0
    hi = max(x_cur * 3.0, 0.1)
    if obj(hi) < obj(hi * 0.5) and hi < 50.0:
        hi *= 2.0
    
    gr = (np.sqrt(5) - 1) / 2
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1 = obj(a1)
    f2 = obj(a2)
    
    for _ in range(35):
        if f1 < f2:
            hi = a2; a2 = a1; f2 = f1
            a1 = hi - gr * (hi - lo); f1 = obj(a1)
        else:
            lo = a1; a1 = a2; f1 = f2
            a2 = lo + gr * (hi - lo); f2 = obj(a2)
    
    x_opt = (lo + hi) / 2
    if obj(0) < obj(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)

def coord_descent(a, max_sweeps, deadline, shuffle=True, seed=0, momentum=False):
    """Coordinate descent with optional momentum."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        if shuffle:
            rng.shuffle(order)
        
        improved = False
        for i in order:
            x_opt = coordinate_update(a, i)
            if momentum and x_opt > 0:
                x_opt = 0.7 * a[i] + 0.3 * x_opt
            if abs(x_opt - a[i]) > 1e-10:
                a[i] = x_opt
                improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not improved:
            no_improve += 10
        if no_improve > 20:
            break
    
    return best_a

def smooth_max_grad(a, tau):
    """Gradient of (1/tau)*log(sum(exp(tau*conv_k)))."""
    n = len(a)
    b = fast_conv(a)
    bmax = b.max()
    eb = np.exp(tau * (b - bmax))
    p = eb / eb.sum()
    
    # grad[i] = 2 * sum_k p[k] * a[k-i] (for valid k-i)
    # This is a cross-correlation: grad = 2 * corr(p, a)
    size = 1
    while size < 4 * n:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(P * np.conj(A), size)
    return 2.0 * corr[:n]

def adam_smooth(a, max_iters, deadline, tau_start=10.0, tau_end=200.0, lr=0.05):
    """Adam on smooth max."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        lr_t = lr / (1 + 0.005 * it)
        
        g = smooth_max_grad(a, tau)
        g = g - g.mean()  # project to sum-zero
        
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** (it + 1))
        vh = v / (1 - b2 ** (it + 1))
        
        a_new = a - lr_t * mh / (np.sqrt(vh) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
    
    return best_a

def perturb_fourier(a, rng, sigma=0.1):
    """Add Fourier-based smooth perturbation."""
    n = len(a)
    # Random smooth perturbation via low-frequency Fourier components
    kmax = max(2, n // 10)
    freqs = rng.integers(1, kmax + 1, size=min(5, n // 4))
    phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
    amps = rng.uniform(0.3, 1.0, size=len(freqs)) * sigma
    t = np.arange(n) / n
    noise = np.zeros(n)
    for f, p, am in zip(freqs, phases, amps):
        noise += am * np.cos(2 * np.pi * f * t + p)
    result = a + noise
    result = np.maximum(result, 1e-6)
    return result / result.sum()

def perturb_random(a, rng, sigma=0.05):
    """Add random noise."""
    result = a + rng.normal(0, sigma, len(a))
    result = np.maximum(result, 1e-6)
    return result / result.sum()

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 1e-6)
