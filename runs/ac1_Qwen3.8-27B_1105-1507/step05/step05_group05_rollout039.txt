def proj_simplex(v):
    """Euclidean projection onto the simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

def coordinate_update_fast(a, b, i):
    """Fast coordinate update using precomputed convolution."""
    n = len(a)
    x = a[i]
    S = a.sum()
    
    k_lo = i
    k_hi = min(2*n-2, i+n-1)
    affected = np.arange(k_lo, k_hi+1)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2*n-2:
        mask = np.ones(2*n-1, dtype=bool)
        mask[k_lo:k_hi+1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    # For each affected k, express b[k] as f(x_new)
    # b[k] = sum_j a[j]*a[k-j]
    # When a[i] changes from x to x_new:
    #   b[k] changes by: (x_new - x) * (a[k-i] + a[i-k] if k-i != i) ... 
    #   Actually: the terms involving a[i] in b[k] are:
    #     a[i]*a[k-i] (when the first index is i)
    #     a[k-i]*a[i] (when the second index is i, i.e., j = k-i)
    #   If k-i == i (i.e., k = 2i): only one term a[i]*a[i] = x^2
    #   If k-i != i: two terms, each x * a[k-i], total 2*x*a[k-i]
    #   If k-i out of range [0,n): no contribution from a[i]
    
    # So b_new[k] = b[k] - old_contribution + new_contribution
    # where old_contribution = x^2 if k==2i, 2*x*a[k-i] if 0<=k-i<n and k!=2i, 0 otherwise
    # and new_contribution = x_new^2 if k==2i, 2*x_new*a[k-i] if 0<=k-i<n and k!=2i, 0 otherwise
    
    # Precompute coefficients
    c = np.zeros(len(affected))  # constant term (b[k] minus old contribution)
    s = np.zeros(len(affected))  # linear coefficient in x_new
    q = np.zeros(len(affected))  # quadratic coefficient (1 if k==2i, else 0)
    
    for idx, k in enumerate(affected):
        ki = k - i  # = k - i
        if 0 <= ki < n:
            if ki == i:  # k = 2i
                c[idx] = b[k] - x*x
                s[idx] = 0.0
                q[idx] = 1.0
            else:
                c[idx] = b[k] - 2*x*a[ki]
                s[idx] = 2*a[ki]
                q[idx] = 0.0
        else:
            c[idx] = b[k]
            s[idx] = 0.0
            q[idx] = 0.0
    
    # Objective: max(base_max, max_k (c_k + s_k*x + q_k*x^2)) / (S - x + x_new)^2
    # Wait, we're NOT normalizing within the coordinate update (that's done after the full sweep)
    # So the objective is just: max(base_max, max_k (c_k + s_k*x_new + q_k*x_new^2))
    
    def obj(x_new):
        if x_new < 0:
            return 1e10
        vals = c + s * x_new + q * x_new * x_new
        m = base_max
        if vals.max() > m:
            m = vals.max()
        return m
    
    # Golden section search
    lo = 0.0
    hi = max(x * 2, np.sqrt(b.max()) + 1, 0.1)
    # ... golden section ...
    
    x_opt = (lo + hi) / 2
    if obj(0) < obj(x_opt):
        x_opt = 0
    
    return x_opt
```

These are the same term (both equal x*a[k-i]) UNLESS i = k-i, in which case it's a single term x^2.

So:
- If i = k-i (k = 2i): contribution is x^2
- If i ≠ k-i and 0 <= k-i < n: contribution is 2*x*a[k-i]
- Otherwise: contribution is 0

When x changes to x_new:
- If k = 2i: change = x_new^2 - x^2
- If 0 <= k-i < n and k ≠ 2i: change = 2*x_new*a[k-i] - 2*x*a[k-i] = 2*(x_new-x)*a[k-i]
- Otherwise: change = 0

So the update is:
b[k] += x_new^2 - x^2 if k == 2*i
b[k] += 2*(x_new-x)*a[k-i] if 0 <= k-i < n and k != 2*i

This is clean and efficient!

Now, for the golden section search, the objective is:
obj(x_new) = max(base_max, max_{k in affected} [b[k] + delta_k(x_new)])

where delta_k(x_new) = (x_new^2 - x^2) if k==2i, 2*(x_new-x)*a[k-i] if 0<=k-i<n and k≠2i, 0 otherwise.

We can precompute:
- For k in affected, the "base" b[k] (before update)
- The coefficient of x_new in delta_k:
  - If k == 2i: delta = x_new^2 - x^2, so it's quadratic with coeff 1, linear 0, constant -x^2
  - If 0 <= k-i < n, k ≠ 2i: delta = 2*(x_new-x)*a[k-i] = 2*a[k-i]*x_new - 2*x*a[k-i], linear with slope 2*a[k-i], intercept -2*x*a[k-i]

So obj(x_new) = max(base_max, max_k [b[k] + coeff_k * x_new + quad_k * x_new^2 + const_k])

This is a piecewise function (max of affine and one quadratic), and we can find its minimum using golden section search.

"""
Multi-phase optimization for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast coordinate descent with incremental convolution updates, periodic perturbation
for escaping local minima, LP mirror descent for large jumps, smooth gradient descent,
and multi-length exploration. Key innovation: vectorized coordinate updates with O(n)
per-coordinate cost via precomputed affine/quadratic decompositions of the convolution.
"""

import numpy as np
from scipy.optimize import linprog, minimize as sp_minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT with fixed size."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def _coord_obj_and_apply(a, b, i):
    """
    Find optimal a[i] given current a and b=conv(a,a).
    Returns (x_opt, bases, slopes, quads, ks, base_max) for potential apply.
    """
    n = len(a)
    x = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    if k_lo > 0 or k_hi < 2 * n - 2:
        base_max = 0.0
        if k_lo > 0:
            base_max = max(base_max, float(b[:k_lo].max()))
        if k_hi < 2 * n - 2:
            base_max = max(base_max, float(b[k_hi + 1:].max()))
    else:
        base_max = 0.0

    ks = np.arange(k_lo, k_hi + 1)
    m = len(ks)
    bases = np.empty(m)
    slopes = np.empty(m)
    quads = np.empty(m)

    for idx in range(m):
        k = ks[idx]
        ki = k - i
        if k == 2 * i:
            bases[idx] = b[k] - x * x
            slopes[idx] = 0.0
            quads[idx] = 1.0
        elif 0 <= ki < n:
            bases[idx] = b[k] - 2.0 * x * a[ki]
            slopes[idx] = 2.0 * a[ki]
            quads[idx] = 0.0
        else:
            bases[idx] = b[k]
            slopes[idx] = 0.0
            quads[idx] = 0.0

    def obj(x_new):
        vals = bases + slopes * x_new + quads * x_new * x_new
        v = vals.max()
        return v if v > base_max else base_max

    lo = 0.0
    hi = max(x * 3.0, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.05)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = obj(c_pt)
    fd = obj(d_pt)

    for _ in range(40):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = obj(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = obj(d_pt)

    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    x_opt = max(0.0, x_opt)

    return x_opt, bases, slopes, quads, ks, base_max


def _apply_coord_update(a, b, i, x_opt, bases, slopes, quads, ks):
    """Apply a coordinate update to a and b in place."""
    for idx in range(len(ks)):
        k = int(ks[idx])
        b[k] = bases[idx] + slopes[idx] * x_opt + quads[idx] * x_opt * x_opt
    a[i] = x_opt


def run_cd(a, max_sweeps, deadline, seed=0, perturb_every=30, perturb_frac=0.15, perturb_sigma=0.08):

    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    b = fast_conv(a)
    best_a = a.copy()
    best_M = float(b.max())
    rng = np.random.default_rng(seed)
    no_improve = 0
    sweep_count = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            x_new, bases, slopes, quads, ks, _ = _coord_obj_and_apply(a, b, i)
            if abs(x_new - a[i]) > 1e-14:
                _apply_coord_update(a, b, i, x_new, bases, slopes, quads, ks)
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        b = fast_conv(a)
        M = float(b.max())

        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

                s = a.sum()

                    b = fast_conv(a)
                no_improve = 0

def smooth_max_grad(a, tau):
    """Gradient of smooth max (log-sum-exp) of convolution."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * pk * a[k - idx]

    return grad


def run_smooth_grad(a, max_iters, deadline, tau_start=5.0, tau_end=100.0, lr=0.03):
    """Adam optimizer on smooth max with simplex projection."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.6)
        lr_t = lr / (1.0 + 0.005 * it)

        grad = smooth_max_grad(a, tau)
        grad = grad - grad.mean()

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = float(fast_conv(a).max())
        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a


def _build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: rows=conv positions, cols=a indices."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = _build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a, max_iters, deadline):
    """LP mirror descent (Frank-Wolfe style)."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        improved = False
        base_t = max(0.05, 0.5 * (1.0 - 0.3 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01]:
            a_trial = np.maximum((1.0 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = float(fast_conv(a_trial).max())
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 12:
                break
        else:
            no_improve = 0

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.04, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.12, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    for freq in [1, 2, 3, 5, 7]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)), 0.01))

    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(0.3 + 0.7 * np.abs(2 * t - 1))

    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Sine-based
    inits.append(np.maximum(1.0 + 0.2 * np.sin(2 * np.pi * 3 * t) + 0.1 * np.sin(2 * np.pi * 7 * t), 0.01))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Fast CD from parent with perturbation
    Phase 2: Multi-length exploration with CD
    Phase 3: LP mirror descent
    Phase 4: Smooth gradient + CD refinement
    Phase 5: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    # === PHASE 1: Fast CD from parent ===
    p1_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Fast CD from parent (~350s)", flush=True)
    print(f"{'='*60}", flush=True)

    n_parent = parent.size
    a1 = run_cd(parent, 10000, min(time.monotonic() + 350, deadline - 400), seed=seed)
    tracker.update(a1)

    # Try a perturbed version
    a1b = np.maximum(a1 + rng.normal(0, 0.03, a1.size), 0.01)
    a1b = a1b / a1b.sum()
    a1b = run_cd(a1b, 5000, min(time.monotonic() + 100, deadline - 350), seed=seed + 1)
    tracker.update(a1b)

    tracker.report("Phase 1 complete")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Multi-length exploration ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: Multi-length exploration (~350s)", flush=True)
    print(f"{'='*60}", flush=True)

    best_ref = tracker.best if tracker.best is not None else parent
    lengths_to_try = [384, 512, 640, 768, 1024]
    # Remove parent length (already done)
    lengths_to_try = [n for n in lengths_to_try if n != n_parent]
    # Add parent length with different init
    lengths_to_try.append(n_parent)

    time_per_len = 350.0 / len(lengths_to_try)
    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 350:
            break
        phase_dl = min(time.monotonic() + time_per_len, deadline - 300)
        if time.monotonic() >= phase_dl:
            continue

        t0 = time.monotonic()
        inits = diverse_inits(n_try, rng, best_ref)
        best_for_n = None
        best_M_for_n = float('inf')

        for idx, a0 in enumerate(inits[:5]):
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            max_sw = max(10, int(2000.0 / max(1, n_try // 128)))
            a_opt = run_cd(a0, max_sw, phase_dl, seed=seed + idx * 13 + n_try)
            M_val = float(fast_conv(a_opt).max())
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n_try:4d}: 2n*max_b={2*n_try*best_M_for_n:.6f} J={J_val:.6f} ({elapsed:.1f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("Phase 2 complete")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: LP Mirror Descent ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: LP Mirror Descent (~250s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 200:
        n_best = tracker.best.size
        est_lp_time = 0.003 * (n_best / 128.0) ** 2
        max_iters = max(15, min(200, int(200.0 / max(0.01, est_lp_time))))
        print(f"  n={n_best}, est LP time={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)

        a_lp = run_lp_mirror(tracker.best, max_iters, min(time.monotonic() + 200, deadline - 150))
        tracker.update(a_lp)

        if time.monotonic() < deadline - 100:
            a_lp2 = run_lp_mirror(a_lp, max(10, max_iters // 2), min(time.monotonic() + 80, deadline - 80))
            tracker.update(a_lp2)

    tracker.report("Phase 3 complete")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Smooth gradient + CD ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Smooth gradient + CD (~200s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 100:
        a_sm = run_smooth_grad(tracker.best, 1500, min(time.monotonic() + 100, deadline - 80),
                               tau_start=5.0, tau_end=80.0, lr=0.04)
        tracker.update(a_sm)

        a_cd = run_cd(tracker.best, 5000, min(time.monotonic() + 80, deadline - 50), seed=seed + 999)
        tracker.update(a_cd)

    tracker.report("Phase 4 complete")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Final polish ===
    p5_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print("PHASE 5: Final polish (remaining time)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 5:
        # Extended CD on best
        a_fin = run_cd(tracker.best, 10000, deadline - 3, seed=seed + 77777,
                       perturb_every=50, perturb_frac=0.1, perturb_sigma=0.05)
        tracker.update(a_fin)

    # Try doubling length if time permits
    if tracker.best is not None and time.monotonic() < deadline - 30:
        new_n = tracker.best.size * 2
        if new_n <= 2048:
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext = run_cd(a_ext, 3000, deadline - 5, seed=seed + 55555)
            tracker.update(a_ext)

    tracker.report("Phase 5 complete")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
