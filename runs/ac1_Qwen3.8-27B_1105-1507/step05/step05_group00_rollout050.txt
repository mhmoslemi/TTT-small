import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=30, x0=None):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    options = {'presolve': True, 'time_limit': time_limit,
               'dual_feasibility_tolerance': 1e-10,
               'primal_feasibility_tolerance': 1e-10}
    if x0 is not None:
        options['presolve'] = True
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options=options, x0=x0)
            if res.status == 0:
                g = np.maximum(res.x, 0)
                s = float(np.sum(g))
                if s > 1e-10:
                    return g / s, g
        except Exception:
            pass
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options=options)
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, g
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=30):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',

        if res.status == 0:

    except Exception:

    return None, None


def golden_section_line_search(a, g, obj_fn, f0, t_range=(0.01, 1.5), tol=0.02, max_iter=15):
    """Golden section line search for t in (a*g blend). Returns best t and point."""
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    invphi2 = (3.0 - np.sqrt(5.0)) / 2.0
    a_t, b_t = t_range
    
    def blend(t):
        x = (1.0 - t) * a + t * g
        x = np.maximum(x, 0)
        s = np.sum(x)
        if s < 1e-10:
            return np.inf, x
        return obj_fn(x / s), x / s
    
    c_t = b_t - invphi * (b_t - a_t)
    d_t = a_t + invphi * (b_t - a_t)
    fc, _ = blend(c_t)
    fd, _ = blend(d_t)
    
    best_t = 0.5
    best_f = f0
    best_x = a.copy()
    
    for _ in range(max_iter):
        if b_t - a_t < tol:
            break
        if fc < fd:
            b_t = d_t
            d_t = c_t
            fd = fc
            c_t = b_t - invphi * (b_t - a_t)
            fc, _ = blend(c_t)
        else:
            a_t = c_t
            c_t = d_t
            fc = fd
            d_t = a_t + invphi * (b_t - a_t)
            fd, _ = blend(d_t)
        
        # Track best
        if fc < best_f:
            best_f = fc
            best_t = c_t
        if fd < best_f:
            best_f = fd
            best_t = d_t
    
    mid_t = (a_t + b_t) / 2.0
    fmid, xmid = blend(mid_t)
    if fmid < best_f:
        best_f = fmid
        best_t = mid_t
        best_x = xmid
    else:
        best_x = (1.0 - best_t) * a + best_t * g
        best_x = np.maximum(best_x, 0)
        s = np.sum(best_x)
        if s > 1e-10:
            best_x = best_x / s
    
    return best_t, best_x, best_f

def smooth_objective_and_grad(a, tau):
    """Compute smooth objective F(a) = 2*n*logsumexp(conv/tau) / (sum a)^2 and its gradient."""
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    # Log-sum-exp for numerical stability
    c = conv / tau
    c_max = float(np.max(c))
    c_shifted = c - c_max
    exp_c = np.exp(c_shifted)
    log_sum = c_max + np.log(np.sum(exp_c))
    w = exp_c / np.sum(exp_c)
    
    # F = 2*n*tau*logsumexp / (sum_a)^2
    # Note: tau * logsumexp = tau * (max(conv)/tau + log(sum(exp((conv-max)/tau))))
    # = max(conv) + tau * log(sum(exp((conv-max)/tau)))
    F = 2.0 * n * tau * log_sum / (sum_a * sum_a)
    
    # Gradient of F w.r.t. a[j]:
    # d(conv[k])/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    # grad_F[j] = 2*n/(sum_a)^2 * sum_k w[k] * d(conv[k])/da[j] - 2*F/sum_a
    #           = 2*n/(sum_a)^2 * (term1[j] + term2[j]) - 4*n*tau*log_sum/(sum_a^3)
    
    # term1[j] = sum_k w[k] * a[k-j] (cross-correlation of w and a)
    # term2[j] = sum_k w[k] * a[j-k] (cross-correlation of w and a[::-1] shifted)
    
    # Compute using FFT:
    # term1: w has length 2n-1, a has length n
    # term1[j] = sum_{k} w[k] * a[k-j] for valid indices
    # This is the cross-correlation of w and a at lag j
    # = (w * a_flipped)[j + n - 1] where a_flipped = a[::-1]
    # Actually, let me use direct FFT convolution:
    # corr_wa[j] = sum_k w[k] * a[k-j]
    # Let's compute: conv(w, a_flipped) where a_flipped[k] = a[n-1-k]
    # conv(w, a_flipped)[m] = sum_k w[k] * a_flipped[m-k] = sum_k w[k] * a[n-1-(m-k)]
    # = sum_k w[k] * a[n-1-m+k]
    # We want sum_k w[k] * a[k-j], so set n-1-m+k = k-j => n-1-m = -j => m = n-1+j
    # So corr_wa[j] = conv(w, a_flipped)[n-1+j]
    
    a_flipped = a[::-1]
    size1 = 1
    while size1 < (2 * n - 1) + n - 1:
        size1 <<= 1
    fw = rfft(w, size1)
    fa_fl = rfft(a_flipped, size1)
    conv_wf = irfft(fw * fa_fl, size1)
    # term1[j] = conv_wf[n-1+j] for j = 0..n-1
    term1 = conv_wf[n - 1:2 * n - 1]
    
    # term2[j] = sum_k w[k] * a[j-k]
    # = sum_k w[k] * a[j-k], let m = j-k, k = j-m: sum_m w[j-m] * a[m]
    # This is (w_rev * a)[j] where w_rev[k] = w[2n-2-k]
    # Alternatively: term2[j] = sum_k w[k] * a[j-k]
    # = conv(w, a)[j] ??? No.
    # conv(w, a)[j] = sum_k w[k] * a[j-k] -- YES that's exactly it!
    # But w has length 2n-1 and a has length n, so conv has length 3n-2
    # We want term2[j] for j=0..n-1, which is conv(w,a)[j]
    
    size2 = 1
    while size2 < (2 * n - 1) + n - 1:
        size2 <<= 1
    fw2 = rfft(w, size2)
    fa2 = rfft(a, size2)
    conv_wa = irfft(fw2 * fa2, size2)
    term2 = conv_wa[:n]
    
    # Full gradient
    grad = (2.0 * n / (sum_a * sum_a)) * (term1 + term2)
    grad -= np.full(n, 4.0 * n * tau * log_sum / (sum_a ** 3))
    
    return F, grad


def smooth_lbfgs(a, time_budget=30.0, tau_start=5.0, tau_end=0.1, verbose=False):
    """Minimize smooth approximation using L-BFGS-B with temperature annealing."""
    n = n_orig = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Anneal tau from tau_start to tau_end
    n_phases = 5
    for phase in range(n_phases):
        if time.monotonic() - start > time_budget - 5:
            break
        frac = phase / max(n_phases - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        def obj_fn(x):
            val, _ = smooth_objective_and_grad(x, tau)
            return val
        
        def grad_fn(x):
            _, g = smooth_objective_and_grad(x, tau)
            return g
        
        # Run L-BFGS-B
        res = minimize(obj_fn, a, jac=grad_fn, method='L-BFGS-B',
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 50, 'ftol': 1e-10, 'gtol': 1e-6})
        
        if res.x is not None:
            a_new = np.maximum(res.x, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
                    if verbose:
                        print(f"        Smooth phase {phase} (tau={tau:.3f}): {best_obj:.6f}")
    
    return best_a

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

    print(f"Starting search: best={best_score:.8f}, n_parent={n_parent}")
    
    rng_master = np.random.default_rng(seed)
    
    # Phase 1: Intensive LP optimization of parent
    print(f"\n=== Phase 1: LP Optimization (n={n_parent}) ===")
    if best_seq is not None:
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
        
        time_budget_p1 = min(300.0, (deadline - start_time) * 0.25)
        a_opt = lp_alternate_improved(a, max_iters=100, time_budget=time_budget_p1, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Phase 1 LP result: {best_score:.8f}")
    
    # Phase 2: Smooth L-BFGS refinement
    print(f"\n=== Phase 2: Smooth L-BFGS (n={n_parent}) ===")
    if best_seq is not None:
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
        
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.15, 120.0)
        a_smooth = smooth_lbfgs(a, time_budget=alloc, tau_start=10.0, tau_end=0.2, verbose=True)
        obj = compute_obj(a_smooth)
        if obj < best_score:
            best_score = obj
            best_seq = a_smooth.copy()
            print(f"  Phase 2 Smooth result: {best_score:.8f}")
        else:
            print(f"  Phase 2 Smooth: no improvement ({obj:.6f} vs {best_score:.6f})")
    
    # Phase 3: LP again on the smooth result
    print(f"\n=== Phase 3: LP Refinement ===")
    if best_seq is not None:
        a = best_seq.copy()
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
        
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.15, 120.0)
        a_opt = lp_alternate_improved(a, max_iters=80, time_budget=alloc, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Phase 3 LP result: {best_score:.8f}")
    
    # Phase 4: Explore different n values
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    n_options = []
    base_n = len(best_seq) if best_seq is not None else 700
    for delta in [-100, -50, -25, 0, 25, 50, 100, 150, 200]:
        n_try = base_n + delta
        if 300 <= n_try <= 1200 and n_try not in n_options:
            n_options.append(n_try)
    n_options.sort()
    
    for n_try in n_options:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        if best_seq is None:
            a_try = np.ones(n_try)
        elif n_try == len(best_seq):
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)
        
        s = np.sum(a_try)
        if s > 1e-10:
            a_try = a_try / s
        
        # Quick optimization: LP + smooth
        alloc = min(remaining * 0.15, 60.0)
        
        a_lp = lp_alternate_improved(a_try, max_iters=30, time_budget=alloc * 0.5)
        obj_lp = compute_obj(a_lp)
        
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_sm = smooth_lbfgs(a_lp, time_budget=min(remaining * 0.1, 20.0))
            obj_sm = compute_obj(a_sm)
            if obj_sm < obj_lp:
                a_try = a_sm
                obj_lp = obj_sm
            else:
                a_try = a_lp
        
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_try.copy()
            print(f"  ** n={n_try}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_lp:.6f} (no improvement)")
    
    # Phase 5: Diversification and re-optimization
    print(f"\n=== Phase 5: Diversification ===")
    if best_seq is not None:
        for div_round in range(3):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            
            rng = np.random.default_rng(seed + div_round * 1000)
            candidates = []
            
            # Generate diverse candidates
            a1, a2 = diversify(best_seq, rng, strength=0.05 + 0.03 * div_round)
            candidates.append((a1, "mult_noise"))
            candidates.append((a2, "shift"))
            
            # Also try a completely random initialization with the same n
            n_cur = len(best_seq)
            a_rand = rng.dirichlet(np.ones(n_cur))
            candidates.append((a_rand, "random"))
            
            # Chirp-based initialization
            t = np.linspace(0, 1, n_cur)
            a_chirp = np.abs(np.exp(2j * np.pi * 0.3 * t ** 2)) + 0.1
            a_chirp = np.maximum(a_chirp, 0)
            a_chirp = a_chirp / np.sum(a_chirp)
            candidates.append((a_chirp, "chirp"))
            
            for a_cand, name in candidates:
                remaining = deadline - time.monotonic()
                if remaining < 15:
                    break
                
                s = np.sum(a_cand)
                if s < 1e-10:
                    continue
                a_cand = a_cand / s
                
                # Quick optimize
                alloc = min(remaining * 0.1, 25.0)
                a_opt = lp_alternate_improved(a_cand, max_iters=20, time_budget=alloc)
                obj = compute_obj(a_opt)
                
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  ** Div {name}: obj={best_score:.8f} **")
                else:
                    print(f"  Div {name}: {obj:.6f}")
    
    # Phase 6: Final intensive polish
    print(f"\n=== Phase 6: Final Polish ===")
    print(f"Best before final: {best_score:.8f}")
    
    if best_seq is not None:
        a_final = best_seq.copy()
        for polish_round in range(5):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            
            a_pol, obj_pol = polish_sequence(a_final, time_budget=min(remaining * 0.2, 40.0))
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_final = a_pol.copy()
                print(f"  Polish round {polish_round}: {best_score:.8f}")
            else:
                print(f"  Polish round {polish_round}: no improvement ({obj_pol:.6f})")
                # Try a small perturbation
                rng = np.random.default_rng(seed + 9000 + polish_round)
                a_pert = a_final * np.exp(rng.normal(0, 0.03, len(a_final)))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    a_pert = lp_alternate_improved(a_pert, max_iters=30, time_budget=min(remaining * 0.1, 20.0))
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_final = a_pert
                        print(f"  Pert round {polish_round}: {best_score:.8f}")
    
    # Phase 7: One more LP pass with extended time
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_f = len(best_seq)
        if n_f <= 900:
            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.4, 120.0)
            print(f"\n=== Phase 7: Final LP (n={n_f}, {alloc:.0f}s) ===")
            a_final = best_seq.copy()
            s = np.sum(a_final)
            if s > 1e-10:
                a_final = a_final / s
            a_opt = lp_alternate_improved(a_final, max_iters=200, time_budget=alloc, verbose=True)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Phase 7 result: {best_score:.8f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    
    return result
