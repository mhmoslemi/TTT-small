"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LSE-based mirror descent with adaptive temperature (primary optimizer)
2. Hybrid LP polishing for final precision
3. Multi-length, multi-start exploration with diverse initializations
4. Coordinate descent for local refinement
Key differentiator from parent: replaces slow LP-heavy main loop with fast
FFT-based mirror descent, then uses LP only for final polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lse_and_grad(a, t):
    """Compute LSE of conv(a,a) and its gradient. Returns (grad, b, lse_val)."""
    n = len(a)
    b = fast_conv(a)
    tb = t * b
    tb_max = tb.max()
    exp_tb = np.exp(tb - tb_max)
    p = exp_tb / exp_tb.sum()
    # grad[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    Pf = np.fft.fft(p, N)
    Af = np.fft.fft(a, N)
    xcorr = np.fft.ifft(Pf * np.conj(Af)).real[:n]
    grad = 2.0 * xcorr
    lse_val = (tb_max + np.log(exp_tb.sum())) / t
    return grad, b, lse_val


def mirror_descent_lse(a, deadline, t_start=3.0, t_end=300.0,
                       max_iters=10000, lr_scale=1.0, verbose=False):
    """Mirror descent on LSE objective with temperature annealing.
    Update: a_new[i] = a[i]*exp(-lr*grad[i]) / Z, then normalize to sum=1."""
    a = np.maximum(a.astype(np.float64).copy(), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    lr = lr_scale / (t_start * max(a.max(), 1e-6) + 1e-10)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        progress = min(1.0, it / max_iters)
        t = t_start * (t_end / t_start) ** progress
        # Adaptive lr: decrease as t increases
        lr = lr_scale / (t * max(a.max(), 1e-6) + 1e-10)

        grad, b, _ = lse_and_grad(a, t)

        # Mirror descent update (exponential weights)
        log_update = -lr * grad
        # Clip for numerical stability
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        s = a_new.sum()
        if s < 1e-10 or not np.isfinite(s):
            a_new = a.copy()
            s = a.sum()
        a_new = a_new / s

        # Line search: accept if J improves
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try half step
            a_half = (a + a_new) / 2.0
            s = a_half.sum()
            if s > 1e-10:
                a_half = a_half / s
                J_half = compute_J(a_half)
                if J_half < best_J:
                    best_J = J_half
                    best_a = a_half.copy()
                    a = a_half.copy()
                else:
                    # Halve learning rate
                    lr *= 0.5
            else:
                lr *= 0.5

        if verbose and it % 2000 == 0:
            print(f"      MD iter={it} t={t:.1f} J={best_J:.8f} lr={lr:.4f}", flush=True)

    return best_a, best_J


def build_lp_matrix_reduced(n, a, active_k):
    """Build sparse matrix for reduced LP constraints."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))


def lp_polish(a, deadline, max_iters=200, top_frac=0.5, verbose=False):
    """LP mirror descent polishing step."""
    a = np.maximum(a.astype(np.float64).copy(), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Select active constraints (top fraction)
        threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
        active_k = np.where(b >= threshold)[0]
        if len(active_k) < 10:
            active_k = np.arange(2 * n - 1)

        A = build_lp_matrix_reduced(n, a, active_k)
        c = -np.ones(n)
        b_ub = np.full(len(active_k), M)
        bounds = [(0, np.inf)] * n

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 3, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0.0)
                if g.sum() > 1e-10:
                    g = g / g.sum()
                    # Verify
                    ag = fast_cross_conv(a, g)
                    if ag.max() <= M * (1.0 + 1e-4):
                        # Line search
                        improved = False
                        for t_step in [0.5, 0.3, 0.2, 0.1, 0.05]:
                            a_trial = (1.0 - t_step) * a + t_step * g
                            a_trial = np.maximum(a_trial, 0.0)
                            s = a_trial.sum()
                            if s < 1e-10:
                                continue
                            a_trial = a_trial / s
                            J_trial = compute_J(a_trial)
                            if J_trial < best_J:
                                best_J = J_trial
                                best_a = a_trial.copy()
                                a = a_trial.copy()
                                improved = True
                                no_improve = 0
                                break
                            elif J_trial < compute_J(a):
                                a = a_trial.copy()
                                no_improve = 0
                                break
                        if not improved:
                            no_improve += 1
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 50:
            break

        if verbose and it % 50 == 0:
            print(f"      LP iter={it} J={best_J:.8f} no_improve={no_improve}", flush=True)

    return best_a, best_J


def coordinate_descent(a, max_sweeps=3, deadline=None):
    """Coordinate descent: for each coordinate, try small perturbations."""
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            a_save = a[i]
            # Try a few values around current
            for scale in [0.5, 0.8, 1.2, 1.5, 2.0, 0.3]:
                a[i] = a_save * scale
                a_nc = np.maximum(a, 0.0)
                s = a_nc.sum()
                if s < 1e-10:
                    a[i] = a_save
                    continue
                a_nc = a_nc / s
                J_try = compute_J(a_nc)
                if J_try < best_J:
                    best_J = J_try
                    a = a_nc.copy()
                    improved = True
                    break
            a[i] = a[i]  # keep the best
        if not improved:
            break

    return a, best_J


def make_palindrome(a):
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(np.maximum(p, 1e-6))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(make_palindrome(p))

    inits.append(np.ones(n))

    # Various structured patterns
    inits.append(np.maximum(0.3 + 0.7 * np.cos(np.pi * (2 * t_arr - 1)), 0.01))
    inits.append(np.maximum(0.5 + 0.5 * np.exp(-8 * (t_arr - 0.5) ** 2), 0.01))
    inits.append(np.maximum(0.2 + 0.8 * (1 - np.abs(2 * t_arr - 1)), 0.01))

    # Multi-frequency
    for f in [2, 3, 4, 5]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * f * t_arr), 0.01))

    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Edge-peaked
    inits.append(np.maximum(0.1 + np.exp(-5 * t_arr ** 2) + np.exp(-5 * (t_arr - 1) ** 2), 0.01))

    # Block structure
    n_blocks = max(4, n // 64)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    # Smooth the blocks
    ksize2 = max(3, n // 32)
    block_smooth = np.convolve(block_seq, np.ones(ksize2) / ksize2, mode='same')
    inits.append(np.maximum(block_smooth, 0.01))

    return inits

        if time.monotonic() < deadline - 20:
            a_pal = make_palindrome(best_a)
            a_pal = np.maximum(a_pal, 1e-6)
            a_pal = a_pal / a_pal.sum()
            a_pal, _ = mirror_descent_lse(a_pal, min(time.monotonic() + 20, deadline - 15),
                                          t_start=10, t_end=150, max_iters=2000)
            update_best(a_pal, "palindrome_md")
            if time.monotonic() < deadline - 10:
                a_pal2, _ = lp_polish(a_pal, min(time.monotonic() + 15, deadline - 8),
                                      max_iters=30, top_frac=0.5)
                update_best(a_pal2, "palindrome_lp")

    # === PHASE 4: Final safety checks ===
    if best_a is None:
        best_a = np.ones(128)
        best_score = compute_J(best_a)

    # One final LP pass on the absolute best if time allows
    if time.monotonic() < deadline - 5:
        print(f"\n  Final LP pass on best (n={best_a.size})...", flush=True)
        a_final, _ = lp_polish(best_a, deadline - 2, max_iters=50, top_frac=0.6)
        update_best(a_final, "final_lp")

    # Ensure valid output
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:
        best_a = np.ones(128)
        best_score = compute_J(best_a)

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size} time={elapsed:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_a]
