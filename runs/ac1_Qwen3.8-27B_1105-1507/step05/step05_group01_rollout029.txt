"""
Peak-autocorrelation minimization via LP alternating, subgradient descent,
and multi-length exploration. Minimizes 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    # Use scipy's toeplitz-like construction but more memory efficient
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M, constraint_indices=None):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    
    if constraint_indices is None:
        A = build_toeplitz(a)
        b_ub = np.full(m, M)
    else:
        A = build_toeplitz(a)
        A = A[constraint_indices, :]
        b_ub = np.full(len(constraint_indices), M)
    
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, 0.0


def subgradient_step(a, step_size):
    """Subgradient descent step to reduce peak autocorrelation."""
    n = len(a)
    conv = compute_conv(a)
    
    # Find the peak positions
    peak_thresh = 0.95 * np.max(conv)
    peak_positions = np.where(conv >= peak_thresh)[0]
    
    # Compute subgradient (weighted average of active constraint gradients)
    grad = np.zeros(n)
    for k in peak_positions:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += a[j]
    
    # Normalize gradient
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return a.copy()
    grad = grad / grad_norm
    
    # Step and project onto simplex {a >= 0, sum(a) = 1}
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    a_new = a_new / s
    return a_new


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    k = np.max(np.where(cond)[0])
    theta = cssv[k] / (k + 1)
    return np.maximum(v - theta, 0)

def resample_sequence(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.001)

"""
Multi-strategy peak-autocorrelation minimizer: LP alternating with sparse
constraints, subgradient descent, peak breaking, spectral shaping, and
multi-length exploration to minimize 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        end = min(j + n, m)
        length = end - j
        rows.extend(range(j, end))
        cols.extend([j] * length)
        vals.extend(a[:length])
    return csr_matrix((vals, (rows, cols)), shape=(m, n))


def solve_lp(a, M, constraint_indices=None, time_limit=3.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for selected k, g >= 0."""
    n = len(a)
    if constraint_indices is None:
        A = build_toeplitz_sparse(a)
        b_ub = np.full(A.shape[0], M)
    else:
        A_full = build_toeplitz_sparse(a)
        A = A_full[constraint_indices, :]
        b_ub = np.full(len(constraint_indices), M)

    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g)), res
    except Exception:
        pass
    return None, 0.0, None


def solve_lp_restricted(a, M, top_k=20, time_limit=2.0):
    """Solve LP using only top-k active constraints (faster)."""
    n = len(a)
    conv = compute_conv(a)
    # Select the k positions where conv is highest (most likely to be binding)
    thresh = np.percentile(conv, 100.0 - 100.0 * top_k / (2 * n - 1))
    active = np.where(conv >= thresh)[0]
    if len(active) == 0:
        active = np.array([argmax_idx])
    return solve_lp(a, M, constraint_indices=active, time_limit=time_limit)


def subgradient_step(a, step_size, top_k=3):
    """Subgradient step targeting the top-k peaks."""
    n = len(a)
    conv = compute_conv(a)
    peak_thresh = conv[np.argsort(conv)[::-1][:top_k]][-1] if top_k <= len(conv) else 0
    peak_positions = np.where(conv >= peak_thresh)[0]

    grad = np.zeros(n)
    for k in peak_positions:
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        for i in range(lo, hi):
            j = k - i
            if 0 <= j < n:
                grad[i] += a[j]

    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return a.copy()
    grad = grad / grad_norm

    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = np.max(np.where(cond)[0])
    theta = cssv[k] / (k + 1)
    return np.maximum(v - theta, 0)


def lp_optimize(a, time_budget=100.0, verbose=False):
    """LP-based alternating optimization with adaptive constraint selection."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 40
    relax = 1.0
    use_restricted = n > 200

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(a)
        M = float(np.max(conv)) * relax

        # Choose LP strategy based on problem size and stall
        if use_restricted and M > 0:
            top_k = min(n, max(10, int(0.05 * (2 * n - 1))))
            g, S, _ = solve_lp_restricted(a, M, top_k=top_k, time_limit=min(2.0, remaining * 0.2))
            # Verify feasibility
            if g is not None and S > 1e-10:
                conv_check = compute_conv(a)  # reuse
                # Check if g is feasible for all constraints
                Ag = np.zeros(2 * n - 1)
                for j in range(n):
                    if g[j] > 1e-12:
                        end = min(j + n, 2 * n - 1)
                        Ag[j:end] += g[j] * a[:end - j]
                if np.max(Ag) > M * 1.001:
                    # Not feasible, fall back to full LP
                    g, S, _ = solve_lp(a, M, time_limit=min(3.0, remaining * 0.3))
        else:
            g, S, _ = solve_lp(a, M, time_limit=min(3.0, remaining * 0.3))

        if g is None or S <= 1.0 + 1e-6:
            stall += 1
            if stall > patience:
                relax = min(relax + 0.03, 1.3)
                if relax > 1.15:
                    rng = np.random.default_rng(hash((n, int(time.monotonic()*1000))) % (2**31))
                    a = a * np.exp(rng.normal(0, 0.06, n))
                    a = np.maximum(a, 0)
                    s2 = np.sum(a)
                    if s2 > 1e-10:
                        a = a / s2
                    else:
                        a = best_a.copy()
                        relax = 1.0
                    stall = 0
                    patience = max(15, patience // 2)
            continue

        # Line search with golden section refinement
        improved = False
        best_t = 0
        best_t_obj = best_obj
        for t in [0.5, 0.7, 0.85, 1.0, 0.3, 0.95, 0.15, 0.6, 0.8]:
            a_try = (1.0 - t) * a + t * g / S
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t = t

        if best_t > 0 and best_t_obj < best_obj - 1e-14:
            a_final = (1.0 - best_t) * a + best_t * g / S
            a_final = np.maximum(a_final, 0)
            s_f = np.sum(a_final)
            if s_f > 1e-10:
                a_final = a_final / s_f
            best_obj = best_t_obj
            best_a = a_final.copy()
            a = a_final
            improved = True
            stall = 0
            relax = max(1.0, relax - 0.01)
        else:
            stall += 1

    return best_a, best_obj


def subgradient_optimize(a, time_budget=50.0):
    """Subgradient descent with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    min_step = 0.0005
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        a_new = subgradient_step(a, step, top_k=rng.integers(1, 6))
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            step = max(min_step, step * 0.97)
            if rng.random() < 0.05:
                a_pert = project_simplex(a + rng.normal(0, 0.01 * n / n, n))
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                a = a_pert

    return best_a, best_obj


def peak_breaking(a, time_budget=20.0):
    """Targeted peak-breaking by reducing contributors to top peaks."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(4242)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        top_k = np.argsort(conv)[::-1][:8]
        k = int(top_k[rng.integers(0, len(top_k))])

        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        contributors = []
        for i in range(lo, hi):
            j = k - i
            contributors.append((i, j, a[i] * a[j]))
        contributors.sort(key=lambda x: -x[2])

        a_new = a.copy()
        for i, j, c in contributors[:15]:
            factor = rng.uniform(0.65, 0.95)
            if a[i] >= a[j]:
                a_new[i] *= factor
            else:
                a_new[j] *= factor

        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def spectral_optimize(a, time_budget=25.0):
    """Frequency domain: flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(9999)

    while time.monotonic() - start < time_budget:
        fa = rfft(a)
        mag = np.abs(fa)
        phase = np.angle(fa)

        strategy = rng.integers(0, 5)
        if strategy == 0:
            target = np.geomean(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
            alpha = rng.uniform(0.1, 0.4)
            mag_new = (1 - alpha) * mag + alpha * target
        elif strategy == 1:
            thresh = np.percentile(mag[mag > 0], 85) if np.any(mag > 0) else 0
            mag_new = np.where(mag > thresh, thresh + (mag - thresh) * 0.3, mag)
        elif strategy == 2:
            mag_new = mag * np.exp(rng.normal(0, 0.04, len(mag)))
        elif strategy == 3:
            freqs = np.linspace(0, 1, len(mag))
            tilt = rng.uniform(0.05, 0.4)
            mag_new = mag * (1.0 + tilt * freqs)
        else:
            # Notch the highest magnitude bin
            peak_bin = np.argmax(mag)
            if peak_bin > 0:
                mag_new = mag.copy()
                mag_new[peak_bin] *= rng.uniform(0.5, 0.8)
            else:
                mag_new = mag * np.exp(rng.normal(0, 0.03, len(mag)))

        fa_new = mag_new * np.exp(1j * phase)
        a_new = irfft(fa_new, n)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def random_init(n, seed):
    """Diverse random initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 7)

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 5)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        nblocks = rng.integers(3, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 5:
        a = 1.0 + 0.3 * rng.normal(0, 1, n)
    else:
        # Piecewise constant with smooth transitions
        nseg = rng.integers(4, 10)
        a = np.zeros(n)
        for seg in range(nseg):
            val = rng.uniform(0.3, 1.7)
            start = seg * n // nseg
            end = min((seg + 1) * n // nseg + n // (2 * nseg), n)
            a[start:end] = val
        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.001)

def combined_optimize(a, time_budget=150.0):
    """Multi-strategy combined optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: LP (majority)
    lp1_budget = time_budget * 0.45
    a1, obj1 = lp_optimize(a, time_budget=lp1_budget)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()

    if time.monotonic() - start >= time_budget:
        return best_a, best_obj

    # Phase 2: Peak breaking
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        pb_budget = min(remaining * 0.15, 15.0)
        a2, obj2 = peak_breaking(best_a, time_budget=pb_budget)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()

    if time.monotonic() - start >= time_budget:
        return best_a, best_obj

    # Phase 3: More LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        lp2_budget = min(remaining * 0.35, 50.0)
        a3, obj3 = lp_optimize(best_a, time_budget=lp2_budget)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()

    if time.monotonic() - start >= time_budget:
        return best_a, best_obj

    # Phase 4: Spectral
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        sp_budget = min(remaining * 0.2, 15.0)
        a4, obj4 = spectral_optimize(best_a, time_budget=sp_budget)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()

    if time.monotonic() - start >= time_budget:
        return best_a, best_obj

    # Phase 5: Final LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        a5, obj5 = lp_optimize(best_a, time_budget=remaining * 0.9)
        if obj5 < best_obj:
            best_obj = obj5
            best_a = a5.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-stage search minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating, subgradient, peak breaking, spectral shaping,
    and multi-length exploration within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.6f} (t={time.monotonic()-start_time:.1f}s)")

    # Determine n values
    if best_seq is not None:
        np_ = len(best_seq)
        n_values = sorted(set([
            max(100, np_ - 150),
            max(100, np_ - 50),
            np_,
            np_ + 50,
            np_ + 100,
            np_ + 200,
            np_ + 300,
        ]))
    else:
        n_values = [300, 400, 500, 600, 700, 800, 1000]

    print(f"n values: {n_values}")

    # Phase 1: Explore n values
    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Allocate time: more for n closer to current best
        if best_seq is not None:
            dist = abs(n - len(best_seq))
            alloc = max(25.0, min(remaining * 0.22, 180.0))
            if dist > 0:
                alloc *= max(0.4, 1.0 - 0.002 * dist)
        else:
            alloc = min(remaining * 0.18, 120.0)
        alloc = max(20.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s rem, alloc={alloc:.0f}s")

        # Initialize from best if available
        if best_seq is not None and abs(n - len(best_seq)) <= 200:
            a_start = resample_sequence(best_seq, n)
        else:
            a_start = random_init(n, seed + n * 7919)

        a_opt, obj_opt = combined_optimize(a_start, time_budget=alloc)
        print(f"  N={n}: obj={obj_opt:.6f}")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: n={n}, score={best_score:.6f} **")

    # Phase 2: Intensive final polishing
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s ===")

        # Multiple polish rounds
        for rnd in range(6):
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            rnd_alloc = min(remaining * 0.25, 120.0)
            strategy = rnd % 3

            if strategy == 0:
                a_cur, obj_cur = lp_optimize(best_seq, time_budget=rnd_alloc)
            elif strategy == 1:
                a_cur, obj_cur = combined_optimize(best_seq, time_budget=rnd_alloc)
            else:
                # Subgradient + LP
                sg_budget = min(rnd_alloc * 0.3, 30.0)
                a_sg, obj_sg = subgradient_optimize(best_seq, time_budget=sg_budget)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                remaining2 = rnd_alloc - (time.monotonic() - (deadline - remaining))
                a_cur, obj_cur = lp_optimize(best_seq, time_budget=max(5.0, remaining2))

            if obj_cur < best_score:
                best_score = obj_cur
                best_seq = a_cur.copy()
                print(f"  Round {rnd} (strat={strategy}): {best_score:.6f}")
            else:
                print(f"  Round {rnd} (strat={strategy}): {obj_cur:.6f} (no gain)")

        # Final peak breaking
        if time.monotonic() < deadline - 15:
            rem = deadline - time.monotonic()
            a_pb, obj_pb = peak_breaking(best_seq, time_budget=min(15.0, rem * 0.3))
            if obj_pb < best_score:
                best_score = obj_pb
                best_seq = a_pb.copy()
                print(f"  Peak break: {best_score:.6f}")

            # Final LP
            rem = deadline - time.monotonic()
            if rem > 5:
                a_fin, obj_fin = lp_optimize(best_seq, time_budget=max(3.0, rem * 0.7))
                if obj_fin < best_score:
                    best_score = obj_fin
                    best_seq = a_fin.copy()
                    print(f"  Final LP: {best_score:.6f}")

    # Validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\nRESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return result
