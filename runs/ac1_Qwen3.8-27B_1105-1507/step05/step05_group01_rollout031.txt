def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:

    sum_a = float(np.sum(a))

Coordinate descent with peak-targeted refinement for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
Uses vectorized 1D subproblem solving, periodic LP refinement,
and multi-strategy exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.optimize import minimize_scalar
from scipy.fft import rfft, irfft
import time

def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def solve_lp_direction(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    
    # Build toeplitz matrix
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0

            if max_time is not None and time.monotonic() - start_time > max_time:
                return best_a, best_obj
            
            aj = a[j]
            S = np.sum(a) - aj

def golden_section_min(f, lo, hi, tol=1e-10, max_iter=100):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~ 0.618
    
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    
    return (lo + hi) / 2

Actually, one more important consideration: the `minimize_scalar` / golden section search might not be the most efficient. Since the function f(t) = max_k (base[k] + lin[k]*t + quad[k]*t²) / (S+t)² is piecewise smooth (the "max" switches which k is active), the minimum occurs either at:
1. A point where two or more k values are tied for the max AND the derivative of the ratio is zero
2. At t = 0
3. At the boundary t = hi

For case 1, I can use a "parametric" approach: for a given target value c, find the minimum t such that max_k (base[k] + lin[k]*t + quad[k]*t²) <= c*(S+t)². This is equivalent to: for all k, base[k] + lin[k]*t + quad[k]*t² <= c*(S+t)².

This gives us: c*t² + (2*c*S - lin[k])*t + (c*S² - base[k]) - quad[k]*t² >= 0 for all k
i.e., (c - quad[k])*t² + (2*c*S - lin[k])*t + (c*S² - base[k]) >= 0 for all k

This is a system of quadratic inequalities. The minimum c for which this is feasible can be found by bisection on c.

For a given c, the feasibility check is: for each k, the quadratic (c-quad[k])*t² + (2*c*S - lin[k])*t + (c*S² - base[k]) must be >= 0 for some common t >= 0.

This is more complex than the 1D search, so let me stick with the golden section approach.

Let me also add one more strategy: "iterative reweighting". The idea is:
1. Compute the convolution
2. Identify the peak lag k*
3. The "gradient" of the objective with respect to a[j] is dominated by the peak
4. Adjust a[j] to reduce the peak: a[j] ← a[j] * exp(-η * ∂/∂a[j] max_conv)

The subgradient at coordinate j for the peak at k* is:
- If k* = 2j: ∂/∂a[j] conv[k*] = 2*a[j]
- If 0 <= k*-j < n: ∂/∂a[j] conv[k*] = 2*a[k*-j]
- Else: 0

So the update would be: a[j] ← a[j] * (1 - η * 2*a[k*-j]) (for the relevant j)

This is a simple subgradient method. It's not as efficient as exact CD, but it's complementary (it only targets the peak, while CD optimizes the full objective).

"""
Hybrid coordinate descent + LP optimization for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses exact 1D subproblem solving via golden section search, periodic LP direction
finding, peak-targeted subgradient steps, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
from functools import partial
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def lp_direction(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax for all k, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0


def f1d_evaluate(t, base, lin, quad_val, quad_pos, S):
    """Evaluate 1D objective: max_k (base[k] + lin[k]*t + quad[k]*t^2) / (S+t)^2."""
    vals = base + lin * t
    if quad_val > 0.0:
        vals = vals.copy()
        vals[quad_pos] += quad_val * t * t
    denom = (S + t) * (S + t)
    if denom < 1e-30:
        return 1e30
    return float(np.max(vals)) / denom


def golden_section_min_1d(base, lin, quad_val, quad_pos, S, lo, hi, tol=1e-12, max_iter=80):
    """Golden section search for min of f1d_evaluate over [lo, hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f1d_evaluate(c, base, lin, quad_val, quad_pos, S)
    fd = f1d_evaluate(d, base, lin, quad_val, quad_pos, S)

    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f1d_evaluate(c, base, lin, quad_val, quad_pos, S)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f1d_evaluate(d, base, lin, quad_val, quad_pos, S)

    return (lo + hi) / 2.0


def cd_sweep(a, conv, time_limit=None, start=None):
    """One Gauss-Seidel coordinate descent sweep with exact 1D subproblems."""
    n = len(a)
    if start is None:
        start = time.monotonic()
    total_sum = float(a.sum())
    k_all = np.arange(2 * n - 1)

    for j in range(n):
        if time_limit is not None and time.monotonic() - start > time_limit:
            break

        aj = a[j]
        S = total_sum - aj
        if S < 1e-15:
            continue

        kmj = k_all - j
        valid = (kmj >= 0) & (kmj < n)

        lin = np.zeros(2 * n - 1)
        mask = valid & (k_all != 2 * j)
        lin[mask] = 2.0 * a[kmj[mask]]

        quad_val = 1.0 if (0 <= 2 * j < 2 * n - 1) else 0.0
        quad_pos = 2 * j if quad_val > 0.0 else -1

        base = conv - lin * aj
        if quad_val > 0.0:
            base[quad_pos] -= quad_val * aj * aj

        hi = max(4.0 * total_sum / n, 1e-6)
        t_opt = golden_section_min_1d(base, lin, quad_val, quad_pos, S, 0.0, hi)
        t_opt = max(0.0, t_opt)

        delta = t_opt - aj
        if abs(delta) > 1e-14:
            conv += lin * delta
            if quad_val > 0.0:
                conv[quad_pos] += quad_val * (t_opt * t_opt - aj * aj)
            a[j] = t_opt
            total_sum += delta

    return a, conv


def cd_multiple_sweeps(a, num_sweeps=10, time_limit=None, start=None):
    """Multiple coordinate descent sweeps."""
    n = len(a)
    a = a.copy()
    conv = compute_conv(a)
    best_obj = compute_obj(a)
    best_a = a.copy()

    if start is None:
        start = time.monotonic()

    for s in range(num_sweeps):
        if time_limit is not None and time.monotonic() - start > time_limit:
            break
        a, conv = cd_sweep(a, conv, time_limit=time_limit, start=start)
        obj = 2.0 * n * float(np.max(conv)) / (float(a.sum()) ** 2)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def peak_subgradient_step(a, conv, eta=0.05):
    """Subgradient step targeting the convolution peak."""
    n = len(a)
    peak_k = int(np.argmax(conv))
    a_new = a.copy()
    total_sum = float(a.sum())

    for j in range(n):
        if 0 <= peak_k - j < n:
            gradient = 2.0 * a[peak_k - j]
            if peak_k == 2 * j:
                gradient = 2.0 * a[j]
            a_new[j] = max(0.0, a[j] * (1.0 - eta * gradient / total_sum))
        elif peak_k - j >= n:
            kj = peak_k - j - n + 1
            if 0 <= kj < n and j != kj:
                pass

    s_new = float(a_new.sum())
    if s_new < 1e-10:
        return a.copy()
    return a_new / s_new * (total_sum / s_new) if s_new > 0 else a.copy()


def lp_polish(a, time_budget=60.0):
    """LP-based alternating optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        relax = 1.0 + 0.03 * min(stall, 20)
        g, S = lp_direction(a, M, relax=relax)

        if g is not None and S > 1.0 + 1e-8:
            improved = False
            for t in [0.5, 0.7, 0.85, 1.0, 0.3, 0.15]:
                a_try = (1.0 - t) * a + t * g / S
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
            if improved:
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 15:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 2**31)
            scale = rng.uniform(0.03, 0.1)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s2 = a.sum()
            if s2 > 1e-10:
                a = a / s2
            stall = 0

    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def random_init(n, seed):
    """Generate a random initial sequence."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 5)
    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 7.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.uniform(0.3, 1.5, n)
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    else:
        a = rng.uniform(0.5, 1.5, n)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
