"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def solve_irls_qp(w, support, max_active_set_iters=200):
    """
    Solve: min a^T H a s.t. sum(a)=1, a_j=0 for j not in support, a_j >= 0 for j in support.
    H_{ij} = w_{i+j} (Hankel matrix).
    
    Returns a (full length vector) or None if infeasible.
    """
    idx = np.array(support)
    size = len(idx)
    if size == 0:
        return None
    
    # Build H_S
    # H_S[i,j] = w[idx[i] + idx[j]]
    H = w[np.ix_(idx, idx)]  # This gives H[i,j] = w[idx[i] + idx[j]]
    
    # Actually, H_{ij} = w_{i+j} where i,j are the actual indices
    # So H_S[i,j] = w[idx[i] + idx[j]]
    
    # Solve H_S * x = 1
    try:
        x = np.linalg.solve(H, np.ones(size))
    except np.linalg.LinAlgError:
        # Use least squares as fallback
        x, _, _, _ = np.linalg.lstsq(H, np.ones(size), rcond=None)
    
    s = x.sum()
    if abs(s) < 1e-15:
        return None
    
    a_S = x / s
    
    # Check non-negativity
    if np.any(a_S < -1e-10):
        return None
    
    # Build full vector
    n = len(w) // 2 + 1  # This isn't right, we need to know n
    # Actually we need to pass n
    return a_S, idx


def irls_iterate(a, alpha=0.5, max_irls_iters=500, deadline=None):
    """
    IRLS with active set QP solving.
    a: non-negative vector, will be normalized to sum 1.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    for it in range(max_irls_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute current autocorrelation
        b = fast_conv(a)
        b_max = b.max()
        
        # Update weights: emphasize peaks
        # w_k = (b_max / b_k)^alpha or w_k = 1/(b_k + eps)^alpha
        eps = 1e-12
        w = (b_max / (b + eps)) ** alpha
        # Or: w = 1.0 / (b + eps) ** alpha
        # Or: w = np.exp(alpha * (b_max - b))  (softmax-like)
        
        # Active set QP
        support = list(range(n))
        a_new = _active_set_qp(w, support, n)
        
        if a_new is None:
            break
        
        M_new = compute_M(a_new)
        if M_new < best_M * (1 - 1e-10):
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try with current a
            a = a_new
            if compute_M(a) > best_M * 1.5:
                a = best_a.copy()
                break
    
    return best_a


def _active_set_qp(w, support, n):
    """
    Solve: min sum_{i,j in support} w_{i+j} * a_i * a_j s.t. sum(a_i) = 1, a_i >= 0.
    Uses active set method.
    """
    idx = np.array(support, dtype=int)
    size = len(idx)
    
    if size == 0:
        return None
    
    # Build H_S: H_S[i,j] = w[idx[i] + idx[j]]
    H = w[idx[:, None] + idx[None, :]]
    
    # Symmetrize (should already be symmetric)
    H = (H + H.T) / 2.0
    
    # Try solving with current support
    for attempt in range(200):
        try:
            x = np.linalg.solve(H, np.ones(size))
        except np.linalg.LinAlgError:
            try:
                x, _, _, _ = np.linalg.lstsq(H, np.ones(size), rcond=None)
            except:
                return None
        
        s = x.sum()
        if abs(s) < 1e-15 or s < 0:
            # Try regularization
            H_reg = H + 1e-8 * np.eye(size)
            try:
                x = np.linalg.solve(H_reg, np.ones(size))
            except:
                return None
            s = x.sum()
            if abs(s) < 1e-15:
                return None
        
        a_S = x / s
        lam = 2.0 / s  # lambda = -2/s from the KKT conditions
        
        # Check non-negativity
        neg_mask = a_S < -1e-10
        if not neg_mask.any():
            # Check if we should add any indices back
            # For now, just return
            a_full = np.zeros(n)
            a_full[idx] = np.maximum(a_S, 0)
            a_full = a_full / a_full.sum()
            return a_full
        
        # Remove the most negative indices
        neg_indices = np.where(neg_mask)[0]
        worst = neg_indices[np.argmin(a_S[neg_mask])]
        support = list(set(support) - {idx[worst]})
        if len(support) < 2:
            return None
        idx = np.array(sorted(support), dtype=int)
        size = len(idx)
        H = w[idx[:, None] + idx[None, :]]
        H = (H + H.T) / 2.0
    
    return None


def irls_variant(a, alpha=0.5, max_irls_iters=500, deadline=None, weight_type="ratio"):
    """
    IRLS with different weight schemes.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    for it in range(max_irls_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        b_min = b.max() * 1e-10  # floor
        
        if weight_type == "ratio":
            w = (b_max / np.maximum(b, b_min)) ** alpha
        elif weight_type == "inverse":
            w = 1.0 / np.maximum(b, b_min) ** alpha
        elif weight_type == "exp":
            w = np.exp(alpha * (b_max - b) / b_max)
        elif weight_type == "uniform":
            w = np.ones(2 * n - 1)
        else:
            w = (b_max / np.maximum(b, b_min)) ** alpha
        
        a_new = _active_set_qp(w, list(range(n)), n)
        
        if a_new is None:
            break
        
        M_new = compute_M(a_new)
        if M_new < best_M * (1 - 1e-10):
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = 0.5 * a + 0.5 * a_new
            a = np.maximum(a, 0)
            a = a / a.sum()
            if compute_M(a) > best_M * 2:
                a = best_a.copy()
                break
    
    return best_a


def simulated_annealing(a, rng, max_iters=2000, deadline=None):
    """
    Simulated annealing with structured moves.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    current_M = best_M
    
    T_start = 0.1
    T_end = 0.001
    cooldown = 0.995
    
    T = T_start
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Generate a move
        move_type = rng.integers(0, 5)
        a_new = a.copy()
        
        if move_type == 0:
            # Add Gaussian noise
            noise = rng.normal(0, 0.1, n)
            a_new = a + noise
        elif move_type == 1:
            # Flip a random position
            idx = rng.integers(0, n)
            a_new[idx] = 0
            # Redistribute mass
            if a_new.sum() > 1e-10:
                a_new = a_new / a_new.sum()
        elif move_type == 2:
            # Shift
            shift = rng.integers(1, min(n // 4, 10))
            a_new = np.roll(a, shift)
        elif move_type == 3:
            # Scale a random subset
            mask = rng.random(n) < 0.3
            factor = rng.uniform(0.5, 2.0)
            a_new[mask] *= factor
        elif move_type == 4:
            # Swap two random positions
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a[j], a[i]
        
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        M_new = compute_M(a_new)
        delta = M_new - current_M
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_new
            current_M = M_new
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
        
        T *= cooldown
    
    return best_a


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    lo = 0.0
    hi = 1.0
    for _ in range(70):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        
        t, M_new = line_search_analytic(a, g)
        
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 100:
            break
        
        if verbose and it % 200 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_structured_init(n, rng, method="random"):
    """Generate structured initial sequence."""
    if method == "random":
        a = rng.random(n) + 0.1
    elif method == "sparse_binary":
        k = max(2, n // 4)
        a = np.zeros(n)
        positions = rng.choice(n, size=k, replace=False)
        a[positions] = 1.0
    elif method == "quadratic_residues":
        # For prime n, use quadratic residues
        a = np.zeros(n)
        # Find a prime near n
        p = n
        while p > 3:
            if all(p % i != 0 for i in range(2, int(p**0.5) + 1)):
                break
            p -= 1
        if p >= 5 and p % 4 == 1:
            # Quadratic residues mod p
            residues = set((i * i) % p for i in range(1, p))
            for r in residues:
                if r < n:
                    a[r] = 1.0
    elif method == "sinusoidal":
        t = np.linspace(0, 2 * np.pi * rng.integers(3, 10), n)
        a = np.abs(np.sin(t)) + 0.1
    elif method == "gaussian_bump":
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(0.1, 0.4) * n
        x = np.arange(n)
        a = np.exp(-((x - center) / width) ** 2) + 0.01
    elif method == "two_bumps":
        c1 = rng.uniform(0.2, 0.4) * n
        c2 = rng.uniform(0.6, 0.8) * n
        w = rng.uniform(0.1, 0.2) * n
        x = np.arange(n)
        a = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2) + 0.01
    else:
        a = rng.random(n) + 0.1
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. IRLS with active-set QP solving (Hankel matrix structure) - novel optimizer
2. Simulated annealing with structured moves for global exploration
3. LP mirror descent with analytic line search for polishing
4. Multi-length exploration with resampling
5. Multi-initialization from structured patterns
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def _active_set_qp(w, n, max_active_set_iters=300):
    """
    Solve: min sum_{i,j} w_{i+j} * a_i * a_j s.t. sum(a)=1, a >= 0.
    H_{ij} = w_{i+j} is a Hankel matrix.
    Uses active set method: start with full support, remove negative variables.
    Returns non-negative a with sum=1, or None.
    """
    support = list(range(n))
    idx = np.array(support, dtype=int)
    size = len(idx)

    for attempt in range(max_active_set_iters):
        if size < 1:
            return None
        # Build H_S: H_S[i,j] = w[idx[i] + idx[j]]
        H = w[idx[:, None] + idx[None, :]]
        H = (H + H.T) / 2.0

        # Solve H_S * x = 1
        try:
            x = np.linalg.solve(H, np.ones(size))
        except np.linalg.LinAlgError:
            try:
                x, _, _, _ = np.linalg.lstsq(H + 1e-10 * np.eye(size), np.ones(size), rcond=None)
            except Exception:
                return None

        s = x.sum()
        if abs(s) < 1e-15:
            return None

        a_S = x / s

        # Check non-negativity
        if np.all(a_S >= -1e-10):
            a_full = np.zeros(n)
            a_full[idx] = np.maximum(a_S, 0)
            s2 = a_full.sum()
            if s2 > 1e-15:
                a_full = a_full / s2
            return a_full

        # Remove the most negative variable
        neg_mask = a_S < -1e-10
        if not neg_mask.any():
            a_full = np.zeros(n)
            a_full[idx] = np.maximum(a_S, 0)
            s2 = a_full.sum()
            if s2 > 1e-15:
                a_full = a_full / s2
            return a_full

        neg_indices = np.where(neg_mask)[0]
        worst_local = neg_indices[np.argmin(a_S[neg_mask])]
        worst_global = idx[worst_local]
        support.remove(int(worst_global))
        idx = np.array(support, dtype=int)
        size = len(idx)

    return None


def irls_optimize(a, alpha=0.5, max_irls_iters=300, deadline=None, weight_type="ratio"):
    """
    IRLS with active set QP solving.
    Iteratively reweights the autocorrelation peaks and solves the resulting QP.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_irls_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        if b_max < 1e-15:
            break

        # Update weights
        floor = b_max * 1e-10
        if weight_type == "ratio":
            w = (b_max / np.maximum(b, floor)) ** alpha
        elif weight_type == "inverse":
            w = 1.0 / np.maximum(b, floor) ** alpha
        elif weight_type == "exp":
            w = np.exp(alpha * (b_max - b) / max(b_max, 1e-15))
        elif weight_type == "peak":
            # Heavier emphasis on the top peaks
            w = np.ones(2 * n - 1)
            top_k = max(1, int(0.1 * len(b)))
            top_indices = np.argsort(b)[-top_k:]
            w[top_indices] = (b_max / b[top_indices]) ** alpha
        else:
            w = (b_max / np.maximum(b, floor)) ** alpha

        a_new = _active_set_qp(w, n)
        if a_new is None:
            break

        M_new = compute_M(a_new)
        if M_new < best_M * (1 - 1e-12):
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            # Damped update
            a = 0.7 * a + 0.3 * a_new
            a = np.maximum(a, 0)
            s2 = a.sum()
            if s2 > 1e-10:
                a = a / s2
            no_improve += 1

        if no_improve > 30:
            break

    return best_a


def irls_multi_alpha(a, alpha_list, max_iters_each, deadline):
    """Try multiple alpha values and weight types, keep the best."""
    best = a.copy()
    best_M = compute_M(a)

    for weight_type in ["ratio", "inverse", "exp", "peak"]:
        for alpha in alpha_list:
            if deadline is not None and time.monotonic() > deadline:
                return best
            a_trial = irls_optimize(a, alpha=alpha, max_irls_iters=max_iters_each,
                                    deadline=deadline, weight_type=weight_type)
            M_trial = compute_M(a_trial)
            if M_trial < best_M:
                best_M = M_trial
                best = a_trial.copy()

    return best


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(70):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            break

        if verbose and it % 200 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def simulated_annealing(a, rng, max_iters=1000, deadline=None):
    """Simulated annealing with structured moves."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s.sum() if False else a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    current_M = best_M

    T = 0.05
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        move_type = rng.integers(0, 6)
        a_new = a.copy()

        if move_type == 0:
            # Gaussian perturbation
            noise = rng.normal(0, 0.08, n)
            a_new = a + noise
        elif move_type == 1:
            # Zero out random positions and redistribute
            k_zero = max(1, int(n * 0.05))
            positions = rng.choice(n, size=k_zero, replace=False)
            a_new[positions] = 0
        elif move_type == 2:
            # Circular shift
            shift = int(rng.integers(1, max(2, n // 8)))
            a_new = np.roll(a, shift)
        elif move_type == 3:
            # Scale a random block
            start = int(rng.integers(0, n - 1))
            length = int(rng.integers(1, max(2, n // 6)))
            end = min(start + length, n)
            factor = rng.uniform(0.3, 3.0)
            a_new[start:end] *= factor
        elif move_type == 4:
            # Swap two blocks
            i = int(rng.integers(0, n // 2))
            j = int(rng.integers(n // 2, n))
            length = int(rng.integers(1, max(2, n // 10)))
            a_new[i:i+length], a_new[j:j+length] = a_new[j:j+length].copy(), a_new[i:i+length].copy()
        else:
            # Inject a spike
            pos = int(rng.integers(0, n))
            a_new[pos] += rng.uniform(0.1, 0.5)

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        M_new = compute_M(a_new)
        delta = M_new - current_M

        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_new
            current_M = M_new
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()

        T *= 0.997

    return best_a


def generate_structured_init(n, rng, method="random"):
    """Generate structured initial sequence."""
    if method == "random":
        a = rng.random(n) + 0.1
    elif method == "sparse_binary":
        k = max(2, n // 4)
        a = np.zeros(n)
        positions = rng.choice(n, size=k, replace=False)
        a[positions] = 1.0
    elif method == "quadratic_residues":
        a = np.zeros(n)
        p = n
        found_prime = False
        for candidate in range(n, max(1, n - 50), -1):
            if candidate >= 5 and all(candidate % i != 0 for i in range(2, int(candidate**0.5) + 1)):
                p = candidate
                found_prime = True
                break
        if found_prime and p >= 5:
            for i in range(1, p):
                r = (i * i) % p
                if r < n:
                    a[r] = 1.0
            if a.sum() < 1:
                a = rng.random(n) + 0.1
        else:
            a = rng.random(n) + 0.1
    elif method == "sinusoidal":
        freq = rng.integers(3, 12)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n)
        a = np.abs(np.sin(2 * np.pi * freq * t / n + phase)) + 0.1
    elif method == "gaussian_bump":
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(0.1, 0.4) * n
        x = np.arange(n) - center
        a = np.exp(-(x / width) ** 2) + 0.01
    elif method == "two_bumps":
        c1 = rng.uniform(0.2, 0.4) * n
        c2 = rng.uniform(0.6, 0.8) * n
        w = rng.uniform(0.1, 0.25) * n
        x = np.arange(n)
        a = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2) + 0.01
    elif method == "alternating":
        a = np.ones(n)
        a[::2] = rng.uniform(0.3, 0.7)
    elif method == "plateau":
        w = rng.uniform(0.2, 0.5) * n
        c = rng.uniform(0.3, 0.7) * n
        x = np.arange(n)
        a = np.where(np.abs(x - c) < w, 1.0, 0.2)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def smooth_gradient_step(a, tau):
    """Gradient of (1/tau)*log(sum exp(tau*(a*a)))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad


def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=150.0):
    """Smooth gradient descent with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = 0.3 * (0.05 / 0.3) ** frac

        grad = smooth_gradient_step(a, tau)
        v = 0.9 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining IRLS (Hankel QP), simulated annealing,
    LP mirror descent, smooth gradient, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = generate_structured_init(512, rng, "random")
        tracker.update(parent, "(default)")

    # ===== PHASE 1: IRLS on parent =====
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: IRLS optimization n={parent.size}", flush=True)
    print(f"{'='*60}", flush=True)

    n0 = parent.size
    a = parent.copy()
    a = a / a.sum()

    phase1_dl = min(t_start + budget_s * 0.15, deadline)
    a_irls = irls_multi_alpha(a, [0.3, 0.5, 0.7, 1.0], 150, phase1_dl)
    tracker.update(a_irls, "(P1 IRLS)")

    # LP MD polish on IRLS result
    if tracker.best is not None and tracker.best.size == n0:
        row_idx, col_idx = build_lp_matrix(n0)
        sub_dl = min(t_start + budget_s * 0.25, deadline)
        a_lp = run_lp_mirror_descent(tracker.best, 10000, sub_dl, row_idx, col_idx, verbose=False)
        tracker.update(a_lp, "(P1 LP-MD)")

        # IRLS again after LP
        if time.monotonic() < deadline:
            sub_dl2 = min(t_start + budget_s * 0.28, deadline)
            a_irls2 = irls_multi_alpha(tracker.best, [0.5, 1.0], 100, sub_dl2)
            tracker.update(a_irls2, "(P1 IRLS2)")

    # ===== PHASE 2: Multi-initialization exploration =====
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-init IRLS + LP", flush=True)
    print(f"{'='*60}", flush=True)

    phase2_dl = min(t_start + budget_s * 0.50, deadline)
    methods = ["random", "sparse_binary", "sinusoidal", "gaussian_bump",
               "two_bumps", "quadratic_residues", "alternating", "plateau"]

    for trial in range(16):
        if time.monotonic() > phase2_dl:
            break
        method = methods[trial % len(methods)]
        n_trial = n0
        a_init = generate_structured_init(n_trial, rng, method)

        # IRLS
        sub_dl = min(time.monotonic() + 15, phase2_dl)
        a_opt = irls_multi_alpha(a_init, [0.5, 1.0], 80, sub_dl)
        tracker.update(a_opt, f"(P2 {method}#{trial})")

        # LP polish
        if tracker.best.size == n_trial and time.monotonic() < phase2_dl:
            r_idx, c_idx = build_lp_matrix(n_trial)
            sub_dl2 = min(time.monotonic() + 20, phase2_dl)
            a_lp = run_lp_mirror_descent(tracker.best, 2000, sub_dl2, r_idx, c_idx, verbose=False)
            tracker.update(a_lp, f"(P2 {method} LPG{trial})")

    # ===== PHASE 3: Simulated annealing + polish =====
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Simulated Annealing", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        phase3_dl = min(t_start + budget_s * 0.65, deadline)
        n_best = tracker.best.size
        r_idx_b, c_idx_b = build_lp_matrix(n_best)

        for r in range(10):
            if time.monotonic() > phase3_dl:
                break

            a_sa = simulated_annealing(tracker.best, rng, max_iters=400,
                                       deadline=min(time.monotonic() + 12, phase3_dl))
            tracker.update(a_sa, f"(P3 SA r={r})")

            if tracker.best.size == n_best and time.monotonic() < phase3_dl:
                sub_dl = min(time.monotonic() + 12, phase3_dl)
                a_lp = run_lp_mirror_descent(tracker.best, 2000, sub_dl, r_idx_b, c_idx_b, verbose=False)
                tracker.update(a_lp, f"(P3 SA+LP r={r})")

                # Quick IRLS
                if time.monotonic() < phase3_dl:
                    a_ir = irls_multi_alpha(tracker.best, [0.5, 1.0], 50,
                                            min(time.monotonic() + 8, phase3_dl))
                    tracker.update(a_ir, f"(P3 SA+IRLS r={r})")

    # ===== PHASE 4: Multi-length exploration =====
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.5, 0.6, 0.75, 1.25, 1.33, 1.5, 2.0, 0.8, 1.2]:
            n_trial = int(base_n * mult)
            if 64 <= n_trial <= 4096 and n_trial != base_n:
                trial_lengths.append(n_trial)
        trial_lengths = sorted(set(trial_lengths))

        phase4_dl = min(t_start + budget_s * 0.82, deadline)
        for n_trial in trial_lengths:
            if time.monotonic() > phase4_dl:
                break

            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()

            sub_dl = min(time.monotonic() + 20, phase4_dl)
            a_opt = irls_multi_alpha(a_init, [0.5, 1.0], 100, sub_dl)
            tracker.update(a_opt, f"(P4 IRLS n={n_trial})")

            if time.monotonic() < phase4_dl:
                r_idx_t, c_idx_t = build_lp_matrix(tracker.best.size)
                sub_dl2 = min(time.monotonic() + 15, phase4_dl)
                a_lp = run_lp_mirror_descent(tracker.best, 2000, sub_dl2, r_idx_t, c_idx_t, verbose=False)
                tracker.update(a_lp, f"(P4 LP n={tracker.best.size})")

    # ===== PHASE 5: Perturbation restarts =====
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Perturbation restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_idx_b, c_idx_b = build_lp_matrix(n_best)
        phase5_dl = min(t_start + budget_s * 0.90, deadline)
        temperature = 0.10

        for r in range(20):
            if time.monotonic() > phase5_dl:
                break

            a = tracker.best.copy().astype(np.float64)
            noise_scale = temperature * 0.01
            a = a + rng.normal(0, noise_scale, n_best)
            a = np.maximum(a, 1e-8)
            a = a / a.sum()

            # Quick smooth gradient
            sub_dl = min(time.monotonic() + 10, phase5_dl)
            a = run_smooth_gradient(a, 150, sub_dl, initial_tau=10, final_tau=80)
            # IRLS
            sub_dl2 = min(time.monotonic() + 10, phase5_dl)
            a = irls_multi_alpha(a, [0.5, 1.0], 60, sub_dl2)
            tracker.update(a, f"(P5 pert r={r} T={temperature:.3f})")

            # LP polish
            if time.monotonic() < phase5_dl:
                sub_dl3 = min(time.monotonic() + 15, phase5_dl)
                a_lp = run_lp_mirror_descent(tracker.best, 2000, sub_dl3, r_idx_b, c_idx_b, verbose=False)
                tracker.update(a_lp, f"(P5 pert+LP r={r})")

            temperature *= 0.85

    # ===== PHASE 6: Final intensive polish =====
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        # IRLS final
        a = tracker.best.copy()
        a = a / a.sum()
        a_ir = irls_multi_alpha(a, [0.3, 0.5, 0.7, 1.0, 1.5, 2.0], 200, deadline)
        tracker.update(a_ir, "(P6 IRLS final)")

        # LP MD final
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)
        a_lp = run_lp_mirror_descent(tracker.best, 50000, deadline, r_f, c_f, verbose=True)
        tracker.update(a_lp, "(P6 LP final)")

        # One more IRLS
        if time.monotonic() < deadline - 5:
            a_ir2 = irls_multi_alpha(tracker.best, [0.5, 1.0, 1.5], 150, deadline)
            tracker.update(a_ir2, "(P6 IRLS after LP)")

            # Final LP
            if time.monotonic() < deadline - 3:
                r_f2, c_f2 = build_lp_matrix(tracker.best.size)
                a_lp2 = run_lp_mirror_descent(tracker.best, 10000, deadline, r_f2, c_f2, verbose=False)
                tracker.update(a_lp2, "(P6 LP final2)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
