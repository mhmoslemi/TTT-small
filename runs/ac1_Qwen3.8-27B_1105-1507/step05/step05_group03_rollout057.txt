"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Dual LP-based mirror descent (fewer constraints = faster LP solves)
2. Aggressive multi-length exploration
3. Hybrid subgradient + LP refinement
4. Diverse restarts with structured perturbations
5. Intensive final polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_structure(n):
    """Precompute row/col indices for the LP constraint matrix A where (a*g)[k] uses A."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_primal(a, M, row_idx, col_idx):
    """Solve primal LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def solve_lp_dual(a, M, row_idx, col_idx):

    The dual has 2n-1 variables and n constraints.

    # Dual: min M * 1^T y s.t. A^T y >= 1, y >= 0
    # A^T is n x (2n-1), (A^T y)[j] = sum_k A[k,j] * y[k] = sum_k a[k-j] * y[k]
    # We need: -A^T y <= -1
    # Build A^T sparsely: (A^T)[j,k] = A[k,j] = a[k-j] if 0<=k-j<n
    # So (A^T)[j,k] is nonzero when 0 <= k-j < n, i.e., j <= k <= j+n-1
    t_rows = []
    t_cols = []
    t_vals = []
    for j in range(n):
        for k in range(2 * n - 1):
            diff = k - j
            if 0 <= diff < n:
                t_rows.append(j)
                t_cols.append(k)
                t_vals.append(a[diff])
    A_T = csr_matrix((t_vals, (t_rows, t_cols)), shape=(n, 2 * n - 1))
    # Dual: min M * sum(y) s.t. A^T y >= 1, y >= 0
    # For linprog (min form): min c^T y s.t. A_ub y <= b_ub, bounds
    c = np.full(2 * n - 1, M)
    A_ub = -A_T
    b_ub = -np.ones(n)
    bounds = [(0, None)] * (2 * n - 1)
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            y = np.maximum(r.x, 0)
            # Recover primal from dual via complementary slackness or just return y
            # We need g, the primal solution. Use strong duality:
            # At optimum, g_j > 0 iff (A^T y)[j] = 1
            # We can approximate g by: g_j = 1/(A^T y)[j] if (A^T y)[j] is close to 1
            # But this is unreliable. Instead, just solve the primal with the dual info.
            # Alternative: use the dual objective value to verify, then solve primal.
            # For now, fall back to primal solve if dual succeeds but we can't recover g.
            pass
    except Exception:
        pass
    return None


def solve_lp_hybrid(a, M, row_idx, col_idx):
    """Try dual first, fall back to primal."""
    g = solve_lp_primal(a, M, row_idx, col_idx)
    if g is not None:
        return g
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def subgradient_direction(a, budget_check=None):
    """
    Compute a weighted subgradient direction.
    Returns g (non-negative, not yet normalized) that points "outward" from the constraint set.
    We use the soft-max weighting of the active constraints.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    tau = 50.0
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    # Subgradient of max(a*a) is sum_k p_k * grad(a*a)[k]
    # grad(a*a)[k] has j-th component = 2 * a[k-j] (if 0<=k-j<n)
    # So the full subgradient is: sg_j = 2 * sum_k p_k * a[k-j] (if valid)
    # = 2 * (p * a)[shifted] ... = 2 * (cross_conv(p, a))[j + offset]
    # Actually: sg_j = 2 * sum_{k: 0<=k-j<n} p_k * a[k-j]
    # Let m = k-j, so k = j+m: sg_j = 2 * sum_m p[j+m] * a[m]
    # This is a cross-correlation type operation.
    # sg = 2 * (p * reversed_a) shifted... Let me just compute it directly via FFT.
    
    # sg_j = 2 * sum_k p_k * a_k_j where a[k-j] is valid
    # This is: sg = 2 * np.correlate(p, a, mode='same') ... not quite.
    # Let me think: sg_j = 2 * sum_k p[k] * a[k-j] for k-j in [0,n-1]
    # = 2 * sum_m a[m] * p[j+m] for j+m in [0, 2n-2]
    # This is the cross-correlation of a and p, evaluated at index j.
    
    # Using FFT: cross-correlation of a and p = ifft(conj(fft(a)) * fft(p))
    # But we want: sum_m a[m] * p[j+m] = (a * flip(p))[j] ... 
    # Actually sum_m a[m] * p[j+m] is the cross-correlation of p and a at lag j.
    # crosscorr(p, a)[j] = sum_m p[m] * conj(a[m-j]) ... this is getting confusing.
    
    # Let me just compute it via the convolution structure:
    # sg_j = 2 * sum_k p[k] * a[k-j] (for valid k-j)
    # Let's define q = flip(a) (so q[i] = a[n-1-i])
    # Then a[k-j] = q[n-1-(k-j)] = q[n-1-k+j]
    # sg_j = 2 * sum_k p[k] * q[n-1-k+j]
    # Let l = n-1-k+j, so k = n-1-j+l
    # sg_j = 2 * sum_l p[n-1-j+l] * q[l]
    # = 2 * (p[n-1-j:...] * q[:...]) ... this is a convolution.
    
    # Simpler: just compute sg_j = 2 * sum_{k=max(j,0)}^{min(2n-2, j+n-1)} p[k] * a[k-j]
    # For efficiency, use FFT:
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    # cross_conv(a, p) gives sum_i a[i]*p[k-i]
    # We want sum_k p[k]*a[k-j] = cross_conv(p, a_reversed)[j + n - 1]
    # Actually let me just use: sum_m a[m] * p[j+m]
    # = (a * flip(p)) evaluated at appropriate position
    # cross_conv(a, flip(p))[j] = sum_i a[i] * flip(p)[j-i] = sum_i a[i] * p[n-1-j+i]
    # Hmm, this is getting complicated. Let me just do it with a loop for clarity
    # (n=512, so at most 1023 iterations, each O(n) = O(n^2) total = ~262K ops, fast enough)
    
    sg = np.zeros(n)
    for m in range(n):
        # sg_j += 2 * p[j+m] * a[m] for valid j+m
        # j ranges from max(0, -(m)) to min(n-1, 2n-2-m)
        # Since m >= 0: j ranges from 0 to min(n-1, 2n-2-m)
        j_max = min(n - 1, 2 * n - 2 - m)
        if j_max >= 0:
            sg[:j_max + 1] += 2.0 * p[m:j_max + 1 + m] * a[m]
    
    return sg


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False, lp_timeout=5):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        g = solve_lp_primal(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-10:
            no_improve += 1
        elif M_new < best_M - 1e-15:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            break

        if verbose and it % 100 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def run_subgradient_avg(a, max_iters, deadline, tau_start=20.0, tau_end=200.0):
    """Subgradient method with averaging for max(a*a) subject to a in simplex."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    a_avg = np.zeros(n)
    step_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        step_size = 0.5 / (1.0 + 0.1 * it)

        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        # Compute subgradient: sg_j = 2 * sum_k p[k] * a[k-j] (if valid)
        # Using the formula: sg = 2 * cross_conv_reversed
        # Let's compute sum_k p[k] * a[k-j] for each j
        # This equals (p * a)[j + offset] in a specific sense
        # cross_conv(p, a)[k] = sum_i p[i] * a[k-i]
        # We want: for each j, sum_k p[k] * a[k-j]
        # Let m = k-j: sum_m p[m+j] * a[m]
        # = sum_m a[m] * p[m+j]
        # This is the cross-correlation of a and p at lag j.
        # xcorr(a, p)[j] = sum_m a[m] * conj(p[m-j]) ... (for real: sum_m a[m] * p[m-j])
        # We want sum_m a[m] * p[m+j] = xcorr(p, a)[j] = sum_m p[m] * a[m-j]
        # Hmm, let me just use FFT directly.
        
        # sum_k p[k] * a[k-j] where k-j must be in [0,n-1] and k in [0,2n-2]
        # = sum_{k=j}^{j+n-1} p[k] * a[k-j] (for valid k)
        # Let's define: for each j, this is the inner product of p[j:j+n] and a[0:n]
        # (with zero-padding where needed)
        
        # Using FFT: define P = fft(p, size), A = fft(a, size)
        # The cross-correlation is: ifft(conj(P) .* A) ... not exactly.
        # Let me just compute it as: result = np.zeros(n)
        # For m in range(n): result += a[m] * p[m:m+n] (with appropriate slicing)
        
        # Efficient computation via FFT:
        # sum_k p[k] * a[k-j] = (p reversed convolved with a) at position (2n-2-j)
        # Actually, let's use: 
        # define q = a (length n), r = p (length 2n-1)
        # We want: c[j] = sum_m q[m] * r[m+j] for j=0..n-1
        # = (q * r_rev)[j + (2n-2) - ... ] 
        # This is getting complicated. For n=512, a direct O(n^2) loop is ~262K ops which is fine.
        
        sg = np.zeros(n)
        for m in range(n):
            am = a[m]
            if am < 1e-15:
                continue
            # sg[j] += 2 * am * p[j+m] for j+m <= 2n-2, j <= n-1
            j_max = min(n - 1, 2 * n - 2 - m)
            if j_max >= 0:
                sg[:j_max + 1] += 2.0 * am * p[m:m + j_max + 1]

        # Projected gradient step on simplex
        a_new = a - step_size * sg
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        a = a_new
        a_avg += a
        step_count += 1

        M = compute_M(a)
        if M < best_M:
            best_M = M
            best_a = a.copy()

        # Also check the average
        if step_count > 10 and step_count % 50 == 0:
            a_avg_norm = a_avg / step_count
            a_avg_norm = np.maximum(a_avg_norm, 0)
            s_avg = a_avg_norm.sum()
            if s_avg > 0.01:
                a_avg_norm = a_avg_norm / s_avg
                M_avg = compute_M(a_avg_norm)
                if M_avg < best_M:
                    best_M = M_avg
                    best_a = a_avg_norm.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_diverse_init(n, rng, method="random"):
    """Generate diverse initial sequences."""
    if method == "random":
        a = rng.random(n)
    elif method == "raised_cosine":
        j = np.arange(n)
        a = np.cos(np.pi * (j - (n - 1) / 2.0) / (n - 1)) ** 2
    elif method == "gaussian":
        j = np.arange(n)
        sigma = n / 4.0
        a = np.exp(-0.5 * ((j - (n - 1) / 2.0) / sigma) ** 2)
    elif method == "sinusoidal":
        j = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * j / (n // 3 + 1))
    elif method == "bimodal":
        j = np.arange(n)
        a = np.exp(-0.5 * ((j - 0.25 * (n-1)) / (n/8.0))**2) + np.exp(-0.5 * ((j - 0.75*(n-1)) / (n/8.0))**2)
    elif method == "uniform_mod":
        a = np.ones(n)
        mod = 1.0 + 0.2 * np.sin(2.0 * np.pi * np.arange(n) * 7 / n)
        a = a * mod
    else:
        a = rng.random(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: LP mirror descent on parent (30%)
    Phase 2: Multi-length exploration (25%)
    Phase 3: Diverse initialization restarts (20%)
    Phase 4: Perturbation + refinement (15%)
    Phase 5: Final intensive polish (10%)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    # Phase 1: LP mirror descent on parent
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent n={parent.size}", flush=True)
    print(f"{'='*60}", flush=True)

    n0 = parent.size
    row_idx, col_idx = build_lp_structure(n0)
    a = parent.copy()
    a = a / a.sum()

    phase1_deadline = min(t_start + budget_s * 0.30, deadline)
    a_opt = run_lp_mirror_descent(a, 50000, phase1_deadline, row_idx, col_idx, verbose=True)
    tracker.update(a_opt, "(P1 LP-MD)")

    # Subgradient average as complementary
    if time.monotonic() < phase1_deadline:
        a_sg = parent.copy() / parent.sum()
        a_sg_opt = run_subgradient_avg(a_sg, 5000, phase1_deadline)
        tracker.update(a_sg_opt, "(P1 subgrad)")

    # Phase 2: Multi-length exploration
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = [128, 192, 256, 320, 384, 448, 512, 576, 640, 768, 896, 1024, 1280, 1536]
        trial_lengths = [n for n in trial_lengths if 64 <= n <= 2048 and n != base_n]
        trial_lengths = sorted(trial_lengths)

        phase2_deadline = min(t_start + budget_s * 0.55, deadline)
        for n_trial in trial_lengths:
            if time.monotonic() > phase2_deadline:
                break
            remaining = phase2_deadline - time.monotonic()
            if remaining < 5:
                break

            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()

            r_idx, c_idx = build_lp_structure(n_trial)
            sub_deadline = min(time.monotonic() + remaining * 0.6, phase2_deadline)
            a_opt = run_lp_mirror_descent(a_init, 3000, sub_deadline, r_idx, c_idx, verbose=False)
            tracker.update(a_opt, f"(P2 n={n_trial})")

            # Try subgradient too for shorter times
            if time.monotonic() < phase2_deadline - 5:
                sub_dl2 = min(time.monotonic() + 10, phase2_deadline)
                a_sg = run_subgradient_avg(a_opt, 1000, sub_dl2)
                tracker.update(a_sg, f"(P2 n={n_trial} sg)")

    # Phase 3: Diverse initialization restarts
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Diverse Initialization Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_idx_b, c_idx_b = build_lp_structure(n_best)
        phase3_deadline = min(t_start + budget_s * 0.75, deadline)

        methods = ["raised_cosine", "gaussian", "sinusoidal", "bimodal", "uniform_mod", "random"]
        for method in methods:
            if time.monotonic() > phase3_deadline:
                break
            remaining = phase3_deadline - time.monotonic()
            if remaining < 10:
                break

            a_init = generate_diverse_init(n_best, rng, method)
            sub_dl = min(time.monotonic() + remaining * 0.4, phase3_deadline)
            a_opt = run_lp_mirror_descent(a_init, 2000, sub_dl, r_idx_b, c_idx_b, verbose=False)
            tracker.update(a_opt, f"(P3 {method})")

    # Phase 4: Perturbation restarts
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_idx_b, c_idx_b = build_lp_structure(n_best)
        temperature = 0.12
        phase4_deadline = min(t_start + budget_s * 0.90, deadline)

        for r in range(30):
            if time.monotonic() > phase4_deadline:
                break

            a = tracker.best.copy().astype(np.float64)
            noise_scale = temperature * 0.02
            a = a + rng.normal(0, noise_scale, n_best)
            a = np.maximum(a, 1e-6)
            a = a / a.sum()

            # Quick subgradient
            sub_dl = min(time.monotonic() + 10, phase4_deadline)
            a = run_subgradient_avg(a, 500, sub_dl)

            # Then LP
            sub_dl2 = min(time.monotonic() + 25, phase4_deadline)
            a_opt = run_lp_mirror_descent(a, 2000, sub_dl2, r_idx_b, c_idx_b, verbose=False)
            tracker.update(a_opt, f"(P4 r={r} T={temperature:.4f})")

            temperature *= 0.87

    # Phase 5: Final intensive LP polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Intensive LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_structure(n_final)
        a = tracker.best.copy()
        a = a / a.sum()

        # Run LP-MD with more patience
        a_opt = run_lp_mirror_descent(a, 100000, deadline, r_f, c_f, verbose=True)
        tracker.update(a_opt, "(P5 final LP)")

        # Also try subgradient averaging on the best
        if time.monotonic() < deadline - 5:
            a_sg = run_subgradient_avg(a_opt, 5000, deadline, tau_start=50, tau_end=500)
            tracker.update(a_sg, "(P5 final sg)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
