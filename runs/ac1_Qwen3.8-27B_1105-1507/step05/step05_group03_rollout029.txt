2. FFT-based gradient computation

4. Multi-length and multi-initialization exploration
5. Final high-tau FISTA polish

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lse_and_grad(a, tau):
    """Compute log-sum-exp objective value and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    p = exp_s / exp_s.sum()
    
    obj = b_max + (max_s + np.log(exp_s.sum())) / tau

    S = 1

    fp = np.fft.rfft(p, S)
    fa = np.fft.rfft(a, S)
    R = np.fft.irfft(np.conj(fp) * fa, S)
    
    grad = np.zeros(n)
    grad[0] = 2.0 * R[0]
    if n > 1:
        grad[1:] = 2.0 * R[S - 1:S - n + 1:-1]
    
    return obj, grad, p

    FISTA on log-sum-exp approximation with gradually increasing tau.
    a_init should be non-negative and will be normalized.
    """

    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    tau_levels = np.geomspace(tau_start, tau_end, tau_steps)
    
    best_a = a.copy()
    best_obj = float('inf')
    
    for tau_idx, tau in enumerate(tau_levels):
        if time.monotonic() > deadline:
            break
        
        # Reset momentum for each tau level (or continue)
        y = a.copy()
        theta = 1.0
        L = 2.0 * n / tau  # Initial Lipschitz estimate
        L = max(L, 1.0)
        
        iters_this_tau = max(50, max_iters // tau_steps)

            if time.monotonic() > deadline:
                break
            
            obj_y, grad_y, p = lse_and_grad(y, tau)
            a_new = project_simplex(y - grad_y / L)
            
            # Backtracking
            obj_new, _, _ = lse_and_grad(a_new, tau)
            if obj_new > obj_y + grad_y @ (a_new - y) + 0.5 * L * np.sum((a_new - y)**2) + 1e-12:

                if obj_new > obj_y + grad_y @ (a_new - y) + 0.5 * L * np.sum((a_new - y)**2) + 1e-12:

            # FISTA update
            theta_new = (1.0 + np.sqrt(1.0 + 4.0 * theta * theta)) / 2.0
            y_new = a_new + ((theta - 1.0) / theta_new) * (a_new - a)
            a = a_new
            theta = theta_new
            
            # Track best (using true max, not log-sum-exp)
            b_true = fast_conv(a)
            true_obj = float(2 * n * b_true.max())  # assuming sum=1
            if true_obj < best_obj:

                best_a = a.copy()
        
        if verbose:

So H_k has at most min(k+1, 2n-1-k, n) non-zero entries on the "anti-diagonal" j+l=k.

The matrix Q = sum_k q_k H_k has entries:
Q[j,l] = sum_k q_k * H_k[j,l] = sum_k q_k * 2 * δ_{j+l,k} = 2 * q_{j+l} (if j+l < 2n-1)

So Q is a **Hankel matrix** with entries Q[j,l] = 2 * q_{j+l}!

This is a beautiful structure! The QP in step 2 becomes:
minimize a^T Q a = 2 * sum_{j,l} a_j * q_{j+l} * a_l
subject to a >= 0, sum(a) = 1

And Q is a Hankel matrix determined by the sequence q = (q_0, q_1, ..., q_{2n-2}).

For Q to be PSD, we need the Hankel matrix [q_{j+l}] to be PSD. This is equivalent to the measure with moments q_k being positive (by the moment problem). If q comes from a probability distribution on {0, 1, ..., 2n-2}, then Q is PSD (it's a moment matrix of a discrete measure).

So the algorithm is:
1. Initialize a (e.g., uniform)
2. Compute b = a*a, find the max
3. Set q = softmax(b / temperature) or q = delta at argmax
4. Solve QP: minimize a^T Q_a subject to a >= 0, sum=1, where Q_a[j,l] = 2*q_{j+l}
5. Update a, go to 2

For step 4, since Q is a Hankel matrix, we can exploit its structure. The QP minimizes a quadratic form defined by a Hankel matrix. This is equivalent to:
minimize 2 * sum_k q_k * (a*a)[k]
subject to a >= 0, sum(a) = 1

This is a convex QP if Q is PSD (which it is if q is a probability distribution, as argued above).

We can solve this with scipy.optimize.minimize (SLSQP or trust-constr) or with a dedicated QP solver.

Actually, since the objective is 2 * sum_k q_k * (a*a)[k], we can compute the gradient as:
grad_j = 2 * sum_k q_k * 2 * a_{k-j} = 4 * sum_k q_k * a_{k-j} (for valid k-j)

This is 4 * (q cross-correlation with a) at specific lags, which can be computed via FFT!

So the gradient of the QP objective is:
grad = 4 * cross_corr(q, a) (evaluated at the right lags)

This is the same structure as the log-sum-exp gradient (with p replaced by q and a factor of 2 difference).

So the QP step is:
minimize 2 * sum_k q_k * (a*a)[k]
subject to a >= 0, sum(a) = 1
with gradient grad_j = 4 * sum_{k: 0≤k-j<n} q_k * a_{k-j}

This can be solved with projected gradient (since the problem is convex when Q is PSD):
a_{t+1} = proj_simplex(a_t - step * grad_t)

Or with FISTA (accelerated projected gradient).

The whole algorithm becomes:
1. a ← initial
2. b ← a*a
3. q ← softmax(b / temp) [or some other way to get q from b]
4. Solve QP (using FISTA on the QP objective) to get new a
5. Repeat

This is essentially the **dual subgradient method** or **bisection method** for the minimax problem!

The choice of q is crucial. If q = delta at argmax, we're doing a single-constraint QP (which is what the previous LP approach was doing, but in a different form). If q = softmax(b/temp), we're doing a smoothed version.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (accelerated proximal gradient) on log-sum-exp with adaptive tau
2. FFT-based gradient and cross-correlation
3. SLSQP active-set polish for final precision
4. Multi-length (384-2048) and multi-initialization exploration
5. High-tau FISTA final polish

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_grad(a, tau):
    """Compute (1/tau)*log(sum(exp(tau*(a*a)))) and its gradient. Returns (obj, grad)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    p = exp_s / exp_s.sum()
    obj = (b_max + max_s / tau + np.log(exp_s.sum()) / tau)

    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] for valid k-j in [0,n)
    # Using FFT cross-correlation: R = IFFT(conj(FFT(p,S)) * FFT(a,S))
    # Then grad[0] = 2*R[0], grad[j] = 2*R[S-j] for j>=1
    S = 1
    while S < 3 * n:
        S <<= 1
    fp = np.fft.rfft(p, S)
    fa = np.fft.rfft(a, S)
    R = np.fft.irfft(np.conj(fp) * fa, S)

    grad = np.zeros(n)
    grad[0] = 2.0 * R[0]
    if n > 1:
        grad[1:] = 2.0 * R[S - 1:S - n + 1:-1]

    return obj, grad


def true_J_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    b = fast_conv(a)
    return float(2 * len(a) * b.max())


def run_fista(a_init, deadline, tau_start=5.0, tau_end=5000.0, num_tau_levels=40,
              max_total_iters=200000, label="", start_time=0):
    """
    FISTA on log-sum-exp with gradually increasing tau.
    a_init: non-negative array (will be normalized to sum=1).
    Returns best a found (normalized to sum=1).
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_val = true_J_normalized(a)

    tau_levels = np.geomspace(tau_start, tau_end, num_tau_levels)
    iters_per_tau = max(200, max_total_iters // num_tau_levels)

    total_iters = 0
    for tau_idx, tau in enumerate(tau_levels):
        if time.monotonic() > deadline or total_iters >= max_total_iters:
            break

        y = a.copy()
        theta = 1.0
        L = 2.0 * n / tau
        L = max(L, 0.5)

        for it in range(iters_per_tau):
            if time.monotonic() > deadline or total_iters >= max_total_iters:
                break
            total_iters += 1

            obj_y, grad_y = lse_grad(y, tau)
            a_new = project_simplex(y - grad_y / L)

            # Backtracking (simplified: one check)
            obj_new, _ = lse_grad(a_new, tau)
            sufficient = obj_new <= obj_y + grad_y @ (a_new - y) + 0.5 * L * np.sum((a_new - a_new - y)**2) + 1e-10
            if not sufficient:
                L *= 2.0
                a_new = project_simplex(y - grad_y / L)
                obj_new, _ = lse_grad(a_new, tau)

            # FISTA momentum update
            theta_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * theta * theta))
            y = a_new + ((theta - 1.0) / theta_new) * (a_new - a)
            a = a_new
            theta = theta_new

            # Track best using true objective
            if it % 10 == 0 or tau_idx == len(tau_levels) - 1:
                val = true_J_normalized(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()

    val = true_J_normalized(a)
    if val < best_val:
        best_val = val
        best_a = a.copy()

    return best_a


def run_slsqp_polish(a_init, deadline, start_time=0, max_rounds=20, label=""):
    """
    Active-set SLSQP polish: minimize max of top-K convolution values.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_val = true_J_normalized(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        # Find active set: indices where b is within 5% of max
        active = np.where(b >= b_max * 0.95)[0]
        if len(active) > 100:
            # Keep top 100
            top_idx = np.argsort(b)[-100:]
            active = np.sort(top_idx)

        # Define smooth objective: log-sum-exp of active values
        tau_smooth = 50.0

        def objective(x):
            x = np.maximum(x, 0)
            sx = x.sum()
            if sx < 1e-10:
                return 1e10
            x = x / sx
            bx = fast_conv(x)
            vals = bx[active]
            vmax = vals.max()
            shifted = tau_smooth * (vals - vmax)
            return float(vmax + np.log(np.exp(shifted).sum()) / tau_smooth)

        def gradient(x):
            x = np.maximum(x, 0)
            sx = x.sum()
            if sx < 1e-10:
                return np.zeros(n)
            x = x / sx
            bx = fast_conv(x)
            vals = bx[active]
            vmax = vals.max()
            shifted = tau_smooth * (vals - vmax)
            max_sh = shifted.max()
            exp_sh = np.exp(shifted - max_sh)
            p_act = exp_sh / exp_sh.sum()

            # Gradient w.r.t. x (before normalization)
            S = 1
            while S < 3 * n:
                S <<= 1
            # Build full p vector
            p_full = np.zeros(2 * n - 1)
            p_full[active] = p_act
            fp = np.fft.rfft(p_full, S)
            fx = np.fft.rfft(x, S)
            R = np.fft.irfft(np.conj(fp) * fx, S)
            grad_raw = np.zeros(n)
            grad_raw[0] = 2.0 * R[0]
            if n > 1:
                grad_raw[1:] = 2.0 * R[S - 1:S - n + 1:-1]
            return grad_raw

        try:
            res = minimize(objective, a, jac=gradient, method='SLSQP',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 300, 'ftol': 1e-12})
            if res.success and res.x is not None:
                a_new = np.maximum(res.x, 0)
                sn = a_new.sum()
                if sn > 1e-10:
                    a_new = a_new / sn
                    val = true_J_normalized(a_new)
                    if val < best_val:
                        best_val = val
                        best_a = a_new.copy()
                        a = a_new
                        print(f"      SLSQP r{rnd}: J={val:.8f} active={len(active)} t={time.monotonic()-start_time:.0f}s {label}", flush=True)
        except Exception:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    print("PHASE 2: SLSQP Active-Set Polish", flush=True)
    print(f"{'='*60}", flush=True)

    phase2_dl = min(t_start + budget_s * 0.45, deadline)
    if tracker.best is not None:
        a_pol = run_slsqp_polish(tracker.best, phase2_dl, start_time=t_start,
                                 max_rounds=15, label="(P2)")
        tracker.update(a_pol, "(P2 SLSQP)")

    # ===== Phase 3: Multi-start with diverse initializations =====
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Multi-Start Diverse Inits", flush=True)
    print(f"{'='*60}", flush=True)

    phase3_dl = min(t_start + budget_s * 0.65, deadline)
    n_ref = tracker.best.size if tracker.best is not None else n_parent
    inits = [
        ("cosine", make_init_cosine(n_ref)),
        ("gauss", make_init_gaussian(n_ref, 0.25)),
        ("flat", make_init_flat(n_ref)),
        ("random", None),
    ]
    for name, a_init in inits:
        if time.monotonic() > phase3_dl:
            break
        if a_init is None:
            a_init = rng.random(n_ref) + 0.1
            a_init = a_init / a_init.sum()
        a_opt = run_fista(a_init, min(time.monotonic() + 60, phase3_dl),
                          tau_start=10, tau_end=5000,
                          num_tau_levels=20, max_total_iters=50000,
                          label=f"{name}", start_time=t_start)
        tracker.update(a_opt, f"(P3 {name})")

    # Also try different lengths from best
    if tracker.best is not None:
        for mult in [0.5, 1.5, 2.0]:
            if time.monotonic() > phase3_dl:
                break
            nl = int(round(tracker.best.size * mult))
            if nl < 128 or nl > 2560:
                continue
            a_init = resample_seq(tracker.best, nl)
            a_opt = run_fista(a_init, min(time.monotonic() + 50, phase3_dl),
                              tau_start=5, tau_end=3000,
                              num_tau_levels=20, max_total_iters=40000,
                              label=f"n={nl}", start_time=t_start)
            tracker.update(a_opt, f"(P3 n={nl})")

    # ===== Phase 4: High-tau FISTA final polish =====
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: High-Tau FISTA Polish", flush=True)
    print(f"{'='*60}", flush=True)

    phase4_dl = min(t_start + budget_s * 0.85, deadline)
    if tracker.best is not None:
        a_init = tracker.best.copy()
        s = a_init.sum()
        a_init = a_init / s
        a_opt = run_fista(a_init, phase4_dl, tau_start=100, tau_end=20000,
                          num_tau_levels=30, max_total_iters=100000,
                          label="(P4)", start_time=t_start)
        tracker.update(a_opt, "(P4 high-tau)")

    # ===== Phase 5: Final SLSQP =====
    print(f"\n{'='*60}", flush=True)
    print("PHASE 5: Final SLSQP", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline:
        a_pol = run_slsqp_polish(tracker.best, deadline, start_time=t_start,
                                 max_rounds=20, label="(P5)")
        tracker.update(a_pol, "(P5 SLSQP)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
