Given f(t) = max_k q_k(t) where q_k(t) = A_k + B_k*t + C_k*t^2, we want to find t* = argmin f(t) for t in [0,1].

Approach:

2. Find the interval [t_i, t_{i+1}] containing the minimum.
3. Within that interval, the active constraint (argmax) might change at most a few times.
4. For each sub-interval with a fixed active constraint, minimize the quadratic exactly.
5. Pick the best overall.

Step 4 is the key. For a fixed active constraint k, the minimum of q_k(t) = A_k + B_k*t + C_k*t^2 is at t = -B_k/(2*C_k) if C_k > 0.

But the active constraint might not be the one that gives the minimum of its own quadratic (because at that t, another constraint might become active and have a higher value).

A practical approach:
1. Evaluate f at several points (say 20 points evenly spaced in [0,1]).
2. Find the point with the minimum f.
3. Do a local refinement:

   b. Minimize that quadratic over the neighborhood.
   c. Check if the solution is valid (the active constraint is still the max at the new point).

(a*g)[k] = sum_{j: 0<=k-j<n} a[j] * g[k-j]

So the coefficient of g[i] in the k-th constraint is a[k-i] (if 0 <= k-i < n).

This means the constraint matrix A has A[k, i] = a[k-i] if 0 <= k-i < n, else 0.

For each k, the non-zero entries are g[max(0, k-n+1)] through g[min(n-1, k)].
The number of non-zero entries per row is min(k+1, n, 2n-1-k).
Total non-zeros: sum_{k=0}^{2n-2} min(k+1, n, 2n-1-k) = n^2 (approximately).

For n=512, that's about 262,000 non-zeros. The LP should handle this fine.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-scale optimization (optimize at small n first, then resample up)
2. LP mirror descent with fast bisection line search
3. Smooth gradient descent for initialization and polish
4. Multi-length exploration with resampling
5. Symmetry exploitation and perturbation restarts
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_indices(n):
    """Precompute row/col indices for the LP constraint matrix A[k,i] = a[k-i]."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp_direction(a, M, row_idx, col_idx, time_limit=3.0):
    """
    Solve LP: max 1^T g s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns g (unnormalized) or None.
    """
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_bisect(a, g):
    """
    Find t in [0,1] minimizing max_k conv((1-t)*a + t*g, (1-t)*a + t*g).
    Uses bisection on the subgradient (60 iterations for high precision).
    """
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = A - 2.0 * B + C

    lo = 0.0
    hi = 1.0
    for _ in range(60):
        t = 0.5 * (lo + hi)
        vals = a_coef + b_coef * t + c_coef * t * t
        idx = np.argmax(vals)
        grad = b_coef[idx] + 2.0 * c_coef[idx] * t
        if grad < 0:
            lo = t
        else:
            hi = t

    t = 0.5 * (lo + hi)
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))

    # Try exact minimization of each candidate active constraint
    # Check a few top constraints
    top_k = np.argsort(a_coef + b_coef * t + c_coef * t * t)[-5:]
    for idx in top_k:
        if c_coef[idx] > 1e-14:
            t_exact = -b_coef[idx] / (2.0 * c_coef[idx])
            if 0.0 <= t_exact <= 1.0:
                val_exact = float(np.max(a_coef + b_coef * t_exact + c_coef * t_exact * t_exact))
                if val_exact < best_val:
                    best_val = val_exact
                    t = t_exact

    return t, best_val


def smooth_gradient_step(a, tau):
    """Compute gradient of smooth max approximation (1/tau)*log(sum exp(tau*conv))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    scaled = tau * (b - b_max)
    scaled -= scaled.max()
    exp_b = np.exp(scaled)
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * pk * a[k - j]

    return grad


def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0):
    """Smooth gradient descent with adaptive tau and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * frac
        lr = 0.5 * (0.02 / 0.5) ** frac

        grad = smooth_gradient_step(a, tau)
        v = 0.85 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()

    return best_a


def run_lp_mirror_descent(a, row_idx, col_idx, deadline, verbose=False, max_iters=100000):
    """
    LP mirror descent with bisection line search.
    Input a should be non-negative, normalized to sum 1.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = float(fast_conv(a).max())
        g = solve_lp_direction(a, M, row_idx, col_idx, time_limit=3.0)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        # Check if g is essentially the same as a (stalled)
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        t, M_new = line_search_bisect(a, g)

        if t < 1e-8:
            no_improve += 1
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                actual_M = float(fast_conv(a_new).max())
                if actual_M < best_M * (1.0 - 1e-12):
                    a = a_new
                    best_M = actual_M
                    best_a = a_new.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 200:
            # Perturb to escape
            rng_local = np.random.default_rng(it)
            a = a + rng_local.normal(0, 0.001, n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - (deadline - 600)  # approximate
            print(f"        iter {it}: M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def gaussian_init(n, width):
    """Generate a Gaussian-like initialization."""
    center = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a / a.sum()
    return a
