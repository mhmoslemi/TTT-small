def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    """Precompute the structure (row/col indices) of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx, time_limit=5):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2 (both normalized)."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False, lp_time=5):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx, time_limit=lp_time)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-15:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            break

        if verbose and it % 500 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def project_nonneg_sum1(v):
    """Project v onto {x >= 0, sum(x) = 1} via bisection on threshold."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    lo = v.min() - 1e-10
    hi = v.max() + 1e-10
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.maximum(v - mid, 0)
        s = projected.sum()
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    projected = np.maximum(v - (lo + hi) / 2, 0)
    s = projected.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return projected / s


def subgradient_of_max_conv(a):
    """Compute subgradient of max_k (a*a)[k] at point a.
    Returns the average subgradient over the active set."""
    n = len(a)
    conv = fast_conv(a)
    max_val = conv.max()
    # Active set with tolerance
    active = conv >= max_val - 1e-10 * max(1.0, abs(max_val))
    n_active = active.sum()
    if n_active == 0:
        active = np.zeros(2 * n - 1, dtype=bool)
        active[np.argmax(conv)] = True
        n_active = 1

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if not active[k]:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * a[k - j]
    grad /= n_active
    return grad, max_val


def run_nesterov_subgrad(a, max_iters, deadline, verbose=False):
    """Nesterov-accelerated subgradient method on max(conv(a,a))."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    x_prev = a.copy()
    x_curr = a.copy()
    alpha = 1.0

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break

        alpha_next = (1.0 + np.sqrt(1.0 + 4.0 * alpha * alpha)) / 2.0
        momentum = (alpha - 1.0) / alpha_next

        y = x_curr + momentum * (x_curr - x_prev)
        y = np.maximum(y, 0.0)
        s = y.sum()
        if s < 1e-10:
            y = np.ones(n) / n
        else:
            y = y / s

        grad, M_y = subgradient_of_max_conv(y)

        # Step size: eta = 2 / (L * it) where L ~ 4 for normalized sequences
        eta = 2.0 / (4.0 * it)
        x_new = y - eta * grad
        x_new = project_nonneg_sum1(x_new)

        x_prev = x_curr
        x_curr = x_new
        alpha = alpha_next

        M_curr = compute_M(x_curr)
        if M_curr < best_M - 1e-14:
            best_M = M_curr
            best_a = x_curr.copy()

        if verbose and it % 500 == 0:
            print(f"      nesterov iter {it}: M={M_curr:.10f} best={best_M:.10f}", flush=True)

    return best_a


def smooth_grad_with_adam(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0, verbose=False):
    """Smooth gradient descent with Adam optimizer and adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.7)
        lr = base_lr * (0.1 + 0.9 * (1.0 - frac))

        # Compute smooth gradient
        b = fast_conv(a)
        b_max = b.max()
        shifted = tau * (b - b_max)
        # Clip for numerical stability
        shifted = np.clip(shifted, -500, 500)
        exp_b = np.exp(shifted)
        p = exp_b / exp_b.sum()

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()

        if verbose and it % 500 == 0:
            print(f"      adam iter {it}: M={M:.10f} best={best_M:.10f} tau={tau:.1f}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_simulated_annealing(a_best, n, r_idx, c_idx, deadline, rng, verbose=False):
    """Simulated annealing with LP polish on accepted moves."""
    best_a = a_best.copy()
    best_M = compute_M(a_best)
    current_M = best_M
    current_a = best_a.copy()

    T = 0.005
    T_min = 0.0001
    cooling = 0.95
    max_iter = 200

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Generate perturbation
        noise = rng.normal(0, 1, n) * T * 0.1
        a_pert = current_a + noise
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        M_pert = compute_M(a_pert)

        # Acceptance criterion
        if M_pert < current_M:
            accept = True
        else:
            delta = M_pert - current_M
            if delta < 1e-15:
                accept = True
            else:
                accept_val = np.exp(-delta / max(T, 1e-15))
                accept = rng.random() < accept_val

        if accept:
            # Quick LP polish (few iterations)
            sub_dl = min(time.monotonic() + 5, deadline)
            a_polished = run_lp_mirror_descent(a_pert, 200, sub_dl, r_idx, c_idx, lp_time=3)
            M_polished = compute_M(a_polished)
            current_a = a_polished
            current_M = M_polished

            if current_M < best_M:
                best_M = current_M
                best_a = current_a.copy()
                if verbose:
                    print(f"      SA iter {it}: new best M={best_M:.10f} T={T:.6f}", flush=True)

        T = max(T * cooling, T_min)

    return best_a


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: LP mirror descent on parent (quick)
    Phase 2: Nesterov subgradient + Adam smooth gradient
    Phase 3: Multi-length exploration (nearby lengths)
    Phase 4: Extended LP polish
    Phase 5: Simulated annealing perturbations
    Phase 6: Final LP polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    # Phase 1: LP mirror descent on parent (quick polish)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent (quick) n={parent.size}", flush=True)
    print(f"{'='*60}", flush=True)

    n0 = parent.size
    row_idx0, col_idx0 = build_lp_matrix(n0)
    a = parent.copy()
    a = a / a.sum()

    phase1_deadline = min(t_start + budget_s * 0.20, deadline)
    a_opt = run_lp_mirror_descent(a, 30000, phase1_deadline, row_idx0, col_idx0, verbose=True, lp_time=5)
    tracker.update(a_opt, "(P1 LP-MD)")

    # Phase 2: Nesterov subgradient + Adam smooth gradient
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Nesterov Subgradient + Adam Gradient", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_cur = tracker.best.size
        if n_cur != n0:
            row_idx2, col_idx2 = build_lp_matrix(n_cur)
        else:
            row_idx2, col_idx2 = row_idx0, col_idx0

        # Nesterov subgradient
        a_nest = tracker.best.copy()
        a_nest = a_nest / a_nest.sum()
        phase2a_deadline = min(t_start + budget_s * 0.35, deadline)
        a_nest_opt = run_nesterov_subgrad(a_nest, 5000, phase2a_deadline, verbose=True)
        tracker.update(a_nest_opt, "(P2a Nesterov)")

        # LP polish after Nesterov
        sub_dl = min(t_start + budget_s * 0.42, deadline)
        a_lp_after_nest = run_lp_mirror_descent(a_nest_opt, 5000, sub_dl, row_idx2, col_idx2, lp_time=5)
        tracker.update(a_lp_after_nest, "(P2a LP after Nesterov)")

        # Adam smooth gradient
        a_adam = tracker.best.copy()
        a_adam = a_adam / a_adam.sum()
        phase2b_deadline = min(t_start + budget_s * 0.50, deadline)
        a_adam_opt = smooth_grad_with_adam(a_adam, 5000, phase2b_deadline,
                                           initial_tau=3.0, final_tau=200.0, verbose=True)
        tracker.update(a_adam_opt, "(P2b Adam)")

        # LP polish after Adam
        sub_dl2 = min(t_start + budget_s * 0.55, deadline)
        a_lp_after_adam = run_lp_mirror_descent(a_adam_opt, 5000, sub_dl2, row_idx2, col_idx2, lp_time=5)
        tracker.update(a_lp_after_adam, "(P2b LP after Adam)")

    # Phase 3: Multi-length exploration (nearby lengths)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration (nearby)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        # Try lengths within +/- 8 of base, plus a few wider
        trial_lengths = []
        for delta in [-8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8]:
            n_trial = base_n + delta
            if 64 <= n_trial <= 4096:
                trial_lengths.append(n_trial)
        # Also try some wider exploration
        for mult in [0.9, 1.1]:
            n_trial = int(base_n * mult)
            if 64 <= n_trial <= 4096 and n_trial not in trial_lengths:
                trial_lengths.append(n_trial)

        trial_lengths = sorted(set(trial_lengths))
        phase3_deadline = min(t_start + budget_s * 0.70, deadline)

        for n_trial in trial_lengths:
            if time.monotonic() > phase3_deadline:
                break

            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()

            r_idx, c_idx = build_lp_matrix(n_trial)
            sub_deadline = min(time.monotonic() + 30, phase3_deadline)
            a_opt = run_lp_mirror_descent(a_init, 3000, sub_deadline, r_idx, c_idx, lp_time=3)
            tracker.update(a_opt, f"(P3 n={n_trial})")

    # Phase 4: Extended LP polish on best
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Extended LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_idx_b, c_idx_b = build_lp_matrix(n_best)
        a = tracker.best.copy()
        a = a / a.sum()
        phase4_deadline = min(t_start + budget_s * 0.82, deadline)
        a_opt = run_lp_mirror_descent(a, 50000, phase4_deadline, r_idx_b, c_idx_b, verbose=True, lp_time=5)
        tracker.update(a_opt, "(P4 LP extended)")

    # Phase 5: Simulated annealing perturbations
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Simulated Annealing", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_idx_b, c_idx_b = build_lp_matrix(n_best)
        phase5_deadline = min(t_start + budget_s * 0.92, deadline)
        a_sa = run_simulated_annealing(tracker.best, n_best, r_idx_b, c_idx_b,
                                       phase5_deadline, rng, verbose=True)
        tracker.update(a_sa, "(P5 SA)")

    # Phase 6: Final LP polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)
        a = tracker.best.copy()
        a = a / a.sum()
        a_opt = run_lp_mirror_descent(a, 50000, deadline, r_f, c_f, verbose=True, lp_time=8)
        tracker.update(a_opt, "(P6 final)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
