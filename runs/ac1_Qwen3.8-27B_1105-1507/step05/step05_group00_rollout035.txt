```python
"""
Optimized step function search using smooth max gradient descent in softmax
parameterization with Nesterov momentum, adaptive temperature annealing,
and periodic LP-based refinement. Targets improvement over parent sequence.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fft_conv(a, b=None):
    """Linear convolution using FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_obj(p):
    """Compute objective 2*n*max(conv(p,p)) for p with sum=1."""
    n = len(p)
    conv = fft_conv(p)
    return float(2.0 * n * np.max(conv))


def softmax(x):
    x = x - np.max(x)
    e = np.exp(x)
    return e / e.sum()


def build_T(w, n):
    """Build gradient matrix T where (T@p)[i] = sum_k w[k]*(p[k-i]+p[i-k])."""
    rows = np.arange(n)[:, None]
    cols = np.arange(n)[None, :]
    T1 = w[rows + cols]
    mask = rows >= cols
    T2 = np.where(mask, w[rows - cols], 0.0)
    return T1 + T2


def gradient_obj(x, tau):
    """Compute gradient of smoothed objective w.r.t. x (softmax params)."""
    n = len(x)
    p = softmax(x)
    conv = fft_conv(p)
    cmax = np.max(conv)
    w = np.exp(np.clip(tau * (conv - cmax), -500, 500))
    w /= w.sum()
    
    T = build_T(w, n)
    grad_p = T @ p
    
    # Softmax chain rule: d/smooth_max/dx_j = sum_i grad_p[i] * dp[i]/dx[j]
    # dp[i]/dx[j] = p[i]*(delta_ij - p[j])
    # So d/dx_j = p[j]*grad_p[j] - p[j]*sum_i grad_p[i]*p[i]
    dot = np.dot(grad_p, p)
    grad_x = p * (grad_p - dot)
    return 2.0 * n * grad_x, p, conv


def optimize_nesterov(p0, time_budget, tau0=50.0, tau1=500.0, lr_init=0.5, verbose=False):
    """
    Nesterov-accelerated gradient descent with adaptive step size.
    Optimizes in softmax parameterization.
    """
    n = len(p0)
    x = np.log(p0 + 1e-30)
    v = np.zeros(n)
    mu = 0.85
    lr = lr_init
    
    best_p = p0.copy()
    best_obj = compute_obj(p0)
    start = time.monotonic()
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        tau = tau0 * (tau1 / tau0) ** (it / max(100, int(time_budget)))
        
        grad_x, p_curr, conv = gradient_obj(x, tau)
        
        v = mu * v - lr * grad_x
        x_new = x + v
        
        p_new = softmax(x_new)
        obj_new = compute_obj(p_new)
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_p = p_new.copy()
            x = x_new
            stall = 0
        else:
            x = x_new  # Still update x (momentum might help escape)
            stall += 1
        
        if stall > 20:
            # Reset momentum and try smaller step
            v = np.zeros(n)
            lr *= 0.5
            mu = 0.8
            stall = 0
        
        if it % 100 == 0:
            lr = max(lr * 0.95, 0.01)
    
    return best_p, best_obj
```
