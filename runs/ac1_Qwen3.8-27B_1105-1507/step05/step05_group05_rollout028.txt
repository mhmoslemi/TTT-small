            slopes[idx] = 0.0

            slopes[idx] = 2 * a[diff]

            slopes[idx] = 0.0

        if new_sum <= 1e-10:
            return 1e10

        m = base_max
        if vals.max() > m:
            m = vals.max()
        if quad_pos >= 0:
            qv = c_vals[quad_pos] + x * x
            if qv > m:
                m = qv
        return m / (new_sum * new_sum)
    
    lo = 0.0

            fb = fa

            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)

    x_opt = (lo + hi) / 2.0

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated smooth gradient descent with adaptive tau annealing
2. Incremental convolution updates for exact coordinate descent (O(n) per update)
3. Multi-length search with diverse initializations including KKT-motivated shapes
4. Periodic interleaving of gradient and coordinate methods
5. Random restarts to escape local minima

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad_fft(a, tau):
    """Compute gradient of smooth max (log-sum-exp) of self-convolution."""
    n = len(a)
    b = fast_conv(a)
    bmax = b.max()
    e = np.exp(tau * (b - bmax))
    p = e / e.sum()
    N = 1
    while N < 3 * n:
        N <<= 1
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(P * np.conj(A), N)
    grad = 2.0 * cross[:n]
    smooth_val = bmax + np.log(e.sum()) / tau
    return grad, smooth_val


def coord_step(a, b, i):
    """Find optimal a[i] via golden section, given current b. Returns new value."""
    n = len(a)
    x0 = a[i]
    S = a.sum()
    klo = i
    khi = min(2 * n - 2, i + n - 1)
    width = khi - klo + 1

    mask = np.ones(2 * n - 1, dtype=bool)
    mask[klo:khi + 1] = False
    bmax_others = b[mask].max() if mask.any() else 0.0

    c = np.zeros(width)
    s = np.zeros(width)
    qpos = -1
    for idx in range(width):
        k = klo + idx
        d = k - i
        if k == 2 * i:
            c[idx] = b[k] - x0 * x0
            s[idx] = 0.0
            qpos = idx
        elif 0 <= d < n:
            c[idx] = b[k] - 2.0 * x0 * a[d]
            s[idx] = 2.0 * a[d]
        else:
            c[idx] = b[k]
            s[idx] = 0.0

    def f(x):
        ns = S - x0 + x
        if ns < 1e-10:
            return 1e10
        m = bmax_others
        t = c + s * x
        tm = t.max()
        if tm > m:
            m = tm
        if qpos >= 0:
            qv = c[qpos] + x * x
            if qv > m:
                m = qv
        return m / (ns * ns)

    lo = 0.0
    hi = max(x0 * 3.0, np.sqrt(max(b.max(), 1e-10)) + 0.5, 0.05)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a1 = hi - gr * (hi - lo)
    b1 = lo + gr * (hi - lo)
    fa = f(a1)
    fb = f(b1)

            hi = b1
            b1 = a1
            fb = fa
            a1 = hi - gr * (hi - lo)
            fa = f(a1)
        else:
            lo = a1
            a1 = b1
            fa = fb
            b1 = lo + gr * (hi - lo)
            fb = f(b1)
    xopt = (lo + hi) / 2.0
    if f(0.0) < f(xopt):
        xopt = 0.0
    return max(0.0, xopt)


def update_b_inplace(b, a, i, x_new):
    """Incrementally update b after changing a[i] to x_new. Modifies b in place."""
    n = len(a)
    x_old = a[i]
    dx = x_new - x_old
    if abs(dx) < 1e-15:
        return
    for k in range(i, min(2 * n - 1, i + n)):

            if k == 2 * i:
                b[k] += x_new * x_new - x_old * x_old
            else:
                b[k] += 2.0 * dx * a[d]
    a[i] = x_new


def run_coord_descent(a, max_sweeps, deadline, seed=0):
    """Coordinate descent with incremental b updates."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
