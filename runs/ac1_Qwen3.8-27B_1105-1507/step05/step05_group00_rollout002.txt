import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

            s = float(np.sum(g))

                best_a_local = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_a_local.copy()
            a = best_a_local
            last_t = best_t_val
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Perturb and restart
                rng = np.random.default_rng(it * 7919 + 12345)
                scale = 0.03 + 0.02 * (stall // 10)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
                best_a = a.copy()
                best_obj = compute_obj(a)

    return best_a


def subgradient_adaptive_temp(a, max_iters=300, time_budget=30.0):
    """
    Subgradient with adaptive temperature: start hot (all peaks matter),
    cool down to focus on max peak. Uses Nesterov momentum.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.85
    step = 0.08

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))

        # Adaptive temperature: starts large (consider many peaks), shrinks
        progress = it / max(max_iters, 1)
        tau = cmax * (0.3 * (1 - progress) + 0.02)

        shifted = np.clip((conv - cmax) / (tau + 1e-15), -300.0, 300.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient computation via FFT
        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

        grad_centered = grad_a - np.mean(grad_a)

        # Nesterov momentum
        v = mu * v - step * grad_centered
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        step *= 0.998

    return best_a


def block_coordinate_moves(a, time_budget=20.0):
    """
    Try block-level moves: swap or blend adjacent blocks to escape local optima.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Try different block sizes
    for block_size in [n // 4, n // 8, max(5, n // 16)]:
        if time.monotonic() - start > time_budget:
            break

        n_blocks = n // block_size
        for _ in range(20):
            if time.monotonic() - start > time_budget:
                break

            rng = np.random.default_rng()
            b1 = rng.integers(0, n_blocks)
            b2 = rng.integers(0, n_blocks)
            if b1 == b2:
                continue

            a_try = a.copy()
            # Swap or blend blocks
            for blend in [0.3, 0.5, 0.7, 1.0]:
                a_try = a.copy()
                ks1 = b1 * block_size
                ks2 = b2 * block_size
                block1 = a_try[ks1:ks1 + block_size].copy()
                block2 = a_try[ks2:ks2 + block_size].copy()
                a_try[ks1:ks1 + block_size] = (1 - blend) * block1 + blend * block2
                a_try[ks2:ks2 + block_size] = (1 - blend) * block2 + blend * block1
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def generate_diverse_starts(n, parent=None, num_starts=8):
    """Generate diverse starting sequences of length n."""
    starts = []

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            starts.append(parent.copy())
        else:
            starts.append(resample_sequence(parent, n))

    # Cosine-modulated
    for freq in [1.0, 2.0, 3.0, 5.0]:
        x = np.arange(n)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * x / n)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
        starts.append(a)

    # Sine-modulated (asymmetric)
    x = np.arange(n)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * 2.0 * x / n + np.pi / 4)
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    starts.append(a)

    # Chirp-like (quadratic phase modulation)
    x = np.arange(n)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * (np.pi * x ** 2 / n + x / n))
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    starts.append(a)

    # Block alternating
    n_blocks = 8
    block_len = n // n_blocks
    rng = np.random.default_rng(42)
    a = np.zeros(n)
    for b in range(n_blocks):
        val = 0.5 + rng.uniform(0, 1.5)
        a[b * block_len:(b + 1) * block_len] = val
    if n % n_blocks != 0:
        a[-(n % n_blocks):] = a[-1]
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    starts.append(a)

    # Random
    rng = np.random.default_rng(123)
    a = rng.uniform(0.1, 1.0, n)
    a = a / np.sum(a)
    starts.append(a)

    # Gaussian-modified
    x = np.linspace(-1, 1, n)
    a = 1.0 + 0.5 * np.exp(-4 * x ** 2)
    a = a / np.sum(a)
    starts.append(a)

    return starts[:num_starts]


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = np.sum(parent)
            if s > 1e-10:
                parent = parent / s
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_single(a, time_budget=60.0, n_max_lp=800):
    """
    Multi-method optimization on a single sequence.
    Returns the best sequence found.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Quick subgradient pass
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5 and n <= 2000:
        a_sg = subgradient_adaptive_temp(a, max_iters=int(remaining * 8),
                                         time_budget=min(remaining * 0.2, 15.0))
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            a = a_sg.copy()

    # LP alternating (if n is manageable)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10 and n <= n_max_lp:
        a_lp = adaptive_lp_alternate(a, max_iters=30,
                                     time_budget=min(remaining * 0.4, 30.0),
                                     use_weighted=True)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            a = a_lp.copy()

    # Another subgradient pass after LP
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_sg2 = subgradient_adaptive_temp(a, max_iters=int(remaining * 10),
                                          time_budget=min(remaining * 0.3, 15.0))
        obj_sg2 = compute_obj(a_sg2)
        if obj_sg2 < best_obj:
            best_obj = obj_sg2
            best_a = a_sg2.copy()

    # Block moves
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_bm = block_coordinate_moves(best_a, time_budget=min(remaining * 0.3, 10.0))
        obj_bm = compute_obj(a_bm)
        if obj_bm < best_obj:
            best_obj = obj_bm
            best_a = a_bm.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses diverse multi-start strategy with adaptive
    time allocation, multi-length exploration, and intensive final polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 500
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}, time_left={deadline - time.monotonic():.0f}s")

    # Phase 1: Multi-start optimization at parent length
    print(f"\n=== Phase 1: Multi-start at n={n_parent} ===")
    starts = generate_diverse_starts(n_parent, parent, num_starts=10)

    # First pass: quick optimization to rank starts
    start_objs = []
    for i, s in enumerate(starts):
        elapsed = time.monotonic() - start_time
        if elapsed > budget_s * 0.3:
            break
        obj = compute_obj(s)
        start_objs.append((obj, i, s))
        print(f"  Start {i}: obj={obj:.6f}")

    # Sort by objective and optimize the best ones
    start_objs.sort(key=lambda x: x[0])
    num_to_optimize = min(6, len(start_objs))

    for rank in range(num_to_optimize):
        elapsed = time.monotonic() - start_time
        if elapsed > budget_s * 0.7:
            break

        obj_init, idx, a_init = start_objs[rank]
        alloc = min(40.0, (deadline - time.monotonic()) * 0.15)

        print(f"  Optimizing start {rank} (init obj={obj_init:.6f}), alloc={alloc:.0f}s")
        a_opt = optimize_single(a_init, time_budget=alloc, n_max_lp=900)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: obj={best_score:.8f} (start {rank}) **")

    # Phase 2: Multi-length exploration
    print(f"\n=== Phase 2: Multi-length exploration ===")
    print(f"Best so far: {best_score:.8f}")

    n_options = []
    # Focus around parent length but also try significantly different lengths
    for delta in [-200, -150, -100, -50, 0, 50, 100, 150, 200, 300]):
        n_try = n_parent + delta
        if 100 <= n_try <= 1200 and n_try not in n_options:
            n_options.append(n_try)
    n_options.sort()

    for n_try in n_options:
        elapsed = time.monotonic() - start_time
        if elapsed > budget_s - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n_try == n_parent:
            a_try = best_seq.copy()
        else:
            a_try = resample_sequence(best_seq, n_try)

        alloc = min(30.0, remaining * 0.15)
        a_opt = optimize_single(a_try, time_budget=alloc, n_max_lp=900)
        obj_opt = compute_obj(a_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n_try}, obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_opt:.6f} (no improvement)")

    # Phase 3: Intensive final polish
    print(f"\n=== Phase 3: Intensive final polish ===")
    print(f"Best before polish: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)
        a_final = best_seq.copy()
        s = np.sum(a_final)
        if s > 1e-10:
            a_final = a_final / s

        for round_idx in range(10):
            if time.monotonic() > deadline - 20:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            alloc = min(40.0, remaining * 0.2)

            # Alternate between LP and subgradient
            if round_idx % 2 == 0 and n_best <= 900:
                a_new = adaptive_lp_alternate(a_final, max_iters=40,
                                              time_budget=alloc, use_weighted=True)
            else:
                a_new = subgradient_adaptive_temp(a_final, max_iters=int(alloc * 12),
                                                  time_budget=alloc)

            obj_new = compute_obj(a_new)
            if obj_new < best_score - 1e-12:
                best_score = obj_new
                best_seq = a_new.copy()
                a_final = a_new.copy()
                print(f"  Round {round_idx}: {best_score:.8f}")
            else:
                # Try perturbation
                rng = np.random.default_rng(round_idx * 1000 + seed)
                a_pert = a_final * np.exp(rng.normal(0, 0.02, n_best))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    a_pert = subgradient_adaptive_temp(a_pert, max_iters=200,
                                                       time_budget=min(alloc * 0.5, 15.0))
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_final = a_pert.copy()
                        print(f"  Pert round {round_idx}: {best_score:.8f}")

    # Phase 4: Final LP polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_final = len(best_seq)
        if n_final <= 800:
            remaining = deadline - time.monotonic()
            if remaining > 20:
                print(f"\n=== Phase 4: Final LP (n={n_final}) ===")
                a_lp_final = adaptive_lp_alternate(best_seq, max_iters=50,
                                                   time_budget=min(remaining * 0.6, 50.0),
                                                   use_weighted=True)
                obj_final = compute_obj(a_lp_final)
                if obj_final < best_score:
                    best_score = obj_final
                    best_seq = a_lp_final.copy()
                    print(f"  Final LP: {best_score:.8f}")

    # Ensure we have a valid result
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 200
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
