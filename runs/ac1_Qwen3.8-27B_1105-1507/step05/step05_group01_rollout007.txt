Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternation, subgradient descent, spectral shaping, and multi-length search.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

def solve_lp(a, M, active_idx=None, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1

        A_full = build_toeplitz(a)
        A_sub = A_full[active_idx, :]
        b_sub = np.full(len(active_idx), M * relax)
    else:
        A_sub = build_toeplitz(a)
        b_sub = np.full(m, M * relax)
    
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_sub, bounds=bounds,
                      method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            duals = None
            if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
                duals = res.ineqlin.marginals
            return g, s, duals

        pass
    return None, 0.0, None
