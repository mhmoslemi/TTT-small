```python
def smooth_max_and_grad(a, tau):
    """Compute smooth max value and gradient."""
    n = len(a)
    c = compute_conv(a)  # length 2n-1
    c_shifted = c - c.max()
    exp_c = np.exp(tau * c_shifted)
    p = exp_c / exp_c.sum()
    smooth_val = c.max() + np.log(exp_c.sum()) / tau
    
    # Gradient
    # grad[i] = 2 * (a * p[::-1])[2n-2-i] for i in [0, n-1]
    size = 1
    while size < 3*n-2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p[::-1], size)
    conv = np.fft.irfft(fa * fp, size)[:3*n-2]
    
    # grad[i] = 2 * conv[2n-2-i]
    idx = 2*n-2 - np.arange(n)
    grad = 2.0 * conv[idx]
    
    return smooth_val, grad, p, c
```

```python
def adam_smooth_descent(a_init, deadline, tau_schedule=None, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    while time.monotonic() < deadline:
        t += 1
        
        # Adaptive tau
        if tau_schedule is not None:
            tau = tau_schedule(t)
        else:
            tau = min(50.0, 5.0 + t * 0.1)
        
        _, grad, p, c = smooth_max_and_grad(a, tau)
        M = c.max()
        current_J = 2.0 * n * M  # since sum(a)=1
        
        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        a = a_new
    
    return best_a, best_J
```

```python
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using Adam-based optimization

def fast_self_conv(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_self_conv(a)
    return float(2.0 * len(a) * c.max() / (s * s))

def smooth_max_grad(a, tau):
    """
    Compute gradient of smooth max of conv(a,a) w.r.t. a.
    Returns (smooth_value, gradient).
    """
    n = len(a)
    c = fast_self_conv(a)
    c_max = c.max()
    shifted = tau * (c - c_max)
    # For numerical stability, subtract max from exponent
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()
    smooth_val = c_max + np.log(exp_c.sum()) / tau
    
    # Gradient: grad[i] = 2 * sum_{m=0}^{n-1} a[m] * p[i+m]
    # Computed via FFT: grad[i] = 2 * (a * p[::-1])[2n-2-i]
    p_flipped = p[::-1]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_flipped, size)
    conv = np.fft.irfft(fa * fp, size)[:3*n-2]
    
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv[idx]
    
    return smooth_val, grad, c

    s = a.sum()
    if s < 1e-10:

            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

def resample_seq(src, dest_n):
```

```python
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_self_conv(a)
    return float(2.0 * len(a) * c.max() / (s * s))


def smooth_max_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Returns (smooth_value, gradient, raw_convolution).
    """
    n = len(a)
    c = fast_self_conv(a)
    c_max = c.max()
    shifted = tau * (c - c_max)
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()
    smooth_val = c_max + np.log(exp_c.sum()) / tau

    # Gradient: grad[i] = 2 * sum_{m=0}^{n-1} a[m] * p[i+m]
    # Computed via: grad[i] = 2 * (a * p[::-1])[2n-2-i]
    p_flipped = p[::-1]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_flipped, size)
    conv = np.fft.irfft(fa * fp, size)[:3 * n - 2]

    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv[idx]

    return smooth_val, grad, c


def exact_subgradient(a):
    """Compute subgradient of max_k conv(a,a)[k] w.r.t. a."""
    n = len(a)
    c = fast_self_conv(a)
    k_star = c.argmax()
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for i in range(lo, hi + 1):
        grad[i] = 2.0 * a[k_star - i]
    return grad, c, k_star


def adam_optimize(a_init, deadline, tau_start=5.0, tau_end=80.0,
                  lr=0.008, beta1=0.9, beta2=0.999, eps=1e-8,
                  restart_interval=400, max_restarts=200,
                  use_multiplicative=False, verbose=False):
    """
    Adam optimization with smooth max gradient and periodic restarts.
    Uses either additive (standard Adam) or multiplicative (entropy mirror) updates.
    """
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = 2.0 * n * fast_self_conv(a).max()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    n_restarts = 0
    last_print = 0

    while time.monotonic() < deadline and n_restarts < max_restarts:
        t += 1

        # Adaptive tau: increase over time for sharper max approximation
        progress = min(1.0, t / 3000.0)
        tau = tau_start + (tau_end - tau_start) * progress

        smooth_val, grad, c = smooth_max_grad(a, tau)
        M = c.max()
        current_J = 2.0 * n * M

        if current_J < best_J - 1e-13:
            best_J = current_J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (t - last_print) > 500:
            print(f"      t={t} J={best_J:.8f} tau={tau:.1f} restarts={n_restarts}", flush=True)
            last_print = t

        if use_multiplicative:
            # Mirror descent with entropy: a[i] <- a[i] * exp(-lr * grad[i])
            grad_norm = grad / (np.linalg.norm(grad) + eps)
            a_new = a * np.exp(-lr * grad_norm)
            a_new = np.maximum(a_new, 1e-15)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
            a = a_new
        else:
            # Standard Adam
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            update = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - update
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
            a = a_new

        # Restart logic
        if no_improve >= restart_interval:
            n_restarts += 1
            rng = np.random.default_rng(t * 31 + n_restarts * 7919)
            # Structured perturbation: mix with random smooth component
            scale = 0.08 / (1.0 + 0.3 * n_restarts)

            # Generate smooth perturbation
            freqs = rng.integers(1, 6, size=3)
            phases = rng.uniform(0, 2 * np.pi, size=3)
            amps = rng.uniform(0.3, 1.0, size=3)
            t_arr = np.linspace(0, 1, n, endpoint=False)
            pert = np.zeros(n)
            for f, ph, amp in zip(freqs, phases, amps):
                pert += amp * np.cos(2.0 * np.pi * f * t_arr + ph)

            a = a + scale * pert
            a = np.maximum(a, 1e-6)
            a = a / a.sum()

            # Reset moments
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

            # Occasionally do a larger jump
            if n_restarts % 10 == 0:
                large_scale = 0.2 / (1.0 + 0.2 * n_restarts)
                a = a * (1.0 + large_scale * rng.normal(0, 1, n))
                a = np.maximum(a, 1e-6)
                a = a / a.sum()

    return best_a, best_J


def lp_direction(a, M):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns the optimal g or None.
    """
    n = len(a)
    # Build sparse convolution matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, np.inf)] * n

    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_polish(a, deadline, max_iters=200):
    """
    LP-based mirror descent polishing: use LP direction for updates.
    Good for final refinement when Adam is stuck.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_self_conv(a)
        M = c.max()

        g = lp_direction(a, M)
        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()

        # Try several step sizes
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            break

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, parent=None, seed=42):
    """Generate diverse initial sequences of length n."""
    rng = np.random.default_rng(seed)
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        p = p / p.sum()
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        # Smoothed parent
        k = max(3, n // 32)
        kern = np.ones(k) / k
        inits.append(np.maximum(np.convolve(p, kern, 'same'), 0.01))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine bell (centered)
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t - 1)), 0.01))

    # Gaussian center
    inits.append(np.maximum(np.exp(-5 * (t - 0.5) ** 2) + 0.05, 0.01))

    # Edge concentrated
    inits.append(np.maximum(np.exp(-8 * t ** 2) + np.exp(-8 * (t - 1) ** 2) + 0.05, 0.01))

    # Sine patterns
    for f in [2, 3, 4, 5]:
        inits.append(np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * f * t), 0.01))

    # Random smooth
    k = max(3, n // 16)
    kern = np.ones(k) / k
    smooth = np.convolve(rng.random(n), kern, 'same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # V-shape
    inits.append(np.maximum(1.0 - 1.5 * np.abs(t - 0.5) + 0.3, 0.01))

    # Inverted V (peak in middle, low at edges) - actually this is the cosine
    # Multi-bump
    n_bumps = 3
    bumps = np.zeros(n)
    for b in range(n_bumps):
        center = (b + 0.5) / n_bumps
        bumps += np.exp(-30 * (t - center) ** 2)
    inits.append(np.maximum(bumps + 0.1, 0.01))

    # Normalize all
    result = []
    for a in inits:
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
        result.append(a)
    return result


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for optimal step function heights:
    Phase 1: Diverse initialization with quick Adam (200s)
    Phase 2: Focused Adam optimization with adaptive tau (700s)
    Phase 3: Multi-length exploration + multiplicative mode (300s)
    Phase 4: LP polishing for final precision (remaining)
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0
    best_a = None
    best_J = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best_J = compute_J(parent)
            best_a = parent.copy()
            print(f"Parent: n={parent.size}, J={best_J:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        n = 512
        parent = np.ones(n) / n
        best_a = parent.copy()
        best_J = compute_J(parent)
        print(f"Default init: n={n}, J={best_J:.8f}", flush=True)

    # === PHASE 1: Diverse initialization ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Diverse Init + Quick Adam (budget: 200s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    p1_deadline = t_start + 200.0
    n_main = best_a.size if best_a.size >= 256 else 512

    inits = make_diverse_inits(n_main, parent, seed=seed)
    print(f"  {len(inits)} inits at n={n_main}", flush=True)

    candidates = []
    time_per = max(8.0, (p1_deadline - time.monotonic()) / len(inits))

    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_deadline - 5:
            break
        sub_dl = min(time.monotonic() + time_per, p1_deadline - 5)
        a_opt, J_opt = adam_optimize(a0, sub_dl,
                                      tau_start=5, tau_end=30,
                                      lr=0.008, restart_interval=150,
                                      max_restarts=20)
        candidates.append((J_opt, a_opt))
        print(f"  init {idx:2d}: J={J_opt:.8f}", flush=True)
        if J_opt < best_J:
            best_J = J_opt
            best_a = a_opt.copy()

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 best: J={candidates[0][0]:.8f}", flush=True)

    # === PHASE 2: Focused optimization ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Focused Adam (budget: 700s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    p2_deadline = t_start + 900.0

    top_k = min(3, len(candidates))
    time_alloc = [0.45, 0.30, 0.25]

    for ci in range(top_k):
        if time.monotonic() > p2_deadline - 20:
            break
        J_init, a_init = candidates[ci]
        remaining = p2_deadline - time.monotonic()
        alloc = remaining * time_alloc[ci]
        alloc = max(60.0, alloc)
        sub_dl = min(time.monotonic() + alloc, p2_deadline - 20)

        print(f"\n  Cand {ci} (J={J_init:.8f}), budget={alloc:.0f}s", flush=True)

        # Standard Adam
        a_opt, J_opt = adam_optimize(a_init, sub_dl,
                                      tau_start=10, tau_end=80,
                                      lr=0.005, restart_interval=400,
                                      max_restarts=100)
        if J_opt < best_J:
            best_J = J_opt
            best_a = a_opt.copy()
            print(f"  ** New best: J={best_J:.8f}", flush=True)
        else:
            print(f"  Cand {ci}: J={J_opt:.8f} (best stays {best_J:.8f})", flush=True)

    # === PHASE 3: Multi-length + multiplicative mode ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Multi-length + Multiplicative (budget: 300s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_deadline = t_start + 1200.0

    if time.monotonic() < p3_deadline - 30:
        # Try multiplicative mode on best
        sub_dl = min(time.monotonic() + 100, p3_deadline - 60)
        a_opt, J_opt = adam_optimize(best_a, sub_dl,
                                      tau_start=20, tau_end=100,
                                      lr=0.003, restart_interval=300,
                                      max_restarts=50,
                                      use_multiplicative=True)
        if J_opt < best_J:
            best_J = J_opt
            best_a = a_opt.copy()
            print(f"  Multiplicative: J={best_J:.8f} (improved!)", flush=True)
        else:
            print(f"  Multiplicative: J={J_opt:.8f} (no improvement)", flush=True)

        # Multi-length
        if time.monotonic() < p3_deadline - 30:
            base_n = best_a.size
            try_ns = [base_n // 2, base_n + 128, base_n * 2]
            try_ns = [x for x in try_ns if 128 <= x <= 4096]

            for n_try in try_ns:
                if time.monotonic() > p3_deadline - 20:
                    break
                if n_try == base_n:
                    continue

                a_init = resample_seq(best_a, n_try)
                a_init = np.maximum(a_init, 1e-6)
                a_init = a_init / a_init.sum()

                sub_dl = min(time.monotonic() + 60, p3_deadline - 20)
                a_opt, J_opt = adam_optimize(a_init, sub_dl,
                                              tau_start=10, tau_end=60,
                                              lr=0.005, restart_interval=200,
                                              max_restarts=30)
                print(f"  n={n_try}: J={J_opt:.8f}", flush=True)
                if J_opt < best_J:
                    best_J = J_opt
                    best_a = a_opt.copy()

    # === PHASE 4: Final LP polishing ===
    print(f"\n{'=' * 60}", flush=True)
    rem = deadline - time.monotonic()
    print(f"PHASE 4: LP Polishing (remaining: {rem:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if time.monotonic() < deadline - 5:
        # High-tau Adam first (quick)
        sub_dl = min(time.monotonic() + rem * 0.4, deadline - 30)
        a_opt, J_opt = adam_optimize(best_a, sub_dl,
                                      tau_start=40, tau_end=200,
                                      lr=0.002, restart_interval=200,
                                      max_restarts=30)
        if J_opt < best_J:
            best_J = J_opt
            best_a = a_opt.copy()
            print(f"  High-tau Adam: J={best_J:.8f}", flush=True)

        # LP polish (slow but precise)
        if time.monotonic() < deadline - 5:
            sub_dl = min(time.monotonic() + (deadline - time.monotonic()), deadline - 3)
            a_opt, J_opt = lp_polish(best_a, sub_dl, max_iters=100)
            if J_opt < best_J:
                best_J = J_opt
                best_a = a_opt.copy()
                print(f"  LP polish: J={best_J:.8f}", flush=True)
            else:
                print(f"  LP polish: J={J_opt:.8f} (no improvement)", flush=True)

    # Final verification
    final_J = compute_J(best_a)
    if final_J < best_J:
        best_J = final_J

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_J:.8f}, n={best_a.size}, time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_a]
```
