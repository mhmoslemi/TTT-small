"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-temperature softmax-smoothed gradient descent with Adam (O(n log n) per step)
2. LP mirror descent for local refinement at the peak
3. Coordinate descent for final polishing
4. Multi-start with continuation method on temperature and periodic restarts
Key innovation: efficient gradient via FFT (correlation + convolution),
continuation method on softmax temperature for navigating rugged landscape.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_gradient(a, tau):
    """
    Compute smooth gradient of L = logsumexp(conv(a,a)/tau)/tau w.r.t. a.
    Returns (grad_L, L, b, p) where grad_L[i] = dL/da_i.
    """
    n = len(a)
    b = fast_conv(a)
    b_scaled = b / tau
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    p = exp_b / exp_b.sum()
    L = (b_max + np.log(exp_b.sum())) / tau

    # Compute gradient via FFT: grad_L[i] = sum_k p[k]*(a[k-i] + a[i-k]) / tau
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)

    # Cross-correlation: ifft(P * conj(A))[i] = sum_k p[k]*a[k-i]
    term1 = np.fft.irfft(P * np.conj(A), size)[:n]
    # Convolution: ifft(P * A)[i] = sum_k p[k]*a[i-k]
    term2 = np.fft.irfft(P * A, size)[:n]

    grad_L = (term1 + term2) / tau
    return grad_L, L, b, p


def adam_smooth_descent(a, deadline, max_iters=50000, lr=0.03,
                        tau_start=None, tau_end=None,
                        beta1=0.9, beta2=0.999, eps=1e-8,
                        restart_every=300, restart_scale=0.08):
    """
    Adam-based smooth gradient descent with continuation on tau.
    Normalizes to sum=1 after each step (simplex projection).
    """
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 0.01:
        return a, compute_J(a)
    a = a / S
    n = len(a)

    if tau_start is None:
        b_init = fast_conv(a)
        tau_start = max(b_init.max(), 1e-10) * 0.3
    if tau_end is None:
        tau_end = max(tau_start * 0.01, 1e-8)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t += 1
        frac = min(1.0, t / max(1, max_iters))
        # Log-linear temperature schedule
        tau = tau_start * (tau_end / tau_start) ** frac
        # Learning rate decay
        lr_t = lr * (0.3 ** frac)

        grad_L, L, b, p = smooth_gradient(a, tau)

        # Full gradient of J (up to positive constant 2n/S^2)
        S_val = a.sum()
        grad_J = grad_L - 2.0 * L / S_val

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_J
        v = beta2 * v + (1 - beta2) * grad_J ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        S_new = a_new.sum()
        if S_new < 1e-10:
            break
        a_new = a_new / S_new

        J_new = compute_J(a_new)

        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Restart if stuck
        if no_improve >= restart_every:
            # Adaptive restart scale
            scale = restart_scale * (0.5 + 0.5 * (1.0 - frac))
            a = a + np.random.normal(0, scale, n)
            a = np.maximum(a, 0.005)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_J

    Each phase initializes from the result of the previous.

    n_phases = len(tau_levels)

        iters = max(1000, int(time_per_phase * 500))  # rough estimate

        a_r, J_r = adam_smooth_descent(a, phase_deadline, max_iters=iters,
                                        lr=0.05, tau_start=tau, tau_end=tau * 0.5,
                                        restart_every=200, restart_scale=0.06)
        if J_r < best_J:
            best_J = J_r
            best_a = a_r.copy()
        a = a_r

    return best_a, best_J


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows_list.append(k)
            cols_list.append(j)
            vals_list.append(a[k - j])
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    vals = np.array(vals_list, dtype=np.float64)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent for local refinement."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        A = build_lp_matrix(n, a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 60, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                if g.sum() > 1e-15:
                    g = g / g.sum()
                    # Line search
                    b_ag = cross_conv(a, g)
                    b_gg = fast_conv(g)

                    def obj(t):
                        b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
                        return b.max()

                    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                             options={'xatol': 1e-8})
                    t_opt = float(result.x)
                    a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
                    s = a_new.sum()
                    if s > 1e-10:
                        a_new = a_new / s
                        J_new = compute_J(a_new)
                        if J_new < best_J - 1e-12:
                            best_J = J_new
                            best_a = a_new.copy()
                            a = a_new
                            no_improve = 0
                        else:
                            no_improve += 1
        except Exception:
            pass

        if no_improve > 30:
            break

    return best_a, best_J


def coordinate_descent_polish(a, deadline, max_passes=10):
    """
    Coordinate descent: for each coordinate, try several candidate values
    and keep the best. Fast version using incremental convolution updates.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = np.random.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            if ai < 1e-10:
                candidates = [0.01, 0.05, 0.1, 0.2, 0.3]
            else:
                candidates = [0.0, ai * 0.3, ai * 0.5, ai * 0.7, ai * 1.3, ai * 1.6, ai * 2.0]

            best_candidate = ai
            best_score = compute_J(a)

            for cand in candidates:
                if abs(cand - ai) < 1e-14:
                    continue
                a[i] = cand
                a_clip = np.maximum(a, 0)
                s_check = a_clip.sum()
                if s_check < 0.01:
                    a[i] = ai
                    continue
                score = compute_J(a_clip / s_check)
                if score < best_score - 1e-12:
                    best_score = score
                    best_candidate = cand

            if best_candidate != ai:
                a[i] = best_candidate
                improved = True

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def simulated_annealing(a, deadline, max_iters=5000, T_start=0.04, T_end=0.001):
    """Simulated annealing with single-coordinate moves."""
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    current_J = best_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        T = T_start * (T_end / T_start) ** frac

        i = np.random.randint(n)
        old_val = a[i]

        if old_val < 0.05:
            new_val = np.random.exponential(0.08)
        else:
            new_val = np.random.normal(old_val, 0.2 * old_val + 0.02)
        new_val = max(0.0, new_val)

        a[i] = new_val
        a_new = np.maximum(a, 0)
        s = a_new.sum()
        if s < 0.01:
            a[i] = old_val
            continue
        a_new = a_new / s
        new_J = compute_J(a_new)

        delta = new_J - current_J
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            current_J = new_J
            if new_J < best_J - 1e-12:
                best_J = new_J
                best_a = a_new.copy()
        else:
            a[i] = old_val

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random
    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())

    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())

    # Bell
    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())

    # Frequency modulated
    for freq in [2, 3, 5]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())

    # Parent perturbation
    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 0.01)
        p = p / p.sum()
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01) / max(1e-10, (np.maximum(p + rng.normal(0, 0.1, n), 0.01)).sum()))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Multi-temperature smooth gradient descent (main workhorse)
    Phase 2: LP mirror descent refinement
    Phase 3: Coordinate descent + simulated annealing
    Phase 4: Multi-start and multi-length exploration
    Phase 5: Final polishing
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512
        update_best(parent)

    # === PHASE 1: Multi-temperature smooth gradient descent ===
    p1_deadline = t_start + 550
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-temperature Smooth GD (n={parent.size}, budget=550s)", flush=True)
    print(f"{'='*60}", flush=True)

    a1 = parent.copy()
    a1 = np.maximum(a1, 0)
    a1 = a1 / a1.sum()

    # Run main continuation
    a_r, J_r = multi_temp_descent(a1, p1_deadline)
    update_best(a_r)
    print(f"  Phase 1 main done: J={J_r:.8f}", flush=True)

    # If we have time, do another round from a different init
    if time.monotonic() < p1_deadline - 60:
        inits = generate_inits(a1.size, rng, parent=parent)
        init_scores = [(compute_J(init), init) for init in inits[:5]]
        init_scores.sort(key=lambda x: x[0])
        for sc, init in init_scores[:3]:
            if time.monotonic() > p1_deadline - 30:
                break
            phase_dl = min(time.monotonic() + 40, p1_deadline)
            a_r2, J_r2 = adam_smooth_descent(init, phase_dl, max_iters=5000,
                                              lr=0.04, tau_start=None, tau_end=None,
                                              restart_every=150, restart_scale=0.05)
            update_best(a_r2)

    print(f"  Phase 1 done: best J={best_score:.8f}", flush=True)

    # === PHASE 2: LP mirror descent refinement ===
    p2_deadline = min(t_start + 750, deadline - 300)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_lp, J_lp = lp_mirror_descent(best_seq, 500, p2_deadline)
        update_best(a_lp)
        print(f"  Phase 2 done: best J={best_score:.8f}", flush=True)

    # === PHASE 3: Coordinate descent + Simulated annealing ===
    p3_deadline = min(t_start + 1000, deadline - 300)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: CD + SA Polishing (budget=250s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        # Coordinate descent
        cd_dl = min(time.monotonic() + 100, p3_deadline)
        a_cd = coordinate_descent_polish(best_seq, cd_dl, max_passes=15)
        update_best(a_cd)

        # Simulated annealing
        sa_dl = min(time.monotonic() + 120, p3_deadline)
        a_sa = simulated_annealing(a_cd, sa_dl, max_iters=8000, T_start=0.03, T_end=0.001)
        update_best(a_sa)

        # Small CD on SA result
        a_cd2 = coordinate_descent_polish(a_sa, min(sa_dl + 30, p3_deadline), max_passes=5)
        update_best(a_cd2)

        print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)

    # === PHASE 4: Multi-length exploration ===
    p4_deadline = min(t_start + 1250, deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration (budget=250s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = sorted(set([
            int(n_base * 0.75), int(n_base * 1.25), int(n_base * 1.5),
            384, 448, 512, 640, 768, 1024, 1280
        ]))
        target_lengths = [L for L in target_lengths if 128 <= L <= 2048 and L != n_base]

        for tgt_n in target_lengths[:6]:
            if time.monotonic() > p4_deadline:
                break
            phase_dl = min(time.monotonic() + 30, p4_deadline)

            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            a_r, J_r = adam_smooth_descent(a_ext, phase_dl, max_iters=2000,
                                            lr=0.04, restart_every=100, restart_scale=0.04)
            update_best(a_r)
            a_cd = coordinate_descent_polish(a_r, min(phase_dl + 5, p4_deadline), max_passes=3)
            update_best(a_cd)

        print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Final polishing ===
    p5_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        # One more smooth GD round with very low temp
        gd_dl = min(time.monotonic() + 80, p5_deadline)
        a_f, J_f = adam_smooth_descent(best_seq, gd_dl, max_iters=10000,
                                        lr=0.02, restart_every=100, restart_scale=0.03)
        update_best(a_f)

        # LP on the result
        lp_dl = min(time.monotonic() + 60, p5_deadline)
        a_lp2, J_lp2 = lp_mirror_descent(a_f, 200, lp_dl)
        update_best(a_lp2)

        # Final CD
        a_cd_final = coordinate_descent_polish(a_lp2, p5_deadline, max_passes=5)
        update_best(a_cd_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
