"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based fast gradient computation via cross-correlation
2. Nesterov-accelerated smooth gradient descent
3. Multi-initialization (cosine, Gaussian, triangular, random)
4. Multi-scale exploration (varying sequence length)
5. LP mirror descent polish
6. Perturbation-based local search

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_M(a):
    return float(fast_conv(a).max())

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def fast_grad(a, tau):
    """Gradient of smooth max via FFT-based cross-correlation."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    # grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # = 2 * correlate(p, a, 'full')[i] for i in [0, n-1]
    grad = 2.0 * np.correlate(p, a, mode='full')[:n]
    return grad

def smooth_max_value(a, tau):
    """Value of smooth max approximation."""
    b = fast_conv(a)
    b_max = b.max()
    return float(b_max + np.log(np.exp(tau * (b - b_max)).sum()) / tau)

def run_smooth_gradient(a, max_iters, deadline, tau_start=1.0, tau_end=50.0, 
                        lr_scale=1.0, momentum=0.9):
    """Nesterov-accelerated smooth gradient descent on the simplex."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    y = a.copy()  # Nesterov momentum
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        
        # Adaptive learning rate
        # Lipschitz constant ~ 4 * tau * max(a)^2 * n (rough bound)
        # Use a simpler heuristic
        lr = lr_scale / (tau * n ** 0.5 * max(a.max(), 1e-6))
        lr = min(lr, 0.5)  # cap
        
        # Nesterov extrapolation
        beta = momentum * (1.0 + frac * 0.1)
        y = a + beta * (a - y) if it > 0 else a.copy()
        y = np.maximum(y, 1e-10)
        
        # Gradient at y
        grad_y = fast_grad(y, tau)
        
        # Step
        a_new = y - lr * grad_y
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        # Track best
        if it % 10 == 0 or it == max_iters - 1:
            M = compute_M(a)
            if M < best_M * (1 - 1e-10):
                best_M = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve > 200:
            break
    
    # Final check
    M = compute_M(a)
    if M < best_M:
        best_M = M
        best_a = a.copy()
    
    return best_a

def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)

def solve_lp(a, M, row_idx, col_idx):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def line_search_analytic(a, g):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val

def cross_conv(a, b):

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]

def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx):

    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        t, M_new = line_search_analytic(a, g)
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:

        return src.copy()

    x = np.arange(n) - (n - 1) / 2
    sigma = sigma_frac * n
    a = np.exp(-x**2 / (2 * sigma**2))

Then sum_j q[i+j] * r[j] = sum_j q[i+j] * r_flip[n-1-j]

Let m = n-1-j, so j = n-1-m:
= sum_m q[i+n-1-m] * r_flip[m]

This is the (i+n-1)-th element of the convolution q ⊛ r_flip:
(q * r_flip)[k] = sum_m q[k-m] * r_flip[m]

So (q * r_flip)[i+n-1] = sum_m q[i+n-1-m] * r_flip[m] = sum_j q[i+j] * r[j] = R[i] ✓

So R[i] = (q * r_flip)[i + n - 1] for i = 0, ..., n-1.

The full convolution q * r_flip has length (2n-1) + n - 1 = 3n-2.
I need elements at positions n-1, n, ..., 2n-2 (the first n elements starting from position n-1).

Using FFT:
```python
N = 1
while N < 3*n - 2:
    N <<= 1
fq = np.fft.rfft(q, N)  # q = p (length 2n-1)
fr = np.fft.rfft(r_flip, N)  # r_flip = a[::-1] (length n)
conv = np.fft.irfft(fq * fr, N)
R = conv[n-1:2*n-1]  # positions n-1 to 2n-2, length n
grad = 2.0 * R
```
