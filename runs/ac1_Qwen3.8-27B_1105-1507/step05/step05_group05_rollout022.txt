    Uses vectorized numpy for O(n) per evaluation of the 1D objective.

def spectral_flatten(a, n_iters=50, alpha_start=0.3, alpha_end=0.8, deadline=None):
    """
    Iteratively flatten the power spectrum of a (Gerchberg-Saxton style).
    Alternates between: (1) flatten |FFT(a)|, (2) project onto non-negative simplex.
    """
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        frac = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        A = np.fft.rfft(a, size)
        amp = np.abs(A)
        phase = A / (amp + 1e-30)

        target_amp = np.sqrt(amp ** 2).mean() ** 0.5  # = sqrt(mean(|A|^2))
        # Actually: mean(|A|^2) is the target power level
        target_amp = np.full_like(amp, np.sqrt(np.mean(amp ** 2)))

        new_amp = (1 - alpha) * amp + alpha * target_amp
        A_new = new_amp * phase

        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a


def subgradient_step(a, step_size, return_subgrad=False):
    """
    One step of subgradient descent on max(conv(a,a)) with sum(a)=1, a>=0.
    Uses the exact subgradient at the peak set.
    """
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    b = fast_conv(a)
    M = b.max()
    # Peak set (with tolerance to include near-peaks)
    tol = 1e-10
    peak_mask = b >= M - tol
    peak_indices = np.where(peak_mask)[0]

    if len(peak_indices) == 0:
        g = np.zeros(n)
    elif len(peak_indices) == 1:
        k = peak_indices[0]
        g = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] = 2.0 * a[j]
    else:
        # Average subgradient over peak set
        g = np.zeros(n)
        for k in peak_indices:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    g[i] += 2.0 * a[j]
        g /= len(peak_indices)

    # Project gradient to tangent space (sum = 0)
    g = g - g.mean()

    # Take step
    a_new = a - step_size * g
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-10:
        return a
    a_new = a_new / s_new

    if return_subgrad:
        return a_new, g, M, len(peak_indices)
    return a_new


def run_subgradient_descent(a, max_iters, deadline, lr_start=0.01, lr_end=0.0001):
    """
    Subgradient descent with adaptive step size (1/k decay).
    """
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        lr = lr_start / (1.0 + lr_start * it)
        a = subgradient_step(a, lr)
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: (a*g)[k] for all k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=10):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, time_limit=15):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64).copy(), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=time_limit)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        improved = False
        base_t = max(0.05, 0.5 * (1 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 12:
                break
        else:
            no_improve = 0

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))

    inits.append(np.exp(-3 * (t - 0.5) ** 2) + 0.1)
    inits.append(0.3 + 0.7 * np.abs(2 * t - 1))

    # Smooth random
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Quadratic residue inspired (Legendre symbol smoothed)
    if n % 2 == 1:
        p_prime = n
        legendre = np.zeros(n)
        for j in range(n):
            legendre[j] = 1 if (j * j) % p_prime == 1 or j == 0 else -1
        legendre[0] = 0
        smooth_leg = np.convolve(legendre, np.ones(5) / 5, mode='same')
        inits.append(np.maximum(1.0 + 0.5 * smooth_leg, 0.01))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining fast coordinate descent, spectral flattening,
    subgradient descent, and LP mirror descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    # === PHASE 1: Spectral Flattening (120s) ===
    p1_deadline = min(time.monotonic() + 120, deadline - 800)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Spectral Flattening", flush=True)
    print(f"{'='*60}", flush=True)

    n_parent = parent.size
    lengths_p1 = [n_parent]
    # Add a few other lengths
    for mult in [0.75, 1.5, 2.0]:
        nl = int(n_parent * mult)
        if 64 <= nl <= 2048 and nl not in lengths_p1:
            if nl % 2 == 0:
                lengths_p1.append(nl)

    for n in lengths_p1:
        if time.monotonic() > p1_deadline:
            break
        a_init = resample_seq(parent, n)
        a_init = np.maximum(a_init, 1e-6)
        a_init = a_init / a_init.sum()

        a_sf = spectral_flatten(a_init, n_iters=80, alpha_start=0.2, alpha_end=0.9,
                                deadline=min(time.monotonic() + 30, deadline - 700))
        tracker.update(a_sf)

    tracker.report("Phase 1 complete")

    # === PHASE 2: Fast Coordinate Descent (500s) ===
    p2_deadline = min(time.monotonic() + 500, deadline - 350)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Fast Coordinate Descent (main workhorse)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths_p2 = [n_parent]
    for nl in [int(n_parent * 0.8), int(n_parent * 1.25), int(n_parent * 1.5)]:
        if 128 <= nl <= 2048 and nl % 64 == 0 and nl not in lengths_p2:
            lengths_p2.append(nl)

    # Allocate more time to parent length
    time_alloc_p2 = {}
    total_alloc = 500
    n_lengths = len(lengths_p2)
    # Parent length gets 50%, others split the rest
    time_alloc_p2[lengths_p2[0]] = int(total_alloc * 0.5)
    remaining = total_alloc - time_alloc_p2[lengths_p2[0]]
    if len(lengths_p2) > 1:
        each = remaining // (len(lengths_p2) - 1)
        for i, nl in enumerate(lengths_p2[1:]):
            time_alloc_p2[nl] = each

    for n in lengths_p2:
        if time.monotonic() > p2_deadline:
            break
        t_alloc = time_alloc_p2.get(n, 60)
        phase_deadline = min(time.monotonic() + t_alloc, deadline - 300)

        if time.monotonic() >= phase_deadline:
            continue

        print(f"\n  --- n={n}, budget={t_alloc}s ---", flush=True)
        inits = diverse_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_for_n = None
        best_score_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_deadline:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            a_opt, score_opt = run_fast_coord_descent(a0, max_sweeps=5000,
                                                      deadline=phase_deadline,
                                                      seed=seed + idx * 13,
                                                      normalize_every=3)
            if score_opt < best_score_for_n:
                best_score_for_n = score_opt
                best_for_n = a_opt.copy()
                # Evaluate with the official function
                tracker.update(a_opt)

        if best_for_n is not None:
            print(f"  n={n:4d}: best_M_ratio={best_score_for_n:.8f}", flush=True)

    tracker.report("Phase 2 complete")

    # === PHASE 3: Subgradient + Coord Hybrid (250s) ===
    p3_deadline = min(time.monotonic() + 250, deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Subgradient + Coord Hybrid", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p3_deadline:
        a_sub = run_subgradient_descent(tracker.best, max_iters=500,
                                        deadline=min(time.monotonic() + 60, p3_deadline - 50),
                                        lr_start=0.008, lr_end=0.00005)
        tracker.update(a_sub)

        # Coordinate polish after subgradient
        a_hyb, _ = run_fast_coord_descent(a_sub, max_sweeps=1000,
                                          deadline=min(time.monotonic() + 80, p3_deadline),
                                          seed=9999, normalize_every=3)
        tracker.update(a_hyb)

        # Try from perturbed best
        a_pert = tracker.best + rng.normal(0, 0.01, tracker.best.size)
        a_pert = np.maximum(a_pert, 0.001)
        a_pert = a_pert / a_pert.sum()
        a_pert_opt, _ = run_fast_coord_descent(a_pert, max_sweeps=500,
                                               deadline=min(time.monotonic() + 60, p3_deadline),
                                               seed=7777, normalize_every=3)
        tracker.update(a_pert_opt)

    tracker.report("Phase 3 complete")

    # === PHASE 4: LP Mirror Descent (200s) ===
    p4_deadline = min(time.monotonic() + 200, deadline - 50)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p4_deadline:
        n_best = tracker.best.size
        if n_best <= 1024:  # LP gets too slow for very large n
            est_lp_time = 0.003 * (n_best / 128) ** 2
            max_iters = max(20, min(100, int(150 / max(0.02, est_lp_time))))
            print(f"  n={n_best}, est LP time={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)

            a_lp = run_lp_mirror_descent(tracker.best, max_iters,
                                         min(time.monotonic() + 150, p4_deadline),
                                         time_limit=10)
            tracker.update(a_lp)

            # Second LP pass from the result
            if time.monotonic() < p4_deadline and a_lp is not None:
                a_lp2 = run_lp_mirror_descent(a_lp, min(30, max_iters), p4_deadline,
                                              time_limit=10)
                tracker.update(a_lp2)
        # Coordinate polish after LP
        if time.monotonic() < p4_deadline:
            a_after_lp, _ = run_fast_coord_descent(tracker.best, max_sweeps=500,
                                                   deadline=p4_deadline,
                                                   seed=55555, normalize_every=2)
            tracker.update(a_after_lp)

    tracker.report("Phase 4 complete")

    # === PHASE 5: Final Polish (remaining time) ===
    p5_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p5_deadline:
        # Multiple quick coordinate descent passes with different seeds
        for s in [111, 222, 333]:
            if time.monotonic() > p5_deadline - 10:
                break
            a_pol, _ = run_fast_coord_descent(tracker.best, max_sweeps=2000,
                                              deadline=min(time.monotonic() + 40, p5_deadline - 10),
                                              seed=s, normalize_every=2)
            tracker.update(a_pol)

        # Try slightly different lengths
        if time.monotonic() < p5_deadline - 30:
            for mult in [1.1, 0.9]:
                new_n = int(tracker.best.size * mult)
                if 128 <= new_n <= 2048 and new_n != tracker.best.size:
                    a_ext = resample_seq(tracker.best, new_n)
                    a_ext = np.maximum(a_ext, 1e-6)
                    a_ext = a_ext / a_ext.sum()
                    a_ext_opt, _ = run_fast_coord_descent(a_ext, max_sweeps=500,
                                                          deadline=min(time.monotonic() + 30, p5_deadline - 5),
                                                          seed=44444, normalize_every=3)
                    tracker.update(a_ext_opt)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    # Final spectral flattening attempt if time
    if tracker.best is not None and time.monotonic() < p5_deadline - 5:
        a_final_sf = spectral_flatten(tracker.best, n_iters=30, alpha_start=0.1, alpha_end=0.3,
                                      deadline=p5_deadline - 2)
        tracker.update(a_final_sf)

    tracker.report("FINAL")
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
