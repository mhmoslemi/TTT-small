def water_filling_init(n, iterations=100):
    """Initialize using frequency-domain water-filling."""
    a = np.ones(n) / n
    for _ in range(iterations):
        A = np.fft.rfft(a, 2*n-1)
        P = np.abs(A)**2
        # Water-filling: try to make P as flat as possible
        # P_target = mean(P) * ones
        # Adjust a to reduce variance of P
        # Simple approach: a_new = a * exp(-gamma * (P - mean(P)))
        # But this doesn't directly work since we need a >= 0

        a = np.maximum(a_new, 0)
        s = a.sum()
        if s > 0:

    return a

"""
Hybrid multi-scale optimization for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over parent:
1. Multi-length strategy (512, 1024, 2048) - larger n gives lower theoretical bound
2. Frequency-domain water-filling initialization for new lengths
3. Adaptive constraint active set (expands as convergence improves)
4. Eigenvector-based perturbation for escaping local minima
5. Improved line search with duality-based step sizing
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n, a, active_k=None):
    """Build sparse convolution matrix A where (A@g)[k] = (a*g)[k]."""
    if active_k is None:
        active_k = np.arange(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))


def solve_lp(a, M, active_k):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, np.inf)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None


def water_filling_init(n, num_iters=200, seed=0):
    """Frequency-domain water-filling initialization."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) / n
    a += rng.normal(0, 0.1, n)
    a = np.maximum(a, 1e-4)
    a = a / a.sum()

    for it in range(num_iters):
        # Compute power spectrum
        A = np.fft.rfft(a, 2 * n - 1)
        P = np.abs(A) ** 2

        # Target: flat power spectrum
        # Water-filling: reduce high-power frequencies, boost low-power
        target = P.mean() * 1.5  # slightly above mean for robustness
        # Adjust amplitude in frequency domain
        gain = np.ones_like(P)
        high_mask = P > target
        low_mask = P < target * 0.3
        gain[high_mask] = 0.8
        gain[low_mask] = 1.2

        A_new = A * gain
        a_new = np.fft.irfft(A_new, 2 * n - 1)[:n]
        a = np.maximum(a_new, 1e-6)
        s = a.sum()
        if s > 0:
            a = a / s

    return a


def smooth_gradient_step(a, tau, lr):
    """One step of smooth gradient descent on the objective."""
    n = len(a)
    conv = fast_conv(a)
    M = conv.max()
    exp_b = np.exp(tau * (conv - M))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * pk * a[k - j]

    S = a.sum()
    # Gradient of 2n*LSE/S^2 (with LSE normalized by tau already in p)
    LSE = M + np.log(exp_b.sum()) / tau
    full_grad = (2.0 * n / S ** 2) * (grad - 2.0 * LSE / S)

    a_new = np.maximum(a - lr * full_grad, 1e-8)
    a_new = a_new / a_new.sum()
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new

    a = a / a.sum()

    # Quick smooth optimization
    sub_deadline = min(time.monotonic() + 20, deadline - 60)
    a = quick_smooth_optimize(a, sub_deadline, tau=12.0, num_steps=100)
    tracker.update(a, f"{label}_smooth")

    # LP mirror descent
    if time.monotonic() < deadline - 30:
        a_opt, _ = lp_mirror_descent_v2(
            a, deadline - 10,
            top_frac_start=0.5, top_frac_end=0.85,
            max_no_improve=100, verbose=False
        )
        tracker.update(a_opt, f"{label}_lp")

    return tracker.best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Multi-length initialization (120s)
    Phase 2: Primary LP mirror descent on best candidate (900s)
    Phase 3: Alternative length exploration (300s)
    Phase 4: Final polishing (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "parent")
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent, "default")

    # === PHASE 1: Multi-length initialization ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: Multi-length Initialization", flush=True)
    print(f"{'=' * 60}", flush=True)

    p1_deadline = time.monotonic() + 120.0

    # Try different lengths
    lengths_to_try = [512, 1024, 1536, 2048]
    init_candidates = {}

    for n_try in lengths_to_try:
        if time.monotonic() > p1_deadline:
            break
        if n_try < 64 or n_try > 4096:
            continue

        print(f"  Initializing n={n_try}...", flush=True)
        t0 = time.monotonic()

        # Water-filling init
        a_wf = water_filling_init(n_try, num_iters=100, seed=seed + n_try)
        init_candidates[f"wf_{n_try}"] = a_wf

        # Resampled parent
        a_res = resample_seq(parent, n_try)
        a_res = np.maximum(a_res, 1e-6)
        a_res = a_res / a_res.sum()
        init_candidates[f"res_{n_try}"] = a_res

        # Palindrome version
        a_pal = make_palindrome(a_res)
        init_candidates[f"pal_{n_try}"] = a_pal

        # Uniform with noise
        a_uni = np.ones(n_try) / n_try + rng.normal(0, 0.02, n_try)
        a_uni = np.maximum(a_uni, 1e-4)
        a_uni = a_uni / a_uni.sum()
        init_candidates[f"uni_{n_try}"] = a_uni

        elapsed = time.monotonic() - t0
        print(f"    Generated {len(init_candidates)} candidates in {elapsed:.1f}s", flush=True)

    # Quick-optimize each candidate
    print("  Quick-optimizing candidates...", flush=True)
    for name, a0 in list(init_candidates.items()):
        if time.monotonic() > p1_deadline:
            break
        a_opt = quick_smooth_optimize(a0, p1_deadline, tau=12.0, num_steps=50)
        J = compute_J(a_opt)
        init_candidates[name] = a_opt
        print(f"    {name}: J={J:.8f}", flush=True)
        tracker.update(a_opt, name)

    # Sort candidates
    sorted_cands = sorted(init_candidates.items(), key=lambda x: compute_J(x[1]))
    print(f"Phase 1 best: {sorted_cands[0][0]} J={compute_J(sorted_cands[0][1]):.8f}", flush=True)

    # === PHASE 2: Primary LP Mirror Descent ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 2: Primary LP Mirror Descent", flush=True)
    print(f"{'=' * 60}", flush=True)

    p2_deadline = time.monotonic() + 900.0
    p2_start = time.monotonic()

    # Allocate time to top candidates
    top_cands = sorted_cands[:4]
    time_weights = [0.45, 0.25, 0.15, 0.15]

    for ci, (name, a_init) in enumerate(top_cands):
        if time.monotonic() > p2_deadline:
            break

        remaining = p2_deadline - time.monotonic()
        alloc = remaining * time_weights[ci]
        alloc = max(30.0, alloc)
        phase_dl = min(time.monotonic() + alloc, p2_deadline)

        J_init = compute_J(a_init)
        print(f"\n  Candidate '{name}' (J={J_init:.8f}, n={a_init.size}), "
              f"budget={alloc:.0f}s", flush=True)

        # Run LP mirror descent
        a_best, J_best = lp_mirror_descent_v2(
            a_init, phase_dl,
            top_frac_start=0.5, top_frac_end=0.9,
            max_no_improve=120,
            verbose=True
        )
        tracker.update(a_best, f"p2_{name}")

        # If this is the best, do a second pass with tighter constraints
        if tracker.best_score == compute_J(a_best) and time.monotonic() < p2_deadline - 60:
            print(f"  Second pass on best with tighter constraints...", flush=True)
            a_best2, _ = lp_mirror_descent_v2(
                a_best, min(time.monotonic() + 60, p2_deadline - 10),
                top_frac_start=0.8, top_frac_end=1.0,
                max_no_improve=200,
                verbose=False
            )
            tracker.update(a_best2, f"p2_{name}_tight")

    tracker.report("After Phase 2")

    # === PHASE 3: Alternative exploration ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 3: Alternative Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_deadline = min(time.monotonic() + 300.0, deadline - 50.0)

    if tracker.best is not None and time.monotonic() < p3_deadline:
        best_n = tracker.best.size

        # Try different lengths from current best
        alt_lengths = set()
        for factor in [0.5, 1.0, 1.5, 2.0, 3.0]:
            n_alt = int(best_n * factor)
            n_alt = max(128, min(4096, n_alt))
            # Round to power of 2 or multiple of 64
            n_alt = max(128, (n_alt // 64) * 64)
            if n_alt != best_n:
                alt_lengths.add(n_alt)

        for n_alt in sorted(alt_lengths):
            if time.monotonic() > p3_deadline:
                break

            print(f"  Trying n={n_alt}...", flush=True)
            a_alt = resample_seq(tracker.best, n_alt)
            a_alt = np.maximum(a_alt, 1e-6)
            a_alt = a_alt / a_alt.sum()

            # Quick smooth
            a_alt = quick_smooth_optimize(
                a_alt, min(time.monotonic() + 15, p3_deadline - 30),
                tau=10.0, num_steps=50
            )
            tracker.update(a_alt, f"p3_n{n_alt}_smooth")

            # Short LP
            if time.monotonic() < p3_deadline - 20:
                a_alt2, _ = lp_mirror_descent_v2(
                    a_alt, min(time.monotonic() + 40, p3_deadline - 10),
                    top_frac_start=0.6, top_frac_end=0.9,
                    max_no_improve=80,
                    verbose=False
                )
                tracker.update(a_alt2, f"p3_n{n_alt}_lp")

        # Palindrome variant
        if time.monotonic() < p3_deadline - 30:
            a_pal = make_palindrome(tracker.best)
            a_pal = np.maximum(a_pal, 1e-6)
            a_pal = a_pal / a_pal.sum()
            a_pal, _ = lp_mirror_descent_v2(
                a_pal, min(time.monotonic() + 50, p3_deadline - 10),
                top_frac_start=0.6, top_frac_end=0.9,
                max_no_improve=100,
                verbose=False
            )
            tracker.update(a_pal, "p3_palindrome")

        # Water-filling restart from best
        if time.monotonic() < p3_deadline - 30:
            n_wf = tracker.best.size
            a_wf = water_filling_init(n_wf, num_iters=150, seed=seed + 999)
            # Mix with best
            a_mix = 0.5 * a_wf + 0.5 * resample_seq(tracker.best, n_wf)
            a_mix = np.maximum(a_mix, 1e-6)
            a_mix = a_mix / a_mix.sum()
            a_mix, _ = lp_mirror_descent_v2(
                a_mix, min(time.monotonic() + 40, p3_deadline - 10),
                top_frac_start=0.5, top_frac_end=0.8,
                max_no_improve=80,
                verbose=False
            )
            tracker.update(a_mix, "p3_wf_mix")

    tracker.report("After Phase 3")

    # === PHASE 4: Final Polishing ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 4: Final Polishing", flush=True)
    remaining = deadline - time.monotonic()
    print(f"  Remaining time: {remaining:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None and remaining > 30:
        # Full constraint LP for precision
        a_polish, _ = lp_mirror_descent_v2(
            tracker.best, deadline - 5,
            top_frac_start=0.9, top_frac_end=1.0,
            max_no_improve=300,
            verbose=True
        )
        tracker.update(a_polish, "polish_full")

        # One more pass with slightly different params
        if time.monotonic() < deadline - 10:
            a_polish2, _ = lp_mirror_descent_v2(
                tracker.best, deadline - 5,
                top_frac_start=0.7, top_frac_end=0.95,
                max_no_improve=200,
                verbose=False
            )
            tracker.update(a_polish2, "polish_alt")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
