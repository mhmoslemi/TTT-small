def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    # This is (a * fliplr(h))[i] in the convolution sense...

        # c = IFFT(FA * conj(FH_reversed)) ... 
        # Actually let me just use: c = np.correlate(active, a, mode='full') and take the right part
        # np.correlate(a, v, mode='full')[i] = sum_j a[j] * v[j - i + n-1]

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = v.copy()
    # We want to find alpha such that sum(max(v - alpha, 0)) = target_sum
    # Bisection on alpha
    v_sorted = np.sort(v)[::-1]
    # Standard algorithm
    w = np.sort(v)[::-1]
    n = len(w)
    cs = np.cumsum(w)
    rho = np.where(w > 0)[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    # Find rho where (w[rho] - (sum(w[:rho+1]) - target_sum)/(rho+1)) >= 0
    # Standard: find largest rho such that w[rho] > (sum(w[:rho+1]) - target_sum) / (rho+1)
    
    # Use the standard O(n log n) algorithm
    tau = np.zeros(n)
    for i in range(1, n + 1):
        r = i - 1
        tau[r] = (cs[r] - target_sum) / (r + 1)
    # Find the rightmost r where w[r] > tau[r]
    valid = (w > tau)
    if not np.any(valid):
        return np.full(n, target_sum / n)
    rho = np.max(np.where(valid)[0])
    theta = (cs[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0)

        # Momentum update

def spectral_flat_init(n, n_iters=100, seed=42):

    rng = np.random.default_rng(seed)

        size <<= 1

        fa = np.fft.rfft(a, size)
        power = np.abs(fa) ** 2
        
        # Target: make the power spectrum "peaked" at DC
        # This makes the convolution flat in time domain
        # Strategy: raise the minimum non-DC power toward the mean
        non_dc_power = power[1:]
        if len(non_dc_power) > 0:
            p_min = non_dc_power.min()
            p_mean = non_dc_power.mean()
            # Push toward flat: new_power = mean + alpha * (power - mean)
            # alpha < 1 reduces variance
            alpha = 0.7
            power_new = power.copy()
            power_new[1:] = p_mean + alpha * (non_dc_power - p_mean)
            power_new = np.maximum(power_new, 0)
            
            mag_new = np.sqrt(power_new)
            phase = np.angle(fa)
            fa_new = mag_new * np.exp(1j * phase)
            a_new = np.fft.irfft(fa_new, size)[:n]
            
            # Enforce non-negativity
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new /= s
                a = a_new
    
    return a

            kq = 2 * i
            valid_kq = (kq <= 2 * n - 2)
            
            # a[i] is at position i in original, which is index (i - (k_lo - i)) in a_slice
            idx_i_in_slice = i - (k_lo - i)
            
            best_x = old
            best_local_M = current_M
            
            for x in candidates:
                x = float(x)
                if x < 0:
                    x = 0.0
                delta = x - old
                if abs(delta) < 1e-14:
                    continue
                
                # Apply: save a[i]'s contribution in a_slice
                orig_val = a_slice[idx_i_in_slice]
                a_slice[idx_i_in_slice] = x
                conv[k_lo:k_hi+1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                
                new_M = conv.max()
                if new_M < best_local_M:
                    best_local_M = new_M
                    best_x = x
                
                # Rollback
                a_slice[idx_i_in_slice] = orig_val
                conv[k_lo:k_hi+1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
            
            # Apply best
            if best_x != old and abs(best_x - old) > 1e-14:
                delta = best_x - old
                conv[k_lo:k_hi+1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_x
                sweep_changed = True
        
        if sweep_changed:
            current_M = conv.max()
            if current_M < best_M - 1e-12:
                best_M = current_M
                best_a = a.copy()
                no_improve_sweeps = 0
            else:
                no_improve_sweeps += 1
        
            # Periodic random perturbation to escape local minima
            if no_improve_sweeps > 50 and sweep % 25 == 0:
                n_pert = max(1, n // 200)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                for idx in pert_idx:
                    a[idx] = max(0, a[idx] * (1 + rng.normal(0, 0.3)))
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a /= s
                conv = fast_conv(a)
                current_M = conv.max()
                if current_M < best_M:
                    best_M = current_M
                    best_a = a.copy()
        else:
            no_improve_sweeps += 1
        
        if no_improve_sweeps > max_no_improve:
            break
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum > 1e-10:
        a /= a_sum
    return a


def spectral_flat_init(n, n_iters=200, seed=42):
    """Initialize using GS-style spectral inversion for flat convolution."""
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Start with a smooth bell shape
    a = np.exp(-4.0 * (t - 0.5) ** 2)
    a = np.maximum(a, 1e-6)
    a /= a.sum()
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    best_a = a.copy()
    best_obj = fast_conv(a).max()
    
    for it in range(n_iters):
        fa = np.fft.rfft(a, size)
        power = np.abs(fa) ** 2
        
        # Target: concentrate power at DC (makes conv flat in time domain)
        # Reduce non-DC power variance
        non_dc = power[1:]
        if len(non_dc) == 0:
            break
        p_mean = non_dc.mean()
        p_std = non_dc.std()
        
        # Soft threshold: reduce peaks in the power spectrum
        if p_std > 1e-10:
            # Compress the dynamic range
            alpha = 0.5 + 0.3 * (it / n_iters)  # increase compression over time
            power_new = power.copy()
            power_new[1:] = p_mean + alpha * (non_dc - p_mean)
            power_new = np.maximum(power_new, p_mean * 0.01)
        else:
            break
        
        mag_new = np.sqrt(power_new)
        phase = np.angle(fa)
        fa_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(fa_new, size)[:n]
        
        # Enforce non-negativity with soft projection
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        obj = fast_conv(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def mirror_descent(a, n_iters=5000, deadline=None):
    """Multiplicative (mirror) descent: a_i <- a_i * exp(-eta * grad_i).
    Naturally maintains non-negativity."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    eta = 0.15
    no_improve = 0
    best_iter = 0
    patience = 300
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        grad, M = compute_subgradient_fft(a)
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            best_iter = it
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            eta *= 0.5
            no_improve = 0
            if eta < 1e-5:
                break
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-10)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new
        
        # Slow eta decay
        eta *= 0.998
    
    return best_a


def projected_subgradient(a, n_iters=5000, deadline=None, eta0=0.1, momentum=0.8):
    """Projected subgradient descent with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    v = np.zeros(n)  # momentum
    eta = eta0
    no_improve = 0
    prev_y = a.copy()  # Nesterov lookahead
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Nesterov step
        theta = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        y = a + ((it) / theta) * (a - prev_y)
        prev_y = a.copy()
        
        grad, M = compute_subgradient_fft(y)
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            # We should track the a that gave best_M, not y
            # For simplicity, also check a
            a_check_M = fast_conv(a).max()
            if a_check_M < best_M:
                best_M = a_check_M
                best_a = a.copy()
            no_improve = 0
        else:
            # Check a as well
            a_check_M = fast_conv(a).max()
            if a_check_M < best_M:
                best_M = a_check_M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve > 500:
            break
        
        # Update
        v = momentum * v - eta * grad
        a_new = a + v
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = _project_simplex(a_new, 1.0)
        
        # Adaptive step size
        M_new = fast_conv(a_new).max()
        if M_new > M + 1e-6:
            eta *= 0.5
            v *= 0.5
            a_new = _project_simplex(a + v, 1.0)
        else:
            eta = min(eta * 1.02, 0.3)
        
        a = a_new
    
    # Final check
    final_M = fast_conv(a).max()
    if final_M < best_M:
        best_a = a.copy()
        best_M = final_M
    
    return best_a


def _project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum} in O(n log n)."""
    n = len(v)
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    
    # Find rho: largest index where w[rho] > (cs[rho] - target_sum) / (rho+1)
    thresh = (cs - target_sum) / np.arange(1, n + 1)
    valid = (w > thresh)
    if not np.any(valid):
        return np.full(n, target_sum / n)
    
    rho = np.max(np.where(valid)[0])
    theta = (cs[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def score_sequence(a):
    """Evaluate sequence and return score."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0 or a.sum() < 0.01:
        return float('inf')
    return float(evaluate_sequence(a.tolist()))


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: Smart coordinate descent from parent (most reliable, ~400s)
    Phase 2: Mirror descent for different local structure (~250s)
    Phase 3: Projected subgradient from spectral inits (~250s)
    Phase 4: Length variants and resampling (~200s)
    Phase 5: Final polishing with intensive CD (~150s)
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 10.0
    rng = np.random.default_rng(seed)
    
    best_seq = np.ones(64) / 64
    best_score = float('inf')
    
    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return False
        sc = score_sequence(a)
        if np.isfinite(sc) and sc < best_score:
            best_seq = a.copy()
            best_score = sc
            elapsed = time.monotonic() - t_start
            print(f"  ** {label}: J={sc:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
            return True
        return False

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            _try(parent, "parent")
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = spectral_flat_init(512, n_iters=100, seed=seed)
        _try(parent, "spectral_init")
    
    n_base = len(parent)
    print(f"\nStarting search: n_base={n_base}, current best={best_score:.6f}", flush=True)
    
    # === PHASE 1: Smart coordinate descent from parent (~400s) ===
    p1_end = min(t_start + 400, deadline - 400)
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Smart CD from parent ({p1_end - time.monotonic():.0f}s budget)", flush=True)
    print(f"{'='*50}", flush=True)
    
    t0 = time.monotonic()
    a_cd = smart_coordinate_descent(parent, max_sweeps=50000, deadline=p1_end,
                                     seed=seed, use_analytic_step=True)
    _try(a_cd, f"smart_cd_parent ({time.monotonic()-t0:.0f}s)")
    
    # Second pass on the improved solution
    if time.monotonic() < p1_end - 30:
        t0 = time.monotonic()
        a_cd2 = smart_coordinate_descent(best_seq, max_sweeps=20000,
                                          deadline=p1_end, seed=seed+7)
        _try(a_cd2, f"smart_cd_pass2 ({time.monotonic()-t0:.0f}s)")
    
    print(f"  Phase 1 best: J={best_score:.8f}", flush=True)
    
    # === PHASE 2: Mirror descent (~250s) ===
    p2_end = min(t_start + 700, deadline - 300)
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: Mirror Descent ({p2_end - time.monotonic():.0f}s budget)", flush=True)
    print(f"{'='*50}", flush=True)
    
    if time.monotonic() < p2_end - 20:
        t0 = time.monotonic()
        a_md = mirror_descent(best_seq, n_iters=5000, deadline=p2_end)
        _try(a_md, f"mirror_descent ({time.monotonic()-t0:.0f}s)")
        
        # CD after mirror
        if time.monotonic() < p2_end - 30:
            t0 = time.monotonic()
            a_cd_pw = smart_coordinate_descent(a_md, max_sweeps=10000,
                                                deadline=p2_end, seed=seed+100)
            _try(a_cd_pw, f"cd_after_mirror ({time.monotonic()-t0:.0f}s)")
    
    print(f"  Phase 2 best: J={best_score:.8f}", flush=True)
    
    # === PHASE 3: Projected subgradient from spectral inits (~250s) ===
    p3_end = min(t_start + 950, deadline - 200)
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Projected Subgradient + Spectral Inits ({p3_end - time.monotonic():.0f}s budget)", flush=True)
    print(f"{'='*50}", flush=True)
    
    for trial in range(4):
        if time.monotonic() > p3_end - 40:
            break
        
        # Generate diverse spectral init
        a_spec = spectral_flat_init(n_base, n_iters=80, seed=seed + 200 + trial * 13)
        
        # Mix with current best
        if best_seq.size != n_base:
            best_res = resample_seq(best_seq, n_base)
        else:
            best_res = best_seq
        a_mix = 0.4 * a_spec + 0.6 * best_res
        a_mix = np.maximum(a_mix, 0)
        a_mix /= a_mix.sum()
        
        t0 = time.monotonic()
        subtree_dl = min(t_start + 150 + trial * 60, p3_end - 20)
        a_proj = projected_subgradient(a_mix, n_iters=3000, deadline=subtree_dl,
                                        eta0=0.12, momentum=0.7)
        _try(a_proj, f"proj_subgrad_trial{trial} ({time.monotonic()-t0:.0f}s)")
        
        # Quick CD polish
        if time.monotonic() < p3_end - 20:
            a_polish = smart_coordinate_descent(a_proj, max_sweeps=5000,
                                                 deadline=p3_end - 10, seed=seed+300+trial)
            _try(a_polish, f"cd_polish_trial{trial}")
    
    print(f"  Phase 3 best: J={best_score:.8f}", flush=True)
    
    # === PHASE 4: Length variants (~150s) ===
    p4_end = min(t_start + 1150, deadline - 100)
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Length Variants ({p4_end - time.monotonic():.0f}s budget)", flush=True)
    print(f"{'='*50}", flush=True)
    
    for new_n in [n_base * 2, n_base // 2, n_base + 128, n_base - 128]:
        if new_n < 64 or new_n > 3000:
            continue
        if time.monotonic() > p4_end - 50:
            break
        
        a_res = resample_seq(best_seq, new_n)
        a_res = np.maximum(a_res, 1e-6)
        a_res /= a_res.sum()
        
        t0 = time.monotonic()
        a_opt = smart_coordinate_descent(a_res, max_sweeps=10000,
                                          deadline=p4_end - 30, seed=seed + 400 + new_n)
        _try(a_opt, f"length_{new_n} ({time.monotonic()-t0:.0f}s)")
    
    print(f"  Phase 4 best: J={best_score:.8f}", flush=True)
    
    # === PHASE 5: Final intensive polishing (~remaining time) ===
    p5_end = deadline
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Final Polish ({p5_end - time.monotonic():.0f}s budget)", flush=True)
    print(f"{'='*50}", flush=True)
    
    # Multiple perturbed restarts of CD
    for restart in range(8):
        if time.monotonic() > p5_end - 15:
            break
        
        # Geometrically increasing perturbation
        noise_scale = 0.02 * (1.5 ** restart)
        a_pert = best_seq + rng.normal(0, noise_scale, len(best_seq))
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert /= s
        
        t0 = time.monotonic()
        subtree_dl = min(t_start + 60 + restart * 30, p5_end - 10)
        a_fin = smart_coordinate_descent(a_pert, max_sweeps=20000,
                                          deadline=subtree_dl, seed=seed + 1000 + restart)
        _try(a_fin, f"final_restart{restart}_s{noise_scale:.3f} ({time.monotonic()-t0:.0f}s)")
    
    # One final mirror descent pass
    if time.monotonic() < p5_end - 30:
        t0 = time.monotonic()
        a_final_md = mirror_descent(best_seq, n_iters=3000, deadline=p5_end - 10)
        _try(a_final_md, f"final_mirror ({time.monotonic()-t0:.0f}s)")
        
        if time.monotonic() < p5_end - 10:
            a_final_cd = smart_coordinate_descent(a_final_md, max_sweeps=5000,
                                                   deadline=p5_end, seed=seed+2000)
            _try(a_final_cd, "final_cd_after_mirror")
    
    print(f"  Phase 5 best: J={best_score:.8f}", flush=True)
    
    # Final report
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL RESULT: J={best_score:.8f} n={len(best_seq)} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    
    return [float(x) for x in best_seq]
